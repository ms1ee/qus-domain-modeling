"""Grid model of the 8x8 city map (800x800 units, 100-unit cells).

Streets run along the grid lines (coordinates that are multiples of 100).
Restaurants and customers are snapped to the centres of the cells, i.e.
coordinates of the form 100 * k + 50.
"""

from dataclasses import dataclass
from typing import List, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
LATTICE_STEP = 50  # routing resolution (half a cell)

Point = Tuple[int, int]


def snap_to_center(value: int, cell_size: int = CELL_SIZE,
                   map_size: int = MAP_SIZE) -> int:
    """Convert a coordinate into the centre of its grid cell.

    Implements ``center = (grid // cell) * cell + cell // 2`` with clamping so
    that the boundary value (for example 800) maps to the last cell.

    Args:
        value: Input coordinate (0 - map_size).
        cell_size: Size of one grid cell.
        map_size: Size of the whole map.

    Returns:
        The integer coordinate of the cell centre along that axis.
    """
    dimension = map_size // cell_size
    index = min(max(int(value) // cell_size, 0), dimension - 1)
    return index * cell_size + cell_size // 2


def cell_index(x: int, y: int, cell_size: int = CELL_SIZE,
               map_size: int = MAP_SIZE) -> Tuple[int, int]:
    """Return the (grid_x, grid_y) index of the cell containing a point.

    Args:
        x: X coordinate.
        y: Y coordinate.
        cell_size: Size of one grid cell.
        map_size: Size of the whole map.

    Returns:
        Tuple of grid indices, clamped to the valid grid range.
    """
    dimension = map_size // cell_size
    gx = min(max(int(x) // cell_size, 0), dimension - 1)
    gy = min(max(int(y) // cell_size, 0), dimension - 1)
    return gx, gy


def is_street_point(x: int, y: int, cell_size: int = CELL_SIZE) -> bool:
    """Return True if the point lies on a street (a grid line).

    Args:
        x: X coordinate.
        y: Y coordinate.
        cell_size: Size of one grid cell.
    """
    return x % cell_size == 0 or y % cell_size == 0


@dataclass
class GridCell:
    """One 100x100 block of the grid.

    ``is_street`` is True for cells on the outer ring of the city, which are
    bordered by the main streets.
    """

    grid_x: int
    grid_y: int
    center_x: int
    center_y: int
    is_street: bool


class GridMap:
    """The 8x8 grid of cells covering the 800x800 coordinate system."""

    def __init__(self, map_size: int = MAP_SIZE, cell_size: int = CELL_SIZE) -> None:
        """Build the grid.

        Args:
            map_size: Width and height of the map in units.
            cell_size: Width and height of one cell in units.

        Raises:
            ValueError: If the sizes are not positive or do not divide evenly.
        """
        if map_size <= 0 or cell_size <= 0 or map_size % cell_size != 0:
            raise ValueError("map_size must be a positive multiple of cell_size")
        self.width = map_size
        self.height = map_size
        self.cell_size = cell_size
        self.dimension = map_size // cell_size
        self.cells: List[List[GridCell]] = []
        last = self.dimension - 1
        for gx in range(self.dimension):
            column: List[GridCell] = []
            for gy in range(self.dimension):
                column.append(GridCell(
                    grid_x=gx,
                    grid_y=gy,
                    center_x=gx * cell_size + cell_size // 2,
                    center_y=gy * cell_size + cell_size // 2,
                    is_street=gx in (0, last) or gy in (0, last),
                ))
            self.cells.append(column)

    def cell_at(self, x: int, y: int) -> GridCell:
        """Return the cell containing the point (x, y)."""
        gx, gy = cell_index(x, y, self.cell_size, self.width)
        return self.cells[gx][gy]

    def contains(self, x: int, y: int) -> bool:
        """Return True if (x, y) lies inside the map boundaries."""
        return 0 <= x <= self.width and 0 <= y <= self.height

    def snap_point(self, x: int, y: int) -> Point:
        """Snap a point to the centre of its grid cell."""
        return (snap_to_center(x, self.cell_size, self.width),
                snap_to_center(y, self.cell_size, self.height))

"""System configuration loaded from a JSON file with sane defaults."""

import json
import logging
import os
from dataclasses import asdict, dataclass, fields

from grid import LATTICE_STEP

logger = logging.getLogger(__name__)

DEFAULT_CONFIG_PATH = "dispatch_config.json"


@dataclass
class SystemConfig:
    """Tunable parameters of the dispatch simulation.

    Attributes:
        recruitment_threshold: Funds level at which a new rider is recruited.
        recruitment_cost: Funds spent when a rider is recruited.
        default_rider_speed: Movement speed in units per tick. Must divide 50
            so that riders always stay on the routing lattice.
        map_size: Map width/height in units.
        cell_size: Grid cell width/height in units.
        order_fee: Funds income earned for each accepted order.
        initial_funds: Funds available when the system starts.
        config_file_path: Path the configuration was loaded from.
    """

    recruitment_threshold: int = 300
    recruitment_cost: int = 300
    default_rider_speed: int = 50
    map_size: int = 800
    cell_size: int = 100
    order_fee: int = 100
    initial_funds: int = 300
    config_file_path: str = DEFAULT_CONFIG_PATH

    def validate(self) -> None:
        """Check parameter consistency.

        Raises:
            ValueError: If any parameter is invalid.
        """
        positive = ("recruitment_threshold", "recruitment_cost",
                    "default_rider_speed", "map_size", "cell_size", "order_fee")
        for name in positive:
            value = getattr(self, name)
            if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
                raise ValueError(f"{name} must be a positive integer")
        if isinstance(self.initial_funds, bool) or not isinstance(self.initial_funds, int):
            raise ValueError("initial_funds must be an integer")
        if self.map_size % self.cell_size != 0:
            raise ValueError("map_size must be a multiple of cell_size")
        if LATTICE_STEP % self.default_rider_speed != 0:
            raise ValueError(f"default_rider_speed must divide {LATTICE_STEP}")

    @classmethod
    def load(cls, path: str = DEFAULT_CONFIG_PATH) -> "SystemConfig":
        """Load configuration from a JSON file.

        Missing files fall back to defaults. Invalid files are logged and the
        defaults are used instead, so the system always starts.

        Args:
            path: Path to the JSON configuration file.

        Returns:
            A validated SystemConfig instance.
        """
        config = cls(config_file_path=path)
        if not os.path.exists(path):
            logger.info("Configuration file %s not found; using defaults", path)
            config.validate()
            return config
        try:
            with open(path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
            if not isinstance(data, dict):
                raise ValueError("top-level JSON value must be an object")
            valid = {f.name for f in fields(cls)} - {"config_file_path"}
            for key, value in data.items():
                if key in valid:
                    setattr(config, key, value)
                else:
                    logger.warning("Ignoring unknown configuration key '%s'", key)
            config.validate()
        except (OSError, json.JSONDecodeError, ValueError, TypeError) as exc:
            logger.error("Invalid configuration %s (%s); using defaults", path, exc)
            config = cls(config_file_path=path)
        return config

    def to_dict(self) -> dict:
        """Return the configuration as a plain dictionary."""
        return asdict(self)

"""Order data model: coordinates, lifecycle status and delivery metrics."""

import itertools
from typing import Any, Dict, Tuple

from globalvar import get_value
from grid import MAP_SIZE, snap_to_center

Point = Tuple[int, int]

_order_ids = itertools.count(1)


def _validate_point(point: Any, label: str) -> Point:
    """Validate an ``[x, y]`` pair of integers within the map range.

    Args:
        point: Two-element sequence of integers.
        label: Human-readable name used in error messages.

    Returns:
        The validated (x, y) tuple.

    Raises:
        ValueError: If the input is malformed or out of range.
    """
    try:
        x, y = point
    except (TypeError, ValueError):
        raise ValueError(f"{label} location must be an [x, y] pair") from None
    for name, value in (("x", x), ("y", y)):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} {name} must be an integer")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(
                f"{label} {name} must be between 0 and {MAP_SIZE}, got {value}")
    return x, y


class Order:
    """A food delivery order from a restaurant to a customer.

    Input coordinates are validated and automatically converted to the centres
    of their grid cells.

    Attributes:
        id: Unique order identifier.
        restaurant_x, restaurant_y: Pickup cell-centre coordinates.
        customer_x, customer_y: Delivery cell-centre coordinates.
        start_time: Simulated tick at which the order was placed.
        completion_time: Tick at which delivery finished (None until done).
        picked_up_time: Tick at which the food was collected.
        distance: Total planned travel distance in units.
        time_taken: Ticks between placement and completion.
        status: One of created, queued, assigned, picking_up, delivering,
            completed.
        rider_id: ID of the assigned rider, or None.
    """

    def __init__(self, res_pos: Any, cus_pos: Any) -> None:
        """Create an order from restaurant and customer locations.

        Args:
            res_pos: ``[x, y]`` restaurant location (0-800).
            cus_pos: ``[x, y]`` customer location (0-800).

        Raises:
            ValueError: If a location is malformed or out of range.
        """
        res_x, res_y = _validate_point(res_pos, "restaurant")
        cus_x, cus_y = _validate_point(cus_pos, "customer")
        self.id: int = next(_order_ids)
        self.restaurant_x = snap_to_center(res_x)
        self.restaurant_y = snap_to_center(res_y)
        self.customer_x = snap_to_center(cus_x)
        self.customer_y = snap_to_center(cus_y)
        self.start_time: int = int(get_value("simulated_time", 0) or 0)
        self.completion_time = None
        self.picked_up_time = None
        self.distance: int = 0
        self.time_taken = None
        self.status: str = "created"
        self.rider_id = None

    @property
    def restaurant_pos(self) -> Point:
        """Restaurant cell-centre coordinates as a tuple."""
        return (self.restaurant_x, self.restaurant_y)

    @property
    def customer_pos(self) -> Point:
        """Customer cell-centre coordinates as a tuple."""
        return (self.customer_x, self.customer_y)

    def assign_to(self, rider_id: int) -> None:
        """Record the rider this order is assigned to.

        Args:
            rider_id: Identifier of the rider.
        """
        self.rider_id = rider_id
        self.status = "assigned"

    def mark_picked_up(self, time: int) -> None:
        """Mark the food as collected at the restaurant.

        Args:
            time: Simulated tick of the pickup.
        """
        self.picked_up_time = time
        self.status = "delivering"

    def add_distance(self, units: int) -> None:
        """Add planned travel distance to the order.

        Args:
            units: Distance in map units.
        """
        self.distance += int(units)

    def complete(self, time: int) -> None:
        """Mark the order as delivered and compute efficiency metrics.

        Args:
            time: Simulated tick of delivery.
        """
        self.completion_time = time
        self.time_taken = time - self.start_time
        self.status = "completed"

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "rider_id": self.rider_id,
            "status": self.status,
        }

    def __repr__(self) -> str:
        return (f"Order(id={self.id}, status={self.status}, "
                f"restaurant={self.restaurant_pos}, customer={self.customer_pos})")

"""Grid-based path planning with a preference for streets.

The routing lattice has a node every 50 units. Edges along street lines
(coordinates that are multiples of 100) are cheap. Entering or leaving the
block of a restaurant or customer (the "driveway", i.e. a rider directly above
or below the target) is allowed but costs more, so routes prefer streets.
"""

import heapq
import itertools
import logging
import math
from dataclasses import dataclass
from typing import Dict, List, Optional, Sequence, Tuple

from grid import CELL_SIZE, LATTICE_STEP, MAP_SIZE, GridMap, is_street_point

logger = logging.getLogger(__name__)

Point = Tuple[int, int]


class PathNotFoundError(Exception):
    """Raised when no route exists between two points."""


@dataclass
class Route:
    """A planned route.

    Attributes:
        route_id: Unique identifier of the route.
        points: Ordered lattice points, including start and goal.
        total_distance: Total Manhattan length in units.
        estimated_ticks: Ticks needed at the planned speed.
        has_driveway: True if the route uses a non-street final approach
            (special case: target directly above or below the rider).
    """

    route_id: int
    points: List[Point]
    total_distance: int
    estimated_ticks: int
    has_driveway: bool = False

    @property
    def special_case(self) -> bool:
        """Alias of ``has_driveway`` used by the rider special-case flag."""
        return self.has_driveway


class PathPlanner:
    """Computes street-preferring routes on the 8x8 grid."""

    STREET_COST = 1
    DRIVEWAY_COST = 2

    def __init__(self, grid: Optional[GridMap] = None) -> None:
        """Create a planner.

        Args:
            grid: Grid map to plan on. A default 8x8 map is created if omitted.
        """
        self.grid = grid if grid is not None else GridMap()
        self.grid_size = self.grid.dimension
        self.prefer_streets = True
        self.step = LATTICE_STEP
        self._route_ids = itertools.count(1)

    def nearest_lattice(self, point: Point) -> Point:
        """Snap a point to the nearest routing-lattice node inside the map."""
        x, y = point
        lx = min(max(int(round(x / self.step)) * self.step, 0), self.grid.width)
        ly = min(max(int(round(y / self.step)) * self.step, 0), self.grid.height)
        return lx, ly

    def plan(self, start: Point, goal: Point, speed: int = 1) -> Route:
        """Plan a route from ``start`` to ``goal``.

        Args:
            start: Starting coordinates.
            goal: Destination coordinates (usually a cell centre).
            speed: Rider speed in units per tick, used for the tick estimate.

        Returns:
            A Route describing the path.

        Raises:
            ValueError: If speed is not positive.
            PathNotFoundError: If no route exists.
        """
        if speed <= 0:
            raise ValueError("speed must be positive")
        start_node = self.nearest_lattice(start)
        goal_node = self.nearest_lattice(goal)
        if start_node == goal_node:
            return Route(next(self._route_ids), [start_node], 0, 0, False)

        previous = self._dijkstra(start_node, goal_node)
        points = [goal_node]
        while points[-1] != start_node:
            points.append(previous[points[-1]])
        points.reverse()

        total = 0
        driveway = False
        for a, b in zip(points, points[1:]):
            total += abs(b[0] - a[0]) + abs(b[1] - a[1])
            if not self._is_street_edge(a, b):
                driveway = True
        ticks = int(math.ceil(total / speed))
        return Route(next(self._route_ids), points, total, ticks, driveway)

    def plan_multi(self, start: Point, destinations: Sequence[Point],
                   speed: int = 1) -> Route:
        """Plan a continuous route through several destinations in order.

        Used for the restaurant-then-customer path of an order.

        Args:
            start: Starting coordinates.
            destinations: Ordered list of waypoints.
            speed: Rider speed in units per tick.

        Returns:
            A single Route covering all legs.
        """
        if speed <= 0:
            raise ValueError("speed must be positive")
        points: List[Point] = [self.nearest_lattice(start)]
        total = 0
        driveway = False
        for destination in destinations:
            leg = self.plan(points[-1], destination, speed)
            points.extend(leg.points[1:])
            total += leg.total_distance
            driveway = driveway or leg.has_driveway
        ticks = int(math.ceil(total / speed))
        return Route(next(self._route_ids), points, total, ticks, driveway)

    # ------------------------------------------------------------------
    # internal helpers
    # ------------------------------------------------------------------
    def _is_street_edge(self, a: Point, b: Point) -> bool:
        """Return True if the lattice edge a-b lies on a street line."""
        if a[1] == b[1]:
            return is_street_point(0, a[1], self.grid.cell_size)
        return is_street_point(a[0], 0, self.grid.cell_size)

    def _neighbors(self, node: Point) -> List[Point]:
        """Return the lattice neighbours of a node that lie inside the map."""
        x, y = node
        result = []
        for dx, dy in ((self.step, 0), (-self.step, 0), (0, self.step), (0, -self.step)):
            nx, ny = x + dx, y + dy
            if self.grid.contains(nx, ny):
                result.append((nx, ny))
        return result

    def _edge_allowed(self, a: Point, b: Point, start: Point, goal: Point) -> bool:
        """Edges are allowed on streets, or as driveways touching start/goal."""
        if self._is_street_edge(a, b):
            return True
        return a in (start, goal) or b in (start, goal)

    def _dijkstra(self, start: Point, goal: Point) -> Dict[Point, Point]:
        """Run Dijkstra's algorithm and return the predecessor map.

        Raises:
            PathNotFoundError: If the goal cannot be reached.
        """
        dist: Dict[Point, int] = {start: 0}
        previous: Dict[Point, Point] = {}
        heap: List[Tuple[int, Point]] = [(0, start)]
        while heap:
            cost_so_far, node = heapq.heappop(heap)
            if node == goal:
                break
            if cost_so_far > dist.get(node, math.inf):
                continue
            for neighbor in self._neighbors(node):
                if not self._edge_allowed(node, neighbor, start, goal):
                    continue
                step_cost = (self.STREET_COST if self._is_street_edge(node, neighbor)
                             else self.DRIVEWAY_COST)
                new_cost = cost_so_far + step_cost
                if new_cost < dist.get(neighbor, math.inf):
                    dist[neighbor] = new_cost
                    previous[neighbor] = node
                    heapq.heappush(heap, (new_cost, neighbor))
        if goal not in dist:
            raise PathNotFoundError(f"no route from {start} to {goal}")
        return previous

"""Rider model: state machine, movement along planned routes and event logs.

State machine:
    IDLE (0)        -> PICKING_UP (1)  when an order is accepted (queued order exists)
    PICKING_UP (1)  -> DELIVERING (2)  when the restaurant is reached
    DELIVERING (2)  -> IDLE (0)        when the customer is reached (or the next
                                       queued order is accepted immediately)
"""

import logging
from typing import Any, Dict, List, Optional, Tuple

from globalvar import get_value
from grid import GridMap
from path_planner import PathNotFoundError, PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

SPAWN_POINT = (400, 400)  # central street intersection
DEFAULT_SPEED = 50
MAX_WAIT_TICKS = 3        # yield limit before deadlock avoidance forces progress
MAX_LOG_ENTRIES = 5000    # bounds memory use during long runs

Point = Tuple[int, int]

_fallback_planner: Optional[PathPlanner] = None


def _configured_speed() -> int:
    """Return the default rider speed from the system configuration."""
    config = get_value("config")
    return int(config.default_rider_speed) if config is not None else DEFAULT_SPEED


def _planner() -> PathPlanner:
    """Return the shared path planner, creating a fallback if needed."""
    global _fallback_planner
    planner = get_value("path_planner")
    if planner is None:
        if _fallback_planner is None:
            _fallback_planner = PathPlanner(GridMap())
        planner = _fallback_planner
    return planner


class Rider:
    """A delivery rider that moves on the grid and serves assigned orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates.
        speed: Movement speed in units per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders; the first one is the current order.
        special_case: True when the current approach is a driveway
            (target directly above or below the rider's route).
        log: Event dictionaries with keys "event" and "order_id" (plus "time",
            "x" and "y").
    """

    def __init__(self, rider_id: int) -> None:
        """Create an idle rider at the spawn point.

        Args:
            rider_id: Unique identifier of the rider.
        """
        self.rider_id: int = int(rider_id)
        self.x: int = SPAWN_POINT[0]
        self.y: int = SPAWN_POINT[1]
        self.speed: int = _configured_speed()
        self.state: int = IDLE
        self.special_case: bool = False
        self.orders: List[Any] = []
        self.target: Optional[Point] = None
        self.path: List[Point] = []
        self.wait_ticks: int = 0
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # properties and helpers
    # ------------------------------------------------------------------
    @property
    def position(self) -> Point:
        """Current coordinates as a tuple."""
        return (self.x, self.y)

    @property
    def current_order(self) -> Any:
        """The order being served, or None when no orders are assigned."""
        return self.orders[0] if self.orders else None

    @property
    def state_name(self) -> str:
        """Human-readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    def _now(self) -> int:
        """Return the current simulated tick."""
        return int(get_value("simulated_time", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event to the rider log, trimming old entries if needed."""
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
        })
        if len(self.log) > MAX_LOG_ENTRIES:
            del self.log[: len(self.log) - MAX_LOG_ENTRIES]
        logger.info("Rider %d: %s (order %s) at (%d,%d)",
                    self.rider_id, event, order_id, self.x, self.y)

    def reset_after_error(self) -> None:
        """Return the rider to a consistent idle state after an internal error.

        Assigned orders are kept so that they are re-accepted on the next tick.
        """
        self.state = IDLE
        self.target = None
        self.path = []
        self.special_case = False
        self.wait_ticks = 0

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [getattr(order, "id", None) for order in self.orders],
        }

    # ------------------------------------------------------------------
    # main tick logic
    # ------------------------------------------------------------------
    def move(self) -> None:
        """Advance the rider by one tick.

        Handles order acceptance, movement along the route, arrival at the
        restaurant and delivery to the customer, including all state
        transitions and their log entries.
        """
        now = self._now()
        if self.state == IDLE:
            if not self.orders:
                return
            self._accept_order(now)
        if self.state == IDLE or self.target is None:
            return
        self._advance(now)
        self._handle_arrivals(now)

    def _accept_order(self, now: int) -> None:
        """Start the pickup leg for the first queued order."""
        order = self.orders[0]
        order.status = "picking_up"
        self.state = PICKING_UP
        self._record("accepted_order", order.id)
        self._plan_to(order.restaurant_pos, order)

    def _plan_to(self, target: Point, order: Any) -> None:
        """Plan a route from the current position to ``target``."""
        try:
            route = _planner().plan(self.position, target, self.speed)
            self.path = list(route.points[1:])
            self.special_case = route.has_driveway
            order.add_distance(route.total_distance)
        except PathNotFoundError as exc:
            logger.error("Rider %d: %s; snapping to target", self.rider_id, exc)
            self.path = [target]
            self.special_case = False
        self.target = target

    def _advance(self, now: int) -> None:
        """Move along the planned path by up to ``speed`` units."""
        budget = self.speed
        while budget > 0 and self.path:
            nxt = self.path[0]
            if self._is_blocked(nxt):
                if self.wait_ticks < MAX_WAIT_TICKS:
                    self.wait_ticks += 1
                    return
                logger.warning("Rider %d: forcing progress to avoid deadlock",
                               self.rider_id)
            self.wait_ticks = 0
            dx = nxt[0] - self.x
            dy = nxt[1] - self.y
            distance = abs(dx) + abs(dy)
            if distance <= budget:
                self.x, self.y = nxt
                self.path.pop(0)
                budget -= distance
            else:
                if dx:
                    self.x += budget if dx > 0 else -budget
                else:
                    self.y += budget if dy > 0 else -budget
                budget = 0

    def _is_blocked(self, nxt: Point) -> bool:
        """Conflict avoidance for the next lattice node.

        A rider yields when another rider occupies the next node and either
        the two riders head towards each other (the higher ID yields) or the
        occupant is stationary.
        """
        for other in get_value("riders", []) or []:
            if other is self or other.position != nxt:
                continue
            if other.path and other.path[0] == self.position \
                    and other.rider_id < self.rider_id:
                return True
            if not other.path:
                return True
        return False

    def _handle_arrivals(self, now: int) -> None:
        """Process arrivals at the current target, possibly several in a row."""
        while self.state != IDLE and self.target is not None \
                and self.position == self.target:
            if self.state == PICKING_UP:
                self._arrive_at_merchant(now)
            elif self.state == DELIVERING:
                self._deliver(now)
            else:
                break

    def _arrive_at_merchant(self, now: int) -> None:
        """Collect the food and start the delivery leg."""
        order = self.orders[0]
        self._record("arrived_at_merchant", order.id)
        self.state = DELIVERING
        order.mark_picked_up(now)
        self._plan_to(order.customer_pos, order)

    def _deliver(self, now: int) -> None:
        """Complete the current order and start the next one if queued."""
        order = self.orders[0]
        self._record("delivered_to_customer", order.id)
        order.complete(now)
        self.orders.pop(0)
        self.state = IDLE
        self.target = None
        self.path = []
        self.special_case = False
        self.wait_ticks = 0
        if self.orders:
            self._accept_order(now)

    def __repr__(self) -> str:
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state_name}, orders={len(self.orders)})")

"""Order management and rider assignment.

Assignment strategy: the closest rider is preferred. Each order already queued
on a rider adds a load penalty, which spreads work across riders (load
balancing) instead of overloading one rider.
"""

import logging
from typing import TYPE_CHECKING, List, Optional, Tuple

from globalvar import get_value
from order_model import Order

if TYPE_CHECKING:  # pragma: no cover - only for type checkers
    from rider import Rider

logger = logging.getLogger(__name__)

LOAD_PENALTY = 150  # distance-equivalent cost of each order already queued


def find_best_rider_for_order(order: Order) -> "Rider":
    """Find the best rider for an order.

    The score of a rider is the Manhattan distance from its position to the
    restaurant plus ``LOAD_PENALTY`` per queued order. Ties are broken by the
    lower rider ID.

    Args:
        order: Order that needs a rider.

    Returns:
        The selected Rider, or None if no riders exist.
    """
    riders = get_value("riders", []) or []
    rest_x, rest_y = order.restaurant_pos
    best = None
    best_key: Optional[Tuple[int, int]] = None
    for rider in riders:
        distance = abs(rider.x - rest_x) + abs(rider.y - rest_y)
        key = (distance + LOAD_PENALTY * len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates orders, assigns them to riders and tracks their status."""

    def __init__(self) -> None:
        """Create an empty order manager."""
        self.orders: List[Order] = []
        self.pending: List[Order] = []

    def create_order(self, restaurant_pos, customer_pos) -> Order:
        """Create and register a new order.

        Args:
            restaurant_pos: ``[x, y]`` restaurant location.
            customer_pos: ``[x, y]`` customer location.

        Returns:
            The created Order.
        """
        order = Order(restaurant_pos, customer_pos)
        self.orders.append(order)
        return order

    def assign(self, order: Order) -> Optional["Rider"]:
        """Assign an order to the best rider without queueing it.

        The order is appended to ``rider.orders`` and recorded on the order.

        Args:
            order: Order to assign.

        Returns:
            The rider that received the order, or None if no rider exists.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Order %d assigned to rider %d", order.id, rider.rider_id)
        return rider

    def submit(self, order: Order) -> Optional["Rider"]:
        """Assign an order, or queue it as pending when no rider is available."""
        rider = self.assign(order)
        if rider is None:
            order.status = "queued"
            self.pending.append(order)
            logger.warning("No rider available; order %d queued", order.id)
        return rider

    def retry_pending(self) -> List[Order]:
        """Try to assign every pending order.

        Returns:
            Orders that were assigned during this call.
        """
        assigned: List[Order] = []
        for order in list(self.pending):
            if self.assign(order) is not None:
                self.pending.remove(order)
                assigned.append(order)
        return assigned

    def active_orders(self) -> List[Order]:
        """Return orders that have not been completed yet."""
        return [o for o in self.orders if o.status != "completed"]

    def completed_orders(self) -> List[Order]:
        """Return completed orders."""
        return [o for o in self.orders if o.status == "completed"]

    # funds and recruitment
    # ------------------------------------------------------------------

        """Add (or subtract, if negative) funds."""
        self.funds = self.funds + int(amount)

    def create_rider(self) -> Rider:
        """Create a new rider with a fresh unique ID and register it."""
        counter = self.rider_id_counter + 1
        set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        logger.info("Recruited rider %d", counter)
        return rider

    def recruit_riders(self) -> List[int]:
        """Recruit riders while funds are at or above the threshold.

        Each recruitment costs ``recruitment_cost`` from the funds.

        Returns:
            IDs of the riders created during this call.
        """
        new_ids: List[int] = []
        threshold = self.config.recruitment_threshold
        cost = self.config.recruitment_cost
        while self.funds >= threshold:
            rider = self.create_rider()
            self.funds = self.funds - cost
            new_ids.append(rider.rider_id)
        return new_ids

    # orders and simulation
    # ------------------------------------------------------------------
    def place_order(self, restaurant_pos, customer_pos) -> Dict[str, Any]:
        """Create an order, collect its fee, recruit riders and assign it.

        Args:
            restaurant_pos: ``[x, y]`` restaurant location.
            customer_pos: ``[x, y]`` customer location.

        Returns:
            Dictionary with keys "order", "new_riders" (list of IDs),
            "rider" (Rider or None) and "route" (Route or None; preview of the
            assigned rider's restaurant-then-customer path).
        """
        order = self.order_manager.create_order(restaurant_pos, customer_pos)
        self.add_funds(self.config.order_fee)
        new_riders = self.recruit_riders()
        rider = self.order_manager.submit(order)
        route = None
        if rider is not None:
            try:
                route = self.planner.plan_multi(
                    rider.position,
                    [order.restaurant_pos, order.customer_pos],
                    rider.speed)
            except PathNotFoundError as exc:
                logger.error("Route preview failed: %s", exc)
        return {
            "order": order,
            "new_riders": new_riders,
            "rider": rider,
            "route": route,
        }

    def tick(self) -> None:
        """Advance the simulation by one tick.

        Increments the clock, retries pending orders, moves every rider and
        recruits riders if funds allow. An error in one rider is logged and
        that rider is reset, so the rest of the system keeps running.
        """
        set_value("simulated_time", self.simulated_time + 1)
        self.order_manager.retry_pending()
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # error recovery: keep the simulation alive
                logger.exception("Rider %d failed; resetting its state",
                                 rider.rider_id)
                rider.reset_after_error()
        self.recruit_riders()

    def status_snapshot(self) -> Dict[str, Any]:
        """Return a summary of the current state."""
        return {
            "simulated_time": self.simulated_time,
            "funds": self.funds,
            "rider_count": len(self.riders),
            "riders": [r.to_dict() for r in self.riders],
            "active_orders": len(self.order_manager.active_orders()),
            "pending_orders": len(self.order_manager.pending),
        }

    # ------------------------------------------------------------------
    # persistence
    # ------------------------------------------------------------------

            path: Destination file path.

            The path the data was written to.

        Raises:
            ValueError: If the file cannot be written.

            "config": self.config.to_dict(),
            "simulated_time": self.simulated_time,
            "funds": self.funds,
            "rider_id_counter": self.rider_id_counter,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.orders],
        }

        try:
            with open(tmp_path, "w", encoding="utf-8") as handle:

        except OSError as exc:
            logger.error("Could not save state to %s: %s", path, exc)
            raise ValueError(f"could not save state to '{path}': {exc}") from None
        logger.info("State saved to %s", path)
        return path

"""Status displays and ASCII map visualisation.

The map is drawn on a character canvas where every routing-lattice node
(every 50 units) occupies one even-indexed cell and the space between two
neighbouring nodes is an odd-indexed cell. This makes streets and route
segments visible as connected lines.

Symbols:
    R  rider          S  restaurant      C  customer
    .  empty space    #  street          *  route node
    -  route segment (horizontal)        |  route segment (vertical)
"""

from typing import Iterable, List, Optional, Sequence, Tuple

from grid import CELL_SIZE, LATTICE_STEP, MAP_SIZE
from path_planner import Route

Point = Tuple[int, int]

LEGEND = ("Legend: R rider  S restaurant  C customer  # street  . block  "
          "* route  -/| route segment\n"
          "Top = y 800, left = x 0. Streets are the lines at multiples of 100; "
          "cell centres are at 50, 150, ... 750.")


class DisplayManager:
    """Renders the map, status, order table and rider logs as text."""

    def __init__(self, map_width: int = MAP_SIZE, map_height: int = MAP_SIZE,
                 cell_size: int = CELL_SIZE) -> None:
        """Create a display manager.

        Args:
            map_width: Map width in units.
            map_height: Map height in units.
            cell_size: Grid cell size in units.
        """
        self.map_width = map_width
        self.map_height = map_height
        self.cell_size = cell_size
        self.step = LATTICE_STEP

    def _to_canvas(self, x: float, y: float) -> Tuple[int, int]:
        """Convert map coordinates to (row, column) canvas indices."""
        col = int(round(2 * x / self.step))
        row = int(round(2 * (self.map_height - y) / self.step))
        return row, col

    def _is_street(self, x: int, y: int) -> bool:
        """Return True if the lattice node lies on a street."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def render_map(self, riders: Iterable = (), orders: Iterable = (),
                   route: Optional[Route] = None) -> str:
        """Render the 8x8 city map with riders, orders and an optional route.

        Args:
            riders: Riders to mark with 'R'.
            orders: Orders whose restaurant ('S') and customer ('C') are shown.
            route: Optional route drawn as connected line segments.

        Returns:
            The map as a multi-line string.
        """
        rows = 2 * self.map_height // self.step + 1
        cols = 2 * self.map_width // self.step + 1
        canvas: List[List[str]] = [["." for _ in range(cols)] for _ in range(rows)]
        nodes_x = self.map_width // self.step + 1
        nodes_y = self.map_height // self.step + 1

        for xi in range(nodes_x):
            for yi in range(nodes_y):
                x, y = xi * self.step, yi * self.step
                row, col = self._to_canvas(x, y)
                if self._is_street(x, y):
                    canvas[row][col] = "#"
                if xi + 1 < nodes_x and y % self.cell_size == 0:
                    r2, c2 = self._to_canvas(x + self.step, y)
                    canvas[(row + r2) // 2][(col + c2) // 2] = "#"
                if yi + 1 < nodes_y and x % self.cell_size == 0:
                    r2, c2 = self._to_canvas(x, y + self.step)
                    canvas[(row + r2) // 2][(col + c2) // 2] = "#"

        if route is not None:
            points = list(route.points)
            for a, b in zip(points, points[1:]):
                ra, ca = self._to_canvas(*a)
                rb, cb = self._to_canvas(*b)
                canvas[(ra + rb) // 2][(ca + cb) // 2] = "-" if a[1] == b[1] else "|"
            for point in points:
                r, c = self._to_canvas(*point)
                canvas[r][c] = "*"

        for order in orders:
            r, c = self._to_canvas(order.restaurant_x, order.restaurant_y)
            canvas[r][c] = "S"
            r, c = self._to_canvas(order.customer_x, order.customer_y)
            canvas[r][c] = "C"

        for rider in riders:
            r, c = self._to_canvas(rider.x, rider.y)
            canvas[r][c] = "R"

        body = "\n".join("".join(row) for row in canvas)
        return f"{body}\n{LEGEND}"

    @staticmethod
    def _cell_label(x: int, y: int) -> str:
        """Return the grid cell label of a coordinate, e.g. '(1,2)'."""
        gx = min(max(x // CELL_SIZE, 0), 7)
        gy = min(max(y // CELL_SIZE, 0), 7)
        return f"({gx},{gy})"

    def render_status(self, system) -> str:
        """Render funds, rider count and per-rider details.

        Args:
            system: SystemManager instance.
        """
        lines = [
            f"Simulated time : {system.simulated_time}",
            f"System funds   : {system.funds}",
            f"Rider count    : {len(system.riders)}",
        ]
        for rider in system.riders:
            order_ids = ", ".join(str(getattr(o, "id", "?")) for o in rider.orders) or "-"
            order_status = ", ".join(
                f"#{o.id}:{o.status}" for o in rider.orders) or "-"
            lines.append(
                f"  Rider {rider.rider_id:>3} | {rider.state_name:<10} | "
                f"pos ({rider.x},{rider.y}) | cell {self._cell_label(rider.x, rider.y)} | "
                f"speed {rider.speed} | orders [{order_ids}] | {order_status}")
        if not system.riders:
            lines.append("  No riders yet. Riders are recruited when funds reach the threshold.")
        return "\n".join(lines)

    def render_orders(self, orders: Sequence) -> str:
        """Render a table of orders.

        Args:
            orders: Order objects to display.
        """
        if not orders:
            return "No orders have been placed yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Distance':>9} {'Taken':>6} {'Rider':>6} {'Status':<12}")
        lines = [header, "-" * len(header)]
        for o in orders:
            taken = "-" if o.time_taken is None else str(o.time_taken)
            rider = "-" if o.rider_id is None else str(o.rider_id)
            lines.append(
                f"{o.id:>4} {o.start_time:>6} "
                f"{'(' + str(o.restaurant_x) + ',' + str(o.restaurant_y) + ')':>12} "
                f"{'(' + str(o.customer_x) + ',' + str(o.customer_y) + ')':>12} "
                f"{o.distance:>9} {taken:>6} {rider:>6} {o.status:<12}")
        return "\n".join(lines)

    def render_rider_log(self, riders: Sequence) -> str:
        """Render every recorded status change of every rider.

        Args:
            riders: Rider objects whose logs are shown.
        """
        if not riders:
            return "No riders yet."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider {rider.rider_id} (current state: {rider.state_name})")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order = "-" if entry["order_id"] is None else str(entry["order_id"])
                lines.append(
                    f"  t={entry['time']:>5}  {entry['event']:<22} "
                    f"order={order:<5} pos=({entry['x']},{entry['y']})")
        return "\n".join(lines)

"""Command parsing, shortcuts, history and input validation."""

import logging

from typing import List, Optional

COORDINATE_LABELS = ["restaurant x", "restaurant y", "customer x", "customer y"]
MAX_TICKS_PER_COMMAND = 1000
MAX_HISTORY = 500


@dataclass
class Command:

    Attributes:
        name: Canonical command name (e.g. "start").
        arguments: Integer arguments (coordinates or tick count).

    arguments: List[int] = field(default_factory=list)

class InputManager:
    """Parses and validates user input and keeps the command history."""

    COMMANDS = {"start", "status", "orders", "riders", "map", "tick",
                "save", "history", "help", "quit", "exit"}
    SHORTCUTS = {
        "s": "start", "st": "status", "o": "orders", "r": "riders",
        "m": "map", "t": "tick", "v": "save", "hi": "history",
        "h": "help", "?": "help", "q": "quit", "x": "exit",

        self.command_history: List[str] = []

    # validation helpers
    # ------------------------------------------------------------------
    @staticmethod
    def parse_int(token: str, label: str) -> int:
        """Convert a token to an integer with a clear error message.

        Raises:
            ValueError: If the token is not an integer.
        """
        try:
            return int(token)
        except ValueError:
            raise ValueError(f"{label} must be an integer, got '{token}'") from None

    @staticmethod
    def validate_coordinate(value: int, label: str) -> int:
        """Ensure a coordinate lies within 0-800.

        Raises:
            ValueError: If the coordinate is out of range.

            raise ValueError(f"{label} must be between 0 and {MAP_SIZE}, got {value}")

    # ------------------------------------------------------------------
    # input reading and history
    # ------------------------------------------------------------------
    def read_line(self, prompt: str = "> ") -> Optional[str]:
        """Read one line from standard input.

        Returns:
            The line, or None on end-of-file or interrupt.

            return input(prompt)
        except (EOFError, KeyboardInterrupt):
            return None

    def resolve_history(self, line: str) -> str:
        """Expand history references such as ``!!`` and ``!3``.

        Raises:
            ValueError: If the reference is invalid.
        """
        text = line.strip()
        if not text.startswith("!"):
            return text
        body = text[1:]
        if body == "!":
            if not self.command_history:
                raise ValueError("there is no previous command to repeat")
            return self.command_history[-1]
        if body.isdigit():
            index = int(body)
            if not 1 <= index <= len(self.command_history):
                raise ValueError(f"no history entry {index}")
            return self.command_history[index - 1]
        raise ValueError("history references are '!!' or '!N'")

    def record(self, line: str) -> None:
        """Append a successfully executed command to the history."""
        self.command_history.append(line)
        if len(self.command_history) > MAX_HISTORY:
            del self.command_history[: len(self.command_history) - MAX_HISTORY]

    # ------------------------------------------------------------------
    # parsing
    # ------------------------------------------------------------------
    def parse(self, line: str) -> Command:
        """Parse and validate one command line.

        Args:
            line: Raw (already history-resolved) input line.

        Returns:
            A Command object.

        Raises:
            ValueError: With a user-facing message when the input is invalid.
        """
        text = line.strip()
        if not text:
            raise ValueError("empty command; type 'help' for a list of commands")
        tokens = text.split()
        typed = tokens[0].lower()
        name = self.SHORTCUTS.get(typed, typed)
        if name not in self.COMMANDS:
            raise ValueError(f"unknown command '{tokens[0]}'; type 'help'")

        arguments: List[int] = []
        args = tokens[1:]
        if name == "start":
            if len(args) not in (0, 4):
                raise ValueError("start takes either no arguments (interactive) or "
                                 "four coordinates: restaurant_x restaurant_y "
                                 "customer_x customer_y")
            for token, label in zip(args, COORDINATE_LABELS):
                value = self.parse_int(token, label)
                arguments.append(self.validate_coordinate(value, label))
        elif name == "tick":
            if len(args) > 1:
                raise ValueError("tick takes at most one argument (number of ticks)")
            count = self.parse_int(args[0], "tick count") if args else 1
            if not 1 <= count <= MAX_TICKS_PER_COMMAND:
                raise ValueError(
                    f"tick count must be between 1 and {MAX_TICKS_PER_COMMAND}")
            arguments.append(count)
        elif name != "save" and args:
            raise ValueError(f"'{name}' does not take arguments")

        return Command(name=name, arguments=arguments, raw_input=text)

"""Command-line entry point for the food delivery dispatch simulator.

Run with:  python src/main.py [--debug]
"""

import argparse
import logging
import sys
from typing import Optional, Tuple

from display import DisplayManager
from globalvar import get_value
from grid import cell_index
from input_manager import COORDINATE_LABELS, Command, InputManager
from system import DEFAULT_STATE_PATH, init_system

logger = logging.getLogger("dispatch")

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy] [s]  Place an order. Without numbers you are prompted for
                           restaurant x/y and customer x/y (0-800 each).
  status              [st] Funds, rider count and every rider's state and orders.
  orders              [o]  Table of all orders with timing and assigned rider.
  riders              [r]  Full status-change log of every rider.
  map                 [m]  ASCII map with riders and active orders.
  tick [n]            [t]  Advance the simulation by n ticks (default 1).
  save [file]         [v]  Save configuration, riders and orders as JSON.
  history             [hi] Show the command history.
  help                [h]  Show this help.
  quit / exit         [q]  Leave the program.
History: '!!' repeats the last command, '!N' repeats history entry N."""


class App:
    """Interactive command loop tying together system, display and input."""

    def __init__(self) -> None:
        """Initialise the system and the user interface components."""
        init_system()
        self.system = get_value("system_manager")
        self.display = DisplayManager()
        self.inputs = InputManager()
        self.handlers = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "save": self._cmd_save,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
            "exit": self._cmd_quit,
        }

    def run(self) -> None:
        """Run the main loop until the user quits or input ends."""
        print("Food Delivery Order Dispatch System (8x8 grid, 800x800 units)")
        print("Type 'help' for commands. Funds: %d" % self.system.funds)
        while self.system.running and self.inputs.running:
            line = self.inputs.read_line("dispatch> ")
            if line is None:
                print()
                break
            if not line.strip():
                continue
            try:
                resolved = self.inputs.resolve_history(line)
                if resolved != line.strip():
                    print(f"Repeating: {resolved}")
                command = self.inputs.parse(resolved)
                self.handlers[command.name](command)
                self.inputs.record(resolved)
            except ValueError as exc:
                print(f"Error: {exc}")
            except Exception:  # keep the simulation alive on unexpected faults
                logger.exception("Unexpected error while executing a command")
                print("Unexpected internal error; system state was preserved. "
                      "Details were written to the log.")
        print("Goodbye.")

    # ------------------------------------------------------------------
    # coordinate prompting
    # ------------------------------------------------------------------
    def _prompt_coordinates(self) -> Optional[Tuple[int, int, int, int]]:
        """Interactively ask for the four coordinates.

        Returns:
            Tuple of (restaurant_x, restaurant_y, customer_x, customer_y), or
            None if the user cancelled or input ended.
        """
        values = []
        for label in COORDINATE_LABELS:
            while True:
                text = self.inputs.read_line(f"  {label} (0-800, 'cancel' to abort): ")
                if text is None or text.strip().lower() == "cancel":
                    return None
                try:
                    value = self.inputs.parse_int(text.strip(), label)
                    values.append(self.inputs.validate_coordinate(value, label))
                    break
                except ValueError as exc:
                    print(f"  Error: {exc}")
        return tuple(values)

    # ------------------------------------------------------------------
    # command handlers
    # ------------------------------------------------------------------
    def _cmd_start(self, command: Command) -> None:
        """Place an order and show the assignment and route."""
        if command.arguments:
            coords = tuple(command.arguments)
        else:
            coords = self._prompt_coordinates()
            if coords is None:
                print("Order entry cancelled.")
                return
        rx, ry, cx, cy = coords
        result = self.system.place_order((rx, ry), (cx, cy))
        order = result["order"]
        rider = result["rider"]
        route = result["route"]

        rgx, rgy = cell_index(rx, ry)
        cgx, cgy = cell_index(cx, cy)
        print(f"Order #{order.id} created at tick {order.start_time}.")
        print(f"  Restaurant: input ({rx},{ry}) -> grid cell ({rgx},{rgy}), "
              f"centre ({order.restaurant_x},{order.restaurant_y})")
        print(f"  Customer:   input ({cx},{cy}) -> grid cell ({cgx},{cgy}), "
              f"centre ({order.customer_x},{order.customer_y})")
        if result["new_riders"]:
            ids = ", ".join(str(i) for i in result["new_riders"])
            print(f"  New rider(s) added: {ids} (funds now {self.system.funds})")
        else:
            print(f"  No new rider added (funds {self.system.funds}, "
                  f"threshold {self.system.config.recruitment_threshold})")
        if rider is None:
            print("  No rider available. The order is queued and will be assigned "
                  "on the next tick.")
            return
        print(f"  Assigned to rider {rider.rider_id} at ({rider.x},{rider.y}); "
              f"speed {rider.speed} units/tick.")
        if route is None:
            print("  Route preview unavailable.")
            return
        print(self.display.render_map(riders=[rider], orders=[order], route=route))
        print(f"  Route: pickup then delivery, {route.total_distance} units, "
              f"about {route.estimated_ticks} ticks at speed {rider.speed}.")
        print(f"  Special approach (driveway): {'yes' if route.has_driveway else 'no'}")

    def _cmd_status(self, command: Command) -> None:
        """Show system funds and every rider's state and orders."""
        print(self.display.render_status(self.system))

    def _cmd_orders(self, command: Command) -> None:
        """Show all orders."""
        print(self.display.render_orders(self.system.order_manager.orders))

    def _cmd_riders(self, command: Command) -> None:
        """Show the status-change log of every rider."""
        print(self.display.render_rider_log(self.system.riders))

    def _cmd_map(self, command: Command) -> None:
        """Show the ASCII map with riders and active orders."""
        active = self.system.order_manager.active_orders()
        print(self.display.render_map(riders=self.system.riders, orders=active))

    def _cmd_tick(self, command: Command) -> None:
        """Advance the simulation."""
        count = command.arguments[0]
        for _ in range(count):
            self.system.tick()
        busy = sum(1 for r in self.system.riders if r.state != 0)
        print(f"Advanced {count} tick(s). Simulated time: {self.system.simulated_time}. "
              f"Riders: {len(self.system.riders)} (busy {busy}), "
              f"pending orders: {len(self.system.order_manager.pending)}.")

    def _cmd_save(self, command: Command) -> None:
        """Save the state to a JSON file."""
        parts = command.raw_input.split(maxsplit=1)
        path = parts[1].strip() if len(parts) > 1 else DEFAULT_STATE_PATH
        saved = self.system.save_state(path)
        print(f"State saved to {saved}")

    def _cmd_history(self, command: Command) -> None:
        """Show previously executed commands."""
        history = self.inputs.command_history
        if not history:
            print("No commands in history yet.")
            return
        for index, entry in enumerate(history, start=1):
            print(f"  {index:>4}  {entry}")

    def _cmd_help(self, command: Command) -> None:
        """Show the command reference."""
        print(HELP_TEXT)

    def _cmd_quit(self, command: Command) -> None:
        """Stop the program."""
        self.system.running = False
        self.inputs.running = False


def main(argv=None) -> int:
    """Program entry point.

    Args:
        argv: Optional list of command-line arguments.

    Returns:
        Process exit code.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--debug", action="store_true",
                        help="enable debug logging to stderr")
    args = parser.parse_args(argv)
    logging.basicConfig(
        level=logging.DEBUG if args.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        stream=sys.stderr,
    )
    app = App()
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

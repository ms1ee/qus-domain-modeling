"""Global variable store shared by all simulator modules.

The store is a simple dictionary wrapped by ``set_value`` and ``get_value``.
Well-known keys:

* ``"riders"``: ``list[Rider]`` - the dynamic rider list (CRUD operations).
* ``"orders"``: ``list[Order]`` - every order created in this session.
* ``"funds"``: ``int`` - system fund balance (may be negative).
* ``"rider_id_counter"``: ``int`` - last issued rider ID.
* ``"order_id_counter"``: ``int`` - last issued order ID.
* ``"tick"``: ``int`` - current simulated time.
* ``"running"``: ``bool`` - main loop control flag.
"""

from typing import Any, Dict

_store: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store to its default values.

    Riders list is empty, rider ID counter is 0, the system is running and
    the simulated clock starts at tick 0.
    """
    global _store
    _store = {
        "riders": [],
        "rider_id_counter": 0,
        "order_id_counter": 0,
        "orders": [],
        "funds": 0,
        "tick": 0,
        "running": True,
    }


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Value to store.
    """
    _store[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` if the key is missing.
    """
    return _store.get(key, default)


def next_counter(key: str) -> int:
    """Increment an integer counter stored under ``key`` and return it.

    Args:
        key: Name of the counter variable.

    Returns:
        The new (incremented) counter value, starting at 1.
    """
    value = int(_store.get(key, 0)) + 1
    _store[key] = value
    return value

"""System parameters and JSON persistence helpers.

Parameters can be overridden by a JSON configuration file. Unknown keys are
ignored with a warning. The current state (funds, riders, orders) can be
written to a JSON state file on exit.
"""

import json
import logging
import os
from typing import Any, Dict

logger = logging.getLogger(__name__)

DEFAULT_CONFIG: Dict[str, Any] = {
    "initial_funds": 300,
    "recruit_threshold": 300,
    "rider_cost": 300,
    "order_revenue": 100,
    "rider_speed": 50,
    "max_orders_per_rider": 3,
    "load_penalty": 150,
    "late_threshold_ticks": 60,
    "late_penalty": 20,
    "max_wait_ticks": 3,
    "state_file": "dispatch_state.json",
    "log_file": "dispatch.log",
}

# Active configuration; modules read values from here at use time.
CONFIG: Dict[str, Any] = dict(DEFAULT_CONFIG)

_POSITIVE_KEYS = ("rider_speed", "max_orders_per_rider")


def load_config(path: str) -> bool:
    """Load parameter overrides from a JSON configuration file.

    The file may either be a flat JSON object or contain the overrides under
    a ``"parameters"`` key.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        True if the file existed and was applied, False if it was absent.

    Raises:
        ValueError: If the JSON is malformed or a value has an invalid type.
        OSError: If the file cannot be read.
    """
    if not os.path.exists(path):
        return False
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError("configuration file must contain a JSON object")
    params = data.get("parameters", data)
    if not isinstance(params, dict):
        raise ValueError("'parameters' must be a JSON object")

    for key, value in params.items():
        if key not in DEFAULT_CONFIG:
            logger.warning("Ignoring unknown configuration key '%s'", key)
            continue
        default = DEFAULT_CONFIG[key]
        if isinstance(default, int):
            if isinstance(value, bool) or not isinstance(value, int):
                raise ValueError(f"'{key}' must be an integer")
            if value < 0 or (key in _POSITIVE_KEYS and value < 1):
                raise ValueError(f"'{key}' is out of the allowed range")
        elif isinstance(default, str) and not isinstance(value, str):
            raise ValueError(f"'{key}' must be a string")
        CONFIG[key] = value
    logger.info("Configuration loaded from %s", path)
    return True


def save_state(path: str, state: Dict[str, Any]) -> None:
    """Write the system state to a JSON file atomically.

    Args:
        path: Destination file path.
        state: JSON-serialisable dictionary describing the system state.

    Raises:
        OSError: If the file cannot be written.
    """
    tmp_path = path + ".tmp"
    with open(tmp_path, "w", encoding="utf-8") as handle:
        json.dump(state, handle, indent=2)
    os.replace(tmp_path, path)
    logger.info("State saved to %s", path)

"""Grid geometry helpers for the 8x8 food delivery map.

The map is an 800x800 coordinate system divided into 8x8 cells of 100x100
units. Streets run along the grid lines (x or y multiple of 100).
Cell centers are located at ``cell * 100 + 50``.
"""

from typing import Tuple

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = 8
CELL_CENTER_OFFSET = 50
COORD_MIN = 0
COORD_MAX = MAP_SIZE

Point = Tuple[int, int]


def validate_coordinate(value: object, name: str = "coordinate") -> int:
    """Validate that ``value`` is an integer within the map range.

    Args:
        value: The value to validate.
        name: Human readable name used in error messages.

    Returns:
        The validated integer.

    Raises:
        ValueError: If the value is not an integer or is out of range.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError(f"{name} must be an integer, got {value!r}")
    if value < COORD_MIN or value > COORD_MAX:
        raise ValueError(
            f"{name} must be between {COORD_MIN} and {COORD_MAX}, got {value}"
        )
    return value


def cell_index(value: int) -> int:
    """Return the grid cell index (0-7) containing a coordinate.

    Coordinates on the far edge (800) are clamped into the last cell.

    Args:
        value: A coordinate in units.

    Returns:
        The cell index along one axis.
    """
    index = value // CELL_SIZE
    return min(max(index, 0), GRID_DIM - 1)


def to_grid_center(x: int, y: int) -> Point:
    """Convert a map coordinate to the center of its grid cell.

    Uses ``center = (cell // 100) * 100 + 50`` on each axis.

    Args:
        x: X coordinate in units.
        y: Y coordinate in units.

    Returns:
        The ``(center_x, center_y)`` tuple.
    """
    return (
        cell_index(x) * CELL_SIZE + CELL_CENTER_OFFSET,
        cell_index(y) * CELL_SIZE + CELL_CENTER_OFFSET,
    )


def manhattan(a: Point, b: Point) -> int:
    """Return the Manhattan distance between two points.

    Args:
        a: First point.
        b: Second point.

    Returns:
        The sum of absolute coordinate differences.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def is_street_x(x: int) -> bool:
    """Return True if a vertical line at ``x`` is a street."""
    return x % CELL_SIZE == 0


def is_street_y(y: int) -> bool:
    """Return True if a horizontal line at ``y`` is a street."""
    return y % CELL_SIZE == 0

"""Grid-based path planning for riders.

Paths are built from axis-aligned segments (right-angle turns only). When
two L-shaped routes are possible, the planner prefers the one whose segments
lie on streets (grid lines). A target lying directly above or below (or left
or right of) the rider is reached by a straight-line movement.
"""

import logging
from typing import List

from grid import Point, is_street_x, is_street_y, manhattan

logger = logging.getLogger(__name__)


class PathPlanner:
    """Compute waypoint routes on the 800x800 grid."""

    def plan(self, start: Point, goal: Point) -> List[Point]:
        """Plan a route from ``start`` to ``goal``.

        The returned list excludes ``start`` and always ends with ``goal``
        (unless the two points are identical, in which case it is empty).

        Args:
            start: Current position.
            goal: Destination position.

        Returns:
            Ordered list of waypoints to visit.
        """
        if start == goal:
            return []
        start_x, start_y = start
        goal_x, goal_y = goal
        # Special case: target directly above/below or left/right.
        if start_x == goal_x or start_y == goal_y:
            return [goal]
        corners = [(goal_x, start_y), (start_x, goal_y)]
        best = max(
            corners,
            key=lambda corner: self._street_score(start, corner, goal),
        )
        logger.debug("Planned route %s -> %s via %s", start, goal, best)
        return [best, goal]

    @classmethod
    def _segment_on_street(cls, a: Point, b: Point) -> bool:
        """Return True if the axis-aligned segment ``a``-``b`` is a street."""
        if a[1] == b[1]:
            return is_street_y(a[1])
        if a[0] == b[0]:
            return is_street_x(a[0])
        return False

    def _street_score(self, start: Point, corner: Point, goal: Point) -> int:
        """Score a candidate corner by the number of street segments."""
        return int(self._segment_on_street(start, corner)) + int(
            self._segment_on_street(corner, goal)
        )

    @staticmethod
    def path_length(points: List[Point]) -> int:
        """Return the total Manhattan length of a polyline.

        Args:
            points: Ordered points, including the starting point.

        Returns:
            Total travelled distance in units.
        """
        total = 0
        for first, second in zip(points, points[1:]):
            total += manhattan(first, second)
        return total

    @staticmethod
    def is_on_street(point: Point) -> bool:
        """Return True if the point lies on a street (grid line)."""
        return is_street_x(point[0]) or is_street_y(point[1])

"""Order data model.

An order is created from user input ``[restaurant_x, restaurant_y,
customer_x, customer_y]`` (each 0-800). Coordinates are converted to grid
cell centers for routing.
"""

from typing import Any, Dict, Optional, Sequence

import globalvar
from grid import (
    Point,
    cell_index,
    manhattan,
    to_grid_center,
    validate_coordinate,
)

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKED_UP = "picked_up"
STATUS_COMPLETED = "completed"


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        res_pos: Raw restaurant coordinate as entered by the user.
        cus_pos: Raw customer coordinate as entered by the user.
        restaurant_cell / customer_cell: Grid cell indices (column, row).
        restaurant_center / customer_center: Routing targets (cell centers).
        distance: Manhattan distance between the two centers.
        status: One of "pending", "assigned", "picked_up", "completed".
        rider_id: ID of the assigned rider, or None.
        start_time: Tick when the order was created.
        assigned_time / pickup_time / completed_time: Event ticks.
        time_taken: Ticks from creation to completion.
        efficiency: Distance covered per tick (units/tick).
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: ``(x, y)`` restaurant coordinate, each 0-800.
            cus_pos: ``(x, y)`` customer coordinate, each 0-800.

        Raises:
            ValueError: If a coordinate is invalid.
        """
        rx, ry = self._parse_point(res_pos, "restaurant")
        cx, cy = self._parse_point(cus_pos, "customer")
        self.id: int = globalvar.next_counter("order_id_counter")
        self.res_pos: Point = (rx, ry)
        self.cus_pos: Point = (cx, cy)
        self.restaurant_cell: Point = (cell_index(rx), cell_index(ry))
        self.customer_cell: Point = (cell_index(cx), cell_index(cy))
        self.restaurant_center: Point = to_grid_center(rx, ry)
        self.customer_center: Point = to_grid_center(cx, cy)
        self.distance: int = manhattan(
            self.restaurant_center, self.customer_center
        )
        self.status: str = STATUS_PENDING
        self.rider_id: Optional[int] = None
        self.start_time: int = int(globalvar.get_value("tick", 0))
        self.assigned_time: Optional[int] = None
        self.pickup_time: Optional[int] = None
        self.completed_time: Optional[int] = None
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None

    @staticmethod
    def _parse_point(point: Sequence[int], label: str) -> Point:
        """Validate a two-element coordinate sequence.

        Args:
            point: Sequence containing x and y.
            label: Name used in error messages.

        Returns:
            Validated ``(x, y)`` tuple.

        Raises:
            ValueError: If the sequence length or values are invalid.
        """
        if len(point) != 2:
            raise ValueError(f"{label} position must have exactly 2 values")
        x = validate_coordinate(point[0], f"{label} x")
        y = validate_coordinate(point[1], f"{label} y")
        return (x, y)

    def assign_to(self, rider_id: int) -> None:
        """Record that the order has been assigned to a rider.

        The caller is responsible for appending the order to the rider's
        ``orders`` list.

        Args:
            rider_id: ID of the rider receiving the order.

        Raises:
            RuntimeError: If the order is not pending.
        """
        if self.status != STATUS_PENDING:
            raise RuntimeError(
                f"order {self.id} cannot be assigned from status {self.status}"
            )
        self.rider_id = rider_id
        self.assigned_time = int(globalvar.get_value("tick", 0))
        self.status = STATUS_ASSIGNED

    def mark_picked_up(self, time: int) -> None:
        """Mark that the rider reached the restaurant and collected the food.

        Args:
            time: Simulated tick of the pickup.
        """
        self.status = STATUS_PICKED_UP
        self.pickup_time = time

    def complete(self, time: int) -> None:
        """Mark the order as delivered and compute efficiency metrics.

        Args:
            time: Simulated tick of the delivery.
        """
        self.status = STATUS_COMPLETED
        self.completed_time = time
        self.time_taken = max(0, time - self.start_time)
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = float(self.distance)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "status": self.status,
            "restaurant_input": list(self.res_pos),
            "customer_input": list(self.cus_pos),
            "restaurant_center": list(self.restaurant_center),
            "customer_center": list(self.customer_center),
            "distance": self.distance,
            "start_time": self.start_time,
            "assigned_time": self.assigned_time,
            "pickup_time": self.pickup_time,
            "completed_time": self.completed_time,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
            "rider_id": self.rider_id,
        }

"""Rider model: state machine, movement and order execution.

Rider states:
    0 - idle (available for new orders)
    1 - picking up (en route to the restaurant)
    2 - delivering (en route to the customer)

Transitions:
    idle -> picking up   when an order is appended to ``orders``
    picking up -> delivering  on arrival at the restaurant
    delivering -> idle   on delivery (or -> picking up for the next order)
"""

import logging
from typing import Any, Dict, List, Optional, Tuple

import config
import globalvar
from grid import Point
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

# Depot where every newly recruited rider starts (a street intersection).
DEPOT: Point = (0, 0)


class Rider:
    """A delivery rider moving on the street grid.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates in units (0-800).
        speed: Movement in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Assigned orders; ``orders[0]`` is the one being executed.
        special_case: True while the rider is waiting for a blocked cell.
        route: Remaining waypoints for the current target.
        log: Chronological list of event dictionaries.
    """

    def __init__(
        self,
        rider_id: int,
        x: int = DEPOT[0],
        y: int = DEPOT[1],
        speed: Optional[int] = None,
    ) -> None:
        """Create a rider and record its spawn event.

        Args:
            rider_id: Unique rider identifier.
            x: Initial x coordinate (defaults to the depot).
            y: Initial y coordinate (defaults to the depot).
            speed: Units per tick; defaults to the configured rider speed.

        Raises:
            ValueError: If the speed is not positive.
        """
        if speed is None:
            speed = int(config.CONFIG["rider_speed"])
        if speed <= 0:
            raise ValueError("rider speed must be positive")
        self.rider_id: int = rider_id
        self.x: int = x
        self.y: int = y
        self.speed: int = speed
        self.state: int = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.route: List[Point] = []
        self.wait_ticks: int = 0
        self.completed_count: int = 0
        self.log: List[Dict[str, Any]] = []
        self._planner = PathPlanner()
        self._record("spawn", None)
        logger.info(
            "Rider %d spawned at (%d, %d)", self.rider_id, self.x, self.y
        )

    @property
    def position(self) -> Point:
        """Current ``(x, y)`` position."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Human readable name of the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def move(self) -> Optional[Any]:
        """Advance the rider by one tick.

        Handles state transitions, movement along the planned route, arrival
        detection, and order completion.

        Returns:
            The completed Order if a delivery finished during this tick,
            otherwise None.
        """
        now = self._now()
        if self.state == STATE_IDLE and self.orders:
            self._start_pickup(now)
        if not self.orders:
            if self.state != STATE_IDLE:
                self._become_idle()
            return None

        order = self.orders[0]
        target = self._target_for_state(order)
        if self.position == target:
            return self._on_arrival(order, now)
        if not self.route:
            self.route = self._planner.plan(self.position, target)
        if not self._travel():
            return None
        if self.position == target:
            return self._on_arrival(order, now)
        return None

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
            "completed_count": self.completed_count,
            "log": list(self.log),
        }

    def _now(self) -> int:
        """Return the current simulated tick."""
        return int(globalvar.get_value("tick", 0))

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log."""
        self.log.append(
            {
                "event": event,
                "order_id": order_id,
                "time": self._now(),
                "rider_id": self.rider_id,
                "x": self.x,
                "y": self.y,
                "state": self.state,
            }
        )

    def _target_for_state(self, order: Any) -> Point:
        """Return the destination matching the current state."""
        if self.state == STATE_PICKING_UP:
            return order.restaurant_center
        return order.customer_center

    def _start_pickup(self, now: int) -> None:
        """Transition idle -> picking up for the first order."""
        self.state = STATE_PICKING_UP
        self.route = []
        self.wait_ticks = 0
        order = self.orders[0]
        self._record("accepted_order", order.id)
        logger.info(
            "Tick %d: rider %d accepted order %d",
            now,
            self.rider_id,
            order.id,
        )

    def _become_idle(self) -> None:
        """Return the rider to the idle state."""
        self.state = STATE_IDLE
        self.route = []
        self.special_case = False
        self.wait_ticks = 0

    def _on_arrival(self, order: Any, now: int) -> Optional[Any]:
        """Handle arrival at the current target.

        Returns:
            The order if it was delivered, otherwise None.
        """
        self.route = []
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.mark_picked_up(now)
            self._record("arrived_at_merchant", order.id)
            logger.info(
                "Tick %d: rider %d picked up order %d",
                now,
                self.rider_id,
                order.id,
            )
            return None
        if self.state == STATE_DELIVERING:
            order.complete(now)
            self._record("delivered_to_customer", order.id)
            logger.info(
                "Tick %d: rider %d delivered order %d",
                now,
                self.rider_id,
                order.id,
            )
            self.orders.pop(0)
            self.completed_count += 1
            if self.orders:
                self._start_pickup(now)
            else:
                self._become_idle()
            return order
        return None

    def _travel(self) -> bool:
        """Move up to ``speed`` units along the planned route.

        Movement is along the axis of the next waypoint. If the destination
        cell is occupied by another rider, the rider waits for a few ticks
        (special case) before proceeding anyway to break deadlocks.

        Returns:
            True if the rider moved (or the deadlock breaker fired),
            False if the rider is waiting.
        """
        x, y = self.x, self.y
        route = list(self.route)
        budget = self.speed
        while budget > 0 and route:
            wx, wy = route[0]
            if (x, y) == (wx, wy):
                route.pop(0)
                continue
            dx = wx - x
            dy = wy - y
            if dx != 0:
                step = min(budget, abs(dx))
                x += step if dx > 0 else -step
            else:
                step = min(budget, abs(dy))
                y += step if dy > 0 else -step
            budget -= step

        if (x, y) != self.position and self._occupied((x, y)):
            self.special_case = True
            self.wait_ticks += 1
            if self.wait_ticks <= int(config.CONFIG["max_wait_ticks"]):
                logger.debug(
                    "Rider %d waiting for occupied cell %s",
                    self.rider_id,
                    (x, y),
                )
                return False
            logger.warning(
                "Rider %d breaking possible deadlock at %s",
                self.rider_id,
                (x, y),
            )

        self.wait_ticks = 0
        self.special_case = False
        self.x, self.y = x, y
        self.route = route
        return True

    def _occupied(self, point: Point) -> bool:
        """Return True if another rider currently occupies ``point``."""
        riders = globalvar.get_value("riders", []) or []
        return any(
            other is not self and other.position == point for other in riders
        )


def describe_position(rider: Rider) -> Tuple[int, int]:
    """Return the rider position as a tuple (helper for display code)."""
    return rider.position

"""Order assignment logic and the order manager.

Assignment strategy: closest eligible rider first, with a load penalty per
already-assigned order so work is spread among riders (competitive
assignment for load balancing).
"""

import logging
from typing import List, Optional

import config
import globalvar
from grid import manhattan
from order_model import STATUS_PENDING, Order
from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Find the best rider for an order.

    Score = distance from the rider to the restaurant
            + load_penalty * number of orders already held.
    Riders already holding ``max_orders_per_rider`` orders are skipped.
    Ties are broken by the lower rider ID.

    Args:
        order: The order needing a rider.

    Returns:
        The best Rider, or None if no rider is eligible.
    """
    max_orders = int(config.CONFIG["max_orders_per_rider"])
    penalty = int(config.CONFIG["load_penalty"])
    best: Optional[Rider] = None
    best_score: Optional[int] = None
    for rider in globalvar.get_value("riders", []) or []:
        if len(rider.orders) >= max_orders:
            continue
        distance = manhattan(rider.position, order.restaurant_center)
        score = distance + penalty * len(rider.orders)
        if (
            best is None
            or score < best_score
            or (score == best_score and rider.rider_id < best.rider_id)
        ):
            best = rider
            best_score = score
    return best

        pending: Orders waiting for an eligible rider.

    def __init__(self) -> None:
        """Create an empty order manager."""

        """Create an order and register it in the global order history.

        Args:
            res_pos: Restaurant ``(x, y)`` input coordinate.
            cus_pos: Customer ``(x, y)`` input coordinate.

        Returns:
            The newly created Order.

        Raises:
            ValueError: If a coordinate is invalid.
        """
        order = Order(res_pos, cus_pos)
        globalvar.get_value("orders").append(order)
        logger.info("Created order %d", order.id)
        return order

    def try_assign(self, order: Order) -> Optional[Rider]:
        """Attempt to assign an order to the best available rider.

        If no rider is eligible, the order is kept in the pending queue.

        Args:
            order: The order to assign.

        Returns:
            The rider that received the order, or None.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            if order not in self.pending and order.status == STATUS_PENDING:
                self.pending.append(order)
            logger.info("Order %d queued (no rider available)", order.id)
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        if order in self.pending:
            self.pending.remove(order)
        logger.info("Order %d assigned to rider %d", order.id, rider.rider_id)
        return rider

    def assign_pending(self) -> List[Order]:
        """Try to assign every pending order.

        Returns:
            The list of orders that were assigned during this call.
        """
        assigned: List[Order] = []
        for order in list(self.pending):
            if self.try_assign(order) is not None:
                assigned.append(order)
        return assigned

"""Core system manager: funds, rider recruitment, order flow and ticks.

``init_system`` sets up the global variables, the initial funds and the
first rider. The ``SystemManager`` coordinates orders, riders and the
simulated clock.
"""

import logging
import math
from typing import Any, Dict, List, Optional, Sequence

import config
import globalvar
from grid import Point
from order import OrderManager
from order_model import (
    STATUS_ASSIGNED,
    STATUS_COMPLETED,
    STATUS_PENDING,
    STATUS_PICKED_UP,
    Order,
)
from path_planner import PathPlanner
from rider import DEPOT, STATE_DELIVERING, STATE_IDLE, STATE_PICKING_UP, Rider

logger = logging.getLogger(__name__)

_manager: Optional["SystemManager"] = None


class SystemManager:
    """Manages funds, riders, orders and the simulated clock."""

    def __init__(self) -> None:
        """Create the manager with its order manager and path planner."""
        self.order_manager = OrderManager()
        self.planner = PathPlanner()

    # ------------------------------------------------------------------
    # Funds, riders and state accessors
    # ------------------------------------------------------------------
    @property
    def funds(self) -> int:
        """Current system fund balance."""
        return int(globalvar.get_value("funds", 0))

    @property
    def tick_count(self) -> int:
        """Current simulated tick."""
        return int(globalvar.get_value("tick", 0))

    @property
    def riders(self) -> List[Rider]:
        """The dynamic rider list."""
        return globalvar.get_value("riders", [])

    def all_orders(self) -> List[Order]:
        """Return every order created in this session."""
        return list(globalvar.get_value("orders", []) or [])

    def active_orders(self) -> List[Order]:
        """Return orders that are not yet completed."""
        return [o for o in self.all_orders() if o.status != STATUS_COMPLETED]

    def add_funds(self, amount: int) -> None:
        """Add ``amount`` (may be negative) to the fund balance."""
        globalvar.set_value("funds", self.funds + int(amount))

    def create_rider(self) -> Rider:
        """Create a new rider at the depot and add it to the rider list."""
        rider_id = globalvar.next_counter("rider_id_counter")
        rider = Rider(rider_id, DEPOT[0], DEPOT[1])
        self.riders.append(rider)
        return rider

    def check_recruitment(self) -> Optional[Rider]:
        """Recruit a rider if funds reach the recruitment threshold.

        Recruiting costs ``rider_cost`` from the fund balance.

        Returns:
            The new Rider, or None if no recruitment happened.
        """
        if self.funds >= int(config.CONFIG["recruit_threshold"]):
            self.add_funds(-int(config.CONFIG["rider_cost"]))
            rider = self.create_rider()
            logger.info("Recruited rider %d", rider.rider_id)
            return rider
        return None

    # ------------------------------------------------------------------
    # Order flow
    # ------------------------------------------------------------------
    def build_route(self, order: Order, rider: Rider) -> List[Point]:
        """Build the full route rider -> restaurant -> customer.

        Args:
            order: The order being planned.
            rider: The rider that will execute it.

        Returns:
            List of points starting at the rider's position.
        """
        start = rider.position
        route: List[Point] = [start]
        route.extend(self.planner.plan(start, order.restaurant_center))
        route.extend(
            self.planner.plan(order.restaurant_center, order.customer_center)
        )
        return route

    def process_order(
        self, res_pos: Sequence[int], cus_pos: Sequence[int]
    ) -> Dict[str, Any]:
        """Handle a ``start`` request: create, fund, recruit and assign.

        Args:
            res_pos: Restaurant ``(x, y)`` input coordinate.
            cus_pos: Customer ``(x, y)`` input coordinate.

        Returns:
            Dictionary with keys ``order``, ``rider`` (assigned or None),
            ``new_rider`` (recruited Rider or None), ``route``, ``speed``,
            ``route_length``, ``eta_ticks`` and ``queue_ahead``.

        Raises:
            ValueError: If a coordinate is invalid.
        """
        order = self.order_manager.create_order(res_pos, cus_pos)
        self.add_funds(int(config.CONFIG["order_revenue"]))
        new_rider = self.check_recruitment()
        rider = self.order_manager.try_assign(order)
        route: List[Point] = []
        length = 0
        eta: Optional[int] = None
        speed: Optional[int] = None
        queue_ahead = 0
        if rider is not None:
            route = self.build_route(order, rider)
            length = PathPlanner.path_length(route)
            speed = rider.speed
            eta = math.ceil(length / speed)
            queue_ahead = len(rider.orders) - 1
        return {
            "order": order,
            "rider": rider,
            "new_rider": new_rider,
            "route": route,
            "speed": speed,
            "route_length": length,
            "eta_ticks": eta,
            "queue_ahead": queue_ahead,
        }

    def tick(self) -> List[Order]:
        """Advance the simulation by one unit of time.

        Every rider moves once, completed deliveries are settled, pending
        orders are retried and recruitment is re-checked.

        Returns:
            Orders completed during this tick.
        """
        now = self.tick_count + 1
        globalvar.set_value("tick", now)
        completed: List[Order] = []
        for rider in list(self.riders):
            try:
                done = rider.move()
            except Exception:  # pylint: disable=broad-except
                logger.exception("Rider %d failed to move", rider.rider_id)
                continue
            if done is not None:
                self._on_order_completed(done)
                completed.append(done)
        self.order_manager.assign_pending()
        self.check_recruitment()
        return completed

    def _on_order_completed(self, order: Order) -> None:
        """Apply late-delivery penalties and log the completion."""
        late = int(config.CONFIG["late_threshold_ticks"])
        if order.time_taken is not None and order.time_taken > late:
            self.add_funds(-int(config.CONFIG["late_penalty"]))
            logger.info("Order %d was late (%d ticks)", order.id,
                        order.time_taken)

    # ------------------------------------------------------------------
    # Consistency checks and recovery
    # ------------------------------------------------------------------
    def verify_state(self) -> List[str]:
        """Check internal consistency and return a list of problems."""
        problems: List[str] = []
        rider_ids = {rider.rider_id for rider in self.riders}
        if len(rider_ids) != len(self.riders):
            problems.append("duplicate rider IDs detected")
        for rider in self.riders:
            if rider.state == STATE_IDLE and rider.orders:
                problems.append(
                    f"rider {rider.rider_id} is idle but holds orders"
                )
            if rider.state != STATE_IDLE and not rider.orders:
                problems.append(
                    f"rider {rider.rider_id} is busy without orders"
                )
            for order in rider.orders:
                if order.status == STATUS_COMPLETED:
                    problems.append(
                        f"rider {rider.rider_id} holds completed order "
                        f"{order.id}"
                    )
        for order in self.all_orders():
            if order.status in (STATUS_ASSIGNED, STATUS_PICKED_UP):
                if order.rider_id not in rider_ids:
                    problems.append(
                        f"order {order.id} references missing rider "
                        f"{order.rider_id}"
                    )
        return problems

    def repair_state(self) -> List[str]:
        """Repair inconsistencies found by ``verify_state``.

        Returns:
            Human readable descriptions of the fixes applied.
        """
        fixes: List[str] = []
        rider_ids = {rider.rider_id: rider for rider in self.riders}
        for rider in self.riders:
            active = [o for o in rider.orders if o.status != STATUS_COMPLETED]
            if len(active) != len(rider.orders):
                rider.orders = active
                fixes.append(f"removed completed orders from rider "
                             f"{rider.rider_id}")
            if rider.orders:
                expected = (
                    STATE_DELIVERING
                    if rider.orders[0].status == STATUS_PICKED_UP
                    else STATE_PICKING_UP
                )
                if rider.state != expected:
                    rider.state = expected
                    rider.route = []
                    fixes.append(f"reset state of rider {rider.rider_id}")
            elif rider.state != STATE_IDLE:
                rider.state = STATE_IDLE
                rider.route = []
                fixes.append(f"set rider {rider.rider_id} to idle")
        for order in self.all_orders():
            if order.status not in (STATUS_ASSIGNED, STATUS_PICKED_UP):
                continue
            holder = rider_ids.get(order.rider_id)
            if holder is None or order not in holder.orders:
                order.status = STATUS_PENDING
                order.rider_id = None
                if order not in self.order_manager.pending:
                    self.order_manager.pending.append(order)
                fixes.append(f"re-queued order {order.id}")
        self.order_manager.pending = [
            o for o in self.order_manager.pending
            if o.status == STATUS_PENDING
        ]
        self.order_manager.assign_pending()
        return fixes

    def export_state(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the whole system."""
        return {
            "tick": self.tick_count,
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "order_id_counter": globalvar.get_value("order_id_counter", 0),
            "config": dict(config.CONFIG),
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in self.all_orders()],
        }


def init_system() -> None:
    """Initialise global variables and the system manager.

    Sets the initial funds, the clock and the empty rider and order lists,
    then performs the automatic recruitment check (which spawns the first
    rider when the initial funds reach the threshold).
    """
    global _manager
    globalvar._init()  # pylint: disable=protected-access
    globalvar.set_value("funds", int(config.CONFIG["initial_funds"]))
    globalvar.set_value("tick", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("riders", [])
    globalvar.set_value("orders", [])
    _manager = SystemManager()
    globalvar.set_value("manager", _manager)
    _manager.check_recruitment()
    logger.info("System initialised with funds %d", _manager.funds)


def get_manager() -> SystemManager:
    """Return the active SystemManager, initialising the system if needed."""
    if _manager is None:
        init_system()
    assert _manager is not None
    return _manager

cell centers at odd indices. Each canvas character represents 50 units, so
``canvas_index = coordinate // 50``.

Symbols: 'R' rider, 'S' restaurant, 'C' customer, '.' empty cell,
'#' street, '*' planned route.
"""

from typing import Iterable, List, Optional, Sequence, Tuple

from grid import CELL_SIZE, GRID_DIM
from order_model import STATUS_COMPLETED, Order
from rider import Rider

CANVAS_SIZE = 2 * GRID_DIM + 1
UNIT = CELL_SIZE // 2
Cell = Tuple[int, int]


class DisplayManager:
    """Renders the map, status snapshots and order/rider reports."""

    # ------------------------------------------------------------------
    # Map rendering
    # ------------------------------------------------------------------
    @staticmethod
    def _blank_canvas() -> List[List[str]]:
        """Create the empty street-grid canvas."""
        return [
            [
                "#" if (row % 2 == 0 or col % 2 == 0) else "."
                for col in range(CANVAS_SIZE)
            ]
            for row in range(CANVAS_SIZE)
        ]

    @staticmethod
    def _to_cell(point: Tuple[int, int]) -> Cell:
        """Convert a map point to canvas ``(row, col)``."""
        x, y = point
        col = min(x // UNIT, CANVAS_SIZE - 1)
        row = min(y // UNIT, CANVAS_SIZE - 1)
        return (row, col)

    @staticmethod
    def _span(start: int, end: int) -> List[int]:
        """Inclusive integer range from ``start`` to ``end`` in either order."""
        step = 1 if end >= start else -1
        return list(range(start, end + step, step))

    def _draw_route(self, canvas: List[List[str]], route: Sequence) -> None:
        """Draw the route as a connected line with '*' characters."""
        for first, second in zip(route, route[1:]):
            row1, col1 = self._to_cell(first)
            row2, col2 = self._to_cell(second)
            for col in self._span(col1, col2):
                self._mark_route(canvas, row1, col)
            for row in self._span(row1, row2):
                self._mark_route(canvas, row, col2)

    @staticmethod
    def _mark_route(canvas: List[List[str]], row: int, col: int) -> None:
        """Mark a route cell unless a symbol already occupies it."""
        if canvas[row][col] in ("#", "."):
            canvas[row][col] = "*"

    def render_map(
        self,
        riders: Iterable[Rider],
        orders: Iterable[Order] = (),
        route: Optional[Sequence] = None,
    ) -> str:
        """Render the 8x8 ASCII map.

        Args:
            riders: Riders to display as 'R'.
            orders: Orders whose restaurant ('S') and customer ('C')
                locations should be shown.
            route: Optional list of points drawn as a connected line.

        Returns:
            Multi-line string containing the map and a legend.
        """
        canvas = self._blank_canvas()
        if route:
            self._draw_route(canvas, route)
        for order in orders:
            row, col = self._to_cell(order.restaurant_center)
            canvas[row][col] = "S"
            row, col = self._to_cell(order.customer_center)
            canvas[row][col] = "C"
        for rider in riders:
            row, col = self._to_cell(rider.position)
            canvas[row][col] = "R"
        lines = ["".join(row) for row in canvas]
        lines.append(
            "Legend: R=Rider S=Restaurant C=Customer #=Street "
            "*=Route .=Empty (origin top-left, x->right, y->down)"
        )
        return "\n".join(lines)

    # ------------------------------------------------------------------
    # Text reports
    # ------------------------------------------------------------------
    @staticmethod
    def rider_line(rider: Rider) -> str:
        """Return a one-line summary of a rider."""
        order_ids = [str(o.id) for o in rider.orders] or ["-"]

"""Command parsing, shortcut resolution, history and input validation."""

import logging
import re
from typing import Callable, List, Optional, Sequence, Tuple

from grid import validate_coordinate

logger = logging.getLogger(__name__)

COMMAND_ALIASES = {
    "start": "start",
    "s": "start",
    "status": "status",
    "st": "status",
    "orders": "orders",
    "o": "orders",
    "riders": "riders",
    "r": "riders",
    "map": "map",
    "m": "map",
    "tick": "tick",
    "t": "tick",
    "help": "help",
    "h": "help",
    "?": "help",
    "quit": "quit",
    "exit": "quit",
    "q": "quit",
}

COORDINATE_NAMES = (
    "restaurant x",
    "restaurant y",
    "customer x",
    "customer y",
)

_INTEGER_PATTERN = re.compile(r"[+-]?\d+")

try:  # readline is optional and only available on some platforms
    import readline  # pylint: disable=unused-import
    _READLINE = readline
except ImportError:  # pragma: no cover - platform dependent
    _READLINE = None


class InputManager:
    """Reads user commands and validates their arguments.

    Attributes:
        prompt: Text shown before each command.
        history: Previously entered commands (newest last).
    """

    def __init__(
        self,
        prompt: str = "dispatch> ",
        input_func: Callable[[str], str] = input,
    ) -> None:
        """Create the input manager.

        Args:
            prompt: Command prompt text.
            input_func: Function used to read a line (injectable for tests).
        """
        self.prompt = prompt
        self.history: List[str] = []
        self._input = input_func
        if _READLINE is not None:
            try:
                _READLINE.set_history_length(200)
            except (AttributeError, OSError):  # pragma: no cover
                pass

    def read_command(self) -> str:
        """Read one command line, handling the ``!!`` repeat shortcut.

        Returns:
            The stripped command text (may be empty).

        Raises:
            EOFError: If input is closed.
            ValueError: If ``!!`` is used with an empty history.
        """
        line = self._input(self.prompt).strip()
        if line == "!!":
            if not self.history:
                raise ValueError("no previous command to repeat")
            line = self.history[-1]
            print(f"Repeating: {line}")
        if line:
            self.history.append(line)
            if _READLINE is not None:
                try:
                    _READLINE.add_history(line)
                except (AttributeError, OSError):  # pragma: no cover
                    pass
        return line

    @staticmethod
    def parse_command(line: str) -> Tuple[str, List[str]]:
        """Split a command line into a canonical command and arguments.

        Args:
            line: Raw command text.

        Returns:
            Tuple ``(command, arguments)``.

        Raises:
            ValueError: If the command is unknown.
        """
        parts = line.split()
        if not parts:
            raise ValueError("empty command")
        command = COMMAND_ALIASES.get(parts[0].lower())
        if command is None:
            raise ValueError(f"unknown command '{parts[0]}'")
        return command, parts[1:]

    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> List[int]:
        """Parse and validate four integer coordinates.

        Accepts whitespace, commas or brackets as separators, for example
        ``100 200 300 400`` or ``[100, 200, 300, 400]``.

        Args:
            tokens: Text pieces containing the coordinates.

        Returns:
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            ValueError: If the count, type or range is invalid.
        """
        text = " ".join(tokens)
        for separator in "[],":
            text = text.replace(separator, " ")
        parts = text.split()
        if len(parts) != 4:
            raise ValueError(
                "expected 4 integer coordinates (restaurant x y, customer "
                f"x y), got {len(parts)} value(s)"
            )
        values: List[int] = []
        for part, name in zip(parts, COORDINATE_NAMES):
            if not _INTEGER_PATTERN.fullmatch(part):
                raise ValueError(f"{name} must be an integer, got '{part}'")
            values.append(validate_coordinate(int(part), name))
        return values

    def request_order_coordinates(self) -> Optional[List[int]]:
        """Interactively ask for order coordinates until valid or cancelled.

        Returns:
            The four validated coordinates, or None if the user cancelled
            with an empty line.
        """
        prompt = (
            "Enter restaurant x y and customer x y (0-800), blank to cancel: "
        )
        while True:
            text = self._input(prompt).strip()
            if not text:
                return None
            try:
                return self.parse_coordinates(text.split())
            except ValueError as exc:
                print(f"Invalid input: {exc}. Please try again.")

"""Command-line entry point for the food delivery dispatch simulator.

Run with::

    python src/main.py [--config config.json] [--state dispatch_state.json]
                       [--debug] [--no-save]

Each loop iteration reads one command, executes it and then advances the
simulated clock by one tick.
"""

import argparse
import logging
import sys
from typing import List, Optional

import config
import globalvar
from display import DisplayManager
from input_manager import InputManager
from system import SystemManager, get_manager, init_system

logger = logging.getLogger("dispatch")

BANNER = (
    "Food Delivery Order Dispatch System\n"
    "8x8 grid, 800x800 units. Type 'help' for commands."
)
MAX_TICK_COMMAND = 1000
CONSISTENCY_CHECK_INTERVAL = 1


def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(description=BANNER.splitlines()[0])
    parser.add_argument(
        "--config", default="config.json",
        help="JSON file with system parameter overrides",
    )
    parser.add_argument(
        "--state", default=None,
        help="JSON file used to save the final state (overrides config)",
    )
    parser.add_argument(
        "--debug", action="store_true", help="enable debug logging"
    )
    parser.add_argument(
        "--no-save", action="store_true", help="do not save state on exit"
    )
    return parser.parse_args(argv)


def setup_logging(log_file: str, debug: bool) -> None:
    """Configure file logging; fall back to stderr if the file is unusable."""
    level = logging.DEBUG if debug else logging.INFO
    fmt = "%(asctime)s %(levelname)s %(name)s: %(message)s"
    try:
        logging.basicConfig(filename=log_file, level=level, format=fmt)
    except OSError:
        logging.basicConfig(level=logging.WARNING, format=fmt)


def run_ticks(manager: SystemManager, display: DisplayManager,
              steps: int) -> None:
    """Advance the simulation and print delivery notifications."""
    for _ in range(steps):
        completed = manager.tick()
        for order in completed:
            print(
                f"[tick {manager.tick_count}] Order #{order.id} delivered by "
                f"rider #{order.rider_id} (took {order.time_taken} tick(s))"
            )


def cmd_start(manager: SystemManager, display: DisplayManager,
              input_mgr: InputManager, tokens: List[str]) -> None:
    """Handle ``start``: place an order and show the assignment."""
    if tokens:
        coords = input_mgr.parse_coordinates(tokens)
    else:
        coords = input_mgr.request_order_coordinates()
    if coords is None:
        print("Order cancelled.")
        return
    result = manager.process_order((coords[0], coords[1]),
                                   (coords[2], coords[3]))
    print(display.render_order_result(result, manager.riders))


def cmd_tick(manager: SystemManager, display: DisplayManager,
             tokens: List[str]) -> None:
    """Handle ``tick [n]``: advance the clock n ticks."""
    steps = 1
    if tokens:
        if len(tokens) != 1 or not tokens[0].isdigit():
            raise ValueError("tick expects a positive integer count")
        steps = int(tokens[0])
    if not 1 <= steps <= MAX_TICK_COMMAND:
        raise ValueError(f"tick count must be between 1 and "
                         f"{MAX_TICK_COMMAND}")
    run_ticks(manager, display, steps)


def execute(command: str, tokens: List[str], manager: SystemManager,
            display: DisplayManager, input_mgr: InputManager) -> bool:
    """Dispatch a parsed command.

    Returns:
        True if the command should advance the clock by one tick.
    """
    if command == "start":
        cmd_start(manager, display, input_mgr, tokens)
    elif command == "status":
        print(display.render_status(manager.funds, manager.tick_count,
                                    manager.riders,
                                    len(manager.order_manager.pending)))
    elif command == "orders":
        print(display.render_orders(manager.all_orders()))
    elif command == "riders":
        print(display.render_rider_log(manager.riders))
    elif command == "map":
        print(display.render_map(manager.riders, manager.active_orders()))
    elif command == "tick":
        cmd_tick(manager, display, tokens)
        return False
    elif command == "help":
        print(display.render_help())
    elif command == "quit":
        globalvar.set_value("running", False)
        return False
    return True


def check_consistency(manager: SystemManager) -> None:
    """Verify state and recover automatically if inconsistencies exist."""
    problems = manager.verify_state()
    if not problems:
        return
    for problem in problems:
        logger.warning("Consistency problem: %s", problem)
    fixes = manager.repair_state()
    for fix in fixes:
        logger.info("Recovery: %s", fix)
    print(f"Note: {len(problems)} state inconsistency(ies) detected and "
          "automatically recovered.")


def save_final_state(manager: SystemManager, path: str) -> None:
    """Persist the system state to a JSON file, reporting any failure."""
    try:
        config.save_state(path, manager.export_state())
        print(f"State saved to {path}.")
    except (OSError, TypeError, ValueError) as exc:
        logger.error("Could not save state: %s", exc)
        print(f"Warning: could not save state: {exc}")


def main(argv: Optional[List[str]] = None) -> int:
    """Run the interactive dispatch simulator.

    Args:
        argv: Optional argument list (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code.
    """
    args = parse_args(argv)
    try:
        if config.load_config(args.config):
            print(f"Configuration loaded from {args.config}.")
    except (OSError, ValueError) as exc:
        print(f"Warning: could not load config '{args.config}': {exc}. "
              "Using default parameters.")
        config.CONFIG.update(config.DEFAULT_CONFIG)
    setup_logging(str(config.CONFIG["log_file"]), args.debug)

    init_system()
    manager = get_manager()
    display = DisplayManager()
    input_mgr = InputManager()
    print(BANNER)
    print(f"Starting funds: {manager.funds}, riders: {len(manager.riders)}")

    while globalvar.get_value("running", True):
        try:
            line = input_mgr.read_command()
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if not line:
            continue
        try:
            command, tokens = input_mgr.parse_command(line)
        except ValueError as exc:
            print(f"Error: {exc}. Type 'help' for the command list.")
            continue

        advance = True
        try:
            advance = execute(command, tokens, manager, display, input_mgr)
        except ValueError as exc:
            print(f"Input error: {exc}")
            advance = False
        except Exception as exc:  # pylint: disable=broad-except
            logger.exception("Unexpected error running '%s'", command)
            print(f"Unexpected internal error: {exc}. Attempting recovery.")
            manager.repair_state()
            advance = False

        check_consistency(manager)
        if advance and globalvar.get_value("running", True):
            run_ticks(manager, display, 1)

    if not args.no_save:
        state_path = args.state or str(config.CONFIG["state_file"])
        save_final_state(manager, state_path)
    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Participant list loading, validation and group statistics."""
from __future__ import annotations

from lottery_system.default_data import DEFAULT_PARTICIPANT_LINES
from lottery_system.models import User, normalize


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building participant list."""
        return self._parse_lines(DEFAULT_PARTICIPANT_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt participant list from the given path."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file where each line is 'Name,Department Group'."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except UnicodeDecodeError:
            return [], [f"File Error: '{path}' is not a valid UTF-8 text file."]
        except OSError as exc:
            reason = exc.strerror or str(exc)
            return [], [f"File Error: cannot read '{path}' ({reason})."]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines, collecting format errors per line number."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip().replace("\uff0c", ",")  # full-width comma
            if not text:
                continue
            parts = text.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department', got '{raw.strip()}'"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group (sorted by group name)."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZE_SPECS
from lottery_system.models import Prize, normalize


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages for skipped prizes."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZE_SPECS:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"Skipped preset prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")
        key = normalize(clean_name)
        if any(normalize(p.name) == key for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups: list[str] = []
        for group in avoid_groups or []:
            text = str(group).strip()
            if text and text not in groups:
                groups.append(text)

        drawer = associated_drawer.strip() if associated_drawer else None
        if not drawer:
            drawer = None

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        try:
            is_permutation = sorted(order) == list(range(1, len(self._prizes) + 1))
        except TypeError:
            return False
        if not is_permutation:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Winner recording, grouping and CSV-style export."""
from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str


class ResultManager:
    """Stores winners in winning order and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for a prize."""
        winner = Winner(user=user, prize_name=prize.name)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, following the order of the given prizes."""
        grouped: dict[str, list[Winner]] = {}
        for prize in prizes:
            grouped[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        by_prize = self.get_winners_by_prize(prizes)
        rows = [winner for winners in by_prize.values() for winner in winners]
        if not rows:
            return False, "No winners to export."

        filename = f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for winner in rows:
                    writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery rule engine: eligibility pools, Cheng Ru fallback, and winner tracking."""
from __future__ import annotations

import random

from lottery_system.models import Prize, User, normalize
from lottery_system.services.result_manager import ResultManager

CHENG_RU_GROUP = "Cheng Ru Group"

DRAWER_CHEN_QIXIAN = "Chen Qixian"
DRAWER_JIN_DUO = "Jin Duo"
DRAWER_MA_JIAQI = "Ma Jiaqi"
DRAWER_LI_BIN = "Li Bin"
DRAWER_OPTIONS: tuple[str, ...] = (DRAWER_CHEN_QIXIAN, DRAWER_JIN_DUO, DRAWER_MA_JIAQI, DRAWER_LI_BIN)

_HISTORY_DRAWERS: tuple[str, ...] = (DRAWER_CHEN_QIXIAN, DRAWER_JIN_DUO)


def is_cheng_ru_member(user: User) -> bool:
    """True if the user belongs to the Cheng Ru group."""
    return normalize(user.department) == normalize(CHENG_RU_GROUP)


def _is_drawer(prize: Prize, drawer: str) -> bool:
    return prize.associated_drawer is not None and normalize(prize.associated_drawer) == normalize(drawer)


def describe_prize_rule(prize: Prize, fallback_active: bool = False) -> str:
    """Build a human-readable description of the rules applied to a prize round."""
    if _is_drawer(prize, DRAWER_LI_BIN):
        parts = ["Only previous winners are excluded (group avoidance not applied)"]
    else:
        parts = ["Previous winners are excluded"]
        if prize.avoid_groups:
            parts.append("avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("no group avoidance")
    if _is_drawer(prize, DRAWER_MA_JIAQI) and fallback_active:
        parts.append("Cheng Ru Group protection active: only Cheng Ru Group members are eligible")
    return "; ".join(parts) + "."


class LotteryManager:
    """Controls eligibility and winner bookkeeping for lottery rounds."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self.last_two_draw_results: list[bool] = []
        self.result_manager = ResultManager()

    def has_won(self, user: User) -> bool:
        """True if this person has already won any prize."""
        return normalize(user.name) in self._winner_keys

    def is_cheng_ru_fallback_active(self) -> bool:
        """True when the last two lottery-person results both contain no Cheng Ru members."""
        return len(self.last_two_draw_results) >= 2 and not any(self.last_two_draw_results[-2:])

    def _is_cheng_ru_fallback_for(self, prize: Prize) -> bool:
        return _is_drawer(prize, DRAWER_MA_JIAQI) and self.is_cheng_ru_fallback_active()

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users eligible to win the given prize right now."""
        pool = [user for user in self._users if not self.has_won(user)]

        # Fallback (Cheng Ru group protection): limit to Cheng Ru members, overriding avoidance groups.
        if self._is_cheng_ru_fallback_for(prize):
            return [user for user in pool if is_cheng_ru_member(user)]

        # Special rule: Li Bin's prizes only exclude previous winners.
        if _is_drawer(prize, DRAWER_LI_BIN):
            return pool

        avoid = {normalize(group) for group in prize.avoid_groups}
        return [user for user in pool if normalize(user.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        """Describe the rules applied to the prize in the current state."""
        return describe_prize_rule(prize, self.is_cheng_ru_fallback_active())

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if no eligible candidate exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winner_keys.add(normalize(winner.name))
        self.result_manager.add_winner(winner, prize)
        return winner

    def finish_round(self, prize: Prize, winners: list[User]) -> None:
        """Record the outcome of a finished round for the fallback history."""
        if any(_is_drawer(prize, drawer) for drawer in _HISTORY_DRAWERS):
            contains_cheng_ru = any(is_cheng_ru_member(winner) for winner in winners)
            self.last_two_draw_results.append(contains_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

class InputReader:
    """Reads stdin in a background thread so control commands can be received during a lottery."""

    def __init__(self, stream: TextIO | None = None) -> None:
        self._stream = stream or sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._thread = threading.Thread(target=self._pump, daemon=True)
        self._thread.start()

    def _pump(self) -> None:
        while True:
            try:
                line = self._stream.readline()
            except (OSError, ValueError):
                line = ""
            if line == "":
                self._queue.put(None)
                return
            self._queue.put(line.strip())

    def read(self) -> str:
        """Blocking read of the next input line. Raises EOFError when input is closed."""
        item = self._queue.get()
        if item is None:
            self._queue.put(None)
            raise EOFError
        return item

    def poll(self) -> str | None:
        """Non-blocking read. Returns None when no input is pending."""
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        if item is None:
            self._queue.put(None)
            return None
        return item


def ordinal(number: int) -> str:
    """Format an integer as an English ordinal, e.g. 1st, 2nd, 11th."""
    if 11 <= number % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def parse_numbers(text: str) -> list[int] | None:
    """Parse whitespace/comma separated integers. Returns None if any token is not an integer."""
    tokens = [token for token in re.split(r"[\s,]+", text.strip()) if token]
    try:
        return [int(token) for token in tokens]
    except ValueError:
        return None

        for index, prize in enumerate(prizes, start=1):
            print(f"{index}. {prize.name} | Quantity: {prize.quantity} | Drawer: {prize.associated_drawer or 'None'}")
            print(f"   Rule: {describe_prize_rule(prize)}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            print("At least two prizes are needed to adjust the sequence.")
            return
        self._print_sequence(prizes)
        numbers = parse_numbers(
            self._ask("Enter the new order as serial numbers (e.g. '2 1 3'): ")
        )
        if numbers is None or not self.prize_manager.reorder_prizes(numbers):
            print(f"Invalid order. Each serial number from 1 to {len(prizes)} must appear exactly once.")
            return
        print("Lottery sequence updated.")
        self._print_sequence(self.prize_manager.get_prizes())

    @staticmethod
    def _print_sequence(prizes: list[Prize]) -> None:
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name} (quantity: {prize.quantity})")

    # ---------------------------------------------------------- lottery module
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery_completed:
            print("The lottery is already completed for the current list. Load a list to start a new lottery.")
            return
        total_quantity = sum(prize.quantity for prize in prizes)
        if total_quantity > len(self.users):
            print(
                f"Warning: total prize quantity ({total_quantity}) exceeds participants "
                f"({len(self.users)}); some rounds may draw fewer winners."
            )
        answer = self._ask("Start the lottery now? (y/n): ").strip().lower()
        if answer not in ("y", "yes"):
            print("Lottery cancelled.")
            return

        if self.lottery is None:
            self.lottery = LotteryManager(self.users)
        print("\nLottery started. Commands: 'pause' to interrupt, 'continue' to resume, 'terminate' to end the current round.")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize)
        self.lottery_completed = True
        print("\nAll lottery rounds completed.")
        self._show_results()

    def _run_round(self, index: int, total: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {index}/{total}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")
        print(f"Drawer: {prize.associated_drawer or 'None'}")
        print(f"Rule: {self.lottery.describe_rules(prize)}")

        winners: list[User] = []
        terminated = False
        for position in range(1, prize.quantity + 1):
            if self._pump_controls(self.draw_interval):
                terminated = True
                break
            print(f"Currently drawing the {ordinal(position)} winner...")
            winner = self.lottery.draw_one(prize)
            if winner is None:
                print("No eligible candidates remain for this prize. The round ends early.")
                break
            winners.append(winner)
            print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}")

        self.lottery.finish_round(prize, winners)
        status = "terminated by operator" if terminated else "completed"
        print(f"Round {index} {status}: {len(winners)}/{prize.quantity} winner(s) drawn.")

    def _pump_controls(self, duration: float) -> bool:
        """Wait for `duration` seconds while handling control commands.

        Returns True if the current round should be terminated.
        """
        deadline = time.monotonic() + duration
        while True:
            command = self.reader.poll()
            if command is None:
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    return False
                time.sleep(min(CONTROL_POLL_SECONDS, remaining))
                continue

            command = command.strip().lower()
            if not command:
                continue
            if command == "pause":
                print("[PAUSED] Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    answer = self.reader.read().strip().lower()
                    if answer == "continue":
                        print("[RESUMED] Lottery resumed.")
                        break
                    if answer == "terminate":
                        print("[TERMINATED] Current round terminated. Remaining draws are skipped.")
                        return True
                    print("Lottery is paused. Type 'continue' or 'terminate'.")
            elif command == "terminate":
                print("[TERMINATED] Current round terminated. Remaining draws are skipped.")
                return True
            elif command == "continue":
                print("Lottery is not paused.")
            else:
                print(f"Unknown command '{command}'. Use 'pause' or 'terminate' during the lottery.")

    # ---------------------------------------------------------- results module
    def _show_results(self) -> None:
        if self.lottery is None:
            print("No lottery has been run yet.")
            return
        prizes = self.prize_manager.get_prizes()
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        total = sum(len(winners) for winners in grouped.values())
        if total == 0:
            print("No winners recorded yet.")
            return
        print("\n=== Winner List ===")
        print(f"{'Name':<20} | {'Department':<28} | Prize")
        for prize_name, winners in grouped.items():
            for winner in winners:
                print(f"{winner.user.name:<20} | {winner.user.department:<28} | {prize_name}")

    def _export_results(self) -> None:
        if self.lottery is None:
            print("No lottery results to export.")
            return
        _, message = self.lottery.result_manager.export_to_csv(self.prize_manager.get_prizes())[0:1] + (
            self.lottery.result_manager.export_to_csv(self.prize_manager.get_prizes())[1],
        )
        print(message)

    app = LotterySystemCLI()
    app.run()

    main()

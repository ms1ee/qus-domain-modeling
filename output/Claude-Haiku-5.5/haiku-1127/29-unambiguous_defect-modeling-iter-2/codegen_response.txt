"""Process-wide variable store for the food delivery dispatch simulator.

Shared state (funds, riders, orders, simulated tick, ...) lives in a
module-level dictionary so that any module can read or write it without
passing objects around explicitly.

Keys used across the program:
    funds (int): system funds, may become negative.
    riders (list[Rider]): active rider objects.
    rider_id_counter (int): last issued rider id (starts at 0).
    is_running (bool): main loop control flag.
    current_tick (int): simulated time in ticks.
    orders (list[Order]): every order created in this session.
    order_history (list[Order]): completed orders in completion order.
    order_id_counter (int): last issued order id.
    max_rider_load (int): maximum concurrent orders per rider.
"""

import logging
from typing import Any, Dict

logger = logging.getLogger(__name__)

_global_dict: Dict[str, Any] = {}


def _init() -> None:
    """Reset the store to the default values of a fresh simulation.

    Returns:
        None. The module-level dictionary is replaced in place.
    """
    global _global_dict
    _global_dict = {
        "funds": 0,
        "riders": [],
        "rider_id_counter": 0,
        "is_running": True,
        "current_tick": 0,
        "orders": [],
        "order_history": [],
        "order_id_counter": 0,
        "max_rider_load": 3,
    }
    logger.debug("Global variable store initialised.")


def set_value(key, value) -> None:
    """Store a value under the given key.

    Args:
        key (str): Name of the global variable.
        value (Any): Value to store.
    """
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key``.

    Args:
        key (str): Name of the global variable.
        default (Any): Value returned when the key does not exist.

    Returns:
        Any: The stored value, or ``default`` if the key is missing.
    """
    return _global_dict.get(key, default)

"""Grid map definitions and coordinate helpers.

The map is an 800x800 coordinate system divided into an 8x8 grid of 100x100
cells. Grid lines (x or y multiple of 100) are streets. Restaurant and
customer locations are the centres of cells (for example x = 50, 150, ...).
All coordinates are integers in the range 0-800.
"""

from typing import Dict, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = 8
CELL_HALF = CELL_SIZE // 2

Point = Tuple[int, int]


def clamp_coordinate(value) -> int:
    """Convert a value to an int and clamp it to the map range 0-800.

    Args:
        value (int | float | str): Raw coordinate.

    Returns:
        int: Clamped integer coordinate.
    """
    return max(0, min(MAP_SIZE, int(value)))


def cell_index(value) -> int:
    """Return the grid index (0-7) of the cell containing a coordinate.

    Args:
        value (int): Coordinate along one axis.

    Returns:
        int: Cell index between 0 and GRID_DIMENSION - 1.
    """
    return min(clamp_coordinate(value) // CELL_SIZE, GRID_DIMENSION - 1)


def to_grid_center(x, y) -> Point:
    """Transform arbitrary coordinates into the centre of their grid cell.

    Implements ``center_x = (grid_x // 100) * 100 + 50``.

    Args:
        x (int): X coordinate (0-800, out-of-range values are clamped).
        y (int): Y coordinate (0-800, out-of-range values are clamped).

    Returns:
        tuple[int, int]: Cell centre coordinates.
    """
    return (cell_index(x) * CELL_SIZE + CELL_HALF,
            cell_index(y) * CELL_SIZE + CELL_HALF)


def cell_corner(x, y) -> Point:
    """Return the bottom-left street intersection of the cell containing a point.

    Args:
        x (int): X coordinate.
        y (int): Y coordinate.

    Returns:
        tuple[int, int]: Intersection coordinates (both multiples of 100).
    """
    return cell_index(x) * CELL_SIZE, cell_index(y) * CELL_SIZE


def is_on_street(x, y) -> bool:
    """Return True if the point lies on a street (grid line).

    Args:
        x (int): X coordinate.
        y (int): Y coordinate.

    Returns:
        bool: True when x or y is a multiple of 100.
    """
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


def is_cell_center(x, y) -> bool:
    """Return True if the point is the centre of a grid cell.

    Args:
        x (int): X coordinate.
        y (int): Y coordinate.

    Returns:
        bool: True for restaurant/customer location points.
    """
    return x % CELL_SIZE == CELL_HALF and y % CELL_SIZE == CELL_HALF


def manhattan(a, b) -> int:
    """Return the Manhattan distance between two points.

    Args:
        a (tuple[int, int]): First point.
        b (tuple[int, int]): Second point.

    Returns:
        int: |ax - bx| + |ay - by|.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class GridCell:
    """One 100x100 cell of the map.

    Attributes:
        grid_x (int): Column index (0-7).
        grid_y (int): Row index (0-7).
        centerX (int): X coordinate of the cell centre.
        centerY (int): Y coordinate of the cell centre.
        isRoad (bool): True, the cell is bounded by streets.
        isLocation (bool): True, the centre can hold a restaurant/customer.
    """

    def __init__(self, grid_x: int, grid_y: int):
        """Create a cell at the given grid index.

        Args:
            grid_x (int): Column index (0-7).
            grid_y (int): Row index (0-7).
        """
        self.grid_x = grid_x
        self.grid_y = grid_y
        self.centerX = grid_x * CELL_SIZE + CELL_HALF
        self.centerY = grid_y * CELL_SIZE + CELL_HALF
        self.isRoad = True
        self.isLocation = True


class GridMap:
    """The 8x8 grid of cells representing the 800x800 city.

    Attributes:
        mapSize (int): Map side length in units.
        cellSize (int): Side length of one cell in units.
        gridDimension (int): Number of cells per row/column.
        cells (dict): Mapping (grid_x, grid_y) -> GridCell.
    """

    def __init__(self):
        """Build all 64 cells of the map."""
        self.mapSize = MAP_SIZE
        self.cellSize = CELL_SIZE
        self.gridDimension = GRID_DIMENSION
        self.cells: Dict[Point, GridCell] = {
            (gx, gy): GridCell(gx, gy)
            for gx in range(GRID_DIMENSION)
            for gy in range(GRID_DIMENSION)
        }

    def get_cell(self, x, y) -> GridCell:
        """Return the cell containing the given coordinate.

        Args:
            x (int): X coordinate.
            y (int): Y coordinate.

        Returns:
            GridCell: The containing cell.
        """
        return self.cells[(cell_index(x), cell_index(y))]

"""Grid-based path planning for riders.

Riders travel along streets (grid lines where x or y is a multiple of 100).
A route is a list of axis-aligned waypoints. Only the final approach into a
restaurant/customer cell and the exit from a cell leave the streets.
Riders never block each other, so no deadlock can occur on the street
network; the planner therefore only needs to guarantee axis-aligned legs.
"""

import logging
from typing import List, Optional, Sequence

from grid_map import (CELL_SIZE, GridMap, Point, cell_corner, is_on_street,
                      to_grid_center, manhattan)

logger = logging.getLogger(__name__)


class PathPlanner:
    """Plans street-first, axis-aligned routes on the 8x8 grid.

    Attributes:
        grid_map (GridMap): The map the planner navigates.
        preferStreets (bool): When True, the planner prefers horizontal
            streets whenever the rider is standing on one.
    """

    def __init__(self, grid_map: Optional[GridMap] = None,
                 preferStreets: bool = True):
        """Create a planner.

        Args:
            grid_map (GridMap | None): Map to navigate; a new one is built
                when omitted.
            preferStreets (bool): Street preference flag.
        """
        self.grid_map = grid_map if grid_map is not None else GridMap()
        self.preferStreets = preferStreets

    def plan(self, start: Sequence[int], goal: Sequence[int]) -> List[Point]:
        """Compute waypoints from ``start`` to the centre of ``goal``'s cell.

        The start point is not included in the result. Every waypoint is
        reached by an axis-aligned move. Special cases handled:

        * Start and goal in the same cell: a direct two-leg move inside the
          cell (no detour onto the streets).
        * Start inside a cell (off street): first exit to the cell's corner.
        * Goal is directly above/below the start in the same column: the
          axis-aligned legs reduce to a single vertical movement.

        Args:
            start (tuple[int, int]): Current position of the rider.
            goal (tuple[int, int]): Target location (any point; it is
                snapped to its cell centre).

        Returns:
            list[tuple[int, int]]: Ordered waypoints, possibly empty when
            the rider is already at the target.
        """
        start = (int(start[0]), int(start[1]))
        goal_center = to_grid_center(goal[0], goal[1])
        if start == goal_center:
            return []

        goal_corner = cell_corner(goal_center[0], goal_center[1])
        waypoints: List[Point] = []

        if not is_on_street(start[0], start[1]):
            if cell_corner(start[0], start[1]) == goal_corner:
                # Same cell: move inside the cell only.
                waypoints.append((goal_center[0], start[1]))
                waypoints.append(goal_center)
                return self._compact(start, waypoints)
            if start[0] == goal_center[0]:
                # Directly above or below the target: straight movement.
                waypoints.append(goal_center)
                return self._compact(start, waypoints)
            start_corner = cell_corner(start[0], start[1])
            waypoints.append((start[0], start_corner[1]))
            waypoints.append(start_corner)
            current = start_corner
        else:
            current = start

        if self.preferStreets and current[1] % CELL_SIZE == 0:
            # On a horizontal street: run along it, then turn onto the
            # vertical street of the goal corner.
            waypoints.append((goal_corner[0], current[1]))
        else:
            # On a vertical street: run along it, then turn onto the
            # horizontal street of the goal corner.
            waypoints.append((current[0], goal_corner[1]))
        waypoints.append(goal_corner)
        waypoints.append((goal_center[0], goal_corner[1]))
        waypoints.append(goal_center)
        return self._compact(start, waypoints)

    @staticmethod
    def _compact(start: Point, points: List[Point]) -> List[Point]:
        """Remove consecutive duplicate waypoints.

        Args:
            start (tuple[int, int]): Starting point (excluded from output).
            points (list[tuple[int, int]]): Raw waypoints.

        Returns:
            list[tuple[int, int]]: Waypoints without repeats.
        """
        result: List[Point] = []
        previous = start
        for point in points:
            if point != previous:
                result.append(point)
                previous = point
        return result

    @staticmethod
    def route_length(start: Sequence[int], waypoints: Sequence[Point]) -> int:
        """Return the total travelled length of a route.

        Args:
            start (tuple[int, int]): Starting point of the route.
            waypoints (list[tuple[int, int]]): Route waypoints.

        Returns:
            int: Sum of Manhattan distances between consecutive points.
        """
        total = 0
        previous = (int(start[0]), int(start[1]))
        for point in waypoints:
            total += manhattan(previous, point)
            previous = point
        return total

"""Order data model.

An order is created from user input ``[restaurant_x, restaurant_y,
customer_x, customer_y]``; both locations are automatically transformed to
the centres of their grid cells.
"""

import logging
from typing import Any, Dict, Optional, Sequence

from globalvar import get_value, set_value
from grid_map import manhattan, to_grid_center

logger = logging.getLogger(__name__)


def _next_order_id() -> int:
    """Issue the next unique order id.

    Returns:
        int: Auto-incremented order id starting at 1.
    """
    counter = int(get_value("order_id_counter", 0) or 0) + 1
    set_value("order_id_counter", counter)
    return counter


class Order:
    """A food delivery order.

    Attributes:
        id (int): Unique order identifier.
        res_pos (tuple[int, int]): Restaurant cell centre.
        cus_pos (tuple[int, int]): Customer cell centre.
        status (str): One of created, assigned, picking_up, delivering,
            completed.
        rider_id (int | None): Id of the rider the order is assigned to.
        start_time (int): Tick at which the order was created.
        completion_time (int | None): Tick at which delivery finished.
        distance (float): Distance actually travelled for this order.
        time_taken (int | None): completion_time - start_time.
    """

    CREATED = "created"
    ASSIGNED = "assigned"
    PICKING_UP = "picking_up"
    DELIVERING = "delivering"
    COMPLETED = "completed"

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]):
        """Create an order and transform its coordinates to grid centres.

        Args:
            res_pos (Sequence[int]): Restaurant ``[x, y]`` in range 0-800.
            cus_pos (Sequence[int]): Customer ``[x, y]`` in range 0-800.

        Raises:
            ValueError: If a position does not have exactly two components.
        """
        if len(res_pos) != 2 or len(cus_pos) != 2:
            raise ValueError("Positions must be [x, y] pairs.")
        self.id = _next_order_id()
        self.res_pos = to_grid_center(res_pos[0], res_pos[1])
        self.cus_pos = to_grid_center(cus_pos[0], cus_pos[1])
        self.status = self.CREATED
        self.rider_id: Optional[int] = None
        self.start_time = int(get_value("current_tick", 0) or 0)
        self.completion_time: Optional[int] = None
        self.distance = 0.0
        self.time_taken: Optional[int] = None
        logger.debug("Created order %s: %s -> %s", self.id,
                     self.res_pos, self.cus_pos)

    @property
    def restaurant_x(self) -> int:
        """int: X coordinate of the restaurant cell centre."""
        return self.res_pos[0]

    @property
    def restaurant_y(self) -> int:
        """int: Y coordinate of the restaurant cell centre."""
        return self.res_pos[1]

    @property
    def customer_x(self) -> int:
        """int: X coordinate of the customer cell centre."""
        return self.cus_pos[0]

    @property
    def customer_y(self) -> int:
        """int: Y coordinate of the customer cell centre."""
        return self.cus_pos[1]

    @property
    def straight_distance(self) -> int:
        """int: Manhattan distance from restaurant to customer."""
        return manhattan(self.res_pos, self.cus_pos)

    @property
    def is_active(self) -> bool:
        """bool: True while the order has not been completed."""
        return self.status != self.COMPLETED

    def assign_to(self, rider_id: int) -> None:
        """Record the rider that the order is assigned to.

        The caller is responsible for appending the order to the rider's
        ``orders`` list (see ``Rider.accept_order``).

        Args:
            rider_id (int): Identifier of the assigned rider.
        """
        self.rider_id = rider_id
        if self.status == self.CREATED:
            self.status = self.ASSIGNED

    def mark_picking_up(self) -> None:
        """Mark that the assigned rider is heading to the restaurant."""
        self.status = self.PICKING_UP

    def mark_delivering(self) -> None:
        """Mark that the food has been collected and is being delivered."""
        self.status = self.DELIVERING

    def add_distance(self, distance: float) -> None:
        """Accumulate distance travelled by the rider for this order.

        Args:
            distance (float): Distance moved in one tick.
        """
        self.distance += float(distance)

    def mark_completed(self, tick: int) -> None:
        """Mark the order as delivered and compute efficiency data.

        Args:
            tick (int): Simulated tick of completion.
        """
        self.status = self.COMPLETED
        self.completion_time = tick
        self.time_taken = tick - self.start_time

    def to_dict(self) -> Dict[str, Any]:
        """Serialise the order for JSON persistence.

        Returns:
            dict: JSON-compatible representation.
        """
        return {
            "id": self.id,
            "status": self.status,
            "rider_id": self.rider_id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

    def __repr__(self) -> str:
        """Return a concise representation for debugging."""
        return (f"Order(id={self.id}, status={self.status}, "
                f"rider_id={self.rider_id})")

"""Order management and distance-first rider assignment."""

import logging
from typing import Any, Dict, List, Optional

from globalvar import get_value
from grid_map import manhattan
from order_model import Order

logger = logging.getLogger(__name__)

# Load-balancing penalty (in distance units) added per order already held.
LOAD_PENALTY = 150
DEFAULT_MAX_RIDER_LOAD = 3


class OrderManager:
    """Creates, assigns, and tracks orders.

    Orders are stored in the global variable ``orders``; completed orders
    are additionally recorded in ``order_history`` by the rider on delivery.
    """

    @property
    def orderIdCounter(self) -> int:
        """int: Last issued order id."""
        return int(get_value("order_id_counter", 0) or 0)

    def create_order(self, res_pos, cus_pos) -> Order:
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos (Sequence[int]): Restaurant ``[x, y]``.
            cus_pos (Sequence[int]): Customer ``[x, y]``.

        Returns:
            Order: The newly created order (already stored globally).
        """
        order = Order(res_pos, cus_pos)
        get_value("orders").append(order)
        return order

    def discard_order(self, order: Order) -> None:
        """Remove an order that could not be dispatched.

        Args:
            order (Order): Order to remove from the global order list.
        """
        orders = get_value("orders")
        if order in orders:
            orders.remove(order)
            logger.info("Discarded undispatched order %s", order.id)

    def assign_order(self, order: Order, rider) -> None:
        """Assign an order to a rider.

        Args:
            order (Order): Order to assign.
            rider (Rider): Rider receiving the order.
        """
        order.assign_to(rider.rider_id)
        rider.accept_order(order)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)

    def active_orders(self) -> List[Order]:
        """Return all orders that are not yet completed.

        Returns:
            list[Order]: Active orders in creation order.
        """
        return [o for o in get_value("orders", []) if o.is_active]

    def all_orders(self) -> List[Order]:
        """Return every order created in this session.

        Returns:
            list[Order]: All orders in creation order.
        """
        return list(get_value("orders", []))

    def completed_orders(self) -> List[Order]:
        """Return completed orders in completion order.

        Returns:
            list[Order]: Delivered orders.
        """
        return list(get_value("order_history", []))

    def to_dict_list(self) -> List[Dict[str, Any]]:
        """Serialise all orders.

        Returns:
            list[dict]: JSON-compatible order records.
        """
        return [o.to_dict() for o in self.all_orders()]


def find_best_rider_for_order(order: Order) -> Optional["Rider"]:
    """Select the best rider for an order using a distance-first rule.

    Each rider with spare capacity is scored by::

        score = manhattan(rider position, restaurant) + LOAD_PENALTY * load

    The lowest score wins, which keeps dispatch close to the restaurant while
    balancing load between riders. Ties are broken by rider id.

    Args:
        order (Order): Order that needs a rider.

    Returns:
        Rider | None: The chosen rider, or None if no rider has spare
        capacity (the caller may then recruit a new rider).
    """
    riders = get_value("riders", []) or []
    max_load = int(get_value("max_rider_load", DEFAULT_MAX_RIDER_LOAD)
                   or DEFAULT_MAX_RIDER_LOAD)
    best = None
    best_key = None
    for rider in riders:
        load = len(rider.orders)
        if load >= max_load:
            continue
        distance = manhattan(rider.position, order.res_pos)
        key = (distance + LOAD_PENALTY * load, rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best

"""Rider model: state machine, movement, and event logging.

Rider states:
    0 = idle, 1 = picking up (heading to the restaurant),
    2 = delivering (heading to the customer).

Events written to ``Rider.log``:
    spawn, accepted_order, arrived_at_merchant, delivered_to_customer,
    recovered.
"""

import logging
import random
from typing import Any, Dict, List, Optional, Tuple

from globalvar import get_value, set_value
from grid_map import CELL_SIZE, GRID_DIMENSION, MAP_SIZE, is_on_street
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

_DEFAULT_PLANNER = PathPlanner()


def _sign(value: int) -> int:
    """Return -1, 0 or 1 according to the sign of ``value``."""
    return (value > 0) - (value < 0)


class Rider:
    """A delivery rider moving on the street grid.

    Attributes:
        rider_id (int): Unique identifier.
        x (int): Current X coordinate.
        y (int): Current Y coordinate.
        speed (int): Distance units travelled per tick.
        state (int): 0 idle, 1 picking up, 2 delivering.
        orders (list[Order]): Assigned orders; index 0 is the active one.
        special_case (bool): True while the rider is inside a location cell
            (off the streets).
        path (list[tuple[int, int]]): Remaining waypoints of the current leg.
        target (tuple[int, int] | None): Current destination.
        log (list[dict]): Chronological status-change records.
    """

    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2
    STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up",
                   DELIVERING: "delivering"}
    MIN_SPEED = 20
    MAX_SPEED = 40

    def __init__(self, rider_id: int, x: Optional[int] = None,
                 y: Optional[int] = None, speed: Optional[int] = None,
                 planner: Optional[PathPlanner] = None):
        """Create a rider, spawning it on a street if no position is given.

        Args:
            rider_id (int): Unique rider identifier.
            x (int | None): Optional starting X coordinate (must be on road).
            y (int | None): Optional starting Y coordinate.
            speed (int | None): Optional speed; random in 20-40 otherwise.
            planner (PathPlanner | None): Planner to use for routing.

        Raises:
            ValueError: If the speed is not positive.
        """
        self.rider_id = int(rider_id)
        if x is None or y is None:
            x, y = self._random_road_point()
        self.x = int(x)
        self.y = int(y)
        self.speed = (int(speed) if speed is not None
                      else random.randint(self.MIN_SPEED, self.MAX_SPEED))
        if self.speed <= 0:
            raise ValueError("Rider speed must be positive.")
        self.state = self.IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.path: List[Tuple[int, int]] = []
        self.target: Optional[Tuple[int, int]] = None
        self.planner = planner or _DEFAULT_PLANNER
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    @staticmethod
    def _random_road_point() -> Tuple[int, int]:
        """Pick a random point on a vertical street.

        Returns:
            tuple[int, int]: Road coordinates (x is a multiple of 100).
        """
        x = random.randint(0, GRID_DIMENSION) * CELL_SIZE
        y = random.randint(0, MAP_SIZE)
        return x, y

    @property
    def position(self) -> Tuple[int, int]:
        """tuple[int, int]: Current (x, y) position."""
        return self.x, self.y

    @staticmethod
    def _current_tick() -> int:
        """Return the simulated tick from the global store."""
        return int(get_value("current_tick", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append a status-change entry to the rider log.

        Args:
            event (str): Event name.
            order_id (int | None): Related order id, if any.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": self._current_tick(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s %s order=%s at (%s,%s)", self.rider_id,
                     event, order_id, self.x, self.y)

    def _update_special_case(self) -> None:
        """Set ``special_case`` when the rider is off the streets."""
        self.special_case = not is_on_street(self.x, self.y)

    def accept_order(self, order) -> None:
        """Accept an order by appending it to ``orders``.

        If the rider is idle it immediately starts the pickup leg.

        Args:
            order (Order): Order to accept.
        """
        self.orders.append(order)
        self._record("accepted_order", order.id)
        if self.state == self.IDLE:
            self._begin_pickup()

    def _begin_pickup(self) -> None:
        """Start the leg from the current position to the active restaurant."""
        order = self.orders[0]
        self.state = self.PICKING_UP
        order.mark_picking_up()
        self.target = order.res_pos
        self.path = self.planner.plan(self.position, self.target)
        self._update_special_case()

    def _set_idle(self) -> None:
        """Return the rider to the idle state and clear the route."""
        self.state = self.IDLE
        self.target = None
        self.path = []
        self._update_special_case()

    def _sync_path_state(self) -> None:
        """Keep the path and target consistent with the rider state."""
        if self.state == self.IDLE:
            self.target = None
            self.path = []

    def _advance(self) -> int:
        """Move along the current path by up to ``speed`` units.

        Moves are axis-aligned: the rider first covers the horizontal part of
        a waypoint, then the vertical part.

        Returns:
            int: Number of distance units actually travelled.
        """
        remaining = self.speed
        moved = 0
        while remaining > 0 and self.path:
            tx, ty = self.path[0]
            dx = tx - self.x
            dy = ty - self.y
            dist = abs(dx) + abs(dy)
            if dist == 0:
                self.path.pop(0)
                continue
            step = min(dist, remaining)
            step_x = min(abs(dx), step)
            step_y = step - step_x
            self.x += _sign(dx) * step_x
            self.y += _sign(dy) * step_y
            remaining -= step
            moved += step
            if (self.x, self.y) == (tx, ty):
                self.path.pop(0)
        return moved

    def _handle_arrival(self) -> None:
        """Process arrival at the current target and advance the state machine.

        Pickup arrival: state 1 -> 2, plan route to the customer.
        Delivery arrival: order completed, state 2 -> 0 or next pickup.
        """
        order = self.orders[0]
        if self.state == self.PICKING_UP:
            self.state = self.DELIVERING
            order.mark_delivering()
            self._record("arrived_at_merchant", order.id)
            self.target = order.cus_pos
            self.path = self.planner.plan(self.position, self.target)
        elif self.state == self.DELIVERING:
            order.mark_completed(self._current_tick())
            self.orders.pop(0)
            self._record("delivered_to_customer", order.id)
            history = get_value("order_history")
            if history is not None:
                history.append(order)
            if self.orders:
                self._begin_pickup()
            else:
                self._set_idle()
        else:
            self._set_idle()
        self._update_special_case()

    def move(self) -> None:
        """Advance the rider by one tick.

        Performs the idle -> picking up -> delivering -> idle transitions,
        moves along the planned route, and accumulates travelled distance on
        the active order.
        """
        if not self.orders:
            if self.state != self.IDLE:
                self._set_idle()
            return

        if self.state == self.IDLE:
            # Orders appended directly to ``orders`` are accepted here.
            self._record("accepted_order", self.orders[0].id)
            self._begin_pickup()

        moved = self._advance()
        if moved and self.orders:
            self.orders[0].add_distance(moved)

        while self.orders and not self.path and self.target is not None:
            self._handle_arrival()
        self._sync_path_state()
        self._update_special_case()

    def recover(self) -> None:
        """Restore a consistent state after an unexpected error.

        The route is re-planned for the active order, or the rider is set
        idle if it has no orders.
        """
        self.path = []
        self.target = None
        if self.orders:
            self._begin_pickup()
            self._record("recovered", self.orders[0].id)
        else:
            self._set_idle()
            self._record("recovered", None)
        logger.warning("Rider %s recovered to a consistent state.",
                       self.rider_id)

    def to_dict(self) -> Dict[str, Any]:
        """Serialise the rider for JSON persistence.

        Returns:
            dict: JSON-compatible representation.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [o.id for o in self.orders],
        }

    def __repr__(self) -> str:
        """Return a concise representation for debugging."""
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state}, orders={len(self.orders)})")

import json
import logging
import os
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from globalvar import get_value, set_value
from order import OrderManager, find_best_rider_for_order

logger = logging.getLogger(__name__)

PROJECT_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
DEFAULT_CONFIG_PATH = os.path.join(PROJECT_ROOT, "config.json")
DEFAULT_STATE_PATH = os.path.join(PROJECT_ROOT, "simulation_state.json")

DEFAULT_CONFIG: Dict[str, int] = {
    "initial_funds": 1000,
    "recruitment_cost": 100,
    "max_rider_load": 3,
    "max_ticks_per_command": 1000,
}


def load_config(path: Optional[str] = None) -> Dict[str, int]:
    """Load system parameters from a JSON file, falling back to defaults.

    Unknown keys are ignored; invalid values are reported and replaced with
    defaults so that a bad configuration never stops the program.

    Args:
        path (str | None): Path to the JSON configuration file.

    Returns:
        dict: Effective configuration values.
    """
    config = dict(DEFAULT_CONFIG)
    path = path or DEFAULT_CONFIG_PATH
    if not os.path.exists(path):
        logger.info("No configuration file at %s; using defaults.", path)
        return config
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, ValueError) as exc:
        logger.warning("Could not read configuration %s: %s", path, exc)
        return config
    if not isinstance(data, dict):
        logger.warning("Configuration %s must be a JSON object.", path)
        return config
    for key in DEFAULT_CONFIG:
        if key not in data:
            continue
        value = data[key]
        if isinstance(value, int) and not isinstance(value, bool) and value >= 0:
            config[key] = value
        else:
            logger.warning("Invalid value for %s in configuration; "
                           "using default.", key)
    return config


class SystemManager:
    """Core coordinator of the simulation.

    Attributes:
        recruitmentCost (int): Cost deducted from funds per new rider.
        orderManager (OrderManager): Order creation and assignment.
        planner (PathPlanner): Shared planner for route projections.
        config (dict): Effective configuration.
    """

    def __init__(self, config: Optional[Dict[str, int]] = None):
        """Initialise global state from configuration.

        Args:
            config (dict | None): Parameter overrides; defaults are used for
                missing keys.
        """
        cfg = dict(DEFAULT_CONFIG)
        cfg.update(config or {})
        self.config = cfg
        self.recruitmentCost = int(cfg["recruitment_cost"])
        self.orderManager = OrderManager()
        self.planner = PathPlanner()
        set_value("funds", int(cfg["initial_funds"]))
        set_value("riders", [])
        set_value("rider_id_counter", 0)
        set_value("is_running", True)
        set_value("current_tick", 0)
        set_value("orders", [])
        set_value("order_history", [])
        set_value("order_id_counter", 0)
        set_value("max_rider_load", int(cfg["max_rider_load"]))

    @property
    def funds(self) -> int:
        """int: Current system funds (may be negative)."""
        return int(get_value("funds", 0))

    @funds.setter
    def funds(self, value: int) -> None:
        set_value("funds", int(value))

    @property
    def riderIdCounter(self) -> int:
        """int: Last issued rider id."""
        return int(get_value("rider_id_counter", 0))

    @riderIdCounter.setter
    def riderIdCounter(self, value: int) -> None:
        set_value("rider_id_counter", int(value))

    @property
    def isRunning(self) -> bool:
        """bool: Main loop control flag."""
        return bool(get_value("is_running", False))

    @isRunning.setter
    def isRunning(self, value: bool) -> None:
        set_value("is_running", bool(value))

    @property
    def currentTick(self) -> int:
        """int: Current simulated tick."""
        return int(get_value("current_tick", 0))

    @currentTick.setter
    def currentTick(self, value: int) -> None:
        set_value("current_tick", int(value))

    @property
    def riders(self) -> List[Rider]:
        """list[Rider]: The dynamic rider list."""
        return get_value("riders")

    def stop(self) -> None:
        """Request that the main loop terminate."""
        self.isRunning = False
        logger.info("System stop requested.")

    def can_afford_rider(self) -> bool:
        """Return True if funds cover the recruitment cost.

        Returns:
            bool: Affordability flag.
        """
        return self.funds >= self.recruitmentCost

    def recruit_rider(self) -> Optional[Rider]:
        """Recruit a new rider if funds allow.

        Returns:
            Rider | None: The new rider, or None when funds are insufficient.
        """
        if not self.can_afford_rider():
            logger.info("Recruitment skipped: insufficient funds (%s).",
                        self.funds)
            return None
        self.funds = self.funds - self.recruitmentCost
        self.riderIdCounter = self.riderIdCounter + 1
        rider = Rider(self.riderIdCounter)
        self.riders.append(rider)
        logger.info("Recruited rider %s; funds now %s.", rider.rider_id,
                    self.funds)
        return rider

    def dispatch_order(self, restaurant, customer
                       ) -> Tuple[Any, Rider, bool]:
        """Create an order and assign it, recruiting a rider if needed.

        Args:
            restaurant (Sequence[int]): Restaurant ``[x, y]``.
            customer (Sequence[int]): Customer ``[x, y]``.

        Returns:
            tuple[Order, Rider, bool]: The order, the assigned rider, and
            True when a new rider was recruited for this order.

        Raises:
            RuntimeError: If no rider has spare capacity and a new rider
                cannot be afforded.
        """
        order = self.orderManager.create_order(restaurant, customer)
        rider = find_best_rider_for_order(order)
        new_rider = False
        if rider is None:
            rider = self.recruit_rider()
            if rider is None:
                self.orderManager.discard_order(order)
                raise RuntimeError(
                    "No rider is available and funds are insufficient "
                    "to recruit a new rider.")
            new_rider = True
        self.orderManager.assign_order(order, rider)
        return order, rider, new_rider

    def tick(self, steps: int = 1) -> int:
        """Advance the simulation by one or more ticks.

        Each tick increments the clock and moves every rider. Exceptions
        raised by a rider are logged and the rider is recovered so that one
        faulty rider cannot stop the simulation.

        Args:
            steps (int): Number of ticks to simulate.

        Returns:
            int: The tick number after advancing.
        """
        for _ in range(max(0, int(steps))):
            self.currentTick = self.currentTick + 1
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception:  # noqa: BLE001 - recovery path
                    logger.exception("Rider %s failed; recovering.",
                                     rider.rider_id)
                    rider.recover()
        return self.currentTick

    def check_consistency(self) -> List[str]:
        """Verify internal invariants and report problems.

        Returns:
            list[str]: Human-readable issues; empty when the state is valid.
        """
        issues: List[str] = []
        ids = [r.rider_id for r in self.riders]
        if len(ids) != len(set(ids)):
            issues.append("duplicate rider ids detected")
        for rider in self.riders:
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    issues.append(f"order {order.id} is held by rider "
                                  f"{rider.rider_id} but records "
                                  f"{order.rider_id}")
                if order.status == order.COMPLETED:
                    issues.append(f"completed order {order.id} still in "
                                  f"rider {rider.rider_id}'s list")
        return issues

    def snapshot(self) -> Dict[str, Any]:
        """Return a JSON-compatible snapshot of the whole system.

        Returns:
            dict: Configuration, funds, tick, riders, and orders.
        """
        return {
            "config": self.config,
            "funds": self.funds,
            "current_tick": self.currentTick,
            "rider_id_counter": self.riderIdCounter,
            "riders": [r.to_dict() for r in self.riders],
            "orders": self.orderManager.to_dict_list(),
        }

    def save_state(self, path: Optional[str] = None) -> str:
        """Persist the system snapshot as JSON using an atomic write.

        Args:
            path (str | None): Destination file; defaults to
                ``simulation_state.json`` in the project root.

        Returns:
            str: Path of the written file.

        Raises:
            OSError: If the file cannot be written.
        """
        path = path or DEFAULT_STATE_PATH
        temp_path = path + ".tmp"
        with open(temp_path, "w", encoding="utf-8") as handle:
            json.dump(self.snapshot(), handle, indent=2)
        os.replace(temp_path, path)
        logger.info("State saved to %s", path)
        return path


def init_system(config_path: Optional[str] = None) -> SystemManager:
    """Initialise global variables and create the system manager.

    Args:
        config_path (str | None): Optional configuration file path.

    Returns:
        SystemManager: The initialised system.
    """
    globalvar._init()
    config = load_config(config_path)
    manager = SystemManager(config)
    logger.info("System initialised with funds %s.", manager.funds)
    return manager

import logging

from globalvar import get_value
from grid_map import GRID_DIMENSION, MAP_SIZE
from rider import Rider

logger = logging.getLogger(__name__)

HALF = 50  # one canvas character represents 50 units


def _sign(value: int) -> int:
    """Return -1, 0 or 1 according to the sign of ``value``."""
    return (value > 0) - (value < 0)


def _axis_path(a: Sequence[int], b: Sequence[int]):
    """Yield every unit point along an axis-aligned segment from a to b.

    Args:
        a (tuple[int, int]): Start point.
        b (tuple[int, int]): End point.

    Yields:
        tuple[int, int]: Points on the segment.
    """
    x0, y0 = a
    x1, y1 = b
    sx = _sign(x1 - x0)
    sy = _sign(y1 - y0)
    for i in range(abs(x1 - x0) + 1):
        yield x0 + sx * i, y0
    for j in range(1, abs(y1 - y0) + 1):
        yield x1, y0 + sy * j


class DisplayManager:
    """Formats status information and renders the ASCII map.

    The 800x800 map is drawn on a 17x17 character canvas: intersections are
    '+', horizontal streets '-', vertical streets '|', and location cells
    (cell centres) are '.' unless occupied by a marker.

    Attributes:
        symbolMap (dict[str, str]): Symbol for each map element.
    """

    CANVAS_SIZE = 2 * GRID_DIMENSION + 1
    ROUTE_REPLACEABLE = ("-", "|", "+", ".", " ")

    def __init__(self):
        """Create the display with the default symbol mapping."""
        self.symbolMap = {
            "empty": ".",
            "street_h": "-",
            "street_v": "|",
            "intersection": "+",
            "restaurant": "M",
            "customer": "C",
            "rider": "R",
            "route": "*",
            "space": " ",
        }

    def _blank_canvas(self) -> List[List[str]]:
        """Build the base canvas containing streets and empty cells.

        Returns:
            list[list[str]]: Canvas rows of symbols.
        """
        size = self.CANVAS_SIZE
        canvas = []
        for row in range(size):
            line = []
            for col in range(size):
                if row % 2 == 0 and col % 2 == 0:
                    line.append(self.symbolMap["intersection"])
                elif row % 2 == 0:
                    line.append(self.symbolMap["street_h"])
                elif col % 2 == 0:
                    line.append(self.symbolMap["street_v"])
                else:
                    line.append(self.symbolMap["empty"])
            canvas.append(line)
        return canvas

    @staticmethod
    def _to_canvas(x: int, y: int) -> Tuple[int, int]:
        """Convert a map coordinate to a canvas (row, col) index.

        Args:
            x (int): X coordinate.
            y (int): Y coordinate.

        Returns:
            tuple[int, int]: Row and column indices.
        """
        col = min(max(int(x), 0), MAP_SIZE) // HALF
        row = min(max(int(y), 0), MAP_SIZE) // HALF
        return row, col

    def _put(self, canvas, x: int, y: int, symbol: str) -> None:
        """Place a symbol at a map coordinate.

        Args:
            canvas (list[list[str]]): Canvas to modify.
            x (int): X coordinate.
            y (int): Y coordinate.
            symbol (str): Symbol to draw.
        """
        row, col = self._to_canvas(x, y)
        canvas[row][col] = symbol

    def _draw_route(self, canvas, route: Sequence[Tuple[int, int]]) -> None:
        """Draw a connected route line with the route symbol.

        Markers such as restaurants, customers, and riders are not
        overwritten.

        Args:
            canvas (list[list[str]]): Canvas to modify.
            route (sequence[tuple[int, int]]): Ordered route points.
        """
        route_symbol = self.symbolMap["route"]
        replaceable = set(self.ROUTE_REPLACEABLE)
        for a, b in zip(route, route[1:]):
            for x, y in _axis_path(a, b):
                row, col = self._to_canvas(x, y)
                if canvas[row][col] in replaceable:
                    canvas[row][col] = route_symbol

    def render_map(self, riders: Iterable[Rider] = (),
                   orders: Iterable[Any] = (),
                   route: Optional[Sequence[Tuple[int, int]]] = None) -> str:
        """Render the ASCII map with riders, orders, and an optional route.

        Args:
            riders (iterable[Rider]): Riders to mark with 'R'.
            orders (iterable[Order]): Active orders; restaurants are marked
                'M' and customers 'C'.
            route (sequence[tuple[int, int]] | None): Route to draw as a
                connected line of '*'. The first point is the start.

        Returns:
            str: Multi-line map text including a legend.
        """
        canvas = self._blank_canvas()
        if route:
            self._draw_route(canvas, route)
        for order in orders:
            if getattr(order, "status", "") == "completed":
                continue
            self._put(canvas, order.res_pos[0], order.res_pos[1],
                      self.symbolMap["restaurant"])
            self._put(canvas, order.cus_pos[0], order.cus_pos[1],
                      self.symbolMap["customer"])
        for rider in riders:
            self._put(canvas, rider.x, rider.y, self.symbolMap["rider"])

        lines = ["     x: 0 (left) -> 800 (right)"]
        for row_index, row in enumerate(canvas):
            if row_index % 2 == 1:
                label = f"{(row_index // 2) * 100 + 50:>4} "
            else:
                label = "     "
            lines.append(label + " ".join(row))
        lines.append("y: 0 (top) -> 800 (bottom); rows labelled at cell centres")
        lines.append("Legend: + intersection  - | street  . location  "
                     "M restaurant  C customer  R rider  * route")
        return "\n".join(lines)

    def format_status(self, system) -> str:
        """Build the system status report.

        Args:
            system (SystemManager): The running system.

        Returns:
            str: Funds, tick, rider table, and consistency information.
        """
        riders = system.riders
        active = [o for o in get_value("orders", []) if o.is_active]
        issues = system.check_consistency()
        lines = [
            "=== System Status ===",
            f"Funds: {system.funds}",
            f"Simulated tick: {system.currentTick}",
            f"Riders: {len(riders)} (recruitment cost {system.recruitmentCost})",
            f"Active orders: {len(active)}",
            f"Consistency: {'OK' if not issues else '; '.join(issues)}",
            "",
            f"{'ID':<6}{'Position':<16}{'State':<14}{'Speed':<7}Orders",
        ]
        for rider in riders:
            state = Rider.STATE_NAMES.get(rider.state, "unknown")
            order_ids = ", ".join(str(o.id) for o in rider.orders) or "-"
            pos = f"({rider.x}, {rider.y})"
            lines.append(f"{'R' + str(rider.rider_id):<6}{pos:<16}"
                         f"{state:<14}{rider.speed:<7}{order_ids}")
        if not riders:
            lines.append("No riders on duty.")
        return "\n".join(lines)

    def format_orders(self, orders: Iterable[Any]) -> str:
        """Build the order detail table.

        Args:
            orders (iterable[Order]): Orders to display.

        Returns:
            str: Formatted table text.
        """
        header = (f"{'ID':<5}{'Start':<7}{'Restaurant':<13}{'Customer':<13}"
                  f"{'Distance':<10}{'Time':<7}{'Rider':<7}Status")
        lines = ["=== Orders ===", header]
        count = 0
        for order in orders:
            count += 1
            time_text = (str(order.time_taken)
                         if order.time_taken is not None else "-")
            rider_text = (f"R{order.rider_id}"
                          if order.rider_id is not None else "-")
            res = f"({order.restaurant_x},{order.restaurant_y})"
            cus = f"({order.customer_x},{order.customer_y})"
            lines.append(f"{order.id:<5}{order.start_time:<7}{res:<13}"
                         f"{cus:<13}{order.distance:<10.1f}{time_text:<7}"
                         f"{rider_text:<7}{order.status}")
        if count == 0:
            lines.append("No orders yet.")
        return "\n".join(lines)

    def format_rider_logs(self, riders: Iterable[Rider]) -> str:
        """Build the chronological rider event log.

        Args:
            riders (iterable[Rider]): Riders whose logs are shown.

        Returns:
            str: One line per status change, ordered by tick.
        """
        entries = []
        for rider in riders:
            entries.extend(rider.log)
        entries.sort(key=lambda e: (e["tick"], e["rider_id"]))
        lines = ["=== Rider Event Log ==="]
        for entry in entries:
            order_text = (f"order {entry['order_id']}"
                          if entry["order_id"] is not None else "-")
            state = Rider.STATE_NAMES.get(entry["state"], "unknown")
            lines.append(f"[tick {entry['tick']:>4}] R{entry['rider_id']:<3} "
                         f"{entry['event']:<22} {order_text:<12} "
                         f"at ({entry['x']}, {entry['y']}) state={state}")
        if not entries:
            lines.append("No rider events yet.")
        return "\n".join(lines)

class InputManager:
    """Parses and executes user commands.

    Attributes:
        system (SystemManager): The system being controlled.
        display (DisplayManager): Renderer for tables and maps.
        commandHistory (list[str]): Commands entered so far.
    """

    def __init__(self, system, display: Optional[DisplayManager] = None,
                 input_func: Optional[Callable[[str], str]] = None,
                 output_func: Optional[Callable[..., Any]] = None):

        Args:
            system (SystemManager): The system to control.
            display (DisplayManager | None): Renderer instance.
            input_func (callable | None): Function used to read input.
            output_func (callable | None): Function used to print output.
        """
        self.system = system
        self.display = display or DisplayManager()
        self.commandHistory: List[str] = []
        self._input = input_func or input
        self._output = output_func or print
        self._planner = PathPlanner()
        self._handlers: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def _print(self, text: str = "") -> None:
        """Print text through the configured output function."""
        self._output(text)

    def run(self) -> None:
        """Run the interactive loop until the system stops."""
        self._print("Food delivery dispatch simulator. Type 'help' for "
                    "commands.")
        while self.system.isRunning:
            try:
                line = self._input("dispatch> ")
            except EOFError:
                self._print()
                self.system.stop()
                break
            except KeyboardInterrupt:
                self._print("\nUse 'quit' to exit.")
                continue
            self.process_line(line)

    def process_line(self, line: str) -> None:
        """Expand shortcuts, record history, and dispatch one command.

        Args:
            line (str): Raw user input.
        """
        line = (line or "").strip()
        if not line:
            return
        try:
            if line == "!!":
                if not self.commandHistory:
                    self._print("No previous command to repeat.")
                    return
                line = self.commandHistory[-1]
            elif line.startswith("!") and line[1:].isdigit():
                index = int(line[1:])
                if not 1 <= index <= len(self.commandHistory):
                    self._print(f"No history entry {index}.")
                    return
                line = self.commandHistory[index - 1]
            self.commandHistory.append(line)
            tokens = line.split()
            name = tokens[0].lower()
            name = COMMAND_ALIASES.get(name, name)
            handler = self._handlers.get(name)
            if handler is None:
                self._print(f"Unknown command '{tokens[0]}'. "
                            f"Type 'help' for the command list.")
                return
            handler(tokens[1:])
        except ValueError as exc:
            self._print(f"Invalid input: {exc}")
        except RuntimeError as exc:
            self._print(f"Error: {exc}")
        except EOFError:
            self._print()
            self.system.stop()
        except Exception:  # noqa: BLE001 - keep the session alive
            logger.exception("Unexpected error while running command.")
            self._print("An unexpected error occurred; system state was "
                        "preserved. See the log for details.")

    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> List[int]:
        """Validate that tokens are integers within the map range.

        Args:
            tokens (sequence[str]): Raw coordinate strings.

        Returns:
            list[int]: Parsed integer coordinates.

        Raises:
            ValueError: If a token is not an integer or is out of range.
        """
        values = []
        for token in tokens:
            try:
                value = int(token)
            except (TypeError, ValueError):
                raise ValueError(f"'{token}' is not an integer.")
            if not 0 <= value <= MAP_SIZE:
                raise ValueError(f"coordinate {value} is out of range "
                                 f"0-{MAP_SIZE}.")
            values.append(value)
        return values

    def _prompt_coordinates(self) -> Optional[List[int]]:
        """Interactively ask for the four order coordinates.

        Returns:
            list[int] | None: The four coordinates, or None if cancelled.
        """
        labels = ["restaurant x", "restaurant y", "customer x", "customer y"]
        values: List[int] = []
        for label in labels:
            while True:
                raw = self._input(
                    f"  {label} (0-{MAP_SIZE}, 'cancel' to abort): ").strip()
                if raw.lower() == "cancel":
                    return None
                try:
                    values.extend(self.parse_coordinates([raw]))
                except ValueError as exc:
                    self._print(f"  Invalid: {exc}")
                    continue
                break
        return values

    def _projected_route(self, rider, order) -> List[tuple]:
        """Build the route from the rider's position through the order.

        Args:
            rider (Rider): Assigned rider.
            order (Order): The new order.

        Returns:
            list[tuple[int, int]]: Route points starting at the rider.
        """
        to_restaurant = self._planner.plan(rider.position, order.res_pos)
        to_customer = self._planner.plan(order.res_pos, order.cus_pos)
        return [rider.position] + to_restaurant + to_customer

    def cmd_start(self, args: List[str]) -> None:
        """Place a new order and show the assignment and route.

        Args:
            args (list[str]): Either empty (interactive prompts) or four
                integers: restaurant x, y and customer x, y.

        Raises:
            ValueError: If the arguments are malformed.
        """
        if args:
            if len(args) != 4:
                raise ValueError("start needs four integers: restaurant_x "
                                 "restaurant_y customer_x customer_y.")
            values = self.parse_coordinates(args)
        else:
            values = self._prompt_coordinates()
            if values is None:
                self._print("Order entry cancelled.")
                return
        restaurant = (values[0], values[1])
        customer = (values[2], values[3])
        cost = self.system.recruitmentCost
        order, rider, new_rider = self.system.dispatch_order(restaurant,
                                                             customer)
        self._print(f"Restaurant grid position: {order.res_pos} | "
                    f"Customer grid position: {order.cus_pos} | "
                    f"Order #{order.id}")
        if new_rider:
            self._print(f"New rider R{rider.rider_id} recruited (cost {cost}, "
                        f"funds left {self.system.funds}).")
        else:
            self._print(f"Assigned to existing rider R{rider.rider_id} "
                        f"(now carrying {len(rider.orders)} order(s)).")
        route = self._projected_route(rider, order)
        length = PathPlanner.route_length(route[0], route[1:])
        eta = math.ceil(length / rider.speed) if length else 0
        self._print(f"Rider R{rider.rider_id}: position {rider.position}, "
                    f"speed {rider.speed} units/tick.")
        self._print(f"Projected route: {length} units, about {eta} tick(s).")
        if len(rider.orders) > 1:
            self._print("Note: the rider has other queued orders; the route "
                        "and ETA above ignore them.")
        self._print(self.display.render_map(riders=[rider],
                                            orders=[order], route=route))

    def cmd_status(self, args: List[str]) -> None:
        """Show system funds, riders, and their states.

        Args:
            args (list[str]): Unused.
        """
        self._print(self.display.format_status(self.system))

    def cmd_orders(self, args: List[str]) -> None:
        """Show details for all orders.

        Args:
            args (list[str]): Unused.
        """
        self._print(self.display.format_orders(
            get_value("orders", []) or []))

    def cmd_riders(self, args: List[str]) -> None:
        """Show the rider status-change event log.

        Args:
            args (list[str]): Unused.
        """
        self._print(self.display.format_rider_logs(self.system.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map with riders and active orders.

        Args:
            args (list[str]): Unused.
        """
        active = [o for o in get_value("orders", []) if o.is_active]
        self._print(self.display.render_map(riders=self.system.riders,
                                            orders=active))

    def cmd_tick(self, args: List[str]) -> None:
        """Advance simulated time.

        Args:
            args (list[str]): Optional number of ticks.

        Raises:
            ValueError: If the count is not a positive integer within the
                configured limit.
        """
        steps = 1
        if args:
            steps = self.parse_coordinates(args[:1]) and int(args[0])
            limit = int(self.system.config.get("max_ticks_per_command", 1000))
            if steps < 1 or steps > limit:
                raise ValueError(f"tick count must be between 1 and {limit}.")
        self.system.tick(steps)
        active = [o for o in get_value("orders", []) if o.is_active]
        completed = len(get_value("order_history", []))
        self._print(f"Advanced {steps} tick(s). Now at tick "
                    f"{self.system.currentTick}. Active orders: {len(active)}, "
                    f"completed: {completed}.")

    def cmd_save(self, args: List[str]) -> None:
        """Save a JSON snapshot of the system.

        Args:
            args (list[str]): Optional destination file path.
        """
        try:
            path = self.system.save_state(args[0] if args else None)
            self._print(f"System state saved to {path}.")
        except OSError as exc:
            self._print(f"Could not save state: {exc}")

    def cmd_history(self, args: List[str]) -> None:
        """List previously entered commands.

        Args:
            args (list[str]): Unused.
        """
        if not self.commandHistory:
            self._print("No commands entered yet.")
            return
        for index, command in enumerate(self.commandHistory, start=1):
            self._print(f"{index:>4}  {command}")

    def cmd_help(self, args: List[str]) -> None:
        """Print the command reference.

        Args:
            args (list[str]): Unused.
        """
        self._print(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Terminate the program.

        Args:
            args (list[str]): Unused.
        """
        self.system.stop()
        self._print("Goodbye.")

"""Entry point for the food delivery dispatch simulator.

Usage:
    python src/main.py [--config PATH] [--debug]
"""

import argparse
import logging
import sys
from typing import Optional, Sequence

from input_manager import InputManager
from system import init_system

logger = logging.getLogger("dispatch")


def parse_args(argv: Optional[Sequence[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv (sequence[str] | None): Arguments; defaults to sys.argv.

    Returns:
        argparse.Namespace: Parsed options.
    """
    parser = argparse.ArgumentParser(
        description="Command-line food delivery dispatch simulator.")
    parser.add_argument("--config", default=None,
                        help="path to a JSON configuration file")
    parser.add_argument("--debug", action="store_true",
                        help="enable debug logging to stderr")
    return parser.parse_args(argv)


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Initialise the system and run the interactive loop.

    Args:
        argv (sequence[str] | None): Command-line arguments.

    Returns:
        int: Process exit code (0 on normal exit, 1 on startup failure).
    """
    args = parse_args(argv)
    logging.basicConfig(
        level=logging.DEBUG if args.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s")
    try:
        import readline  # noqa: F401 - enables line editing and history
    except ImportError:
        pass
    try:
        system = init_system(args.config)
    except Exception:  # noqa: BLE001 - report startup failure cleanly
        logger.exception("Failed to initialise the system.")
        print("Startup failed; see the log for details.")
        return 1
    InputManager(system).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

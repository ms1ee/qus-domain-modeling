"""Grid-based path planning for riders on the 800x800 street map.

The map is divided into 100x100 unit cells. Streets are the grid lines where
either the x or the y coordinate is a multiple of 100. Riders prefer to travel
along streets, using right-angle turns between straight segments.
"""

from typing import List, Sequence, Tuple

Point = Tuple[int, int]

MAP_SIZE = 800
CELL_SIZE = 100


def is_on_street(x: int, y: int) -> bool:
    """Return True if the point lies on a street (grid line).

    Args:
        x: X coordinate.
        y: Y coordinate.

    Returns:
        True when ``x`` or ``y`` is aligned with a grid line.
    """
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


def _clamp(value: int) -> int:
    """Clamp a coordinate into the map bounds."""
    return max(0, min(MAP_SIZE, value))


class PathPlanner:
    """Computes routes and per-tick movement steps on the street grid."""

    @staticmethod
    def plan(start: Point, end: Point) -> List[Point]:
        """Plan a street-following route between two points.

        The route moves horizontally first (along the street through the
        start point), then turns at a right angle towards the end point.

        Args:
            start: Starting coordinate.
            end: Destination coordinate.

        Returns:
            List of waypoints including both endpoints.
        """
        sx, sy = start
        ex, ey = end
        if sx == ex or sy == ey:
            return [(sx, sy), (ex, ey)]
        return [(sx, sy), (ex, sy), (ex, ey)]

    @classmethod
    def plan_route(cls, points: Sequence[Point]) -> List[Point]:
        """Plan a continuous route through several destinations in order.

        Args:
            points: Ordered list of coordinates to visit, e.g.
                ``[rider_position, restaurant, customer]``.

        Returns:
            Concatenated list of waypoints with duplicate joints removed.
        """
        if not points:
            return []
        route: List[Point] = [(int(points[0][0]), int(points[0][1]))]
        for nxt in points[1:]:
            target = (int(nxt[0]), int(nxt[1]))
            segment = cls.plan(route[-1], target)
            route.extend(segment[1:])
        return route

    @staticmethod
    def route_length(route: Sequence[Point]) -> int:
        """Return the total Manhattan length of a route.

        Args:
            route: Waypoints of the route.

        Returns:
            Sum of Manhattan distances between consecutive waypoints.
        """
        return sum(
            abs(b[0] - a[0]) + abs(b[1] - a[1])
            for a, b in zip(route, route[1:])
        )

    @staticmethod
    def next_position(x: int, y: int, tx: int, ty: int,
                      speed: int) -> Tuple[int, int, bool]:
        """Compute the new position after moving up to ``speed`` units.

        Movement rules:
            * If the rider is directly above or below the target (same x), it
              moves vertically and the special-case flag is set.
            * If the rider is on a horizontal street, it moves horizontally.
            * Otherwise it moves vertically along the current vertical street.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            tx: Target x coordinate.
            ty: Target y coordinate.
            speed: Maximum distance travelled this tick.

        Returns:
            Tuple ``(new_x, new_y, special_case)``.
        """
        special = False
        remaining = max(0, int(speed))
        while remaining > 0 and (x, y) != (tx, ty):
            if x == tx:
                special = True
                step = min(remaining, abs(ty - y))
                y += step if ty > y else -step
                remaining -= step
            elif y % CELL_SIZE == 0 or y == ty:
                step = min(remaining, abs(tx - x))
                x += step if tx > x else -step
                remaining -= step
            else:
                step = min(remaining, abs(ty - y))
                y += step if ty > y else -step
                remaining -= step
        return _clamp(x), _clamp(y), special

"""Order data model and coordinate conversion helpers.

User input coordinates (0-800) are snapped to the top-left corner of the
100x100 grid cell that contains them. Cell index is clamped to the 8x8 grid.
"""

import itertools
from typing import Any, Dict, Optional, Tuple

import globalvar

GRID_CELL_SIZE = 100
GRID_CELLS = 8
MAX_CELL_INDEX = GRID_CELLS - 1

_id_counter = itertools.count(1)


def snap_to_cell(value: int) -> int:
    """Convert a coordinate to the top-left corner of its grid cell.

    Args:
        value: Input coordinate on one axis.

    Returns:
        Grid coordinate (multiple of 100) of the containing cell.
    """
    v = max(0, int(value))
    index = min(v // GRID_CELL_SIZE, MAX_CELL_INDEX)
    return index * GRID_CELL_SIZE


def to_grid_cell(x: int, y: int) -> Tuple[int, int]:
    """Convert an input coordinate pair to its grid cell top-left corner.

    Args:
        x: Input x coordinate.
        y: Input y coordinate.

    Returns:
        Tuple ``(grid_x, grid_y)``.
    """
    return snap_to_cell(x), snap_to_cell(y)


def cell_center(grid_x: int, grid_y: int) -> Tuple[int, int]:
    """Return the center point of the cell whose top-left corner is given.

    Uses ``center_x = (grid_x // 100) * 100 + 50``.

    Args:
        grid_x: X coordinate of the cell.
        grid_y: Y coordinate of the cell.

    Returns:
        Tuple ``(center_x, center_y)``.
    """
    return (grid_x // GRID_CELL_SIZE) * GRID_CELL_SIZE + 50, \
        (grid_y // GRID_CELL_SIZE) * GRID_CELL_SIZE + 50


def manhattan(a: Tuple[int, int], b: Tuple[int, int]) -> int:
    """Return the Manhattan distance between two points.

    Args:
        a: First point.
        b: Second point.

    Returns:
        ``|ax - bx| + |ay - by|``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        res_pos: Restaurant grid position (top-left of its cell).
        cus_pos: Customer grid position (top-left of its cell).
        status: One of ``"created"``, ``"assigned"``, ``"picked_up"``,
            ``"completed"``.
        rider_id: ID of the assigned rider, or None.
        start_time: Tick at which the order was created.
        distance: Manhattan distance from restaurant to customer.
        completion_time: Tick at which delivery finished, or None.
        time_taken: Ticks between creation and completion, or None.
        efficiency: Distance covered per tick, or None.
    """

    CREATED = "created"
    ASSIGNED = "assigned"
    PICKED_UP = "picked_up"
    COMPLETED = "completed"

    def __init__(self, res_pos: Tuple[int, int],
                 cus_pos: Tuple[int, int]) -> None:
        """Create an order from raw input coordinates.

        Args:
            res_pos: Restaurant coordinate ``(x, y)``.
            cus_pos: Customer coordinate ``(x, y)``.
        """
        self.id: int = next(_id_counter)
        self.res_pos: Tuple[int, int] = to_grid_cell(*res_pos)
        self.cus_pos: Tuple[int, int] = to_grid_cell(*cus_pos)
        self.status: str = self.CREATED
        self.rider_id: Optional[int] = None
        self.start_time: int = globalvar.get_value("tick", 0) or 0
        self.distance: int = manhattan(self.res_pos, self.cus_pos)
        self.completion_time: Optional[int] = None
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = self.ASSIGNED

    def mark_picked_up(self) -> None:
        """Mark the order as picked up at the restaurant."""
        self.status = self.PICKED_UP

    def mark_completed(self, tick: int) -> None:
        """Mark the order as delivered and record efficiency data.

        Args:
            tick: Current simulated tick at delivery time.
        """
        self.status = self.COMPLETED
        self.completion_time = tick
        self.time_taken = max(0, tick - self.start_time)
        self.efficiency = self.distance / max(1, self.time_taken)

    def to_dict(self) -> Dict[str, Any]:
        """Serialize the order to a JSON-compatible dictionary.

        Returns:
            Dictionary representation of the order.
        """
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "distance": self.distance,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

"""Rider model: state transitions, movement and event logging."""

import logging
from typing import Any, Dict, List, Optional, Sequence, Tuple

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

RIDER_SPEED = 20
SPAWN_X = 400
SPAWN_Y = 400


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    State values:
        0 (IDLE): available for new orders.
        1 (PICKING_UP): en route to the restaurant.
        2 (DELIVERING): en route to the customer.

    Log entries (``rider.log``) are dicts with at least the keys
    ``"event"`` and ``"order_id"``. Events are ``"spawn"``,
    ``"accepted_order"``, ``"arrived_at_merchant"`` and
    ``"delivered_to_customer"``.
    """

    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2
    STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the spawn point, idle, with no orders.

        Args:
            rider_id: Unique identifier of the rider.
        """
        self.rider_id: int = int(rider_id)
        self.x: int = SPAWN_X
        self.y: int = SPAWN_Y
        self.speed: int = RIDER_SPEED
        self.state: int = self.IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    @property
    def position(self) -> Tuple[int, int]:
        """Current ``(x, y)`` position."""
        return self.x, self.y

    @property
    def state_name(self) -> str:
        """Human-readable name of the current state."""
        return self.STATE_NAMES.get(self.state, "unknown")

    def active_orders(self) -> List[Any]:
        """Return orders assigned to this rider that are not yet completed.

        Returns:
            List of active :class:`order_model.Order` objects in FIFO order.
        """
        return [o for o in self.orders if o.status != "completed"]

    def current_order(self) -> Optional[Any]:
        """Return the order currently being served, if any.

        Returns:
            The first active order, or None.
        """
        active = self.active_orders()
        return active[0] if active else None

    def _now(self) -> int:
        """Return the current simulated tick."""
        return globalvar.get_value("tick", 0) or 0

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider's log.

        Args:
            event: Event name.
            order_id: Related order ID, or None.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s order=%s", self.rider_id, event, order_id)

    def _advance(self, target: Sequence[int]) -> bool:
        """Move one tick towards a target along the street grid.

        Args:
            target: Destination coordinate.

        Returns:
            True if the rider is at the target after this movement.
        """
        tx, ty = int(target[0]), int(target[1])
        if (self.x, self.y) == (tx, ty):
            self.special_case = False
            return True
        nx, ny, special = PathPlanner.next_position(
            self.x, self.y, tx, ty, self.speed)
        self.x, self.y = nx, ny
        self.special_case = special
        return (nx, ny) == (tx, ty)

    def move(self) -> None:
        """Advance the rider by one tick.

        Performs state transitions and records log entries:
            * Idle -> Picking Up when an order is assigned ("accepted_order").
            * Picking Up -> Delivering on reaching the restaurant
              ("arrived_at_merchant").
            * Delivering -> Picking Up (or Idle) on delivering to the
              customer ("delivered_to_customer").
        """
        order = self.current_order()
        if order is None:
            if self.state != self.IDLE:
                self.state = self.IDLE
                self.special_case = False
            return

        if self.state == self.IDLE:
            self.state = self.PICKING_UP
            self._record("accepted_order", order.id)

        if self.state == self.PICKING_UP:
            if self._advance(order.res_pos):
                order.mark_picked_up()
                self.state = self.DELIVERING
                self._record("arrived_at_merchant", order.id)
            return

        if self.state == self.DELIVERING:
            if self._advance(order.cus_pos):
                order.mark_completed(self._now())
                self._record("delivered_to_customer", order.id)
                next_order = self.current_order()
                if next_order is not None:
                    self._record("accepted_order", next_order.id)
                    self.state = self.PICKING_UP
                else:
                    self.state = self.IDLE
                    self.special_case = False

    def to_dict(self) -> Dict[str, Any]:
        """Serialize the rider to a JSON-compatible dictionary.

        Returns:
            Dictionary representation of the rider and its log.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [o.id for o in self.orders],
            "log": list(self.log),
        }

"""Order creation, assignment and the best-rider selection algorithm."""

import logging
from typing import TYPE_CHECKING, List, Optional, Tuple

import globalvar
from order_model import Order, manhattan

if TYPE_CHECKING:  # pragma: no cover
    from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional["Rider"]:
    """Select the rider that should receive an order.

    Selection criteria, in priority order:
        1. Fewest current (active) orders.
        2. Closest to the restaurant by Manhattan distance.
        3. Lowest rider ID (deterministic tie-break).

    Args:
        order: The order to be assigned.

    Returns:
        The chosen :class:`rider.Rider`, or None if no riders exist.
    """
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None
    target = order.res_pos
    return min(
        riders,
        key=lambda r: (
            len(r.active_orders()),
            manhattan(r.position, target),
            r.rider_id,
        ),
    )


class OrderManager:
    """Creates orders, keeps the order registry and assigns pending orders."""

    def create_order(self, restaurant_xy: Tuple[int, int],
                     customer_xy: Tuple[int, int]) -> Order:
        """Create a new order and register it in the global order list.

        Args:
            restaurant_xy: Restaurant input coordinate ``(x, y)``.
            customer_xy: Customer input coordinate ``(x, y)``.

        Returns:
            The newly created :class:`order_model.Order`.
        """
        order = Order(restaurant_xy, customer_xy)
        globalvar.get_value("orders").append(order)
        logger.info("Order %s created: %s -> %s",
                    order.id, order.res_pos, order.cus_pos)
        return order

    def all_orders(self) -> List[Order]:
        """Return every order ever created, in creation order."""
        return list(globalvar.get_value("orders", []) or [])

    def pending_orders(self) -> List[Order]:
        """Return orders that have not yet been assigned to a rider."""
        return [o for o in self.all_orders() if o.status == Order.CREATED]

    def get_order(self, order_id: int) -> Optional[Order]:
        """Look up an order by its ID.

        Args:
            order_id: Identifier of the order.

        Returns:
            The matching order, or None.
        """
        for order in self.all_orders():
            if order.id == order_id:
                return order
        return None

    def assign_pending_orders(self) -> List[Tuple[Order, "Rider"]]:
        """Assign every pending order to the best available rider.

        Returns:
            List of ``(order, rider)`` pairs that were assigned in this call.
        """
        assigned: List[Tuple[Order, "Rider"]] = []
        for order in self.pending_orders():
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            rider.orders.append(order)
            order.assign_to(rider.rider_id)
            assigned.append((order, rider))
            logger.info("Order %s assigned to rider %s",
                        order.id, rider.rider_id)
        return assigned

"""System-level resource management: funds, riders and simulation ticks."""

import json
import logging
import os
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from order import OrderManager
from rider import Rider

logger = logging.getLogger(__name__)

DEFAULT_PARAMETERS: Dict[str, int] = {
    "initial_funds": 1000,
    "recruit_funds_threshold": 300,
    "recruit_cost": 500,
}


def init_system() -> None:
    """Initialize core global parameters on system startup.

    Sets funds to 1000, an empty rider list, the rider ID counter to 0, the
    running flag to True, the tick to 0 and an empty order list.
    """
    globalvar._init()
    globalvar.set_value("funds", DEFAULT_PARAMETERS["initial_funds"])
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    logger.info("System initialized")


class SystemManager:
    """Coordinates funds, rider recruitment, order dispatch and ticks."""

    def __init__(self) -> None:
        """Initialize the system and its order manager."""
        init_system()
        self.params: Dict[str, int] = dict(DEFAULT_PARAMETERS)
        self.order_manager = OrderManager()

    # ----- global state accessors -------------------------------------
    @property
    def funds(self) -> int:
        """Current system funds (may be negative)."""
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value: int) -> None:
        globalvar.set_value("funds", value)

    @property
    def riders(self) -> List[Rider]:
        """List of all riders currently in service."""
        return globalvar.get_value("riders", [])

    @property
    def tick_count(self) -> int:
        """Number of simulated ticks elapsed."""
        return globalvar.get_value("tick", 0)

    @property
    def running(self) -> bool:
        """Whether the main loop should keep running."""
        return bool(globalvar.get_value("running", False))

    def stop(self) -> None:
        """Signal the main loop to stop."""
        globalvar.set_value("running", False)

    def get_rider(self, rider_id: int) -> Optional[Rider]:
        """Find a rider by ID.

        Args:
            rider_id: Identifier of the rider.

        Returns:
            The matching rider, or None.
        """
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def idle_riders(self) -> List[Rider]:
        """Return riders that currently have no active orders."""
        return [r for r in self.riders if not r.active_orders()]

    def pending_count(self) -> int:
        """Return the number of orders waiting for assignment."""
        return len(self.order_manager.pending_orders())

    # ----- recruitment -----------------------------------------------
    def add_rider(self) -> Rider:
        """Recruit a new rider, deducting the recruitment cost.

        Returns:
            The newly created rider.
        """
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        self.funds = self.funds - self.params["recruit_cost"]
        logger.info("Recruited rider %s (funds now %s)", counter, self.funds)
        return rider

    def maybe_recruit(self) -> int:
        """Recruit riders while funds allow and pending orders exceed idle riders.

        Returns:
            Number of riders recruited.
        """
        count = 0
        while (self.funds >= self.params["recruit_funds_threshold"]
               and self.pending_count() > len(self.idle_riders())):
            self.add_rider()
            count += 1
        return count

    # ----- dispatch --------------------------------------------------
    def dispatch(self) -> List[Any]:
        """Recruit if needed and assign all pending orders.

        Returns:
            List of ``(order, rider)`` pairs assigned in this call.
        """
        self.maybe_recruit()
        return self.order_manager.assign_pending_orders()

    def submit_order(self, restaurant_xy: Tuple[int, int],
                     customer_xy: Tuple[int, int]) -> Dict[str, Any]:
        """Create an order, recruit if necessary and assign it.

        Args:
            restaurant_xy: Restaurant input coordinate.
            customer_xy: Customer input coordinate.

        Returns:
            Dict with keys ``"order"``, ``"new_rider"`` (bool) and
            ``"rider"`` (assigned Rider or None).
        """
        order = self.order_manager.create_order(restaurant_xy, customer_xy)
        recruited = self.maybe_recruit()
        self.order_manager.assign_pending_orders()
        rider = None
        if order.rider_id is not None:
            rider = self.get_rider(order.rider_id)
        return {"order": order, "new_rider": recruited > 0, "rider": rider}

    def tick(self) -> int:
        """Advance the simulation by one tick.

        Increments the tick counter, dispatches pending orders and moves
        every rider. Errors for a single rider are logged and do not stop
        the simulation.

        Returns:
            Number of log events produced by riders during this tick.
        """
        globalvar.set_value("tick", self.tick_count + 1)
        self.dispatch()
        events = 0
        for rider in list(self.riders):
            before = len(rider.log)
            try:
                rider.move()
            except Exception:  # error recovery: keep the system running
                logger.exception("Rider %s failed to move", rider.rider_id)
            events += len(rider.log) - before
        return events

    # ----- persistence -----------------------------------------------
    def save_state(self, path: str) -> None:
        """Write parameters, riders and orders to a JSON file.

        Args:
            path: Destination file path.
        """
        data = {
            "parameters": dict(self.params),
            "system": {
                "funds": self.funds,
                "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
                "tick": self.tick_count,
            },
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.all_orders()],
        }
        tmp_path = path + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        os.replace(tmp_path, path)
        logger.info("State saved to %s", path)

    def load_config(self, path: str) -> Dict[str, int]:
        """Load system parameters from a JSON configuration file.

        Only known parameter keys are applied. If ``initial_funds`` is
        provided, system funds are reset to that value.

        Args:
            path: Path of the JSON configuration file.

        Returns:
            The effective parameter dictionary.

        Raises:
            ValueError: If the file does not contain valid parameters.
        """
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
        params = data.get("parameters", data)
        if not isinstance(params, dict):
            raise ValueError("Configuration 'parameters' must be an object")
        for key in DEFAULT_PARAMETERS:
            if key in params:
                value = params[key]
                if not isinstance(value, int) or isinstance(value, bool):
                    raise ValueError(f"Parameter '{key}' must be an integer")
                self.params[key] = value
        if "initial_funds" in params:
            self.funds = self.params["initial_funds"]
        logger.info("Configuration loaded from %s", path)
        return dict(self.params)

"""Status display and ASCII map visualisation.

The ASCII map is 16x16 characters. Each character covers a 50x50 unit area,
so the displayed area spans 0-750 on both axes. Symbols:
    R = rider, S = restaurant, C = customer, # = street, . = empty,
    * = planned route line.
"""

from typing import Any, Iterable, List, Optional, Sequence, Tuple

MAP_CELLS = 16
MAP_UNIT = 50
CELL_SIZE = 100

SYMBOL_RIDER = "R"
SYMBOL_RESTAURANT = "S"
SYMBOL_CUSTOMER = "C"
SYMBOL_EMPTY = "."
SYMBOL_STREET = "#"
SYMBOL_ROUTE = "*"


class DisplayManager:
    """Builds textual status tables and the command-line map."""

    @staticmethod
    def _index(value: float) -> int:
        """Map a coordinate to a map character index (clamped)."""
        return min(max(int(value), 0) // MAP_UNIT, MAP_CELLS - 1)

    def render_status(self, funds: int, riders: Sequence[Any],
                      tick: int, pending: int) -> str:
        """Render a snapshot of system funds, riders and their orders.

        Args:
            funds: Current system funds.
            riders: Riders to display.
            tick: Current simulated tick.
            pending: Number of unassigned orders.

        Returns:
            Multi-line status text.
        """
        lines = [
            f"=== System Status (tick {tick}) ===",
            f"System funds: {funds}",
            f"Pending (unassigned) orders: {pending}",
            f"Riders: {len(riders)}",
        ]
        if riders:
            lines.append(
                f"{'ID':<5}{'Position':<14}{'Status':<14}"
                f"{'Active':<8}Order IDs")
            for rider in riders:
                active = rider.active_orders()
                order_ids = [o.id for o in rider.orders]
                lines.append(
                    f"#{rider.rider_id:<4}"
                    f"{str((rider.x, rider.y)):<14}"
                    f"{rider.state_name:<14}"
                    f"{len(active):<8}{order_ids}")
        return "\n".join(lines)

    def render_orders(self, orders: Iterable[Any]) -> str:
        """Render a table of all orders.

        Args:
            orders: Orders to display.

        Returns:
            Multi-line table text.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        lines = [
            f"{'ID':<5}{'Start':<7}{'Restaurant':<13}{'Customer':<13}"
            f"{'Dist':<6}{'Taken':<7}{'Rider':<7}Status",
        ]
        for o in orders:
            taken = "-" if o.time_taken is None else str(o.time_taken)
            rider = "-" if o.rider_id is None else f"#{o.rider_id}"
            lines.append(
                f"{o.id:<5}{o.start_time:<7}{str(o.res_pos):<13}"
                f"{str(o.cus_pos):<13}{o.distance:<6}{taken:<7}"
                f"{rider:<7}{o.status}")
            if o.completion_time is not None:
                lines.append(
                    f"      completed at tick {o.completion_time}, "
                    f"efficiency {o.efficiency:.2f} units/tick")
        return "\n".join(lines)

    def render_rider_logs(self, riders: Sequence[Any]) -> str:
        """Render every status-change record for all riders.

        Args:
            riders: Riders whose logs should be displayed.

        Returns:
            Multi-line log text.
        """
        if not riders:
            return "No riders."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"--- Rider #{rider.rider_id} ---")
            for entry in rider.log:
                order = ("-" if entry["order_id"] is None
                         else f"order #{entry['order_id']}")
                lines.append(
                    f"  t={entry['time']:<5}{entry['event']:<22}{order:<12}"
                    f"at ({entry['x']}, {entry['y']}) "
                    f"state={entry['state']}")
        return "\n".join(lines)

    def render_map(self, riders: Sequence[Any] = (),
                   orders: Sequence[Any] = (),
                   route: Optional[Sequence[Tuple[int, int]]] = None) -> str:
        """Render the 16x16 ASCII map with riders, orders and an optional route.

        Args:
            riders: Riders to mark with 'R'.
            orders: Orders whose restaurant ('S') and customer ('C') to mark.
            route: Optional list of waypoints drawn as a connected line.

        Returns:
            Multi-line map text including a legend.
        """
        size = MAP_CELLS
        grid = [[SYMBOL_EMPTY] * size for _ in range(size)]
        for row in range(size):
            for col in range(size):
                x, y = col * MAP_UNIT, row * MAP_UNIT
                if x % CELL_SIZE == 0 or y % CELL_SIZE == 0:
                    grid[row][col] = SYMBOL_STREET

        def mark(x: float, y: float, symbol: str) -> None:
            grid[self._index(y)][self._index(x)] = symbol

        if route:
            for (x1, y1), (x2, y2) in zip(route, route[1:]):
                dist = abs(x2 - x1) + abs(y2 - y1)
                steps = max(1, dist // 10)
                for i in range(steps + 1):
                    t = i / steps
                    mark(x1 + (x2 - x1) * t, y1 + (y2 - y1) * t, SYMBOL_ROUTE)

        for order in orders:
            mark(order.res_pos[0], order.res_pos[1], SYMBOL_RESTAURANT)
            mark(order.cus_pos[0], order.cus_pos[1], SYMBOL_CUSTOMER)

        for rider in riders:
            mark(rider.x, rider.y, SYMBOL_RIDER)

        lines = []
        for row in range(size):
            label = f"{row * MAP_UNIT:>4} "
            lines.append(label + " ".join(grid[row]))
        lines.append("")
        lines.append("Legend: R=rider S=restaurant C=customer "
                     "#=street .=empty *=route")
        lines.append("Each character = 50 units; top-left is (0, 0).")
        return "\n".join(lines)

"""Command parsing, shortcut handling and input validation."""

import logging
from typing import List, Sequence, Tuple

logger = logging.getLogger(__name__)

COORD_MIN = 0
COORD_MAX = 1000

COMMANDS = ("start", "status", "orders", "riders", "map", "tick",
            "save", "help", "quit", "exit")

ALIASES = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "sv": "save",
    "h": "help",
    "?": "help",
    "q": "quit",
    "x": "exit",
}

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy] [s]  Create an order: restaurant (rx, ry) and customer
                           (cx, cy), coordinates 0-1000. Prompts if omitted.
  status [st]              Show system funds, riders and their orders.
  orders [o]               Show all orders with time, distance and rider.
  riders [r]               Show every rider status change log.
  map [m]                  Show the 16x16 ASCII map of riders and orders.
  tick [n] [t]             Advance the simulation by n ticks (default 1).
  save [path] [sv]         Save state to a JSON file (default dispatch_state.json).
  help [h, ?]              Show this help.
  quit [q] / exit [x]      Leave the program.
"""


class InputError(ValueError):
    """Raised when user input is malformed or out of range."""


class InputManager:
    """Parses user commands and validates coordinate input."""

    def __init__(self) -> None:
        """Create the input manager with an empty command history."""
        self.history: List[str] = []

    def parse_command(self, line: str) -> Tuple[str, List[str]]:
        """Split a raw input line into a canonical command and arguments.

        Args:
            line: Raw line typed by the user.

        Returns:
            Tuple ``(command, args)``. The command is an empty string for
            blank input and is resolved through shortcut aliases.
        """
        parts = line.strip().split()
        if not parts:
            return "", []
        cmd = parts[0].lower()
        cmd = ALIASES.get(cmd, cmd)
        return cmd, parts[1:]

    def validate_coordinates(self, tokens: Sequence[str]) -> List[int]:
        """Validate four integer coordinates in the range 0-1000.

        Args:
            tokens: Raw text tokens: restaurant x, y, customer x, y.

        Returns:
            List of four integers.

        Raises:
            InputError: If the count, type or range is invalid.
        """
        if len(tokens) != 4:
            raise InputError(
                "expected 4 integers: restaurant_x restaurant_y "
                "customer_x customer_y")
        values: List[int] = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise InputError(f"'{token}' is not an integer") from None
            if not COORD_MIN <= value <= COORD_MAX:
                raise InputError(
                    f"coordinate {value} out of range "
                    f"[{COORD_MIN}, {COORD_MAX}]")
            values.append(value)
        return values

    def parse_tick_count(self, args: Sequence[str]) -> int:
        """Parse the optional tick count argument for the tick command.

        Args:
            args: Command arguments.

        Returns:
            Number of ticks to run (1-10000).

        Raises:
            InputError: If the value is not an integer or out of range.
        """
        if not args:
            return 1
        try:
            steps = int(args[0])
        except ValueError:
            raise InputError(f"'{args[0]}' is not an integer") from None
        if not 1 <= steps <= 10000:
            raise InputError("tick count must be between 1 and 10000")
        return steps

    def record_history(self, line: str) -> None:
        """Remember a non-empty command line for history recall.

        Args:
            line: The command line to record.
        """
        if line.strip():
            self.history.append(line.strip())

    def setup_readline(self) -> None:
        """Enable arrow-key line editing and history when readline exists."""
        try:
            import readline  # noqa: F401  (side effect: enables editing)
        except ImportError:
            logger.debug("readline not available; history disabled")

"""Command-line entry point for the Food Delivery Order Dispatch simulator.

Usage:
    python src/main.py [config.json]
"""

import logging
import math
import os
import sys
from typing import List, Optional

import globalvar
from display import DisplayManager
from input_manager import HELP_TEXT, InputError, InputManager
from path_planner import PathPlanner
from system import SystemManager

DEFAULT_STATE_FILE = "dispatch_state.json"
logger = logging.getLogger("dispatch")


class Application:
    """Interactive command loop tying all modules together."""

    def __init__(self) -> None:
        """Create the system, display and input managers."""
        self.system = SystemManager()
        self.display = DisplayManager()
        self.inp = InputManager()
        self.inp.setup_readline()

    # ----- main loop --------------------------------------------------
    def run(self) -> None:
        """Run the interactive loop until quit/exit or end of input."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.running:
            try:
                line = input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            if not line.strip():
                continue
            self.inp.record_history(line)
            cmd, args = self.inp.parse_command(line)
            try:
                self.handle(cmd, args)
            except InputError as exc:
                print(f"Input error: {exc}")
            except Exception:  # error recovery: keep the session alive
                logger.exception("Unexpected error handling '%s'", cmd)
                print("An unexpected error occurred; system state preserved.")

    def handle(self, cmd: str, args: List[str]) -> None:
        """Dispatch a parsed command to its handler.

        Args:
            cmd: Canonical command name.
            args: Command arguments.
        """
        handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
            "": lambda _args: None,
        }
        handler = handlers.get(cmd)
        if handler is None:
            raise InputError(f"unknown command '{cmd}'. Type 'help'.")
        handler(args)

    # ----- commands ---------------------------------------------------
    def cmd_start(self, args: List[str]) -> None:
        """Create an order from four coordinates and assign a rider.

        Args:
            args: Optional four coordinates; prompts if omitted.
        """
        if not args:
            print("Enter: restaurant_x restaurant_y customer_x customer_y")
            tokens = input("start> ").split()
        else:
            tokens = args
        rx, ry, cx, cy = self.inp.validate_coordinates(tokens)

        result = self.system.submit_order((rx, ry), (cx, cy))
        order = result["order"]
        rider = result["rider"]

        print(f"Order #{order.id} created.")
        print(f"  Restaurant: input ({rx}, {ry}) -> grid {order.res_pos}")
        print(f"  Customer:   input ({cx}, {cy}) -> grid {order.cus_pos}")
        print(f"  New rider added: {'Yes' if result['new_rider'] else 'No'}")
        if rider is None:
            print("  No rider available (insufficient funds); order is pending.")
            return

        route = PathPlanner.plan_route(
            [rider.position, order.res_pos, order.cus_pos])
        length = PathPlanner.route_length(route)
        eta = math.ceil(length / rider.speed) if rider.speed else 0
        print(f"  Assigned rider: #{rider.rider_id} (state: {rider.state_name})")
        print(f"  Speed: {rider.speed} units/tick")
        print(f"  Estimated time: {eta} tick(s) for {length} units "
              f"(excludes previously queued orders)")
        print(self.display.render_map(
            riders=[rider], orders=[order], route=route))

    def cmd_status(self, _args: List[str]) -> None:
        """Print system funds, riders and their order lists."""
        print(self.display.render_status(
            funds=self.system.funds,
            riders=self.system.riders,
            tick=self.system.tick_count,
            pending=self.system.pending_count(),
        ))

    def cmd_orders(self, _args: List[str]) -> None:
        """Print details of all orders."""
        print(self.display.render_orders(self.system.order_manager.all_orders()))

    def cmd_riders(self, _args: List[str]) -> None:
        """Print the status-change log of every rider."""
        print(self.display.render_rider_logs(self.system.riders))

    def cmd_map(self, _args: List[str]) -> None:
        """Print the ASCII map of riders and orders."""
        print(self.display.render_map(
            riders=self.system.riders,
            orders=self.system.order_manager.all_orders()))

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation by one or more ticks.

        Args:
            args: Optional number of ticks.
        """
        steps = self.inp.parse_tick_count(args)
        events = 0
        for _ in range(steps):
            events += self.system.tick()
        print(f"Advanced {steps} tick(s). Current tick: "
              f"{self.system.tick_count}. Rider events: {events}.")

    def cmd_save(self, args: List[str]) -> None:
        """Save the current state to a JSON file.

        Args:
            args: Optional output path.
        """
        path = args[0] if args else DEFAULT_STATE_FILE
        self.system.save_state(path)
        print(f"State saved to {path}.")

    def cmd_help(self, _args: List[str]) -> None:
        """Print the command help text."""
        print(HELP_TEXT)

    def cmd_quit(self, _args: List[str]) -> None:
        """Stop the main loop."""
        self.system.stop()
        print("Goodbye.")


def configure_logging() -> None:
    """Configure logging level from the DISPATCH_LOG environment variable."""
    level_name = os.environ.get("DISPATCH_LOG", "WARNING").upper()
    level = getattr(logging, level_name, logging.WARNING)
    logging.basicConfig(
        level=level,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        stream=sys.stderr,
    )


def main(argv: Optional[List[str]] = None) -> int:
    """Program entry point.

    Args:
        argv: Command-line arguments (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code.
    """
    configure_logging()
    args = sys.argv[1:] if argv is None else argv
    app = Application()
    if args:
        try:
            app.system.load_config(args[0])
        except (OSError, ValueError) as exc:
            print(f"Could not load configuration: {exc}. Using defaults.")
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

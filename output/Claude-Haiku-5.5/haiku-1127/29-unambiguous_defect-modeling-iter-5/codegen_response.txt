"""Process-wide key/value store shared by the simulator modules.

The store holds shared runtime objects such as the rider list (``"riders"``),
the simulated clock (``"simulated_time"``) and the system manager (``"system"``).
"""

_global_dict = {}


def _init():
    """Reset the global store to its initial state.

    After initialization the store contains an empty rider list under the key
    ``"riders"`` and a simulated clock set to 0 under ``"simulated_time"``.
    """
    global _global_dict
    _global_dict = {}
    _global_dict["riders"] = []
    _global_dict["simulated_time"] = 0


def set_value(key, value):
    """Store ``value`` under ``key``.

    Args:
        key: Hashable key identifying the global value.
        value: Any object to store.
    """
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key``.

    Args:
        key: Key to look up.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` if the key is absent.
    """
    return _global_dict.get(key, default)

"""Grid-based path planning for the 8x8 delivery map.

The map is an 800x800 coordinate system divided into an 8x8 grid of
100x100 cells. Streets lie on the grid lines (any coordinate that is a
multiple of 100). Restaurants and customers are located on street
intersections. Routes are right-angle paths that always travel along streets.
"""

import math

GRID_SIZE = 8
CELL_SIZE = 100
MAP_SIZE = GRID_SIZE * CELL_SIZE


def manhattan(a, b):
    """Return the Manhattan distance between two ``(x, y)`` points.

    Args:
        a: First point as a sequence ``(x, y)``.
        b: Second point as a sequence ``(x, y)``.

    Returns:
        Integer Manhattan distance.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class Route:
    """A planned right-angle route.

    Attributes:
        waypoints (list[tuple[int, int]]): Ordered points, including the start
            point and the destination.
        total_distance (int): Total Manhattan length of the route.
        estimated_time (int): Estimated number of ticks needed to travel it.
    """

    def __init__(self, waypoints, total_distance, estimated_time):
        """Create a route.

        Args:
            waypoints: Ordered list of ``(x, y)`` points.
            total_distance: Total route length in map units.
            estimated_time: Estimated travel time in ticks.
        """
        self.waypoints = list(waypoints)
        self.total_distance = total_distance
        self.estimated_time = estimated_time


class Grid:
    """Description of the grid used by the simulator.

    Attributes:
        width (int): Number of cells horizontally.
        height (int): Number of cells vertically.
        cell_size (int): Edge length of one cell in map units.
    """

    def __init__(self, width=GRID_SIZE, height=GRID_SIZE, cell_size=CELL_SIZE):
        """Create a grid description.

        Args:
            width: Number of cells horizontally.
            height: Number of cells vertically.
            cell_size: Edge length of each cell.
        """
        self.width = width
        self.height = height
        self.cell_size = cell_size

    @property
    def map_size(self):
        """Return the side length of the map in map units."""
        return self.width * self.cell_size

    def cell_type(self, x, y):
        """Classify a point of the map.

        Args:
            x: X coordinate (0..map_size).
            y: Y coordinate (0..map_size).

        Returns:
            ``"intersection"``, ``"street"`` or ``"block"``.

        Raises:
            ValueError: If the point lies outside the map.
        """
        if not (0 <= x <= self.map_size and 0 <= y <= self.map_size):
            raise ValueError(f"point ({x}, {y}) is outside the map")
        on_x = x % self.cell_size == 0
        on_y = y % self.cell_size == 0
        if on_x and on_y:
            return "intersection"
        if on_x or on_y:
            return "street"
        return "block"


class PathPlanner:
    """Plans right-angle, street-following routes on the grid.

    Attributes:
        grid_size (int): Number of cells per axis.
        cell_size (int): Edge length of one cell.
        map_size (int): Side length of the map.
        grid (Grid): Grid description.
    """

    def __init__(self, grid_size=GRID_SIZE, cell_size=CELL_SIZE):
        """Create a planner.

        Args:
            grid_size: Number of cells per axis.
            cell_size: Edge length of one cell in map units.
        """
        self.grid_size = grid_size
        self.cell_size = cell_size
        self.map_size = grid_size * cell_size
        self.grid = Grid(grid_size, grid_size, cell_size)

    def is_valid_point(self, x, y):
        """Return True if ``(x, y)`` lies within the map bounds."""
        return 0 <= x <= self.map_size and 0 <= y <= self.map_size

    def is_on_street(self, x, y):
        """Return True if ``(x, y)`` lies on a street (grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def is_intersection(self, x, y):
        """Return True if ``(x, y)`` is a street intersection."""
        return x % self.cell_size == 0 and y % self.cell_size == 0

    def snap_axis(self, value):
        """Snap one coordinate to the nearest grid line within the map.

        Args:
            value: Raw coordinate (any number).

        Returns:
            Integer coordinate that is a multiple of the cell size.
        """
        value = min(max(int(value), 0), self.map_size)
        snapped = int(round(value / self.cell_size)) * self.cell_size
        return min(max(snapped, 0), self.map_size)

    def snap(self, x, y):
        """Transform raw coordinates into a valid street intersection.

        Args:
            x: Raw X coordinate.
            y: Raw Y coordinate.

        Returns:
            Tuple ``(x, y)`` of snapped integer coordinates.
        """
        return self.snap_axis(x), self.snap_axis(y)

    def plan(self, start, end, speed=1):
        """Plan a right-angle route from ``start`` to ``end``.

        The route follows the street the start point lies on, then turns at
        most once to reach the destination intersection. Straight routes are
        produced when start and end share a street.

        Args:
            start: Current ``(x, y)`` position, which must lie on a street.
            end: Destination ``(x, y)`` intersection.
            speed: Units travelled per tick, used for the time estimate.

        Returns:
            A :class:`Route` describing the path.

        Raises:
            ValueError: If the points are outside the map, the destination is
                not an intersection, or the start is not on a street.
        """
        sx, sy = start
        ex, ey = end
        if not self.is_valid_point(sx, sy) or not self.is_valid_point(ex, ey):
            raise ValueError(f"route points outside map: {start} -> {end}")
        if not self.is_intersection(ex, ey):
            raise ValueError("destination must be a street intersection")
        if (sx, sy) == (ex, ey):
            return Route([(sx, sy)], 0, 0)

        c = self.cell_size
        if sy % c == 0:
            corner = (ex, sy)  # travel along the current horizontal street first
        elif sx % c == 0:
            corner = (sx, ey)  # travel along the current vertical street first
        else:
            raise ValueError(f"start point {start} is not on a street")

        waypoints = [(sx, sy)]
        if corner != waypoints[-1] and corner != (ex, ey):
            waypoints.append(corner)
        waypoints.append((ex, ey))

        total = sum(manhattan(waypoints[i], waypoints[i + 1])
                    for i in range(len(waypoints) - 1))
        speed = max(1, int(speed))
        estimated = math.ceil(total / speed)
        return Route(waypoints, total, estimated)

"""Rider model: state transitions, movement along streets and event logging."""

import globalvar
from path_planner import PathPlanner, GRID_SIZE

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {
    IDLE: "idle",
    PICKING_UP: "picking up",
    DELIVERING: "delivering",
}

DEFAULT_SPEED = 60

_PLANNER = PathPlanner()


def spawn_position(rider_id):
    """Return a deterministic starting street intersection for a rider.

    Args:
        rider_id: Unique rider identifier.

    Returns:
        Tuple ``(x, y)`` of an intersection on the map.
    """
    nodes_per_axis = GRID_SIZE + 1
    index = (rider_id * 7) % (nodes_per_axis * nodes_per_axis)
    return (index % nodes_per_axis) * 100, (index // nodes_per_axis) * 100


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id (int): Unique identifier.
        x (int): Current X coordinate.
        y (int): Current Y coordinate.
        speed (int): Distance units travelled per tick.
        state (int): 0 = idle, 1 = picking up, 2 = delivering.
        orders (list): Assigned orders; the first entry is the active order.
        special_case (bool): True while the rider is between intersections.
        log (list[dict]): Status change records with keys ``event``,
            ``order_id``, ``rider_id``, ``time``, ``x`` and ``y``.
    """

    def __init__(self, rider_id, speed=DEFAULT_SPEED):
        """Create a rider at a spawn point on a street.

        Args:
            rider_id: Unique rider identifier.
            speed: Movement speed in units per tick (positive integer).

        Raises:
            ValueError: If the speed is not a positive integer.
        """
        if isinstance(speed, bool) or not isinstance(speed, int) or speed <= 0:
            raise ValueError("rider speed must be a positive integer")
        self.rider_id = rider_id
        self.x, self.y = spawn_position(rider_id)
        self.speed = speed
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._waypoints = []
        self._target = None
        self._log("spawn", None)

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    @property
    def position(self):
        """Return the current ``(x, y)`` position."""
        return self.x, self.y

    @property
    def state_name(self):
        """Return a human-readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    @staticmethod
    def _now():
        """Return the current simulated time from the global store."""
        return globalvar.get_value("simulated_time", 0)

    def _log(self, event, order):
        """Append a status change record to :attr:`log`."""
        self.log.append({
            "event": event,
            "order_id": order.id if order is not None else None,
            "rider_id": self.rider_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
        })

    def _set_target(self, target):
        """Plan a route from the current position to ``target``."""
        route = _PLANNER.plan((self.x, self.y), target, self.speed)
        self._waypoints = route.waypoints[1:]

    def reset_navigation(self):
        """Clear any planned route, used for error recovery."""
        self._waypoints = []
        self._target = None

    def summary(self):
        """Return a dictionary describing the rider's current status."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "state_name": self.state_name,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
        }

    # ------------------------------------------------------------------
    # Movement and state transitions
    # ------------------------------------------------------------------
    def _advance(self):
        """Move up to ``speed`` units along the planned waypoints."""
        budget = self.speed
        while budget > 0 and self._waypoints:
            tx, ty = self._waypoints[0]
            dx = tx - self.x
            dy = ty - self.y
            distance = abs(dx) + abs(dy)
            if distance <= budget:
                self.x, self.y = tx, ty
                budget -= distance
                self._waypoints.pop(0)
            else:
                if dx != 0:
                    self.x += budget if dx > 0 else -budget
                else:
                    self.y += budget if dy > 0 else -budget
                budget = 0

    def _begin_pickup(self):
        """Transition from idle (or finished order) to picking up."""
        order = self.orders[0]
        self.state = PICKING_UP
        order.status = "picking_up"
        self._log("accepted_order", order)
        self._set_target(order.restaurant_pos)

    def _handle_arrival(self):
        """Process reaching the current target and return completed orders."""
        if not self.orders:
            self.state = IDLE
            self.reset_navigation()
            return []

        order = self.orders[0]
        if self.state == PICKING_UP:
            self._log("arrived_at_merchant", order)
            self.state = DELIVERING
            order.status = "delivering"
            self._set_target(order.customer_pos)
            return []

        # DELIVERING: the customer has been reached.
        order.complete(self._now())
        self._log("delivered_to_customer", order)
        self.orders.pop(0)
        self.reset_navigation()
        if self.orders:
            self._begin_pickup()
        else:
            self.state = IDLE
        return [order]

    def move(self):
        """Advance the rider by one tick.

        Idle riders with assigned orders start picking up. Riders in motion
        advance along their route; reaching the restaurant switches them to
        delivering, and reaching the customer completes the order.

        Returns:
            List of orders completed during this tick.
        """
        completed = []
        if self.state == IDLE:
            if not self.orders:
                return completed
            self._begin_pickup()

        if self.state not in (PICKING_UP, DELIVERING):
            return completed

        self._advance()
        if not self._waypoints:
            completed = self._handle_arrival()
        self.special_case = not _PLANNER.is_intersection(self.x, self.y)
        return completed

import itertools

from path_planner import PathPlanner, manhattan, MAP_SIZE

STATUS_CREATED = "created"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"

_PLANNER = PathPlanner()
_order_ids = itertools.count(1)


def _parse_point(point, label):
    """Validate a raw ``[x, y]`` point supplied by the user.

    Args:
        point: Sequence of two integers.
        label: Name used in error messages.

    Returns:
        Tuple ``(x, y)`` of integers.

    Raises:
        ValueError: If the point is malformed or out of range.
    """
    try:
        x, y = point
    except (TypeError, ValueError):
        raise ValueError(f"{label} location must contain exactly two coordinates")
    for value in (x, y):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} coordinates must be integers")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(
                f"{label} coordinates must be between 0 and {MAP_SIZE}, got {value}")
    return x, y


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id (int): Unique order identifier.
        restaurant_x, restaurant_y (int): Snapped restaurant intersection.
        customer_x, customer_y (int): Snapped customer intersection.
        raw_restaurant (tuple): Restaurant point as entered by the user.
        raw_customer (tuple): Customer point as entered by the user.
        start_time (int): Simulated time when the order was created.
        completion_time (int | None): Simulated time of delivery.
        distance (int): Manhattan distance from restaurant to customer.
        time_taken (int | None): Ticks from creation to delivery.
        rider_id (int | None): Assigned rider identifier.
        status (str): Lifecycle status, ``"completed"`` after delivery.
    """

    def __init__(self, res_pos, cus_pos, start_time=0):
        """Create an order and normalize its coordinates.

        Args:
            res_pos: Restaurant point ``[x, y]`` with values in 0..800.
            cus_pos: Customer point ``[x, y]`` with values in 0..800.
            start_time: Simulated time at creation.

        Raises:
            ValueError: If either point is malformed or out of range.
        """
        raw_res = _parse_point(res_pos, "restaurant")
        raw_cus = _parse_point(cus_pos, "customer")
        self.id = next(_order_ids)
        self.raw_restaurant = raw_res
        self.raw_customer = raw_cus
        self.restaurant_x, self.restaurant_y = _PLANNER.snap(*raw_res)
        self.customer_x, self.customer_y = _PLANNER.snap(*raw_cus)
        self.start_time = start_time
        self.completion_time = None
        self.distance = manhattan(self.restaurant_pos, self.customer_pos)
        self.time_taken = None
        self.rider_id = None
        self.status = STATUS_CREATED

    @property
    def order_id(self):
        """Alias for :attr:`id`."""
        return self.id

    @property
    def restaurant_pos(self):
        """Return the snapped restaurant position as ``(x, y)``."""
        return self.restaurant_x, self.restaurant_y

    @property
    def customer_pos(self):
        """Return the snapped customer position as ``(x, y)``."""
        return self.customer_x, self.customer_y

    @property
    def efficiency(self):
        """Return delivery distance per tick, or 0.0 if not yet completed."""
        if not self.time_taken:
            return 0.0
        return self.distance / self.time_taken

    def assign_to(self, rider_id):
        """Record the rider this order has been assigned to.

        Args:
            rider_id: Identifier of the assigned rider.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def complete(self, time):
        """Mark the order as delivered and record timing data.

        Args:
            time: Simulated time of delivery.
        """
        self.completion_time = time
        self.time_taken = time - self.start_time
        self.status = STATUS_COMPLETED

    def to_dict(self):
        """Return a JSON-serializable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "rider_id": self.rider_id,
            "status": self.status,
        }

"""Order management: creation, rider assignment and completion tracking."""

import globalvar
from order_model import Order, STATUS_COMPLETED
from path_planner import manhattan
from rider import Rider  # noqa: F401  (imported for type reference)

MAX_ORDERS_PER_RIDER = 3
LOAD_PENALTY = 150


def find_best_rider_for_order(order):
    """Select the most suitable rider for an order.

    The score of a rider is the Manhattan distance from its current position
    to the restaurant plus a penalty per already-assigned order, which balances
    load between riders. The lowest score wins; ties go to the lower rider ID.

    Args:
        order: The :class:`Order` to assign.

    Returns:
        The chosen :class:`Rider`, or ``None`` if no rider has spare capacity.
    """
    riders = globalvar.get_value("riders") or []
    best = None
    best_score = None
    for rider in riders:
        if len(rider.orders) >= MAX_ORDERS_PER_RIDER:
            continue
        distance = manhattan(rider.position, order.restaurant_pos)
        score = distance + LOAD_PENALTY * len(rider.orders)
        if (best is None or score < best_score
                or (score == best_score and rider.rider_id < best.rider_id)):
            best = rider
            best_score = score
    return best


class OrderManager:
    """Creates, assigns and tracks orders.

    Attributes:
        order_counter (int): Number of orders created so far.
        orders (list[Order]): All orders in creation order.
    """

    def __init__(self):
        """Create an empty order manager."""
        self.order_counter = 0
        self.orders = []

    def create_order(self, restaurant_pos, customer_pos, start_time=0):
        """Create and register a new order.

        Args:
            restaurant_pos: Restaurant point ``[x, y]``.
            customer_pos: Customer point ``[x, y]``.
            start_time: Simulated creation time.

        Returns:
            The newly created :class:`Order`.
        """
        order = Order(restaurant_pos, customer_pos, start_time)
        self.orders.append(order)
        self.order_counter += 1
        return order

    @staticmethod
    def assign_order(order, rider):
        """Assign an order to a rider by appending it to ``rider.orders``.

        Args:
            order: The order to assign.
            rider: The receiving rider.
        """
        order.assign_to(rider.rider_id)
        rider.orders.append(order)

    def assign_best(self, order):
        """Find the best rider for ``order`` and assign it.

        Args:
            order: The order to assign.

        Returns:
            The rider that received the order, or ``None``.
        """
        rider = find_best_rider_for_order(order)
        if rider is not None:
            self.assign_order(order, rider)
        return rider

    @staticmethod
    def complete_order(order, time):
        """Mark an order as completed at ``time``.

        Args:
            order: The completed order.
            time: Simulated completion time.
        """
        order.complete(time)

    def completed_orders(self):
        """Return the list of completed orders."""
        return [o for o in self.orders if o.status == STATUS_COMPLETED]

    def find_order(self, order_id):
        """Return the order with ``order_id`` or ``None``."""
        for order in self.orders:
            if order.id == order_id:
                return order
        return None

"""System-level coordination: funds, rider recruitment, ticks and persistence."""

import json
import logging

import globalvar
from order import OrderManager, find_best_rider_for_order
from order_model import STATUS_COMPLETED
from path_planner import PathPlanner, MAP_SIZE, GRID_SIZE
from rider import Rider, IDLE, PICKING_UP, DELIVERING

logger = logging.getLogger(__name__)

DEFAULT_FUNDS = 1000
DEFAULT_RECRUITMENT_COST = 200
DEFAULT_MAX_RIDERS = 20
DELIVERY_FEE = 30
OPERATING_COST_PER_BUSY_RIDER = 1


class SystemManager:
    """Core manager for funds, riders, orders and the simulated clock.

    Attributes:
        funds (int): Current system balance (may be negative).
        rider_id_counter (int): Last assigned rider ID.
        is_running (bool): Controls the main loop.
        simulated_time (int): Number of elapsed ticks.
        rider_recruitment_cost (int): Cost deducted per recruited rider.
        max_riders (int): Maximum number of riders that may be recruited.
        planner (PathPlanner): Shared path planner.
        order_manager (OrderManager): Order creation and tracking.
        riders (list[Rider]): The rider list, shared with the global store.
    """

    def __init__(self, funds=DEFAULT_FUNDS,
                 recruitment_cost=DEFAULT_RECRUITMENT_COST,
                 max_riders=DEFAULT_MAX_RIDERS):
        """Create the system manager.

        Args:
            funds: Initial system funds.
            recruitment_cost: Cost of recruiting one rider.
            max_riders: Maximum number of riders allowed.
        """
        self.funds = int(funds)
        self.rider_id_counter = 0
        self.is_running = True
        self.simulated_time = 0
        self.rider_recruitment_cost = int(recruitment_cost)
        self.max_riders = int(max_riders)
        self.planner = PathPlanner()
        self.order_manager = OrderManager()
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        self.riders = riders
        globalvar.set_value("simulated_time", self.simulated_time)

    # ------------------------------------------------------------------
    # Rider management
    # ------------------------------------------------------------------
    def has_idle_rider(self):
        """Return True if at least one rider is idle with no orders."""
        return any(r.state == IDLE and not r.orders for r in self.riders)

    def can_recruit(self):
        """Return True if a new rider can be recruited now."""
        return (len(self.riders) < self.max_riders
                and self.funds >= self.rider_recruitment_cost)

    def recruit_rider(self):
        """Recruit a new rider if funds and capacity allow.

        Returns:
            The new :class:`Rider`, or ``None`` if recruitment is not possible.
        """
        if not self.can_recruit():
            return None
        self.rider_id_counter += 1
        self.funds -= self.rider_recruitment_cost
        rider = Rider(self.rider_id_counter)
        self.riders.append(rider)
        logger.info("Recruited rider %d", rider.rider_id)
        return rider

    def get_rider(self, rider_id):
        """Return the rider with ``rider_id`` or ``None``."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    # ------------------------------------------------------------------
    # Order handling
    # ------------------------------------------------------------------
    def submit_order(self, restaurant_pos, customer_pos):
        """Create an order and assign it to the best available rider.

        A new rider is recruited when no idle rider exists or no rider can
        accept the order and funds allow recruitment.

        Args:
            restaurant_pos: Restaurant point ``[x, y]`` (0..800).
            customer_pos: Customer point ``[x, y]`` (0..800).

        Returns:
            Dictionary with keys ``order``, ``assigned``, ``new_rider``,
            ``rider``, ``route`` (list of waypoints), ``speed``,
            ``estimated_time`` and ``reason``.
        """
        order = self.order_manager.create_order(
            restaurant_pos, customer_pos, self.simulated_time)

        new_rider = False
        rider = find_best_rider_for_order(order)
        if rider is None or not self.has_idle_rider():
            recruited = self.recruit_rider()
            if recruited is not None:
                new_rider = True
                rider = find_best_rider_for_order(order)

        if rider is None:
            return {
                "order": order,
                "assigned": False,
                "new_rider": False,
                "rider": None,
                "route": [],
                "speed": 0,
                "estimated_time": 0,
                "reason": "no rider available and recruitment is not possible",
            }

        to_restaurant = self.planner.plan(rider.position, order.restaurant_pos,
                                          rider.speed)
        to_customer = self.planner.plan(order.restaurant_pos, order.customer_pos,
                                        rider.speed)
        route = to_restaurant.waypoints + to_customer.waypoints[1:]
        total = to_restaurant.total_distance + to_customer.total_distance
        estimated = -(-total // rider.speed)

        self.order_manager.assign_order(order, rider)
        return {
            "order": order,
            "assigned": True,
            "new_rider": new_rider,
            "rider": rider,
            "route": route,
            "speed": rider.speed,
            "estimated_time": estimated,
            "reason": "",
        }

    def active_orders(self):
        """Return all orders that are not yet completed."""
        return [o for o in self.order_manager.orders
                if o.status != STATUS_COMPLETED]

    # ------------------------------------------------------------------
    # Simulation clock
    # ------------------------------------------------------------------
    def tick(self):
        """Advance the simulation by one unit of time.

        Every rider moves once. Completed deliveries pay a fee and busy riders
        incur an operating cost. Errors in a single rider are logged and the
        system state is repaired without stopping the simulation.

        Returns:
            Net income generated during this tick.
        """
        self.simulated_time += 1
        globalvar.set_value("simulated_time", self.simulated_time)
        income = 0
        cost = 0
        for rider in list(self.riders):
            busy = rider.state != IDLE
            try:
                completed = rider.move()
            except Exception:  # error recovery: keep the simulation running
                logger.exception("Rider %s failed during tick", rider.rider_id)
                self.verify_consistency()
                continue
            income += DELIVERY_FEE * len(completed)
            if busy:
                cost += OPERATING_COST_PER_BUSY_RIDER
        self.funds += income - cost
        return income - cost

    def run_ticks(self, count):
        """Advance the simulation by ``count`` ticks.

        Args:
            count: Number of ticks to simulate (non-negative).

        Returns:
            Net income accumulated over all ticks.
        """
        total = 0
        for _ in range(count):
            total += self.tick()
        return total

    def stop(self):
        """Stop the main simulation loop."""
        self.is_running = False
        globalvar.set_value("is_running", False)

    # ------------------------------------------------------------------
    # Consistency and persistence
    # ------------------------------------------------------------------
    def verify_consistency(self):
        """Check and repair the state of all riders.

        Returns:
            List of human-readable descriptions of repaired issues.
        """
        issues = []
        seen_ids = set()
        for rider in self.riders:
            if rider.rider_id in seen_ids:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen_ids.add(rider.rider_id)

            if not self.planner.is_valid_point(rider.x, rider.y):
                rider.x, rider.y = self.planner.snap(rider.x, rider.y)
                rider.reset_navigation()
                issues.append(f"rider {rider.rider_id} moved back into the map")
            elif not self.planner.is_on_street(rider.x, rider.y):
                rider.x, rider.y = self.planner.snap(rider.x, rider.y)
                rider.reset_navigation()
                issues.append(f"rider {rider.rider_id} snapped onto a street")

            finished = [o for o in rider.orders if o.status == STATUS_COMPLETED]
            if finished:
                rider.orders[:] = [o for o in rider.orders
                                   if o.status != STATUS_COMPLETED]
                issues.append(f"rider {rider.rider_id} removed completed orders")

            if rider.state in (PICKING_UP, DELIVERING) and not rider.orders:
                rider.state = IDLE
                rider.reset_navigation()
                issues.append(f"rider {rider.rider_id} reset to idle")
        for issue in issues:
            logger.warning("Consistency repair: %s", issue)
        return issues

    def snapshot(self):
        """Return a JSON-serializable snapshot of the whole system."""
        return {
            "funds": self.funds,
            "simulated_time": self.simulated_time,
            "rider_id_counter": self.rider_id_counter,
            "rider_recruitment_cost": self.rider_recruitment_cost,
            "max_riders": self.max_riders,
            "riders": [
                dict(r.summary(), log=r.log) for r in self.riders
            ],
            "orders": [o.to_dict() for o in self.order_manager.orders],
        }

    def save_snapshot(self, path):
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(self.snapshot(), handle, indent=2)


def load_config(path):
    """Load system parameters from a JSON configuration file.

    Args:
        path: Path to the JSON file.

    Returns:
        Dictionary of parameters, empty if the file is missing or invalid.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not load configuration %s: %s", path, exc)
        return {}
    if not isinstance(data, dict):
        logger.warning("Configuration %s must contain a JSON object", path)
        return {}
    return data


def init_system(config_path=None):
    """Initialize global state and create the system manager.

    Args:
        config_path: Optional JSON configuration file with keys
            ``initial_funds``, ``rider_recruitment_cost`` and ``max_riders``.

    Returns:
        The initialized :class:`SystemManager`.
    """
    globalvar._init()
    config = load_config(config_path) if config_path else {}
    globalvar.set_value("riders", [])
    manager = SystemManager(
        funds=config.get("initial_funds", DEFAULT_FUNDS),
        recruitment_cost=config.get("rider_recruitment_cost",
                                    DEFAULT_RECRUITMENT_COST),
        max_riders=config.get("max_riders", DEFAULT_MAX_RIDERS),
    )
    globalvar.set_value("system", manager)
    globalvar.set_value("is_running", True)
    return manager

from path_planner import MAP_SIZE, CELL_SIZE, GRID_SIZE
from order_model import STATUS_COMPLETED
from rider import STATE_NAMES


class DisplayManager:
    """Renders status text, tables and the ASCII map.

    Attributes:
        map_width (int): Number of cells horizontally.
        map_height (int): Number of cells vertically.
    """

    def __init__(self, map_width=GRID_SIZE, map_height=GRID_SIZE):
        """Create a display manager.

        Args:
            map_width: Number of cells horizontally.
            map_height: Number of cells vertically.
        """
        self.map_width = map_width
        self.map_height = map_height

    @staticmethod
    def _to_canvas(x, y):
        """Convert map coordinates to ``(row, col)`` canvas indices.

        Intersections land on even rows and columns; street segments between
        them land on odd indices; blocks land on odd row and odd column.
        """
        col = int(round(x * 2 / CELL_SIZE))
        row = int(round((MAP_SIZE - y) * 2 / CELL_SIZE))
        return row, col

    def render_riders(self, riders):
        """Return a table describing every rider."""
        if not riders:
            return "No riders recruited yet."
        lines = ["Riders:"]
        for rider in riders:
            orders = ", ".join(f"#{o.id}" for o in rider.orders) or "none"
            lines.append(
                f"  Rider {rider.rider_id:>3} | pos ({rider.x:>3}, {rider.y:>3}) | "
                f"state: {STATE_NAMES.get(rider.state, 'unknown'):<10} | "
                f"speed: {rider.speed} | orders: {len(rider.orders)} [{orders}]"
            )
        return "\n".join(lines)

    def render_status(self, system):
        """Return a snapshot of funds, riders and order lists."""
        active = system.active_orders()
        lines = [
            f"Simulated time : {system.simulated_time}",
            f"System funds   : {system.funds}",
            f"Riders         : {len(system.riders)} (max {system.max_riders})",
            f"Recruit cost   : {system.rider_recruitment_cost}",
            f"Active orders  : {len(active)}",
            self.render_riders(system.riders),
        ]
        return "\n".join(lines)

    def render_orders(self, orders):
        """Return a table with details for all orders."""
        if not orders:
            return "No orders yet."
        lines = ["Orders:"]
        for o in orders:
            rider = f"rider {o.rider_id}" if o.rider_id is not None else "unassigned"
            done = o.time_taken if o.time_taken is not None else "-"
            lines.append(
                f"  #{o.id:<4} | start t={o.start_time:<5} | "
                f"restaurant ({o.restaurant_x}, {o.restaurant_y}) | "
                f"customer ({o.customer_x}, {o.customer_y}) | "
                f"distance {o.distance:<4} | time taken {done} | "
                f"{rider} | {o.status}"
            )
        return "\n".join(lines)

    def render_map(self, riders, orders, route=None):
        """Render the ASCII map with streets, orders, riders and an optional route.

        Symbols: ``+`` intersection, ``-``/``|`` street, ``*`` route,
        ``R`` restaurant, ``C`` customer, ``O`` rider, blank space for blocks.

        Args:
            riders: Riders to draw.
            orders: Orders whose restaurant/customer locations are drawn.
            route: Optional list of ``(x, y)`` waypoints to highlight.

        Returns:
            Multi-line string with the map and a legend.
        """
        rows = 2 * self.map_height + 1
        cols = 2 * self.map_width + 1
        canvas = [[" "] * cols for _ in range(rows)]

        for gy in range(self.map_height + 1):
            for gx in range(self.map_width + 1):
                r, c = self._to_canvas(gx * CELL_SIZE, gy * CELL_SIZE)
                canvas[r][c] = "+"
                if gx < self.map_width:
                    canvas[r][c + 1] = "-"
                if gy < self.map_height:
                    canvas[r + 1][c] = "|"

        if route:
            for a, b in zip(route, route[1:]):
                r1, c1 = self._to_canvas(*a)
                r2, c2 = self._to_canvas(*b)
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        canvas[r][c] = "*"

        def put(x, y, symbol):
            r, c = self._to_canvas(x, y)
            if 0 <= r < rows and 0 <= c < cols:
                canvas[r][c] = symbol

        for order in orders:
            if order.status == STATUS_COMPLETED:
                continue
            put(order.restaurant_x, order.restaurant_y, "R")
            put(order.customer_x, order.customer_y, "C")
        for rider in riders:
            put(rider.x, rider.y, "O")

        lines = []
        for r, row in enumerate(canvas):
            if r % 2 == 0:
                y_value = MAP_SIZE - (r // 2) * CELL_SIZE
                label = f"{y_value:>4} "
            else:
                label = "     "
            lines.append(label + "".join(row).rstrip())
        lines.append("     x: 0 (left) .. 800 (right); y: 0 (bottom) .. 800 (top)")
        lines.append("Legend: + intersection  -/| street  * route  "
                     "R restaurant  C customer  O rider")
        return "\n".join(lines)

"""Command parsing, validation and the interactive command loop."""

import logging

try:  # Enables line editing and history keys on supported platforms.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    pass

from path_planner import MAP_SIZE

logger = logging.getLogger(__name__)

SHORTCUTS = {
    "n": "start",
    "s": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "h": "help",
    "q": "quit",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant x/y, customer x/y, 0-800).
                       Prompts for values if omitted.
  status               Show funds, riders and their order lists.
  orders               Show details for all orders.
  riders               Show the status-change log for all riders.
  map                  Show the ASCII map with riders and active orders.
  tick [n]             Advance the simulation by n ticks (default 1).
  save [path]          Save a JSON snapshot (default snapshot.json).
  history              Show command history.
  help                 Show this help.
  quit / exit          Leave the program.
Shortcuts: n=start s=status o=orders r=riders m=map t=tick h=help q=quit
History:   !! repeats the last command, !N repeats command number N."""

COORD_NAMES = ["restaurant x", "restaurant y", "customer x", "customer y"]


class InputManager:
    """Parses user commands and dispatches them to the system.

    Attributes:
        system (SystemManager): The simulation being controlled.
        display (DisplayManager): Renderer for output.
        command_history (list[str]): Previously executed command lines.
    """

    def __init__(self, system, display, input_func=None, output_func=None):
        """Create the input manager.

        Args:
            system: The :class:`SystemManager` instance.
            display: The :class:`DisplayManager` instance.
            input_func: Callable used to read a line (defaults to ``input``).
            output_func: Callable used to print text (defaults to ``print``).
        """
        self.system = system
        self.display = display
        self.command_history = []
        self._input = input_func or input
        self._output = output_func or print
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    # ------------------------------------------------------------------
    # Parsing and validation
    # ------------------------------------------------------------------
    @staticmethod
    def parse_coordinate_tokens(tokens):
        """Validate four coordinate tokens.

        Args:
            tokens: Sequence of four strings.

        Returns:
            List of four integers in the range 0..800.

        Raises:
            ValueError: If the count, type or range is invalid.
        """
        if len(tokens) != 4:
            raise ValueError("expected four integers: restaurant x y, customer x y")
        values = []
        for name, token in zip(COORD_NAMES, tokens):
            try:
                value = int(token)
            except ValueError:
                raise ValueError(f"{name} must be an integer, got '{token}'")
            if not 0 <= value <= MAP_SIZE:
                raise ValueError(
                    f"{name} must be between 0 and {MAP_SIZE}, got {value}")
            values.append(value)
        return values

    def _prompt_coordinates(self):
        """Interactively ask for the four coordinates.

        Returns:
            List of four integers, or ``None`` if the user cancels.
        """
        values = []
        for name in COORD_NAMES:
            while True:
                raw = self._input(f"{name} (0-{MAP_SIZE}, or 'cancel'): ").strip()
                if raw.lower() == "cancel":
                    return None
                try:
                    values.extend(self.parse_coordinate_tokens(
                        [raw] + ["0"] * (4 - len(values) - 1)
                        if False else [str(self._check_single(name, raw))]
                        + ["0"] * 3))
                    break
                except ValueError as exc:
                    self._output(f"Error: {exc}")
        return values

    @staticmethod
    def _check_single(name, raw):
        """Validate one coordinate string and return its integer value."""
        try:
            value = int(raw)
        except ValueError:
            raise ValueError(f"{name} must be an integer, got '{raw}'")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"{name} must be between 0 and {MAP_SIZE}, got {value}")
        return value

    def _resolve_history(self, line):
        """Expand ``!!`` or ``!N`` into a previous command line.

        Returns:
            The resolved command line, or ``None`` on error.
        """
        if line == "!!":
            if not self.command_history:
                self._output("Error: no previous command.")
                return None
            return self.command_history[-1]
        if line[1:].isdigit():
            index = int(line[1:])
            if 1 <= index <= len(self.command_history):
                return self.command_history[index - 1]
        self._output(f"Error: no history entry for '{line}'.")
        return None

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def process(self, line):
        """Execute one command line.

        Args:
            line: Raw user input.

        Returns:
            False if the program should exit, True otherwise.
        """
        line = line.strip()
        if not line:
            return True
        if line.startswith("!"):
            resolved = self._resolve_history(line)
            if resolved is None:
                return True
            line = resolved

        self.command_history.append(line)
        tokens = line.split()
        name = tokens[0].lower()
        command = SHORTCUTS.get(name, name)
        handler = self._handlers.get(command)
        if handler is None:
            self._output(f"Unknown command '{tokens[0]}'. Type 'help' for commands.")
            return True
        try:
            result = handler(tokens[1:])
            return result is not False
        except ValueError as exc:
            self._output(f"Error: {exc}")
        except Exception:  # recovery: keep the session alive
            logger.exception("Unexpected error handling '%s'", line)
            self._output("Unexpected internal error; system state was checked and repaired.")
            self.system.verify_consistency()
        return True

    def run(self):
        """Run the interactive loop until the user quits."""
        self._output("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.is_running:
            try:
                line = self._input("> ")
            except (EOFError, KeyboardInterrupt):
                self._output("")
                break
            if not self.process(line):
                break

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def cmd_start(self, args):
        """Create an order and display the assignment result."""
        if args:
            values = self.parse_coordinate_tokens(args)
        else:
            values = self._prompt_coordinates()
            if values is None:
                self._output("Order cancelled.")
                return
        restaurant = (values[0], values[1])
        customer = (values[2], values[3])
        result = self.system.submit_order(restaurant, customer)
        order = result["order"]
        rx, ry = order.restaurant_pos
        cx, cy = order.customer_pos
        self._output(f"Order #{order.id} created.")
        self._output(f"Restaurant grid point: ({rx}, {ry})")
        self._output(f"Customer grid point:   ({cx}, {cy})")
        if result["new_rider"]:
            rider = result["rider"]
            self._output(f"New rider R{rider.rider_id} was recruited "
                         f"(cost {self.system.rider_recruitment_cost}). "
                         f"Funds: {self.system.funds}")
        else:
            self._output("No new rider was recruited.")
        if not result["assigned"]:
            self._output(f"Order not assigned: {result['reason']}.")
            return
        rider = result["rider"]
        self._output(f"Assigned to rider R{rider.rider_id}: speed {result['speed']} "
                     f"units/tick, estimated time {result['estimated_time']} ticks.")
        self._output(self.display.render_map(
            self.system.riders, self.system.active_orders(), route=result["route"]))

    def cmd_status(self, args):
        """Show the system status snapshot."""
        self._output(self.display.render_status(self.system))

    def cmd_orders(self, args):
        """Show all order details."""
        self._output(self.display.render_orders(self.system.order_manager.orders))

    def cmd_riders(self, args):
        """Show every recorded rider status change in time order."""
        entries = []
        for rider in self.system.riders:
            entries.extend(rider.log)
        if not entries:
            self._output("No rider events recorded.")
            return
        entries.sort(key=lambda e: (e["time"], e["rider_id"]))
        lines = ["Rider event log:"]
        for e in entries:
            order = f"order #{e['order_id']}" if e["order_id"] is not None else "-"
            lines.append(
                f"  t={e['time']:<5} | Rider {e['rider_id']:>3} | "
                f"{e['event']:<20} | {order:<12} | pos ({e['x']}, {e['y']})")
        self._output("\n".join(lines))

    def cmd_map(self, args):
        """Show the ASCII map."""
        self._output(self.display.render_map(self.system.riders,
                                             self.system.active_orders()))

    def cmd_tick(self, args):
        """Advance the simulation clock."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise ValueError("tick count must be a positive integer")
            if count <= 0 or count > 10000:
                raise ValueError("tick count must be between 1 and 10000")
        net = self.system.run_ticks(count)
        completed = len(self.system.order_manager.completed_orders())
        self._output(f"Advanced {count} tick(s). Time: {self.system.simulated_time}, "
                     f"net income: {net}, funds: {self.system.funds}, "
                     f"completed orders: {completed}.")

    def cmd_save(self, args):
        """Save a JSON snapshot of the system."""
        path = args[0] if args else "snapshot.json"
        try:
            self.system.save_snapshot(path)
        except OSError as exc:
            raise ValueError(f"could not write '{path}': {exc}")
        self._output(f"Snapshot saved to {path}.")

    def cmd_history(self, args):
        """Print the command history."""
        if not self.command_history:
            self._output("No commands in history.")
            return
        for index, entry in enumerate(self.command_history, start=1):
            self._output(f"  {index:>3}  {entry}")

    def cmd_help(self, args):
        """Print the help text."""
        self._output(HELP_TEXT)

    def cmd_quit(self, args):
        """Stop the program."""
        self.system.stop()
        self._output("Goodbye.")
        return False

"""Entry point for the food delivery dispatch simulator.

Run with ``python src/main.py [--config config.json]``.
"""

import argparse
import logging
import os
import sys

_SRC_DIR = os.path.dirname(os.path.abspath(__file__))
if _SRC_DIR not in sys.path:
    sys.path.insert(0, _SRC_DIR)

from display import DisplayManager  # noqa: E402
from input_manager import InputManager  # noqa: E402
from system import init_system  # noqa: E402


def parse_args(argv=None):
    """Parse command-line arguments.

    Args:
        argv: Optional list of arguments (defaults to ``sys.argv[1:]``).

    Returns:
        Parsed :class:`argparse.Namespace`.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=None,
                        help="JSON file with initial_funds, rider_recruitment_cost "
                             "and max_riders")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    return parser.parse_args(argv)


def main(argv=None):
    """Run the interactive simulator.

    Args:
        argv: Optional command-line arguments.

    Returns:
        Process exit code.
    """
    args = parse_args(argv)
    logging.basicConfig(
        level=logging.DEBUG if args.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    system = init_system(args.config)
    manager = InputManager(system, DisplayManager())
    manager.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Data loading utilities for the Beijing Subway network dataset.

The dataset is stored in ``subway_data.json`` next to this module. The path is
resolved from the location of this file, so the program can be started from
any working directory.
"""

import json
import os
from typing import Any, Dict, List

# Absolute path of src/subway_data.json, valid regardless of the working directory.
DATA_FILE_PATH: str = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)

# Top-level keys that every subway dataset must provide.
REQUIRED_KEYS = ("stations", "connections")


def load_subway_data(path: str) -> dict:
    """Load the subway network dataset from a JSON file.

    Args:
        path: Path of the JSON file to read.

    Returns:
        The parsed JSON content as a dictionary with the keys
        ``stations`` and ``connections``.

    Raises:
        OSError: If the file cannot be opened.
        ValueError: If the JSON content does not follow the expected structure.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data: Any = json.load(handle)

    if not isinstance(data, dict):
        raise ValueError(f"Subway data in '{path}' must be a JSON object")

    for key in REQUIRED_KEYS:
        if key not in data:
            raise ValueError(f"Subway data in '{path}' is missing the key '{key}'")
        if not isinstance(data[key], list):
            raise ValueError(f"Key '{key}' in '{path}' must be a JSON array")

    return data


def save_subway_data(data: dict, path: str) -> None:
    """Persist the subway network dataset to a JSON file.

    Args:
        data: Dataset dictionary with ``stations`` and ``connections`` keys.
        path: Destination file path. Missing parent directories are created.
    """
    directory = os.path.dirname(os.path.abspath(path))
    os.makedirs(directory, exist_ok=True)
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")


def list_required_keys() -> List[str]:
    """Return the top-level keys required in the dataset file."""
    return list(REQUIRED_KEYS)


def describe_dataset(data: Dict[str, Any]) -> str:
    """Return a short human-readable summary of a loaded dataset."""
    return (
        f"{len(data.get('stations', []))} station records, "
        f"{len(data.get('connections', []))} connection records"
    )

"""Core domain models for the Beijing Subway network and route results.

This module contains the weighted undirected graph used by the path
calculation module, along with the lightweight data classes describing
queries and computed routes.
"""

import sys
from dataclasses import dataclass, field
from typing import Dict, List, Optional


class Edge:
    """A weighted, undirected connection between two adjacent stations.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance between the stations in meters.
        line: Name of the subway line operating this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source: str, dest: str, distance: float, line: str) -> None:
        if distance < 0:
            raise ValueError(f"Edge distance must be non-negative, got {distance}")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self) -> float:
        """Edge weight used by shortest path algorithms (equals the distance)."""
        return self.distance

    def other_endpoint(self, node: str) -> str:
        """Return the endpoint of this edge that is not ``node``.

        Raises:
            ValueError: If ``node`` is not an endpoint of this edge.
        """
        if node == self.source:
            return self.dest
        if node == self.dest:
            return self.source
        raise ValueError(f"Node '{node}' is not an endpoint of {self!r}")

    def __repr__(self) -> str:
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph of subway stations using adjacency lists.

    Station numbers are assigned by the caller (see ``SubwayNetwork``); the
    graph only manages vertices and edges. The ``station_lines`` dictionary
    maps official station names to the lines serving them.

    Attributes:
        station_lines: Mapping of station name to the list of line names.
    """

    def __init__(self) -> None:
        self._adjacency: Dict[str, List[Edge]] = {}
        self._edge_count: int = 0
        self.station_lines: Dict[str, List[str]] = {}

    # ------------------------------------------------------------------
    # Properties and statistics
    # ------------------------------------------------------------------
    @property
    def vertex_count(self) -> int:
        """Number of stations (vertices) in the graph."""
        return len(self._adjacency)

    @property
    def edge_count(self) -> int:
        """Number of undirected connections (edges) in the graph."""
        return self._edge_count

    @property
    def memory_usage_mb(self) -> float:
        """Approximate memory footprint of the graph data structure in MB."""
        total = sys.getsizeof(self._adjacency)
        for key, edges in self._adjacency.items():
            total += sys.getsizeof(key) + sys.getsizeof(edges)
        for edge in self.edges():
            total += sys.getsizeof(edge) + sys.getsizeof(edge.distance)
        for name, lines in self.station_lines.items():
            total += sys.getsizeof(name) + sys.getsizeof(lines)
        return total / (1024.0 * 1024.0)

    # ------------------------------------------------------------------
    # Vertex management
    # ------------------------------------------------------------------
    def add_vertex(self, number: str) -> None:
        """Add a station vertex identified by ``number`` if it is not present."""
        self._adjacency.setdefault(number, [])

    def has_vertex(self, number: str) -> bool:
        """Return True if the station number exists in the graph."""
        return number in self._adjacency

    def vertices(self) -> List[str]:
        """Return all station numbers in insertion order."""
        return list(self._adjacency.keys())

    # ------------------------------------------------------------------
    # Station line information
    # ------------------------------------------------------------------
    def set_station_lines(self, station_name: str, lines: List[str]) -> None:
        """Set the list of lines serving a station (keyed by station name)."""
        self.station_lines[station_name] = list(lines)

    def get_station_lines(self, station_name: str) -> List[str]:
        """Return a copy of the lines serving a station, or an empty list."""
        return list(self.station_lines.get(station_name, []))

    # ------------------------------------------------------------------
    # Edge management
    # ------------------------------------------------------------------
    def add_edge(self, edge: Edge) -> None:
        """Add an undirected edge, creating its endpoints when necessary."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adjacency[edge.source].append(edge)
        if edge.dest != edge.source:
            self._adjacency[edge.dest].append(edge)
        self._edge_count += 1

    def adjacent(self, number: str) -> List[Edge]:
        """Return the edges incident to a station (read-only use recommended)."""
        return self._adjacency.get(number, [])

    def edges(self) -> List[Edge]:
        """Return every unique edge in the graph exactly once."""
        seen = set()
        result: List[Edge] = []
        for edges in self._adjacency.values():
            for edge in edges:
                if id(edge) not in seen:
                    seen.add(id(edge))
                    result.append(edge)
        return result

    def find_edges(self, number_a: str, number_b: str) -> List[Edge]:
        """Return all edges connecting two stations (in either direction)."""
        pair = {number_a, number_b}
        return [
            edge
            for edge in self._adjacency.get(number_a, [])
            if {edge.source, edge.dest} == pair
        ]

    def lines_at(self, number: str) -> List[str]:
        """Return the sorted unique line names of edges incident to a station."""
        return sorted({edge.line for edge in self._adjacency.get(number, [])})

"""Shortest path calculation for the subway network.

Implements a binary-heap priority queue (``MinPQ``), path configuration
(``PathfindingConfig``) and Dijkstra's algorithm (``ShortestPath``).

The search runs over states ``(station number, current line)`` so that line
changes (transfers) can be counted, limited and penalised. This allows the
three query modes (shortest distance, least transfers, comprehensive), the
maximum transfer limit and the line preference to be supported by a single
Dijkstra implementation with time complexity O((V+E) log V).
"""

import heapq
import itertools
from typing import Dict, List, Optional, Tuple

from models import Edge, Graph

# Supported path query modes.
MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Cost parameters (expressed in meters for comparability with distances).
TRANSFER_PENALTY_METERS = 1500.0      # used by comprehensive mode
PREFERENCE_PENALTY_METERS = 800.0     # applied to edges on non-preferred lines
TRANSFER_TIE_BREAK_METERS = 0.001     # tiny cost so that unnecessary transfers are avoided
LEAST_TRANSFER_WEIGHT = 1e9           # dominates distance in least transfer mode

State = Tuple[str, str]


class MinPQ:
    """Min priority queue implemented on top of a binary heap.

    Items with equal keys are returned in insertion order.
    """

    def __init__(self) -> None:
        self._heap: List[Tuple[float, int, object]] = []
        self._counter = itertools.count()

    @property
    def size(self) -> int:
        """Number of items currently stored in the queue."""
        return len(self._heap)

    def __len__(self) -> int:
        return len(self._heap)

    def is_empty(self) -> bool:
        """Return True when the queue holds no items."""
        return not self._heap

    def insert(self, key: float, item: object) -> None:
        """Insert an item with the given priority key in O(log n)."""
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def min_key(self) -> float:
        """Return the smallest key without removing it.

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("min_key() called on an empty MinPQ")
        return self._heap[0][0]

    def del_min(self) -> Tuple[float, object]:
        """Remove and return ``(key, item)`` with the smallest key in O(log n).

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("del_min() called on an empty MinPQ")
        key, _, item = heapq.heappop(self._heap)
        return key, item


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of ``shortest_distance``, ``least_transfer``, ``comprehensive``.
        max_transfers: Maximum number of allowed transfers, or None for no limit.
        preferred_lines: Lines that are preferred; empty list means no preference.
    """

    def __init__(
        self,
        mode: str = MODE_SHORTEST_DISTANCE,
        max_transfers: Optional[int] = None,
        preferred_lines: Optional[List[str]] = None,
    ) -> None:
        if mode not in VALID_MODES:
            raise ValueError(f"Unsupported mode '{mode}'. Choose from {VALID_MODES}")
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines: List[str] = list(preferred_lines) if preferred_lines else []

    def is_preferred(self, line: str) -> bool:
        """Return True when the line is acceptable without penalty."""
        return not self.preferred_lines or line in self.preferred_lines


class ShortestPath:
    """Dijkstra-based shortest path computation on a subway graph.

    Args:
        graph: The subway network graph.
        source: Station number of the origin.
        config: Path configuration; defaults to ``PathfindingConfig()``.

    Attributes:
        dist_to: Mapping of station number to the physical distance (meters)
            of the best path found to that station.
    """

    def __init__(
        self,
        graph: Graph,
        source: str,
        config: Optional[PathfindingConfig] = None,
    ) -> None:
        if not graph.has_vertex(source):
            raise ValueError(f"Source station '{source}' is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to: Dict[str, float] = {}
        self._score: Dict[State, float] = {}
        self._distance: Dict[State, float] = {}
        self._transfers: Dict[State, int] = {}
        self._prev: Dict[State, Optional[Tuple[State, Optional[Edge]]]] = {}
        self._best_state: Dict[str, State] = {}
        self._run()

    # ------------------------------------------------------------------
    # Cost model
    # ------------------------------------------------------------------
    def _increment(self, distance_inc: float, transfer_inc: int, penalty: float) -> float:
        """Return the additive score increment of one search step."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return transfer_inc * LEAST_TRANSFER_WEIGHT + distance_inc + penalty
        if mode == MODE_COMPREHENSIVE:
            return distance_inc + transfer_inc * TRANSFER_PENALTY_METERS + penalty
        return distance_inc + transfer_inc * TRANSFER_TIE_BREAK_METERS + penalty

    def _relax(
        self,
        pq: MinPQ,
        from_state: State,
        to_state: State,
        edge: Optional[Edge],
        distance_inc: float,
        transfer_inc: int,
        penalty: float,
    ) -> None:
        """Relax a transition between two search states."""
        new_transfers = self._transfers[from_state] + transfer_inc
        max_transfers = self.config.max_transfers
        if max_transfers is not None and new_transfers > max_transfers:
            return
        new_score = self._score[from_state] + self._increment(distance_inc, transfer_inc, penalty)
        if to_state not in self._score or new_score < self._score[to_state]:
            self._score[to_state] = new_score
            self._distance[to_state] = self._distance[from_state] + distance_inc
            self._transfers[to_state] = new_transfers
            self._prev[to_state] = (from_state, edge)
            pq.insert(new_score, to_state)

    # ------------------------------------------------------------------
    # Dijkstra main loop
    # ------------------------------------------------------------------
    def _run(self) -> None:
        """Execute Dijkstra's algorithm over (station, line) states."""
        pq = MinPQ()
        for line in self.graph.lines_at(self.source):
            state = (self.source, line)
            self._score[state] = 0.0
            self._distance[state] = 0.0
            self._transfers[state] = 0
            self._prev[state] = None
            pq.insert(0.0, state)

        finalized = set()
        while not pq.is_empty():
            _, state = pq.del_min()
            if state in finalized:
                continue
            finalized.add(state)
            node, line = state

            # Transfer to another line served by the same interchange station.
            for other_line in self.graph.lines_at(node):
                if other_line != line:
                    self._relax(pq, state, (node, other_line), None, 0.0, 1, 0.0)

            # Ride along an edge operated by the current line.
            for edge in self.graph.adjacent(node):
                if edge.line != line:
                    continue
                next_node = edge.other_endpoint(node)
                penalty = 0.0 if self.config.is_preferred(line) else PREFERENCE_PENALTY_METERS
                self._relax(pq, state, (next_node, line), edge, edge.distance, 0, penalty)

        # Choose the best state for every reached station.
        for state, score in self._score.items():
            node = state[0]
            best = self._best_state.get(node)
            if best is None or score < self._score[best]:
                self._best_state[node] = state
        for node, state in self._best_state.items():
            self.dist_to[node] = self._distance[state]

    # ------------------------------------------------------------------
    # Public query API
    # ------------------------------------------------------------------
    def has_path(self, target: str) -> bool:
        """Return True if ``target`` is reachable under the configuration."""
        return target in self._best_state

    def path_with_edges(self, target: str) -> Tuple[List[str], List[Edge]]:
        """Reconstruct the path from the source to ``target``.

        Returns:
            A tuple ``(stations, edges)`` where ``stations`` is the ordered list
            of station numbers (transfers do not repeat a station) and ``edges``
            are the traversed edges in order. Both are empty when unreachable.
        """
        if target not in self._best_state:
            return [], []
        state: Optional[State] = self._best_state[target]
        nodes: List[str] = []
        edges: List[Edge] = []
        while state is not None:
            nodes.append(state[0])
            previous = self._prev[state]
            if previous is None:
                break
            prev_state, edge = previous
            if edge is not None:
                edges.append(edge)
            state = prev_state
        nodes.reverse()
        edges.reverse()

        stations: List[str] = []
        for node in nodes:
            if not stations or stations[-1] != node:
                stations.append(node)
        return stations, edges

    def get_transfer_count(self, target: str) -> int:
        """Return the number of line transfers on the best path to ``target``."""
        state = self._best_state.get(target)
        if state is None:
            return 0
        return self._transfers[state]

"""Fare and travel time calculation for Beijing Subway trips.

Fare structure (distance in kilometers, fractional distances are rounded up):
    0-6 km    : 3 yuan
    6-12 km   : 4 yuan
    12-22 km  : 5 yuan
    22-32 km  : 6 yuan
    beyond 32 km: +1 yuan for every additional 20 km (or part thereof)
"""

import math
from dataclasses import dataclass
from typing import Tuple


@dataclass(frozen=True)
class FareTier:
    """A distance band of the progressive fare table."""

    min_distance_km: int
    max_distance_km: int
    fare_yuan: int


# Progressive fare tiers, ordered by distance.
FARE_TIERS: Tuple[FareTier, ...] = (
    FareTier(0, 6, 3),
    FareTier(6, 12, 4),
    FareTier(12, 22, 5),
    FareTier(22, 32, 6),
)

# Extended fare settings for trips longer than the last progressive tier.
EXTENSION_THRESHOLD_KM = 32
EXTENSION_BLOCK_KM = 20
EXTENSION_INCREMENT_YUAN = 1

# Parameters for the estimated travel time.
AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


def ceil_distance_km(distance: float) -> int:
    """Convert a distance in meters to whole kilometers, rounding up.

    Raises:
        ValueError: If the distance is negative.
    """
    if distance < 0:
        raise ValueError(f"Distance must be non-negative, got {distance}")
    return int(math.ceil(distance / 1000.0))


def Sum_money(distance: float) -> int:
    """Calculate the subway fare in yuan for a trip of ``distance`` meters.

    Args:
        distance: Travelled distance in meters.

    Returns:
        The fare in yuan as an integer.

    Raises:
        ValueError: If the distance is negative.
    """
    km = ceil_distance_km(distance)
    for tier in FARE_TIERS:
        if km <= tier.max_distance_km:
            return tier.fare_yuan
    extra_km = km - EXTENSION_THRESHOLD_KM
    blocks = int(math.ceil(extra_km / EXTENSION_BLOCK_KM))
    return FARE_TIERS[-1].fare_yuan + blocks * EXTENSION_INCREMENT_YUAN


def calculate_estimated_time(distance: float, transfer_count: int) -> float:
    """Estimate the travel time in minutes.

    The estimate combines in-vehicle time at an average speed with a fixed
    penalty for every transfer.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers on the route.

    Returns:
        Estimated travel time in minutes.

    Raises:
        ValueError: If the distance or transfer count is negative.
    """
    if distance < 0:
        raise ValueError(f"Distance must be non-negative, got {distance}")
    if transfer_count < 0:
        raise ValueError(f"Transfer count must be non-negative, got {transfer_count}")
    travel_minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    return travel_minutes + transfer_count * TRANSFER_TIME_MINUTES

"""Subway network management: station registry, graph construction and data updates.

``SubwayNetwork`` owns the station-to-number mapping (``station_to_number``),
the weighted ``Graph`` and the operations used by the data management module
(adding lines, modifying stations, updating distances and persisting data).
"""

from typing import Dict, List, Optional

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
from models import Edge, Graph


class SubwayNetwork:
    """Beijing Subway network built from the ``subway_data.json`` dataset.

    Attributes:
        graph: The weighted undirected graph of stations.
        station_to_number: Mapping of official station name to station number.
    """

    def __init__(self, data: Optional[dict] = None) -> None:
        if data is None:
            data = load_subway_data(DATA_FILE_PATH)
        self.graph = Graph()
        self.station_to_number: Dict[str, str] = {}
        self._number_to_name: Dict[str, str] = {}
        self._station_order: List[str] = []
        self._counter = 0
        self._build(data)

    @classmethod
    def from_file(cls, path: str = DATA_FILE_PATH) -> "SubwayNetwork":
        """Create a network by loading the dataset from ``path``."""
        return cls(load_subway_data(path))

    # ------------------------------------------------------------------
    # Construction helpers
    # ------------------------------------------------------------------
    def _build(self, data: dict) -> None:
        """Populate the graph from the raw dataset."""
        for station in data.get("stations", []):
            name = str(station["name"]).strip()
            lines = [str(line).strip() for line in station.get("lines", [])]
            self._register_station(name)
            merged = self.graph.get_station_lines(name)
            for line in lines:
                if line not in merged:
                    merged.append(line)
            self.graph.set_station_lines(name, merged)

        for conn in data.get("connections", []):
            self._add_edge_by_name(
                str(conn["source"]).strip(),
                str(conn["dest"]).strip(),
                conn["distance"],
                str(conn["line"]).strip(),
            )

    def _new_number(self) -> str:
        """Generate the next unique station number."""
        self._counter += 1
        return f"S{self._counter:04d}"

    def _register_station(self, name: str) -> str:
        """Ensure a station exists and return its station number."""
        if name in self.station_to_number:
            return self.station_to_number[name]
        number = self._new_number()
        self.station_to_number[name] = number
        self._number_to_name[number] = name
        self._station_order.append(name)
        self.graph.add_vertex(number)
        self.graph.station_lines.setdefault(name, [])
        return number

    def _add_edge_by_name(self, source: str, dest: str, distance: float, line: str) -> Edge:
        """Create an edge between two stations identified by name."""
        if not isinstance(distance, (int, float)) or isinstance(distance, bool):
            raise ValueError(f"Distance between '{source}' and '{dest}' must be numeric")
        number_a = self._register_station(source)
        number_b = self._register_station(dest)
        edge = Edge(number_a, number_b, distance, line)
        self.graph.add_edge(edge)
        return edge

    # ------------------------------------------------------------------
    # Lookup API
    # ------------------------------------------------------------------
    def station_names(self) -> List[str]:
        """Return all official station names in registration order."""
        return list(self._station_order)

    def station_exists(self, name: str) -> bool:
        """Return True if the exact station name is registered."""
        return name in self.station_to_number

    def number_of(self, name: str) -> str:
        """Return the station number for a station name.

        Raises:
            KeyError: If the station does not exist.
        """
        if name not in self.station_to_number:
            raise KeyError(f"Unknown station '{name}'")
        return self.station_to_number[name]

    def name_of(self, number: str) -> str:
        """Return the station name for a station number.

        Raises:
            KeyError: If the station number does not exist.
        """
        if number not in self._number_to_name:
            raise KeyError(f"Unknown station number '{number}'")
        return self._number_to_name[number]

    def standardize_name(self, name: str) -> Optional[str]:
        """Map user input to the official station name.

        Matching ignores surrounding whitespace, inner spaces and letter case.

        Returns:
            The official station name, or None if no station matches.
        """
        if not name:
            return None
        cleaned = name.strip()
        if cleaned in self.station_to_number:
            return cleaned
        key = _normalize(cleaned)
        for official in self._station_order:
            if _normalize(official) == key:
                return official
        return None

    def lines_of_station(self, name: str) -> List[str]:
        """Return the lines serving a station (copy)."""
        return self.graph.get_station_lines(name)

    def line_names(self) -> List[str]:
        """Return all line names known to the network, sorted."""
        lines = set()
        for station_lines in self.graph.station_lines.values():
            lines.update(station_lines)
        for edge in self.graph.edges():
            lines.add(edge.line)
        return sorted(lines)

    def stations_on_line(self, line: str) -> List[str]:
        """Return the station names served by a line, in registration order."""
        return [name for name in self._station_order if line in self.lines_of_station(name)]

    def transfer_stations(self) -> List[str]:
        """Return the names of interchange stations (served by multiple lines)."""
        return [name for name in self._station_order if len(self.lines_of_station(name)) > 1]

    # ------------------------------------------------------------------
    # Data update API
    # ------------------------------------------------------------------
    def add_line(self, line_name: str, station_names: List[str], distances: List[float]) -> None:
        """Add a new line described by an ordered station sequence.

        Args:
            line_name: Name of the new line.
            station_names: Ordered station names along the new line.
            distances: Distance in meters between consecutive stations.

        Raises:
            ValueError: If the input is inconsistent or the line already exists.
        """
        line_name = line_name.strip()
        if not line_name:
            raise ValueError("Line name must not be empty")
        if line_name in self.line_names():
            raise ValueError(f"Line '{line_name}' already exists")
        names = [name.strip() for name in station_names]
        if len(names) < 2:
            raise ValueError("A line needs at least two stations")
        if len({_normalize(name) for name in names}) != len(names):
            raise ValueError("Station sequence of a line must not contain duplicates")
        if len(distances) != len(names) - 1:
            raise ValueError("Number of distances must be one less than number of stations")
        for distance in distances:
            if distance <= 0:
                raise ValueError("Distances must be positive")

        for name in names:
            if name in self.station_to_number:
                lines = self.lines_of_station(name)
                if line_name not in lines:
                    lines.append(line_name)
                self.graph.set_station_lines(name, lines)
            else:
                self._register_station(name)
                self.graph.set_station_lines(name, [line_name])

        for index, distance in enumerate(distances):
            self._add_edge_by_name(names[index], names[index + 1], distance, line_name)

    def modify_station(
        self,
        old_name: str,
        new_name: Optional[str] = None,
        lines: Optional[List[str]] = None,
    ) -> str:
        """Rename a station and/or replace the lines serving it.

        Returns:
            The current (possibly new) official name of the station.

        Raises:
            ValueError: If the station does not exist or the new name is taken.
        """
        old = old_name.strip()
        if old not in self.station_to_number:
            raise ValueError(f"Station '{old}' does not exist")
        current = old
        if new_name is not None and new_name.strip() and new_name.strip() != old:
            new = new_name.strip()
            if new in self.station_to_number:
                raise ValueError(f"Station '{new}' already exists")
            number = self.station_to_number.pop(old)
            self.station_to_number[new] = number
            self._number_to_name[number] = new
            index = self._station_order.index(old)
            self._station_order[index] = new
            self.graph.station_lines[new] = self.graph.station_lines.pop(old, [])
            current = new
        if lines is not None:
            self.graph.set_station_lines(current, [str(line).strip() for line in lines])
        return current

    def update_distance(
        self,
        name_a: str,
        name_b: str,
        distance: float,
        line: Optional[str] = None,
    ) -> int:
        """Update the distance of connection(s) between two adjacent stations.

        Returns:
            The number of connections updated.

        Raises:
            ValueError: If the distance is invalid or no matching connection exists.
        """
        if distance <= 0:
            raise ValueError("Distance must be positive")
        number_a = self.number_of(name_a.strip())
        number_b = self.number_of(name_b.strip())
        edges = self.graph.find_edges(number_a, number_b)
        if line is not None:
            edges = [edge for edge in edges if edge.line == line]
        if not edges:
            raise ValueError(
                f"No connection found between '{name_a}' and '{name_b}'"
            )
        for edge in edges:
            edge.distance = distance
        return len(edges)

    # ------------------------------------------------------------------
    # Persistence
    # ------------------------------------------------------------------
    def to_dict(self) -> dict:
        """Export the network in the ``subway_data.json`` structure."""
        stations = [
            {"name": name, "lines": self.lines_of_station(name)}
            for name in self._station_order
        ]
        connections = [
            {
                "source": self.name_of(edge.source),
                "dest": self.name_of(edge.dest),
                "distance": edge.distance,
                "line": edge.line,
            }
            for edge in self.graph.edges()
        ]
        return {"stations": stations, "connections": connections}

    def save(self, path: str = DATA_FILE_PATH) -> None:
        """Persist the current network to a JSON file."""
        save_subway_data(self.to_dict(), path)

    """Normalize a station name for tolerant comparison."""
    return "".join(name.split()).casefold()

"""Data integrity checks for the subway network.

Implements the ``--check_data`` validations: station coverage of the main
lines, connection integrity (connectivity, line consistency, duplicates) and
accuracy of distance data.
"""

from collections import deque
from typing import List, Set, Tuple

from network import SubwayNetwork

# Main lines whose representative presence must be verified.
REQUIRED_MAIN_LINES: Tuple[str, ...] = (
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
)

# Plausible range of a single inter-station distance in meters.
MIN_SEGMENT_METERS = 1
MAX_SEGMENT_METERS = 15000


class DataChecker:
    """Runs integrity checks against a ``SubwayNetwork``.

    Attributes:
        error_messages: Human-readable descriptions of every detected problem.
    """

    def __init__(self, network: SubwayNetwork) -> None:
        self.network = network
        self.error_messages: List[str] = []

    @property
    def passed(self) -> bool:
        """True when no integrity problems were found by the last run."""
        return not self.error_messages

    def run(self) -> bool:
        """Execute all checks and return True if the data passed."""
        self.error_messages = []
        self._check_station_coverage()
        self._check_connection_integrity()
        self._check_distance_accuracy()
        return self.passed

    def _error(self, message: str) -> None:
        self.error_messages.append(message)

    def _check_station_coverage(self) -> None:
        """Verify that main lines and their stations are represented."""
        if not self.network.station_names():
            self._error("No stations are defined in the network data")
            return
        lines: Set[str] = set(self.network.line_names())
        for line in REQUIRED_MAIN_LINES:
            if line not in lines:
                self._error(f"Main line '{line}' is not covered by the network data")
            elif len(self.network.stations_on_line(line)) < 2:
                self._error(f"Main line '{line}' has fewer than two stations")
        for name in self.network.station_names():
            if not self.network.lines_of_station(name):
                self._error(f"Station '{name}' does not serve any line")

    def _check_connection_integrity(self) -> None:
        """Verify station references, line consistency, duplicates and connectivity."""
        graph = self.network.graph
        seen_keys: Set[Tuple[str, str, str]] = set()
        for edge in graph.edges():
            src = self.network.name_of(edge.source)
            dst = self.network.name_of(edge.dest)
            if src == dst:
                self._error(f"Connection on line '{edge.line}' starts and ends at '{src}'")
            for name in (src, dst):
                if name not in graph.station_lines:
                    self._error(f"Connection references unknown station '{name}'")
                elif edge.line not in self.network.lines_of_station(name):
                    self._error(
                        f"Station '{name}' is connected on line '{edge.line}' "
                        f"but does not list that line"
                    )
            key = (min(src, dst), max(src, dst), edge.line)
            if key in seen_keys:
                self._error(
                    f"Duplicate connection between '{src}' and '{dst}' on line '{edge.line}'"
                )
            seen_keys.add(key)

        vertices = graph.vertices()
        if len(vertices) <= 1:
            return
        visited = {vertices[0]}
        queue = deque([vertices[0]])
        while queue:
            node = queue.popleft()
            for edge in graph.adjacent(node):
                neighbour = edge.other_endpoint(node)
                if neighbour not in visited:
                    visited.add(neighbour)
                    queue.append(neighbour)
        unreachable = [self.network.name_of(v) for v in vertices if v not in visited]
        if unreachable:
            preview = ", ".join(unreachable[:10])
            self._error(
                f"Network is not connected; {len(unreachable)} station(s) unreachable: {preview}"
            )

    def _check_distance_accuracy(self) -> None:
        """Verify that every connection distance is numeric and plausible."""
        for edge in self.network.graph.edges():
            src = self.network.name_of(edge.source)
            dst = self.network.name_of(edge.dest)
            distance = edge.distance
            if not isinstance(distance, (int, float)) or isinstance(distance, bool):
                self._error(f"Distance between '{src}' and '{dst}' is not numeric")
            elif distance < MIN_SEGMENT_METERS or distance > MAX_SEGMENT_METERS:
                self._error(
                    f"Distance between '{src}' and '{dst}' on line '{edge.line}' "
                    f"is {distance} m, outside the plausible range "
                    f"[{MIN_SEGMENT_METERS}, {MAX_SEGMENT_METERS}] m"
                )

"""Route planning service combining path search, segmentation and fare calculation.

``find_route`` turns a Dijkstra result into a ``Route`` object containing the
station sequence, line segments, transfer details, total distance, fare and
estimated travel time.
"""

from typing import List, Optional, Tuple

from fare_calculator import Sum_money, calculate_estimated_time
from models import Edge, Route, RouteSegment, TransferPoint
from network import SubwayNetwork
from pathfinding import PathfindingConfig, ShortestPath


class RouteNotFoundError(Exception):
    """Raised when no route satisfies the query and its constraints."""


def find_route(
    network: SubwayNetwork,
    start_name: str,
    end_name: str,
    config: Optional[PathfindingConfig] = None,
) -> Route:
    """Compute the optimal route between two official station names.

    Args:
        network: The loaded subway network.
        start_name: Official name of the origin station.
        end_name: Official name of the destination station.
        config: Path query configuration (mode, transfer limit, preference).

    Returns:
        A fully populated ``Route``.

    Raises:
        RouteNotFoundError: If no path exists or the transfer limit cannot be met.
        KeyError: If a station name is unknown.
    """
    config = config if config is not None else PathfindingConfig()
    source_number = network.number_of(start_name)
    target_number = network.number_of(end_name)

    search = ShortestPath(network.graph, source_number, config)
    if not search.has_path(target_number):
        raise RouteNotFoundError(f"No route found from '{start_name}' to '{end_name}'")

    node_numbers, edges = search.path_with_edges(target_number)
    names = [network.name_of(number) for number in node_numbers]
    segments, transfers = _build_segments(names, edges)

    transfer_count = len(transfers)
    if config.max_transfers is not None and transfer_count > config.max_transfers:
        raise RouteNotFoundError(
            f"No route found within {config.max_transfers} transfer(s); "
            f"the best route requires {transfer_count}"
        )

    total_distance = float(sum(edge.distance for edge in edges))
    lines_involved: List[str] = []
    for segment in segments:
        if segment.line_name not in lines_involved:
            lines_involved.append(segment.line_name)

    return Route(
        stations=names,
        segments=segments,
        transfers=transfers,
        total_distance_meters=total_distance,
        transfer_count=transfer_count,
        estimated_time_minutes=calculate_estimated_time(total_distance, transfer_count),
        fare_yuan=Sum_money(total_distance),
        lines_involved=lines_involved,
    )


def _build_segments(
    names: List[str], edges: List[Edge]
) -> Tuple[List[RouteSegment], List[TransferPoint]]:
    """Group consecutive edges on the same line into segments and derive transfers.

    The station list has exactly one more element than the edge list, because
    transfers do not repeat a station in the reconstructed path.

    Raises:
        RuntimeError: If the station and edge sequences are inconsistent.
    """
    segments: List[RouteSegment] = []
    transfers: List[TransferPoint] = []
    if not edges:
        return segments, transfers
    if len(names) != len(edges) + 1:
        raise RuntimeError("Inconsistent path reconstruction: station/edge count mismatch")

    start_index = 0
    for index, edge in enumerate(edges):
        is_last = index == len(edges) - 1
        if is_last or edges[index + 1].line != edge.line:
            segment_edges = edges[start_index : index + 1]
            segments.append(
                RouteSegment(
                    line_name=edge.line,
                    start_station=names[start_index],
                    end_station=names[index + 1],
                    distance_meters=float(sum(item.distance for item in segment_edges)),
                    hop_count=len(segment_edges),
                )
            )
            start_index = index + 1

    for position in range(len(segments) - 1):
        current = segments[position]
        following = segments[position + 1]
        transfers.append(
            TransferPoint(
                station_name=current.end_station,
                from_line=current.line_name,
                to_line=following.line_name,
            )
        )
    return segments, transfers

"""Path visualization and report export.

Renders a ``Route`` as console text or structured JSON and writes reports to
an optional output file.

import json
import os
from typing import Optional

from models import Query, Route

SUPPORTED_FORMATS = ("text", "json")
SEPARATOR = "-" * 60


def format_distance(meters: float) -> str:
    """Format a distance using meters for short and kilometers for long distances."""
    if meters < 1000:
        return f"{meters:.0f} m"
    return f"{meters / 1000.0:.2f} km"


def format_distance_both(meters: float) -> str:
    """Format a distance in both meters and kilometers, primary unit first."""
    if meters < 1000:
        return f"{meters:.0f} m ({meters / 1000.0:.2f} km)"
    return f"{meters / 1000.0:.2f} km ({meters:.0f} m)"


def build_summary(route: Route) -> dict:
    """Extract the key summary information of a route."""
    return {
        "total_distance_meters": round(route.total_distance_meters, 1),
        "total_distance_km": round(route.total_distance_km, 2),
        "fare_yuan": route.fare_yuan,
        "transfer_count": route.transfer_count,
        "estimated_time_minutes": round(route.estimated_time_minutes, 1),
    }


def format_text(route: Route, query: Query) -> str:
    """Render the route as human-readable console text."""
    lines = []
    lines.append("=" * 60)
    lines.append(f"Route: {query.start_station} -> {query.end_station}")
    lines.append(f"Query mode: {query.mode}")
    if query.max_transfers is not None:
        lines.append(f"Maximum transfers: {query.max_transfers}")
    if query.line_preference:
        lines.append(f"Line preference: {query.line_preference}")
    lines.append(SEPARATOR)

    lines.append(f"Station sequence ({len(route.stations)} stations):")
    for index, name in enumerate(route.stations, start=1):
        lines.append(f"  {index:>3}. {name}")
    lines.append(SEPARATOR)

    lines.append("Line segments:")
    for segment in route.segments:
        lines.append(
            f"  [{segment.line_name}] {segment.start_station} -> {segment.end_station} "
            f"({segment.hop_count} hop(s), {format_distance(segment.distance_meters)})"
        )

    lines.append("Transfer details:")
    if route.transfers:
        for transfer in route.transfers:
            lines.append(
                f"  Transfer at {transfer.station_name}: "
                f"{transfer.from_line} -> {transfer.to_line}"
            )
    else:
        lines.append("  No transfer required")
    lines.append(f"Lines involved: {', '.join(route.lines_involved)}")
    lines.append(f"Transfer count: {route.transfer_count}")
    lines.append(SEPARATOR)

    lines.append(f"Total distance: {format_distance_both(route.total_distance_meters)}")
    lines.append(f"Estimated time: {route.estimated_time_minutes:.1f} minutes")
    lines.append(f"Fare: {route.fare_yuan} yuan")
    lines.append("=" * 60)
    return "\n".join(lines)


def format_json(route: Route, query: Query) -> str:
    """Render the route as a JSON document for programmatic integration."""
    summary = build_summary(route)
    payload = {
        "query": {
            "start_station": query.start_station,
            "end_station": query.end_station,
            "mode": query.mode,
            "max_transfers": query.max_transfers,
            "line_preference": query.line_preference,
        },
        "path": list(route.stations),
        "distance": {
            "meters": summary["total_distance_meters"],
            "kilometers": summary["total_distance_km"],
        },
        "fare": {"yuan": route.fare_yuan},
        "estimated_time_minutes": summary["estimated_time_minutes"],
        "transfer_count": route.transfer_count,
        "transfers": [
            {
                "station": transfer.station_name,
                "from_line": transfer.from_line,
                "to_line": transfer.to_line,
            }
            for transfer in route.transfers
        ],
        "lines": list(route.lines_involved),
        "segments": [
            {
                "line": segment.line_name,
                "start_station": segment.start_station,
                "end_station": segment.end_station,
                "distance_meters": round(segment.distance_meters, 1),
                "hop_count": segment.hop_count,
            }
            for segment in route.segments
        ],
    }
    return json.dumps(payload, ensure_ascii=False, indent=2)


def render_report(route: Route, query: Query, format_name: str) -> str:
    """Render a route in the requested format ('text' or 'json')."""
    if format_name == "text":
        return format_text(route, query)
    if format_name == "json":
        return format_json(route, query)
    raise ValueError(f"Unsupported output format '{format_name}'")


class Report:
    """A rendered path report that can be exported to a file.

    Attributes:
        format: Output format, 'text' or 'json'.
        output_file: Optional destination file path.
        content: The rendered report text.
    """

    def __init__(self, format_name: str, output_file: Optional[str] = None, content: str = "") -> None:
        if format_name not in SUPPORTED_FORMATS:
            raise ValueError(f"Unsupported output format '{format_name}'")
        self.format = format_name
        self.output_file = output_file
        self.content = content

    @classmethod
    def from_route(
        cls,
        route: Route,
        query: Query,
        format_name: str,
        output_file: Optional[str] = None,
    ) -> "Report":
        """Render a route and wrap it in a ``Report``."""
        return cls(format_name, output_file, render_report(route, query, format_name))

    def write(self) -> Optional[str]:
        """Write the content to the output file, if one is configured.

        Returns:
            The path written to, or None when no output file is configured.

        Raises:
            OSError: If the file cannot be written.
        """
        if not self.output_file:
            return None
        directory = os.path.dirname(os.path.abspath(self.output_file))
        os.makedirs(directory, exist_ok=True)
        with open(self.output_file, "w", encoding="utf-8") as handle:
            handle.write(self.content)
            if not self.content.endswith("\n"):
                handle.write("\n")
        return self.output_file

"""Command-line interface for the Beijing Subway fare calculation system.

Supports continuous interactive queries in the form
"Starting Station; Ending Station", command-line configuration of output,
mode, transfer limits and line preference, and the ``--check_data`` mode.

import sys

INPUT_FORMAT_HINT = "Input format incorrect, please use 'Starting Station; Ending Station' format"
EXIT_COMMANDS = ("quit", "exit", "q")


class QueryError(Exception):
    """A user-facing error raised while processing a query."""


def _non_negative_int(value: str) -> int:
    """argparse type helper that accepts only non-negative integers."""
    try:
        number = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError(f"invalid integer value: '{value}'") from exc
    if number < 0:
        raise argparse.ArgumentTypeError("value must be a non-negative integer")
    return number


def build_arg_parser() -> argparse.ArgumentParser:
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (Dijkstra shortest path).",
    )
    parser.add_argument(
        "--output_file",
        default=None,
        help="Export query results to this file (overwritten by each query).",
    )
    parser.add_argument(
        "--format",
        dest="output_format",
        choices=list(SUPPORTED_FORMATS),
        default="text",
        help="Output format: text or json (default: text).",
    )
    parser.add_argument(
        "--mode",
        choices=list(VALID_MODES),
        default=MODE_SHORTEST_DISTANCE,
        help="Path query mode (default: shortest_distance).",
    )
    parser.add_argument(
        "--max_transfers",
        type=_non_negative_int,
        default=None,
        help="Maximum number of allowed transfers.",
    )
    parser.add_argument(
        "--line_preference",
        default=None,
        help="Preferred line(s), e.g. 'Line 1' or 'Line 1,Line 2'.",
    )
    parser.add_argument(
        "--check_data",
        action="store_true",
        help="Run the data integrity check and exit.",
    )
    return parser


def parse_query_text(text: str) -> List[str]:
    """Split a 'Starting Station; Ending Station' input into two parts.

    Raises:
        QueryError: If the input does not contain exactly one semicolon or
            either side is empty.
    """
    if text.count(";") != 1:
        raise QueryError(INPUT_FORMAT_HINT)
    parts = [part.strip() for part in text.split(";")]
    if not parts[0] or not parts[1]:
        raise QueryError(INPUT_FORMAT_HINT)
    return parts


def parse_line_preference(value: Optional[str]) -> List[str]:
    """Parse a comma-separated line preference string into a list."""
    if not value:
        return []
    return [part.strip() for part in value.split(",") if part.strip()]


def answer_query(network: SubwayNetwork, text: str, options: argparse.Namespace) -> Report:
    """Validate a query string, compute the route and build the report.

    Raises:
        QueryError: For format, station, logic or path errors.
    """
    start_raw, end_raw = parse_query_text(text)

    start = network.standardize_name(start_raw)
    if start is None:
        raise QueryError(
            f"Starting station '{start_raw}' does not exist, please check the station name"
        )
    end = network.standardize_name(end_raw)
    if end is None:
        raise QueryError(
            f"Ending station '{end_raw}' does not exist, please check the station name"
        )
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")

    preferred = parse_line_preference(options.line_preference)
    config = PathfindingConfig(
        mode=options.mode,
        max_transfers=options.max_transfers,
        preferred_lines=preferred,
    )
    query = Query(
        start_station=start,
        end_station=end,
        mode=options.mode,
        max_transfers=options.max_transfers,
        line_preference=options.line_preference,
    )
    try:
        route = find_route(network, start, end, config)
    except RouteNotFoundError as exc:
        raise QueryError(str(exc)) from exc
    return Report.from_route(route, query, options.output_format, options.output_file)


def run_check_data(network: SubwayNetwork) -> int:
    """Run the data integrity check, print the result and return an exit code."""
    checker = DataChecker(network)
    passed = checker.run()
    if passed:
        print("Data integrity check passed.")
        return 0
    print(f"Data integrity check failed with {len(checker.error_messages)} problem(s):")
    for message in checker.error_messages:
        print(f"  - {message}")
    return 1


def run_interactive(network: SubwayNetwork, options: argparse.Namespace) -> int:
    """Run the continuous query loop until the user quits or input ends."""
    print("Beijing Subway fare calculator.")
    print("Enter 'Starting Station; Ending Station' to query (or 'quit' to exit).")
    while True:
        try:
            text = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = text.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        try:
            report = answer_query(network, text, options)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # Defensive: keep the session alive on unexpected failures.
            print(f"Error: unexpected failure while calculating the route: {exc}")
            continue

        print(report.content)
        if options.output_file:
            try:
                path = report.write()
                print(f"Result saved to {path}")
            except OSError as exc:
                print(f"Error: could not write output file: {exc}")
    return 0


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Entry point of the CLI. Returns the process exit code."""
    parser = build_arg_parser()
    options = parser.parse_args(argv)

    try:
        network = SubwayNetwork.from_file(DATA_FILE_PATH)
    except (OSError, ValueError, KeyError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if options.check_data:
        return run_check_data(network)

    available = set(network.line_names())
    for line in parse_line_preference(options.line_preference):
        if line not in available:
            print(
                f"Warning: line '{line}' is not in the network data; the preference has no effect.",
                file=sys.stderr,
            )

    return run_interactive(network, options)

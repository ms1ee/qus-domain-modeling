"""Process-wide key/value storage shared by the simulator modules.

The simulator keeps its shared state (funds, riders, orders, clock) in a
single dictionary so that every module can reach it through ``get_value``

from typing import Any, Dict

_global_dict: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global storage to its default layout.

    The list containers used across modules are always created here so that
    ``get_value("riders")`` and ``get_value("orders")`` return lists.

    global _global_dict
    _global_dict = {
        "riders": [],
        "orders": [],
        "current_tick": 0,

    Args:
        key: Name of the global variable.

        key: Name of the global variable.

"""Grid-based path planning for the food delivery simulator.

The city is an 800x800 unit map divided into an 8x8 grid of 100x100 cells.
Streets run along the grid lines (a coordinate that is a multiple of 100).
Restaurant and customer locations sit on grid corners, so every route can be
expressed as straight street segments joined by right-angle turns.

Riders never block each other: they may share streets, which avoids
deadlocks by construction. Path conflicts are resolved by always preferring
the route whose legs both lie on streets.
"""

import math
from typing import List, Tuple

Point = Tuple[int, int]

DEFAULT_GRID_SIZE = 8
DEFAULT_CELL_SIZE = 100
DEFAULT_MAP_SIZE = DEFAULT_GRID_SIZE * DEFAULT_CELL_SIZE


def snap_to_cell(value: int, cell_size: int = DEFAULT_CELL_SIZE,
                 map_size: int = DEFAULT_MAP_SIZE) -> int:
    """Snap a coordinate to the top-left corner of its grid cell.

    Example: ``150 -> 100`` and ``799 -> 700``. Results are clamped to the map.

    Args:
        value: Raw coordinate.
        cell_size: Size of one grid cell in units.
        map_size: Size of the map in units.

    Returns:
        The snapped integer coordinate in ``[0, map_size]``.
    """
    snapped = (int(value) // cell_size) * cell_size
    return max(0, min(snapped, map_size))


class PathPlanner:
    """Plans street-following routes and single movement steps.

    Attributes:
        grid_size: Number of grid cells per side (8).
        cell_size: Size of a grid cell in units (100).
        map_size: Total map size in units.
    """

    def __init__(self, grid_size: int = DEFAULT_GRID_SIZE,
                 cell_size: int = DEFAULT_CELL_SIZE) -> None:
        """Create a planner for a square grid.

        Args:
            grid_size: Number of cells per side.
            cell_size: Size of each cell in units.
        """
        self.grid_size = grid_size
        self.cell_size = cell_size
        self.map_size = grid_size * cell_size

    def is_on_street(self, x: int, y: int) -> bool:
        """Return True if the point lies on a street (a grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def is_intersection(self, x: int, y: int) -> bool:
        """Return True if the point is a grid intersection."""
        return x % self.cell_size == 0 and y % self.cell_size == 0

    def is_within_map(self, x: int, y: int) -> bool:
        """Return True if the point lies inside the map bounds."""
        return 0 <= x <= self.map_size and 0 <= y <= self.map_size

    def _leg_is_street(self, start: Point, end: Point) -> bool:
        """Return True if a straight leg runs along a street line."""
        if start[1] == end[1]:
            return start[1] % self.cell_size == 0
        if start[0] == end[0]:
            return start[0] % self.cell_size == 0
        return False

    def plan(self, start: Point, goal: Point) -> List[Point]:
        """Plan a street-following route from ``start`` to ``goal``.

        Straight routes are returned as two points. Otherwise an L-shaped
        route with one right-angle turn is produced. When both L-shapes are
        possible, the one whose legs both lie on streets is preferred, and
        horizontal-first is used as the tie breaker.

        Args:
            start: Starting point.
            goal: Destination point.

        Returns:
            List of waypoints including ``start`` and ``goal``.
        """
        start = (int(start[0]), int(start[1]))
        goal = (int(goal[0]), int(goal[1]))
        if start == goal:
            return [start]
        if start[0] == goal[0] or start[1] == goal[1]:
            return [start, goal]

        corner_h = (goal[0], start[1])
        corner_v = (start[0], goal[1])
        score_h = (self._leg_is_street(start, corner_h)
                   + self._leg_is_street(corner_h, goal))
        score_v = (self._leg_is_street(start, corner_v)
                   + self._leg_is_street(corner_v, goal))
        corner = corner_v if score_v > score_h else corner_h
        return [start, corner, goal]

    def next_step(self, x: int, y: int, goal_x: int, goal_y: int,
                  budget: int) -> Tuple[int, int, int]:
        """Compute one movement step toward a goal along streets.

        Special cases:
            * Target directly above/below: moves purely vertically.
            * Target directly left/right: moves purely horizontally.
            * Otherwise horizontal movement is preferred on horizontal
              streets, and vertical movement on vertical streets.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            goal_x: Target x coordinate.
            goal_y: Target y coordinate.
            budget: Maximum distance that may be covered this tick.

        Returns:
            Tuple ``(new_x, new_y, distance_used)``.
        """
        dx = goal_x - x
        dy = goal_y - y
        if (dx == 0 and dy == 0) or budget <= 0:
            return x, y, 0

        c = self.cell_size
        if dx != 0 and y % c == 0:
            horizontal = True
        elif dy != 0 and x % c == 0:
            horizontal = False
        elif dx != 0:
            horizontal = True
        else:
            horizontal = False

        if horizontal:
            step = min(budget, abs(dx))
            return x + (1 if dx > 0 else -1) * step, y, step
        step = min(budget, abs(dy))
        return x, y + (1 if dy > 0 else -1) * step, step

    @staticmethod
    def route_length(points: List[Point]) -> int:
        """Return the total Manhattan length of a polyline route."""
        total = 0
        for (ax, ay), (bx, by) in zip(points, points[1:]):
            total += abs(bx - ax) + abs(by - ay)
        return total

    @staticmethod
    def estimate_ticks(distance: float, speed: int) -> int:
        """Return the number of ticks needed to cover ``distance``.

        Args:
            distance: Distance in units.
            speed: Speed in units per tick (must be positive).

        Returns:
            Ceiling of ``distance / speed``.
        """
        if speed <= 0:
            raise ValueError("speed must be positive")
        return max(0, math.ceil(distance / speed))

"""Order data model for the food delivery simulator."""

import itertools
from typing import Any, Dict, Optional, Sequence, Tuple

import globalvar
from path_planner import snap_to_cell

STATUS_CREATED = "created"
STATUS_ASSIGNED = "assigned"
STATUS_PICKED_UP = "picked_up"
STATUS_COMPLETED = "completed"
ACTIVE_STATUSES = (STATUS_ASSIGNED, STATUS_PICKED_UP)

_order_ids = itertools.count(1)


class Order:
    """A food delivery order from a restaurant to a customer.

    Coordinates supplied by the user are snapped to the top-left corner of
    their grid cell, so restaurant and customer locations always lie on
    street intersections.

    Attributes:
        id: Unique, auto-incremented order identifier.
        restaurant_x: Snapped restaurant x coordinate.
        restaurant_y: Snapped restaurant y coordinate.
        customer_x: Snapped customer x coordinate.
        customer_y: Snapped customer y coordinate.
        status: One of "created", "assigned", "picked_up", "completed".
        rider_id: Identifier of the assigned rider, or None.
        start_time: Tick at which the order was created.
        completion_time: Tick at which the order was delivered, or None.
        distance: Manhattan distance from restaurant to customer.
        time_taken: Ticks from creation to delivery, or None.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order from raw coordinates.

        Args:
            res_pos: Restaurant coordinates ``[x, y]`` in the range 0-800.
            cus_pos: Customer coordinates ``[x, y]`` in the range 0-800.
        """
        rx, ry = int(res_pos[0]), int(res_pos[1])
        cx, cy = int(cus_pos[0]), int(cus_pos[1])
        self.id: int = next(_order_ids)
        self.restaurant_x: int = snap_to_cell(rx)
        self.restaurant_y: int = snap_to_cell(ry)
        self.customer_x: int = snap_to_cell(cx)
        self.customer_y: int = snap_to_cell(cy)
        self.status: str = STATUS_CREATED
        self.rider_id: Optional[int] = None
        self.start_time: int = int(globalvar.get_value("current_tick", 0) or 0)
        self.completion_time: Optional[int] = None
        self.distance: float = float(
            abs(self.customer_x - self.restaurant_x)
            + abs(self.customer_y - self.restaurant_y)
        )
        self.time_taken: Optional[int] = None

    @property
    def restaurant_pos(self) -> Tuple[int, int]:
        """Restaurant grid point as a tuple."""
        return (self.restaurant_x, self.restaurant_y)

    @property
    def customer_pos(self) -> Tuple[int, int]:
        """Customer grid point as a tuple."""
        return (self.customer_x, self.customer_y)

    @property
    def is_active(self) -> bool:
        """True while the order is assigned and not yet delivered."""
        return self.status in ACTIVE_STATUSES

    @property
    def efficiency(self) -> Optional[float]:
        """Delivery efficiency expressed as distance covered per tick."""
        if self.time_taken is None:
            return None
        return self.distance / max(self.time_taken, 1)

    def assign_to(self, rider_id: int) -> None:
        """Bind the order to a rider and mark it as assigned.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_picked_up(self) -> None:
        """Mark the order as picked up at the restaurant."""
        self.status = STATUS_PICKED_UP

    def mark_completed(self, now: int) -> None:
        """Mark the order as delivered and record efficiency data.

        Args:
            now: Tick at which delivery happened.
        """
        self.status = STATUS_COMPLETED
        self.completion_time = now
        self.time_taken = max(0, now - self.start_time)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serializable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

"""Rider model: state machine, movement and delivery log.

Rider states:
    0 - Idle (available for new orders)
    1 - Picking up (en route to the restaurant)
    2 - Delivering (en route to the customer)

State transitions:
    Idle (0) -> Picking up (1)    when an assigned order is accepted
    Picking up (1) -> Delivering (2)  on arrival at the restaurant
    Delivering (2) -> Picking up (1)  after delivery, if more orders wait
    Delivering (2) -> Idle (0)        after delivery, if no orders wait
"""

import logging
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from order_model import ACTIVE_STATUSES
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

RIDER_SPEED = 20
START_X = 400
START_Y = 400

_planner = PathPlanner()


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x: Current x coordinate.
        y: Current y coordinate.
        speed: Movement speed in units per tick.
        state: Current state (0 idle, 1 picking up, 2 delivering).
        orders: All orders ever assigned to this rider (history included).
        special_case: Flag set while moving in a straight line to the target.
        log: Chronological list of event dictionaries. Each entry contains
            "event" and "order_id" plus "tick", "rider_id", "x", "y", "state".
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the default start position (400, 400).

        Args:
            rider_id: Unique identifier for the rider.
        """
        self.rider_id: int = int(rider_id)
        self.x: int = START_X
        self.y: int = START_Y
        self.speed: int = RIDER_SPEED
        self.state: int = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    @property
    def position(self) -> Tuple[int, int]:
        """Current location as a tuple."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Human-readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    def current_order(self) -> Optional[Any]:
        """Return the order currently being served, or None.

        The first active order in the rider's list is always the one being
        served, so orders are handled in assignment sequence.
        """
        for order in self.orders:
            if order.status in ACTIVE_STATUSES:
                return order
        return None

    def active_order_count(self) -> int:
        """Return how many assigned orders are not yet delivered."""
        return sum(1 for order in self.orders if order.status in ACTIVE_STATUSES)

    def move(self) -> None:
        """Advance the rider by one tick.

        State transitions and their log entries happen here. The rider
        spends its speed budget walking along streets toward the current
        target. Arrival at a restaurant or customer is processed immediately,
        and any leftover budget is spent heading to the next target.
        """
        now = self._now()
        self.special_case = False

        if self.state == STATE_IDLE:
            order = self.current_order()
            if order is None:
                return
            self.state = STATE_PICKING_UP
            self._record("accepted_order", order.id)

        budget = self.speed
        while budget > 0:
            order = self.current_order()
            if order is None:
                self._become_idle()
                break
            tx, ty = self._target_for(order)
            if (self.x, self.y) == (tx, ty):
                self._arrive(order, now)
                continue
            self.special_case = (self.x == tx or self.y == ty)
            nx, ny, used = _planner.next_step(self.x, self.y, tx, ty, budget)
            if used <= 0:
                break
            self.x, self.y = nx, ny
            budget -= used

    def _target_for(self, order: Any) -> Tuple[int, int]:
        """Return the current destination for the given order."""
        if self.state == STATE_PICKING_UP:
            return order.restaurant_pos
        return order.customer_pos

    def _arrive(self, order: Any, now: int) -> None:
        """Process arrival at the current destination.

        Args:
            order: The order being served.
            now: The current tick.
        """
        if self.state == STATE_PICKING_UP:
            order.mark_picked_up()
            self.state = STATE_DELIVERING
            self._record("arrived_at_merchant", order.id)
        elif self.state == STATE_DELIVERING:
            order.mark_completed(now)
            self._record("delivered_to_customer", order.id)
            next_order = self.current_order()
            if next_order is not None:
                self.state = STATE_PICKING_UP
                self._record("accepted_order", next_order.id)
            else:
                self._become_idle()

    def _become_idle(self) -> None:
        """Release the rider so it can take new orders."""
        self.state = STATE_IDLE
        self.special_case = False

    def _now(self) -> int:
        """Return the current simulation tick."""
        return int(globalvar.get_value("current_tick", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append a status-change entry to the rider log.

        Args:
            event: Event name (spawn, accepted_order, arrived_at_merchant,
                delivered_to_customer).
            order_id: Related order identifier, or None.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": self._now(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %d: %s (order %s)", self.rider_id, event, order_id)

"""Order management and rider assignment."""

import logging
import math
from typing import List, Optional, Tuple

import globalvar
from order_model import Order, STATUS_ASSIGNED, STATUS_CREATED, STATUS_PICKED_UP
from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Choose the rider that should serve ``order``.

    The rider with the fewest current (undelivered) orders is preferred. Ties
    are broken by the Euclidean distance between the rider and the restaurant:
    ``sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2)``.

    Args:
        order: The order that needs a rider.

    Returns:
        The selected Rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    best: Optional[Rider] = None
    best_key: Optional[Tuple[int, float]] = None
    for rider in riders:
        distance = math.sqrt(
            (rider.x - order.restaurant_x) ** 2
            + (rider.y - order.restaurant_y) ** 2
        )
        key = (rider.active_order_count(), distance)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates, assigns and tracks delivery orders.

    Attributes:
        orders: Every order created in this session (shared global list).
    """

    def __init__(self) -> None:
        """Bind to the shared order list held in global storage."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        self.orders: List[Order] = orders

    def create_order(self, restaurant_pos, customer_pos) -> Order:
        """Create an order and register it.

        Args:
            restaurant_pos: ``(x, y)`` restaurant coordinates.
            customer_pos: ``(x, y)`` customer coordinates.

        Returns:
            The newly created Order.
        """
        order = Order(restaurant_pos, customer_pos)
        self.orders.append(order)
        logger.info("Order %d created", order.id)
        return order

    def assign_order(self, order: Order) -> Optional[Rider]:
        """Assign an order to the best available rider.

        Args:
            order: Order to assign.

        Returns:
            The rider that received the order, or None if no rider exists.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        logger.info("Order %d assigned to rider %d", order.id, rider.rider_id)
        return rider

    def dispatch_pending(self) -> List[Tuple[int, int]]:
        """Try to assign every queued (unassigned) order.

        Returns:
            List of ``(order_id, rider_id)`` pairs that were assigned.
        """
        assigned: List[Tuple[int, int]] = []
        for order in self.orders:
            if order.status != STATUS_CREATED:
                continue
            rider = self.assign_order(order)
            if rider is not None:
                assigned.append((order.id, rider.rider_id))
        return assigned

    def pending_orders(self) -> List[Order]:
        """Return orders that have not yet been picked up."""
        return [o for o in self.orders
                if o.status in (STATUS_CREATED, STATUS_ASSIGNED)]

    def active_orders(self) -> List[Order]:
        """Return orders that are not yet delivered."""
        return [o for o in self.orders
                if o.status in (STATUS_CREATED, STATUS_ASSIGNED, STATUS_PICKED_UP)]

    def find_order(self, order_id: int) -> Optional[Order]:
        """Return the order with ``order_id``, or None."""
        for order in self.orders:
            if order.id == order_id:
                return order
        return None

"""System manager: funds, rider recruitment, simulated clock and persistence."""

import json
import logging
import math
import os
from typing import Any, Dict, List, Optional

import globalvar
from order import OrderManager
from order_model import STATUS_ASSIGNED, STATUS_CREATED, STATUS_PICKED_UP
from path_planner import PathPlanner
from rider import (
    STATE_DELIVERING,

    STATE_PICKING_UP,
    Rider,
)

logger = logging.getLogger(__name__)

CONFIG_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "config.json")

DEFAULT_CONFIG: Dict[str, int] = {
    "initial_funds": 1000,
    "recruit_threshold": 300,
    "recruit_cost": 500,
    "grid_size": 8,
    "cell_size": 100,
    "min_coordinate": 0,
    "max_coordinate": 800,
}


def load_config(path: str = CONFIG_PATH) -> Dict[str, int]:
    """Load system parameters from a JSON file, falling back to defaults.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        A dictionary containing every known configuration key.
    """
    config = dict(DEFAULT_CONFIG)
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError:
        logger.info("Config file %s not found; using defaults", path)
        return config
    except (OSError, ValueError) as exc:
        logger.warning("Could not read config %s (%s); using defaults", path, exc)
        return config

    if isinstance(data, dict):
        for key in DEFAULT_CONFIG:
            if key in data:
                try:
                    config[key] = int(data[key])
                except (TypeError, ValueError):
                    logger.warning("Ignoring invalid config value for %s", key)
    return config


def init_system() -> "SystemManager":
    """Reset global state and create a fully initialized SystemManager.

    Returns:
        The initialized SystemManager.
    """
    globalvar._init()
    config = load_config()
    return SystemManager(config)

    def current_tick(self, value: int) -> None:
        globalvar.set_value("current_tick", int(value))

    def idle_count(self) -> int:
        """Return the number of riders in the idle state."""
        return sum(1 for r in self.riders if r.state == STATE_IDLE)

    def should_recruit(self, pending_count: int) -> bool:
        """Decide whether a new rider should be recruited.

        Args:
            pending_count: Number of orders waiting for pickup.

        Returns:
            True if funds are at least the threshold and pending orders
            exceed idle riders.
        """
        return (self.funds >= self.config["recruit_threshold"]
                and pending_count > self.idle_count())

    def recruit_rider(self) -> Rider:
        """Create a new rider and deduct the recruitment cost.

        Returns:
            The newly recruited Rider.
        """
        self.rider_id_counter = self.rider_id_counter + 1
        rider = Rider(self.rider_id_counter)
        self.riders.append(rider)
        self.funds = self.funds - self.config["recruit_cost"]
        logger.info("Recruited rider %d", rider.rider_id)
        return rider

    def find_rider(self, rider_id: Optional[int]) -> Optional[Rider]:
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def build_route(self, rider: Rider, order: Any) -> List[tuple]:
        """Build the route from the rider's position through pickup to delivery.

        Args:
            rider: Rider that will serve the order.
            order: Order being planned.

        Returns:
            Ordered list of waypoints.
        """
        first = self.planner.plan(rider.position, order.restaurant_pos)
        second = self.planner.plan(order.restaurant_pos, order.customer_pos)
        return first + second[1:]

    def place_order(self, restaurant_x: int, restaurant_y: int,
                    customer_x: int, customer_y: int) -> Dict[str, Any]:
        """Create an order, possibly recruit a rider, and dispatch it.

        Args:
            restaurant_x: Restaurant x coordinate (0-800).
            restaurant_y: Restaurant y coordinate (0-800).
            customer_x: Customer x coordinate (0-800).
            customer_y: Customer y coordinate (0-800).

        Returns:
            Dictionary with keys: order, restaurant, customer,
            recruited_rider, rider, route, route_length, eta_ticks, queued.

        Raises:
            ValueError: If a coordinate is outside the allowed range.
        """
        lo = self.config["min_coordinate"]
        hi = self.config["max_coordinate"]
        for value in (restaurant_x, restaurant_y, customer_x, customer_y):
            if not lo <= int(value) <= hi:
                raise ValueError(f"coordinate {value} outside {lo}-{hi}")

        order = self.order_manager.create_order(
            (restaurant_x, restaurant_y), (customer_x, customer_y))

        pending = len(self.order_manager.pending_orders())
        recruited: Optional[Rider] = None
        if self.should_recruit(pending):
            recruited = self.recruit_rider()

        rider = self.order_manager.assign_order(order)
        route: List[tuple] = []
        route_length = 0
        eta: Optional[int] = None
        if rider is not None:
            route = self.build_route(rider, order)
            route_length = self.planner.route_length(route)
            eta = self.planner.estimate_ticks(route_length, rider.speed)

        return {
            "order": order,
            "restaurant": order.restaurant_pos,
            "customer": order.customer_pos,
            "recruited_rider": recruited,
            "rider": rider,
            "route": route,
            "route_length": route_length,
            "eta_ticks": eta,
            "queued": rider is None,
        }

    def tick(self) -> None:
        """Advance the simulation by one unit of time.

        Order:
            1. Increment the clock.
            2. Recruit a rider if queued orders justify it.
            3. Dispatch queued orders.
            4. Move every rider. Exceptions are logged and the loop continues,
               so one faulty rider cannot stop the simulation.
        """
        self.current_tick = self.current_tick + 1

        queued = [o for o in self.order_manager.orders if o.status == STATUS_CREATED]
        if queued and self.should_recruit(len(self.order_manager.pending_orders())):
            self.recruit_rider()

        try:
            self.order_manager.dispatch_pending()
        except Exception:  # pragma: no cover - defensive recovery
            logger.exception("Dispatch failed; state preserved")

        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                logger.exception("Rider %s failed to move", rider.rider_id)

    def run_ticks(self, count: int) -> None:
        """Advance the simulation by ``count`` ticks."""
        for _ in range(count):
            self.tick()

    def stop(self) -> None:
        """Signal the main loop to exit."""
        self.running = False

    def check_consistency(self) -> List[str]:
        """Check internal state invariants.

        Returns:
            Human-readable descriptions of every inconsistency found. An empty
            list means the state is consistent.
        """
        issues: List[str] = []
        if not isinstance(self.funds, int):
            issues.append("funds is not an integer")

        ids = [r.rider_id for r in self.riders]
        if len(ids) != len(set(ids)):
            issues.append("duplicate rider identifiers detected")
        if ids and max(ids) > self.rider_id_counter:
            issues.append("rider_id_counter is behind the largest rider id")

        for rider in self.riders:
            active = rider.active_order_count()
            current = rider.current_order()
            if rider.state != STATE_IDLE and active == 0:
                issues.append(f"Rider #{rider.rider_id} is busy without orders")
            if rider.state == STATE_IDLE and any(
                    o.status == STATUS_PICKED_UP for o in rider.orders):
                issues.append(f"Rider #{rider.rider_id} is idle while carrying food")
            if rider.state == STATE_PICKING_UP and current is not None \
                    and current.status != STATUS_ASSIGNED:
                issues.append(f"Rider #{rider.rider_id} state/order mismatch")
            if rider.state == STATE_DELIVERING and current is not None \
                    and current.status != STATUS_PICKED_UP:
                issues.append(f"Rider #{rider.rider_id} state/order mismatch")
            if not (0 <= rider.x <= 800 and 0 <= rider.y <= 800):
                issues.append(f"Rider #{rider.rider_id} is off the map")

        for order in self.order_manager.orders:
            if order.status in (STATUS_ASSIGNED, STATUS_PICKED_UP):
                rider = self.find_rider(order.rider_id)
                if rider is None or order not in rider.orders:
                    issues.append(f"Order #{order.id} is not held by its rider")
        return issues

    def save_state(self, path: str) -> None:
        """Persist parameters, riders and order history as JSON.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.

            "funds": self.funds,
            "rider_id_counter": self.rider_id_counter,
            "current_tick": self.current_tick,
            "riders": [
                {
                    "rider_id": r.rider_id,
                    "x": r.x,
                    "y": r.y,
                    "speed": r.speed,
                    "state": r.state,
                    "special_case": r.special_case,
                    "order_ids": [o.id for o in r.orders],
                    "log": r.log,
                }
                for r in self.riders

            "orders": [o.to_dict() for o in self.order_manager.orders],
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)

from typing import Any, Iterable, List, Optional, Sequence, Tuple

from rider import STATE_NAMES

Point = Tuple[int, int]


class DisplayManager:
    """Formats the simulator state for the command line.

    The ASCII map samples the 800x800 unit map every 50 units, producing a
    17x17 character canvas. Intersections of the 8x8 cell grid are drawn at
    even positions, so streets appear as continuous lines of '#'.

    Attributes:
        map_size: Size of the map in units.
        resolution: Units represented by one character.
        symbol_rider: Symbol for riders ('R').
        symbol_restaurant: Symbol for restaurants ('S').
        symbol_customer: Symbol for customers ('C').
        symbol_empty: Symbol for off-street empty space ('.').
        symbol_street: Symbol for streets ('#').
        symbol_route: Symbol for a planned route ('*').
    """

    def __init__(self, map_size: int = 800, cell_size: int = 100,
                 resolution: int = 50) -> None:
        """Configure the renderer.

        Args:
            map_size: Map size in units.
            cell_size: Grid cell size in units.
            resolution: Units per character.
        """
        self.map_size = map_size
        self.cell_size = cell_size
        self.resolution = resolution
        self.symbol_rider = "R"
        self.symbol_restaurant = "S"
        self.symbol_customer = "C"
        self.symbol_empty = "."
        self.symbol_street = "#"
        self.symbol_route = "*"

    def render_status(self, funds: int, riders: Sequence[Any]) -> str:
        """Render the system snapshot: funds and each rider's details.

        Args:
            funds: Current system funds.
            riders: Riders to describe.

        Returns:
            Multi-line status text.
        """
        lines = [
            "=== System Status ===",
            f"System funds: {funds}",
            f"Rider count: {len(riders)}",
        ]
        for rider in riders:
            order_ids = [o.id for o in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id}: location ({rider.x}, {rider.y}) | "
                f"status: {STATE_NAMES.get(rider.state, 'unknown')} | "
                f"active orders: {rider.active_order_count()} | "
                f"order list: {order_ids if order_ids else 'none'}"
            )
        return "\n".join(lines)

    def render_orders(self, orders: Sequence[Any]) -> str:
        """Render a table of all orders.

        Args:
            orders: Orders to display.

        Returns:
            Multi-line table text.
        """
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Dist':>7} {'Taken':>6} {'Rider':>6} {'Status':>10}")
        lines = ["=== Orders ===", header, "-" * len(header)]
        for o in orders:
            rest = f"({o.restaurant_x},{o.restaurant_y})"
            cust = f"({o.customer_x},{o.customer_y})"
            taken = "-" if o.time_taken is None else str(o.time_taken)
            rider = "-" if o.rider_id is None else str(o.rider_id)
            lines.append(
                f"{o.id:>4} {o.start_time:>6} {rest:>12} {cust:>12} "
                f"{o.distance:>7.0f} {taken:>6} {rider:>6} {o.status:>10}"
            )
        return "\n".join(lines)

    def render_rider_log(self, riders: Sequence[Any]) -> str:
        """Render every recorded status change for all riders.

        Args:
            riders: Riders whose logs should be shown.

        Returns:
            Multi-line log text.
        """
        if not riders:
            return "No riders."
        lines = ["=== Rider Event Log ==="]
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id}:")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order = "-" if entry["order_id"] is None else f"#{entry['order_id']}"
                lines.append(
                    f"  tick {entry['tick']:>4} | {entry['event']:<20} | "
                    f"order {order:>5} | pos ({entry['x']}, {entry['y']}) | "
                    f"state {STATE_NAMES.get(entry['state'], 'unknown')}"
                )
        return "\n".join(lines)

    def render_map(self, riders: Iterable[Any] = (), orders: Iterable[Any] = (),
                   route: Optional[List[Point]] = None) -> str:
        """Render the ASCII map with riders, orders and an optional route.

        Args:
            riders: Riders to mark with 'R'.
            orders: Orders whose restaurants ('S') and customers ('C') are marked.
            route: Optional polyline drawn with '*'.

        Returns:
            Multi-line map text.
        """
        steps = self.map_size // self.resolution
        size = steps + 1
        canvas = [[self.symbol_empty] * size for _ in range(size)]

        for row in range(size):
            y = (steps - row) * self.resolution
            for col in range(size):
                x = col * self.resolution
                if x % self.cell_size == 0 or y % self.cell_size == 0:
                    canvas[row][col] = self.symbol_street

        if route:
            for a, b in zip(route, route[1:]):
                for px, py in self._segment_points(a, b):
                    self._plot(canvas, px, py, self.symbol_route)
            if len(route) == 1:
                self._plot(canvas, route[0][0], route[0][1], self.symbol_route)

        for order in orders:
            self._plot(canvas, order.restaurant_x, order.restaurant_y,
                       self.symbol_restaurant)
            self._plot(canvas, order.customer_x, order.customer_y,
                       self.symbol_customer)

        for rider in riders:
            self._plot(canvas, rider.x, rider.y, self.symbol_rider)

        lines = ["=== Map (y=800 at top, x=0 at left) ==="]
        for row in range(size):
            y = (steps - row) * self.resolution
            label = f"{y:>4}" if y % 200 == 0 else "    "
            lines.append(label + " " + " ".join(canvas[row]))
        axis = "     " + " ".join(
            f"{x:<1}" if (i % 4 == 0) else " " for i, x in enumerate(range(size)))
        lines.append("     x: 0 ... 800 (ticks every 200 units)")
        lines.append(
            f"Legend: {self.symbol_rider}=rider {self.symbol_restaurant}=restaurant "
            f"{self.symbol_customer}=customer {self.symbol_street}=street "
            f"{self.symbol_empty}=empty {self.symbol_route}=route"
        )
        return "\n".join(lines)

    def _plot(self, canvas: List[List[str]], x: int, y: int, symbol: str) -> None:
        """Place ``symbol`` on the canvas at the sampled location of (x, y)."""
        steps = self.map_size // self.resolution
        col = int(round(x / self.resolution))
        row = steps - int(round(y / self.resolution))
        if 0 <= row <= steps and 0 <= col <= steps:
            canvas[row][col] = symbol

    def _segment_points(self, a: Point, b: Point) -> List[Point]:
        """Return sampled points along a straight segment from a to b."""
        ax, ay = a
        bx, by = b
        dx, dy = bx - ax, by - ay
        length = max(abs(dx), abs(dy))
        count = length // self.resolution
        sx = (dx > 0) - (dx < 0)
        sy = (dy > 0) - (dy < 0)
        return [(ax + sx * self.resolution * i, ay + sy * self.resolution * i)
                for i in range(count + 1)]

"""Command parsing, shortcuts, history and input validation."""

import re
from typing import List, Tuple

COORDINATE_LABELS = (
    "restaurant x",
    "restaurant y",
    "customer x",
    "customer y",
)

COMMAND_ALIASES = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "hist": "history",
    "?": "help",
    "h": "help",
    "q": "quit",
    "x": "quit",
    "exit": "quit",
}

_INTEGER_PATTERN = re.compile(r"^[+-]?\d+$")


class InputError(ValueError):
    """Raised when user input cannot be accepted."""


class InputManager:
    """Parses commands and validates user-supplied values.

    Attributes:
        command_history: Previously executed commands, oldest first.
        min_coordinate: Smallest accepted coordinate.
        max_coordinate: Largest accepted coordinate.
    """

    def __init__(self, min_coordinate: int = 0, max_coordinate: int = 800,
                 history_limit: int = 200) -> None:
        """Create the input manager.

        Args:
            min_coordinate: Smallest accepted coordinate.
            max_coordinate: Largest accepted coordinate.
            history_limit: Maximum number of remembered commands.
        """
        self.command_history: List[str] = []
        self.min_coordinate = min_coordinate
        self.max_coordinate = max_coordinate
        self.history_limit = history_limit

    def resolve(self, line: str) -> str:
        """Expand history shortcuts (``!!`` and ``!n``).

        Args:
            line: Raw user input.

        Returns:
            The command text to execute.

        Raises:
            InputError: If the referenced history entry does not exist.
        """
        text = line.strip()
        if text == "!!":
            if not self.command_history:
                raise InputError("no previous command to repeat")
            return self.command_history[-1]
        if text.startswith("!") and text[1:].isdigit():
            index = int(text[1:])
            if not 1 <= index <= len(self.command_history):
                raise InputError(f"no history entry #{index}")
            return self.command_history[index - 1]
        return text

    def record(self, line: str) -> None:
        """Append a command to the history, trimming old entries."""
        if not line:
            return
        self.command_history.append(line)
        overflow = len(self.command_history) - self.history_limit
        if overflow > 0:
            del self.command_history[:overflow]

    @staticmethod
    def parse_command(text: str) -> Tuple[str, List[str]]:
        """Split a command line into a lower-case name and its arguments.

        Args:
            text: Command text.

        Returns:
            Tuple ``(name, args)``; name is "" for blank input.
        """
        parts = text.split()
        if not parts:
            return "", []
        name = parts[0].lower()
        return COMMAND_ALIASES.get(name, name), parts[1:]

    def parse_coordinates(self, tokens: List[str]) -> List[int]:
        """Validate four integer coordinates.

        Commas and spaces may both separate values.

        Args:
            tokens: Raw argument tokens.

        Returns:
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            InputError: On wrong count, non-integer text, or out-of-range value.
        """
        cleaned: List[str] = []
        for token in tokens:
            cleaned.extend(token.replace(",", " ").split())
        if len(cleaned) != 4:
            raise InputError(
                f"expected 4 integers (restaurant x y, customer x y), got {len(cleaned)}")

        values: List[int] = []
        for label, token in zip(COORDINATE_LABELS, cleaned):
            if not _INTEGER_PATTERN.match(token):
                raise InputError(f"{label} '{token}' is not an integer")
            value = int(token)
            if not self.min_coordinate <= value <= self.max_coordinate:
                raise InputError(
                    f"{label} {value} is out of range "
                    f"{self.min_coordinate}-{self.max_coordinate}")
            values.append(value)
        return values

    def parse_positive_int(self, tokens: List[str], default: int, maximum: int) -> int:
        """Parse an optional positive integer argument.

        Args:
            tokens: Argument tokens (only the first is used).
            default: Value used when no argument is given.
            maximum: Largest accepted value.

        Returns:
            The parsed integer.

        Raises:
            InputError: If the value is not a positive integer within range.
        """
        if not tokens:
            return default
        token = tokens[0]
        if not token.isdigit() or int(token) <= 0:
            raise InputError(f"'{token}' is not a positive integer")
        value = int(token)
        if value > maximum:
            raise InputError(f"value {value} exceeds maximum {maximum}")
        return value

  status (st)              Show system funds, riders and their orders.

  save [file]              Save state to a JSON file (default dispatch_state.json).

Shortcuts: !! repeats the last command, !n repeats history entry n."""


class Application:

    Attributes:
        system: The SystemManager holding simulation state.
        inputs: Parser and validator for user input.
        display: Renderer for status text and maps.
    """

    def __init__(self) -> None:
        """Initialize the system, input parser and display."""
        self.system: SystemManager = init_system()
        self.inputs = InputManager(
            min_coordinate=self.system.config["min_coordinate"],
            max_coordinate=self.system.config["max_coordinate"],
        )
        self.display = DisplayManager(
            map_size=self.system.config["grid_size"] * self.system.config["cell_size"],
            cell_size=self.system.config["cell_size"],
        )
        self.commands: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "history": self.cmd_history,
            "save": self.cmd_save,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def run(self) -> None:
        """Run the main loop until the user quits or input ends."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands.")
        while self.system.running:
            try:
                line = input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            self.execute(line)
        print("Goodbye.")

    def execute(self, line: str) -> None:
        """Parse and run a single command line, reporting errors gracefully.

        Args:
            line: Raw user input.
        """
        try:
            text = self.inputs.resolve(line)
        except InputError as exc:
            print(f"Error: {exc}")
            return
        if not text:
            return
        self.inputs.record(text)

        name, args = self.inputs.parse_command(text)
        handler = self.commands.get(name)
        if handler is None:
            print(f"Unknown command '{name}'. Type 'help' for the command list.")
            return
        try:
            handler(args)
        except InputError as exc:
            print(f"Input error: {exc}")
        except Exception:  # Recovery: keep the simulation alive.
            logging.getLogger(__name__).exception("Command '%s' failed", name)
            print("An internal error occurred; the system state was preserved.")

    def cmd_start(self, args: List[str]) -> None:

        print(f"  Assigned to Rider #{rider.rider_id} "

    def cmd_status(self, args: List[str]) -> None:

        """Show every rider's status-change log."""
        print(self.display.render_rider_log(self.system.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map with riders and active orders."""
        active = self.system.order_manager.active_orders()
        print(self.display.render_map(self.system.riders, active))

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation clock."""
        count = self.inputs.parse_positive_int(args, default=1, maximum=1000)
        self.system.run_ticks(count)
        riders = self.system.riders
        busy = sum(1 for r in riders if r.state != 0)
        print(f"Advanced {count} tick(s). Current tick: {self.system.current_tick}. "
              f"Busy riders: {busy}/{len(riders)}.")

    def cmd_history(self, args: List[str]) -> None:
        """List previously entered commands."""
        if not self.inputs.command_history:
            print("No command history.")
            return
        for index, entry in enumerate(self.inputs.command_history, start=1):
            print(f"  {index:>3}  {entry}")

    def cmd_save(self, args: List[str]) -> None:
        """Save the current state to a JSON file."""
        path = args[0] if args else "dispatch_state.json"
        try:
            self.system.save_state(path)
        except OSError as exc:
            print(f"Could not save state: {exc}")
            return
        print(f"State saved to {path}.")

    def cmd_help(self, args: List[str]) -> None:
        """Print the command reference."""
        print(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the main loop."""
        self.system.stop()

    """Configure logging and start the interactive application."""

        level=logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        stream=sys.stderr,
    )
    Application().run()


if __name__ == "__main__":

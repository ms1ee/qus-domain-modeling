"""Participant list loading, validation and group extraction."""

from __future__ import annotations

import os

from lottery_system.default_data import DEFAULT_LIST_TEXT
from lottery_system.models import User


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.source_mode: str | None = None
        self.file_path: str | None = None
        self.is_validated: bool = False

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        self.source_mode = "default"
        self.file_path = None
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list uploaded by the user."""
        self.source_mode = "custom"
        self.file_path = path
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it. Returns (valid users, error messages)."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines and '#' comments are skipped."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            stripped = raw.strip()
            if not stripped or stripped.startswith("#"):
                continue
            parts = [part.strip() for part in stripped.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{stripped}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence kept) and duplicate entries."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        self.is_validated = True
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of members."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

"""Prize library management and prize ordering."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the ordered list of prizes. Starts with no prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the draw order.

        Raises ValueError when the input is invalid.
        """
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(p.name == clean_name for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups = self._clean_groups(avoid_groups)
        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def set_avoid_groups(self, name: str, avoid_groups: list[str]) -> bool:
        """Replace the avoidance groups of an existing prize."""
        for prize in self._prizes:
            if prize.name == name:
                prize.avoid_groups = self._clean_groups(avoid_groups)
                return True
        return False

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current draw order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if len(order) != count:
            return False
        if set(order) != set(range(1, count + 1)):
            return False
        if any(not isinstance(pos, int) or isinstance(pos, bool) for pos in order):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in order]
        return True

    @staticmethod
    def _clean_groups(groups: list[str] | None) -> list[str]:
        if not groups:
            return []
        cleaned: list[str] = []
        for group in groups:
            text = (group or "").strip()
            if text and text not in cleaned:
                cleaned.append(text)
        return cleaned

"""Lottery rule engine: eligibility filtering, special rules and draws."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User

CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
MA_JIAQI = "Ma Jiaqi"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"


def _norm(text: str | None) -> str:
    return (text or "").strip().casefold()


def _is_drawer(prize: Prize, drawer_name: str) -> bool:
    return _norm(prize.associated_drawer) == _norm(drawer_name) and bool(
        _norm(prize.associated_drawer)
    )


def _is_cheng_ru(user: User) -> bool:
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


class LotteryManager:
    """Draws winners while enforcing the lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # Results (True = contained Cheng Ru Group member) of the two most recent
        # draws by lottery persons Chen Qixian / Jin Duo.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.Random()

    @property
    def winner_names(self) -> set[str]:
        return set(self._winner_names)

    def is_relaxed(self, prize: Prize) -> bool:
        """Li Bin as lottery person: relaxed filtering (avoidance groups ignored)."""
        return _is_drawer(prize, LI_BIN)

    def is_cheng_ru_protected(self, prize: Prize) -> bool:
        """Ma Jiaqi's prize is limited to Cheng Ru Group when the two previous
        Chen Qixian / Jin Duo draws both contained no Cheng Ru Group member."""
        return (
            _is_drawer(prize, MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = [u for u in self._users if u.name not in self._winner_names]

        if self.is_cheng_ru_protected(prize):
            protected = [u for u in available if _is_cheng_ru(u)]
            if protected:
                return protected  # overrides avoidance groups

        if self.is_relaxed(prize):
            return available

        avoided = {_norm(group) for group in prize.avoid_groups}
        return [u for u in available if _norm(u.department) not in avoided]

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single eligible winner, or None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_names.add(winner.name)
        return winner

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applying to a prize round."""
        rules = ["previous winners are excluded"]
        if self.is_relaxed(prize):
            rules.append("lottery person Li Bin: relaxed rules, avoidance groups not applied")
        elif prize.avoid_groups:
            rules.append("avoidance groups excluded: " + ", ".join(prize.avoid_groups))
        if self.is_cheng_ru_protected(prize):
            rules.append(
                "Cheng Ru Group protection active: only Cheng Ru Group members are eligible"
            )
        if prize.associated_drawer:
            rules.append(f"lottery person: {prize.associated_drawer}")
        return "; ".join(rules)

    def finish_round(self, prize: Prize, winners: list[User]) -> None:
        """Update fallback-mechanism state after a round has ended."""
        if _is_drawer(prize, CHEN_QIXIAN) or _is_drawer(prize, JIN_DUO):
            has_cheng_ru = any(_is_cheng_ru(u) for u in winners)
            self.last_two_draw_results.append(has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self.is_cheng_ru_protected(prize):
            # The protection window has been consumed by this Ma Jiaqi round.
            self.last_two_draw_results = []

"""Winner recording, grouping, sorting and CSV-style export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


def prize_level(name: str) -> int:
    """Prize level for sorting: First=1 (highest), Second=2, Third=3, others last."""
    lowered = (name or "").strip().casefold()
    for keyword, level in (("first", 1), ("second", 2), ("third", 3)):
        if lowered.startswith(keyword):
            return level
    return 99


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_order: int
    win_time: datetime

    @property
    def prize_name(self) -> str:
        return self.prize.name


class ResultManager:
    """Keeps the complete winner list of the lottery."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(
            user=user,
            prize=prize,
            winning_order=len(self._winners) + 1,
            win_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """All winners in global winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Winners grouped by prize name, each list in winning order."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            grouped.setdefault(winner.prize.name, []).append(winner)
        return grouped

    def get_sorted_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Winners sorted by prize level (high to low), then prize order, then winning order."""
        index = {prize.name: pos for pos, prize in enumerate(prizes)}
        return sorted(
            self._winners,
            key=lambda w: (
                prize_level(w.prize.name),
                index.get(w.prize.name, len(index)),
                w.winning_order,
            ),
        )

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        now = datetime.now()
        path = os.path.join(os.getcwd(), f"awards_result_{now.strftime('%Y%m%d')}.txt")
        header = [
            "Prize Level",
            "Prize Name",
            "Winning Order",
            "Name",
            "Department",
            "Win Time",
        ]
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(header)
                for winner in self.get_sorted_winners(prizes):
                    level = prize_level(winner.prize.name)
                    writer.writerow(
                        [
                            level if level != 99 else "",
                            winner.prize.name,
                            winner.winning_order,
                            winner.user.name,
                            winner.user.department,
                            winner.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Command-line interaction for the team building lottery."""

from __future__ import annotations

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager, prize_level

BANNER = (
    "=" * 60
    + "\n  Company Team Building - Smart Lottery System\n"
    + "=" * 60
)


def ask(prompt: str) -> str:
    return input(prompt).strip()


def ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def print_group_info(lm: ListManager, users: list[User]) -> None:
    info = lm.get_group_info(users)
    print("\nDepartment groups:")
    for department, count in info.items():
        print(f"  {department}: {count} member(s)")
    print(f"  Total: {len(users)} participant(s)")


def load_participants(lm: ListManager) -> list[User] | None:
    """Participant list loading with validation. Returns None when user exits."""
    while True:
        print("\nParticipant list loading:")
        print("  1. Load built-in default team building list")
        print("  2. Upload custom txt list (one 'Name,Department' per line)")
        print("  0. Exit")
        choice = ask("Select an option: ")
        if choice == "0":
            return None
        if choice == "1":
            users, errors = lm.load_default_list()
        elif choice == "2":
            path = ask("Enter the txt file path: ")
            users, errors = lm.load_custom_list(path)
        else:
            print("Invalid option.")
            continue

        if errors:
            print(f"\nFound {len(errors)} invalid record(s):")
            for error in errors:
                print(f"  - {error}")
            print("Options: [r] remove invalid records and continue, [a] abort upload")
            if ask("Your choice: ").lower() != "r":
                print("Upload terminated.")
                continue
            print("Invalid records removed.")

        valid, duplicates = lm.validate_users(users)
        if duplicates:
            print(f"\nFound {len(duplicates)} duplicate name(s):")
            for dup in duplicates:
                print(f"  - {dup.name} ({dup.department})")
            print("Options: [r] remove duplicates and continue, [a] abort upload")
            if ask("Your choice: ").lower() != "r":
                print("Upload terminated.")
                continue
            print("Duplicates removed.")

        if not valid:
            print("No valid participants found.")
            continue

        print(f"\nLoaded {len(valid)} valid participant(s).")
        print_group_info(lm, valid)
        return valid


def show_prizes(pm: PrizeManager) -> None:
    prizes = pm.get_prizes()
    if not prizes:
        print("No prizes configured.")
        return
    print("\nPrize draw order:")
    for pos, prize in enumerate(prizes, start=1):
        avoid = ", ".join(prize.avoid_groups) or "None"
        drawer = prize.associated_drawer or "None"
        print(
            f"  {pos}. {prize.name} | Quantity: {prize.quantity} | "
            f"Avoid groups: {avoid} | Lottery person: {drawer}"
        )


def add_custom_prize(pm: PrizeManager) -> None:
    name = ask("Prize name: ")
    try:
        quantity = int(ask("Quantity: "))
    except ValueError:
        print("Error: quantity must be an integer.")
        return
    groups_text = ask("Avoidance groups (comma-separated, blank for none): ")
    groups = [g.strip() for g in groups_text.split(",") if g.strip()]
    drawer = ask("Associated lottery person (blank for none): ")
    try:
        pm.add_prize(name, quantity, groups, drawer or None)
        print("Prize added.")
    except ValueError as exc:
        print(f"Error: {exc}")


def reorder_prizes(pm: PrizeManager) -> None:
    show_prizes(pm)
    if not pm.get_prizes():
        return
    raw = ask("Enter the new order as serial numbers separated by spaces (e.g. 2 1 3): ")
    try:
        order = [int(token) for token in raw.split()]
    except ValueError:
        print("Invalid input: serial numbers must be integers.")
        return
    if pm.reorder_prizes(order):
        print("Prize order updated.")
        show_prizes(pm)
    else:
        print("Invalid order: every serial number must appear exactly once.")


def set_avoid_groups(pm: PrizeManager) -> None:
    show_prizes(pm)
    prizes = pm.get_prizes()
    if not prizes:
        return
    try:
        pos = int(ask("Enter the serial number of the prize: "))
    except ValueError:
        print("Invalid serial number.")
        return
    if not 1 <= pos <= len(prizes):
        print("Serial number out of range.")
        return
    groups_text = ask("Avoidance groups (comma-separated, blank for none): ")
    groups = [g.strip() for g in groups_text.split(",") if g.strip()]
    if pm.set_avoid_groups(prizes[pos - 1].name, groups):
        print("Avoidance groups updated.")


def manage_prizes(pm: PrizeManager) -> None:
    while True:
        print("\nPrize configuration:")
        print("  1. Load preset prize library")
        print("  2. Add custom prize")
        print("  3. Show prizes (draw order)")
        print("  4. Reorder prizes")
        print("  5. Set avoidance groups for a prize")
        print("  0. Back")
        choice = ask("Select an option: ")
        if choice == "1":
            errors = pm.load_default_prizes()
            if errors:
                for error in errors:
                    print(f"  - {error}")
            else:
                print("Preset prizes loaded.")
        elif choice == "2":
            add_custom_prize(pm)
        elif choice == "3":
            show_prizes(pm)
        elif choice == "4":
            reorder_prizes(pm)
        elif choice == "5":
            set_avoid_groups(pm)
        elif choice == "0":
            return
        else:
            print("Invalid option.")


def wait_while_paused() -> bool:
    """Handle paused state. Returns True if the round should be terminated."""
    print("Lottery paused.")
    while True:
        cmd = ask("Type 'continue' to resume or 'terminate' to end this round: ").lower()
        if cmd == "continue":
            print("Lottery resumed.")
            return False
        if cmd == "terminate":
            return True
        print("Unknown command.")


def run_lottery(prizes, lottery: LotteryManager, results: ResultManager) -> None:
    total_rounds = len(prizes)
    for round_no, prize in enumerate(prizes, start=1):
        print(f"\n==== Round {round_no}/{total_rounds}: {prize.name} ====")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {lottery.describe_rules(prize)}")

        round_winners: list[User] = []
        terminated = False
        while len(round_winners) < prize.quantity and not terminated:
            position = len(round_winners) + 1
            print(f"Currently drawing the {ordinal(position)} winner")
            cmd = ask("Press Enter to draw, or type 'pause' / 'terminate': ").lower()
            if cmd == "terminate":
                terminated = True
            elif cmd == "pause":
                terminated = wait_while_paused()
            elif cmd == "":
                user = lottery.draw_one(prize)
                if user is None:
                    print("No eligible participants remain for this prize. Round ends.")
                    break
                round_winners.append(user)
                results.add_winner(user, prize)
                print(
                    f"*** Congratulations {user.name} ({user.department}) "
                    f"for winning {prize.name}! ***"
                )
            else:
                print("Unknown command. Use Enter, 'pause', or 'terminate'.")

        lottery.finish_round(prize, round_winners)

        print(f"\nRound results for {prize.name}:")
        if round_winners:
            for order, user in enumerate(round_winners, start=1):
                print(f"  {order}. {user.name} ({user.department})")
        else:
            print("  No winners drawn.")
        if terminated:
            print(
                f"Round terminated early: {len(round_winners)} of "
                f"{prize.quantity} winner(s) drawn. Moving to next round."
            )


def print_final_summary(prizes, results: ResultManager) -> None:
    print("\n==== Final Winner List (by prize level, high to low) ====")
    by_prize = results.get_winners_by_prize(prizes)
    if not any(by_prize.values()):
        print("No winners.")
        return
    ordered = sorted(enumerate(prizes), key=lambda t: (prize_level(t[1].name), t[0]))
    for _, prize in ordered:
        winners = by_prize.get(prize.name, [])
        if not winners:
            continue
        print(f"\n{prize.name}:")
        for winner in winners:
            print(
                f"  #{winner.winning_order}  {winner.user.name} "
                f"({winner.user.department})"
            )


def main() -> None:
    print(BANNER)
    lm = ListManager()
    pm = PrizeManager()
    try:
        users = load_participants(lm)
        if users is None:
            print("Goodbye.")
            return

        while True:
            print("\nMain menu:")
            print("  1. View department groups")
            print("  2. Prize configuration")
            print("  3. Start lottery")
            print("  0. Exit")
            choice = ask("Select an option: ")
            if choice == "1":
                print_group_info(lm, users)
            elif choice == "2":
                manage_prizes(pm)
            elif choice == "3":
                prizes = pm.get_prizes()
                if not prizes:
                    print("No prizes configured. Please add prizes first.")
                    continue
                lottery = LotteryManager(users)
                results = ResultManager()

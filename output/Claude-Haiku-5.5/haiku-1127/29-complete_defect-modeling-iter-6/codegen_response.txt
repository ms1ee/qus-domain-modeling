"""Global variable storage shared by the food delivery dispatch modules.

The storage is a module-level dictionary. ``_init()`` resets it to the
default contents and is called by ``system.init_system()`` at startup.

Default keys:
    riders (list[Rider]): dynamic list of all riders currently in service.
    current_tick (int): simulated time, advanced once per tick.
    system (SystemManager): the running system manager instance.
"""

from typing import Any, Dict

_global_dict: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global storage to its default values.

    The rider list starts empty and the simulated time starts at tick 0.
    """
    global _global_dict
    _global_dict = {"riders": [], "current_tick": 0}


def set_value(key: str, value: Any) -> None:
    """Store a value under the given key.

    Args:
        key: Name of the global variable.
        value: Value to store.
    """
    _global_dict[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under the given key.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` if the key is missing.
    """
    return _global_dict.get(key, default)

"""System parameter defaults and optional JSON configuration loading.

Parameters can be customised by placing a JSON object in
``dispatch_config.json`` in the working directory. Unknown keys and keys
with the wrong value type are ignored with a warning.
"""

import copy
import json
import logging
import os
from typing import Any, Dict

logger = logging.getLogger(__name__)

CONFIG_FILE = "dispatch_config.json"

DEFAULT_CONFIG: Dict[str, Any] = {
    "initial_funds": 300,
    "recruit_threshold": 300,
    "recruit_cost": 300,
    "delivery_fee": 100,
    "default_rider_speed": 50,
    "load_penalty": 100,
    "max_ticks_per_order": 5000,
    "save_file": "dispatch_state.json",
    "log_file": "dispatch.log",
    "log_level": "INFO",
}


def load_config(path: str = CONFIG_FILE) -> Dict[str, Any]:
    """Load system parameters, falling back to defaults on any problem.

    Args:
        path: Path of the JSON configuration file.

    Returns:
        A dictionary containing every known configuration key.
    """
    config = copy.deepcopy(DEFAULT_CONFIG)
    if not os.path.exists(path):
        return config
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read config %s: %s; using defaults", path, exc)
        return config
    if not isinstance(data, dict):
        logger.warning("Config %s must contain a JSON object; using defaults", path)
        return config
    for key, value in data.items():
        if key not in DEFAULT_CONFIG:
            logger.warning("Ignoring unknown config key '%s'", key)
            continue
        expected = type(DEFAULT_CONFIG[key])
        if not isinstance(value, expected):
            logger.warning("Config key '%s' must be of type %s; ignored",
                           key, expected.__name__)
            continue
        config[key] = value
    return config

"""Grid coordinate system for the 800x800 delivery map.

The map is an 8x8 grid of 100x100 unit cells. Streets are the grid lines
where an x or y coordinate is a multiple of 100. Order locations are
converted to cell centres using ``center = (value // 100) * 100 + 50``.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = MAP_SIZE // CELL_SIZE  # 8 x 8 = 64 cells
MIN_COORDINATE = 0
MAX_COORDINATE = MAP_SIZE


class GridCell:
    """A single 100x100 cell of the map.

    Attributes:
        grid_x: Column index (0-7).
        grid_y: Row index (0-7).
        center_x: X coordinate of the cell centre.
        center_y: Y coordinate of the cell centre.
        is_street: Always False for cell interiors; streets are the grid
            lines that bound the cells.
    """

    def __init__(self, grid_x: int, grid_y: int) -> None:
        """Create a cell and compute its centre point.

        Args:
            grid_x: Column index of the cell.
            grid_y: Row index of the cell.
        """
        self.grid_x = grid_x
        self.grid_y = grid_y
        self.center_x = grid_x * CELL_SIZE + CELL_SIZE // 2
        self.center_y = grid_y * CELL_SIZE + CELL_SIZE // 2
        self.is_street = False


class Grid:
    """The 8x8 grid of cells covering the map."""

    def __init__(self, map_size: int = MAP_SIZE,
                 cell_size: int = CELL_SIZE) -> None:
        """Build the grid.

        Args:
            map_size: Width and height of the map in units.
            cell_size: Width and height of one cell in units.

        Raises:
            ValueError: If the map size is not a multiple of the cell size.
        """
        if cell_size <= 0 or map_size % cell_size != 0:
            raise ValueError("map_size must be a multiple of cell_size")
        self.map_size = map_size
        self.cell_size = cell_size
        self.dimension = map_size // cell_size
        self.cells = [[GridCell(gx, gy) for gy in range(self.dimension)]
                      for gx in range(self.dimension)]

    def cell_index(self, value: int) -> int:
        """Return the grid index containing a coordinate (clamped to the map).

        Args:
            value: A coordinate along one axis.

        Returns:
            Index in ``range(dimension)``. The map edge 800 maps to the last
            cell.
        """
        index = int(value) // self.cell_size
        return min(max(index, 0), self.dimension - 1)

    def cell_at(self, x: int, y: int) -> GridCell:
        """Return the cell that contains the point (x, y)."""
        return self.cells[self.cell_index(x)][self.cell_index(y)]

    def is_street_point(self, x: int, y: int) -> bool:
        """Return True if (x, y) lies on a street (a grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def is_intersection(self, x: int, y: int) -> bool:
        """Return True if (x, y) is a street crossing point."""
        return x % self.cell_size == 0 and y % self.cell_size == 0


def to_cell_center(value: int) -> int:
    """Convert a raw coordinate to the centre of its grid cell.

    Implements ``center = (value // 100) * 100 + 50``. Values at the map edge
    (800) are placed in the last cell so the centre stays within the map.

    Args:
        value: Raw coordinate in the range 0-800.

    Returns:
        The cell centre coordinate along the same axis.
    """
    index = min(max(int(value), 0) // CELL_SIZE, GRID_DIMENSION - 1)
    return index * CELL_SIZE + CELL_SIZE // 2


def validate_coordinate(value, label: str = "coordinate") -> int:
    """Check that a value is an integer within the map range.

    Args:
        value: The value to check.
        label: Name used in error messages.

    Returns:
        The validated integer.

    Raises:
        ValueError: If the value is not an integer or is out of range.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError(f"{label} must be an integer, got {value!r}")
    if not MIN_COORDINATE <= value <= MAX_COORDINATE:
        raise ValueError(
            f"{label} {value} is outside the range "
            f"{MIN_COORDINATE}-{MAX_COORDINATE}")
    return value

"""Grid-based path planning and step-wise movement.

Routes are Manhattan (axis-aligned) polylines. The planner prefers routes
that travel along streets (grid lines) and handles the special case where
the target lies directly above, below or beside the start point by moving
in a straight line. Riders do not block each other, so no path conflicts or
deadlocks can occur in the tick-based model.
"""

import logging
import math
from typing import List, Sequence, Tuple

from grid import CELL_SIZE, Grid

logger = logging.getLogger(__name__)

Point = Tuple[int, int]


def _sign(value: int) -> int:
    """Return -1, 0 or 1 according to the sign of value."""
    return (value > 0) - (value < 0)


def _path_length(start: Point, waypoints: Sequence[Point]) -> int:
    """Return the Manhattan length of a polyline starting at ``start``."""
    total = 0
    current = start
    for point in waypoints:
        total += abs(point[0] - current[0]) + abs(point[1] - current[1])
        current = point
    return total


class Route:
    """A planned route from a start point through a list of waypoints.

    Attributes:
        start: Starting point (x, y).
        waypoints: Ordered list of corner/end points (x, y).
        speed: Speed used to estimate the duration, in units per tick.
        total_distance: Total Manhattan length of the route.
        estimated_ticks: Number of ticks needed at the given speed.
    """

    def __init__(self, start: Point, waypoints: Sequence[Point],
                 speed: int) -> None:
        """Create a route and compute its distance and duration.

        Args:
            start: Starting point.
            waypoints: Points to visit in order.
            speed: Movement speed in units per tick.
        """
        self.start = (int(start[0]), int(start[1]))
        self.waypoints: List[Point] = [(int(p[0]), int(p[1]))
                                       for p in waypoints]
        self.speed = int(speed)
        self.total_distance = float(_path_length(self.start, self.waypoints))
        if self.speed > 0:
            self.estimated_ticks = int(math.ceil(self.total_distance /
                                                 self.speed))
        else:
            self.estimated_ticks = 0


class PathPlanner:
    """Computes routes and moves riders along them on the grid."""

    def __init__(self, grid: Grid = None) -> None:
        """Create a planner.

        Args:
            grid: Grid used for street checks. A default 8x8 grid is created
                if none is given.
        """
        self.grid = grid if grid is not None else Grid()

    def plan(self, start: Point, goal: Point, speed: int) -> Route:
        """Plan a route from start to goal.

        Special cases:
            * Start equals goal: empty route.
            * Goal directly above/below or beside the start (same x or y):
              straight-line movement with a single waypoint.
            * Otherwise: an L-shaped route with two candidate corners. The
              corner whose legs run along streets for the greatest length is
              chosen; ties keep the horizontal-first corner.

        Args:
            start: Current position of the rider.
            goal: Target point (restaurant or customer cell centre).
            speed: Rider speed in units per tick.

        Returns:
            A Route object.
        """
        sx, sy = start
        gx, gy = goal
        if (sx, sy) == (gx, gy):
            return Route(start, [], speed)
        if sx == gx or sy == gy:
            # Directly above/below or beside the target: straight movement.
            return Route(start, [(gx, gy)], speed)
        corner_a = (gx, sy)  # horizontal first
        corner_b = (sx, gy)  # vertical first
        cost_a = (self.off_street_length(start, corner_a) +
                  self.off_street_length(corner_a, goal))
        cost_b = (self.off_street_length(start, corner_b) +
                  self.off_street_length(corner_b, goal))
        corner = corner_a if cost_a <= cost_b else corner_b
        logger.debug("Planned route %s -> %s via %s", start, goal, corner)
        return Route(start, [corner, (gx, gy)], speed)

    def off_street_length(self, a: Point, b: Point) -> int:
        """Return the length of an axis-aligned segment that is off-street.

        A horizontal segment is on a street when its y is a grid line; a
        vertical segment is on a street when its x is a grid line.
        """
        ax, ay = a
        bx, by = b
        if ay == by:
            return 0 if ay % CELL_SIZE == 0 else abs(bx - ax)
        if ax == bx:
            return 0 if ax % CELL_SIZE == 0 else abs(by - ay)
        return abs(bx - ax) + abs(by - ay)

    @staticmethod
    def step(x: int, y: int, waypoints: Sequence[Point],
             speed: int) -> Tuple[int, int, List[Point], int]:
        """Move up to ``speed`` units along a list of waypoints.

        Movement inside each segment is horizontal first, then vertical.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            waypoints: Remaining waypoints (not modified).
            speed: Maximum distance to move this tick.

        Returns:
            Tuple ``(new_x, new_y, remaining_waypoints, distance_moved)``.
        """
        remaining = [(int(p[0]), int(p[1])) for p in waypoints]
        budget = int(speed)
        moved = 0
        while budget > 0 and remaining:
            tx, ty = remaining[0]
            if (x, y) == (tx, ty):
                remaining.pop(0)
                continue
            dx = tx - x
            dy = ty - y
            mx = min(budget, abs(dx))
            x += mx * _sign(dx)
            budget -= mx
            moved += mx
            my = min(budget, abs(dy))
            y += my * _sign(dy)
            budget -= my
            moved += my
            if (x, y) == (tx, ty):
                remaining.pop(0)
        return x, y, remaining, moved

"""Rider model: state machine, movement and order execution.

State machine:
    IDLE (0) -> PICKING_UP (1) when the first assigned order is accepted.
    PICKING_UP (1) -> DELIVERING (2) on arrival at the restaurant.
    DELIVERING (2) -> IDLE (0) on delivery to the customer.

Log entries are appended to ``Rider.log`` as dictionaries with the keys
``event``, ``order_id``, ``tick``, ``rider_id``, ``x``, ``y`` and ``state``.
Events: "spawn", "accepted_order", "arrived_at_merchant",
"delivered_to_customer".
"""

import logging
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from grid import CELL_SIZE
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {
    IDLE: "idle",
    PICKING_UP: "picking up",
    DELIVERING: "delivering",
}

DEFAULT_RIDER_SPEED = 50


class Rider:
    """A delivery rider that moves on the grid and executes orders.

    Attributes:
        rider_id: Unique integer identifier.
        x, y: Current integer coordinates.
        speed: Movement speed in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Assigned orders; ``orders[0]`` is the one being served.
        special_case: True while the rider is off-street (inside a cell).
        log: Chronological list of status-change entries.
    """

    def __init__(self, rider_id: int, x: int = 0, y: int = 0,
                 speed: int = DEFAULT_RIDER_SPEED,
                 planner: Optional[PathPlanner] = None) -> None:
        """Create a rider and record its spawn event.

        Args:
            rider_id: Unique identifier.
            x: Initial x coordinate.
            y: Initial y coordinate.
            speed: Movement speed in units per tick.
            planner: Shared path planner; a new one is created if omitted.
        """
        self.rider_id = int(rider_id)
        self.x = int(x)
        self.y = int(y)
        self.speed = int(speed)
        self.state = IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.planner = planner if planner is not None else PathPlanner()
        self.log: List[Dict[str, Any]] = []
        self._waypoints: List[Tuple[int, int]] = []
        self._record("spawn", None)

    @property
    def position(self) -> Tuple[int, int]:
        """Current position as an (x, y) tuple."""
        return (self.x, self.y)

    def is_idle(self) -> bool:
        """Return True when the rider is available for new work."""
        return self.state == IDLE

    def _now(self) -> int:
        """Return the current simulated tick from the global store."""
        tick = globalvar.get_value("current_tick", 0)
        return tick if isinstance(tick, int) else 0

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append a status-change entry to the rider log."""
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": self._now(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %d: %s (order %s)", self.rider_id, event,
                     order_id)

    def _distance_to(self, point: Tuple[int, int]) -> int:
        """Return the Manhattan distance from the rider to a point."""
        return abs(point[0] - self.x) + abs(point[1] - self.y)

    def _reset_to_idle(self) -> None:
        """Set the rider to idle and drop any planned waypoints."""
        self.state = IDLE
        self._waypoints = []
        self.special_case = False

    def _start_pickup(self) -> None:
        """Accept the first order and plan the route to the restaurant.

        With several assigned orders, the unpicked orders are sorted by
        distance from the rider so that the nearest restaurant is visited
        first (multi-destination ordering).
        """
        if len(self.orders) > 1:
            self.orders.sort(key=lambda o: self._distance_to(o.restaurant_pos))
        order = self.orders[0]
        self.state = PICKING_UP
        self._record("accepted_order", order.id)
        route = self.planner.plan(self.position, order.restaurant_pos,
                                  self.speed)
        self._waypoints = list(route.waypoints)
        order.record_route(self.position, route.waypoints)

    def _advance(self) -> None:
        """Move up to ``speed`` units along the planned waypoints."""
        x, y, remaining, moved = self.planner.step(
            self.x, self.y, self._waypoints, self.speed)
        self.x, self.y = x, y
        self._waypoints = remaining
        if self.orders:
            self.orders[0].distance += moved
        self.special_case = (x % CELL_SIZE != 0 and y % CELL_SIZE != 0)

    def _handle_arrival(self) -> Optional[Any]:
        """Process arrival at the current target.

        Returns:
            The completed Order if a delivery finished, otherwise None.
        """
        order = self.orders[0]
        if self.state == PICKING_UP:
            order.mark_picked_up()
            self.state = DELIVERING
            self._record("arrived_at_merchant", order.id)
            route = self.planner.plan(self.position, order.customer_pos,
                                      self.speed)
            self._waypoints = list(route.waypoints)
            order.record_route(self.position, route.waypoints)
            return None
        if self.state == DELIVERING:
            order.complete(self._now())
            self.orders.pop(0)
            self._record("delivered_to_customer", order.id)
            self._reset_to_idle()
            logger.info("Rider %d delivered order %d", self.rider_id,
                        order.id)
            return order
        return None

    def move(self) -> List[Any]:
        """Advance the rider by one tick.

        Performs state transitions, movement and arrival handling.

        Returns:
            List of orders completed during this tick.
        """
        completed: List[Any] = []
        if not self.orders:
            if self.state != IDLE:
                logger.warning(
                    "Rider %d in state %d without orders; resetting to idle",
                    self.rider_id, self.state)
                self._reset_to_idle()
            return completed
        if self.state == IDLE:
            self._start_pickup()
        if self._waypoints:
            self._advance()
        while self.orders and self.state != IDLE and not self._waypoints:
            finished = self._handle_arrival()
            if finished is not None:
                completed.append(finished)
        return completed

"""Order data model.

An order stores the restaurant and customer locations converted to grid
cell centres, its lifecycle status and its delivery statistics.
"""

import itertools
from typing import Any, Dict, List, Optional, Sequence, Tuple

from grid import CELL_SIZE, GRID_DIMENSION, to_cell_center, validate_coordinate

_ID_SEQUENCE = itertools.count(1)


def _parse_position(position: Sequence[int], label: str) -> Tuple[int, int]:
    """Validate an (x, y) pair and return it as integers.

    Args:
        position: Pair of raw coordinates.
        label: Location name used in error messages.

    Returns:
        Tuple of validated (x, y).

    Raises:
        ValueError: If the pair is malformed or out of range.
    """
    try:
        x, y = position
    except (TypeError, ValueError):
        raise ValueError(f"{label} position must be a pair (x, y)") from None
    return (validate_coordinate(x, f"{label} x"),
            validate_coordinate(y, f"{label} y"))


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order identifier.
        restaurant_x, restaurant_y: Restaurant cell centre.
        customer_x, customer_y: Customer cell centre.
        start_time: Tick at which the order was created.
        completion_time: Tick of delivery, or None.
        distance: Total distance travelled for this order (units).
        time_taken: Ticks from creation to delivery, or None.
        status: "pending", "assigned", "picked_up" or "completed".
        rider_id: Identifier of the assigned rider, or None.
        route_points: Planned route points (restaurant leg then customer
            leg), used for the map display.
    """

    PENDING = "pending"
    ASSIGNED = "assigned"
    PICKED_UP = "picked_up"
    COMPLETED = "completed"

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int],
                 start_time: int = 0) -> None:
        """Create an order from raw restaurant and customer coordinates.

        Args:
            res_pos: Restaurant (x, y) in the range 0-800.
            cus_pos: Customer (x, y) in the range 0-800.
            start_time: Tick at which the order is created.

        Raises:
            ValueError: If a coordinate is invalid.
        """
        rx, ry = _parse_position(res_pos, "restaurant")
        cx, cy = _parse_position(cus_pos, "customer")
        self.id = next(_ID_SEQUENCE)
        self.restaurant_x = to_cell_center(rx)
        self.restaurant_y = to_cell_center(ry)
        self.customer_x = to_cell_center(cx)
        self.customer_y = to_cell_center(cy)
        self.start_time = int(start_time)
        self.completion_time: Optional[int] = None
        self.distance = 0.0
        self.time_taken: Optional[int] = None
        self.status = Order.PENDING
        self.rider_id: Optional[int] = None
        self.route_points: List[Tuple[int, int]] = []

    @property
    def restaurant_pos(self) -> Tuple[int, int]:
        """Restaurant cell centre as (x, y)."""
        return (self.restaurant_x, self.restaurant_y)

    @property
    def customer_pos(self) -> Tuple[int, int]:
        """Customer cell centre as (x, y)."""
        return (self.customer_x, self.customer_y)

    @property
    def efficiency(self) -> float:
        """Distance per tick for a completed order (0.0 if not completed)."""
        if not self.time_taken:
            return 0.0
        return self.distance / self.time_taken

    def restaurant_cell(self) -> Tuple[int, int]:
        """Return the restaurant grid cell indices (column, row)."""
        return (min(self.restaurant_x // CELL_SIZE, GRID_DIMENSION - 1),
                min(self.restaurant_y // CELL_SIZE, GRID_DIMENSION - 1))

    def customer_cell(self) -> Tuple[int, int]:
        """Return the customer grid cell indices (column, row)."""
        return (min(self.customer_x // CELL_SIZE, GRID_DIMENSION - 1),
                min(self.customer_y // CELL_SIZE, GRID_DIMENSION - 1))

    def assign_to(self, rider_id: int) -> None:
        """Record the assigned rider and mark the order as assigned.

        Args:
            rider_id: Identifier of the rider taking the order.
        """
        self.rider_id = int(rider_id)
        self.status = Order.ASSIGNED

    def mark_picked_up(self) -> None:
        """Mark the food as collected at the restaurant."""
        self.status = Order.PICKED_UP

    def complete(self, tick: int) -> None:
        """Mark the order as delivered and record efficiency data.

        Args:
            tick: Tick at which the delivery happened.
        """
        self.completion_time = int(tick)
        self.time_taken = self.completion_time - self.start_time
        self.status = Order.COMPLETED

    def record_route(self, start: Tuple[int, int],
                     waypoints: Sequence[Tuple[int, int]]) -> None:
        """Append planned route points for the map display.

        Args:
            start: Point the leg starts from. Only stored for the first leg.
            waypoints: Waypoints of the leg.
        """
        if not self.route_points:
            self.route_points.append((int(start[0]), int(start[1])))
        self.route_points.extend((int(p[0]), int(p[1])) for p in waypoints)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": round(self.distance, 2),
            "time_taken": self.time_taken,
            "efficiency": round(self.efficiency, 3),
            "status": self.status,
            "rider_id": self.rider_id,
        }

"""Order management and the closest-rider assignment algorithm.

Assignment picks the rider with the smallest score, where the score is the
Manhattan distance from the rider to the restaurant plus a load penalty per
order already held by that rider. This keeps the distance-first strategy
while balancing load between riders.
"""

import logging
from typing import Any, List, Optional, Sequence

import globalvar
from order_model import Order

logger = logging.getLogger(__name__)

DEFAULT_LOAD_PENALTY = 100


def find_best_rider_for_order(order: Order,
                              load_penalty: int = DEFAULT_LOAD_PENALTY
                              ) -> Optional[Any]:
    """Find the best rider for an order.

    Args:
        order: The order to assign.
        load_penalty: Score added per order already held by a rider.

    Returns:
        The selected Rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    best = None
    best_key = None
    rx, ry = order.restaurant_pos
    for rider in riders:
        distance = abs(rider.x - rx) + abs(rider.y - ry)
        key = (distance + load_penalty * len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates, assigns and tracks orders.

    Attributes:
        order_counter: Number of orders created so far.
        orders: History of all orders.
        pending: Orders waiting for a rider.
        load_penalty: Load-balancing penalty used by the assignment.
    """

    def __init__(self, load_penalty: int = DEFAULT_LOAD_PENALTY) -> None:
        """Create an empty order manager.

        Args:
            load_penalty: Load-balancing penalty for rider selection.
        """
        self.order_counter = 0
        self.orders: List[Order] = []
        self.pending: List[Order] = []
        self.load_penalty = int(load_penalty)

    def create_order(self, res_pos: Sequence[int], cus_pos: Sequence[int],
                     tick: int = 0) -> Order:
        """Create and register a new order.

        Args:
            res_pos: Restaurant (x, y), 0-800.
            cus_pos: Customer (x, y), 0-800.
            tick: Current simulated tick.

        Returns:
            The created Order.

        Raises:
            ValueError: If a coordinate is invalid.
        """
        order = Order(res_pos, cus_pos, start_time=tick)
        self.order_counter += 1
        self.orders.append(order)
        logger.info("Created order %d at tick %d", order.id, tick)
        return order

    def _bind(self, order: Order, rider: Any) -> None:
        """Attach an order to a rider."""
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        if order in self.pending:
            self.pending.remove(order)
        logger.info("Order %d assigned to rider %d", order.id, rider.rider_id)

    def assign_order(self, order: Order) -> Optional[Any]:
        """Assign an order to the best available rider.

        If no rider exists, the order is queued as pending.

        Args:
            order: A pending order.

        Returns:
            The assigned Rider, or None if the order is pending.

        Raises:
            RuntimeError: If the order has already been assigned.
        """
        if order.status != Order.PENDING:
            raise RuntimeError(
                f"Order {order.id} cannot be assigned from status "
                f"'{order.status}'")
        rider = find_best_rider_for_order(order, self.load_penalty)
        if rider is None:
            if order not in self.pending:
                self.pending.append(order)
            logger.warning("No rider available; order %d is pending", order.id)
            return None
        self._bind(order, rider)
        return rider

    def try_assign_pending(self) -> int:
        """Try to assign every pending order.

        Returns:
            Number of orders assigned in this call.
        """
        assigned = 0
        for order in list(self.pending):
            rider = find_best_rider_for_order(order, self.load_penalty)
            if rider is None:
                break
            self._bind(order, rider)
            assigned += 1
        return assigned

    def active_orders(self) -> List[Order]:
        """Return orders that are not yet completed."""
        return [o for o in self.orders if o.status != Order.COMPLETED]

    def get_order(self, order_id: int) -> Optional[Order]:
        """Return the order with the given id, or None."""
        for order in self.orders:
            if order.id == order_id:
                return order
        return None

"""System manager: funds, rider recruitment, simulation clock and persistence.

Every call to ``SystemManager.tick()`` is one unit of simulated time. Within a
tick the system first assigns pending orders, then moves every rider, credits
the delivery fee for completed orders and finally checks the recruitment
trigger.
"""

import json
import logging
from typing import Any, Dict, Optional, Tuple

import globalvar
from config import DEFAULT_CONFIG
from order import OrderManager
from order_model import Order
from path_planner import PathPlanner
from rider import Rider

logger = logging.getLogger(__name__)

SPAWN_POINT: Tuple[int, int] = (400, 400)  # a street intersection at map centre


class SystemManager:
    """Core manager for funds, riders, orders and the simulation clock.

    Attributes:
        config: Active system parameters.
        funds: System fund balance (integer, may be negative).
        rider_id_counter: Last rider id issued (ids start at 1).
        is_running: Main-loop control flag.
        current_tick: Simulated time.
        planner: Shared path planner.
        order_manager: The OrderManager owned by the system.
    """

    def __init__(self, config: Optional[Dict[str, Any]] = None) -> None:
        """Create the system manager with the given parameters.

        Args:
            config: Parameter overrides; missing keys use defaults.
        """
        self.config: Dict[str, Any] = dict(DEFAULT_CONFIG)
        if config:
            self.config.update(config)
        self.funds = int(self.config["initial_funds"])
        self.rider_id_counter = 0
        self.is_running = True
        self.current_tick = 0
        self.planner = PathPlanner()
        self.order_manager = OrderManager(
            load_penalty=int(self.config["load_penalty"]))
        globalvar.set_value("riders", [])
        globalvar.set_value("current_tick", 0)

    @property
    def riders(self) -> list:
        """The dynamic list of riders stored in the global variables."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    def create_rider(self) -> Rider:
        """Create a new idle rider at the spawn point and register it.

        Returns:
            The new Rider.
        """
        self.rider_id_counter += 1
        rider = Rider(self.rider_id_counter, x=SPAWN_POINT[0],
                      y=SPAWN_POINT[1],
                      speed=int(self.config["default_rider_speed"]),
                      planner=self.planner)
        self.riders.append(rider)
        logger.info("Recruited rider %d", rider.rider_id)
        return rider

    def recruit_rider_if_funded(self) -> Optional[Rider]:
        """Recruit a rider when funds reach the recruitment threshold.

        The recruitment cost is deducted from the funds.

        Returns:
            The new Rider, or None if funds are below the threshold.
        """
        if self.funds >= int(self.config["recruit_threshold"]):
            self.funds -= int(self.config["recruit_cost"])
            return self.create_rider()
        return None

    def submit_order(self, res_pos, cus_pos
                     ) -> Tuple[Order, Optional[Rider], Optional[Rider]]:
        """Create an order, assign it and run the simulation until delivery.

        Args:
            res_pos: Restaurant (x, y), 0-800.
            cus_pos: Customer (x, y), 0-800.

        Returns:
            Tuple ``(order, new_rider, assigned_rider)``. ``new_rider`` is
            the rider recruited for this order (or None). ``assigned_rider``
            is the rider chosen for the order (or None if still pending).

        Raises:
            ValueError: If a coordinate is invalid.
        """
        new_rider = self.recruit_rider_if_funded()
        order = self.order_manager.create_order(res_pos, cus_pos,
                                                self.current_tick)
        assigned = self.order_manager.assign_order(order)
        self.run_until_completed(order)
        return order, new_rider, assigned

    def tick(self) -> None:
        """Advance the simulation by one unit of time.

        Raises:
            RuntimeError: If the system has been shut down.
        """
        if not self.is_running:
            raise RuntimeError("System is not running")
        self.current_tick += 1
        globalvar.set_value("current_tick", self.current_tick)
        self.order_manager.try_assign_pending()
        for rider in list(self.riders):
            for order in rider.move():
                self._on_order_completed(order)
        self.recruit_rider_if_funded()

    def _on_order_completed(self, order: Order) -> None:
        """Release resources and credit the delivery fee for an order."""
        self.funds += int(self.config["delivery_fee"])
        logger.info("Order %d completed; funds now %d", order.id, self.funds)

    def run_ticks(self, count: int) -> None:
        """Advance the simulation by ``count`` ticks."""
        for _ in range(int(count)):
            self.tick()

    def run_until_completed(self, order: Order) -> bool:
        """Run ticks until the order is delivered or the limit is reached.

        Args:
            order: The order to wait for.

        Returns:
            True if the order was completed.
        """
        limit = int(self.config["max_ticks_per_order"])
        ticks = 0
        while order.status != Order.COMPLETED and ticks < limit:
            self.tick()
            ticks += 1
        return order.status == Order.COMPLETED

    def rider_by_id(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with the given id, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the system state."""
        return {
            "funds": self.funds,
            "rider_id_counter": self.rider_id_counter,
            "current_tick": self.current_tick,
            "config": self.config,
            "riders": [
                {
                    "rider_id": r.rider_id,
                    "x": r.x,
                    "y": r.y,
                    "speed": r.speed,
                    "state": r.state,
                    "special_case": r.special_case,
                    "order_ids": [o.id for o in r.orders],
                    "log": r.log,
                }
                for r in self.riders
            ],
            "orders": [o.to_dict() for o in self.order_manager.orders],
            "pending_order_ids": [o.id for o in self.order_manager.pending],
        }

    def save(self, path: Optional[str] = None) -> bool:
        """Write the system state to a JSON file.

        Args:
            path: Destination file; defaults to the configured save file.

        Returns:
            True on success, False if the file could not be written.
        """
        target = path or str(self.config["save_file"])
        try:
            with open(target, "w", encoding="utf-8") as handle:
                json.dump(self.to_dict(), handle, indent=2)
        except (OSError, TypeError, ValueError) as exc:
            logger.error("Failed to save state to %s: %s", target, exc)
            return False
        logger.info("Saved state to %s", target)
        return True

    def shutdown(self, path: Optional[str] = None) -> None:
        """Stop the main loop and persist the state."""
        self.save(path)
        self.is_running = False


def init_system(config: Optional[Dict[str, Any]] = None) -> SystemManager:
    """Initialise global variables and create the system manager.

    Args:
        config: Optional parameter overrides.

    Returns:
        The initialised SystemManager, also stored as global "system".
    """
    globalvar._init()
    manager = SystemManager(config)
    globalvar.set_value("system", manager)
    logger.info("System initialised")
    return manager

"""Status display and ASCII map rendering.

The map is drawn on a 17x17 character canvas. Each character represents
50 units, so cell centres (50 + 100k) fall on odd indices and street lines
(multiples of 100) fall on even indices. Symbols:
    R rider, S restaurant, C customer, . empty cell centre,
    # street, * planned route.
"""

from typing import Iterable, List, Optional, Sequence, Tuple

from grid import CELL_SIZE, GRID_DIMENSION
from order_model import Order
from rider import STATE_NAMES

SYMBOL_RIDER = "R"
SYMBOL_RESTAURANT = "S"
SYMBOL_CUSTOMER = "C"
SYMBOL_EMPTY = "."
SYMBOL_STREET = "#"
SYMBOL_ROUTE = "*"

STEP = CELL_SIZE // 2
LEGEND = ("Legend: R rider, S restaurant, C customer, . cell centre, "
          "# street, * route (1 char = 50 units)")


class DisplayManager:
    """Formats status information and renders the ASCII map.

    Attributes:
        map_width: Canvas width in characters.
        map_height: Canvas height in characters.
    """

    def __init__(self) -> None:
        """Create the display manager with the 17x17 canvas size."""
        self.map_width = GRID_DIMENSION * 2 + 1
        self.map_height = self.map_width

    def _index(self, value: int) -> int:
        """Convert a coordinate to a canvas index (clamped)."""
        index = (int(value) + STEP // 2) // STEP
        return min(max(index, 0), self.map_width - 1)

    def _point(self, point: Tuple[int, int]) -> Tuple[int, int]:
        """Convert an (x, y) point to a canvas (column, row) pair."""
        return self._index(point[0]), self._index(point[1])

    def render_map(self, riders: Iterable = (),
                   orders: Sequence[Order] = (),
                   route: Optional[Sequence[Tuple[int, int]]] = None) -> str:
        """Render the ASCII map.

        Args:
            riders: Riders to mark with 'R'.
            orders: Orders whose restaurant ('S') and customer ('C') are
                marked.
            route: Route points drawn as connected line segments with '*'.

        Returns:
            The multi-line map string.
        """
        width, height = self.map_width, self.map_height
        canvas: List[List[str]] = []
        for row in range(height):
            line = []
            for col in range(width):
                if col % 2 == 1 and row % 2 == 1:
                    line.append(SYMBOL_EMPTY)
                else:
                    line.append(SYMBOL_STREET)
            canvas.append(line)

        if route:
            points = [self._point(p) for p in route]
            for (c0, r0), (c1, r1) in zip(points, points[1:]):
                for col in range(min(c0, c1), max(c0, c1) + 1):
                    self._mark(canvas, col, r0, SYMBOL_ROUTE)
                for row in range(min(r0, r1), max(r0, r1) + 1):
                    self._mark(canvas, c1, row, SYMBOL_ROUTE)

        for order in orders:
            if order.status == Order.COMPLETED:
                continue
            col, row = self._point(order.restaurant_pos)
            canvas[row][col] = SYMBOL_RESTAURANT
            col, row = self._point(order.customer_pos)
            canvas[row][col] = SYMBOL_CUSTOMER

        for rider in riders:
            col, row = self._point(rider.position)
            canvas[row][col] = SYMBOL_RIDER

        return "\n".join("".join(line) for line in canvas)

    @staticmethod
    def _mark(canvas: List[List[str]], col: int, row: int,
              symbol: str) -> None:
        """Draw a route symbol without overwriting entity symbols."""
        if canvas[row][col] in (SYMBOL_EMPTY, SYMBOL_STREET):
            canvas[row][col] = symbol

    @staticmethod
    def legend() -> str:
        """Return the map legend text."""
        return LEGEND

    def format_status(self, system, riders: Sequence) -> str:
        """Return the system snapshot: funds, riders and their orders.

        Args:
            system: The SystemManager.
            riders: Riders to list.

        Returns:
            Multi-line status text.
        """
        pending = len(system.order_manager.pending)
        lines = [
            f"Simulated tick : {system.current_tick}",
            f"System funds   : {system.funds}",
            f"Riders         : {len(riders)}",
            f"Pending orders : {pending}",
        ]
        for rider in riders:
            state = STATE_NAMES.get(rider.state, "unknown")
            order_text = ", ".join(
                f"#{o.id}({o.status})" for o in rider.orders) or "-"
            lines.append(
                f"  Rider #{rider.rider_id} [{state}] at "
                f"({rider.x}, {rider.y}) speed {rider.speed} "
                f"orders: {order_text}")
        return "\n".join(lines)

    def format_orders(self, orders: Sequence[Order]) -> str:
        """Return a table of order details.

        Args:
            orders: Orders to list.

        Returns:
            Multi-line table text.
        """
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} "
                  f"{'Customer':>12} {'Dist':>8} {'Ticks':>6} "

        return "\n".join(lines)

    def format_riders(self, riders: Sequence) -> str:
        """Return the status-change history of every rider.

    def format_start_result(self, order: Order, rider, new_rider) -> str:

        if rider is not None:

        else:
            lines.append("Assigned rider: none available (order pending)")
        if order.status == Order.COMPLETED:
            lines.append(
                f"Completed at tick {order.completion_time}; "
                f"time taken {order.time_taken} ticks; "
                f"distance {order.distance:.1f} units")
        else:
            lines.append(f"Status: {order.status}")
        return "\n".join(lines)

"""Command parsing, shortcut keys, command history and input validation."""

import logging
from typing import Callable, List, Tuple

from grid import MAX_COORDINATE, MIN_COORDINATE

logger = logging.getLogger(__name__)

COMMAND_ALIASES = {
    "start": "start", "s": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "history": "history", "hist": "history",
    "save": "save",
    "help": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}

COORDINATE_LABELS = ("restaurant_x", "restaurant_y",
                     "customer_x", "customer_y")

PROMPT_TEXT = ("Enter restaurant_x restaurant_y customer_x customer_y "
               "(0-800, or 'cancel'): ")


class InputManager:
    """Reads, parses and validates user commands.

    Attributes:
        command_history: Raw command lines entered so far.
        min_coordinate: Smallest accepted coordinate.
        max_coordinate: Largest accepted coordinate.
    """

    def __init__(self, min_coordinate: int = MIN_COORDINATE,
                 max_coordinate: int = MAX_COORDINATE,
                 input_func: Callable[[str], str] = input,
                 output_func: Callable[[str], None] = print) -> None:
        """Create the input manager.

        Args:
            min_coordinate: Smallest accepted coordinate value.
            max_coordinate: Largest accepted coordinate value.
            input_func: Function used to read a line (injectable).
            output_func: Function used to print messages (injectable).
        """
        self.command_history: List[str] = []
        self.min_coordinate = min_coordinate
        self.max_coordinate = max_coordinate
        self.input_func = input_func
        self.output_func = output_func

    def read_command(self, prompt: str = "dispatch> ") -> str:
        """Read one command line. ``!!`` repeats the previous command.

        Args:
            prompt: Prompt shown to the user.

        Returns:
            The stripped command text (may be empty).

        Raises:
            ValueError: If ``!!`` is used with an empty history.
        """
        raw = self.input_func(prompt).strip()
        if raw == "!!":
            if not self.command_history:
                raise ValueError("No previous command to repeat.")
            raw = self.command_history[-1]
            self.output_func(f"Repeating: {raw}")
        if raw:
            self.command_history.append(raw)
        return raw

    def parse_command(self, line: str) -> Tuple[str, List[str]]:
        """Split a command line into a canonical command name and arguments.

        Args:
            line: The raw command line.

        Returns:
            Tuple ``(command_name, arguments)``.

        Raises:
            ValueError: If the command is empty or unknown.
        """
        tokens = line.split()
        if not tokens:
            raise ValueError("Empty command.")
        name = COMMAND_ALIASES.get(tokens[0].lower())
        if name is None:
            raise ValueError(
                f"Unknown command '{tokens[0]}'. Type 'help' for the list.")
        return name, tokens[1:]

    def validate_coordinate(self, text, label: str = "coordinate") -> int:
        """Validate that text is an integer within the coordinate range.

        Args:
            text: Text typed by the user.
            label: Field name for error messages.

        Returns:
            The integer value.

        Raises:
            ValueError: If the text is not an integer or is out of range.
        """
        try:
            value = int(str(text).strip())
        except ValueError:
            raise ValueError(
                f"{label} must be an integer, got '{text}'") from None
        if not self.min_coordinate <= value <= self.max_coordinate:
            raise ValueError(
                f"{label} {value} is out of range "
                f"{self.min_coordinate}-{self.max_coordinate}")
        return value

    def parse_coordinates(self, tokens: List[str]) -> List[int]:
        """Return the four validated order coordinates.

        If no tokens are given the user is prompted. Values may be separated
        by spaces or commas.

        Args:
            tokens: Arguments following the start command.

        Returns:
            List ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            ValueError: If the input is cancelled, malformed or out of range.
        """
        if tokens:
            text = " ".join(tokens)
        else:
            text = self.input_func(PROMPT_TEXT).strip()
            if text.lower() == "cancel":
                raise ValueError("Order cancelled.")
        parts = text.replace(",", " ").split()
        if len(parts) != 4:
            raise ValueError(
                f"Expected 4 integer coordinates, got {len(parts)}. "
                "Format: restaurant_x restaurant_y customer_x customer_y")
        return [self.validate_coordinate(part, label)
                for part, label in zip(parts, COORDINATE_LABELS)]

    def help_text(self) -> str:
        """Return the command help text."""
        return "\n".join([
            "Commands (shortcut in brackets):",
            "  start [rx ry cx cy] [s]  Place an order (restaurant, customer)",
            "  status [st]              Funds, riders and their orders",
            "  orders [o]               Details of all orders",
            "  riders [r]               Status-change log of all riders",
            "  map [m]                  Show the ASCII map",
            "  tick [n] [t]             Advance simulated time n ticks",
            "  history [hist]           Show command history",
            "  save [path]              Save state to a JSON file",
            "  help [?]                 Show this help",
            "  quit [exit, q]           Save and exit",
            "  !!                       Repeat the previous command",

"""Command-line entry point for the Food Delivery Order Dispatch System.

Run with ``python src/main.py`` (or ``python -m src.main`` from the project
root). The main loop reads commands; simulated time advances through the
``start`` and ``tick`` commands, one tick per unit of time.
"""

import logging
import os
import sys

_SRC_DIR = os.path.dirname(os.path.abspath(__file__))
if _SRC_DIR not in sys.path:
    sys.path.insert(0, _SRC_DIR)

from config import load_config  # noqa: E402
from display import DisplayManager  # noqa: E402
from grid import MAX_COORDINATE, MIN_COORDINATE  # noqa: E402
from input_manager import InputManager  # noqa: E402
from order_model import Order  # noqa: E402
from system import init_system  # noqa: E402

logger = logging.getLogger("dispatch")

BANNER = ("Food Delivery Order Dispatch System\n"
          "Type 'help' for commands, 'quit' to exit.")


def setup_logging(config):
    """Configure file logging from the configuration.

    Falls back to warnings on stderr if the log file cannot be opened.

    Args:
        config: System configuration dictionary.
    """
    level = getattr(logging, str(config.get("log_level", "INFO")).upper(),
                    logging.INFO)
    fmt = "%(asctime)s %(levelname)s %(name)s: %(message)s"
    try:
        logging.basicConfig(filename=config["log_file"], level=level,
                            format=fmt)
    except (OSError, ValueError):
        logging.basicConfig(level=logging.WARNING, format=fmt)


class Application:
    """Interactive command loop wiring the managers together."""

    def __init__(self, config, input_func=input, output_func=print):
        """Initialise the system and the interface components.

        Args:
            config: System configuration dictionary.
            input_func: Function used to read input (injectable).
            output_func: Function used to print output (injectable).
        """
        self.config = config
        self.out = output_func
        self.system = init_system(config)
        self.display = DisplayManager()
        self.cli = InputManager(MIN_COORDINATE, MAX_COORDINATE,
                                input_func=input_func,
                                output_func=output_func)
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "history": self.cmd_history,
            "save": self.cmd_save,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def run(self):
        """Run the main loop until the user quits or input ends."""
        self.out(BANNER)
        while self.system.is_running:
            try:
                line = self.cli.read_command()
            except (EOFError, KeyboardInterrupt):
                self.out("")
                self.system.shutdown(str(self.config["save_file"]))
                break
            except ValueError as exc:
                self.out(f"Error: {exc}")
                continue
            if not line:
                continue
            try:
                name, args = self.cli.parse_command(line)
                self._handlers[name](args)
            except ValueError as exc:
                self.out(f"Error: {exc}")
            except Exception:  # keep the session alive on unexpected errors
                logger.exception("Unexpected error handling '%s'", line)
                self.out("Unexpected internal error; state preserved. "
                         "See the log file for details.")
        self.out("Goodbye.")

    def cmd_start(self, args):
        """Place an order, run it to completion and show the route."""
        coords = self.cli.parse_coordinates(args)
        restaurant = (coords[0], coords[1])
        customer = (coords[2], coords[3])
        order, new_rider, rider = self.system.submit_order(restaurant,
                                                           customer)
        self.out(self.display.format_start_result(order, rider, new_rider))
        self.out(self.display.render_map(self.system.riders, [order],
                                         order.route_points))
        self.out(self.display.legend())
        if order.status != Order.COMPLETED:
            self.out("The order is still pending. Use 'tick' to advance time "
                     "once a rider is available.")

    def cmd_status(self, args):
        """Show funds, rider count and each rider's state and orders."""
        riders = self.system.riders
        self.out(self.display.format_status(self.system, riders))
        self.out(self.display.render_map(
            riders, self.system.order_manager.active_orders()))
        self.out(self.display.legend())

    def cmd_orders(self, args):
        """Show the details of every order."""
        self.out(self.display.format_orders(
            self.system.order_manager.orders))

    def cmd_riders(self, args):
        """Show the status-change log of every rider."""
        self.out(self.display.format_riders(self.system.riders))

    def cmd_map(self, args):
        """Show the ASCII map with riders and active orders."""
        self.out(self.display.render_map(
            self.system.riders, self.system.order_manager.active_orders()))
        self.out(self.display.legend())

    def cmd_tick(self, args):
        """Advance the simulated clock by one or more ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise ValueError(
                    f"tick count must be an integer, got '{args[0]}'") from None
        if not 1 <= count <= 10000:
            raise ValueError("tick count must be between 1 and 10000")
        self.system.run_ticks(count)
        self.out(f"Advanced {count} tick(s). Current tick: "
                 f"{self.system.current_tick}. Funds: {self.system.funds}.")

    def cmd_history(self, args):
        """Print the command history."""
        history = self.cli.command_history
        if not history:
            self.out("No commands yet.")
            return
        for index, command in enumerate(history, start=1):
            self.out(f"{index:>4}  {command}")

    def cmd_save(self, args):
        """Save the system state to a JSON file."""
        path = args[0] if args else None
        if self.system.save(path):
            self.out(f"State saved to {path or self.config['save_file']}.")
        else:
            self.out("Error: could not save state; see the log file.")

    def cmd_help(self, args):
        """Print the command list."""
        self.out(self.cli.help_text())

    def cmd_quit(self, args):
        """Save the state and stop the main loop."""
        self.system.shutdown(str(self.config["save_file"]))
        self.out("Saving state and exiting.")


def main(argv=None):
    """Program entry point.

    Args:
        argv: Unused; reserved for future command-line options.

    Returns:
        Process exit code.
    """
    config = load_config()
    setup_logging(config)
    logger.info("Starting dispatch system")
    app = Application(config)
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Loading of the Beijing Subway network dataset.

The dataset is stored in ``subway_data.json`` located next to this module, so
the program works regardless of the current working directory.
"""

import json
import os

DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read or is malformed."""


def load_subway_data(path=None):
    """Load the subway dataset from a JSON file.

    Args:
        path: Path of the JSON file. Defaults to ``DATA_FILE_PATH``.

    Returns:
        dict: The JSON content with at least the keys ``stations`` and
        ``connections``.

    Raises:
        DataLoadError: If the file is missing, not valid JSON, or lacks the
        required top-level keys.
    """
    if path is None:
        path = DATA_FILE_PATH
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"data file is not valid JSON: {path} ({exc})") from exc
    except OSError as exc:
        raise DataLoadError(f"cannot read data file: {path} ({exc})") from exc

    if not isinstance(data, dict):
        raise DataLoadError("data file root must be a JSON object")
    for key in ("stations", "connections"):
        if key not in data or not isinstance(data[key], list):
            raise DataLoadError(f"data file is missing list field '{key}'")
    return data

"""Graph data structures for the Beijing Subway network.

The network is a weighted undirected graph. Stations are identified by
integer numbers that are assigned by the caller (see ``data_manager``), and
each edge stores the subway line it belongs to.
"""


class Edge:
    """An undirected weighted edge between two station numbers.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Distance between the stations in meters.
        line: Name of the subway line operating this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by path algorithms (equal to the distance)."""
        return self.distance

    def other(self, node):
        """Return the endpoint opposite to ``node``.

        Raises:
            ValueError: If ``node`` is not an endpoint of this edge.
        """
        if node == self.source:
            return self.dest
        if node == self.dest:
            return self.source
        raise ValueError(f"node {node} is not an endpoint of edge {self!r}")

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph of subway stations.

    Station numbers are assigned by the caller through ``register_station``.
    The graph keeps the mapping between station names and numbers, the lines
    serving each station, and adjacency lists of edges.
    """

    def __init__(self):
        self.station_to_number = {}
        self.station_lines = {}
        self._number_to_station = {}
        self._adjacency = {}
        self._edges = []

    # ----- station management -------------------------------------------
    def register_station(self, station_name, number):
        """Register a station name with its number (a graph node).

        Raises:
            ValueError: If the name or number is already registered.
        """
        if station_name in self.station_to_number:
            raise ValueError(f"station '{station_name}' is already registered")
        if number in self._adjacency:
            raise ValueError(f"station number {number} is already in use")
        self.station_to_number[station_name] = number
        self._number_to_station[number] = station_name
        self._adjacency[number] = []
        self.station_lines.setdefault(station_name, [])

    def set_station_lines(self, station_name, lines):
        """Set the list of lines serving a station.

        Raises:
            KeyError: If the station is not registered.
        """
        if station_name not in self.station_to_number:
            raise KeyError(f"unknown station '{station_name}'")
        self.station_lines[station_name] = list(lines)

    def has_station(self, station_name):
        """Return True if the station name is registered."""
        return station_name in self.station_to_number

    def contains_number(self, number):
        """Return True if the station number is a node of this graph."""
        return number in self._adjacency

    def get_number(self, station_name):
        """Return the station number for a station name."""
        return self.station_to_number[station_name]

    def get_name(self, number):
        """Return the station name for a station number."""
        return self._number_to_station[number]

    def station_names(self):
        """Return all station names sorted alphabetically."""
        return sorted(self.station_to_number)

    def is_transfer_station(self, station_name):
        """Return True if more than one line serves the station."""
        return len(self.station_lines.get(station_name, [])) > 1

    # ----- edge management -----------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge to the graph.

        Raises:
            KeyError: If an endpoint is not a registered station number.
            ValueError: If the edge is a self-loop.
        """
        if edge.source not in self._adjacency or edge.dest not in self._adjacency:
            raise KeyError(f"edge endpoints {edge.source}, {edge.dest} are not registered")
        if edge.source == edge.dest:
            raise ValueError("self-loop edges are not allowed")
        self._adjacency[edge.source].append(edge)
        self._adjacency[edge.dest].append(edge)
        self._edges.append(edge)

    def adjacent(self, number):
        """Return the list of edges incident to a station number."""
        return self._adjacency.get(number, [])

    def neighbors(self, number):
        """Return the station numbers adjacent to a station number."""
        return [edge.other(number) for edge in self.adjacent(number)]

    def edges(self):
        """Return a copy of the list of all edges."""
        return list(self._edges)

    def find_edges(self, number_a, number_b):
        """Return all edges connecting two station numbers (any order)."""
        return [
            edge
            for edge in self.adjacent(number_a)
            if {edge.source, edge.dest} == {number_a, number_b}
        ]

    @property
    def node_count(self):
        """Number of stations (nodes) in the graph."""
        return len(self._adjacency)

    @property
    def edge_count(self):
        """Number of connections (edges) in the graph."""
        return len(self._edges)

"""Path-finding algorithms for the subway network.

Provides a binary-heap priority queue (``MinPQ``), a configuration object
(``PathfindingConfig``) and a Dijkstra-based shortest path computation
(``ShortestPath``). The search runs over states ``(station, arriving line)``
so that line changes (transfers) can be counted, limited and penalised.
"""

from models import Graph  # noqa: F401  (documented dependency of ShortestPath)

TRANSFER_PENALTY_M = 500.0      # extra cost per transfer in comprehensive mode
PREFERENCE_PENALTY_M = 1000.0   # extra cost per edge on a non-preferred line

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")
DEFAULT_MODE = "shortest_distance"


class PathNotFoundError(Exception):
    """Raised when no path satisfies the query constraints."""


class PathfindingConfig:
    """Query configuration for path computation.

    Attributes:
        mode: One of ``shortest_distance``, ``least_transfer``, ``comprehensive``.
        max_transfers: Maximum allowed number of transfers, or None for no limit.
        preferred_lines: List of line names that are preferred.
    """

    def __init__(self, mode=DEFAULT_MODE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"unsupported mode '{mode}', expected one of {VALID_MODES}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer or None")
            if max_transfers < 0:
                raise ValueError("max_transfers must not be negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Min-priority queue implemented as a binary heap.

    Items are stored with comparable keys; ``del_min`` returns the entry with
    the smallest key. Insertion and removal run in O(log n).
    """

    def __init__(self):
        self._heap = []  # list of (key, item) tuples

    @property
    def heap_size(self):
        """Number of entries currently in the queue."""
        return len(self._heap)

    def is_empty(self):
        """Return True if the queue has no entries."""
        return not self._heap

    def min_key(self):
        """Return the smallest key without removing the entry."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._heap[0][0]

    def insert(self, key, item):
        """Insert an item with the given key."""
        self._heap.append((key, item))
        self._sift_up(len(self._heap) - 1)

    def del_min(self):
        """Remove and return the ``(key, item)`` pair with the smallest key."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        top = self._heap[0]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._sift_down(0)
        return top

    def _sift_up(self, index):
        heap = self._heap
        entry = heap[index]
        while index > 0:
            parent = (index - 1) // 2
            if entry[0] < heap[parent][0]:
                heap[index] = heap[parent]
                index = parent
            else:
                break
        heap[index] = entry

    def _sift_down(self, index):
        heap = self._heap
        size = len(heap)
        entry = heap[index]
        while True:
            child = 2 * index + 1
            if child >= size:
                break
            if child + 1 < size and heap[child + 1][0] < heap[child][0]:
                child += 1
            if heap[child][0] < entry[0]:
                heap[index] = heap[child]
                index = child
            else:
                break
        heap[index] = entry


class ShortestPath:
    """Dijkstra-based path computation from a single source station.

    Attributes:
        dist_to: Mapping from station number to the actual travelled distance
            (meters) of the best path found to that station.
    """

    def __init__(self, graph, source, config=None):
        if not graph.contains_number(source):
            raise ValueError(f"source station number {source} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._states = {}   # state key -> state record
        self._best = {}     # station number -> best state key
        self._pq = MinPQ()
        self._compute()

    # ----- cost model ------------------------------------------------------
    def _cost(self, distance, transfers, preference_penalty):
        """Return the comparable cost tuple of a partial path."""
        mode = self.config.mode
        if mode == "shortest_distance":
            return (distance + preference_penalty, distance)
        if mode == "least_transfer":
            return (float(transfers), distance + preference_penalty)
        # comprehensive: distance plus a penalty for every transfer
        return (distance + preference_penalty + TRANSFER_PENALTY_M * transfers, distance)

    def _state_key(self, node, line, transfers):
        """Return the search-state key. Transfer count is part of the key only
        when a transfer limit is set, so that limited searches stay correct."""
        if self.config.max_transfers is None:
            return (node, line)
        return (node, line, transfers)

    # ----- Dijkstra --------------------------------------------------------
    def _compute(self):
        graph = self.graph
        limit = self.config.max_transfers
        preferred = set(self.config.preferred_lines)

        start_key = self._state_key(self.source, None, 0)
        start = {
            "node": self.source,
            "line": None,
            "transfers": 0,
            "dist": 0.0,
            "pref": 0.0,
            "cost": self._cost(0.0, 0, 0.0),
            "prev": None,
            "edge": None,
        }
        self._states[start_key] = start
        self._pq.insert(start["cost"], start_key)

        settled = set()
        while not self._pq.is_empty():
            cost, key = self._pq.del_min()
            if key in settled:
                continue
            record = self._states[key]
            if cost != record["cost"]:
                continue  # stale queue entry
            settled.add(key)

            node = record["node"]
            if node not in self._best:
                self._best[node] = key

            for edge in graph.adjacent(node):
                nxt = edge.other(node)
                transferred = record["line"] is not None and edge.line != record["line"]
                new_transfers = record["transfers"] + (1 if transferred else 0)
                if limit is not None and new_transfers > limit:
                    continue
                penalty = PREFERENCE_PENALTY_M if (preferred and edge.line not in preferred) else 0.0
                new_dist = record["dist"] + edge.weight
                new_pref = record["pref"] + penalty
                new_cost = self._cost(new_dist, new_transfers, new_pref)
                next_key = self._state_key(nxt, edge.line, new_transfers)
                existing = self._states.get(next_key)
                if existing is None or new_cost < existing["cost"]:
                    self._states[next_key] = {
                        "node": nxt,
                        "line": edge.line,
                        "transfers": new_transfers,
                        "dist": new_dist,
                        "pref": new_pref,
                        "cost": new_cost,
                        "prev": key,
                        "edge": edge,
                    }
                    self._pq.insert(new_cost, next_key)

        for node, key in self._best.items():
            self.dist_to[node] = self._states[key]["dist"]

    # ----- queries ---------------------------------------------------------
    def has_path_to(self, target):
        """Return True if a path from the source to ``target`` exists."""
        return target in self._best

    def path_with_edges(self, target):
        """Return ``(station_numbers, edges)`` of the best path to ``target``.

        ``station_numbers`` starts with the source and ends with the target;
        ``edges`` has one fewer element and ``edges[i]`` joins
        ``station_numbers[i]`` and ``station_numbers[i + 1]``.
        Returns ``([], [])`` if the target is unreachable.
        """
        key = self._best.get(target)
        if key is None:
            return [], []
        nodes = []
        edges = []
        record = self._states[key]
        while record is not None:
            nodes.append(record["node"])
            if record["edge"] is not None:
                edges.append(record["edge"])
            prev_key = record["prev"]
            record = self._states[prev_key] if prev_key is not None else None
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        """Return the number of line changes on the best path to ``target``."""
        _, edges = self.path_with_edges(target)
        return sum(
            1 for index in range(1, len(edges)) if edges[index - 1].line != edges[index].line
        )

"""Fare calculation for the Beijing Subway fare structure.

Fare tiers:
  * Basic tier:       0-6 km       -> 3 yuan
  * Progressive tier: 6-12 km      -> 4 yuan
                      12-22 km     -> 5 yuan
                      22-32 km     -> 6 yuan
  * Extended tier:    >= 32 km     -> 7 yuan, +1 yuan for every further 20 km
"""

import math

# (min_distance_m, max_distance_m, fare_yuan); a distance d belongs to the
# tier where min <= d < max.
BASE_FARE_TIERS = (
    (0, 6000, 3),
    (6000, 12000, 4),
    (12000, 22000, 5),
    (22000, 32000, 6),
)
EXTENDED_START_M = 32000
EXTENDED_BASE_YUAN = 7
EXTENDED_STEP_M = 20000

AVERAGE_SPEED_KMH = 35.0        # average in-service speed used for time estimates
TRANSFER_TIME_MIN = 5.0         # minutes per transfer (walking and waiting)
BOARDING_TIME_MIN = 3.0         # fixed boarding/alighting overhead in minutes


def _validate_distance(distance):
    """Validate a distance in meters and return it as float."""
    if isinstance(distance, bool):
        raise ValueError("distance must be a number")
    try:
        value = float(distance)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"distance must be a number, got {distance!r}") from exc
    if math.isnan(value) or math.isinf(value):
        raise ValueError("distance must be a finite number")
    if value < 0:
        raise ValueError("distance must not be negative")
    return value


def Sum_money(distance):
    """Return the subway fare in yuan for a travelled distance.

    Args:
        distance: Travelled distance in meters (fractional values allowed).

    Returns:
        int: Fare in yuan.

    Raises:
        ValueError: If the distance is negative or not a number.
    """
    value = _validate_distance(distance)
    for min_m, max_m, fare in BASE_FARE_TIERS:
        if value < max_m:
            return fare
    # Extended tier: fare increases by 1 yuan per started 20 km block
    # beyond 32 km.
    return EXTENDED_BASE_YUAN + int((value - EXTENDED_START_M) // EXTENDED_STEP_M)


def get_fare_tier_name(distance):
    """Return the name of the fare tier applicable to a distance in meters."""
    value = _validate_distance(distance)
    if value < 6000:
        return "Basic"
    if value < EXTENDED_START_M:
        return "Progressive"
    return "Extended"


def calculate_estimated_time(distance, transfer_count):
    """Estimate the travel time in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers on the route.

    Returns:
        float: Estimated travel time in minutes.
    """
    value = _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int):
        raise ValueError("transfer_count must be an integer")
    if transfer_count < 0:
        raise ValueError("transfer_count must not be negative")
    in_motion = (value / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    return in_motion + transfer_count * TRANSFER_TIME_MIN + BOARDING_TIME_MIN

"""Data management for the Beijing Subway network.

Loads the dataset, builds the ``Graph`` used by the path algorithms, runs
data integrity checks, and supports updates (adding lines, modifying station
information, updating distances) followed by graph rebuilds.
"""

import json
import os
from collections import deque

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

# Representative (station, line) pairs that must be present in the dataset.
REPRESENTATIVE_STATIONS = (
    ("Pingguoyuan", "Line 1"),
    ("Xizhimen", "Line 2"),
    ("Beijing Zoo", "Line 4"),
    ("Songjiazhuang", "Line 5"),
    ("Shuangjing", "Line 6"),
    ("Dongzhimen", "Line 2"),
)

# Plausibility range for inter-station distances (meters).
MIN_PLAUSIBLE_DISTANCE_M = 100
MAX_PLAUSIBLE_DISTANCE_M = 8000


class DataManager:
    """Manage the subway dataset and the graph built from it."""

    def __init__(self, data_source_path=None):
        self.data_source_path = data_source_path or DATA_FILE_PATH
        self.data = None
        self.graph = None

    # ----- loading and graph construction ---------------------------------
    def load(self):
        """Load the dataset from the data source path and return it."""
        self.data = load_subway_data(self.data_source_path)
        self.graph = None
        return self.data

    def _require_data(self):
        if self.data is None:
            self.load()
        return self.data

    def build_graph(self):
        """Build the weighted undirected graph from the dataset."""
        data = self._require_data()
        graph = Graph()

        names = []
        seen = set()

        def add_name(name):
            if name not in seen:
                seen.add(name)
                names.append(name)

        for station in data["stations"]:
            add_name(station["name"])
        for conn in data["connections"]:
            add_name(conn["source"])
            add_name(conn["dest"])

        for number, name in enumerate(names):
            graph.register_station(name, number)

        lines_by_station = {name: [] for name in names}
        for station in data["stations"]:
            for line in station.get("lines", []):
                if line not in lines_by_station[station["name"]]:
                    lines_by_station[station["name"]].append(line)
        for conn in data["connections"]:
            for endpoint in (conn["source"], conn["dest"]):
                if conn["line"] not in lines_by_station[endpoint]:
                    lines_by_station[endpoint].append(conn["line"])
        for name in names:
            graph.set_station_lines(name, lines_by_station[name])

        for conn in data["connections"]:
            graph.add_edge(
                Edge(
                    graph.get_number(conn["source"]),
                    graph.get_number(conn["dest"]),
                    conn["distance"],
                    conn["line"],
                )
            )

        self.graph = graph
        return graph

    def _refresh(self):
        """Rebuild the graph after the dataset has changed."""
        if self.graph is not None:
            self.build_graph()

    def get_lines(self):
        """Return a sorted list of all line names in the dataset."""
        data = self._require_data()
        lines = set()
        for station in data["stations"]:
            lines.update(station.get("lines", []))
        for conn in data["connections"]:
            lines.add(conn["line"])
        return sorted(lines)

    def get_line_connections(self, line):
        """Return all connections operated by a line."""
        data = self._require_data()
        return [conn for conn in data["connections"] if conn["line"] == line]

    # ----- integrity check -------------------------------------------------
    def check_integrity(self):
        """Run data integrity checks.

        Returns:
            list[str]: Human-readable issue descriptions; empty if no issues.
        """
        data = self._require_data()
        issues = []

        stations = data.get("stations", [])
        connections = data.get("connections", [])

        station_lines = {}
        names = []
        for index, station in enumerate(stations):
            name = station.get("name")
            if not name:
                issues.append(f"station #{index} has no name")
                continue
            if name in station_lines:
                issues.append(f"duplicate station name '{name}'")
                continue
            station_lines[name] = list(station.get("lines", []))
            names.append(name)

        # Connection integrity and distance accuracy.
        seen = set()
        adjacency = {name: set() for name in station_lines}
        for index, conn in enumerate(connections):
            try:
                src = conn["source"]
                dst = conn["dest"]
                dist = conn["distance"]
                line = conn["line"]
            except (KeyError, TypeError):
                issues.append(f"connection #{index} is missing required fields")
                continue

            if src == dst:
                issues.append(f"connection on {line} is a self-loop at '{src}'")
            for endpoint in (src, dst):
                if endpoint not in station_lines:
                    issues.append(
                        f"connection {src} - {dst} ({line}) references unknown station '{endpoint}'"
                    )
                elif station_lines[endpoint] and line not in station_lines[endpoint]:
                    issues.append(f"station '{endpoint}' is not listed on {line}")

            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                issues.append(f"connection {src} - {dst} ({line}) has invalid distance {dist!r}")
            elif not MIN_PLAUSIBLE_DISTANCE_M <= dist <= MAX_PLAUSIBLE_DISTANCE_M:
                issues.append(
                    f"connection {src} - {dst} ({line}) distance {dist} m is outside the "
                    f"plausible range {MIN_PLAUSIBLE_DISTANCE_M}-{MAX_PLAUSIBLE_DISTANCE_M} m"
                )

            key = (frozenset((src, dst)), line)
            if key in seen:
                issues.append(f"duplicate connection {src} - {dst} on {line}")
            seen.add(key)

            if src in adjacency and dst in adjacency:
                adjacency[src].add(dst)
                adjacency[dst].add(src)

        # Connectivity.
        if names:
            visited = {names[0]}
            queue = deque([names[0]])
            while queue:
                current = queue.popleft()
                for nxt in adjacency.get(current, ()):
                    if nxt not in visited:
                        visited.add(nxt)
                        queue.append(nxt)
            for name in names:
                if name not in visited:
                    issues.append(f"station '{name}' is not connected to the rest of the network")

        # Coverage of representative stations.
        for station, line in REPRESENTATIVE_STATIONS:
            if station not in station_lines:
                issues.append(f"representative station '{station}' ({line}) is missing")
            elif line not in station_lines[station]:
                issues.append(f"representative station '{station}' is not listed on {line}")

        # Lines that have stations but no connections.
        connection_lines = {conn.get("line") for conn in connections}
        station_line_names = set()
        for lines in station_lines.values():
            station_line_names.update(lines)
        for line in sorted(station_line_names - connection_lines):
            issues.append(f"line '{line}' has stations but no connections")

        return issues

    # ----- updates ---------------------------------------------------------
    def add_line(self, line_name, stations, distances):
        """Add a new line with an ordered station sequence.

        Args:
            line_name: Name of the new line.
            stations: Ordered list of station names.
            distances: Distances in meters between consecutive stations
                (length ``len(stations) - 1``).

        Raises:
            ValueError: If the input is inconsistent or the line exists.
        """
        data = self._require_data()
        if not line_name:
            raise ValueError("line name must not be empty")
        if line_name in self.get_lines():
            raise ValueError(f"line '{line_name}' already exists")
        if len(stations) < 2:
            raise ValueError("a line needs at least two stations")
        if len(set(stations)) != len(stations):
            raise ValueError("a line cannot contain the same station twice")
        if len(distances) != len(stations) - 1:
            raise ValueError("expected one distance per consecutive station pair")
        if any(not isinstance(d, int) or d <= 0 for d in distances):
            raise ValueError("distances must be positive integers (meters)")

        entries = {station["name"]: station for station in data["stations"]}
        for name in stations:
            if name not in entries:
                entry = {"name": name, "lines": [line_name]}
                data["stations"].append(entry)
                entries[name] = entry
            elif line_name not in entries[name].setdefault("lines", []):
                entries[name]["lines"].append(line_name)

        for index, dist in enumerate(distances):
            data["connections"].append(
                {
                    "source": stations[index],
                    "dest": stations[index + 1],
                    "distance": dist,
                    "line": line_name,
                }
            )
        self._refresh()

    def update_station(self, name, new_name=None, lines=None):
        """Modify a station's name and/or lines.

        Raises:
            KeyError: If the station does not exist.
            ValueError: If the new name already exists.
        """
        data = self._require_data()
        entry = next((s for s in data["stations"] if s["name"] == name), None)
        if entry is None:
            raise KeyError(f"unknown station '{name}'")
        if lines is not None:
            entry["lines"] = list(lines)
        if new_name and new_name != name:
            if any(s["name"] == new_name for s in data["stations"]):
                raise ValueError(f"station '{new_name}' already exists")
            entry["name"] = new_name
            for conn in data["connections"]:
                if conn["source"] == name:
                    conn["source"] = new_name
                if conn["dest"] == name:
                    conn["dest"] = new_name
        self._refresh()

    def update_distance(self, source, dest, distance):
        """Update the distance of every connection between two stations.

        Returns:
            int: Number of connections updated.

        Raises:
            KeyError: If no connection exists between the stations.
            ValueError: If the distance is not a positive number.
        """
        data = self._require_data()
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("distance must be a positive number of meters")
        count = 0
        for conn in data["connections"]:
            if {conn["source"], conn["dest"]} == {source, dest}:
                conn["distance"] = distance
                count += 1
        if count == 0:
            raise KeyError(f"no connection between '{source}' and '{dest}'")
        self._refresh()
        return count

    def save(self, path=None):
        """Write the current dataset to a JSON file."""
        data = self._require_data()
        target = path or self.data_source_path
        directory = os.path.dirname(os.path.abspath(target))
        os.makedirs(directory, exist_ok=True)
        with open(target, "w", encoding="utf-8") as handle:
            json.dump(data, handle, ensure_ascii=False, indent=2)
        return target

"""Path visualization, summary generation and report export."""

import json
import os

from fare_calculator import Sum_money, calculate_estimated_time, get_fare_tier_name
from pathfinding import PathfindingConfig, PathNotFoundError, ShortestPath


def compute_route(graph, start_name, end_name, config=None):
    """Compute the route between two stations and build the result record.

    Raises:
        ValueError: For invalid station names or identical endpoints.
        PathNotFoundError: If no route satisfies the constraints.
    """
    config = config if config is not None else PathfindingConfig()
    if start_name == end_name:
        raise ValueError("Starting station cannot be the same as the ending station")
    for name in (start_name, end_name):
        if not graph.has_station(name):
            raise ValueError(f"station '{name}' does not exist")

    source = graph.get_number(start_name)
    target = graph.get_number(end_name)
    sp = ShortestPath(graph, source, config)
    if not sp.has_path_to(target):
        if config.max_transfers is not None:
            raise PathNotFoundError(
                f"no route from '{start_name}' to '{end_name}' within "
                f"{config.max_transfers} transfer(s)"
            )
        raise PathNotFoundError(f"no route from '{start_name}' to '{end_name}'")

    nodes, edges = sp.path_with_edges(target)
    return build_route_result(graph, nodes, edges, start_name, end_name, config)


def build_route_result(graph, nodes, edges, start_name, end_name, config):
    """Build a JSON-serialisable dictionary describing a route."""
    names = [graph.get_name(number) for number in nodes]
    total_m = float(sum(edge.distance for edge in edges))

    transfers = []
    segments = []
    segment_start = 0
    for index in range(1, len(edges)):
        if edges[index - 1].line != edges[index].line:
            segments.append(_make_segment(names, edges, segment_start, index, edges[index - 1].line))
            transfers.append(
                {
                    "station": names[index],
                    "from_line": edges[index - 1].line,
                    "to_line": edges[index].line,
                }
            )
            segment_start = index
    if edges:
        segments.append(
            _make_segment(names, edges, segment_start, len(names) - 1, edges[-1].line)
        )

    lines_involved = []
    for segment in segments:
        if segment["line"] not in lines_involved:
            lines_involved.append(segment["line"])

    transfer_count = len(transfers)
    fare = Sum_money(total_m)
    tier = get_fare_tier_name(total_m)
    estimated = calculate_estimated_time(total_m, transfer_count)
    total_km = round(total_m / 1000.0, 3)

    return {
        "query": {
            "start": start_name,
            "end": end_name,
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "line_preference": list(config.preferred_lines),
        },
        "path": names,
        "segments": segments,
        "transfers": transfers,
        "lines": lines_involved,
        "distance": {"meters": round(total_m, 1), "kilometers": total_km},
        "fare": {"yuan": fare, "tier": tier},
        "transfer_count": transfer_count,
        "estimated_time_minutes": round(estimated, 1),
        "summary": {
            "total_distance_km": total_km,
            "total_fare_yuan": fare,
            "transfer_count": transfer_count,
            "estimated_time_minutes": round(estimated, 1),
        },
    }


def _make_segment(names, edges, start, end, line):
    """Create a description of a run of stations served by one line."""
    distance = sum(edges[k].distance for k in range(start, end))
    return {
        "line": line,
        "start": names[start],
        "end": names[end],
        "stations": names[start:end + 1],
        "stop_count": end - start,
        "distance_m": float(distance),
    }


def format_text(result):
    """Render a route result as console-formatted text."""
    query = result["query"]
    out = []
    out.append("=" * 60)
    out.append(f"Route: {query['start']} -> {query['end']}")
    out.append(f"Mode: {query['mode']}")
    if query["max_transfers"] is not None:
        out.append(f"Max transfers: {query['max_transfers']}")
    if query["line_preference"]:
        out.append(f"Preferred lines: {', '.join(query['line_preference'])}")
    out.append("-" * 60)

    out.append(f"Station sequence ({len(result['path'])} stations):")
    for index, name in enumerate(result["path"], start=1):
        out.append(f"  {index:>3}. {name}")

    out.append("Line segments:")
    for segment in result["segments"]:
        out.append(
            f"  [{segment['line']}] {segment['start']} -> {segment['end']} "
            f"({segment['stop_count']} stops, {segment['distance_m']:.0f} m)"
        )

    out.append("Transfers:")
    if not result["transfers"]:
        out.append("  None (direct trip)")
    for transfer in result["transfers"]:
        out.append(
            f"  - at {transfer['station']}: {transfer['from_line']} -> {transfer['to_line']}"
        )
    out.append(f"Lines involved: {', '.join(result['lines'])}")

    out.append("-" * 60)
    distance = result["distance"]
    out.append(f"Total distance: {distance['meters']:.0f} m ({distance['kilometers']:.2f} km)")
    out.append(f"Number of transfers: {result['transfer_count']}")
    out.append(f"Estimated time: {result['estimated_time_minutes']:.1f} minutes")
    out.append(f"Fare: {result['fare']['yuan']} yuan")
    out.append("=" * 60)
    return "\n".join(out)


def format_json(result):
    """Render a route result as a JSON document."""
    return json.dumps(result, ensure_ascii=False, indent=2)


def render_report(result, fmt="text"):
    """Render a route result in the requested format ('text' or 'json')."""
    if fmt == "json":
        return format_json(result)
    if fmt == "text":
        return format_text(result)
    raise ValueError(f"unsupported output format '{fmt}'")


def export_results(results, fmt, output_file):
    """Write all query results of a session to a file.

    JSON output is a single JSON array; text output concatenates the reports.
    """
    if fmt == "json":
        content = json.dumps(results, ensure_ascii=False, indent=2)
    else:
        content = "\n\n".join(format_text(result) for result in results)
    directory = os.path.dirname(os.path.abspath(output_file))
    os.makedirs(directory, exist_ok=True)
    with open(output_file, "w", encoding="utf-8") as handle:
        handle.write(content + "\n")
    return output_file

"""Command-line interface for the Beijing Subway fare calculation system.

Run with:  python src/cli.py [--format text|json] [--mode MODE]
           [--max_transfers N] [--line_preference LINE]
           [--output_file PATH] [--check_data]
"""

import argparse
import re
import sys

from data_loader import DataLoadError
from data_manager import DataManager
from fare_calculator import Sum_money  # noqa: F401  (fare is computed inside report)
from pathfinding import DEFAULT_MODE, VALID_MODES, PathfindingConfig, PathNotFoundError
from report import compute_route, export_results, render_report

QUERY_SEPARATORS = r"[;；,，]"
QUIT_COMMANDS = ("quit", "exit", "q")
HELP_COMMANDS = ("help", "?")

HELP_TEXT = """Enter a query in the form 'Starting Station; Ending Station'
(a comma may also be used as separator).
Commands: help, quit"""


class QueryError(Exception):
    """Raised for invalid user input."""


def _normalize(text):
    """Normalise a name for lenient matching (case and separators ignored)."""
    return "".join(ch for ch in text.lower() if ch.isalnum())


def build_name_index(graph):
    """Map normalised station names to official station names."""
    return {_normalize(name): name for name in graph.station_names()}


def resolve_station(raw, name_index, label):
    """Standardise a user-entered station name against the official names."""
    key = _normalize(raw)
    if not key or key not in name_index:
        raise QueryError(f"{label} station '{raw.strip()}' does not exist, please check the station name")
    return name_index[key]


def parse_query(text):
    """Split a query into (starting station, ending station) strings."""
    parts = [part.strip() for part in re.split(QUERY_SEPARATORS, text)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise QueryError(
            "Input format incorrect, please use 'Starting Station; Ending Station' format"
        )
    return parts[0], parts[1]


def parse_line_preference(value, known_lines):
    """Parse a comma-separated list of preferred lines (e.g. 'Line 1,Line 2')."""
    if not value:
        return []
    index = {_normalize(line): line for line in known_lines}
    preferred = []
    for part in re.split(r"[,，;；]", value):
        token = part.strip()
        if not token:
            continue
        key = _normalize(token)
        if key in index:
            line = index[key]
        elif "line" + key in index:
            line = index["line" + key]
        else:
            raise ValueError(
                f"unknown line '{token}'. Known lines: {', '.join(known_lines)}"
            )
        if line not in preferred:
            preferred.append(line)
    return preferred


def parse_args(argv=None):
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (CLI)."
    )
    parser.add_argument("--output_file", default=None,
                        help="export query results to this file")
    parser.add_argument("--format", choices=["text", "json"], default="text",
                        help="output format (default: text)")
    parser.add_argument("--mode", choices=list(VALID_MODES), default=DEFAULT_MODE,
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="maximum number of transfers allowed")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line(s), e.g. 'Line 1' or 'Line 1,Line 2'")
    parser.add_argument("--check_data", action="store_true",
                        help="run data integrity check and exit")
    return parser.parse_args(argv)


def process_query(text, graph, name_index, config):
    """Parse, validate and compute a single query; return the result record."""
    start_raw, end_raw = parse_query(text)
    start = resolve_station(start_raw, name_index, "Starting")
    end = resolve_station(end_raw, name_index, "Ending")
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")
    return compute_route(graph, start, end, config)


def run_interactive(graph, config, fmt, output_file):
    """Run the continuous query loop until the user quits."""
    name_index = build_name_index(graph)
    results = []
    print("Beijing Subway fare calculator. " + HELP_TEXT)
    while True:
        try:
            line = input("\nQuery> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = line.strip()
        if not text:
            continue
        lowered = text.lower()
        if lowered in QUIT_COMMANDS:
            break
        if lowered in HELP_COMMANDS:
            print(HELP_TEXT)
            continue

        try:
            result = process_query(text, graph, name_index, config)
            rendered = render_report(result, fmt)
        except QueryError as exc:
            print(f"Input error: {exc}")
            continue
        except PathNotFoundError as exc:
            print(f"Path error: {exc}")
            continue
        except (ValueError, KeyError, ArithmeticError) as exc:
            print(f"Calculation error: {exc}")
            continue
        except Exception as exc:  # last-resort guard to keep the session alive
            print(f"Unexpected error while processing query: {exc}")
            continue

        print(rendered)
        if output_file:
            results.append(result)
            try:
                export_results(results, fmt, output_file)
                print(f"Results exported to {output_file}")
            except OSError as exc:
                print(f"Export error: cannot write '{output_file}': {exc}")
    return 0


def main(argv=None):
    """Entry point. Returns the process exit code."""
    args = parse_args(argv)

    manager = DataManager()
    try:
        manager.load()
        graph = manager.build_graph()
    except DataLoadError as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        issues = manager.check_integrity()
        if not issues:
            print("Data integrity check passed: no issues found.")
            print(f"Stations: {graph.node_count}, connections: {graph.edge_count}")
            return 0
        print(f"Data integrity check found {len(issues)} issue(s):")
        for issue in issues:
            print(f"  - {issue}")
        return 1

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must not be negative", file=sys.stderr)
        return 2

    try:
        preferred = parse_line_preference(args.line_preference, manager.get_lines())
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    try:
        config = PathfindingConfig(args.mode, args.max_transfers, preferred)
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    return run_interactive(graph, config, args.format, args.output_file)


if __name__ == "__main__":
    sys.exit(main())

"""Grid geometry helpers for the 800x800 map.

The map is an 8x8 grid of 100x100 unit cells. Riders, restaurants and
customers are always placed at cell centers, computed as
``(coordinate // 100) * 100 + 50``.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = 8
CELL_HALF = 50
COORD_MIN = 0
COORD_MAX = MAP_SIZE


def validate_coordinate(value):
    """Validate a single integer coordinate.

    Args:
        value: The coordinate value to validate.

    Returns:
        int: The validated coordinate.

    Raises:
        TypeError: If ``value`` is not an integer (booleans are rejected).
        ValueError: If ``value`` lies outside the range 0-800.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise TypeError(f"coordinate {value!r} is not an integer")
    if not COORD_MIN <= value <= COORD_MAX:
        raise ValueError(
            f"coordinate {value} is out of range {COORD_MIN}-{COORD_MAX}"
        )
    return value


def cell_index(value):
    """Return the grid index (0-7) that contains a coordinate value.

    Abnormal values are clamped into the map. The far boundary value 800 is
    mapped into the last cell.

    Args:
        value: Coordinate in units (0-800).

    Returns:
        int: Grid index between 0 and 7 inclusive.
    """
    clamped = min(max(int(value), COORD_MIN), MAP_SIZE - 1)
    return clamped // CELL_SIZE


def cell_center(x, y):
    """Convert a coordinate pair to the center point of its grid cell.

    Args:
        x: Horizontal coordinate in units.
        y: Vertical coordinate in units.

    Returns:
        tuple[int, int]: The cell center ``(center_x, center_y)``.
    """
    return (cell_index(x) * CELL_SIZE + CELL_HALF,
            cell_index(y) * CELL_SIZE + CELL_HALF)


def cell_of(x, y):
    """Return the grid cell indices ``(gx, gy)`` of a coordinate pair."""
    return (cell_index(x), cell_index(y))


def manhattan(a, b):
    """Return the Manhattan distance between two points.

    Args:
        a: Point ``(x, y)``.
        b: Point ``(x, y)``.

    Returns:
        int: Sum of absolute axis differences.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def path_length(start, points):
    """Return the total Manhattan length of a poly-line.

    Args:
        start: Starting point ``(x, y)``.
        points: Ordered list of waypoints ``(x, y)`` following ``start``.

    Returns:
        int: Total travelled distance in units.
    """
    total = 0
    current = start
    for point in points:
        total += manhattan(current, point)
        current = point
    return total

"""Grid-based path planning for riders.

Routes follow the streets of the 8x8 grid: a rider always moves along one
axis at a time, preferring horizontal movement first at a right-angle turn.
A simple priority rule prevents deadlocks between riders that want to occupy
the same position.
"""

import grid

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2


class PathPlanner:
    """Compute street-following routes and resolve movement conflicts."""

    def plan(self, start, goal):
        """Plan a route from ``start`` to ``goal``.

        If the two points share an x or y coordinate (the target is directly
        above, below, left or right), a straight single-segment route is
        returned. Otherwise the route makes exactly one right-angle turn.

        Args:
            start: Current point ``(x, y)``.
            goal: Destination point ``(x, y)``.

        Returns:
            list[tuple[int, int]]: Waypoints after ``start`` ending at
            ``goal``. Empty if already at the goal.
        """
        sx, sy = start
        gx, gy = goal
        if (sx, sy) == (gx, gy):
            return []
        if sx == gx or sy == gy:
            return [(gx, gy)]
        corner = (gx, sy)
        return [corner, (gx, gy)]

    def plan_route(self, start, stops):
        """Plan a continuous multi-destination route.

        Args:
            start: Starting point ``(x, y)``.
            stops: Ordered destinations, e.g. ``[restaurant, customer]``.

        Returns:
            list[tuple[int, int]]: Concatenated waypoints after ``start``.
        """
        points = []
        current = start
        for stop in stops:
            leg = self.plan(current, stop)
            points.extend(leg)
            current = stop
        return points

    def route_length(self, start, points):
        """Return the total length of a route in units."""
        return grid.path_length(start, points)

    @staticmethod
    def step(x, y, target, budget):
        """Move one axis-aligned step toward ``target``.

        Movement is performed along the first axis that still differs from
        the target, limited by ``budget``.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            target: Target waypoint ``(x, y)``.
            budget: Maximum units that may be moved this tick.

        Returns:
            tuple[int, int, int]: New ``(x, y)`` and the distance moved.
        """
        tx, ty = target
        if x != tx:
            delta = tx - x
            moved = min(budget, abs(delta))
            return (x + moved if delta > 0 else x - moved), y, moved
        if y != ty:
            delta = ty - y
            moved = min(budget, abs(delta))
            return x, (y + moved if delta > 0 else y - moved), moved
        return x, y, 0

    @staticmethod
    def is_blocked(mover, next_pos, riders):
        """Decide whether ``mover`` must yield before entering ``next_pos``.

        A rider yields only to a lower-id rider that is currently in motion
        and occupies the target position. Because waiting is only ever done
        toward strictly lower ids, no cycle of waiting riders can form, which
        rules out deadlock.

        Args:
            mover: The rider that wants to move.
            next_pos: Position ``(x, y)`` the mover would enter.
            riders: Iterable of all riders.

        Returns:
            bool: True if the mover must wait this tick.
        """
        target = (next_pos[0], next_pos[1])
        for other in riders:
            if other is mover:
                continue
            if other.rider_id >= mover.rider_id:
                continue
            if other.state == STATE_IDLE:
                continue
            if (other.x, other.y) == target:
                return True
        return False

"""Order data model.

An order holds the restaurant and customer positions as entered by the user
(0-800 units) and the grid cell centers they are converted to.
"""

import itertools

import globalvar
import grid

PENDING = "pending"
ASSIGNED = "assigned"
PICKING_UP = "picking_up"
DELIVERING = "delivering"
COMPLETED = "completed"

_id_sequence = itertools.count(1)


def _parse_position(pos, label):
    """Validate an ``(x, y)`` pair of integer coordinates.

    Args:
        pos: Sequence of two values.
        label: Human-readable name used in error messages.

    Returns:
        tuple[int, int]: Validated coordinates.

    Raises:
        ValueError: If the pair is malformed or out of range.
    """
    try:
        x, y = pos
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{label} position must be a pair (x, y)") from exc
    return (grid.validate_coordinate(x), grid.validate_coordinate(y))


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        restaurant_pos: Restaurant coordinates as entered (0-800).
        customer_pos: Customer coordinates as entered (0-800).
        res_grid: Restaurant grid cell center.
        cus_grid: Customer grid cell center.
        status: One of pending, assigned, picking_up, delivering, completed.
        rider_id: Identifier of the assigned rider, or None.
        start_tick: Simulation tick at which the order was created.
        complete_tick: Simulation tick of delivery, or None.
        distance: Accumulated planned travel distance in units.
        settled: True once the order's revenue has been booked.
    """

    def __init__(self, res_pos, cus_pos):
        """Create an order from raw coordinates.

        Args:
            res_pos: Restaurant position ``(x, y)`` in 0-800.
            cus_pos: Customer position ``(x, y)`` in 0-800.

        Raises:
            ValueError: If a coordinate is invalid or out of range.
            TypeError: If a coordinate is not an integer.
        """
        self.id = next(_id_sequence)
        self.restaurant_pos = _parse_position(res_pos, "restaurant")
        self.customer_pos = _parse_position(cus_pos, "customer")
        self.res_grid = grid.cell_center(*self.restaurant_pos)
        self.cus_grid = grid.cell_center(*self.customer_pos)
        self.status = PENDING
        self.rider_id = None
        self.start_tick = globalvar.get_value("tick", 0)
        self.complete_tick = None
        self.distance = 0
        self.settled = False

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = ASSIGNED

    def complete(self, tick):
        """Mark the order as delivered.

        Args:
            tick: Simulation tick at which delivery finished.
        """
        self.status = COMPLETED
        self.complete_tick = tick

    @property
    def time_taken(self):
        """Ticks from creation to delivery, or None if not completed."""
        if self.complete_tick is None:
            return None
        return self.complete_tick - self.start_tick

    @property
    def efficiency(self):
        """Distance covered per tick, or None if not completed."""
        taken = self.time_taken
        if not taken:
            return None
        return self.distance / taken

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant_pos": list(self.restaurant_pos),
            "customer_pos": list(self.customer_pos),
            "res_grid": list(self.res_grid),
            "cus_grid": list(self.cus_grid),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_tick": self.start_tick,
            "complete_tick": self.complete_tick,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "settled": self.settled,
        }

"""Rider entity: movement, state transitions and event logging.

State machine:
    idle (0) -> picking up (1) when an order is accepted
    picking up (1) -> delivering (2) on arrival at the restaurant
    delivering (2) -> idle (0) after delivery (or -> picking up for the next
    queued order)

Every transition is recorded in ``rider.log`` as a dict with the keys
``event``, ``order_id``, ``time``, ``x``, ``y`` and ``state``.
"""

import logging

import globalvar
import grid
import order_model
from path_planner import (PathPlanner, STATE_IDLE, STATE_PICKING_UP,
                          STATE_DELIVERING)

logger = logging.getLogger("dispatch.rider")

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}
DEFAULT_SPEED = 50


class Rider:
    """A delivery rider that moves on the street grid.

    Attributes:
        rider_id: Unique identifier.
        x: Current x coordinate.
        y: Current y coordinate.
        speed: Movement speed in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Assigned orders, the first one being the active order.
        special_case: True when the rider yielded or is in a special state.
        route: Remaining waypoints of the current leg.
        log: Chronological list of event dictionaries.
        total_distance: Total units travelled.
    """

    def __init__(self, rider_id, x=50, y=50, speed=DEFAULT_SPEED):
        """Create a rider and log its spawn event.

        Args:
            rider_id: Unique identifier.
            x: Initial x coordinate (defaults to a cell center).
            y: Initial y coordinate (defaults to a cell center).
            speed: Movement speed in units per tick, must be positive.

        Raises:
            ValueError: If speed is not positive.
        """
        if int(speed) <= 0:
            raise ValueError("rider speed must be a positive integer")
        self.rider_id = int(rider_id)
        self.x = int(x)
        self.y = int(y)
        self.speed = int(speed)
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.route = []
        self.log = []
        self.total_distance = 0
        self._planner = PathPlanner()
        self._record("spawn", None)
        logger.info("Rider #%d spawned at (%d, %d)", self.rider_id,
                    self.x, self.y)

    @property
    def position(self):
        """Current position as a tuple ``(x, y)``."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human-readable name of the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def _now(self):
        """Return the current simulation tick."""
        return globalvar.get_value("tick", 0)

    def _all_riders(self):
        """Return the list of all riders from global storage."""
        return globalvar.get_value("riders") or []

    def _record(self, event, order_id):
        """Append an event entry to the rider log.

        Args:
            event: Event name such as ``accepted_order``.
            order_id: Related order identifier, or None.
        """
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def _plan_to(self, target, order):
        """Plan the route to ``target`` and add its length to the order."""
        points = self._planner.plan(self.position, target)
        order.distance += grid.path_length(self.position, points)
        self.route = points

    def _begin_pickup(self):
        """Start the pickup leg for the first order in the queue."""
        order = self.orders[0]
        self.state = STATE_PICKING_UP
        order.status = order_model.PICKING_UP
        self._plan_to(order.res_grid, order)
        self._record("accepted_order", order.id)
        logger.info("Rider #%d accepted order #%d", self.rider_id, order.id)

    def _finish_leg(self):
        """Handle arrival at the current leg destination."""
        if not self.orders:
            self.state = STATE_IDLE
            self.route = []
            return
        order = self.orders[0]
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.status = order_model.DELIVERING
            self._plan_to(order.cus_grid, order)
            self._record("arrived_at_merchant", order.id)
            logger.info("Rider #%d picked up order #%d",
                        self.rider_id, order.id)
        elif self.state == STATE_DELIVERING:
            self.orders.pop(0)
            order.complete(self._now())
            self._record("delivered_to_customer", order.id)
            logger.info("Rider #%d delivered order #%d",
                        self.rider_id, order.id)
            if self.orders:
                self._begin_pickup()
            else:
                self.state = STATE_IDLE
                self.route = []
                self.special_case = False

    def _advance(self):
        """Move along the current route using this tick's speed budget."""
        budget = self.speed
        self.special_case = False
        riders = self._all_riders()
        while budget > 0 and self.route:
            target = self.route[0]
            nx, ny, moved = self._planner.step(self.x, self.y, target, budget)
            if moved == 0:
                self.route.pop(0)
                continue
            if self._planner.is_blocked(self, (nx, ny), riders):
                self.special_case = True
                logger.debug("Rider #%d yielded at (%d, %d)",
                             self.rider_id, self.x, self.y)
                return
            self.x, self.y = nx, ny
            self.total_distance += moved
            budget -= moved
            if (self.x, self.y) == target:
                self.route.pop(0)

    def move(self):
        """Advance the rider by one tick.

        Handles the idle-to-picking-up transition when an order is assigned,
        moves along the planned route, and performs arrival transitions
        (restaurant and customer) when a leg completes.
        """
        if self.state == STATE_IDLE:
            if not self.orders:
                self.special_case = False
                return
            self._begin_pickup()
        self._advance()
        while not self.route and self.state != STATE_IDLE:
            self._finish_leg()

    def to_dict(self):
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "state_name": self.state_name,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
            "total_distance": self.total_distance,
            "log": list(self.log),
        }

"""Order management: assignment, pending queue, completion and settlement."""

import logging

import globalvar
import grid
from order_model import Order, COMPLETED

logger = logging.getLogger("dispatch.order")

# Distance-equivalent penalty per already-queued order, used to balance load
# among riders that are roughly equally close.
LOAD_PENALTY = 50


def find_best_rider_for_order(order):
    """Find the best rider for an order using a distance-first strategy.

    The closest rider (Manhattan distance from its current position to the
    restaurant) wins. Each order already queued on a rider adds a load
    penalty so that work is spread across riders. Ties are broken by the
    lower rider id.

    Args:
        order: The order needing a rider.

    Returns:
        Rider: The selected rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    best = None
    best_key = None
    for rider in riders:
        distance = grid.manhattan(rider.position, order.res_grid)
        key = (distance + LOAD_PENALTY * len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Create, assign and settle orders.

    Attributes:
        pending: Orders waiting for a rider to become available.
        active: Orders assigned but not yet settled.
        history: Orders whose completion has been settled.
    """

    def __init__(self):
        """Initialise empty order queues."""
        self.pending = []
        self.active = []
        self.history = []

    def create_order(self, res_pos, cus_pos):
        """Create an order and register it in global storage.

        Args:
            res_pos: Restaurant ``(x, y)`` in 0-800.
            cus_pos: Customer ``(x, y)`` in 0-800.

        Returns:
            Order: The newly created order.
        """
        order = Order(res_pos, cus_pos)
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        logger.info("Order #%d created: restaurant %s customer %s",
                    order.id, order.restaurant_pos, order.customer_pos)
        return order

    def get_all_orders(self):
        """Return every order ever created."""
        return globalvar.get_value("orders") or []

    def assign(self, order):
        """Assign an order to the best rider, or queue it if none exists.

        Args:
            order: The order to assign.

        Returns:
            Rider: The rider that received the order, or None if the order
            was placed in the pending queue.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            if order not in self.pending:
                self.pending.append(order)
            logger.warning("No rider available for order #%d", order.id)
            return None
        if order in self.pending:
            self.pending.remove(order)
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        if order not in self.active:
            self.active.append(order)
        logger.info("Order #%d assigned to rider #%d",
                    order.id, rider.rider_id)
        return rider

    def dispatch_pending(self):
        """Try to assign every pending order.

        Returns:
            list[tuple[Order, Rider]]: Orders that were newly assigned.
        """
        assigned = []
        for order in list(self.pending):
            rider = self.assign(order)
            if rider is not None:
                assigned.append((order, rider))
        return assigned

    def settle_completed(self):
        """Book revenue for completed orders and release their resources.

        Revenue is ``order_revenue`` minus a fuel cost of
        ``fuel_cost_per_100_units`` for every 100 units travelled. Funds may
        become negative.

        Returns:
            list[Order]: Orders settled during this call.
        """
        config = globalvar.get_value("config") or {}
        revenue = int(config.get("order_revenue", 150))
        fuel_rate = int(config.get("fuel_cost_per_100_units", 2))
        settled = []
        for order in list(self.active):
            if order.status != COMPLETED or order.settled:
                continue
            fuel_cost = (order.distance // 100) * fuel_rate
            net = revenue - fuel_cost
            globalvar.set_value("funds",
                                (globalvar.get_value("funds") or 0) + net)
            order.settled = True
            self.active.remove(order)
            self.history.append(order)
            settled.append(order)
            logger.info("Order #%d settled, net %d", order.id, net)
        return settled

"""System-wide state, funds, rider recruitment and the simulation clock."""

import json
import logging
import os

import globalvar
import grid

    "initial_funds": 300,
    "rider_hire_threshold": 300,

def init_system():
    """Initialise the global variables used by the system.

    Sets an empty rider list, rider id counter 0, running state True, the
    simulation tick to 0, the default configuration and starting funds.
    """
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("config", dict(DEFAULT_CONFIG))
    globalvar.set_value("funds", DEFAULT_CONFIG["initial_funds"])

class SystemManager:

    """

    def __init__(self):
        """Create the system manager and its order manager."""

    def _config(self):
        """Return the effective configuration merged with defaults."""
        config = dict(DEFAULT_CONFIG)
        stored = globalvar.get_value("config") or {}
        config.update(stored)
        return config

    @property
    def funds(self):
        """Current system fund balance (may be negative)."""
        return globalvar.get_value("funds", 0)

    def add_funds(self, amount):
        """Add (or subtract, if negative) an amount of funds.

        Args:
            amount: Integer amount to add.
        """
        globalvar.set_value("funds", self.funds + int(amount))

    @property
    def tick_count(self):
        """Current simulation tick."""
        return globalvar.get_value("tick", 0)

    @property
    def riders(self):
        """The list of all riders."""
        return globalvar.get_value("riders")

    @property
    def running(self):
        """True while the main loop should keep running."""
        return bool(globalvar.get_value("running", False))

        """Request that the main loop terminate."""
        globalvar.set_value("running", False)

    def get_rider(self, rider_id):

        Args:
            rider_id: Identifier to look up.

        counter = (globalvar.get_value("rider_id_counter") or 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        index = (counter - 1) % (grid.GRID_DIM * grid.GRID_DIM)
        x = (index % grid.GRID_DIM) * grid.CELL_SIZE + grid.CELL_HALF
        y = (index // grid.GRID_DIM) * grid.CELL_SIZE + grid.CELL_HALF
        speed = int(self._config()["rider_speed"])
        rider = Rider(counter, x, y, speed)
        self.riders.append(rider)
        return rider

    def remove_rider(self, rider_id):
        """Remove an idle rider that has no assigned orders.

        Args:
            rider_id: Identifier of the rider to remove.

        Raises:
            KeyError: If no such rider exists.
            RuntimeError: If the rider is busy.
        """
        rider = self.get_rider(rider_id)
        if rider is None:
            raise KeyError(f"rider #{rider_id} not found")
        if rider.orders or rider.state != 0:
            raise RuntimeError(f"rider #{rider_id} is busy")
        self.riders.remove(rider)
        logger.info("Rider #%d removed", rider_id)

    def try_recruit(self):
        """Recruit a rider when funds reach the hiring threshold.

        Returns:
            Rider: The recruited rider, or None if funds are below threshold.
        """
        config = self._config()
        if self.funds >= int(config["rider_hire_threshold"]):
            self.add_funds(-int(config["rider_hire_cost"]))
            rider = self.create_rider()
            logger.info("Recruited rider #%d", rider.rider_id)
            return rider
        return None

    def tick(self):
        """Advance the simulation by one tick.

        Order of operations: advance the clock, move every rider, settle
        completed orders, recruit if funds allow, then dispatch pending
        orders.
        """
        globalvar.set_value("tick", self.tick_count + 1)
        for rider in list(self.riders):
            rider.move()
        self.orders.settle_completed()
        self.try_recruit()
        self.orders.dispatch_pending()

    def load_config(self, path):
        """Load configuration overrides from a JSON file.

        Unknown keys and invalid values are ignored. Missing files and
        malformed JSON leave the defaults in place.

        Args:
            path: Path to the JSON configuration file.

        Returns:
            bool: True if the file was read successfully.
        """
        if not os.path.exists(path):
            return False
        try:
            with open(path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
        except (OSError, json.JSONDecodeError) as exc:
            logger.error("Failed to load config %s: %s", path, exc)
            return False
        if not isinstance(data, dict):
            logger.error("Config %s must contain a JSON object", path)
            return False

        config = self._config()
        for key in DEFAULT_CONFIG:
            value = data.get(key)
            if isinstance(value, bool) or not isinstance(value, int):
                continue
            if value < 0:
                continue
            if key == "rider_speed" and value == 0:
                continue
            config[key] = value
        globalvar.set_value("config", config)

        if self.tick_count == 0 and not self.riders \
                and "initial_funds" in data:
            globalvar.set_value("funds", config["initial_funds"])
        logger.info("Configuration loaded from %s", path)
        return True

    def save_state(self, path):
        """Persist configuration, funds, riders and orders as JSON.

        Args:
            path: Destination file path.

        Returns:
            bool: True if the file was written successfully.
        """
        state = {
            "config": self._config(),
            "funds": self.funds,
            "tick": self.tick_count,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict()
                       for order in self.orders.get_all_orders()],
        }
        try:
            with open(path, "w", encoding="utf-8") as handle:
                json.dump(state, handle, indent=2)
        except OSError as exc:
            logger.error("Failed to save state to %s: %s", path, exc)
            return False
        logger.info("State saved to %s", path)
        return True

"""Text rendering for the dispatch simulator.

The ASCII map uses a 15x15 canvas: even canvas positions are grid
intersections (cell centers) and odd positions are the streets between them.
"""

import grid
from order_model import COMPLETED
from rider import STATE_NAMES


class DisplayManager:
    """Render status tables, order tables, rider logs and the ASCII map."""

    CANVAS_SIZE = grid.GRID_DIM * 2 - 1
    LEGEND = ("Legend: R=Rider  S=Restaurant  C=Customer  "
              ".=intersection  #=street  *=planned route")

    def _canvas_pos(self, point):
        """Return the ``(row, col)`` canvas position of a map point.

        Row 0 is the top of the map, which corresponds to grid row 7.
        """
        gx, gy = grid.cell_of(point[0], point[1])
        return 2 * (grid.GRID_DIM - 1 - gy), 2 * gx

    def _put(self, canvas, point, symbol):
        """Place a marker symbol on the canvas at a map point."""
        row, col = self._canvas_pos(point)
        canvas[row][col] = symbol

    def _draw_route(self, canvas, start, end):
        """Draw an axis-aligned route segment with ``*`` characters."""
        r1, c1 = self._canvas_pos(start)
        r2, c2 = self._canvas_pos(end)
        for row in range(min(r1, r2), max(r1, r2) + 1):
            for col in range(min(c1, c2), max(c1, c2) + 1):
                canvas[row][col] = "*"

    def render_map(self, riders, orders=None, route=None, start=None):
        """Render the ASCII map with riders, orders and an optional route.

        Args:
            riders: Iterable of riders to mark with ``R``.
            orders: Iterable of orders; non-completed orders show ``S``
                (restaurant) and ``C`` (customer).
            route: Optional list of waypoints drawn as a connected line.
            start: Optional starting point of the route.

        Returns:
            str: Multi-line map text.
        """
        size = self.CANVAS_SIZE
        canvas = []
        for row in range(size):
            line = []
            for col in range(size):
                if row % 2 == 0 and col % 2 == 0:
                    line.append(".")
                elif row % 2 == 0 or col % 2 == 0:
                    line.append("#")
                else:
                    line.append(" ")
            canvas.append(line)

        if route:
            points = ([start] if start is not None else []) + list(route)
            for first, second in zip(points, points[1:]):
                self._draw_route(canvas, first, second)
            if len(points) == 1:
                self._draw_route(canvas, points[0], points[0])

        for order in orders or []:
            if order.status == COMPLETED:
                continue
            self._put(canvas, order.cus_grid, "C")
            self._put(canvas, order.res_grid, "S")
        for rider in riders or []:
            self._put(canvas, rider.position, "R")

        lines = ["  " + "".join(
            f"{col // 2 if col % 2 == 0 else ' '} " for col in range(size))]
        for row_index, row in enumerate(canvas):
            label = (str(grid.GRID_DIM - 1 - row_index // 2)
                     if row_index % 2 == 0 else " ")
            lines.append(f"{label} " + "".join(f"{ch} " for ch in row))
        lines.append("(columns = grid x index, rows = grid y index)")
        lines.append(self.LEGEND)
        return "\n".join(lines)

    def render_status(self, funds, riders, pending_count, tick):
        """Render a snapshot of the system state.

        Args:
            funds: Current fund balance.
            riders: List of riders.
            pending_count: Number of orders waiting for a rider.
            tick: Current simulation tick.

        Returns:
            str: Status text.
        """
        lines = [
            f"Simulation tick: {tick}",
            f"System funds: {funds}",
            f"Riders: {len(riders)}",
            f"Pending orders (no rider yet): {pending_count}",
        ]
        for rider in riders:
            order_ids = [order.id for order in rider.orders]
            flag = " [special case]" if rider.special_case else ""
            lines.append(
                f"  Rider #{rider.rider_id}: position ({rider.x}, {rider.y}), "
                f"state {rider.state_name}, speed {rider.speed}, "
                f"orders {order_ids}{flag}")
        return "\n".join(lines)

    def render_orders(self, orders):
        """Render a table of all orders.

        Args:
            orders: Iterable of Order objects.

        Returns:
            str: Table text.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        lines = [
            f"{'ID':<5}{'Start':<7}{'Restaurant':<13}{'Customer':<13}"
            f"{'Distance':<10}{'Taken':<7}{'Rider':<7}{'Status':<12}"
            f"{'Eff.':<8}"
        ]
        for order in orders:
            taken = order.time_taken
            taken_text = "-" if taken is None else str(taken)
            rider_text = "-" if order.rider_id is None else f"#{order.rider_id}"
            eff = order.efficiency
            eff_text = "-" if eff is None else f"{eff:.2f}"
            lines.append(
                f"{order.id:<5}{order.start_tick:<7}"
                f"{str(order.restaurant_pos):<13}"
                f"{str(order.customer_pos):<13}"
                f"{order.distance:<10}{taken_text:<7}{rider_text:<7}"
                f"{order.status:<12}{eff_text:<8}")
        return "\n".join(lines)

    def render_rider_logs(self, riders):
        """Render the recorded status changes of every rider.

        Args:
            riders: Iterable of riders.

        Returns:
            str: Log text grouped by rider.
        """
        riders = list(riders)
        if not riders:
            return "No riders yet."
        lines = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} ({rider.state_name}):")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order_text = ("-" if entry["order_id"] is None
                              else f"order #{entry['order_id']}")
                state_name = STATE_NAMES.get(entry["state"], "unknown")
                lines.append(
                    f"  t={entry['time']:<6} {entry['event']:<22}"
                    f"{order_text:<12} at ({entry['x']}, {entry['y']}) "
                    f"[{state_name}]")
        return "\n".join(lines)

    def render_assignment(self, rider, order, route, distance, ticks):
        """Render the summary shown after an order is assigned.

        Args:
            rider: The assigned rider.
            order: The new order.
            route: Waypoints of the planned route after the rider position.
            distance: Total route length in units.
            ticks: Estimated number of ticks to complete the route.

        Returns:
            str: Summary text followed by the route map.
        """
        queued = len(rider.orders) - 1
        queue_text = (f" (queued behind {queued} order(s))"
                      if queued > 0 else "")
        lines = [
            f"Order #{order.id} assigned to rider #{rider.rider_id}"
            f"{queue_text}.",
            f"  Rider speed: {rider.speed} units/tick, "
            f"state: {rider.state_name}",
            f"  Restaurant grid point: {order.res_grid}",
            f"  Customer grid point: {order.cus_grid}",
            f"  Route distance: {distance} units, "
            f"estimated time: {ticks} tick(s)",
            "Route map:",
            self.render_map([rider], [order], route, rider.position),
        ]
        return "\n".join(lines)

"""Command parsing, shortcut resolution, history and input validation."""

import grid

try:  # Enables arrow-key line editing and history where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    readline = None

COMMANDS = {
    "start", "status", "orders", "riders", "map", "tick", "save",
    "history", "help", "quit", "exit",
}

SHORTCUTS = {
    "a": "start",
    "s": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "v": "save",
    "y": "history",
    "h": "help",
    "?": "help",
    "q": "quit",
    "x": "exit",
}


class InputManager:
    """Read, expand and parse user commands.

    Attributes:
        history: Ordered list of previously entered command lines.
    """

    def __init__(self, reader=input):
        """Create the input manager.

        Args:
            reader: Callable used to read a line, taking a prompt argument.
        """
        self._reader = reader
        self.history = []

    def read_line(self, prompt="dispatch> "):
        """Read one raw line from the user.

        Args:
            prompt: Prompt text.

        Returns:
            str: The stripped input line.
        """
        return self._reader(prompt).strip()

    def record(self, line):
        """Add a non-empty command line to the history."""
        if line.strip():
            self.history.append(line.strip())

    def expand_history(self, line):
        """Expand ``!!`` (last command) and ``!n`` (n-th command) references.

        Args:
            line: Raw command line.

        Returns:
            str: The expanded command line.

        Raises:
            IndexError: If the referenced history entry does not exist.
        """
        stripped = line.strip()
        if stripped == "!!":
            if not self.history:
                raise IndexError("no previous command in history")
            return self.history[-1]
        if stripped.startswith("!") and stripped[1:].isdigit():
            index = int(stripped[1:])
            if index < 1 or index > len(self.history):
                raise IndexError(f"no history entry {index}")
            return self.history[index - 1]
        return stripped

    def parse_command(self, line):
        """Split a command line into a canonical command name and arguments.

        Args:
            line: Expanded command line.

        Returns:
            tuple[str | None, list[str]]: Command name and argument tokens.

        Raises:
            ValueError: If the command is unknown.
        """
        parts = line.split()
        if not parts:
            return None, []
        word = parts[0].lower()
        name = SHORTCUTS.get(word, word)
        if name not in COMMANDS:
            raise ValueError(f"unknown command '{parts[0]}'. Type 'help'.")
        return name, parts[1:]

    @staticmethod
    def parse_coordinate_token(token):
        """Convert one token to a validated integer coordinate.

        Args:
            token: Text token from the user.

        Returns:
            int: The validated coordinate.

        Raises:
            ValueError: If the token is not an integer or is out of range.
        """
        try:
            value = int(token)
        except (TypeError, ValueError) as exc:
            raise ValueError(f"'{token}' is not an integer") from exc
        return grid.validate_coordinate(value)

    def parse_coordinates(self, tokens):
        """Parse four coordinates ``[rx, ry, cx, cy]``.

        Args:
            tokens: Sequence of four text tokens.

        Returns:
            list[int]: Validated coordinates.

        Raises:
            ValueError: If the count is wrong or a value is invalid.
        """
        tokens = list(tokens)
        if len(tokens) != 4:
            raise ValueError("expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y")
        return [self.parse_coordinate_token(token) for token in tokens]

    def prompt_coordinates(self, writer=print):
        """Interactively ask for the four order coordinates.

        Invalid entries are reported and re-requested. Typing ``cancel``
        aborts the order.

        Args:
            writer: Callable used to print messages.

        Returns:
            list[int] | None: Validated coordinates, or None if cancelled.
        """
        labels = ["restaurant x", "restaurant y",
                  "customer x", "customer y"]
        values = []
        for label in labels:
            while True:
                raw = self.read_line(f"  {label} (0-800, 'cancel' to abort): ")
                if raw.lower() == "cancel":
                    return None
                try:
                    values.append(self.parse_coordinate_token(raw))
                    break
                except ValueError as exc:
                    writer(f"  Invalid input: {exc}. Try again.")
        return values

"""Command-line entry point for the food delivery dispatch simulator.

Each iteration of the main loop processes one command and then advances the
simulation by one tick.
"""

import logging
import math
import sys

import globalvar
import grid
from display import DisplayManager
from input_manager import InputManager
from path_planner import PathPlanner
from system import SystemManager, init_system

CONFIG_PATH = "config.json"
STATE_PATH = "dispatch_state.json"
LOG_PATH = "dispatch.log"
MAX_TICK_COMMAND = 1000

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy] [a]  Place an order: restaurant then customer grid
                           coordinates (0-800). Prompts if omitted.
  status [s]               Funds, rider count and each rider's state/orders.
  orders [o]               Details of all orders.
  riders [r]               Status-change log of every rider.
  map [m]                  ASCII map of riders and active orders.
  tick [t] [n]             Advance the simulation by n extra ticks (default 1).
  save [v] [path]          Save state to JSON (default dispatch_state.json).
  history [y]              Show command history. Use !! or !n to repeat.
  help [h, ?]              Show this help.
  quit [q] / exit [x]      Leave the program.
Every command is followed by one simulation tick."""


class DispatchApp:
    """Interactive controller that wires the modules together."""

    def __init__(self, reader=input, writer=print):
        """Create the application.

        Args:
            reader: Callable used to read input lines.
            writer: Callable used to print output lines.
        """
        self.system = SystemManager()
        self.inputs = InputManager(reader)
        self.display = DisplayManager()
        self.planner = PathPlanner()
        self._out = writer
        self._log = logging.getLogger("dispatch.app")
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def say(self, text=""):
        """Print a line of output."""
        self._out(text)

    def startup(self):
        """Initialise the system, load configuration and recruit if able."""
        init_system()
        self.system.load_config(CONFIG_PATH)
        rider = self.system.try_recruit()
        self.say("=== Food Delivery Dispatch Simulator ===")
        if rider is not None:
            self.say(f"Funds reached the hiring threshold. "
                     f"Rider #{rider.rider_id} recruited.")
        self.say("Type 'help' for commands.")

    def run(self):
        """Run the main loop until the system is stopped."""
        self.startup()
        while self.system.running:
            try:
                line = self.inputs.read_line()
            except (EOFError, KeyboardInterrupt):
                self.say()
                break
            self.execute(line)
            if self.system.running:
                self._safe_tick()
        self.say("Goodbye.")

    def _safe_tick(self):
        """Advance one tick, recovering from unexpected errors."""
        try:
            self.system.tick()
        except Exception:  # noqa: BLE001 - keep the simulation alive
            self._log.exception("Error during simulation tick")
            self.say("Internal error during tick; state has been kept.")

    def execute(self, line):
        """Expand, validate and dispatch one command line.

        Args:
            line: Raw command line typed by the user.
        """
        try:
            expanded = self.inputs.expand_history(line)
        except IndexError as exc:
            self.say(f"History error: {exc}")
            return
        if not expanded:
            return
        self.inputs.record(expanded)
        try:
            name, args = self.inputs.parse_command(expanded)
        except ValueError as exc:
            self.say(f"Error: {exc}")
            return
        if name is None:
            return
        handler = self._handlers[name]
        try:
            handler(args)
        except EOFError:
            raise
        except Exception as exc:  # noqa: BLE001 - graceful recovery
            self._log.exception("Command '%s' failed", name)
            self.say(f"Error while running '{name}': {exc}")

    def cmd_start(self, args):
        """Handle ``start``: create an order and assign a rider."""
        if args:
            try:
                coords = self.inputs.parse_coordinates(args)
            except ValueError as exc:
                self.say(f"Invalid input: {exc}")
                return
        else:
            self.say("Enter restaurant and customer coordinates (0-800).")
            coords = self.inputs.prompt_coordinates(self.say)
            if coords is None:
                self.say("Order cancelled.")
                return

        rx, ry, cx, cy = coords
        new_rider = self.system.try_recruit()
        order = self.system.orders.create_order((rx, ry), (cx, cy))
        self.say(f"Order #{order.id} created.")
        self.say(f"  Restaurant: input ({rx}, {ry}) -> grid {order.res_grid}")
        self.say(f"  Customer:   input ({cx}, {cy}) -> grid {order.cus_grid}")
        if new_rider is not None:
            self.say(f"New rider #{new_rider.rider_id} added to the team.")

        rider = self.system.orders.assign(order)
        if rider is None:
            self.say("No rider available. The order is queued (pending).")
            return
        route = self.planner.plan_route(rider.position,
                                        [order.res_grid, order.cus_grid])
        distance = grid.path_length(rider.position, route)
        ticks = math.ceil(distance / rider.speed) if distance else 0
        self.say(self.display.render_assignment(
            rider, order, route, distance, ticks))

    def cmd_status(self, _args):
        """Handle ``status``: show funds, riders and pending orders."""
        self.say(self.display.render_status(
            self.system.funds, self.system.riders,
            len(self.system.orders.pending), self.system.tick_count))

    def cmd_orders(self, _args):
        """Handle ``orders``: show the order table."""
        self.say(self.display.render_orders(self.system.orders.get_all_orders()))

    def cmd_riders(self, _args):
        """Handle ``riders``: show every rider's status-change log."""
        self.say(self.display.render_rider_logs(self.system.riders))

    def cmd_map(self, _args):
        """Handle ``map``: show the ASCII map."""
        self.say(self.display.render_map(self.system.riders,
                                         self.system.orders.active))

    def cmd_tick(self, args):
        """Handle ``tick [n]``: advance the simulation by n extra ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                self.say("Invalid input: tick count must be an integer.")
                return
            if count < 1 or count > MAX_TICK_COMMAND:
                self.say(f"Tick count must be between 1 and "
                         f"{MAX_TICK_COMMAND}.")
                return
        for _ in range(count):
            self._safe_tick()
        self.say(f"Advanced {count} tick(s). Current tick: "
                 f"{self.system.tick_count}.")

    def cmd_save(self, args):
        """Handle ``save [path]``: persist the current state."""
        path = args[0] if args else STATE_PATH
        if self.system.save_state(path):
            self.say(f"State saved to {path}.")
        else:
            self.say(f"Could not save state to {path}. See dispatch.log.")

    def cmd_history(self, _args):
        """Handle ``history``: list previously entered commands."""
        if not self.inputs.history:
            self.say("History is empty.")
            return
        for index, entry in enumerate(self.inputs.history, start=1):
            self.say(f"  {index:>3}  {entry}")

    def cmd_help(self, _args):
        """Handle ``help``: show the command reference."""
        self.say(HELP_TEXT)

    def cmd_quit(self, _args):
        """Handle ``quit``/``exit``: stop the main loop."""
        self.system.stop()


def main():
    """Configure logging and run the interactive application.

    Returns:
        int: Process exit code.
    """
    try:
        logging.basicConfig(
            filename=LOG_PATH,
            level=logging.INFO,
            format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        )
    except OSError:
        logging.basicConfig(level=logging.WARNING)
    app = DispatchApp()
    try:
        app.run()
    except Exception:  # noqa: BLE001 - last-resort reporting
        logging.getLogger("dispatch.app").exception("Fatal error")
        print("A fatal error occurred. See dispatch.log for details.")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

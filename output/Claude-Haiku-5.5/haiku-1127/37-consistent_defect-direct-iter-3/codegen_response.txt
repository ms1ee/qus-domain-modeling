def normalize_text(value: str | None) -> str:
    """Return a case-insensitive, whitespace-trimmed key for comparisons."""

"""Participant list loading, verification and group extraction."""

from __future__ import annotations

import re
from pathlib import Path

from lottery_system.default_data import DEFAULT_LIST_LINES
from lottery_system.models import User

_SEPARATOR_PATTERN = re.compile(r"[,;]")


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'Name,Department' or 'Name;Department'."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        try:
            content = self._read_text(path)
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}' ({exc})"]
        return self._parse_lines(content.splitlines())

    @staticmethod
    def _read_text(path) -> str:
        raw = Path(path).read_bytes()
        for encoding in ("utf-8-sig", "gbk"):
            try:
                return raw.decode(encoding)
            except UnicodeDecodeError:
                continue
        return raw.decode("utf-8", errors="replace")

    @staticmethod
    def _parse_lines(lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw_line in enumerate(lines, start=1):
            line = raw_line.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SEPARATOR_PATTERN.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {line_no}: Format Error - '{line}' "
                    "(expected 'Name,Department Group' or 'Name;Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). The first occurrence of a name is kept."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group -> number of people (first-seen order)."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize, normalize_text


def _clean_groups(avoid_groups) -> list[str]:
    cleaned: list[str] = []
    for group in avoid_groups or []:
        if group is None:
            continue
        text = str(group).strip()
        if text and text not in cleaned:
            cleaned.append(text)
    return cleaned


def _clean_drawer(associated_drawer: str | None) -> str | None:
    if associated_drawer is None:
        return None
    text = associated_drawer.strip()
    return text or None


class PrizeManager:
    """Manages the ordered prize list. Starts with no prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(groups), drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        """Append a prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")
        if self.find_prize(clean_name) is not None:
            raise ValueError(f"Prize '{clean_name}' already exists.")
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=_clean_groups(avoid_groups),
            associated_drawer=_clean_drawer(associated_drawer),
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def find_prize(self, name: str) -> Prize | None:
        key = normalize_text(name)
        for prize in self._prizes:
            if normalize_text(prize.name) == key:
                return prize
        return None

    def configure_prize(self, name: str, avoid_groups=None, associated_drawer=None) -> bool:
        """Update avoidance groups and lottery person of an existing prize."""
        prize = self.find_prize(name)
        if prize is None:
            return False
        prize.avoid_groups = _clean_groups(avoid_groups)
        prize.associated_drawer = _clean_drawer(associated_drawer)
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order holds 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            if len(order) != count or sorted(order) != list(range(1, count + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    normalize_text,
)


def is_cheng_ru_member(user: User) -> bool:
    """True if the user belongs to the Cheng Ru group."""
    return normalize_text(user.department) == normalize_text(CHENG_RU_GROUP)


class LotteryManager:
    """Runs draws while enforcing the lottery rules.

    Rules:
    - Previous winners are always excluded (anti-duplication, any prize type).
    - Members of a prize's avoidance groups are drawn only after other members.
    - Lottery person Li Bin: only previous winners are excluded; avoidance groups are ignored.
    - Cheng Ru protection: if the last two draws by Chen Qixian / Jin Duo contained no
      Cheng Ru group member, Ma Jiaqi's prize pool is limited to Cheng Ru group members.
    """

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # True means the draw winner belonged to the Cheng Ru group.
        self.last_two_draw_results: list[bool] = []

    def has_won(self, user: User) -> bool:
        return user.name in self._winner_names

    def mark_winner(self, user: User) -> None:
        self._winner_names.add(user.name)

    def get_available_users(self) -> list[User]:
        """Users who have not won yet."""
        return [user for user in self._users if user.name not in self._winner_names]

    def _cheng_ru_protection_active(self, prize: Prize) -> bool:
        return (
            normalize_text(prize.associated_drawer) == normalize_text(MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = self.get_available_users()
        drawer = normalize_text(prize.associated_drawer)

        # Special rule: Li Bin only filters previous winners.
        if drawer == normalize_text(LI_BIN):
            return available

        # Fallback protection: force Cheng Ru group members when protection is active.
        if self._cheng_ru_protection_active(prize):
            protected = [user for user in available if is_cheng_ru_member(user)]
            if protected:
                return protected

        # Avoidance groups: non-avoided members first; avoided members only afterwards.
        avoid = {normalize_text(group) for group in prize.avoid_groups}
        preferred = [user for user in available if normalize_text(user.department) not in avoid]
        return preferred if preferred else available

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible user remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self.mark_winner(winner)
        self._record_draw(prize, winner)
        return winner

    def _record_draw(self, prize: Prize, winner: User) -> None:
        drawer = normalize_text(prize.associated_drawer)
        if drawer in (normalize_text(CHEN_QIXIAN), normalize_text(JIN_DUO)):
            self.last_two_draw_results.append(is_cheng_ru_member(winner))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == normalize_text(MA_JIAQI):
            # The protection window is consumed by Ma Jiaqi's draw.
            self.last_two_draw_results = []

    @staticmethod
    def describe_rule(prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        drawer = normalize_text(prize.associated_drawer)
        parts: list[str] = []
        if drawer == normalize_text(LI_BIN):
            parts.append(
                "Previous winners are excluded; avoidance groups are NOT applied "
                "for this lottery person"
            )
        else:
            parts.append("Previous winners are excluded")
            if prize.avoid_groups:
                groups = ", ".join(prize.avoid_groups)
                parts.append(
                    f"members of avoidance group(s) [{groups}] are drawn only after other members"
                )
            if drawer == normalize_text(MA_JIAQI):
                parts.append(
                    "Cheng Ru Group protection: if the last two draws by Chen Qixian and "
                    "Jin Duo had no Cheng Ru Group winner, only Cheng Ru Group members are eligible"
                )
        return "; ".join(parts) + "."

"""Winner recording, ordering, display grouping and CSV export."""

from __future__ import annotations

import csv
import re
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path

from lottery_system.models import Prize, User

_LEVEL_PATTERN = re.compile(r"^\s*(First|Second|Third)\s+Prize\b", re.IGNORECASE)
_LEVEL_RANKS = {"first": 1, "second": 2, "third": 3}
_CUSTOM_RANK = 4


def prize_level_rank(prize_name: str) -> int:
    """Lower rank means higher level: First=1, Second=2, Third=3, others=4."""
    match = _LEVEL_PATTERN.match(prize_name or "")
    if match is None:
        return _CUSTOM_RANK
    return _LEVEL_RANKS[match.group(1).lower()]


@dataclass
class Winner:
    user: User
    prize: Prize
    won_at: datetime
    sequence: int

    @property
    def prize_name(self) -> str:
        return self.prize.name


class ResultManager:
    """Keeps the complete winner list in winning order."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(
            user=user,
            prize=prize,
            won_at=datetime.now(),
            sequence=len(self._winners) + 1,
        )
        self._winners.append(winner)
        return winner

    def has_winners(self) -> bool:
        return bool(self._winners)

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    @staticmethod
    def _sorted_prizes(prizes: list[Prize]) -> list[Prize]:
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda item: (prize_level_rank(item[1].name), item[0]))
        return [prize for _, prize in indexed]

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Winners grouped by prize, levels ordered high to low, winning order within a prize."""
        result: dict[str, list[Winner]] = {}
        for prize in self._sorted_prizes(prizes):
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export winners to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        if not self._winners:
            return False, "No winners to export."

        grouped = self.get_winners_by_prize(prizes)
        ordered: list[Winner] = []
        seen: set[int] = set()
        for winners in grouped.values():
            for winner in winners:
                ordered.append(winner)
                seen.add(id(winner))
        # Winners whose prize is not in the given list are still exported.
        for winner in self._winners:
            if id(winner) not in seen:
                ordered.append(winner)

        file_path = Path.cwd() / f"awards_result_{datetime.now():%Y%m%d}.txt"
        try:
            with open(file_path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in ordered:
                    writer.writerow([
                        winner.user.name,
                        winner.user.department,
                        winner.prize_name,
                        winner.won_at.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {file_path}"

def ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery_manager: LotteryManager | None = None
        self.completed_prizes: set[str] = set()
        self._handlers = {
            "1": self.load_list_menu,
            "2": self.show_groups,
            "3": self.prize_menu,
            "4": self.start_lottery,
            "5": self.show_results,
            "6": self.export_results,
        }

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _available_groups(self) -> list[str]:
        if not self.users:
            return []
        return list(self.list_manager.get_group_info(self.users).keys())

    # ---------------------------------------------------------------- main loop
    def run(self) -> None:
        print("Company Team Building - Smart Lottery System")
        try:
            while True:
                print(MAIN_MENU)
                choice = self._ask("Select an option: ")
                if choice == "0":
                    print("Goodbye!")
                    return
                handler = self._handlers.get(choice)
                if handler is None:
                    print("Invalid option, please try again.")
                else:
                    handler()
        except (EOFError, KeyboardInterrupt):
            print("\nLottery system closed.")

    # ------------------------------------------------------------ list module
    def load_list_menu(self) -> None:
        if self.result_manager.has_winners():
            print("The lottery already has results; the participant list can no longer be changed.")
            return
        print("1. Built-in default team building list")
        print("2. Upload custom txt list (each line: Name,Department Group)")
        choice = self._ask("Select list source: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter the txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Invalid option.")
            return

        final_users = self._verify_users(users, errors)
        if final_users is None:
            print("List upload terminated. Current list is unchanged.")
            return
        if not final_users:
            print("No valid participants found. Current list is unchanged.")
            return

        self.users = final_users
        self.lottery_manager = LotteryManager(self.users)
        self.completed_prizes.clear()
        print(f"Loaded {len(final_users)} participant(s).")
        self.show_groups()

    def _verify_users(self, users: list[User], errors: list[str]) -> list[User] | None:
        """Report errors/duplicates. Returns the valid users, or None if terminated."""
        if errors:
            print(f"Found {len(errors)} format error(s):")
            for error in errors:
                print(f"  {error}")
        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  Duplicate: {user.name} ({user.department})")
        if not errors and not duplicates:
            return valid

        print("Options: 'r' = remove invalid data and continue, 't' = terminate upload")
        while True:
            choice = self._ask("Your choice (r/t): ").lower()
            if choice == "r":
                return valid
            if choice == "t":
                return None
            print("Please enter 'r' or 't'.")

    def show_groups(self) -> None:

            return
        info = self.list_manager.get_group_info(self.users)

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) or "None"
            drawer = prize.associated_drawer or "None"

    def _ask_groups(self) -> list[str]:
        groups = self._available_groups()
        if groups:
            print("Available department groups:")
            for index, group in enumerate(groups, start=1):
                print(f"  {index}. {group}")
        text = self._ask("Avoidance groups (numbers or names, comma-separated; blank for none): ")
        result: list[str] = []
        for token in (part.strip() for part in text.split(",")):
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                name = groups[int(token) - 1]
            else:

            if name not in result:

    def _ask_drawer(self) -> str | None:

        if not text:
            return None

            return None

            print("No prizes configured.")

        text = self._ask("Select prize number to configure: ")

            print("Invalid prize number.")
            return
        prize = prizes[int(text) - 1]
        groups = self._ask_groups()
        drawer = self._ask_drawer()
        if self.prize_manager.configure_prize(prize.name, groups, drawer):
            print("Prize configuration updated.")
        else:
            print("Prize not found.")

    def reorder_prizes_menu(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self.show_prizes()
        text = self._ask("Enter new order as serial numbers separated by spaces (e.g. 3 1 2): ")
        try:
            order = [int(token) for token in text.split()]
        except ValueError:
            print("Order must contain only serial numbers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
            self.show_prizes()
        else:
            print("Invalid order: each prize number must appear exactly once.")

    # ------------------------------------------------------ lottery control
    def start_lottery(self) -> None:
        if not self.users or self.lottery_manager is None:
            print("Load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Configure at least one prize first.")
            return

        print("\nLottery sequence:")
        self.show_prizes()
        for round_no, prize in enumerate(prizes, start=1):
            if prize.name in self.completed_prizes:
                print(f"Round {round_no} ({prize.name}) already completed; skipping.")
                continue
            self._run_round(round_no, prize)

        print("\nAll rounds finished.")
        self.show_results()

    def _run_round(self, round_no: int, prize) -> None:
        print("\n" + "=" * 64)
        print(f"Round {round_no}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {LotteryManager.describe_rule(prize)}")
        print("=" * 64)
        print("Commands: Enter = draw next winner | pause | continue | terminate (skip round)")

        drawn: list = []
        paused = False
        while len(drawn) < prize.quantity:
            position = len(drawn) + 1
            status = " [PAUSED]" if paused else ""
            command = self._ask(
                f"Currently drawing the {ordinal(position)} winner{status} > "
            ).lower()

            if command == "pause":
                if paused:
                    print("Lottery is already paused.")
                else:
                    paused = True
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to skip the round.")
                continue
            if command == "continue":
                if paused:
                    paused = False
                    print("Lottery resumed.")
                else:
                    print("Lottery is not paused.")
                continue
            if command == "terminate":
                print(f"Round terminated. {len(drawn)} of {prize.quantity} winner(s) drawn.")
                break
            if command != "":
                print("Unknown command. Use Enter, pause, continue or terminate.")
                continue
            if paused:
                print("Lottery is paused. Type 'continue' to resume.")
                continue

            user = self.lottery_manager.draw(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            winner = self.result_manager.add_winner(user, prize)
            drawn.append(winner)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        print(f"\nRound {round_no} results ({prize.name}):")
        if not drawn:
            print("  (no winners in this round)")
        for index, winner in enumerate(drawn, start=1):
            print(f"  {index}. {winner.user.name} ({winner.user.department})")
        self.completed_prizes.add(prize.name)

    # -------------------------------------------------------- results module
    def show_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not self.result_manager.has_winners():
            print("No winners yet.")
            return
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n==================== Winner List ====================")
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}")
            if not winners:
                print("  (no winners)")
            for index, winner in enumerate(winners, start=1):
                print(
                    f"  {index}. {winner.user.name} ({winner.user.department})"
                    f"  [{winner.won_at:%H:%M:%S}]"
                )

    def export_results(self) -> None:
        success, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(message)


def main() -> None:
    LotterySystemCLI().run()

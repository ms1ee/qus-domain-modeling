"""Core data models for the lottery system."""

from __future__ import annotations

import re
from dataclasses import dataclass, field

_ORDINAL_LEVELS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
}

UNRANKED_LEVEL = 99


@dataclass
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round.

    avoid_groups: department names whose members must not win this prize.
    associated_drawer: name of the lottery person in charge of this prize, or None.
    """

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None


def prize_level(prize: Prize) -> int:
    """Return the ranking level of a prize (1 = highest).

    The level is derived from an ordinal word contained in the prize name
    (e.g. "First Prize - ..." -> 1). Prizes without an ordinal word are
    unranked and receive a large level value so they sort after ranked prizes.
    """
    for token in re.findall(r"[A-Za-z]+", prize.name):
        level = _ORDINAL_LEVELS.get(token.lower())
        if level is not None:
            return level
    return UNRANKED_LEVEL

"""Participant list loading, validation and group extraction."""

from __future__ import annotations

import re

from lottery_system.default_data import DEFAULT_PARTICIPANTS
from lottery_system.models import User

_SPLIT_PATTERN = re.compile(r"[,\t]")


class ListManager:
    """Loads participant lists from the built-in default or custom txt files.

    Custom file format: one participant per line, ``name,department``
    (a tab may be used instead of the comma). Blank lines and lines starting
    with ``#`` are ignored.
    """

    def __init__(self) -> None:
        self.users: list[User] = []

    def load_default_list(self) -> tuple[list[User], list[str]]:
        users = [User(name=name, department=dept) for name, dept in DEFAULT_PARTICIPANTS]
        self.users = list(users)
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        users, errors = self._load_from_path(path)
        self.users = list(users)
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"Unable to read file {path}: {exc}"]

        for line_no, raw_line in enumerate(lines, start=1):
            line = raw_line.strip()
            if not line or line.startswith("#"):
                continue
            parts = [part.strip() for part in _SPLIT_PATTERN.split(line)]
            if len(parts) != 2 or not all(parts):
                errors.append(
                    f"Line {line_no}: Format Error - expected 'name,department', got '{line}'"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid entries (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items(), key=lambda item: item[0]))

"""Prize library management and lottery sequence ordering."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the prize list. Starts empty; prizes are kept in lottery order."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | str | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        if avoid_groups is None:
            raw_groups: list[str] = []
        elif isinstance(avoid_groups, str):
            raw_groups = avoid_groups.split(",")
        else:
            raw_groups = list(avoid_groups)

        groups: list[str] = []
        for group in raw_groups:
            group = group.strip()
            if group and group not in groups:
                groups.append(group)

        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. ``order`` holds 1-based current positions in the new order."""
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(positions) != list(range(1, len(self._prizes) + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        return True

"""Winner recording, summary grouping and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, prize_level


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize: Prize
    winning_order: int


class ResultManager:
    """Keeps the winner list in winning time order and produces summaries."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(user=user, prize=prize, winning_order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered from highest to lowest prize level.

        Within the same prize, winners keep their winning time order.
        """
        ordered_prizes = sorted(prizes, key=prize_level)
        result: dict[str, list[Winner]] = {}
        for prize in ordered_prizes:
            if prize.name in result:
                continue
            result[prize.name] = [w for w in self._winners if w.prize.name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        if not self._winners:
            return False, "No winners to export."

        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize Name", "Name", "Department", "Winning Order"])
                for prize_name, winners in self.get_winners_by_prize(prizes).items():
                    for winner in winners:
                        writer.writerow(
                            [
                                prize_name,
                                winner.user.name,
                                winner.user.department,
                                winner.winning_order,
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

CHENG_RU_GROUP = "Cheng Ru group"
LI_BIN_NAME = "Li Bin"
CHEN_QIXIAN_NAME = "Chen Qixian"
JIN_DUO_NAME = "Jin Duo"
MA_JIAQI_NAME = "Ma Jiaqi"


class LotteryManager:
    """Runs individual draws while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Cheng Ru outcome (True = at least one Cheng Ru member won) of the
        # most recent draws made by Chen Qixian / Jin Duo (max two entries).
        self.last_two_draw_results: list[bool] = []
        self._winner_names: set[str] = set()
        self._rng = random.Random()

    # ----- rule helpers -------------------------------------------------

    def _is_fallback_active(self, prize: Prize) -> bool:
        return (
            prize.associated_drawer == MA_JIAQI_NAME
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = [u for u in self.users if u.name not in self._winner_names]

        # Special rule: Li Bin's rounds relax filtering (group avoidance ignored).
        if prize.associated_drawer == LI_BIN_NAME:
            return available

        # Fallback (Cheng Ru group protection): forced to Cheng Ru members,
        # even if Cheng Ru is an avoidance group of this prize.
        if self._is_fallback_active(prize):
            protected = [u for u in available if u.department == CHENG_RU_GROUP]
            if protected:
                return protected

        return [u for u in available if u.department not in prize.avoid_groups]

    def has_eligible(self, prize: Prize) -> bool:
        return len(self._get_eligible_pool(prize)) > 0

    def describe_rules(self, prize: Prize) -> str:
        rules = ["previous winners excluded"]
        drawer = prize.associated_drawer
        if drawer == LI_BIN_NAME:
            rules.append("special rule for Li Bin: group avoidance relaxed")
        elif prize.avoid_groups:
            rules.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if drawer == MA_JIAQI_NAME and self._is_fallback_active(prize):
            rules.append(
                "Cheng Ru group protection active: only Cheng Ru group members are eligible"
            )
        return "; ".join(rules)

    # ----- drawing ------------------------------------------------------

    def draw_one(self, prize: Prize) -> User | None:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winner_names.add(user.name)
        self.result_manager.add_winner(user, prize)
        self._update_fallback_state(prize, user)
        return user

    def _update_fallback_state(self, prize: Prize, user: User) -> None:
        drawer = prize.associated_drawer
        if drawer in (CHEN_QIXIAN_NAME, JIN_DUO_NAME):
            self.last_two_draw_results.append(user.department == CHENG_RU_GROUP)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def end_round(self, prize: Prize) -> None:
        """Must be called when a round finishes; resets fallback state after Ma Jiaqi's round."""
        if prize.associated_drawer == MA_JIAQI_NAME:
            self.last_two_draw_results = []

from __future__ import annotations

from lottery_system.models import Prize, User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager

MAIN_MENU = """
==============================================
   Team Building Smart Lottery System
==============================================
1. Load participant list
2. View department groups
3. Prize configuration
4. Start lottery
5. Export results
0. Exit
"""

PRIZE_MENU = """
--- Prize configuration ---
1. Add custom prize
2. Load default prize library
3. View prizes
4. Reorder prizes (lottery sequence)
0. Back
"""


def ordinal(n: int) -> str:
    """Return the English ordinal form of a positive integer (1st, 2nd, 11th...)."""
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Drives the menu-based lottery workflow."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.lottery_prizes: list[Prize] = []

    # ----- main loop ----------------------------------------------------

    def run(self) -> None:
        try:
            while True:
                print(MAIN_MENU)
                choice = self._ask("Select an option: ")
                if choice == "1":
                    self.load_list()
                elif choice == "2":
                    self.view_groups()
                elif choice == "3":
                    self.prize_menu()
                elif choice == "4":
                    self.start_lottery()
                elif choice == "5":
                    self.export_results()
                elif choice == "0":
                    print("Goodbye!")
                    return
                else:
                    print("Invalid option, please try again.")
        except (EOFError, KeyboardInterrupt):
            print("\nExiting.")

    def _ask(self, prompt: str) -> str:
        return input(prompt).strip()

    def _ask_int(self, prompt: str, minimum: int = 1) -> int:
        while True:
            raw = self._ask(prompt)
            try:
                value = int(raw)
            except ValueError:
                print("Please enter a whole number.")
                continue
            if value < minimum:
                print(f"Please enter a number greater than or equal to {minimum}.")
                continue
            return value

    def _reset_lottery(self) -> None:
        self.lottery = None
        self.lottery_prizes = []

    # ----- list management ----------------------------------------------

    def load_list(self) -> None:
        print("1. Built-in default team building list")
        print("2. Custom txt file (format: name,department per line)")
        mode = self._ask("Select loading mode: ")
        if mode == "1":
            users, errors = self.list_manager.load_default_list()
        elif mode == "2":
            path = self._ask("Enter the txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Invalid option.")
            return

        if errors:
            print(f"Found {len(errors)} invalid line(s):")
            for error in errors:
                print(f"  {error}")
            command = self._ask(
                "Type 'skip' to ignore invalid lines and continue, or 'cancel' to abort: "
            ).lower()
            if command != "skip":
                print("Upload cancelled.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"Found {len(duplicates)} duplicate name(s):")
            for dup in duplicates:
                print(f"  {dup.name} ({dup.department})")
            command = self._ask(
                "Type 'remove' to drop duplicates (keep first occurrence), or 'cancel' to abort: "
            ).lower()
            if command != "remove":
                print("Upload cancelled.")
                return

        if not valid:
            print("No valid participants were loaded.")
            return

        self.users = valid
        self._reset_lottery()
        print(f"Loaded {len(valid)} participants.")

    def view_groups(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current department groups:")
        for department, count in info.items():
            print(f"  {department}: {count} people")
        print(f"Total: {len(self.users)} people")

    # ----- prize configuration ------------------------------------------

    def prize_menu(self) -> None:
        while True:
            print(PRIZE_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                self._add_prize_interactive()
            elif choice == "2":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    print("Some default prizes could not be added:")
                    for error in errors:
                        print(f"  {error}")
                else:
                    print("Default prize library loaded.")
                self._reset_lottery()
            elif choice == "3":
                self._show_prizes()
            elif choice == "4":
                self._reorder_prizes_interactive()
            elif choice == "0":
                return
            else:
                print("Invalid option, please try again.")

    def _add_prize_interactive(self) -> None:
        name = self._ask("Prize name: ")
        quantity = self._ask_int("Quantity: ", minimum=1)
        if self.users:
            groups = ", ".join(self.list_manager.get_group_info(self.users).keys())
            print(f"Available groups: {groups}")
        avoid_raw = self._ask("Avoid groups (comma-separated, blank for none): ")
        drawer = self._ask("Associated lottery person (blank for none): ")
        avoid_groups = [g for g in avoid_raw.split(",") if g.strip()]
        try:
            prize = self.prize_manager.add_prize(name, quantity, avoid_groups, drawer or None)
        except ValueError as exc:
            print(f"Error: {exc}")
            return
        self._reset_lottery()
        print(f"Prize added: {prize.name} (quantity {prize.quantity}).")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Prizes (lottery sequence):")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            print(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"avoid groups: {avoid} | lottery person: {drawer}"
            )

    def _reorder_prizes_interactive(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            print("At least two prizes are required to reorder.")
            return
        self._show_prizes()
        raw = self._ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(item) for item in raw.split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._reset_lottery()
            print("Lottery sequence updated.")
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ----- lottery process ----------------------------------------------

    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lottery = LotteryManager(self.users)
        self.lottery = lottery
        self.lottery_prizes = list(prizes)

        print("\n=== Lottery started ===")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(lottery, round_no, prize)

        print("\n=== Final winner list ===")
        results = lottery.result_manager.get_winners_by_prize(self.lottery_prizes)
        for prize_name, winners in results.items():
            print(f"{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(f"  {winner.user.name} ({winner.user.department})")
        print("=== Lottery finished ===")

    def _run_round(self, lottery: LotteryManager, round_no: int, prize: Prize) -> None:
        drawer = prize.associated_drawer or "Anyone"
        print(f"\n=== Round {round_no}: {prize.name} ===")
        print(f"Quantity: {prize.quantity} | Lottery person: {drawer}")
        print(f"Rules: {lottery.describe_rules(prize)}")

        round_winners: list[User] = []
        for _ in range(prize.quantity):
            if not lottery.has_eligible(prize):
                print("No eligible participants remain for this prize. Round ends.")
                break
            action = self._await_draw_command(len(round_winners) + 1)
            if action == "terminate":
                print("Round terminated. Skipping remaining draws.")
                break
            winner = lottery.draw_one(prize)
            if winner is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            round_winners.append(winner)
            print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}")

        lottery.end_round(prize)

        print(f"Round {round_no} results:")
        if round_winners:
            for index, winner in enumerate(round_winners, start=1):
                print(f"  {index}. {winner.name} ({winner.department})")
        else:
            print("  No winners in this round.")

    def _await_draw_command(self, draw_number: int) -> str:
        """Wait for the operator before the next draw.

        Returns "draw" to proceed or "terminate" to end the current round.
        """
        print(f"Currently drawing the {ordinal(draw_number)} winner...")
        while True:
            command = self._ask("[Enter]=draw | pause | terminate > ").lower()
            if command in ("", "continue"):
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    paused_command = self._ask("[paused] continue | terminate > ").lower()
                    if paused_command == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if paused_command == "terminate":
                        return "terminate"
                    print("Unknown command. Type 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    # ----- export -------------------------------------------------------

    def export_results(self) -> None:
        if self.lottery is None:
            print("No lottery results to export. Please run the lottery first.")
            return
        _, message = self.lottery.result_manager.export_to_csv(self.lottery_prizes)
        print(message)


def main() -> None:
    LotteryCLI().run()

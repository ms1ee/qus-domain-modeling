"""Core data models."""

from __future__ import annotations

from dataclasses import dataclass, field

CUSTOM_PRIZE_LEVEL = 999
_LEVEL_KEYWORDS = (("first", 1), ("second", 2), ("third", 3))


@dataclass(frozen=True)
class User:
    """A participant with name and department group."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize configured for a lottery round.

    avoid_groups: department groups whose members are drawn only after other members.
    associated_drawer: the lottery person bound to this prize (or None).
    """

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    @property
    def level(self) -> int:
        """Prize level derived from the name prefix (First=1, Second=2, Third=3).

        Custom prizes get a lower priority level (larger number)."""
        lowered = self.name.strip().lower()
        for keyword, value in _LEVEL_KEYWORDS:
            if lowered.startswith(keyword):
                return value
        return CUSTOM_PRIZE_LEVEL

"""Participant list loading, validation and group extraction."""

from __future__ import annotations

import re

from lottery_system.constants import DEFAULT_LIST_LINES
from lottery_system.models import User

_SPLIT_PATTERN = re.compile(r"[,;]")


class ListManager:
    """Loads participant lists from the built-in default or a custom txt file."""

    def __init__(self) -> None:
        self.users: list[User] = []
        self.source_mode: str | None = None
        self.file_path: str | None = None
        self.is_valid: bool = False
        self.validation_errors: list[str] = []

    def load_default_list(self) -> tuple[list[User], list[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self._update_state("default", None, users, errors)
        return users, errors

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        users, errors = self._load_from_path(path)
        self._update_state("custom", path, users, errors)
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except UnicodeDecodeError:
            return [], [f"File Error: file is not valid UTF-8 text: {path}"]
        except OSError as exc:
            return [], [f"File Error: cannot read {path} ({exc})"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            stripped = raw.strip()
            if not stripped:
                continue
            parts = [part.strip() for part in _SPLIT_PATTERN.split(stripped)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{stripped}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def _update_state(
        self,
        mode: str,
        path: str | None,
        users: list[User],
        errors: list[str],
    ) -> None:
        self.source_mode = mode
        self.file_path = path
        self.users = list(users)
        self.validation_errors = list(errors)
        self.is_valid = not errors and bool(users)

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Keep the first occurrence of each name; later repeats are duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management: presets, custom prizes, avoidance groups and ordering."""

from __future__ import annotations

from lottery_system.constants import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes (order = lottery round sequence)."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, associated_drawer=drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        groups = self._clean_groups(avoid_groups)
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        prize = Prize(name=clean_name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def set_avoid_groups(self, prize_name: str, avoid_groups: list[str] | None) -> bool:
        for prize in self._prizes:
            if prize.name == prize_name:
                prize.avoid_groups = self._clean_groups(avoid_groups)
                return True
        return False

    def reorder_prizes(self, order: list[int]) -> bool:
        """order: 1-based current positions, listed in the new order."""
        count = len(self._prizes)
        try:
            normalized = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(normalized) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in normalized]
        return True

    @staticmethod
    def _clean_groups(groups: list[str] | None) -> list[str]:
        cleaned: list[str] = []
        for group in groups or []:
            if group is None:
                continue
            text = str(group).strip()
            if text and text not in cleaned:
                cleaned.append(text)
        return cleaned

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.constants import CHEN_QIXIAN, CHENG_RU_GROUP, JIN_DUO, LI_BIN, MA_JIAQI
from lottery_system.models import Prize, User


class LotteryManager:
    """Draws winners while enforcing all lottery rules.

    last_two_draw_results: records, for the most recent draws by the lottery persons
    Chen Qixian and Jin Duo, whether the winner belonged to the Cheng Ru group.
    """

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        self.last_two_draw_results: list[bool] = []

    @property
    def winner_names(self) -> set[str]:
        return set(self._winner_names)

    def is_fallback_active(self, prize: Prize) -> bool:
        """Cheng Ru protection applies to Ma Jiaqi's draws when the last two
        Chen Qixian / Jin Duo results contain no Cheng Ru group members."""
        if prize.associated_drawer != MA_JIAQI:
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def _get_available_users(self) -> list[User]:
        return [user for user in self._users if user.name not in self._winner_names]

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Primary pool: non-winners, excluding the prize's avoidance groups.

        Special rules:
        - Li Bin: only previous winners are filtered (avoidance groups not applied).
        - Fallback active (Ma Jiaqi): pool forced to Cheng Ru group members.
        """
        available = self._get_available_users()
        if self.is_fallback_active(prize):
            forced = [user for user in available if user.department == CHENG_RU_GROUP]
            if forced:
                return forced
        if prize.associated_drawer == LI_BIN:
            return available
        avoid = set(prize.avoid_groups)
        return [user for user in available if user.department not in avoid]

    def _get_deferred_pool(self, prize: Prize) -> list[User]:
        """Avoidance-group members: still eligible, drawn only after other members."""
        if prize.associated_drawer == LI_BIN or not prize.avoid_groups:
            return []
        avoid = set(prize.avoid_groups)
        return [user for user in self._get_available_users() if user.department in avoid]

    def _get_eligible_pool_with_fallback(self, prize: Prize) -> list[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = self._get_deferred_pool(prize)
        return pool

    def draw_one(self, prize: Prize) -> User | None:
        pool = self._get_eligible_pool_with_fallback(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winner_names.add(winner.name)
        self._record_draw(prize, winner)
        return winner

    def _record_draw(self, prize: Prize, winner: User) -> None:
        drawer = prize.associated_drawer
        if drawer in (CHEN_QIXIAN, JIN_DUO):
            self.last_two_draw_results.append(winner.department == CHENG_RU_GROUP)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == MA_JIAQI:
            self.last_two_draw_results = []

"""Winner recording, grouping, ordering and CSV-style export."""

from __future__ import annotations

import csv
import io
import os
from dataclasses import dataclass
from datetime import date, datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize_name: str
    prize_level: int
    sequence: int
    win_time: datetime


def ordered_prizes(prizes: list[Prize]) -> list[Prize]:
    """Prizes sorted by level from high to low; ties keep configured order."""
    indexed = sorted(enumerate(prizes), key=lambda item: (item[1].level, item[0]))
    return [prize for _, prize in indexed]


class ResultManager:
    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=prize.level,
            sequence=len(self._winners) + 1,
            win_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        by_prize = self.get_winners_by_prize(prizes)
        buffer = io.StringIO()
        writer = csv.writer(buffer, lineterminator="\n")
        writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
        for prize in ordered_prizes(prizes):
            for winner in by_prize.get(prize.name, []):
                writer.writerow(
                    [
                        winner.user.name,
                        winner.user.department,
                        prize.name,
                        winner.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                    ]
                )
        file_name = f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                handle.write(buffer.getvalue())
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery process control: rounds, pause / continue / terminate and live display."""

from __future__ import annotations

from datetime import datetime
from typing import Callable

from lottery_system.constants import CHENG_RU_GROUP, CHEN_QIXIAN, JIN_DUO, LI_BIN, MA_JIAQI
from lottery_system.models import Prize, User
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager

CMD_DRAW = "draw"
CMD_PAUSE = "pause"
CMD_CONTINUE = "continue"
CMD_TERMINATE = "terminate"

STATUS_NOT_STARTED = "NOT_STARTED"
STATUS_RUNNING = "RUNNING"
STATUS_PAUSED = "PAUSED"
STATUS_COMPLETED = "COMPLETED"


def ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def describe_rules(prize: Prize) -> str:
    rules = ["Previous winners are excluded"]
    drawer = prize.associated_drawer
    if drawer == LI_BIN:
        rules.append("Special rule for Li Bin: only previous winners are filtered (avoidance groups not applied)")
    elif prize.avoid_groups:
        rules.append(
            "Members of avoidance groups ("
            + ", ".join(prize.avoid_groups)
            + ") are drawn only after other members"
        )
    if drawer == MA_JIAQI:
        rules.append(
            "Cheng Ru group protection: if the last two draws by "
            f"{CHEN_QIXIAN}/{JIN_DUO} had no {CHENG_RU_GROUP} winners, "
            f"this draw is limited to {CHENG_RU_GROUP} members"
        )
    if drawer:
        rules.append(f"Lottery person: {drawer}")
    return "; ".join(rules)


class LotteryProcess:
    def __init__(
        self,
        users: list[User],
        prizes: list[Prize],
        input_func: Callable[[str], str] = input,
        output_func: Callable[[str], None] = print,
    ) -> None:
        self.status: str = STATUS_NOT_STARTED
        self.current_round_index: int = 0
        self.start_time: datetime | None = None
        self.commands: list[tuple[str, datetime]] = []
        self.prizes: list[Prize] = list(prizes)
        self.lottery = LotteryManager(users)
        self.results = ResultManager()
        self._ask = input_func
        self._say = output_func

    def run(self) -> ResultManager:
        self.status = STATUS_RUNNING
        self.start_time = datetime.now()
        for index, prize in enumerate(self.prizes, start=1):
            self.current_round_index = index
            self._run_round(index, prize)
        self.status = STATUS_COMPLETED
        return self.results

    def _run_round(self, index: int, prize: Prize) -> None:
        self._say("")
        self._say(f"=== Round {index}: {prize.name} ===")
        self._say(f"Quantity: {prize.quantity}")
        self._say(f"Rules: {describe_rules(prize)}")
        round_winners: list[User] = []
        drawn = 0
        while drawn < prize.quantity:
            action = self._await_action()
            if action == CMD_TERMINATE:
                self._say("Round terminated; skipping to the next round.")
                break
            position = drawn + 1
            self._say(f"Currently drawing the {ordinal(position)} winner...")
            winner = self.lottery.draw_one(prize)
            if winner is None:
                self._say("No eligible participants remain for this prize. Round ends.")
                break
            self.results.add_winner(winner, prize)
            round_winners.append(winner)
            drawn += 1
            self._say(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}")
        self._say(f"Round {index} results:")
        if not round_winners:
            self._say("  (no winners)")
        for position, user in enumerate(round_winners, start=1):
            self._say(f"  {position}. {user.name} ({user.department})")

    def _await_action(self) -> str:
        while True:
            command = self._ask(
                "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
            ).strip().lower()
            if command == "":
                return CMD_DRAW
            if command == CMD_TERMINATE:
                self._record(CMD_TERMINATE)
                return CMD_TERMINATE
            if command == CMD_CONTINUE:
                self._say("Lottery is not paused.")
                continue
            if command == CMD_PAUSE:
                self._record(CMD_PAUSE)
                self.status = STATUS_PAUSED
                self._say("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("> ").strip().lower()
                    if resume == CMD_CONTINUE:
                        self._record(CMD_CONTINUE)
                        self.status = STATUS_RUNNING
                        self._say("Lottery resumed.")
                        break
                    if resume == CMD_TERMINATE:
                        self._record(CMD_TERMINATE)
                        self.status = STATUS_RUNNING
                        return CMD_TERMINATE
                    self._say("Lottery is paused. Type 'continue' or 'terminate'.")
                continue
            self._say("Unknown command. Press Enter to draw, or type 'pause' / 'terminate'.")

    def _record(self, command: str) -> None:
        self.commands.append((command, datetime.now()))

        self._say = output_func

        self.prize_manager = PrizeManager()

        self.results: ResultManager | None = None

        self._say("=== Company Team Building Smart Lottery System ===")

            self._print_main_menu()
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self._load_default_list()
            elif choice == "2":
                self._upload_custom_list()
            elif choice == "3":
                self._view_groups()
            elif choice == "4":
                self._prize_menu()
            elif choice == "5":
                self._start_lottery()
            elif choice == "6":
                self._view_results()
            elif choice == "7":
                self._export_results()
            elif choice == "0":
                self._say("Goodbye.")
                return
            else:
                self._say("Invalid option, please try again.")

    def _print_main_menu(self) -> None:
        self._say("")
        self._say("---------------- Main Menu ----------------")
        self._say("1. Load built-in team building list")
        self._say("2. Upload custom txt list")
        self._say("3. View department groups")
        self._say("4. Prize configuration")
        self._say("5. Start lottery")
        self._say("6. View winner results")
        self._say("7. Export results to file")

        self._finalize_users(users, errors)

    def _finalize_users(self, users: list[User], errors: list[str]) -> None:
        if not users:
            self._say("Upload failed: no valid data loaded.")
            for error in errors:
                self._say(f"  {error}")
            return
        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            self._say("Verification found problems:")
            for error in errors:
                self._say(f"  {error}")
            for duplicate in duplicates:
                self._say(f"  Duplicate Name: {duplicate.name} ({duplicate.department})")
            choice = self._ask(
                "Enter 'r' to remove invalid/duplicate data and keep valid entries, "
                "or 'q' to terminate the upload: "
            ).strip().lower()
            if choice != "r":
                self._say("Upload terminated. Previous list unchanged.")
                return
        self.users = valid
        self._say(f"Loaded {len(self.users)} participants.")
        self._view_groups()

    def _view_groups(self) -> None:
        if not self.users:
            self._say("No participant list loaded.")
            return
        groups = self.list_manager.get_group_info(self.users)
        self._say("Department groups:")
        for index, (group, count) in enumerate(groups.items(), start=1):
            self._say(f"  {index}. {group}: {count} people")

            if choice == "1":
                errors = self.prize_manager.load_default_prizes()

        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return prizes
        self._say("Current lottery sequence:")
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self._say(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"lottery person: {drawer} | avoidance groups: {groups}"
            )
        return prizes

    def _add_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity_text = self._ask("Quantity: ").strip()
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._say("Quantity must be a positive integer.")
            return
        groups = self._select_groups("Avoidance groups (numbers or names, comma separated; blank for none): ")
        drawer = self._ask("Associated lottery person (blank for none): ").strip()
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer or None)
        except ValueError as exc:
            self._say(f"Failed to add prize: {exc}")
            return
        self._say(f"Prize added: {prize.name} (quantity {prize.quantity}).")

    def _set_prize_avoid_groups(self) -> None:
        prizes = self._view_prizes()
        if not prizes:
            return
        text = self._ask("Select prize number: ").strip()
        if not text.isdigit() or not (1 <= int(text) <= len(prizes)):
            self._say("Invalid prize number.")
            return
        prize = prizes[int(text) - 1]
        groups = self._select_groups("Avoidance groups (numbers or names, comma separated; blank for none): ")
        if self.prize_manager.set_avoid_groups(prize.name, groups):
            self._say(f"Avoidance groups updated for {prize.name}.")
        else:
            self._say("Failed to update avoidance groups.")

    def _reorder_prizes(self) -> None:
        prizes = self._view_prizes()
        if not prizes:
            return
        text = self._ask(
            "Enter the new sequence as current serial numbers separated by spaces (e.g. 2 1 3): "
        )
        tokens = text.replace(",", " ").split()
        try:
            order = [int(token) for token in tokens]
        except ValueError:
            self._say("Invalid input: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._say("Lottery sequence updated.")
            self._view_prizes()
        else:
            self._say("Invalid order: each serial number must appear exactly once.")

    def _select_groups(self, prompt: str) -> list[str]:
        group_names = list(self.list_manager.get_group_info(self.users).keys()) if self.users else []
        if group_names:
            self._say("Available groups:")
            for index, group in enumerate(group_names, start=1):
                self._say(f"  {index}. {group}")
        text = self._ask(prompt).strip()
        selected: list[str] = []
        for token in re.split(r"[,，]", text):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(group_names):
                name = group_names[int(token) - 1]
            elif token in group_names:
                name = token
            else:
                self._say(f"  Ignored unknown group: {token}")
                continue
            if name not in selected:
                selected.append(name)
        return selected

    # ---------- lottery ----------
    def _start_lottery(self) -> None:
        if not self.users:
            self._say("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("Please configure prizes first.")
            return
        confirm = self._ask("Start the lottery now? (y/n): ").strip().lower()
        if confirm != "y":
            self._say("Lottery cancelled.")
            return
        process = LotteryProcess(self.users, prizes, input_func=self._ask, output_func=self._say)
        results = process.run()
        self.results = results
        self.result_prizes = prizes
        self._say("")
        self._say("Lottery finished. Final winner list:")
        self._print_summary()

    # ---------- results ----------
    def _view_results(self) -> None:
        if self.results is None:
            self._say("No lottery results yet.")
            return
        self._print_summary()

    def _print_summary(self) -> None:
        if self.results is None:
            return
        by_prize = self.results.get_winners_by_prize(self.result_prizes)
        for prize in ordered_prizes(self.result_prizes):
            self._say(f"{prize.name}:")
            winners = by_prize.get(prize.name, [])
            if not winners:
                self._say("  (no winners)")
            for position, winner in enumerate(winners, start=1):
                self._say(f"  {position}. {winner.user.name} ({winner.user.department})")

    def _export_results(self) -> None:
        if self.results is None:
            self._say("No lottery results to export.")
            return
        _, message = self.results.export_to_csv(self.result_prizes)
        self._say(message)


def main() -> None:
    try:
        LotteryCLI().run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting the lottery system.")

"""Global variable store shared across the simulator modules.

The store is a simple process-wide dictionary. Values are accessed with
:func:`set_value` and :func:`get_value`. Calling :func:`_init` resets every
key to its default value.
"""

from constants import INITIAL_FUNDS

_store = {}


def _init():
    """Reset the global store to the default system parameters.

    Keys initialised:
        funds (int): system fund balance.
        riders (list[Rider]): active rider list.
        rider_id_counter (int): last issued rider id.
        running (bool): main loop control flag.
        tick (int): current simulated time.
        orders (list[Order]): full order history.
        pending_orders (list[Order]): orders waiting for a rider.
    """
    _store.clear()
    _store.update({
        "funds": INITIAL_FUNDS,
        "riders": [],
        "rider_id_counter": 0,
        "running": True,
        "tick": 0,
        "orders": [],
        "pending_orders": [],
    })


def set_value(key, value):
    """Store ``value`` under ``key``.

    Args:
        key: Hashable identifier of the global value.
        value: Any object to store.
    """
    _store[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key``.

    Args:
        key: Identifier of the global value.
        default: Value returned when ``key`` is missing.

    Returns:
        The stored value, or ``default``.
    """
    return _store.get(key, default)

"""Grid-based path planning for riders on the 8x8 street grid.

Movement rules (evaluated in order each step):
    1. Directly above/below the target (x equal): move vertically only.
    2. On the target row (y equal): move horizontally only.
    3. On a horizontal street (y is a multiple of 100): move horizontally.
    4. On a vertical street (x is a multiple of 100): move vertically.
    5. Inside a block: first move vertically to the nearest horizontal street.

Streets are preferred; the only off-street movement is the final approach
to a cell-centre target, which is the special "directly above or below" case.
"""

from constants import CELL_SIZE, MAP_SIZE


class PathPlanner:
    """Stateless helper that computes rider movement along the street grid."""

    @staticmethod
    def is_on_street(x, y):
        """Return True when the point lies on a street (grid line).

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            bool: True if x or y is a multiple of the cell size.
        """
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def nearest_street(value):
        """Return the nearest street coordinate to ``value`` within the map.

        Args:
            value: A coordinate along one axis.

        Returns:
            int: Nearest multiple of CELL_SIZE in the range [0, MAP_SIZE].
        """
        street = int(round(value / CELL_SIZE)) * CELL_SIZE
        return min(max(street, 0), MAP_SIZE)

    @staticmethod
    def _step(value, target, speed):
        """Move ``value`` toward ``target`` by at most ``speed``, never overshooting."""
        delta = target - value
        if abs(delta) <= speed:
            return target
        return value + speed if delta > 0 else value - speed

    @classmethod
    def next_position(cls, x, y, tx, ty, speed):
        """Compute the position after one tick of movement toward a target.

        Args:
            x: Current X coordinate.
            y: Current Y coordinate.
            tx: Target X coordinate.
            ty: Target Y coordinate.
            speed: Maximum units to move this tick (>= 1).

        Returns:
            tuple[int, int]: The new (x, y) position.

        Raises:
            ValueError: If speed is less than 1.
        """
        if speed < 1:
            raise ValueError("speed must be at least 1 unit per tick")
        if x == tx and y == ty:
            return x, y
        # 1. Directly above or below the target: vertical approach.
        if x == tx:
            return x, cls._step(y, ty, speed)
        # 2. On the target row: horizontal approach.
        if y == ty:
            return cls._step(x, tx, speed), y
        # 3. On a horizontal street: travel along it.
        if y % CELL_SIZE == 0:
            return cls._step(x, tx, speed), y
        # 4. On a vertical street: travel along it.
        if x % CELL_SIZE == 0:
            return x, cls._step(y, ty, speed)
        # 5. Inside a block: reach the nearest horizontal street first.
        return x, cls._step(y, cls.nearest_street(y), speed)

    @classmethod
    def project_route(cls, start, stops, speed, max_steps=5000):
        """Project the tick-by-tick route through a sequence of stops.

        Args:
            start: (x, y) starting position.
            stops: Iterable of (x, y) destinations visited in order.
            speed: Units per tick.
            max_steps: Safety limit on the number of simulated steps per leg.

        Returns:
            tuple[list[tuple[int, int]], int]: The visited points (including
            the start) and the number of ticks required.

        Raises:
            RuntimeError: If a leg cannot be completed within ``max_steps``.
        """
        x, y = start
        points = [(x, y)]
        ticks = 0
        for tx, ty in stops:
            steps = 0
            while (x, y) != (tx, ty):
                x, y = cls.next_position(x, y, tx, ty, speed)
                points.append((x, y))
                ticks += 1
                steps += 1
                if steps > max_steps:
                    raise RuntimeError("route projection did not converge")
        return points, ticks

"""Order data model and coordinate conversion."""

import itertools

import globalvar
from constants import (
    CELL_SIZE,
    GRID_DIM,
    MAP_SIZE,
    ORDER_ASSIGNED,
    ORDER_COMPLETED,
    ORDER_PENDING,
)

_ORDER_IDS = itertools.count(1)


def validate_point(x, y):
    """Validate that ``x`` and ``y`` are integers inside the map.

    Args:
        x: X coordinate.
        y: Y coordinate.

    Raises:
        ValueError: If a value is not an integer or is outside 0..MAP_SIZE.
    """
    for value in (x, y):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError("coordinates must be integers")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"coordinate {value} is out of range 0-{MAP_SIZE}")


def to_cell_center(x, y):
    """Convert an input coordinate to the centre point of its grid cell.

    Uses ``center = (value // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2``.
    Coordinates equal to MAP_SIZE belong to the last cell.

    Args:
        x: X coordinate in 0..MAP_SIZE.
        y: Y coordinate in 0..MAP_SIZE.

    Returns:
        tuple[int, int]: Cell centre (cx, cy).
    """
    validate_point(x, y)
    cx = min(x // CELL_SIZE, GRID_DIM - 1) * CELL_SIZE + CELL_SIZE // 2
    cy = min(y // CELL_SIZE, GRID_DIM - 1) * CELL_SIZE + CELL_SIZE // 2
    return cx, cy


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id (int): Unique order identifier.
        res_pos (tuple[int, int]): Restaurant cell centre.
        cus_pos (tuple[int, int]): Customer cell centre.
        input_res (tuple[int, int]): Restaurant coordinate as entered.
        input_cus (tuple[int, int]): Customer coordinate as entered.
        status (str): One of pending/assigned/picking_up/delivering/completed.
        rider_id (int | None): Assigned rider identifier.
        start_time (int): Tick when the order was created.
        pickup_time (int | None): Tick when the rider reached the restaurant.
        completion_time (int | None): Tick when delivery completed.
        distance (int): Manhattan distance between restaurant and customer centres.
        time_taken (int | None): Ticks from creation to completion.
    """

    def __init__(self, res_pos, cus_pos):
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: Sequence [restaurant_x, restaurant_y] (0-800).
            cus_pos: Sequence [customer_x, customer_y] (0-800).

        Raises:
            ValueError: If the input is malformed or out of range.
        """
        res = self._validate_pair(res_pos, "restaurant")
        cus = self._validate_pair(cus_pos, "customer")
        self.id = next(_ORDER_IDS)
        self.input_res = res
        self.input_cus = cus
        self.res_pos = to_cell_center(*res)
        self.cus_pos = to_cell_center(*cus)
        self.status = ORDER_PENDING
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.pickup_time = None
        self.completion_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))
        self.time_taken = None

    @staticmethod
    def _validate_pair(pos, label):
        """Validate an (x, y) pair and return it as a tuple of ints."""
        try:
            x, y = pos
        except (TypeError, ValueError):
            raise ValueError(f"{label} location must contain exactly two values")
        validate_point(x, y)
        return int(x), int(y)

    def assign_to(self, rider_id):
        """Record the rider this order is assigned to.

        The caller is responsible for appending the order to ``rider.orders``.

        Args:
            rider_id: Identifier of the assigned rider.
        """
        self.rider_id = rider_id
        if self.status == ORDER_PENDING:
            self.status = ORDER_ASSIGNED

    def mark_picked_up(self, time):
        """Record arrival at the restaurant.

        Args:
            time: Current simulation tick.
        """
        from constants import ORDER_DELIVERING
        self.pickup_time = time
        self.status = ORDER_DELIVERING

    def mark_completed(self, time):
        """Record delivery completion and compute the time taken.

        Args:
            time: Current simulation tick.
        """
        self.completion_time = time
        self.time_taken = max(time - self.start_time, 0)
        self.status = ORDER_COMPLETED

    @property
    def efficiency(self):
        """Delivery efficiency in distance units per tick, or None if unfinished."""
        if not self.time_taken:
            return None
        return round(self.distance / self.time_taken, 2)

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.input_res),
            "customer": list(self.input_cus),
            "restaurant_center": list(self.res_pos),
            "customer_center": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

    def __repr__(self):
        return f"Order(id={self.id}, status={self.status}, rider={self.rider_id})"

"""Rider model: state machine, movement and event logging."""

import logging

import globalvar
from constants import (
    MAX_BLOCKED_WAIT,
    MAP_SIZE,
    ORDER_ASSIGNED,
    ORDER_PICKING_UP,
    RIDER_SPEED,
    RIDER_START_X,
    RIDER_START_Y,
    STATE_DELIVERING,
    STATE_IDLE,
    STATE_PICKING_UP,
)
from path_planner import PathPlanner

logger = logging.getLogger(__name__)


class Rider:
    """A delivery rider that moves along the street grid and serves orders.

    Attributes:
        rider_id (int): Unique identifier.
        x (int), y (int): Current position.
        speed (int): Units moved per tick.
        state (int): 0 idle, 1 picking up, 2 delivering.
        orders (list[Order]): Assigned orders; index 0 is the current order.
        special_case (bool): Flag set while the rider is waiting on a blocked tile.
        log (list[dict]): Status change history. Each entry has keys
            "event", "order_id", "time", "rider_id", "x", "y", "state".
        distance_travelled (int): Total Manhattan distance moved.
        wait_ticks (int): Consecutive ticks spent waiting on a blocked tile.
    """

    def __init__(self, rider_id, x=RIDER_START_X, y=RIDER_START_Y, speed=RIDER_SPEED):
        """Create a new idle rider and log its spawn event.

        Args:
            rider_id: Unique integer identifier.
            x: Initial X coordinate (default 435).
            y: Initial Y coordinate (default 385).
            speed: Units per tick (default 10).

        Raises:
            ValueError: If the identifier or speed is invalid.
        """
        if isinstance(rider_id, bool) or not isinstance(rider_id, int):
            raise ValueError("rider_id must be an integer")
        if speed < 1:
            raise ValueError("speed must be at least 1")
        self.rider_id = rider_id
        self.x = int(x)
        self.y = int(y)
        self.speed = int(speed)
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.distance_travelled = 0
        self.wait_ticks = 0
        self._log("spawn", None)

    def __repr__(self):
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state}, orders={len(self.orders)})")

    @property
    def position(self):
        """Current (x, y) position as a tuple."""
        return self.x, self.y

    def current_order(self):
        """Return the order currently being served, or None."""
        return self.orders[0] if self.orders else None

    def current_target(self):
        """Return the target point for the rider's current state, or None.

        Idle riders have no target. Picking-up riders head to the restaurant;
        delivering riders head to the customer.
        """
        order = self.current_order()
        if order is None or self.state == STATE_IDLE:
            return None
        if self.state == STATE_PICKING_UP:
            return order.res_pos
        if self.state == STATE_DELIVERING:
            return order.cus_pos
        return None

    def _now(self):
        return globalvar.get_value("tick", 0)

    def _log(self, event, order_id):
        """Append a status-change record to the rider log."""
        entry = {
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s event=%s order=%s", self.rider_id, event, order_id)

    def _other_rider_at(self, x, y):
        """Return True if another rider occupies (x, y)."""
        for other in globalvar.get_value("riders", []) or []:
            if other is not self and other.x == x and other.y == y:
                return True
        return False

    def _optimize_queue(self):
        """Greedily reorder not-yet-picked-up orders by nearest restaurant.

        This is the multi-destination optimisation: starting from the rider's
        current position, the nearest restaurant is served first, then the
        next nearest from there.
        """
        pending = [o for o in self.orders if o.status == ORDER_ASSIGNED]
        if len(pending) <= 1:
            return
        in_progress = [o for o in self.orders if o.status != ORDER_ASSIGNED]
        ordered = []
        cx, cy = self.x, self.y
        remaining = list(pending)
        while remaining:
            nxt = min(
                remaining,
                key=lambda o: (abs(o.res_pos[0] - cx) + abs(o.res_pos[1] - cy), o.id),
            )
            remaining.remove(nxt)
            ordered.append(nxt)
            cx, cy = nxt.res_pos
        self.orders = in_progress + ordered

    def _start_pickup(self):
        """Transition Idle -> Picking Up for the next queued order."""
        self._optimize_queue()
        self.state = STATE_PICKING_UP
        order = self.orders[0]
        order.status = ORDER_PICKING_UP
        self._log("accepted_order", order.id)

    def _step_towards(self, target):
        """Advance one tick toward ``target`` with blocked-tile handling."""
        tx, ty = target
        nx, ny = PathPlanner.next_position(self.x, self.y, tx, ty, self.speed)
        if (nx, ny) == (self.x, self.y):
            return
        if self._other_rider_at(nx, ny) and self.wait_ticks < MAX_BLOCKED_WAIT:
            # Conflict avoidance: yield the tile for a few ticks.
            self.wait_ticks += 1
            self.special_case = True
            return
        # Deadlock avoidance: after MAX_BLOCKED_WAIT ticks the rider proceeds.
        self.special_case = (nx == tx and ny != ty) or (ny == ty and nx != tx)
        self.wait_ticks = 0
        self.distance_travelled += abs(nx - self.x) + abs(ny - self.y)
        self.x, self.y = nx, ny
        self.x = min(max(self.x, 0), MAP_SIZE)
        self.y = min(max(self.y, 0), MAP_SIZE)

    def _handle_arrival(self):
        """Handle reaching the current target (restaurant or customer)."""
        order = self.current_order()
        if order is None:
            self.state = STATE_IDLE
            return
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.mark_picked_up(self._now())
            self._log("arrived_at_merchant", order.id)
        elif self.state == STATE_DELIVERING:
            self.orders.pop(0)
            order.mark_completed(self._now())
            self.state = STATE_IDLE
            self._log("delivered_to_customer", order.id)

    def move(self):
        """Advance the rider by one tick.

        Handles the Idle -> Picking Up transition when an order is queued,
        moves toward the current target, and performs the Picking Up ->
        Delivering and Delivering -> Idle transitions on arrival.
        """
        if self.state == STATE_IDLE:
            if not self.orders:
                return
            self._start_pickup()
        target = self.current_target()
        if target is None:
            return
        self._step_towards(target)
        if (self.x, self.y) == target:
            self._handle_arrival()

"""Order assignment and order management.

Provides :func:`find_best_rider_for_order` (closest-rider assignment with a
load-balancing penalty), :func:`assign_order`, and :class:`OrderManager`,
which validates input, dispatches orders, recruits riders and tracks
pending orders.
"""

import logging
import math

import globalvar
from constants import LOAD_PENALTY, ORDER_COMPLETED
from order_model import Order

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Select the best rider for an order.

    The score of each rider is the Euclidean distance from the rider to the
    restaurant plus LOAD_PENALTY for every order already queued on that rider.
    The lowest score wins; ties go to the lower rider_id. This favours the
    closest rider while balancing load among competing riders.

    Args:
        order: The Order to assign.

    Returns:
        Rider | None: The chosen rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders", []) or []
    best = None
    best_score = None
    for rider in riders:
        distance = math.hypot(rider.x - order.res_pos[0], rider.y - order.res_pos[1])
        score = distance + LOAD_PENALTY * len(rider.orders)
        if (best is None or score < best_score
                or (score == best_score and rider.rider_id < best.rider_id)):
            best = rider
            best_score = score
    return best


def assign_order(order, rider):
    """Assign ``order`` to ``rider`` by appending it to the rider's order list.

    Args:
        order: Order to assign.
        rider: Rider receiving the order.
    """
    rider.orders.append(order)
    order.assign_to(rider.rider_id)
    logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)


class OrderManager:
    """Creates, dispatches and tracks orders.

    Args:
        system_manager: The SystemManager used for rider recruitment.
    """

    def __init__(self, system_manager):
        self.system = system_manager

    @staticmethod
    def pending_list():
        """Return the live list of pending (unassigned) orders."""
        return globalvar.get_value("pending_orders", [])

    @staticmethod
    def order_history():
        """Return the live list of all orders ever created."""
        return globalvar.get_value("orders", [])

    def create_order(self, coords):
        """Create and register an order from four integer coordinates.

        Args:
            coords: Sequence [restaurant_x, restaurant_y, customer_x, customer_y].

        Returns:
            Order: The new order.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        values = list(coords)
        if len(values) != 4:
            raise ValueError("an order needs exactly four coordinates")
        order = Order(values[0:2], values[2:4])
        self.order_history().append(order)
        logger.info("Order %s created", order.id)
        return order

    def _try_assign(self, order):
        """Attempt to recruit if needed, pick a rider and assign the order.

        Returns:
            dict | None: Result information, or None if no rider is available.
        """
        recruited = self.system.try_recruit(len(self.pending_list()))
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        if order in self.pending_list():
            self.pending_list().remove(order)
        assign_order(order, rider)
        return {
            "order": order,
            "rider": rider,
            "recruited": recruited,
        }

    def dispatch(self, order):
        """Queue a new order and try to assign it immediately.

        Args:
            order: The Order to dispatch.

        Returns:
            dict: Keys "order", "rider" (None if not assigned), "recruited"
            (Rider or None) and "restaurant"/"customer" centre points.
        """
        self.pending_list().append(order)
        result = self._try_assign(order)
        if result is None:
            result = {"order": order, "rider": None, "recruited": None}
        result["restaurant"] = order.res_pos
        result["customer"] = order.cus_pos
        return result

    def process_pending(self):
        """Retry assignment for every pending order.

        Returns:
            list[dict]: Results for orders that were assigned in this call.
        """
        assigned = []
        for order in list(self.pending_list()):
            result = self._try_assign(order)
            if result is not None:
                assigned.append(result)
        return assigned

    def active_orders(self):
        """Return orders that are not yet completed."""
        return [o for o in self.order_history() if o.status != ORDER_COMPLETED]

"""System-level state management: funds, rider recruitment and simulation clock."""

import json
import logging

import globalvar

    CELL_SIZE,

    RECRUIT_COST,
    RECRUIT_MIN_FUNDS,

)

def init_system():
    """Reset global state to the initial system parameters.

    Sets funds to 1000, an empty rider list, rider id counter to 0, the
    running flag to True, and the simulated clock to 0.
    """
    globalvar._init()
    logger.info("System initialised: funds=%s", globalvar.get_value("funds"))


def load_config(path):
    """Load a JSON configuration file.

    Args:
        path: Path to the JSON file.

    Returns:
        dict: Parsed configuration, or an empty dict if the file is missing
        or invalid.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError:
        logger.warning("Config file %s not found; using defaults", path)
        return {}
    except (OSError, json.JSONDecodeError) as exc:
        logger.error("Could not read config %s: %s", path, exc)
        return {}
    if not isinstance(data, dict):
        logger.error("Config %s must contain a JSON object", path)
        return {}
    return data


def apply_config(config):
    """Apply supported configuration keys to the global store.

    Supported keys:
        initial_funds (int): starting fund balance.

    Args:
        config: Dictionary returned by :func:`load_config`.
    """
    funds = config.get("initial_funds")
    if isinstance(funds, int) and not isinstance(funds, bool):
        globalvar.set_value("funds", funds)
        logger.info("Initial funds set from config: %s", funds)


class SystemManager:
    """Manages funds, riders, recruitment, the simulation clock and persistence."""

    # Funds ----------------------------------------------------------------
    @property
    def funds(self):
        """Current system fund balance (may be negative)."""
        return globalvar.get_value("funds", 0)

    def _set_funds(self, value):
        globalvar.set_value("funds", int(value))

    # Riders ---------------------------------------------------------------
    @staticmethod
    def rider_list():
        """Return the live rider list."""
        return globalvar.get_value("riders", [])

    def idle_riders(self):
        """Return riders whose state is idle."""
        return [r for r in self.rider_list() if r.state == STATE_IDLE]

    def create_rider(self):
        """Create a new rider with an auto-incremented id and add it to the list.

        Returns:
            Rider: The newly created rider.
        """
        rider_id = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", rider_id)
        rider = Rider(rider_id)
        self.rider_list().append(rider)
        logger.info("Rider %s recruited", rider_id)
        return rider

    def try_recruit(self, pending_count):
        """Recruit a rider when pending orders exceed idle riders and funds allow.

        Args:
            pending_count: Number of orders currently waiting for assignment.

        Returns:
            Rider | None: The new rider, or None if no recruitment happened.
        """
        if pending_count <= len(self.idle_riders()):
            return None
        if self.funds < RECRUIT_MIN_FUNDS:
            logger.debug("Recruitment skipped: funds %s below %s",
                         self.funds, RECRUIT_MIN_FUNDS)
            return None
        self._set_funds(self.funds - RECRUIT_COST)
        return self.create_rider()

    # Simulation -----------------------------------------------------------
    @staticmethod
    def current_tick():
        """Return the current simulated tick."""
        return globalvar.get_value("tick", 0)

    def tick(self):
        """Advance the simulation by one unit of time.

        Every rider moves once. Exceptions raised by a rider are logged and the
        rider is reset to a consistent state so the simulation continues.

        Returns:
            int: The new tick value.
        """
        now = self.current_tick() + 1
        globalvar.set_value("tick", now)
        for rider in list(self.rider_list()):
            try:
                rider.move()
            except Exception:  # noqa: BLE001 - recovery is intentional
                logger.exception("Rider %s failed at tick %s; recovering",
                                 rider.rider_id, now)
                rider.wait_ticks = 0
                if not rider.orders:
                    rider.state = STATE_IDLE
        self.check_consistency()
        return now

    def check_consistency(self):
        """Repair common state inconsistencies.

        Removes completed orders from rider queues, forces riders without
        orders back to idle, and clamps coordinates to the map.

        Returns:
            int: Number of repairs performed.
        """
        repairs = 0
        for rider in self.rider_list():
            before = len(rider.orders)
            rider.orders = [o for o in rider.orders if o.status != ORDER_COMPLETED]
            if len(rider.orders) != before:
                repairs += 1
            if not rider.orders and rider.state != STATE_IDLE:
                rider.state = STATE_IDLE
                repairs += 1
            clamped_x = min(max(rider.x, 0), MAP_SIZE)
            clamped_y = min(max(rider.y, 0), MAP_SIZE)
            if (clamped_x, clamped_y) != (rider.x, rider.y):
                rider.x, rider.y = clamped_x, clamped_y
                repairs += 1
        if repairs:
            logger.warning("Consistency check performed %s repair(s)", repairs)
        return repairs

    # Persistence ----------------------------------------------------------
    def save_state(self, path):
        """Write system parameters, riders and order history to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        data = {
            "config": {
                "funds": self.funds,
                "recruit_cost": RECRUIT_COST,
                "recruit_min_funds": RECRUIT_MIN_FUNDS,
                "grid_cell_size": CELL_SIZE,
                "map_size": MAP_SIZE,
            },
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "tick": self.current_tick(),
            "riders": [
                {
                    "rider_id": r.rider_id,
                    "x": r.x,
                    "y": r.y,
                    "speed": r.speed,
                    "state": r.state,
                    "special_case": r.special_case,
                    "order_ids": [o.id for o in r.orders],
                    "distance_travelled": r.distance_travelled,
                }
                for r in self.rider_list()
            ],
            "orders": [o.to_dict() for o in globalvar.get_value("orders", [])],
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        logger.info("System state saved to %s", path)

class DisplayManager:
    """Formats the system status, order and rider tables, and ASCII maps."""

    # Helpers --------------------------------------------------------------
    @staticmethod
    def _to_index(value):
        """Map a coordinate to a canvas index (nearest 50-unit step)."""
        index = (value + CANVAS_STEP // 2) // CANVAS_STEP
        return min(max(index, 0), CANVAS_SIZE - 1)

    @staticmethod
    def _base_canvas():
        """Create the street canvas: even rows/columns are streets ('#')."""
        return [
            ["#" if (row % 2 == 0 or col % 2 == 0) else "." for col in range(CANVAS_SIZE)]
            for row in range(CANVAS_SIZE)
        ]

    def _place(self, canvas, x, y, symbol):
        """Place ``symbol`` at the canvas cell that corresponds to (x, y)."""
        canvas[self._to_index(y)][self._to_index(x)] = symbol

    @staticmethod
    def _canvas_text(canvas):
        """Render a canvas with coordinate labels on each row and column."""
        header = "      " + " ".join(f"{c * CANVAS_STEP % 1000:>3}" if False else " " for c in range(CANVAS_SIZE))
        lines = []
        for row_index, row in enumerate(canvas):
            label = f"{row_index * CANVAS_STEP:>4} "
            lines.append(label + " ".join(row))
        axis = "     " + " ".join(
            f"{(c * CANVAS_STEP) // 100}" if c % 2 == 0 else " "
            for c in range(CANVAS_SIZE)
        )
        return "\n".join(lines + ["x: 0..800 (100-unit ticks shown by '#' columns)", axis]) \
            if header is not None else ""

    # Textual reports ------------------------------------------------------
    @staticmethod
    def format_funds(funds):
        """Return a one-line fund balance string."""
        return f"System funds: {funds}"

        Returns:
            str: Multi-line report.
        """
        riders = list(riders)
        lines = [
            f"Tick: {tick}",
            self.format_funds(funds),
            f"Riders: {len(riders)}  Pending orders: {pending_count}",
        ]
        if not riders:
            lines.append("No riders on duty.")
        for rider in riders:
            order_ids = [o.id for o in rider.orders] or "none"
            lines.append(

                f"status={STATE_NAMES.get(rider.state, rider.state)} "

    def render_map(self, riders, orders):

        Args:
            riders: Iterable of Rider objects.
            orders: Iterable of Order objects (completed orders are hidden).

        Returns:
            str: Map text with a legend.

        canvas = self._base_canvas()
        for order in orders:

        Args:

            ticks: Projected number of ticks.

        for x, y in points:
            self._place(canvas, x, y, "*")
        self._place(canvas, *order.res_pos, "S")
        self._place(canvas, *order.cus_pos, "C")
        if points:
            self._place(canvas, *points[0], "R")
        header = [
            f"Route for order {order.id}: restaurant {order.input_res} -> "
            f"grid centre {order.res_pos}",
            f"  customer {order.input_cus} -> grid centre {order.cus_pos}",
            f"  Speed: {speed} units/tick | Projected time: {ticks} ticks | "
            f"Route length: {length} units",
        ]
        return "\n".join(header) + "\n" + self._format_canvas(canvas) + (
            "\nLegend: R=Rider S=Restaurant C=Customer *=route #=street .=empty"
        )

    @staticmethod
    def _format_canvas(canvas):
        """Format a canvas with y-axis labels and an x-axis ruler."""
        lines = []
        for row_index, row in enumerate(canvas):
            label = f"{row_index * CANVAS_STEP:>4} "
            lines.append(label + " ".join(row))
        ruler = "     " + " ".join(
            (str((col * CANVAS_STEP) // 100) if col % 2 == 0 else " ")
            for col in range(CANVAS_SIZE)
        )
        lines.append(ruler)
        lines.append("     (x axis: hundreds of units; y axis labelled in units)")
        return "\n".join(lines)

"""Command parsing, input validation and command history."""

import logging
import re

from constants import MAP_SIZE

logger = logging.getLogger(__name__)

COMMAND_ALIASES = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "run": "run",
    "h": "help",
    "?": "help",
    "hist": "history",
    "q": "quit",
    "x": "exit",
}

COMMAND_HELP = {
    "start": "start [rx ry cx cy]  Create an order (restaurant then customer coordinates, 0-800)",
    "status": "status              Show funds, riders and their current state",
    "orders": "orders              Show all orders with timing and assignment",
    "riders": "riders              Show every recorded rider status change",
    "map": "map                 Show the ASCII map (R rider, S restaurant, C customer)",
    "tick": "tick [n]             Advance the simulation by n ticks (default 1)",
    "run": "run                 Advance until all active orders are delivered",
    "history": "history             Show recent commands",
    "help": "help                Show this help",
    "quit": "quit / exit         Exit the program",
}


class InputManager:
    """Reads, parses and validates user commands.

    Args:
        reader: Callable used to read a line (default: built-in input).
        output: Callable used to print messages (default: built-in print).
        history_limit: Maximum number of commands kept in history.
    """

    def __init__(self, reader=input, output=print, history_limit=100):
        self._reader = reader
        self._output = output
        self.history = []
        self.history_limit = history_limit
        self._setup_readline()

    @staticmethod
    def _setup_readline():
        """Enable line editing and arrow-key history when readline is available."""
        try:
            import readline  # noqa: F401  (side effect: enables editing)
        except ImportError:
            logger.debug("readline not available; line editing disabled")

    def read_line(self, prompt):
        """Read a raw line from the user and strip surrounding whitespace.

        Raises:
            EOFError: When input is closed.
        """
        return self._reader(prompt).strip()

    def _remember(self, line):
        """Add a command to the history, respecting the size limit."""
        self.history.append(line)
        if len(self.history) > self.history_limit:
            del self.history[0]

    @staticmethod
    def parse_command(line):
        """Split a command line into a canonical command name and arguments.

        Args:
            line: Raw command text.

        Returns:
            tuple[str, list[str]]: Command name (lower case) and arguments.
        """
        parts = line.split()
        if not parts:
            return "", []
        word = parts[0].lower()
        return COMMAND_ALIASES.get(word, word), parts[1:]

    def read_command(self, prompt="dispatch> "):
        """Read and parse one command, supporting the '!!' repeat shortcut.

        Returns:
            tuple[str, list[str]]: Canonical command and its arguments. An
            empty command name is returned for blank input.

        Raises:
            EOFError: When input is closed.
        """
        line = self.read_line(prompt)
        if line == "!!":
            if not self.history:
                self._output("No previous command to repeat.")
                return "", []
            line = self.history[-1]
            self._output(f"Repeating: {line}")
        if line:
            self._remember(line)
        return self.parse_command(line)

    @staticmethod
    def parse_coordinates(value):
        """Parse four integer coordinates from text or a token list.

        Args:
            value: A string such as "100 200 300 400" or "100,200,300,400",
                or a sequence of tokens.

        Returns:
            list[int]: Four coordinates in the range 0..MAP_SIZE.

        Raises:
            ValueError: If the count, type or range is invalid.
        """
        if isinstance(value, str):
            tokens = [t for t in re.split(r"[\s,]+", value.strip()) if t]
        else:
            tokens = [str(t) for t in value]
        if len(tokens) != 4:
            raise ValueError(
                f"expected 4 integer coordinates (restaurant x y, customer x y), got {len(tokens)}"
            )
        coords = []
        for token in tokens:
            try:
                number = int(token)
            except ValueError:
                raise ValueError(f"'{token}' is not an integer")
            if not 0 <= number <= MAP_SIZE:
                raise ValueError(f"coordinate {number} is out of range 0-{MAP_SIZE}")
            coords.append(number)
        return coords

    def prompt_coordinates(self, max_attempts=3):
        """Interactively ask for four coordinates with retry on invalid input.

        Args:
            max_attempts: Number of invalid entries allowed before giving up.

        Returns:
            list[int] | None: Valid coordinates, or None if the user cancelled
            (blank line) or exhausted attempts.
        """
        for attempt in range(max_attempts):
            text = self.read_line(
                "Enter restaurant x y and customer x y (0-800), blank to cancel: "
            )
            if not text:
                self._output("Order cancelled.")
                return None
            try:
                return self.parse_coordinates(text)
            except ValueError as exc:
                remaining = max_attempts - attempt - 1
                self._output(f"Invalid input: {exc}.")
                if remaining:
                    self._output(f"Please try again ({remaining} attempt(s) left).")
        self._output("Too many invalid attempts; order cancelled.")
        return None

    def help_text(self):
        """Return the help text listing all commands."""
        lines = ["Available commands:"]
        lines.extend(f"  {COMMAND_HELP[name]}" for name in COMMAND_HELP)
        lines.append("Shortcuts: s=start st=status o=orders r=riders m=map t=tick "
                     "h=help q=quit, '!!' repeats the last command.")
        return "\n".join(lines)

    def history_text(self):
        """Return the numbered command history."""
        if not self.history:
            return "History is empty."
        return "\n".join(f"{i:>3}: {cmd}" for i, cmd in enumerate(self.history, 1))

Run from the project root with:

class DispatchApp:
    """Interactive application that wires the modules together.

    Args:
        input_manager: InputManager used to read commands (optional).
        config_path: Optional JSON configuration file to load at startup.
        save_path: Optional JSON file to write system state to on exit.
        output: Callable used for printing (default: print).
    """

    def __init__(self, input_manager=None, config_path=None, save_path=None,
                 output=print):
        self.out = output
        self.input = input_manager or InputManager(output=output)
        self.save_path = save_path
        self.display = DisplayManager()
        init_system()
        if config_path:
            apply_config(load_config(config_path))
        self.system = SystemManager()
        self.orders = OrderManager(self.system)
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "run": self.cmd_run,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    # Main loop --------------------------------------------------------------
    def run(self):
        """Run the interactive command loop until quit/exit or end of input."""
        self.out(BANNER)
        while globalvar.get_value("running", True):
            try:
                cmd, args = self.input.read_command()
            except (EOFError, KeyboardInterrupt):
                self.out("")
                self.cmd_quit([])
                break
            except Exception as exc:  # noqa: BLE001
                logger.exception("Input error")
                self.out(f"Error reading input: {exc}")
                continue
            if not cmd:
                continue
            self.execute(cmd, args)
        self.shutdown()

    def execute(self, cmd, args):
        """Dispatch a parsed command, guarding against unexpected errors."""
        handler = self._handlers.get(cmd)
        if handler is None:
            self.out(f"Unknown command '{cmd}'. Type 'help' for the command list.")
            return
        try:
            handler(args)
        except Exception as exc:  # noqa: BLE001 - keep the simulation alive
            logger.exception("Command %s failed", cmd)
            self.out(f"Error while running '{cmd}': {exc}. State was preserved.")

    def shutdown(self):
        """Save state when requested and report the exit."""
        if self.save_path:
            try:
                self.system.save_state(self.save_path)
                self.out(f"State saved to {self.save_path}.")
            except OSError as exc:
                self.out(f"Could not save state: {exc}")
        self.out("Goodbye.")

    # Commands ---------------------------------------------------------------
    def cmd_start(self, args):
        """Create an order, assign a rider and show the projected route."""
        if args:
            try:
                coords = self.input.parse_coordinates(args)
            except ValueError as exc:
                self.out(f"Invalid input: {exc}.")
                return
        else:
            coords = self.input.prompt_coordinates()
            if coords is None:
                return
        try:
            order = self.orders.create_order(coords)
        except ValueError as exc:
            self.out(f"Invalid order: {exc}.")
            return

        result = self.orders.dispatch(order)
        self.out(f"Order {order.id} created at tick {order.start_time}.")
        self.out(f"  Restaurant: input {order.input_res} -> grid centre {order.res_pos}")
        self.out(f"  Customer:   input {order.input_cus} -> grid centre {order.cus_pos}")
        rider = result["rider"]
        if result["recruited"] is not None:
            self.out(f"  New rider added: R{result['recruited'].rider_id} "
                     f"(recruitment cost deducted, funds now {self.system.funds}).")
        else:
            self.out("  No new rider added.")
        if rider is None:
            self.out(f"  No rider available (funds {self.system.funds}). "
                     f"Order {order.id} is pending and will be retried on each tick.")
            return
        self.out(f"  Assigned to R{rider.rider_id} "
                 f"(now at ({rider.x},{rider.y}), status: queued).")
        points, ticks = PathPlanner.project_route(
            rider.position, [order.res_pos, order.cus_pos], rider.speed)
        self.out(self.display.render_route(order, points, rider.speed, ticks))

    def cmd_status(self, args):
        """Show funds, riders and their state."""
        self.out(self.display.render_status(
            self.system.funds,
            self.system.rider_list(),
            self.system.current_tick(),

    def cmd_orders(self, args):
        """Show all orders with timing and assignment details."""
        self.out(self.display.render_orders(self.orders.order_history()))

    def cmd_riders(self, args):
        """Show every recorded rider status change."""

    def cmd_map(self, args):

                                         self.orders.order_history()))

    def _advance(self, count):
        """Advance the simulation by ``count`` ticks."""
        for _ in range(count):
            self.orders.process_pending()
            self.system.tick()

    def cmd_tick(self, args):
        """Advance the simulation by one or more ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                self.out("Usage: tick [n] where n is a positive integer.")
                return
            if not 1 <= count <= MAX_SIM_TICKS:
                self.out(f"n must be between 1 and {MAX_SIM_TICKS}.")
                return
        self._advance(count)
        self.out(f"Advanced {count} tick(s). Current tick: {self.system.current_tick()}.")

    def cmd_run(self, args):
        """Advance until every active order has been delivered."""
        start = self.system.current_tick()
        steps = 0
        while self.orders.active_orders() and steps < MAX_SIM_TICKS:
            self._advance(1)
            steps += 1
        if self.orders.active_orders():
            self.out(f"Stopped after {steps} ticks; orders still active.")
        else:

def main(argv=None):
    """Parse command-line arguments and run the application.

    Args:

    Returns:

    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", help="JSON file with system parameters")
    parser.add_argument("--save", help="JSON file to write system state to on exit")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    options = parser.parse_args(argv)

    logging.basicConfig(
        level=logging.DEBUG if options.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )

        app = DispatchApp(config_path=options.config, save_path=options.save)

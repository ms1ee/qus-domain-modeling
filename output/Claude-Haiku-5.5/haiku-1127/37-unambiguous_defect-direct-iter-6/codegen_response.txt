from __future__ import annotations

from ..defaults import DEFAULT_USER_LINES
from ..models import User


def _norm(text: str) -> str:
    return " ".join((text or "").split()).casefold()


class ListManager:
    """Loads and verifies participant lists.

    Supported custom file format: one participant per line, ``Name,Department``.
    Blank lines and lines starting with ``#`` are ignored.
    """

    def __init__(self) -> None:
        pass

    def load_default_list(self) -> tuple[list[User], list[str]]:
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "rb") as handle:
                raw = handle.read()
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        text = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                text = raw.decode(encoding)
                break
            except UnicodeDecodeError:
                continue
        if text is None:
            return [], [f"File Error: unable to decode '{path}' (please use UTF-8 or GBK)"]

        return self._parse_lines(text.splitlines())

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw_line in enumerate(lines, start=1):
            line = raw_line.strip()
            if not line or line.startswith("#"):
                continue
            parts = [part.strip() for part in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' (expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). The first occurrence of a name is kept."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = _norm(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from ..defaults import DEFAULT_PRIZES
from ..models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"Prize Error for '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")
        if any(p.name.casefold() == name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = (group or "").strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer = (associated_drawer or "").strip() or None
        prize = Prize(name=name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """order holds 1-based current positions in the desired new sequence."""
        count = len(self._prizes)
        if not isinstance(order, list) or len(order) != count:
            return False
        if not all(isinstance(i, int) and not isinstance(i, bool) for i in order):
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

"""Winner records, grouped result summary and CSV-style export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from ..models import Prize, User

_LEVEL_WORDS = (
    "first", "second", "third", "fourth", "fifth",
    "sixth", "seventh", "eighth", "ninth", "tenth",
)


def prize_level_rank(name: str) -> int:
    """Lower rank means higher prize level (e.g. 'First Prize' -> 1)."""
    for token in re.findall(r"[a-z]+", name.casefold()):
        if token in _LEVEL_WORDS:
            return _LEVEL_WORDS.index(token) + 1
    return len(_LEVEL_WORDS) + 1


@dataclass
class Winner:
    user: User
    prize_name: str


class ResultManager:
    """Keeps the winner list in winning order."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        self._winners.append(Winner(user=user, prize_name=prize.name))

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Prize names ordered from high to low level; winners in winning order."""
        indexed = sorted(
            enumerate(prizes),
            key=lambda item: (prize_level_rank(item[1].name), item[0]),
        )
        result: dict[str, list[Winner]] = {}
        for _, prize in indexed:
            if prize.name in result:
                continue
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        grouped = self.get_winners_by_prize(prizes)
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize", "Name", "Department", "Order"])
                for prize_name, winners in grouped.items():
                    for order, winner in enumerate(winners, start=1):
                        writer.writerow(
                            [prize_name, winner.user.name, winner.user.department, order]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery rule engine: filtering, special rules, fallback and anti-duplication."""

from __future__ import annotations

import random

from ..models import Prize, User
from .result_manager import ResultManager

CHENG_RU_GROUP = "Cheng Ru Group"
SPECIAL_DRAWER = "Li Bin"
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_DRAWERS = ("Chen Qixian", "Jin Duo")
HISTORY_LENGTH = 2


def _norm(text: str | None) -> str:
    return " ".join((text or "").split()).casefold()


def is_cheng_ru_member(user: User) -> bool:
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


class LotteryManager:
    """Draws winners for prizes while applying the lottery rules.

    last_two_draw_results: outcomes of the most recent individual draws made by
    Chen Qixian or Jin Duo (True if the drawn person is a Cheng Ru Group member).
    Ma Jiaqi's draws consume this history.
    """

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self.result_manager = ResultManager()
        self.last_two_draw_results: list[bool] = []

    # ----- helpers -----
    @property
    def winners(self):
        return self.result_manager.get_all_winners()

    def _available_users(self) -> list[User]:
        return [u for u in self._users if _norm(u.name) not in self._winner_keys]

    @staticmethod
    def _drawer_key(prize: Prize) -> str:
        return _norm(prize.associated_drawer)

    def _fallback_pool(self, prize: Prize, available: list[User]) -> list[User] | None:
        """Cheng Ru group protection: returns the forced pool, or None if not active."""
        if self._drawer_key(prize) != _norm(FALLBACK_DRAWER):
            return None
        if len(self.last_two_draw_results) < HISTORY_LENGTH or any(self.last_two_draw_results):
            return None
        members = [u for u in available if is_cheng_ru_member(u)]
        return members or None

    def is_fallback_active(self, prize: Prize) -> bool:
        return self._fallback_pool(prize, self._available_users()) is not None

    # ----- rule engine -----
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = self._available_users()

        fallback = self._fallback_pool(prize, available)
        if fallback is not None:
            return fallback

        if self._drawer_key(prize) == _norm(SPECIAL_DRAWER):
            # Relaxed rules for Li Bin: avoidance groups are not applied.
            return available

        avoid = {_norm(g) for g in prize.avoid_groups}
        return [u for u in available if _norm(u.department) not in avoid]

    def _record_draw(self, prize: Prize, user: User) -> None:
        self._winner_keys.add(_norm(user.name))
        self.result_manager.add_winner(user, prize)

        key = self._drawer_key(prize)
        if key in {_norm(d) for d in PROTECTED_DRAWERS}:
            self.last_two_draw_results.append(is_cheng_ru_member(user))
            del self.last_two_draw_results[:-HISTORY_LENGTH]
        elif key == _norm(FALLBACK_DRAWER):
            self.last_two_draw_results.clear()

    def draw_one(self, prize: Prize) -> User | None:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self._record_draw(prize, user)
        return user

    def describe_rules(self, prize: Prize) -> str:
        key = self._drawer_key(prize)
        parts: list[str] = []
        if key == _norm(SPECIAL_DRAWER):
            parts.append(
                "Special rule for Li Bin: relaxed filtering, avoidance groups ignored; "
                "previous winners still excluded"
            )
        else:
            parts.append("Previous winners excluded")
            if prize.avoid_groups:
                parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
            else:
                parts.append("no avoidance groups")
        if key == _norm(FALLBACK_DRAWER):
            parts.append(
                "Cheng Ru Group protection: if the last two draws by Chen Qixian/Jin Duo "
                "had no Cheng Ru Group winner, the pool is limited to Cheng Ru Group members"
            )
        if not prize.associated_drawer:
            parts.append("no specific lottery person")
        else:
            parts.append(f"lottery person: {prize.associated_drawer}")
        return "; ".join(parts)

"""Command-line interface: list setup, prize setup, lottery control and results."""

from __future__ import annotations

import queue
import sys
import threading
import time

from .models import Prize, User
from .services.list_manager import ListManager
from .services.lottery_manager import LotteryManager
from .services.prize_manager import PrizeManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class CommandReader:
    """Reads stdin in a background thread so commands can be polled during the lottery.

    A None item in the queue marks end of input.
    """

    def __init__(self) -> None:
        self._queue: queue.Queue = queue.Queue()
        self._thread = threading.Thread(target=self._run, daemon=True)
        self._thread.start()

    def _run(self) -> None:
        try:
            for line in sys.stdin:
                self._queue.put(line.rstrip("\r\n"))
        except (OSError, ValueError):
            pass
        finally:
            self._queue.put(None)

    def _take(self, line):
        if line is None:
            raise EOFError("Input closed")
        return line

    def ask(self, prompt: str = "") -> str:
        print(prompt, end="", flush=True)
        return self._take(self._queue.get())

    def wait(self) -> str:
        return self._take(self._queue.get())

    def poll(self) -> str | None:
        """Return the next queued line, or None if nothing is waiting."""
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        return self._take(item)


class LotterySystemApp:
    def __init__(self, reader: CommandReader, draw_delay: float = 0.8) -> None:
        self.reader = reader
        self.draw_delay = draw_delay
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery_manager: LotteryManager | None = None

    # ----- input helpers -----
    def _ask(self, prompt: str) -> str:
        return self.reader.ask(prompt)

    def _ask_yes_no(self, prompt: str) -> bool:
        while True:
            answer = self._ask(prompt).strip().lower()
            if answer in ("y", "yes"):
                return True
            if answer in ("n", "no"):
                return False
            print("Please answer 'y' or 'n'.")

    def _ask_choice(self, prompt: str, options: tuple[str, ...]) -> str:
        while True:
            answer = self._ask(prompt).strip().lower()
            if answer in options:
                return answer
            print("Invalid choice, please try again.")

    # ----- step 1: participant list -----
    def _setup_users(self) -> bool:
        while True:
            print("\n=== Step 1: Participant List ===")
            print("1. Load built-in default team building list")
            print("2. Upload custom txt list (Name,Department per line)")
            print("q. Quit")
            choice = self._ask("Select: ").strip().lower()

            if choice == "1":
                users, errors = self.list_manager.load_default_list()
            elif choice == "2":
                path = self._ask("Path to txt file: ").strip().strip('"').strip("'")
                users, errors = self.list_manager.load_custom_list(path)
            elif choice == "q":
                return False
            else:
                print("Invalid option.")
                continue

            verified = self._verify_list(users, errors)
            if verified is None:
                print("Upload terminated.")
                continue
            if not verified:
                print("No valid participants loaded.")
                continue

            self.users = verified
            print(f"Loaded {len(self.users)} participant(s).")
            self._show_groups()
            return True

    def _verify_list(self, users: list[User], errors: list[str]) -> list[User] | None:
        """Returns the verified user list, or None if the upload was terminated."""
        if errors:
            print(f"Found {len(errors)} format error(s):")
            for error in errors:
                print("  " + error)
            choice = self._ask_choice(
                "[r] remove invalid data and continue / [t] terminate upload: ", ("r", "t")
            )
            if choice == "t":
                return None

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  {user.name} ({user.department})")
            choice = self._ask_choice(
                "[r] remove duplicates (keep first occurrence) and continue / [t] terminate upload: ",
                ("r", "t"),
            )
            if choice == "t":
                return None
            return valid
        return valid

    def _show_groups(self) -> None:
        groups = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for group, count in groups.items():
            print(f"  {group}: {count} people")
        print(f"  Total: {sum(groups.values())} people")

    # ----- step 2: prizes -----
    def _setup_prizes(self) -> None:
        print("\n=== Step 2: Prize Configuration ===")
        if self._ask_yes_no("Load preset prizes? (y/n): "):
            for error in self.prize_manager.load_default_prizes():
                print("  " + error)

        while not self.prize_manager.get_prizes() or self._ask_yes_no(
            "Add a custom prize? (y/n): "
        ):
            self._add_custom_prize()

        self._show_prizes()
        self._configure_order()

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        if not name:
            print("Prize name cannot be empty.")
            return

        while True:
            raw = self._ask("Quantity: ").strip()
            if raw.isdigit() and int(raw) > 0:
                quantity = int(raw)
                break
            print("Quantity must be a positive integer.")

        groups = self._select_avoid_groups()
        drawer = self._ask("Associated lottery person (Enter to skip): ").strip() or None

        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Cannot add prize: {exc}")

    def _select_avoid_groups(self) -> list[str]:
        groups = sorted(self.list_manager.get_group_info(self.users).keys())
        print("Available department groups:")
        for index, group in enumerate(groups, start=1):
            print(f"  {index}. {group}")
        raw = self._ask("Avoid groups (numbers or names, comma-separated; Enter for none): ").strip()
        if not raw:
            return []

        selected: list[str] = []
        for token in raw.split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                group = groups[int(token) - 1]
            else:
                group = next((g for g in groups if g.casefold() == token.casefold()), None)
                if group is None:
                    print(f"Unknown group '{token}', ignored.")
                    continue
            if group not in selected:
                selected.append(group)
        return selected

    def _show_prizes(self) -> None:
        print("Current round order:")
        for index, prize in enumerate(self.prize_manager.get_prizes(), start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            print(f"  {index}. {prize.name} | qty {prize.quantity} | avoid: {avoid} | drawer: {drawer}")

    def _configure_order(self) -> None:
        while True:
            raw = self._ask(
                "Enter new round order as serial numbers (e.g. 2 1 3), or Enter to keep: "
            ).strip()
            if not raw:
                return
            try:
                order = [int(token) for token in raw.replace(",", " ").split()]
            except ValueError:
                print("Please enter integers only.")
                continue
            if self.prize_manager.reorder_prizes(order):
                print("Round order updated.")
                self._show_prizes()
                return
            print("Invalid order: each serial number must appear exactly once.")

    # ----- step 3: lottery -----
    def _handle_commands(self) -> str | None:
        """Process queued operator commands. Returns 'terminate' to skip the round."""
        while True:
            line = self.reader.poll()
            if line is None:
                return None
            command = line.strip().lower()
            if not command:
                continue
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to skip this round.")
                while True:
                    reply = self.reader.wait().strip().lower()
                    if reply == "continue":
                        print("Lottery resumed.")
                        break
                    if reply == "terminate":
                        return "terminate"
                    print("While paused, type 'continue' or 'terminate'.")
                continue
            if command == "terminate":
                return "terminate"
            if command == "continue":
                print("Lottery is not paused.")
                continue
            print(f"Unknown command '{line.strip()}'. Available: pause, continue, terminate.")

    def _run_round(self, index: int, prize: Prize, total: int) -> None:
        manager = self.lottery_manager
        assert manager is not None
        print(f"\n=== Round {index}/{total}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {manager.describe_rules(prize)}")

        won: list[User] = []
        for pick in range(1, prize.quantity + 1):
            if self._handle_commands() == "terminate":
                print("Round terminated by operator; skipping remaining picks.")
                break

            print(f"Currently drawing the {ordinal(pick)} winner...")
            time.sleep(self.draw_delay)

            if self._handle_commands() == "terminate":
                print("Round terminated by operator; skipping remaining picks.")
                break

            if manager.is_fallback_active(prize):
                print("  [Cheng Ru Group protection active: pool limited to Cheng Ru Group members]")
            user = manager.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize.")
                break
            won.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        print(f"--- Round {index} results ({len(won)}/{prize.quantity}) ---")
        for number, user in enumerate(won, start=1):
            print(f"  {number}. {user.name} ({user.department})")

    def _run_lottery(self) -> None:
        if not self._ask_yes_no("Start the lottery now? (y/n): "):
            return

        self.lottery_manager = LotteryManager(self.users)
        prizes = self.prize_manager.get_prizes()
        print("\nLottery started. Commands: pause / continue / terminate")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, prize, len(prizes))
        print("\nAll rounds completed.")

    def _show_final_results(self) -> None:
        assert self.lottery_manager is not None
        prizes = self.prize_manager.get_prizes()
        grouped = self.lottery_manager.result_manager.get_winners_by_prize(prizes)
        print("\n=== Final Results ===")
        for prize_name, winners in grouped.items():
            print(prize_name)
            if not winners:
                print("  (no winners)")
            for number, winner in enumerate(winners, start=1):
                print(f"  {number}. {winner.user.name} ({winner.user.department})")

        if self._ask_yes_no("Export winner list to file? (y/n): "):
            _, message = self.lottery_manager.result_manager.export_to_csv(prizes)[0:1] + (
                self.lottery_manager.result_manager.export_to_csv(prizes)[1],
            )
            print(message)

    # ----- main flow -----
    def run(self) -> int:
        print("=== Company Team Building Smart Lottery System ===")
        if not self._setup_users():
            print("Goodbye.")
            return 0
        self._setup_prizes()
        self._run_lottery()
        if self.lottery_manager is not None:
            self._show_final_results()
        print("Goodbye.")
        return 0

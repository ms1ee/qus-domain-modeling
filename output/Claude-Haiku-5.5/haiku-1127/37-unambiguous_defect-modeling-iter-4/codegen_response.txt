def _read_file_lines(path: str) -> tuple[list[str] | None, str | None]:
    """Read file lines trying UTF-8 (with BOM) first, then GBK."""
    for encoding in ("utf-8-sig", "gbk"):
        try:
            with open(path, "r", encoding=encoding) as handle:
                return handle.read().splitlines(), None
        except UnicodeDecodeError:
            continue
        except OSError as exc:
            return None, f"File Error: cannot read '{path}': {exc}"
    return None, f"File Error: unable to decode '{path}' (please use UTF-8 or GBK)"


class ListManager:
    """Loads, validates and summarizes participant lists.

    Custom txt format: one participant per line, "Name,Department".
    Blank lines are ignored.
    """

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list from the given path."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file into users and format error messages."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        lines, error = _read_file_lines(path)
        if error is not None:
            return [], [error]
        return self._parse_lines(lines or [])

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse "Name,Department" lines into users and format errors."""
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in re.split(SEPARATOR_PATTERN, line, maxsplit=1)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_number}: '{line}' "
                    f"(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of members."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

import re

from lottery_system.models import Prize

DEFAULT_PRIZES: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 10, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 5, ["Technology Department"], "Jin Duo"),
    ("Technology Department Exclusive Coupon", 3, ["Product Department"], "Li Bin"),
    ("First Prize - 1 Day Annual Leave", 2, [], "Ma Jiaqi"),
]

_LEVEL_KEYWORDS = {"first": 3, "second": 2, "third": 1}


def infer_level(name: str) -> int:
    """Infer prize level from its name: First=3, Second=2, Third=1, otherwise 0."""
    match = re.search(r"\b(first|second|third)\b", name.lower())
    if match is None:
        return 0
    return _LEVEL_KEYWORDS[match.group(1)]


class PrizeManager:
    """Manages the ordered list of prizes for a lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Load the preset prize library. Returns error messages, if any."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a new prize. Raises ValueError on invalid input."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        prize = Prize(name=name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        prize.level = infer_level(name)
        self._prizes.append(prize)

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order contains 1-based current positions in the new order."""
        try:
            normalized = [int(value) for value in order]
        except (TypeError, ValueError):
            return False
        if sorted(normalized) != list(range(1, len(self._prizes) + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in normalized]
        return True

    def update_avoid_groups(self, position: int, groups: list[str]) -> bool:
        """Set avoidance groups for the prize at 1-based position."""
        if not 1 <= position <= len(self._prizes):
            return False
        cleaned = [g.strip() for g in groups if g and g.strip()]
        self._prizes[position - 1].avoid_groups = cleaned
        return True

    def remove_prize(self, position: int) -> bool:
        """Remove the prize at 1-based position."""
        if not 1 <= position <= len(self._prizes):
            return False
        del self._prizes[position - 1]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User

CHENG_RU_KEYWORD = "cheng ru"
PROTECTED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"
RELAXED_DRAWER = "Li Bin"


def is_cheng_ru(department: str) -> bool:
    """Return True if the department belongs to the Cheng Ru group."""
    return CHENG_RU_KEYWORD in department.strip().lower()


class LotteryManager:
    """Runs lottery draws while enforcing the lottery rules.

    Rules:
    - Previous winners are never drawn again.
    - Members of the prize's avoidance groups are excluded (basic rule).
    - Drawer "Li Bin" uses relaxed filtering (avoidance groups ignored).
    - Cheng Ru fallback: if the last two draws by Chen Qixian / Jin Duo contained
      no Cheng Ru member, the next draw by Ma Jiaqi is limited to Cheng Ru members
      (overriding avoidance groups). After Ma Jiaqi draws, the window is reset.
    """

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.winner_names: set[str] = set()
        self.last_two_draw_results: list[bool] = []

    def is_fallback_active(self) -> bool:
        """Return True when the Cheng Ru protection must apply to Ma Jiaqi's draw."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users eligible to win the given prize right now."""
        pool = [user for user in self.users if user.name not in self.winner_names]
        drawer = prize.associated_drawer

        if drawer == FALLBACK_TARGET_DRAWER and self.is_fallback_active():
            return [user for user in pool if is_cheng_ru(user.department)]

        if drawer == RELAXED_DRAWER:
            return pool

        avoided = {group.strip().lower() for group in prize.avoid_groups}
        return [user for user in pool if user.department.strip().lower() not in avoided]

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None

        winner = random.choice(pool)
        self.winner_names.add(winner.name)

        drawer = prize.associated_drawer
        if drawer in PROTECTED_DRAWERS:
            self._record_draw(is_cheng_ru(winner.department))
        elif drawer == FALLBACK_TARGET_DRAWER:
            self.last_two_draw_results.clear()
        return winner

    def _record_draw(self, contains_cheng_ru: bool) -> None:
        """Keep only the results of the last two protected draws."""
        self.last_two_draw_results = (self.last_two_draw_results + [contains_cheng_ru])[-2:]

    def describe_rule(self, prize: Prize) -> str:
        """Return a human readable description of the rules applied to the prize."""
        parts: list[str] = ["Previous winners excluded"]
        drawer = prize.associated_drawer

        if drawer == RELAXED_DRAWER:
            parts.append("Special rule (Li Bin): avoidance groups relaxed")
        elif drawer == FALLBACK_TARGET_DRAWER and self.is_fallback_active():
            parts.append(
                "Cheng Ru group protection active: only Cheng Ru Group members eligible "
                "(avoidance groups overridden)"
            )
        elif prize.avoid_groups:
            parts.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")

        if drawer:
            parts.append(f"Drawer: {drawer}")
        return "; ".join(parts)

"""Winner tracking, grouping, sorting and CSV-style export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_order: int = 0
    win_time: datetime = field(default_factory=datetime.now)


class ResultManager:
    """Stores winners and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a new winner with the next global winning order."""
        winner = Winner(user=user, prize=prize, winning_order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, keeping winning time order."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize.name in result:
                result[winner.prize.name].append(winner)
        return result

    def get_sorted_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Sort winners by prize level (high to low), then prize order, then winning order."""
        position = {prize.name: index for index, prize in enumerate(prizes)}
        fallback = len(position)
        return sorted(
            self._winners,
            key=lambda w: (-w.prize.level, position.get(w.prize.name, fallback), w.winning_order),
        )

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the sorted winner list to awards_result_YYYYMMDD.txt in the current directory."""
        if not self._winners:
            return False, "No winners to export."

        file_name = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(
                    ["Prize Level", "Prize Name", "Winning Order", "Name", "Department", "Win Time"]
                )
                for winner in self.get_sorted_winners(prizes):
                    writer.writerow(
                        [
                            winner.prize.level,
                            winner.prize.name,
                            winner.winning_order,
                            winner.user.name,
                            winner.user.department,
                            winner.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Result successfully exported to {path}"

def ordinal(number: int) -> str:
    """Format an integer as an English ordinal (1st, 2nd, 3rd, 4th, 11th...)."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:

    def _ask_int(self, prompt: str) -> int:
        while True:
            text = self._ask(prompt).strip()
            try:
                value = int(text)
            except ValueError:
                print("Please enter a valid integer.")
                continue
            if value <= 0:
                print("Please enter a positive integer.")
                continue
            return value

    # --------------------------------------------------------------- main loop

        print("=" * 60)
        print("   Company Team Building - Smart Lottery System")
        print("=" * 60)
        while True:
            self._print_main_menu()

            if choice == "1":
                self._load_default_list()
            elif choice == "2":
                self._load_custom_list()
            elif choice == "3":
                self._show_groups()
            elif choice == "4":
                self._prize_menu()
            elif choice == "5":
                self._start_lottery()
            elif choice == "6":
                self._export_results()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    def _verify_and_store(self, users: list[User], errors: list[str]) -> None:

            f"Duplicate Name: {user.name} ({user.department}) - duplicate entry ignored"
            for user in duplicates
        )

            self.users = valid_users
            print(f"List loaded successfully. Total participants: {len(self.users)}")
            self._show_groups()

        print("Verification found the following issues:")

            print(f"  - {problem}")

            print("No valid participants remain. Upload terminated; list unchanged.")
            return

        choice = self._ask(
            "Enter 'r' to remove invalid data and continue, 't' to terminate upload: "
        ).strip().lower()
        if choice == "r":
            self.users = valid_users
            print(f"Invalid data removed. Total participants: {len(self.users)}")
            self._show_groups()
        else:
            print("Upload terminated; list unchanged.")

            return

            print("No prizes configured. Please configure prizes first.")

                print("  No winners in this round.")

        self._print_final_summary(prizes)

    def _run_round(self, lottery: LotteryManager, prize: Prize, results: ResultManager) -> list[Winner]:
        print(f"\n===== Round: {prize.name} | Quantity: {prize.quantity} =====")
        print(f"Rules: {lottery.describe_rule(prize)}")

        round_winners: list[Winner] = []
        drawn = 0
        while drawn < prize.quantity:
            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            command = self._ask("> ").strip().lower()

            if command == "pause":
                command = self._wait_while_paused()

            if command == "terminate":
                print("Current round terminated; remaining slots skipped.")
                break
            if command not in ("", "continue"):
                print("Unknown command. Press Enter to draw, 'pause' or 'terminate'.")
                continue

            winner_user = lottery.draw_one(prize)
            if winner_user is None:
                print("No eligible candidates remain for this prize. Round ends.")
                break

            drawn += 1
            winner = results.add_winner(winner_user, prize)
            round_winners.append(winner)
            print(
                f"Congratulations {winner_user.name} ({winner_user.department}) "
                f"for winning {prize.name}!"
            )
        return round_winners

    def _wait_while_paused(self) -> str:
        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = self._ask("[PAUSED] > ").strip().lower()
            if command in ("continue", "terminate"):
                if command == "continue":
                    print("Lottery resumed.")
                return command
            print("Unknown command. Type 'continue' or 'terminate'.")

    def _print_final_summary(self, prizes: list[Prize]) -> None:
        if self.result_manager is None:
            return
        print("\n========== Final Winner List ==========")
        ordered_prizes = [
            prize for _, prize in sorted(
                enumerate(prizes), key=lambda item: (-item[1].level, item[0])
            )
        ]
        grouped = self.result_manager.get_winners_by_prize(ordered_prizes)
        any_winner = False
        for prize in ordered_prizes:
            winners = grouped.get(prize.name, [])
            if not winners:
                continue
            any_winner = True
            print(f"\n[{prize.name}]")
            for winner in winners:
                print(
                    f"  {winner.winning_order}. {winner.user.name} "
                    f"({winner.user.department})"
                )
        if not any_winner:
            print("No winners were drawn.")

    # ---------------------------------------------------------------- export
    def _export_results(self) -> None:
        if self.result_manager is None:
            print("No lottery results yet. Please run the lottery first.")
            return
        success, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(message)


def main() -> None:
    """Application entry point."""
    try:
        LotteryCLI().run()
    except (KeyboardInterrupt, EOFError):
        print("\nExiting lottery system.")

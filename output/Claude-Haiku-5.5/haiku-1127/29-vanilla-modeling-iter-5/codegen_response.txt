    """Load system parameters from a JSON file.

"""Grid coordinate system for the simulated 800x800 map.

The map is split into an 8x8 grid of 100x100 cells. Input coordinates are
converted to the center point of their cell. Points whose x or y coordinate
is a multiple of 100 lie on a street.
"""

from dataclasses import dataclass
from typing import List, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE  # 8


def is_valid_coordinate(value: object) -> bool:
    """Return True when ``value`` is an integer in the range 0..MAP_SIZE.

    Args:
        value: Candidate coordinate.

    Returns:
        Validity flag.
    """
    return isinstance(value, int) and not isinstance(value, bool) and 0 <= value <= MAP_SIZE


def validate_point(x: object, y: object) -> None:
    """Raise ``ValueError`` if either coordinate is outside the map.

    Args:
        x: X coordinate.
        y: Y coordinate.

    Raises:
        ValueError: If a coordinate is not an integer in range.
    """
    if not is_valid_coordinate(x) or not is_valid_coordinate(y):
        raise ValueError(
            f"coordinates must be integers in the range 0-{MAP_SIZE}, got ({x}, {y})"
        )


def cell_index(value: int) -> int:
    """Return the grid index (0..7) of a coordinate, clamping the edge value.

    Args:
        value: Coordinate in map units.

    Returns:
        Index of the grid cell along that axis.
    """
    return min(max(int(value) // CELL_SIZE, 0), GRID_DIM - 1)


def cell_center(value: int) -> int:
    """Return the center coordinate of the cell containing ``value``.

    Implements ``center = (value // 100) * 100 + 50`` with edge clamping.

    Args:
        value: Coordinate in map units.

    Returns:
        Center coordinate of the containing cell.
    """
    return cell_index(value) * CELL_SIZE + CELL_SIZE // 2


def to_grid_center(x: int, y: int) -> Tuple[int, int]:
    """Convert an input point to the center of its grid cell.

    Args:
        x: Input X coordinate (0-800).
        y: Input Y coordinate (0-800).

    Returns:
        Tuple ``(center_x, center_y)``.

    Raises:
        ValueError: If the coordinates are invalid.
    """
    validate_point(x, y)
    return cell_center(x), cell_center(y)


def is_street_point(x: int, y: int) -> bool:
    """Return True if the point lies on a street (grid line).

    Args:
        x: X coordinate.
        y: Y coordinate.

    Returns:
        True when x or y is a multiple of the cell size.
    """
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


def nearest_street_line(value: float) -> int:
    """Return the nearest grid line coordinate to ``value``.

    Args:
        value: Coordinate in map units.

    Returns:
        Nearest multiple of CELL_SIZE within the map bounds.
    """
    line = int(round(value / CELL_SIZE)) * CELL_SIZE
    return min(max(line, 0), MAP_SIZE)


@dataclass
class GridCell:
    """A single cell of the 8x8 grid.

    Attributes:
        grid_x: Column index (0-7).
        grid_y: Row index (0-7).
        center_x: X coordinate of the cell center.
        center_y: Y coordinate of the cell center.
        is_street: Whether the cell center lies on a road.
    """

    grid_x: int
    grid_y: int
    center_x: int
    center_y: int
    is_street: bool


class Grid:
    """The 800x800 map partitioned into 64 cells.

    Attributes:
        map_size: Map side length in units.
        cell_size: Cell side length in units.
        dim: Number of cells per row/column.
        cells: All grid cells in row-major order.
    """

    def __init__(self, map_size: int = MAP_SIZE, cell_size: int = CELL_SIZE) -> None:
        """Create the grid.

        Args:
            map_size: Map side length in units.
            cell_size: Cell side length in units.
        """
        self.map_size = map_size
        self.cell_size = cell_size
        self.dim = map_size // cell_size
        self.cells: List[GridCell] = []
        for gy in range(self.dim):
            for gx in range(self.dim):
                center_x = gx * cell_size + cell_size // 2
                center_y = gy * cell_size + cell_size // 2
                self.cells.append(
                    GridCell(gx, gy, center_x, center_y, is_street_point(center_x, center_y))
                )

    def cell_at(self, x: int, y: int) -> GridCell:
        """Return the cell containing the point ``(x, y)``.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            The matching GridCell.
        """
        gx = min(max(x // self.cell_size, 0), self.dim - 1)
        gy = min(max(y // self.cell_size, 0), self.dim - 1)
        return self.cells[gy * self.dim + gx]

"""Grid-based path planning for riders.

Routes prefer streets: a rider first snaps onto the nearest vertical street,
travels along vertical streets to the nearest horizontal street next to the
goal, follows that street horizontally, and makes a short final approach to
the target. Riders never block each other, so no deadlock can arise.
"""

from typing import List, Optional, Tuple

from grid import CELL_SIZE, Grid, nearest_street_line

Point = Tuple[int, int]


class PathPlanner:
    """Plans and executes right-angle routes on the grid.

    Attributes:
        grid: The grid the planner operates on.
    """

    def __init__(self, grid: Optional[Grid] = None) -> None:
        """Create a planner.

        Args:
            grid: Optional grid instance; a default 8x8 grid is created otherwise.
        """
        self.grid = grid if grid is not None else Grid()

    def plan(self, start: Point, goal: Point) -> List[Point]:
        """Compute the waypoint list from ``start`` to ``goal``.

        Every consecutive pair of points shares either the x or the y
        coordinate, so each leg is a straight street segment or a final
        approach segment.

        Args:
            start: Current position.
            goal: Target position.

        Returns:
            Ordered waypoints, excluding ``start`` and ending at ``goal``.
            An empty list means the rider is already at the goal.
        """
        sx, sy = start
        gx, gy = goal
        waypoints: List[Point] = []
        cx, cy = sx, sy

        # Special case: an off-street rider first snaps onto a vertical street.
        if cx % CELL_SIZE != 0:
            cx = nearest_street_line(cx)
            waypoints.append((cx, cy))

        if cx == gx:
            # Directly above or below the target: straight movement.
            waypoints.append((gx, gy))
        else:
            street_y = nearest_street_line(gy)
            waypoints.append((cx, street_y))
            waypoints.append((gx, street_y))
            waypoints.append((gx, gy))

        return _remove_repeats(start, waypoints)

    def step(
        self, x: int, y: int, route: List[Point], speed: int
    ) -> Tuple[int, int, List[Point]]:
        """Advance along ``route`` by at most ``speed`` units.

        Args:
            x: Current X coordinate.
            y: Current Y coordinate.
            route: Remaining waypoints.
            speed: Maximum Manhattan distance to travel this tick.

        Returns:
            Tuple ``(new_x, new_y, remaining_route)``.
        """
        remaining = int(speed)
        pending = list(route)
        if remaining <= 0:
            return x, y, pending
        while remaining > 0 and pending:
            tx, ty = pending[0]
            dx = tx - x
            dy = ty - y
            dist = abs(dx) + abs(dy)
            if dist <= remaining:
                x, y = tx, ty
                remaining -= dist
                pending.pop(0)
            else:
                if dx != 0:
                    x += remaining if dx > 0 else -remaining
                else:
                    y += remaining if dy > 0 else -remaining
                remaining = 0
        return x, y, pending


def _remove_repeats(start: Point, points: List[Point]) -> List[Point]:
    """Drop consecutive duplicate waypoints, including repeats of ``start``.

    Args:
        start: Starting point.
        points: Candidate waypoints.

    Returns:
        Filtered waypoint list.
    """
    result: List[Point] = []
    prev = start
    for point in points:
        if point != prev:
            result.append(point)
            prev = point
    return result

"""Order data model.

An order is described by ``[restaurant_x, restaurant_y, customer_x, customer_y]``.
Input coordinates are converted to grid cell centers on creation.
"""

import itertools
import math
from typing import Any, Dict, Optional, Sequence, Tuple

from grid import to_grid_center

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_COMPLETED = "completed"

_ORDER_IDS = itertools.count(1)


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order identifier.
        restaurant_x: Restaurant X (grid cell center).
        restaurant_y: Restaurant Y (grid cell center).
        customer_x: Customer X (grid cell center).
        customer_y: Customer Y (grid cell center).
        start_time: Tick at which the order was created.
        completion_time: Tick at which delivery finished, or None.
        distance: Euclidean distance between restaurant and customer.
        time_taken: Ticks between creation and completion, or None.
        rider_id: Identifier of the assigned rider, or None.
        status: "pending", "assigned" or "completed".
    """

    def __init__(
        self,
        res_pos: Sequence[int],
        cus_pos: Sequence[int],
        start_time: int = 0,
    ) -> None:
        """Create an order from raw input points.

        Args:
            res_pos: Restaurant point ``(x, y)`` in the range 0-800.
            cus_pos: Customer point ``(x, y)`` in the range 0-800.
            start_time: Creation tick.

        Raises:
            ValueError: If a point has the wrong shape or invalid coordinates.
        """
        res_x, res_y = _parse_point(res_pos, "restaurant")
        cus_x, cus_y = _parse_point(cus_pos, "customer")
        self.id: int = next(_ORDER_IDS)
        self.restaurant_x, self.restaurant_y = to_grid_center(res_x, res_y)
        self.customer_x, self.customer_y = to_grid_center(cus_x, cus_y)
        self.start_time: int = int(start_time)
        self.completion_time: Optional[int] = None
        self.distance: float = round(
            math.hypot(self.customer_x - self.restaurant_x, self.customer_y - self.restaurant_y),
            2,
        )
        self.time_taken: Optional[int] = None
        self.rider_id: Optional[int] = None
        self.status: str = STATUS_PENDING

    @property
    def res_pos(self) -> Tuple[int, int]:
        """Restaurant grid point as a tuple."""
        return self.restaurant_x, self.restaurant_y

    @property
    def cus_pos(self) -> Tuple[int, int]:
        """Customer grid point as a tuple."""
        return self.customer_x, self.customer_y

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to the rider with ``rider_id``.

        Args:
            rider_id: Identifier of the rider.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_completed(self, tick: int) -> None:
        """Record delivery completion and release the order.

        Args:
            tick: Simulation tick at which the delivery finished.
        """
        self.completion_time = int(tick)
        self.time_taken = self.completion_time - self.start_time
        self.status = STATUS_COMPLETED

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order.

        Returns:
            Dictionary with the order's attributes.
        """
        return {
            "id": self.id,
            "status": self.status,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "rider_id": self.rider_id,
        }


def _parse_point(point: Sequence[int], label: str) -> Tuple[int, int]:
    """Validate that ``point`` is a pair of integers.

    Args:
        point: Candidate point.
        label: Human readable name used in error messages.

    Returns:
        Tuple ``(x, y)``.

    Raises:
        ValueError: If the point does not have exactly two integer values.
    """
    if len(point) != 2:
        raise ValueError(f"{label} point must contain exactly two coordinates")
    x, y = point
    return x, y

"""Rider model: state machine, movement and order execution.

Rider states:
    0 idle, 1 picking up (en route to restaurant), 2 delivering (en route to customer).

Log events recorded in ``Rider.log``:
    "spawn", "accepted_order", "arrived_at_merchant", "delivered_to_customer".
"""

import logging
from typing import TYPE_CHECKING, Any, Dict, List, Optional, Tuple

import config
import globalvar
from path_planner import PathPlanner

if TYPE_CHECKING:  # pragma: no cover
    from order_model import Order

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

LOG_LIMIT = 2000
Point = Tuple[int, int]


def _current_tick() -> int:
    """Return the current simulation tick from the global store.

    Returns:
        The tick counter, or 0 if it has not been set.
    """
    value = globalvar.get_value("current_tick")
    return int(value) if value is not None else 0


class Rider:
    """A delivery rider.

    Attributes:
        rider_id: Unique identifier.
        x: Current X coordinate.
        y: Current Y coordinate.
        speed: Movement speed in units per tick.
        state: Current status enum (0, 1 or 2).
        orders: Assigned orders; the first element is the active order.
        special_case: Flag set while aligned with the current target.
        log: Status change history (list of dictionaries).
        route: Remaining waypoints for the current leg.
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the configured start position in idle state.

        Args:
            rider_id: Unique rider identifier.
        """
        self.rider_id: int = int(rider_id)
        self.x: int = int(config.CONFIG["rider_start_x"])
        self.y: int = int(config.CONFIG["rider_start_y"])
        self.speed: int = int(config.CONFIG["rider_speed"])
        self.state: int = STATE_IDLE
        self.orders: List["Order"] = []
        self.special_case: bool = False
        self.log: List[Dict[str, Any]] = []
        self.route: List[Point] = []
        self._route_goal: Optional[Point] = None
        self._planner = PathPlanner()
        self._record("spawn", None)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an entry to the rider's status log.

        Args:
            event: Event name.
            order_id: Related order identifier, or None.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": _current_tick(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        if len(self.log) > LOG_LIMIT:
            del self.log[: len(self.log) - LOG_LIMIT]
        logger.debug("rider %s: %s (order %s)", self.rider_id, event, order_id)

    def reset_route(self) -> None:
        """Discard the planned route so that it is recomputed on the next move."""
        self.route = []
        self._route_goal = None

    def current_target(self) -> Optional[Point]:
        """Return the active destination.

        Returns:
            Restaurant point while picking up, customer point while delivering,
            restaurant point when idle with orders, or None without orders.
        """
        if not self.orders:
            return None
        order = self.orders[0]
        if self.state == STATE_DELIVERING:
            return order.cus_pos
        return order.res_pos

    def move(self) -> None:
        """Advance the rider by one tick.

        Handles the state transitions:
            idle -> picking up when an order is present,
            picking up -> delivering on arrival at the restaurant,
            delivering -> idle on delivery to the customer.
        """
        if not self.orders:
            if self.state != STATE_IDLE:
                self.state = STATE_IDLE
            self.reset_route()
            self.special_case = False
            return

        order = self.orders[0]
        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP
            self.reset_route()
            self._record("accepted_order", order.id)

        target = self.current_target()
        if target is None:  # pragma: no cover - guarded above
            return
        if self._route_goal != target:
            self.route = self._planner.plan((self.x, self.y), target)
            self._route_goal = target

        self.x, self.y, self.route = self._planner.step(
            self.x, self.y, self.route, self.speed
        )
        self.special_case = self.x == target[0] or self.y == target[1]
        if self.route:
            return

        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            self.reset_route()
            self._record("arrived_at_merchant", order.id)
        elif self.state == STATE_DELIVERING:
            order.mark_completed(_current_tick())
            if order in self.orders:
                self.orders.remove(order)
            self.state = STATE_IDLE
            self.reset_route()
            self.special_case = False
            self._record("delivered_to_customer", order.id)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider.

        Returns:
            Dictionary with the rider's attributes and assigned order ids.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [order.id for order in self.orders],
            "log": list(self.log),
        }

"""Order assignment and order lifecycle management.

``find_best_rider_for_order`` implements distance-first assignment using the
Euclidean distance from each rider to the restaurant. ``OrderManager`` keeps
the pending and historical order lists.
"""

import logging
import math
from typing import List, Optional

import config
import globalvar
from order_model import Order
from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the closest rider that still has capacity for ``order``.

    Distance is ``sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2)``.
    Riders already holding ``max_orders_per_rider`` orders are skipped, which
    balances load across riders. Ties are broken by fewer orders, then rider id.

    Args:
        order: The order to be assigned.

    Returns:
        The selected Rider, or None when no rider has spare capacity.
    """
    riders = globalvar.get_value("riders") or []
    capacity = int(config.CONFIG["max_orders_per_rider"])
    best: Optional[Rider] = None
    best_key = None
    for rider in riders:
        if len(rider.orders) >= capacity:
            continue
        distance = math.hypot(rider.x - order.restaurant_x, rider.y - order.restaurant_y)
        key = (distance, len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates, assigns and tracks orders.

    Attributes:
        all_orders: Every order created in this session.
        pending_orders: Orders waiting for a rider.
    """

    def __init__(self) -> None:
        """Create an empty order manager."""
        self.all_orders: List[Order] = []
        self.pending_orders: List[Order] = []

    @property
    def pending_count(self) -> int:
        """Number of orders waiting for assignment."""
        return len(self.pending_orders)

    def create_order(self, res_pos, cus_pos) -> Order:
        """Create a new order stamped with the current tick and queue it.

        Args:
            res_pos: Restaurant point ``(x, y)``.
            cus_pos: Customer point ``(x, y)``.

        Returns:
            The created Order.
        """
        tick = globalvar.get_value("current_tick") or 0
        order = Order(res_pos, cus_pos, start_time=tick)
        self.all_orders.append(order)
        self.pending_orders.append(order)
        logger.info("order %s created at tick %s", order.id, tick)
        return order

    def assign_pending_orders(self) -> List[Order]:
        """Assign every pending order that can be matched to a rider.

        Returns:
            Orders assigned during this call.
        """
        assigned: List[Order] = []
        for order in list(self.pending_orders):
            rider = find_best_rider_for_order(order)
            if rider is None:
                continue
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
            self.pending_orders.remove(order)
            assigned.append(order)
            logger.info("order %s assigned to rider %s", order.id, rider.rider_id)
        return assigned

    def active_orders(self) -> List[Order]:
        """Return orders that have not been completed yet.

        Returns:
            List of pending or assigned orders.
        """
        return [o for o in self.all_orders if o.status != "completed"]

    def completed_orders(self) -> List[Order]:
        """Return completed orders.

        Returns:
            List of completed orders.
        """
        return [o for o in self.all_orders if o.status == "completed"]

"""System-level state management and simulation tick loop.

``SystemManager`` owns funds, the rider id counter, the running flag and the
order manager. ``init_system`` resets global storage, loads configuration and
returns a ready-to-use manager.
"""

import json
import logging
from typing import Any, Dict, List, Optional

import config
import globalvar
from order import OrderManager
from order_model import STATUS_COMPLETED
from rider import STATE_IDLE, Rider

logger = logging.getLogger(__name__)


class SystemManager:
    """Core coordinator for funds, riders, orders and the simulated clock.

    Attributes:
        funds: Current system funds (may be negative).
        rider_id_counter: Last assigned rider id.
        is_running: Main loop control flag.
        order_manager: The OrderManager instance.
    """

    def __init__(self) -> None:
        """Create the manager with configured initial funds."""
        self.funds: int = int(config.CONFIG["initial_funds"])
        self.rider_id_counter: int = 0
        self.is_running: bool = True
        self.order_manager = OrderManager()
        globalvar.set_value("riders", [])
        globalvar.set_value("current_tick", 0)

    @property
    def riders(self) -> List[Rider]:
        """The rider list stored under the global key 'riders'."""
        return globalvar.get_value("riders") or []

    @property
    def current_tick(self) -> int:
        """Current simulation tick."""
        return int(globalvar.get_value("current_tick") or 0)

    def find_rider(self, rider_id: Optional[int]) -> Optional[Rider]:
        """Look up a rider by identifier.

        Args:
            rider_id: Rider identifier.

        Returns:
            The matching Rider or None.
        """
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def create_rider(self) -> Rider:
        """Create a new idle rider and add it to the rider list.

        Returns:
            The created Rider.
        """
        self.rider_id_counter += 1
        rider = Rider(self.rider_id_counter)
        self.riders.append(rider)
        logger.info("rider %s recruited", rider.rider_id)
        return rider

    def idle_rider_count(self) -> int:
        """Count riders with state idle and no assigned orders.

        Returns:
            Number of available riders.
        """
        return sum(1 for r in self.riders if r.state == STATE_IDLE and not r.orders)

    def can_recruit(self) -> bool:
        """Check the recruitment trigger.

        Returns:
            True when funds are at least the recruitment threshold and the
            number of pending orders exceeds the number of idle riders.
        """
        threshold = int(config.CONFIG["recruit_min_funds"])
        return (
            self.funds >= threshold
            and self.order_manager.pending_count > self.idle_rider_count()
        )

    def recruit_riders(self) -> int:
        """Recruit riders while the recruitment trigger holds.

        Each recruit costs ``rider_cost`` from system funds.

        Returns:
            Number of riders added.
        """
        added = 0
        cost = int(config.CONFIG["rider_cost"])
        while self.can_recruit():
            self.funds -= cost
            self.create_rider()
            added += 1
        return added

    def submit_order(self, res_pos, cus_pos):
        """Create an order, recruit if needed and attempt assignment.

        Args:
            res_pos: Restaurant point ``(x, y)``.
            cus_pos: Customer point ``(x, y)``.

        Returns:
            The created Order (status is 'assigned' or 'pending').
        """
        order = self.order_manager.create_order(res_pos, cus_pos)
        self.recruit_riders()
        self.order_manager.assign_pending_orders()
        return order

    def check_consistency(self) -> List[str]:
        """Verify and automatically repair state inconsistencies.

        Returns:
            Descriptions of the issues found (empty when consistent).
        """
        issues: List[str] = []
        seen_ids = set()
        for rider in self.riders:
            if rider.rider_id in seen_ids:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen_ids.add(rider.rider_id)
            completed = [o for o in rider.orders if o.status == STATUS_COMPLETED]
            if completed:
                issues.append(f"rider {rider.rider_id} held completed orders; removed")
                rider.orders = [o for o in rider.orders if o.status != STATUS_COMPLETED]
            if not rider.orders and rider.state != STATE_IDLE:
                issues.append(f"rider {rider.rider_id} had no orders but was not idle; reset")
                rider.state = STATE_IDLE
                rider.reset_route()
        return issues

    def tick(self) -> None:
        """Advance the simulation by one tick.

        Order of operations: increment clock, recruit riders, assign pending
        orders, move every rider, and run consistency checks.
        """
        globalvar.set_value("current_tick", self.current_tick + 1)
        self.recruit_riders()
        self.order_manager.assign_pending_orders()
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # error recovery: keep the simulation running
                logger.exception("rider %s failed to move; resetting route", rider.rider_id)
                rider.reset_route()
        for issue in self.check_consistency():
            logger.warning(issue)

    def shutdown(self) -> None:
        """Stop the main loop."""
        self.is_running = False

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the system.

        Returns:
            Dictionary with funds, counters, riders and order history.
        """
        return {
            "funds": self.funds,
            "rider_id_counter": self.rider_id_counter,
            "current_tick": self.current_tick,
            "is_running": self.is_running,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.all_orders],
        }

    def save_state(self, path: Optional[str] = None) -> str:
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination path; defaults to the configured state file.

        Returns:
            The path that was written.
        """
        target = path or str(config.CONFIG["state_file"])
        with open(target, "w", encoding="utf-8") as handle:
            json.dump(self.to_dict(), handle, indent=2)
        return target


def init_system(config_path: Optional[str] = None) -> SystemManager:
    """Initialise global storage, configuration and the system manager.

    Args:
        config_path: Optional path to a JSON configuration file.

    Returns:
        A ready SystemManager instance.
    """
    globalvar._init()
    config.load_config(config_path)
    return SystemManager()

"""Status displays and ASCII map rendering.

Map legend:
    R rider, S restaurant, C customer, # street, . empty space, * route.

The map is drawn at 50-unit resolution so that cell centers (e.g. 150) and
street lines (e.g. 200) both land on distinct characters. The y axis points
upwards.
"""

from typing import List, Optional, Sequence, Tuple

from grid import MAP_SIZE
from order_model import Order, STATUS_COMPLETED
from rider import STATE_NAMES, Rider

CANVAS_STEP = 50
CANVAS_MAX = MAP_SIZE // CANVAS_STEP  # 16
CANVAS_SIZE = CANVAS_MAX + 1
LEGEND = "Legend: R rider, S restaurant, C customer, # street, . empty, * route"


def _axis_samples(a: int, b: int, step: int = 10) -> List[int]:
    """Return sample coordinates between ``a`` and ``b`` inclusive.

    Args:
        a: First coordinate.
        b: Second coordinate.
        step: Sampling distance.

    Returns:
        Sorted list of sample values.
    """
    lo, hi = sorted((a, b))
    values = list(range(lo, hi + 1, step))
    if not values or values[-1] != hi:
        values.append(hi)
    return values


def _segment_points(a: Tuple[int, int], b: Tuple[int, int]) -> List[Tuple[int, int]]:
    """Sample a Manhattan path from ``a`` to ``b`` (horizontal then vertical).

    Args:
        a: Start point.
        b: End point.

    Returns:
        Sampled points along the path.
    """
    (x0, y0), (x1, y1) = a, b
    points = [(x, y0) for x in _axis_samples(x0, x1)]
    points.extend((x1, y) for y in _axis_samples(y0, y1))
    return points


def _to_cell(x: int, y: int) -> Tuple[int, int]:
    """Convert a map coordinate to canvas (row, column).

    Args:
        x: Map X coordinate.
        y: Map Y coordinate.

    Returns:
        Tuple ``(row, col)`` where row 0 is the top of the map.
    """
    col = min(max(x, 0) // CANVAS_STEP, CANVAS_MAX)
    row = CANVAS_MAX - min(max(y, 0) // CANVAS_STEP, CANVAS_MAX)
    return row, col


class DisplayManager:
    """Renders status text, order tables, rider logs and the ASCII map.

    Attributes:
        system: The SystemManager whose state is displayed.
    """

    def __init__(self, system) -> None:
        """Create a display bound to a system manager.

        Args:
            system: SystemManager instance.
        """
        self.system = system

    def render_status(self) -> str:
        """Build the status snapshot text.

        Returns:
            Multi-line status report.
        """
        riders = self.system.riders
        lines = [
            f"System funds: {self.system.funds}",
            f"Current tick: {self.system.current_tick}",
            f"Riders: {len(riders)} (idle: {self.system.idle_rider_count()})",
            f"Pending orders: {self.system.order_manager.pending_count}",
        ]
        for rider in riders:
            order_ids = ", ".join(str(o.id) for o in rider.orders) or "none"
            lines.append(
                f"  Rider #{rider.rider_id}: position ({rider.x}, {rider.y}) | "
                f"status: {STATE_NAMES[rider.state]} | speed: {rider.speed} | "
                f"orders: {len(rider.orders)} [{order_ids}]"
            )
        return "\n".join(lines)

    def render_orders(self) -> str:
        """Build the order table.

        Returns:
            Formatted table of all orders.
        """
        orders = self.system.order_manager.all_orders
        if not orders:
            return "No orders yet."
        header = (
            f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
            f"{'Distance':>9} {'Time':>6} {'Rider':>6} {'Status':>10}"
        )
        lines = [header, "-" * len(header)]
        for order in orders:
            rider = "-" if order.rider_id is None else str(order.rider_id)
            time_taken = "-" if order.time_taken is None else str(order.time_taken)
            lines.append(
                f"{order.id:>4} {order.start_time:>6} "
                f"{str((order.restaurant_x, order.restaurant_y)):>12} "
                f"{str((order.customer_x, order.customer_y)):>12} "
                f"{order.distance:>9.2f} {time_taken:>6} {rider:>6} {order.status:>10}"
            )
        return "\n".join(lines)

    def render_rider_log(self) -> str:
        """Build the status-change history for every rider.

        Returns:
            Multi-line history text.
        """
        lines: List[str] = []
        for rider in self.system.riders:
            lines.append(f"Rider #{rider.rider_id}:")
            if not rider.log:
                lines.append("  (no records)")
            for entry in rider.log:
                order = "-" if entry["order_id"] is None else str(entry["order_id"])
                lines.append(
                    f"  tick {entry['tick']:>5} | {entry['event']:<22} | order {order:>4} | "
                    f"pos ({entry['x']}, {entry['y']}) | state {STATE_NAMES[entry['state']]}"
                )
        return "\n".join(lines) if lines else "No riders yet."

    def render_map(self, trail: Optional[Sequence[Tuple[int, int]]] = None) -> str:
        """Render the ASCII map.

        Args:
            trail: Optional route points drawn as '*'.

        Returns:
            Multi-line map text including legend.
        """
        canvas = self._blank_canvas()
        if trail:
            self._draw_route(canvas, trail)
        for order in self.system.order_manager.active_orders():
            self._place(canvas, order.restaurant_x, order.restaurant_y, "S")
            self._place(canvas, order.customer_x, order.customer_y, "C")
        for rider in self.system.riders:
            self._place(canvas, rider.x, rider.y, "R")
        return self._frame(canvas) + "\n" + LEGEND

    def render_route_map(
        self,
        order: Order,
        trail: Sequence[Tuple[int, int]],
        rider: Optional[Rider],
        new_rider_added: int,
    ) -> str:
        """Render the summary shown after an order is accepted and completed.

        Args:
            order: The order that was processed.
            trail: Route points visited by the assigned rider.
            rider: The assigned rider, if any.
            new_rider_added: Number of riders recruited for this order.

        Returns:
            Multi-line summary including the route map.
        """
        lines = [
            f"Order #{order.id} processed.",
            f"Restaurant grid point: ({order.restaurant_x}, {order.restaurant_y})",
            f"Customer grid point: ({order.customer_x}, {order.customer_y})",
            f"New rider added: {'yes (' + str(new_rider_added) + ')' if new_rider_added else 'no'}",
        ]
        if rider is not None:
            lines.append(
                f"Assigned rider: #{rider.rider_id} | speed: {rider.speed} units/tick"
            )
        lines.append(
            f"Distance: {order.distance:.2f} | time taken: "
            f"{order.time_taken if order.time_taken is not None else '-'} ticks"
        )
        lines.append(self.render_map(trail))
        return "\n".join(lines)

    def _blank_canvas(self) -> List[List[str]]:
        """Create the base canvas with streets and empty cells.

        Returns:
            2-D list of characters.
        """
        canvas: List[List[str]] = []
        for row in range(CANVAS_SIZE):
            y = (CANVAS_MAX - row) * CANVAS_STEP
            line = []
            for col in range(CANVAS_SIZE):
                x = col * CANVAS_STEP
                line.append("#" if x % 100 == 0 or y % 100 == 0 else ".")
            canvas.append(line)
        return canvas

    def _draw_route(self, canvas: List[List[str]], trail: Sequence[Tuple[int, int]]) -> None:
        """Draw connected route segments onto the canvas.

        Args:
            canvas: Canvas to modify in place.
            trail: Ordered route points.
        """
        points: List[Tuple[int, int]] = [trail[0]]
        for prev, cur in zip(trail, trail[1:]):
            points.extend(_segment_points(prev, cur))
        for x, y in points:
            row, col = _to_cell(x, y)
            if canvas[row][col] in ".#":
                canvas[row][col] = "*"

    def _place(self, canvas: List[List[str]], x: int, y: int, symbol: str) -> None:
        """Place a marker symbol at a map coordinate.

        Args:
            canvas: Canvas to modify.
            x: Map X coordinate.
            y: Map Y coordinate.
            symbol: Single-character marker.
        """
        row, col = _to_cell(x, y)
        canvas[row][col] = symbol

    def _frame(self, canvas: List[List[str]]) -> str:
        """Add row labels and axis notes to the canvas.

        Args:
            canvas: Completed canvas.

        Returns:
            Printable map text.
        """
        lines = []
        for row, chars in enumerate(canvas):
            y = (CANVAS_MAX - row) * CANVAS_STEP
            label = f"{y:>4}" if y % 100 == 0 else "    "
            lines.append(f"{label} |{''.join(chars)}|")
        lines.append("     x: 0 (left) .. 800 (right), y: 0 (bottom) .. 800 (top)")
        return "\n".join(lines)

"""Command parsing, validation and the interactive command loop.

Commands:
    start [rx ry cx cy]   create an order and run it to completion
    status                system snapshot
    orders                order table
    riders                rider status-change history
    map                   ASCII map
    tick [n]              advance the simulation by n ticks
    save [path]           write a JSON snapshot
    history               list previous commands
    help                  show help
    quit / exit           leave the program

Shortcuts: s=start, st=status, o=orders, r=riders, m=map, t=tick,
sv=save, h or ?=help, q/e=quit/exit, !!=repeat last, !n=repeat history entry n.
"""

import logging
from typing import Dict, List, Optional, Callable

import config
from display import DisplayManager
from grid import MAP_SIZE
from order_model import STATUS_COMPLETED, STATUS_PENDING
from rider import Rider

try:  # enables arrow-key editing and history on platforms that support it
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - Windows without pyreadline
    readline = None

logger = logging.getLogger(__name__)

ALIASES: Dict[str, str] = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "sv": "save",
    "h": "help",
    "?": "help",
    "q": "quit",
    "e": "exit",
}

MAX_TICK_BATCH = 10000

HELP_TEXT = """Commands:
  start [rx ry cx cy]  order from restaurant (rx, ry) to customer (cx, cy), 0-800
  status               system funds, riders and their orders
  orders               all orders with times, distance and rider
  riders               rider status-change history
  map                  ASCII map (R rider, S restaurant, C customer, # street)
  tick [n]             advance simulation by n ticks (default 1)
  save [path]          save a JSON snapshot
  history              show command history
  help                 show this help
  quit | exit          leave the program
Shortcuts: s, st, o, r, m, t, sv, h, ?, q, e, !!, !n"""


def parse_coordinates(tokens: List[str]) -> List[int]:
    """Convert four tokens into validated integer coordinates.

    Args:
        tokens: Raw string tokens.

    Returns:
        List of four integers.

    Raises:
        ValueError: If a token is not an integer or is outside 0-800.
    """
    if len(tokens) != 4:
        raise ValueError("expected 4 coordinates: restaurant_x restaurant_y customer_x customer_y")
    values: List[int] = []
    for token in tokens:
        try:
            value = int(token)
        except ValueError:
            raise ValueError(f"'{token}' is not an integer") from None
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"{value} is outside the allowed range 0-{MAP_SIZE}")
        values.append(value)
    return values


class InputManager:
    """Reads commands from standard input and dispatches them.

    Attributes:
        system: The SystemManager being driven.
        display: The DisplayManager used for output.
        command_history: Executed command lines, oldest first.
    """

    def __init__(self, system, display: Optional[DisplayManager] = None) -> None:
        """Create the input manager.

        Args:
            system: SystemManager instance.
            display: Optional DisplayManager; one is created otherwise.
        """
        self.system = system
        self.display = display if display is not None else DisplayManager(system)
        self.command_history: List[str] = []
        self._handlers: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self) -> None:
        """Run the interactive main loop until the system stops."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.is_running:
            try:
                line = input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            try:
                self.handle_line(line)
            except Exception:  # error recovery: report and continue
                logger.exception("command failed")
                print("An unexpected error occurred; the system state was preserved.")
        print("Goodbye.")

    def handle_line(self, line: str) -> None:
        """Parse and execute one command line.

        Args:
            line: Raw user input.
        """
        line = line.strip()
        if not line:
            return
        if line == "!!":
            if not self.command_history:
                print("No previous command.")
                return
            line = self.command_history[-1]
        elif line.startswith("!") and line[1:].isdigit():
            index = int(line[1:])
            if not 1 <= index <= len(self.command_history):
                print(f"No history entry {index}.")
                return
            line = self.command_history[index - 1]

        self.command_history.append(line)
        tokens = line.split()
        name = tokens[0].lower()
        command = ALIASES.get(name, name)
        handler = self._handlers.get(command)
        if handler is None:
            print(f"Unknown command '{tokens[0]}'. Type 'help' for a list of commands.")
            return
        handler(tokens[1:])

    def cmd_start(self, args: List[str]) -> None:
        """Create an order and simulate until it is delivered.

        Args:
            args: Four coordinate tokens, or none to be prompted.
        """
        if not args:
            args = input("Enter restaurant_x restaurant_y customer_x customer_y: ").split()
        try:
            coords = parse_coordinates(args)
        except ValueError as exc:
            print(f"Invalid input: {exc}")
            return

        riders_before = len(self.system.riders)
        order = self.system.submit_order((coords[0], coords[1]), (coords[2], coords[3]))
        new_riders = len(self.system.riders) - riders_before
        if order.status == STATUS_PENDING:
            print(
                f"Order #{order.id} queued: no rider available "
                f"(funds: {self.system.funds}). Use 'tick' to advance time."
            )
            return

        rider = self.system.find_rider(order.rider_id)
        trail = [(rider.x, rider.y)] if rider is not None else []
        limit = int(config.CONFIG["max_ticks_per_order"])
        ticks = 0
        while order.status != STATUS_COMPLETED and ticks < limit:
            self.system.tick()
            ticks += 1
            rider = self.system.find_rider(order.rider_id)
            if rider is not None:
                trail.append((rider.x, rider.y))

        if order.status != STATUS_COMPLETED:
            print(f"Order #{order.id} not completed within {limit} ticks.")
            print(self.display.render_map(trail))
            return
        print(self.display.render_route_map(order, trail, rider, new_riders))

    def cmd_status(self, args: List[str]) -> None:
        """Print the system snapshot.

        Args:
            args: Ignored.
        """
        print(self.display.render_status())

    def cmd_orders(self, args: List[str]) -> None:
        """Print the order table.

        Args:
            args: Ignored.
        """
        print(self.display.render_orders())

    def cmd_riders(self, args: List[str]) -> None:
        """Print each rider's status-change history.

        Args:
            args: Ignored.
        """
        print(self.display.render_rider_log())

    def cmd_map(self, args: List[str]) -> None:
        """Print the ASCII map.

        Args:
            args: Ignored.
        """
        print(self.display.render_map())

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation.

        Args:
            args: Optional number of ticks.
        """
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                print(f"'{args[0]}' is not an integer.")
                return
            if not 1 <= count <= MAX_TICK_BATCH:
                print(f"Tick count must be between 1 and {MAX_TICK_BATCH}.")
                return
        for _ in range(count):
            self.system.tick()
        completed = len(self.system.order_manager.completed_orders())
        print(
            f"Advanced {count} tick(s). Current tick: {self.system.current_tick}. "
            f"Completed orders: {completed}. Funds: {self.system.funds}."
        )

    def cmd_save(self, args: List[str]) -> None:
        """Save a JSON snapshot.

        Args:
            args: Optional destination path.
        """
        try:
            path = self.system.save_state(args[0] if args else None)
        except OSError as exc:
            print(f"Could not save state: {exc}")
            return
        print(f"State saved to {path}.")

    def cmd_history(self, args: List[str]) -> None:
        """Print the command history.

        Args:
            args: Ignored.
        """
        if not self.command_history:
            print("No commands yet.")
            return
        for index, entry in enumerate(self.command_history, start=1):
            print(f"{index:>4}  {entry}")

    def cmd_help(self, args: List[str]) -> None:
        """Print help text.

        Args:
            args: Ignored.
        """
        print(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the main loop.

        Args:
            args: Ignored.
        """
        self.system.shutdown()

"""Entry point for the Food Delivery Order Dispatch simulator.

Usage:
    python src/main.py [--config path/to/config.json] [--debug]
"""

import argparse
import logging
import sys
from typing import List, Optional

from display import DisplayManager
from input_manager import InputManager
from system import init_system


def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: Optional argument list; defaults to ``sys.argv[1:]``.

    Returns:
        Parsed namespace.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=None, help="path to a JSON configuration file")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    return parser.parse_args(argv)


def main(argv: Optional[List[str]] = None) -> int:
    """Run the simulator.

    Args:
        argv: Optional argument list.

    Returns:
        Process exit code.
    """
    args = parse_args(argv)
    system = init_system(args.config)

    import config

    level_name = "DEBUG" if args.debug else str(config.CONFIG.get("log_level", "WARNING")).upper()
    logging.basicConfig(
        level=getattr(logging, level_name, logging.WARNING),
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )

    display = DisplayManager(system)
    manager = InputManager(system, display)
    manager.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

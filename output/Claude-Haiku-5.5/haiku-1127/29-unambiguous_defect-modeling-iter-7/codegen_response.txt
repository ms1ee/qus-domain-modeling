"""Process-wide key/value store used to share runtime state between modules.

The store holds shared values such as the rider list (key ``"riders"``) and
the simulated clock (key ``"current_tick"``). Call :func:`_init` to reset the
store to its default contents.
"""

from typing import Any, Dict

_global_vars: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store and install the default keys.

    Returns:
        None
    """
    global _global_vars
    _global_vars = {
        "riders": [],
        "current_tick": 0,
    }


def set_value(key: str, value: Any) -> None:
    """Store a value under the given key.

    Args:
        key: Name of the global variable.
        value: Value to store.

    Returns:
        None
    """
    _global_vars[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Fetch a value from the global store.

    Args:
        key: Name of the global variable.
        default: Value returned when the key is missing.

    Returns:
        The stored value, or ``default`` when the key does not exist.
    """
    return _global_vars.get(key, default)

"""Grid geometry for the 800x800 food delivery map.

The map is an 8x8 arrangement of 100x100 cells. Points whose coordinates are
both multiples of 100 are *locations* (restaurants, customers and street
intersections). Points where exactly one coordinate is a multiple of 100 lie
on a *street*. All other points are inside city blocks and are not drivable.
"""

import math
from typing import List, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
DIMENSIONS = MAP_SIZE // CELL_SIZE
POINT_STEP = 50  # Sampling resolution used for road sampling and ASCII map.


def _is_int(value) -> bool:
    """Return True when ``value`` is an int but not a bool.

    Args:
        value: Any object.

    Returns:
        bool: Whether the value is a genuine integer.
    """
    return isinstance(value, int) and not isinstance(value, bool)


class Grid:
    """Coordinate helper describing the dispatch grid."""

    def __init__(self, map_size: int = MAP_SIZE, cell_size: int = CELL_SIZE):
        """Create a grid description.

        Args:
            map_size: Side length of the square map in units.
            cell_size: Side length of one grid cell in units.

        Raises:
            ValueError: If the sizes are not positive or do not divide evenly.
        """
        if map_size <= 0 or cell_size <= 0 or map_size % cell_size != 0:
            raise ValueError("map_size must be a positive multiple of cell_size")
        self.map_size = map_size
        self.cell_size = cell_size
        self.dimensions = map_size // cell_size
        self.half_cell = cell_size // 2

    def is_valid_coord(self, x, y) -> bool:
        """Check that both coordinates are integers inside the map.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            bool: True when the point lies within 0..map_size on both axes.
        """
        return (
            _is_int(x)
            and _is_int(y)
            and 0 <= x <= self.map_size
            and 0 <= y <= self.map_size
        )

    def clamp(self, value) -> int:
        """Clamp a value into the map range.

        Args:
            value: Numeric coordinate.

        Returns:
            int: Value limited to 0..map_size.
        """
        return max(0, min(self.map_size, int(value)))

    def snap_to_location(self, x, y) -> Tuple[int, int]:
        """Transform arbitrary coordinates to the nearest grid location.

        Locations are the points at integer hundreds (e.g. 100, 200).

        Args:
            x: X coordinate (may be any integer in range).
            y: Y coordinate.

        Returns:
            Tuple[int, int]: Snapped (x, y) location within the map.
        """
        cs = self.cell_size
        snapped_x = math.floor(x / cs + 0.5) * cs
        snapped_y = math.floor(y / cs + 0.5) * cs
        return self.clamp(snapped_x), self.clamp(snapped_y)

    def is_location(self, x: int, y: int) -> bool:
        """Return True if the point is a grid location (intersection).

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            bool: Whether both coordinates are multiples of the cell size.
        """
        return x % self.cell_size == 0 and y % self.cell_size == 0

    def is_horizontal_street(self, y: int) -> bool:
        """Return True if the given y value lies on a horizontal street.

        Args:
            y: Y coordinate.

        Returns:
            bool: Whether y is a grid line.
        """
        return y % self.cell_size == 0

    def is_vertical_street(self, x: int) -> bool:
        """Return True if the given x value lies on a vertical street.

        Args:
            x: X coordinate.

        Returns:
            bool: Whether x is a grid line.
        """
        return x % self.cell_size == 0

    def is_road(self, x: int, y: int) -> bool:
        """Return True if the point lies on any street.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            bool: Whether the point is drivable.
        """
        return self.is_horizontal_street(y) or self.is_vertical_street(x)

    def cell_of(self, x: int, y: int) -> Tuple[int, int]:
        """Return the (grid_x, grid_y) index of the cell containing a point.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            Tuple[int, int]: Cell indices in 0..dimensions-1.
        """
        gx = min(x // self.cell_size, self.dimensions - 1)
        gy = min(y // self.cell_size, self.dimensions - 1)
        return gx, gy

    def cell_center(self, x: int, y: int) -> Tuple[int, int]:
        """Return the center point of the cell containing a point.

        Uses ``center_x = (grid_x // 100) * 100 + 50``.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            Tuple[int, int]: Cell center coordinates.
        """
        gx, gy = self.cell_of(x, y)
        return gx * self.cell_size + self.half_cell, gy * self.cell_size + self.half_cell

    def locations(self) -> List[Tuple[int, int]]:
        """Return every grid location (intersection) on the map.

        Returns:
            List[Tuple[int, int]]: All lattice points.
        """
        cs = self.cell_size
        return [
            (x, y)
            for x in range(0, self.map_size + 1, cs)
            for y in range(0, self.map_size + 1, cs)
        ]

    def road_points(self, step: int = POINT_STEP) -> List[Tuple[int, int]]:
        """Return sampled points that lie on streets.

        Args:
            step: Sampling resolution.

        Returns:
            List[Tuple[int, int]]: Drivable sample points.
        """
        return [
            (x, y)
            for x in range(0, self.map_size + 1, step)
            for y in range(0, self.map_size + 1, step)
            if self.is_road(x, y)
        ]

"""Grid-based path planning for riders.

Routes always travel along streets: a rider on a horizontal street moves
horizontally first, otherwise it moves vertically first. Right-angle turns
happen at street intersections, and straight-line movement is used when the
rider is directly above or below its target.
"""

import math
from dataclasses import dataclass, field
from typing import List, Sequence, Tuple

from grid import Grid

Point = Tuple[int, int]


@dataclass
class Route:
    """A planned path made of connected waypoints.

    Attributes:
        waypoints: Ordered corner points from start to final destination.
        total_distance: Manhattan length of the route in units.
        estimated_time: Estimated number of ticks needed to traverse it.
    """

    waypoints: List[Point] = field(default_factory=list)
    total_distance: int = 0
    estimated_time: int = 0


def _move_toward(current: int, target: int, step: int) -> int:
    """Move a single coordinate toward a target by at most ``step``.

    Args:
        current: Current coordinate value.
        target: Target coordinate value.
        step: Maximum movement magnitude.

    Returns:
        int: New coordinate value, never overshooting the target.
    """
    if current < target:
        return min(current + step, target)
    if current > target:
        return max(current - step, target)
    return current


def _unique_consecutive(points: Sequence[Point]) -> List[Point]:
    """Remove consecutive duplicate points from a list.

    Args:
        points: Sequence of points.

    Returns:
        List[Point]: Points with consecutive duplicates removed.
    """
    result: List[Point] = []
    for point in points:
        if not result or result[-1] != point:
            result.append(point)
    return result


class PathPlanner:
    """Computes street-following routes and per-tick movement steps."""

    def __init__(self, grid: Grid = None):
        """Create a planner.

        Args:
            grid: Grid description to use. A default 800x800 grid is created
                when omitted.
        """
        self.grid = grid or Grid()
        self.algorithm = "street-first-manhattan"

    def next_step(
        self, x: int, y: int, target_x: int, target_y: int, speed: int
    ) -> Tuple[int, int, bool]:
        """Advance one tick toward a target along the street network.

        Args:
            x: Current X coordinate.
            y: Current Y coordinate.
            target_x: Target X coordinate.
            target_y: Target Y coordinate.
            speed: Maximum units to move this tick.

        Returns:
            Tuple[int, int, bool]: New (x, y) and whether the target was reached.
        """
        if x == target_x and y == target_y:
            return x, y, True
        step = max(1, int(speed))
        if self.grid.is_horizontal_street(y) and x != target_x:
            x = _move_toward(x, target_x, step)
        elif y != target_y:
            y = _move_toward(y, target_y, step)
        else:
            x = _move_toward(x, target_x, step)
        return x, y, (x == target_x and y == target_y)

    def plan(self, start: Point, goal: Point, speed: int = 1) -> Route:
        """Plan a single-leg street route from start to goal.

        Args:
            start: Starting point.
            goal: Destination point (a grid location).
            speed: Rider speed used for the time estimate.

        Returns:
            Route: Waypoints, Manhattan distance and estimated ticks.
        """
        sx, sy = start
        gx, gy = goal
        if self.grid.is_horizontal_street(sy) and sx != gx:
            corner = (gx, sy)
        else:
            corner = (sx, gy)
        waypoints = _unique_consecutive([start, corner, goal])
        distance = abs(gx - sx) + abs(gy - sy)
        return Route(
            waypoints=waypoints,
            total_distance=distance,
            estimated_time=math.ceil(distance / max(1, speed)),
        )

    def plan_multi(self, start: Point, stops: Sequence[Point], speed: int = 1) -> Route:
        """Plan a continuous route visiting several stops in order.

        Args:
            start: Starting point.
            stops: Ordered destinations (e.g. restaurant then customer).
            speed: Rider speed used for the time estimate.

        Returns:
            Route: Combined route across all legs.
        """
        waypoints: List[Point] = [start]
        total = 0
        current = start
        for stop in stops:
            leg = self.plan(current, stop, speed)
            waypoints.extend(leg.waypoints[1:])
            total += leg.total_distance
            current = stop
        waypoints = _unique_consecutive(waypoints)
        return Route(
            waypoints=waypoints,
            total_distance=total,
            estimated_time=math.ceil(total / max(1, speed)),
        )

"""Order data model for the dispatch system."""

import itertools
from typing import Sequence, Tuple

from grid import Grid

ORDER_STATUSES = ("pending", "assigned", "picking_up", "delivering", "completed")

_ORDER_ID_SEQUENCE = itertools.count(1)
_GRID = Grid()


def _validated_point(point: Sequence[int], label: str) -> Tuple[int, int]:
    """Validate a user supplied (x, y) pair.

    Args:
        point: Two-element sequence of integers.
        label: Human readable name used in error messages.

    Returns:
        Tuple[int, int]: The validated coordinates.

    Raises:
        ValueError: If the pair is malformed, non-integer or out of range.
    """
    if len(point) != 2:
        raise ValueError(f"{label} location must have exactly two coordinates")
    x, y = point
    if not (isinstance(x, int) and isinstance(y, int)) or isinstance(x, bool) or isinstance(y, bool):
        raise ValueError(f"{label} coordinates must be integers")
    if not _GRID.is_valid_coord(x, y):
        raise ValueError(
            f"{label} coordinates must be within 0-{_GRID.map_size}, got ({x}, {y})"
        )
    return x, y


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        res_pos: Restaurant location snapped to the grid, (x, y).
        cus_pos: Customer location snapped to the grid, (x, y).
        status: One of ``ORDER_STATUSES``; ``"completed"`` after delivery.
        rider_id: Identifier of the assigned rider, or None.
        start_time: Tick at which the order was created.
        completion_time: Tick at which the order was delivered, or None.
        distance: Manhattan distance between restaurant and customer.
        time_taken: Ticks from creation to delivery, or None.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int], start_time: int = 0):
        """Create an order and transform its coordinates to grid locations.

        Args:
            res_pos: Restaurant coordinates (x, y), range 0-800.
            cus_pos: Customer coordinates (x, y), range 0-800.
            start_time: Tick at which the order is created.

        Raises:
            ValueError: If coordinates are malformed or out of range.
        """
        rx, ry = _validated_point(res_pos, "restaurant")
        cx, cy = _validated_point(cus_pos, "customer")
        self.id = next(_ORDER_ID_SEQUENCE)
        self.res_pos = _GRID.snap_to_location(rx, ry)
        self.cus_pos = _GRID.snap_to_location(cx, cy)
        self.restaurant_x, self.restaurant_y = self.res_pos
        self.customer_x, self.customer_y = self.cus_pos
        self.status = "pending"
        self.rider_id = None
        self.start_time = int(start_time)
        self.completion_time = None
        self.distance = abs(self.restaurant_x - self.customer_x) + abs(
            self.restaurant_y - self.customer_y
        )
        self.time_taken = None

    @classmethod
    def from_sequence(cls, values: Sequence[int], start_time: int = 0) -> "Order":
        """Create an order from the ``[restaurant_x, restaurant_y, customer_x, customer_y]`` format.

        Args:
            values: Four integers.
            start_time: Tick at which the order is created.

        Returns:
            Order: The new order.

        Raises:
            ValueError: If the sequence does not contain exactly four values.
        """
        if len(values) != 4:
            raise ValueError("order requires four coordinates")
        return cls(values[0:2], values[2:4], start_time)

    def assign_to(self, rider_id: int) -> None:
        """Record assignment to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.

        Returns:
            None
        """
        self.rider_id = rider_id
        self.status = "assigned"

    def mark_picking_up(self) -> None:
        """Mark that the rider is heading to the restaurant.

        Returns:
            None
        """
        self.status = "picking_up"

    def mark_delivering(self) -> None:
        """Mark that the rider has picked up and is heading to the customer.

        Returns:
            None
        """
        self.status = "delivering"

    def complete(self, tick: int) -> None:
        """Mark the order as delivered and record efficiency data.

        Args:
            tick: Simulation tick at which delivery happened.

        Returns:
            None
        """
        self.status = "completed"
        self.completion_time = int(tick)
        self.time_taken = self.completion_time - self.start_time

    def is_completed(self) -> bool:
        """Return True if the order has been delivered.

        Returns:
            bool: Completion flag.
        """
        return self.status == "completed"

    def to_dict(self) -> dict:
        """Serialize the order to a JSON-compatible dictionary.

        Returns:
            dict: Order attributes.
        """
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

"""Rider model: state machine, movement and order execution."""

import logging
import random

import globalvar
from grid import Grid
from path_planner import PathPlanner

LOGGER = logging.getLogger(__name__)

DEFAULT_SPEED = 20
STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}
MAX_LOG_ENTRIES = 10000

_GRID = Grid()
_PLANNER = PathPlanner(_GRID)


def _current_tick() -> int:
    """Return the simulated clock value from the global store.

    Returns:
        int: Current tick.
    """
    return int(globalvar.get_value("current_tick", 0) or 0)


class Rider:
    """A delivery rider that moves on the street network and executes orders.

    Attributes:
        rider_id: Unique identifier.
        x: Current X coordinate.
        y: Current Y coordinate.
        speed: Movement speed in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Assigned orders; the first entry is the active order.
        special_case: True while travelling directly above/below the target.
        log: Chronological status-change records. Each entry has the keys
            ``event`` and ``order_id`` plus tick, position and state details.
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider on a random street point.

        Args:
            rider_id: Unique identifier assigned by the system.
        """
        self.rider_id = int(rider_id)
        self.x, self.y = random.choice(_GRID.road_points())
        self.speed = DEFAULT_SPEED
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    @property
    def position(self):
        """Return the current position.

        Returns:
            Tuple[int, int]: (x, y).
        """
        return self.x, self.y

    @property
    def state_name(self) -> str:
        """Return a human readable state name.

        Returns:
            str: Name of the current state.
        """
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def is_idle(self) -> bool:
        """Return True when the rider can accept a new order.

        Returns:
            bool: Idle state with no assigned orders.
        """
        return self.state == STATE_IDLE and not self.orders

    def _record(self, event: str, order_id) -> None:
        """Append a status change entry to the rider log.

        Args:
            event: Event name such as ``"accepted_order"``.
            order_id: Related order identifier, or None.

        Returns:
            None
        """
        self.log.append(
            {
                "event": event,
                "order_id": order_id,
                "tick": _current_tick(),
                "rider_id": self.rider_id,
                "x": self.x,
                "y": self.y,
                "state": self.state,
            }
        )
        if len(self.log) > MAX_LOG_ENTRIES:
            del self.log[: len(self.log) - MAX_LOG_ENTRIES]

    def _advance(self, target) -> bool:
        """Move one tick toward a target along the streets.

        Args:
            target: Destination (x, y).

        Returns:
            bool: True if the rider reached the target this tick.
        """
        tx, ty = target
        nx, ny, arrived = _PLANNER.next_step(self.x, self.y, tx, ty, self.speed)
        self.x, self.y = nx, ny
        self.special_case = nx == tx and ny != ty
        return arrived

    def _begin_order(self, order) -> None:
        """Start working on the given order (idle -> picking up).

        Args:
            order: The order to start.

        Returns:
            None
        """
        self.state = STATE_PICKING_UP
        self.special_case = False
        order.mark_picking_up()
        self._record("accepted_order", order.id)

    def _recover(self) -> None:
        """Restore a consistent state after an unexpected error.

        Returns:
            None
        """
        if not self.orders:
            self.state = STATE_IDLE
            self.special_case = False
        elif self.state == STATE_IDLE:
            self._begin_order(self.orders[0])

    def move(self) -> None:
        """Advance this rider by one tick, handling state transitions.

        Transitions:
            idle -> picking up when an order is present;
            picking up -> delivering on arrival at the restaurant;
            delivering -> idle (or picking up the next order) on delivery.

        Returns:
            None
        """
        try:
            self._advance_state()
        except Exception:  # pragma: no cover - defensive recovery path
            LOGGER.exception("Rider %s failed during move; recovering", self.rider_id)
            self._recover()

    def _advance_state(self) -> None:
        """Perform the state machine step for one tick.

        Returns:
            None
        """
        if not self.orders:
            if self.state != STATE_IDLE:
                self.state = STATE_IDLE
                self.special_case = False
            return

        if self.state == STATE_IDLE:
            self._begin_order(self.orders[0])

        if self.state == STATE_PICKING_UP:
            order = self.orders[0]
            if self._advance(order.res_pos):
                self.state = STATE_DELIVERING
                order.mark_delivering()
                self._record("arrived_at_merchant", order.id)
            return

        if self.state == STATE_DELIVERING:
            order = self.orders[0]
            if self._advance(order.cus_pos):
                order.complete(_current_tick())
                self.orders.pop(0)
                self._record("delivered_to_customer", order.id)
                self.special_case = False
                if self.orders:
                    self._begin_order(self.orders[0])
                else:
                    self.state = STATE_IDLE

    def to_dict(self) -> dict:
        """Serialize the rider to a JSON-compatible dictionary.

        Returns:
            dict: Rider attributes and order identifiers.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "state_name": self.state_name,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
            "log": list(self.log),
        }

"""Order management and distance-first rider assignment."""

from typing import TYPE_CHECKING, List, Optional, Sequence

import globalvar
from order_model import Order

if TYPE_CHECKING:  # pragma: no cover
    from rider import Rider

DISTANCE_WEIGHT = 1
LOAD_PENALTY = 200  # Distance-equivalent penalty per active order (load balancing).


def find_best_rider_for_order(order: Order) -> "Rider":
    """Select the most suitable rider for an order.

    The score is the Manhattan distance from the rider to the restaurant plus a
    load penalty per already assigned order. The lowest score wins; ties are
    broken by the smaller rider ID.

    Args:
        order: Order that needs a rider.

    Returns:
        Rider: The chosen rider, or None when no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    rx, ry = order.res_pos
    best = None
    best_score = None
    for rider in riders:
        distance = abs(rider.x - rx) + abs(rider.y - ry)
        score = (
            DISTANCE_WEIGHT * distance + LOAD_PENALTY * len(rider.orders),
            rider.rider_id,
        )
        if best_score is None or score < best_score:
            best = rider
            best_score = score
    return best


class OrderManager:
    """Creates, assigns and tracks orders.

    Attributes:
        orders: All orders created during the session.
    """

    def __init__(self) -> None:
        """Create an empty order manager."""
        self.orders: List[Order] = []

    def create_order(self, res_pos: Sequence[int], cus_pos: Sequence[int], tick: int = 0) -> Order:
        """Create and register a new order.

        Args:
            res_pos: Restaurant coordinates.
            cus_pos: Customer coordinates.
            tick: Current simulated tick.

        Returns:
            Order: The newly created order.
        """
        order = Order(res_pos, cus_pos, start_time=tick)
        self.orders.append(order)
        return order

    def assign_order(self, order: Order) -> Optional["Rider"]:
        """Assign a pending order to the best rider.

        Args:
            order: A pending order.

        Returns:
            Rider: The rider now responsible for the order, or None if no
            rider is available.

        Raises:
            ValueError: If the order is not pending.
        """
        if order.status != "pending":
            raise ValueError(f"order {order.id} is already {order.status}")
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider

    def discard(self, order: Order) -> None:
        """Remove a pending order that could not be assigned.

        Args:

import json

DEFAULT_PARAMETERS: Dict[str, int] = {
    "initial_funds": 5000,
    "rider_recruitment_cost": 500,
    "max_riders": 20,
    "rider_speed": 20,
}


class SystemConfig:
    """Loads system parameters from and saves state to a JSON file.

    Attributes:
        config_file_path: Path of the JSON configuration file.
        parameters: Effective system parameters.
        saved_state: State dictionary read from the file, if any.
    """

    def __init__(self, config_file_path: str = "config.json") -> None:
        """Create a configuration holder with default parameters.

        Args:
            config_file_path: Location of the JSON file.
        """
        self.config_file_path = config_file_path
        self.parameters: Dict[str, Any] = dict(DEFAULT_PARAMETERS)
        self.saved_state: Optional[dict] = None

    def load(self) -> Dict[str, Any]:
        """Load parameters from the configuration file when it exists.

        Invalid or unreadable files are logged and the defaults are kept.

        Returns:
            Dict[str, Any]: Effective parameters.
        """
        if not os.path.exists(self.config_file_path):
            LOGGER.info("No configuration file at %s; using defaults", self.config_file_path)
            return dict(self.parameters)
        try:
            with open(self.config_file_path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
        except (OSError, json.JSONDecodeError) as exc:
            LOGGER.error("Could not read configuration: %s", exc)
            return dict(self.parameters)
        if not isinstance(data, dict):
            LOGGER.error("Configuration root must be an object; using defaults")
            return dict(self.parameters)
        params = data.get("parameters", data)
        if isinstance(params, dict):
            for key, default in DEFAULT_PARAMETERS.items():
                value = params.get(key, default)
                if isinstance(value, int) and not isinstance(value, bool) and value >= 0:
                    self.parameters[key] = value
                else:
                    LOGGER.warning("Invalid value for %s; using %s", key, default)
        state = data.get("state")
        self.saved_state = state if isinstance(state, dict) else None
        return dict(self.parameters)

    def save(self, state: dict) -> str:
        """Write parameters and a state snapshot to the configuration file.

        Args:
            state: JSON-compatible snapshot of the simulation.

        Returns:
            str: Path of the written file.

        Raises:
            OSError: If the file cannot be written.
        """
        payload = {"parameters": self.parameters, "state": state}
        tmp_path = self.config_file_path + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as handle:
            json.dump(payload, handle, indent=2)
        os.replace(tmp_path, self.config_file_path)
        return self.config_file_path

"""Core system manager: funds, rider recruitment, order submission and clock."""

import logging
from typing import Dict, List, Optional

import globalvar
from config import DEFAULT_PARAMETERS
from grid import Grid
from order import OrderManager
from order_model import Order
from path_planner import PathPlanner, Route
from rider import STATE_IDLE, Rider

LOGGER = logging.getLogger(__name__)


class SystemManager:
    """Manages funds, riders, orders and the simulation clock.

    Attributes:
        funds: System funds (may go negative).
        rider_id_counter: Last rider ID issued.
        running_state: True while the main loop should keep running.
        current_tick: Simulated time.
        rider_recruitment_cost: Cost deducted for each new rider.
        max_riders: Upper bound on rider count.
        rider_speed: Speed given to newly recruited riders.
        grid: Map description.
        planner: Path planner used for route previews.
        order_manager: Order creation and assignment component.
    """

    def __init__(self, parameters: Optional[Dict] = None) -> None:
        """Create the system with the supplied or default parameters.

        Args:
            parameters: Optional overrides for DEFAULT_PARAMETERS.
        """
        params = dict(DEFAULT_PARAMETERS)
        params.update(parameters or {})
        self.funds = int(params["initial_funds"])
        self.rider_id_counter = 0
        self.running_state = True
        self.current_tick = 0
        self.rider_recruitment_cost = int(params["rider_recruitment_cost"])
        self.max_riders = int(params["max_riders"])
        self.rider_speed = int(params["rider_speed"])
        self.grid = Grid()
        self.planner = PathPlanner(self.grid)
        self.order_manager = OrderManager()
        globalvar.set_value("riders", [])
        globalvar.set_value("current_tick", 0)

    @property
    def riders(self) -> List[Rider]:
        """Return the shared rider list.

        Returns:
            List[Rider]: Riders currently managed by the system.
        """
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    def has_idle_rider(self) -> bool:
        """Return True if at least one rider has no assigned work.

        Returns:
            bool: Idle availability flag.
        """
        return any(rider.is_idle for rider in self.riders)

    def can_recruit(self) -> bool:
        """Return True when funds allow and the rider cap is not reached.

        Returns:
            bool: Recruitment feasibility.
        """
        return len(self.riders) < self.max_riders and self.funds >= self.rider_recruitment_cost

    def recruit_rider(self) -> Optional[Rider]:
        """Create a new rider and charge the recruitment cost.

        Returns:
            Rider: The new rider, or None if recruitment is not possible.
        """
        if not self.can_recruit():
            return None
        self.rider_id_counter += 1
        rider = Rider(self.rider_id_counter)
        rider.speed = self.rider_speed
        self.funds -= self.rider_recruitment_cost
        self.riders.append(rider)
        LOGGER.info("Recruited rider %s (funds now %s)", rider.rider_id, self.funds)
        return rider

    def submit_order(self, res_pos, cus_pos) -> dict:
        """Create an order, recruit if necessary, assign it and plan its route.

        Args:
            res_pos: Restaurant coordinates (x, y).
            cus_pos: Customer coordinates (x, y).

        Returns:
            dict: Keys ``order`` (Order), ``rider`` (Rider), ``new_rider``
            (bool) and ``route`` (Route).

        Raises:
            ValueError: If coordinates are invalid.
            RuntimeError: If no rider is available and none can be recruited.
        """
        order: Order = self.order_manager.create_order(res_pos, cus_pos, self.current_tick)
        new_rider = False
        try:
            if not self.has_idle_rider() and self.recruit_rider() is not None:
                new_rider = True
            if not self.riders:
                raise RuntimeError("No riders available and insufficient funds to recruit one.")
            rider = self.order_manager.assign_order(order)
            if rider is None:
                raise RuntimeError("No rider could accept the order.")
        except Exception:
            self.order_manager.discard(order)
            raise
        route: Route = self.planner.plan_multi(
            rider.position, [order.res_pos, order.cus_pos], rider.speed
        )
        LOGGER.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return {"order": order, "rider": rider, "new_rider": new_rider, "route": route}

    def tick(self) -> List[str]:
        """Advance the simulation by one tick.

        Returns:
            List[str]: Consistency issues detected (and repaired) this tick.
        """
        self.current_tick += 1
        globalvar.set_value("current_tick", self.current_tick)
        for rider in list(self.riders):
            rider.move()
        return self.check_consistency()

    def run_ticks(self, count: int) -> List[str]:
        """Advance the simulation several ticks.

        Args:
            count: Number of ticks to run.

        Returns:
            List[str]: All consistency issues found during the run.
        """
        issues: List[str] = []
        for _ in range(count):
            issues.extend(self.tick())
        return issues

    def check_consistency(self) -> List[str]:
        """Verify and repair internal state consistency.

        Returns:
            List[str]: Human readable descriptions of issues found.
        """
        issues: List[str] = []
        seen_ids = set()
        for rider in self.riders:
            if rider.rider_id in seen_ids:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen_ids.add(rider.rider_id)
            if rider.state == STATE_IDLE and rider.orders:
                issues.append(f"rider {rider.rider_id} idle with orders; repairing")
                rider._recover()
            completed = [o for o in rider.orders if o.is_completed()]
            if completed:
                issues.append(f"rider {rider.rider_id} holds completed orders; removing")
                rider.orders = [o for o in rider.orders if not o.is_completed()]
        if len(self.riders) > self.max_riders:
            issues.append("rider count exceeds configured maximum")
        for message in issues:
            LOGGER.warning(message)
        return issues

    def snapshot(self) -> dict:
        """Return a JSON-compatible snapshot of the whole simulation.

        Returns:
            dict: Funds, clock, rider and order data.
        """
        return {
            "funds": self.funds,
            "current_tick": self.current_tick,
            "rider_id_counter": self.rider_id_counter,
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in self.order_manager.orders],
        }

    def shutdown(self) -> None:
        """Stop the main loop.

        Returns:
            None
        """
        self.running_state = False


def init_system(parameters: Optional[Dict] = None) -> SystemManager:
    """Reset global state and create a ready-to-use system manager.

    Args:
        parameters: Optional parameter overrides.

    Returns:
        SystemManager: The initialized system.
    """
    globalvar._init()
    return SystemManager(parameters)

from typing import Iterable, List, Optional, Sequence, Tuple

from grid import POINT_STEP, Grid
from rider import Rider

Point = Tuple[int, int]


class DisplayManager:
    """Renders maps, status tables and logs as text.

    Attributes:
        grid: Map description.
        symbol_map: Symbols used on the ASCII map.
    """

    def __init__(self, grid: Optional[Grid] = None) -> None:
        """Create a display manager.

        Args:
            grid: Map description; a default 800x800 grid when omitted.
        """
        self.grid = grid or Grid()
        self.symbol_map = {
            "empty": "+",
            "street_h": "-",
            "street_v": "|",
            "block": " ",
            "restaurant": "R",
            "customer": "C",
            "route": "*",
        }

    def _size(self) -> int:
        """Return the canvas dimension (one character per 50 units).

        Returns:
            int: Number of rows/columns on the canvas.
        """
        return self.grid.map_size // POINT_STEP + 1

    def _base_canvas(self) -> List[List[str]]:
        """Build the empty street canvas.

        Returns:
            List[List[str]]: 2D character grid.
        """
        n = self._size()
        canvas = [[" "] * n for _ in range(n)]
        for row in range(n):
            for col in range(n):
                x, y = col * POINT_STEP, row * POINT_STEP
                if self.grid.is_location(x, y):
                    key = "empty"
                elif self.grid.is_horizontal_street(y):
                    key = "street_h"
                elif self.grid.is_vertical_street(x):
                    key = "street_v"
                else:
                    key = "block"
                canvas[row][col] = self.symbol_map[key]
        return canvas

    def _cell(self, point: Point) -> Tuple[int, int]:
        """Convert a coordinate to canvas (row, col).

        Args:
            point: (x, y) coordinate.

        Returns:
            Tuple[int, int]: (row, col) index.
        """
        return point[1] // POINT_STEP, point[0] // POINT_STEP

    def _overlay_route(self, canvas: List[List[str]], waypoints: Sequence[Point]) -> None:
        """Mark route segments on the canvas with the route symbol.

        Args:
            canvas: Canvas to modify in place.
            waypoints: Ordered route corners.

        Returns:
            None
        """
        symbol = self.symbol_map["route"]
        cells = [(x // POINT_STEP, y // POINT_STEP) for x, y in waypoints]
        for (c1, r1), (c2, r2) in zip(cells, cells[1:]):
            if r1 == r2:
                targets = [(r1, c) for c in range(min(c1, c2), max(c1, c2) + 1)]
            elif c1 == c2:
                targets = [(r, c1) for r in range(min(r1, r2), max(r1, r2) + 1)]
            else:
                continue
            for r, c in targets:
                if canvas[r][c] in ("-", "|", "+"):
                    canvas[r][c] = symbol

    def _put(self, canvas: List[List[str]], point: Point, symbol: str) -> None:
        """Place a symbol at a coordinate if it lies on the canvas.

        Args:
            canvas: Canvas to modify.
            point: (x, y) coordinate.
            symbol: Character to place.

        Returns:
            None
        """
        row, col = self._cell(point)
        if 0 <= row < len(canvas) and 0 <= col < len(canvas[0]):
            canvas[row][col] = symbol

    def _format(self, canvas: List[List[str]]) -> str:
        """Convert the canvas to text with y-axis labels.

        Args:
            canvas: Character grid.

        Returns:
            str: Multi-line map text.
        """
        lines = []
        for row_index, row in enumerate(canvas):
            y = row_index * POINT_STEP
            label = f"{y:>3} " if y % self.grid.cell_size == 0 else "    "
            lines.append(label + " ".join(row))
        lines.append("    x: 0 .. %d (lattice every %d units)" % (
            self.grid.map_size, self.grid.cell_size))
        return "\n".join(lines)

    def render_map(
        self,
        riders: Iterable[Rider],
        orders: Iterable = (),
        route=None,
    ) -> str:
        """Render riders, active orders and an optional route on the map.

        Args:
            riders: Riders to draw (digit = rider ID mod 10).
            orders: Orders whose restaurants (R) and customers (C) are drawn.
            route: Optional Route to highlight with ``*``.

        Returns:
            str: ASCII map.
        """
        canvas = self._base_canvas()
        if route is not None:
            self._overlay_route(canvas, route.waypoints)
        for order in orders:
            if order.is_completed():
                continue
            self._put(canvas, order.res_pos, self.symbol_map["restaurant"])
            self._put(canvas, order.cus_pos, self.symbol_map["customer"])
        for rider in riders:
            self._put(canvas, rider.position, str(rider.rider_id % 10))
        return self._format(canvas)

    def render_legend(self) -> str:
        """Return the map legend.

        Returns:
            str: Legend text.
        """
        return (
            "Legend: + intersection  - | street  R restaurant  C customer  "
            "* route  0-9 rider (id mod 10)"
        )

    def render_status(self, system) -> str:
        """Render a snapshot of funds, clock and all riders.

        Args:
            system: SystemManager instance.

        Returns:
            str: Status text.
        """
        lines = [
            f"Tick: {system.current_tick}",
            f"System funds: {system.funds}",
            f"Riders: {len(system.riders)} (max {system.max_riders})",
            f"Active orders: {len(system.order_manager.active_orders())}",
        ]
        for rider in system.riders:

    def render_orders(self, orders: Sequence) -> str:

        if not orders:

from typing import List, Tuple

from grid import MAP_SIZE

COMMAND_ALIASES = {
    "start": "start",
    "s": "start",
    "status": "status",
    "st": "status",
    "orders": "orders",
    "o": "orders",
    "riders": "riders",
    "r": "riders",
    "map": "map",
    "m": "map",
    "tick": "tick",
    "t": "tick",
    "history": "history",
    "h": "history",
    "help": "help",
    "?": "help",
    "quit": "quit",
    "exit": "quit",
    "q": "quit",
}

MAX_TICK_STEP = 1000


class InputManager:
    """Parses user commands and validates their arguments.

    Attributes:
        command_history: Commands entered so far, oldest first.
    """

    def __init__(self) -> None:
        """Create an input manager with empty history."""
        self.command_history: List[str] = []

    def expand(self, line: str) -> str:
        """Resolve history shortcuts (``!!`` and ``!n``).

        Args:
            line: Raw user input.

        Returns:
            str: The command to execute.

        Raises:
            ValueError: If the referenced history entry does not exist.
        """
        text = line.strip()
        if text == "!!":
            if not self.command_history:
                raise ValueError("No previous command in history.")
            return self.command_history[-1]
        if text.startswith("!") and text[1:].isdigit():
            index = int(text[1:])
            if index < 1 or index > len(self.command_history):
                raise ValueError(f"History entry {index} does not exist.")
            return self.command_history[index - 1]
        return text

    def record(self, line: str) -> None:
        """Append a command to the history.

        Args:
            line: Command text.

        Returns:
            None
        """
        if line:
            self.command_history.append(line)

    def parse(self, line: str) -> Tuple[str, List[str]]:
        """Split a command line into a canonical command name and arguments.

        Args:
            line: Command text.

        Returns:
            Tuple[str, List[str]]: Canonical command and argument tokens.

        Raises:
            ValueError: If the command is unknown.
        """
        tokens = line.split()
        if not tokens:
            return "", []
        word = tokens[0].lower()
        if word not in COMMAND_ALIASES:
            raise ValueError(f"Unknown command '{word}'. Type 'help' for a list of commands.")
        return COMMAND_ALIASES[word], tokens[1:]

    @staticmethod
    def parse_coordinates(tokens: List[str], expected: int = 4) -> List[int]:
        """Parse and validate integer coordinates.

        Accepts space or comma separated values, optionally wrapped in brackets.

        Args:
            tokens: Argument tokens.
            expected: Required number of values.

        Returns:
            List[int]: Validated coordinates.

        Raises:
            ValueError: If the count, type or range is wrong.
        """
        text = " ".join(tokens).replace("[", " ").replace("]", " ").replace(",", " ")
        parts = text.split()
        if len(parts) != expected:
            raise ValueError(
                f"expected {expected} integers: restaurant_x restaurant_y "
                f"customer_x customer_y (got {len(parts)})"
            )
        values = []
        for part in parts:
            try:
                value = int(part)
            except ValueError:
                raise ValueError(f"'{part}' is not an integer") from None
            if not 0 <= value <= MAP_SIZE:
                raise ValueError(f"coordinate {value} is outside the range 0-{MAP_SIZE}")
            values.append(value)
        return values

    @staticmethod
    def parse_tick_count(tokens: List[str]) -> int:
        """Parse the optional step count for the ``tick`` command.

        Args:
            tokens: Argument tokens.

        Returns:
            int: Number of ticks (default 1).

        Raises:
            ValueError: If the value is not a positive integer within limits.
        """
        if not tokens:
            return 1
        try:
            count = int(tokens[0])
        except ValueError:
            raise ValueError(f"'{tokens[0]}' is not an integer") from None
        if not 1 <= count <= MAX_TICK_STEP:
            raise ValueError(f"tick count must be between 1 and {MAX_TICK_STEP}")
        return count

        """Create the command-line interface.

        Args:
            system: Simulation core.
            inputs: Input manager.
            display: Display manager.
            config: Configuration helper.
            persist: Save state to the configuration file on exit.
        """
        self.system = system
        self.inputs = inputs
        self.display = display
        self.config = config
        self.persist = persist
        self._handlers: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def run(self) -> None:
        """Run the interactive loop until the system stops.

        Returns:
            None
        """
        print(BANNER)
        while self.system.running_state:
            try:
                raw = input(self.PROMPT)
            except (EOFError, KeyboardInterrupt):
                print()
                self.cmd_quit([])
                break
            try:
                line = self.inputs.expand(raw)
            except ValueError as exc:
                print(f"Error: {exc}")
                continue
            if not line:
                continue
            self.inputs.record(line)
            try:
                self.dispatch(line)
            except ValueError as exc:
                print(f"Error: {exc}")
            except Exception:  # Keep the simulation alive on unexpected errors.
                LOGGER.exception("Unexpected error while handling %r", line)
                print("Internal error occurred; system state was preserved. See log.")

    def dispatch(self, line: str) -> None:
        """Parse and execute one command line.

        Args:
            line: Command text.

        Returns:
            None
        """
        command, args = self.inputs.parse(line)
        if not command:
            return
        self._handlers[command](args)

    def cmd_start(self, args: List[str]) -> None:
        """Submit a new order.

        Args:
            args: Optional four coordinates; prompted for when omitted.

        Returns:
            None
        """
        tokens = args
        if not tokens:
            try:
                tokens = input("Enter restaurant_x restaurant_y customer_x customer_y: ").split()
            except (EOFError, KeyboardInterrupt):
                print()
                return
        coords = self.inputs.parse_coordinates(tokens, 4)
        result = self.system.submit_order((coords[0], coords[1]), (coords[2], coords[3]))
        print(self.display.render_order_summary(result))
        print(
            self.display.render_map(
                self.system.riders,
                self.system.order_manager.orders,
                result["route"],
            )
        )
        print(self.display.render_legend())

    def cmd_status(self, args: List[str]) -> None:
        """Show the system status snapshot.

        Args:
            args: Unused.

        Returns:
            None
        """
        print(self.display.render_status(self.system))

    def cmd_orders(self, args: List[str]) -> None:
        """Show all order details.

        Args:
            args: Unused.

        Returns:
            None
        """
        print(self.display.render_orders(self.system.order_manager.orders))

    def cmd_riders(self, args: List[str]) -> None:
        """Show every rider's status-change log.

        Args:
            args: Unused.

        Returns:
            None
        """
        print(self.display.render_rider_logs(self.system.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map with riders and active orders.

        Args:
            args: Unused.

        Returns:
            None
        """
        print(
            self.display.render_map(self.system.riders, self.system.order_manager.orders)
        )
        print(self.display.render_legend())

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation.

        Args:
            args: Optional number of ticks.

        Returns:
            None
        """
        count = self.inputs.parse_tick_count(args)
        issues = self.system.run_ticks(count)
        active = len(self.system.order_manager.active_orders())
        print(f"Advanced {count} tick(s). Current tick: {self.system.current_tick}. "
              f"Active orders: {active}.")
        for issue in issues:
            print(f"Warning: {issue}")

    def cmd_history(self, args: List[str]) -> None:
        """Print command history.

        Args:
            args: Unused.

        Returns:
            None
        """
        if not self.inputs.command_history:
            print("History is empty.")
            return
        for index, command in enumerate(self.inputs.command_history, start=1):
            print(f"{index:>4}  {command}")

    def cmd_help(self, args: List[str]) -> None:
        """Print help text.

        Args:
            args: Unused.

        Returns:
            None
        """
        print(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Save state (when enabled) and stop the main loop.

        Args:
            args: Unused.

        Returns:
            None
        """
        if self.system.running_state and self.persist:
            try:
                path = self.config.save(self.system.snapshot())
                print(f"State saved to {path}.")
            except OSError as exc:
                print(f"Warning: could not save state: {exc}")
        self.system.shutdown()
        print("Goodbye.")


def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: Argument list; defaults to ``sys.argv[1:]``.

    Returns:
        argparse.Namespace: Parsed options.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default="config.json", help="JSON configuration file")
    parser.add_argument("--no-save", action="store_true", help="do not save state on exit")
    parser.add_argument(
        "--log-level",
        default="WARNING",
        choices=["DEBUG", "INFO", "WARNING", "ERROR"],
        help="logging verbosity",
    )
    return parser.parse_args(argv)


def main(argv: Optional[List[str]] = None) -> int:
    """Program entry point.

    Args:
        argv: Optional argument list.

    Returns:
        int: Process exit code.
    """
    args = parse_args(argv)
    logging.basicConfig(
        level=getattr(logging, args.log_level),
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    config = SystemConfig(args.config)
    parameters = config.load()
    system = init_system(parameters)
    cli = DispatchCLI(
        system=system,
        inputs=InputManager(),
        display=DisplayManager(),
        config=config,
        persist=not args.no_save,
    )
    cli.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Configuration loading for the food delivery dispatch simulator.

System parameters are read from an optional JSON file. Missing or invalid
entries fall back to the built-in defaults so the program always starts.
"""

import dataclasses
import json
import logging
import os
from dataclasses import dataclass
from typing import Any, Dict, Optional

logger = logging.getLogger("dispatch.config")

PROJECT_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
DEFAULT_CONFIG_PATH = os.path.join(PROJECT_ROOT, "config.json")
DEFAULT_STATE_PATH = os.path.join(PROJECT_ROOT, "dispatch_state.json")


@dataclass
class Configuration:
    """System parameters for the dispatch simulator.

    Attributes:
        initial_funds: Starting balance of the system (may go negative later).
        default_rider_speed: Base rider speed in units per tick.
        recruitment_cost: Funds deducted when a new rider is recruited.
        delivery_fee: Funds credited for each completed order.
        rider_upkeep: Funds deducted per tick for each busy rider.
        max_riders: Upper limit on the number of riders.
        max_orders_per_rider: Maximum active orders a single rider may hold.
        save_on_exit: Whether to save the state file when the program exits.
        state_file_path: JSON file used to persist the system state.
        log_level: Logging level name (e.g. "WARNING", "DEBUG").
        config_file_path: Path of the configuration file that was read.
    """

    initial_funds: int = 5000
    default_rider_speed: int = 50
    recruitment_cost: int = 300
    delivery_fee: int = 150
    rider_upkeep: int = 1
    max_riders: int = 20
    max_orders_per_rider: int = 3
    save_on_exit: bool = True
    state_file_path: str = DEFAULT_STATE_PATH
    log_level: str = "WARNING"
    config_file_path: str = DEFAULT_CONFIG_PATH

    @classmethod
    def load(cls, path: Optional[str] = None) -> "Configuration":
        """Load configuration from a JSON file, using defaults as fallback.

        Args:
            path: Optional path to the JSON configuration file.

        Returns:
            A Configuration instance with values from the file applied.
        """
        config_path = path or DEFAULT_CONFIG_PATH
        cfg = cls(config_file_path=config_path)
        if not os.path.exists(config_path):
            return cfg
        try:
            with open(config_path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
        except (OSError, json.JSONDecodeError) as exc:
            logger.warning(
                "Could not read configuration %s (%s); using defaults.",
                config_path, exc,
            )
            return cfg
        if not isinstance(data, dict):
            logger.warning("Configuration root must be an object; using defaults.")
            return cfg
        for fld in dataclasses.fields(cls):
            if fld.name == "config_file_path" or fld.name not in data:
                continue
            value = data[fld.name]
            current = getattr(cfg, fld.name)
            is_bool_mismatch = isinstance(current, int) and isinstance(value, bool)
            if isinstance(value, type(current)) and not is_bool_mismatch:
                setattr(cfg, fld.name, value)
            else:
                logger.warning("Ignoring invalid value for '%s'.", fld.name)
        return cfg

    def to_dict(self) -> Dict[str, Any]:
        """Return the configuration as a plain dictionary.

        Returns:
            Dictionary of configuration names to values.
        """
        return dataclasses.asdict(self)

    global _global_dict

    Args:
        key: Name of the global variable.

    """
    _global_dict[key] = value

    """Retrieve a value from the global store.

"""Grid map and coordinate utilities for the food delivery simulator.

The map is 800x800 units divided into an 8x8 grid of 100x100 cells.

* Street points: points whose x or y coordinate lies on a grid line
  (``x % 100 == 0`` or ``y % 100 == 0``). Riders move only along streets.
* Location points: restaurant and customer positions. They are cell centre
  points, ``center = (v // 100) * 100 + 50``, so ``x % 100 == 50`` and
  ``y % 100 == 50``. Each location point is adjacent to four street points.

Riders move on a 50-unit lattice, so every lattice point is either a street
point or a location point.
"""

from dataclasses import dataclass
from typing import List, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = 8
LATTICE_STEP = 50

Point = Tuple[int, int]


@dataclass
class Cell:
    """A single 100x100 grid cell.

    Attributes:
        grid_x: Column index of the cell (0-7).
        grid_y: Row index of the cell (0-7).
        is_road: True when the cell boundary is part of the road network.
        is_location_point: True when the cell centre is a location point.
        center_x: X coordinate of the cell centre.
        center_y: Y coordinate of the cell centre.
    """

    grid_x: int
    grid_y: int
    is_road: bool
    is_location_point: bool
    center_x: int
    center_y: int


def manhattan(a: Point, b: Point) -> int:
    """Return the Manhattan distance between two points.

    Args:
        a: First point (x, y).
        b: Second point (x, y).

    Returns:
        Sum of absolute coordinate differences.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def snap_value(value: int, map_length: int = MAP_SIZE,
               cell_size: int = CELL_SIZE) -> int:
    """Snap one coordinate to the centre of its grid cell.

    Args:
        value: Raw coordinate in the range 0-800.
        map_length: Total map length.
        cell_size: Size of one grid cell.

    Returns:
        Centre coordinate of the cell containing ``value``.
    """
    clamped = min(max(int(value), 0), map_length)
    size = map_length // cell_size
    index = min(clamped // cell_size, size - 1)
    return index * cell_size + cell_size // 2


def snap_point(x: int, y: int) -> Point:
    """Snap an (x, y) point to the nearest grid cell centre.

    Args:
        x: Raw x coordinate.
        y: Raw y coordinate.

    Returns:
        Grid cell centre point (x, y).
    """
    return snap_value(x), snap_value(y)


class GridMap:
    """An 8x8 grid representing an 800x800 coordinate system."""

    def __init__(self, size: int = GRID_DIMENSION,
                 cell_size: int = CELL_SIZE) -> None:
        """Create the grid and its 64 cells.

        Args:
            size: Number of cells per row and column.
            cell_size: Edge length of one cell in units.
        """
        self.size = size
        self.cell_size = cell_size
        self.map_length = size * cell_size
        self.cells: List[Cell] = []
        half = cell_size // 2
        for grid_y in range(size):
            for grid_x in range(size):
                self.cells.append(
                    Cell(
                        grid_x=grid_x,
                        grid_y=grid_y,
                        is_road=True,
                        is_location_point=True,
                        center_x=grid_x * cell_size + half,
                        center_y=grid_y * cell_size + half,
                    )
                )

    def cell_at(self, grid_x: int, grid_y: int) -> Cell:
        """Return the cell at the given grid indices.

        Args:
            grid_x: Column index (0 to size-1).
            grid_y: Row index (0 to size-1).

        Returns:
            The Cell object.

        Raises:
            IndexError: If the indices are outside the grid.
        """
        if not (0 <= grid_x < self.size and 0 <= grid_y < self.size):
            raise IndexError(f"cell ({grid_x}, {grid_y}) is outside the grid")
        return self.cells[grid_y * self.size + grid_x]

    def in_bounds(self, x: int, y: int) -> bool:
        """Check whether a point lies inside the map.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            True if 0 <= x, y <= map_length.
        """
        return 0 <= x <= self.map_length and 0 <= y <= self.map_length

    def is_street(self, x: int, y: int) -> bool:
        """Check whether a point lies on the street network.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            True if the point is on a grid line and inside the map.
        """
        if not self.in_bounds(x, y):
            return False
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def is_location(self, x: int, y: int) -> bool:
        """Check whether a point is a restaurant/customer location point.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            True if the point is a cell centre.
        """
        if not self.in_bounds(x, y):
            return False
        half = self.cell_size // 2
        return x % self.cell_size == half and y % self.cell_size == half

    def snap(self, x: int, y: int) -> Point:
        """Snap a raw point to the centre of its grid cell.

        Args:
            x: Raw x coordinate.
            y: Raw y coordinate.

        Returns:
            Cell centre point.
        """
        return (snap_value(x, self.map_length, self.cell_size),
                snap_value(y, self.map_length, self.cell_size))

    def lattice_points(self) -> List[Point]:
        """Return every 50-unit lattice point of the map.

        Returns:
            List of (x, y) lattice points.
        """
        points: List[Point] = []
        for y in range(0, self.map_length + 1, LATTICE_STEP):
            for x in range(0, self.map_length + 1, LATTICE_STEP):
                points.append((x, y))
        return points

"""Grid-based path planning for riders.

Riders travel along the street network. A Dijkstra search on the 50-unit
lattice minimises the number of steps first and the number of right-angle
turns second, which prioritises straight streets. Location points (cell
centres) can only be used as the start or goal of a path, so riders never
cut through a block.
"""

import heapq
import random
from typing import List, Optional, Sequence, Tuple

from grid_map import LATTICE_STEP, GridMap, Point

DIRECTIONS = (
    (LATTICE_STEP, 0),
    (-LATTICE_STEP, 0),
    (0, LATTICE_STEP),
    (0, -LATTICE_STEP),
)
NO_DIRECTION = (0, 0)
STEP_COST = 10
TURN_COST = 1


class PathPlanner:
    """Plan grid-based routes between points on the map."""

    def __init__(self, grid: Optional[GridMap] = None) -> None:
        """Create a planner for the given grid.

        Args:
            grid: Grid map to plan on. A default 8x8 map is created if None.
        """
        self.grid = grid if grid is not None else GridMap()
        self._street_points: List[Point] = [
            p for p in self.grid.lattice_points() if self.grid.is_street(*p)
        ]

    @property
    def street_points(self) -> List[Point]:
        """Return a copy of all street lattice points.

        Returns:
            List of street points.
        """
        return list(self._street_points)

    def _passable(self, point: Point, goal: Point) -> bool:
        """Return True if a point may be entered during the search.

        Args:
            point: Candidate point.
            goal: Destination of the current search.

        Returns:
            True for street points and for the goal itself.
        """
        return self.grid.is_street(point[0], point[1]) or point == goal

    def find_path(self, start: Point, goal: Point) -> List[Point]:
        """Find a street-following path from start to goal.

        The returned list starts with ``start`` and ends with ``goal``. An
        empty list is returned when no path exists.

        Args:
            start: Starting lattice point.
            goal: Destination lattice point.

        Returns:
            Ordered list of lattice points forming the path.
        """
        start = (int(start[0]), int(start[1]))
        goal = (int(goal[0]), int(goal[1]))
        if start == goal:
            return [start]

        start_state = (start, NO_DIRECTION)
        best = {start_state: 0}
        prev = {}
        heap = [(0, start, NO_DIRECTION)]
        end_state = None

        while heap:
            cost, current, direction = heapq.heappop(heap)
            state = (current, direction)
            if cost > best.get(state, float("inf")):
                continue
            if current == goal:
                end_state = state
                break
            for nd in DIRECTIONS:
                nxt = (current[0] + nd[0], current[1] + nd[1])
                if not self._passable(nxt, goal):
                    continue
                step = STEP_COST
                if direction != NO_DIRECTION and nd != direction:
                    step += TURN_COST
                new_cost = cost + step
                key = (nxt, nd)
                if new_cost < best.get(key, float("inf")):
                    best[key] = new_cost
                    prev[key] = state
                    heapq.heappush(heap, (new_cost, nxt, nd))

        if end_state is None:
            return []

        path: List[Point] = []
        state = end_state
        while True:
            path.append(state[0])
            if state == start_state:
                break
            state = prev[state]
        path.reverse()
        return path

    def plan_route(self, points: Sequence[Point]) -> List[Point]:
        """Plan a continuous route through several destinations in order.

        Args:
            points: Ordered list of waypoints, e.g. rider, restaurant, customer.

        Returns:
            Concatenated lattice path visiting every waypoint in order.
        """
        route: List[Point] = []
        for a, b in zip(points, points[1:]):
            segment = self.find_path(a, b) or [a, b]
            if not route:
                route.extend(segment)
            else:
                route.extend(segment[1:])
        if not route and points:
            route = [points[0]]
        return route

    @staticmethod
    def path_length(path: Sequence[Point]) -> int:
        """Return the length of a path in map units.

        Args:
            path: Ordered list of lattice points.

        Returns:
            Number of steps multiplied by the lattice step size.
        """
        return max(0, len(path) - 1) * LATTICE_STEP

    def random_road_point(self) -> Point:
        """Return a random street point for placing a rider.

        Returns:
            A street lattice point.
        """
        return random.choice(self._street_points)

    def straight_line_ok(self, start: Point, goal: Point) -> bool:
        """Return True when start and goal share a row or column.

        Such targets are reached by a straight run, the special case of a
        rider directly above or below its target.

        Args:
            start: Starting point.
            goal: Destination point.

        Returns:
            True if the two points are vertically or horizontally aligned.
        """
        return start[0] == goal[0] or start[1] == goal[1]

"""Rider model: movement, state transitions and service logging."""

import random
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from grid_map import LATTICE_STEP, manhattan
from path_planner import PathPlanner

Point = Tuple[int, int]

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

MAX_WAIT_TICKS = 2
_PLANNER = PathPlanner()


def _current_tick() -> int:
    """Return the current simulation tick from the global store."""
    return globalvar.get_value("current_tick") or 0


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id: Unique identifier of the rider.
        x: Current x coordinate.
        y: Current y coordinate.
        speed: Movement speed in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Active orders assigned to this rider.
        special_case: True while on a location point or aligned with target.
        log: Event records with keys "event", "order_id", "tick", "x", "y".
        route: Remaining lattice points toward the current destination.
        distance_traveled: Total distance moved in map units.
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider on a random road point.

        Args:
            rider_id: Unique identifier assigned by the system.
        """
        self.rider_id = rider_id
        self.x, self.y = _PLANNER.random_road_point()
        base_speed = globalvar.get_value("default_rider_speed") or 50
        self.speed = int(base_speed) * random.choice((1, 2))
        self.state = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.log: List[Dict[str, Any]] = []
        self.route: List[Point] = []
        self.distance_traveled = 0
        self._target: Optional[Point] = None
        self._target_kind: Optional[str] = None
        self._target_order: Any = None
        self._wait_ticks = 0
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------
    def has_route(self) -> bool:
        """Return True if the rider still has lattice points to travel.

        Returns:
            True when the remaining route is non-empty.
        """
        return bool(self.route)

    def move(self) -> None:
        """Advance the rider by one tick.

        Accepts newly assigned orders, chooses the next stop, moves along
        the route, and handles arrival events and state transitions.
        """
        self._accept_new_orders()
        if not self.orders:
            self._clear_target()
            self.refresh_state()
            return
        if self._target is None:
            self._choose_next_stop()
        if self._target is not None:
            self._advance()
            if not self.route and (self.x, self.y) == self._target:
                self._arrive()
        self.refresh_state()

    def refresh_state(self) -> None:
        """Recompute the rider state from its orders and current target."""
        if not self.orders:
            self.state = STATE_IDLE
        elif self._target_kind == "dropoff":
            self.state = STATE_DELIVERING
        elif self._target_kind == "pickup":
            self.state = STATE_PICKING_UP
        elif any(o.status == "delivering" for o in self.orders):
            self.state = STATE_DELIVERING
        else:
            self.state = STATE_PICKING_UP

    def reset_route(self) -> None:
        """Clear the current destination so it is re-planned next tick.

        Used by the system to recover from unexpected failures.
        """
        self._clear_target()
        self.refresh_state()

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log.

        Args:
            event: Event name, e.g. "accepted_order".
            order_id: Related order identifier or None.
        """
        self.log.append({
            "event": event,
            "order_id": order_id,
            "tick": _current_tick(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
        })

    def _accept_new_orders(self) -> None:
        """Accept orders that were appended to the rider but not yet started."""
        for order in self.orders:
            if order.status in ("created", "assigned"):
                order.assign_to(self.rider_id)
                order.status = "picking_up"
                self._record("accepted_order", order.id)

    def _clear_target(self) -> None:
        """Forget the current destination and remaining route."""
        self._target = None
        self._target_kind = None
        self._target_order = None
        self.route = []

    def _choose_next_stop(self) -> None:
        """Pick the nearest pending stop among all active orders.

        Pickups are available for orders not yet collected; drop-offs are
        available only after the restaurant has been visited. Choosing the
        nearest stop implements the multi-destination optimisation.
        """
        here = (self.x, self.y)
        candidates = []
        for order in self.orders:
            if order.status in ("created", "assigned", "picking_up"):
                candidates.append(("pickup", order.res_pos, order))
            elif order.status == "delivering":
                candidates.append(("dropoff", order.cus_pos, order))
        if not candidates:
            return
        kind, target, order = min(
            candidates, key=lambda c: (manhattan(here, c[1]), c[2].id)
        )
        path = _PLANNER.find_path(here, target)
        if not path:
            # Fallback when no street path exists: keep progressing directly.
            path = [here, target]
        self._target_kind = kind
        self._target = target
        self._target_order = order
        self.route = path[1:]
        self.special_case = (
            _PLANNER.grid.is_location(self.x, self.y)
            or _PLANNER.straight_line_ok(here, target)
        )

    def _blocked(self, point: Point) -> bool:
        """Return True if a stationary rider occupies the given point.

        Args:
            point: Lattice point the rider wants to enter next.

        Returns:
            True if another rider without a route sits on that point.
        """
        for other in globalvar.get_value("riders") or []:
            if other is self:
                continue
            if (other.x, other.y) == point and not other.has_route():
                return True
        return False

    def _advance(self) -> None:
        """Move along the route by up to speed // LATTICE_STEP steps.

        A rider waits up to MAX_WAIT_TICKS when the next point is occupied
        by a stationary rider; after that it proceeds to avoid deadlock.
        """
        budget = max(1, self.speed // LATTICE_STEP)
        while budget > 0 and self.route:
            nxt = self.route[0]
            if self._blocked(nxt):
                self._wait_ticks += 1
                if self._wait_ticks <= MAX_WAIT_TICKS:
                    break
            self.route.pop(0)
            self._wait_ticks = 0
            self.x, self.y = nxt
            self.distance_traveled += LATTICE_STEP
            budget -= 1
        self.special_case = (
            _PLANNER.grid.is_location(self.x, self.y)
            or (self._target is not None
                and _PLANNER.straight_line_ok((self.x, self.y), self._target))
        )

    def _arrive(self) -> None:
        """Handle arrival at the current target.

        At a restaurant the order moves to the delivering stage. At the
        customer the order is completed, removed from the rider, and the fee
        is credited to the system funds.
        """
        order = self._target_order
        if order is None:
            self._clear_target()
            return
        if self._target_kind == "pickup":
            order.status = "delivering"
            self._record("arrived_at_merchant", order.id)
        else:
            order.complete()
            if order in self.orders:
                self.orders.remove(order)
            fee = globalvar.get_value("delivery_fee") or 0
            globalvar.set_value("funds", (globalvar.get_value("funds") or 0) + fee)
            self._record("delivered_to_customer", order.id)
        self._clear_target()

"""Order data model: validation, coordinate snapping and lifecycle."""

import itertools
from typing import Any, Dict, Sequence

import globalvar
from grid_map import MAP_SIZE, Point, snap_point

_ORDER_IDS = itertools.count(1)


def _current_tick() -> int:
    """Return the current simulation tick from the global store."""
    return globalvar.get_value("current_tick") or 0


def _validate_point(point: Sequence[int]) -> Point:
    """Validate a raw (x, y) pair.

    Args:
        point: Two integers in the range 0-800.

    Returns:
        The validated (x, y) tuple.

    Raises:
        TypeError: If a value is not an integer.
        ValueError: If the pair has the wrong length or is out of range.
    """
    if len(point) != 2:
        raise ValueError("a point must contain exactly two coordinates")
    x, y = point
    for value in (x, y):
        if not isinstance(value, int) or isinstance(value, bool):
            raise TypeError(f"coordinate {value!r} is not an integer")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"coordinate {value} is outside 0-{MAP_SIZE}")
    return int(x), int(y)


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        res_pos: Snapped restaurant grid cell centre (x, y).
        cus_pos: Snapped customer grid cell centre (x, y).
        coordinates: [restaurant_x, restaurant_y, customer_x, customer_y].
        start_time: Tick at which the order was created.
        completion_time: Tick at which the order was delivered, or None.
        distance: Planned route distance in map units.
        time_taken: Ticks between creation and completion, or None.
        status: created, assigned, picking_up, delivering, completed or
            rejected.
        rider_id: Identifier of the assigned rider, or None.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order and snap its coordinates to grid cell centres.

        Args:
            res_pos: Restaurant position as (x, y), each in 0-800.

from typing import List, Optional

import globalvar
from grid_map import manhattan
from order_model import Order
from rider import Rider

LOAD_PENALTY = 150


def find_best_rider_for_order(order: Order) -> Rider:
    """Select the best rider for an order.

    The score is the Manhattan distance from the rider to the restaurant plus
    a load penalty per active order, so nearby and lightly loaded riders are
    preferred. Riders already at their order limit are skipped.

    Args:
        order: The order that needs a rider.

    Returns:
        The selected Rider, or None if no rider can take the order.
    """
    riders = globalvar.get_value("riders") or []
    max_orders = globalvar.get_value("max_orders_per_rider") or 3
    best = None
    best_key = None
    for rider in riders:
        load = len(rider.orders)
        if load >= max_orders:
            continue
        distance = manhattan((rider.x, rider.y), order.res_pos)
        key = (distance + load * LOAD_PENALTY, rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates, assigns and tracks orders."""

    @property
    def orders(self) -> List[Order]:
        """Return the list of all orders stored in the global store.

        Returns:
            List of Order objects.
        """
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, res_pos, cus_pos) -> Order:
        """Create an order and register it.

        Args:
            res_pos: Restaurant position (x, y).
            cus_pos: Customer position (x, y).

        Returns:
            The newly created Order.
        """
        order = Order(res_pos, cus_pos)
        self.orders.append(order)
        return order

    def assign(self, order: Order) -> Optional[Rider]:
        """Assign an order to the best available rider.

        Args:
            order: Order to assign.

        Returns:
            The rider that received the order, or None if none is available.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        return rider

    def discard(self, order: Order) -> None:
        """Mark an order as rejected and remove it from the registry.

        Args:
            order: Order to discard.
        """
        order.status = "rejected"
        if order in self.orders:
            self.orders.remove(order)

    def active_orders(self) -> List[Order]:
        """Return orders that are still in progress.

        Returns:
            Orders whose status is not completed or rejected.
        """
        return [o for o in self.orders if o.status not in ("completed", "rejected")]

    def completed_orders(self) -> List[Order]:
        """Return orders that have been delivered.

        Returns:
            Orders with status completed.
        """
        return [o for o in self.orders if o.status == "completed"]

class SystemManager:
    """Core coordinator for funds, riders, orders and simulated time.

    Funds, riders, the rider ID counter, the running flag and the current
    tick are stored in the global variable store so every module shares them.
    """

    def __init__(self, config: Configuration) -> None:
        """Create the manager.

        Args:
            config: System configuration.
        """
        self.config = config
        self.grid = GridMap()
        self.planner = PathPlanner(self.grid)
        self.order_manager = OrderManager()
        self.recruitment_cost = int(config.recruitment_cost)

    # ------------------------------------------------------------------
    # Global-backed properties
    # ------------------------------------------------------------------
    @property
    def funds(self) -> int:
        """System fund balance (may be negative)."""
        return int(globalvar.get_value("funds") or 0)

    @funds.setter
    def funds(self, value: int) -> None:

    @property
    def riders(self) -> List[Rider]:
        """List of all riders."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    def rider_id_counter(self) -> int:
        """Last rider ID handed out."""
        return int(globalvar.get_value("rider_id_counter") or 0)

    def rider_id_counter(self, value: int) -> None:
        globalvar.set_value("rider_id_counter", int(value))

    def is_running(self) -> bool:
        """Whether the main loop should keep running."""
        return bool(globalvar.get_value("is_running"))

    @is_running.setter
    def is_running(self, value: bool) -> None:
        globalvar.set_value("is_running", bool(value))

    @property
    def current_tick(self) -> int:
        """Current simulated time."""
        return int(globalvar.get_value("current_tick") or 0)

    @current_tick.setter
    def current_tick(self, value: int) -> None:
        globalvar.set_value("current_tick", int(value))

    # ------------------------------------------------------------------
    # Lookups
    # ------------------------------------------------------------------
    def get_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with the given identifier.

        Args:
            rider_id: Rider identifier.

        Returns:
            The Rider, or None if not found.
        """
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def find_order(self, order_id: int) -> Optional[Order]:
        """Return the order with the given identifier.

        Args:
            order_id: Order identifier.

        Returns:
            The Order, or None if not found.
        """
        for order in self.order_manager.orders:
            if order.id == order_id:
                return order
        return None

    # ------------------------------------------------------------------
    # Recruitment and dispatch
    # ------------------------------------------------------------------
    def recruit_rider(self) -> Optional[Rider]:
        """Recruit a new rider if funds and the rider limit allow it.

        Returns:
            The new Rider, or None if recruitment was not possible.
        """
        if len(self.riders) >= self.config.max_riders:
            return None
        if self.funds < self.recruitment_cost:
            return None
        self.rider_id_counter = self.rider_id_counter + 1
        rider = Rider(self.rider_id_counter)
        self.riders.append(rider)
        self.funds = self.funds - self.recruitment_cost
        logger.info("Recruited rider #%s", rider.rider_id)
        return rider

    def submit_order(self, restaurant_pos, customer_pos) -> Dict[str, Any]:
        """Create an order and assign it to a rider.

        A new rider is recruited when no rider is idle and recruitment is
        possible. The best rider is then chosen by distance and load.

        Args:
            restaurant_pos: Restaurant raw position (x, y).
            customer_pos: Customer raw position (x, y).

        Returns:
            Dictionary describing the accepted order and its route.

        Raises:
            DispatchError: If no rider can take the order.
        """
        order = self.order_manager.create_order(restaurant_pos, customer_pos)
        new_rider = False
        if not any(not r.orders for r in self.riders):
            if self.recruit_rider() is not None:
                new_rider = True
        rider = self.order_manager.assign(order)
        if rider is None:
            self.order_manager.discard(order)
            raise DispatchError(
                "no rider is available and recruitment is not possible "
                "(insufficient funds or rider limit reached)."
            )
        route = self.planner.plan_route(
            [(rider.x, rider.y), order.res_pos, order.cus_pos]
        )
        order.distance = float(self.planner.path_length(route))
        eta = math.ceil(order.distance / rider.speed) if rider.speed > 0 else 0
        logger.info("Order #%s assigned to rider #%s", order.id, rider.rider_id)
        return {
            "order": order,
            "rider": rider,
            "order_id": order.id,
            "restaurant": order.res_pos,
            "customer": order.cus_pos,
            "new_rider": new_rider,
            "speed": rider.speed,
            "route": route,
            "distance": order.distance,
            "estimated_ticks": eta,
        }

    # ------------------------------------------------------------------
    # Simulation
    # ------------------------------------------------------------------
    def tick(self) -> None:
        """Advance the simulation by one time unit.

        Every rider moves once, upkeep is charged for busy riders, and the
        state is checked for consistency. Failures in one rider do not stop
        the others.
        """
        self.current_tick = self.current_tick + 1
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # noqa: BLE001 - recover per rider
                logger.exception("Rider #%s failed; resetting its route.",
                                 rider.rider_id)
                rider.reset_route()
        busy = sum(1 for r in self.riders if r.orders)
        self.funds = self.funds - busy * self.config.rider_upkeep
        for issue in self.check_consistency():
            logger.warning(issue)

    def check_consistency(self) -> List[str]:
        """Verify and repair consistency between riders and orders.

        Returns:
            List of human-readable descriptions of issues found and fixed.
        """
        issues: List[str] = []
        seen_ids = set()
        for rider in self.riders:
            if rider.rider_id in seen_ids:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen_ids.add(rider.rider_id)
            for order in list(rider.orders):
                if order.status in ("completed", "rejected"):
                    rider.orders.remove(order)
                    issues.append(
                        f"removed finished order #{order.id} "
                        f"from rider #{rider.rider_id}"
                    )
                elif order.rider_id != rider.rider_id:
                    issues.append(
                        f"order #{order.id} rider mismatch; "
                        f"set to rider #{rider.rider_id}"
                    )
                    order.rider_id = rider.rider_id
            before = rider.state
            rider.refresh_state()
            if before != rider.state:
                issues.append(
                    f"rider #{rider.rider_id} state corrected {before}->{rider.state}"
                )
        for order in self.order_manager.active_orders():
            if order.rider_id is not None and order.rider_id not in seen_ids:
                issues.append(f"order #{order.id} references missing rider")
        return issues

    # ------------------------------------------------------------------
    # Persistence
    # ------------------------------------------------------------------
    def save_state(self, path: Optional[str] = None) -> str:
        """Save configuration, riders and order history to a JSON file.

        The file is written to a temporary name first and then renamed, so
        an interrupted save does not corrupt the previous state.

        Args:
            path: Destination file; defaults to the configured state path.

        Returns:
            The path that was written.

        Raises:
            OSError: If the file cannot be written.
        """
        target = path or self.config.state_file_path
        data = {
            "saved_at": time.strftime("%Y-%m-%d %H:%M:%S"),
            "config": self.config.to_dict(),
            "funds": self.funds,
            "current_tick": self.current_tick,
            "rider_id_counter": self.rider_id_counter,
            "riders": [
                {
                    "rider_id": r.rider_id,
                    "x": r.x,
                    "y": r.y,
                    "speed": r.speed,
                    "state": r.state,
                    "special_case": r.special_case,
                    "order_ids": [o.id for o in r.orders],
                    "distance_traveled": r.distance_traveled,
                    "log": r.log,
                }
                for r in self.riders
            ],
            "orders": [o.to_dict() for o in self.order_manager.orders],
        }
        tmp_path = target + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        os.replace(tmp_path, target)
        logger.info("State saved to %s", target)
        return target

class DisplayManager:

    def __init__(self, grid: Optional[GridMap] = None) -> None:
        """Create the display manager.

        Args:
            grid: Grid map to render. A default 8x8 map is created if None.
        """
        self.grid = grid if grid is not None else GridMap()
        self.map_symbols: Dict[str, str] = {
            "empty": " ",
            "street_h": "-",
            "street_v": "|",
            "intersection": "+",
            "restaurant": "R",
            "customer": "C",
            "both": "X",
            "route": "*",

    # ------------------------------------------------------------------
    # Map
    # ------------------------------------------------------------------
    def _place_marker(self, canvas: List[List[str]], pos: Point, kind: str) -> None:
        """Place a restaurant or customer marker on the canvas.

        Args:
            canvas: Character grid being drawn.
            pos: Location point.
            kind: "restaurant" or "customer".
        """
        x, y = pos
        ix, iy = x // LATTICE_STEP, y // LATTICE_STEP
        current = canvas[iy][ix]
        symbol = self.map_symbols[kind]
        restaurant = self.map_symbols["restaurant"]
        customer = self.map_symbols["customer"]
        both = self.map_symbols["both"]
        if current == both or (current in (restaurant, customer) and current != symbol):
            symbol = both
        canvas[iy][ix] = symbol

    def _axis_header(self, n: int) -> str:
        """Build a header line with x-axis labels in hundreds.

        Args:
            n: Number of lattice points per row.

        Returns:
            The header string.
        """
        chars = [" "] * (2 * n - 1)
        for ix in range(n):
            x = ix * LATTICE_STEP
            if x % CELL_SIZE == 0:
                chars[2 * ix] = str(x // CELL_SIZE)
        return " " * 5 + "".join(chars)

    def render_map(self, routes: Optional[Iterable[Sequence[Point]]] = None,
                   riders: Optional[Iterable[Any]] = None,
                   orders: Optional[Iterable[Any]] = None) -> str:
        """Render the ASCII map.

        Riders are drawn with the last digit of their ID, restaurants with R,
        customers with C, and routes with '*'. The top row is y=800.

        Args:
            routes: Optional lists of lattice points to draw as routes.
            riders: Optional riders to display.
            orders: Optional active orders whose locations are displayed.

        Returns:
            Multi-line string containing the map and legend.
        """
        step = LATTICE_STEP
        grid = self.grid
        n = grid.map_length // step + 1
        canvas = [[self.map_symbols["empty"] for _ in range(n)] for _ in range(n)]

        for iy in range(n):
            for ix in range(n):
                x, y = ix * step, iy * step
                if not grid.is_street(x, y):
                    continue
                if x % CELL_SIZE == 0 and y % CELL_SIZE == 0:
                    symbol = self.map_symbols["intersection"]
                elif y % CELL_SIZE == 0:
                    symbol = self.map_symbols["street_h"]
                else:
                    symbol = self.map_symbols["street_v"]
                canvas[iy][ix] = symbol

        for route in routes or []:
            for x, y in route:
                if grid.is_street(x, y):
                    canvas[y // step][x // step] = self.map_symbols["route"]

        for order in orders or []:
            self._place_marker(canvas, order.res_pos, "restaurant")
            self._place_marker(canvas, order.cus_pos, "customer")

        for rider in riders or []:
            canvas[rider.y // step][rider.x // step] = str(rider.rider_id % 10)

        lines = [self._axis_header(n)]
        for iy in reversed(range(n)):
            y = iy * step
            label = f"{y:4d} " if y % CELL_SIZE == 0 else "     "
            lines.append(label + " ".join(canvas[iy]))
        lines.append("")
        lines.append(
            "Legend: R=restaurant C=customer X=both 0-9=rider id (mod 10) "
            "*=route +=intersection -=horizontal street |=vertical street"
        )
        lines.append("Axes: x labels are hundreds (left->right), y labels are hundreds (bottom->top).")
        return "\n".join(lines)

class InputManager:
    """Parses user commands, validates arguments and keeps history."""

    def __init__(self, history_limit: int = 200,
                 input_func: Callable[[str], str] = input) -> None:
        """Create the input manager.

        Args:
            history_limit: Maximum number of commands kept in history.
            input_func: Function used to read a line (replaceable for I/O).
        """
        self.command_history: List[str] = []
        self._history_limit = history_limit
        self._input = input_func
        self._setup_readline()

    def _setup_readline(self) -> None:
        """Enable arrow-key history when readline is available."""
        if readline is None:
            return
        try:
            readline.set_history_length(self._history_limit)
        except Exception:  # noqa: BLE001 - optional feature
            pass

    def record(self, line: str) -> None:
        """Add a command to the history, trimming old entries.

        Args:
            line: Command text.
        """
        self.command_history.append(line)
        overflow = len(self.command_history) - self._history_limit
        if overflow > 0:
            del self.command_history[:overflow]

    def expand_history(self, raw: str) -> str:
        """Expand history shortcuts such as '!!' and '!3'.

        Args:
            raw: Raw input line.

        Returns:
            The expanded command text.

        Raises:
            InputError: If the referenced history entry does not exist.
        """
        if raw == "!!":
            if not self.command_history:
                raise InputError("no previous command in history.")
            return self.command_history[-1]
        if raw.startswith("!") and raw[1:].isdigit():
            index = int(raw[1:])
            if not 1 <= index <= len(self.command_history):
                raise InputError(f"history entry {index} does not exist.")
            return self.command_history[index - 1]
        return raw

    def read_command(self, prompt: str = "dispatch> ") -> str:
        """Read, expand and record one command line.

        Args:
            prompt: Prompt text shown to the user.

        Returns:
            The command line (may be empty).

        Raises:
            EOFError: If input is closed.
            InputError: If a history shortcut cannot be expanded.
        """
        raw = self._input(prompt).strip()
        line = self.expand_history(raw)
        if line:
            self.record(line)
        return line

    def parse(self, line: str) -> Tuple[str, List[str]]:
        """Split a command line into a canonical command name and arguments.

        Args:
            line: Command text.

        Returns:
            Tuple of (command name, argument tokens).

        Raises:
            InputError: If the command is empty or unknown.
        """
        tokens = line.replace(",", " ").split()
        if not tokens:
            raise InputError("empty command. Type 'help' for the command list.")
        name = tokens[0].lower()
        name = ALIASES.get(name, name)
        if name not in COMMANDS:
            raise InputError(
                f"unknown command '{tokens[0]}'. Type 'help' for the command list."
            )
        return name, tokens[1:]

    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> List[int]:
        """Validate four integer coordinates.

        Args:
            tokens: Exactly four tokens: restaurant x and y, customer x and y.

        Returns:
            The four integers.

        Raises:
            InputError: If the count, type or range is invalid.
        """
        if len(tokens) != 4:
            raise InputError(
                "expected 4 integers: restaurant_x restaurant_y customer_x customer_y"
            )
        values: List[int] = []
        for token in tokens:
            try:
                value = int(token)
            except (TypeError, ValueError):
                raise InputError(f"'{token}' is not an integer.") from None
            if not 0 <= value <= MAP_SIZE:
                raise InputError(
                    f"coordinate {value} is out of range 0-{MAP_SIZE}."
                )
            values.append(value)
        return values

    def prompt_coordinates(self) -> List[int]:
        """Interactively ask for the four coordinates.

        Returns:
            The four validated integers.

        Raises:
            InputError: If any value is invalid.
        """
        labels = ("restaurant x", "restaurant y", "customer x", "customer y")
        tokens = [self._input(f"  {label} (0-{MAP_SIZE}): ").strip() for label in labels]
        return self.parse_coordinates(tokens)

    @staticmethod
    def parse_tick_count(tokens: Sequence[str]) -> int:
        """Parse the optional tick count argument of the tick command.

        Args:
            tokens: Argument tokens.

        Returns:
            Number of ticks to run (default 1).

        Raises:
            InputError: If the value is not a positive integer within limits.
        """
        if not tokens:
            return 1
        try:
            count = int(tokens[0])
        except ValueError:
            raise InputError(f"'{tokens[0]}' is not an integer.") from None
        if not 1 <= count <= MAX_TICKS_PER_COMMAND:
            raise InputError(
                f"tick count must be between 1 and {MAX_TICKS_PER_COMMAND}."
            )
        return count

"""Command-line entry point for the food delivery dispatch simulator.

Run with: ``python src/main.py``
"""

import logging
import sys
from typing import Dict, List, Optional

from config import Configuration
from display import DisplayManager
from input_manager import HELP_TEXT, InputError, InputManager
from system import DispatchError, SystemManager, init_system

logger = logging.getLogger("dispatch")


class DispatchApp:
    """Interactive application wiring the system, display and input layers."""

    def __init__(self, config: Configuration,
                 input_manager: Optional[InputManager] = None) -> None:
        """Initialise the system and the user interface.

        Args:
            config: Loaded configuration.
            input_manager: Optional input manager (created if None).
        """
        self.config = config
        self.system: SystemManager = init_system(config)
        self.display = DisplayManager(self.system.grid)
        self.inputs = input_manager or InputManager()
        self._handlers = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
            "exit": self._cmd_quit,
        }

    def run(self) -> int:
        """Run the command loop until the user quits.

        Returns:
            Process exit code (0 on normal exit).
        """
        print("Food Delivery Order Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands.")
        while self.system.is_running:
            try:
                line = self.inputs.read_command()
                if not line:
                    continue
                name, args = self.inputs.parse(line)
                self._handlers[name](args)
            except (EOFError, KeyboardInterrupt):
                print()
                self._cmd_quit([])
                break
            except InputError as exc:
                print(f"[input error] {exc}")
            except DispatchError as exc:
                print(f"[dispatch error] {exc}")
            except Exception:  # noqa: BLE001 - keep the program alive
                logger.exception("Unexpected failure while handling a command")
                print("[error] unexpected failure; system state was preserved.")
        return 0

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def _cmd_start(self, args: List[str]) -> None:
        """Create an order from restaurant and customer coordinates.

        Args:
            args: Optional four coordinate tokens.
        """
        if args:
            coords = self.inputs.parse_coordinates(args)
        else:
            print("Enter restaurant and customer positions (integers 0-800):")
            coords = self.inputs.prompt_coordinates()
        restaurant = (coords[0], coords[1])
        customer = (coords[2], coords[3])
        result = self.system.submit_order(restaurant, customer)
        order = result["order"]
        rider = result["rider"]
        print(f"Order #{order.id} accepted at tick {order.start_time}.")
        print(f"  Restaurant input {restaurant} -> grid point {result['restaurant']}")
        print(f"  Customer input   {customer} -> grid point {result['customer']}")
        print(f"  New rider recruited: {'yes' if result['new_rider'] else 'no'}")
        print(f"  Assigned rider: #{rider.rider_id} (speed {result['speed']} units/tick)")
        print(f"  Planned distance: {result['distance']:.0f} units, "
              f"estimated time: {result['estimated_ticks']} ticks")
        print("  Route map (advance time with 'tick'):")
        print(self.display.render_map(
            routes=[result["route"]], riders=[rider], orders=[order]
        ))

    def _cmd_status(self, args: List[str]) -> None:
        """Print the system status snapshot.

        Args:
            args: Unused.
        """
        print(self.display.render_status(
            self.system.funds, self.system.riders, self.system.current_tick
        ))

    def _cmd_orders(self, args: List[str]) -> None:
        """Print the order table.

        Args:
            args: Unused.
        """
        print(self.display.render_orders(self.system.order_manager.orders))

    def _cmd_riders(self, args: List[str]) -> None:
        """Print every rider's status-change log.

        Args:
            args: Unused.
        """
        print(self.display.render_rider_logs(self.system.riders))

    def _cmd_map(self, args: List[str]) -> None:
        """Print the ASCII map with riders, orders and active routes.

        Args:
            args: Unused.
        """
        routes = [[(r.x, r.y)] + list(r.route) for r in self.system.riders if r.route]
        print(self.display.render_map(
            routes=routes,
            riders=self.system.riders,
            orders=self.system.order_manager.active_orders(),
        ))

    def _cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation by one or more ticks.

        Args:
            args: Optional tick count.
        """
        count = self.inputs.parse_tick_count(args)
        for _ in range(count):
            self.system.tick()
        print(f"Advanced {count} tick(s); current tick {self.system.current_tick}. "
              f"Funds: {self.system.funds}. Active orders: "
              f"{len(self.system.order_manager.active_orders())}.")

    def _cmd_history(self, args: List[str]) -> None:
        """Print the command history.

        Args:
            args: Unused.
        """
        if not self.inputs.command_history:
            print("History is empty.")
            return
        for index, entry in enumerate(self.inputs.command_history, start=1):
            print(f"  {index:>3}  {entry}")

    def _cmd_help(self, args: List[str]) -> None:
        """Print the help text.

        Args:
            args: Unused.
        """
        print(HELP_TEXT)

    def _cmd_quit(self, args: List[str]) -> None:
        """Save state if enabled and stop the main loop.

        Args:
            args: Unused.
        """
        if self.config.save_on_exit:
            try:
                path = self.system.save_state()
                print(f"State saved to {path}.")
            except OSError as exc:
                print(f"[warning] could not save state: {exc}")
        self.system.is_running = False
        print("Goodbye.")


def main() -> int:
    """Program entry point.

    Returns:
        Process exit code.
    """
    config = Configuration.load()
    level = getattr(logging, str(config.log_level).upper(), logging.WARNING)
    logging.basicConfig(
        level=level,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    app = DispatchApp(config)
    return app.run()


if __name__ == "__main__":
    sys.exit(main())

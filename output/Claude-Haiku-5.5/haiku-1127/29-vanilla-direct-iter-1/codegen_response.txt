"""Grid-based path planning for riders on the 8x8 street grid.

Routes are built from axis-aligned segments. When two corner choices exist
the planner prefers the corner whose segments run along streets, i.e. lines
where x or y is a multiple of CELL_SIZE. A rider that is directly above or
below its target simply moves in a straight line (special case).
"""

from typing import List, Tuple

from config import CELL_SIZE, GRID_SIZE

Point = Tuple[int, int]


class PathPlanner:
    """Computes routes and single-tick movement along the street grid."""

    @staticmethod
    def to_cell_center(x: int, y: int) -> Point:
        """Convert a raw coordinate to the centre point of its grid cell.

        Uses ``center = (grid // 100) * 100 + 50``. Out-of-range values are
        clamped to the map boundary.

        Args:
            x: Raw x coordinate.
            y: Raw y coordinate.

        Returns:
            Tuple ``(center_x, center_y)``.
        """
        max_index = GRID_SIZE - 1
        gx = min(max(int(x), 0) // CELL_SIZE, max_index)
        gy = min(max(int(y), 0) // CELL_SIZE, max_index)
        half = CELL_SIZE // 2
        return (gx * CELL_SIZE + half, gy * CELL_SIZE + half)

    @staticmethod
    def is_street_point(point: Point) -> bool:
        """Return True if the point lies on a street (grid line).

        Args:
            point: Coordinate tuple.
        """
        x, y = point
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def manhattan(a: Point, b: Point) -> int:
        """Return the Manhattan distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    def plan(self, start: Point, goal: Point) -> List[Point]:
        """Build an axis-aligned waypoint list from ``start`` to ``goal``.

        Args:
            start: Starting coordinate.
            goal: Destination coordinate.

        Returns:
            Waypoints including both endpoints. A single-element list is
            returned when start equals goal.
        """
        start = (int(start[0]), int(start[1]))
        goal = (int(goal[0]), int(goal[1]))
        if start == goal:
            return [start]

        sx, sy = start
        gx, gy = goal
        # Directly above/below or left/right of the target: straight line.
        if sx == gx or sy == gy:
            return [start, goal]

        horizontal_corner = (gx, sy)  # move along x first
        vertical_corner = (sx, gy)  # move along y first

        # Count how many of the two segments travel along a street.
        h_score = int(sy % CELL_SIZE == 0) + int(gx % CELL_SIZE == 0)
        v_score = int(sx % CELL_SIZE == 0) + int(gy % CELL_SIZE == 0)
        corner = vertical_corner if v_score > h_score else horizontal_corner
        return [start, corner, goal]

    def route_length(self, start: Point, goal: Point) -> int:
        """Return the total length of the planned route in coordinate units."""
        path = self.plan(start, goal)
        return sum(self.manhattan(a, b) for a, b in zip(path, path[1:]))

    def next_position(self, start: Point, goal: Point, budget: int) -> Point:
        """Advance from ``start`` towards ``goal`` by at most ``budget`` units.

        Every step moves monotonically towards the goal, so the rider never
        oscillates and always converges.

        Args:
            start: Current coordinate.
            goal: Target coordinate.
            budget: Maximum number of coordinate units to travel.

        Returns:
            The new coordinate after movement.
        """
        pos = (int(start[0]), int(start[1]))
        remaining = int(budget)
        for waypoint in self.plan(pos, goal)[1:]:
            while remaining > 0 and pos != waypoint:
                dx = waypoint[0] - pos[0]
                dy = waypoint[1] - pos[1]
                if dx != 0:
                    step = min(abs(dx), remaining)
                    pos = (pos[0] + step if dx > 0 else pos[0] - step, pos[1])
                else:
                    step = min(abs(dy), remaining)
                    pos = (pos[0], pos[1] + step if dy > 0 else pos[1] - step)
                remaining -= step
            if remaining <= 0:
                break
        return pos

"""Order data model.

An order is described by ``[restaurant_x, restaurant_y, customer_x,
customer_y]`` with coordinates in the range 0-800. Coordinates are converted
to the centre point of their grid cell on creation.
"""

import itertools
from typing import Any, Dict, List, Optional, Sequence, Tuple

from config import MAP_SIZE
from globalvar import get_value
from path_planner import PathPlanner

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKED_UP = "picked_up"
STATUS_COMPLETED = "completed"

_order_ids = itertools.count(1)


def validate_point(point: Sequence[Any]) -> Tuple[int, int]:
    """Validate a raw ``(x, y)`` coordinate pair.

    Args:
        point: Sequence with exactly two integer values in the range 0-800.

    Returns:
        The validated coordinate as a tuple of ints.

    Raises:
        ValueError: If the shape, type or range of the coordinate is invalid.
    """
    if not isinstance(point, (tuple, list)) or len(point) != 2:
        raise ValueError("a location must consist of exactly two coordinates")
    x, y = point
    for value in (x, y):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"coordinate {value!r} is not an integer")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(
                f"coordinate {value} is out of range 0-{MAP_SIZE}"
            )
    return (x, y)


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        raw_res_pos: Restaurant coordinate as entered by the user.
        raw_cus_pos: Customer coordinate as entered by the user.
        res_pos: Restaurant grid-cell centre used for routing.
        cus_pos: Customer grid-cell centre used for routing.
        status: One of pending, assigned, picked_up, completed.
        rider_id: Identifier of the assigned rider, or None.
        start_time: Tick at which the order was created.
        pickup_time: Tick at which the rider picked the order up.
        complete_time: Tick at which the order was delivered.
        duration: Ticks between creation and delivery.
        distance: Route length from restaurant to customer.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order and convert its coordinates to cell centres.

        Args:
            res_pos: Restaurant coordinate ``(x, y)`` in range 0-800.
            cus_pos: Customer coordinate ``(x, y)`` in range 0-800.

        Raises:
            ValueError: If any coordinate is invalid.
        """
        self.id: int = next(_order_ids)
        self.raw_res_pos: Tuple[int, int] = validate_point(res_pos)
        self.raw_cus_pos: Tuple[int, int] = validate_point(cus_pos)
        planner = PathPlanner()
        self.res_pos: Tuple[int, int] = planner.to_cell_center(*self.raw_res_pos)
        self.cus_pos: Tuple[int, int] = planner.to_cell_center(*self.raw_cus_pos)
        self.status: str = STATUS_PENDING
        self.rider_id: Optional[int] = None
        self.start_time: int = int(get_value("time", 0))
        self.pickup_time: Optional[int] = None
        self.complete_time: Optional[int] = None
        self.duration: Optional[int] = None
        self.distance: int = planner.route_length(self.res_pos, self.cus_pos)

    def assign_to(self, rider_id: int) -> None:
        """Record that the order has been given to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_picked_up(self, time: int) -> None:
        """Record arrival of the rider at the restaurant.

        Args:
            time: Current simulated tick.
        """
        self.pickup_time = time
        self.status = STATUS_PICKED_UP

    def mark_completed(self, time: int) -> None:
        """Record delivery of the order and compute efficiency data.

        Args:
            time: Current simulated tick.
        """
        self.complete_time = time
        self.duration = time - self.start_time
        self.status = STATUS_COMPLETED

    def to_list(self) -> List[int]:
        """Return the order in ``[restaurant_x, restaurant_y, customer_x, customer_y]`` form."""
        return [self.raw_res_pos[0], self.raw_res_pos[1],
                self.raw_cus_pos[0], self.raw_cus_pos[1]]

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "status": self.status,
            "rider_id": self.rider_id,
            "raw_restaurant": list(self.raw_res_pos),
            "raw_customer": list(self.raw_cus_pos),
            "restaurant_center": list(self.res_pos),
            "customer_center": list(self.cus_pos),
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "duration": self.duration,
            "distance": self.distance,
        }

"""Rider model: movement, state transitions and event logging.

Rider states:
    0 - idle
    1 - picking up (en route to the restaurant)
    2 - delivering (en route to the customer)
"""

import logging
from typing import TYPE_CHECKING, Any, Dict, List, Optional, Tuple

from config import (
    MAX_WAIT_TICKS,
    RIDER_SPAWN_X,
    RIDER_SPAWN_Y,
    RIDER_SPEED,
    STATE_DELIVERING,
    STATE_IDLE,
    STATE_PICKING_UP,
)
from globalvar import get_value
from path_planner import PathPlanner

if TYPE_CHECKING:  # pragma: no cover
    from order_model import Order

logger = logging.getLogger(__name__)

Point = Tuple[int, int]


class Rider:
    """A delivery rider that serves one or more orders.

    Attributes:
        rider_id: Unique identifier.
        x: Current x coordinate.
        y: Current y coordinate.
        speed: Units moved per tick.
        state: Current state (0 idle, 1 picking up, 2 delivering).
        orders: Assigned orders; index 0 is the order currently served.
        special_case: True while the rider is aligned with its target and
            moves in a straight line.
        log: Chronological list of event dictionaries with keys ``event``
            and ``order_id`` plus ``time``, ``x``, ``y``, ``state``, ``orders``.
        total_distance: Total distance travelled in coordinate units.
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the spawn location in the idle state.

        Args:
            rider_id: Unique identifier assigned by the system.
        """
        self.rider_id: int = rider_id
        self.x: int = RIDER_SPAWN_X
        self.y: int = RIDER_SPAWN_Y
        self.speed: int = RIDER_SPEED
        self.state: int = STATE_IDLE
        self.orders: List["Order"] = []
        self.special_case: bool = False
        self.log: List[Dict[str, Any]] = []
        self.total_distance: int = 0
        self._wait_ticks: int = 0
        self._planner = PathPlanner()
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # Properties and helpers
    # ------------------------------------------------------------------
    @property
    def position(self) -> Point:
        """Current ``(x, y)`` position."""
        return (self.x, self.y)

    @property
    def current_order(self) -> Optional["Order"]:
        """The order currently being served, or None."""
        return self.orders[0] if self.orders else None

    def _now(self) -> int:
        """Return the current simulated tick."""
        return int(get_value("time", 0))

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log."""
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "orders": [order.id for order in self.orders],
        })

    def _become_idle(self) -> None:
        """Release the rider so it can accept new orders."""
        self.state = STATE_IDLE
        self.special_case = False
        self._wait_ticks = 0

    def _current_target(self) -> Point:
        """Return the coordinate the rider is currently heading to."""
        order = self.current_order
        if order is None:
            return self.position
        if self.state == STATE_PICKING_UP:
            return order.res_pos
        return order.cus_pos

    def _is_occupied(self, point: Point) -> bool:
        """Return True if another active rider occupies ``point``."""
        for other in get_value("riders") or []:
            if other is self:
                continue
            if other.state != STATE_IDLE and other.position == point:
                return True
        return False

    # ------------------------------------------------------------------
    # Order handling
    # ------------------------------------------------------------------
    def accept_order(self, order: "Order") -> None:
        """Assign an order to this rider.

        The order is appended to ``orders``; an idle rider moves to the
        picking-up state.

        Args:
            order: The order to assign.
        """
        self.orders.append(order)
        order.assign_to(self.rider_id)
        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP
        self._record("accepted_order", order.id)

    def _arrive_at_merchant(self, order: "Order") -> None:
        """Handle arrival at the restaurant: switch to delivering."""
        self.state = STATE_DELIVERING
        self.special_case = False
        order.mark_picked_up(self._now())
        self._record("arrived_at_merchant", order.id)

    def _deliver(self, order: "Order") -> None:
        """Handle delivery to the customer and release the order."""
        order.mark_completed(self._now())
        self.orders.pop(0)
        self._record("delivered_to_customer", order.id)
        if self.orders:
            self.state = STATE_PICKING_UP
            self.special_case = False
        else:
            self._become_idle()

    def _check_arrival(self) -> bool:
        """Process arrival at the current target.

        Returns:
            True if an arrival event was handled.
        """
        order = self.current_order
        if order is None:
            return False
        if self.state == STATE_PICKING_UP and self.position == order.res_pos:
            self._arrive_at_merchant(order)
            return True
        if self.state == STATE_DELIVERING and self.position == order.cus_pos:
            self._deliver(order)
            return True
        return False

    # ------------------------------------------------------------------
    # Movement
    # ------------------------------------------------------------------
    def move(self) -> None:
        """Advance the rider by one tick.

        Moves up to ``speed`` units along the street grid towards the current
        target, handles arrival at the restaurant and at the customer, and
        waits a limited number of ticks when the next cell is occupied to
        avoid conflicts and deadlocks.
        """
        if not self.orders:
            if self.state != STATE_IDLE:
                self._become_idle()
            return
        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP

        if self._check_arrival():
            return

        target = self._current_target()
        self.special_case = self.x == target[0] or self.y == target[1]
        new_pos = self._planner.next_position(self.position, target, self.speed)

        if new_pos != self.position:
            if self._is_occupied(new_pos) and self._wait_ticks < MAX_WAIT_TICKS:
                self._wait_ticks += 1
                logger.debug("Rider %s waiting at %s", self.rider_id, self.position)
                return
            self._wait_ticks = 0
            self.total_distance += PathPlanner.manhattan(self.position, new_pos)
            self.x, self.y = new_pos

        self._check_arrival()

    # ------------------------------------------------------------------
    # Serialisation
    # ------------------------------------------------------------------
    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [order.id for order in self.orders],
            "total_distance": self.total_distance,
            "log": list(self.log),
        }

"""Order assignment: distance-first rider selection and order management."""

import logging
import math
from typing import TYPE_CHECKING, List, Optional, Sequence

from globalvar import get_value, set_value
from order_model import Order

if TYPE_CHECKING:  # pragma: no cover
    from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional["Rider"]:
    """Select the rider closest to the order's restaurant.

    Distance is Euclidean: ``sqrt((rider_x - restaurant_x)^2 +
    (rider_y - restaurant_y)^2)``. Ties are broken by the smaller number of
    assigned orders (load balancing), then by rider id.

    Args:
        order: The order that needs a rider.

    Returns:
        The best rider, or None if no rider exists.
    """
    best: Optional["Rider"] = None
    best_key = None
    for rider in get_value("riders") or []:
        distance = math.sqrt(
            (rider.x - order.res_pos[0]) ** 2 + (rider.y - order.res_pos[1]) ** 2
        )
        key = (distance, len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best, best_key = rider, key
    return best


class OrderManager:
    """Creates, assigns and tracks orders.

    Pending orders are kept in the ``pending_orders`` global; every created
    order is also appended to the ``orders`` history global.
    """

    def __init__(self) -> None:
        """Ensure the order containers exist in global storage."""
        if get_value("pending_orders") is None:
            set_value("pending_orders", [])
        if get_value("orders") is None:
            set_value("orders", [])

    @property
    def pending(self) -> List[Order]:
        """Orders not yet assigned to a rider."""
        return get_value("pending_orders")

    @property
    def history(self) -> List[Order]:
        """All orders created during the session."""
        return get_value("orders")

    def create_order(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> Order:
        """Create an order from raw coordinates and queue it for dispatch.

        Args:
            res_pos: Restaurant ``(x, y)``.
            cus_pos: Customer ``(x, y)``.

        Returns:
            The newly created order.

        Raises:
            ValueError: If a coordinate is invalid.
        """
        order = Order(res_pos, cus_pos)
        self.pending.append(order)
        self.history.append(order)
        logger.info("Order %s created", order.id)
        return order

    def assign_pending_orders(self) -> List[Order]:
        """Assign every pending order to its best rider when possible.

        Returns:
            The list of orders that were assigned during this call.
        """
        assigned: List[Order] = []
        for order in list(self.pending):
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            rider.accept_order(order)
            self.pending.remove(order)
            assigned.append(order)
            logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return assigned

    def completed_orders(self) -> List[Order]:
        """Return all orders whose status is completed."""
        return [order for order in self.history if order.status == "completed"]

    def active_orders(self) -> List[Order]:
        """Return all orders that are not yet completed."""
        return [order for order in self.history if order.status != "completed"]

"""System management: funds, rider recruitment, dispatch and state persistence."""

import json
import logging
import os
from typing import Any, Dict, List, Tuple

from config import (
    CELL_SIZE,
    GRID_SIZE,
    INITIAL_FUNDS,
    MAP_SIZE,
    RIDER_COST,
    RIDER_MIN_FUNDS,
    RIDER_SPAWN_X,
    RIDER_SPAWN_Y,
    RIDER_SPEED,
    STATE_IDLE,
    STATE_PICKING_UP,
)
from globalvar import _init, get_value, set_value
from order import OrderManager
from order_model import Order
from rider import Rider

logger = logging.getLogger(__name__)


class SystemManager:
    """Core coordinator for funds, riders, orders and the simulation clock."""

    def __init__(self) -> None:
        """Create the manager bound to the global state."""
        self.order_manager = OrderManager()

    # ------------------------------------------------------------------
    # Accessors
    # ------------------------------------------------------------------
    @property
    def funds(self) -> int:
        """Current system funds."""
        return int(get_value("funds", 0))

    @property
    def riders(self) -> List[Rider]:
        """The live rider list."""
        return get_value("riders")

    @property
    def running(self) -> bool:
        """Main loop control flag."""
        return bool(get_value("running", False))

    @property
    def time(self) -> int:
        """Current simulated tick."""
        return int(get_value("time", 0))

    def stop(self) -> None:
        """Request the main loop to terminate."""
        set_value("running", False)

    # ------------------------------------------------------------------
    # Rider management
    # ------------------------------------------------------------------
    def idle_riders(self) -> List[Rider]:
        """Return riders in the idle state."""
        return [rider for rider in self.riders if rider.state == STATE_IDLE]

    def create_rider(self) -> Rider:
        """Create and register a new rider with the next unique id.

        Returns:
            The newly created rider.
        """
        rider_id = int(get_value("rider_id_counter", 0)) + 1
        set_value("rider_id_counter", rider_id)
        rider = Rider(rider_id)
        self.riders.append(rider)
        logger.info("Rider %s recruited", rider_id)
        return rider

    def recruit_if_needed(self) -> List[Rider]:
        """Recruit riders while funds allow and pending orders exceed idle riders.

        Each recruit costs ``RIDER_COST`` from system funds.

        Returns:
            The riders recruited during this call.
        """
        new_riders: List[Rider] = []
        while (self.funds >= RIDER_MIN_FUNDS
               and len(self.order_manager.pending) > len(self.idle_riders())):
            set_value("funds", self.funds - RIDER_COST)
            new_riders.append(self.create_rider())
        return new_riders

    def dispatch(self) -> Tuple[List[Rider], List[Order]]:
        """Recruit if necessary and assign pending orders.

        Returns:
            Tuple ``(new_riders, assigned_orders)``.
        """
        new_riders = self.recruit_if_needed()
        assigned = self.order_manager.assign_pending_orders()
        return new_riders, assigned

    # ------------------------------------------------------------------
    # Simulation clock
    # ------------------------------------------------------------------
    def tick(self) -> Dict[str, Any]:
        """Advance the simulation by one unit of time.

        Every rider moves once, pending orders are dispatched and the state
        is checked for consistency. Exceptions are caught per rider so that a
        single failure does not stop the simulation.

        Returns:
            Summary dictionary with keys ``time``, ``events``, ``new_riders``,
            ``assigned``, ``errors`` and ``fixes``.
        """
        set_value("time", self.time + 1)
        events: List[Tuple[int, Dict[str, Any]]] = []
        errors: List[str] = []

        for rider in list(self.riders):
            before = len(rider.log)
            try:
                rider.move()
            except Exception as exc:  # error recovery
                logger.exception("Rider %s failed during tick", rider.rider_id)
                errors.append(f"rider #{rider.rider_id}: {exc}")
            for entry in rider.log[before:]:
                events.append((rider.rider_id, entry))

        new_riders: List[Rider] = []
        assigned: List[Order] = []
        try:
            new_riders, assigned = self.dispatch()
        except Exception as exc:  # error recovery
            logger.exception("Dispatch failed during tick")
            errors.append(f"dispatch: {exc}")

        fixes = self.check_consistency()
        return {
            "time": self.time,
            "events": events,
            "new_riders": new_riders,
            "assigned": assigned,
            "errors": errors,
            "fixes": fixes,
        }

    def check_consistency(self) -> List[str]:
        """Repair inconsistent rider states.

        Returns:
            Human readable descriptions of repaired problems.
        """
        fixes: List[str] = []
        for rider in self.riders:
            if rider.orders and rider.state == STATE_IDLE:
                rider.state = STATE_PICKING_UP
                fixes.append(f"rider #{rider.rider_id} had orders but was idle")
            elif not rider.orders and rider.state != STATE_IDLE:
                rider.state = STATE_IDLE
                rider.special_case = False
                fixes.append(f"rider #{rider.rider_id} had no orders but was busy")
        return fixes

    # ------------------------------------------------------------------
    # Persistence
    # ------------------------------------------------------------------
    def snapshot(self) -> Dict[str, Any]:
        """Return the full system state as a JSON-serialisable dictionary."""
        return {
            "system_parameters": {
                "initial_funds": INITIAL_FUNDS,
                "rider_cost": RIDER_COST,
                "rider_min_funds": RIDER_MIN_FUNDS,
                "rider_spawn": [RIDER_SPAWN_X, RIDER_SPAWN_Y],
                "rider_speed": RIDER_SPEED,
                "map_size": MAP_SIZE,
                "cell_size": CELL_SIZE,
                "grid_size": GRID_SIZE,
            },
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": int(get_value("rider_id_counter", 0)),
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in self.order_manager.history],
        }

    def save_state(self, path: str) -> str:
        """Write the system state to a JSON file.

        Args:
            path: Destination file path.

        Returns:
            The absolute path of the written file.

        Raises:
            OSError: If the file cannot be written.
        """
        directory = os.path.dirname(os.path.abspath(path))
        os.makedirs(directory, exist_ok=True)
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(self.snapshot(), handle, indent=2)
        return os.path.abspath(path)

class DisplayManager:
    """Produces text reports and the command-line map visualisation.

    The map uses one character per ``MAP_RESOLUTION`` units, so the 800x800
    area becomes a 33x33 character canvas in which street lines fall every
    four characters (every 100 units).
    """

    LEGEND = "Legend: R=Rider  S=Restaurant  C=Customer  #=Street  *=Route  .=Empty"

    def __init__(self) -> None:
        """Create the display manager."""
        self._planner = PathPlanner()
        self._n = MAP_SIZE // MAP_RESOLUTION + 1

    # ------------------------------------------------------------------
    # Canvas helpers
    # ------------------------------------------------------------------
    def _to_rc(self, point: Point) -> Tuple[int, int]:
        """Convert a coordinate to ``(row, column)`` on the canvas.

        Row 0 is the top line, i.e. the largest y value.
        """
        x, y = point
        col = min(max(x, 0) // MAP_RESOLUTION, self._n - 1)
        row = self._n - 1 - min(max(y, 0) // MAP_RESOLUTION, self._n - 1)
        return row, col

    def _new_canvas(self) -> Canvas:
        """Create an empty canvas with street lines drawn as '#'."""
        n = self._n
        canvas = [["." for _ in range(n)] for _ in range(n)]
        for col in range(n):
            if (col * MAP_RESOLUTION) % CELL_SIZE == 0:
                for row in range(n):
                    canvas[row][col] = "#"
        for row in range(n):
            if ((n - 1 - row) * MAP_RESOLUTION) % CELL_SIZE == 0:
                for col in range(n):
                    canvas[row][col] = "#"
        return canvas

    def _place(self, canvas: Canvas, point: Point, symbol: str) -> None:
        """Draw a single symbol at a coordinate."""
        row, col = self._to_rc(point)
        canvas[row][col] = symbol

    def _draw_rect(self, canvas: Canvas, r1: int, c1: int, r2: int, c2: int) -> None:
        """Draw '*' over an axis-aligned span of canvas cells."""
        for row in range(min(r1, r2), max(r1, r2) + 1):
            for col in range(min(c1, c2), max(c1, c2) + 1):
                canvas[row][col] = "*"

    def _draw_polyline(self, canvas: Canvas, points: Sequence[Point]) -> None:
        """Draw a connected line segment graph through the given points."""
        for a, b in zip(points, points[1:]):
            r1, c1 = self._to_rc(a)
            r2, c2 = self._to_rc(b)
            if r1 != r2 and c1 != c2:
                # Not axis aligned: go through the corner.
                self._draw_rect(canvas, r1, c1, r1, c2)
                self._draw_rect(canvas, r1, c2, r2, c2)
            else:
                self._draw_rect(canvas, r1, c1, r2, c2)

    @staticmethod
    def cell_label(point: Point) -> str:
        """Return a coordinate with its grid cell index, e.g. ``(150,50) cell[1,0]``."""
        x, y = point
        return f"({x},{y}) cell[{x // CELL_SIZE},{y // CELL_SIZE}]"

    def render_map(
        self,
        riders: Sequence[Rider],
        orders: Sequence[Order],
        routes: Optional[Sequence[Sequence[Point]]] = None,
    ) -> str:
        """Render the ASCII map of riders, restaurants, customers and routes.

        Args:
            riders: Riders to draw as 'R'.
            orders: Orders whose restaurant ('S') and customer ('C') are drawn.
                Completed orders are skipped.
            routes: Optional list of polylines drawn as '*'.

        Returns:
            Multi-line string containing the map and a legend.
        """
        canvas = self._new_canvas()
        for route in routes or []:
            self._draw_polyline(canvas, route)
        for order in orders:
            if order.status == STATUS_COMPLETED:
                continue
            self._place(canvas, order.res_pos, "S")
            self._place(canvas, order.cus_pos, "C")
        for rider in riders:
            self._place(canvas, rider.position, "R")

        lines = []
        for row, cells in enumerate(canvas):
            y = (self._n - 1 - row) * MAP_RESOLUTION
            label = f"{y:>3} " if y % CELL_SIZE == 0 else "    "
            lines.append(label + "".join(cells))
        lines.append(f"    x: 0 to {MAP_SIZE} (1 char = {MAP_RESOLUTION} units, "
                     f"street every {CELL_SIZE} units)")
        lines.append("    " + self.LEGEND)
        return "\n".join(lines)

    # Reports
    # ------------------------------------------------------------------
    def status_report(self, system: Any) -> str:
        """Build the status snapshot of funds, riders and orders.

        Args:
            system: SystemManager instance.
        """
        riders = system.riders
        active = system.order_manager.active_orders()
        lines = [

            f"System funds: {system.funds}",

    def orders_report(self, orders: Sequence[Order]) -> str:

        Args:

        if not orders:

                  f"{'Distance':>9} {'Time':>6} {'Rider':>6} {'Status':<10}")

            restaurant = f"({order.raw_res_pos[0]},{order.raw_res_pos[1]})"

            rider = f"#{order.rider_id}" if order.rider_id is not None else "-"

                f"{customer:>14} {order.distance:>9} {taken:>6} "
                f"{rider:>6} {order.status:<10}"
            )
        return "\n".join(lines)

    def rider_log_report(self, riders: Sequence[Rider]) -> str:

        for rider in riders:
            lines.append(f"--- Rider #{rider.rider_id} ---")

                lines.append(

                    f"status={STATE_NAMES[entry['state']]}"

    def start_report(self, order: Order, rider: Rider, new_rider: bool) -> str:

        total = self._planner.route_length(rider.position, order.res_pos) + \
            self._planner.route_length(order.res_pos, order.cus_pos)
        ticks = math.ceil(total / rider.speed) if rider.speed > 0 else 0
        lines = [
            f"Order #{order.id} accepted by rider #{rider.rider_id}.",
            f"New rider added: {'yes' if new_rider else 'no'}",
            f"Restaurant input ({order.raw_res_pos[0]},{order.raw_res_pos[1]}) "
            f"-> grid {self.cell_label(order.res_pos)}",
            f"Customer   input ({order.raw_cus_pos[0]},{order.raw_cus_pos[1]}) "
            f"-> grid {self.cell_label(order.cus_pos)}",
            f"Rider speed: {rider.speed} units/tick",
            f"Route length: {total} units, estimated time: {ticks} ticks",
            "Route map:",
            self.render_map([rider], [order], routes=[pickup, delivery]),
        ]
        return "\n".join(lines)

    def queued_report(self, order: Order, funds: int) -> str:
        """Build the message shown when an order cannot be assigned yet.

            order: The queued order.

        return (f"Order #{order.id} is queued: no rider is available and "
                f"funds ({funds}) are insufficient to recruit one.")

import logging
import re
from typing import Any, Callable, Dict, List, Optional

try:  # Enables line editing and arrow-key history where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    pass

from config import DEFAULT_SAVE_PATH, MAP_SIZE, MAX_INPUT_ATTEMPTS
from display import DisplayManager
from order_model import Order
from rider import Rider
from system import SystemManager

logger = logging.getLogger(__name__)

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant then customer, 0-800)
  status  (s)          System funds, riders and map
  orders  (o)          Details of all orders
  riders  (r)          Status-change log of every rider
  map     (m)          ASCII map of riders and active orders
  tick [n] (t)         Advance the simulation n ticks (default 1)
  save [path]          Save system state to JSON
  history (h)          Show command history
  help    (?)          Show this help
  quit    (q)/exit     Exit the program
Every command except help/history/tick/quit advances the simulation by one tick."""

COMMAND_ALIASES = {
    "s": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "h": "history",
    "?": "help",
    "q": "quit",
    "x": "exit",
}

NO_AUTO_TICK = {"help", "history", "tick", "quit", "exit", "unknown"}


def parse_coordinates(text: str) -> List[int]:
    """Parse four integer coordinates from user text.

    Accepts whitespace, commas or brackets as separators.

    Args:
        text: Raw user input, e.g. ``"100 200 300 400"``.

    Returns:
        List ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

    Raises:
        ValueError: If the count, type or range of the values is invalid.
    """
    tokens = [tok for tok in re.split(r"[\s,\[\]]+", text.strip()) if tok]
    if len(tokens) != 4:
        raise ValueError(
            f"expected 4 integers (restaurant_x restaurant_y customer_x customer_y), "
            f"got {len(tokens)}"
        )
    values: List[int] = []
    for token in tokens:
        try:
            values.append(int(token))
        except ValueError:
            raise ValueError(f"'{token}' is not an integer") from None
    for value in values:
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"coordinate {value} is out of range 0-{MAP_SIZE}")
    return values


class InputManager:
    """Reads commands, dispatches them and drives the simulation loop."""

    def __init__(
        self,
        system: SystemManager,
        display: DisplayManager,
        input_func: Callable[[str], str] = input,
        output_func: Callable[[str], Any] = print,
    ) -> None:
        """Create the input manager.

        Args:
            system: The system manager controlling the simulation.
            display: Display manager used for all output reports.
            input_func: Function used to read a line (injectable for tests).
            output_func: Function used to write a line of output.
        """
        self.system = system
        self.display = display
        self._input = input_func
        self._print = output_func
        self.history: List[str] = []
        self._auto_tick = True
        self._handlers: Dict[str, Callable[[List[str]], bool]] = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "save": self._cmd_save,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
            "exit": self._cmd_quit,
        }

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> None:
        """Run the interactive loop until quit, EOF or interrupt.

        Each iteration processes one command and then advances the
        simulation by one tick (unless the command does not advance time).
        """
        self._print(HELP_TEXT)
        while self.system.running:
            try:
                line = self._input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self._print("")
                break
            line = line.strip()
            if not line:
                continue
            self.history.append(line)
            self._auto_tick = True
            try:
                keep_running = self.execute(line)
            except (EOFError, KeyboardInterrupt):
                self._print("")
                break
            except Exception as exc:  # graceful error recovery
                logger.exception("Command failed: %s", line)
                self._print(f"Error: {exc}. The command was ignored.")
                keep_running = True
            if not keep_running:
                break
            if self._auto_tick:
                self._run_tick()
        self._print("Simulation ended. Goodbye!")

    def execute(self, line: str) -> bool:
        """Parse and run a single command line.

        Args:
            line: Command text entered by the user.

        Returns:
            False if the program should exit, True otherwise.
        """
        parts = line.split()
        name = parts[0].lower()
        command = COMMAND_ALIASES.get(name, name)
        handler = self._handlers.get(command)
        if handler is None:
            self._auto_tick = False
            self._print(f"Unknown command '{parts[0]}'. Type 'help' for a list of commands.")
            return True
        return handler(parts[1:])

    # ------------------------------------------------------------------
    # Tick handling
    # ------------------------------------------------------------------
    def _run_tick(self) -> None:
        """Advance the simulation by one tick and print notable events."""
        result = self.system.tick()
        time = result["time"]
        for rider_id, entry in result["events"]:
            order_id = entry["order_id"]
            if entry["event"] == "arrived_at_merchant":
                message = f"Rider #{rider_id} reached the restaurant for order #{order_id} and is delivering it."
            elif entry["event"] == "delivered_to_customer":
                message = f"Rider #{rider_id} delivered order #{order_id}."
            else:
                message = f"Rider #{rider_id}: {entry['event']} (order #{order_id})."
            self._print(f"[t={time}] {message}")
        for order in result["assigned"]:
            self._print(f"[t={time}] Order #{order.id} assigned to rider #{order.rider_id}.")
        for error in result["errors"]:
            self._print(f"[t={time}] Recovered from error: {error}")
        for fix in result["fixes"]:
            self._print(f"[t={time}] State repaired: {fix}")

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    def _try_parse(self, text: str) -> Optional[List[int]]:
        """Parse coordinates, printing an error message on failure."""
        try:
            return parse_coordinates(text)
        except ValueError as exc:
            self._print(f"Invalid input: {exc}.")
            return None

    def _find_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with the given id, if it exists."""
        for rider in self.system.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    # ------------------------------------------------------------------
    # Command handlers (each returns True to keep running)
    # ------------------------------------------------------------------
    def _cmd_start(self, args: List[str]) -> bool:
        """Create an order from four coordinates and dispatch it."""
        values: Optional[List[int]] = None
        if args:
            values = self._try_parse(" ".join(args))
            if values is None:
                return True
        else:
            self._print("Enter: restaurant_x restaurant_y customer_x customer_y (0-800)")
            for _ in range(MAX_INPUT_ATTEMPTS):
                text = self._input("start> ")
                values = self._try_parse(text)
                if values is not None:
                    break
            else:
                self._print("Too many invalid attempts; the order was cancelled.")
                return True

        try:
            order = self.system.order_manager.create_order(
                (values[0], values[1]), (values[2], values[3])
            )
        except ValueError as exc:
            self._print(f"Invalid order: {exc}.")
            return True

        new_riders, _assigned = self.system.dispatch()
        if order.rider_id is None:
            self._print(self.display.queued_report(order, self.system.funds))
            return True
        rider = self._find_rider(order.rider_id)
        if rider is None:  # state consistency guard
            self._print(f"Error: rider #{order.rider_id} not found for order #{order.id}.")
            return True
        self._print(self.display.start_report(order, rider, bool(new_riders)))
        return True

    def _cmd_status(self, _args: List[str]) -> bool:
        """Show a snapshot of funds, riders and the map."""
        self._print(self.display.status_report(self.system))
        return True

    def _cmd_orders(self, _args: List[str]) -> bool:
        """Show details of all orders."""
        self._print(self.display.orders_report(self.system.order_manager.history))
        return True

    def _cmd_riders(self, _args: List[str]) -> bool:
        """Show the status-change log of all riders."""
        self._print(self.display.rider_log_report(self.system.riders))
        return True

    def _cmd_map(self, _args: List[str]) -> bool:
        """Show the ASCII map of riders and active orders."""
        self._print(self.display.render_map(
            self.system.riders, self.system.order_manager.active_orders()))
        return True

    def _cmd_tick(self, args: List[str]) -> bool:
        """Advance the simulation a number of ticks."""
        self._auto_tick = False
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                self._print(f"Invalid tick count '{args[0]}': must be an integer.")
                return True
            if not 1 <= count <= 1000:
                self._print("Tick count must be between 1 and 1000.")
                return True
        for _ in range(count):
            self._run_tick()
        return True

    def _cmd_save(self, args: List[str]) -> bool:
        """Save the system state to a JSON file."""
        self._auto_tick = False
        path = args[0] if args else DEFAULT_SAVE_PATH
        try:
            saved = self.system.save_state(path)
        except OSError as exc:
            self._print(f"Could not save state: {exc}.")
            return True
        self._print(f"State saved to {saved}.")
        return True

    def _cmd_history(self, _args: List[str]) -> bool:
        """Show the command history."""
        self._auto_tick = False
        if not self.history:
            self._print("No commands yet.")
        for index, line in enumerate(self.history, start=1):
            self._print(f"{index:>4}  {line}")
        return True

    def _cmd_help(self, _args: List[str]) -> bool:
        """Show the command help."""
        self._auto_tick = False
        self._print(HELP_TEXT)
        return True

    def _cmd_quit(self, _args: List[str]) -> bool:
        """Stop the simulation and exit."""
        self._auto_tick = False
        self.system.stop()
        return False

from __future__ import annotations

import re
from dataclasses import dataclass, field

CUSTOM_PRIZE_LEVEL = 99

_LEVEL_PATTERNS = [
    (re.compile(r"\bfirst\b", re.IGNORECASE), 1),
    (re.compile(r"\bsecond\b", re.IGNORECASE), 2),
    (re.compile(r"\bthird\b", re.IGNORECASE), 3),
]


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize that can be drawn in a lottery round.

    avoid_groups: department groups whose members are drawn after other members.
    associated_drawer: the lottery person associated with this prize (or None).
    """

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    @property
    def level(self) -> int:
        """Prize level derived from the name: 1 = highest. Custom prizes rank last."""
        for pattern, value in _LEVEL_PATTERNS:
            if pattern.search(self.name):
                return value
        return CUSTOM_PRIZE_LEVEL

    @property
    def is_custom(self) -> bool:
        return self.level == CUSTOM_PRIZE_LEVEL


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Return prizes ordered from the highest level to the lowest (stable)."""
    return sorted(prizes, key=lambda prize: prize.level)

_SEPARATOR_PATTERN = re.compile(r"[,;]")


class ListManager:
    """Loads, validates and summarizes participant lists."""

    def __init__(self) -> None:
        self._users: list[User] = []

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        users = [User(name, department) for name, department in DEFAULT_PARTICIPANTS]
        self._users = list(users)
        return list(users), []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group' (or ';')."""
        users, errors = self._load_from_path(path)
        self._users = list(users)
        return list(users), errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.readlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]

        for line_number, raw_line in enumerate(lines, start=1):
            line = raw_line.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SEPARATOR_PATTERN.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {line_number}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of members per department group."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.models import Prize

DEFAULT_PRIZES: list[tuple[str, int]] = [
    ("Third Prize - Movie Tickets", 5),
    ("Second Prize - Shopping Cards", 3),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns error messages for skipped prizes."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError(f"Prize '{clean_name}': quantity must be a positive integer.")
        if any(existing.name.casefold() == clean_name.casefold() for existing in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]
        drawer = (associated_drawer or "").strip() or None
        prize = Prize(name=clean_name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def set_rules(
        self,
        index: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> bool:
        """Update avoidance groups and associated lottery person of the prize at a 0-based index."""
        if index < 0 or index >= len(self._prizes):
            return False
        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]
        drawer = (associated_drawer or "").strip() or None
        prize = self._prizes[index]
        prize.avoid_groups = groups
        prize.associated_drawer = drawer
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order contains 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            if sorted(order) != list(range(1, count + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility pools, special rules and anti-duplication."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User

LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
CHENG_RU_GROUP = "Cheng Ru Group"


def _norm(value: str | None) -> str:
    return (value or "").strip().casefold()


def _is_cheng_ru_member(user: User) -> bool:
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


class LotteryManager:
    """Runs draws for prizes while enforcing the lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winners: list[User] = []
        self._won_names: set[str] = set()
        # Results of the most recent draws by Chen Qixian / Jin Duo:
        # True means the draw produced a Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []

    @property
    def winners(self) -> list[User]:
        return list(self._winners)

    def has_won(self, user: User) -> bool:
        return _norm(user.name) in self._won_names

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the candidates for the next draw of the given prize.

        Priority:
        1. Previous winners are always excluded (anti-duplication).
        2. Fallback protection: if Ma Jiaqi is the lottery person and the last two
           results of Chen Qixian / Jin Duo contain no Cheng Ru group member, the
           pool is limited to Cheng Ru group members (ignoring avoidance groups).
        3. Special rule: Li Bin only filters previous winners.
        4. Otherwise members of avoidance groups are excluded in the first tier;
           they become eligible only after all other members are exhausted.
        """
        drawer = _norm(prize.associated_drawer)
        available = [user for user in self._users if not self.has_won(user)]

        if drawer == _norm(LI_BIN):
            return available

        if (
            drawer == _norm(MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        ):
            protected = [user for user in available if _is_cheng_ru_member(user)]
            if protected:
                return protected

        avoid = {_norm(group) for group in prize.avoid_groups}
        primary = [user for user in available if _norm(user.department) not in avoid]
        if primary:
            return primary
        return [user for user in available if _norm(user.department) in avoid]

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None when no candidate is eligible."""
        drawer = _norm(prize.associated_drawer)
        pool = self._get_eligible_pool(prize)
        winner = random.choice(pool) if pool else None

        if drawer in (_norm(CHEN_QIXIAN), _norm(JIN_DUO)):
            self.last_two_draw_results = (
                self.last_two_draw_results
                + [winner is not None and _is_cheng_ru_member(winner)]
            )[-2:]
        elif drawer == _norm(MA_JIAQI):
            # Protection applies to the lottery person immediately following Chen/Jin.
            self.last_two_draw_results = []

        if winner is not None:
            self._winners.append(winner)
            self._won_names.add(_norm(winner.name))
        return winner

"""Winner recording, summarization and CSV-style export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, sort_prizes_by_level


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    winning_time: datetime
    winning_order: int

    @property
    def name(self) -> str:
        return self.user.name

    @property
    def department(self) -> str:
        return self.user.department


class ResultManager:
    """Keeps the complete winner list of a lottery session."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._order_counter = 0

    def add_winner(self, user: User, prize: Prize | str) -> Winner:
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        self._order_counter += 1
        winner = Winner(
            user=user,
            prize_name=prize_name,
            winning_time=datetime.now(),
            winning_order=self._order_counter,
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map each prize name (in the given order) to its winners in winning order."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            name = prize.name if isinstance(prize, Prize) else str(prize)
            matched = [winner for winner in self._winners if winner.prize_name == name]
            result[name] = sorted(matched, key=lambda winner: winner.winning_order)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to awards_result_YYYYMMDD.txt in the current directory."""
        try:
            ordered_prizes = sort_prizes_by_level(prizes)
            grouped = self.get_winners_by_prize(ordered_prizes)
            date_text = datetime.now().strftime("%Y%m%d")
            path = os.path.join(os.getcwd(), f"awards_result_{date_text}.txt")
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        writer.writerow(
                            [
                                winner.name,
                                winner.department,
                                prize_name,
                                winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                            ]
                        )
            return True, f"Result successfully exported to {path}"
        except OSError as exc:
            return False, f"Export failed: {exc}"

def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _split_values(text: str) -> list[str]:
    return [part.strip() for part in re.split(r"[,;]", text) if part.strip()]


def describe_rules(prize: Prize) -> str:
    """Build a human readable description of the rules applied to a prize round."""
    parts: list[str] = []
    drawer = (prize.associated_drawer or "").strip()
    if drawer.casefold() == "li bin":
        parts.append("only previous winners are excluded (avoidance groups not applied)")
    else:
        parts.append("previous winners are excluded")
        if prize.avoid_groups:
            parts.append(
                "members of avoidance groups "
                f"[{', '.join(prize.avoid_groups)}] are drawn after other members"
            )
    if drawer.casefold() == "ma jiaqi":
        parts.append(
            "Cheng Ru Group protection: if the last two results of Chen Qixian and Jin Duo "
            "contain no Cheng Ru Group member, the pool is limited to Cheng Ru Group members"
        )
    if drawer:
        parts.append(f"associated lottery person: {drawer}")
    return "; ".join(parts)

    def _review_and_accept(self, users: list[User], errors: list[str]) -> bool:
        valid, duplicates = self.list_manager.validate_users(users)
        problems = list(errors)
        problems += [
            f"Duplicate name: {user.name} ({user.department}) - entry will be ignored"
            for user in duplicates
        ]

        if not problems:
            if not valid:
                print("The list is empty. Nothing was loaded.")
                return False
            self._accept_list(valid)
            return True

        print(f"\n{len(problems)} problem(s) found:")
        for problem in problems:
            print(f"  - {problem}")
        if not valid:
            print("No valid participants remain. Upload terminated.")
            return False

        print("Options: [r] remove all invalid/duplicate data and keep the rest")
        print("         [m] remove specific names manually from the valid list")
        print("         [t] terminate upload")
        while True:
            choice = self._ask("Select [r/m/t]: ").strip().lower()
            if choice == "r":
                self._accept_list(valid)
                return True
            if choice == "m":
                names = {name.casefold() for name in _split_values(

                remaining = [user for user in valid if user.name.casefold() not in names]
                if not remaining:
                    print("No valid participants remain. Upload terminated.")
                    return False
                self._accept_list(remaining)
                return True

    def _run_round(
        self,

        total_rounds: int,
        prize: Prize,

        print(f"Rule: {describe_rules(prize)}")
        print("-" * 60)

        round_winners: list[Winner] = []
        for draw_index in range(1, prize.quantity + 1):
            command = self._wait_for_command()
            if command == "terminate":
                print("Current round terminated. Skipping to the next round.")
                break

            print(f"Currently drawing the {_ordinal(draw_index)} winner...")
            user = lottery.draw(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            winner = self.results.add_winner(user, prize)
            round_winners.append(winner)
            print(
                f"Congratulations {user.name} ({user.department}) for winning {prize.name}!"
            )

        print(f"\nRound {round_number} results ({len(round_winners)} winner(s)):")
        if not round_winners:
            print("  (no winners in this round)")
        for winner in round_winners:
            print(f"  {winner.winning_order}. {winner.name} ({winner.department}) - {prize.name}")

    def _wait_for_command(self) -> str:
        """Return 'draw' or 'terminate'. Supports pause/continue/terminate commands."""
        while True:
            command = self._ask(
                "[Enter] draw next | 'pause' | 'terminate' > "
            ).strip().lower()
            if command in ("", "draw", "continue"):
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("paused > ").strip().lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if resume == "terminate":
                        return "terminate"
                    print("Type 'continue' or 'terminate'.")
            print("Unknown command. Press Enter to draw, or type 'pause' or 'terminate'.")

    # ---------------------------------------------------------- results
    def _print_results(self, prizes: list[Prize]) -> None:
        grouped = self.results.get_winners_by_prize(sort_prizes_by_level(prizes))
        if not any(grouped.values()):
            print("No winners recorded.")
            return
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            print(f"\n{prize_name}:")
            for winner in winners:
                print(
                    f"  {winner.winning_order}. {winner.name} ({winner.department}) "
                    f"- {winner.winning_time.strftime('%Y-%m-%d %H:%M:%S')}"
                )

    def _menu_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not self.results.get_all_winners():
            print("No results yet. Run the lottery first.")
            return
        print("\nResults")
        print("  1. Show winner list")
        print("  2. Export winner list to txt (CSV format)")
        print("  0. Back")
        choice = self._ask("Select: ").strip()
        if choice == "1":
            self._print_results(prizes)
        elif choice == "2":
            _, message = self.results.export_to_csv(prizes)
            print(message)

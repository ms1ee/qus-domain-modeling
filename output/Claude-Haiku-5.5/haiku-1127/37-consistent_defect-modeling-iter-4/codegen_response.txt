"""Small text helpers shared by the services."""

from __future__ import annotations


def normalize_text(text: str | None) -> str:
    """Normalize text for case-insensitive, whitespace-tolerant comparison."""
    return " ".join(str(text or "").split()).casefold()


def same_text(first: str | None, second: str | None) -> bool:
    """Return True if two names/groups are equal after normalization."""
    return normalize_text(first) == normalize_text(second)


def contains_group(department: str | None, groups: list[str]) -> bool:
    """Return True if the department matches any of the given groups."""
    return any(same_text(department, group) for group in groups)

from __future__ import annotations

import os
import re
from datetime import datetime

from lottery_system.constants import DEFAULT_PARTICIPANTS
from lottery_system.models import User
from lottery_system.utils import normalize_text

# Accepts ASCII and full-width commas / semicolons as field separators.
_FIELD_SEPARATORS = re.compile(r"[,;\uff0c\uff1b]")


class ListManager:
    """Loads, validates and summarizes participant lists."""

    def __init__(self) -> None:
        self.source_type: str | None = None
        self.file_path: str | None = None
        self.loaded_at: datetime | None = None
        self.is_valid: bool = False

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building participant list."""
        users = [User(name, department) for name, department in DEFAULT_PARTICIPANTS]
        self.source_type = "default"
        self.file_path = None
        self.loaded_at = datetime.now()
        self.is_valid = bool(users)
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list in the format 'Name,Department Group'."""
        users, errors = self._load_from_path(path)
        self.source_type = "custom"
        self.file_path = path
        self.loaded_at = datetime.now()
        self.is_valid = bool(users) and not errors
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file. Returns parsed users and error messages."""
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        lines: list[str] | None = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=encoding) as handle:
                    lines = handle.read().splitlines()
                break
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"File Error: cannot read '{path}': {exc}"]

        if lines is None:
            return [], [f"File Error: unsupported text encoding in '{path}'"]

        users: list[User] = []
        errors: list[str] = []
        for line_number, raw_line in enumerate(lines, start=1):
            text = raw_line.strip()
            if not text:
                continue
            parts = [part.strip() for part in _FIELD_SEPARATORS.split(text)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_number}: '{text}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users and duplicates.

        The first occurrence of each name is kept as valid; any later
        occurrence of the same name is reported as a duplicate.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the member count of every department group."""
        info: dict[str, int] = {}
        canonical: dict[str, str] = {}
        for user in users:
            key = normalize_text(user.department)
            display = canonical.setdefault(key, user.department.strip())
            info[display] = info.get(display, 0) + 1
        return info

"""Prize library management."""

from __future__ import annotations

import re

from lottery_system.constants import DEFAULT_PRIZES, ORDINAL_LEVELS
from lottery_system.models import Prize
from lottery_system.utils import normalize_text

_ORDINAL_PATTERN = re.compile(r"^\s*(first|second|third|fourth|fifth)\b", re.IGNORECASE)


class PrizeManager:
    """Manages the prize library and the lottery sequence of prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Load the preset prize library. Returns error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | str | None = None,
        associated_drawer: str | None = None,
        level: int | None = None,
    ) -> Prize:
        """Add a prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize already exists: {clean_name}")

        if isinstance(avoid_groups, str):
            avoid_groups = avoid_groups.split(",")
        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]

        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None

        if level is None:
            level = self._infer_level(clean_name)

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
            level=level,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. ``order`` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if len(order) != count:
            return False
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        return True

    def _find(self, name: str) -> Prize | None:
        key = normalize_text(name)
        for prize in self._prizes:
            if normalize_text(prize.name) == key:
                return prize
        return None

    def _infer_level(self, name: str) -> int:
        match = _ORDINAL_PATTERN.match(name)
        if match:
            return ORDINAL_LEVELS[match.group(1).lower()]
        existing = [prize.level for prize in self._prizes]
        return max(existing + [0, len(DEFAULT_PRIZES)]) + 1

"""Winner recording, result summaries and CSV/txt export."""

from __future__ import annotations

import csv
import io
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.constants import EXPORT_FILE_PREFIX, TIME_FORMAT
from lottery_system.models import Prize, User
from lottery_system.utils import normalize_text


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_order: int = 0
    winning_time: datetime = field(default_factory=datetime.now)

    @property
    def prize_name(self) -> str:
        return self.prize.name


class ResultManager:
    """Keeps the winner list and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._won_names: set[str] = set()

    def has_won(self, user: User | str) -> bool:
        """Return True if the person (by name) has already won a prize."""
        name = user.name if isinstance(user, User) else user
        return normalize_text(name) in self._won_names

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner. Raises ValueError if the person already won."""
        if self.has_won(user):
            raise ValueError(f"{user.name} has already won a prize.")
        winner = Winner(
            user=user,
            prize=prize,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        self._won_names.add(normalize_text(user.name))
        return winner

    def get_all_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low)."""
        result: dict[str, list[Winner]] = {}
        for prize in self._sorted_prizes(prizes):
            result[prize.name] = [w for w in self._winners if w.prize.name == prize.name]
        return result

    def ordered_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Flatten winners by prize level (high to low), then winning order."""
        ordered: list[Winner] = []
        for prize_winners in self.get_winners_by_prize(prizes).values():
            ordered.extend(prize_winners)
        return ordered

    def export_to_csv(self, prizes: list[Prize], directory: str | None = None) -> tuple[bool, str]:
        """Export winners to ``awards_result_YYYYMMDD.txt`` in CSV format."""
        date_text = datetime.now().strftime("%Y%m%d")
        target_dir = directory or os.getcwd()
        path = os.path.join(target_dir, f"{EXPORT_FILE_PREFIX}{date_text}.txt")
        try:
            buffer = io.StringIO()
            writer = csv.writer(buffer)
            writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
            for winner in self.ordered_winners(prizes):
                writer.writerow([
                    winner.user.name,
                    winner.user.department,
                    winner.prize.name,
                    winner.winning_time.strftime(TIME_FORMAT),
                ])
            with open(path, "w", encoding="utf-8", newline="") as handle:
                handle.write(buffer.getvalue())
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

    @staticmethod
    def _sorted_prizes(prizes: list[Prize]) -> list[Prize]:
        return sorted(prizes, key=lambda prize: prize.level)

"""Lottery rule engine: eligibility, avoidance, special rules and fallback."""

from __future__ import annotations

import random

from lottery_system.constants import CHEN_QIXIAN, CHENG_RU_GROUP, JIN_DUO, LI_BIN, MA_JIAQI
from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager
from lottery_system.utils import contains_group, same_text


class LotteryManager:
    """Executes lottery draws according to the configured rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Cheng Ru group results (True = winners contained a Cheng Ru member)
        # of the most recent rounds conducted by Chen Qixian or Jin Duo.
        self.last_two_draw_results: list[bool] = []
        self._rng = random.Random()

    @property
    def users(self) -> list[User]:
        return list(self._users)

    # ----- helpers -------------------------------------------------------

    def _remaining(self) -> list[User]:
        """Participants who have not won any prize yet."""
        return [user for user in self._users if not self.result_manager.has_won(user)]

    @staticmethod
    def _drawer_is(prize: Prize, name: str) -> bool:
        return prize.associated_drawer is not None and same_text(prize.associated_drawer, name)

    def _is_li_bin_round(self, prize: Prize) -> bool:
        return self._drawer_is(prize, LI_BIN)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return same_text(user.department, CHENG_RU_GROUP)

    @staticmethod
    def _is_avoided(user: User, prize: Prize) -> bool:
        return contains_group(user.department, prize.avoid_groups)

    def _is_fallback_active(self, prize: Prize) -> bool:
        """Cheng Ru protection: Ma Jiaqi's pool is forced to Cheng Ru members
        when the last two Chen Qixian / Jin Duo results had no Cheng Ru member."""
        return (
            self._drawer_is(prize, MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    # ----- eligibility ---------------------------------------------------

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Primary eligible pool (excludes previous winners and avoidance groups)."""
        remaining = self._remaining()
        if self._is_fallback_active(prize):
            return [user for user in remaining if self._is_cheng_ru(user)]
        if self._is_li_bin_round(prize):
            return remaining
        return [user for user in remaining if not self._is_avoided(user, prize)]

    def _get_avoidance_pool(self, prize: Prize) -> list[User]:
        """Avoidance-group members, drawn only after the primary pool is exhausted."""
        if self._is_fallback_active(prize) or self._is_li_bin_round(prize):
            return []
        return [user for user in self._remaining() if self._is_avoided(user, prize)]

    # ----- drawing -------------------------------------------------------

    def draw_next(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = self._get_avoidance_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self.result_manager.add_winner(winner, prize)
        return winner

    def complete_round(self, prize: Prize, winners: list[User]) -> None:
        """Update fallback history after a round finishes."""
        if self._drawer_is(prize, CHEN_QIXIAN) or self._drawer_is(prize, JIN_DUO):
            round_has_cheng_ru = any(self._is_cheng_ru(user) for user in winners)
            self.last_two_draw_results.append(round_has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def rule_description(self, prize: Prize) -> str:
        """Human-readable description of the rules for a round."""
        parts: list[str] = []
        if self._is_li_bin_round(prize):
            parts.append("Only previous winners are excluded; avoidance groups do not apply.")
        elif self._is_fallback_active(prize):
            parts.append(
                "Cheng Ru Group protection active: only Cheng Ru Group members are eligible."
            )
        else:
            parts.append("Previous winners are excluded.")
            if prize.avoid_groups:
                parts.append(
                    "Members of avoidance groups ("
                    + ", ".join(prize.avoid_groups)
                    + ") are drawn only after other members."
                )
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}.")
        return " ".join(parts)

"""Non-blocking stdin channel used for lottery control commands."""

from __future__ import annotations

import queue
import sys
import threading


class _Sentinel:
    """Marker object with a readable representation."""

    def __init__(self, label: str) -> None:
        self._label = label

    def __repr__(self) -> str:
        return f"<{self._label}>"


NO_INPUT = _Sentinel("NO_INPUT")
END_OF_INPUT = _Sentinel("END_OF_INPUT")


class InputChannel:
    """Reads stdin in a background thread so commands can be polled during a lottery."""

    def __init__(self, stream=None) -> None:
        self._stream = stream if stream is not None else sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._thread = threading.Thread(target=self._read_loop, name="stdin-reader", daemon=True)
        self._thread.start()

    def _read_loop(self) -> None:
        try:
            for line in self._stream:
                self._queue.put(line.strip())
        except (OSError, ValueError):
            pass
        finally:
            self._queue.put(END_OF_INPUT)

    def get(self):
        """Blocking read. Returns a line or END_OF_INPUT."""
        item = self._queue.get()
        if item is END_OF_INPUT:
            self._queue.put(END_OF_INPUT)
        return item

    def poll(self):
        """Non-blocking read. Returns a line, END_OF_INPUT, or NO_INPUT."""
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return NO_INPUT
        if item is END_OF_INPUT:
            self._queue.put(END_OF_INPUT)
        return item

def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryApp:
    """Interactive command-line application."""

    def __init__(self, channel: InputChannel | None = None) -> None:
        self.channel = channel if channel is not None else InputChannel()
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()

        self.lottery: LotteryManager | None = None

    # ----- main loop -----------------------------------------------------

        print("=" * 60)
        print("Company Team Building - Smart Lottery System")
        print("=" * 60)
        try:
            while True:
                self._print_main_menu()
                choice = self._ask("Select an option: ").strip()
                if choice == "1":
                    self._load_participants()
                elif choice == "2":
                    self._show_groups()
                elif choice == "3":
                    self._configure_prizes()
                elif choice == "4":
                    self._run_lottery()
                elif choice == "5":
                    self._show_results()
                elif choice == "6":
                    self._export_results()
                elif choice == "0":
                    print("Goodbye!")
                    break
                else:
                    print("Invalid option, please try again.")

            print("\nInput closed. Exiting.")
        except KeyboardInterrupt:
            print("\nExiting.")

                print("Invalid option.")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        quantity_text = self._ask("Quantity: ").strip()
        try:
            quantity = int(quantity_text)
        except ValueError:
            print("Error: quantity must be an integer.")
            return
        if self.users:
            info = self.list_manager.get_group_info(self.users)
            print("Available groups: " + ", ".join(info.keys()))
        groups_text = self._ask("Avoidance groups (comma-separated, blank for none): ")
        groups = [group.strip() for group in groups_text.split(",") if group.strip()]
        drawer = self._ask("Associated lottery person (blank for none): ").strip() or None
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            print(f"Error: {exc}")
            return
        print(f"Prize added: {prize.name} (quantity {prize.quantity}).")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            print(
                f"{index}. {prize.name} | Quantity: {prize.quantity} | Level: {prize.level} "
                f"| Avoid groups: {groups} | Lottery person: {drawer}"
            )

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        text = self._ask(f"Enter new order as serial numbers (1-{len(prizes)}), e.g. 2 1 3: ")
        try:
            order = [int(token) for token in text.replace(",", " ").split()]
        except ValueError:
            print("Invalid order: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ----- lottery process -----------------------------------------------

    def _run_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return

        self.lottery = LotteryManager(self.users)
        print("\nLottery starts!")
        print("Commands during drawing: 'pause' | 'continue' | 'terminate' (ends current round).")
        for number, prize in enumerate(prizes, start=1):
            self._run_round(number, prize)
        print("\nAll lottery rounds have been processed.")
        self._show_results()

    def _run_round(self, number: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {number}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {self.lottery.rule_description(prize)}")

        drawn: list[User] = []
        terminated = False
        for order in range(1, prize.quantity + 1):
            if self._checkpoint():
                terminated = True
                break
            print(f"Currently drawing the {_ordinal(order)} winner...", flush=True)
            time.sleep(DRAW_DELAY_SECONDS)
            if self._checkpoint():
                terminated = True
                break
            user = self.lottery.draw_next(prize)
            if user is None:
                print("No eligible participants remain for this prize.")
                break
            drawn.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self.lottery.complete_round(prize, drawn)

        if terminated:
            print(f"Round terminated by command. {len(drawn)} of {prize.quantity} winner(s) drawn.")
        elif len(drawn) < prize.quantity:
            print(f"Only {len(drawn)} of {prize.quantity} winner(s) could be drawn.")
        else:
            print("Round complete.")

        if drawn:
            print("Round results (in winning order):")
            for index, user in enumerate(drawn, start=1):
                print(f"  {index}. {user.name} ({user.department})")

    def _checkpoint(self) -> bool:
        """Process pending control commands. Returns True if the round must end."""
        item = self.channel.poll()
        while item is not NO_INPUT:
            if item is END_OF_INPUT:
                raise EOFError
            command = item.strip().lower()
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    reply = self.channel.get()
                    if reply is END_OF_INPUT:
                        raise EOFError
                    reply = reply.strip().lower()
                    if reply == "continue":
                        print("Lottery resumed.")
                        break
                    if reply == "terminate":
                        return True
                    if reply:
                        print("Lottery is paused. Type 'continue' or 'terminate'.")
            elif command == "terminate":
                return True
            elif command == "continue":
                print("Lottery is not paused.")
            elif command:
                print(f"Unknown command '{item}'. Available: pause, continue, terminate.")
            item = self.channel.poll()
        return False

    # ----- results -------------------------------------------------------

    def _show_results(self) -> None:
        if self.lottery is None:
            print("No lottery has been run yet.")
            return
        prizes = self.prize_manager.get_prizes()
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        print("\n=== Winner List ===")
        total = 0
        for prize_name, winners in grouped.items():
            print(f"\n[{prize_name}]")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                total += 1
                print(
                    f"  {winner.winning_order}. {winner.user.name} ({winner.user.department}) "
                    f"- {winner.winning_time.strftime('%Y-%m-%d %H:%M:%S')}"
                )
        print(f"\nTotal winners: {total}")

    def _export_results(self) -> None:
        if self.lottery is None:
            print("No lottery has been run yet.")
            return
        prizes = self.prize_manager.get_prizes()
        _, message = self.lottery.result_manager.export_to_csv(prizes)
        print(message)


def main() -> None:
    """Entry point for the lottery system."""
    LotteryApp().run()

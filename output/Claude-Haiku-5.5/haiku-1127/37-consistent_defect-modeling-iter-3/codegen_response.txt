from __future__ import annotations

import os
import re

from lottery_system.models import User

DEFAULT_LIST_LINES: list[str] = [
    "Zhang San;Technology Department",
    "Li Bin;Product Department",
    "Wang Wu;Marketing Department",
    "Chen Qixian;Cheng Ru Group",
    "Jin Duo;Technology Department",
    "Ma Jiaqi;Cheng Ru Group",
    "Zhao Liu;Finance Department",
    "Sun Qi;Human Resources Department",
    "Zhou Ba;Product Department",
    "Wu Jiu;Marketing Department",
    "Zheng Shi;Cheng Ru Group",
    "Feng Yi;Technology Department",
    "Chen Er;Product Department",
    "Chu San;Finance Department",
    "Wei Si;Human Resources Department",
    "Jiang Wu;Cheng Ru Group",
    "Shen Liu;Marketing Department",
    "Han Qi;Technology Department",
    "Yang Ba;Product Department",
    "Zhu Jiu;Finance Department",
]

_SPLIT_PATTERN = re.compile(r"[,;]")


class ListManager:
    """Loads, validates and summarizes participant lists."""

    def __init__(self) -> None:
        self.source_mode: str = ""
        self.file_path: str = ""
        self.validation_status: str = "not validated"

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        self.source_mode = "default"
        self.file_path = ""
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.validation_status = "valid" if not errors else "invalid"
        return users, errors

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: "Name,Department" or "Name;Department"."""
        self.source_mode = "custom"
        self.file_path = (path or "").strip().strip('"')
        return self._load_from_path(self.file_path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            lines = self._read_lines(path)
        except OSError as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        users, errors = self._parse_lines(lines)
        self.validation_status = "valid" if not errors else "invalid"
        return users, errors

    @staticmethod
    def _read_lines(path: str) -> list[str]:
        with open(path, "rb") as handle:
            raw = handle.read()
        for encoding in ("utf-8-sig", "gbk"):
            try:
                return raw.decode(encoding).splitlines()
            except UnicodeDecodeError:
                continue
        raise OSError("unsupported text encoding (expected UTF-8 or GBK)")

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw_line in enumerate(lines, start=1):
            line = raw_line.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SPLIT_PATTERN.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates).

        The first occurrence of each name is kept as valid; later occurrences
        with the same name (case-insensitive) are returned as duplicates.
        """
        valid: list[User] = []
        duplicates: list[User] = []
        seen: set[str] = set()
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department name -> number of members, in first-seen order."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

import re

from lottery_system.models import UNRANKED_LEVEL, Prize

PRESET_PRIZES: list[tuple[str, int]] = [
    ("Third Prize - Movie Tickets", 5),
    ("Second Prize - Shopping Cards", 3),
    ("First Prize - 1 Day Annual Leave", 1),
]

_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3, "fourth": 4, "fifth": 5}
_LEVEL_PATTERN = re.compile(r"^\s*(First|Second|Third|Fourth|Fifth)\s+Prize", re.IGNORECASE)


def parse_level(name: str) -> int:
    """Derive prize level from its name prefix (e.g. 'First Prize ...' -> 1)."""
    match = _LEVEL_PATTERN.match(name or "")
    if not match:
        return UNRANKED_LEVEL
    return _LEVEL_WORDS[match.group(1).lower()]


class PrizeManager:
    """Manages the prize list and lottery round order. Starts with no prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in PRESET_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the lottery sequence."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name must not be empty.")
        if any(p.name.casefold() == name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")

        groups: list[str] = []
        for group in avoid_groups or []:
            group = (group or "").strip()
            if group and group not in groups:
                groups.append(group)

        drawer = (associated_drawer or "").strip() or None

        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
            level=parse_level(name),
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            normalized = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if len(normalized) != count or sorted(normalized) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in normalized]
        return True

"""Lottery rule engine: eligibility filtering, avoidance, special rules and fallback."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User

PROTECTED_GROUP = "Cheng Ru Group"
FALLBACK_TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


def _norm(text: str | None) -> str:
    return (text or "").strip().casefold()


def is_protected_group(user: User) -> bool:
    """True if the user belongs to the protected 'Cheng Ru group'."""
    return _norm(user.department) == _norm(PROTECTED_GROUP)


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.winners: list[User] = []
        self._winner_names: set[str] = set()
        # Records, for the last two trigger-drawer rounds, whether the round
        # contained Cheng Ru group members (True) or not (False).
        self.last_two_draw_results: list[bool] = []
        self.draw_index: int = 0
        self.status: str = "idle"

    def has_won(self, user: User) -> bool:
        return _norm(user.name) in self._winner_names

    def _available(self) -> list[User]:
        """Users who have not won any prize yet (anti-duplication)."""
        return [user for user in self.users if not self.has_won(user)]

    def is_fallback_active(self, prize: Prize) -> bool:
        """Fallback applies to Ma Jiaqi's rounds when the last two trigger
        rounds (Chen Qixian / Jin Duo) both contained no Cheng Ru group members."""
        return (
            _norm(prize.associated_drawer) == _norm(FALLBACK_TARGET_DRAWER)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _split_tiers(self, prize: Prize) -> tuple[list[User], list[User]]:
        """Return (primary_pool, deferred_pool).

        Primary pool is drawn first. Deferred pool holds members of the prize's
        avoidance groups and is only drawn after the primary pool is exhausted.
        """
        available = self._available()

        if self.is_fallback_active(prize):
            protected = [user for user in available if is_protected_group(user)]
            if protected:
                return protected, []

        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            # Special rule: only previous winners are filtered out.
            return available, []

        avoid = {_norm(group) for group in prize.avoid_groups}
        primary = [user for user in available if _norm(user.department) not in avoid]
        deferred = [user for user in available if _norm(user.department) in avoid]
        return primary, deferred

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """All users currently eligible for the prize, primary members first."""
        primary, deferred = self._split_tiers(prize)
        return primary + deferred

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no one is eligible."""
        primary, deferred = self._split_tiers(prize)
        pool = primary or deferred
        if not pool:
            return None
        winner = random.choice(pool)
        self.winners.append(winner)
        self._winner_names.add(_norm(winner.name))
        self.draw_index += 1
        return winner

    def finish_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Update fallback history after a round has completed."""
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(name) for name in FALLBACK_TRIGGER_DRAWERS}:
            hit = any(is_protected_group(user) for user in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_TARGET_DRAWER):
            # The fallback has been consumed by this round.
            self.last_two_draw_results.clear()

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        parts: list[str] = []
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}.")
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            parts.append("Only previous winners are excluded; avoidance groups do not apply.")
        else:
            parts.append("Previous winners are excluded.")
            if prize.avoid_groups:
                groups = ", ".join(prize.avoid_groups)
                parts.append(
                    f"Members of avoidance group(s) [{groups}] are drawn after other members."
                )
        if self.is_fallback_active(prize):
            parts.append(
                f"Fallback active: only {PROTECTED_GROUP} members are eligible."
            )
        return " ".join(parts)

"""Winner recording, result summarization and CSV export."""

from __future__ import annotations

import csv
import datetime
import io
import os
from dataclasses import dataclass

from lottery_system.models import Prize, User

TIME_FORMAT = "%Y-%m-%d %H:%M:%S"


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    prize_level: int
    winning_order: int
    winning_time: datetime.datetime


class ResultManager:
    """Maintains the winner list in real time and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._won_names: set[str] = set()

    def add_winner(self, user: User, prize: Prize) -> Winner:
        key = user.name.strip().casefold()
        if key in self._won_names:
            raise ValueError(f"{user.name} has already won a prize.")
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=prize.level,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.datetime.now(),
        )
        self._winners.append(winner)
        self._won_names.add(key)
        return winner

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map prize name -> winners in winning time order."""
        return {
            prize.name: [w for w in self._winners if w.prize_name == prize.name]
            for prize in prizes
        }

    def get_sorted_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Winners ordered by prize level (high to low), then winning order."""
        by_prize = self.get_winners_by_prize(prizes)
        ordered_prizes = sorted(prizes, key=lambda prize: prize.level)
        result: list[Winner] = []
        for prize in ordered_prizes:
            result.extend(sorted(by_prize[prize.name], key=lambda w: w.winning_order))
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export winners to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        winners = self.get_sorted_winners(prizes)
        if not winners:
            return False, "No winners to export."

        now = datetime.datetime.now()
        path = os.path.join(os.getcwd(), f"awards_result_{now:%Y%m%d}.txt")

        buffer = io.StringIO()
        writer = csv.writer(buffer, lineterminator="\n")
        writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
        for winner in winners:
            writer.writerow(
                [
                    winner.user.name,
                    winner.user.department,
                    winner.prize_name,
                    winner.winning_time.strftime(TIME_FORMAT),
                ]
            )

        try:
            with open(path, "w", encoding="utf-8-sig", newline="") as handle:
                handle.write(buffer.getvalue())
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

def ordinal(number: int) -> str:
    """Return 1 -> '1st', 2 -> '2nd', 11 -> '11th', etc."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def ask(prompt: str) -> str:
    return input(prompt)


def ask_int(prompt: str, minimum: int = 1) -> int:
    while True:
        text = ask(prompt).strip()
        try:
            value = int(text)
        except ValueError:
            print("Please enter a whole number.")
            continue
        if value < minimum:
            print(f"Please enter a number >= {minimum}.")
            continue
        return value


def ask_yes_no(prompt: str) -> bool:
    return ask(prompt).strip().lower() in ("y", "yes")


def parse_int_list(text: str) -> list[int]:
    """Parse '2 1 3' or '2,1,3' into [2, 1, 3]. Raises ValueError on bad input."""
    tokens = [token for token in re.split(r"[,\s]+", text.strip()) if token]
    return [int(token) for token in tokens]

        print("Lottery person:")
        print("  0. None")
        for index, name in enumerate(DRAWER_OPTIONS, start=1):
            print(f"  {index}. {name}")
        print(f"  {len(DRAWER_OPTIONS) + 1}. Other (type a name)")
        while True:
            choice = ask("Select a lottery person: ").strip()
            if choice == "0":
                return None
            if choice.isdigit():
                number = int(choice)
                if 1 <= number <= len(DRAWER_OPTIONS):
                    return DRAWER_OPTIONS[number - 1]
                if number == len(DRAWER_OPTIONS) + 1:
                    custom = ask("Enter the lottery person's name: ").strip()
                    return custom or None
            print("Invalid selection.")

    # ---------------------------------------------------------------- lottery
    def run_lottery(self) -> None:
        assert self.lottery is not None
        prizes = self.prize_manager.get_prizes()
        print(SEPARATOR)
        print("Lottery starts!")
        print("Commands: Enter or 'continue' = next draw | 'pause' = pause | "
              "'terminate' = end current round")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, prize)
        print(SEPARATOR)
        print("All lottery rounds completed.")

    def _run_round(self, round_no: int, prize) -> None:
        assert self.lottery is not None
        print(SEPARATOR)
        print(f"Round {round_no}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {self.lottery.describe_rule(prize)}")

        round_winners: list[User] = []
        while len(round_winners) < prize.quantity:
            command = ask("Press Enter to draw (or 'pause' / 'terminate'): ").strip().lower()
            if command == "terminate":
                print("Current round terminated.")
                break
            if command == "pause":
                self.lottery.status = "paused"
                if self._handle_pause():
                    print("Current round terminated.")
                    break
                self.lottery.status = "running"
            elif command not in ("", "continue"):
                print("Unknown command.")
                continue

            position = len(round_winners) + 1
            print(f"Currently drawing the {ordinal(position)} winner...")
            user = self.lottery.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize.")
                break
            self.results.add_winner(user, prize)
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        self.lottery.finish_round(prize, round_winners)
        self.lottery.status = "idle"

        if round_winners:
            print(f"Round {round_no} results:")
            for position, user in enumerate(round_winners, start=1):
                print(f"  {position}. {user.name} ({user.department})")
        else:
            print("No winners were drawn in this round.")
        if len(round_winners) < prize.quantity:
            print(f"Note: {len(round_winners)} of {prize.quantity} prize(s) awarded.")

    def _handle_pause(self) -> bool:
        """Wait until 'continue' (returns False) or 'terminate' (returns True)."""
        print("Lottery paused.")
        while True:
            command = ask("Type 'continue' to resume or 'terminate' to end this round: ")
            command = command.strip().lower()
            if command in ("continue", ""):
                print("Lottery resumed.")
                return False
            if command == "terminate":
                return True
            print("Unknown command.")

    # ---------------------------------------------------------------- results
    def show_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        by_prize = self.results.get_winners_by_prize(prizes)
        print(SEPARATOR)
        print("FINAL WINNER LIST (by prize level, high to low)")
        print(SEPARATOR)
        if not any(by_prize.values()):
            print("No winners.")
            return
        for prize in sorted(prizes, key=lambda p: p.level):
            winners = by_prize[prize.name]
            if not winners:
                continue
            print(f"{prize.name}:")
            for winner in winners:
                print(
                    f"  {winner.winning_order}. {winner.user.name} "
                    f"({winner.user.department}) - {prize.name}"
                )

    def export_results(self) -> None:
        if not ask_yes_no("Export the winner list to file? (y/n): "):
            return
        _, message = self.results.export_to_csv(self.prize_manager.get_prizes())
        print(message)

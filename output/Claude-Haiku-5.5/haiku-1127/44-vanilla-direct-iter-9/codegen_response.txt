"""Loading of the Beijing Subway network dataset.

The dataset lives in ``subway_data.json`` next to this module. The path is
resolved relative to this file so the program works from any working
directory.
"""

import json
import os
from typing import Any, Dict

DATA_FILE_PATH: str = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)

REQUIRED_KEYS = ("stations", "connections")


def load_subway_data(path: str) -> Dict[str, Any]:
    """Load the subway dataset from a JSON file.

    Args:
        path: Path to the JSON file containing ``stations`` and ``connections``.

    Returns:
        The parsed JSON content as a dictionary.

    Raises:
        OSError: If the file cannot be opened.
        ValueError: If the JSON is malformed or lacks the required structure.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)

    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")

    for key in REQUIRED_KEYS:
        if not isinstance(data.get(key), list):
            raise ValueError(f"Subway data file must contain a list under '{key}'")

    return data

"""Graph data structures for the Beijing Subway network.

Stations are identified by integer numbers assigned by the caller (see
``subway_network.SubwayNetwork``). The graph is an undirected weighted graph
whose adjacency lists hold shared ``Edge`` objects.
"""

from typing import Dict, Hashable, List, Optional


class Edge:
    """An undirected, weighted connection between two stations on one line.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance in meters.
        line: Name of the subway line this connection belongs to.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source: Hashable, dest: Hashable, distance: float, line: str):
        if distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self) -> float:
        """Edge weight used by path algorithms (equal to the distance)."""
        return self.distance

    @weight.setter
    def weight(self, value: float) -> None:
        if value < 0:
            raise ValueError("Edge weight must be non-negative")
        self.distance = value

    def other(self, vertex: Hashable) -> Hashable:
        """Return the endpoint opposite to ``vertex``.

        Raises:
            ValueError: If ``vertex`` is not an endpoint of this edge.
        """
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex!r} is not an endpoint of this edge")

    def __repr__(self) -> str:
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph with adjacency lists.

    Also keeps ``station_lines``, a mapping from station name to the list of
    lines serving that station, which is used for transfer-station handling.
    """

    def __init__(self) -> None:
        self._adj: Dict[Hashable, List[Edge]] = {}
        self.station_lines: Dict[str, List[str]] = {}

    # ----- node management -------------------------------------------------
    def add_node(self, vertex: Hashable) -> None:
        """Add a vertex if it does not already exist."""
        self._adj.setdefault(vertex, [])

    def has_node(self, vertex: Hashable) -> bool:
        """Return True if the vertex exists in the graph."""
        return vertex in self._adj

    def nodes(self) -> List[Hashable]:
        """Return all vertices as a list."""
        return list(self._adj.keys())

    @property
    def num_nodes(self) -> int:
        """Number of vertices."""
        return len(self._adj)

    def adj(self, vertex: Hashable) -> List[Edge]:
        """Return the adjacency list of ``vertex``. Callers must not mutate it.

        Raises:
            KeyError: If the vertex does not exist.
        """
        if vertex not in self._adj:
            raise KeyError(f"Vertex {vertex!r} is not in the graph")
        return self._adj[vertex]

    # ----- edge management -------------------------------------------------
    def add_edge(self, edge: Edge) -> None:
        """Insert an undirected edge into the adjacency lists of both endpoints."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self._adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self._adj[edge.dest].append(edge)

    def find_edges(self, u: Hashable, v: Hashable, line: Optional[str] = None) -> List[Edge]:
        """Return all edges between ``u`` and ``v``, optionally filtered by line."""
        result: List[Edge] = []
        for edge in self._adj.get(u, []):
            if edge.other(u) == v and (line is None or edge.line == line):
                result.append(edge)
        return result

    def remove_edge(self, u: Hashable, v: Hashable, line: Optional[str] = None) -> int:
        """Remove edges between ``u`` and ``v`` (optionally on one line).

        Returns:
            The number of edges removed.
        """
        targets = self.find_edges(u, v, line)
        for edge in targets:
            for node in (edge.source, edge.dest):
                self._adj[node] = [e for e in self._adj[node] if e is not edge]
        return len(targets)

    def edges(self) -> List[Edge]:
        """Return every distinct edge exactly once."""
        seen = set()
        result: List[Edge] = []
        for edge_list in self._adj.values():
            for edge in edge_list:
                if id(edge) not in seen:
                    seen.add(id(edge))
                    result.append(edge)
        return result

    @property
    def num_edges(self) -> int:
        """Number of distinct edges."""
        return len(self.edges())

    # ----- station line information ----------------------------------------
    def set_station_lines(self, station_name: str, lines: List[str]) -> None:
        """Set the list of lines serving a station."""
        self.station_lines[station_name] = list(lines)

    def get_station_lines(self, station_name: str) -> List[str]:
        """Return a copy of the lines serving a station (empty if unknown)."""
        return list(self.station_lines.get(station_name, []))

"""Route search for the Beijing Subway network.

Implements Dijkstra's algorithm with a binary-heap priority queue (``MinPQ``).
The search runs over states ``(station, incoming_line)`` so that line changes
(transfers) can be counted and optimized exactly.

Supported query modes:
    shortest_distance -- minimize total distance (meters)
    least_transfer    -- minimize transfers, then distance
    comprehensive     -- minimize distance plus a time-equivalent transfer penalty
"""

from dataclasses import dataclass, field
from typing import Dict, List, Optional, Sequence, Tuple

from models import Edge, Graph

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# 5 minutes per transfer at 1.5 minutes per kilometer ~= 3.33 km of travel.
TRANSFER_EQUIVALENT_METERS = 3333.0
# Soft penalty applied to edges that are not on a preferred line.
LINE_PREFERENCE_PENALTY_METERS = 1000.0


class MinPQ:
    """Binary min-heap priority queue.

    Items are stored with comparable keys; ``del_min`` returns the item with
    the smallest key. Insertion and removal run in O(log n).
    """

    def __init__(self) -> None:
        self._keys: list = []
        self._items: list = []

    def __len__(self) -> int:
        return len(self._keys)

    def size(self) -> int:
        """Number of elements in the queue."""
        return len(self._keys)

    def is_empty(self) -> bool:
        """Return True if the queue holds no elements."""
        return not self._keys

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._keys:
            raise IndexError("priority queue is empty")
        return self._keys[0]

    def min_item(self):
        """Return the item with the smallest key without removing it."""
        if not self._items:
            raise IndexError("priority queue is empty")
        return self._items[0]

    def insert(self, key, item) -> None:
        """Insert an item with the given priority key."""
        self._keys.append(key)
        self._items.append(item)
        self._swim(len(self._keys) - 1)

    def del_min(self):
        """Remove and return a ``(key, item)`` pair with the smallest key."""
        if not self._keys:
            raise IndexError("priority queue underflow")
        key, item = self._keys[0], self._items[0]
        last = len(self._keys) - 1
        self._swap(0, last)
        self._keys.pop()
        self._items.pop()
        if self._keys:
            self._sink(0)
        return key, item

    def _greater(self, i: int, j: int) -> bool:
        return self._keys[i] > self._keys[j]

    def _swap(self, i: int, j: int) -> None:
        self._keys[i], self._keys[j] = self._keys[j], self._keys[i]
        self._items[i], self._items[j] = self._items[j], self._items[i]

    def _swim(self, k: int) -> None:
        """Move the element at index ``k`` up until heap order holds."""
        while k > 0:
            parent = (k - 1) // 2
            if self._greater(parent, k):
                self._swap(parent, k)
                k = parent
            else:
                break

    def _sink(self, k: int) -> None:
        """Move the element at index ``k`` down until heap order holds."""
        n = len(self._keys)
        while True:
            child = 2 * k + 1
            if child >= n:
                break
            if child + 1 < n and self._greater(child, child + 1):
                child += 1
            if not self._greater(k, child):
                break
            self._swap(k, child)
            k = child


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of ``shortest_distance``, ``least_transfer``, ``comprehensive``.
        max_transfers: Maximum allowed number of transfers, or None for no limit.
        preferred_lines: Lines that are favoured; other lines carry a soft penalty.
    """

    def __init__(
        self,
        mode: str = MODE_SHORTEST_DISTANCE,
        max_transfers: Optional[int] = None,
        preferred_lines: Optional[Sequence[str]] = None,
    ) -> None:
        if mode not in VALID_MODES:
            raise ValueError(f"Unknown query mode '{mode}'")
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines: List[str] = list(preferred_lines or [])


class _Label:
    """Best known information for one search state."""

    __slots__ = ("key", "distance", "cost", "transfers", "prev", "edge")

    def __init__(self, key, distance, cost, transfers, prev, edge):
        self.key = key              # priority used for ordering
        self.distance = distance    # actual travelled distance (meters)
        self.cost = cost            # distance plus preference penalties
        self.transfers = transfers  # number of line changes so far
        self.prev = prev            # predecessor state, None for the source
        self.edge = edge            # edge used to reach this state


State = Tuple[int, Optional[str]]


class ShortestPath:
    """Dijkstra-based search from a source station.

    Attributes:
        source: Station number the search starts from.
        config: The ``PathfindingConfig`` used for the search.
        dist_to: Mapping from reachable station number to actual distance (m).
    """

    def __init__(self, graph: Graph, source: int, config: Optional[PathfindingConfig] = None):
        if not graph.has_node(source):
            raise ValueError(f"Source station {source} is not in the graph")
        self._graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self._labels: Dict[State, _Label] = {}
        self._best_state: Dict[int, State] = {}
        self.dist_to: Dict[int, float] = {}
        self._run()

    def _make_key(self, cost: float, transfers: int):
        """Build the ordering key for the configured query mode."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, cost)
        if mode == MODE_COMPREHENSIVE:
            return (cost + transfers * TRANSFER_EQUIVALENT_METERS,)
        return (cost,)

    def _run(self) -> None:
        """Execute Dijkstra's algorithm over (station, line) states."""
        start: State = (self.source, None)
        self._labels[start] = _Label(self._make_key(0, 0), 0, 0, 0, None, None)

        pq = MinPQ()
        pq.insert(self._labels[start].key, start)

        max_transfers = self.config.max_transfers
        preferred = set(self.config.preferred_lines)

        while not pq.is_empty():
            key, state = pq.del_min()
            label = self._labels[state]
            if key != label.key:
                continue  # stale queue entry
            station, line = state
            for edge in self._graph.adj(station):
                nxt = edge.other(station)
                transfers = label.transfers
                if line is not None and edge.line != line:
                    transfers += 1
                if max_transfers is not None and transfers > max_transfers:
                    continue
                distance = label.distance + edge.distance
                penalty = 0.0
                if preferred and edge.line not in preferred:
                    penalty = LINE_PREFERENCE_PENALTY_METERS
                cost = label.cost + edge.distance + penalty
                new_key = self._make_key(cost, transfers)
                nstate: State = (nxt, edge.line)
                existing = self._labels.get(nstate)
                if existing is None or new_key < existing.key:
                    self._labels[nstate] = _Label(new_key, distance, cost, transfers, state, edge)
                    pq.insert(new_key, nstate)

        # Select the best state for every station reached.
        for state, label in self._labels.items():
            station = state[0]
            current = self._best_state.get(station)
            if current is None or label.key < self._labels[current].key:
                self._best_state[station] = state
        for station, state in self._best_state.items():
            self.dist_to[station] = self._labels[state].distance

    def has_path_to(self, target: int) -> bool:
        """Return True if ``target`` is reachable within the configured limits."""
        return target in self._best_state

    def path_with_edges(self, target: int) -> Tuple[List[int], List[Edge]]:
        """Return the station numbers and edges of the best path to ``target``.

        Returns ``([], [])`` if the target is unreachable.
        """
        state = self._best_state.get(target)
        if state is None:
            return [], []
        stations: List[int] = []
        edges: List[Edge] = []
        current: Optional[State] = state
        while current is not None:
            label = self._labels[current]
            stations.append(current[0])
            if label.prev is None:
                break
            edges.append(label.edge)
            current = label.prev
        stations.reverse()
        edges.reverse()
        return stations, edges

    def get_transfer_count(self, target: int) -> int:
        """Return the number of transfers on the best path to ``target``.

        Returns -1 if the target is unreachable.
        """
        state = self._best_state.get(target)
        if state is None:
            return -1
        return self._labels[state].transfers


@dataclass
class RouteResult:
    """Outcome of a successful route query.

    Attributes:
        station_numbers: Ordered station numbers from start to end.
        edges: Ordered edges traversed between consecutive stations.
        total_distance: Sum of edge distances in meters.
        transfer_count: Number of line changes along the route.
        lines: Unique line names in order of first use.
    """

    station_numbers: List[int]
    edges: List[Edge]
    total_distance: float
    transfer_count: int
    lines: List[str] = field(default_factory=list)


def _unique_lines(edges: Sequence[Edge]) -> List[str]:
    """Return line names in order of first appearance."""
    lines: List[str] = []
    for edge in edges:
        if edge.line not in lines:
            lines.append(edge.line)
    return lines


def compute_route(
    graph: Graph,
    source: int,
    target: int,
    config: Optional[PathfindingConfig] = None,
) -> Optional[RouteResult]:
    """Compute a route between two stations.

    Returns:
        A ``RouteResult``, or None if no route satisfies the configuration.

    Raises:
        ValueError: If the source or target station is not in the graph.
    """
    if not graph.has_node(target):
        raise ValueError(f"Target station {target} is not in the graph")
    search = ShortestPath(graph, source, config)
    if not search.has_path_to(target):
        return None
    numbers, edges = search.path_with_edges(target)
    total = sum(edge.distance for edge in edges)
    return RouteResult(
        station_numbers=numbers,
        edges=edges,
        total_distance=total,
        transfer_count=search.get_transfer_count(target),
        lines=_unique_lines(edges),
    )

"""Fare calculation based on the Beijing Subway fare structure.

Fare tiers (distance intervals are left-closed, right-open):
    [0, 6) km      -> 3 yuan
    [6, 12) km     -> 4 yuan
    [12, 22) km    -> 5 yuan
    [22, 32) km    -> 6 yuan
    [32, 52) km    -> 7 yuan
    [52, 72) km    -> 8 yuan   (+1 yuan per additional 20 km)

Fractional meters are rounded up (ceiling) before tier lookup.
"""

import math
from typing import List, Tuple

# (upper bound in meters, fare for distances below that bound)
TIER_TABLE: List[Tuple[int, int]] = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_START_METERS = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP_METERS = 20000

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def _validate_distance(distance) -> None:
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of meters")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("distance must be a finite number")
    if distance < 0:
        raise ValueError("distance must be non-negative")


def Sum_money(distance) -> int:
    """Return the subway fare in yuan for a travel distance in meters.

    Args:
        distance: Travelled distance in meters (fractions are rounded up).

    Returns:
        The fare in yuan as an integer.

    Raises:
        TypeError: If the distance is not a number.
        ValueError: If the distance is negative, NaN or infinite.
    """
    _validate_distance(distance)
    meters = math.ceil(distance)
    for upper_bound, fare in TIER_TABLE:
        if meters < upper_bound:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_START_METERS) // EXTENDED_STEP_METERS


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimate travel time in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers.

    Returns:
        Estimated time in minutes (about 1.5 min/km plus 5 min per transfer).
    """
    _validate_distance(distance)
    if transfer_count < 0:
        raise ValueError("transfer_count must be non-negative")
    return (distance / 1000.0) * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""Subway network modelling and data management.

``SubwayNetwork`` converts the JSON dataset into a ``Graph`` whose vertices are
station numbers, keeps the name/number mappings (``station_to_number``), and
offers data update and integrity-check operations.
"""

import re
from typing import Any, Callable, Dict, Iterable, List, Optional, Sequence, Set

from models import Edge, Graph

# Main lines that must be present in the dataset.
MAIN_LINES: List[str] = [
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
]

# Representative stations expected on some main lines (used for coverage checks).
REPRESENTATIVE_STATIONS: Dict[str, List[str]] = {
    "Line 1": ["Pingguoyuan", "Tiananmen East"],
    "Line 2": ["Xizhimen", "Dongzhimen"],
    "Line 10": ["Shuangjing"],
}

MIN_PLAUSIBLE_DISTANCE_M = 100
MAX_PLAUSIBLE_DISTANCE_M = 10000

_DIGITS = re.compile(r"\d+")


def _normalize(text: str) -> str:
    """Lower-case and remove all whitespace."""
    return re.sub(r"\s+", "", text).lower()


def _station_key(text: str) -> str:
    """Normalized key used for station-name standardization."""
    key = _normalize(text)
    if key.endswith("站"):
        key = key[:-1]
    return key


def _line_sort_key(line: str):
    """Sort 'Line 2' before 'Line 10', and named lines after numbered lines."""
    match = _DIGITS.search(line)
    if match:
        return (0, int(match.group()), line)
    return (1, 0, line)


def _is_number(value: Any) -> bool:
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def _count_components(nodes: Iterable[int], neighbors: Callable[[int], Iterable[int]]) -> int:
    """Count connected components among ``nodes`` using ``neighbors``."""
    visited: Set[int] = set()
    count = 0
    for start in nodes:
        if start in visited:
            continue
        count += 1
        visited.add(start)
        stack = [start]
        while stack:
            current = stack.pop()
            for nxt in neighbors(current):
                if nxt not in visited:
                    visited.add(nxt)
                    stack.append(nxt)
    return count


class SubwayNetwork:
    """In-memory model of the subway network.

    Attributes:
        graph: Weighted undirected ``Graph`` keyed by station numbers.
        station_to_number: Mapping from official station name to station number.
        number_to_station: Reverse mapping from station number to name.
    """

    def __init__(self, data: Optional[Dict[str, Any]] = None) -> None:
        self.graph = Graph()
        self.station_to_number: Dict[str, int] = {}
        self.number_to_station: Dict[int, str] = {}
        self._line_stations: Dict[str, Set[str]] = {}
        self._next_number = 0
        if data is not None:
            self._load(data)

    # ----- loading ---------------------------------------------------------
    def _load(self, data: Dict[str, Any]) -> None:
        """Populate the network from a parsed dataset."""
        if not isinstance(data, dict):
            raise ValueError("Subway data must be a JSON object")
        stations = data.get("stations")
        connections = data.get("connections")
        if not isinstance(stations, list) or not isinstance(connections, list):
            raise ValueError("Subway data must contain 'stations' and 'connections' lists")

        for entry in stations:
            if not isinstance(entry, dict):
                raise ValueError("Each station entry must be an object")
            name = entry.get("name")
            lines = entry.get("lines", [])
            if not isinstance(name, str) or not name.strip():
                raise ValueError("Every station needs a non-empty 'name'")
            if not isinstance(lines, list) or not all(isinstance(l, str) for l in lines):
                raise ValueError(f"Station '{name}' has an invalid 'lines' value")
            self.add_station(name, lines)

        for entry in connections:
            if not isinstance(entry, dict):
                raise ValueError("Each connection entry must be an object")
            try:
                self.add_connection(
                    entry["source"], entry["dest"], entry["distance"], entry["line"]
                )
            except KeyError as exc:
                raise ValueError(f"Connection entry is missing field {exc}") from exc

    # ----- internal helpers ------------------------------------------------
    def _require_number(self, name: str) -> int:
        """Return the station number for an exact official name."""
        if not isinstance(name, str) or name.strip() not in self.station_to_number:
            raise ValueError(f"Unknown station '{name}'")
        return self.station_to_number[name.strip()]

    def _index_line(self, line: str, *station_names: str) -> None:
        self._line_stations.setdefault(line, set()).update(station_names)

    def _rebuild_line_index(self) -> None:
        """Recompute the line -> stations index from graph data."""
        self._line_stations = {}
        for name in self.station_to_number:
            for line in self.graph.get_station_lines(name):
                self._index_line(line, name)
        for edge in self.graph.edges():
            self._index_line(
                edge.line,
                self.number_to_station[edge.source],
                self.number_to_station[edge.dest],
            )

    # ----- station and connection management -------------------------------
    def add_station(self, name: str, lines: Sequence[str] = ()) -> int:
        """Add a station (or merge lines into an existing one).

        Returns:
            The station number.
        """
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Station name must be a non-empty string")
        name = name.strip()
        clean_lines = list(dict.fromkeys(l.strip() for l in lines if l and l.strip()))

        if name in self.station_to_number:
            number = self.station_to_number[name]
            current = self.graph.get_station_lines(name)
            for line in clean_lines:
                if line not in current:
                    current.append(line)
            self.graph.set_station_lines(name, current)
        else:
            number = self._next_number
            self._next_number += 1
            self.graph.add_node(number)
            self.graph.set_station_lines(name, clean_lines)
            self.station_to_number[name] = number
            self.number_to_station[number] = name

        for line in clean_lines:
            self._index_line(line, name)
        return number

    def add_connection(self, source: str, dest: str, distance, line: str) -> bool:
        """Add a connection between two existing stations.

        Returns:
            True if added, False if an identical connection already exists.
        """
        u = self._require_number(source)
        v = self._require_number(dest)
        if u == v:
            raise ValueError("A connection cannot join a station to itself")
        if not _is_number(distance) or distance <= 0:
            raise ValueError(f"Invalid distance {distance!r} for connection")
        if not isinstance(line, str) or not line.strip():
            raise ValueError("Connection line must be a non-empty string")
        line = line.strip()
        if self.graph.find_edges(u, v, line):
            return False
        self.graph.add_edge(Edge(u, v, distance, line))
        self._index_line(line, self.number_to_station[u], self.number_to_station[v])
        return True

    def add_line(self, line_name: str, station_names: Sequence[str], distances: Sequence[float]) -> None:
        """Add a new line as an ordered sequence of stations.

        Args:
            line_name: Name of the new line (must not already exist).
            station_names: Ordered station names; unknown names are created.
            distances: Distances in meters between consecutive stations.

        Raises:
            ValueError: If the input is inconsistent.
        """
        if not isinstance(line_name, str) or not line_name.strip():
            raise ValueError("Line name must be a non-empty string")
        name = line_name.strip()
        if self.resolve_line(name) is not None:
            raise ValueError(f"Line '{name}' already exists")

        stations = [s.strip() if isinstance(s, str) else "" for s in station_names]
        if len(stations) < 2:
            raise ValueError("A line needs at least two stations")
        if any(not s for s in stations):
            raise ValueError("Station names must be non-empty")
        if len(set(stations)) != len(stations):
            raise ValueError("A line cannot contain duplicate stations")
        if len(distances) != len(stations) - 1:
            raise ValueError("Expected one distance per pair of consecutive stations")
        for distance in distances:
            if not _is_number(distance) or distance <= 0:
                raise ValueError(f"Invalid distance {distance!r}")

        for station in stations:
            self.add_station(station, [name])
        for index, distance in enumerate(distances):
            self.add_connection(stations[index], stations[index + 1], distance, name)

    def modify_station(self, name: str, new_name: Optional[str] = None,
                       lines: Optional[Sequence[str]] = None) -> str:
        """Rename a station and/or replace its list of lines.

        Returns:
            The final station name.
        """
        number = self._require_number(name)
        old = self.number_to_station[number]
        final = old
        if new_name is not None:
            candidate = new_name.strip() if isinstance(new_name, str) else ""
            if not candidate:
                raise ValueError("New station name must be a non-empty string")
            if candidate != old and candidate in self.station_to_number:
                raise ValueError(f"Station '{candidate}' already exists")
            final = candidate
        if lines is not None:
            if not all(isinstance(l, str) for l in lines):
                raise ValueError("Lines must be a list of strings")
            new_lines = list(dict.fromkeys(l.strip() for l in lines if l.strip()))
        else:
            new_lines = self.graph.get_station_lines(old)

        if final != old:
            del self.station_to_number[old]
            self.station_to_number[final] = number
            self.number_to_station[number] = final
            self.graph.station_lines.pop(old, None)
        self.graph.set_station_lines(final, new_lines)
        self._rebuild_line_index()
        return final

    def update_distance(self, source: str, dest: str, distance, line: Optional[str] = None) -> int:
        """Update the distance of the connection(s) between two stations.

        Returns:
            The number of connections updated.

        Raises:
            ValueError: If no connection exists, the distance is invalid, or the
                pair is connected by several lines and ``line`` was not given.
        """
        u = self._require_number(source)
        v = self._require_number(dest)
        if not _is_number(distance) or distance <= 0:
            raise ValueError(f"Invalid distance {distance!r}")
        edges = self.graph.find_edges(u, v, line.strip() if line else None)
        if not edges:
            raise ValueError(f"No connection between '{source}' and '{dest}'")
        if line is None and len({e.line for e in edges}) > 1:
            raise ValueError("Several lines connect these stations; specify the line")
        for edge in edges:
            edge.distance = distance
        return len(edges)

    def resolve_station(self, name: str) -> Optional[str]:
        """Standardize a user-entered station name to its official form.

        Matching ignores case, whitespace and a trailing '站'.

        Returns:
            The official name, or None if no unique match exists.
        """
        if not isinstance(name, str):
            return None
        stripped = name.strip()
        if stripped in self.station_to_number:
            return stripped
        key = _station_key(stripped)
        if not key:
            return None
        matches = [official for official in self.station_to_number if _station_key(official) == key]
        return matches[0] if len(matches) == 1 else None

    def resolve_line(self, name: str) -> Optional[str]:
        """Standardize a user-entered line name, or return None if unknown."""
        if not isinstance(name, str):
            return None
        stripped = name.strip()
        if stripped in self._line_stations:
            return stripped
        key = _normalize(stripped)
        matches = [line for line in self._line_stations if _normalize(line) == key]
        return matches[0] if len(matches) == 1 else None

    def has_station(self, name: str) -> bool:
        """Return True if the exact official name exists."""
        return isinstance(name, str) and name.strip() in self.station_to_number

    def station_number(self, name: str) -> int:
        """Return the station number for an official name."""
        return self._require_number(name)

    def station_name(self, number: int) -> str:
        """Return the official name for a station number."""
        return self.number_to_station[number]

    def station_names(self) -> List[str]:
        """Return all official station names, sorted."""
        return sorted(self.station_to_number)

    def line_names(self) -> List[str]:
        """Return all line names in natural order."""
        return sorted(self._line_stations, key=_line_sort_key)

    def line_station_names(self, line: str) -> List[str]:
        """Return the sorted names of stations served by a line."""
        return sorted(self._line_stations.get(line, set()))

    def lines_of_station(self, name: str) -> List[str]:
        """Return the lines serving a station."""
        return self.graph.get_station_lines(self._require_number_name(name))

    def is_transfer_station(self, name: str) -> bool:
        """Return True if a station is served by more than one line."""
        return len(self.lines_of_station(name)) > 1

    def _require_number_name(self, name: str) -> str:
        self._require_number(name)
        return name.strip()

    # ----- data integrity --------------------------------------------------
    def check_integrity(
        self,
        required_lines: Optional[Sequence[str]] = None,
        representative_stations: Optional[Dict[str, List[str]]] = None,
    ) -> List[str]:
        """Run data integrity checks.

        Checks station coverage of main lines and representative stations,
        station line assignments, connection consistency, distance accuracy and
        line/network connectivity.

        Returns:
            A list of human-readable issue descriptions (empty if healthy).
        """
        required = MAIN_LINES if required_lines is None else list(required_lines)
        representative = (
            REPRESENTATIVE_STATIONS if representative_stations is None else representative_stations
        )
        issues: List[str] = []

        # Coverage of main lines.
        known_lines = set(self._line_stations)
        for line in required:
            if line not in known_lines:
                issues.append(f"Coverage: main line '{line}' is missing")
            elif len(self._line_stations[line]) < 2:
                issues.append(f"Coverage: main line '{line}' has fewer than 2 stations")

        # Representative stations.
        for line, stations in representative.items():
            for station in stations:
                if station not in self.station_to_number:
                    issues.append(f"Coverage: representative station '{station}' of {line} is missing")
                elif line not in self.graph.get_station_lines(station):
                    issues.append(f"Coverage: station '{station}' is not listed on {line}")

        # Station data.
        for name in sorted(self.station_to_number):
            if not self.graph.get_station_lines(name):
                issues.append(f"Station data: '{name}' has no line assignment")

        # Connection integrity and distance accuracy.
        for edge in self.graph.edges():
            a = self.number_to_station[edge.source]
            b = self.number_to_station[edge.dest]
            label = f"{a} - {b} ({edge.line})"
            for end in (a, b):
                if edge.line not in self.graph.get_station_lines(end):
                    issues.append(f"Connection: '{end}' is not listed on the line of {label}")
            if not _is_number(edge.distance) or edge.distance <= 0:
                issues.append(f"Distance: {label} has invalid distance {edge.distance!r}")
            elif not MIN_PLAUSIBLE_DISTANCE_M <= edge.distance <= MAX_PLAUSIBLE_DISTANCE_M:
                issues.append(
                    f"Distance: {label} distance {edge.distance} m is outside the plausible range"
                )

        # Per-line connectivity.
        for line in self.line_names():
            numbers = [self.station_to_number[s] for s in self.line_station_names(line)]
            count = _count_components(
                numbers,
                lambda n, ln=line: [e.other(n) for e in self.graph.adj(n) if e.line == ln],
            )
            if count > 1:
                issues.append(f"Connectivity: {line} is split into {count} disconnected segments")

        # Whole-network connectivity.
        all_nodes = self.graph.nodes()
        if all_nodes:
            count = _count_components(all_nodes, lambda n: [e.other(n) for e in self.graph.adj(n)])
            if count > 1:
                issues.append(f"Connectivity: network is split into {count} disconnected components")

        return issues

    def to_data(self) -> Dict[str, Any]:
        """Export the current network in the dataset JSON structure."""
        stations = [
            {"name": name, "lines": self.graph.get_station_lines(name)}
            for name in sorted(self.station_to_number)
        ]
        connections = [
            {
                "source": self.number_to_station[edge.source],
                "dest": self.number_to_station[edge.dest],
                "distance": edge.distance,
                "line": edge.line,
            }
            for edge in self.graph.edges()
        ]
        return {"stations": stations, "connections": connections}

"""Path visualization and report export.

Builds console text reports and structured JSON reports from a ``RouteResult``.
"""

import os
from typing import Any, Dict, List, Sequence

from fare_calculator import calculate_estimated_time
from models import Edge
from pathfinding import RouteResult


def format_distance(meters: float) -> str:
    """Format a distance showing meters for short and kilometers for long trips.

    Examples: ``850 m (0.85 km)`` and ``12.35 km (12350 m)``.
    """
    if meters < 1000:
        return f"{meters:.0f} m ({meters / 1000:.2f} km)"
    return f"{meters / 1000:.2f} km ({meters:.0f} m)"


def build_segments(station_names: Sequence[str], edges: Sequence[Edge]) -> List[Dict[str, Any]]:
    """Group consecutive stations travelled on the same line into segments."""
    if not edges:
        return []
    segments: List[Dict[str, Any]] = []
    current: Dict[str, Any] = {"line": edges[0].line, "stations": [station_names[0]]}
    for index, edge in enumerate(edges):
        if edge.line != current["line"]:
            segments.append(current)
            current = {"line": edge.line, "stations": [station_names[index]]}
        current["stations"].append(station_names[index + 1])
    segments.append(current)
    for segment in segments:
        segment["start"] = segment["stations"][0]
        segment["end"] = segment["stations"][-1]
        segment["stops"] = len(segment["stations"]) - 1
    return segments


def find_transfers(station_names: Sequence[str], edges: Sequence[Edge]) -> List[Dict[str, str]]:
    """Identify transfer stations where the travelled line changes."""
    transfers: List[Dict[str, str]] = []
    for index in range(1, len(edges)):
        if edges[index - 1].line != edges[index].line:
            transfers.append(
                {
                    "station": station_names[index],
                    "from_line": edges[index - 1].line,
                    "to_line": edges[index].line,
                }
            )
    return transfers


def summarize_route(route: RouteResult, fare: int) -> Dict[str, Any]:
    """Extract key route information (distance, fare, transfers, time)."""
    distance = route.total_distance
    return {
        "distance_meters": distance,
        "distance_kilometers": round(distance / 1000.0, 2),
        "fare": fare,
        "transfer_count": route.transfer_count,
        "estimated_time_minutes": round(
            calculate_estimated_time(distance, route.transfer_count), 1
        ),
        "station_count": len(route.station_numbers),
        "lines": list(route.lines),
    }


def render_text_report(query: Dict[str, Any], route: RouteResult,
                       station_names: Sequence[str], fare: int) -> str:
    """Render a human-readable console report."""
    summary = summarize_route(route, fare)
    max_transfers = query.get("max_transfers")
    preference = query.get("line_preference")
    lines: List[str] = [
        "=" * 56,
        "Beijing Subway Fare Query Result",
        "=" * 56,
        f"Starting station : {query.get('start')}",
        f"Ending station   : {query.get('end')}",
        f"Query mode       : {query.get('mode')}",
        f"Max transfers    : {max_transfers if max_transfers is not None else 'unlimited'}",
        f"Line preference  : {preference if preference else 'none'}",
        "",
        f"Station sequence ({len(station_names)} stations):",
    ]
    for index, name in enumerate(station_names, start=1):
        lines.append(f"  {index:>3}. {name}")

    lines.append("")
    lines.append("Line segments:")
    segments = build_segments(station_names, route.edges)
    if segments:
        for segment in segments:
            lines.append(
                f"  [{segment['line']}] {segment['start']} -> {segment['end']} "
                f"({segment['stops']} stops)"
            )
    else:
        lines.append("  (no travel required)")

    lines.append("")
    lines.append("Transfers:")
    transfers = find_transfers(station_names, route.edges)
    if transfers:
        for transfer in transfers:
            lines.append(
                f"  - {transfer['station']}: {transfer['from_line']} -> {transfer['to_line']}"
            )
    else:
        lines.append("  No transfer required")

    lines.append("")
    lines.append(f"Lines involved   : {', '.join(route.lines) if route.lines else 'none'}")
    lines.append(f"Total distance   : {format_distance(route.total_distance)}")
    lines.append(f"Transfer count   : {route.transfer_count}")
    lines.append(f"Estimated time   : {summary['estimated_time_minutes']} minutes")
    lines.append(f"Fare: {fare} yuan")
    return "\n".join(lines)


def build_json_report(query: Dict[str, Any], route: RouteResult,
                      station_names: Sequence[str], fare: int) -> Dict[str, Any]:
    """Build a structured report suitable for programmatic integration."""
    summary = summarize_route(route, fare)
    return {
        "query": query,
        "path": list(station_names),
        "distance": {
            "meters": summary["distance_meters"],
            "kilometers": summary["distance_kilometers"],
        },
        "fare": fare,
        "estimated_time_minutes": summary["estimated_time_minutes"],
        "transfer_count": route.transfer_count,
        "transfers": find_transfers(station_names, route.edges),
        "lines": list(route.lines),
        "segments": build_segments(station_names, route.edges),
    }


def export_report(content: str, path: str) -> None:
    """Write report content to a file (UTF-8), creating parent folders if needed.

    Raises:
        OSError: If the file cannot be written.
    """
    directory = os.path.dirname(os.path.abspath(path))
    os.makedirs(directory, exist_ok=True)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(content)
        handle.write("\n")

"""Command-line interface for the Beijing Subway fare calculator.

Runs an interactive loop that accepts "Starting Station; Ending Station"
queries, validates them, computes the route and fare, and prints or exports
the result. Supports one-shot options such as ``--check_data``.
"""

import argparse
import json
from typing import Optional, Tuple

from data_loader import DATA_FILE_PATH, load_subway_data
from fare_calculator import Sum_money
from pathfinding import (
    MODE_COMPREHENSIVE,
    MODE_LEAST_TRANSFER,
    MODE_SHORTEST_DISTANCE,
    PathfindingConfig,
    compute_route,
)
from subway_network import SubwayNetwork
from visualizer import build_json_report, export_report, render_text_report

EXIT_COMMANDS = ("quit", "exit", "q")
INPUT_FORMAT_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"
PROMPT = "Enter query 'Starting Station; Ending Station' (or 'quit' to exit): "


class QueryError(Exception):
    """Raised when a user query cannot be parsed."""


def parse_query(raw: str) -> Tuple[str, str]:
    """Split a raw query on the semicolon separator.

    Raises:
        QueryError: If the format is not 'Starting Station; Ending Station'.
    """
    parts = raw.split(";")
    if len(parts) != 2:
        raise QueryError(INPUT_FORMAT_MESSAGE)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryError(INPUT_FORMAT_MESSAGE)
    return start, end


def build_parser() -> argparse.ArgumentParser:
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(
        prog="beijing_subway_fare",
        description="Beijing Subway fare calculation system.",
    )
    parser.add_argument("--output_file", default=None,
                        help="Export query results to this file (overwritten each query).")
    parser.add_argument("--format", dest="output_format", choices=["text", "json"],
                        default="text", help="Output format: text or json (default: text).")
    parser.add_argument("--mode", choices=[MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE],
                        default=MODE_SHORTEST_DISTANCE,
                        help="Path query mode (default: shortest_distance).")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="Maximum number of allowed transfers.")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line name, e.g. 'Line 1'.")
    parser.add_argument("--check_data", action="store_true",
                        help="Run the data integrity check and exit.")
    return parser


def load_network() -> SubwayNetwork:
    """Load the subway dataset from the default data file."""
    return SubwayNetwork(load_subway_data(DATA_FILE_PATH))


def run_integrity_check(network: SubwayNetwork) -> int:
    """Print the integrity report. Returns 0 if healthy, 1 otherwise."""
    issues = network.check_integrity()
    print(f"Stations: {len(network.station_to_number)}, "
          f"connections: {network.graph.num_edges}, lines: {len(network.line_names())}")
    if not issues:
        print("Data integrity check passed: no issues found.")
        return 0
    print(f"Data integrity check found {len(issues)} issue(s):")
    for issue in issues:
        print(f"  - {issue}")
    return 1


def process_query(
    network: SubwayNetwork,
    raw: str,
    config: PathfindingConfig,
    output_format: str = "text",
    output_file: Optional[str] = None,
) -> bool:
    """Handle one query end-to-end. Returns True on success."""
    try:
        start_raw, end_raw = parse_query(raw)
    except QueryError as exc:
        print(exc)
        return False

    start = network.resolve_station(start_raw)
    if start is None:
        print(f"Starting station '{start_raw}' does not exist, please check the station name")
        return False
    end = network.resolve_station(end_raw)
    if end is None:
        print(f"Ending station '{end_raw}' does not exist, please check the station name")
        return False
    if start == end:
        print("Starting station cannot be the same as the ending station")
        return False

    try:
        route = compute_route(
            network.graph,
            network.station_number(start),
            network.station_number(end),
            config,
        )
    except Exception as exc:  # defensive: any path computation failure
        print(f"Path calculation failed: {exc}")
        return False
    if route is None:
        print(f"No route found from '{start}' to '{end}' under the current settings "
              f"(check --max_transfers and --mode).")
        return False

    try:
        fare = Sum_money(route.total_distance)
    except (TypeError, ValueError) as exc:
        print(f"Fare calculation failed: {exc}")
        return False

    names = [network.station_name(n) for n in route.station_numbers]
    query = {
        "start": start,
        "end": end,
        "mode": config.mode,
        "max_transfers": config.max_transfers,
        "line_preference": config.preferred_lines[0] if config.preferred_lines else None,
    }

    if output_format == "json":
        content = json.dumps(build_json_report(query, route, names, fare),
                             ensure_ascii=False, indent=2)
    else:
        content = render_text_report(query, route, names, fare)

    print(content)

    if output_file:
        try:
            export_report(content, output_file)
            print(f"Result exported to {output_file}")
        except OSError as exc:
            print(f"Failed to write output file: {exc}")
            return False
    return True


def main(argv=None) -> int:
    """Program entry point. Returns the process exit code."""
    parser = build_parser()
    args = parser.parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be non-negative")

    try:
        network = load_network()
    except (OSError, ValueError, TypeError) as exc:
        print(f"Failed to load subway data: {exc}")
        return 1

    if args.check_data:
        return run_integrity_check(network)

    preferred = []
    if args.line_preference:
        canonical = network.resolve_line(args.line_preference)
        if canonical is None:
            print(f"Line preference '{args.line_preference}' does not exist, "
                  f"available lines: {', '.join(network.line_names())}")
            return 2
        preferred = [canonical]

    try:
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=preferred,
        )
    except ValueError as exc:
        print(f"Invalid configuration: {exc}")
        return 2

    print("Beijing Subway Fare Calculator")
    print("Enter queries as 'Starting Station; Ending Station'. Type 'quit' to exit.")

    while True:
        try:
            raw = input(PROMPT)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        try:
            process_query(network, text, config, args.output_format, args.output_file)
        except Exception as exc:  # keep the session alive on unexpected errors
            print(f"Unexpected error while processing query: {exc}")
        print()

    return 0

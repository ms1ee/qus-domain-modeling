"""Process-wide key/value storage shared by the dispatch simulator modules.

The store holds the global system variables described in the PRD, such as
the system funds, the rider list (key ``"riders"``), the rider ID counter,
the running flag, the simulated tick and the order history.

_global_dict = {}


def _init():
    """Reset the global variable store to an empty dictionary.

    Called at import time and by ``system.init_system`` to start from a
    clean state.
    """
    global _global_dict
    _global_dict = {}


def set_value(key, value):
    """Store ``value`` under ``key``.

    Args:
        key: Hashable identifier of the global variable.
        value: Any Python object to store.
    """
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key``.

    Args:
        key: Identifier of the global variable.
        default: Value returned when ``key`` has not been set.

    Returns:
        The stored value, or ``default`` if the key does not exist.

    return _global_dict.get(key, default)

"""Grid-based path planning for the food delivery simulator.

The city is an 800x800 unit map divided into 8x8 cells of 100x100 units.
Grid lines (coordinates that are multiples of 100) are streets. Restaurant
and customer locations sit on integer-hundred points, which are street
intersections. Riders always move along streets: horizontally or vertically,
turning at right angles only at intersections.

Route convention (shared by movement and route drawing so they never
disagree): when moving from a point ``a`` to a point ``b``, the rider goes
horizontally first if ``a`` lies on a horizontal street (``a.y % 100 == 0``),
otherwise vertically first.
"""

import logging
import math

logger = logging.getLogger(__name__)

CELL_SIZE = 100
MAP_SIZE = 800


def _step_towards(current, target, speed):
    """Move ``current`` toward ``target`` by at most ``speed`` units.

    Args:
        current: Current coordinate value.
        target: Target coordinate value.
        speed: Maximum step size (positive integer).

    Returns:
        The new coordinate value, never overshooting ``target``.
    """
    delta = target - current
    if abs(delta) <= speed:
        return target
    return current + speed if delta > 0 else current - speed


class PathPlanner:
    """Pathfinding logic for riders on the street grid.

    All methods are stateless; the class groups the grid-based algorithm,
    the special-case handling and the multi-destination route construction.
    """

    @staticmethod
    def snap_point(x, y):
        """Convert raw coordinates to the top-left corner of their grid cell.

        Each coordinate ``v`` becomes ``(v // 100) * 100`` and is clamped to
        the map range [0, 800]. Values outside the map are clamped, so the
        result is always a valid integer-hundred point.

        Args:
            x: Raw x coordinate.
            y: Raw y coordinate.

        Returns:
            Tuple ``(snapped_x, snapped_y)`` of integers.
        """
        def snap(value):
            cell = int(value) // CELL_SIZE * CELL_SIZE
            return max(0, min(MAP_SIZE, cell))

        return snap(x), snap(y)

    @staticmethod
    def manhattan(a, b):
        """Return the Manhattan (street) distance between two points.

        Args:
            a: Tuple ``(x, y)``.
            b: Tuple ``(x, y)``.

        Returns:
            Integer distance ``|ax - bx| + |ay - by|``.
        """
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def euclidean(a, b):
        """Return the straight-line distance ``sqrt(dx^2 + dy^2)``.

        Args:
            a: Tuple ``(x, y)``.
            b: Tuple ``(x, y)``.

        Returns:
            Float Euclidean distance.
        """
        return math.hypot(a[0] - b[0], a[1] - b[1])

    @staticmethod
    def is_on_street(point):
        """Return True if the point lies on a street (grid line).

        Args:
            point: Tuple ``(x, y)``.

        Returns:
            True when ``x`` or ``y`` is a multiple of the cell size.
        """
        return point[0] % CELL_SIZE == 0 or point[1] % CELL_SIZE == 0

    @staticmethod
    def corner_point(start, end):
        """Return the right-angle corner used when travelling ``start`` to ``end``.

        Args:
            start: Tuple ``(x, y)`` where the leg begins.
            end: Tuple ``(x, y)`` where the leg finishes.

        Returns:
            Tuple ``(x, y)`` of the turning point. It equals ``start`` or
            ``end`` when the two points are already aligned.
        """
        if start[1] % CELL_SIZE == 0:
            return end[0], start[1]
        return start[0], end[1]

    @staticmethod
    def next_position(x, y, target_x, target_y, speed):
        """Advance a rider one tick toward a target along the streets.

        Special cases handled:
          * Target directly above/below (same x): move vertically only.
          * Target directly left/right (same y): move horizontally only.
          * General case: follow the route convention (see module docstring).

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            target_x: Target x coordinate.
            target_y: Target y coordinate.
            speed: Movement budget for this tick (units/tick, positive).

        Returns:
            Tuple ``(new_x, new_y, arrived)`` where ``arrived`` is True when
            the rider stands exactly on the target after this step.

        Raises:
            ValueError: If ``speed`` is not positive.
        """
        if speed <= 0:
            raise ValueError("speed must be a positive integer")
        if x == target_x and y == target_y:
            return x, y, True

        if y == target_y:
            new_x, new_y = _step_towards(x, target_x, speed), y
        elif x == target_x:
            new_x, new_y = x, _step_towards(y, target_y, speed)
        elif y % CELL_SIZE == 0:
            new_x, new_y = _step_towards(x, target_x, speed), y
        else:
            new_x, new_y = x, _step_towards(y, target_y, speed)

        arrived = new_x == target_x and new_y == target_y
        return new_x, new_y, arrived

    @classmethod
    def build_route(cls, start, stops):
        """Build the ordered list of route points through several stops.

        Used for multi-destination paths, e.g. start -> restaurant -> customer.

        Args:
            start: Tuple ``(x, y)`` where the rider currently is.
            stops: Iterable of ``(x, y)`` destinations, visited in order.

        Returns:
            List of ``(x, y)`` points starting with ``start``, including the
            right-angle corners between consecutive points.
        """
        current = tuple(start)
        points = [current]
        for stop in stops:
            stop = tuple(stop)
            corner = cls.corner_point(current, stop)
            if corner != current and corner != stop:
                points.append(corner)
            if stop != points[-1]:
                points.append(stop)
            current = stop
        return points

    @classmethod
    def route_length(cls, points):
        """Return the total street length of a route.

        Args:
            points: List of ``(x, y)`` route points.

        Returns:
            Integer total Manhattan length of consecutive segments.
        """
        return sum(cls.manhattan(a, b) for a, b in zip(points, points[1:]))

    @classmethod
    def estimate_ticks(cls, points, speed):
        """Estimate the number of ticks needed to travel a route.

        Args:
            points: List of ``(x, y)`` route points.
            speed: Rider speed in units per tick (positive).

        Returns:
            Integer number of ticks (rounded up).

        Raises:
            ValueError: If ``speed`` is not positive.
        """
        if speed <= 0:
            raise ValueError("speed must be a positive integer")
        return int(math.ceil(cls.route_length(points) / speed))

"""Order data model.

An order carries a restaurant location and a customer location, both snapped
to integer-hundred street intersections. Its lifecycle is:

    pending -> assigned -> picking_up -> delivering -> completed
"""

import logging

from path_planner import PathPlanner

logger = logging.getLogger(__name__)

ORDER_PENDING = "pending"
ORDER_ASSIGNED = "assigned"
ORDER_PICKING_UP = "picking_up"
ORDER_DELIVERING = "delivering"
ORDER_COMPLETED = "completed"
ACTIVE_ORDER_STATUSES = (ORDER_ASSIGNED, ORDER_PICKING_UP, ORDER_DELIVERING)


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique positive integer identifier.
        res_pos: Restaurant grid point ``(x, y)`` (snapped).
        cus_pos: Customer grid point ``(x, y)`` (snapped).
        status: One of the ``ORDER_*`` status strings.
        rider_id: ID of the assigned rider, or None while pending.
        created_tick: Simulation tick when the order was created.
        assigned_tick: Tick when the order was assigned to a rider.
        pickup_tick: Tick when the rider reached the restaurant.
        completed_tick: Tick when the order was delivered.
        distance: Street distance from restaurant to customer.
        time_taken: Ticks from creation to completion (None if unfinished).
        efficiency: Distance covered per tick (None if unfinished).
    """

    _counter = 0

    def __init__(self, res_pos, cus_pos, created_tick=0):
        """Create a pending order.

        Args:
            res_pos: Restaurant coordinates ``(x, y)``, range 0-800.
            cus_pos: Customer coordinates ``(x, y)``, range 0-800.
            created_tick: Simulation tick of creation.
        """
        Order._counter += 1
        self.id = Order._counter
        self.res_pos = PathPlanner.snap_point(res_pos[0], res_pos[1])
        self.cus_pos = PathPlanner.snap_point(cus_pos[0], cus_pos[1])
        self.status = ORDER_PENDING
        self.rider_id = None
        self.created_tick = int(created_tick)
        self.assigned_tick = None
        self.pickup_tick = None
        self.completed_tick = None
        self.distance = PathPlanner.manhattan(self.res_pos, self.cus_pos)
        self.time_taken = None
        self.efficiency = None

    def assign_to(self, rider_id, tick=None):
        """Assign the order to a rider by ID.

        The caller is responsible for appending the order to ``rider.orders``.

        Args:
            rider_id: ID of the rider receiving the order.
            tick: Optional simulation tick of the assignment.

        Raises:
            ValueError: If the order is not pending.
        """
        if self.status != ORDER_PENDING:
            raise ValueError(
                f"Order {self.id} cannot be assigned from status '{self.status}'."
            )
        self.rider_id = rider_id
        self.status = ORDER_ASSIGNED
        self.assigned_tick = tick
        logger.info("Order %d assigned to rider %s", self.id, rider_id)

    def accept(self):
        """Move the order from assigned to picking up (rider accepted it).

        Raises:
            ValueError: If the order is not in the assigned state.
        """
        if self.status != ORDER_ASSIGNED:
            raise ValueError(f"Order {self.id} cannot be accepted from '{self.status}'.")
        self.status = ORDER_PICKING_UP

    def mark_picked_up(self, tick):
        """Record arrival at the restaurant and start delivery.

        Args:
            tick: Simulation tick of the arrival.

        Raises:
            ValueError: If the order is not in the picking up state.
        """
        if self.status != ORDER_PICKING_UP:
            raise ValueError(f"Order {self.id} is not being picked up.")
        self.status = ORDER_DELIVERING
        self.pickup_tick = tick

    def complete(self, tick):
        """Mark the order as delivered and record efficiency data.

        Args:
            tick: Simulation tick of delivery.

        Raises:
            ValueError: If the order is not being delivered.
        """
        if self.status != ORDER_DELIVERING:
            raise ValueError(f"Order {self.id} is not being delivered.")
        self.status = ORDER_COMPLETED
        self.completed_tick = tick
        self.time_taken = tick - self.created_tick
        self.efficiency = (
            self.distance / self.time_taken if self.time_taken > 0 else 0.0
        )
        logger.info("Order %d completed at tick %d", self.id, tick)

    def to_dict(self):
        """Serialize the order into a JSON-compatible dictionary.

        Returns:
            Dictionary with all order attributes.
        """
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_tick": self.created_tick,
            "assigned_tick": self.assigned_tick,
            "pickup_tick": self.pickup_tick,
            "completed_tick": self.completed_tick,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from a dictionary produced by ``to_dict``.

        Args:
            data: Dictionary with order attributes.

        Returns:
            A restored ``Order`` instance with its original ID.
        """
        order = cls(data["res_pos"], data["cus_pos"], data.get("created_tick", 0))
        order.id = int(data["id"])
        Order._counter = max(Order._counter, order.id)
        order.status = data.get("status", ORDER_PENDING)
        order.rider_id = data.get("rider_id")
        order.assigned_tick = data.get("assigned_tick")
        order.pickup_tick = data.get("pickup_tick")
        order.completed_tick = data.get("completed_tick")
        order.time_taken = data.get("time_taken")
        order.efficiency = data.get("efficiency")
        return order

    def __repr__(self):
        return (
            f"Order(id={self.id}, status={self.status!r}, "
            f"res={self.res_pos}, cus={self.cus_pos})"
        )

"""Rider model: state machine, movement and per-rider event log.

State machine (rider.state):
    0 Idle        -> 1 Picking Up  : an assigned order is accepted (move()).
    1 Picking Up  -> 2 Delivering  : the rider reaches the restaurant.
    2 Delivering  -> 1 Picking Up  : an order is delivered and another is queued.
    2 Delivering  -> 0 Idle        : an order is delivered and none remain.
"""

import logging

from globalvar import get_value
from order_model import (
    ACTIVE_ORDER_STATUSES,
    ORDER_ASSIGNED,
    ORDER_COMPLETED,
    ORDER_DELIVERING,
    ORDER_PICKING_UP,
)
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

RIDER_SPEED = 20
SPAWN_POSITION = (400, 400)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "Idle",
    STATE_PICKING_UP: "Picking Up",
    STATE_DELIVERING: "Delivering",
}


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id: Unique integer identifier.
        x: Current x coordinate (0-800).
        y: Current y coordinate (0-800).
        speed: Movement speed in units per tick (fixed at 20).
        state: 0 idle, 1 picking up, 2 delivering.
        orders: All orders ever assigned to this rider (history kept).
        special_case: True while the rider is directly above or below its target.
        log: List of event dictionaries with keys "event", "order_id", "tick",
            "rider_id", "x", "y" and "state".
    """

    def __init__(self, rider_id, x=SPAWN_POSITION[0], y=SPAWN_POSITION[1]):
        """Create a rider and record its spawn event.

        Args:
            rider_id: Unique integer identifier.
            x: Initial x coordinate (default 400).
            y: Initial y coordinate (default 400).
        """
        self.rider_id = int(rider_id)
        self.x = int(x)
        self.y = int(y)
        self.speed = RIDER_SPEED
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    @property
    def state_name(self):
        """Human-readable name of the current state."""
        return STATE_NAMES.get(self.state, "Unknown")

    def position(self):
        """Return the current position as a tuple ``(x, y)``."""
        return self.x, self.y

    def active_orders(self):
        """Return orders that are assigned or in progress (not completed)."""
        return [o for o in self.orders if o.status in ACTIVE_ORDER_STATUSES]

    def is_idle(self):
        """Return True when the rider has no active orders."""
        return not self.active_orders()

    def current_order(self):
        """Return the order the rider is currently serving, or None.

        Orders are served first-in, first-out.
        """
        for order in self.orders:
            if order.status in (ORDER_PICKING_UP, ORDER_DELIVERING):
                return order
        return None

    def _now(self):
        """Return the current simulation tick from the global store."""
        return get_value("tick", 0) or 0

    def _record(self, event, order_id):
        """Append an event entry to the rider log."""
        self.log.append({
            "event": event,
            "order_id": order_id,
            "tick": self._now(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def _accept_new_orders(self):
        """Accept every newly assigned order and log the acceptance."""
        for order in self.orders:
            if order.status == ORDER_ASSIGNED:
                order.accept()
                self._record("accepted_order", order.id)

    def _sync_state(self):
        """Derive the rider state from the current order."""
        current = self.current_order()
        if current is None:
            self.state = STATE_IDLE
        elif current.status == ORDER_PICKING_UP:
            self.state = STATE_PICKING_UP
        else:
            self.state = STATE_DELIVERING

    def move(self):
        """Advance the rider by one tick.

        Accepts newly assigned orders, moves toward the current target
        (restaurant when picking up, customer when delivering) and performs
        the arrival state transitions with their log entries.

        Returns:
            True if the rider had a target and moved this tick, else False.
        """
        self._accept_new_orders()
        self._sync_state()

        current = self.current_order()
        if current is None:
            self.special_case = False
            return False

        if current.status == ORDER_PICKING_UP:
            target = current.res_pos
        else:
            target = current.cus_pos

        new_x, new_y, arrived = PathPlanner.next_position(
            self.x, self.y, target[0], target[1], self.speed
        )
        self.x, self.y = new_x, new_y
        self.special_case = (new_x == target[0] and new_y != target[1])

        if arrived:
            if current.status == ORDER_PICKING_UP:
                current.mark_picked_up(self._now())
                self._sync_state()
                self._record("arrived_at_merchant", current.id)
            else:
                current.complete(self._now())
                self._sync_state()
                self._record("delivered_to_customer", current.id)
        return True

    def to_dict(self):
        """Serialize the rider into a JSON-compatible dictionary.

        Returns:
            Dictionary with rider attributes, order IDs and the event log.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [o.id for o in self.orders],
            "log": list(self.log),
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        """Rebuild a rider from ``to_dict`` output.

        Args:
            data: Dictionary produced by ``to_dict``.
            orders_by_id: Mapping of order ID to restored ``Order`` objects.

        Returns:
            A restored ``Rider`` instance.
        """
        rider = cls(int(data["rider_id"]), int(data["x"]), int(data["y"]))
        rider.state = int(data.get("state", STATE_IDLE))
        rider.special_case = bool(data.get("special_case", False))
        rider.orders = [
            orders_by_id[oid] for oid in data.get("order_ids", []) if oid in orders_by_id
        ]
        rider.log = list(data.get("log", []))
        return rider

    def __repr__(self):
        return (
            f"Rider(id={self.rider_id}, pos=({self.x}, {self.y}), "
            f"state={self.state_name!r}, orders={len(self.orders)})"
        )

"""Order creation, rider assignment and order bookkeeping."""

import logging

from globalvar import get_value, set_value
from order_model import ORDER_PENDING, Order
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

COORD_MIN = 0
COORD_MAX = 1000
COORD_LABELS = ("restaurant_x", "restaurant_y", "customer_x", "customer_y")


def _riders():
    """Return the global rider list, creating it if missing."""
    riders = get_value("riders")
    if riders is None:
        riders = []
        set_value("riders", riders)
    return riders


def _orders():
    """Return the global order history list, creating it if missing."""
    orders = get_value("orders")
    if orders is None:
        orders = []
        set_value("orders", orders)
    return orders


def validate_order_coordinates(values):
    """Validate the four order coordinates.

    Args:
        values: Sequence ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

    Returns:
        A list of four validated integers.

    Raises:
        ValueError: If the count is wrong, a value is not an integer, or a
            value lies outside 0-1000.
    """
    values = list(values)
    if len(values) != 4:
        raise ValueError(f"Expected 4 coordinates, got {len(values)}.")
    result = []
    for label, value in zip(COORD_LABELS, values):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} must be an integer.")
        if not COORD_MIN <= value <= COORD_MAX:
            raise ValueError(
                f"{label} must be between {COORD_MIN} and {COORD_MAX}, got {value}."
            )
        result.append(value)
    return result


def find_best_rider_for_order(order):
    """Select the best rider for an order using a distance-first strategy.

    The rider closest to the restaurant (Euclidean distance
    ``sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2)``) is
    chosen. Ties are broken by the rider's current number of active orders,
    then by rider ID.

    Args:
        order: The ``Order`` to assign.

    Returns:
        The selected ``Rider``, or None if no riders exist.
    """
    riders = _riders()
    if not riders:
        return None
    best = None
    best_key = None
    for rider in riders:
        distance = PathPlanner.euclidean(rider.position(), order.res_pos)
        key = (distance, len(rider.active_orders()), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    logger.debug("Order %d -> rider %s (distance %.1f)",
                 order.id, best.rider_id, best_key[0])
    return best


class OrderManager:
    """Creates, assigns and queries orders."""

    def create_order(self, restaurant, customer, tick=0):
        """Validate coordinates and create a pending order.

        Args:
            restaurant: ``(x, y)`` restaurant coordinates, range 0-1000.
            customer: ``(x, y)`` customer coordinates, range 0-1000.
            tick: Current simulation tick.

        Returns:
            The newly created ``Order`` (also stored in the order history).

        Raises:
            ValueError: If any coordinate is invalid.
        """
        res = validate_order_coordinates(list(restaurant) + list(customer))
        order = Order((res[0], res[1]), (res[2], res[3]), created_tick=tick)
        _orders().append(order)
        logger.info("Order %d created", order.id)
        return order

    def assign(self, order, rider, tick=0):
        """Assign a pending order to a rider.

        Args:
            order: A pending ``Order``.
            rider: The ``Rider`` receiving the order.
            tick: Current simulation tick.

        Raises:
            ValueError: If the order is not pending.
        """
        if order.status != ORDER_PENDING:
            raise ValueError(f"Order {order.id} is not pending.")
        order.assign_to(rider.rider_id, tick)
        rider.orders.append(order)

    def pending_orders(self):
        """Return all orders still waiting for a rider, oldest first."""
        return [o for o in _orders() if o.status == ORDER_PENDING]

    def all_orders(self):
        """Return a copy of the full order history."""
        return list(_orders())

    def completed_orders(self):
        """Return all completed orders."""
        return [o for o in _orders() if o.status == "completed"]

import logging
import os

# Configurable parameters (may be overridden through the JSON config file).
PARAMS = {
    "initial_funds": 1000,

def init_system():
    """Initialize the global variables of the dispatch system.

    Sets system funds, an empty rider list, the rider ID counter, the
    running state, the simulated tick and an empty order history.
    """
    globalvar._init()
    set_value("funds", PARAMS["initial_funds"])
    set_value("riders", [])

        return sum(1 for r in self.riders if r.is_idle())

            rider = find_best_rider_for_order(order)

"""Status display, order/rider reports and ASCII map rendering.

Map layout: the 800x800 map is drawn on a 17x17 character canvas. Each
character covers 50 units, so grid nodes (multiples of 100) fall on even
rows and columns. Symbols:
    '#'  street (grid line)          '.'  empty cell interior
    'S'  restaurant (active order)   'C'  customer (active order)
    'R'  rider                       '*'  planned route
"""

from order_model import ORDER_COMPLETED
from rider import STATE_NAMES

CANVAS_SIZE = 17
CELL_UNITS = 50
MAX_CELL_INDEX = CANVAS_SIZE - 1


def _segment_points(start, end):
    """Return sample points along an axis-aligned segment, including ends."""
    if start[0] == end[0]:
        low, high = sorted((start[1], end[1]))
        points = [(start[0], v) for v in range(low, high + 1, 10)]
        points.append((start[0], high))
    elif start[1] == end[1]:
        low, high = sorted((start[0], end[0]))
        points = [(v, start[1]) for v in range(low, high + 1, 10)]
        points.append((high, start[1]))
    else:
        points = [tuple(start), tuple(end)]
    return points


def _format_table(headers, rows):
    """Format rows into an aligned text table."""
    widths = [len(h) for h in headers]
    for row in rows:
        for i, cell in enumerate(row):
            widths[i] = max(widths[i], len(str(cell)))
    lines = ["  ".join(h.ljust(widths[i]) for i, h in enumerate(headers))]
    lines.append("  ".join("-" * w for w in widths))
    for row in rows:
        lines.append("  ".join(str(c).ljust(widths[i]) for i, c in enumerate(row)))
    return "\n".join(lines)


class DisplayManager:
    """Renders status screens and the command-line map."""

    LEGEND = (
        "Legend: R=Rider  S=Restaurant  C=Customer  #=Street  "
        ".=Empty  *=Route\n"
        "Grid nodes are every 100 units; labels show node indices (0-8)."
    )

    @staticmethod
    def _cell(x, y):
        """Map a coordinate to a canvas ``(row, col)`` pair."""
        col = min(int(x) // CELL_UNITS, MAX_CELL_INDEX)
        row = min(int(y) // CELL_UNITS, MAX_CELL_INDEX)
        return row, col

    def _put(self, canvas, point, symbol):
        """Place a symbol on the canvas at ``point``."""
        row, col = self._cell(point[0], point[1])
        canvas[row][col] = symbol

    def render_map(self, riders=(), orders=(), route=None):
        """Render the ASCII map with riders, active orders and an optional route.

        Args:
            riders: Iterable of riders to mark with 'R'.
            orders: Iterable of orders; non-completed ones are marked 'S'/'C'.
            route: Optional list of ``(x, y)`` points drawn with '*'.

        Returns:
            Multi-line string of the map with labels.
        """
        canvas = [
            ["#" if (r % 2 == 0 or c % 2 == 0) else "." for c in range(CANVAS_SIZE)]
            for r in range(CANVAS_SIZE)
        ]
        if route and len(route) > 1:
            for start, end in zip(route, route[1:]):
                for point in _segment_points(start, end):
                    self._put(canvas, point, "*")
        for order in orders:
            if order.status == ORDER_COMPLETED:
                continue
            self._put(canvas, order.res_pos, "S")
            self._put(canvas, order.cus_pos, "C")
        for rider in riders:
            self._put(canvas, (rider.x, rider.y), "R")

        ruler = "   " + "".join(
            str(c // 2) if c % 2 == 0 else " " for c in range(CANVAS_SIZE)
        )
        lines = [ruler]
        for r, row in enumerate(canvas):
            label = f"{r // 2:>2} " if r % 2 == 0 else "   "
            lines.append(label + "".join(row))
        return "\n".join(lines)

        """Return the map legend text."""
        return self.LEGEND

"""Command parsing, shortcut resolution, input validation and history."""

import logging

logger = logging.getLogger(__name__)

COORD_MIN = 0
COORD_MAX = 1000
COORD_LABELS = ("restaurant_x", "restaurant_y", "customer_x", "customer_y")

COMMAND_ALIASES = {
    "start": "start", "s": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "run": "run",
    "save": "save",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}


class InputManager:
    """Reads and validates user input, keeping a command history."""

    def __init__(self, reader=None, history_size=100):
        """Create the input manager.

        Args:
            reader: Callable taking a prompt and returning a line (defaults to input).
            history_size: Maximum number of remembered commands.
        """
        self._reader = reader or input
        self.history = []
        self.history_size = history_size
        self._enable_readline()

    @staticmethod
    def _enable_readline():
        """Enable line editing and up/down history when readline is available."""
        try:
            import readline  # noqa: F401  (side effect: enables history keys)
        except ImportError:
            logger.debug("readline not available; history keys disabled")

    def _remember(self, line):
        """Add a command to the in-memory history."""
        self.history.append(line)
        if len(self.history) > self.history_size:
            del self.history[0]

    def read_line(self, prompt):
        """Read one line from the user and strip surrounding whitespace.

        Args:
            prompt: Prompt string shown to the user.

        Returns:
            The stripped line (may be empty).
        """
        stripped = self._reader(prompt).strip()
        if stripped:
            self._remember(stripped)
        return stripped

    @staticmethod
    def parse_command(line):
        """Resolve a command word (or shortcut) and split its arguments.

        Args:
            line: A non-empty input line.

        Returns:
            Tuple ``(command_name, arguments)``.

        Raises:
            ValueError: If the command is unknown.
        """
        tokens = line.split()
        if not tokens:
            raise ValueError("Empty command.")
        word = tokens[0].lower()
        name = COMMAND_ALIASES.get(word)
        if name is None:
            raise ValueError(f"Unknown command '{word}'. Type 'help' for a command list.")
        return name, tokens[1:]

    @staticmethod
    def parse_integer(text, label):
        """Parse an integer token with a descriptive error message.

        Args:
            text: Token text.
            label: Name of the value, used in the error message.

        Returns:
            The parsed integer.

        Raises:
            ValueError: If the text is not an integer.
        """
        try:
            return int(text)
        except ValueError:
            raise ValueError(f"{label} must be an integer, got '{text}'.") from None

    @staticmethod
    def validate_coordinate(value, label):
        """Check that a coordinate lies within 0-1000.

        Args:
            value: Integer coordinate.
            label: Name of the value, used in the error message.

        Returns:
            The validated value.

        Raises:
            ValueError: If the value is out of range.
        """
        if not COORD_MIN <= value <= COORD_MAX:
            raise ValueError(
                f"{label} must be between {COORD_MIN} and {COORD_MAX}, got {value}."
            )
        return value

    def parse_coordinates(self, tokens):
        """Parse and validate the four order coordinates.

        Args:
            tokens: List of four string tokens.

        Returns:
            List of four validated integers.

        Raises:
            ValueError: On wrong count, non-integer or out-of-range values.
        """
        tokens = list(tokens)
        if len(tokens) != 4:
            raise ValueError(
                "Expected 4 integer coordinates: restaurant_x restaurant_y "
                f"customer_x customer_y (got {len(tokens)})."
            )
        values = []
        for label, token in zip(COORD_LABELS, tokens):
            value = self.parse_integer(token, label)
            values.append(self.validate_coordinate(value, label))
        return values

    def prompt_coordinates(self, prompt):
        """Ask repeatedly for four coordinates until valid input or cancel.

        Args:
            prompt: Prompt text shown to the user.

        Returns:
            List of four validated integers, or None if the user enters nothing.
        """
        while True:
            line = self._reader(prompt).strip()
            if not line:
                return None
            self._remember(line)
            tokens = line.replace(",", " ").split()
            try:
                return self.parse_coordinates(tokens)
            except ValueError as exc:
                print(f"[Error] {exc} Please try again, or press Enter to cancel.")

        """Run the main command loop until quit or end of input."""
        self.display.show_message("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.is_running():
            try:
                line = self.inputs.read_line("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                self.system.stop()
                break
            if not line:
                continue
            try:
                name, args = self.inputs.parse_command(line)
                self.handlers[name](args)
            except ValueError as exc:
                self.display.show_error(str(exc))
            except Exception as exc:  # recovery: keep the simulation alive
                logger.exception("Unexpected error while handling '%s'", line)
                self.display.show_error(f"Unexpected error: {exc}. State preserved.")
                self._recover()
        self.shutdown()

    def _recover(self):
        """Run a consistency check after an unexpected failure and report it."""
        issues = self.system.check_consistency()
        for issue in issues:
            logger.warning("Consistency issue: %s", issue)

    def shutdown(self):
        """Save the configuration when persistence is enabled and say goodbye."""
        if self.config_path:
            try:
                self.system.save_config(self.config_path)
                self.display.show_message(f"State saved to {self.config_path}.")
            except OSError as exc:
                self.display.show_error(f"Could not save configuration: {exc}")

        )
        if new_rider:
            self.display.show_message(
                f"New rider added: Rider #{rider.rider_id if rider else '?'} "
                f"(cost {self.system_recruit_cost()}, funds now {self.system.funds})."
            )
        else:
            self.display.show_message("New rider added: No")

        if rider is None:
            self.display.show_message(
                f"Order #{order.id} is pending: no rider available "
                f"(funds {self.system.funds}, need >= 300 to recruit)."
            )
            return

        route = PathPlanner.build_route(
            (rider.x, rider.y), [order.res_pos, order.cus_pos]
        )
        ticks = PathPlanner.estimate_ticks(route, rider.speed)
        self.display.show_message(
            f"Order #{order.id} assigned to Rider #{rider.rider_id}."
        )
        self.display.show_route(rider, order, route, ticks)

    def system_recruit_cost(self):
        """Return the configured recruitment cost."""
        from system import PARAMS
        return PARAMS["recruit_cost"]

    def cmd_status(self, args):
        """Show the system snapshot."""
        self.display.show_status(
            self.system.funds,
            self.system.riders,
            self.system.pending_order_count(),
            self.system.check_consistency(),
        )

    def cmd_orders(self, args):
        """Show the order table."""
        self.display.show_orders(self.system.order_manager.all_orders())

    def cmd_riders(self, args):
        """Show each rider's recorded event history."""
        self.display.show_rider_history(self.system.riders)

    def cmd_map(self, args):
        """Show the ASCII map with riders and active orders."""
        active = [o for o in self.system.order_manager.all_orders()
                  if o.status in ACTIVE_ORDER_STATUSES or o.status == "pending"]
        self.display.show_map(self.system.riders, active)

    def cmd_tick(self, args):
        """Advance the simulation by one or more ticks."""
        steps = 1
        if args:
            steps = self.inputs.parse_integer(args[0], "tick count")
            if not 1 <= steps <= MAX_TICK_STEP:
                raise ValueError(f"tick count must be between 1 and {MAX_TICK_STEP}.")
        completed = 0
        for _ in range(steps):
            completed += self.system.tick()
        self.display.show_message(
            f"Advanced {steps} tick(s). Now at tick {self.system.current_tick()}. "
            f"Orders completed: {completed}. Active orders: "
            f"{sum(1 for o in self.system.order_manager.all_orders() if o.status in ACTIVE_ORDER_STATUSES)}."
        )

    def cmd_run(self, args):
        """Advance the simulation until no active orders remain."""
        steps = 0
        completed = 0
        while self.system.has_active_orders() and steps < MAX_RUN_TICKS:
            completed += self.system.tick()
            steps += 1
        if self.system.has_active_orders():
            self.display.show_error(f"Stopped after {MAX_RUN_TICKS} ticks; orders still active.")
        self.display.show_message(
            f"Run finished after {steps} tick(s). Orders completed: {completed}. "
            f"Current tick: {self.system.current_tick()}."
        )

    def cmd_save(self, args):
        """Save the configuration to a file."""
        path = args[0] if args else self.config_path
        if not path:
            raise ValueError("No file path given and no default configuration file set.")
        try:
            self.system.save_config(path)
        except OSError as exc:
            raise ValueError(f"Could not save to {path}: {exc}") from exc
        self.display.show_message(f"State saved to {path}.")

    def cmd_help(self, args):
        """Print the command reference."""
        print(HELP_TEXT)

    def cmd_quit(self, args):
        """Stop the main loop; state is saved on shutdown."""
        self.system.stop()


def main(argv=None):
    """Parse command-line arguments and run the simulator.

    Args:
        argv: Optional argument list (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code (0 on normal exit).
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=DEFAULT_CONFIG,
                        help="JSON configuration/state file ('' disables persistence)")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    args = parser.parse_args(argv)

    logging.basicConfig(
        level=logging.DEBUG if args.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    app = DispatchApp(args.config)
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

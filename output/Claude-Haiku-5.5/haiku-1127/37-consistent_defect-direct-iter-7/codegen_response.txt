"""Participant list loading, validation and group extraction."""

from __future__ import annotations

import os
import re

from lottery_system.defaults import DEFAULT_USER_LINES
from lottery_system.models import User

_SEPARATOR_RE = re.compile(r"[,;]")


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'Name,Department' (';' also accepted)."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        path = str(path).strip().strip('"').strip("'")
        if not path or not os.path.isfile(path):
            return [], [f"File Error: '{path}' does not exist or is not a file."]

        lines: list[str] | None = None
        last_error: Exception | None = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=encoding) as handle:
                    lines = handle.read().splitlines()
                break
            except UnicodeDecodeError as exc:
                last_error = exc
            except OSError as exc:
                return [], [f"File Error: cannot read '{path}': {exc}"]

        if lines is None:
            return [], [f"File Error: cannot decode '{path}': {last_error}"]

        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for lineno, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SEPARATOR_RE.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {lineno}: '{line}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Remove duplicate names. Returns (valid users, duplicate entries).

        The first occurrence of each name is kept as valid; later occurrences
        are reported as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        """Return department -> number of people, in first-seen order."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management and ordering."""

from __future__ import annotations

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(f"Failed to add default prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name,
        quantity,
        avoid_groups=None,
        associated_drawer=None,
    ) -> Prize:
        """Append a prize to the lottery sequence.

        Raises ValueError on invalid name, quantity, or duplicate name.
        """
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name must not be empty.")
        name = name.strip()
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")

        if isinstance(quantity, bool):
            raise ValueError("Quantity must be a positive integer.")
        try:
            quantity = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Quantity must be a positive integer.") from None
        if quantity < 1:
            raise ValueError("Quantity must be a positive integer.")

        groups: list[str] = []
        for group in avoid_groups or []:
            group = str(group).strip()
            if group and group not in groups:
                groups.append(group)

        drawer = None
        if associated_drawer is not None:
            drawer = str(associated_drawer).strip() or None

        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery order (a copy of the list)."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        order: 1-based current positions, listed in the new order.
        Returns False if order is not a permutation of 1..n.
        """
        n = len(self._prizes)
        try:
            indexes = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if len(indexes) != n or sorted(indexes) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in indexes]
        return True

"""Winner recording, ranking and CSV export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import User

_ORDINAL_WORDS = [
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
    ("sixth", 6),
    ("seventh", 7),
    ("eighth", 8),
    ("ninth", 9),
    ("tenth", 10),
]


def prize_level(name: str, index: int) -> int:
    """Return the prize level (1 = highest). Unknown levels sort after known ones,
    keeping their configured position."""
    lower = name.casefold()
    for word, rank in _ORDINAL_WORDS:
        if re.search(rf"\b{word}\b", lower):
            return rank
    match = re.search(r"\b(\d+)(st|nd|rd|th)\b", lower)
    if match:
        return int(match.group(1))
    return 1000 + index


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    sequence: int
    timestamp: str


class ResultManager:
    """Maintains the winner list in real time."""

    def __init__(self) -> None:
        self.winners: list[Winner] = []

    def add_winner(self, user: User, prize) -> Winner:
        """Record a winner. Prize may be a Prize object or a prize name.

        Raises ValueError if the person has already won (anti-duplication).
        """
        if any(w.user.name == user.name for w in self.winners):
            raise ValueError(f"{user.name} has already won a prize.")
        prize_name = prize if isinstance(prize, str) else prize.name
        winner = Winner(
            user=user,
            prize_name=prize_name,
            sequence=len(self.winners) + 1,
            timestamp=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        )
        self.winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        """Map prize name -> winners in winning-time order."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            result[prize.name] = [w for w in self.winners if w.prize_name == prize.name]
        return result

    def get_ranked_winners(self, prizes) -> list[Winner]:
        """Winners sorted by prize level (high to low), then winning time order."""
        prize_list = list(prizes)
        ranked = sorted(
            enumerate(prize_list),
            key=lambda item: (prize_level(item[1].name, item[0]), item[0]),
        )
        result: list[Winner] = []
        for _, prize in ranked:
            result.extend(w for w in self.winners if w.prize_name == prize.name)
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export the ranked winner list to current_dir/awards_result_YYYYMMDD.txt."""
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in self.get_ranked_winners(prizes):
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.timestamp,
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery rule engine."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager, Winner

CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"
LOTTERY_PERSONS = [CHEN_QIXIAN, JIN_DUO, MA_JIAQI, LI_BIN]

CHENG_RU_KEYWORD = "cheng ru"


def _norm(text) -> str:
    return " ".join(str(text).split()).casefold()


def is_cheng_ru_group(department: str) -> bool:
    """True if the department is the Cheng Ru group."""
    return CHENG_RU_KEYWORD in _norm(department)


class LotteryManager:
    """Applies the lottery rules and tracks winners and fallback history."""

    def __init__(self, users) -> None:
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Results of the last two rounds drawn by Chen Qixian / Jin Duo.
        # True means that round produced a Cheng Ru group winner.
        self.last_two_draw_results: list[bool] = []
        self._round_prize: Prize | None = None
        self._round_cheng_ru_hit = False

    # ----- helpers -----

    def _drawer_is(self, prize: Prize, drawer: str) -> bool:
        return _norm(prize.associated_drawer or "") == _norm(drawer)

    def _is_special_person(self, prize: Prize) -> bool:
        return self._drawer_is(prize, LI_BIN)

    def _is_protection_active(self, prize: Prize) -> bool:
        return (
            self._drawer_is(prize, MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _is_avoided(self, user: User, prize: Prize) -> bool:
        avoid = {_norm(group) for group in prize.avoid_groups}
        return _norm(user.department) in avoid

    def _candidates(self) -> list[User]:
        winner_names = {w.user.name for w in self.result_manager.winners}
        return [u for u in self.users if u.name not in winner_names]

    def _build_pool(self, prize: Prize) -> tuple[list[User], bool]:
        """Return (pool, flat). flat=True means no avoidance tiering applies."""
        candidates = self._candidates()

        if self._is_protection_active(prize):
            protected = [u for u in candidates if is_cheng_ru_group(u.department)]
            if protected:
                return protected, True

        if self._is_special_person(prize):
            return candidates, True

        primary = [u for u in candidates if not self._is_avoided(u, prize)]
        secondary = [u for u in candidates if self._is_avoided(u, prize)]
        return primary + secondary, False

    # ----- public API -----

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Eligible participants for the prize.

        - Previous winners are always excluded.
        - Li Bin special rule: only previous winners are excluded.
        - Cheng Ru protection: when active, pool is limited to Cheng Ru members.
        - Otherwise avoidance-group members remain eligible but are placed
          after all other members.
        """
        pool, _ = self._build_pool(prize)
        return pool

    def draw_next(self, prize: Prize) -> Winner | None:
        """Draw one winner for the prize. Returns None if no one is eligible."""
        pool, flat = self._build_pool(prize)
        if not pool:
            return None

        if flat:
            choices = pool
        else:
            primary = [u for u in pool if not self._is_avoided(u, prize)]
            choices = primary or pool

        user = random.choice(choices)
        winner = self.result_manager.add_winner(user, prize)

        drawer_triggers = self._drawer_is(prize, CHEN_QIXIAN) or self._drawer_is(
            prize, JIN_DUO
        )
        if drawer_triggers and is_cheng_ru_group(user.department):
            self._round_cheng_ru_hit = True
        return winner

    def start_round(self, prize: Prize) -> None:
        self._round_prize = prize
        self._round_cheng_ru_hit = False

    def end_round(self, prize: Prize) -> None:
        """Finish a round and record its result for the fallback rule."""
        if self._drawer_is(prize, CHEN_QIXIAN) or self._drawer_is(prize, JIN_DUO):
            self.last_two_draw_results.append(self._round_cheng_ru_hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        self._round_prize = None
        self._round_cheng_ru_hit = False

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable rule description for a prize round."""
        rules: list[str] = []
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}.")

        if self._is_special_person(prize):
            rules.append(
                "Only previous winners are excluded; avoidance groups do not apply."
            )
        else:
            rules.append("Previous winners are excluded.")
            if prize.avoid_groups:
                rules.append(
                    "Members of avoidance group(s) "
                    f"{', '.join(prize.avoid_groups)} are drawn after all other members."
                )
            if self._drawer_is(prize, MA_JIAQI):
                rules.append(
                    "Cheng Ru Group protection: if neither of the last two Chen Qixian/"
                    "Jin Duo rounds produced a Cheng Ru Group winner, the pool is "
                    "limited to Cheng Ru Group members."
                )
                if self._is_protection_active(prize):
                    rules.append("Cheng Ru Group protection is ACTIVE for this round.")
        return " ".join(rules)

"""Command-line interface for the team building smart lottery."""

from __future__ import annotations

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LOTTERY_PERSONS, LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager, Winner

MENU = """
========== Team Building Smart Lottery ==========
1. Load participant list
2. View department groups
3. Configure prizes
4. Start lottery
5. View / export results
0. Exit
"""


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotterySystemCLI:
    def __init__(self, input_func=input, output_func=print) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.lottery: LotteryManager | None = None

    # ----- basic helpers -----

    def _ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    def run(self) -> None:
        try:
            while True:
                self._print(MENU)
                choice = self._ask("Select an option: ")
                if choice == "1":
                    self._menu_load_list()
                elif choice == "2":
                    self._menu_view_groups()
                elif choice == "3":
                    self._menu_prizes()
                elif choice == "4":
                    self._start_lottery()
                elif choice == "5":
                    self._menu_results()
                elif choice == "0":
                    self._print("Goodbye!")
                    return
                else:
                    self._print("Invalid option, please try again.")
        except (EOFError, KeyboardInterrupt):
            self._print("\nExiting.")

    # ----- participant list -----

    def _menu_load_list(self) -> None:
        self._print("1. Built-in team building list")
        self._print("2. Custom txt list (each line: Name,Department)")
        self._print("0. Back")
        choice = self._ask("Select: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

        for error in errors:
            self._print(error)

        valid, duplicates = self.list_manager.validate_users(users)
        for dup in duplicates:
            self._print(f"Duplicate Name: '{dup.name}' ({dup.department}) - later entry ignored.")

        if errors or duplicates:
            self._print(
                f"Detected {len(errors)} format error(s) and {len(duplicates)} duplicate(s)."
            )
            self._print("Enter 'r' to remove invalid data and continue, or 'q' to terminate upload.")
            while True:
                action = self._ask("Choice (r/q): ").lower()
                if action == "r":
                    break
                if action == "q":
                    self._print("Upload terminated; participant list unchanged.")
                    return
                self._print("Please enter 'r' or 'q'.")

        if not valid:
            self._print("No valid participants found.")
            return

        self.users = valid
        self.lottery = None
        self._print(f"Loaded {len(valid)} participant(s).")

    def _groups(self) -> list[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    def _menu_view_groups(self) -> None:
        if not self.users:
            self._print("Please load the participant list first.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._print("\nDepartment groups:")
        for index, (dept, count) in enumerate(info.items(), start=1):
            self._print(f"  {index}. {dept}: {count} people")
        self._print(f"Total: {len(self.users)} people")

    # ----- prizes -----

    def _menu_prizes(self) -> None:
        while True:
            self._print("\n--- Prize configuration ---")
            self._print("1. Load preset prizes")
            self._print("2. Add custom prize")
            self._print("3. Reorder prizes")
            self._print("4. View prizes")
            self._print("0. Back")
            choice = self._ask("Select: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                for error in errors:
                    self._print(error)
                if not errors:
                    self._print("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._reorder_prizes()
            elif choice == "4":
                self._show_prizes()
            elif choice == "0":
                return
            else:
                self._print("Invalid option.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self._print(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"avoid groups: {groups} | lottery person: {drawer}"
            )

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            self._print("Prize name cannot be empty.")
            return
        quantity_text = self._ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._print("Quantity must be an integer.")
            return
        groups = self._choose_avoid_groups()
        drawer = self._choose_drawer()
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            self._print(f"Failed to add prize: {exc}")
            return
        self._print(f"Prize '{name}' added.")

    def _choose_avoid_groups(self) -> list[str]:
        groups = self._groups()
        if groups:
            self._print("Department groups:")
            for index, dept in enumerate(groups, start=1):
                self._print(f"  {index}. {dept}")
        text = self._ask(
            "Avoidance groups (numbers or names, comma-separated; blank for none): "
        )
        result: list[str] = []
        for token in text.split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                value = groups[int(token) - 1]
            else:
                value = token
            if value not in result:
                result.append(value)
        return result

    def _choose_drawer(self) -> str | None:
        self._print("Lottery persons:")
        for index, person in enumerate(LOTTERY_PERSONS, start=1):
            self._print(f"  {index}. {person}")
        text = self._ask("Associated lottery person (number or name; blank for none): ")
        if not text:
            return None
        if text.isdigit() and 1 <= int(text) <= len(LOTTERY_PERSONS):
            return LOTTERY_PERSONS[int(text) - 1]
        return text

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            self._print("At least two prizes are needed to reorder.")
            return
        self._show_prizes()
        text = self._ask(
            "Enter new order as serial numbers separated by spaces (e.g. 2 1 3): "
        )
        try:
            order = [int(token) for token in text.split()]
        except ValueError:
            self._print("Serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Prize order updated.")
            self._show_prizes()
        else:
            self._print("Invalid order: every serial number must appear exactly once.")

    # ----- lottery process -----

    def _start_lottery(self) -> None:
        if not self.users:
            self._print("Please load the participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure prizes first.")
            return
        if self.lottery and self.lottery.result_manager.winners:
            answer = self._ask("Previous results will be cleared. Continue? (y/n): ").lower()
            if answer != "y":
                return

        self.lottery = LotteryManager(self.users)
        total = len(prizes)
        self._print("\nCommands before each draw: [Enter]/continue = draw, pause = pause, terminate = end this round.")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)

        self._print("\n========== Lottery finished ==========")
        self._print_results(self.lottery.result_manager, prizes)

    def _run_round(self, index: int, total: int, prize) -> None:
        assert self.lottery is not None
        self._print(
            f"\n----- Round {index}/{total}: {prize.name} | Quantity: {prize.quantity} -----"
        )
        self._print(f"Rule: {self.lottery.describe_rules(prize)}")

        self.lottery.start_round(prize)
        drawn = 0
        while drawn < prize.quantity:
            if not self._wait_for_command():
                self._print("Round terminated by user.")
                break
            self._print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            winner = self.lottery.draw_next(prize)
            if winner is None:
                self._print("No eligible participants remain for this prize.")
                break
            drawn += 1
            user = winner.user
            self._print(
                f"Congratulations {user.name} ({user.department}) for winning {prize.name}!"
            )
        self.lottery.end_round(prize)

        round_winners = [
            w for w in self.lottery.result_manager.winners if w.prize_name == prize.name
        ]
        self._print(
            f"Round {index} finished: {len(round_winners)}/{prize.quantity} winner(s)."
        )

    def _wait_for_command(self) -> bool:
        """Handle control commands before a draw. Returns False to terminate the round."""
        command = self._ask("[Enter] draw | pause | terminate > ").lower()
        while True:
            if command == "pause":
                self._print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    command = self._ask("[paused] continue / terminate > ").lower()
                    if command == "continue":
                        self._print("Lottery resumed.")
                        return True
                    if command == "terminate":
                        return False
                    self._print("While paused, only 'continue' or 'terminate' is accepted.")
            if command == "terminate":
                return False
            if command in ("", "continue"):
                return True
            self._print("Unknown command. Use Enter, 'pause' or 'terminate'.")
            command = self._ask("[Enter] draw | pause | terminate > ").lower()

    # ----- results -----

    def _print_results(self, result_manager: ResultManager, prizes) -> None:
        ranked: list[Winner] = result_manager.get_ranked_winners(prizes)
        if not ranked:
            self._print("No winners yet.")
            return
        self._print(f"{'Name':<16}{'Department':<30}{'Prize Name':<40}{'Time'}")
        for winner in ranked:
            self._print(
                f"{winner.user.name:<16}{winner.user.department:<30}"
                f"{winner.prize_name:<40}{winner.timestamp}"
            )

    def _menu_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        while True:
            self._print("\n--- Results ---")
            self._print("1. View winner list")
            self._print("2. Export winner list (txt/CSV)")
            self._print("0. Back")
            choice = self._ask("Select: ")
            if choice == "1":
                if self.lottery is None:
                    self._print("No lottery has been run yet.")
                else:
                    self._print_results(self.lottery.result_manager, prizes)
            elif choice == "2":
                if self.lottery is None:
                    self._print("No lottery has been run yet.")
                    continue
                _, message = self.lottery.result_manager.export_to_csv(prizes)
                self._print(message)
            elif choice == "0":
                return
            else:
                self._print("Invalid option.")


def main() -> None:
    LotterySystemCLI().run()

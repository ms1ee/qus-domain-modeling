def _parse_lines(lines) -> tuple[list[User], list[str]]:
    """Parse 'Name,Department' lines. Blank lines are skipped.

    Returns the valid users and a list of error messages for malformed lines.
    """
    users: list[User] = []
    errors: list[str] = []
    for lineno, raw in enumerate(lines, start=1):
        line = raw.strip()
        if not line:
            continue
        parts = line.split(",")
        if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
            errors.append(
                f"Line {lineno}: Format Error - expected 'Name,Department', got '{line}'"
            )
            continue
        users.append(User(parts[0].strip(), parts[1].strip()))
    return users, errors


class ListManager:
    """Loads, validates and summarises participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return _parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.readlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except UnicodeDecodeError:
            return [], [f"File Error: {path} is not valid UTF-8 text"]
        except OSError as exc:
            return [], [f"File Error: cannot read {path} ({exc})"]
        return _parse_lines(lines)

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid_users, duplicates).

        The first occurrence of each name is valid; later occurrences of the
        same name are reported as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the department groups and the number of people in each."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.models import Prize

DEFAULT_PRIZES: list[tuple[str, int]] = [
    ("Third Prize - Movie Tickets", 5),
    ("Second Prize - Shopping Cards", 3),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns error messages, if any."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(f"Default prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize to the end of the sequence. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups: list[str] = []
        for group in avoid_groups or []:
            group = (group or "").strip()
            if group and group not in groups:
                groups.append(group)

        drawer = None
        if associated_drawer and associated_drawer.strip():
            drawer = associated_drawer.strip()

        prize = Prize(clean_name, quantity, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        try:
            order_list = list(order)
        except TypeError:
            return False
        if not all(isinstance(index, int) and not isinstance(index, bool) for index in order_list):
            return False
        if sorted(order_list) != list(range(1, len(self._prizes) + 1)):
            return False
        self._prizes = [self._prizes[index - 1] for index in order_list]
        return True

"""Winner recording, grouping and CSV-style export."""

import csv
import os
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str


class ResultManager:
    """Maintains the real-time winner list."""

    def __init__(self):
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        """Record a winner. Raises ValueError if the person has already won."""
        if user.name in self.winner_names():
            raise ValueError(f"{user.name} has already won a prize.")
        self._winners.append(Winner(user, prize.name))

    def winner_names(self) -> set[str]:
        """Names of everyone who has won so far."""
        return {winner.user.name for winner in self._winners}

    def get_all_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, keeping the order of `prizes`."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export winners to ./awards_result_YYYYMMDD.txt in CSV format."""
        path = os.path.join(
            os.getcwd(), f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
        )
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for prize in prizes:
                    for winner in grouped[prize.name]:
                        writer.writerow(
                            [winner.user.name, winner.user.department, winner.prize_name]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery rule engine: eligibility filtering, drawing and fallback protection."""

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

CHENG_RU_GROUP = "Cheng Ru group"
LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
FALLBACK_DRAWERS = (CHEN_QIXIAN, JIN_DUO)


class LotteryManager:
    """Runs lottery draws while applying all rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None):
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Results (True = contained a Cheng Ru group member) of the most recent
        # rounds run by Chen Qixian / Jin Duo, newest last. Max length 2.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.Random()

    def _unrestricted_pool(self) -> list[User]:
        """Everyone who has not won yet."""
        winners = self.result_manager.winner_names()
        return [user for user in self.users if user.name not in winners]

    def _fallback_active(self) -> bool:
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _cheng_ru_fallback_pool(self, prize: Prize, pool: list[User]) -> list[User] | None:
        """Cheng Ru group protection pool for Ma Jiaqi's prizes, or None if not applicable."""
        if prize.associated_drawer != MA_JIAQI or not self._fallback_active():
            return None
        members = [user for user in pool if user.department == CHENG_RU_GROUP]
        return members or None

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        pool = self._unrestricted_pool()

        # Special rule: Li Bin only filters previous winners.
        if prize.associated_drawer == LI_BIN:
            return pool

        # Fallback rule: forcibly limit to Cheng Ru group members, overriding avoidance.
        fallback = self._cheng_ru_fallback_pool(prize, pool)
        if fallback is not None:
            return fallback

        if prize.avoid_groups:
            avoided = set(prize.avoid_groups)
            pool = [user for user in pool if user.department not in avoided]
        return pool

    def has_candidates(self, prize: Prize) -> bool:
        """Whether at least one person is eligible for the prize right now."""
        return bool(self._get_eligible_pool(prize))

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one eligible winner and record them. Returns None if no candidates."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self.result_manager.add_winner(winner, prize)
        return winner

    def record_round_result(self, prize: Prize, winners: list[User]) -> None:
        """Update fallback state after a round. Only Chen Qixian / Jin Duo rounds count."""
        if prize.associated_drawer not in FALLBACK_DRAWERS:
            return
        contains_cheng_ru = any(user.department == CHENG_RU_GROUP for user in winners)
        self.last_two_draw_results = (self.last_two_draw_results + [contains_cheng_ru])[-2:]

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize."""
        parts: list[str] = []
        if prize.associated_drawer:
            parts.append(f"lottery person: {prize.associated_drawer}")
        if prize.associated_drawer == LI_BIN:
            parts.append("only previous winners are excluded (avoidance groups not applied)")
        elif self._cheng_ru_fallback_pool(prize, self._unrestricted_pool()) is not None:
            parts.append(
                f"Cheng Ru group protection active: only '{CHENG_RU_GROUP}' members are eligible"
            )
        else:
            parts.append("previous winners are excluded")
            if prize.avoid_groups:
                parts.append("avoided groups: " + ", ".join(prize.avoid_groups))
        return "; ".join(parts)

"""Command-line interface: list setup, prize configuration and lottery control."""

import queue
import sys
import threading
import time

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LI_BIN, LotteryManager
from lottery_system.services.prize_manager import PrizeManager

DEFAULT_DRAW_DELAY = 0.5
_EOF = object()


class InputClosed(Exception):
    """Raised when standard input has been closed."""


class CommandReader:
    """Reads stdin in a background thread so commands can be polled during a draw."""

    def __init__(self, stream=None):
        self._stream = stream if stream is not None else sys.stdin
        self._queue: queue.Queue = queue.Queue()
        thread = threading.Thread(target=self._pump, daemon=True)
        thread.start()

    def _pump(self) -> None:
        try:
            for line in self._stream:
                self._queue.put(line.rstrip("\r\n"))
        except (OSError, ValueError):
            pass
        finally:
            self._queue.put(_EOF)

    def read(self) -> str:
        """Blocking read of one line. Raises InputClosed at end of input."""
        item = self._queue.get()
        if item is _EOF:
            self._queue.put(_EOF)
            raise InputClosed()
        return item

    def poll(self) -> str | None:
        """Non-blocking read. Returns None if no line is waiting."""
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        if item is _EOF:
            self._queue.put(_EOF)
            return None
        return item


def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _parse_numbers(text: str) -> list[int] | None:
    """Parse whitespace/comma separated integers. Returns None if any token is invalid."""
    tokens = text.replace(",", " ").split()
    try:
        return [int(token) for token in tokens]
    except ValueError:
        return None


class LotterySystemCLI:
    """Interactive menu-driven controller for the lottery system."""

    def __init__(self, reader: CommandReader | None = None, draw_delay: float = DEFAULT_DRAW_DELAY):
        self.reader = reader if reader is not None else CommandReader()
        self.draw_delay = draw_delay
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self._actions = {
            "1": self.load_default_list,
            "2": self.upload_custom_list,
            "3": self.view_groups,
            "4": self.load_default_prizes,
            "5": self.add_custom_prize,
            "6": self.manage_prize_order,
            "7": self.start_lottery,
            "8": self.export_results,
        }

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str) -> str:
        print(prompt, end="", flush=True)
        return self.reader.read().strip()

    def _show_menu(self) -> None:
        print()
        print("=" * 56)
        print("   Team Building Smart Lottery System")
        print("=" * 56)
        print("  1. Load built-in default team building list")
        print("  2. Upload custom txt list (Name,Department per line)")
        print("  3. View department groups and headcounts")
        print("  4. Load default prize library")
        print("  5. Add custom prize")
        print("  6. View prizes / adjust lottery sequence")
        print("  7. Start lottery")
        print("  8. Export winner list (txt, CSV format)")
        print("  0. Exit")
        print("-" * 56)

    def _group_names(self) -> list[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    def _print_groups(self) -> None:
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for index, (department, count) in enumerate(info.items(), start=1):
            print(f"  {index}. {department}: {count} people")

    def _print_prizes(self, prizes) -> None:
        for index, prize in enumerate(prizes, start=1):
            drawer = prize.associated_drawer or "-"
            avoid = ", ".join(prize.avoid_groups) or "-"
            print(
                f"  {index}. {prize.name} | quantity: {prize.quantity} "
                f"| lottery person: {drawer} | avoid groups: {avoid}"
            )

    # ------------------------------------------------------------ main loop
    def run(self) -> None:
        print("Welcome to the Team Building Smart Lottery System.")
        while True:
            self._show_menu()
            choice = self._ask("Select an option: ")
            if choice == "0":
                print("Goodbye!")
                return
            action = self._actions.get(choice)
            if action is None:
                print("Invalid option, please try again.")
                continue
            action()

    # -------------------------------------------------------- list management
    def _accept_list(self, users: list[User], errors: list[str]) -> bool:
        if errors:
            print(f"Found {len(errors)} problematic line(s):")
            for error in errors:
                print(f"  - {error}")
            choice = self._ask(
                "Enter 'r' to remove invalid data and continue, or 't' to terminate the upload: "
            ).lower()
            if choice != "r":
                print("Upload terminated.")
                return False
            print("Invalid data removed.")

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  - {user.name} ({user.department})")
            choice = self._ask(
                "Enter 'r' to remove duplicates and continue, or 't' to terminate the upload: "
            ).lower()
            if choice != "r":
                print("Upload terminated.")
                return False
            users = valid

        if not users:
            print("No valid participants found. The list was not loaded.")
            return False

        self.users = users
        print(f"Loaded {len(users)} participant(s).")
        self._print_groups()
        return True

    def load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._accept_list(users, errors)

    def upload_custom_list(self) -> None:
        path = self._ask("Enter the path of the txt file: ").strip('"').strip("'")
        users, errors = self.list_manager.load_custom_list(path)
        self._accept_list(users, errors)

    def view_groups(self) -> None:
        if not self.users:
            print("No participant list loaded yet.")
            return
        self._print_groups()

    # ---------------------------------------------------------- prize config
    def load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            print("Some default prizes could not be loaded:")
            for error in errors:
                print(f"  - {error}")
        else:
            print("Default prize library loaded.")

    def add_custom_prize(self) -> None:
        name = self._ask("Prize name (blank to cancel): ")
        if not name:
            print("Cancelled.")
            return

        quantity_text = self._ask("Prize quantity: ")
        if not quantity_text.isdigit() or int(quantity_text) < 1:
            print("Quantity must be a positive integer. Prize not added.")
            return

        drawer = self._ask("Associated lottery person (blank for none): ") or None

        avoid_groups: list[str] = []
        groups = self._group_names()
        if groups:
            print("Department groups:")
            for index, group in enumerate(groups, start=1):
                print(f"  {index}. {group}")
            text = self._ask("Avoidance group numbers, comma separated (blank for none): ")
            if text:
                indexes = _parse_numbers(text)
                if indexes is None or any(not 1 <= index <= len(groups) for index in indexes):
                    print("Invalid group selection. Prize not added.")
                    return
                avoid_groups = [groups[index - 1] for index in dict.fromkeys(indexes)]

        if drawer == LI_BIN and avoid_groups:
            print("Note: the Li Bin rule ignores avoidance groups.")

        try:
            prize = self.prize_manager.add_prize(name, int(quantity_text), avoid_groups, drawer)
        except ValueError as exc:
            print(f"Error: {exc}")
            return
        print(f"Prize added: {prize.name} (quantity: {prize.quantity})")

    def manage_prize_order(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured yet.")
            return
        print("Current lottery sequence:")
        self._print_prizes(prizes)
        text = self._ask("Enter new order as serial numbers, e.g. '3 1 2' (blank to keep): ")
        if not text:
            return
        order = _parse_numbers(text)
        if not order or not self.prize_manager.reorder_prizes(order):
            print("Invalid order. Each prize number must appear exactly once.")
            return
        print("Lottery sequence updated:")
        self._print_prizes(self.prize_manager.get_prizes())

    # --------------------------------------------------------- lottery control
    def _check_controls(self) -> bool:
        """Process queued control commands. Returns False if the current round should stop."""
        command = self.reader.poll()
        while command is not None:
            command = command.strip().lower()
            if command == "pause":
                print("[PAUSED] Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    answer = self.reader.read().strip().lower()
                    if answer == "continue":
                        print("[RESUMED]")
                        break
                    if answer == "terminate":
                        print("[TERMINATED] Current round ended early.")
                        return False
                    print("Lottery is paused. Type 'continue' or 'terminate'.")
            elif command == "terminate":
                print("[TERMINATED] Current round ended early.")
                return False
            elif command == "continue":
                print("Lottery is already running.")
            elif command:
                print(f"Unknown command '{command}'. Available: pause, continue, terminate.")
            command = self.reader.poll()
        return True

    def _run_round(self, index: int, total: int, prize) -> None:
        assert self.lottery is not None
        print()
        print(f"=== Round {index}/{total}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {self.lottery.describe_rule(prize)}")

        round_winners: list[User] = []
        if not self.lottery.has_candidates(prize):
            print("No eligible candidates for this prize; round skipped.")
        else:
            for position in range(1, prize.quantity + 1):
                if not self._check_controls():
                    break
                if not self.lottery.has_candidates(prize):
                    print("No eligible candidates remain; round ends early.")
                    break
                print(f"Currently drawing the {_ordinal(position)} winner...", flush=True)
                time.sleep(self.draw_delay)
                if not self._check_controls():
                    break
                winner = self.lottery.draw_one(prize)
                if winner is None:
                    print("No eligible candidates remain; round ends early.")
                    break
                round_winners.append(winner)
                print(
                    f"Congratulations {winner.name} ({winner.department}) "
                    f"for winning {prize.name}!"
                )

        self.lottery.record_round_result(prize, round_winners)

        print(f"--- Round results: {prize.name} ---")
        if not round_winners:
            print("  No winners in this round.")
        for number, winner in enumerate(round_winners, start=1):
            print(f"  {number}. {winner.name} ({winner.department})")

    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is not None:
            print("A lottery has already been run in this session. Restart the program to run again.")
            return

        self.lottery = LotteryManager(self.users)
        print()
        print("Lottery starts! During a draw you may type: pause / continue / terminate")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize)

        self._show_final_results(prizes)

    def _show_final_results(self, prizes) -> None:
        assert self.lottery is not None
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        print()
        print("=" * 56)
        print("   Final Winner List")
        print("=" * 56)
        for prize in prizes:
            print(f"{prize.name}:")
            winners = grouped[prize.name]
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(f"  - {winner.user.name} ({winner.user.department})")

        answer = self._ask("Export the winner list now? (y/n): ").lower()
        if answer == "y":
            self._export(prizes)

    # ------------------------------------------------------------- results
    def _export(self, prizes) -> None:
        assert self.lottery is not None
        _, message = self.lottery.result_manager.export_to_csv(prizes)
        print(message)

    def export_results(self) -> None:
        if self.lottery is None:
            print("No lottery results to export yet.")
            return
        self._export(self.prize_manager.get_prizes())

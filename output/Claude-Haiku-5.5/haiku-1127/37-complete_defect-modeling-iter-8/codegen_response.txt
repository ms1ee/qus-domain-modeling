class ListManager:
    """Loads, validates and summarizes participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        try:
            text = self._read_text(path)
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(text.splitlines())

    @staticmethod
    def _read_text(path: str) -> str:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                return handle.read()
        except UnicodeDecodeError:
            with open(path, "r", encoding="gbk") as handle:
                return handle.read()

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for lineno, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in line.replace("\uff0c", ",").split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {lineno}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). Only the first occurrence of a name is valid."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to member count."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from lottery_system.models import Prize


PRESET_PRIZES: list[tuple[str, int]] = [
    ("Third Prize - Movie Tickets", 10),
    ("Second Prize - Shopping Cards", 5),
    ("First Prize - 1 Day Annual Leave", 2),
]


class PrizeManager:
    """Manages the prize library and the lottery sequence."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _has_name(self, name: str) -> bool:
        key = name.strip().casefold()
        return any(prize.name.strip().casefold() == key for prize in self._prizes)

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns error messages for presets that could not be added."""
        errors: list[str] = []
        for name, quantity in PRESET_PRIZES:
            if self._has_name(name):
                errors.append(f"Prize already exists, skipped: '{name}'")
                continue
            self.add_prize(name, quantity)
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize to the library. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._has_name(clean_name):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups: list[str] = []
        for group in avoid_groups or []:
            g = str(group).strip()
            if g and g not in groups:
                groups.append(g)

        drawer = associated_drawer.strip() if associated_drawer else None
        if drawer == "":
            drawer = None

        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        n = len(self._prizes)
        if len(order) != n:
            return False
        if not all(isinstance(i, int) and not isinstance(i, bool) for i in order):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

from __future__ import annotations

import random

from lottery_system.models import Prize, User


CHENG_RU_GROUP = "Cheng Ru Group"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"


def _norm(value: str | None) -> str:
    return (value or "").strip().casefold()


def _same(a: str | None, b: str | None) -> bool:
    return _norm(a) == _norm(b) and _norm(a) != ""


class LotteryManager:
    """Executes lottery rounds and enforces the lottery rule engine."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._rng = rng if rng is not None else random.Random()
        self._winners: list[User] = []
        self._winner_keys: set[str] = set()
        # Rolling record (max 2 entries) of whether the last rounds run by
        # Chen Qixian / Jin Duo contained a Cheng Ru Group member.
        self.last_two_draw_results: list[bool] = []
        self._round_prize: Prize | None = None
        self._round_winners: list[User] = []
        self._round_quota: int = 0

    # ------------------------------------------------------------------ state
    @property
    def winners(self) -> list[User]:
        return list(self._winners)

    @property
    def is_round_active(self) -> bool:
        return self._round_prize is not None

    def _is_winner(self, user: User) -> bool:
        return _norm(user.name) in self._winner_keys

    def _is_fallback_active(self, prize: Prize) -> bool:
        """Cheng Ru protection: applies to Ma Jiaqi's prize when the last two
        Chen Qixian / Jin Duo rounds contained no Cheng Ru Group member."""
        if not _same(prize.associated_drawer, MA_JIAQI):
            return False
        if len(self.last_two_draw_results) < 2:
            return False
        return not any(self.last_two_draw_results)

    # ------------------------------------------------------------ rule engine
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        candidates = [user for user in self._users if not self._is_winner(user)]

        # Special rule: Li Bin's prize only excludes previous winners.
        if _same(prize.associated_drawer, LI_BIN):
            return candidates

        # Fallback rule: pool forcibly limited to Cheng Ru Group members,
        # even if that group is an avoidance group of this prize.
        if self._is_fallback_active(prize):
            return [u for u in candidates if _same(u.department, CHENG_RU_GROUP)]

        avoid = {_norm(group) for group in prize.avoid_groups}
        return [u for u in candidates if _norm(u.department) not in avoid]

    def describe_rule(self, prize: Prize) -> str:
        parts: list[str] = []
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}")
        if _same(prize.associated_drawer, LI_BIN):
            parts.append(
                "Special rule: only previous winners are excluded; avoidance groups do not apply"
            )
        elif self._is_fallback_active(prize):
            parts.append("Fallback rule active: only Cheng Ru Group members are eligible")
        elif prize.avoid_groups:
            parts.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No group avoidance")
        parts.append("Previous winners are excluded")
        return "; ".join(parts)

    # ---------------------------------------------------------- round control
    def start_round(self, prize: Prize) -> None:
        if self._round_prize is not None:
            raise RuntimeError("A lottery round is already in progress.")
        self._round_prize = prize
        self._round_winners = []
        self._round_quota = prize.quantity

    def draw_next(self) -> User | None:
        """Draw the next winner of the active round. Returns None if no draw is possible."""
        if self._round_prize is None:
            raise RuntimeError("No lottery round is in progress.")
        if len(self._round_winners) >= self._round_quota:
            return None
        pool = self._get_eligible_pool(self._round_prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._round_winners.append(user)
        self._winners.append(user)
        self._winner_keys.add(_norm(user.name))
        return user

    def end_round(self) -> list[User]:
        """Close the active round, update fallback records and return its winners."""
        prize = self._round_prize
        winners = list(self._round_winners)
        if prize is None:
            return []

        drawer = prize.associated_drawer
        if _same(drawer, CHEN_QIXIAN) or _same(drawer, JIN_DUO):
            has_cheng_ru = any(_same(u.department, CHENG_RU_GROUP) for u in winners)
            self.last_two_draw_results.append(has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif _same(drawer, MA_JIAQI):
            # Ma Jiaqi's round consumes the consecutive-miss window.
            self.last_two_draw_results = []

        self._round_prize = None
        self._round_winners = []
        self._round_quota = 0
        return winners

    def draw(self, prize: Prize) -> list[User]:
        """Convenience: run a full round without interaction."""
        self.start_round(prize)
        while self.draw_next() is not None:
            pass
        return self.end_round()

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    draw_order: int = 0
    win_time: datetime = field(default_factory=datetime.now)

    @property
    def name(self) -> str:
        return self.user.name

    @property
    def department(self) -> str:
        return self.user.department


class ResultManager:
    """Stores winners and exports results."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        self._winners.append(
            Winner(user=user, prize_name=prize_name, draw_order=len(self._winners) + 1)
        )

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        stamp = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{stamp}.txt")

        grouped = self.get_winners_by_prize(prizes)
        ordered: list[Winner] = []
        for prize in prizes:
            ordered.extend(grouped.get(prize.name, []))
        listed = {id(w) for w in ordered}
        ordered.extend(w for w in self._winners if id(w) not in listed)

        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department Group", "Prize Name"])
                for winner in ordered:
                    writer.writerow([winner.name, winner.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

from __future__ import annotations

from lottery_system.models import Prize, User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager

MENU = """
==================== Team Building Lottery ====================
 1. Load participant list
 2. View department groups
 3. Load preset prizes
 4. Add custom prize
 5. View / reorder prizes
 6. Start lottery
 7. Show winners
 8. Export results (CSV txt)
 0. Exit
==============================================================="""


def ordinal(n: int) -> str:
    if 11 <= n % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Command-line interface controlling the whole lottery process."""

    def __init__(self, input_func=input, output=print) -> None:
        self._input = input_func
        self._out = output
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.list_loaded = False
        self.lottery_run = False
        self.last_prizes: list[Prize] = []

    # ----------------------------------------------------------- helpers
    def _print(self, text: str = "") -> None:
        self._out(text)

    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    def _group_names(self) -> list[str]:
        return sorted(self.list_manager.get_group_info(self.users).keys())

    # -------------------------------------------------------------- main
    def run(self) -> None:
        self._print("Welcome to the Team Building Smart Lottery System.")
        try:
            while True:
                self._print(MENU)
                choice = self._ask("Select an option: ").strip()
                if choice == "1":
                    self.load_list()
                elif choice == "2":
                    self.view_groups()
                elif choice == "3":
                    self.load_presets()
                elif choice == "4":
                    self.add_prize()
                elif choice == "5":
                    self.view_and_reorder_prizes()
                elif choice == "6":
                    self.start_lottery()
                elif choice == "7":
                    self.show_winners()
                elif choice == "8":
                    self.export_results()
                elif choice == "0":
                    self._print("Goodbye.")
                    break
                else:
                    self._print("Invalid option, please try again.")
        except (EOFError, KeyboardInterrupt):
            self._print("\nExiting.")

    # ------------------------------------------------------- list module
    def load_list(self) -> None:
        self._print("1) Built-in default list")
        self._print("2) Custom txt file (each line: Name,Department Group)")
        choice = self._ask("Select source: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip().strip('"')
            users, errors = self.list_manager.load_custom_list(path)
        else:
            self._print("Invalid source selection.")
            return

        for error in errors:
            self._print(error)

        valid, duplicates = self.list_manager.validate_users(users)
        for dup in duplicates:
            self._print(f"Duplicate Name: '{dup.name}' ({dup.department}) - repeated entry")

        if errors or duplicates:
            self._print("1) Remove invalid / duplicate data and continue")
            self._print("2) Terminate upload")
            action = self._ask("Select action: ").strip()
            if action != "1":
                self._print("List upload terminated. No changes made.")
                return

        if not valid:
            self._print("No valid participants found. List not loaded.")
            return

        self.users = valid
        self.list_loaded = True
        self._print(f"Loaded {len(valid)} valid participants.")
        self.view_groups()

    def view_groups(self) -> None:
        if not self.list_loaded:
            self._print("No participant list loaded yet.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._print("Department groups:")
        for group in sorted(info):
            self._print(f"  - {group}: {info[group]} people")

    # ----------------------------------------------------- prize module
    def load_presets(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for error in errors:
            self._print(error)
        self._print("Preset prizes loaded.")

    def _parse_groups(self, raw: str) -> list[str] | None:
        groups_list = self._group_names()
        tokens = [t.strip() for t in raw.replace("\uff0c", ",").split(",") if t.strip()]
        result: list[str] = []
        for token in tokens:
            if token.isdigit() and groups_list and 1 <= int(token) <= len(groups_list):
                group = groups_list[int(token) - 1]
            elif token.isdigit() and groups_list:
                self._print(f"Invalid group number: {token}")
                return None
            else:
                group = token
            if group not in result:
                result.append(group)
        return result

    def add_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        if not name:
            self._print("Prize name must not be empty.")
            return
        qty_raw = self._ask("Quantity: ").strip()
        if not qty_raw.isdigit() or int(qty_raw) <= 0:
            self._print("Quantity must be a positive integer.")
            return
        quantity = int(qty_raw)

        groups_list = self._group_names()
        if groups_list:
            self._print("Department groups:")
            for idx, group in enumerate(groups_list, 1):
                self._print(f"  {idx}. {group}")
        raw_groups = self._ask(
            "Avoid groups (comma-separated numbers or names, blank for none): "
        )
        avoid = self._parse_groups(raw_groups)
        if avoid is None:
            return

        drawer = self._ask("Associated lottery person (blank for none): ").strip() or None

        try:
            self.prize_manager.add_prize(name, quantity, avoid, drawer)
        except ValueError as exc:
            self._print(f"Error: {exc}")
            return
        self._print(f"Prize '{name}' added.")

    def view_and_reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._print_prizes(prizes)
        raw = self._ask("Enter new order as serial numbers (e.g. '2 1 3'), blank to keep: ").strip()
        if not raw:
            return
        try:
            order = [int(t) for t in raw.replace(",", " ").split()]
        except ValueError:
            self._print("Invalid input: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Prize sequence updated.")
            self._print_prizes(self.prize_manager.get_prizes())
        else:
            self._print("Invalid order: must be a permutation of all serial numbers.")

    def _print_prizes(self, prizes: list[Prize]) -> None:
        self._print("Lottery sequence:")
        for idx, prize in enumerate(prizes, 1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self._print(
                f"  {idx}. {prize.name} | quantity: {prize.quantity} | "
                f"avoid groups: {avoid} | lottery person: {drawer}"
            )

    # ------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.list_loaded:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure prizes first.")
            return

        self.result_manager = ResultManager()
        self.last_prizes = prizes
        lottery = LotteryManager(self.users)
        self.lottery_run = True

        total_rounds = len(prizes)
        for idx, prize in enumerate(prizes, 1):
            self._print("")
            self._print(f"----- Round {idx}/{total_rounds}: {prize.name} -----")
            self._print(f"Quantity: {prize.quantity}")
            self._print(f"Rule: {lottery.describe_rule(prize)}")

            lottery.start_round(prize)
            status = self._run_round(lottery, prize)
            winners = lottery.end_round()

            self._print(f"Round results ({prize.name}):")
            if winners:
                for order, user in enumerate(winners, 1):
                    self._print(f"  {order}. {user.name} ({user.department})")
            else:
                self._print("  No winners in this round.")
            if status == "terminated":
                self._print("Round terminated by operator; moving to next round.")
            elif status == "exhausted":
                self._print("No eligible participants remain for this prize; round skipped.")

        self._print("")
        self._print("================ Lottery finished ================")
        self.show_winners()

    def _run_round(self, lottery: LotteryManager, prize: Prize) -> str:
        drawn = 0
        while drawn < prize.quantity:
            nxt = ordinal(drawn + 1)
            cmd = self._ask(
                f"[Enter] draw the {nxt} winner | 'pause' | 'terminate': "
            ).strip().lower()

            if cmd == "terminate":
                return "terminated"
            if cmd == "pause":
                self._print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    sub = self._ask("(paused) > ").strip().lower()
                    if sub == "continue":
                        self._print("Lottery resumed.")
                        break
                    if sub == "terminate":
                        return "terminated"
                    self._print("Unknown command. Use 'continue' or 'terminate'.")
                continue
            if cmd != "":
                self._print("Unknown command.")
                continue

            self._print(f"Currently drawing the {nxt} winner...")
            user = lottery.draw_next()
            if user is None:
                return "exhausted"
            drawn += 1
            self.result_manager.add_winner(user, prize)
            self._print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        return "completed"

    # ------------------------------------------------------- results
    def show_winners(self) -> None:
        if not self.lottery_run:
            self._print("No lottery results yet.")
            return
        prizes = self.last_prizes
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self._print("")
        self._print("Complete winner list:")
        self._print(f"{'Name':<20} {'Department':<28} Prize")
        total = 0
        for prize in prizes:
            for winner in grouped.get(prize.name, []):
                total += 1
                self._print(f"{winner.name:<20} {winner.department:<28} {winner.prize_name}")
        if total == 0:
            self._print("  (no winners)")

        if not self.lottery_run:

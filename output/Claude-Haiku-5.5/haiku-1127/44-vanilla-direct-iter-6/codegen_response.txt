"""Data loading utilities for the Beijing Subway network.

The network dataset lives in ``subway_data.json`` next to this module. The
file has the structure::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str, "distance": int, "line": str}]}

Distances are expressed in meters.
"""

import json
import os

# Absolute path of src/subway_data.json, valid regardless of the working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway dataset from a JSON file.

    Args:
        path: Path to the JSON data file. Defaults to ``DATA_FILE_PATH``.

    Returns:
        dict: The JSON content of the file with ``stations`` and
        ``connections`` keys.

    Raises:
        FileNotFoundError: If the file does not exist.
        ValueError: If the file is not valid JSON or lacks required keys.
    """
    file_path = DATA_FILE_PATH if path is None else os.fspath(path)
    try:
        with open(file_path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise FileNotFoundError(f"Subway data file not found: {file_path}") from exc
    except json.JSONDecodeError as exc:
        raise ValueError(f"Subway data file is not valid JSON: {exc}") from exc

    if not isinstance(data, dict) or "stations" not in data or "connections" not in data:
        raise ValueError("Subway data must contain 'stations' and 'connections' keys")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write the subway dataset back to a JSON file.

    Args:
        data: The dataset dictionary.
        path: Destination file path. Defaults to ``DATA_FILE_PATH``.

    Raises:
        OSError: If the file cannot be written.
    """
    file_path = DATA_FILE_PATH if path is None else os.fspath(path)
    with open(file_path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")

"""Indexed binary min-heap priority queue used by Dijkstra's algorithm.

Each item is stored at most once. Keys can be decreased (or increased) in
O(log n) time via ``put``, which is what Dijkstra's relaxation step needs.
Keys may be any mutually comparable objects, such as numbers or tuples.
"""


class MinPQ:
    """A min priority queue backed by a binary heap with an item index."""

    def __init__(self):
        self._heap = []   # heap-ordered list of items
        self._pos = {}    # item -> index of the item in self._heap
        self._key = {}    # item -> current priority key

    def __len__(self):
        return len(self._heap)

    def __contains__(self, item):
        return item in self._pos

    def size(self):
        """Return the number of items in the queue."""
        return len(self._heap)

    def is_empty(self):
        """Return True if the queue contains no items."""
        return not self._heap

    def put(self, item, key):
        """Insert an item, or update its key if it is already queued.

        Args:
            item: Any hashable item.
            key: Priority key; smaller keys are served first.
        """
        if item in self._pos:
            index = self._pos[item]
            old_key = self._key[item]
            self._key[item] = key
            if key < old_key:
                self._swim(index)
            elif old_key < key:
                self._sink(index)
            return
        self._key[item] = key
        self._heap.append(item)
        self._pos[item] = len(self._heap) - 1
        self._swim(len(self._heap) - 1)

    def min_item(self):
        """Return the item with the smallest key without removing it."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._heap[0]

    def min_key(self):
        """Return the smallest key without removing its item."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._key[self._heap[0]]

    def del_min(self):
        """Remove and return the ``(item, key)`` pair with the smallest key.

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("priority queue is empty")
        item = self._heap[0]
        key = self._key[item]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._pos[last] = 0
            self._sink(0)
        del self._pos[item]
        del self._key[item]
        return item, key

    # ----- heap helpers -------------------------------------------------

    def _less(self, i, j):
        """Compare the keys stored at heap positions i and j."""
        return self._key[self._heap[i]] < self._key[self._heap[j]]

    def _swap(self, i, j):
        """Swap two heap positions and keep the position index up to date."""
        self._heap[i], self._heap[j] = self._heap[j], self._heap[i]
        self._pos[self._heap[i]] = i
        self._pos[self._heap[j]] = j

    def _swim(self, k):
        """Move the element at position k up until the heap property holds."""
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._swap(k, parent)
                k = parent
            else:
                break

    def _sink(self, k):
        """Move the element at position k down until the heap property holds."""
        n = len(self._heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and self._less(right, left):
                child = right
            if self._less(child, k):
                self._swap(k, child)
                k = child
            else:
                break

"""Graph data models for the Beijing Subway network.

The network is a weighted undirected graph. Station names are mapped to
integer station numbers by ``SubwayNetwork``; the ``Graph`` itself works only
with station numbers, as required by the interface.
"""


class Edge:
    """An undirected weighted edge between two station numbers.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance between the stations in meters.
        line: Name of the subway line operating this connection.
        weight: Alias of ``distance`` used as the graph weight.
    """

    __slots__ = ("source", "dest", "_distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self._distance = self._validate_distance(distance)
        self.line = line

    @staticmethod
    def _validate_distance(distance):
        """Ensure the distance is a non-negative number."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("distance must be a number of meters")
        if distance < 0:
            raise ValueError("distance must be non-negative")
        return distance

    @property
    def distance(self):
        """Distance between the endpoints in meters."""
        return self._distance

    @distance.setter
    def distance(self, value):
        self._distance = self._validate_distance(value)

    @property
    def weight(self):
        """Graph weight of the edge (equal to the distance in meters)."""
        return self._distance

    @weight.setter
    def weight(self, value):
        self._distance = self._validate_distance(value)

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``.

        Raises:
            ValueError: If ``vertex`` is not an endpoint of this edge.
        """
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"vertex {vertex} is not an endpoint of this edge")

    def __repr__(self):
        return (f"Edge({self.source}, {self.dest}, distance={self._distance}, "
                f"line={self.line!r})")


class Graph:
    """Weighted undirected graph with adjacency lists.

    Station numbers are assigned by the caller. The graph also keeps a
    ``station_lines`` mapping from station name to the lines serving it, which
    is used to identify transfer stations.
    """

    def __init__(self):
        self._adj = {}          # station number -> list[Edge]
        self._edges = []        # all edges, in insertion order
        self.station_lines = {}  # station name -> list[str]

    # ----- station metadata ---------------------------------------------

    def set_station_lines(self, station_name, lines):
        """Record the subway lines serving a station."""
        self.station_lines[station_name] = list(lines)

    # ----- vertices and edges -------------------------------------------

    def add_vertex(self, v):
        """Add a station number to the graph if it is not already present."""
        self._adj.setdefault(v, [])

    def has_vertex(self, v):
        """Return True if the station number is part of the graph."""
        return v in self._adj

    def add_edge(self, edge):
        """Add an undirected edge, creating its endpoints if necessary."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adj[edge.source].append(edge)
        self._adj[edge.dest].append(edge)
        self._edges.append(edge)

    def adj(self, v):
        """Return a copy of the list of edges incident to station ``v``."""
        return list(self._adj.get(v, []))

    def vertices(self):
        """Return the list of station numbers in the graph."""
        return list(self._adj.keys())

    def edges(self):
        """Return the list of all edges."""
        return list(self._edges)

    def num_vertices(self):
        """Return the number of stations in the graph."""
        return len(self._adj)

    def num_edges(self):
        """Return the number of edges in the graph."""
        return len(self._edges)

    def find_edge(self, u, v, line=None):
        """Find an edge between ``u`` and ``v`` (optionally on a given line).

        Returns:
            Edge or None: The first matching edge, or None if none exists.
        """
        for edge in self._adj.get(u, []):
            if edge.other(u) == v and (line is None or edge.line == line):
                return edge
        return None

    def remove_edge(self, u, v, line=None):
        """Remove the first edge between ``u`` and ``v`` matching ``line``.

        Returns:
            bool: True if an edge was removed, False otherwise.
        """
        edge = self.find_edge(u, v, line)
        if edge is None:
            return False
        self._adj[edge.source].remove(edge)
        if edge.dest != edge.source:
            self._adj[edge.dest].remove(edge)
        self._edges.remove(edge)
        return True


class SubwayNetwork:
    """Subway network combining the graph with name/number mappings.

    Attributes:
        graph: The ``Graph`` holding the station connections.
        station_to_number: Mapping from station name to station number.
        number_to_station: Mapping from station number to station name.
    """

    def __init__(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}

    def add_station(self, name, lines=()):
        """Add a station if needed and merge the given lines into it.

        Returns:
            int: The station number of the (new or existing) station.
        """
        if name not in self.station_to_number:
            number = len(self.number_to_station)
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
            self.graph.set_station_lines(name, [])
        current = self.graph.station_lines[name]
        for line in lines:
            if line not in current:
                current.append(line)
        return self.station_to_number[name]

    def has_station(self, name):
        """Return True if a station with this exact name exists."""
        return name in self.station_to_number

    def number_of(self, name):
        """Return the station number for a station name."""
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the station name for a station number."""
        return self.number_to_station[number]

    def station_names(self):
        """Return all station names sorted alphabetically."""
        return sorted(self.station_to_number)

    def lines_of(self, name):
        """Return the lines serving a station."""
        return list(self.graph.station_lines.get(name, []))

    def lines(self):
        """Return all subway line names in sorted order."""
        found = set()
        for line_list in self.graph.station_lines.values():
            found.update(line_list)
        return sorted(found)

    def is_transfer_station(self, name):
        """Return True if the station is served by more than one line."""
        return len(self.lines_of(name)) > 1

    def stations_on_line(self, line):
        """Return the names of stations served by ``line``, sorted by number."""
        result = []
        for number in sorted(self.number_to_station):
            name = self.number_to_station[number]
            if line in self.graph.station_lines.get(name, []):
                result.append(name)
        return result


def build_network(data):
    """Build a ``SubwayNetwork`` from the parsed subway dataset.

    Stations are numbered in the order of first appearance. Duplicate
    connections between the same stations on the same line (for example the
    two directions of one track segment) are merged into a single undirected
    edge.

    Args:
        data: dict with ``stations`` and ``connections`` lists.

    Returns:
        SubwayNetwork: The constructed network.
    """
    network = SubwayNetwork()
    for station in data.get("stations", []):
        network.add_station(station["name"], station.get("lines", []))

    for conn in data.get("connections", []):
        line = conn["line"]
        src = network.add_station(conn["source"], [line])
        dst = network.add_station(conn["dest"], [line])
        if network.graph.find_edge(src, dst, line) is None:
            network.graph.add_edge(Edge(src, dst, conn["distance"], line))
    return network

"""Fare and travel-time calculation for the Beijing Subway.

Fare structure (distances in meters, left-closed / right-open intervals):
    [0, 6000)          -> 3 yuan
    [6000, 12000)      -> 4 yuan
    [12000, 22000)     -> 5 yuan
    [22000, 32000)     -> 6 yuan
    [32000, ...)       -> 7 yuan, plus 1 yuan for every further 20 km
                          (32-52 km: 7 yuan, 52-72 km: 8 yuan, ...)
Fractional distances are rounded up to the next whole meter before the tier
lookup, so a boundary value belongs to the next higher tier.
"""

import math

# (exclusive upper bound in meters, fare in yuan) for the base and progressive tiers.
FARE_TIERS = (
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
)

EXTENSION_START_M = 32000      # start of the extended tier
EXTENSION_STEP_M = 20000       # every 20 km beyond the extension start...
EXTENSION_STEP_FARE = 1        # ...adds 1 yuan
EXTENSION_BASE_FARE = 7        # fare for the first extended segment (32-52 km)

# Travel time model
MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def Sum_money(distance):
    """Compute the subway fare for a travelled distance.

    Args:
        distance: Travelled distance in meters (int or float). Fractional
            values are rounded up to the next whole meter.

    Returns:
        int: The fare in yuan.

    Raises:
        TypeError: If ``distance`` is not a real number.
        ValueError: If ``distance`` is negative or NaN.
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of meters")
    if math.isnan(distance):
        raise ValueError("distance must not be NaN")
    if distance < 0:
        raise ValueError("distance must be non-negative")

    meters = math.ceil(distance)
    for upper_bound, fare in FARE_TIERS:
        if meters < upper_bound:
            return fare
    extra_steps = (meters - EXTENSION_START_M) // EXTENSION_STEP_M
    return EXTENSION_BASE_FARE + extra_steps * EXTENSION_STEP_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimate the travel time in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of line transfers.

    Returns:
        float: Estimated travel time in minutes (about 1.5 min per km plus
        about 5 min per transfer).

    Raises:
        ValueError: If the distance or transfer count is negative.
    """
    if distance < 0:
        raise ValueError("distance must be non-negative")
    if transfer_count < 0:
        raise ValueError("transfer_count must be non-negative")
    kilometers = distance / 1000.0
    return kilometers * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

<file path="src/pathfinding.py">
"""Path calculation for the Beijing Subway network.

The module implements Dijkstra's algorithm on an expanded state space. A state
is a pair ``(station_number, arriving_line)``: this lets the search count line
transfers exactly. The priority queue is the indexed binary heap ``MinPQ``,
giving O((V + E) log V) behaviour.

Supported query modes:
    shortest_distance -- minimise total travelled distance (meters)
    least_transfer    -- minimise transfers, then distance
    comprehensive     -- minimise distance + transfer_penalty * transfers
Optional settings:
    max_transfers     -- paths with more transfers are discarded
    preferred_lines   -- edges on other lines receive a penalty in the search
                         cost (the reported distance stays the real distance)
"""

import math
from dataclasses import dataclass, field

from min_pq import MinPQ

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Cost (in meters) of each transfer in comprehensive mode.
DEFAULT_TRANSFER_PENALTY_M = 1000.0
# Extra search cost (in meters) for riding a line outside the preferred list.
DEFAULT_PREFERENCE_PENALTY_M = 500.0


class PathNotFoundError(Exception):
    """Raised when no path satisfies the query constraints."""


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of ``VALID_MODES``.
        max_transfers: Maximum allowed transfers, or None for no limit.
        preferred_lines: List of preferred line names (empty means no preference).
        transfer_penalty: Cost per transfer in comprehensive mode, in meters.
        preference_penalty: Extra search cost per non-preferred edge, in meters.
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None,
                 transfer_penalty=DEFAULT_TRANSFER_PENALTY_M,
                 preference_penalty=DEFAULT_PREFERENCE_PENALTY_M):
        if mode not in VALID_MODES:
            raise ValueError(f"mode must be one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if (isinstance(max_transfers, bool) or not isinstance(max_transfers, int)
                    or max_transfers < 0):
                raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])
        self.transfer_penalty = float(transfer_penalty)
        self.preference_penalty = float(preference_penalty)

    def penalty_for_edge(self, edge):
        """Return the extra search cost of riding ``edge``."""
        if not self.preferred_lines or edge.line in self.preferred_lines:
            return 0.0
        return self.preference_penalty

    def key(self, cost, transfers):
        """Build the priority key used by Dijkstra for a search state.

        Every component is non-decreasing along a path, so the keys are
        monotone and Dijkstra's algorithm remains correct.
        """
        if self.mode == MODE_SHORTEST_DISTANCE:
            return (cost, transfers)
        if self.mode == MODE_LEAST_TRANSFER:
            return (transfers, cost)
        return (cost + self.transfer_penalty * transfers, transfers)


class _StateInfo:
    """Bookkeeping for one search state (station, arriving line)."""

    __slots__ = ("cost", "distance", "transfers", "key", "prev", "edge")

    def __init__(self, cost, distance, transfers, key, prev, edge):
        self.cost = cost              # search cost including penalties
        self.distance = distance      # real travelled distance in meters
        self.transfers = transfers    # number of line changes so far
        self.key = key                # priority key
        self.prev = prev              # predecessor state, or None
        self.edge = edge              # edge used to reach this state, or None


class ShortestPath:
    """Single-source shortest paths computed with Dijkstra's algorithm.

    Attributes:
        dist_to: dict station number -> real distance (meters) of the best path.
        transfers_to: dict station number -> transfer count of the best path.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"unknown source station number: {source}")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self.transfers_to = {}
        self._best_state = {}   # station number -> best finalized state
        self._info = {}         # state -> _StateInfo
        self._run()

    def _run(self):
        """Execute Dijkstra's algorithm over (station, line) states."""
        cfg = self.config
        start = (self.source, None)
        start_info = _StateInfo(0.0, 0, 0, cfg.key(0.0, 0), None, None)
        self._info[start] = start_info

        pq = MinPQ()
        pq.put(start, start_info.key)
        while not pq.is_empty():
            state, _ = pq.del_min()
            info = self._info[state]
            node, line = state

            # The first state popped for a station has the minimum key for it.
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = info.distance
                self.transfers_to[node] = info.transfers

            for edge in self.graph.adj(node):
                transfers = info.transfers
                if line is not None and edge.line != line:
                    transfers += 1  # changing lines at this station
                if cfg.max_transfers is not None and transfers > cfg.max_transfers:
                    continue

                cost = info.cost + edge.distance + cfg.penalty_for_edge(edge)
                distance = info.distance + edge.distance
                key = cfg.key(cost, transfers)
                next_state = (edge.other(node), edge.line)

                old = self._info.get(next_state)
                if old is not None and old.key <= key:
                    continue
                self._info[next_state] = _StateInfo(cost, distance, transfers,
                                                    key, state, edge)
                pq.put(next_state, key)

    def has_path_to(self, target):
        """Return True if station ``target`` is reachable under the constraints."""
        return target in self._best_state

    def distance_to(self, target):
        """Return the travelled distance in meters, or ``math.inf`` if unreachable."""
        if target not in self._best_state:
            return math.inf
        return self.dist_to[target]

    def get_transfer_count(self, target):
        """Return the number of transfers on the best path, or -1 if unreachable."""
        return self.transfers_to.get(target, -1)

    def path_with_edges(self, target):
        """Return the station numbers and edges of the best path to ``target``.

        Returns:
            tuple: ``(stations, edges)`` where ``stations`` runs from the source
            to ``target`` and ``edges`` has one fewer element. Both lists are
            empty if the target is unreachable.
        """
        state = self._best_state.get(target)
        nodes, edges = [], []
        while state is not None:
            info = self._info[state]
            nodes.append(state[0])
            if info.edge is not None:
                edges.append(info.edge)
            state = info.prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        """Return the station numbers of the best path to ``target``."""
        return self.path_with_edges(target)[0]


@dataclass
class Route:
    """Result of a path query, with all information needed for reporting.

    Attributes:
        start: Starting station name.
        end: Ending station name.
        mode: Query mode used.
        stations: Ordered station names from start to end.
        edges: Edges traversed, one per consecutive station pair.
        segments: Line segments: list of ``{"line": str, "stations": [str]}``.
        transfer_stations: List of ``{"station", "from_line", "to_line"}``.
        lines: Unique line names in the order they are ridden.
        distance: Total distance in meters.
        transfer_count: Number of transfers.
    """

    start: str
    end: str
    mode: str
    stations: list = field(default_factory=list)
    edges: list = field(default_factory=list)
    segments: list = field(default_factory=list)
    transfer_stations: list = field(default_factory=list)
    lines: list = field(default_factory=list)
    distance: float = 0
    transfer_count: int = 0


def _build_segments(names, edges):
    """Group consecutive edges on the same line into segments."""
    segments = []
    if not edges:
        return segments
    current = {"line": edges[0].line, "stations": [names[0]]}
    for index, edge in enumerate(edges):
        if edge.line != current["line"]:
            segments.append(current)
            current = {"line": edge.line, "stations": [names[index]]}
        current["stations"].append(names[index + 1])
    segments.append(current)
    return segments


def plan_route(network, start, end, config=None):
    """Compute the route between two stations.

    Args:
        network: ``SubwayNetwork`` instance.
        start: Starting station name (must exist in the network).
        end: Ending station name (must exist in the network).
        config: ``PathfindingConfig`` or None for defaults.

    Returns:
        Route: The computed route.

    Raises:
        KeyError: If a station name is unknown.
        PathNotFoundError: If no path satisfies the constraints.
    """
    cfg = config if config is not None else PathfindingConfig()
    source = network.number_of(start)
    target = network.number_of(end)

    sp = ShortestPath(network.graph, source, cfg)
    if not sp.has_path_to(target):
        message = f"No route found from '{start}' to '{end}'"
        if cfg.max_transfers is not None:

            "station": segments[index]["stations"][0],
            "from_line": segments[index - 1]["line"],
            "to_line": segments[index]["line"],

    for segment in segments:
        if segment["line"] not in lines:
            lines.append(segment["line"])

        end=end,

"""Data integrity checks and data update helpers for the subway dataset.

All update helpers operate on the parsed dataset dictionary (the structure
returned by ``data_loader.load_subway_data``) and return it after modification.
Use ``data_loader.save_subway_data`` to persist the changes.
"""

from collections import defaultdict

# Lines that must be represented in the dataset.
REQUIRED_MAIN_LINES = (
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
)
MIN_STATIONS_PER_MAIN_LINE = 2
MAX_REASONABLE_DISTANCE_M = 8000


class IntegrityReport:
    """Collects errors and warnings found by ``check_data_integrity``."""

    def __init__(self):
        self.errors = []
        self.warnings = []

    @property
    def ok(self):
        """True when no errors were found (warnings are allowed)."""
        return not self.errors

    def add_error(self, message):
        self.errors.append(message)

    def add_warning(self, message):
        self.warnings.append(message)

    def render(self):
        """Return a human-readable text summary of the report."""
        lines = ["Data integrity check: " + ("PASSED" if self.ok else "FAILED")]
        lines.append(f"Errors: {len(self.errors)}")
        for message in self.errors:
            lines.append(f"  [ERROR] {message}")
        lines.append(f"Warnings: {len(self.warnings)}")
        for message in self.warnings:
            lines.append(f"  [WARN] {message}")
        return "\n".join(lines)


def _is_valid_distance(value):
    return (isinstance(value, (int, float)) and not isinstance(value, bool)
            and value > 0)


def check_data_integrity(data):
    """Run integrity checks on the subway dataset.

    Checks performed:
        * structure and station name uniqueness
        * station coverage of the main lines
        * connection references, distance accuracy and duplicates
        * bidirectional consistency of connections
        * network connectivity

    Args:
        data: The parsed dataset dictionary.

    Returns:
        IntegrityReport: The findings.
    """
    report = IntegrityReport()
    if not isinstance(data, dict):
        report.add_error("Dataset must be a JSON object")
        return report

    stations = data.get("stations")
    connections = data.get("connections")
    if not isinstance(stations, list):
        report.add_error("'stations' must be a list")
        stations = []
    if not isinstance(connections, list):
        report.add_error("'connections' must be a list")
        connections = []

    # ----- station data -------------------------------------------------
    station_lines = {}
    for index, station in enumerate(stations):
        if not isinstance(station, dict) or not station.get("name"):
            report.add_error(f"Station entry #{index} has no name")
            continue
        name = station["name"]
        if name in station_lines:
            report.add_error(f"Duplicate station name: {name}")
        lines = station.get("lines")
        if not isinstance(lines, list) or not lines:
            report.add_warning(f"Station '{name}' has no line affiliation")
            station_lines[name] = set()
        else:
            station_lines[name] = set(lines)

    # ----- connection data ----------------------------------------------
    distance_map = {}
    for index, conn in enumerate(connections):
        if not isinstance(conn, dict):
            report.add_error(f"Connection entry #{index} is not an object")
            continue
        missing = [k for k in ("source", "dest", "distance", "line") if k not in conn]
        if missing:
            report.add_error(f"Connection entry #{index} is missing fields: {missing}")
            continue
        src, dst, line, dist = conn["source"], conn["dest"], conn["line"], conn["distance"]
        label = f"{src} -> {dst} ({line})"
        if src not in station_lines or dst not in station_lines:
            report.add_error(f"Connection {label} references an unknown station")
            continue
        if not _is_valid_distance(dist):
            report.add_error(f"Connection {label} has an invalid distance: {dist!r}")
            continue
        if dist > MAX_REASONABLE_DISTANCE_M:
            report.add_warning(f"Connection {label} distance {dist} m looks too large")
        if line not in station_lines[src] or line not in station_lines[dst]:
            report.add_warning(f"Connection {label}: line not listed for both stations")
        key = (src, dst, line)
        if key in distance_map:
            report.add_error(f"Duplicate connection: {label}")
        distance_map[key] = dist

    # ----- bidirectional consistency ------------------------------------
    for (src, dst, line), dist in distance_map.items():
        reverse = (dst, src, line)
        if reverse not in distance_map:
            report.add_warning(f"Connection {src} -> {dst} ({line}) has no reverse entry")
        elif distance_map[reverse] != dist:
            report.add_error(
                f"Distance mismatch between {src} and {dst} on {line}: "
                f"{dist} m vs {distance_map[reverse]} m")

    # ----- line coverage ------------------------------------------------
    for main_line in REQUIRED_MAIN_LINES:
        count = sum(1 for lines in station_lines.values() if main_line in lines)
        if count == 0:
            report.add_error(f"Main line '{main_line}' has no stations in the dataset")
        elif count < MIN_STATIONS_PER_MAIN_LINE:
            report.add_warning(f"Main line '{main_line}' has only {count} station(s)")

    # ----- connectivity -------------------------------------------------
    adjacency = defaultdict(set)
    for src, dst, _ in distance_map:
        adjacency[src].add(dst)
        adjacency[dst].add(src)
    visited = set()
    components = 0
    for start in sorted(station_lines):
        if start in visited:
            continue
        components += 1
        stack = [start]
        visited.add(start)
        while stack:
            node = stack.pop()
            for neighbour in adjacency[node]:
                if neighbour not in visited:
                    visited.add(neighbour)
                    stack.append(neighbour)
    if components > 1:
        report.add_warning(f"The network is split into {components} disconnected groups")

    return report


def add_line(data, line_name, station_names, distances):
    """Add a new line with its ordered stations and inter-station distances.

    Existing stations are reused (and gain the new line); unknown stations are
    created. Connections are added in both directions.

    Args:
        data: The parsed dataset dictionary (modified in place).
        line_name: Name of the new line.
        station_names: Ordered list of station names on the line.
        distances: List of ``len(station_names) - 1`` distances in meters.

    Returns:
        dict: The modified dataset.

    Raises:
        ValueError: If the input is inconsistent or the line already exists.
    """
    line_name = " ".join(str(line_name).split())
    if not line_name:
        raise ValueError("line name must not be empty")
    if len(station_names) < 2:
        raise ValueError("a line needs at least two stations")
    if len(set(station_names)) != len(station_names):
        raise ValueError("a station appears more than once on the line")
    if len(distances) != len(station_names) - 1:
        raise ValueError("expected one distance per consecutive station pair")
    for dist in distances:
        if isinstance(dist, bool) or not isinstance(dist, int) or dist <= 0:
            raise ValueError(f"invalid distance: {dist!r}")
    if any(conn.get("line") == line_name for conn in data["connections"]):
        raise ValueError(f"line already exists: {line_name}")

    index = {station["name"]: station for station in data["stations"]}
    for name in station_names:
        station = index.get(name)
        if station is None:
            station = {"name": name, "lines": [line_name]}
            data["stations"].append(station)
            index[name] = station
        elif line_name not in station["lines"]:
            station["lines"].append(line_name)

    pairs = zip(station_names, station_names[1:])
    for (first, second), dist in zip(pairs, distances):
        data["connections"].append(
            {"source": first, "dest": second, "distance": dist, "line": line_name})
        data["connections"].append(
            {"source": second, "dest": first, "distance": dist, "line": line_name})
    return data


def modify_station(data, name, new_name=None, lines=None):
    """Rename a station and/or replace its line list.

    Args:
        data: The parsed dataset dictionary (modified in place).
        name: Current station name.
        new_name: Optional new name; connections are updated accordingly.
        lines: Optional replacement list of lines.

    Returns:
        dict: The modified dataset.

    Raises:
        KeyError: If the station does not exist.
        ValueError: If the new name is already used.
    """
    station = next((s for s in data["stations"] if s["name"] == name), None)
    if station is None:
        raise KeyError(f"unknown station: {name}")
    if new_name and new_name != name:
        if any(s["name"] == new_name for s in data["stations"]):
            raise ValueError(f"station already exists: {new_name}")
        station["name"] = new_name
        for conn in data["connections"]:
            if conn["source"] == name:
                conn["source"] = new_name
            if conn["dest"] == name:
                conn["dest"] = new_name
    if lines is not None:
        station["lines"] = list(lines)
    return data


def update_distance(data, source, dest, distance, line=None):
    """Update the distance between two adjacent stations (both directions).

    Args:
        data: The parsed dataset dictionary (modified in place).
        source: One station name.
        dest: The adjacent station name.
        distance: New distance in meters (positive integer).
        line: Optional line filter when several lines connect the pair.

    Returns:
        int: Number of connection entries updated.

    Raises:
        ValueError: If the distance is invalid.
        KeyError: If no matching connection exists.
    """
    if isinstance(distance, bool) or not isinstance(distance, int) or distance <= 0:
        raise ValueError("distance must be a positive integer number of meters")
    count = 0
    for conn in data["connections"]:
        if line is not None and conn["line"] != line:
            continue
        forward = conn["source"] == source and conn["dest"] == dest
        backward = conn["source"] == dest and conn["dest"] == source
        if forward or backward:
            conn["distance"] = distance
            count += 1
    if count == 0:
        raise KeyError(f"no connection between {source} and {dest}")
    return count

def format_distance(meters):
    """Format a distance: meters for short trips, kilometers for long trips."""
    if meters < 1000:
        return f"{round(meters)} m"
    return f"{meters / 1000:.2f} km ({round(meters)} m)"


def build_summary(route):
    """Extract the key figures of a route.

    Returns:
        dict with total distance (m and km), fare, transfer count and
        estimated time in minutes.
    """
    fare = Sum_money(route.distance)
    estimated = calculate_estimated_time(route.distance, route.transfer_count)
    return {
        "total_distance_m": route.distance,
        "total_distance_km": round(route.distance / 1000.0, 3),
        "fare": fare,
        "transfer_count": route.transfer_count,
        "estimated_time_minutes": round(estimated, 1),
    }


def build_report_data(route, query_info):
    """Build the structured report used by both text and JSON renderers.

    Args:
        route: ``pathfinding.Route`` instance.
        query_info: dict with ``mode``, ``max_transfers`` and ``line_preference``.

    Returns:
        dict: Report content.
    """
    summary = build_summary(route)
    return {
        "query": {
            "start": route.start,
            "end": route.end,
            "mode": query_info.get("mode", route.mode),
            "max_transfers": query_info.get("max_transfers"),
            "line_preference": query_info.get("line_preference"),
        },
        "path": list(route.stations),
        "distance": {
            "meters": route.distance,
            "kilometers": summary["total_distance_km"],
            "display": format_distance(route.distance),
        },
        "fare": summary["fare"],
        "estimated_time_minutes": summary["estimated_time_minutes"],
        "transfer_count": route.transfer_count,
        "lines": list(route.lines),
        "segments": [{"line": seg["line"], "stations": list(seg["stations"])}
                     for seg in route.segments],
        "transfers": [dict(item) for item in route.transfer_stations],
    }

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import sys

from data_loader import load_subway_data

from models import build_network

    VALID_MODES,
    PathfindingConfig,
    PathNotFoundError,
    plan_route,
)

EXIT_COMMANDS = ("quit", "exit", "q")
QUERY_PROMPT = "Enter query as 'Starting Station; Ending Station' (or 'quit' to exit): "


class QueryError(Exception):
    """Base class for user-facing query errors."""


class InputFormatError(QueryError):
    """The query text is not in 'Starting Station; Ending Station' form."""


class StationNotFoundError(QueryError):
    """A station name does not match any station in the network."""


class QueryLogicError(QueryError):
    """The query is logically invalid (for example identical endpoints)."""


def _normalize_key(name):
    """Normalize a station name for loose matching."""
    text = unicodedata.normalize("NFKC", name)
    text = " ".join(text.split()).casefold()
    if len(text) > 1 and text.endswith("站"):
        text = text[:-1].rstrip()
    return text


def _clean(text):
    """Trim and collapse whitespace, applying NFKC normalization."""
    return " ".join(unicodedata.normalize("NFKC", text).split())


class StationIndex:
    """Resolves user-typed station names to official network station names."""

    def __init__(self, network):
        self._exact = set(network.station_to_number)
        self._by_key = {}
        for name in network.station_names():
            self._by_key.setdefault(_normalize_key(name), name)

    def resolve(self, raw):
        """Return the official station name for ``raw``, or None if unknown."""
        cleaned = _clean(raw)
        if cleaned in self._exact:
            return cleaned
        return self._by_key.get(_normalize_key(cleaned))


def parse_query(text):
    """Split 'Starting Station; Ending Station' into two raw names.

    Raises:
        InputFormatError: If the separator is missing, repeated, or a name is empty.
    """
    message = "Input format incorrect, please use 'Starting Station; Ending Station' format"
    normalized = text.replace("；", ";")
    if normalized.count(";") != 1:
        raise InputFormatError(message)
    start, end = (part.strip() for part in normalized.split(";"))
    if not start or not end:
        raise InputFormatError(message)
    return start, end


def validate_query(raw_start, raw_end, index):
    """Validate station names and the logical constraints of a query.

    Returns:
        tuple: ``(start, end)`` official station names.

    Raises:
        StationNotFoundError: If a station does not exist.
        QueryLogicError: If the start and end stations are the same.
    """
    start = index.resolve(raw_start)
    if start is None:
        raise StationNotFoundError(
            f"Starting station '{raw_start}' does not exist, please check the station name")
    end = index.resolve(raw_end)
    if end is None:
        raise StationNotFoundError(
            f"Ending station '{raw_end}' does not exist, please check the station name")
    if start == end:
        raise QueryLogicError("Starting station cannot be the same as the ending station")
    return start, end


def parse_line_preference(text, known_lines):
    """Parse a comma-separated line preference string.

    Returns:
        tuple: ``(preferred_lines, unknown_tokens)``.
    """
    if not text:
        return [], []
    lookup = {line.casefold(): line for line in known_lines}
    preferred, unknown = [], []
    for token in text.split(","):
        token = _clean(token)
        if not token:
            continue
        canonical = lookup.get(token.casefold())
        if canonical is None:
            unknown.append(token)
            preferred.append(token)
        else:
            preferred.append(canonical)
    return preferred, unknown


def process_query(text, network, index, config, query_info, output_format):
    """Run one query end to end and return the rendered report text.

    Raises:
        QueryError: For input, station, path or fare errors.
    """
    raw_start, raw_end = parse_query(text)
    start, end = validate_query(raw_start, raw_end, index)

    try:
        route = plan_route(network, start, end, config)
    except PathNotFoundError as exc:
        raise QueryError(str(exc)) from exc
    except Exception as exc:  # defensive: unexpected path calculation failure
        raise QueryError(f"Path calculation failed: {exc}") from exc

    try:
        report = build_report_data(route, query_info)
    except (TypeError, ValueError) as exc:
        raise QueryError(f"Fare calculation failed: {exc}") from exc

    return render_json(report) if output_format == "json" else render_text(report)


def run_interactive(network, index, config, query_info, output_format, output_file):
    """Run the continuous query loop until the user exits.

    Returns:
        int: Process exit code.
    """
    while True:
        try:
            line = input(QUERY_PROMPT)
        except (EOFError, KeyboardInterrupt):
            print()
            return 0

        text = line.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            return 0

        try:
            content = process_query(text, network, index, config,
                                    query_info, output_format)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # keep the session alive on unexpected errors
            print(f"Error: unexpected failure while answering the query: {exc}")
            continue

        print(content)
        if output_file:
            try:
                export_report(content, output_file)
                print(f"Result saved to {output_file}")
            except OSError as exc:
                print(f"Error: could not write output file: {exc}")
        print()


def parse_args(argv=None):

    parser = argparse.ArgumentParser(

    Returns:

    return run_interactive(network, index, config, query_info,

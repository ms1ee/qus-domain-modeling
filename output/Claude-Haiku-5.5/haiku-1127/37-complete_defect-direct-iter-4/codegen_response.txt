from __future__ import annotations

from lottery_system.models import User

DEFAULT_LIST_LINES: list[str] = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Wang Wu,Product Department",
    "Zhao Liu,Product Department",
    "Sun Qi,Marketing Department",
    "Zhou Ba,Marketing Department",
    "Wu Jiu,Operations Department",
    "Zheng Shi,Operations Department",
    "Chen Qixian,Cheng Ru Group",
    "Jin Duo,Cheng Ru Group",
    "Ma Jiaqi,Cheng Ru Group",
    "Xu Ming,Cheng Ru Group",
    "Li Bin,Human Resources Department",
    "Hu Hong,Finance Department",
    "Guo Lei,Finance Department",
    "He Fang,Technology Department",
]


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it. Returns (users, errors)."""
        try:
            with open(path, "rb") as f:
                raw = f.read()
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        text: str | None = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                text = raw.decode(encoding)
                break
            except UnicodeDecodeError:
                continue
        if text is None:
            return [], [f"File Error: '{path}' is neither UTF-8 nor GBK text"]

        return self._parse_lines(text.splitlines())

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Returns (users, errors)."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            name = parts[0].strip()
            department = parts[1].strip()
            if not name or not department:
                errors.append(
                    f"Line {line_no}: Format Error - name and department group must not be empty"
                )
                continue
            users.append(User(name=name, department=department))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Remove duplicate names. The first occurrence is kept as valid;
        later occurrences are returned as duplicates."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group -> number of people."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from lottery_system.models import Prize

DEFAULT_PRIZES: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 10, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 5, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the ordered list of prizes for a lottery session."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"Cannot add preset prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int):
            raise ValueError("quantity must be an integer")
        if quantity < 1:
            raise ValueError("quantity must be at least 1")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"prize '{clean_name}' already exists")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer else None
        if drawer == "":
            drawer = None

        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order.
        Returns True on success, False if the order is not a valid permutation."""
        n = len(self._prizes)
        if not isinstance(order, list) or len(order) != n:
            return False
        if any(isinstance(i, bool) or not isinstance(i, int) for i in order):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

"""Lottery rule engine: eligibility pools, fallback protection and draws."""
from __future__ import annotations

import random

from lottery_system.models import Prize, User

CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"


def _norm(text: str) -> str:
    return " ".join(text.split()).casefold()


def _is_drawer(prize: Prize, drawer_name: str) -> bool:
    return prize.associated_drawer is not None and _norm(prize.associated_drawer) == _norm(drawer_name)


def _is_cheng_ru(user: User) -> bool:
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


class LotteryManager:
    """Runs lottery draws for a participant list while enforcing all rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winners: list[User] = []
        self._winner_names: set[str] = set()
        # Results of the most recent rounds drawn by Chen Qixian / Jin Duo:
        # True means the round produced at least one Cheng Ru Group winner.
        self.last_two_draw_results: list[bool] = []
        self._round_has_cheng_ru: bool = False

    @property
    def winners(self) -> list[User]:
        """Winners in the order they were drawn."""
        return list(self._winners)

    def is_fallback_active(self) -> bool:
        """Fallback (Cheng Ru protection) is active when the last two
        Chen Qixian / Jin Duo rounds both had no Cheng Ru Group winners."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the candidates eligible for the next draw of `prize`."""
        # Basic filtering: exclude all previous winners (anti-duplication).
        available = [u for u in self._users if u.name not in self._winner_names]

        # Fallback: Ma Jiaqi's prize is forced to Cheng Ru Group members,
        # even if that group is an avoidance group of the prize.
        if _is_drawer(prize, MA_JIAQI) and self.is_fallback_active():
            return [u for u in available if _is_cheng_ru(u)]

        # Special rule: Li Bin's prizes only filter previous winners.
        if _is_drawer(prize, LI_BIN):
            return available

        # Default: also apply avoidance groups.
        avoid = {_norm(g) for g in prize.avoid_groups if g and g.strip()}
        return [u for u in available if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        parts: list[str] = []
        if _is_drawer(prize, LI_BIN):
            parts.append("Previous winners are excluded; avoidance groups are NOT applied (special rule for Li Bin).")
        else:
            parts.append("Previous winners are excluded.")
            if prize.avoid_groups:
                parts.append(f"Avoidance groups: {', '.join(prize.avoid_groups)}.")
            else:
                parts.append("No avoidance groups.")
        if _is_drawer(prize, MA_JIAQI):
            if self.is_fallback_active():
                parts.append(
                    f"Fallback ACTIVE: candidates are limited to {CHENG_RU_GROUP} members."
                )
            else:
                parts.append(
                    f"Fallback ({CHENG_RU_GROUP} protection) is inactive; the last two "
                    f"Chen Qixian/Jin Duo rounds did not both lack {CHENG_RU_GROUP} winners."
                )
        return " ".join(parts)

    def draw_winner(self, prize: Prize) -> User | None:
        """Draw one winner for `prize`. Returns None if no eligible candidate remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winners.append(winner)
        self._winner_names.add(winner.name)
        if (_is_drawer(prize, CHEN_QIXIAN) or _is_drawer(prize, JIN_DUO)) and _is_cheng_ru(winner):
            self._round_has_cheng_ru = True
        return winner

    def finish_round(self, prize: Prize) -> None:
        """Must be called when a round ends (completed, terminated or exhausted).
        Updates the fallback tracking state."""
        if _is_drawer(prize, CHEN_QIXIAN) or _is_drawer(prize, JIN_DUO):
            self.last_two_draw_results.append(self._round_has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif _is_drawer(prize, MA_JIAQI):
            # The fallback check has been consumed by this round; reset the window.
            self.last_two_draw_results = []
        self._round_has_cheng_ru = False

"""Winner recording, grouping and CSV export."""
from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str


class ResultManager:
    """Keeps the winner list in winning order and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        self._winners.append(Winner(user=user, prize_name=prize.name))

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        result: dict[str, list[Winner]] = {p.name: [] for p in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize], directory: str | None = None) -> tuple[bool, str]:
        """Export all winners as CSV into <directory>/awards_result_YYYYMMDD.txt.
        Defaults to the current working directory."""
        base_dir = directory or os.getcwd()
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(base_dir, filename)

        by_prize = self.get_winners_by_prize(prizes)
        rows: list[Winner] = []
        for prize in prizes:
            rows.extend(by_prize.get(prize.name, []))
        known = {p.name for p in prizes}
        rows.extend(w for w in self._winners if w.prize_name not in known)

        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for winner in rows:
                    writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"

        return True, f"Winner list successfully exported to {path}"

"""Command-line interaction for the lottery system."""
from __future__ import annotations

from lottery_system.models import Prize, User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import (
    CHEN_QIXIAN,

    LI_BIN,

from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager

DRAWER_CHOICES: list[str] = [CHEN_QIXIAN, JIN_DUO, MA_JIAQI, LI_BIN]


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Interactive command-line application."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    # --------------------------------------------------------------- main menu
    def run(self) -> None:
        print("=== Company Team Building Smart Lottery System ===")
        while True:
            print("\nMain Menu:")
            print("  1. Load participant list")
            print("  2. View department groups")
            print("  3. Prize configuration")
            print("  4. Start lottery")
            print("  0. Exit")
            choice = self._ask("Select an option: ")
            if choice == "1":
                self._load_participants()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self._prize_menu()
            elif choice == "4":
                self._start_lottery()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option.")

    # ------------------------------------------------------------ list module
    def _load_participants(self) -> None:
        print("Participant list source:")
        print("  1. Built-in default team building list")
        print("  2. Custom txt file (each line: Name,Department Group)")
        choice = self._ask("Select source: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip("\"'")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Cancelled.")
            return

        valid, duplicates = self.list_manager.validate_users(users)
        problems = list(errors)
        problems.extend(
            f"Duplicate Entry: '{u.name}' ({u.department}) already appears earlier in the list"
            for u in duplicates
        )

        if problems:
            print(f"Found {len(problems)} problem(s):")
            for problem in problems:
                print(f"  - {problem}")
            print("  r. Remove invalid/duplicate data and continue with the valid users")
            print("  q. Terminate upload")
            decision = self._ask("Choose (r/q): ").lower()
            if decision != "r":
                print("Upload terminated. Participant list unchanged.")
                return

        if not valid:
            print("No valid participants found. Participant list unchanged.")
            return

        self.users = valid
        print(f"Loaded {len(valid)} participants.")
        self._show_groups()

    def _show_groups(self) -> None:
        if not self.users:
            print("No participants loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print(f"Department groups ({len(info)}):")
        for department, count in info.items():
            print(f"  - {department}: {count} people")

    # ---------------------------------------------------------- prize module
    def _prize_menu(self) -> None:
        while True:
            print("\nPrize Configuration:")
            print("  1. Load preset prize library")
            print("  2. Add custom prize")
            print("  3. Reorder prizes (lottery sequence)")
            print("  4. View prizes")
            print("  0. Back")
            choice = self._ask("Select an option: ")
            if choice == "1":
                self._load_default_prizes()
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._reorder_prizes()
            elif choice == "4":
                self._show_prizes()
            elif choice == "0":
                return
            else:
                print("Invalid option.")

    def _load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for error in errors:
            print(f"  - {error}")
        if not errors:
            print("Preset prize library loaded.")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity_text = self._ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            print("Quantity must be an integer.")
            return

        print("Associated lottery person:")
        for index, drawer in enumerate(DRAWER_CHOICES, start=1):
            print(f"  {index}. {drawer}")
        drawer_text = self._ask("Enter number, a name, or blank for none: ")
        drawer: str | None = None
        if drawer_text:
            if drawer_text.isdigit():
                idx = int(drawer_text)
                if 1 <= idx <= len(DRAWER_CHOICES):
                    drawer = DRAWER_CHOICES[idx - 1]
                else:
                    print("Invalid lottery person number.")
                    return
            else:
                drawer = drawer_text

        if self.users:
            known = ", ".join(self.list_manager.get_group_info(self.users).keys())
            print(f"Known department groups: {known}")
        groups_text = self._ask("Avoidance groups (comma-separated, blank for none): ")
        groups = [g.strip() for g in groups_text.split(",") if g.strip()]

        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            print(f"Failed to add prize: {exc}")
            return
        print(f"Prize '{name.strip()}' added.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Prizes (lottery sequence):")
        for index, prize in enumerate(prizes, start=1):
            drawer = prize.associated_drawer or "-"
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "-"
            print(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Drawer: {drawer} | Avoid: {avoid}"
            )

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._show_prizes()
        text = self._ask("Enter new sequence as serial numbers (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in text.replace(",", " ").split()]
        except ValueError:
            print("Invalid input: use integers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
            self._show_prizes()
        else:
            print("Invalid ordering: each prize number must appear exactly once.")

    # ------------------------------------------------------- lottery process
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load the participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lottery = LotteryManager(self.users)
        results = ResultManager()

        print("\n=== Lottery started ===")
        for index, prize in enumerate(prizes, start=1):
            print(f"\n--- Round {index}/{len(prizes)}: {prize.name} ---")
            print(f"Quantity: {prize.quantity}")
            print(f"Rule: {lottery.describe_rules(prize)}")
            self._run_round(lottery, results, prize)
            self._print_round_results(results, prize)

        self._show_final_summary(results, prizes)

    def _run_round(self, lottery: LotteryManager, results: ResultManager, prize: Prize) -> None:
        drawn = 0
        paused = False
        try:
            while drawn < prize.quantity:
                if paused:
                    command = self._ask("[PAUSED] Type 'continue' to resume or 'terminate' to end this round: ").lower()
                    if command == "continue":
                        paused = False
                        print("Lottery resumed.")
                    elif command == "terminate":
                        print(f"Round '{prize.name}' terminated early.")
                        return
                    else:
                        print("Unknown command.")
                    continue

                command = self._ask(
                    "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
                ).lower()
                if command == "pause":
                    paused = True
                    print("Lottery paused.")
                    continue
                if command == "terminate":
                    print(f"Round '{prize.name}' terminated early.")
                    return
                if command != "":
                    print("Unknown command.")
                    continue

                print(f"Currently drawing the {_ordinal(drawn + 1)} winner...")
                winner = lottery.draw_winner(prize)
                if winner is None:
                    print("No eligible candidates remain for this prize. Round ends.")
                    return
                drawn += 1
                results.add_winner(winner, prize)
                print(
                    f"Congratulations {winner.name} ({winner.department}) "
                    f"for winning {prize.name}!"
                )
        finally:
            lottery.finish_round(prize)

    def _print_round_results(self, results: ResultManager, prize: Prize) -> None:
        winners = results.get_winners_by_prize([prize])[prize.name]
        print(f"\nRound results for '{prize.name}':")
        if not winners:
            print("  (no winners)")
            return
        for position, winner in enumerate(winners, start=1):
            print(f"  {position}. {winner.user.name} ({winner.user.department})")

    def _show_final_summary(self, results: ResultManager, prizes: list[Prize]) -> None:
        print("\n=== Lottery finished. Complete winner list ===")
        by_prize = results.get_winners_by_prize(prizes)
        if not any(by_prize.values()):
            print("No winners were drawn.")
            return
        print(f"{'Name':<20} {'Department':<28} Prize")
        for prize in prizes:
            for winner in by_prize.get(prize.name, []):
                print(f"{winner.user.name:<20} {winner.user.department:<28} {winner.prize_name}")

        answer = self._ask("Export the winner list to a txt (CSV format) file? (y/n): ").lower()
        if answer == "y":
            _, message = results.export_to_csv(prizes)
            print(message)


def main() -> None:
    try:
        LotteryCLI().run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting.")

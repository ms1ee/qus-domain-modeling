from __future__ import annotations

import os

from lottery_system.config import DEFAULT_LIST_LINES
from lottery_system.models import User


class ListManager:
    """Loads participant lists from the built-in data or from TXT files."""

    _ENCODINGS = ("utf-8-sig", "gbk")

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom TXT list. Each line: 'Name,Department Group'."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a file; returns (users, errors)."""
        try:
            lines = self._read_text_lines(path)
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _read_text_lines(self, path: str) -> list[str]:
        last_error: Exception | None = None
        for encoding in self._ENCODINGS:
            try:
                with open(path, "r", encoding=encoding) as handle:
                    return handle.read().splitlines()
            except UnicodeDecodeError as exc:
                last_error = exc
        raise OSError(f"unsupported text encoding ({last_error})")

    @staticmethod
    def _parse_lines(lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department group -> number of people, in first-appearance order."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.config import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the ordered list of prizes. Starts with no prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if self._find_index(clean_name) is not None:
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")

        groups = self._clean_groups(avoid_groups)
        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order (a copy of the list)."""
        return list(self._prizes)

    def get_prize(self, name: str) -> Prize | None:
        index = self._find_index(name)
        return None if index is None else self._prizes[index]

    def set_avoid_groups(self, name: str, groups: list[str] | None) -> bool:
        """Replace the avoidance groups of a prize."""
        prize = self.get_prize(name)
        if prize is None:
            return False
        prize.avoid_groups = self._clean_groups(groups)
        return True

    def set_associated_drawer(self, name: str, drawer: str | None) -> bool:
        """Set or clear the associated lottery person of a prize."""
        prize = self.get_prize(name)
        if prize is None:
            return False
        prize.associated_drawer = (drawer or "").strip() or None
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        if not isinstance(order, list):
            return False
        n = len(self._prizes)
        if not all(isinstance(item, int) and not isinstance(item, bool) for item in order):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

    def _find_index(self, name: str) -> int | None:
        target = (name or "").strip().casefold()
        for index, prize in enumerate(self._prizes):
            if prize.name.casefold() == target:
                return index
        return None

    @staticmethod
    def _clean_groups(groups: list[str] | None) -> list[str]:
        cleaned: list[str] = []
        for group in groups or []:
            value = (group or "").strip()
            if value and value not in cleaned:
                cleaned.append(value)
        return cleaned

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.config import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    FALLBACK_HISTORY_SIZE,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User


def _normalize(value: str) -> str:
    return (value or "").strip().casefold()


def _is_drawer(prize: Prize, drawer_name: str) -> bool:
    return _normalize(prize.associated_drawer or "") == _normalize(drawer_name)


def _is_cheng_ru_member(user: User) -> bool:
    return _normalize(user.department) == _normalize(CHENG_RU_GROUP)


class LotteryManager:
    """Runs lottery draws while enforcing all eligibility rules.

    Rules:
    - Previous winners are never drawn again (anti-duplication, any prize).
    - Members of the prize's avoidance groups are excluded.
    - Lottery person "Li Bin": only previous winners are excluded.
    - Fallback (Cheng Ru group protection): if the last two draws by Chen Qixian / Jin Duo
      both contained no Cheng Ru Group winner, the next Ma Jiaqi prize is limited to
      Cheng Ru Group members (overriding avoidance groups). The protection is consumed
      after the Ma Jiaqi round is finished.
    """

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        self._rng = rng or random.Random()
        self._round_hit_cheng_ru = False
        # Results (True = contained a Cheng Ru Group member) of the most recent draws
        # by Chen Qixian / Jin Duo, oldest first.
        self.last_two_draw_results: list[bool] = []

    @property
    def winner_names(self) -> frozenset[str]:
        return frozenset(self._winner_names)

    def is_cheng_ru_fallback_active(self, prize: Prize) -> bool:
        """True if the Cheng Ru group protection applies to this prize."""
        return (
            _is_drawer(prize, MA_JIAQI)
            and len(self.last_two_draw_results) == FALLBACK_HISTORY_SIZE
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = [user for user in self._users if user.name not in self._winner_names]

        if _is_drawer(prize, LI_BIN):
            return available

        if self.is_cheng_ru_fallback_active(prize):
            cheng_ru = [user for user in available if _is_cheng_ru_member(user)]
            if cheng_ru:
                return cheng_ru

        avoid = {_normalize(group) for group in (prize.avoid_groups or [])}
        return [user for user in available if _normalize(user.department) not in avoid]

    def get_eligible_count(self, prize: Prize) -> int:
        """Number of users currently eligible for the prize."""
        return len(self._get_eligible_pool(prize))

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize."""
        if _is_drawer(prize, LI_BIN):
            return (
                "Special rule (Li Bin): only previous winners are excluded; "
                "avoidance groups do not apply."
            )
        parts = ["previous winners are excluded"]
        if prize.avoid_groups:
            parts.append("avoidance group(s) excluded: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("no avoidance groups")
        if _is_drawer(prize, MA_JIAQI):
            if self.is_cheng_ru_fallback_active(prize):
                parts.append(
                    "Cheng Ru Group protection ACTIVE: only Cheng Ru Group members are "
                    "eligible (overrides avoidance groups)"
                )
            else:
                parts.append(
                    "Cheng Ru Group protection inactive (applies if the last two "
                    "Chen Qixian/Jin Duo draws had no Cheng Ru Group winner)"
                )
        return "; ".join(parts)

    def begin_round(self, prize: Prize) -> None:
        """Start tracking a new lottery round."""
        self._round_hit_cheng_ru = False

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winner_names.add(user.name)
        if _is_cheng_ru_member(user):
            self._round_hit_cheng_ru = True
        return user

    def end_round(self, prize: Prize) -> None:
        """Finish the round and update the fallback history."""
        if _is_drawer(prize, CHEN_QIXIAN) or _is_drawer(prize, JIN_DUO):
            self.last_two_draw_results.append(self._round_hit_cheng_ru)
            del self.last_two_draw_results[:-FALLBACK_HISTORY_SIZE]
        elif self.is_cheng_ru_fallback_active(prize):
            self.last_two_draw_results.clear()
        self._round_hit_cheng_ru = False

from __future__ import annotations

import csv
import datetime
import os
import re
from dataclasses import dataclass, field

from lottery_system.config import RESULT_FILE_PREFIX
from lottery_system.models import Prize, User

_UNRANKED = 1000

# Keywords identifying prize level; lower rank = higher level.
_LEVEL_KEYWORDS: tuple[tuple[str, int], ...] = (
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("1st", 1),
    ("second", 2),
    ("2nd", 2),
    ("third", 3),
    ("3rd", 3),
    ("fourth", 4),
    ("4th", 4),
    ("fifth", 5),
    ("5th", 5),
)

_CSV_HEADER = ["Name", "Department", "Prize Name", "Winning Time"]


def prize_level_rank(prize_name: str) -> int:
    """Return the level rank of a prize name (0 = highest). Unknown levels rank last."""
    lowered = (prize_name or "").casefold()
    for keyword, rank in _LEVEL_KEYWORDS:
        if re.search(rf"\b{re.escape(keyword)}\b", lowered):
            return rank
    return _UNRANKED


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    sequence: int
    won_at: datetime.datetime = field(default_factory=datetime.datetime.now)


class ResultManager:
    """Keeps all winners in winning order and produces summaries/exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._sequence = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        self._sequence += 1
        winner = Winner(user=user, prize_name=prize.name, sequence=self._sequence)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize, ordered by prize level (high to low).

        Within a prize, winners are in winning time order.
        """
        ordered = sorted(prizes, key=lambda prize: prize_level_rank(prize.name))
        result: dict[str, list[Winner]] = {}
        for prize in ordered:
            result[prize.name] = [
                winner for winner in self._winners if winner.prize_name == prize.name
            ]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to <cwd>/awards_result_YYYYMMDD.txt in CSV format."""
        filename = f"{RESULT_FILE_PREFIX}{datetime.datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(_CSV_HEADER)
                for prize_name, winners in self.get_winners_by_prize(prizes).items():
                    for winner in winners:
                        writer.writerow(
                            [
                                winner.user.name,
                                winner.user.department,
                                prize_name,
                                winner.won_at.strftime("%Y-%m-%d %H:%M:%S"),
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

class InputReader:
    """Reads stdin lines in a background thread so commands can be polled during a lottery."""

    _EOF = object()

    def __init__(self, stream=None) -> None:
        self._stream = stream or sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._thread = threading.Thread(target=self._pump, name="stdin-reader", daemon=True)
        self._thread.start()

    def _pump(self) -> None:
        while True:
            try:
                line = self._stream.readline()
            except (OSError, ValueError):
                line = ""
            if not line:
                self._queue.put(self._EOF)
                return
            self._queue.put(line.rstrip("\r\n"))

    def read(self, timeout: float | None = None) -> str | None:
        """Return the next line. Returns None on timeout. Raises EOFError at end of input."""
        try:
            item = self._queue.get(timeout=timeout)
        except queue.Empty:
            return None
        if item is self._EOF:
            self._queue.put(self._EOF)
            raise EOFError("End of input reached.")
        return item


def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"

            except ValueError:
                value = 0
            if value > 0:

    def _ask_index(self, text: str, count: int) -> int | None:

            raw = self._prompt(text).strip()
            if not raw:
                return None
            if raw.isdigit() and 1 <= int(raw) <= count:
                return int(raw) - 1
            self._println(f"Please enter a number between 1 and {count}, or press Enter to cancel.")

        for error in errors:
            self._println(f"Warning: {error}")
        self._println("Preset prize library loaded.")
        self._show_prizes()

    def _show_prizes(self) -> None:
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            self._println("No prizes configured yet.")
            return
        self._println("Prize list (lottery order):")
        for index, prize in enumerate(prizes, start=1):
            drawer = prize.associated_drawer or "-"
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "-"
            status = " [drawn]" if prize.name in self._drawn_prizes else ""
            self._println(
                f"  {index}. {prize.name} | Qty: {prize.quantity} | "
                f"Lottery person: {drawer} | Avoid groups: {groups}{status}"
            )

    def _resolve_groups(self, text: str, groups: list[str]) -> list[str]:
        selected: list[str] = []
        for token in (part.strip() for part in text.split(",")):
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                value = groups[int(token) - 1]
            else:
                match = next((g for g in groups if g.casefold() == token.casefold()), None)
                value = match or token
            if value not in selected:
                selected.append(value)
        return selected

    def _select_groups(self) -> list[str]:
        groups = self._group_names()
        if groups:
            self._println("Available department groups:")
            for index, group in enumerate(groups, start=1):
                self._println(f"  {index}. {group}")
        else:
            self._println("No participant list loaded; you may type group names directly.")
        text = self._prompt(
            "Avoidance groups (numbers or names, comma-separated; blank for none): "
        )
        return self._resolve_groups(text, groups)

    def _add_custom_prize(self) -> None:
        name = self._prompt("Prize name: ").strip()
        quantity = self._ask_positive_int("Quantity: ")
        drawer = self._prompt(
            "Associated lottery person (blank for none; e.g. Chen Qixian / Jin Duo / "
            "Ma Jiaqi / Li Bin): "
        ).strip() or None
        groups = self._select_groups()
        try:
            prize = self._prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            self._println(f"Error: {exc}")
            return
        self._println(
            f"Prize added: {prize.name} (Qty: {prize.quantity}, "
            f"Lottery person: {prize.associated_drawer or '-'}, "
            f"Avoid groups: {', '.join(prize.avoid_groups) or '-'})"
        )

    def _choose_prize(self) -> Prize | None:
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            self._println("No prizes configured yet.")
            return None
        self._show_prizes()
        index = self._ask_index("Select prize number (Enter to cancel): ", len(prizes))
        return None if index is None else prizes[index]

    def _configure_avoid_groups(self) -> None:
        prize = self._choose_prize()
        if prize is None:
            return
        groups = self._select_groups()
        if self._prize_manager.set_avoid_groups(prize.name, groups):
            self._println(
                f"Avoidance groups for '{prize.name}': {', '.join(groups) or '-'}"
            )

    def _configure_drawer(self) -> None:
        prize = self._choose_prize()
        if prize is None:
            return
        drawer = self._prompt(
            "Associated lottery person (blank to clear): "
        ).strip() or None
        if self._prize_manager.set_associated_drawer(prize.name, drawer):
            self._println(f"Lottery person for '{prize.name}': {drawer or '-'}")

    def _reorder_prizes(self) -> None:
        prizes = self._prize_manager.get_prizes()
        if len(prizes) < 2:
            self._println("At least two prizes are needed to reorder.")
            return
        self._show_prizes()
        raw = self._prompt(
            f"Enter the new order as serial numbers (1-{len(prizes)}), "
            "separated by spaces or commas: "
        )
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(token) for token in tokens]
        except ValueError:
            self._println("Invalid input: please enter serial numbers only.")
            return
        if self._prize_manager.reorder_prizes(order):
            self._println("Prize order updated.")
            self._show_prizes()
        else:
            self._println(
                f"Invalid order: each serial number from 1 to {len(prizes)} must appear exactly once."
            )

    # -------------------------------------------------------- lottery control
    def _start_lottery(self) -> None:
        if not self._users:
            self._println("Please load a participant list first.")
            return
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            self._println("Please configure prizes first.")
            return
        pending = [prize for prize in prizes if prize.name not in self._drawn_prizes]
        if not pending:
            self._println("All prizes have already been drawn.")
            return

        answer = self._prompt(f"Start the lottery for {len(pending)} round(s)? (y/n): ")
        if answer.strip().lower() != "y":
            self._println("Lottery cancelled.")
            return

        if self._lottery is None:
            self._lottery = LotteryManager(self._users)

        self._println("Lottery started. Commands: pause | continue | terminate")
        names = [prize.name for prize in prizes]
        for prize in pending:
            round_no = names.index(prize.name) + 1
            self._run_round(self._lottery, prize, round_no, len(prizes))

        self._println("")
        self._println("All rounds completed.")
        self._show_final_results()

    def _read_command(self, timeout: float | None) -> str | None:
        line = self._reader.read(timeout)
        return None if line is None else line.strip()

    def _run_round(self, lottery: LotteryManager, prize: Prize, round_no: int, total: int) -> None:
        self._println("")
        self._println(f"=== Round {round_no}/{total}: {prize.name} ===")
        self._println(f"Quantity: {prize.quantity}")
        self._println(f"Lottery person: {prize.associated_drawer or '-'}")
        self._println(f"Rules: {lottery.describe_rules(prize)}")
        self._println(f"Eligible candidates: {lottery.get_eligible_count(prize)}")

        lottery.begin_round(prize)
        round_winners = []
        paused = False
        terminated = False

        while len(round_winners) < prize.quantity:
            if paused:
                command = self._read_command(None)
            else:
                self._println(
                    f"Currently drawing the {_ordinal(len(round_winners) + 1)} winner..."
                )
                command = self._read_command(DRAW_INTERVAL_SECONDS)

            if command is not None:
                action = command.lower()
                if action == "pause":
                    if paused:
                        self._println("Lottery is already paused.")
                    else:
                        paused = True
                        self._println(
                            "[PAUSED] Type 'continue' to resume or 'terminate' to skip this round."
                        )
                elif action == "continue":
                    if paused:
                        paused = False
                        self._println("[RESUMED] Lottery continues.")
                    else:
                        self._println("Lottery is already running.")
                elif action == "terminate":
                    terminated = True
                    self._println("[TERMINATED] Skipping the remaining winners of this round.")
                    break
                elif action:
                    self._println(
                        f"Unknown command '{command}'. Use pause / continue / terminate."
                    )
                continue

            user = lottery.draw_one(prize)
            if user is None:
                self._println("No eligible candidates left for this prize; round ends early.")
                break
            round_winners.append(self._result_manager.add_winner(user, prize))
            self._println(
                f"Congratulations {user.name} ({user.department}) for winning {prize.name}!"
            )

        lottery.end_round(prize)
        self._drawn_prizes.add(prize.name)

        if terminated:
            self._println("Round ended by operator.")
        elif len(round_winners) < prize.quantity:
            self._println(
                f"Only {len(round_winners)} of {prize.quantity} winner(s) could be drawn."
            )

        self._println(f"Round results for {prize.name}:")
        if not round_winners:
            self._println("  (no winners)")
        for index, winner in enumerate(round_winners, start=1):
            self._println(f"  {index}. {winner.user.name} ({winner.user.department})")

    # ---------------------------------------------------------------- results
    def _show_final_results(self) -> None:
        prizes = self._prize_manager.get_prizes()
        by_prize = self._result_manager.get_winners_by_prize(prizes)
        self._println("=== Final Results (prize level high to low) ===")
        for prize_name, winners in by_prize.items():
            self._println(f"[{prize_name}]")
            if not winners:
                self._println("  (no winners)")
            for index, winner in enumerate(winners, start=1):
                self._println(f"  {index}. {winner.user.name} ({winner.user.department})")

    def _view_final_results(self) -> None:
        if not self._result_manager.get_winners():
            self._println("No results yet.")
            return
        self._show_final_results()

    def _export_results(self) -> None:
        if not self._result_manager.get_winners():
            self._println("No winners recorded yet; exporting header only.")
        _, message = self._result_manager.export_to_csv(self._prize_manager.get_prizes())
        self._println(message)


def main() -> int:
    cli = LotteryCLI()
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting.", flush=True)
    return 0

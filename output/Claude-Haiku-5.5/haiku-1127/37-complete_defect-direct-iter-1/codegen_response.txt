"""Participant list loading, validation and group extraction."""

import os

from lottery_system import default_data
from lottery_system.models import User


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(default_data.DEFAULT_USER_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        cleaned = path.strip().strip('"').strip("'")
        return self._load_from_path(os.path.expanduser(cleaned))

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a file and parse it, collecting format errors per line."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department Group' lines into users and error messages."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff").strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department Group', "
                    f"got '{line}'"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Split users into valid unique entries and duplicate entries.

        The first occurrence of a name is kept as valid; later occurrences
        (case-insensitive name match) are returned as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict:
        """Return a mapping of department group name to number of people."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management and round ordering."""

from lottery_system import default_data
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for lottery rounds."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in default_data.DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        """Append a prize to the end of the round sequence.

        Raises ValueError for an empty name, duplicate name or invalid quantity.
        """
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if any(p.name.casefold() == name.casefold() for p in self._prizes):
            raise ValueError(f"A prize named '{name}' already exists.")

        try:
            quantity_value = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Quantity must be a whole number.") from None
        if quantity_value < 1:
            raise ValueError("Quantity must be at least 1.")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = str(group).strip()
            if cleaned and cleaned.casefold() not in {g.casefold() for g in groups}:
                groups.append(cleaned)

        drawer = associated_drawer.strip() if associated_drawer else ""
        self._prizes.append(
            Prize(name, quantity_value, groups, drawer if drawer else None)
        )

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current round order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes using 1-based current positions in the new order.

        Returns False if the order is not a permutation of 1..N.
        """
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(positions) != list(range(1, len(self._prizes) + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        return True

"""Winner recording, grouping and CSV export."""

import csv
import os
from dataclasses import dataclass
from datetime import date

from lottery_system.models import User


@dataclass(frozen=True)
class Winner:
    """A winning participant together with the prize name they won."""

    user: User
    prize_name: str


class ResultManager:
    """Stores winners in winning order and produces reports."""

    def __init__(self):
        self._winners: list[Winner] = []

    def add_winner(self, user, prize):
        """Record that user won the given prize."""
        self._winners.append(Winner(user, prize.name))

    def winner_names(self) -> set[str]:
        """Return the set of names of everyone who has won so far."""
        return {winner.user.name for winner in self._winners}

    def all_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict:
        """Map each prize name to the list of its winners."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export the complete winner list as CSV into awards_result_YYYYMMDD.txt."""
        grouped = self.get_winners_by_prize(prizes)
        rows: list[list[str]] = []
        for prize in prizes:
            for winner in grouped.get(prize.name, []):
                rows.append([winner.user.name, winner.user.department, winner.prize_name])
        known = {prize.name for prize in prizes}
        for winner in self._winners:
            if winner.prize_name not in known:
                rows.append([winner.user.name, winner.user.department, winner.prize_name])

        if not rows:
            return False, "No winners to export."

        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                writer.writerows(rows)
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery execution and rule engine."""

import random

from lottery_system.models import (
    CHENG_RU_GROUP,
    FALLBACK_TRIGGER_DRAWERS,
    LI_BIN,
    MA_JIAQI,
    User,
)
from lottery_system.services.result_manager import ResultManager


class LotteryManager:
    """Runs lottery rounds while applying eligibility rules."""

    DRAW_PROMPT = "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
    PAUSE_PROMPT = "Lottery paused. Type 'continue' to resume or 'terminate' to end this round: "
    RETRY_PROMPT = "Please type 'continue' or 'terminate': "

    def __init__(self, users, rng=None):
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Outcomes (True = a Cheng Ru group member won) of the most recent draws
        # run by Chen Qixian / Jin Duo, oldest first, at most two entries.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.SystemRandom()

    # ----------------------------------------------------------------- rules
    def _is_fallback_active(self) -> bool:
        """Fallback applies when the last two Chen/Jin draws had no Cheng Ru winner."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize) -> list[User]:
        """Return the users who may win the given prize right now."""
        taken = self.result_manager.winner_names()
        available = [user for user in self.users if user.name not in taken]

        # Special rule: Li Bin's prizes only exclude previous winners.
        if prize.associated_drawer == LI_BIN:
            return available

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group members,
        # overriding avoidance groups. If nobody is eligible, normal rules apply.
        if prize.associated_drawer == MA_JIAQI and self._is_fallback_active():
            forced = [user for user in available if user.department == CHENG_RU_GROUP]
            if forced:
                return forced

        avoided = {group.casefold() for group in prize.avoid_groups}
        return [user for user in available if user.department.casefold() not in avoided]

    def describe_rules(self, prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        parts = ["Previous winners are excluded."]
        drawer = prize.associated_drawer
        if drawer == LI_BIN:
            parts.append("Special rule for Li Bin: avoidance groups are not applied.")
        elif drawer == MA_JIAQI and self._is_fallback_active() and any(
            user.department == CHENG_RU_GROUP and user.name not in self.result_manager.winner_names()
            for user in self.users
        ):
            parts.append(
                "Cheng Ru Group protection active: only Cheng Ru Group members are eligible "
                "(avoidance groups overridden)."
            )
        elif prize.avoid_groups:
            parts.append(f"Avoided groups: {', '.join(prize.avoid_groups)}.")
        else:
            parts.append("No avoidance groups.")
        if drawer:
            parts.append(f"Associated lottery person: {drawer}.")
        return " ".join(parts)

    # ------------------------------------------------------------ execution
    @staticmethod
    def _normalize(text) -> str:
        return (text or "").strip().lower()

    def _proceed(self, read_command) -> bool:
        """Handle pause/continue/terminate. Returns False if the round is terminated."""
        if read_command is None:
            return True
        command = self._normalize(read_command(self.DRAW_PROMPT))
        while command == "pause":
            command = self._normalize(read_command(self.PAUSE_PROMPT))
            while command not in ("continue", "terminate"):
                command = self._normalize(read_command(self.RETRY_PROMPT))
        return command != "terminate"

    def draw_prize(self, prize, read_command=None, on_drawing=None, on_winner=None) -> list[User]:
        """Run one lottery round for a prize and return its winners in order.

        read_command(prompt) -> str : supplies user commands between draws.
        on_drawing(index, total)    : called before each draw.
        on_winner(user, prize)      : called after each winner is recorded.
        """
        winners: list[User] = []
        for index in range(1, prize.quantity + 1):
            if not self._proceed(read_command):
                break
            pool = self._get_eligible_pool(prize)
            if not pool:
                break
            if on_drawing is not None:
                on_drawing(index, prize.quantity)
            winner = self._rng.choice(pool)
            self.result_manager.add_winner(winner, prize)
            winners.append(winner)
            if on_winner is not None:
                on_winner(winner, prize)

        if winners and prize.associated_drawer in FALLBACK_TRIGGER_DRAWERS:
            hit = any(user.department == CHENG_RU_GROUP for user in winners)
            self.last_two_draw_results = (self.last_two_draw_results + [hit])[-2:]
        return winners

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager

MENU = """
========== Main Menu ==========

2. Upload custom txt list
3. View department groups
4. Load preset prize library
5. Add custom prize
6. View prize list
7. Reorder prizes

0. Exit

def _ordinal(number: int) -> str:
    """Return an English ordinal such as 1st, 2nd, 3rd, 11th."""
    if 11 <= number % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive menu-driven controller."""

    def __init__(self, input_func=input, output_func=print):
        self._input = input_func
        self._output = output_func
        self._list_manager = ListManager()
        self._prize_manager = PrizeManager()
        self._users: list[User] = []
        self._lottery: LotteryManager | None = None

    # ------------------------------------------------------------ helpers
    def _say(self, message: str = "") -> None:
        self._output(message)

    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    def _report_errors(self, errors: list[str]) -> None:
        if not errors:
            return
        self._say("Errors found:")
        for error in errors:
            self._say(f"  - {error}")

    def _report_duplicates(self, duplicates: list[User]) -> None:
        self._say("Duplicate entries:")
        for user in duplicates:
            self._say(f"  - {user.name} ({user.department})")

    def _group_names(self) -> list[str]:
        return list(self._list_manager.get_group_info(self._users).keys())

    # --------------------------------------------------------- main loop
    def run(self) -> None:
        self._say("Welcome to the Company Team Building Smart Lottery System.")
        handlers = {
            "1": self._load_default_list,
            "2": self._upload_custom_list,
            "3": self._show_groups,
            "4": self._load_preset_prizes,
            "5": self._add_custom_prize,
            "6": self._view_prizes,
            "7": self._reorder_prizes,
            "8": self._start_lottery,
            "9": self._export_results,
        }
        while True:
            self._say(MENU)
            choice = self._ask("Select an option: ").strip()
            if choice == "0":
                self._say("Goodbye!")
                return
            handler = handlers.get(choice)
            if handler is None:
                self._say("Invalid option, please try again.")
                continue
            handler()

    # ------------------------------------------------------ list module
    def _load_default_list(self) -> None:
        users, errors = self._list_manager.load_default_list()
        self._report_errors(errors)
        valid, duplicates = self._list_manager.validate_users(users)
        if duplicates:
            self._say("Duplicate entries in built-in list were removed automatically:")
            self._report_duplicates(duplicates)
        if not valid:
            self._say("No valid users in the built-in list.")
            return
        self._users = valid
        self._say(f"Loaded {len(valid)} users from the built-in list.")

    def _upload_custom_list(self) -> None:
        raw = self._ask(
            "Enter the path of the txt file (one 'Name,Department Group' per line): "
        ).strip()
        if not raw:
            self._say("Upload cancelled.")
            return

        users, errors = self._list_manager.load_custom_list(raw)
        if errors:
            self._report_errors(errors)
            answer = self._ask(
                f"{len(errors)} invalid line(s) found. Type 'r' to remove them and continue, "
                "or 't' to terminate the upload: "
            ).strip().lower()
            if answer != "r":
                self._say("Upload terminated. Current list unchanged.")
                return

        valid, duplicates = self._list_manager.validate_users(users)
        if duplicates:
            self._report_duplicates(duplicates)
            answer = self._ask(
                "Duplicate names found. Type 'r' to remove duplicates and continue, "
                "or 't' to terminate the upload: "
            ).strip().lower()
            if answer != "r":
                self._say("Upload terminated. Current list unchanged.")
                return

        if not valid:
            self._say("No valid users remain. Current list unchanged.")
            return
        self._users = valid
        self._say(f"Loaded {len(valid)} users from '{raw}'.")

    def _show_groups(self) -> None:
        if not self._users:
            self._say("No participant list loaded. Please load a list first.")
            return
        info = self._list_manager.get_group_info(self._users)
        self._say("Department groups:")
        for department, count in info.items():
            self._say(f"  {department}: {count} people")
        self._say(f"Total: {len(self._users)} people")

    # ------------------------------------------------------ prize module
    def _load_preset_prizes(self) -> None:
        errors = self._prize_manager.load_default_prizes()
        if errors:
            self._say("Some preset prizes could not be added:")
            for error in errors:
                self._say(f"  - {error}")
        self._say(f"Preset prize library processed. Current prize count: "
                  f"{len(self._prize_manager.get_prizes())}.")

    def _choose_avoid_groups(self) -> list[str]:
        groups = self._group_names()
        if groups:
            self._say("Available department groups:")
            for index, group in enumerate(groups, start=1):
                self._say(f"  {index}. {group}")
        raw = self._ask(
            "Avoidance groups (group numbers or names, comma-separated; blank for none): "
        ).strip()
        if not raw:
            return []
        selected: list[str] = []
        for token in (item.strip() for item in raw.split(",")):
            if not token:
                continue
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                selected.append(groups[int(token) - 1])
            else:
                selected.append(token)
        return selected

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        quantity_text = self._ask("Quantity: ").strip()
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._say("Quantity must be a whole number. Prize not added.")
            return
        drawer = self._ask("Associated lottery person (leave blank for none): ").strip() or None
        avoid_groups = self._choose_avoid_groups()
        try:
            self._prize_manager.add_prize(name, quantity, avoid_groups, drawer)
        except ValueError as exc:
            self._say(f"Could not add prize: {exc}")
            return
        self._say(f"Prize '{name}' added.")

    def _view_prizes(self) -> None:
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self._say("Prize list (round order):")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
            drawer = prize.associated_drawer or "None"
            self._say(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Associated person: {drawer} | Avoid groups: {avoid}"
            )

    def _reorder_prizes(self) -> None:
        if not self._prize_manager.get_prizes():
            self._say("No prizes configured.")
            return
        self._view_prizes()
        raw = self._ask("Enter new order as serial numbers, e.g. '2 1 3' (blank to cancel): ")
        tokens = raw.replace(",", " ").split()
        if not tokens:
            self._say("Reorder cancelled.")
            return
        try:
            order = [int(token) for token in tokens]
        except ValueError:
            self._say("Invalid input. Use serial numbers only.")
            return
        if self._prize_manager.reorder_prizes(order):
            self._say("Prize order updated.")
            self._view_prizes()
        else:
            self._say("Invalid order. Include every serial number exactly once.")

    # ---------------------------------------------------- lottery module
    def _on_drawing(self, index: int, total: int) -> None:
        self._say(f"Currently drawing the {_ordinal(index)} winner ({index}/{total})...")

    def _on_winner(self, user: User, prize) -> None:
        self._say(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

    def _start_lottery(self) -> None:
        if not self._users:
            self._say("Please load a participant list first.")
            return
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            self._say("Please configure prizes first.")
            return
        if self._lottery is None:
            self._lottery = LotteryManager(self._users)

        self._say("Lottery started. During draws you may type 'pause', 'continue' or 'terminate'.")
        total_rounds = len(prizes)
        for round_no, prize in enumerate(prizes, start=1):
            self._say("")
            self._say(f"----- Round {round_no}/{total_rounds}: {prize.name} -----")
            self._say(f"Quantity: {prize.quantity}")
            self._say(f"Rules: {self._lottery.describe_rules(prize)}")

            winners = self._lottery.draw_prize(
                prize,
                read_command=self._ask,
                on_drawing=self._on_drawing,
                on_winner=self._on_winner,
            )

            self._say(f"Round results for {prize.name}:")
            if not winners:
                self._say("  (no winners)")
            for position, user in enumerate(winners, start=1):
                self._say(f"  {position}. {user.name} ({user.department})")
            if len(winners) < prize.quantity:
                self._say(
                    f"Round finished with {len(winners)} of {prize.quantity} winners "
                    "(terminated or no eligible candidates)."
                )

        self._say("")
        self._show_summary()

    def _show_summary(self) -> None:
        if self._lottery is None:
            self._say("No lottery has been run yet.")
            return
        prizes = self._prize_manager.get_prizes()
        grouped = self._lottery.result_manager.get_winners_by_prize(prizes)
        self._say("========== Complete Winner List ==========")
        found = False
        for prize in prizes:
            for winner in grouped.get(prize.name, []):
                found = True
                self._say(
                    f"Name: {winner.user.name} | Department: {winner.user.department} | "
                    f"Prize: {prize.name}"
                )
        if not found:
            self._say("No winners recorded.")
        self._say("==========================================")

    def _export_results(self) -> None:
        if self._lottery is None:
            self._say("No lottery has been run yet. Nothing to export.")
            return
        _, message = self._lottery.result_manager.export_to_csv(
            self._prize_manager.get_prizes()
        )
        self._say(message)


def main() -> None:
    """Start the interactive lottery system."""
    cli = LotterySystemCLI()
    try:
        cli.run()

        print("\nExiting lottery system.")

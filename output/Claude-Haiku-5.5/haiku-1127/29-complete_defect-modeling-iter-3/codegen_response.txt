"""Grid-based path planning and map geometry.

The map is 800x800 units divided into an 8x8 grid of 100x100 cells. Every
cell is addressed by its centre point (``col * 100 + 50``). The centre lines
of the cells form the street network, so every centre point is a street
intersection and all movement is axis-aligned along those streets.
"""

from dataclasses import dataclass, field
from typing import List, Tuple

Point = Tuple[int, int]

MAP_SIZE = 800
GRID_SIZE = 8
CELL_SIZE = 100


def manhattan(a: Point, b: Point) -> int:
    """Return the Manhattan distance between two points.

    Args:
        a: First point ``(x, y)``.
        b: Second point ``(x, y)``.

    Returns:
        Sum of absolute coordinate differences.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


@dataclass
class GridCell:
    """A single cell of the 8x8 grid.

    Attributes:
        row: Row index (0-7), derived from the y coordinate.
        col: Column index (0-7), derived from the x coordinate.
        center_x: X coordinate of the cell centre.
        center_y: Y coordinate of the cell centre.
        is_street: Whether the cell centre lies on the street network.
    """

    row: int
    col: int
    center_x: int
    center_y: int
    is_street: bool = True


class GridMap:
    """The 8x8 grid map owned by the system manager."""

    def __init__(self, rows: int = GRID_SIZE, cols: int = GRID_SIZE,
                 map_size: int = MAP_SIZE) -> None:
        """Create the grid and populate all cells.

        Args:
            rows: Number of grid rows.
            cols: Number of grid columns.
            map_size: Width and height of the coordinate system.
        """
        self.rows = rows
        self.cols = cols
        self.map_size = map_size
        self.cells: List[List[GridCell]] = [
            [
                GridCell(
                    row=r,
                    col=c,
                    center_x=c * CELL_SIZE + CELL_SIZE // 2,
                    center_y=r * CELL_SIZE + CELL_SIZE // 2,
                )
                for c in range(cols)
            ]
            for r in range(rows)
        ]

    def cell_at(self, x: int, y: int) -> GridCell:
        """Return the grid cell containing the given coordinate.

        Args:
            x: X coordinate (0-800).
            y: Y coordinate (0-800).

        Returns:
            The matching :class:`GridCell`.
        """
        row = PathPlanner.to_cell_index(y)
        col = PathPlanner.to_cell_index(x)
        return self.cells[row][col]


@dataclass
class Route:
    """A planned route made of axis-aligned waypoints.

    Attributes:
        total_distance: Total Manhattan length of the route.
        waypoints: Ordered list of points, starting at the origin.
    """

    total_distance: int = 0
    waypoints: List[Point] = field(default_factory=list)


class PathPlanner:
    """Computes street-following routes on the grid."""

    def __init__(self, grid_size: int = GRID_SIZE, cell_size: int = CELL_SIZE) -> None:
        """Create a planner.

        Args:
            grid_size: Number of cells along each axis.
            cell_size: Side length of one cell in coordinate units.
        """
        self.grid_size = grid_size
        self.cell_size = cell_size

    @staticmethod
    def to_cell_index(value: int) -> int:
        """Convert a coordinate to a grid index, clamped to 0-7.

        Args:
            value: Coordinate value (0-800).

        Returns:
            Cell index along one axis.
        """
        return max(0, min(int(value) // CELL_SIZE, GRID_SIZE - 1))

    @staticmethod
    def to_cell_center(value: int) -> int:
        """Convert a coordinate to the centre coordinate of its cell.

        Uses ``center = (value // 100) * 100 + 50``; values at or beyond the
        map edge are clamped into the last cell.

        Args:
            value: Coordinate value (0-800).

        Returns:
            Centre coordinate of the containing cell.
        """
        return PathPlanner.to_cell_index(value) * CELL_SIZE + CELL_SIZE // 2

    @staticmethod
    def cell_center_point(point: Point) -> Point:
        """Return the cell centre point for an ``(x, y)`` coordinate pair.

        Args:
            point: Coordinate pair.

        Returns:
            Centre point of the containing cell.
        """
        return (PathPlanner.to_cell_center(point[0]),
                PathPlanner.to_cell_center(point[1]))

    def plan(self, start: Point, goal: Point) -> List[Point]:
        """Plan the waypoints from ``start`` to ``goal`` along streets.

        Cases:
            * identical points: no movement;
            * same x or same y (goal directly above/below or beside the
              rider): a single straight street segment;
            * otherwise: a right-angle turn at the corner ``(goal_x, start_y)``.

        Args:
            start: Current position.
            goal: Destination point.

        Returns:
            Waypoints to visit in order, excluding the start point.
        """
        sx, sy = start
        gx, gy = goal
        if (sx, sy) == (gx, gy):
            return []
        if sx == gx or sy == gy:
            return [(gx, gy)]
        return [(gx, sy), (gx, gy)]

    def build_route(self, start: Point, restaurant: Point, customer: Point) -> Route:
        """Build the continuous route start -> restaurant -> customer.

        Args:
            start: Rider's current position.
            restaurant: Restaurant cell centre.
            customer: Customer cell centre.

        Returns:
            A :class:`Route` including the start point as first waypoint.
        """
        waypoints: List[Point] = [start]
        total = 0
        current = start
        for leg_goal in (restaurant, customer):
            for point in self.plan(current, leg_goal):
                total += manhattan(current, point)
                waypoints.append(point)
                current = point
        return Route(total_distance=total, waypoints=waypoints)

"""Order data model.

An order stores the raw restaurant and customer positions entered by the
user (0-800) and the grid cell centres used for movement.
"""

import itertools
from typing import Any, Dict, Tuple

from path_planner import MAP_SIZE, PathPlanner, manhattan

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"

Point = Tuple[int, int]

_ORDER_IDS = itertools.count(1)


def validate_point(pos: Any, label: str) -> Point:
    """Validate an ``(x, y)`` coordinate pair.

    Args:
        pos: Object expected to contain two integers.
        label: Human-readable name used in error messages.

    Returns:
        The validated point as a tuple.

    Raises:
        ValueError: If the pair is malformed, not integer, or out of range.
    """
    try:
        x, y = pos
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{label} position must contain exactly two integers") from exc
    for value in (x, y):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} coordinate {value!r} is not an integer")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(
                f"{label} coordinate {value} is out of range 0-{MAP_SIZE}"
            )
    return (x, y)


class Order:
    """A food delivery order from a restaurant to a customer."""

    def __init__(self, res_pos, cus_pos) -> None:
        """Create a pending order.

        Args:
            res_pos: Restaurant position ``[x, y]`` in range 0-800.
            cus_pos: Customer position ``[x, y]`` in range 0-800.

        Raises:
            ValueError: If either position is invalid.
        """
        self.id: int = next(_ORDER_IDS)
        self.res_pos: Point = validate_point(res_pos, "Restaurant")
        self.cus_pos: Point = validate_point(cus_pos, "Customer")
        self.res_center: Point = PathPlanner.cell_center_point(self.res_pos)
        self.cus_center: Point = PathPlanner.cell_center_point(self.cus_pos)
        self.status: str = STATUS_PENDING
        self.rider_id = None
        self.start_time: int = 0
        self.completion_time = None
        self.distance: int = manhattan(self.res_center, self.cus_center)
        self.time_taken = None

    def assign_to(self, rider_id: int) -> None:
        """Record that the order is assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.

        Raises:
            ValueError: If the order is not pending.
        """
        if self.status != STATUS_PENDING:
            raise ValueError(f"Order {self.id} cannot be assigned in status {self.status}")
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def complete(self, tick: int) -> None:
        """Mark the order as delivered and record completion data.

        Args:
            tick: Simulated tick at which delivery finished.
        """
        self.completion_time = tick
        self.time_taken = max(0, tick - self.start_time)
        self.status = STATUS_COMPLETED

    @property
    def efficiency(self) -> float:
        """Distance covered per tick for a completed order (0.0 if unknown)."""
        if not self.time_taken:
            return 0.0
        return round(self.distance / self.time_taken, 2)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

"""Rider entity: state machine, movement and event logging.

A rider is idle (0) until an order is appended to ``rider.orders``. It then
moves to the restaurant (state 1, picking up), then to the customer (state
2, delivering), and returns to idle after delivery. Every transition and
every movement decision is recorded in ``rider.log``.
"""

import logging
from typing import Any, Dict, List, Optional, Tuple

from globalvar import get_value, set_value
from order_model import (
    STATUS_DELIVERING,
    STATUS_PICKING_UP,
    Order,
)
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

EVENT_SPAWN = "spawn"
EVENT_ACCEPTED = "accepted_order"
EVENT_ARRIVED = "arrived_at_merchant"
EVENT_DELIVERED = "delivered_to_customer"

SPAWN_POSITION: Tuple[int, int] = (50, 50)
DEFAULT_SPEED = 10


def _sign(value: int) -> int:
    """Return -1, 0 or 1 according to the sign of ``value``."""
    return (value > 0) - (value < 0)


class Rider:
    """A delivery rider that moves on the street grid."""

    def __init__(self, rider_id: int) -> None:
        """Create an idle rider at the spawn position.

        Args:
            rider_id: Unique rider identifier.
        """
        self.rider_id: int = rider_id
        self.x, self.y = SPAWN_POSITION
        self.speed: int = int(get_value("rider_speed", DEFAULT_SPEED))
        self.state: int = STATE_IDLE
        self.orders: List[Order] = []
        self.special_case: bool = False
        self.log: List[Dict[str, Any]] = []
        self.path: List[Tuple[int, int]] = []
        self.stall_count: int = 0
        self._planner = PathPlanner()
        self._record(EVENT_SPAWN, None)

    # ------------------------------------------------------------------
    # Logging helpers
    # ------------------------------------------------------------------
    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append a log entry describing the current rider status.

        Args:
            event: Event name (spawn, accepted_order, ...).
            order_id: Related order identifier, or None.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": int(get_value("current_tick", 0)),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s (order=%s)", self.rider_id, event, order_id)

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------
    def move(self) -> None:
        """Advance the rider by one tick.

        Handles idle-to-picking-up activation, movement along the planned
        path using up to ``speed`` units, arrival at the restaurant, and
        delivery to the customer. A rider waits when a lower-id rider blocks
        its next waypoint, up to ``stall_limit`` ticks, to avoid deadlock.
        """
        if self.state == STATE_IDLE:
            if not self.orders:
                self.special_case = False
                return
            self._activate_order()

        budget = self.speed
        while budget > 0 and self.orders and self.state != STATE_IDLE:
            if self.path:
                if self._is_blocked():
                    break
                budget = self._step(budget)
            elif self.state == STATE_PICKING_UP:
                self._arrive_at_merchant()
            else:
                self._complete_delivery()

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "order_ids": [order.id for order in self.orders],
            "special_case": self.special_case,
            "log": list(self.log),
        }

    # ------------------------------------------------------------------
    # State transitions
    # ------------------------------------------------------------------
    def _activate_order(self) -> None:
        """Start picking up the first queued order (idle -> picking up)."""
        order = self.orders[0]
        self.state = STATE_PICKING_UP
        order.status = STATUS_PICKING_UP
        self._set_path(order.res_center)
        self._record(EVENT_ACCEPTED, order.id)

    def _arrive_at_merchant(self) -> None:
        """Handle arrival at the restaurant (picking up -> delivering)."""
        order = self.orders[0]
        self.state = STATE_DELIVERING
        order.status = STATUS_DELIVERING
        self._record(EVENT_ARRIVED, order.id)
        self._set_path(order.cus_center)

    def _complete_delivery(self) -> None:
        """Finish the current order and release the rider.

        Completes the order, records the delivery, credits the delivery fee
        to system funds, and either becomes idle or starts the next order.
        """
        order = self.orders.pop(0)
        order.complete(int(get_value("current_tick", 0)))
        self._record(EVENT_DELIVERED, order.id)
        fee = int(get_value("delivery_fee", 100))
        set_value("funds", int(get_value("funds", 0)) + fee)
        self.path = []
        self.stall_count = 0
        self.special_case = False
        self.state = STATE_IDLE
        if self.orders:
            self._activate_order()

    # ------------------------------------------------------------------
    # Movement helpers
    # ------------------------------------------------------------------
    def _set_path(self, target: Tuple[int, int]) -> None:
        """Plan a path to ``target`` and update the special-case flag.

        ``special_case`` is True when the target is directly above/below
        (same x) or directly beside (same y) the rider, i.e. a straight run.

        Args:
            target: Destination cell centre.
        """
        self.path = self._planner.plan((self.x, self.y), target)
        self.stall_count = 0
        self.special_case = bool(self.path) and (
            self.x == target[0] or self.y == target[1]
        )

    def _is_blocked(self) -> bool:
        """Return True if this rider should yield the next waypoint.

        A rider yields to a lower-id rider standing on its next waypoint.
        After ``stall_limit`` waiting ticks it proceeds anyway, which
        prevents indefinite deadlock.

        Returns:
            Whether movement should pause this tick.
        """
        if not self.path:
            return False
        wx, wy = self.path[0]
        limit = int(get_value("stall_limit", 3))
        for other in get_value("riders") or []:
            if other is self:
                continue
            if other.x == wx and other.y == wy and other.rider_id < self.rider_id:
                if self.stall_count < limit:
                    self.stall_count += 1
                    return True
                self.stall_count = 0
                return False
        self.stall_count = 0
        return False

    def _step(self, budget: int) -> int:
        """Move toward the next waypoint using at most ``budget`` units.

        Args:
            budget: Remaining movement units for this tick.

        Returns:
            Movement units left unused after this step.
        """
        tx, ty = self.path[0]
        dx = tx - self.x
        dy = ty - self.y
        remaining = budget
        move_x = min(remaining, abs(dx))
        self.x += move_x * _sign(dx)
        remaining -= move_x
        move_y = min(remaining, abs(dy))
        self.y += move_y * _sign(dy)
        remaining -= move_y
        if (self.x, self.y) == (tx, ty):
            self.path.pop(0)
        return remaining

"""Order management: rider assignment and pending-order handling."""

import logging
from typing import List, Optional

from globalvar import get_value
from order_model import STATUS_PENDING, Order
from rider import Rider

logger = logging.getLogger(__name__)

LOAD_PENALTY = 100


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the best rider for an order.

    The primary criterion is the distance from the rider's current position
    to the restaurant (closest rider first). Each already-queued order adds
    a load penalty so that work is balanced across riders. Riders already at
    ``max_orders_per_rider`` are skipped.

    Args:
        order: The order to assign.

    Returns:
        The chosen :class:`Rider`, or None if no rider has spare capacity.
    """
    max_load = int(get_value("max_orders_per_rider", 3))
    rx, ry = order.res_center
    best: Optional[Rider] = None
    best_key = None
    for rider in get_value("riders") or []:
        load = len(rider.orders)
        if load >= max_load:
            continue
        distance = abs(rider.x - rx) + abs(rider.y - ry)
        key = (distance + LOAD_PENALTY * load, rider.rider_id)
        if best_key is None or key < best_key:
            best, best_key = rider, key
    return best


class OrderManager:
    """Creates, assigns and tracks orders."""

    def __init__(self) -> None:
        """Create an empty order manager."""
        self.orders: List[Order] = []

    def create_order(self, res_pos, cus_pos, tick: int) -> Order:
        """Create and register a new pending order.

        Args:
            res_pos: Restaurant ``[x, y]``.
            cus_pos: Customer ``[x, y]``.
            tick: Simulated time of creation.

        Returns:
            The new :class:`Order`.
        """
        order = Order(res_pos, cus_pos)
        order.start_time = tick
        self.orders.append(order)
        logger.info("Created order %s at tick %s", order.id, tick)
        return order

    def assign(self, order: Order) -> Optional[Rider]:
        """Assign an order to the best available rider.

        Args:
            order: Pending order to assign.

        Returns:
            The rider the order was given to, or None if none is available.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

    def retry_pending(self) -> List[Order]:
        """Try to assign every pending order.

        Returns:
            Orders that were assigned during this call.
        """
        assigned: List[Order] = []
        for order in self.pending_orders():
            if self.assign(order) is not None:
                assigned.append(order)
        return assigned

    def pending_orders(self) -> List[Order]:
        """Return orders that still wait for a rider."""
        return [o for o in self.orders if o.status == STATUS_PENDING]

    def active_orders(self) -> List[Order]:
        """Return orders that are not yet completed."""
        return [o for o in self.orders if o.status != "completed"]

    def get_order(self, order_id: int) -> Optional[Order]:
        """Look up an order by identifier.

        Args:
            order_id: Order identifier.

        Returns:
            The matching order, or None.
        """
        for order in self.orders:
            if order.id == order_id:
                return order
        return None

import json
import logging
import math
import os

    @property
    def rider_id_counter(self) -> int:
        """Last issued rider identifier."""
        return int(get_value("rider_id_counter", 0))

    @property
    def current_tick(self) -> int:
        """Current simulated tick."""
        return int(get_value("current_tick", 0))

    @property
    def is_running(self) -> bool:
        """Whether the main loop should keep running."""
        return bool(get_value("is_running", True))

        """Create a new idle rider with a unique identifier.

        new_id = self.rider_id_counter + 1
        set_value("rider_id_counter", new_id)
        rider = Rider(new_id)
        self.riders.append(rider)
        logger.info("Recruited rider %s", new_id)

        The recruitment cost is deducted from the funds.

        Returns:

        if self.funds >= threshold:

    # ------------------------------------------------------------------
    def submit_order(self, res_pos, cus_pos) -> Dict[str, Any]:
        """Create an order, assign it and plan its route.

        Args:
            res_pos: Restaurant ``[x, y]`` (0-800).
            cus_pos: Customer ``[x, y]`` (0-800).

        Returns:
            Dictionary with keys ``order``, ``rider`` (or None), ``new_rider``,
            ``route`` (or None) and ``estimated_ticks`` (or None).

        Raises:
            ValueError: If a coordinate is invalid.
        """
        new_rider = self.check_recruitment()
        order = self.order_manager.create_order(res_pos, cus_pos, self.current_tick)
        rider = self.order_manager.assign(order)
        route: Optional[Route] = None
        estimated: Optional[int] = None
        if rider is not None:
            route = self.planner.build_route(
                (rider.x, rider.y), order.res_center, order.cus_center
            )
            estimated = math.ceil(route.total_distance / rider.speed) if rider.speed else None
        return {
            "order": order,
            "rider": rider,
            "new_rider": new_rider,
            "route": route,
            "estimated_ticks": estimated,
        }

    def run_until_complete(self, order: Order, max_ticks: Optional[int] = None) -> int:
        """Advance the simulation until the order is delivered.

        Args:
            order: Order to wait for.
            max_ticks: Maximum number of ticks to simulate.

        Returns:
            Number of ticks simulated.
        """
        limit = max_ticks or int(get_value("max_ticks_per_order", 10000))
        start = self.current_tick
        while order.status != STATUS_COMPLETED and self.current_tick - start < limit:
            self.tick()
        if order.status != STATUS_COMPLETED:
            logger.warning("Order %s not completed within %s ticks", order.id, limit)
        return self.current_tick - start

    # ------------------------------------------------------------------
    # Simulation clock
    # ------------------------------------------------------------------
    def tick(self) -> None:
        """Advance the simulation by one tick.

        Recruits riders if funded, retries pending orders, and moves every
        rider. Exceptions raised by one rider are logged and that rider is
        reset to a consistent state so the rest of the system keeps running.
        """
        set_value("current_tick", self.current_tick + 1)
        self.check_recruitment()
        self.order_manager.retry_pending()
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # noqa: BLE001 - recovery path
                logger.exception("Rider %s failed during tick; recovering", rider.rider_id)
                self._recover_rider(rider)

    @staticmethod
    def _recover_rider(rider: Rider) -> None:
        """Restore a rider to a consistent state after an unexpected error."""
        rider.path = []
        rider.stall_count = 0
        rider.special_case = False
        if not rider.orders:
            rider.state = STATE_IDLE
        elif rider.state == STATE_IDLE:
            rider.state = STATE_PICKING_UP

    # ------------------------------------------------------------------
    # Persistence
    # ------------------------------------------------------------------
    def save_state(self, path: str = STATE_FILE) -> None:
        """Write parameters, riders and order history to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        state = {
            "parameters": {k: get_value(k) for k in DEFAULT_PARAMETERS},
            "funds": self.funds,
            "current_tick": self.current_tick,
            "rider_id_counter": self.rider_id_counter,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.orders],
        }
        tmp_path = path + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as fh:
            json.dump(state, fh, indent=2)
        os.replace(tmp_path, path)
        logger.info("System state saved to %s", path)


def load_config(path: str = CONFIG_FILE) -> bool:
    """Load system parameters from a JSON configuration file.

    The file may contain a ``parameters`` object or the parameters at top
    level. Unknown keys and invalid values are ignored with a warning.

    Args:
        path: Configuration file path.

    Returns:
        True if the file was read, otherwise False.
    """
    if not os.path.exists(path):
        logger.info("No configuration file at %s; using defaults", path)
        return False
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except (OSError, json.JSONDecodeError) as exc:
        logger.error("Could not read configuration %s: %s", path, exc)
        return False
    if not isinstance(data, dict):
        logger.error("Configuration %s must contain a JSON object", path)
        return False
    params = data.get("parameters", data)
    if not isinstance(params, dict):
        return False
    for key in DEFAULT_PARAMETERS:
        if key not in params:
            continue
        try:
            value = int(params[key])
        except (TypeError, ValueError):
            logger.warning("Parameter %s has an invalid value; ignored", key)
            continue
        if value < 0 or (key in POSITIVE_PARAMETERS and value == 0):
            logger.warning("Parameter %s out of range; ignored", key)
            continue
        set_value(key, value)
    return True


def init_system() -> SystemManager:
    """Initialise global state, load configuration and build the system.

    Returns:
        A ready-to-use :class:`SystemManager`.
    """
    globalvar._init()
    load_config(CONFIG_FILE)
    set_value("funds", int(get_value("initial_funds", 300)))
    system = SystemManager()
    logger.info("System initialised with %s rider(s)", len(system.riders))
    return system

from typing import List, Optional, Sequence, Tuple

from order_model import STATUS_COMPLETED, Order

Cell = Tuple[int, int]  # (row, col)


class DisplayManager:
    """Renders the 8x8 map, the route graph and status tables as text."""

    SYMBOL_RIDER = "R"
    SYMBOL_RESTAURANT = "S"
    SYMBOL_CUSTOMER = "C"
    SYMBOL_EMPTY = "."
    SYMBOL_STREET = "#"

    def __init__(self, grid_size: int = GRID_SIZE) -> None:
        """Create a display manager.

        Args:
            grid_size: Number of cells along each axis.
        """
        self.grid_size = grid_size

    # ------------------------------------------------------------------
    # Geometry helpers
    # ------------------------------------------------------------------
    @staticmethod
    def cell_of(point: Tuple[int, int]) -> Cell:
        """Return the ``(row, col)`` grid cell for a coordinate point."""
        return (PathPlanner.to_cell_index(point[1]), PathPlanner.to_cell_index(point[0]))

    @staticmethod
    def _line_cells(a: Cell, b: Cell) -> List[Cell]:
        """Return all cells on the axis-aligned line from ``a`` to ``b``."""
        (r1, c1), (r2, c2) = a, b
        if r1 == r2:
            step = 1 if c2 >= c1 else -1
            return [(r1, c) for c in range(c1, c2 + step, step)]
        if c1 == c2:
            step = 1 if r2 >= r1 else -1
            return [(r, c1) for r in range(r1, r2 + step, step)]
        return [a, b]

    @classmethod
    def route_cells(cls, waypoints: Sequence[Tuple[int, int]]) -> List[Cell]:
        """Expand route waypoints into the ordered list of grid cells they cross.

        Args:
            waypoints: Route waypoints in coordinate space.

        Returns:
            Ordered cells without consecutive duplicates.
        """
        cells: List[Cell] = []
        prev: Optional[Cell] = None
        for point in waypoints:
            cur = cls.cell_of(point)
            if prev is None:
                cells.append(cur)
            else:
                cells.extend(cls._line_cells(prev, cur)[1:])
            prev = cur
        return cells

    # ------------------------------------------------------------------
    # Map rendering
    # ------------------------------------------------------------------
    def render_map(self, riders: Sequence[Rider], orders: Sequence[Order],
                   route: Optional[Route] = None) -> str:
        """Render the 8x8 map with riders, restaurants, customers and streets.

        Symbols: ``R`` rider, ``S`` restaurant, ``C`` customer, ``#`` cell
        crossed by the highlighted route, ``.`` empty. Entities take priority
        over street markings.

        Args:
            riders: Riders to display.
            orders: Orders whose active restaurants and customers are shown.
            route: Optional route to highlight.

        Returns:
            Multi-line string.
        """
        n = self.grid_size
        grid = [[self.SYMBOL_EMPTY] * n for _ in range(n)]
        if route is not None:
            for r, c in self.route_cells(route.waypoints):
                grid[r][c] = self.SYMBOL_STREET
        for order in orders:
            if order.status == STATUS_COMPLETED:
                continue
            r, c = self.cell_of(order.res_center)
            grid[r][c] = self.SYMBOL_RESTAURANT
            r, c = self.cell_of(order.cus_center)
            grid[r][c] = self.SYMBOL_CUSTOMER
        for rider in riders:
            r, c = self.cell_of((rider.x, rider.y))
            grid[r][c] = self.SYMBOL_RIDER

        lines = ["    " + " ".join(str(c) for c in range(n))]
        for r in range(n):
            lines.append(f"{r:>3} " + " ".join(grid[r]))
        lines.append("Legend: R=rider S=restaurant C=customer #=street(route) .=empty")
        return "\n".join(lines)

    def render_route_graph(self, route: Route, restaurant: Tuple[int, int],
                           customer: Tuple[int, int], rider_pos: Tuple[int, int]) -> str:
        """Render a route as a connected line-segment graph on the ASCII map.

        Each grid cell is a node drawn at an even canvas position; route
        segments are drawn with ``-`` (horizontal) and ``|`` (vertical).

        Args:
            route: Planned route.
            restaurant: Restaurant centre point.
            customer: Customer centre point.
            rider_pos: Rider start point (drawn as R).

        Returns:
            Multi-line string.
        """
        n = self.grid_size
        size = 2 * n - 1
        canvas = [[" "] * size for _ in range(size)]
        for r in range(n):
            for c in range(n):
                canvas[2 * r][2 * c] = self.SYMBOL_EMPTY
        cells = self.route_cells(route.waypoints)
        for r, c in cells:
            canvas[2 * r][2 * c] = self.SYMBOL_STREET
        for a, b in zip(cells, cells[1:]):
            mid_r, mid_c = a[0] + b[0], a[1] + b[1]
            canvas[mid_r][mid_c] = "-" if a[0] == b[0] else "|"
        for point, symbol in ((restaurant, self.SYMBOL_RESTAURANT),
                              (customer, self.SYMBOL_CUSTOMER),
                              (rider_pos, self.SYMBOL_RIDER)):
            r, c = self.cell_of(point)
            canvas[2 * r][2 * c] = symbol

        header = "   " + "".join(str(c) if c % 2 == 0 else " " for c in range(size))
        lines = [header]
        for row in range(size):
            label = f"{row // 2:>2} " if row % 2 == 0 else "   "
            lines.append(label + "".join(canvas[row]))
        lines.append("Legend: R=rider start S=restaurant C=customer #=route node "
                     "-/|=route segment")
        return "\n".join(lines)

    # ------------------------------------------------------------------
    # Tables
    # ------------------------------------------------------------------
    @staticmethod
    def state_name(state: int) -> str:
        """Return the human-readable name for a rider state."""
        return STATE_NAMES.get(state, "unknown")

    def render_status(self, funds: int, riders: Sequence[Rider],
                      orders: Sequence[Order]) -> str:
        """Render the system snapshot: funds, riders and active orders.

        Args:
            funds: Current fund balance.
            riders: All riders.
            orders: All orders (completed ones are filtered out).

        Returns:
            Multi-line string.
        """
        lines = [f"System funds: {funds}", f"Riders: {len(riders)}"]
        for rider in riders:
            order_ids = [o.id for o in rider.orders] or "none"
            lines.append(
                f"  Rider {rider.rider_id:<3} pos=({rider.x},{rider.y}) "
                f"state={self.state_name(rider.state)} speed={rider.speed} "
                f"orders={order_ids} special_case={rider.special_case}"
            )
        active = [o for o in orders if o.status != STATUS_COMPLETED]
        lines.append(f"Active orders: {len(active)}")
        for order in active:
            rider_text = order.rider_id if order.rider_id is not None else "-"
            lines.append(
                f"  Order #{order.id} status={order.status} rider={rider_text} "
                f"restaurant={order.res_pos} customer={order.cus_pos}"
            )
        return "\n".join(lines)

    def render_orders(self, orders: Sequence[Order]) -> str:
        """Render a table of all orders.

        Args:
            orders: Orders to display.

        Returns:
            Multi-line string.
        """
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Dist':>6} {'Taken':>6} {'Rider':>6} {'Status':>11}")
        lines = [header, "-" * len(header)]
        for o in orders:
            taken = o.time_taken if o.time_taken is not None else "-"
            rider_text = o.rider_id if o.rider_id is not None else "-"
            lines.append(
                f"{o.id:>4} {o.start_time:>6} {str(o.res_pos):>12} {str(o.cus_pos):>12} "
                f"{o.distance:>6} {str(taken):>6} {str(rider_text):>6} {o.status:>11}"
            )
        return "\n".join(lines)

    def render_rider_logs(self, riders: Sequence[Rider]) -> str:
        """Render the recorded status changes of every rider.

        Args:
            riders: Riders whose logs should be displayed.

        Returns:
            Multi-line string.
        """
        if not riders:
            return "No riders."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider {rider.rider_id}:")
            if not rider.log:
                lines.append("  (no records)")
            for entry in rider.log:
                order_text = f" order=#{entry['order_id']}" if entry["order_id"] else ""
                lines.append(
                    f"  [tick {entry['tick']:>5}] {entry['event']:<20}{order_text} "
                    f"pos=({entry['x']},{entry['y']}) "
                    f"state={self.state_name(entry['state'])}"
                )
        return "\n".join(lines)

def parse_coordinates(tokens: Sequence[str]) -> Tuple[int, int, int, int]:
    """Parse and validate four integer coordinates.

    Args:
        tokens: Exactly four string tokens.

    Returns:
        Tuple ``(restaurant_x, restaurant_y, customer_x, customer_y)``.

    Raises:
        ValueError: If the count, type or range is wrong.
    """
    if len(tokens) != 4:
        raise ValueError(f"expected 4 integers, got {len(tokens)}")
    values: List[int] = []
    for token in tokens:
        try:
            value = int(token)
        except ValueError as exc:
            raise ValueError(f"'{token}' is not an integer") from exc
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"coordinate {value} is out of range 0-{MAP_SIZE}")
        values.append(value)
    return values[0], values[1], values[2], values[3]


def _cell_text(point: Tuple[int, int]) -> str:
    """Describe a point as grid cell (col,row) and its centre."""
    col = PathPlanner.to_cell_index(point[0])
    row = PathPlanner.to_cell_index(point[1])
    return f"cell (col {col}, row {row}) centre {PathPlanner.cell_center_point(point)}"


class InputManager:
    """Reads user commands, dispatches them to the system and prints results."""

    def __init__(self, system) -> None:
        """Create the command interface.

        Args:
            system: The :class:`system.SystemManager` to control.
        """
        self.system = system
        self.display = DisplayManager()
        self.command_history: List[str] = []
        self._handlers: Dict[str, Callable[[List[str]], None]] = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
        }

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> None:
        """Run the interactive loop until the user quits."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.is_running:
            try:
                line = input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            self.handle_line(line)

    def handle_line(self, line: str) -> None:
        """Process one line of user input with error recovery.

        Args:
            line: Raw input line.
        """
        text = line.strip()
        if not text:
            return
        if text == "!!":
            if not self.command_history:
                print("No previous command.")
                return
            text = self.command_history[-1]
            print(f"Repeating: {text}")
        self.command_history.append(text)
        try:
            self.dispatch(text)
        except ValueError as exc:
            print(f"Input error: {exc}")
        except Exception as exc:  # noqa: BLE001 - keep the loop alive
            logger.exception("Command failed: %s", text)
            print(f"Unexpected error: {exc}. The system state has been preserved.")

    def dispatch(self, text: str) -> None:
        """Parse a command line and invoke its handler.

        Args:
            text: Command text.

        Raises:
            ValueError: If the command is unknown.
        """
        tokens = text.replace(",", " ").split()
        name = ALIASES.get(tokens[0].lower())
        if name is None:
            raise ValueError(f"unknown command '{tokens[0]}'. Type 'help' for commands.")
        self._handlers[name](tokens[1:])

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def _cmd_start(self, args: List[str]) -> None:
        """Create and dispatch an order, then run it to completion."""
        if not args:
            prompt = "Enter restaurant_x restaurant_y customer_x customer_y (0-800): "
            args = input(prompt).replace(",", " ").split()
        rx, ry, cx, cy = parse_coordinates(args)
        result = self.system.submit_order((rx, ry), (cx, cy))
        order = result["order"]
        rider = result["rider"]

        print(f"Order #{order.id} created at tick {order.start_time}")
        print(f"Restaurant: {_cell_text(order.res_center)}")
        print(f"Customer:   {_cell_text(order.cus_center)}")
        print(f"New rider added: {'yes' if result['new_rider'] else 'no'}")
        if rider is None:
            print("No rider available. The order is pending and will be assigned "
                  "when a rider becomes free.")
            return
        print(f"Assigned rider {rider.rider_id} (speed {rider.speed} units/tick)")
        route = result["route"]
        print(self.display.render_route_graph(
            route, order.res_center, order.cus_center, (rider.x, rider.y)))
        print(f"Route distance: {route.total_distance} units, "
              f"estimated time: {result['estimated_ticks']} tick(s)")
        self.system.run_until_complete(order)
        if order.status == STATUS_COMPLETED:
            print(f"Delivered at tick {order.completion_time}; time taken "
                  f"{order.time_taken} tick(s); efficiency {order.efficiency} units/tick")
        else:
            print("Delivery did not finish within the tick limit; check 'orders'.")

    def _cmd_status(self, args: List[str]) -> None:
        """Print the system snapshot."""
        print(self.display.render_status(
            self.system.funds, self.system.riders, self.system.order_manager.orders))

    def _cmd_orders(self, args: List[str]) -> None:
        """Print details of all orders."""
        print(self.display.render_orders(self.system.order_manager.orders))

    def _cmd_riders(self, args: List[str]) -> None:
        """Print the status-change log of all riders."""
        print(self.display.render_rider_logs(self.system.riders))

    def _cmd_map(self, args: List[str]) -> None:
        """Print the ASCII map."""
        print(self.display.render_map(self.system.riders, self.system.order_manager.orders))

    def _cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation by one or more ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError as exc:
                raise ValueError(f"'{args[0]}' is not an integer") from exc
        if not 1 <= count <= 10000:
            raise ValueError("tick count must be between 1 and 10000")
        for _ in range(count):
            self.system.tick()
        print(f"Advanced {count} tick(s); current tick {self.system.current_tick}")

    def _cmd_history(self, args: List[str]) -> None:
        """Print the command history."""
        if not self.command_history:
            print("History is empty.")
            return
        for idx, cmd in enumerate(self.command_history, start=1):
            print(f"{idx:>4}  {cmd}")

    def _cmd_help(self, args: List[str]) -> None:
        """Print the help text."""
        print(HELP_TEXT)

    def _cmd_quit(self, args: List[str]) -> None:
        """Stop the main loop."""
        self.system.is_running = False
        print("Goodbye.")

"""Command-line entry point for the Food Delivery Order Dispatch Simulator."""

import argparse
import logging
import sys

from input_manager import InputManager
from system import STATE_FILE, init_system


def configure_logging(debug: bool) -> None:
    """Configure root logging.

    Args:
        debug: When True, log at DEBUG level; otherwise WARNING.
    """
    logging.basicConfig(
        level=logging.DEBUG if debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )


def main(argv=None) -> int:
    """Run the simulator.

    Args:
        argv: Optional argument list (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    parser.add_argument("--state-file", default=STATE_FILE,
                        help="path of the JSON file where state is saved on exit")
    args = parser.parse_args(argv)
    configure_logging(args.debug)
    logger = logging.getLogger("main")

    try:
        system = init_system()
    except Exception as exc:  # noqa: BLE001
        logger.exception("Initialisation failed")
        print(f"Fatal error during initialisation: {exc}")
        return 1

    try:
        InputManager(system).run()
    finally:
        try:
            system.save_state(args.state_file)
            print(f"State saved to {args.state_file}")
        except OSError as exc:
            print(f"Warning: could not save state: {exc}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

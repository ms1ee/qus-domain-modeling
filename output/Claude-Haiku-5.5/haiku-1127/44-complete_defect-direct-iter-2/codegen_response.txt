"""Binary min-heap priority queue used by the Dijkstra implementation.

The queue stores (key, item) pairs and always returns the pair with the
smallest key first. Keys may be any mutually comparable values, including
tuples, which allows lexicographic cost comparison.

Insertion and removal run in O(log n) time, which gives Dijkstra's algorithm
its O((V + E) log V) bound when combined with lazy deletion.
"""


class MinPQ:
    """A binary min-heap keyed priority queue (lazy deletion friendly)."""

    def __init__(self):
        # Parallel arrays: _keys[i] is the priority of _items[i].
        self._keys = []
        self._items = []

    def __len__(self):
        return len(self._keys)

    def size(self):
        """Return the number of entries currently in the queue."""
        return len(self._keys)

    def is_empty(self):
        """Return True when the queue holds no entries."""
        return not self._keys

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._keys:
            raise IndexError("priority queue is empty")
        return self._keys[0]

    def min_item(self):
        """Return the item with the smallest key without removing it."""
        if not self._items:
            raise IndexError("priority queue is empty")
        return self._items[0]

    def insert(self, key, item):
        """Insert an item with the given priority key."""
        self._keys.append(key)
        self._items.append(item)
        self._swim(len(self._keys) - 1)

    def del_min(self):
        """Remove and return the (key, item) pair with the smallest key."""
        if not self._keys:
            raise IndexError("priority queue underflow")
        top_key = self._keys[0]
        top_item = self._items[0]
        last_key = self._keys.pop()
        last_item = self._items.pop()
        if self._keys:
            self._keys[0] = last_key
            self._items[0] = last_item
            self._sink(0)
        return top_key, top_item

    # ------------------------------------------------------------------
    # Heap maintenance helpers
    # ------------------------------------------------------------------
    def _less(self, i, j):
        return self._keys[i] < self._keys[j]

    def _swap(self, i, j):
        self._keys[i], self._keys[j] = self._keys[j], self._keys[i]
        self._items[i], self._items[j] = self._items[j], self._items[i]

    def _swim(self, k):
        """Move the element at index k up until heap order is restored."""
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._swap(k, parent)
                k = parent
            else:
                break

    def _sink(self, k):
        """Move the element at index k down until heap order is restored."""
        n = len(self._keys)
        while True:
            child = 2 * k + 1
            if child >= n:
                break
            if child + 1 < n and self._less(child + 1, child):
                child += 1
            if self._less(child, k):
                self._swap(child, k)
                k = child
            else:
                break

"""Core graph data structures for the subway network.

The Graph stores an undirected weighted adjacency list keyed by integer
station numbers. Station numbers are assigned by the caller (see
subway_network.SubwayNetwork), not by the Graph itself.
"""

import math


class Edge:
    """An undirected weighted edge between two station numbers.

    Attributes:
        source: station number of one endpoint.
        dest: station number of the other endpoint.
        distance: physical distance in meters (also used as the weight).
        line: name of the subway line that operates this segment.
    """

    def __init__(self, source, dest, distance, line):
        if source == dest:
            raise ValueError("self-loop edges are not allowed")
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("edge distance must be a number")
        if not math.isfinite(distance) or distance < 0:
            raise ValueError("edge distance must be a finite non-negative number")
        if not isinstance(line, str) or not line:
            raise ValueError("edge line must be a non-empty string")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by shortest-path algorithms (equals distance)."""
        return self.distance

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"vertex {vertex} is not an endpoint of this edge")

    def __repr__(self):
        return (f"Edge({self.source}, {self.dest}, "
                f"distance={self.distance}, line={self.line!r})")


class Graph:
    """Weighted undirected graph with adjacency lists.

    Attributes:
        station_lines: dict mapping station name -> list of line names the
            station belongs to (used for transfer detection).
    """

    def __init__(self):
        self._adj = {}
        self._edge_count = 0
        self.station_lines = {}

    def add_vertex(self, vertex):
        """Register a vertex (station number) if it is not already present."""
        self._adj.setdefault(vertex, [])

    def has_vertex(self, vertex):
        """Return True if the vertex exists in the graph."""
        return vertex in self._adj

    def set_station_lines(self, station_name, lines):
        """Set the list of lines that serve a station (stored as a copy)."""
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an undirected edge; it is registered in both endpoint lists."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adj[edge.source].append(edge)
        self._adj[edge.dest].append(edge)
        self._edge_count += 1

    def adjacent(self, vertex):
        """Return the list of edges incident to ``vertex`` (do not mutate)."""
        return self._adj.get(vertex, [])

    def vertices(self):
        """Return the list of all vertex numbers."""
        return list(self._adj.keys())

    def edges(self):
        """Return each undirected edge exactly once."""
        seen = set()
        result = []
        for edge_list in self._adj.values():
            for edge in edge_list:
                if id(edge) not in seen:
                    seen.add(id(edge))
                    result.append(edge)
        return result

    def num_vertices(self):
        """Return the number of vertices."""
        return len(self._adj)

    def num_edges(self):
        """Return the number of undirected edges."""
        return self._edge_count

"""Subway network model: station numbering, graph construction, data
management (line/station/distance updates) and integrity checks.
"""

import math
from collections import deque

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

# Representative stations per main line, used by the coverage check.
REPRESENTATIVE_STATIONS = {
    "Line 1": ["苹果园", "天安门东", "国贸"],
    "Line 2": ["西直门", "建国门", "东直门"],
    "Line 4": ["圆明园", "北京大学东门"],
    "Line 5": ["天通苑北", "北新桥"],
    "Line 6": ["金安桥"],
    "Line 8": ["鼓楼大街"],
    "Line 10": ["亮马桥"],
    "Line 13": ["回龙观"],
    "Line 14": ["张郭庄"],
    "Line 15": ["望京西"],
    "Yizhuang Line": ["宋家庄"],
}

# Distances outside this range (meters) are reported as suspicious.
MIN_PLAUSIBLE_DISTANCE_M = 100
MAX_PLAUSIBLE_DISTANCE_M = 10000


def _validate_distance(distance):
    """Ensure a distance is a positive finite number and return it."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError(f"distance {distance!r} is not a number")
    if not math.isfinite(distance) or distance <= 0:
        raise ValueError(f"distance {distance!r} must be a positive number")
    return distance


def _fold(text):
    """Normalized key for case/space-insensitive station name matching."""
    return "".join(text.split()).casefold()


class SubwayNetwork:
    """Holds the subway graph plus the name <-> number station mapping.

    Attributes:
        graph: the weighted undirected Graph (vertices are station numbers).
        station_to_number: dict station name -> station number.
        number_to_name: list where index is the station number.
        connections: list of connection records (dicts with source, dest,
            distance, line).
    """

    def __init__(self, data=None):
        if data is None:
            data = load_subway_data(DATA_FILE_PATH)
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_name = []
        self.connections = []
        self.line_stations = {}
        self._declared_stations = set()
        self._declared_lines = {}
        self._undeclared_stations = set()
        self._load_problems = []
        self._folded_index = {}
        self._load(data)

    # ------------------------------------------------------------------
    # Construction
    # ------------------------------------------------------------------
    def _load(self, data):
        """Populate the network from the parsed JSON dataset."""
        for station in data.get("stations", []):
            try:
                name = station["name"]
                lines = station.get("lines", [])
                if not isinstance(name, str) or not name.strip():
                    raise ValueError("empty station name")
                if not isinstance(lines, list):
                    raise ValueError("'lines' must be a list")
                name = name.strip()
                clean_lines = [str(line).strip() for line in lines if str(line).strip()]
            except (KeyError, TypeError, ValueError, AttributeError) as exc:
                self._load_problems.append(f"ERROR: invalid station record {station!r}: {exc}")
                continue
            self._declared_stations.add(name)
            self._declared_lines[name] = list(dict.fromkeys(clean_lines))
            self._ensure_station(name, clean_lines)

        for conn in data.get("connections", []):
            try:
                self._connect(conn["source"], conn["dest"], conn["distance"], conn["line"])
            except (KeyError, TypeError, ValueError, AttributeError) as exc:
                self._load_problems.append(f"ERROR: invalid connection {conn!r}: {exc}")

    def _ensure_station(self, name, lines=()):
        """Return the station number for ``name``, creating the station if needed.

        Any lines given are merged into the station's line list.
        """
        if name not in self.station_to_number:
            number = len(self.number_to_name)
            self.station_to_number[name] = number
            self.number_to_name.append(name)
            self.graph.add_vertex(number)
            self.graph.set_station_lines(name, [])
            self._folded_index.setdefault(_fold(name), name)
        current = list(self.graph.station_lines.get(name, []))
        for line in lines:
            if line not in current:
                current.append(line)
        self.graph.set_station_lines(name, current)
        return self.station_to_number[name]

    def _connect(self, src, dst, distance, line):
        """Create an undirected connection between two stations on a line."""
        if not all(isinstance(x, str) for x in (src, dst, line)):
            raise ValueError("station and line names must be strings")
        src, dst, line = src.strip(), dst.strip(), line.strip()
        if not src or not dst or not line:
            raise ValueError("empty station or line name")
        if src == dst:
            raise ValueError("connection endpoints must differ")
        _validate_distance(distance)
        for name in (src, dst):
            if name not in self._declared_stations:
                self._undeclared_stations.add(name)
        a = self._ensure_station(src, [line])
        b = self._ensure_station(dst, [line])
        self.graph.add_edge(Edge(a, b, distance, line))
        self.connections.append({"source": src, "dest": dst,
                                 "distance": distance, "line": line})
        self.line_stations.setdefault(line, set()).update((src, dst))

    # ------------------------------------------------------------------
    # Queries
    # ------------------------------------------------------------------
    def station_count(self):
        """Return the number of stations in the network."""
        return len(self.station_to_number)

    def station_exists(self, name):
        """Return True if the exact station name exists."""
        return name in self.station_to_number

    def get_number(self, name):
        """Return the station number for an exact station name."""
        return self.station_to_number[name]

    def get_name(self, number):
        """Return the station name for a station number."""
        return self.number_to_name[number]

    def standardize_station_name(self, raw):
        """Map user input to an official station name, or None if unknown.

        Matching ignores surrounding whitespace, letter case, inner spaces and
        a trailing "站" character.
        """
        if not isinstance(raw, str):
            return None
        text = raw.strip()
        if not text:
            return None
        candidates = [text]
        if text.endswith("站") and len(text) > 1:
            candidates.append(text[:-1].strip())
        for candidate in candidates:
            if candidate in self.station_to_number:
                return candidate
        for candidate in candidates:
            key = _fold(candidate)
            if key and key in self._folded_index:
                return self._folded_index[key]
        return None

    def lines_of(self, name):
        """Return the list of lines serving a station."""
        return list(self.graph.station_lines.get(name, []))

    def all_line_names(self):
        """Return the sorted list of all known line names."""
        names = set(self.line_stations.keys())
        for lines in self.graph.station_lines.values():
            names.update(lines)
        return sorted(names)

    def transfer_stations(self):
        """Return names of stations served by two or more lines."""
        return [name for name in self.station_to_number
                if len(self.graph.station_lines.get(name, [])) >= 2]

    # ------------------------------------------------------------------
    # Data update mechanisms
    # ------------------------------------------------------------------
    def _require_station(self, name):
        std = self.standardize_station_name(name)
        if std is None:
            raise ValueError(f"unknown station '{name}'")
        return std

    def add_line(self, line_name, station_names, distances):
        """Add a new line as an ordered station sequence with distances.

        Args:
            line_name: name of the new line (must not exist yet).
            station_names: ordered list of at least two station names. Unknown
                names create new stations.
            distances: list of distances in meters, one per consecutive pair.

        Raises:
            ValueError: on invalid input (nothing is modified in that case).
        """
        if not isinstance(line_name, str) or not line_name.strip():
            raise ValueError("line name must be a non-empty string")
        line_name = line_name.strip()
        if line_name in self.all_line_names():
            raise ValueError(f"line '{line_name}' already exists")
        names = [s.strip() if isinstance(s, str) else "" for s in station_names]
        if len(names) < 2 or any(not n for n in names):
            raise ValueError("a line needs at least two named stations")
        if len(set(names)) != len(names):
            raise ValueError("station sequence must not repeat a station")
        if len(distances) != len(names) - 1:
            raise ValueError("one distance is required per consecutive station pair")
        for d in distances:
            _validate_distance(d)

        for name in names:
            self._declared_stations.add(name)
            declared = list(self._declared_lines.get(name, []))
            if line_name not in declared:
                declared.append(line_name)
            self._declared_lines[name] = declared
            self._ensure_station(name, [line_name])
        for i, d in enumerate(distances):
            self._connect(names[i], names[i + 1], d, line_name)

    def update_station_lines(self, name, lines):
        """Replace the list of lines serving an existing station."""
        std = self._require_station(name)
        clean = list(dict.fromkeys(str(l).strip() for l in lines if str(l).strip()))
        self.graph.set_station_lines(std, clean)
        self._declared_stations.add(std)
        self._declared_lines[std] = list(clean)
        return clean

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance between two adjacent stations.

        Args:
            station_a, station_b: station names (standardized internally).
            distance: new distance in meters (positive number).
            line: optional line name restricting which connection to update.

        Returns:
            The number of connections updated.

        Raises:
            ValueError: if the distance is invalid or no connection exists.
        """
        _validate_distance(distance)
        na = self._require_station(station_a)
        nb = self._require_station(station_b)
        ia, ib = self.station_to_number[na], self.station_to_number[nb]
        count = 0
        for edge in self.graph.adjacent(ia):
            if edge.other(ia) == ib and (line is None or edge.line == line):
                edge.distance = distance
                count += 1
        for rec in self.connections:
            if {rec["source"], rec["dest"]} == {na, nb} and (line is None or rec["line"] == line):
                rec["distance"] = distance
        if count == 0:
            raise ValueError(f"no connection between '{na}' and '{nb}'")
        return count

    # ------------------------------------------------------------------
    # Integrity checks
    # ------------------------------------------------------------------
    def check_integrity(self):
        """Run data integrity checks.

        Returns:
            A list of message strings prefixed with "ERROR:" or "WARNING:".
            An empty list means no problems were detected.
        """
        problems = list(self._load_problems)
        lines = self.all_line_names()

        # 1. Coverage of representative stations on main lines.
        for line, reps in REPRESENTATIVE_STATIONS.items():
            if line not in lines:
                problems.append(f"ERROR: main line '{line}' has no data")
            for rep in reps:
                if rep not in self.station_to_number:
                    problems.append(
                        f"ERROR: representative station '{rep}' of {line} is missing")

        # 2. Connection relationship integrity.
        for name in sorted(self._undeclared_stations):
            problems.append(
                f"WARNING: station '{name}' is used by connections but not declared in stations")
        for rec in self.connections:
            for end in (rec["source"], rec["dest"]):
                if end in self._declared_stations:
                    declared = self._declared_lines.get(end, [])
                    if rec["line"] not in declared:
                        problems.append(
                            f"WARNING: connection {rec['source']}-{rec['dest']} is on "
                            f"'{rec['line']}' but station '{end}' does not list that line")
        seen = set()
        for rec in self.connections:
            key = (tuple(sorted((rec["source"], rec["dest"]))), rec["line"])
            if key in seen:
                problems.append(
                    f"WARNING: duplicate connection {rec['source']}-{rec['dest']} on {rec['line']}")
            seen.add(key)
        for name, number in self.station_to_number.items():
            if not self.graph.adjacent(number):
                problems.append(f"WARNING: station '{name}' has no connections")
        n = len(self.number_to_name)
        visited = set()
        components = 0
        for start in range(n):
            if start in visited:
                continue
            components += 1
            visited.add(start)
            queue = deque([start])
            while queue:
                v = queue.popleft()
                for edge in self.graph.adjacent(v):
                    w = edge.other(v)
                    if w not in visited:
                        visited.add(w)
                        queue.append(w)
        if components > 1:
            problems.append(f"WARNING: network has {components} disconnected components")

        # 3. Distance accuracy.
        for rec in self.connections:
            d = rec["distance"]
            label = f"{rec['source']}-{rec['dest']} ({rec['line']})"
            if not isinstance(d, int):
                problems.append(f"ERROR: distance of {label} is not an integer number of meters")
            if d < MIN_PLAUSIBLE_DISTANCE_M or d > MAX_PLAUSIBLE_DISTANCE_M:
                problems.append(
                    f"WARNING: distance of {label} = {d} m is outside the plausible range")
        return problems

"""Shortest-path computation using Dijkstra's algorithm with a binary heap.

The search runs over states (station number, arriving line). This allows
exact transfer counting, the least-transfer mode and the max-transfer limit.
Each state is settled at most once, and states are kept in a MinPQ, so the
running time is O((V + E) log V) for the expanded state graph.
"""

from min_pq import MinPQ

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance (meters) charged for each transfer in comprehensive mode.
TRANSFER_PENALTY_M = 1000
# Equivalent distance (meters) charged for each segment not on a preferred line.
PREFERENCE_PENALTY_M = 500


class PathfindingConfig:
    """Query configuration for path computation.

    Attributes:
        mode: one of "shortest_distance", "least_transfer", "comprehensive".
        max_transfers: maximum allowed number of transfers, or None for no limit.
        preferred_lines: list of line names to prefer (empty means no preference).
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []
        self.validate()

    def validate(self):
        """Raise ValueError/TypeError if the configuration is invalid."""
        if self.mode not in MODES:
            raise ValueError(f"unknown mode '{self.mode}'")
        if self.max_transfers is not None:
            if (isinstance(self.max_transfers, bool)
                    or not isinstance(self.max_transfers, int)
                    or self.max_transfers < 0):
                raise ValueError("max_transfers must be a non-negative integer or None")
        for line in self.preferred_lines:
            if not isinstance(line, str):
                raise TypeError("preferred lines must be strings")


class ShortestPath:
    """Single-source shortest path computation.

    Args:
        graph: the Graph to search.
        source: station number of the starting station.
        config: optional PathfindingConfig (defaults to shortest distance).

    Attributes:
        dist_to: dict station number -> actual travelled distance in meters
            of the best path found to that station (only reachable stations).
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"source station {source} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()
        self.dist_to = {}
        self._keys = {}       # state -> cost key used for ordering
        self._info = {}       # state -> (distance_m, transfers, preference_penalty)
        self._edge_to = {}    # state -> (previous state, edge used)
        self._best_state = {}  # station number -> best state reaching it
        self._run()

    # ------------------------------------------------------------------
    # Cost model
    # ------------------------------------------------------------------
    def _cost_key(self, dist, transfers, pref):
        """Return the comparable cost key of a state for the current mode."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, dist + pref)
        if mode == MODE_COMPREHENSIVE:
            return (dist + pref + TRANSFER_PENALTY_M * transfers,)
        return (dist + pref,)

    # ------------------------------------------------------------------
    # Dijkstra main loop
    # ------------------------------------------------------------------
    def _run(self):
        prefs = set(self.config.preferred_lines)
        max_transfers = self.config.max_transfers
        start = (self.source, None)
        self._info[start] = (0, 0, 0)
        self._keys[start] = self._cost_key(0, 0, 0)

        pq = MinPQ()
        pq.insert(self._keys[start], start)
        settled = set()

        while not pq.is_empty():
            key, state = pq.del_min()
            if state in settled or key != self._keys[state]:
                continue  # stale heap entry
            settled.add(state)
            node, line = state
            dist, transfers, pref = self._info[state]

            for edge in self.graph.adjacent(node):
                nxt = edge.other(node)
                new_transfers = transfers + (1 if line is not None and line != edge.line else 0)
                if max_transfers is not None and new_transfers > max_transfers:
                    continue
                new_dist = dist + edge.distance
                new_pref = pref + (PREFERENCE_PENALTY_M
                                   if prefs and edge.line not in prefs else 0)
                next_state = (nxt, edge.line)
                if next_state in settled:
                    continue
                new_key = self._cost_key(new_dist, new_transfers, new_pref)
                old_key = self._keys.get(next_state)
                if old_key is None or new_key < old_key:
                    self._keys[next_state] = new_key
                    self._info[next_state] = (new_dist, new_transfers, new_pref)
                    self._edge_to[next_state] = (state, edge)
                    pq.insert(new_key, next_state)

        for state in settled:
            node = state[0]
            if node not in self._best_state or self._keys[state] < self._keys[self._best_state[node]]:
                self._best_state[node] = state
        for node, state in self._best_state.items():
            self.dist_to[node] = self._info[state][0]

    # ------------------------------------------------------------------
    # Public queries
    # ------------------------------------------------------------------
    def has_path_to(self, target):
        """Return True if ``target`` is reachable under the current constraints."""
        return target in self._best_state

    def distance_to(self, target):
        """Return the travelled distance in meters to ``target`` (KeyError if unreachable)."""
        return self.dist_to[target]

    def path_with_edges(self, target):
        """Return the station sequence and edge sequence from source to target.

        Returns:
            (list of station numbers, list of Edge). The edge list has one
            element fewer than the station list. Unreachable targets return
            ([], []).
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        chain_states = [state]
        chain_edges = []
        while state in self._edge_to:
            prev_state, edge = self._edge_to[state]
            chain_edges.append(edge)
            state = prev_state
            chain_states.append(state)
        chain_states.reverse()
        chain_edges.reverse()
        nodes = [s[0] for s in chain_states]
        return nodes, chain_edges

    def get_transfer_count(self, target):
        """Return the number of line changes on the best path to ``target``.

        Returns 0 if the target is unreachable.
        """
        if target not in self._best_state:
            return 0
        return self._info[self._best_state[target]][1]

"""Fare and travel-time calculation for the Beijing Subway.

Fare structure (distance in kilometers, rounded up):
    0-6 km    : 3 yuan
    6-12 km   : 4 yuan
    12-22 km  : 5 yuan
    22-32 km  : 6 yuan
    > 32 km   : 6 yuan plus 1 yuan for every further (or partial) 20 km
"""

import math

# (upper bound in km inclusive, fare in yuan)
BASE_FARE_TABLE = ((6, 3), (12, 4), (22, 5), (32, 6))
EXTENSION_STEP_KM = 20
EXTENSION_STEP_FARE = 1

AVERAGE_SPEED_KMH = 35.0    # average operating speed of the subway
BOARDING_TIME_MIN = 3.0     # fixed time for entering and boarding
TRANSFER_TIME_MIN = 5.0     # walking and waiting time per transfer


def _check_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of meters")
    if not math.isfinite(distance):
        raise ValueError("distance must be finite")
    if distance < 0:
        raise ValueError("distance must not be negative")


def Sum_money(distance):
    """Return the fare in yuan for a journey of ``distance`` meters.

    Fractional kilometers are rounded up (ceiling), so any part of a
    kilometer is charged as a full kilometer for tier selection.
    """
    _check_distance(distance)
    km = math.ceil(distance / 1000)  # integer kilometers, rounded up
    for upper_km, fare in BASE_FARE_TABLE:
        if km <= upper_km:
            return fare
    last_upper, last_fare = BASE_FARE_TABLE[-1]
    extra_km = km - last_upper
    extra_steps = (extra_km + EXTENSION_STEP_KM - 1) // EXTENSION_STEP_KM
    return last_fare + extra_steps * EXTENSION_STEP_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimate the travel time in minutes.

    Args:
        distance: total distance in meters.
        transfer_count: number of line transfers.

    Returns:
        Estimated time in minutes (float, one decimal place).
    """
    _check_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise ValueError("transfer_count must be a non-negative integer")
    riding_min = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    total = BOARDING_TIME_MIN + riding_min + transfer_count * TRANSFER_TIME_MIN
    return round(total, 1)

"""Path visualization, summary generation and report export.

Produces:
    * the complete station sequence,
    * line segments and transfer details,
    * distance summaries (meters for short, kilometers for long distances),
    * console text and JSON representations.
"""

import json

from fare_calculator import Sum_money, calculate_estimated_time

MODE_DESCRIPTIONS = {
    "shortest_distance": "Shortest distance",
    "least_transfer": "Minimum transfers",
    "comprehensive": "Comprehensive optimal",
}


def _normalize_number(value):
    """Return an int for integral floats so that output stays tidy."""
    if isinstance(value, float) and value.is_integer():
        return int(value)
    return value


def format_distance(distance_m):
    """Format a distance: meters for short trips, kilometers for long trips."""
    if distance_m < 1000:
        return f"{distance_m} m"
    return f"{distance_m / 1000:.2f} km ({distance_m} m)"


def meters_to_km(distance_m):
    """Convert meters to kilometers rounded to two decimals."""
    return round(distance_m / 1000.0, 2)


def involved_lines(edges):
    """Return the line names used by the path, in order of first appearance."""
    result = []
    for edge in edges:
        if edge.line not in result:
            result.append(edge.line)
    return result


def build_segments(names, edges):
    """Group consecutive edges on the same line into segments.

    Each segment is a dict with: line, from, to, stations, stop_count, distance_m.
    """
    if not edges:
        return []

    def make_segment(first, last):
        # Segment covers edges[first:last], i.e. stations names[first..last].
        stations = names[first:last + 1]
        distance = _normalize_number(sum(edges[k].distance for k in range(first, last)))
        return {
            "line": edges[first].line,
            "from": stations[0],
            "to": stations[-1],
            "stations": stations,
            "stop_count": len(stations) - 1,
            "distance_m": distance,
        }

    segments = []
    seg_start = 0
    current_line = edges[0].line
    for i, edge in enumerate(edges):
        if edge.line != current_line:
            segments.append(make_segment(seg_start, i))
            seg_start = i
            current_line = edge.line
    segments.append(make_segment(seg_start, len(edges)))
    return segments


def build_transfers(names, edges):
    """Return the list of transfer points (station and the two lines involved)."""
    transfers = []
    for i in range(1, len(edges)):
        if edges[i - 1].line != edges[i].line:
            transfers.append({
                "station": names[i],
                "from_line": edges[i - 1].line,
                "to_line": edges[i].line,
            })
    return transfers


def count_transfers(edges):
    """Count line changes along the path."""
    return sum(1 for i in range(1, len(edges)) if edges[i - 1].line != edges[i].line)


def generate_summary(names, edges, transfer_count=None):
    """Extract the key path summary: distance, fare, transfers and time."""
    distance_m = _normalize_number(sum(e.distance for e in edges))
    if transfer_count is None:
        transfer_count = count_transfers(edges)
    fare = Sum_money(distance_m)
    estimated_time = calculate_estimated_time(distance_m, transfer_count)
    return {
        "distance_m": distance_m,
        "distance_km": meters_to_km(distance_m),
        "fare": fare,
        "transfer_count": transfer_count,
        "estimated_time_min": estimated_time,
    }


def build_report(start, end, config, names, edges, transfer_count=None):
    """Assemble the complete report dictionary for one query.

    Args:
        start, end: official station names of the query.
        config: the PathfindingConfig used for the query.
        names: station names along the path, from start to end.
        edges: Edge objects along the path.
        transfer_count: optional transfer count from the path search.
    """
    summary = generate_summary(names, edges, transfer_count)
    return {
        "query": {
            "start": start,
            "end": end,
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "line_preference": list(config.preferred_lines),
        },
        "path": list(names),
        "segments": build_segments(names, edges),
        "transfers": build_transfers(names, edges),
        "lines": involved_lines(edges),
        "distance_m": summary["distance_m"],
        "distance_km": summary["distance_km"],
        "fare": summary["fare"],
        "transfer_count": summary["transfer_count"],
        "estimated_time_min": summary["estimated_time_min"],
    }


def format_text(report):
    """Render a report as human-readable console text."""
    query = report["query"]
    rule = "=" * 60
    out = [rule, "Beijing Subway Fare Query Result", rule]
    out.append(f"Start station : {query['start']}")
    out.append(f"End station   : {query['end']}")
    out.append(f"Query mode    : {MODE_DESCRIPTIONS.get(query['mode'], query['mode'])}")
    if query["max_transfers"] is not None:
        out.append(f"Max transfers : {query['max_transfers']}")
    if query["line_preference"]:
        out.append("Line preference: " + ", ".join(query["line_preference"]))
    out.append("")

    out.append(f"Station sequence ({len(report['path'])} stations):")
    for index, name in enumerate(report["path"], start=1):
        out.append(f"  {index:>3}. {name}")
    out.append("")

    out.append("Line segments:")
    for seg in report["segments"]:
        out.append(f"  [{seg['line']}] {seg['from']} -> {seg['to']} "
                   f"({seg['stop_count']} stops, {format_distance(seg['distance_m'])})")
    out.append("")

    out.append("Transfers:")
    if report["transfers"]:
        for t in report["transfers"]:
            out.append(f"  {t['station']}: {t['from_line']} -> {t['to_line']}")
    else:
        out.append("  None (direct route)")
    out.append("")

    out.append(f"Total distance : {format_distance(report['distance_m'])}")
    out.append(f"Transfers      : {report['transfer_count']}")
    out.append(f"Estimated time : {report['estimated_time_min']} min")
    out.append(f"Lines involved : {', '.join(report['lines'])}")
    out.append(f"Fare: {report['fare']} yuan")
    out.append(rule)
    return "\n".join(out)


def format_json(report):
    """Render a report as a JSON document for programmatic use."""
    return json.dumps(report, ensure_ascii=False, indent=2)

"""Command-line entry point for the Beijing Subway fare calculation system.

Usage examples:
    python src/main.py
    python src/main.py --mode least_transfer --format json
    python src/main.py --max_transfers 2 --line_preference "Line 1,Line 2"
    python src/main.py --output_file results.txt
    python src/main.py --check_data
"""

import argparse
import json
import os
import sys

_SRC_DIR = os.path.dirname(os.path.abspath(__file__))
if _SRC_DIR not in sys.path:
    sys.path.insert(0, _SRC_DIR)

from data_loader import DATA_FILE_PATH, load_subway_data  # noqa: E402
from subway_network import SubwayNetwork  # noqa: E402
from pathfinding import MODES, PathfindingConfig, ShortestPath  # noqa: E402
from path_report import build_report, format_json, format_text  # noqa: E402

EXIT_COMMANDS = {"q", "quit", "exit"}
QUERY_FORMAT_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"
SEPARATOR = "-" * 60


class QueryError(Exception):
    """A user-facing error raised while processing one query."""


def parse_query(text):
    """Split 'Starting Station; Ending Station' into two raw station names."""
    normalized = text.replace("\uff1b", ";")  # accept full-width semicolon too
    parts = normalized.split(";")
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise QueryError(QUERY_FORMAT_MESSAGE)
    return parts[0].strip(), parts[1].strip()


def resolve_station(network, raw, role):
    """Standardize a station name or raise a user-facing error."""
    name = network.standardize_station_name(raw)
    if name is None:
        raise QueryError(f"{role} station '{raw}' does not exist, please check the station name")
    return name


def run_query(network, start_raw, end_raw, config):
    """Compute the route and report for one query.

    Returns:
        The report dictionary produced by path_report.build_report.

    Raises:
        QueryError: for any user-facing failure.
    """
    start = resolve_station(network, start_raw, "Starting")
    end = resolve_station(network, end_raw, "Ending")
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")

    try:
        search = ShortestPath(network.graph, network.get_number(start), config)
        target = network.get_number(end)
        if not search.has_path_to(target):
            raise QueryError(
                f"No route found from '{start}' to '{end}' under the current constraints")
        node_numbers, edges = search.path_with_edges(target)
        transfer_count = search.get_transfer_count(target)
    except QueryError:
        raise
    except Exception as exc:  # defensive: any unexpected path error
        raise QueryError(f"Path calculation failed: {exc}") from exc

    names = [network.get_name(n) for n in node_numbers]
    try:
        return build_report(start, end, config, names, edges, transfer_count)
    except Exception as exc:
        raise QueryError(f"Fare calculation failed: {exc}") from exc


def render(report, fmt):
    """Render a report in the requested output format."""
    return format_json(report) if fmt == "json" else format_text(report)


def save_output(path, fmt, session):
    """Persist the session results to ``path``.

    Text results are appended; JSON results are rewritten as a JSON array
    containing every result of the current session.

    Returns:
        (True, None) on success, or (False, error message) on failure.
    """
    try:
        if fmt == "json":
            with open(path, "w", encoding="utf-8") as fh:
                json.dump(session, fh, ensure_ascii=False, indent=2)
                fh.write("\n")
        else:
            with open(path, "a", encoding="utf-8") as fh:
                fh.write(format_text(session[-1]))
                fh.write("\n" + SEPARATOR + "\n")
        return True, None
    except OSError as exc:
        return False, str(exc)


def parse_preferences(value):
    """Turn a comma separated --line_preference value into a list of lines."""
    if not value:
        return []
    return [part.strip() for part in value.split(",") if part.strip()]


def run_data_check(network):
    """Print the data integrity report and return an exit code."""
    problems = network.check_integrity()
    print(f"Stations: {network.station_count()}  "
          f"Connections: {len(network.connections)}  "
          f"Lines: {len(network.all_line_names())}")
    if not problems:
        print("Data integrity check passed: no problems found.")
        return 0
    for problem in problems:
        print(problem)
    errors = [p for p in problems if p.startswith("ERROR")]
    print(f"Data integrity check finished: {len(errors)} error(s), "
          f"{len(problems) - len(errors)} warning(s).")
    return 1 if errors else 0


def run_interactive(network, args, preferred):
    """Continuous query loop: keeps asking until the user quits."""
    session = []
    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' (or 'q' to quit).")
    while True:
        try:
            raw_line = input("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw_line.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        try:
            start_raw, end_raw = parse_query(text)
            config = PathfindingConfig(mode=args.mode,
                                       max_transfers=args.max_transfers,
                                       preferred_lines=preferred)
            report = run_query(network, start_raw, end_raw, config)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # keep the session alive on unexpected errors
            print(f"Error: unexpected failure while processing the query: {exc}")
            continue

        print(render(report, args.format))
        if args.output_file:
            session.append(report)
            ok, message = save_output(args.output_file, args.format, session)
            if ok:
                print(f"Result saved to {args.output_file}")
            else:
                print(f"Error: could not write output file: {message}")
    print("Goodbye.")
    return 0


def build_arg_parser():
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (CLI).")
    parser.add_argument("--output_file", default=None,
                        help="export query results to this file")
    parser.add_argument("--format", choices=["text", "json"], default="text",
                        help="output format (default: text)")
    parser.add_argument("--mode", choices=list(MODES), default="shortest_distance",
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="maximum number of allowed transfers")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line(s), e.g. \"Line 1\" or \"Line 1,Line 2\"")
    parser.add_argument("--check_data", action="store_true",
                        help="run data integrity checks and exit")
    return parser


def main(argv=None):
    """Program entry point. Returns the process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")

    try:
        network = SubwayNetwork(load_subway_data(DATA_FILE_PATH))
    except Exception as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        return run_data_check(network)

    preferred = parse_preferences(args.line_preference)
    known_lines = set(network.all_line_names())
    for line in preferred:
        if line not in known_lines:
            print(f"Warning: line preference '{line}' is not a known line", file=sys.stderr)

    return run_interactive(network, args, preferred)


if __name__ == "__main__":
    sys.exit(main())

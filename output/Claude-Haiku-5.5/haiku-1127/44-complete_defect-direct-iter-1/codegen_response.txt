"""Data loading utilities for the Beijing Subway dataset.

The dataset lives in ``subway_data.json`` next to this module. The path is
resolved relative to this file so the program works from any working directory.
"""

import json
import os

# Absolute path of src/subway_data.json, valid regardless of the current working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH) -> dict:
    """Load the subway dataset from a JSON file.

    Args:
        path: Path of the JSON file. Defaults to DATA_FILE_PATH.

    Returns:
        The parsed JSON content as a dict containing the "stations" and
        "connections" lists.

    Raises:
        FileNotFoundError: If the file does not exist.
        ValueError: If the file is not valid JSON or has an unexpected structure.
    """
    with open(path, "r", encoding="utf-8") as handle:
        try:
            data = json.load(handle)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid JSON in subway data file '{path}': {exc}") from exc

    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")
    for key in ("stations", "connections"):
        if not isinstance(data.get(key), list):
            raise ValueError(f"Subway data must contain a list under '{key}'")
    return data


def save_subway_data(data, path=DATA_FILE_PATH) -> None:
    """Persist the subway dataset to a JSON file.

    Args:
        data: Dataset dict with "stations" and "connections" lists.
        path: Destination file path. Defaults to DATA_FILE_PATH.
    """
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")

"""Core graph data structures for the subway network.

Stations are identified by integer numbers that are assigned by the caller
(see network.SubwayNetwork). The graph is an undirected weighted graph whose
edge weight is the physical distance in meters.
"""


class Edge:
    """An undirected weighted edge between two stations.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance in meters.
        weight: Edge weight used by path algorithms (equal to distance).
        line: Name of the subway line this connection belongs to.
    """

    def __init__(self, source, dest, distance, line):
        if distance < 0:
            raise ValueError(f"Edge distance must be non-negative, got {distance}")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def other(self, vertex):
        """Return the endpoint of this edge that is not ``vertex``.

        Raises:
            ValueError: If ``vertex`` is not an endpoint of this edge.
        """
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex} is not an endpoint of edge {self!r}")

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Attributes:
        adj: dict mapping station number -> list of incident Edge objects.
        station_lines: dict mapping station name -> list of line names.
    """

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_vertex(self, vertex):
        """Register a station number as a vertex (no-op if already present)."""
        self.adj.setdefault(vertex, [])

    def add_edge(self, edge):
        """Add an undirected edge, registering both endpoints as vertices."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self.adj[edge.dest].append(edge)

    def remove_edge(self, edge):
        """Remove an edge from both endpoint adjacency lists."""
        for vertex in {edge.source, edge.dest}:
            self.adj[vertex] = [e for e in self.adj.get(vertex, []) if e is not edge]

    def adjacent(self, vertex):
        """Return the list of edges incident to ``vertex``."""
        return self.adj.get(vertex, [])

    def vertices(self):
        """Return the list of all station numbers in the graph."""
        return list(self.adj.keys())

    def edges(self):
        """Return every distinct edge exactly once."""
        seen = set()
        result = []
        for edge_list in self.adj.values():
            for edge in edge_list:
                if id(edge) not in seen:
                    seen.add(id(edge))
                    result.append(edge)
        return result

    def vertex_count(self):
        """Return the number of vertices."""
        return len(self.adj)

    def edge_count(self):
        """Return the number of distinct edges."""
        return len(self.edges())

    def set_station_lines(self, station_name, lines):
        """Record the list of lines serving a station (by name)."""
        self.station_lines[station_name] = list(lines)

    def get_station_lines(self, station_name):
        """Return a copy of the list of lines serving a station (by name)."""
        return list(self.station_lines.get(station_name, []))

"""Shortest path computation using Dijkstra's algorithm with a priority queue.

The search runs over states (station number, line used to arrive). This lets
the algorithm count line changes (transfers) exactly and handle multi-line
interchange stations correctly.

Supported modes:
    shortest_distance: minimize total distance in meters.
    least_transfer:    minimize transfers first, then distance.
    comprehensive:     minimize distance plus a penalty per transfer.

Line preferences add a penalty to distance for edges on non-preferred lines.
"""

import heapq
import itertools

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")
DEFAULT_MODE = "shortest_distance"
TRANSFER_PENALTY_M = 500.0      # equivalent distance penalty per transfer (comprehensive mode)
PREFERENCE_PENALTY_M = 300.0    # penalty per edge on a non-preferred line


class PathfindingConfig:
    """Configuration for a path query.

    Attributes:
        mode: One of VALID_MODES.
        max_transfers: Maximum allowed transfers, or None for no limit.
        preferred_lines: List of preferred line names (may be empty).
    """

    def __init__(self, mode=DEFAULT_MODE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Unknown mode '{mode}', expected one of {VALID_MODES}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer or None")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class MinPQ:
    """Min priority queue backed by a binary heap.

    Items with equal keys are served in insertion order thanks to a counter
    tie-breaker, so the items themselves never need to be comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        """Insert an item with the given priority key."""
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self):
        """Remove and return the (key, item) pair with the smallest key."""
        if not self._heap:
            raise IndexError("del_min from an empty priority queue")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._heap:
            raise IndexError("min_key from an empty priority queue")
        return self._heap[0][0]

    def is_empty(self):
        """Return True if the queue holds no items."""
        return not self._heap

    def size(self):
        """Return the number of items in the queue."""
        return len(self._heap)


class ShortestPath:
    """Dijkstra-based shortest path search from a single source station.

    Attributes:
        dist_to: dict mapping reachable station number -> actual distance in meters.
        transfers_to: dict mapping reachable station number -> transfer count.
    """

    def __init__(self, graph, source, config=None):
        if source not in graph.adj:
            raise ValueError(f"Unknown source station number: {source}")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()

        self.dist_to = {}
        self.transfers_to = {}
        self._state_for = {}   # station -> best settled state (station, line)
        self._info = {}        # state -> (distance, transfers, accumulated penalty)
        self._best_key = {}    # state -> best priority key found so far
        self._edge_to = {}     # state -> (previous state, Edge)

        self._run()

    def _make_key(self, effective_distance, transfers):
        """Build the priority key for a state according to the query mode."""
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, effective_distance)
        if mode == "comprehensive":
            return (effective_distance + TRANSFER_PENALTY_M * transfers,)
        return (effective_distance,)

    def _run(self):
        """Execute Dijkstra's algorithm over (station, line) states."""
        preferred = set(self.config.preferred_lines)
        max_transfers = self.config.max_transfers

        start = (self.source, None)
        self._info[start] = (0.0, 0, 0.0)
        self._best_key[start] = self._make_key(0.0, 0)

        pq = MinPQ()
        pq.insert(self._best_key[start], start)
        settled = set()

        while not pq.is_empty():
            key, state = pq.del_min()
            if state in settled or key != self._best_key[state]:
                continue  # stale queue entry
            settled.add(state)

            station, line = state
            dist, transfers, penalty = self._info[state]

            # The first settled state of a station is its optimal state.
            if station not in self._state_for:
                self._state_for[station] = state
                self.dist_to[station] = dist
                self.transfers_to[station] = transfers

            for edge in self.graph.adjacent(station):
                nxt = edge.other(station)

                # A transfer happens when the line changes (not at the origin).
                new_transfers = transfers
                if line is not None and edge.line != line:
                    new_transfers += 1
                if max_transfers is not None and new_transfers > max_transfers:
                    continue

                new_dist = dist + edge.weight
                new_penalty = penalty
                if preferred and edge.line not in preferred:
                    new_penalty += PREFERENCE_PENALTY_M

                nstate = (nxt, edge.line)
                if nstate in settled:
                    continue

                nkey = self._make_key(new_dist + new_penalty, new_transfers)
                best = self._best_key.get(nstate)
                if best is None or nkey < best:
                    self._best_key[nstate] = nkey
                    self._info[nstate] = (new_dist, new_transfers, new_penalty)
                    self._edge_to[nstate] = (state, edge)
                    pq.insert(nkey, nstate)

    def has_path_to(self, target):
        """Return True if ``target`` is reachable from the source (within limits)."""
        return target in self._state_for

    def distance_to(self, target):
        """Return the distance in meters to ``target``, or infinity if unreachable."""
        return self.dist_to.get(target, float("inf"))

    def get_transfer_count(self, target):
        """Return the number of transfers on the path to ``target``.

        Returns -1 if the target is unreachable.
        """
        return self.transfers_to.get(target, -1)

    def path_with_edges(self, target):
        """Reconstruct the path from the source to ``target``.

        Returns:
            (stations, edges): ``stations`` is the ordered list of station numbers
            from source to target; ``edges`` is the ordered list of Edge objects
            between consecutive stations. Both are empty if unreachable.
        """
        if target not in self._state_for:
            return [], []

        stations = []
        edges = []
        state = self._state_for[target]
        while True:
            stations.append(state[0])
            if state not in self._edge_to:
                break
            prev_state, edge = self._edge_to[state]
            edges.append(edge)
            state = prev_state

        stations.reverse()
        edges.reverse()
        return stations, edges

"""Fare and travel time calculation for the Beijing Subway.

Fare structure (distance in kilometers, rounded up):
    0-6 km:   3 yuan
    6-12 km:  4 yuan
    12-22 km: 5 yuan
    22-32 km: 6 yuan
    beyond 32 km: 6 yuan + 1 yuan for every additional (started) 20 km
"""

import math

# (inclusive upper bound in km, fare in yuan), in ascending order.
FARE_TIERS = ((6, 3), (12, 4), (22, 5), (32, 6))
EXTENSION_START_KM = 32
EXTENSION_STEP_KM = 20
EXTENSION_STEP_FARE = 1

AVERAGE_SPEED_KMH = 35.0    # average operating speed including stops
TRANSFER_TIME_MIN = 4.0     # walking/waiting time per transfer
BOARDING_TIME_MIN = 2.0     # fixed boarding overhead per trip


def Sum_money(distance) -> int:
    """Return the subway fare in yuan for a trip of ``distance`` meters.

    Fractional kilometers are rounded up (ceiling) before mapping to a tier.

    Raises:
        TypeError: If distance is not a number.
        ValueError: If distance is negative, NaN or infinite.
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError(f"distance must be a number, got {type(distance).__name__}")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("distance must be a finite number")
    if distance < 0:
        raise ValueError("distance must be non-negative")

    km = math.ceil(distance / 1000)

    for upper_km, fare in FARE_TIERS:
        if km <= upper_km:
            return fare

    extra_km = km - EXTENSION_START_KM
    steps = math.ceil(extra_km / EXTENSION_STEP_KM)
    base_fare = FARE_TIERS[-1][1]
    return base_fare + steps * EXTENSION_STEP_FARE


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimate the travel time in minutes.

    Args:
        distance: Total distance in meters.
        transfer_count: Number of transfers on the route.

    Returns:
        Estimated travel time in minutes, rounded to one decimal place.

    Raises:
        ValueError: If distance or transfer_count is negative.
    """
    if distance < 0:
        raise ValueError("distance must be non-negative")
    if transfer_count < 0:
        raise ValueError("transfer_count must be non-negative")

    travel_minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    total = BOARDING_TIME_MIN + travel_minutes + transfer_count * TRANSFER_TIME_MIN
    return round(total, 1)

"""Subway network modeling and data management.

This module builds the weighted graph from the dataset, assigns integer
station numbers, resolves user-supplied station names, supports data updates
(adding lines, modifying stations, updating distances), persists changes, and
runs data integrity checks.
"""

import difflib
import re
import unicodedata

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
from models import Edge, Graph

# Main lines that must be present in the dataset.
MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

# Representative stations used for coverage checks (warnings only).
REPRESENTATIVE_STATIONS = {
    "Line 1": ["苹果园", "西单", "国贸"],
    "Line 2": ["西直门", "东直门"],
    "Line 4": ["圆明园"],
    "Line 5": ["宋家庄"],
    "Line 6": ["南锣鼓巷"],
    "Line 8": ["鼓楼大街"],
    "Line 10": ["国贸"],
    "Line 13": ["大钟寺"],
}

MAX_REASONABLE_DISTANCE_M = 5000
MIN_REASONABLE_DISTANCE_M = 100
_NAME_SUFFIXES = ("station", "站")


class StationNotFoundError(ValueError):
    """Raised when a station name does not match any official station."""

    def __init__(self, name, suggestions=None):
        super().__init__(f"Station '{name}' does not exist")
        self.name = name
        self.suggestions = list(suggestions or [])


def _clean(value):
    """Return a stripped string, or an empty string for non-string values."""
    if isinstance(value, str):
        return value.strip()
    return ""


def normalize_station_name(name):
    """Return a canonical lookup key for a station name.

    The key ignores case, whitespace, hyphens/underscores, full-width variants
    and a trailing "station" or "站" suffix.
    """
    text = unicodedata.normalize("NFKC", str(name)).strip().lower()
    text = re.sub(r"[\s_\-]+", "", text)
    for suffix in _NAME_SUFFIXES:
        if text.endswith(suffix) and len(text) > len(suffix):
            text = text[: -len(suffix)]
            break
    return text


def _validate_distance(value):
    """Convert a distance value to a positive integer number of meters."""
    if isinstance(value, bool):
        raise ValueError(f"Invalid distance: {value!r}")
    try:
        number = int(value)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"Invalid distance: {value!r}") from exc
    if number <= 0 or (isinstance(value, float) and value != number):
        raise ValueError(f"Distance must be a positive integer number of meters, got {value!r}")
    return number


class SubwayNetwork:
    """Subway network model built from the dataset.

    Attributes:
        data: The raw dataset dict (source of truth for updates).
        data_path: File path used when saving.
        station_to_number: dict mapping station name -> station number.
        number_to_station: dict mapping station number -> station name.
        graph: The Graph instance built from the dataset.
    """

    def __init__(self, data=None, data_path=DATA_FILE_PATH):
        if data is None:
            data = load_subway_data(data_path)
        self.data = data
        self.data_path = data_path
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self._name_index = {}
        self.rebuild()

    # ------------------------------------------------------------------
    # Construction
    # ------------------------------------------------------------------
    def rebuild(self):
        """(Re)build the graph and lookup tables from ``self.data``."""
        station_lines = {}

        # Stations declared in the station list (order preserved).
        for entry in self.data.get("stations", []):
            if not isinstance(entry, dict):
                continue
            name = _clean(entry.get("name"))
            if not name:
                continue
            lines = station_lines.setdefault(name, [])
            for line in entry.get("lines") or []:
                line = _clean(line)
                if line and line not in lines:
                    lines.append(line)

        # Stations that appear only in connections are added with their connection line.
        connections = [c for c in self.data.get("connections", []) if isinstance(c, dict)]
        for conn in connections:
            line = _clean(conn.get("line"))
            for key in ("source", "dest"):
                name = _clean(conn.get(key))
                if name and name not in station_lines:
                    station_lines[name] = [line] if line else []

        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for number, (name, lines) in enumerate(station_lines.items()):
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
            self.graph.set_station_lines(name, lines)

        for conn in connections:
            src = self.station_to_number.get(_clean(conn.get("source")))
            dst = self.station_to_number.get(_clean(conn.get("dest")))
            if src is None or dst is None or src == dst:
                continue
            try:
                distance = int(conn.get("distance"))
            except (TypeError, ValueError):
                continue
            if distance < 0:
                continue
            self.graph.add_edge(Edge(src, dst, distance, _clean(conn.get("line"))))

        self._name_index = {}
        for name in self.station_to_number:
            self._name_index.setdefault(normalize_station_name(name), name)

    # ------------------------------------------------------------------
    # Lookup
    # ------------------------------------------------------------------
    def station_names(self):
        """Return all official station names."""
        return list(self.station_to_number.keys())

    def resolve_station(self, name):
        """Map a user-supplied name to the official station name.

        Raises:
            StationNotFoundError: If no station matches; includes suggestions.
        """
        if not isinstance(name, str):
            raise StationNotFoundError(str(name))
        stripped = name.strip()
        if stripped in self.station_to_number:
            return stripped
        key = normalize_station_name(stripped)
        if key in self._name_index:
            return self._name_index[key]
        raise StationNotFoundError(stripped, self.suggest(stripped))

    def suggest(self, name, limit=3):
        """Return up to ``limit`` official station names similar to ``name``."""
        key = normalize_station_name(name)
        matches = difflib.get_close_matches(key, list(self._name_index.keys()), n=limit, cutoff=0.6)
        return [self._name_index[m] for m in matches]

    def station_number(self, name):
        """Return the station number for an official or user-supplied name."""
        return self.station_to_number[self.resolve_station(name)]

    def station_name(self, number):
        """Return the official station name for a station number."""
        return self.number_to_station[number]

    def station_lines(self, name):
        """Return the lines serving a station."""
        return self.graph.get_station_lines(self.resolve_station(name))

    def line_stations(self):
        """Return a dict mapping line name -> sorted list of station names."""
        result = {}
        for name, lines in self.graph.station_lines.items():
            for line in lines:
                result.setdefault(line, set()).add(name)
        for conn in self.data.get("connections", []):
            if not isinstance(conn, dict):
                continue
            line = _clean(conn.get("line"))
            if not line:
                continue
            for key in ("source", "dest"):
                name = _clean(conn.get(key))
                if name in self.station_to_number:
                    result.setdefault(line, set()).add(name)
        return {line: sorted(names) for line, names in result.items()}

    def line_names(self):
        """Return the sorted list of all line names."""
        return sorted(self.line_stations().keys())

    def transfer_stations(self):
        """Return a dict of interchange stations (name -> lines) serving more than one line."""
        return {
            name: list(lines)
            for name, lines in self.graph.station_lines.items()
            if len(lines) > 1
        }

    # ------------------------------------------------------------------
    # Data updates
    # ------------------------------------------------------------------
    def _find_station_entry(self, name):
        """Return the station record dict for ``name`` or None."""
        for entry in self.data.get("stations", []):
            if isinstance(entry, dict) and _clean(entry.get("name")) == name:
                return entry
        return None

    def add_line(self, line_name, station_names, distances):
        """Add a new line described by an ordered station list and segment distances.

        Args:
            line_name: Name of the new line.
            station_names: Ordered list of station names on the line.
            distances: Distances in meters between consecutive stations
                (length must be len(station_names) - 1).

        Returns:
            The canonical ordered list of station names on the new line.

        Raises:
            ValueError: On invalid or conflicting input.
        """
        line_name = _clean(line_name)
        if not line_name:
            raise ValueError("Line name must not be empty")
        if line_name in self.line_names():
            raise ValueError(f"Line '{line_name}' already exists")

        names = []
        for raw in station_names:
            name = _clean(raw)
            if not name:
                raise ValueError("Station names must not be empty")
            try:
                canonical = self.resolve_station(name)
            except StationNotFoundError:
                canonical = name
            if canonical in names:
                raise ValueError(f"Station '{canonical}' appears more than once on the line")
            names.append(canonical)

        if len(names) < 2:
            raise ValueError("A line needs at least two stations")
        distances = list(distances)
        if len(distances) != len(names) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        parsed = [_validate_distance(d) for d in distances]

        stations = self.data.setdefault("stations", [])
        connections = self.data.setdefault("connections", [])
        for name in names:
            entry = self._find_station_entry(name)
            if entry is None:
                stations.append({"name": name, "lines": [line_name]})
            else:
                lines = entry.setdefault("lines", [])
                if line_name not in lines:
                    lines.append(line_name)

        for index, distance in enumerate(parsed):
            connections.append({
                "source": names[index],
                "dest": names[index + 1],
                "distance": distance,
                "line": line_name,
            })

        self.rebuild()
        return names

    def update_station(self, name, new_name=None, lines=None):
        """Modify a station's name and/or line affiliations.

        Args:
            name: Existing station name (user input is resolved).
            new_name: Optional new official name.
            lines: Optional new list of line names.

        Returns:
            The final station name.
        """
        canonical = self.resolve_station(name)
        stations = self.data.setdefault("stations", [])
        entry = self._find_station_entry(canonical)
        if entry is None:
            entry = {"name": canonical, "lines": []}
            stations.append(entry)

        if lines is not None:
            cleaned = []
            for line in lines:
                value = _clean(line)
                if value and value not in cleaned:
                    cleaned.append(value)
            entry["lines"] = cleaned

        target = canonical
        if new_name is not None:
            target = _clean(new_name)
            if not target:
                raise ValueError("New station name must not be empty")
            if target != canonical:
                key = normalize_station_name(target)
                if key in self._name_index and self._name_index[key] != canonical:
                    raise ValueError(f"Station name '{target}' already exists")
                entry["name"] = target
                for conn in self.data.get("connections", []):
                    if not isinstance(conn, dict):
                        continue
                    if _clean(conn.get("source")) == canonical:
                        conn["source"] = target
                    if _clean(conn.get("dest")) == canonical:
                        conn["dest"] = target

        self.rebuild()
        return target

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance of the connection(s) between two stations.

        Args:
            source: One station name.
            dest: The adjacent station name.
            distance: New distance in meters (positive integer).
            line: Optional line name to restrict the update.

        Returns:
            The number of connection records updated.

        Raises:
            ValueError: If no matching connection exists or distance is invalid.
        """
        src = self.resolve_station(source)
        dst = self.resolve_station(dest)
        value = _validate_distance(distance)
        count = 0
        for conn in self.data.get("connections", []):
            if not isinstance(conn, dict):
                continue
            pair = (_clean(conn.get("source")), _clean(conn.get("dest")))
            if set(pair) != {src, dst}:
                continue
            if line is not None and _clean(conn.get("line")) != _clean(line):
                continue
            conn["distance"] = value
            count += 1
        if count == 0:
            raise ValueError(f"No connection found between '{src}' and '{dst}'")
        self.rebuild()
        return count

    def save(self, path=None):
        """Persist the current dataset to ``path`` (defaults to the loaded file)."""
        save_subway_data(self.data, path or self.data_path)

    # ------------------------------------------------------------------
    # Integrity checks
    # ------------------------------------------------------------------
    def check_integrity(self):
        """Run data integrity checks.

        Checks performed:
            * Coverage: main lines exist and representative stations are present.
            * Records: station names are unique and well formed.
            * Connections: endpoints exist, lines are valid, distances are sane.
            * Connectivity: each line forms a connected chain; no isolated stations.

        Returns:
            dict with keys "ok" (bool), "errors" (list[str]), "warnings" (list[str]),
            "station_count", "connection_count" and "line_count".
        """
        errors = []
        warnings = []

        stations_data = self.data.get("stations", [])
        connections = self.data.get("connections", [])

        declared = set()
        record_lines = {}
        for index, entry in enumerate(stations_data):
            if not isinstance(entry, dict):
                errors.append(f"Station record #{index} is not an object")
                continue
            raw_name = entry.get("name")
            name = _clean(raw_name)
            if not name:
                errors.append(f"Station record #{index} has no name")
                continue
            if raw_name != name:
                warnings.append(f"Station name '{raw_name}' has leading or trailing whitespace")
            if name in declared:
                errors.append(f"Duplicate station record for '{name}'")
            declared.add(name)
            lines = entry.get("lines")
            if not isinstance(lines, list) or not lines:
                warnings.append(f"Station '{name}' has no line affiliation")
                lines = []
            record_lines[name] = {_clean(l) for l in lines}

        # Coverage checks.
        lines_map = self.line_stations()
        for line in MAIN_LINES:
            if not lines_map.get(line):
                errors.append(f"Main line '{line}' is missing from the dataset")
        normalized_present = {normalize_station_name(n) for n in declared}
        for line, reps in REPRESENTATIVE_STATIONS.items():
            for rep in reps:
                if normalize_station_name(rep) not in normalized_present:
                    warnings.append(f"Representative station '{rep}' of {line} is not in the dataset")

        # Connection checks.
        pair_counts = {}
        valid_adjacency = {}
        for index, conn in enumerate(connections):
            label = f"Connection #{index}"
            if not isinstance(conn, dict):
                errors.append(f"{label} is not an object")
                continue
            src = _clean(conn.get("source"))
            dst = _clean(conn.get("dest"))
            line = _clean(conn.get("line"))
            if not src or not dst:
                errors.append(f"{label} is missing its source or dest station")
                continue
            if src == dst:
                errors.append(f"{label} connects station '{src}' to itself")
            if src not in declared:
                errors.append(f"{label}: station '{src}' is not in the station list")
            if dst not in declared:
                errors.append(f"{label}: station '{dst}' is not in the station list")
            if not line:
                errors.append(f"{label} ({src} - {dst}) has no line")

            distance = conn.get("distance")
            distance_ok = False
            if isinstance(distance, bool) or not isinstance(distance, (int, float)):
                errors.append(f"{label} ({src} - {dst}) has a non-numeric distance {distance!r}")
            elif distance <= 0:
                errors.append(f"{label} ({src} - {dst}) has a non-positive distance {distance}")
            else:
                distance_ok = True
                if isinstance(distance, float) and distance != int(distance):
                    warnings.append(f"{label} ({src} - {dst}) has a fractional distance {distance}")
                if distance > MAX_REASONABLE_DISTANCE_M:
                    warnings.append(f"{label} ({src} - {dst}) distance {distance} m seems too large")
                elif distance < MIN_REASONABLE_DISTANCE_M:
                    warnings.append(f"{label} ({src} - {dst}) distance {distance} m seems too small")

            for name in (src, dst):
                if line and name in record_lines and line not in record_lines[name]:
                    warnings.append(f"Station '{name}' does not list line '{line}' in its line affiliations")

            key = (frozenset((src, dst)), line)
            pair_counts[key] = pair_counts.get(key, 0) + 1

            if distance_ok and src != dst and src in declared and dst in declared and line:
                valid_adjacency.setdefault(line, {}).setdefault(src, set()).add(dst)
                valid_adjacency[line].setdefault(dst, set()).add(src)

        for (pair, line), count in pair_counts.items():
            if count > 1:
                names = " - ".join(sorted(pair))
                warnings.append(f"Duplicate connection {names} on line '{line}' ({count} records)")

        # Connectivity checks per line.
        for line, names in lines_map.items():
            adjacency = valid_adjacency.get(line, {})
            visited = set()
            components = 0
            for start in names:
                if start in visited:
                    continue
                components += 1
                queue = [start]
                visited.add(start)
                while queue:
                    current = queue.pop()
                    for neighbor in adjacency.get(current, ()):
                        if neighbor not in visited:
                            visited.add(neighbor)
                            queue.append(neighbor)
            if components > 1:
                warnings.append(f"Line '{line}' is split into {components} disconnected segments")

        for name, number in self.station_to_number.items():
            if not self.graph.adjacent(number):
                warnings.append(f"Station '{name}' has no connections")

        return {
            "ok": not errors,
            "errors": errors,
            "warnings": warnings,
            "station_count": len(self.station_to_number),
            "connection_count": self.graph.edge_count(),
            "line_count": len(lines_map),
        }

"""Route query service: ties network, path search, fare and time together.

The service validates station names, runs the shortest path search, and
builds a JSON-serializable result dictionary used by the report and CLI layers.
"""

from fare_calculator import Sum_money, calculate_estimated_time
from network import StationNotFoundError
from pathfinding import PathfindingConfig, ShortestPath

FORMAT_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class RouteQueryError(Exception):
    """Base class for all query errors."""


class InvalidQueryError(RouteQueryError):
    """Raised when the query input is malformed or logically invalid."""


class EndpointNotFoundError(RouteQueryError):
    """Raised when the starting or ending station does not exist."""

    def __init__(self, side, name, suggestions=None):
        self.side = side
        self.name = name
        self.suggestions = list(suggestions or [])
        super().__init__(f"{side} station '{name}' does not exist, please check the station name")


class PathNotFoundError(RouteQueryError):
    """Raised when no route satisfies the query constraints."""


class FareCalculationError(RouteQueryError):
    """Raised when the fare cannot be calculated."""


def parse_query_input(text):
    """Split 'Starting Station; Ending Station' input into two names.

    Full-width semicolons are accepted as well.

    Returns:
        (start_name, end_name)

    Raises:
        InvalidQueryError: If the format is incorrect.
    """
    if not isinstance(text, str):
        raise InvalidQueryError(FORMAT_MESSAGE)
    normalized = text.replace("\uff1b", ";")
    if normalized.count(";") != 1:
        raise InvalidQueryError(FORMAT_MESSAGE)
    start, end = (part.strip() for part in normalized.split(";"))
    if not start or not end:
        raise InvalidQueryError(FORMAT_MESSAGE)
    return start, end


class RouteService:
    """Executes route queries against a SubwayNetwork."""

    def __init__(self, network):
        self.network = network

    def _resolve(self, name, side):
        """Resolve a station name or raise EndpointNotFoundError."""
        try:
            return self.network.resolve_station(name)
        except StationNotFoundError as exc:
            raise EndpointNotFoundError(side, exc.name, exc.suggestions) from exc

    def query(self, start_name, end_name, config=None):
        """Compute the route between two stations.

        Args:
            start_name: Starting station name (user input).
            end_name: Ending station name (user input).
            config: PathfindingConfig; defaults to shortest-distance mode.

        Returns:
            dict with keys: query, path, segments, transfers, transfer_count,
            lines, distance_m, distance_km, fare, estimated_time_min.

        Raises:
            EndpointNotFoundError, InvalidQueryError, PathNotFoundError,
            FareCalculationError, RouteQueryError.
        """
        config = config if config is not None else PathfindingConfig()

        start = self._resolve(start_name, "Starting")
        end = self._resolve(end_name, "Ending")
        if start == end:
            raise InvalidQueryError("Starting station cannot be the same as the ending station")

        try:
            source_number = self.network.station_number(start)
            target_number = self.network.station_number(end)
            finder = ShortestPath(self.network.graph, source_number, config)
        except Exception as exc:  # defensive: unexpected path calculation failure
            raise RouteQueryError(f"Path calculation failed: {exc}") from exc

        if not finder.has_path_to(target_number):
            if config.max_transfers is not None:
                raise PathNotFoundError(
                    f"No route found from '{start}' to '{end}' within {config.max_transfers} transfer(s)"
                )
            raise PathNotFoundError(f"No route found from '{start}' to '{end}'")

        try:
            numbers, edges = finder.path_with_edges(target_number)
            transfers = finder.get_transfer_count(target_number)
        except Exception as exc:
            raise RouteQueryError(f"Path calculation failed: {exc}") from exc

        names = [self.network.station_name(n) for n in numbers]
        distance_m = sum(edge.distance for edge in edges)

        try:
            fare = Sum_money(distance_m)
        except (TypeError, ValueError) as exc:
            raise FareCalculationError(f"Fare calculation failed: {exc}") from exc

        try:
            estimated = calculate_estimated_time(distance_m, transfers)
        except (TypeError, ValueError) as exc:
            raise RouteQueryError(f"Time estimation failed: {exc}") from exc

        segments, transfer_details = self._build_segments(names, edges)
        lines = []
        for segment in segments:
            if segment["line"] not in lines:
                lines.append(segment["line"])

        return {
            "query": {
                "start": start,
                "end": end,
                "mode": config.mode,
                "max_transfers": config.max_transfers,
                "preferred_lines": list(config.preferred_lines),
            },
            "path": names,
            "segments": segments,
            "transfers": transfer_details,
            "transfer_count": transfers,
            "lines": lines,
            "distance_m": distance_m,
            "distance_km": round(distance_m / 1000.0, 2),
            "fare": fare,
            "estimated_time_min": estimated,
        }

    @staticmethod
    def _build_segments(names, edges):
        """Group consecutive edges on the same line into segments and list transfers.

        Returns:
            (segments, transfer_details)
        """
        segments = []
        transfer_details = []
        for index, edge in enumerate(edges):
            if not segments or segments[-1]["line"] != edge.line:
                if segments:
                    transfer_details.append({
                        "station": names[index],
                        "from_line": segments[-1]["line"],
                        "to_line": edge.line,
                    })
                segments.append({
                    "line": edge.line,
                    "stations": [names[index]],
                    "distance_m": 0,
                })
            segment = segments[-1]
            segment["stations"].append(names[index + 1])
            segment["distance_m"] += edge.distance
        return segments, transfer_details

"""Path visualization and export: console text, JSON and path summaries."""

import json


def format_distance(distance_m):
    """Format a distance using meters for short trips and kilometers for long trips.

    Short distances (< 1000 m) are shown in meters followed by kilometers;
    longer distances are shown in kilometers followed by meters.
    """
    if distance_m < 1000:
        return f"{distance_m} m ({distance_m / 1000:.2f} km)"
    return f"{distance_m / 1000:.2f} km ({distance_m} m)"


def generate_path_summary(result):
    """Extract key summary information from a query result.

    Returns:
        dict with total_distance_km, total_fare, transfer_count and estimated_time_min.
    """
    return {
        "total_distance_km": result["distance_km"],
        "total_fare": result["fare"],
        "transfer_count": result["transfer_count"],
        "estimated_time_min": result["estimated_time_min"],
    }


def format_text_report(result):
    """Render a query result as human-readable console text."""
    summary = generate_path_summary(result)
    query = result["query"]
    separator = "-" * 60
    lines = []

    lines.append("=" * 60)
    lines.append(f"Route: {query['start']} -> {query['end']}")
    lines.append(f"Mode: {query['mode']}")
    if query["max_transfers"] is not None:
        lines.append(f"Max transfers: {query['max_transfers']}")
    if query["preferred_lines"]:
        lines.append("Preferred lines: " + ", ".join(query["preferred_lines"]))
    lines.append(separator)

    lines.append(f"Stations ({len(result['path'])}):")
    for index, name in enumerate(result["path"], start=1):
        lines.append(f"  {index:>3}. {name}")
    lines.append(separator)

    lines.append("Line segments:")
    for segment in result["segments"]:
        stops = len(segment["stations"]) - 1
        lines.append(
            f"  [{segment['line']}] {segment['stations'][0]} -> {segment['stations'][-1]}"
            f" ({stops} stops, {format_distance(segment['distance_m'])})"
        )

    lines.append("Transfers:")
    if not result["transfers"]:
        lines.append("  None (direct travel)")
    else:
        for transfer in result["transfers"]:
            lines.append(
                f"  {transfer['station']}: {transfer['from_line']} -> {transfer['to_line']}"
            )
    lines.append("Lines involved: " + ", ".join(result["lines"]))
    lines.append(separator)

    lines.append(f"Total distance: {format_distance(result['distance_m'])}")
    lines.append(f"Transfer count: {summary['transfer_count']}")
    lines.append(f"Estimated time: {summary['estimated_time_min']} min")
    lines.append(f"Fare: {result['fare']} yuan")
    lines.append("=" * 60)
    return "\n".join(lines)


def format_json_report(result):
    """Render a query result as an indented JSON document."""
    return json.dumps(result, ensure_ascii=False, indent=2)


def render_result(result, output_format):
    """Render a result in the requested format ("text" or "json")."""
    if output_format == "json":
        return format_json_report(result)
    return format_text_report(result)

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import sys

from data_loader import load_subway_data
from fare_calculator import Sum_money  # noqa: F401  (fare module is part of the runtime)
from network import SubwayNetwork
from pathfinding import PathfindingConfig, VALID_MODES

    EndpointNotFoundError,
    FareCalculationError,
    FORMAT_MESSAGE,
    InvalidQueryError,
    PathNotFoundError,
    RouteQueryError,
    RouteService,
    parse_query_input,

EXIT_COMMANDS = ("quit", "exit", "q")


def build_arg_parser():
    """Create the argument parser for the CLI."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (shortest path and fare query)."
    )
    parser.add_argument("--output_file", default=None,
                        help="append query results to this file")
    parser.add_argument("--format", dest="output_format", choices=["text", "json"], default="text",
                        help="output format: text or json (default: text)")
    parser.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="maximum number of allowed transfers")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line(s), e.g. 'Line 1' or 'Line 1,Line 2'")
    parser.add_argument("--check_data", action="store_true",
                        help="run data integrity check and exit")
    return parser


def parse_line_preference(value):
    """Convert the --line_preference argument into a list of line names."""
    if not value:
        return []
    return [part.strip() for part in value.split(",") if part.strip()]


def format_integrity_report(report):
    """Render the data integrity check result as text."""
    lines = []
    status = "PASSED" if report["ok"] else "FAILED"
    lines.append(f"Data integrity check: {status}")
    lines.append(f"Stations: {report['station_count']}")
    lines.append(f"Connections: {report['connection_count']}")
    lines.append(f"Lines: {report['line_count']}")
    for message in report["errors"]:
        lines.append(f"ERROR: {message}")
    for message in report["warnings"]:
        lines.append(f"WARNING: {message}")
    if report["ok"] and not report["warnings"]:
        lines.append("No problems found.")
    return "\n".join(lines)


def _format_error(exc):
    """Build the user-facing message for a query error."""
    message = str(exc)
    if isinstance(exc, EndpointNotFoundError) and exc.suggestions:
        message += ". Did you mean: " + ", ".join(exc.suggestions) + "?"
    return message


def append_to_file(path, content):
    """Append rendered output to a file, separated by a blank line."""
    with open(path, "a", encoding="utf-8") as handle:
        handle.write(content)
        handle.write("\n\n")


def handle_query(service, text, config, output_format, output_file=None, output_func=print):
    """Process one query line. Returns True on success, False on error."""
    try:
        start, end = parse_query_input(text)
        result = service.query(start, end, config)
    except EndpointNotFoundError as exc:
        output_func(f"Error: {_format_error(exc)}")
        return False
    except InvalidQueryError as exc:
        output_func(f"Error: {exc}")
        return False
    except PathNotFoundError as exc:
        output_func(f"Error: {exc}")
        return False
    except FareCalculationError as exc:
        output_func(f"Error: {exc}")
        return False
    except RouteQueryError as exc:
        output_func(f"Error: {exc}")
        return False
    except Exception as exc:  # last-resort guard so the session keeps running
        output_func(f"Error: unexpected failure while processing query: {exc}")
        return False

    rendered = render_result(result, output_format)
    output_func(rendered)

    if output_file:
        try:
            append_to_file(output_file, rendered)
        except OSError as exc:
            output_func(f"Warning: could not write to output file '{output_file}': {exc}")
    return True


def run_interactive(service, config, output_format, output_file=None,
                    input_func=input, output_func=print):
    """Run the continuous query loop until the user exits.

    Returns:
        The number of successful queries.
    """
    successes = 0
    output_func("Enter a query as 'Starting Station; Ending Station' (type 'quit' to exit).")
    while True:
        try:
            line = input_func("> ")
        except (EOFError, KeyboardInterrupt):
            output_func("")
            break
        text = line.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        if handle_query(service, text, config, output_format, output_file, output_func):
            successes += 1
    return successes


def main(argv=None):
    """Entry point for the CLI. Returns the process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")

    try:
        data = load_subway_data()
    except (OSError, ValueError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    try:
        network = SubwayNetwork(data)
    except Exception as exc:
        print(f"Error: failed to build subway network: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        report = network.check_integrity()
        print(format_integrity_report(report))
        return 0 if report["ok"] else 1

    preferred = parse_line_preference(args.line_preference)
    known_lines = set(network.line_names())
    unknown = [line for line in preferred if line not in known_lines]
    if unknown:
        print("Warning: unknown preferred line(s): " + ", ".join(unknown), file=sys.stderr)

    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=preferred)
    except ValueError as exc:
        parser.error(str(exc))

    service = RouteService(network)
    print("Beijing Subway fare calculator")
    print(f"Input format: {FORMAT_MESSAGE}" if False else "Input format: 'Starting Station; Ending Station'")
    run_interactive(service, config, args.output_format, args.output_file)
    return 0


if __name__ == "__main__":
    sys.exit(main())

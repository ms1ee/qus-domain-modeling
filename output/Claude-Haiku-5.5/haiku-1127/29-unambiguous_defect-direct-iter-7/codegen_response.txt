"""Coordinate system helpers for the 800x800 delivery map.

The map is an 8x8 grid of 100x100 cells. Streets run along the lines where
x or y is a multiple of 100. Points at integer hundreds are street
intersections where restaurants, customers and riders are located.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE      # 8 cells per axis
LATTICE_DIM = GRID_DIM + 1            # 9 street lines per axis
MIN_COORD = 0
MAX_COORD = MAP_SIZE


class CoordinateError(ValueError):
    """Raised when a coordinate is not an integer or lies outside the map."""


def validate_coordinate(value):
    """Check that ``value`` is an integer within the map bounds.

    Args:
        value: The coordinate to check.

    Returns:
        The validated integer coordinate.

    Raises:
        CoordinateError: If the value is not an integer or is out of range.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise CoordinateError(f"coordinate must be an integer, got {value!r}")
    if not MIN_COORD <= value <= MAX_COORD:
        raise CoordinateError(
            f"coordinate {value} is out of range {MIN_COORD}-{MAX_COORD}"
        )
    return value


def snap_axis(value):
    """Round one axis value to the nearest integer-hundred street line.

    Args:
        value: A numeric coordinate.

    Returns:
        An integer multiple of 100 within the map bounds.
    """
    value = max(MIN_COORD, min(MAX_COORD, int(value)))
    snapped = ((value + CELL_SIZE // 2) // CELL_SIZE) * CELL_SIZE
    return max(MIN_COORD, min(MAX_COORD, snapped))


def snap_to_grid(x, y):
    """Adjust a point to the nearest street intersection.

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.

    Returns:
        Tuple ``(x, y)`` of integer hundreds.
    """
    return snap_axis(x), snap_axis(y)


def cell_index(x, y):
    """Return the (column, row) index of the grid cell containing a point.

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.

    Returns:
        Tuple ``(column, row)`` with values in 0-7.
    """
    col = min(max(int(x) // CELL_SIZE, 0), GRID_DIM - 1)
    row = min(max(int(y) // CELL_SIZE, 0), GRID_DIM - 1)
    return col, row


def cell_center(x, y):
    """Return the center point of the cell containing a point.

    Uses ``center = (grid // 100) * 100 + 50`` for each axis.

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.

    Returns:
        Tuple ``(center_x, center_y)``.
    """
    col, row = cell_index(x, y)
    return col * CELL_SIZE + CELL_SIZE // 2, row * CELL_SIZE + CELL_SIZE // 2


def is_on_street(x, y):
    """Return True if the point lies on a street (grid line).

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.
    """
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


def manhattan(a, b):
    """Return the Manhattan distance between two points.

    Args:
        a: Tuple ``(x, y)``.
        b: Tuple ``(x, y)``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Grid-based path planning along streets.

Movement always follows the street network: horizontal moves first, then
vertical moves. A target lying directly above, below, left or right of the
rider is reached by a straight-line move (the special case). Riders never
block one another, and every move is monotone toward its target, so no
deadlock can occur.
"""

import math

import grid


class PathPlanner:
    """Static helpers for route computation and movement along streets."""

    @staticmethod
    def step_towards(x, y, tx, ty, budget):
        """Advance a point along the street grid by at most ``budget`` units.

        The horizontal leg is walked first, then the vertical leg. If the
        horizontal leg ends mid-tick, the remaining budget continues onto the
        vertical leg, so turns are right angles.

        Args:
            x: Current horizontal coordinate.
            y: Current vertical coordinate.
            tx: Target horizontal coordinate.
            ty: Target vertical coordinate.
            budget: Maximum distance to travel this tick.

        Returns:
            Tuple ``(new_x, new_y, leftover_budget)``.
        """
        while budget > 0 and (x, y) != (tx, ty):
            if x != tx:
                step = min(budget, abs(tx - x))
                x += step if tx > x else -step
            else:
                step = min(budget, abs(ty - y))
                y += step if ty > y else -step
            budget -= step
        return x, y, budget

    @staticmethod
    def is_straight_line(start, end):
        """Return True if ``end`` is directly above/below or left/right of ``start``.

        Args:
            start: Tuple ``(x, y)``.
            end: Tuple ``(x, y)``.
        """
        return start[0] == end[0] or start[1] == end[1]

    @staticmethod
    def plan_route(start, end):
        """Return the corner points of a street route from start to end.

        Straight-line targets (special case) produce a two-point route.
        Otherwise a single right-angle corner is inserted, horizontal first.

        Args:
            start: Tuple ``(x, y)``.
            end: Tuple ``(x, y)``.

        Returns:
            List of corner points beginning with ``start``.
        """
        points = [tuple(start)]
        sx, sy = start
        ex, ey = end
        if sx != ex and sy != ey:
            points.append((ex, sy))
        if points[-1] != (ex, ey):
            points.append((ex, ey))
        return points

    @staticmethod
    def plan_multi_route(start, stops):
        """Return a continuous route from start through several stops.

        Args:
            start: Tuple ``(x, y)`` of the starting point.
            stops: Iterable of tuples ``(x, y)`` visited in order.

        Returns:
            List of corner points with consecutive duplicates removed.
        """
        points = [tuple(start)]
        current = tuple(start)
        for stop in stops:
            stop = tuple(stop)
            for point in PathPlanner.plan_route(current, stop)[1:]:
                if point != points[-1]:
                    points.append(point)
            current = stop
        return points

    @staticmethod
    def route_length(points):
        """Return the total Manhattan length of a corner-point route.

        Args:
            points: List of tuples ``(x, y)``.
        """
        return sum(
            grid.manhattan(a, b) for a, b in zip(points, points[1:])
        )

    @staticmethod
    def ticks_needed(distance, speed):
        """Return the number of ticks needed to cover a distance.

        Args:
            distance: Distance in map units.
            speed: Speed in units per tick (must be positive).

        Raises:
            ValueError: If speed is not positive.
        """
        if speed <= 0:
            raise ValueError("speed must be positive")
        return int(math.ceil(distance / speed))

"""Order data model: a single delivery request and its lifecycle."""

import globalvar
import grid

PENDING = "pending"
ASSIGNED = "assigned"
PICKING_UP = "picking_up"
DELIVERING = "delivering"
COMPLETED = "completed"


def _next_order_id():
    """Return the next unique order ID and advance the global counter."""
    value = int(globalvar.get_value("order_counter", 0)) + 1
    globalvar.set_value("order_counter", value)
    return value


def _now():
    """Return the current simulated time."""
    return int(globalvar.get_value("time", 0))


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique positive integer identifier.
        raw_input: The four user-entered coordinates
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.
        res_pos: Restaurant location snapped to a grid intersection.
        cus_pos: Customer location snapped to a grid intersection.
        status: One of "pending", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: ID of the assigned rider, or None.
        created_time: Simulated time when the order was created.
        assigned_time: Simulated time of assignment.
        accepted_time: Simulated time the rider started pickup.
        pickup_time: Simulated time the rider reached the restaurant.
        complete_time: Simulated time the order was delivered.
        distance: Manhattan distance from restaurant to customer.
    """

    def __init__(self, res_pos, cus_pos):
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: Pair ``(x, y)`` of the restaurant location (0-800).
            cus_pos: Pair ``(x, y)`` of the customer location (0-800).

        Raises:
            ValueError: If a location is malformed or out of range.
        """
        res = self._parse_point(res_pos, "restaurant")
        cus = self._parse_point(cus_pos, "customer")
        self.id = _next_order_id()
        self.raw_input = list(res + cus)
        self.res_pos = grid.snap_to_grid(*res)
        self.cus_pos = grid.snap_to_grid(*cus)
        self.status = PENDING
        self.rider_id = None
        self.created_time = _now()
        self.assigned_time = None
        self.accepted_time = None
        self.pickup_time = None
        self.complete_time = None
        self.distance = grid.manhattan(self.res_pos, self.cus_pos)

    @staticmethod
    def _parse_point(point, label):
        """Validate a coordinate pair and return it as integers.

        Args:
            point: Pair of coordinates.
            label: Name used in error messages.

        Raises:
            ValueError: If the point is malformed or out of range.
        """
        try:
            x, y = point
        except (TypeError, ValueError):
            raise ValueError(f"{label} location must be an (x, y) pair")
        return grid.validate_coordinate(x), grid.validate_coordinate(y)

    def assign_to(self, rider_id):
        """Record that this order is assigned to the rider with ``rider_id``.

        Args:
            rider_id: The identifier of the assigned rider.
        """
        self.rider_id = rider_id
        self.status = ASSIGNED
        self.assigned_time = _now()

    def mark_accepted(self, now):
        """Mark that the rider has started the pickup leg.

        Args:
            now: Current simulated time.
        """
        self.status = PICKING_UP
        self.accepted_time = now

    def mark_picked_up(self, now):
        """Mark that the rider has arrived at the restaurant.

        Args:
            now: Current simulated time.
        """
        self.status = DELIVERING
        self.pickup_time = now

    def mark_completed(self, now):
        """Mark the order as delivered to the customer.

        Args:
            now: Current simulated time.
        """
        self.status = COMPLETED
        self.complete_time = now

    def time_taken(self):
        """Return ticks from creation to delivery, or None if unfinished."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.created_time

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "raw_input": self.raw_input,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_time": self.created_time,
            "assigned_time": self.assigned_time,
            "accepted_time": self.accepted_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
        }

"""Rider model: state machine, street movement and order execution."""

import logging
import random

import globalvar
import grid
from path_planner import PathPlanner

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}
SPEED_CHOICES = (50, 100)

logger = logging.getLogger("dispatch.rider")


class Rider:
    """A delivery rider that moves along streets and executes orders.

    Attributes:
        rider_id: Unique integer identifier.
        x: Current horizontal coordinate.
        y: Current vertical coordinate.
        speed: Movement speed in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Queue of assigned orders; ``orders[0]`` is the active order.
        special_case: True when the current target lies on the same row or
            column as the rider (straight-line movement).
        log: List of status-change dictionaries with keys including
            "event" and "order_id".
        distance_travelled: Total distance covered in map units.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        """Create a rider and place it on a street intersection.

        Args:
            rider_id: Unique identifier for the rider.
            x: Optional starting x coordinate (snapped to a street).
            y: Optional starting y coordinate (snapped to a street).
            speed: Optional speed in units per tick.
        """
        if x is None or y is None:
            x = random.randrange(grid.LATTICE_DIM) * grid.CELL_SIZE
            y = random.randrange(grid.LATTICE_DIM) * grid.CELL_SIZE
        x, y = grid.snap_to_grid(x, y)
        self.rider_id = rider_id
        self.x = x
        self.y = y
        self.speed = int(speed) if speed else random.choice(SPEED_CHOICES)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.distance_travelled = 0
        self._record("spawn", None)

    @property
    def position(self):
        """Return the current position as a tuple ``(x, y)``."""
        return self.x, self.y

    @property
    def state_name(self):
        """Return a human-readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def is_available(self):
        """Return True if the rider is idle and has no assigned orders."""
        return self.state == IDLE and not self.orders

    def current_target(self):
        """Return the point the rider is heading to, or None if no orders."""
        if not self.orders:
            return None
        order = self.orders[0]
        if self.state == DELIVERING:
            return order.cus_pos
        return order.res_pos

    def _is_straight_line(self, target):
        """Return True if the target shares the rider's row or column."""
        return self.x == target[0] or self.y == target[1]

    def move(self):
        """Advance the rider by one tick and apply state transitions.

        Transitions:
            Idle with an order -> Picking up (event "accepted_order").
            Picking up, arrived at restaurant -> Delivering
                (event "arrived_at_merchant").
            Delivering, arrived at customer -> Idle
                (event "delivered_to_customer"); the order is completed and
                removed from the rider's queue.
        """
        now = int(globalvar.get_value("time", 0))
        if not self.orders:
            if self.state != IDLE:
                logger.warning("rider %s had state %s without orders; resetting",
                               self.rider_id, self.state)
                self.state = IDLE
            self.special_case = False
            return

        order = self.orders[0]
        if self.state == IDLE:
            self.state = PICKING_UP
            order.mark_accepted(now)
            self._record("accepted_order", order.id)
            return

        target = self.current_target()
        self.special_case = self._is_straight_line(target)
        nx, ny, _ = PathPlanner.step_towards(self.x, self.y, target[0], target[1],
                                             self.speed)
        self.distance_travelled += grid.manhattan(self.position, (nx, ny))
        self.x, self.y = nx, ny
        if self.position != tuple(target):
            return

        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.mark_picked_up(now)
            self._record("arrived_at_merchant", order.id)
        else:
            self.orders.pop(0)
            order.mark_completed(now)
            self.state = IDLE
            self.special_case = False
            self._record("delivered_to_customer", order.id)

    def _record(self, event, order_id, **extra):
        """Append a status-change entry to the rider log.

        Args:
            event: Event name such as "spawn" or "accepted_order".
            order_id: Related order ID, or None.
            **extra: Additional fields to store with the entry.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "time": int(globalvar.get_value("time", 0)),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        entry.update(extra)
        self.log.append(entry)
        logger.debug("rider %s event %s order %s", self.rider_id, event, order_id)

    def to_dict(self):
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "order_ids": [o.id for o in self.orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

"""Order assignment: rider selection and order lifecycle orchestration."""

import logging

import globalvar
import grid
from order_model import Order
from path_planner import PathPlanner
from system import SystemManager

LOAD_PENALTY = 150

logger = logging.getLogger("dispatch.order")


def rider_cost(rider, order):
    """Return the assignment cost of giving ``order`` to ``rider``.

    The cost is the distance from the rider to the restaurant plus a penalty
    for each order the rider already carries, which balances the load.

    Args:
        rider: A Rider instance.
        order: An Order instance.
    """
    distance = grid.manhattan(rider.position, order.res_pos)
    return distance + LOAD_PENALTY * len(rider.orders)


def find_best_rider_for_order(order):
    """Select the rider with the lowest assignment cost for an order.

    Ties are broken by fewer assigned orders, then by lower rider ID.

    Args:
        order: The Order to assign.

    Returns:
        The selected Rider.

    Raises:
        LookupError: If no riders exist.
    """
    riders = globalvar.get_value("riders", [])
    if not riders:
        raise LookupError("no riders are available")
    return min(
        riders,
        key=lambda r: (rider_cost(r, order), len(r.orders), r.rider_id),
    )


class OrderManager:
    """Create, assign and summarise delivery orders."""

    @staticmethod
    def create_order(res_pos, cus_pos):
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: Pair ``(x, y)`` for the restaurant (0-800).
            cus_pos: Pair ``(x, y)`` for the customer (0-800).

        Returns:
            A new Order instance (not yet assigned).
        """
        return Order(res_pos, cus_pos)

    @staticmethod
    def estimate_eta(rider, order):
        """Estimate ticks until the order is delivered by ``rider``.

        Args:
            rider: The candidate Rider.
            order: The Order being considered.

        Returns:
            Estimated number of ticks.
        """
        queued = sum(o.distance for o in rider.orders)
        legs = grid.manhattan(rider.position, order.res_pos) + order.distance
        return PathPlanner.ticks_needed(queued + legs, rider.speed)

    @staticmethod
    def assign_order(order):
        """Assign an order to the best available rider.

        Recruits a new rider first if the system has no idle capacity and
        funds allow.

        Args:
            order: The Order to assign.

        Returns:
            Tuple ``(rider, new_rider, eta_ticks)`` where ``new_rider`` is
            True if a rider was recruited for this order.

        Raises:
            LookupError: If no rider could be found or recruited.
        """
        new_rider = SystemManager.ensure_capacity()
        rider = find_best_rider_for_order(order)
        eta = OrderManager.estimate_eta(rider, order)
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        globalvar.get_value("orders").append(order)
        logger.info("order %s assigned to rider %s", order.id, rider.rider_id)
        return rider, new_rider, eta

"""System-level state management: funds, recruitment, clock and persistence."""

import json
import logging
import os
import random

import globalvar
from rider import Rider, IDLE

DEFAULT_CONFIG = {
    "initial_funds": 5000,
    "recruit_cost": 500,
    "max_riders": 20,
    "random_seed": None,
}

logger = logging.getLogger("dispatch.system")


class SystemManager:
    """Static helpers for managing funds, riders and the simulation clock."""

    @staticmethod
    def get_funds():
        """Return the current system fund balance."""
        return int(globalvar.get_value("funds", 0))

    @staticmethod
    def adjust_funds(amount):
        """Add ``amount`` (possibly negative) to the system funds.

        Args:
            amount: Integer amount to add.

        Returns:
            The new fund balance.
        """
        funds = SystemManager.get_funds() + int(amount)
        globalvar.set_value("funds", funds)
        return funds

    @staticmethod
    def rider_count():
        """Return the number of riders currently in the system."""
        return len(globalvar.get_value("riders", []))

    @staticmethod
    def recruit_rider():
        """Create a new rider, charge the recruitment cost and register it.

        Returns:
            The newly created Rider.
        """
        counter = int(globalvar.get_value("rider_id_counter", 0)) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        globalvar.get_value("riders").append(rider)
        SystemManager.adjust_funds(-int(globalvar.get_value("recruit_cost", 500)))
        logger.info("recruited rider %s", counter)
        return rider

    @staticmethod
    def ensure_capacity():
        """Recruit a rider if no rider is idle and funds and limits allow.

        Returns:
            True if a new rider was recruited, otherwise False.
        """
        riders = globalvar.get_value("riders", [])
        if any(r.is_available() for r in riders):
            return False
        if len(riders) >= int(globalvar.get_value("max_riders", 20)):
            logger.warning("maximum rider count reached")
            return False
        cost = int(globalvar.get_value("recruit_cost", 500))
        if SystemManager.get_funds() < cost:
            logger.warning("insufficient funds to recruit a rider")
            return False
        SystemManager.recruit_rider()
        return True

    @staticmethod
    def tick():
        """Advance the simulation by one tick and move every rider.

        Returns:
            The new simulated time.
        """
        now = int(globalvar.get_value("time", 0)) + 1
        globalvar.set_value("time", now)
        for rider in list(globalvar.get_value("riders", [])):
            try:
                rider.move()
            except Exception:  # pragma: no cover - recovery path
                logger.exception("error moving rider %s; resetting it",
                                 rider.rider_id)
                rider.state = IDLE
        return now

    @staticmethod
    def is_running():
        """Return True while the main loop should keep running."""
        return bool(globalvar.get_value("running", True))

    @staticmethod
    def shutdown():
        """Stop the main loop."""
        globalvar.set_value("running", False)

    @staticmethod
    def check_consistency():
        """Check internal state invariants and return a list of issues.

        Returns:
            List of human-readable problem descriptions (empty if healthy).
        """
        issues = []
        seen = set()
        for rider in globalvar.get_value("riders", []):
            if rider.rider_id in seen:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen.add(rider.rider_id)
            if rider.state != IDLE and not rider.orders:
                issues.append(f"rider {rider.rider_id} is {rider.state_name} "
                              "without orders")
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    issues.append(f"order {order.id} is queued on rider "
                                  f"{rider.rider_id} but assigned to "
                                  f"{order.rider_id}")
        return issues

    @staticmethod
    def save_state(path):
        """Write the configuration, riders and orders to a JSON file.

        Args:
            path: Destination file path.

"""Status displays and ASCII map visualisation."""

import globalvar
import grid
from rider import STATE_NAMES

MAP_CELLS = grid.GRID_DIM            # 8
MAP_NODES = grid.LATTICE_DIM         # 9


class DisplayManager:
    """Render the map, system status, order list and rider logs as text."""

    @staticmethod
    def _cell(point):
        """Return the (row, col) character position for a map point.

        Intersections sit at even row/column indices. Odd indices hold the
        street segments between intersections. Row 0 is the top (y = 800).
        """
        gx = min(max(int(round(point[0] / grid.CELL_SIZE)), 0), MAP_CELLS)
        gy = min(max(int(round(point[1] / grid.CELL_SIZE)), 0), MAP_CELLS)
        return 2 * (MAP_CELLS - gy), 2 * gx

    @classmethod
    def render_map(cls, route=None, riders=None):
        """Render the 8x8 grid as an ASCII map.

        Symbols: '.' intersection, '-' and '|' streets, 'M' restaurant,
        'C' customer, 'R' rider, '*' planned route.

        Args:
            route: Optional list of corner points to highlight.
            riders: Optional list of riders; defaults to the global list.

        Returns:
            Multi-line string containing the map and legend.
        """
        if riders is None:
            riders = globalvar.get_value("riders", [])
        size = 2 * MAP_CELLS + 1
        canvas = [[" "] * size for _ in range(size)]

        for gx in range(MAP_NODES):
            for gy in range(MAP_NODES):
                r, c = cls._cell((gx * grid.CELL_SIZE, gy * grid.CELL_SIZE))
                canvas[r][c] = "."
                if gx < MAP_CELLS:
                    canvas[r][c + 1] = "-"
                if gy < MAP_CELLS:
                    canvas[r - 1][c] = "|"

        if route:
            for a, b in zip(route, route[1:]):
                r, c = cls._cell(a)
                rb, cb = cls._cell(b)
                canvas[r][c] = "*"
                while (r, c) != (rb, cb):
                    if r != rb:
                        r += 1 if rb > r else -1
                    if c != cb:
                        c += 1 if cb > c else -1
                    canvas[r][c] = "*"

        for rider in riders:
            for order in rider.orders:
                r, c = cls._cell(order.res_pos)
                canvas[r][c] = "M"
                r, c = cls._cell(order.cus_pos)
                canvas[r][c] = "C"
        for rider in riders:
            r, c = cls._cell(rider.position)
            canvas[r][c] = "R"

        lines = []
        for index, row in enumerate(canvas):
            if index % 2 == 0:
                y_value = grid.MAP_SIZE - (index // 2) * grid.CELL_SIZE
                label = f"{y_value:>4} "
            else:
                label = "     "
            lines.append(label + "".join(row))
        lines.append("     x: 0 (left) .. 800 (right); y: 0 (bottom) .. 800 (top)")
        lines.append("     Legend: R rider, M restaurant, C customer, "
                     ". intersection, - | street, * planned route")
        return "\n".join(lines)

    @staticmethod
    def render_status(funds, riders, time_now, issues=None):
        """Render a snapshot of system funds and every rider.

        Args:
            funds: Current fund balance.
            riders: List of Rider objects.
            time_now: Current simulated time.
            issues: Optional list of consistency problems.

        Returns:
            Multi-line status string.
        """
        lines = [
            f"Simulated time: {time_now} tick(s)",
            f"System funds: {funds}",
            f"Riders: {len(riders)}",
        ]
        for rider in riders:
            order_ids = [o.id for o in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id}: position ({rider.x}, {rider.y}), "
                f"status {STATE_NAMES.get(rider.state, 'unknown')}, "
                f"speed {rider.speed}, orders {len(order_ids)} {order_ids}"
            )
        if issues:
            lines.append("Consistency issues:")
            lines.extend(f"  ! {issue}" for issue in issues)
        return "\n".join(lines)

    @staticmethod
    def render_orders(orders, time_now):
        """Render a table of all orders.

        Args:
            orders: List of Order objects.
            time_now: Current simulated time (for unfinished orders).

        Returns:
            Multi-line table string.
        """
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} | {'Start':>5} | {'Restaurant':<12} | "
                  f"{'Customer':<12} | {'Dist':>5} | {'Taken':>10} | "
                  f"{'Rider':>5} | Status")
        lines = [header, "-" * len(header)]
        for order in orders:
            if order.time_taken() is not None:
                taken = f"{order.time_taken()} tick(s)"
            else:
                taken = f"({time_now - order.created_time} so far)"
            rider = f"#{order.rider_id}" if order.rider_id is not None else "-"
            lines.append(
                f"{order.id:>4} | {order.created_time:>5} | "
                f"{str(order.res_pos):<12} | {str(order.cus_pos):<12} | "
                f"{order.distance:>5} | {taken:>10} | {rider:>5} | {order.status}"
            )
        return "\n".join(lines)

    @staticmethod
    def render_rider_log(riders):
        """Render every status-change entry of all riders in time order.

        Args:
            riders: List of Rider objects.

        Returns:
            Multi-line log string.
        """
        entries = []
        for rider in riders:
            entries.extend(rider.log)
        if not entries:
            return "No rider events recorded."
        entries.sort(key=lambda e: (e["time"], e["rider_id"]))
        lines = []
        for entry in entries:
            order = (f"order #{entry['order_id']}"
                     if entry["order_id"] is not None else "no order")
            lines.append(
                f"t={entry['time']:>4} | Rider #{entry['rider_id']:<3} | "
                f"{entry['event']:<20} | {order:<10} | "
                f"at ({entry['x']}, {entry['y']}) | "
                f"{STATE_NAMES.get(entry['state'], 'unknown')}"
            )
        return "\n".join(lines)

"""Command parsing, validation, history and the interactive main loop."""

import logging
import re

import globalvar
import grid
from display import DisplayManager
from order import OrderManager
from path_planner import PathPlanner
from system import SystemManager

try:  # Enables arrow-key editing and history where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    readline = None

SHORTCUTS = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "w": "wait",
    "sv": "save",
    "hi": "history",
    "h": "help",
    "?": "help",
    "q": "quit",
    "x": "exit",
}

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy]  [s]  Create an order (restaurant then customer, 0-800)
  status               [st] Show funds, riders and their orders
  orders               [o]  Show all orders
  riders               [r]  Show every rider status change
  map                  [m]  Show the ASCII map
  wait [n]             [w]  Advance the simulation n extra ticks (default 1)
  save [path]          [sv] Save state to a JSON file
  history              [hi] Show command history
  !! / !n                   Repeat the last / n-th command
  help                 [h]  Show this help
  quit | exit          [q]  Leave the program
Every command line advances the simulation by one tick."""

logger = logging.getLogger("dispatch.input")


class InputError(Exception):
    """Raised for malformed user input."""


class InputManager:
    """Parse and execute user commands for the dispatch simulator."""

    def __init__(self, input_func=input, output_func=print):
        """Create the manager.

        Args:
            input_func: Callable used to read a line (default ``input``).
            output_func: Callable used to print text (default ``print``).
        """
        self.history = []
        self._input = input_func
        self._output = output_func
        self.commands = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "wait": self.cmd_wait,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    @staticmethod
    def parse_coordinates(tokens):
        """Parse four integer coordinates from tokens or a text string.

        Args:
            tokens: List of strings, or a single string. Separators may be
                spaces or commas.

        Returns:
            List of four validated integers.

        Raises:
            InputError: If the count, type or range is wrong.
        """
        text = " ".join(tokens) if isinstance(tokens, (list, tuple)) else tokens
        parts = [p for p in re.split(r"[\s,]+", text.strip()) if p]
        if len(parts) != 4:
            raise InputError(
                f"expected 4 coordinates [restaurant_x restaurant_y "
                f"customer_x customer_y], got {len(parts)}"
            )
        values = []
        for part in parts:
            try:
                value = int(part)
            except ValueError:
                raise InputError(f"'{part}' is not an integer")
            try:
                grid.validate_coordinate(value)
            except grid.CoordinateError as exc:
                raise InputError(str(exc))
            values.append(value)
        return values

    def _expand_history(self, line):
        """Expand '!!' and '!n' references to earlier commands."""
        if line == "!!":
            if not self.history:
                raise InputError("no previous command to repeat")
            return self.history[-1]
        match = re.fullmatch(r"!(\d+)", line)
        if match:
            index = int(match.group(1))
            if not 1 <= index <= len(self.history):
                raise InputError(f"no command number {index} in history")
            return self.history[index - 1]
        return line

    def execute(self, line):
        """Execute one command line and return its textual output.

        Args:
            line: Raw input line.

        Returns:
            Output text (possibly empty).
        """
        line = (line or "").strip()
        if not line:
            return ""
        try:
            line = self._expand_history(line)
        except InputError as exc:
            return f"Input error: {exc}"
        self.history.append(line)
        tokens = line.split()
        name = SHORTCUTS.get(tokens[0].lower(), tokens[0].lower())
        handler = self.commands.get(name)
        if handler is None:
            return f"Unknown command '{tokens[0]}'. Type 'help' for commands."
        try:
            return handler(tokens[1:])
        except EOFError:
            raise
        except InputError as exc:
            return f"Input error: {exc}"
        except LookupError as exc:
            return f"Cannot complete request: {exc}"
        except ValueError as exc:
            return f"Error: {exc}"
        except Exception as exc:  # error recovery: keep the session alive
            logger.exception("unexpected error executing '%s'", line)
            return f"Internal error ({exc}); system state preserved."

    def run(self):
        """Run the interactive main loop until the system stops.

        Each loop iteration reads one command, executes it, and then
        advances the simulation by one tick.
        """
        self._output("Food delivery dispatch simulator. Type 'help' for commands.")
        while SystemManager.is_running():
            try:
                line = self._input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self._output("")
                SystemManager.shutdown()
                break
            text = self.execute(line)
            if text:
                self._output(text)
            if not SystemManager.is_running():
                break
            SystemManager.tick()

    def cmd_start(self, args):
        """Create an order and assign it to the best rider."""
        if not args:
            args = self._input(
                "Enter restaurant_x restaurant_y customer_x customer_y: "
            ).split()
        coords = self.parse_coordinates(args)
        order = OrderManager.create_order(coords[:2], coords[2:])
        rider, new_rider, eta = OrderManager.assign_order(order)
        route = PathPlanner.plan_multi_route(
            grid.snap_to_grid(rider.x, rider.y),
            [order.res_pos, order.cus_pos],
        )
        recruited = (f"yes (rider #{rider.rider_id})" if new_rider else "no")
        lines = [
            f"Order #{order.id} created and assigned.",
            f"Restaurant: input ({coords[0]}, {coords[1]}) -> grid {order.res_pos}",
            f"Customer:   input ({coords[2]}, {coords[3]}) -> grid {order.cus_pos}",
            f"New rider added: {recruited}",
            f"Assigned rider: #{rider.rider_id} at ({rider.x}, {rider.y}), "
            f"speed {rider.speed} units/tick",
            f"Estimated delivery time: {eta} tick(s), distance "
            f"{order.distance} units",
            "Route map (rider -> restaurant -> customer):",
            DisplayManager.render_map(route=route, riders=[rider]),
        ]
        return "\n".join(lines)

    def cmd_status(self, args):
        """Show system funds, riders and consistency checks."""
        riders = globalvar.get_value("riders", [])
        return DisplayManager.render_status(
            SystemManager.get_funds(),
            riders,
            int(globalvar.get_value("time", 0)),
            SystemManager.check_consistency(),
        )

    def cmd_orders(self, args):
        """Show every order and its details."""
        return DisplayManager.render_orders(
            globalvar.get_value("orders", []),
            int(globalvar.get_value("time", 0)),
        )

    def cmd_riders(self, args):
        """Show every rider status-change record."""
        return DisplayManager.render_rider_log(globalvar.get_value("riders", []))

    def cmd_map(self, args):
        """Show the ASCII map with riders and active orders."""
        return DisplayManager.render_map()

    def cmd_wait(self, args):
        """Advance the simulation by extra ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise InputError("wait expects a whole number of ticks")
        if not 1 <= count <= 10000:
            raise InputError("wait count must be between 1 and 10000")
        for _ in range(count):
            SystemManager.tick()
            if not SystemManager.is_running():
                break
        return f"Advanced {count} tick(s); simulated time is now " \
               f"{globalvar.get_value('time', 0)}."

    def cmd_save(self, args):
        """Save the system state to a JSON file."""
        path = args[0] if args else "dispatch_state.json"
        try:
            SystemManager.save_state(path)
        except OSError as exc:
            return f"Could not save state to {path}: {exc}"
        return f"State saved to {path}."

    def cmd_history(self, args):
        """List previously entered commands."""
        if not self.history:
            return "History is empty."
        return "\n".join(f"{i:>4}  {cmd}" for i, cmd in enumerate(self.history, 1))

    def cmd_help(self, args):
        """Show the command reference."""
        return HELP_TEXT

    def cmd_quit(self, args):
        """Stop the simulation loop."""
        SystemManager.shutdown()
        return "Goodbye."

"""Entry point for the food delivery dispatch simulator.

Usage:
    python src/main.py [--config CONFIG_JSON] [--debug]
"""

import argparse
import logging
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import system  # noqa: E402
from input_manager import InputManager  # noqa: E402


def parse_args(argv=None):
    """Parse command-line arguments.

    Args:
        argv: Optional list of arguments (defaults to ``sys.argv[1:]``).

    Returns:
        The parsed ``argparse.Namespace``.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=None,
                        help="path to a JSON file with system parameters")
    parser.add_argument("--debug", action="store_true",
                        help="enable debug logging to stderr")
    return parser.parse_args(argv)


def main(argv=None):
    """Run the simulator.

    Args:
        argv: Optional list of command-line arguments.

    Returns:
        Process exit code.
    """
    args = parse_args(argv)
    logging.basicConfig(
        level=logging.DEBUG if args.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    system.init_system(args.config)
    InputManager().run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

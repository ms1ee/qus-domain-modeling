"""Grid-based path planning on the 800x800 street map.

The map is an 8x8 grid of 100x100 cells. Lines whose x or y coordinate is a
multiple of 100 are streets, and points where both coordinates are multiples
of 100 are intersections. Restaurants and customers are snapped to
intersections, so every route is a sequence of right-angle turns along
streets. Riders never block each other (they may share a street), which
means no path-conflict deadlock can arise from waiting.
"""

import math
from typing import List, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
MAX_PATH_STEPS = 10000

Point = Tuple[int, int]


def snap_coordinate(value: int) -> int:
    """Clamp a coordinate to the map and move it to its cell's top-left corner.

    Args:
        value: Raw coordinate value.

    Returns:
        Coordinate in the range 0-800 that is a multiple of 100.
    """
    clamped = min(max(int(value), 0), MAP_SIZE)
    return (clamped // CELL_SIZE) * CELL_SIZE


class PathPlanner:
    """Computes street-following waypoints between map points."""

    @staticmethod
    def snap_point(x: int, y: int) -> Point:
        """Snap a raw point to the top-left corner of its grid cell.

        Args:
            x: Raw x coordinate.
            y: Raw y coordinate.

        Returns:
            Tuple ``(x, y)`` of snapped coordinates.
        """
        return (snap_coordinate(x), snap_coordinate(y))

    @staticmethod
    def is_street(x: int, y: int) -> bool:
        """Return True if the point lies on a street (grid line).

        Args:
            x: x coordinate.
            y: y coordinate.
        """
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def is_intersection(x: int, y: int) -> bool:
        """Return True if the point is a street intersection.

        Args:
            x: x coordinate.
            y: y coordinate.
        """
        return x % CELL_SIZE == 0 and y % CELL_SIZE == 0

    @staticmethod
    def manhattan(a: Point, b: Point) -> int:
        """Return the Manhattan distance between two points.

        Args:
            a: First point.
            b: Second point.
        """
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def euclidean(a: Point, b: Point) -> float:
        """Return the Euclidean distance between two points.

        Args:
            a: First point.
            b: Second point.
        """
        return math.hypot(a[0] - b[0], a[1] - b[1])

    @classmethod
    def next_waypoint(cls, position: Point, target: Point) -> Tuple[Point, bool]:
        """Compute the next axis-aligned waypoint on the way to a target.

        Movement prefers streets: a rider on a horizontal street moves along
        it until it is directly above or below the target, then turns onto
        the vertical street. A rider on a vertical street moves along it
        until it reaches the target's row, then turns onto the horizontal
        street.

        Args:
            position: Current ``(x, y)`` position.
            target: Destination ``(x, y)`` point.

        Returns:
            A tuple ``(waypoint, special_case)``. ``special_case`` is True when
            the rider is directly above or below the target and moves straight
            toward it.
        """
        x, y = position
        tx, ty = target
        if (x, y) == (tx, ty):
            return (tx, ty), False
        if y % CELL_SIZE == 0:
            if x != tx:
                return (tx, y), False
            # Directly above/below the target: special straight-line movement.
            return (x, ty), True
        if y != ty:
            return (x, ty), x == tx
        return (tx, y), False

    @classmethod
    def trace(cls, start: Point, target: Point) -> List[Point]:
        """Return the list of corner points from ``start`` to ``target``.

        Args:
            start: Starting point.
            target: Destination point.

        Returns:
            List of points starting with ``start`` and ending with ``target``.
        """
        pos = (int(start[0]), int(start[1]))
        goal = (int(target[0]), int(target[1]))
        path = [pos]
        for _ in range(MAX_PATH_STEPS):
            if pos == goal:
                break
            waypoint, _ = cls.next_waypoint(pos, goal)
            if waypoint == pos:
                break
            path.append(waypoint)
            pos = waypoint
        return path

    @classmethod
    def route(cls, start: Point, targets: List[Point]) -> List[Point]:
        """Build a continuous route through an ordered list of targets.

        Args:
            start: Starting point.
            targets: Ordered destinations to visit.

        Returns:
            Combined list of route points (no duplicate joints).
        """
        points: List[Point] = [(int(start[0]), int(start[1]))]
        for target in targets:
            leg = cls.trace(points[-1], target)
            points.extend(leg[1:])
        return points

    @staticmethod
    def path_length(points: List[Point]) -> int:
        """Return the total street length of a list of route points.

        Args:
            points: Ordered route points.
        """
        return sum(
            PathPlanner.manhattan(a, b) for a, b in zip(points, points[1:])
        )

"""Order data model."""

from typing import Any, Dict, Optional, Sequence

import globalvar
from path_planner import PathPlanner

ORDER_PENDING = "pending"
ORDER_ASSIGNED = "assigned"
ORDER_PICKED_UP = "picked_up"
ORDER_COMPLETED = "completed"


def _next_order_id() -> int:
    """Return a new unique order ID and store the counter globally."""
    counter = int(globalvar.get_value("order_id_counter", 0) or 0) + 1
    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        res_pos: Restaurant grid point ``(x, y)`` (top-left of its cell).
        cus_pos: Customer grid point ``(x, y)`` (top-left of its cell).
        status: One of "pending", "assigned", "picked_up", "completed".
        rider_id: ID of the assigned rider, or None.
        start_time: Tick at which the order was created.
        assigned_time: Tick at which the order was assigned.
        pickup_time: Tick at which the rider collected the food.
        completion_time: Tick at which the order was delivered.
        distance: Street distance between restaurant and customer.
        time_taken: Ticks from creation to delivery, once completed.
        efficiency: Distance per tick (delivery efficiency), once completed.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order and snap its coordinates to grid points.

        Args:
            res_pos: Raw restaurant coordinates ``(x, y)``.
            cus_pos: Raw customer coordinates ``(x, y)``.
        """
        self.id = _next_order_id()
        self.res_pos = PathPlanner.snap_point(res_pos[0], res_pos[1])
        self.cus_pos = PathPlanner.snap_point(cus_pos[0], cus_pos[1])
        self.status = ORDER_PENDING
        self.rider_id: Optional[int] = None
        self.start_time = globalvar.current_tick()
        self.assigned_time: Optional[int] = None
        self.pickup_time: Optional[int] = None
        self.completion_time: Optional[int] = None
        self.distance = PathPlanner.manhattan(self.res_pos, self.cus_pos)
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to a rider.

        Args:
            rider_id: ID of the rider receiving the order.

        Raises:
            ValueError: If the order is not pending.
        """
        if self.status != ORDER_PENDING:
            raise ValueError(
                f"Order #{self.id} cannot be assigned from status '{self.status}'."
            )
        self.rider_id = rider_id
        self.status = ORDER_ASSIGNED
        self.assigned_time = globalvar.current_tick()

    def mark_picked_up(self, tick: Optional[int] = None) -> None:
        """Record that the rider collected the food at the restaurant.

        Args:
            tick: Tick of the pickup; defaults to the current tick.

        Raises:
            ValueError: If the order is not in the assigned state.
        """
        if self.status != ORDER_ASSIGNED:
            raise ValueError(
                f"Order #{self.id} cannot be picked up from status '{self.status}'."
            )
        self.status = ORDER_PICKED_UP
        self.pickup_time = globalvar.current_tick() if tick is None else tick

    def complete(self, tick: Optional[int] = None) -> None:
        """Mark the order as delivered and compute delivery statistics.

        Args:
            tick: Tick of the delivery; defaults to the current tick.

        Raises:
            ValueError: If the order has not been picked up.
        """
        if self.status != ORDER_PICKED_UP:
            raise ValueError(
                f"Order #{self.id} cannot be completed from status '{self.status}'."
            )
        end = globalvar.current_tick() if tick is None else tick
        self.status = ORDER_COMPLETED
        self.completion_time = end
        self.time_taken = end - self.start_time
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = None

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "assigned_time": self.assigned_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

"""Rider model: street-based movement, state transitions and event log.

Rider state enumeration:
    0 = idle, 1 = picking up (heading to a restaurant),
    2 = delivering (carrying food to a customer).

A rider serves its order list first-in first-out. The first active order
determines the current target: the restaurant while the order is "assigned",
the customer once it is "picked_up".
"""

import logging
from typing import Any, Dict, List, Optional

import globalvar
from order_model import ORDER_ASSIGNED, ORDER_COMPLETED, ORDER_PICKED_UP
from path_planner import Point, PathPlanner

logger = logging.getLogger(__name__)

DEFAULT_SPEED = 20
DEFAULT_START: Point = (400, 400)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_LABELS = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

EVENT_SPAWN = "spawn"
EVENT_ACCEPTED = "accepted_order"
EVENT_ARRIVED = "arrived_at_merchant"
EVENT_DELIVERED = "delivered_to_customer"

ACTIVE_STATUSES = (ORDER_ASSIGNED, ORDER_PICKED_UP)


def _sign(value: int) -> int:
    """Return -1, 0 or 1 according to the sign of ``value``."""
    return (value > 0) - (value < 0)


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates.
        speed: Movement budget per tick (units/tick).
        state: 0 idle, 1 picking up, 2 delivering.
        orders: All orders ever assigned, including completed ones.
        special_case: True while moving straight toward a target that is
            directly above or below the rider.
        log: List of status-change dictionaries with keys including
            "event" and "order_id".
    """

    def __init__(
        self,
        rider_id: int,
        x: int = DEFAULT_START[0],
        y: int = DEFAULT_START[1],
    ) -> None:
        """Create a rider and record its spawn event.

        Args:
            rider_id: Unique rider identifier.
            x: Initial x coordinate (defaults to 400).
            y: Initial y coordinate (defaults to 400).
        """
        self.rider_id = rider_id
        self.x = int(x)
        self.y = int(y)
        self.speed = DEFAULT_SPEED
        self.state = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.log: List[Dict[str, Any]] = []
        self._accepted_ids = set()
        self._record(EVENT_SPAWN, None)

    # ------------------------------------------------------------------
    # Queries
    # ------------------------------------------------------------------
    @property
    def position(self) -> Point:
        """Return the current ``(x, y)`` position."""
        return (self.x, self.y)

    def active_orders(self) -> List[Any]:
        """Return orders that are assigned or picked up but not completed."""
        return [o for o in self.orders if o.status in ACTIVE_STATUSES]

    def active_order_count(self) -> int:
        """Return the number of active (unfinished) orders."""
        return len(self.active_orders())

    def is_idle(self) -> bool:
        """Return True if the rider is available for new orders."""
        return self.state == STATE_IDLE

    def current_order(self) -> Optional[Any]:
        """Return the order the rider is currently serving, if any."""
        for order in self.orders:
            if order.status in ACTIVE_STATUSES:
                return order
        return None

    def route_targets(self) -> List[Point]:
        """Return the ordered list of points the rider must visit.

        Restaurants are included only for orders not yet picked up.
        """
        targets: List[Point] = []
        for order in self.active_orders():
            if order.status == ORDER_ASSIGNED:
                targets.append(order.res_pos)
            targets.append(order.cus_pos)
        return targets

    # ------------------------------------------------------------------
    # Commands
    # ------------------------------------------------------------------
    def on_order_assigned(self, order: Any) -> None:
        """Register a newly assigned order: log acceptance and update state.

        Args:
            order: The order that was appended to ``self.orders``.
        """
        self._sync_assignments()

    def move(self) -> None:
        """Advance the rider by one tick.

        Accepts newly assigned orders, travels up to ``speed`` units along
        streets, and handles arrival at restaurants (pickup) and customers
        (delivery) along the way. Every state change is logged.
        """
        self._sync_assignments()
        budget = self.speed
        while True:
            order = self.current_order()
            if order is None:
                break
            target = order.res_pos if order.status == ORDER_ASSIGNED else order.cus_pos
            if self.position == target:
                self._arrive(order)
                continue
            if budget <= 0:
                break
            waypoint, special = PathPlanner.next_waypoint(self.position, target)
            self.special_case = special
            if waypoint == self.position:
                break
            budget = self._travel(waypoint, budget)
        self.state = self._derive_state()

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _derive_state(self) -> int:
        """Compute the rider state from its active orders."""
        active = self.active_orders()
        if not active:
            return STATE_IDLE
        if active[0].status == ORDER_PICKED_UP:
            return STATE_DELIVERING
        return STATE_PICKING_UP

    def _sync_assignments(self) -> None:
        """Log an "accepted_order" event for each newly assigned order."""
        for order in self.orders:
            if order.status in (ORDER_ASSIGNED, ORDER_PICKED_UP, ORDER_COMPLETED):
                if order.id not in self._accepted_ids:
                    self._accepted_ids.add(order.id)
                    self.state = self._derive_state()
                    self._record(EVENT_ACCEPTED, order.id)

    def _arrive(self, order: Any) -> None:
        """Handle arrival at the restaurant or the customer location."""
        if order.status == ORDER_ASSIGNED:
            order.mark_picked_up(globalvar.current_tick())
            self.state = self._derive_state()
            self._record(EVENT_ARRIVED, order.id)
        else:
            order.complete(globalvar.current_tick())
            self.state = self._derive_state()
            self._record(EVENT_DELIVERED, order.id)

    def _travel(self, waypoint: Point, budget: int) -> int:
        """Move along a single axis toward ``waypoint`` using the budget.

        Returns:
            The remaining movement budget.
        """
        dx = waypoint[0] - self.x
        dy = waypoint[1] - self.y
        if dx and dy:
            raise ValueError("Waypoint must be axis-aligned with the rider.")
        distance = abs(dx) + abs(dy)
        step = min(distance, budget)
        if dx:
            self.x += _sign(dx) * step
        else:
            self.y += _sign(dy) * step
        return budget - step

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append a status-change entry to the rider log."""
        entry = {
            "time": globalvar.current_tick(),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.info(
            "tick=%s rider=%s event=%s order=%s pos=(%s,%s) state=%s",
            entry["time"],
            self.rider_id,
            event,
            order_id,
            self.x,
            self.y,
            self.state,
        )

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [o.id for o in self.orders],
        }

"""Order creation, rider assignment and order flow orchestration."""

import logging
import math
from typing import Any, Dict, List, Optional

import globalvar
from order_model import ORDER_PENDING, Order
from path_planner import MAP_SIZE, PathPlanner
from rider import Rider

logger = logging.getLogger(__name__)

COORD_MIN = 0
COORD_MAX = 1000


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the best rider for an order.

    The rider with the fewest active orders is preferred. Ties are broken by
    the Euclidean distance from the rider to the restaurant, then by rider ID.

    Args:
        order: The order needing a rider.

    Returns:
        The chosen Rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    best: Optional[Rider] = None
    best_key = None
    for rider in riders:
        key = (
            rider.active_order_count(),
            PathPlanner.euclidean(rider.position, order.res_pos),
            rider.rider_id,
        )
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates orders, assigns them to riders and reports the planned route.

    Attributes:
        system: The SystemManager used for rider recruitment.
    """

    def __init__(self, system: Any) -> None:
        """Bind the manager to a SystemManager.

        Args:
            system: SystemManager instance providing ``maybe_recruit``.
        """
        self.system = system

    @staticmethod
    def all_orders() -> List[Order]:
        """Return the list of every order created so far."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def pending_orders(self) -> List[Order]:
        """Return orders that have not yet been assigned to a rider."""
        return [o for o in self.all_orders() if o.status == ORDER_PENDING]

    @staticmethod
    def _validate(value: int, label: str) -> int:
        """Validate a raw coordinate against the accepted input range."""
        if not isinstance(value, int) or isinstance(value, bool):
            raise ValueError(f"{label} must be an integer.")
        if not COORD_MIN <= value <= COORD_MAX:
            raise ValueError(
                f"{label} = {value} is out of range {COORD_MIN}-{COORD_MAX}."
            )
        return value

    def create_order(self, res_x: int, res_y: int, cus_x: int, cus_y: int) -> Order:
        """Validate coordinates, create an order and register it.

        Args:
            res_x: Restaurant x coordinate.
            res_y: Restaurant y coordinate.
            cus_x: Customer x coordinate.
            cus_y: Customer y coordinate.

        Returns:
            The newly created pending Order.

        Raises:
            ValueError: If any coordinate is not an integer in range.
        """
        self._validate(res_x, "restaurant_x")
        self._validate(res_y, "restaurant_y")
        self._validate(cus_x, "customer_x")
        self._validate(cus_y, "customer_y")
        order = Order((res_x, res_y), (cus_x, cus_y))
        self.all_orders().append(order)
        logger.info("Created order %s res=%s cus=%s", order.id, order.res_pos, order.cus_pos)
        return order

    def assign(self, order: Order, rider: Rider) -> None:
        """Assign an order to a rider.

        Args:
            order: Pending order to assign.
            rider: Rider receiving the order.
        """
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        rider.on_order_assigned(order)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)

    def submit(self, res_x: int, res_y: int, cus_x: int, cus_y: int) -> Dict[str, Any]:
        """Create an order, possibly recruit a rider, and assign it.

        Args:
            res_x: Restaurant x coordinate.
            res_y: Restaurant y coordinate.
            cus_x: Customer x coordinate.
            cus_y: Customer y coordinate.

        Returns:
            Dictionary with keys: "order", "restaurant", "customer",
            "new_rider" (Rider or None), "rider" (Rider or None), "route"
            (list of points), "distance" (int) and "eta" (ticks or None).
        """
        order = self.create_order(res_x, res_y, cus_x, cus_y)
        new_rider = self.system.maybe_recruit(len(self.pending_orders()))
        rider = find_best_rider_for_order(order)
        result: Dict[str, Any] = {
            "order": order,
            "restaurant": order.res_pos,
            "customer": order.cus_pos,
            "new_rider": new_rider,
            "rider": None,
            "route": [],
            "distance": 0,
            "eta": None,
        }
        if rider is None:
            logger.warning("No rider available for order %s; left pending", order.id)
            return result
        self.assign(order, rider)
        route = PathPlanner.route(rider.position, rider.route_targets())
        distance = PathPlanner.path_length(route)
        eta = math.ceil(distance / rider.speed) if rider.speed > 0 else None
        result.update(rider=rider, route=route, distance=distance, eta=eta)
        return result

    @staticmethod
    def map_limit() -> int:
        """Return the maximum map coordinate."""
        return MAP_SIZE

        globalvar.set_value("funds", int(value))

    @property
    def riders(self) -> List[Rider]:
        """The dynamic list of riders."""
        return globalvar.get_value("riders")

    def add_funds(self, amount: int) -> int:
        """Add an amount to system funds and return the new balance."""
        self.funds = self.funds + int(amount)
        return self.funds

    def find_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with the given ID, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def remove_rider(self, rider_id: int) -> bool:
        """Remove a rider from the list.

        Returns:
            True if a rider was removed.
        """
        rider = self.find_rider(rider_id)
        if rider is None:
            return False
        self.riders.remove(rider)
        return True

    def idle_riders(self) -> List[Rider]:
        """Return riders whose state is idle."""
        return [r for r in self.riders if r.is_idle()]

    def create_rider(self) -> Rider:
        """Recruit a new rider at (400, 400), deducting the recruitment cost.

        Returns:
            The newly created Rider.
        """
        counter = int(globalvar.get_value("rider_id_counter", 0) or 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        self.funds = self.funds - RECRUIT_COST
        logger.info("Recruited rider %s; funds now %s", counter, self.funds)
        return rider

    def maybe_recruit(self, pending_count: int) -> Optional[Rider]:
        """Recruit a rider if funds allow and pending orders exceed idle riders.

        Args:
            pending_count: Number of orders waiting for a rider.

        Returns:
            The new Rider, or None if no recruitment happened.
        """
        if self.funds >= RECRUIT_MIN_FUNDS and pending_count > len(self.idle_riders()):
            return self.create_rider()
        return None

    # ------------------------------------------------------------------
    # Simulation
    # ------------------------------------------------------------------
    def tick(self) -> int:
        """Advance the simulation by one tick.

        Every rider moves once, then the global tick counter is incremented.

        Returns:
            The new tick value.
        """
        for rider in list(self.riders):
            rider.move()
        new_tick = globalvar.current_tick() + 1
        globalvar.set_value("tick", new_tick)
        return new_tick

    def is_running(self) -> bool:
        """Return the main-loop running flag."""
        return bool(globalvar.get_value("running", False))

    def stop(self) -> None:
        """Clear the running flag so the main loop exits."""
        globalvar.set_value("running", False)

    def check_consistency(self) -> List[str]:
        """Check internal state consistency and return a list of problems.

        Returns:
            List of human-readable issue descriptions (empty if consistent).
        """
        issues: List[str] = []
        if not isinstance(self.funds, int):
            issues.append("System funds are not an integer.")
        ids = [r.rider_id for r in self.riders]
        if len(set(ids)) != len(ids):
            issues.append("Duplicate rider IDs detected.")
        for rider in self.riders:
            active = rider.active_orders()
            if active and rider.state == 0:
                issues.append(f"Rider #{rider.rider_id} has active orders but is idle.")
            if not active and rider.state != 0:
                issues.append(f"Rider #{rider.rider_id} has no orders but state={rider.state}.")
            for order in active:
                if order.rider_id != rider.rider_id:
                    issues.append(
                        f"Order #{order.id} is listed for rider #{rider.rider_id} "
                        f"but assigned to rider #{order.rider_id}."
                    )
        return issues

    def snapshot(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the system state."""
        orders = globalvar.get_value("orders") or []
        return {
            "tick": globalvar.current_tick(),
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "parameters": {
                "initial_funds": INITIAL_FUNDS,
                "recruit_min_funds": RECRUIT_MIN_FUNDS,
                "recruit_cost": RECRUIT_COST,
                "map_size": 800,
                "cell_size": 100,
            },
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in orders],
        }

    def save_state(self, path: str) -> str:
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination file path.

        Returns:
            The path that was written.
        """
        data = self.snapshot()
        tmp_path = path + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        os.replace(tmp_path, path)
        logger.info("State saved to %s", path)
        return path

class DisplayManager:
    """Builds text output for status, orders, rider logs and the map.

    The ASCII map uses one character per 50 units, so the 800x800 area is
    drawn as 17x17 nodes (16 cells in each direction plus the boundary).
    Legend: R rider, S restaurant, C customer, # street, * route, . empty.

    Attributes:
        system: The SystemManager providing riders and funds.
    """

    MAP_STEP = 50
    MAP_CELLS = 17

    def __init__(self, system: Any) -> None:
        """Bind the display to a SystemManager.

        Args:
            system: SystemManager instance.
        """
        self.system = system

    @staticmethod
    def _all_orders() -> List[Any]:
        """Return all orders recorded in the global store."""
        return list(globalvar.get_value("orders") or [])

    def render_status(self) -> str:
        """Return a snapshot of funds, riders and their order lists."""
        riders = self.system.riders
        pending = [o for o in self._all_orders() if o.status == ORDER_PENDING]
        lines = [
            "=== System Status ===",
            f"Tick: {globalvar.current_tick()}",
            f"System funds: {self.system.funds}",
            f"Riders: {len(riders)}",
            f"Pending orders: {len(pending)}",
        ]
        if not riders:
            lines.append("No riders on duty.")
        for rider in riders:
            lines.append(
                f"Rider #{rider.rider_id}: location=({rider.x}, {rider.y}) "
                f"status={STATE_LABELS[rider.state]} "
                f"active_orders={rider.active_order_count()} "
                f"total_orders={len(rider.orders)}"
            )
            for order in rider.orders:
                lines.append(
                    f"    order #{order.id}: {order.status} "
                    f"restaurant={order.res_pos} customer={order.cus_pos}"
                )
        return "\n".join(lines)

    def render_orders(self) -> str:
        """Return a table of all orders with delivery details."""
        orders = self._all_orders()
        header = (
            f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
            f"{'Dist':>6} {'Status':>10} {'Taken':>6} {'Rider':>6} {'Eff':>7}"
        )
        if not orders:
            return "No orders yet."
        lines = ["=== Orders ===", header]
        for o in orders:
            taken = "-" if o.time_taken is None else str(o.time_taken)
            eff = "-" if o.efficiency is None else f"{o.efficiency:.2f}"
            rider = "-" if o.rider_id is None else f"#{o.rider_id}"
            lines.append(
                f"{o.id:>4} {o.start_time:>6} {str(o.res_pos):>12} {str(o.cus_pos):>12} "
                f"{o.distance:>6} {o.status:>10} {taken:>6} {rider:>6} {eff:>7}"
            )
        return "\n".join(lines)

    def render_rider_log(self) -> str:
        """Return every recorded status change of every rider."""
        lines = ["=== Rider Status Log ==="]
        any_entry = False
        for rider in self.system.riders:
            for entry in rider.log:
                any_entry = True
                order = "-" if entry["order_id"] is None else f"#{entry['order_id']}"
                lines.append(
                    f"t={entry['time']:>4} rider #{entry['rider_id']:<3} "
                    f"{entry['event']:<22} order={order:<5} "
                    f"at ({entry['x']}, {entry['y']}) "
                    f"state={STATE_LABELS[entry['state']]}"
                )
        if not any_entry:
            lines.append("No rider events recorded.")
        return "\n".join(lines)

    @classmethod
    def _axis_index(cls, value: float) -> int:
        """Map a coordinate to a column/row index of the ASCII map."""
        idx = (int(round(value)) + cls.MAP_STEP // 2) // cls.MAP_STEP
        return min(max(idx, 0), cls.MAP_CELLS - 1)

    @classmethod
    def _grid_index(cls, x: float, y: float) -> Tuple[int, int]:
        """Return ``(col, row)`` for a map coordinate; row 0 is the top (y=800)."""
        col = cls._axis_index(x)
        row = cls.MAP_CELLS - 1 - cls._axis_index(y)
        return col, row

    @classmethod
    def _walk(cls, a: Point, b: Point) -> List[Tuple[int, int]]:
        """Return grid cells visited when walking axis-aligned from a to b."""
        c, r = cls._grid_index(a[0], a[1])
        c1, r1 = cls._grid_index(b[0], b[1])
        cells = [(c, r)]
        while (c, r) != (c1, r1):
            if c != c1:
                c += 1 if c1 > c else -1
            else:
                r += 1 if r1 > r else -1
            cells.append((c, r))
        return cells

    def render_map(self, route: Optional[Sequence[Point]] = None) -> str:
        """Render the ASCII map with streets, riders, orders and an optional route.

        Args:
            route: Optional ordered route points to draw as a connected line.

        Returns:
            Multi-line string containing the map and a legend.
        """
        n = self.MAP_CELLS
        grid = [["." for _ in range(n)] for _ in range(n)]
        for row in range(n):
            y = (n - 1 - row) * self.MAP_STEP
            for col in range(n):
                x = col * self.MAP_STEP
                if PathPlanner.is_street(x, y):
                    grid[row][col] = "#"
        if route:
            for a, b in zip(route, route[1:]):
                for col, row in self._walk(a, b):
                    grid[row][col] = "*"
        for order in self._all_orders():
            if order.status == ORDER_COMPLETED:
                continue
            col, row = self._grid_index(*order.res_pos)
            grid[row][col] = "S"
            col, row = self._grid_index(*order.cus_pos)
            grid[row][col] = "C"
        for rider in self.system.riders:
            col, row = self._grid_index(rider.x, rider.y)
            grid[row][col] = "R"

        lines = ["Map (y increases upward, 1 char = 50 units):"]
        for row in range(n):
            y = (n - 1 - row) * self.MAP_STEP
            label = f"{y:>4} " if y % 200 == 0 else "     "
            lines.append(label + " ".join(grid[row]))
        lines.append("     x: 0 (left) to 800 (right)")
        lines.append("Legend: R rider  S restaurant  C customer  # street  * route  . empty")
        return "\n".join(lines)

    def render_start_result(self, result: dict) -> str:
        """Render the response to a ``start`` command.

        Args:
            result: Dictionary returned by ``OrderManager.submit``.

        Returns:
            Formatted multi-line text.
        """
        order = result["order"]
        lines = [
            f"Order #{order.id} created at tick {order.start_time}",
            f"Restaurant grid point: {order.res_pos}",
            f"Customer grid point: {order.cus_pos}",
            f"Street distance: {order.distance} units",
        ]
        rider = result["rider"]
        if rider is None:
            lines.append(
                f"No rider available (funds: {self.system.funds}). Order remains pending."
            )
            return "\n".join(lines)
        new_rider = result["new_rider"]
        if new_rider is not None:
            lines.append(f"New rider added: #{new_rider.rider_id} (funds: {self.system.funds})")
        else:
            lines.append("New rider added: no")
        lines.append(
            f"Assigned rider: #{rider.rider_id} at ({rider.x}, {rider.y}) "
            f"speed {rider.speed} units/tick, active orders {rider.active_order_count()}"
        )
        eta = result["eta"]
        lines.append(
            f"Route length: {result['distance']} units; estimated time: "
            f"{eta if eta is not None else '-'} ticks"
        )
        lines.append(self.render_map(route=result["route"]))
        return "\n".join(lines)

from typing import List, Optional, Tuple

try:  # Enables line editing and history on platforms that provide readline.

COORD_MAX = 1000
MAX_TICKS_PER_COMMAND = 10000
HISTORY_LIMIT = 200

COORD_LABELS = ("restaurant_x", "restaurant_y", "customer_x", "customer_y")

COMMAND_ALIASES = {
    "start": "start",
    "n": "start",
    "status": "status",
    "st": "status",
    "orders": "orders",
    "o": "orders",
    "riders": "riders",
    "r": "riders",
    "map": "map",

class InputError(ValueError):
    """Raised when user input cannot be parsed or fails validation."""


class InputManager:
    """Parses user input lines and validates their arguments.

    Attributes:
        history: Ordered list of accepted command lines (most recent last).
    """

    def __init__(self) -> None:
        """Create an empty command history."""
        self.history: List[str] = []

    def read_line(self, prompt: str) -> Optional[str]:
        """Read one line from standard input.

        Returns:
            The entered text, or None on end-of-file or Ctrl+C.
        """
        try:
            return input(prompt)
        except (EOFError, KeyboardInterrupt):
            print()
            return None

    @staticmethod
    def split_tokens(text: Optional[str]) -> List[str]:
        """Split a line into tokens, treating commas as separators."""
        if not text:
            return []
        return text.replace(",", " ").split()

    def _remember(self, text: str) -> None:
        """Append a command to history, keeping a bounded length."""
        self.history.append(text)
        if len(self.history) > HISTORY_LIMIT:
            del self.history[0]

    def parse_command(self, line: Optional[str]) -> Tuple[str, List[str]]:
        """Parse a command line into a canonical command name and arguments.

        Supports shortcut aliases and ``!!`` to repeat the last command.

        Args:
            line: Raw input line.

        Returns:
            Tuple ``(command, args)``; command is "" for an empty line.

        Raises:
            InputError: If the command is unknown or there is nothing to repeat.
        """
        text = (line or "").strip()
        if not text:
            return "", []
        if text == "!!":
            if not self.history:
                raise InputError("No previous command to repeat.")
            text = self.history[-1]
        tokens = self.split_tokens(text)
        name = tokens[0].lower()
        if name not in COMMAND_ALIASES:
            raise InputError(f"Unknown command '{name}'. Type 'help' for a list of commands.")
        self._remember(text)
        return COMMAND_ALIASES[name], tokens[1:]

    @staticmethod
    def parse_coordinates(tokens: List[str]) -> List[int]:
        """Validate four integer coordinates in the range 0-1000.

        Args:
            tokens: Exactly four tokens in the order restaurant_x, restaurant_y,
                customer_x, customer_y.

        Returns:
            List of four ints.

        Raises:
            InputError: On wrong count, non-integer or out-of-range values.
        """
        if len(tokens) != 4:
            raise InputError(
                "Expected 4 integer coordinates: restaurant_x restaurant_y "
                "customer_x customer_y."
            )
        values: List[int] = []
        for label, token in zip(COORD_LABELS, tokens):
            try:
                value = int(token)
            except ValueError:
                raise InputError(f"{label} must be an integer, got '{token}'.") from None
            if not COORD_MIN <= value <= COORD_MAX:
                raise InputError(
                    f"{label} = {value} is out of range {COORD_MIN}-{COORD_MAX}."
                )
            values.append(value)
        return values

    @staticmethod
    def parse_tick_count(tokens: List[str]) -> int:
        """Parse the optional tick count argument of the ``tick`` command.

        Returns:
            Number of ticks to run (default 1).

        Raises:
            InputError: If the argument is not a positive integer within limits.
        """
        if not tokens:
            return 1
        try:
            count = int(tokens[0])
        except ValueError:
            raise InputError(f"Tick count must be an integer, got '{tokens[0]}'.") from None
        if not 1 <= count <= MAX_TICKS_PER_COMMAND:
            raise InputError(f"Tick count must be between 1 and {MAX_TICKS_PER_COMMAND}.")
        return count

"""Command-line entry point for the Food Delivery Order Dispatch simulator.

Each successfully executed command advances simulated time by one tick,
except ``help``, ``history``, ``save``, ``tick`` (which advances by its own
count) and ``quit``.
"""

import logging
import sys
from typing import Any, Callable, Dict, List

import globalvar
from display import DisplayManager
from input_manager import InputError, InputManager
from order import OrderManager
from system import init_system

LOG_FILE = "dispatch.log"
DEFAULT_SAVE_FILE = "dispatch_state.json"

logger = logging.getLogger("dispatch")

BANNER = """==============================================
 Food Delivery Order Dispatch Simulator
 Type 'help' for commands, 'quit' to exit.
=============================================="""

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant then customer, 0-1000).
                       Prompts for the four values if omitted. Shortcut: n
  status (st)          Show system funds, riders and their order lists
  orders (o)           Show all orders with delivery details
  riders (r)           Show the status-change log of every rider
  map (m)              Show the ASCII map (R rider, S restaurant, C customer)
  tick [n] (t)         Advance simulated time by n ticks (default 1)
  save [file]          Save state as JSON (default dispatch_state.json)
  history              Show command history
  !!                   Repeat the last command
  help (h, ?)          Show this help
  quit (q, exit)       Exit the program
Each executed command (except help/history/save/tick/quit) advances time by one tick."""


class DispatchApp:
    """Interactive application wiring the managers together.

    Attributes:
        system: SystemManager holding funds, riders and simulated time.
        orders: OrderManager for order creation and assignment.
        display: DisplayManager for text and map output.
        inputs: InputManager for parsing and validation.
    """

    def __init__(self) -> None:
        """Initialise the global state and all managers."""
        self.system = init_system()
        self.orders = OrderManager(self.system)
        self.display = DisplayManager(self.system)
        self.inputs = InputManager()
        self._handlers: Dict[str, Callable[[List[str]], bool]] = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "save": self._cmd_save,
            "history": self._cmd_history,
            "help": self._cmd_help,
        }

    # ------------------------------------------------------------------
    # Command handlers: each returns True if the command advances time.
    # ------------------------------------------------------------------
    def _cmd_start(self, args: List[str]) -> bool:
        tokens = args
        if not tokens:
            raw = self.inputs.read_line(
                "Enter restaurant_x restaurant_y customer_x customer_y: "
            )
            tokens = self.inputs.split_tokens(raw)
        res_x, res_y, cus_x, cus_y = self.inputs.parse_coordinates(tokens)
        result = self.orders.submit(res_x, res_y, cus_x, cus_y)
        print(self.display.render_start_result(result))
        return True

    def _cmd_status(self, args: List[str]) -> bool:
        print(self.display.render_status())
        return True

    def _cmd_orders(self, args: List[str]) -> bool:
        print(self.display.render_orders())
        return True

    def _cmd_riders(self, args: List[str]) -> bool:
        print(self.display.render_rider_log())
        return True

    def _cmd_map(self, args: List[str]) -> bool:
        print(self.display.render_map())
        return True

    def _cmd_tick(self, args: List[str]) -> bool:
        count = self.inputs.parse_tick_count(args)
        for _ in range(count):
            self.system.tick()
        print(f"Advanced {count} tick(s). Current tick: {globalvar.current_tick()}")
        return False

    def _cmd_save(self, args: List[str]) -> bool:
        path = args[0] if args else DEFAULT_SAVE_FILE
        saved = self.system.save_state(path)
        print(f"State saved to {saved}")
        return False

    def _cmd_history(self, args: List[str]) -> bool:
        if not self.inputs.history:
            print("No command history.")
        for index, entry in enumerate(self.inputs.history, start=1):
            print(f"{index:>4}  {entry}")
        return False

    def _cmd_help(self, args: List[str]) -> bool:
        print(HELP_TEXT)
        return False

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> None:
        """Run the interactive loop until quit or end of input."""
        print(BANNER)
        while globalvar.get_value("running", True):
            line = self.inputs.read_line("dispatch> ")
            if line is None:
                print("Goodbye.")
                self.system.stop()
                break
            try:
                command, args = self.inputs.parse_command(line)
            except InputError as exc:
                print(f"Error: {exc}")
                continue
            if not command:
                continue
            if command == "quit":
                print("Goodbye.")
                self.system.stop()
                break
            handler = self._handlers[command]
            try:
                advance = handler(args)
            except InputError as exc:
                print(f"Error: {exc}")
                continue
            except Exception:  # Error recovery: keep the system state intact.
                logger.exception("Unexpected error while running '%s'", command)
                print("Unexpected internal error; state preserved. See dispatch.log.")
                continue
            if advance:
                self.system.tick()
            for issue in self.system.check_consistency():
                logger.warning(issue)
                print(f"Warning: {issue}")


def main(argv: Any = None) -> int:
    """Configure logging, then run the interactive dispatcher.

    Args:
        argv: Unused; reserved for future command-line options.

    Returns:
        Process exit code.
    """
    try:
        logging.basicConfig(
            filename=LOG_FILE,
            level=logging.INFO,
            format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        )
    except OSError:
        logging.basicConfig(level=logging.WARNING)
    try:
        app = DispatchApp()
    except Exception:
        logger.exception("Failed to initialise the system")
        print("Fatal: could not initialise the system.")
        return 1
    try:
        app.run()
    except KeyboardInterrupt:
        print("\nInterrupted. Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

from __future__ import annotations

import re
from dataclasses import dataclass, field

UNRANKED_LEVEL = 1000

_LEVEL_WORDS = {
    "grand": 0,
    "special": 0,
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}

_WORD_RE = re.compile(
    r"^\s*(grand|special|first|second|third|fourth|fifth|sixth|seventh|eighth|ninth|tenth)\b",
    re.IGNORECASE,
)
_ORDINAL_RE = re.compile(r"^\s*(\d+)(st|nd|rd|th)\b", re.IGNORECASE)


def normalize_text(value: str) -> str:
    """Normalize a name or department for case-insensitive comparison."""
    return " ".join(str(value).split()).casefold()


def prize_level(name: str) -> int:
    """Return the rank of a prize level; smaller numbers mean higher prizes.

    Recognises names starting with words such as "First", "Second", "Third"
    or ordinal numbers such as "1st". Unknown names are ranked last.
    """
    match = _WORD_RE.match(name or "")
    if match:
        return _LEVEL_WORDS[match.group(1).lower()]
    match = _ORDINAL_RE.match(name or "")
    if match:
        return int(match.group(1))
    return UNRANKED_LEVEL


@dataclass(frozen=True)
class User:
    """A lottery participant."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that can be drawn in a lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

_SEPARATOR_RE = re.compile(r"[,;]")


def _parse_line(line_no: int, raw: str) -> tuple[User | None, str | None]:
    """Parse a single 'Name,Department' line.

    Returns (user, None) for valid lines, (None, error) for invalid lines and
    (None, None) for blank lines which are silently skipped.
    """
    text = raw.strip()
    if not text:
        return None, None
    parts = [part.strip() for part in _SEPARATOR_RE.split(text)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None, (
            f"Line {line_no}: Format Error - expected 'Name,Department Group', "
            f"got {text!r}"
        )
    return User(name=parts[0], department=parts[1]), None


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in team building participant list."""
        users = [User(name=name, department=dept) for name, dept in DEFAULT_USERS_DATA]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load participants from a custom txt file."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file where each line is 'Name,Department Group'."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.readlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]

        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            user, error = _parse_line(line_no, raw.rstrip("\r\n"))
            if error is not None:
                errors.append(error)
            elif user is not None:
                users.append(user)
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users and duplicates (by name).

        The first occurrence of a name is kept as valid; later occurrences are
        returned as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group."""
        canonical: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            key = normalize_text(user.department)
            display = canonical.setdefault(key, user.department)
            counts[display] = counts.get(display, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.models import Prize, normalize_text

DEFAULT_PRIZES_DATA: list[tuple[str, int]] = [
    ("Third Prize - Movie Tickets", 5),
    ("Second Prize - Shopping Cards", 3),
    ("First Prize - 1 Day Annual Leave", 1),
]


def _clean_groups(groups: list[str] | None) -> list[str]:
    """Strip, deduplicate and drop empty avoidance group names."""
    result: list[str] = []
    seen: set[str] = set()
    for group in groups or []:
        text = str(group).strip()
        if not text:
            continue
        key = normalize_text(text)
        if key not in seen:
            seen.add(key)
            result.append(text)
    return result


class PrizeManager:
    """Maintains the ordered list of prizes for a lottery session."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES_DATA:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(f"Cannot add default prize '{name}': {exc}")
        return errors

    def _find_duplicate(self, name: str) -> bool:
        key = normalize_text(name)
        return any(normalize_text(p.name) == key for p in self._prizes)

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("prize quantity must be a positive integer")
        if self._find_duplicate(clean_name):
            raise ValueError(f"prize '{clean_name}' already exists")
        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=_clean_groups(avoid_groups),
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery order."""
        return list(self._prizes)

    def set_avoid_groups(self, index: int, groups: list[str]) -> bool:
        """Replace the avoidance groups of the prize at a 0-based index."""
        if not 0 <= index < len(self._prizes):
            return False
        self._prizes[index].avoid_groups = _clean_groups(groups)
        return True

    def set_associated_drawer(self, index: int, drawer: str | None) -> bool:
        """Set the associated lottery person of the prize at a 0-based index."""
        if not 0 <= index < len(self._prizes):
            return False
        self._prizes[index].associated_drawer = (drawer or "").strip() or None
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        ``order`` contains 1-based current positions in the desired new order.
        Returns False if ``order`` is not a permutation of all positions.
        """
        count = len(self._prizes)
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in positions]
        return True

"""Lottery rule engine: eligibility, avoidance, special rules and fallback."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User, normalize_text

DRAWER_OPTIONS: tuple[str, ...] = ("Chen Qixian", "Jin Duo", "Ma Jiaqi", "Li Bin")

_SPECIAL_DRAWER = "li bin"
_FALLBACK_DRAWERS = ("chen qixian", "jin duo")
_PROTECTED_DRAWER = "ma jiaqi"
_PROTECTED_GROUP_KEYWORD = "cheng ru"


class LotteryManager:
    """Draws winners for prizes while enforcing the lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self.users: list[User] = list(users)
        # Results of the most recent rounds of Chen Qixian / Jin Duo:
        # True if that round produced at least one Cheng Ru group winner.
        self.last_two_draw_results: list[bool] = []
        self._drawn: set[str] = set()
        self._rng = rng if rng is not None else random.Random()

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _name_key(name: str) -> str:
        return normalize_text(name)

    @staticmethod
    def is_cheng_ru_member(user: User) -> bool:
        """Return True if the user belongs to the Cheng Ru group."""
        return _PROTECTED_GROUP_KEYWORD in normalize_text(user.department)

    @staticmethod
    def _drawer_key(prize: Prize) -> str:
        return normalize_text(prize.associated_drawer or "")

    @staticmethod
    def _is_avoided(user: User, prize: Prize) -> bool:
        groups = {normalize_text(group) for group in prize.avoid_groups}
        return normalize_text(user.department) in groups

    def has_drawn(self, user: User) -> bool:
        """Return True if the user has already won a prize."""
        return self._name_key(user.name) in self._drawn

    def remaining_users(self) -> list[User]:
        """Return all users who have not won yet."""
        return [user for user in self.users if not self.has_drawn(user)]

    def is_protection_active(self) -> bool:
        """Return True if the Cheng Ru group protection must be applied."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    # ------------------------------------------------------------ rule engine
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the primary pool of users eligible for the prize.

        - Previous winners are always excluded.
        - Li Bin: avoidance groups are not applied.
        - Ma Jiaqi with protection active: only Cheng Ru group members.
        - Otherwise members of the prize's avoidance groups are excluded from
          the primary pool; they are drawn only after the other members
          (see ``_get_deferred_pool``).
        """
        available = self.remaining_users()
        drawer = self._drawer_key(prize)

        if drawer == _SPECIAL_DRAWER:
            return available

        if drawer == _PROTECTED_DRAWER and self.is_protection_active():
            forced = [user for user in available if self.is_cheng_ru_member(user)]
            if forced:
                return forced

        return [user for user in available if not self._is_avoided(user, prize)]

    def _get_deferred_pool(self, prize: Prize) -> list[User]:
        """Return avoidance-group members who may be drawn after the primary pool."""
        if self._drawer_key(prize) == _SPECIAL_DRAWER:
            return []
        return [user for user in self.remaining_users() if self._is_avoided(user, prize)]

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = self._get_deferred_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._drawn.add(self._name_key(winner.name))
        return winner

    def finish_round(self, prize: Prize, winners: list[User]) -> None:
        """Update fallback state after a prize round has finished."""
        drawer = self._drawer_key(prize)
        if drawer in _FALLBACK_DRAWERS:
            has_protected = any(self.is_cheng_ru_member(user) for user in winners)
            self.last_two_draw_results.append(has_protected)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _PROTECTED_DRAWER:
            self.last_two_draw_results = []

    def rule_description(self, prize: Prize) -> str:
        """Return a human-readable description of the rules for the prize."""
        drawer = (prize.associated_drawer or "").strip()
        drawer_key = normalize_text(drawer)
        parts: list[str] = []

        if drawer_key == _SPECIAL_DRAWER:
            parts.append("Only previous winners are excluded; avoidance groups do not apply.")
        else:
            parts.append("Previous winners are excluded.")
            if prize.avoid_groups:
                groups = ", ".join(prize.avoid_groups)
                parts.append(
                    f"Members of avoidance groups ({groups}) are drawn only after other members."
                )
            if drawer_key == _PROTECTED_DRAWER:
                if self.is_protection_active():
                    parts.append(
                        "Cheng Ru Group protection is ACTIVE: only Cheng Ru Group members are eligible."
                    )
                else:
                    parts.append(
                        "Cheng Ru Group protection applies if the last two rounds of "
                        "Chen Qixian and Jin Duo had no Cheng Ru Group winners."
                    )

        if drawer:
            parts.insert(0, f"Lottery person: {drawer}.")
        else:
            parts.insert(0, "Lottery person: anyone.")
        return " ".join(parts)

"""Winner recording, grouping, and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, prize_level

TIME_FORMAT = "%Y-%m-%d %H:%M:%S"


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    won_at: datetime
    sequence: int


class ResultManager:
    """Maintains the winner list in real time."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for a prize."""
        winner = Winner(
            user=user,
            prize_name=prize.name,
            won_at=datetime.now(),
            sequence=len(self._winners) + 1,
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    @staticmethod
    def _sorted_prizes(prizes: list[Prize]) -> list[Prize]:
        return sorted(prizes, key=lambda prize: prize_level(prize.name))

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low)."""
        result: dict[str, list[Winner]] = {}
        for prize in self._sorted_prizes(prizes):
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def _ordered_winners(self, prizes: list[Prize]) -> list[Winner]:
        ordered: list[Winner] = []
        for prize in self._sorted_prizes(prizes):
            ordered.extend(w for w in self._winners if w.prize_name == prize.name)
        return ordered

    def export_to_csv(self, prizes: list[Prize], directory: str | None = None) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        winners = self._ordered_winners(prizes)
        if not winners:
            return False, "No winners to export."

        file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(directory or os.getcwd(), file_name)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in winners:
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.won_at.strftime(TIME_FORMAT),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def ordinal(number: int) -> str:
    """Return an English ordinal string such as 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
        rng: random.Random | None = None,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self._rng = rng
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []

    # ------------------------------------------------------------ utilities
    def _ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    def _ask_positive_int(self, prompt: str) -> int:
        while True:
            raw = self._ask(prompt)
            try:
                value = int(raw)
                if value > 0:
                    return value
            except ValueError:
                pass
            self._print("Please enter a positive whole number.")

    def _group_names(self) -> list[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    def run(self) -> int:
        self._print("=" * 60)
        self._print("  Company Team Building Smart Lottery System")
        self._print("=" * 60)
        try:
            if not self._load_list_stage():
                self._print("Goodbye.")
                return 0
            self._main_menu()
        except (EOFError, KeyboardInterrupt):
            self._print("\nSession ended.")
        return 0

    def _load_list_stage(self) -> bool:
        while True:
            self._print("\n--- Load Participant List ---")
            self._print("1. Use built-in default team building list")
            self._print("2. Upload custom txt list (format: Name,Department Group)")
            self._print("0. Exit")
            choice = self._ask("Select: ")

            if choice == "0":
                return False
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
            elif choice == "2":
                path = self._ask("Enter path of txt file: ")
                users, errors = self.list_manager.load_custom_list(path)
            else:
                self._print("Invalid option.")
                continue

            accepted = self._verify_list(users, errors)
            if accepted is None:
                self._print("Upload terminated.")
                continue
            if not accepted:
                self._print("No valid participants were loaded. Please try again.")
                continue

            self._print(f"Loaded {len(self.users)} valid participants.")

            self._print("Associated lottery person updated.")
        else:
            self._print("Invalid option.")

    def _choose_drawer(self) -> str | None:
        self._print("Associated lottery person (optional):")
        for index, drawer in enumerate(DRAWER_OPTIONS, start=1):
            self._print(f"  {index}. {drawer}")
        self._print("  0. None (anyone)")
        raw = self._ask("Select number, or type a custom name (blank = none): ")
        if not raw or raw == "0":
            return None
        if raw.isdigit():
            index = int(raw)
            if 1 <= index <= len(DRAWER_OPTIONS):
                return DRAWER_OPTIONS[index - 1]
            self._print("Invalid number; no lottery person assigned.")
            return None
        return raw

    def _choose_groups(self) -> list[str]:
        groups = self._group_names()
        if not groups:
            self._print("No participant list loaded; no avoidance groups can be set.")
            return []
        self._print("Available department groups:")
        for index, group in enumerate(groups, start=1):
            self._print(f"  {index}. {group}")
        raw = self._ask("Avoidance groups (numbers or names, comma-separated; blank = none): ")
        return self._resolve_groups(raw, groups)

    def _resolve_groups(self, raw: str, groups: list[str]) -> list[str]:
        selected: list[str] = []
        for token in raw.split(","):
            text = token.strip()
            if not text:
                continue
            if text.isdigit() and 1 <= int(text) <= len(groups):
                match = groups[int(text) - 1]
            else:
                match = next(
                    (g for g in groups if normalize_text(g) == normalize_text(text)), None
                )
                if match is None:
                    self._print(f"Warning: unknown group '{text}' ignored.")
                    continue
            if match not in selected:
                selected.append(match)
        return selected

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._print("Current lottery sequence:")
        for index, prize in enumerate(prizes, start=1):
            drawer = prize.associated_drawer or "Anyone"
            avoid = ", ".join(prize.avoid_groups) or "None"
            self._print(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"lottery person: {drawer} | avoid groups: {avoid}"
            )

    def _reorder_flow(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._show_prizes()
        raw = self._ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(item) for item in raw.split()]
        except ValueError:
            self._print("Invalid input: use whole numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Lottery sequence updated.")
            self._show_prizes()
        else:
            self._print("Invalid order: each serial number must appear exactly once.")

    # ------------------------------------------------------------ lottery
    def _start_lottery(self) -> None:
        if not self.users:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure prizes first.")
            return

        self.result_manager = ResultManager()
        lottery = LotteryManager(self.users, rng=self._rng)
        total = len(prizes)
        self._print("\n##### Lottery started #####")
        for round_no, prize in enumerate(prizes, start=1):
            self._print(f"\n##### Round {round_no}/{total} #####")
            winners = self._run_round(lottery, prize)
            lottery.finish_round(prize, winners)
            self._show_round_results(winners)

        self._show_summary(prizes)
        self._export_prompt(prizes)

    def _run_round(self, lottery: LotteryManager, prize: Prize) -> list[User]:
        winners: list[User] = []
        self._print(
            f"Prize: {prize.name} | Quantity: {prize.quantity} | "
            f"Lottery person: {prize.associated_drawer or 'Anyone'}"
        )
        self._print(f"Rules: {lottery.rule_description(prize)}")

        for position in range(1, prize.quantity + 1):
            prompt = (
                f"Press Enter to draw the {ordinal(position)} winner "
                f"(or 'pause' / 'terminate'): "
            )
            if not self._control_gate(prompt):
                self._print("Round terminated.")
                break

            self._print(f"Currently drawing the {ordinal(position)} winner...")
            winner = lottery.draw_one(prize)
            if winner is None:
                self._print("No eligible participants remain for this prize. Round ends.")
                break

            self.result_manager.add_winner(winner, prize)
            winners.append(winner)
            self._print(
                f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}"
            )
        return winners

    def _control_gate(self, prompt: str) -> bool:
        """Handle lottery control commands. Returns True to draw, False to end the round."""
        while True:
            command = self._ask(prompt).lower()
            if command in ("", "continue"):
                return True
            if command == "terminate":
                return False
            if command == "pause":
                return self._wait_while_paused()
            self._print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    def _wait_while_paused(self) -> bool:
        self._print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = self._ask("[paused] > ").lower()
            if command == "continue":
                self._print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            self._print("Type 'continue' to resume or 'terminate' to end this round.")

    def _show_round_results(self, winners: list[User]) -> None:
        if not winners:
            self._print("No winners were drawn in this round.")
            return
        self._print("Round results (winning order):")
        for index, user in enumerate(winners, start=1):
            self._print(f"  {index}. {user.name} ({user.department})")

    def _show_summary(self, prizes: list[Prize]) -> None:
        self._print("\n##### Final Winner List #####")
        grouped = self.result_manager.get_winners_by_prize(prizes)
        for prize_name, winners in grouped.items():
            self._print(f"[{prize_name}]")
            if not winners:
                self._print("  (no winners)")
                continue
            for index, winner in enumerate(winners, start=1):
                self._print(
                    f"  {index}. {winner.user.name} | {winner.user.department} | {winner.prize_name}"
                )

    def _export_prompt(self, prizes: list[Prize]) -> None:
        answer = self._ask("Export winner list to file? [y/n]: ").lower()
        if answer in ("y", "yes"):
            _, message = self.result_manager.export_to_csv(prizes)
            self._print(message)


def main() -> int:
    """Application entry point."""
    return LotterySystemCLI().run()

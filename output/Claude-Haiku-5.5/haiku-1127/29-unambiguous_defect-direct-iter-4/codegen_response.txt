"""Grid-based path planning for the food delivery simulator.

Coordinate system:
    * Map is 800x800 units divided into an 8x8 grid of 100x100 cells.
    * Streets lie on grid lines: a point is on a street when its x or y
      coordinate is a multiple of 100.
    * Intersections are points where both coordinates are multiples of 100
      (a 9x9 lattice of 81 nodes).
    * Order locations are converted to cell centers:
      ``center = (cell_index * 100 + 50)``.

Routes travel along streets between intersections (breadth-first search on
the lattice, which always prefers right-angle, street-aligned moves). The
final approach from a cell corner into a cell center is a single right-angle
turn.
"""

import logging
from collections import deque

logger = logging.getLogger(__name__)

MAP_SIZE = 800
CELL_SIZE = 100
GRID_SIZE = 8
CELL_HALF = CELL_SIZE // 2


def to_point(value):
    """Convert a 2-item sequence into an ``(int, int)`` tuple.

    Args:
        value: Sequence containing x and y.

    Returns:
        Tuple ``(x, y)`` of integers.
    """
    x, y = value
    return (int(x), int(y))


def clamp_coordinate(value):
    """Clamp a numeric coordinate into the map range ``[0, MAP_SIZE]``.

    Args:
        value: Number to clamp.

    Returns:
        Integer coordinate within the map bounds.
    """
    number = int(round(float(value)))
    return max(0, min(MAP_SIZE, number))


def cell_index(coordinate):
    """Return the grid cell index (0-7) containing ``coordinate``.

    Args:
        coordinate: Map coordinate in units.

    Returns:
        Cell index clamped to ``[0, GRID_SIZE - 1]``.
    """
    return min(max(int(coordinate) // CELL_SIZE, 0), GRID_SIZE - 1)


def to_cell_center(x, y):
    """Map an input coordinate pair onto the center of its grid cell.

    Implements ``center_x = (grid_x // 100) * 100 + 50``.

    Args:
        x: Input x coordinate.
        y: Input y coordinate.

    Returns:
        Tuple ``(center_x, center_y)``.
    """
    return (cell_index(x) * CELL_SIZE + CELL_HALF,
            cell_index(y) * CELL_SIZE + CELL_HALF)


def cell_corner(x, y):
    """Return the lower-left street intersection of the cell containing (x, y).

    Args:
        x: Map x coordinate.
        y: Map y coordinate.

    Returns:
        Tuple ``(corner_x, corner_y)`` of an intersection.
    """
    return (cell_index(x) * CELL_SIZE, cell_index(y) * CELL_SIZE)


def is_street(x, y):
    """Return True if the point lies on a street (grid line)."""
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


def is_intersection(x, y):
    """Return True if the point is a street intersection."""
    return x % CELL_SIZE == 0 and y % CELL_SIZE == 0


def manhattan(a, b):
    """Return the Manhattan distance between two points.

    Args:
        a: First point ``(x, y)``.
        b: Second point ``(x, y)``.

    Returns:
        Integer Manhattan distance.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def all_intersections():
    """Return all 81 street intersections of the map.

    Returns:
        List of ``(x, y)`` tuples in row-major order.
    """
    steps = range(GRID_SIZE + 1)
    return [(i * CELL_SIZE, j * CELL_SIZE) for i in steps for j in steps]


def l_path(src, dst):
    """Return right-angle waypoints from ``src`` to ``dst`` (horizontal first).

    The returned list excludes ``src`` and ends at ``dst``.

    Args:
        src: Start point.
        dst: Destination point.

    Returns:
        List of waypoints (may be empty when src == dst).
    """
    points = []
    if src[0] != dst[0]:
        points.append((dst[0], src[1]))
    if src[1] != dst[1]:
        points.append((dst[0], dst[1]))
    return points


def route_length(start, waypoints):
    """Return the total Manhattan length of a waypoint route.

    Args:
        start: Starting point.
        waypoints: Ordered list of points.

    Returns:
        Total length in map units.
    """
    total = 0
    previous = start
    for point in waypoints:
        total += manhattan(previous, point)
        previous = point
    return total


def _dedupe(points):
    """Remove consecutive duplicate points from a waypoint list."""
    result = []
    for point in points:
        if not result or result[-1] != point:
            result.append(point)
    return result


def _neighbors(node):
    """Yield the street-adjacent intersections of ``node``."""
    x, y = node
    for dx, dy in ((CELL_SIZE, 0), (-CELL_SIZE, 0), (0, CELL_SIZE), (0, -CELL_SIZE)):
        nx, ny = x + dx, y + dy
        if 0 <= nx <= MAP_SIZE and 0 <= ny <= MAP_SIZE:
            yield (nx, ny)


class PathPlanner:
    """Compute street-first routes between points on the 8x8 grid.

    The planner handles three kinds of starting positions:

    * On an intersection: routing begins from the node itself.
    * On a street segment: the rider first walks to one of the two bounding
      intersections (whichever yields the shorter total route).
    * Off the street (e.g. a cell center): the rider first leaves to the cell
      corner using a right-angle move.

    Destinations are reached through the cell corner of their cell and a final
    right-angle approach, so cell centers are entered from the street.
    """

    def plan(self, start, goal):
        """Return the ordered waypoints from ``start`` to ``goal``.

        Args:
            start: Current position ``(x, y)``.
            goal: Destination ``(x, y)``.

        Returns:
            List of waypoints ending at ``goal``. Empty if already there.
        """
        start = to_point(start)
        goal = to_point(goal)
        if start == goal:
            return []

        access = cell_corner(goal[0], goal[1])
        approach = l_path(access, goal)

        best = None
        for entry, prelude in self._entry_candidates(start):
            nodes = self._bfs(entry, access)
            if nodes is None:
                continue
            cost = route_length(start, prelude) + (len(nodes) - 1) * CELL_SIZE
            if best is None or cost < best[0]:
                best = (cost, prelude, nodes)

        if best is None:
            logger.warning("No street route from %s to %s; using direct path",
                           start, goal)
            return _dedupe(l_path(start, goal))

        _, prelude, nodes = best
        return _dedupe(prelude + nodes[1:] + approach)

    def _entry_candidates(self, start):
        """Return candidate ``(entry_node, prelude_points)`` pairs for a start."""
        x, y = start
        if is_intersection(x, y):
            return [(start, [])]
        if is_street(x, y):
            if x % CELL_SIZE == 0:
                low = (x, (y // CELL_SIZE) * CELL_SIZE)
                high = (x, low[1] + CELL_SIZE)
            else:
                low = ((x // CELL_SIZE) * CELL_SIZE, y)
                high = (low[0] + CELL_SIZE, y)
            candidates = []
            for node in (low, high):
                if 0 <= node[0] <= MAP_SIZE and 0 <= node[1] <= MAP_SIZE:
                    candidates.append((node, [node]))
            return candidates
        corner = cell_corner(x, y)
        return [(corner, l_path(start, corner))]

    @staticmethod
    def _bfs(source, target):
        """Breadth-first search over the intersection lattice.

        Returns:
            List of intersections from ``source`` to ``target`` inclusive,
            or None if unreachable.
        """
        if source == target:
            return [source]
        parent = {source: None}
        queue = deque([source])
        while queue:
            node = queue.popleft()
            for neighbor in _neighbors(node):
                if neighbor in parent:
                    continue
                parent[neighbor] = node
                if neighbor == target:
                    path = []
                    current = target
                    while current is not None:
                        path.append(current)
                        current = parent[current]
                    path.reverse()
                    return path
                queue.append(neighbor)
        return None

import itertools
import logging

import globalvar
from path_planner import to_cell_center

logger = logging.getLogger(__name__)

DELIVERY_FEE = 300
_ORDER_IDS = itertools.count(1)


def _current_tick():
    """Return the current simulated tick from the global store."""
    return globalvar.get_value("tick") or 0


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        res_pos: Restaurant location as grid cell center ``(x, y)``.
        cus_pos: Customer location as grid cell center ``(x, y)``.
        status: One of ``"pending"``, ``"assigned"``, ``"picking_up"``,
            ``"delivering"``, ``"completed"``.
        rider_id: Id of the assigned rider or None.
        start_time: Tick when the order was created.
        pickup_time: Tick when the rider reached the restaurant.
        finish_time: Tick when the order was delivered.
        distance: Distance travelled by the rider while carrying the order.
    """

    def __init__(self, res_pos, cus_pos):
        """Create an order and convert its coordinates to cell centers.

        Args:
            res_pos: Restaurant coordinates ``(x, y)`` in 0-800.
            cus_pos: Customer coordinates ``(x, y)`` in 0-800.
        """
        self.id = next(_ORDER_IDS)
        self.res_pos = to_cell_center(int(res_pos[0]), int(res_pos[1]))
        self.cus_pos = to_cell_center(int(cus_pos[0]), int(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = _current_tick()
        self.pickup_time = None
        self.finish_time = None
        self.distance = 0

    def assign_to(self, rider_id):
        """Record the rider this order was assigned to.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def mark_completed(self, finish_tick):
        """Mark the order as delivered and record completion data.

        Args:
            finish_tick: Simulated tick of delivery.

        Returns:
            Delivery fee earned for the order.
        """
        self.status = "completed"
        self.finish_time = finish_tick
        logger.info("Order %s completed at tick %s", self.id, finish_tick)
        return DELIVERY_FEE

    @property
    def time_taken(self):
        """Ticks elapsed from creation to delivery, or None if unfinished."""
        if self.finish_time is None:
            return None
        return self.finish_time - self.start_time

    def efficiency(self):
        """Return distance units covered per tick, or None if unavailable."""
        taken = self.time_taken
        if not taken:
            return None
        return round(self.distance / taken, 2)

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "finish_time": self.finish_time,
            "distance": self.distance,
        }

"""Rider model: state machine, movement along streets and event logging."""

import logging

import globalvar
from path_planner import PathPlanner, all_intersections

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

BASE_SPEED = 50
SPEED_STEP = 10
SPEED_VARIANTS = 4


def _current_tick():
    """Return the current simulated tick from the global store."""
    return globalvar.get_value("tick") or 0


def _spawn_point(rider_id):
    """Pick a deterministic starting intersection (on a road) for a rider."""
    nodes = all_intersections()
    return nodes[(rider_id * 17) % len(nodes)]


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current position in map units.
        speed: Movement budget in units per tick.
        state: 0=idle, 1=picking up, 2=delivering.
        orders: Assigned orders; the first one is currently being served.
        special_case: True while the rider is yielding or in a special state.
        target: Current destination point or None.
        waypoints: Remaining route points toward ``target``.
        distance_travelled: Total distance moved.
        wait_ticks: Number of ticks spent yielding to other riders.
        log: List of dict entries with keys ``event`` and ``order_id``.
    """

    def __init__(self, rider_id):
        """Create a rider on a road with a reasonable speed, in idle state.

        Args:
            rider_id: Unique integer identifier.
        """
        self.rider_id = int(rider_id)
        self.x, self.y = _spawn_point(self.rider_id)
        self.speed = BASE_SPEED + SPEED_STEP * (self.rider_id % SPEED_VARIANTS)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.target = None
        self.waypoints = []
        self.distance_travelled = 0
        self.wait_ticks = 0
        self.log = []
        self._record("spawn", None)
        logger.info("Rider %s spawned at (%s, %s)", self.rider_id, self.x, self.y)

    @property
    def position(self):
        """Current position as an ``(x, y)`` tuple."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human-readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    def is_available(self):
        """Return True if the rider is idle and has no assigned orders."""
        return self.state == IDLE and not self.orders

    def set_route(self, destination):
        """Plan a street-first route from the current position to ``destination``.

        Args:
            destination: Target ``(x, y)`` point.
        """
        self.target = (int(destination[0]), int(destination[1]))
        self.waypoints = PathPlanner().plan(self.position, self.target)

    def move(self):
        """Advance the rider by one tick.

        Performs state transitions:
            * Idle -> Picking up when an order is assigned (``accepted_order``).
            * Picking up -> Delivering on reaching the restaurant
              (``arrived_at_merchant``).
            * Delivering -> Idle on reaching the customer
              (``delivered_to_customer``).
        """
        if self.state != IDLE and not self.orders:
            logger.warning("Rider %s had no orders while in state %s; resetting",
                           self.rider_id, self.state)
            self._reset_to_idle()

        if self.state == IDLE:
            if not self.orders:
                self.special_case = False
                return
            order = self.orders[0]
            order.status = "picking_up"
            self.state = PICKING_UP
            self.set_route(order.res_pos)
            self._record("accepted_order", order.id)

        if self.state in (PICKING_UP, DELIVERING):
            self._advance()

        if self.state == PICKING_UP and self.position == self.target:
            order = self.orders[0]
            order.pickup_time = _current_tick()
            order.status = "delivering"
            self.state = DELIVERING
            self._record("arrived_at_merchant", order.id)
            self.set_route(order.cus_pos)
        elif self.state == DELIVERING and self.position == self.target:
            self._complete_current_order()

    def _advance(self):
        """Spend the speed budget moving along the waypoint list."""
        budget = self.speed
        moved = 0
        while budget > 0 and self.waypoints:
            nxt = self.waypoints[0]
            if self.position == nxt:
                self.waypoints.pop(0)
                continue
            if self._is_blocked(nxt):
                self.special_case = True
                self.wait_ticks += 1
                break
            self.special_case = False
            dx = nxt[0] - self.x
            dy = nxt[1] - self.y
            if dx:
                step = min(budget, abs(dx))
                self.x += step if dx > 0 else -step
            else:
                step = min(budget, abs(dy))
                self.y += step if dy > 0 else -step
            budget -= step
            moved += step
        if moved:
            self.distance_travelled += moved
            for order in self.orders:
                order.distance += moved

    def _is_blocked(self, point):
        """Check whether a higher-priority moving rider occupies ``point``.

        Priority is given by rider id (lower id goes first). Only moving riders
        block others, so a chain of waits always ends and deadlock cannot occur.
        """
        for other in globalvar.get_value("riders") or []:
            if other is self or other.rider_id > self.rider_id:
                continue
            if other.state == IDLE:
                continue
            if other.position == point:
                return True
        return False

    def _complete_current_order(self):
        """Finish the order being delivered and release the rider."""
        order = self.orders.pop(0)
        fee = order.mark_completed(_current_tick())
        funds = globalvar.get_value("funds") or 0
        globalvar.set_value("funds", funds + fee)
        self._reset_to_idle()
        self._record("delivered_to_customer", order.id)

    def _reset_to_idle(self):
        """Return the rider to the idle state and clear the route."""
        self.state = IDLE
        self.target = None
        self.waypoints = []
        self.special_case = False

    def _record(self, event, order_id):
        """Append a status-change entry to the rider's log."""
        self.log.append({
            "time": _current_tick(),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def to_dict(self):
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
        }

"""Order creation, rider assignment and pending-order management."""

import logging

import globalvar
from order_model import Order
from path_planner import manhattan

logger = logging.getLogger(__name__)

LOAD_PENALTY = 300


def find_best_rider_for_order(order):
    """Select the best rider for an order.

    The score is the Manhattan distance from the rider to the restaurant plus
    a load penalty per already assigned order, so nearby, lightly loaded riders
    win (distance-first with load balancing). Ties go to the lower list index.

    Args:
        order: The :class:`order_model.Order` to assign.

    Returns:
        The selected :class:`rider.Rider`, or None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    best = None
    best_score = None
    for rider in riders:
        score = manhattan(rider.position, order.res_pos) + LOAD_PENALTY * len(rider.orders)
        if best is None or score < best_score:
            best = rider
            best_score = score
    return best


class OrderManager:
    """Creates orders, dispatches them to riders and tracks their lifecycle."""

    def __init__(self):
        """Ensure the global order registries exist."""
        if globalvar.get_value("orders") is None:
            globalvar.set_value("orders", [])
        if globalvar.get_value("pending_orders") is None:
            globalvar.set_value("pending_orders", [])

    def create_order(self, res_x, res_y, cus_x, cus_y):
        """Validate coordinates, create an order and register it.

        Args:
            res_x: Restaurant x (0-800).
            res_y: Restaurant y (0-800).
            cus_x: Customer x (0-800).
            cus_y: Customer y (0-800).

        Returns:
            The created :class:`order_model.Order`.

        Raises:
            ValueError: If any coordinate is outside 0-800.
        """
        for name, value in (("restaurant x", res_x), ("restaurant y", res_y),
                            ("customer x", cus_x), ("customer y", cus_y)):
            if not 0 <= int(value) <= 800:
                raise ValueError(f"{name} must be between 0 and 800, got {value}")
        order = Order((res_x, res_y), (cus_x, cus_y))
        globalvar.get_value("orders").append(order)
        logger.info("Order %s created", order.id)
        return order

    def _recruit_if_needed(self):
        """Recruit a rider when none is available and funds allow.

        Returns:
            True if a new rider was recruited.
        """
        riders = globalvar.get_value("riders") or []
        if any(rider.is_available() for rider in riders):
            return False
        system = globalvar.get_value("system")
        if system is None:
            return False
        return system.recruit_rider() is not None

    def dispatch(self, order):
        """Assign an order to the best available rider.

        Args:
            order: Order to dispatch.

        Returns:
            Tuple ``(rider, new_rider_added)``. ``rider`` is None when the order
            stays pending because no rider could be assigned.
        """
        new_rider = self._recruit_if_needed()
        rider = find_best_rider_for_order(order)
        pending = globalvar.get_value("pending_orders")
        if rider is None:
            order.status = "pending"
            if order not in pending:
                pending.append(order)
            logger.warning("Order %s is pending: no rider available", order.id)
            return None, new_rider
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        if order in pending:
            pending.remove(order)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider, new_rider

    def retry_pending(self):
        """Try to dispatch all pending orders again.

        Returns:
            List of orders that were successfully assigned in this call.
        """
        assigned = []
        for order in list(globalvar.get_value("pending_orders") or []):
            rider, _ = self.dispatch(order)
            if rider is not None:
                assigned.append(order)
        return assigned

    @staticmethod
    def all_orders():
        """Return all orders created in this session."""
        return list(globalvar.get_value("orders") or [])

    @staticmethod
    def pending_orders():
        """Return orders waiting for a rider."""
        return list(globalvar.get_value("pending_orders") or [])

    def completed_orders(self):
        """Return all completed orders."""
        return [order for order in self.all_orders() if order.status == "completed"]

class SystemManager:

            The new fund balance.
        """
        new_value = self.funds + int(amount)
        globalvar.set_value("funds", new_value)
        return new_value

    def get_rider(self, rider_id):
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def recruit_rider(self):
        """Create a new rider if funds and capacity allow.

        Returns:
            The new :class:`rider.Rider`, or None if recruitment is not possible.
        """
        if len(self.riders) >= MAX_RIDERS:
            logger.info("Recruitment skipped: maximum of %s riders reached", MAX_RIDERS)
            return None
        if self.funds < RECRUIT_COST:
            logger.info("Recruitment skipped: insufficient funds (%s)", self.funds)
            return None
        new_id = (globalvar.get_value("rider_id_counter") or 0) + 1
        globalvar.set_value("rider_id_counter", new_id)
        self.add_funds(-RECRUIT_COST)
        rider = Rider(new_id)
        self.riders.append(rider)
        logger.info("Recruited rider %s; funds now %s", new_id, self.funds)
        return rider

    def remove_rider(self, rider_id):
        """Remove an available (idle, no orders) rider.

        Returns:
            True if the rider was removed.
        """
        rider = self.get_rider(rider_id)
        if rider is None or not rider.is_available():
            return False
        self.riders.remove(rider)
        return True

    def tick(self):
        """Advance the simulation by one tick.

        Charges operating costs for busy riders and moves every rider. Errors in
        one rider are logged and do not stop the others (error recovery).

        Returns:
            The new tick value.
        """
        globalvar.set_value("tick", self.tick_count + 1)
        busy = sum(1 for rider in self.riders if rider.state != IDLE)
        if busy:
            self.add_funds(-OPERATING_COST_PER_BUSY_TICK * busy)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                logger.exception("Rider %s failed during tick; state reset",
                                 rider.rider_id)
                rider.state = IDLE
                rider.waypoints = []
                rider.target = None
        return self.tick_count

    def check_consistency(self):
        """Verify and repair internal state.

        Removes completed orders still referenced by riders and resets riders
        that are busy without any orders.

        Returns:
            List of human-readable messages describing repairs made.
        """
        issues = []
        for rider in self.riders:
            completed = [o for o in rider.orders if o.status == "completed"]
            if completed:
                rider.orders = [o for o in rider.orders if o.status != "completed"]
                issues.append(f"Removed {len(completed)} completed order(s) from rider {rider.rider_id}")
            if rider.state != IDLE and not rider.orders:
                rider.state = IDLE
                rider.waypoints = []
                rider.target = None
                issues.append(f"Reset rider {rider.rider_id} to idle (no orders)")
        return issues

import logging

from path_planner import CELL_SIZE, GRID_SIZE, MAP_SIZE, PathPlanner, route_length
from rider import DELIVERING, IDLE, PICKING_UP

logger = logging.getLogger(__name__)

UNIT = CELL_SIZE // 2
CANVAS_SIZE = GRID_SIZE * 2 + 1
SYMBOL_RIDER = "R"
SYMBOL_RESTAURANT = "M"
SYMBOL_CUSTOMER = "C"
SYMBOL_ROUTE = "*"
SYMBOL_MARKERS = (SYMBOL_RIDER, SYMBOL_RESTAURANT, SYMBOL_CUSTOMER)
LEGEND = ("Legend: R=rider  M=restaurant  C=customer  "
          "-,|,+=streets  .=cell center  *=route  (x,y: 0-800, y grows downward)")


def point_to_canvas(x, y):
    """Convert a map point into ``(row, col)`` on the 17x17 canvas.

    Each canvas cell represents 50x50 map units, so intersections sit on even
    rows/columns and cell centers on odd rows/columns.
    """
    col = min(max(int(x) // UNIT, 0), CANVAS_SIZE - 1)
    row = min(max(int(y) // UNIT, 0), CANVAS_SIZE - 1)
    return row, col


def _line_cells(a, b):
    """Return canvas cells on a straight (or near-straight) line from a to b."""
    (r0, c0), (r1, c1) = a, b
    steps = max(abs(r1 - r0), abs(c1 - c0))
    cells = []
    for k in range(steps + 1):
        if steps:
            r = r0 + (r1 - r0) * k // steps
            c = c0 + (c1 - c0) * k // steps
        else:
            r, c = r0, c0
        cells.append((r, c))
    return cells


class DisplayManager:
    """Renders the system status, order and rider tables and the ASCII map."""

    @staticmethod
    def blank_canvas():
        """Return the empty street grid as a list of character rows."""
        canvas = []
        for row in range(CANVAS_SIZE):
            line = []
            for col in range(CANVAS_SIZE):
                if row % 2 == 0 and col % 2 == 0:
                    char = "+"
                elif row % 2 == 0:
                    char = "-"
                elif col % 2 == 0:
                    char = "|"
                else:
                    char = "."
                line.append(char)
            canvas.append(line)
        return canvas

    @staticmethod
    def _stamp(canvas, point, symbol):
        """Place ``symbol`` on the canvas at the cell that contains ``point``."""
        row, col = point_to_canvas(point[0], point[1])
        canvas[row][col] = symbol

    @staticmethod
    def _draw_route(canvas, start, points):
        """Draw a connected line segment graph through ``points`` with ``*``."""
        previous = start
        for point in points:
            r0, c0 = point_to_canvas(*previous)
            r1, c1 = point_to_canvas(*point)
            for r, c in _line_cells((r0, c0), (r1, c1)):
                if canvas[r][c] not in SYMBOL_MARKERS:
                    canvas[r][c] = SYMBOL_ROUTE
            previous = point

    @staticmethod
    def _render_canvas(canvas):
        """Convert the canvas into text lines with y-axis labels."""
        lines = [f"Map: x 0..{MAP_SIZE} (left to right), y 0..{MAP_SIZE} (top to bottom)"]
        for row_index, row in enumerate(canvas):
            label = f"{row_index * UNIT:>4} " if row_index % 2 == 0 else "     "
            lines.append(label + "".join(row))
        return "\n".join(lines)

    def render_map(self, riders=None, orders=None):
        """Render the current map with riders and active orders.

        Args:
            riders: Iterable of riders to show.
            orders: Iterable of orders; completed orders are hidden.

        Returns:
            Multi-line string containing the ASCII map and legend.
        """
        canvas = self.blank_canvas()
        for order in orders or []:
            if order.status == "completed":
                continue
            self._stamp(canvas, order.res_pos, SYMBOL_RESTAURANT)
            self._stamp(canvas, order.cus_pos, SYMBOL_CUSTOMER)
        for rider in riders or []:
            self._stamp(canvas, rider.position, SYMBOL_RIDER)
        return self._render_canvas(canvas) + "\n" + LEGEND

    def render_route(self, rider, order):
        """Render the route a rider will take to serve ``order``.

        Includes the rider speed and an estimated number of ticks, and draws the
        route on the ASCII map.

        Args:
            rider: The assigned :class:`rider.Rider`.
            order: The :class:`order_model.Order`.

        Returns:
            Multi-line description and map string.
        """
        planner = PathPlanner()
        if rider.state == DELIVERING:
            points = list(rider.waypoints)
        elif rider.state == PICKING_UP:
            points = list(rider.waypoints) + planner.plan(order.res_pos, order.cus_pos)
        else:
            points = planner.plan(rider.position, order.res_pos) + \
                planner.plan(order.res_pos, order.cus_pos)
        length = route_length(rider.position, points)
        speed = max(1, rider.speed)
        ticks = -(-length // speed) if length else 0

        canvas = self.blank_canvas()
        self._draw_route(canvas, rider.position, points)
        self._stamp(canvas, order.res_pos, SYMBOL_RESTAURANT)
        self._stamp(canvas, order.cus_pos, SYMBOL_CUSTOMER)
        self._stamp(canvas, rider.position, SYMBOL_RIDER)

        header = (
            f"Order #{order.id} assigned to Rider #{rider.rider_id} "
            f"(state: {rider.state_name})\n"
            f"Restaurant cell center: {order.res_pos}   "
            f"Customer cell center: {order.cus_pos}\n"
            f"Rider speed: {rider.speed} units/tick   "
            f"Route length: {length} units   "
            f"Estimated time: {ticks} tick(s)"
        )
        return header + "\n" + self._render_canvas(canvas) + "\n" + LEGEND

    @staticmethod
    def format_status(funds, riders, tick):
        """Format the system snapshot: funds, tick and each rider's state.

        Args:
            funds: Current fund balance.
            riders: Iterable of riders.
            tick: Current simulated tick.

        Returns:
            Multi-line string.
        """
        riders = list(riders or [])
        lines = [f"Tick: {tick} | System funds: {funds} | Riders: {len(riders)}"]
        if not riders:
            lines.append("No riders recruited yet.")
        for rider in riders:
            order_ids = [order.id for order in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id:<3} state={rider.state_name:<11} "
                f"pos=({rider.x},{rider.y}) speed={rider.speed} "
                f"orders={len(rider.orders)} {order_ids} "
                f"special={'yes' if rider.special_case else 'no'}"
            )
        return "\n".join(lines)

    @staticmethod
    def format_orders(orders):
        """Format a table of order details.

        Args:
            orders: Iterable of orders.

        Returns:
            Multi-line string table.
        """
        orders = list(orders or [])
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Rider':>6} {'Status':<12} {'Dist':>6} {'Time':>6} {'Eff':>6}")

        for order in orders:
            rider = "-" if order.rider_id is None else f"#{order.rider_id}"
            taken = order.time_taken

                )
        return "\n".join(lines) if lines else "No riders recruited yet."

"""Command parsing, shortcut handling, history and input validation."""

import logging

logger = logging.getLogger(__name__)

COORD_MIN = 0
COORD_MAX = 800
COORD_NAMES = ("restaurant x", "restaurant y", "customer x", "customer y")

COMMAND_ALIASES = {
    "start": "start", "s": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick", "step": "tick", "next": "tick",
    "save": "save",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}


class InputManager:
    """Reads user input, expands history shortcuts and validates values.

    Shortcuts:
        ``!!`` repeats the previous command; ``!n`` repeats history entry n.
    """

    def __init__(self, input_func=input):
        """Create the manager.

        Args:
            input_func: Callable used to read a line (injectable for testing).
        """
        self._input = input_func
        self.history = []

    def read_line(self, prompt="> "):
        """Read one command line, expanding history shortcuts.

        Args:
            prompt: Prompt text.

        Returns:
            The command text (possibly empty).

        Raises:
            ValueError: If a history shortcut cannot be resolved.
        """
        raw = self._input(prompt).strip()
        if not raw:
            return ""
        if raw == "!!":
            if not self.history:
                raise ValueError("no previous command in history")
            raw = self.history[-1]
        elif raw.startswith("!") and raw[1:].isdigit():
            index = int(raw[1:])
            if not 1 <= index <= len(self.history):
                raise ValueError(f"history entry {index} does not exist")
            raw = self.history[index - 1]
        self.history.append(raw)
        return raw

    @staticmethod
    def parse_command(line):
        """Split a command line into a canonical command name and arguments.

        Args:
            line: Raw command text.

        Returns:
            Tuple ``(command, args)``; command is empty for blank input.
        """
        parts = line.split()
        if not parts:
            return "", []
        name = parts[0].lower()
        return COMMAND_ALIASES.get(name, name), parts[1:]

    @staticmethod
    def parse_int(token, name):
        """Parse an integer token with a clear error message.

        Raises:
            ValueError: If the token is not an integer.
        """
        try:
            return int(token)
        except (TypeError, ValueError):
            raise ValueError(f"{name} must be an integer, got {token!r}") from None

    @staticmethod
    def validate_coordinate(name, value):
        """Ensure a coordinate lies within the map range.

        Raises:
            ValueError: If the value is outside 0-800.
        """
        if not COORD_MIN <= value <= COORD_MAX:
            raise ValueError(f"{name} must be between {COORD_MIN} and {COORD_MAX}, got {value}")
        return value

    def collect_coordinates(self, tokens):
        """Return four validated coordinates from arguments or prompts.

        Args:
            tokens: Arguments following the ``start`` command. Either empty
                (prompt for each value) or exactly four integers.

        Returns:
            List of four integers in the order restaurant x, y, customer x, y.

        Raises:
            ValueError: If the argument count or values are invalid.
        """
        if tokens:
            if len(tokens) != 4:
                raise ValueError("start expects 4 integers: restaurant_x restaurant_y "
                                 "customer_x customer_y")
            return [self.validate_coordinate(name, self.parse_int(token, name))
                    for token, name in zip(tokens, COORD_NAMES)]

        values = []
        print(f"Enter coordinates in the range {COORD_MIN}-{COORD_MAX}:")
        for name in COORD_NAMES:
            while True:
                raw = self._input(f"  {name}: ").strip()
                try:
                    value = self.validate_coordinate(name, self.parse_int(raw, name))
                except ValueError as exc:
                    print(f"Error: {exc}")
                    continue
                values.append(value)
                break
        return values

import json
import logging
import os

import globalvar

logger = logging.getLogger(__name__)

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
CONFIG_PATH = os.path.join(BASE_DIR, "config.json")
DEFAULT_STATE_PATH = os.path.join(BASE_DIR, "state.json")


def load_config(path=CONFIG_PATH):
    """Load system parameters from a JSON configuration file.

    Args:
        path: Path to the configuration file.

    Returns:
        Dictionary of parameters; empty if the file is missing or invalid.
    """
    if not os.path.exists(path):
        logger.info("No configuration file at %s; using defaults", path)
        return {}
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read configuration %s: %s", path, exc)
        return {}
    if not isinstance(data, dict):
        logger.warning("Configuration %s must contain a JSON object", path)
        return {}
    return data


def build_snapshot():
    """Build a JSON-serialisable snapshot of the current simulation state."""
    riders = globalvar.get_value("riders") or []
    orders = globalvar.get_value("orders") or []
    return {
        "tick": globalvar.get_value("tick") or 0,
        "funds": globalvar.get_value("funds") or 0,
        "rider_id_counter": globalvar.get_value("rider_id_counter") or 0,
        "riders": [rider.to_dict() for rider in riders],
        "orders": [order.to_dict() for order in orders],
    }


def save_state(path=DEFAULT_STATE_PATH):
    """Write the simulation snapshot to a JSON file.

    Args:
        path: Destination file path.

    Returns:
        The path written to.

    Raises:
        OSError: If the file cannot be written.

        json.dump(snapshot, handle, indent=2)

    return path

Run with::

    python src/main.py

Commands:
    start [rx ry cx cy]  create an order (prompts if no coordinates given)
    status               show funds and every rider's state and orders

    help                 show this help
    quit | exit          leave the program

import globalvar
import persistence
import system
from input_manager import InputManager
from order import OrderManager

logger = logging.getLogger("dispatch")

HELP_TEXT = __doc__


def _enable_history():
    """Enable line editing and arrow-key history when readline is available."""
    try:
        import readline  # noqa: F401  (imported for its side effects)
    except ImportError:
        logger.debug("readline not available; history shortcuts still work")


class DispatchApp:
    """Interactive application that routes commands to the managers."""

    def __init__(self, input_manager=None):
        """Create the application and bind the command handlers.

        Args:
            input_manager: Optional :class:`InputManager` (injectable).
        """
        self.inputs = input_manager or InputManager()
        self.orders = OrderManager()
        self.display = display_module.DisplayManager()
        self.system = globalvar.get_value("system")
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def run(self):
        """Run the main command loop until the running flag is cleared."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while globalvar.get_value("running"):
            try:
                line = self.inputs.read_line("dispatch> ")
            except ValueError as exc:
                print(f"Error: {exc}")
                continue
            except (EOFError, KeyboardInterrupt):
                print()
                break
            if not line:
                continue
            command, args = InputManager.parse_command(line)
            handler = self.handlers.get(command)
            if handler is None:
                print(f"Unknown command: {command!r}. Type 'help' for a list.")
                continue
            try:
                handler(args)
            except ValueError as exc:
                print(f"Input error: {exc}")
            except Exception:
                logger.exception("Unexpected error while running %r", command)
                print("An unexpected error occurred; the system state was preserved.")
        print("Goodbye.")

    def cmd_start(self, args):
        """Create an order and dispatch it to a rider, showing the route."""
        coords = self.inputs.collect_coordinates(args)
        order = self.orders.create_order(*coords)
        rider, new_rider = self.orders.dispatch(order)
        print(f"Order #{order.id}: restaurant cell center {order.res_pos}, "
              f"customer cell center {order.cus_pos}")
        if new_rider:
            print(f"New rider recruited (#{rider.rider_id if rider else '?'}). "
                  f"System funds: {self.system.funds}")
        if rider is None:
            print("No rider is available; the order is pending and will be retried on each tick.")
            return
        print(self.display.render_route(rider, order))

    def cmd_status(self, args):
        """Show the system snapshot."""
        print(self.display.format_status(self.system.funds, self.system.riders,
                                         self.system.tick_count))

    def cmd_orders(self, args):
        """Show the order table."""
        print(self.display.format_orders(self.orders.all_orders()))

    def cmd_riders(self, args):
        """Show the status-change log for every rider."""
        print(self.display.format_rider_logs(self.system.riders))

    def cmd_map(self, args):
        """Show the ASCII map with riders and active orders."""
        print(self.display.render_map(self.system.riders, self.orders.all_orders()))

    def cmd_tick(self, args):
        """Advance the simulation clock by one or more ticks."""
        count = 1
        if args:
            count = InputManager.parse_int(args[0], "tick count")
        if not 1 <= count <= 1000:
            raise ValueError("tick count must be between 1 and 1000")
        before = len(self.orders.completed_orders())
        for _ in range(count):
            self.system.tick()
            self.orders.retry_pending()
            for issue in self.system.check_consistency():
                logger.warning("Consistency repair: %s", issue)
        after = len(self.orders.completed_orders())
        print(f"Tick is now {self.system.tick_count}. Orders completed in this step: "
              f"{after - before}. Pending orders: {len(self.orders.pending_orders())}.")

    def cmd_save(self, args):
        """Save the simulation state to a JSON file."""
        path = args[0] if args else persistence.DEFAULT_STATE_PATH
        try:
            persistence.save_state(path)
        except OSError as exc:
            print(f"Could not save state: {exc}")
            return
        print(f"State saved to {path}")

    def cmd_help(self, args):
        """Print the command reference."""
        print(HELP_TEXT)

    def cmd_quit(self, args):
        """Stop the main loop."""

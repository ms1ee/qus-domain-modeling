from lottery_system.default_data import DEFAULT_USER_LINES
from lottery_system.models import User


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list (each line: "Name,Department Group")."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        """Read a txt file and parse its lines into users and format errors."""
        try:
            with open(path, "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = [part.strip() for part in text.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {line_no}: Format Error - '{text}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Split users into valid users (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict:
        """Return a mapping of department group -> number of people."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.default_data import DEFAULT_PRIZE_SPECS
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for the event."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for spec in DEFAULT_PRIZE_SPECS:
            try:
                self.add_prize(
                    spec["name"],
                    spec["quantity"],
                    list(spec["avoid_groups"]),
                    spec["associated_drawer"],
                )
            except ValueError as exc:
                errors.append(f"{spec['name']}: {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        """Add a prize at the end of the lottery sequence."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        prize = Prize(name=name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Winner recording, grouping and CSV-style export."""

import csv
from dataclasses import dataclass
from datetime import date
from pathlib import Path

from lottery_system.models import Prize, User


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str


class ResultManager:
    """Keeps all winners in winning order."""

    def __init__(self):
        self._winners: list[Winner] = []

    @property
    def winners(self) -> list[Winner]:
        return list(self._winners)

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(user=user, prize_name=prize.name)
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes) -> dict:
        """Return {prize name: [Winner, ...]} for the given prizes, in prize order."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt (CSV)."""
        grouped = self.get_winners_by_prize(prizes)
        rows = [winner for prize in prizes for winner in grouped.get(prize.name, [])]
        if not rows:
            return False, "No winners to export."

        path = Path.cwd() / f"awards_result_{date.today():%Y%m%d}.txt"
        try:
            with open(path, "w", newline="", encoding="utf-8") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Name", "Department", "Prize"])
                for winner in rows:
                    writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery execution and rule engine."""

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
CHENG_RU_GROUP = "Cheng Ru Group"

# Lottery persons whose round results feed the Cheng Ru protection check.
TRACKED_DRAWERS = (CHEN_QIXIAN, JIN_DUO)


def _norm(text) -> str:
    return " ".join(str(text).strip().split()).lower()


def _is_cheng_ru(user: User) -> bool:
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


def _user_key(user: User) -> str:
    return _norm(user.name)


class LotteryManager:
    """Runs lottery rounds while applying the lottery rule engine."""

    def __init__(self, users, rng=None):
        self._users: list[User] = list(users)
        self._rng = rng if rng is not None else random.Random()
        self._winner_keys: set[str] = set()
        self.winners: list[User] = []
        self.result_manager = ResultManager()
        # Round results (True = the round contained Cheng Ru group members)
        # for the last two rounds drawn by Chen Qixian / Jin Duo.
        self.last_two_draw_results: list[bool] = []
        self._round_prize: Prize | None = None
        self._round_winners: list[User] = []

    # ----- rule helpers -------------------------------------------------

    @staticmethod
    def _is_drawer(prize: Prize, drawer_name: str) -> bool:
        return prize.associated_drawer is not None and _norm(prize.associated_drawer) == _norm(drawer_name)

    def fallback_active(self, prize: Prize) -> bool:
        """Cheng Ru protection: if neither of the last two Chen/Jin rounds had
        Cheng Ru members, Ma Jiaqi's prize pool is limited to Cheng Ru members."""
        return (
            self._is_drawer(prize, MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def is_winner(self, user: User) -> bool:
        return _user_key(user) in self._winner_keys

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        # Anti-duplication: previous winners are always excluded.
        pool = [user for user in self._users if not self.is_winner(user)]

        # Special rule: Li Bin only filters previous winners.
        if self._is_drawer(prize, LI_BIN):
            return pool

        # Fallback (Cheng Ru protection) overrides avoidance groups.
        if self.fallback_active(prize):
            return [user for user in pool if _is_cheng_ru(user)]

        avoid = {_norm(group) for group in prize.avoid_groups}
        return [user for user in pool if _norm(user.department) not in avoid]

    # ----- round control -----------------------------------------------

    def start_round(self, prize: Prize) -> None:
        self._round_prize = prize
        self._round_winners = []

    def draw_next(self) -> User | None:
        """Draw one winner for the current round. Returns None if no one is eligible."""
        if self._round_prize is None:
            raise RuntimeError("No lottery round in progress. Call start_round() first.")
        pool = self._get_eligible_pool(self._round_prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winner_keys.add(_user_key(user))
        self.winners.append(user)
        self._round_winners.append(user)
        self.result_manager.add_winner(user, self._round_prize)
        return user

    def finish_round(self) -> list[User]:
        """Close the current round, update fallback history and return its winners."""
        prize = self._round_prize
        if prize is None:
            return []
        if any(self._is_drawer(prize, drawer) for drawer in TRACKED_DRAWERS):
            had_cheng_ru = any(_is_cheng_ru(user) for user in self._round_winners)
            self.last_two_draw_results.append(had_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        result = list(self._round_winners)
        self._round_prize = None
        self._round_winners = []
        return result

    @property
    def round_winners(self) -> list[User]:
        return list(self._round_winners)

def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotterySystemCLI:
    """Interactive controller for the whole lottery workflow."""

    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.completed_prizes: set[str] = set()

    # ----- helpers ------------------------------------------------------

    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _ask_positive_int(self, prompt: str) -> int:
        while True:
            raw = self._ask(prompt)
            try:
                value = int(raw)
                if value >= 1:
                    return value
            except ValueError:
                pass
            print("Please enter a positive integer.")

    def _confirm_remove_or_terminate(self, kind: str) -> bool:
        while True:
            choice = self._ask(f"[R] Remove {kind} data and continue / [T] Terminate upload: ").lower()
            if choice == "r":
                return True
            if choice == "t":
                return False
            print("Please enter R or T.")

    # ----- main loop ----------------------------------------------------

    def run(self) -> None:
        print(BANNER)
        while True:
            print(MAIN_MENU)
            choice = self._ask("Select an option: ")
            try:
                if choice == "1":
                    self.load_participants()
                elif choice == "2":
                    self.view_groups()
                elif choice == "3":
                    self.prize_settings()
                elif choice == "4":
                    self.start_lottery()
                elif choice == "5":
                    self.show_winners()
                elif choice == "6":
                    self.export_results()
                elif choice == "0":
                    print("Goodbye.")
                    return
                else:
                    print("Invalid option, please try again.")
            except (EOFError, KeyboardInterrupt):
                print("\nExiting.")
                return

    # ----- list management ---------------------------------------------

    def load_participants(self) -> None:
        print("1. Built-in default team building list")
        print("2. Custom txt file (each line: Name,Department Group)")
        choice = self._ask("Choose list source: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip('"').strip("'")
            if not path:
                print("No path given.")
                return
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Invalid choice.")
            return

        if errors:
            print(f"Found {len(errors)} invalid entr{'y' if len(errors) == 1 else 'ies'}:")
            for error in errors:
                print(f"  {error}")
            if not self._confirm_remove_or_terminate("invalid"):
                print("Upload terminated.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  {user.name} ({user.department})")
            if not self._confirm_remove_or_terminate("duplicate"):
                print("Upload terminated.")
                return

        if not valid:
            print("No valid participants were loaded.")
            return

        self.users = valid
        self.lottery = None
        self.completed_prizes = set()
        print(f"Loaded {len(valid)} participant(s). Previous lottery results were reset.")

    def view_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current valid groups:")
        for index, (department, count) in enumerate(info.items(), start=1):
            print(f"  {index}. {department}: {count} people")
        print(f"Total: {len(self.users)} people")

    # ----- prize management --------------------------------------------

    def prize_settings(self) -> None:
        while True:
            print(PRIZE_MENU)
            choice = self._ask("Select an option: ").lower()
            if choice == "a":
                self.view_prizes()
            elif choice == "b":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    print("Some default prizes could not be added:")
                    for error in errors:
                        print(f"  {error}")
                else:
                    print("Default prize library loaded.")
            elif choice == "c":
                self.add_custom_prize()
            elif choice == "d":
                self.reorder_prizes()
            elif choice == "0":
                return
            else:
                print("Invalid option.")

    def view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                  f"lottery person: {prize.associated_drawer or 'None'} | "
                  f"avoid groups: {', '.join(prize.avoid_groups) or 'None'}")

    def _choose_avoid_groups(self) -> list[str]:
        if not self.users:
            print("No participant list loaded; no avoidance groups can be selected.")
            return []
        info = self.list_manager.get_group_info(self.users)
        groups = list(info.keys())
        print("Available groups:")
        for index, department in enumerate(groups, start=1):
            print(f"  {index}. {department}")
        raw = self._ask("Avoidance groups (numbers or names, comma-separated; Enter for none): ")
        selected: list[str] = []
        for token in [t.strip() for t in raw.split(",") if t.strip()]:
            if token.isdigit() and 1 <= int(token) <= len(groups):
                name = groups[int(token) - 1]
            else:
                name = next((g for g in groups if g.lower() == token.lower()), token)
                if name == token and token not in groups:
                    print(f"  Warning: '{token}' is not a current group name; kept anyway.")
            if name not in selected:
                selected.append(name)
        return selected

    def add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity = self._ask_positive_int("Quantity: ")
        drawer = self._ask("Associated lottery person (Enter to skip): ") or None
        groups = self._choose_avoid_groups()
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer)
            print(f"Prize added: {prize.name} (quantity {prize.quantity}).")
        except ValueError as exc:
            print(f"Cannot add prize: {exc}")

    def reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self.view_prizes()
        raw = self._ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in raw.replace(",", " ").split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery order updated.")
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ----- lottery ------------------------------------------------------

    def _describe_rules(self, prize: Prize) -> str:
        if prize.associated_drawer is None:
            base = "Lottery person: not specified. Rule: previous winners excluded"
        else:
            base = f"Lottery person: {prize.associated_drawer}. "
            if prize.associated_drawer.strip().lower() == LI_BIN.lower():
                base += "Rule: previous winners excluded only (avoidance groups not applied)"
            elif self.lottery is not None and self.lottery.fallback_active(prize):
                base += "Rule: Cheng Ru Group protection active - only Cheng Ru Group members eligible"
            elif prize.associated_drawer.strip().lower() == MA_JIAQI.lower() and not prize.avoid_groups:
                base += "Rule: previous winners excluded"
            else:
                base += "Rule: previous winners excluded"
        if prize.avoid_groups and not (
            prize.associated_drawer and prize.associated_drawer.strip().lower() == LI_BIN.lower()
        ):
            base += f"; avoid groups: {', '.join(prize.avoid_groups)}"
        return base

    def _wait_until_resumed(self) -> bool:
        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = self._ask("> ").lower()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Please type 'continue' or 'terminate'.")

    def _run_round(self, prize: Prize) -> None:
        lottery = self.lottery
        lottery.start_round(prize)
        print(f"Prize: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {self._describe_rules(prize)}")
        print("Commands: Enter = draw next | pause | terminate")

        drawn = 0
        while drawn < prize.quantity:
            command = self._ask(
                f"[{ordinal(drawn + 1)} winner] Press Enter to draw (pause / terminate): "
            ).lower()
            if command == "terminate":
                print("Round terminated manually; remaining winners skipped.")
                break
            if command == "pause":
                if not self._wait_until_resumed():
                    print("Round terminated manually; remaining winners skipped.")
                    break
                continue
            if command == "continue":
                print("Lottery is not paused.")
                continue
            if command:
                print("Unknown command.")
                continue

            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = lottery.draw_next()
            if user is None:
                print("No eligible participants remain for this prize; round ends early.")
                break
            drawn += 1
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        winners = lottery.finish_round()
        print(f"Round finished: {len(winners)} of {prize.quantity} winner(s) drawn.")
        for index, user in enumerate(winners, start=1):
            print(f"  {index}. {user.name} ({user.department})")

    def start_lottery(self) -> None:
        if not self.users:
            print("Please load the participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Please set up prizes first.")
            return
        pending = [prize for prize in prizes if prize.name not in self.completed_prizes]
        if not pending:
            print("All prizes have already been drawn.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)

        print(f"Starting lottery with {len(pending)} round(s).")
        for index, prize in enumerate(pending, start=1):
            print(f"\n########## Round {index}/{len(pending)} ##########")
            self._run_round(prize)
            self.completed_prizes.add(prize.name)

        print("\nAll rounds completed. Winner summary:")
        self.show_winners()

    # ----- results ------------------------------------------------------

    def show_winners(self) -> None:
        if self.lottery is None:
            print("No lottery results yet.")
            return
        grouped = self.lottery.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        for prize_name, winners in grouped.items():
            print(f"{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:

"""Grid-based path planning for the 8x8 delivery map.

The map is an 800x800 unit area divided into 8x8 cells of 100x100 units.
Grid lines (coordinates that are multiples of 100) are streets.  Restaurant
and customer locations are cell centres (``100 * k + 50``).  Routes are made
of axis-aligned legs only and prefer to travel along the nearest street.
"""

from __future__ import annotations

from typing import List, Sequence, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = 8

Point = Tuple[int, int]


class PathPlanner:
    """Stateless helper providing grid coordinates, routes and movement.

    Route rules:
        * A target directly above/below (same x) or on the same row (same y)
          is reached by a straight line (the "special case").
        * Otherwise the rider first moves to the nearest street row, travels
          along that street to the target column, and finally leaves the
          street towards the target.
    """

    @staticmethod
    def clamp(value: int, low: int = 0, high: int = MAP_SIZE) -> int:
        """Clamp ``value`` into the inclusive range ``[low, high]``."""
        return max(low, min(high, int(value)))

    @staticmethod
    def cell_index(value: int) -> int:
        """Return the grid index (0-7) of a coordinate value."""
        return min(max(int(value), 0) // CELL_SIZE, GRID_DIM - 1)

    @classmethod
    def cell_of(cls, x: int, y: int) -> Tuple[int, int]:
        """Return the ``(column, row)`` grid cell containing ``(x, y)``."""
        return cls.cell_index(x), cls.cell_index(y)

    @classmethod
    def cell_center(cls, x: int, y: int) -> Point:
        """Convert a coordinate to the centre of its grid cell.

        Uses ``center = (value // 100) * 100 + 50`` per axis.
        """
        col, row = cls.cell_of(x, y)
        half = CELL_SIZE // 2
        return col * CELL_SIZE + half, row * CELL_SIZE + half

    @staticmethod
    def is_street(x: int, y: int) -> bool:
        """Return True if the point lies on a street (grid line)."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @classmethod
    def nearest_street(cls, value: int) -> int:
        """Return the grid line closest to ``value`` (clamped to the map)."""
        return cls.clamp(((int(value) + CELL_SIZE // 2) // CELL_SIZE) * CELL_SIZE)

    @staticmethod
    def is_straight(start: Sequence[int], target: Sequence[int]) -> bool:
        """Return True when start and target share a column or a row.

        This is the "directly above or below" special case.
        """
        return start[0] == target[0] or start[1] == target[1]

    @classmethod
    def build_route(cls, start: Sequence[int], target: Sequence[int]) -> List[Point]:
        """Build the list of waypoints from ``start`` to ``target``.

        Args:
            start: Current ``(x, y)`` position.
            target: Destination ``(x, y)`` position.

        Returns:
            Waypoints in travel order, excluding the start point. The last
            waypoint is the target. An empty list means already there.
        """
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        if (sx, sy) == (tx, ty):
            return []
        if cls.is_straight((sx, sy), (tx, ty)):
            return [(tx, ty)]
        street_y = cls.nearest_street(sy)
        candidates = [(sx, street_y), (tx, street_y), (tx, ty)]
        route: List[Point] = []
        previous = (sx, sy)
        for point in candidates:
            if point != previous:
                route.append(point)
                previous = point
        return route

    @staticmethod
    def advance(x: int, y: int, route: List[Point], speed: int) -> Point:
        """Move along ``route`` by up to ``speed`` units.

        Reached waypoints are removed from ``route`` in place.  Movement is
        always axis-aligned.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            route: Mutable list of remaining waypoints.
            speed: Maximum Manhattan distance to travel this tick.

        Returns:
            The new ``(x, y)`` position.
        """
        remaining = int(speed)
        while remaining > 0 and route:
            tx, ty = route[0]
            dx = tx - x
            dy = ty - y
            distance = abs(dx) + abs(dy)
            if distance == 0:
                route.pop(0)
                continue
            step = min(remaining, distance)
            if dx != 0:
                x += step if dx > 0 else -step
            else:
                y += step if dy > 0 else -step
            remaining -= step
            if (x, y) == (tx, ty):
                route.pop(0)
        return x, y

"""Rider model: state machine, movement and order execution."""

from __future__ import annotations

import logging
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_START: Tuple[int, int] = (435, 385)
DEFAULT_SPEED = 10


class Rider:
    """A delivery rider that moves on the grid and executes orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates.
        speed: Movement speed in units per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders (``Order`` objects).
        special_case: True while moving straight to a directly above/below
            target.
        log: Event records with keys ``event`` and ``order_id`` (plus tick,
            position and state).
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the configured start position (default 435,385).

        Args:
            rider_id: Unique rider identifier.
        """
        start = globalvar.get_value("rider_start") or DEFAULT_START
        self.rider_id: int = int(rider_id)
        self.x: int = int(start[0])
        self.y: int = int(start[1])
        self.speed: int = int(globalvar.get_value("rider_speed") or DEFAULT_SPEED)
        self.state: int = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.route: List[Tuple[int, int]] = []
        self.current_order: Optional[Any] = None
        self._accepted_ids: set = set()
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    # ------------------------------------------------------------------ #
    # Read-only helpers
    # ------------------------------------------------------------------ #
    @property
    def position(self) -> Tuple[int, int]:
        """Return the current ``(x, y)`` position."""
        return self.x, self.y

    @property
    def state_name(self) -> str:
        """Return a human readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def is_idle(self) -> bool:
        """Return True when the rider has no state activity and no orders."""
        return self.state == STATE_IDLE and not self.orders

    # ------------------------------------------------------------------ #
    # Logging
    # ------------------------------------------------------------------ #
    @staticmethod
    def _now() -> int:
        """Return the current simulation tick."""
        return int(globalvar.get_value("current_tick", 0))

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log."""
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": self._now(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider #%d %s order=%s at (%d,%d)",
                     self.rider_id, event, order_id, self.x, self.y)

    # ------------------------------------------------------------------ #
    # Simulation
    # ------------------------------------------------------------------ #
    def move(self) -> None:
        """Advance the rider by one tick.

        Idle riders accept the next assigned order (state 0 -> 1) without
        moving in that tick.  Picking-up riders move towards the restaurant
        and switch to delivering (1 -> 2) on arrival.  Delivering riders move
        to the customer and complete the order (2 -> 0), then immediately
        accept the next queued order if one exists.
        """
        self.special_case = False
        if self.state == STATE_IDLE:
            self._accept_next_order()
            return
        self._advance()

    def _accept_next_order(self) -> bool:
        """Accept the first assigned order that has not been started yet.

        Returns:
            True if an order was accepted.
        """
        for order in self.orders:
            if order.id in self._accepted_ids:
                continue
            self._accepted_ids.add(order.id)
            self.current_order = order
            self.state = STATE_PICKING_UP
            self.route = []
            order.status = "picking_up"
            self._record("accepted_order", order.id)
            return True
        return False

    def _advance(self) -> None:
        """Move one tick along the current leg and handle arrivals."""
        order = self.current_order
        if order is None:
            logger.warning("Rider #%d is busy without an order; resetting.", self.rider_id)
            self.state = STATE_IDLE
            self.route = []
            return

        target = order.res_pos if self.state == STATE_PICKING_UP else order.cus_pos
        if not self.route:
            if self.position == target:
                self._arrive()
                return
            self.route = PathPlanner.build_route(self.position, target)
            self.special_case = PathPlanner.is_straight(self.position, target)

        self.x, self.y = PathPlanner.advance(self.x, self.y, self.route, self.speed)
        if not self.route and self.position == target:
            self._arrive()

    def _arrive(self) -> None:
        """Handle arrival at the current leg's destination."""
        order = self.current_order
        now = self._now()
        if self.state == STATE_PICKING_UP:
            order.mark_picked_up(now)
            self.state = STATE_DELIVERING
            self.route = []
            self._record("arrived_at_merchant", order.id)
            return

        if self.state == STATE_DELIVERING:
            order.complete(now)
            self._record("delivered_to_customer", order.id)
            if order in self.orders:
                self.orders.remove(order)
            self._accepted_ids.discard(order.id)
            self.current_order = None
            self.route = []
            self.state = STATE_IDLE
            self._accept_next_order()

    def recover(self) -> None:
        """Reset transient movement data after an unexpected error."""
        self.route = []
        if self.current_order is None and self.state != STATE_IDLE:
            self.state = STATE_IDLE
        logger.warning("Rider #%d recovered to state %s.", self.rider_id, self.state_name)

"""Order data model: coordinate conversion, lifecycle and efficiency data."""

from __future__ import annotations

import itertools
import math
from typing import Any, Dict, Optional, Sequence

from path_planner import PathPlanner


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order identifier.
        raw_res_pos, raw_cus_pos: Coordinates exactly as entered by the user.
        res_pos, cus_pos: Grid cell centres used for routing.
        status: ``pending`` -> ``assigned`` -> ``picking_up`` ->
            ``delivering`` -> ``completed``.
        rider_id: Identifier of the assigned rider (or None).
        start_tick, pickup_tick, end_tick: Lifecycle timestamps.
        distance: Straight-line distance between the two cell centres.
    """

    _id_sequence = itertools.count(1)

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int],
                 start_tick: int = 0) -> None:
        """Create an order and convert its coordinates to grid centres.

        Args:
            res_pos: ``(restaurant_x, restaurant_y)`` in range 0-800.
            cus_pos: ``(customer_x, customer_y)`` in range 0-800.
            start_tick: Simulation tick at creation.
        """
        self.id: int = next(Order._id_sequence)
        self.raw_res_pos = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus_pos = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = PathPlanner.cell_center(*self.raw_res_pos)
        self.cus_pos = PathPlanner.cell_center(*self.raw_cus_pos)
        self.distance = round(math.hypot(self.cus_pos[0] - self.res_pos[0],
                                         self.cus_pos[1] - self.res_pos[1]), 1)
        self.status: str = "pending"
        self.rider_id: Optional[int] = None
        self.start_tick: int = int(start_tick)
        self.pickup_tick: Optional[int] = None
        self.end_tick: Optional[int] = None

    def assign_to(self, rider_id: int) -> None:
        """Record the assigned rider and mark the order as assigned.

        The caller is responsible for appending the order to ``rider.orders``.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = int(rider_id)
        self.status = "assigned"

    def mark_picked_up(self, tick: int) -> None:
        """Record arrival at the restaurant.

        Args:
            tick: Simulation tick of the pickup.
        """
        self.pickup_tick = int(tick)
        self.status = "delivering"

    def complete(self, tick: int) -> None:
        """Mark the order as delivered.

        Args:
            tick: Simulation tick of the delivery.
        """
        self.end_tick = int(tick)
        self.status = "completed"

    @property
    def duration(self) -> Optional[int]:
        """Ticks from creation to completion, or None if not completed."""
        if self.end_tick is None:
            return None
        return self.end_tick - self.start_tick

    @property
    def efficiency(self) -> Optional[float]:
        """Delivery efficiency as distance units per tick, or None."""
        duration = self.duration
        if duration is None:
            return None
        return round(self.distance / max(duration, 1), 2)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "raw_restaurant": list(self.raw_res_pos),
            "raw_customer": list(self.raw_cus_pos),
            "restaurant_center": list(self.res_pos),
            "customer_center": list(self.cus_pos),
            "distance": self.distance,
            "status": self.status,
            "rider_id": self.rider_id,
            "start_tick": self.start_tick,
            "pickup_tick": self.pickup_tick,
            "end_tick": self.end_tick,
            "duration": self.duration,
            "efficiency": self.efficiency,
        }

"""Order management: validation, creation, and rider assignment."""

from __future__ import annotations

import logging
import math
from typing import Any, List, Optional, Sequence

import globalvar
from order_model import Order
from rider import Rider

logger = logging.getLogger(__name__)

MIN_COORDINATE = 0
MAX_COORDINATE = 800
COORDINATE_LABELS = ("restaurant x", "restaurant y", "customer x", "customer y")


def validate_coordinates(values: Sequence[Any]) -> List[int]:
    """Validate and normalise ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

    Args:
        values: Four values that must be integers in the range 0-800.

    Returns:
        The four coordinates as ``int``.

    Raises:
        ValueError: If the count, type or range is invalid.
    """
    try:
        items = list(values)
    except TypeError as exc:
        raise ValueError("Expected four coordinates") from exc
    if len(items) != 4:
        raise ValueError(
            "Expected 4 coordinates [restaurant_x, restaurant_y, customer_x, "
            f"customer_y], got {len(items)}")

    result: List[int] = []
    for label, raw in zip(COORDINATE_LABELS, items):
        if isinstance(raw, bool):
            raise ValueError(f"{label} must be an integer, got {raw!r}")
        if isinstance(raw, float):
            if not raw.is_integer():
                raise ValueError(f"{label} must be an integer, got {raw}")
            raw = int(raw)
        elif isinstance(raw, str):
            try:
                raw = int(raw.strip())
            except ValueError as exc:
                raise ValueError(f"{label} must be an integer, got '{raw}'") from exc
        elif not isinstance(raw, int):
            raise ValueError(f"{label} must be an integer, got {raw!r}")
        if not MIN_COORDINATE <= raw <= MAX_COORDINATE:
            raise ValueError(
                f"{label} {raw} is out of range {MIN_COORDINATE}-{MAX_COORDINATE}")
        result.append(raw)
    return result


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Return the rider closest to the order's restaurant.

    Distance is Euclidean: ``sqrt((rx - restaurant_x)^2 + (ry - restaurant_y)^2)``.
    Ties are broken by the smaller number of assigned orders (load balancing),
    then by rider ID.

    Args:
        order: The order needing a rider.

    Returns:
        The best rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    best: Optional[Rider] = None
    best_key = None
    rest_x, rest_y = order.res_pos
    for rider in riders:
        distance = math.hypot(rider.x - rest_x, rider.y - rest_y)
        key = (round(distance, 6), len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


def assign_order(order: Order) -> Optional[Rider]:
    """Assign a pending order to the best rider.

    Args:
        order: Order to assign.

    Returns:
        The rider that received the order, or None if no rider is available.
    """
    if order.status != "pending":
        return None
    rider = find_best_rider_for_order(order)
    if rider is None:
        logger.warning("No rider available for order #%d", order.id)
        return None
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    logger.info("Order #%d assigned to rider #%d", order.id, rider.rider_id)
    return rider


class OrderManager:
    """Facade for order creation, lookup and assignment."""

    @staticmethod
    def create_order(coordinates: Sequence[Any], start_tick: Optional[int] = None) -> Order:
        """Validate coordinates, create an order and register it.

        Args:
            coordinates: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.
            start_tick: Creation tick; defaults to the current simulation tick.

        Returns:
            The new ``Order``.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        rest_x, rest_y, cus_x, cus_y = validate_coordinates(coordinates)
        if start_tick is None:
            start_tick = int(globalvar.get_value("current_tick", 0))
        order = Order((rest_x, rest_y), (cus_x, cus_y), start_tick)
        globalvar.get_value("orders").append(order)
        logger.info("Order #%d created: restaurant=%s customer=%s",
                    order.id, order.res_pos, order.cus_pos)
        return order

    @staticmethod
    def all_orders() -> List[Order]:
        """Return every order ever created (history included)."""
        return list(globalvar.get_value("orders") or [])

    @staticmethod
    def pending_orders() -> List[Order]:
        """Return orders still waiting for a rider."""
        return [o for o in OrderManager.all_orders() if o.status == "pending"]

    @staticmethod
    def active_orders() -> List[Order]:
        """Return orders that are not yet completed."""
        return [o for o in OrderManager.all_orders() if o.status != "completed"]

    @staticmethod
    def assign(order: Order) -> Optional[Rider]:
        """Assign ``order`` using :func:`assign_order`."""
        return assign_order(order)

from __future__ import annotations

import copy
import json
import logging

from order_model import Order

CONFIG_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)), "config.json")

DEFAULT_CONFIG: Dict[str, Any] = {
    "initial_funds": 1000,
    "recruit_cost": 300,
    "recruit_threshold": 300,
    "rider_speed": 10,
    "rider_start": [435, 385],
    "max_ticks_per_run": 10000,
    "state_file": "dispatch_state.json",
}


def load_config(path: Optional[str] = None) -> Dict[str, Any]:
    """Load system parameters from a JSON file, falling back to defaults.

    Unknown keys are ignored.  Invalid files produce a warning and defaults.

    Args:
        path: Path to the JSON config file (defaults to ``src/config.json``).

    Returns:
        The merged configuration dictionary.
    """
    config = copy.deepcopy(DEFAULT_CONFIG)
    path = path or CONFIG_FILE
    if not os.path.exists(path):
        return config
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read config %s: %s; using defaults.", path, exc)
        return config
    if not isinstance(data, dict):
        logger.warning("Config %s is not a JSON object; using defaults.", path)
        return config
    for key, value in data.items():
        if key in DEFAULT_CONFIG:
            config[key] = value
    try:
        config["initial_funds"] = int(config["initial_funds"])
        config["recruit_cost"] = int(config["recruit_cost"])
        config["recruit_threshold"] = int(config["recruit_threshold"])
        config["rider_speed"] = max(1, int(config["rider_speed"]))
        config["rider_start"] = [int(config["rider_start"][0]), int(config["rider_start"][1])]
        config["max_ticks_per_run"] = max(1, int(config["max_ticks_per_run"]))
    except (TypeError, ValueError, IndexError, KeyError) as exc:
        logger.warning("Invalid config values (%s); using defaults.", exc)
        return copy.deepcopy(DEFAULT_CONFIG)
    return config


def init_system(config_path: Optional[str] = None) -> None:
    """Initialise the global state for a fresh simulation run.

    Args:
        config_path: Optional path to a JSON configuration file.
    """
    config = load_config(config_path)
    globalvar._init()
    globalvar.set_value("config", config)
    globalvar.set_value("system_funds", config["initial_funds"])
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("system_running", True)
    globalvar.set_value("orders", [])
    globalvar.set_value("current_tick", 0)
    globalvar.set_value("rider_speed", config["rider_speed"])
    globalvar.set_value("rider_start", tuple(config["rider_start"]))
    logger.info("System initialised with funds=%d", config["initial_funds"])

    def get_rider(self, rider_id: Optional[int]) -> Optional[Rider]:
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def idle_rider_count(self) -> int:
        """Return the number of riders that are idle and have no orders."""
        return sum(1 for rider in self.riders if rider.is_idle())

    # ------------------------------------------------------------------ #
    # Recruitment
    # ------------------------------------------------------------------ #
    def recruit_rider(self) -> Optional[Rider]:
        """Create a new rider if funds allow.

        Returns:
            The new rider, or None if funds are below the threshold.
        """
        threshold = int(self.config["recruit_threshold"])
        cost = int(self.config["recruit_cost"])
        if self.funds < threshold:
            return None
        self._set_funds(self.funds - cost)
        counter = int(globalvar.get_value("rider_id_counter", 0)) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        logger.info("Recruited rider #%d (cost %d, funds now %d)", counter, cost, self.funds)
        return rider

    def ensure_capacity(self) -> List[Rider]:
        """Recruit riders while pending orders outnumber idle riders.

        Returns:
            List of riders recruited during this call.
        """
        recruited: List[Rider] = []
        while len(OrderManager.pending_orders()) > self.idle_rider_count():
            rider = self.recruit_rider()
            if rider is None:
                break
            recruited.append(rider)
        return recruited

    def dispatch_pending(self) -> Tuple[List[Rider], List[Order]]:
        """Recruit if needed, then assign every pending order.

        Returns:
            Tuple of (riders recruited, orders that were assigned).
        """
        recruited = self.ensure_capacity()
        assigned: List[Order] = []
        for order in OrderManager.pending_orders():
            if assign_order(order) is not None:
                assigned.append(order)
        return recruited, assigned

    # ------------------------------------------------------------------ #
    # Orders and simulation
    # ------------------------------------------------------------------ #
    def submit_order(self, coordinates: Sequence[Any]) -> Dict[str, Any]:
        """Create an order from user input and dispatch it.

        Args:
            coordinates: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            Dictionary with keys ``order``, ``rider``, ``new_riders``,
            ``distance`` (rider to restaurant), ``route`` (predicted points),
            and ``eta`` (predicted ticks, or None if the rider is busy).

        Raises:
            ValueError: If the coordinates are invalid.
        """
        validate_coordinates(coordinates)
        order = OrderManager.create_order(coordinates, self.tick_count)
        recruited, _ = self.dispatch_pending()
        rider = self.get_rider(order.rider_id) if order.rider_id is not None else None

        route: List[Tuple[int, int]] = []
        eta: Optional[int] = None
        distance = 0.0
        if rider is not None:
            distance = math.hypot(rider.x - order.res_pos[0], rider.y - order.res_pos[1])
            if rider.state == STATE_IDLE and rider.orders == [order]:
                route, eta = self.predict_route(rider, order)
        return {
            "order": order,
            "rider": rider,
            "new_riders": recruited,
            "distance": distance,
            "route": route,
            "eta": eta,
        }

    @staticmethod
    def predict_route(rider: Rider, order: Order) -> Tuple[List[Tuple[int, int]], int]:
        """Predict the path and duration of an order for an idle rider.

        The prediction follows the same rules as :meth:`Rider.move`: one tick
        to accept the order, then movement to the restaurant and to the
        customer.

        Args:
            rider: The idle rider that would execute the order.
            order: The order to simulate.

        Returns:
            Tuple of (list of positions visited, estimated ticks).
        """
        x, y = rider.position
        points: List[Tuple[int, int]] = [(x, y)]
        ticks = 1  # acceptance tick (no movement)
        for target in (order.res_pos, order.cus_pos):
            route = PathPlanner.build_route((x, y), target)
            if not route:
                ticks += 1
                continue
            while route:
                x, y = PathPlanner.advance(x, y, route, rider.speed)
                points.append((x, y))
                ticks += 1
        return points, ticks

    def tick(self) -> None:
        """Advance the simulation clock by one tick.

        Pending orders are dispatched first, then every rider moves.  A
        failing rider is recovered without stopping the other riders.
        """
        globalvar.set_value("current_tick", self.tick_count + 1)
        self.dispatch_pending()
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # error recovery: keep the simulation running
                logger.exception("Error while moving rider #%d", rider.rider_id)
                rider.recover()

    def has_active_orders(self) -> bool:
        """Return True while any order has not been completed."""
        return bool(OrderManager.active_orders())

    def run_until_idle(self, max_ticks: Optional[int] = None) -> int:
        """Tick until every order is completed.

        Args:
            max_ticks: Safety limit; defaults to the configured value.

        Returns:
            Number of ticks elapsed.

        Raises:
            RuntimeError: If the limit is reached with orders still active.
        """
        limit = int(max_ticks or self.config["max_ticks_per_run"])
        start = self.tick_count
        while self.has_active_orders():
            if self.tick_count - start >= limit:
                raise RuntimeError(
                    f"orders still active after {limit} ticks (no rider may be available)")
            self.tick()
        return self.tick_count - start

    # ------------------------------------------------------------------ #
    # Consistency and persistence
    # ------------------------------------------------------------------ #
    def check_consistency(self) -> List[str]:
        """Verify internal invariants and return a list of problems found."""
        problems: List[str] = []
        if not isinstance(self.funds, int):
            problems.append("system funds is not an integer")
        ids = [rider.rider_id for rider in self.riders]
        if len(ids) != len(set(ids)):
            problems.append("duplicate rider IDs detected")
        for rider in self.riders:
            if not (0 <= rider.x <= 800 and 0 <= rider.y <= 800):
                problems.append(f"rider #{rider.rider_id} is outside the map")
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    problems.append(
                        f"order #{order.id} is held by rider #{rider.rider_id} "
                        f"but records rider #{order.rider_id}")
        for order in OrderManager.active_orders():
            if order.rider_id is not None and self.get_rider(order.rider_id) is None:
                problems.append(f"order #{order.id} references missing rider #{order.rider_id}")
        return problems

    def save_state(self, path: Optional[str] = None) -> str:
        """Persist configuration, riders and order history as JSON.

        Args:
            path: Destination file; defaults to ``config["state_file"]``.

        Returns:
            The path that was written.

        Raises:
            OSError: If the file cannot be written.
        """
        path = path or str(self.config["state_file"])
        data = {
            "config": self.config,
            "system_funds": self.funds,
            "current_tick": self.tick_count,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [
                {
                    "rider_id": r.rider_id,
                    "x": r.x,
                    "y": r.y,
                    "speed": r.speed,
                    "state": r.state,
                    "order_ids": [o.id for o in r.orders],
                    "special_case": r.special_case,
                    "log": r.log,
                }
                for r in self.riders
            ],
            "orders": [o.to_dict() for o in OrderManager.all_orders()],
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        logger.info("State saved to %s", path)
        return path

"""Status display and ASCII map rendering."""

from __future__ import annotations

from typing import Any, Iterable, List, Optional, Sequence, Tuple

from path_planner import CELL_SIZE, MAP_SIZE

Point = Tuple[int, int]


class DisplayManager:
    """Builds text output for the status, order, rider and map views.

    The ASCII map uses one character per 50 units, so the 8x8 cell grid is
    drawn on a 17x17 canvas: even indices are streets (grid lines), odd
    indices are cell centres.  Restaurants are cell centres (``S``) and
    customers are cell centres (``C``).
    """

    STREET = "#"
    EMPTY = "."
    RIDER = "R"
    RESTAURANT = "S"
    CUSTOMER = "C"
    ROUTE = "*"
    UNITS_PER_CHAR = CELL_SIZE // 2
    CANVAS_SIZE = 2 * (MAP_SIZE // CELL_SIZE) + 1

    @classmethod
    def to_canvas(cls, value: float) -> int:
        """Convert a map coordinate to a canvas index."""
        index = int(round(value / cls.UNITS_PER_CHAR))
        return max(0, min(cls.CANVAS_SIZE - 1, index))

    @classmethod
    def canvas_pos(cls, point: Sequence[float]) -> Tuple[int, int]:
        """Return the ``(row, column)`` canvas index for an ``(x, y)`` point."""
        x, y = point[0], point[1]
        return cls.to_canvas(y), cls.to_canvas(x)

    def render_map(self, riders: Iterable[Any] = (), orders: Iterable[Any] = (),
                   route: Optional[List[Point]] = None) -> str:
        """Render the ASCII map with riders, orders and an optional route.

        Args:
            riders: Riders to mark with ``R``.
            orders: Orders whose restaurant (``S``) and customer (``C``)
                locations are shown while not completed.
            route: Optional list of points drawn as a connected line of ``*``.

        Returns:
            Multi-line string.
        """
        size = self.CANVAS_SIZE
        canvas = [
            [self.STREET if (row % 2 == 0 or col % 2 == 0) else self.EMPTY
             for col in range(size)]
            for row in range(size)
        ]
        if route:
            self._draw_route(canvas, route)
        for order in orders:
            if order.status == "completed":
                continue
            row, col = self.canvas_pos(order.res_pos)
            canvas[row][col] = self.RESTAURANT
            row, col = self.canvas_pos(order.cus_pos)
            canvas[row][col] = self.CUSTOMER
        for rider in riders:
            row, col = self.canvas_pos(rider.position)
            canvas[row][col] = self.RIDER

        lines = [" ".join(row) for row in canvas]
        header = "Map (1 char = 50 units; top-left = (0,0); x ->, y down)"
        legend = "Legend: R rider  S restaurant  C customer  # street  . cell  * route"
        return "\n".join([header, *lines, legend])

    def _draw_route(self, canvas: List[List[str]], route: List[Point]) -> None:
        """Draw connected line segments through ``route`` on the canvas."""
        first_row, first_col = self.canvas_pos(route[0])
        if canvas[first_row][first_col] in (self.STREET, self.EMPTY):
            canvas[first_row][first_col] = self.ROUTE
        for (x1, y1), (x2, y2) in zip(route, route[1:]):
            dx = x2 - x1
            dy = y2 - y1
            steps = max(1, max(abs(dx), abs(dy)) // 5)
            for step in range(steps + 1):
                x = round(x1 + dx * step / steps)
                y = round(y1 + dy * step / steps)
                row, col = self.canvas_pos((x, y))
                if canvas[row][col] in (self.STREET, self.EMPTY):
                    canvas[row][col] = self.ROUTE

    def render_status(self, funds: int, riders: Sequence[Any]) -> str:
        """Render the system snapshot: funds and every rider's details."""
        lines = [f"System funds: {funds}", f"Riders: {len(riders)}"]
        if not riders:
            lines.append("  (no riders recruited yet)")
        for rider in riders:
            order_ids = [o.id for o in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id}: position=({rider.x},{rider.y}) "
                f"state={rider.state_name} speed={rider.speed} "
                f"orders={len(order_ids)} order_ids={order_ids}"
                f"{' [special case]' if rider.special_case else ''}")
        return "\n".join(lines)

    def render_orders(self, orders: Sequence[Any]) -> str:
        """Render a table with details for every order."""
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4}  {'Start':>6}  {'Restaurant':<12}  {'Customer':<12}  "
                  f"{'Distance':>8}  {'Taken':>6}  {'Rider':>6}  Status")
        lines = [header, "-" * len(header)]
        for order in orders:
            restaurant = f"({order.res_pos[0]},{order.res_pos[1]})"
            customer = f"({order.cus_pos[0]},{order.cus_pos[1]})"
            taken = str(order.duration) if order.duration is not None else "-"
            rider = f"#{order.rider_id}" if order.rider_id is not None else "-"
            lines.append(
                f"{order.id:>4}  {order.start_tick:>6}  {restaurant:<12}  {customer:<12}  "
                f"{order.distance:>8.1f}  {taken:>6}  {rider:>6}  {order.status}")
        return "\n".join(lines)

    def render_rider_logs(self, riders: Sequence[Any]) -> str:
        """Render every recorded status change for all riders."""
        if not riders:
            return "No riders recruited yet."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} ({len(rider.log)} events):")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order = f"order #{entry['order_id']}" if entry["order_id"] is not None else "-"
                lines.append(
                    f"  tick {entry['tick']:>6} | {entry['event']:<20} | {order:<12} | "
                    f"pos ({entry['x']},{entry['y']}) | state {entry['state']}")
        return "\n".join(lines)

"""Command parsing, shortcut resolution, command history and input validation."""

from __future__ import annotations

import logging
from typing import Callable, List, Optional, Tuple

logger = logging.getLogger(__name__)

try:  # Enables arrow-key history on platforms that provide readline.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - Windows without pyreadline
    readline = None  # type: ignore[assignment]

MIN_COORDINATE = 0
MAX_COORDINATE = 800


class InputManager:
    """Reads user lines, resolves shortcut keys and validates coordinates."""

    ALIASES = {
        "start": "start", "s": "start",
        "status": "status", "st": "status",
        "orders": "orders", "o": "orders",
        "riders": "riders", "r": "riders",
        "map": "map", "m": "map",
        "tick": "tick", "t": "tick",
        "run": "run", "u": "run",
        "history": "history", "hist": "history",
        "help": "help", "h": "help", "?": "help",
        "quit": "quit", "exit": "quit", "q": "quit",
        "!!": "!!",
    }

    MAX_HISTORY = 200

    def __init__(self, reader: Callable[[str], str] = input) -> None:
        """Create the input manager.

        Args:
            reader: Function used to read a line (``input`` by default).
        """
        self._reader = reader
        self.history: List[str] = []

    def read_line(self, prompt: str = "dispatch> ") -> str:
        """Read one stripped line of user input."""
        return self._reader(prompt).strip()

    def parse(self, line: str) -> Tuple[str, List[str]]:
        """Split a line into a canonical command name and its arguments.

        Args:
            line: Raw user input.

        Returns:
            Tuple of (command name, argument tokens). The command name is
            empty if the line is blank.
        """
        tokens = line.split()
        if not tokens:
            return "", []
        word = tokens[0].lower()
        return self.ALIASES.get(word, word), tokens[1:]

    def record(self, line: str) -> None:
        """Store a command in the in-session history."""
        if not line or line == "!!":
            return
        if self.history and self.history[-1] == line:
            return
        self.history.append(line)
        if len(self.history) > self.MAX_HISTORY:
            del self.history[0]

    def last_command(self) -> Optional[str]:
        """Return the most recent recorded command, if any."""
        return self.history[-1] if self.history else None

    @staticmethod
    def parse_coordinates(tokens: List[str]) -> List[int]:
        """Parse four integer coordinates from command arguments.

        Accepts whitespace or comma separators and optional brackets.

        Args:
            tokens: Argument tokens.

        Returns:
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            ValueError: With a descriptive message if the input is invalid.
        """
        text = " ".join(tokens).replace(",", " ").replace("[", " ").replace("]", " ")
        parts = text.split()
        if len(parts) != 4:
            raise ValueError(
                "expected four integers: restaurant_x restaurant_y customer_x customer_y "
                f"(got {len(parts)} value(s))")
        values: List[int] = []
        labels = ("restaurant x", "restaurant y", "customer x", "customer y")
        for label, part in zip(labels, parts):
            try:
                value = int(part)
            except ValueError as exc:
                raise ValueError(f"{label} '{part}' is not an integer") from exc
            if not MIN_COORDINATE <= value <= MAX_COORDINATE:
                raise ValueError(
                    f"{label} {value} is out of range {MIN_COORDINATE}-{MAX_COORDINATE}")
            values.append(value)
        return values

    def prompt_coordinates(self) -> List[int]:
        """Interactively ask for the four coordinates and validate them."""
        line = self._reader(
            "Enter restaurant_x restaurant_y customer_x customer_y (0-800): ").strip()
        return self.parse_coordinates(line.split())

"""Command-line entry point for the Food Delivery Order Dispatch simulator.

Run with::

    python src/main.py [--config PATH] [--debug]

from __future__ import annotations

import argparse
import logging
import sys
from typing import Callable, Dict, List, Optional

import globalvar
from display import DisplayManager
from input_manager import InputManager
from order_model import Order
from path_planner import PathPlanner
from system import SystemManager, init_system

logger = logging.getLogger("dispatch")

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant then customer, 0-800).
                       Shortcut: s. Prompts if no coordinates are given.
  status               System funds, riders and their orders.       (st)
  orders               All orders: start, locations, distance, time. (o)
  riders               Every rider status change log.                (r)
  map                  ASCII map with riders and active orders.      (m)
  tick [n]             Advance the simulation by n ticks (default 1). (t)
  run                  Advance until all orders are completed.        (u)
  history              Show command history.                         (hist)
  !!                   Repeat the last command.
  help                 Show this help.                               (h, ?)
  quit / exit          Save state and exit.                          (q)
"""


class DispatchApp:
    """Interactive application that dispatches commands to the core modules."""

    def __init__(self, reader: Callable[[str], str] = input,
                 output: Callable[..., None] = print) -> None:
        """Create the application.

        Args:
            reader: Line reader (``input`` by default; injectable for tests).
            output: Output function (``print`` by default).
        """
        self.inputs = InputManager(reader)
        self.display = DisplayManager()
        self.out = output
        self.sm: Optional[SystemManager] = None
        self.commands: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "run": self.cmd_run,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    # ------------------------------------------------------------------ #
    # Lifecycle
    # ------------------------------------------------------------------ #
    def start(self, config_path: Optional[str] = None) -> None:
        """Initialise the system and print the welcome banner."""
        init_system(config_path)
        self.sm = SystemManager()
        self.out("Food Delivery Order Dispatch System")
        self.out(f"Funds: {self.sm.funds} | Map: 800x800 (8x8 cells) | Type 'help'.")

    def loop(self) -> None:
        """Run the read-evaluate loop until the system stops running."""
        while globalvar.get_value("system_running", False):
            try:
                line = self.inputs.read_line()
            except EOFError:
                self.out("")
                break
            except KeyboardInterrupt:
                self.out("")
                break
            if not line:
                continue
            if line == "!!":
                last = self.inputs.last_command()
                if last is None:
                    self.out("No previous command to repeat.")
                    continue
                self.out(f"Repeating: {last}")
                line = last
            self.inputs.record(line)
            command, args = self.inputs.parse(line)
            handler = self.commands.get(command)
            if handler is None:
                self.out(f"Unknown command '{command}'. Type 'help' for the command list.")
                continue
            try:
                handler(args)
            except ValueError as exc:
                self.out(f"Input error: {exc}")
            except RuntimeError as exc:
                self.out(f"Error: {exc}")
            except Exception:  # keep the session alive on unexpected errors
                logger.exception("Unexpected error while running '%s'", command)
                self.out("Unexpected internal error; system state has been preserved.")
            self._check_consistency()

    def shutdown(self) -> None:
        """Save state to disk and stop the main loop."""
        if self.sm is not None:
            try:
                path = self.sm.save_state()
                self.out(f"State saved to {path}")
            except OSError as exc:
                logger.error("Could not save state: %s", exc)
                self.out(f"Could not save state: {exc}")
        globalvar.set_value("system_running", False)
        self.out("Goodbye.")

    def _check_consistency(self) -> None:
        """Log any internal consistency problems."""
        if self.sm is None:
            return
        for problem in self.sm.check_consistency():
            logger.warning("Consistency check: %s", problem)

    # ------------------------------------------------------------------ #
    # Helpers
    # ------------------------------------------------------------------ #
    @staticmethod
    def _cell_label(point) -> str:
        """Return the ``(column,row)`` grid cell label for a centre point."""
        col, row = PathPlanner.cell_of(point[0], point[1])
        return f"({col},{row})"

    def _require_system(self) -> SystemManager:
        """Return the system manager or raise if not initialised."""
        if self.sm is None:
            raise RuntimeError("system is not initialised")
        return self.sm

    # ------------------------------------------------------------------ #
    # Commands
    # ------------------------------------------------------------------ #
    def cmd_start(self, args: List[str]) -> None:
        """Create and dispatch an order from four coordinates."""
        sm = self._require_system()
        coords = InputManager.parse_coordinates(args) if args else self.inputs.prompt_coordinates()
        result = sm.submit_order(coords)
        order: Order = result["order"]
        rider = result["rider"]

        self.out(f"Order #{order.id} created at tick {order.start_tick}.")
        self.out(f"  Restaurant: input {order.raw_res_pos} -> grid cell "

        self.out(f"  Customer:   input {order.raw_cus_pos} -> grid cell "
                 f"{self._cell_label(order.cus_pos)}, centre {order.cus_pos}")
        self.out(f"  Straight-line distance: {order.distance} units")
        for new_rider in result["new_riders"]:
            self.out(f"  New rider added: #{new_rider.rider_id} at {new_rider.position} "

        if rider is None:
            self.out("  No rider available: order remains pending (insufficient funds to recruit).")

        self.out(f"  Assigned to rider #{rider.rider_id} "

            self.out(f"  Rider #{rider.rider_id} is busy with {queued} earlier order(s); "

    def cmd_status(self, args: List[str]) -> None:

        self.out(self.display.render_status(sm.funds, sm.riders))
        self.out(f"Clock tick: {sm.tick_count} | Active orders: {len(sm.riders and [o for r in sm.riders for o in r.orders]) if sm.riders else 0}")

    def cmd_orders(self, args: List[str]) -> None:

        from order import OrderManager  # local import keeps module graph simple
        self._require_system()
        self.out(self.display.render_orders(OrderManager.all_orders()))

        sm = self._require_system()
        self.out(self.display.render_rider_logs(sm.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Print the ASCII map with riders and active orders."""
        from order import OrderManager
        sm = self._require_system()
        self.out(self.display.render_map(sm.riders, OrderManager.active_orders()))

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the clock by one or more ticks."""
        sm = self._require_system()
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError as exc:
                raise ValueError("tick count must be an integer") from exc
        if not 1 <= count <= 10000:
            raise ValueError("tick count must be between 1 and 10000")
        for _ in range(count):
            sm.tick()
        from order import OrderManager
        self.out(f"Advanced {count} tick(s). Clock: {sm.tick_count} | "
                 f"Active orders: {len(OrderManager.active_orders())}")

    def cmd_run(self, args: List[str]) -> None:
        """Advance until every order has been completed."""
        sm = self._require_system()
        elapsed = sm.run_until_idle()
        self.out(f"All orders completed after {elapsed} tick(s). Clock: {sm.tick_count}.")

    def cmd_history(self, args: List[str]) -> None:
        """Print the session command history."""
        if not self.inputs.history:
            self.out("History is empty.")
            return
        for index, line in enumerate(self.inputs.history, start=1):
            self.out(f"  {index:>3}  {line}")

    def cmd_help(self, args: List[str]) -> None:
        """Print the command help."""
        self.out(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the main loop; state is saved during shutdown."""
        globalvar.set_value("system_running", False)

import os
import re

from lottery_system import config
from lottery_system.models import User

_SEPARATOR_RE = re.compile(r"[,;，；]")


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self):
        self.source = None
        self.status = "EMPTY"
        self.load_mode = None

    def load_default_list(self):
        """Load the built-in default team building list."""
        users, errors = self._parse_lines(config.DEFAULT_LIST_TEXT.splitlines())
        self.source = "built-in default list"
        self.load_mode = "default"
        self.status = "LOADED" if users else "FAILED"
        return users, errors

    def load_custom_list(self, path):
        """Load a custom txt list from the given path."""
        users, errors = self._load_from_path(path)
        self.source = path
        self.load_mode = "custom"
        self.status = "LOADED" if users else "FAILED"
        return users, errors

    def _load_from_path(self, path):
        """Read and parse a txt file. Each line: 'Name,Department Group'."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines):
        users = []
        errors = []
        for line_no, raw in enumerate(lines, start=1):
            content = raw.strip()
            if not content:
                continue
            parts = [part.strip() for part in _SEPARATOR_RE.split(content)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{content}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users):
        """Return (valid_users, duplicates). The first occurrence of a name is kept."""
        seen = set()
        valid_users = []
        duplicates = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users):
        """Return a mapping of department name to number of members."""
        counts = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

import re

from lottery_system import config
from lottery_system.models import Prize

_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3, "fourth": 4, "fifth": 5}
_LEVEL_RE = re.compile(r"\s*(first|second|third|fourth|fifth)\b", re.IGNORECASE)


def infer_level(name):
    """Infer prize level from the leading word of the name (First = 1 ...)."""
    match = _LEVEL_RE.match(name or "")
    if match:
        return _LEVEL_WORDS[match.group(1).lower()]
    return 99


class PrizeManager:
    """Manages prizes and their lottery order. Starts with no prizes."""

    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        """Add the preset prize library. Returns a list of error messages."""
        errors = []
        for name, quantity, drawer in config.DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, None, drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if any(prize.name == name for prize in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")
        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(
            Prize(
                name=name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
                level=infer_level(name),
            )
        )

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        """Reorder prizes. order contains 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            items = list(order)
        except TypeError:
            return False
        if len(items) != count:
            return False
        if any(isinstance(item, bool) or not isinstance(item, int) for item in items):
            return False
        if set(items) != set(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in items]
        return True

"""Lottery rule engine."""

import random

from lottery_system import config
from lottery_system.models import FallbackRule


def _norm(text):
    return (text or "").strip().lower()


class LotteryManager:
    """Applies lottery rules and performs draws.

    Rules:
    - Previous winners are always excluded (anti-duplication).
    - Lottery person 'Li Bin' only excludes previous winners.
    - Members of a prize's avoidance groups remain eligible, but are only drawn
      after all other eligible members have been drawn.
    - Cheng Ru group protection: if the last two draws made by Chen Qixian / Jin Duo
      contained no Cheng Ru group member, the next draw by Ma Jiaqi is limited to
      Cheng Ru group members (overriding avoidance groups). The history is reset
      once Ma Jiaqi's round ends.
    """

    def __init__(self, users):
        self._users = list(users)
        self._winner_names = set()
        self.last_two_draw_results = []
        self.fallback_rule = FallbackRule()
        self._rng = random.Random()

    def get_remaining_users(self):
        return [user for user in self._users if user.name not in self._winner_names]

    def _fallback_active(self, drawer_norm):
        rule = self.fallback_rule
        if not rule.active or drawer_norm != _norm(config.FALLBACK_TARGET_DRAWER):
            return False
        limit = rule.consecutive_miss_limit
        history = self.last_two_draw_results
        if len(history) < limit:
            return False
        return not any(history[-limit:])

    def _get_eligible_pool(self, prize):
        """Return the users from which the next winner of this prize is drawn."""
        available = self.get_remaining_users()
        drawer = _norm(prize.associated_drawer)

        if drawer == _norm(config.SPECIAL_DRAWER_LI_BIN):
            return available

        if self._fallback_active(drawer):
            protected = _norm(self.fallback_rule.protected_group)
            protected_members = [
                user for user in available if _norm(user.department) == protected
            ]
            if protected_members:
                return protected_members

        avoid = {_norm(group) for group in prize.avoid_groups if _norm(group)}
        primary = [user for user in available if _norm(user.department) not in avoid]
        if primary:
            return primary
        return [user for user in available if _norm(user.department) in avoid]

    def has_eligible(self, prize):
        return bool(self._get_eligible_pool(prize))

    def _record_draw_result(self, hit_protected_group):
        limit = self.fallback_rule.consecutive_miss_limit
        self.last_two_draw_results.append(bool(hit_protected_group))
        self.last_two_draw_results = self.last_two_draw_results[-limit:]

    def draw_one(self, prize):
        """Draw one winner for the prize. Returns User or None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        winner = self._rng.choice(pool) if pool else None

        drawer = _norm(prize.associated_drawer)
        trigger_drawers = {_norm(name) for name in config.FALLBACK_TRIGGER_DRAWERS}
        if drawer in trigger_drawers:
            hit = winner is not None and _norm(winner.department) == _norm(
                self.fallback_rule.protected_group
            )
            self._record_draw_result(hit)

        if winner is not None:
            self._winner_names.add(winner.name)
        return winner

    def end_round(self, prize):
        """Called when a round finishes. Resets the fallback history after Ma Jiaqi's round."""
        if _norm(prize.associated_drawer) == _norm(config.FALLBACK_TARGET_DRAWER):
            self.last_two_draw_results = []

    def describe_rule(self, prize):
        """Human readable rule description for a round."""
        parts = []
        drawer = _norm(prize.associated_drawer)
        if not drawer:
            parts.append("No associated lottery person.")
        if drawer == _norm(config.SPECIAL_DRAWER_LI_BIN):
            parts.append(
                "Lottery person Li Bin: only previous winners are excluded; "
                "avoidance groups do not apply."
            )
        else:
            parts.append("Previous winners are excluded.")
            if prize.avoid_groups:
                parts.append(
                    "Avoidance group(s) "
                    + ", ".join(prize.avoid_groups)
                    + " remain eligible but are drawn after other members."
                )
            if self._fallback_active(drawer):
                parts.append(
                    f"Cheng Ru group protection active: pool limited to {config.CHENG_RU_GROUP} members."
                )
        if drawer in {_norm(name) for name in config.FALLBACK_TRIGGER_DRAWERS}:
            parts.append("Results feed the Cheng Ru group protection check.")
        return " ".join(parts)

"""Winner recording, summary and CSV export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import User


@dataclass
class Winner:
    user: User
    prize_name: str
    prize_level: int
    winning_order: int
    win_time: datetime

    @property
    def name(self):
        return self.user.name

    @property
    def department(self):
        return self.user.department


class ResultManager:
    """Keeps the winner list and exports it."""

    def __init__(self):
        self._winners = []
        self._counter = 0

    def add_winner(self, user, prize):
        self._counter += 1
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=getattr(prize, "level", 99),
            winning_order=self._counter,
            win_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self):
        return list(self._winners)

    def get_ordered_winners(self, prizes=None):
        """Winners sorted by prize level (high to low), then winning order."""
        return sorted(self._winners, key=lambda w: (w.prize_level, w.winning_order))

    def get_winners_by_prize(self, prizes):
        """Return {prize name: [Winner, ...]} in the order of the given prizes."""
        grouped = {prize.name: [] for prize in prizes}
        for winner in sorted(self._winners, key=lambda w: w.winning_order):
            grouped.setdefault(winner.prize_name, []).append(winner)
        return grouped

    def export_to_csv(self, prizes):
        """Export the full winner list to current_dir/awards_result_YYYYMMDD.txt."""
        now = datetime.now()
        path = os.path.join(os.getcwd(), f"awards_result_{now:%Y%m%d}.txt")
        ordered = self.get_ordered_winners(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in ordered:
                    writer.writerow(
                        [
                            winner.name,
                            winner.department,
                            winner.prize_name,
                            winner.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from lottery_system.services.list_manager import ListManager

SEPARATOR = "-" * 60


def ordinal(number):
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    def __init__(self, input_func=input, output_func=print):
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None
        self._paused = False
        self._running = True

    # ------------------------------------------------------------------ menus
    def run(self):
        self._print("Welcome to the Team Building Smart Lottery System")
        handlers = {
            "1": self._menu_load_list,
            "2": self._show_groups,
            "3": self._menu_prizes,
            "4": self._menu_reorder,
            "5": self._start_lottery,
            "6": self._menu_results,
            "0": self._exit,
        }
        try:
            while self._running:
                self._show_main_menu()
                choice = self._ask("Select an option: ").strip()
                handler = handlers.get(choice)
                if handler is None:
                    self._print("Invalid option, please try again.")
                    continue
                handler()
        except (EOFError, KeyboardInterrupt):
            self._print("\nExiting the system.")

    def _show_main_menu(self):
        self._print(SEPARATOR)
        self._print("1. Load participant list")
        self._print("2. View department groups")
        self._print("3. Configure prizes")
        self._print("4. Adjust prize order")
        self._print("5. Start lottery")
        self._print("6. View results / export")
        self._print("0. Exit")
        self._print(SEPARATOR)

    def _exit(self):
        self._running = False
        self._print("Goodbye.")

    def _ask(self, prompt):
        return self._input(prompt)

    def _ask_positive_int(self, prompt):
        while True:
            raw = self._ask(prompt).strip()
            try:
                value = int(raw)
            except ValueError:
                self._print("Please enter a valid integer.")
                continue
            if value <= 0:
                self._print("Please enter a positive integer.")
                continue
            return value

    # ------------------------------------------------------- list management
    def _menu_load_list(self):
        self._print("1. Built-in default list")
        self._print("2. Upload custom txt list (each line: Name,Department Group)")
        self._print("0. Cancel")
        choice = self._ask("Select a source: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter the path of the txt file: ").strip().strip('"')
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return
        self._finalize_list_load(users, errors)

    def _finalize_list_load(self, users, errors):
        valid_users, duplicates = self.list_manager.validate_users(users)

        if not users and errors:
            self._print("Load failed:")
            for error in errors:
                self._print(f"  {error}")
            return

        if errors:
            self._print("Invalid entries detected:")
            for error in errors:
                self._print(f"  {error}")
        if duplicates:
            self._print("Duplicate names detected (only the first occurrence is kept):")
            for user in duplicates:
                self._print(f"  Duplicate Name: {user.name} ({user.department})")

        if errors or duplicates:
            choice = self._ask(
                "Type 'r' to remove invalid data and keep the rest, "
                "or 't' to terminate the upload: "
            ).strip().lower()
            if choice != "r":
                self._print("Upload terminated. The previous list is unchanged.")
                return

        if not valid_users:
            self._print("No valid participants found. The previous list is unchanged.")
            return

        self.users = valid_users
        self.lottery = None
        self._print(f"Loaded {len(valid_users)} participant(s) from {self.list_manager.source}.")

    def _show_groups(self):
        if not self.users:
            self._print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._print("Department groups:")
        for index, (department, count) in enumerate(sorted(info.items()), start=1):
            self._print(f"  {index}. {department}: {count} member(s)")

    # ------------------------------------------------------- prize management
    def _menu_prizes(self):

    def _add_custom_prize(self):
        name = self._ask("Prize name: ").strip()
        quantity = self._ask_positive_int("Quantity: ")
        drawer = self._ask("Associated lottery person (optional, Enter to skip): ").strip() or None

        raw_groups = self._ask("Avoidance groups (comma-separated, Enter for none): ")

    def _show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) or "-"
            drawer = prize.associated_drawer or "-"
            self._print(
                f"{index}. {prize.name} | quantity: {prize.quantity} | "
                f"lottery person: {drawer} | avoid groups: {avoid}"
            )

    def _menu_reorder(self):
        self._show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = self._ask("Enter the new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in raw.split()]
        except ValueError:
            self._print("Invalid input: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Prize order updated.")
            self._show_prizes()
        else:
            self._print("Invalid order: every serial number must appear exactly once.")

    # ------------------------------------------------------ lottery process
    def _start_lottery(self):
        if not self.users:
            self._print("Please load the participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure prizes first.")
            return
        if self.result_manager.get_all_winners():
            choice = self._ask("Previous results will be cleared. Continue? (y/n): ").strip().lower()
            if choice != "y":
                return
            self.result_manager = ResultManager()

        self.lottery = LotteryManager(self.users)
        self._paused = False
        total = len(prizes)
        self._print(f"Starting lottery: {total} round(s), {len(self.users)} participant(s).")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)
            self.lottery.end_round(prize)

        self._print(SEPARATOR)
        self._print("Lottery finished.")
        self._show_summary(prizes)
        choice = self._ask("Export the winner list now? (y/n): ").strip().lower()
        if choice == "y":
            self._export(prizes)

    def _run_round(self, index, total, prize):
        self._print(SEPARATOR)
        self._print(f"Round {index}/{total}: {prize.name}")
        self._print(f"Quantity: {prize.quantity}")
        self._print(f"Rule: {self.lottery.describe_rule(prize)}")
        self._print(SEPARATOR)

        self._paused = False
        round_winners = []
        for position in range(1, prize.quantity + 1):
            command = self._await_draw_command()
            if command == "terminate":
                self._print("Round terminated by operator. Remaining winners are skipped.")
                break
            if not self.lottery.has_eligible(prize):
                self._print("No eligible participants remain for this prize. Round ends.")
                break
            self._print(f"Currently drawing the {ordinal(position)} winner...")
            user = self.lottery.draw_one(prize)
            if user is None:
                self._print("No eligible participants remain for this prize. Round ends.")
                break
            winner = self.result_manager.add_winner(user, prize)
            round_winners.append(winner)
            self._print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self._print(f"Round {index} results:")
        if not round_winners:
            self._print("  No winners in this round.")
        for winner in round_winners:
            self._print(
                f"  {winner.winning_order}. {winner.name} ({winner.department}) - {winner.prize_name}"
            )

    def _await_draw_command(self):
        """Wait for the operator. Returns 'draw' or 'terminate'."""
        while True:
            if self._paused:
                prompt = "Paused. Type 'continue' to resume or 'terminate' to end this round: "
            else:
                prompt = "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
            command = self._ask(prompt).strip().lower()

            if command == "terminate":
                self._paused = False
                return "terminate"
            if command == "pause":
                if self._paused:
                    self._print("Lottery is already paused.")
                else:
                    self._paused = True
                    self._print("Lottery paused.")
                continue
            if command == "continue":
                if not self._paused:
                    self._print("Lottery is not paused.")
                    continue
                self._paused = False
                self._print("Lottery resumed.")
                return "draw"
            if command == "":
                if self._paused:
                    self._print("Lottery is paused. Type 'continue' to resume.")
                    continue
                return "draw"
            self._print(f"Unknown command '{command}'. Use Enter, 'pause', 'continue' or 'terminate'.")

    # ------------------------------------------------------------- results
    def _menu_results(self):
        prizes = self.prize_manager.get_prizes()
        self._show_summary(prizes)
        if self.result_manager.get_all_winners():
            choice = self._ask("Export the winner list to a txt/CSV file? (y/n): ").strip().lower()
            if choice == "y":
                self._export(prizes)

    def _show_summary(self, prizes):
        if not self.result_manager.get_all_winners():
            self._print("No winners recorded yet.")
            return
        sorted_prizes = sorted(prizes, key=lambda prize: prize.level)
        grouped = self.result_manager.get_winners_by_prize(sorted_prizes)
        self._print("=== Winner List (by prize level, high to low) ===")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            self._print(f"{prize_name}:")
            for winner in winners:
                self._print(
                    f"  {winner.winning_order}. {winner.name} ({winner.department}) "
                    f"- {winner.win_time:%Y-%m-%d %H:%M:%S}"
                )

    def _export(self, prizes):
        success, message = self.result_manager.export_to_csv(prizes)
        self._print(message)

def normalize(text: str | None) -> str:
    """Normalize a name or department for case-insensitive, whitespace-tolerant comparison."""
    if text is None:
        return ""
    return " ".join(str(text).split()).casefold()

class ListManager:
    """Loads, validates and summarizes participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        if not isinstance(path, str) or not path.strip():
            return [], ["File Error: no file path provided"]
        if not path.strip().lower().endswith(".txt"):
            return [], ["File Error: only .txt files are supported"]
        return self._load_from_path(path.strip())

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a list file, collecting format errors per line."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = text.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{text}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            name = " ".join(parts[0].split())
            department = " ".join(parts[1].split())
            users.append(User(name=name, department=department))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique entries and duplicate entries.

        The first occurrence of each name is kept as valid; later occurrences
        are returned as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"Cannot load default prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the lottery sequence."""
        clean_name = " ".join(str(name or "").split())
        if not clean_name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        key = normalize(clean_name)
        if any(normalize(p.name) == key for p in self._prizes):
            raise ValueError(f"prize '{clean_name}' already exists")

        groups: list[str] = []
        for group in avoid_groups or []:
            group_name = " ".join(str(group).split())
            if group_name and group_name not in groups:
                groups.append(group_name)

        drawer = " ".join(str(associated_drawer).split()) if associated_drawer else None
        if not drawer:
            drawer = None

        prize = Prize(name=clean_name, quantity=quantity, avoid_groups=groups,
                      associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        n = len(self._prizes)
        try:
            indexes = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(indexes) != n or sorted(indexes) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in indexes]
        return True

from __future__ import annotations

import random

from ..models import (

    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    normalize,

class LotteryManager:
    """Runs draws for a set of users while enforcing lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self._rng = rng or random.Random()
        self.winners: list[User] = []
        # Each entry: whether the result of a consecutive lottery person
        # (Chen Qixian or Jin Duo) contained any Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []

    # ----- helpers -----
    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return normalize(user.department) == normalize(CHENG_RU_GROUP)

    def _available_users(self) -> list[User]:
        """Users who have not won yet (anti-duplication)."""
        return [u for u in self._users if normalize(u.name) not in self._winner_keys]

    def _fallback_active(self) -> bool:
        """True when the last two lottery persons' results contained no Cheng Ru members."""
        return len(self.last_two_draw_results) >= 2 and not any(self.last_two_draw_results[-2:])

    def is_winner(self, user: User) -> bool:
        return normalize(user.name) in self._winner_keys

    # ----- rule engine -----
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users eligible to win the given prize in the current draw."""
        available = self._available_users()
        drawer = normalize(prize.associated_drawer) if prize.associated_drawer else ""

        # Special rule: Li Bin only filters previous winners.
        if drawer == normalize(LI_BIN):
            return available

        # Fallback (Cheng Ru group protection) for Ma Jiaqi's prizes.
        if drawer == normalize(MA_JIAQI) and self._fallback_active():
            protected = [u for u in available if self._is_cheng_ru(u)]
            if protected:
                return protected

        avoid = {normalize(g) for g in prize.avoid_groups}
        return [u for u in available if normalize(u.department) not in avoid]

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winner_keys.add(normalize(user.name))
        self.winners.append(user)
        return user

    def finish_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Record the outcome of a finished round for fallback tracking."""
        drawer = normalize(prize.associated_drawer) if prize.associated_drawer else ""
        if drawer in (normalize(CHEN_QIXIAN), normalize(JIN_DUO)):
            contains_cheng_ru = any(self._is_cheng_ru(u) for u in round_winners)
            self.last_two_draw_results.append(contains_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

from __future__ import annotations

import csv
import io
import os
from dataclasses import dataclass
from datetime import datetime

from ..models import Prize, User


@dataclass
class Winner:
    """A winning record: the winner and the name of the prize won."""

    user: User
    prize_name: str


class ResultManager:
    """Keeps the list of winners and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        prize_name = getattr(prize, "name", prize)
        self._winners.append(Winner(user=user, prize_name=str(prize_name)))

    def get_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map each prize name (in the given order) to its winners."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list as CSV into the current directory."""
        try:
            grouped = self.get_winners_by_prize(prizes)
            buffer = io.StringIO()
            writer = csv.writer(buffer, lineterminator="\n")
            writer.writerow(["Name", "Department", "Prize"])
            for prize in prizes:
                for winner in grouped.get(prize.name, []):
                    writer.writerow([winner.user.name, winner.user.department, winner.prize_name])

            file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
            path = os.path.join(os.getcwd(), file_name)
            with open(path, "w", encoding="utf-8", newline="") as handle:
                handle.write(buffer.getvalue())
            return True, f"Results successfully exported to {path}"
        except OSError as exc:
            return False, f"Export failed: {exc}"

def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def describe_rule(prize: Prize) -> str:
    """Human-readable rule description for a prize round."""
    parts = ["previous winners are excluded"]
    drawer = prize.associated_drawer
    if drawer:
        parts.insert(0, f"lottery person: {drawer}")
    if drawer and normalize(drawer) == normalize(LI_BIN):
        parts.append("special rule: avoidance groups are not applied")
    else:
        if prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if drawer and normalize(drawer) == normalize(MA_JIAQI):
            parts.append("Cheng Ru Group protection fallback may apply")
    return "; ".join(parts)


class LotteryCLI:
    """Interactive console application."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery_completed = False

    # ----- io helpers -----
    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    def _say(self, message: str = "") -> None:
        self._print(message)

    def _confirm(self, prompt: str) -> bool:
        answer = self._ask(f"{prompt} (y/n): ").strip().lower()
        return answer in ("y", "yes")

    # ----- main loop -----
    def run(self) -> None:
        self._say("=== Company Team Building Smart Lottery System ===")
        while True:
            self._show_menu()
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self._load_default_list()
            elif choice == "2":
                self._load_custom_list()
            elif choice == "3":
                self._view_groups()
            elif choice == "4":
                self._load_default_prizes()
            elif choice == "5":
                self._add_prize()
            elif choice == "6":
                self._view_prizes()
            elif choice == "7":
                self._reorder_prizes()
            elif choice == "8":
                self._start_lottery()
            elif choice == "9":
                self._view_results()
            elif choice == "10":
                self._export_results()
            elif choice == "0":
                self._say("Goodbye!")
                return
            else:
                self._say("Invalid option, please try again.")

    def _show_menu(self) -> None:
        self._say("")
        self._say("---------------- Main Menu ----------------")
        self._say("1. Load built-in default team building list")
        self._say("2. Upload custom txt list (Name,Department Group)")
        self._say("3. View department groups and headcount")
        self._say("4. Load preset prize library")
        self._say("5. Add custom prize")
        self._say("6. View prize list")
        self._say("7. Set prize lottery sequence")
        self._say("8. Start lottery")
        self._say("9. View results")
        self._say("10. Export results to txt (CSV format)")
        self._say("0. Exit")

    # ----- list management -----
    def _load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._apply_loaded_users(users, errors)

    def _load_custom_list(self) -> None:
        path = self._ask("Enter the path of the txt file: ").strip().strip('"')
        users, errors = self.list_manager.load_custom_list(path)
        self._apply_loaded_users(users, errors)

    def _apply_loaded_users(self, users: list, errors: list[str]) -> None:
        if errors:
            self._say("Problems found in the list:")
            for error in errors:
                self._say(f"  - {error}")
            if not self._confirm("Remove invalid entries and continue?"):
                self._say("Upload terminated. Current list is unchanged.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self._say("Duplicate names detected:")
            for user in duplicates:
                self._say(f"  - {user.name} ({user.department})")
            if not self._confirm("Remove duplicate entries and continue?"):
                self._say("Upload terminated. Current list is unchanged.")
                return

        if not valid:
            self._say("No valid users to load. Current list is unchanged.")
            return

        self.users = valid
        self._say(f"Loaded {len(valid)} users successfully.")

    def _view_groups(self) -> None:
        if not self.users:
            self._say("No list loaded yet.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._say("Current department groups:")
        for group, count in info.items():
            self._say(f"  - {group}: {count} people")
        self._say(f"Total: {len(self.users)} people in {len(info)} groups")

    # ----- prize management -----
    def _load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            for error in errors:
                self._say(f"  - {error}")
        else:
            self._say("Preset prize library loaded.")

    def _select_avoid_groups(self) -> list[str]:
        if not self.users:
            self._say("No list loaded; no avoidance groups can be selected.")
            return []
        groups = list(self.list_manager.get_group_info(self.users).keys())
        self._say("Available groups:")
        for index, group in enumerate(groups, start=1):
            self._say(f"  {index}. {group}")
        raw = self._ask("Select avoidance groups by number, comma-separated (Enter for none): ").strip()
        selected: list[str] = []
        for token in raw.replace(" ", "").split(","):
            if not token:
                continue
            if not token.isdigit() or not 1 <= int(token) <= len(groups):
                self._say(f"Ignoring invalid selection: '{token}'")
                continue
            group = groups[int(token) - 1]
            if group not in selected:
                selected.append(group)
        return selected

    def _add_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        if not name:
            self._say("Prize name must not be empty.")
            return
        quantity_text = self._ask("Quantity: ").strip()
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._say("Quantity must be an integer.")
            return
        if quantity <= 0:
            self._say("Quantity must be positive.")
            return
        drawer = self._ask("Associated lottery person (Enter for none): ").strip() or None
        avoid_groups = self._select_avoid_groups()
        try:
            self.prize_manager.add_prize(name, quantity, avoid_groups, drawer)
            self._say(f"Prize '{name}' added.")
        except ValueError as exc:
            self._say(f"Failed to add prize: {exc}")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self._say("Prize sequence:")
        for index, prize in enumerate(prizes, start=1):
            drawer = prize.associated_drawer or "None"
            avoid = ", ".join(prize.avoid_groups) or "None"
            self._say(f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                      f"Drawer: {drawer} | Avoid groups: {avoid}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self._view_prizes()
        raw = self._ask("Enter new order as serial numbers (e.g. 2 1 3): ").strip()
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(t) for t in tokens]
        except ValueError:
            self._say("Invalid input; sequence unchanged.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._say("Prize sequence updated.")
        else:
            self._say("Invalid order; each serial number must be used exactly once.")

    # ----- lottery process -----
    def _start_lottery(self) -> None:
        if not self.users:
            self._say("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("Please configure prizes first.")
            return
        if self.result_manager.get_winners():
            if not self._confirm("Previous results will be cleared. Start a new lottery?"):
                return

        self.result_manager = ResultManager()
        self.lottery_completed = False
        lottery = LotteryManager(self.users)

        self._say("")
        self._say(f"Lottery started: {len(prizes)} rounds.")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, prize, lottery)

        self.lottery_completed = True
        self._say("")
        self._say("All rounds completed.")
        self._show_summary(prizes)

    def _run_round(self, index: int, prize: Prize, lottery: LotteryManager) -> None:
        self._say("")
        self._say(f"===== Round {index}: {prize.name} =====")
        self._say(f"Quantity: {prize.quantity}")
        self._say(f"Rule: {describe_rule(prize)}")

        round_winners = []
        drawn = 0
        while drawn < prize.quantity:
            self._say(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            command = self._ask("Press Enter to draw, or type 'pause' / 'terminate': ").strip().lower()

            if command == "terminate":
                self._say("Round terminated manually; skipping the rest of this round.")
                break
            if command == "pause":
                if not self._handle_pause():
                    self._say("Round terminated manually; skipping the rest of this round.")
                    break
                continue
            if command not in ("", "continue"):
                self._say("Unknown command.")
                continue

            user = lottery.draw(prize)
            if user is None:
                self._say("No eligible candidates remain for this prize. Round ends.")
                break
            drawn += 1
            round_winners.append(user)
            self.result_manager.add_winner(user, prize)
            self._say(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        lottery.finish_round(prize, round_winners)

        self._say(f"Round {index} results (winning order):")
        if round_winners:
            for position, user in enumerate(round_winners, start=1):
                self._say(f"  {position}. {user.name} ({user.department})")
        else:
            self._say("  No winners in this round.")

    def _handle_pause(self) -> bool:
        """Pause the lottery. Returns True to resume, False to terminate the round."""
        self._say("Lottery paused.")
        while True:
            command = self._ask("Type 'continue' to resume or 'terminate' to end this round: ").strip().lower()
            if command == "continue":
                self._say("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            self._say("Unknown command.")

    def _show_summary(self, prizes: list[Prize]) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self._say("")
        self._say("===== Complete Winner List =====")
        for prize in prizes:
            self._say(f"{prize.name}:")
            winners = grouped.get(prize.name, [])
            if not winners:
                self._say("  (no winners)")
            for winner in winners:
                self._say(f"  - {winner.user.name} | {winner.user.department} | {winner.prize_name}")

    # ----- results -----
    def _view_results(self) -> None:
        if not self.result_manager.get_winners():
            self._say("No results yet.")
            return
        self._show_summary(self.prize_manager.get_prizes())

    def _export_results(self) -> None:
        if not self.result_manager.get_winners():
            self._say("No results to export yet.")
            return
        success, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self._say(message)


def main() -> None:
    cli = LotteryCLI()
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting the system.")


if __name__ == "__main__":

_SEPARATOR_PATTERN = re.compile(r"[,\t，]")


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'name,department'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = [part.strip() for part in _SEPARATOR_PATTERN.split(text)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{text}' "
                    f"(expected 'name,department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates).

        The first occurrence of each name is kept as valid; later occurrences
        of the same name are reported as duplicates.
        """
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of people (first-seen order)."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.models import CHEN_QIXIAN, JIN_DUO, LI_BIN, MA_JIAQI, Prize

DEFAULT_PRIZE_PRESETS: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 10, [], None),
    ("Second Prize - Shopping Cards", 5, [], CHEN_QIXIAN),
    ("First Prize - 1 Day Annual Leave", 2, [], LI_BIN),
    ("Technology Department Exclusive Coupon", 3, ["Product Department"], JIN_DUO),
    ("Cheng Ru Group Special Gift", 2, ["Technology Department"], MA_JIAQI),

class PrizeManager:
    """Manages the ordered list of prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages (empty on success)."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZE_PRESETS:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize to the end of the lottery sequence.

        Raises ValueError if the input is invalid.
        """
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")

        groups: list[str] = []
        for group in avoid_groups or []:
            text = (group or "").strip()
            if text and text not in groups:
                groups.append(text)

        drawer = (associated_drawer or "").strip() or None
        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order.

        Returns True on success, False if `order` is not a permutation of 1..n.
        """
        n = len(self._prizes)
        try:
            if len(order) != n or sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)


class LotteryManager:
    """Performs individual draws while enforcing all lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        self._winners: list[User] = []
        self._rng = rng if rng is not None else random.Random()
        # Results (True = contained a Cheng Ru Group member) of the most recent
        # draws by Chen Qixian / Jin Duo, at most the last two entries.
        self.last_two_draw_results: list[bool] = []

    @property
    def winners(self) -> list[User]:
        """All winners drawn so far, in winning order."""
        return list(self._winners)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return user.department.strip().lower() == CHENG_RU_GROUP.lower()

    def is_fallback_active(self) -> bool:
        """True when the last two Chen/Jin draws both had no Cheng Ru Group winner."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the candidates eligible for the next draw of `prize`."""
        # Basic filter: exclude all previous winners (anti-duplication).
        available = [user for user in self._users if user.name not in self._winner_names]
        drawer = (prize.associated_drawer or "").strip()

        # Fallback (Cheng Ru Group protection): forced pool, overrides avoid groups.
        if drawer == MA_JIAQI and self.is_fallback_active():
            return [user for user in available if self._is_cheng_ru(user)]

        # Special rule for Li Bin: relaxed filtering, avoid groups are not applied.
        if drawer == LI_BIN:
            return available

        # Basic filter: exclude members of the prize's avoidance groups.
        avoid = {group.strip().lower() for group in prize.avoid_groups}
        return [user for user in available if user.department.strip().lower() not in avoid]

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for `prize`. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_names.add(winner.name)
        self._winners.append(winner)
        return winner

    def finish_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Update fallback state after a prize round has finished."""
        drawer = (prize.associated_drawer or "").strip()
        if drawer in (CHEN_QIXIAN, JIN_DUO):
            hit = any(self._is_cheng_ru(user) for user in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == MA_JIAQI:
            # The Ma Jiaqi round consumes the fallback state.
            self.last_two_draw_results = []

"""Winner recording, grouping, and export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    sequence: int


def _level_rank(prize_name: str) -> int:
    """Lower rank means higher prize level. Unrecognized names rank last."""
    lowered = prize_name.lower()
    for rank, keyword in enumerate(("first", "second", "third"), start=1):
        if keyword in lowered:
            return rank
    return 99


class ResultManager:
    """Keeps the complete list of winners for the current lottery."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._sequence = 0

    def add_winner(self, user: User, prize: Prize) -> None:
        self._sequence += 1
        self._winners.append(Winner(user=user, prize_name=prize.name, sequence=self._sequence))

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize, ordered by prize level (high to low),
        then by winning time within the same prize."""
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda item: (_level_rank(item[1].name), item[0]))
        result: dict[str, list[Winner]] = {}
        for _, prize in indexed:
            result[prize.name] = sorted(
                (winner for winner in self._winners if winner.prize_name == prize.name),
                key=lambda winner: winner.sequence,
            )
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        grouped = self.get_winners_by_prize(prizes)
        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8-sig", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize Name", "Name", "Department", "Winning Order"])
                for prize_name, winners in grouped.items():
                    for order, winner in enumerate(winners, start=1):
                        writer.writerow(
                            [prize_name, winner.user.name, winner.user.department, order]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

class InputReader:
    """Reads stdin in a background thread so lottery commands can be polled during draws."""

    def __init__(self, stream=None) -> None:
        self._stream = stream if stream is not None else sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._thread = threading.Thread(target=self._pump, name="stdin-reader", daemon=True)
        self._thread.start()

    def _pump(self) -> None:
        try:
            for line in self._stream:
                self._queue.put(line.rstrip("\r\n"))
        except (OSError, ValueError):
            pass
        self._queue.put(_EOF)

    def read_line(self, prompt: str = "") -> str:
        """Blocking read of one line (stripped)."""
        if prompt:
            print(prompt, end="", flush=True)
        item = self._queue.get()
        if item is _EOF:
            self._queue.put(_EOF)
            raise EOFError("Input closed")
        return item.strip()

    def poll(self) -> str | None:
        """Non-blocking read. Returns None when no input is pending."""
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        if item is _EOF:
            self._queue.put(_EOF)
            raise EOFError("Input closed")
        return item.strip()


def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"

    def _handle_commands(self) -> bool:
        """Process pending control commands. Returns True if the current round should end."""
        while True:
            command = self._reader.poll()
            if command is None:
                return False
            command = command.lower()
            if not command:
                continue
            if command == "pause":
                print("[PAUSED] Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    response = self._reader.read_line().lower()
                    if response == "continue":
                        print("[RESUMED]")
                        break
                    if response == "terminate":
                        return True
                    print("Type 'continue' or 'terminate'.")
            elif command == "terminate":
                return True
            elif command == "continue":
                print("Lottery is not paused.")
            else:
                print(f"Unknown command '{command}'. Use 'pause' or 'terminate'.")

    def _describe_rules(self, prize: Prize, lottery: LotteryManager) -> str:
        parts: list[str] = []
        drawer = prize.associated_drawer
        parts.append(f"Lottery person: {drawer}" if drawer else "Lottery person: none")
        if drawer == LI_BIN:
            parts.append("relaxed filtering (avoid groups not applied)")
        else:
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            parts.append(f"avoid groups: {avoid}")
        parts.append("previous winners excluded")
        if drawer == MA_JIAQI and lottery.is_fallback_active():
            parts.append("Cheng Ru Group protection ACTIVE: only Cheng Ru Group members eligible")
        return "; ".join(parts)

    def _run_round(self, index: int, total: int, prize: Prize, lottery: LotteryManager) -> None:
        print(f"\n--- Round {index}/{total}: {prize.name} ---")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {self._describe_rules(prize, lottery)}")
        answer = self._ask("Press Enter to start this round (type 'terminate' to skip): ")
        if answer.lower() == "terminate":
            print("Round skipped.")
            return

        print("Lottery in progress. Commands: 'pause', 'continue', 'terminate'.")
        round_winners: list[User] = []
        terminated = False
        for draw_no in range(1, prize.quantity + 1):
            if self._handle_commands():
                terminated = True
                break
            print(f"Currently drawing the {_ordinal(draw_no)} winner...", flush=True)
            time.sleep(DRAW_DELAY_SECONDS)
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize. Round ends early.")
                break
            round_winners.append(user)
            self._results.add_winner(user, prize)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        lottery.finish_round(prize, round_winners)
        if terminated:
            print("Round terminated manually. Remaining winners skipped.")

        print(f"\nRound {index} results ({len(round_winners)} winner(s)):")
        if not round_winners:
            print("  (none)")
        for position, user in enumerate(round_winners, start=1):
            print(f"  {position}. {user.name} ({user.department})")

    def _show_summary(self, prizes: list[Prize]) -> None:
        grouped = self._results.get_winners_by_prize(prizes)
        print("\n=== Final Winner List (by prize level) ===")
        for prize_name, winners in grouped.items():
            print(f"\n[{prize_name}] ({len(winners)} winner(s))")
            if not winners:
                print("  (none)")
            for position, winner in enumerate(winners, start=1):
                print(f"  {position}. {winner.user.name} ({winner.user.department})")

    # -------------------------------------------------------------- export
    def _export_results(self, prizes: list[Prize]) -> None:
        _, message = self._results.export_to_csv(prizes)
        print(message)

    def _export_last_results(self) -> None:
        if self._last_prizes is None:
            print("No lottery has been run yet.")
            return
        self._export_results(self._last_prizes)


def main() -> None:
    """Application entry point."""
    try:
        LotterySystemCLI().run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting lottery system.")

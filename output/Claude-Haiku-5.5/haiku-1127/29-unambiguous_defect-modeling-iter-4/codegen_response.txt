"""Grid-based path planning for the 800x800 map.

Coordinate model:
    * The map spans 0..800 on both axes with 100-unit cells (8x8 cells).
    * Points whose x or y is a multiple of 100 lie on a street.
    * Points where both coordinates are multiples of 100 are intersections
      (restaurant and customer locations are always intersections).
    * Riders may stand anywhere on a street, including mid-block positions
      that are multiples of 50.
"""

import random


class PathPlanner:
    """Plans street-following routes on the grid.

    The planner prefers streets: every route segment keeps the traveller on
    a grid line, and moves are either purely horizontal or purely vertical.
    """

    prefers_streets = True
    MAP_SIZE = 800
    CELL_SIZE = 100
    SPAWN_STEP = 50

    @staticmethod
    def is_on_street(x, y):
        """Return True if the point ``(x, y)`` lies on a street grid line."""
        return x % PathPlanner.CELL_SIZE == 0 or y % PathPlanner.CELL_SIZE == 0

    @staticmethod
    def next_step(x, y, tx, ty):
        """Return the next single-unit position moving from ``(x, y)`` to ``(tx, ty)``.

        The rule keeps the traveller on streets:
            * On a vertical street (x is a multiple of 100) move vertically
              until the target's y is reached ("directly above or below" case).
            * Otherwise move horizontally along the current horizontal street.
            * Finally move vertically along the target's vertical street.

        Args:
            x, y: Current position.
            tx, ty: Target position (an intersection).

        Returns:
            Tuple ``(nx, ny)`` of the next position. Returns ``(x, y)`` when
            already at the target.
        """
        if (x, y) == (tx, ty):
            return x, y
        cell = PathPlanner.CELL_SIZE
        if x % cell == 0 and y != ty:
            return x, y + (1 if ty > y else -1)
        if x != tx:
            return x + (1 if tx > x else -1), y
        return x, y + (1 if ty > y else -1)

    def plan(self, start, goal):
        """Plan a street-following route from ``start`` to ``goal``.

        Args:
            start: Tuple ``(x, y)`` of the start point.
            goal: Tuple ``(x, y)`` of the destination intersection.

        Returns:
            List of waypoints (tuples) including ``start`` and ``goal``.
        """
        sx, sy = start
        gx, gy = goal
        if sx == gx or sy == gy:
            return self._dedupe([tuple(start), tuple(goal)])
        if sx % self.CELL_SIZE == 0:
            corner = (sx, gy)
        else:
            corner = (gx, sy)
        return self._dedupe([tuple(start), corner, tuple(goal)])

    def plan_route(self, start, stops):
        """Plan a continuous route through several destinations in order.

        Args:
            start: Starting point ``(x, y)``.
            stops: Ordered list of destination points.

        Returns:
            List of waypoints covering the whole multi-stop route.
        """
        points = [tuple(start)]
        current = tuple(start)
        for stop in stops:
            segment = self.plan(current, stop)
            points.extend(segment[1:])
            current = tuple(stop)
        return points

    @staticmethod
    def path_length(points):
        """Return the total Manhattan length of a waypoint list."""
        total = 0
        for (ax, ay), (bx, by) in zip(points, points[1:]):
            total += abs(bx - ax) + abs(by - ay)
        return total

    @staticmethod
    def estimate_ticks(distance, speed):
        """Return the number of ticks needed to cover ``distance`` at ``speed``.

        Args:
            distance: Route length in map units.
            speed: Rider speed in units per tick.

        Returns:
            Integer number of ticks (ceiling division), 0 if distance is 0.
        """
        if speed <= 0 or distance <= 0:
            return 0
        return -(-distance // speed)

    @staticmethod
    def random_road_point(rng=random):
        """Return a random integer point located on a road.

        Args:
            rng: Random number generator exposing ``random``, ``randint``
                and ``randrange``.

        Returns:
            Tuple ``(x, y)`` on a street.
        """
        step = PathPlanner.SPAWN_STEP
        cell = PathPlanner.CELL_SIZE
        max_block = PathPlanner.MAP_SIZE // cell
        if rng.random() < 0.5:
            return rng.randint(0, max_block) * cell, rng.randrange(0, PathPlanner.MAP_SIZE + 1, step)
        return rng.randrange(0, PathPlanner.MAP_SIZE + 1, step), rng.randint(0, max_block) * cell

    @staticmethod
    def _dedupe(points):
        """Remove consecutive duplicate waypoints."""
        result = []
        for point in points:
            if not result or result[-1] != point:
                result.append(point)
        return result

"""Rider model: state machine, movement, and event logging."""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger("dispatch.rider")

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_SPEED = 50
DEFAULT_DELIVERY_FEE = 30


def _config_value(key, default):
    """Read a configuration value, falling back to ``default``."""
    config = globalvar.get_value("config") or {}
    return config.get(key, default)


def _current_tick():
    """Return the current simulated tick."""
    return globalvar.get_value("tick", 0)


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    Attributes:
        rider_id (int): Unique identifier.
        x, y (int): Current coordinates.
        speed (int): Movement speed in units per tick.
        state (int): 0 = idle, 1 = picking up, 2 = delivering.
        orders (list): Assigned orders; ``orders[0]`` is the active one.
        special_case (bool): True while directly above/below the target.
        log (list[dict]): Status change records with keys ``time``,
            ``event``, ``order_id``, ``x``, ``y``, ``state`` and ``rider_id``.
    """

    def __init__(self, rider_id):
        """Create a rider at a random road position in idle state.

        Args:
            rider_id: Unique integer identifier for the rider.
        """
        self.rider_id = rider_id
        self.x, self.y = PathPlanner.random_road_point()
        self.speed = int(_config_value("rider_speed", DEFAULT_SPEED))
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    @property
    def position(self):
        """Return the current position as ``(x, y)``."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Return a human readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    def _record(self, event, order_id):
        """Append a status change entry to the rider log."""
        entry = {
            "time": _current_tick(),
            "event": event,
            "order_id": order_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "rider_id": self.rider_id,
        }
        self.log.append(entry)
        logger.debug("Rider %s t=%s %s order=%s", self.rider_id, entry["time"], event, order_id)

    def move(self):
        """Advance the rider by one tick.

        Handles the state transitions:
            * Idle -> Picking up when an order is assigned ("accepted_order").
            * Picking up -> Delivering on reaching the restaurant
              ("arrived_at_merchant").
            * Delivering -> Idle on reaching the customer
              ("delivered_to_customer").
        """
        if not self.orders:
            self.state = STATE_IDLE
            self.special_case = False
            return

        order = self.orders[0]
        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP
            order.status = "picking_up"
            self._record("accepted_order", order.id)

        target = order.restaurant_pos if self.state == STATE_PICKING_UP else order.customer_pos
        self._advance_toward(target)

        if self.position != target:
            return

        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.status = "delivering"
            self._record("arrived_at_merchant", order.id)
        else:
            self._deliver(order)

    def _advance_toward(self, target):
        """Move up to ``speed`` units along the street grid toward ``target``.

        Riders never block each other, so movement cannot deadlock. A guard
        stops movement if the planner ever reports no progress.
        """
        tx, ty = target
        budget = self.speed
        while budget > 0 and self.position != target:
            nx, ny = PathPlanner.next_step(self.x, self.y, tx, ty)
            if (nx, ny) == self.position:
                logger.warning("Rider %s made no progress toward %s", self.rider_id, target)
                break
            self.x, self.y = nx, ny
            budget -= 1
        self.special_case = self.position != target and self.x == tx

    def _deliver(self, order):
        """Complete the active order, release the rider, and credit the fee."""
        self.orders.pop(0)
        order.complete(_current_tick())
        fee = int(_config_value("delivery_fee", DEFAULT_DELIVERY_FEE))
        globalvar.set_value("funds", globalvar.get_value("funds", 0) + fee)
        self.state = STATE_IDLE
        self.special_case = False
        self._record("delivered_to_customer", order.id)

    def to_dict(self):
        """Return a JSON-serializable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
        }

"""Order data model and coordinate normalization."""

import itertools

import globalvar

MAP_SIZE = 800
CELL_SIZE = 100

_id_counter = itertools.count(1)


def snap_coordinate(value):
    """Convert a raw coordinate to the nearest street intersection.

    Args:
        value: Integer coordinate in the range 0-800.

    Returns:
        Integer multiple of 100 within 0-800.

    Raises:
        TypeError: If ``value`` is not an integer.
        ValueError: If ``value`` is outside 0-800.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise TypeError(f"Coordinate must be an integer, got {value!r}.")
    if value < 0 or value > MAP_SIZE:
        raise ValueError(f"Coordinate {value} is outside the range 0-{MAP_SIZE}.")
    snapped = ((value + CELL_SIZE // 2) // CELL_SIZE) * CELL_SIZE
    return min(max(snapped, 0), MAP_SIZE)


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id (int): Unique order identifier.
        restaurant_pos (tuple[int, int]): Snapped pickup location.
        customer_pos (tuple[int, int]): Snapped delivery location.
        status (str): "pending", "assigned", "picking_up", "delivering",
            or "completed".
        rider_id (int | None): Assigned rider identifier.
        start_time (int): Tick at which the order was created.
        completion_time (int | None): Tick at which the order was delivered.
        distance (int): Manhattan distance between restaurant and customer.
        time_taken (int | None): Ticks between creation and completion.
    """

    def __init__(self, res_pos, cus_pos):
        """Create an order and normalize its coordinates.

        Args:
            res_pos: ``(x, y)`` restaurant coordinates in 0-800.
            cus_pos: ``(x, y)`` customer coordinates in 0-800.
        """
        self.id = next(_id_counter)
        self.restaurant_pos = (snap_coordinate(res_pos[0]), snap_coordinate(res_pos[1]))
        self.customer_pos = (snap_coordinate(cus_pos[0]), snap_coordinate(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.completion_time = None
        self.distance = (
            abs(self.restaurant_pos[0] - self.customer_pos[0])
            + abs(self.restaurant_pos[1] - self.customer_pos[1])
        )
        self.time_taken = None

    def assign_to(self, rider_id):
        """Record that this order was assigned to the rider ``rider_id``."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick):
        """Mark the order as delivered at ``tick`` and compute time taken."""
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time

    def to_dict(self):
        """Return a JSON-serializable snapshot of the order."""
        return {
            "id": self.id,
            "restaurant_pos": list(self.restaurant_pos),
            "customer_pos": list(self.customer_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

"""Order assignment and order lifecycle management."""

import logging

import globalvar
from order_model import Order

logger = logging.getLogger("dispatch.order")

DEFAULT_MAX_ORDERS = 2


def _max_orders_per_rider():
    """Return the configured maximum queue length per rider."""
    config = globalvar.get_value("config") or {}
    return int(config.get("max_orders_per_rider", DEFAULT_MAX_ORDERS))


def find_best_rider_for_order(order):
    """Select the most suitable rider for ``order``.

    Strategy (distance-first with load balancing):
        * Only riders whose order queue is not full are candidates.
        * The rider with the smallest Manhattan distance to the restaurant
          wins; ties are broken by the shorter order queue, then by rider id.

    Args:
        order: The ``Order`` to be assigned.

    Returns:
        The chosen ``Rider``, or ``None`` if no rider can take the order.
    """
    limit = _max_orders_per_rider()
    rx, ry = order.restaurant_pos
    best = None
    best_key = None
    for rider in globalvar.get_value("riders", []):
        if len(rider.orders) >= limit:
            continue
        distance = abs(rider.x - rx) + abs(rider.y - ry)
        key = (distance, len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates, assigns, cancels, and tracks orders."""

    def create_order(self, coords):
        """Create and register an order from four integer coordinates.

        Args:
            coords: Sequence ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            The newly created ``Order``.

        Raises:
            ValueError: If the sequence length is wrong or a coordinate is
                outside 0-800.
        """
        if len(coords) != 4:
            raise ValueError("An order needs exactly four coordinates.")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        logger.info("Order %s created: %s -> %s", order.id, order.restaurant_pos, order.customer_pos)
        return order

    def assign(self, order, rider):
        """Append ``order`` to ``rider``'s queue and mark it assigned."""
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

    def dispatch(self, order):
        """Find the best rider for ``order`` and assign it.

        Returns:
            The assigned ``Rider``, or ``None`` if no rider is available.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        return self.assign(order, rider)

    def cancel(self, order):
        """Remove a rejected order from the order registry."""
        orders = globalvar.get_value("orders", [])
        if order in orders:
            orders.remove(order)
        logger.info("Order %s cancelled", order.id)

    def active_orders(self):
        """Return orders that have not been completed yet."""
        return [o for o in globalvar.get_value("orders", []) if o.status != "completed"]

    def completed_orders(self):
        """Return orders that have been delivered."""
        return [o for o in globalvar.get_value("orders", []) if o.status == "completed"]

"""System initialization, configuration, and resource management."""

import json
import logging
import os
import random

import globalvar
from rider import Rider

logger = logging.getLogger("dispatch.system")

CONFIG_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), os.pardir, "config.json")

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruitment_cost": 200,
    "rider_speed": 50,
    "delivery_fee": 30,
    "max_orders_per_rider": 2,
}


def load_config(path=CONFIG_PATH):
    """Load system parameters from a JSON file, falling back to defaults.

    Unknown keys are ignored; non-integer or negative values are rejected.

    Args:
        path: Location of the JSON configuration file.

    Returns:
        Dictionary of configuration values.
    """
    config = dict(DEFAULT_CONFIG)
    if not os.path.isfile(path):
        return config
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read config %s (%s); using defaults.", path, exc)
        return config
    if not isinstance(data, dict):
        logger.warning("Config %s is not a JSON object; using defaults.", path)
        return config
    for key in DEFAULT_CONFIG:
        value = data.get(key)
        if isinstance(value, int) and not isinstance(value, bool) and value >= 0:
            config[key] = value
    if config["rider_speed"] < 1:
        config["rider_speed"] = DEFAULT_CONFIG["rider_speed"]
    if config["max_orders_per_rider"] < 1:
        config["max_orders_per_rider"] = DEFAULT_CONFIG["max_orders_per_rider"]
    return config


def init_system():
    """Initialize all global parameters for a fresh simulation.

    Sets system funds, an empty rider list, the rider id counter (0), the
    running flag (True), the simulated tick (0), and an empty order list.
    """
    globalvar._init()
    config = load_config()
    globalvar.set_value("config", config)
    globalvar.set_value("funds", int(config["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("orders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("is_running", True)
    globalvar.set_value("tick", 0)
    logger.info("System initialized with funds=%s", config["initial_funds"])


class SystemManager:
    """Facade over global state: funds, riders, tick, and running flag."""

    def __init__(self):
        """Attach to the global state, initializing it if necessary."""
        if globalvar.get_value("config") is None:
            init_system()

    @property
    def funds(self):
        """Current system fund balance (may be negative)."""
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def rider_id_counter(self):
        """Last issued rider identifier."""
        return globalvar.get_value("rider_id_counter", 0)

    @property
    def is_running(self):
        """Whether the main loop should keep running."""
        return bool(globalvar.get_value("is_running", True))

    @is_running.setter
    def is_running(self, value):
        globalvar.set_value("is_running", bool(value))

    @property
    def current_tick(self):
        """Current simulated time in ticks."""
        return globalvar.get_value("tick", 0)

    @property
    def recruitment_cost(self):
        """Fixed cost deducted when a rider is recruited."""
        return int(self._config().get("recruitment_cost", DEFAULT_CONFIG["recruitment_cost"]))

    @property
    def riders(self):
        """The dynamic list of riders."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    def _config(self):
        """Return the active configuration dictionary."""
        return globalvar.get_value("config") or {}

    def can_recruit(self):
        """Return True if the system funds cover one recruitment."""
        return self.funds >= self.recruitment_cost

    def recruit_rider(self):
        """Create a new rider, deduct the recruitment cost, and register it.

        Returns:
            The newly created ``Rider``.
        """
        new_id = self.rider_id_counter + 1
        globalvar.set_value("rider_id_counter", new_id)
        self.funds = self.funds - self.recruitment_cost
        rider = Rider(new_id)
        self.riders.append(rider)
        logger.info("Recruited rider %s (funds now %s)", new_id, self.funds)
        return rider

    def remove_rider(self, rider):
        """Remove an idle rider from the roster (CRUD delete support)."""
        if rider.orders:
            raise ValueError("Cannot remove a rider who still has orders.")
        self.riders.remove(rider)

    def advance_tick(self):
        """Advance simulated time by one tick and move every rider.

        Returns:
            The new current tick value.
        """
        globalvar.set_value("tick", self.current_tick + 1)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # keep the simulation alive on unexpected faults
                logger.exception("Rider %s failed to move; resetting it to idle.", rider.rider_id)
                rider.orders.clear()
                rider.state = 0
                rider.special_case = False
        return self.current_tick

    def save_state(self, path="dispatch_state.json"):
        """Persist configuration, funds, riders, and order history as JSON.

        Args:
            path: Destination file path.

        Returns:
            The path that was written.
        """
        from order import OrderManager  # local import avoids import cycle at load time

        data = {
            "config": self._config(),
            "funds": self.funds,
            "tick": self.current_tick,
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in globalvar.get_value("orders", [])],
        ()
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        logger.info("State saved to %s", path)
        return path

"""Status displays and ASCII map visualization."""

from rider import STATE_NAMES

GRID_STEP = 50
MAP_CELLS = 17  # 0..800 in steps of 50


class DisplayManager:
    """Renders the simulator state as text for the command line.

    Attributes:
        symbol_map (dict[str, str]): Symbols used on the ASCII map.
    """

    def __init__(self):
        """Initialize the symbol mapping."""
        self.symbol_map = {
            "rider": "@",
            "restaurant": "S",
            "customer": "C",
            "route": "*",
            "empty": " ",
            "street_h": "-",
            "street_v": "|",
            "intersection": "+",
        }

    def _base_symbol(self, x, y):
        """Return the background symbol for a map point."""
        on_x = x % 100 == 0
        on_y = y % 100 == 0
        if on_x and on_y:
            return self.symbol_map["intersection"]
        if on_y:
            return self.symbol_map["street_h"]
        if on_x:
            return self.symbol_map["street_v"]
        return self.symbol_map["empty"]

    def render_map(self, riders, orders, route=None):
        """Render the 8x8 cell map with riders, orders, and an optional route.

        Each character represents 50 units. Intersections are shown with
        ``+``; restaurants (``S``) and customers (``C``) are shown for active
        orders; riders are shown as ``@``; and a route is drawn with ``*``.

        Args:
            riders: Iterable of ``Rider`` objects.
            orders: Iterable of ``Order`` objects (completed ones are skipped).
            route: Optional list of waypoints ``(x, y)``.

        Returns:
            Multi-line string with the map and a legend.
        """
        cells = []
        for row in range(MAP_CELLS):
            y = row * GRID_STEP
            cells.append([self._base_symbol(col * GRID_STEP, y) for col in range(MAP_CELLS)])

        if route:
            for x, y in route[1:-1]:
                cells[y // GRID_STEP][x // GRID_STEP] = self.symbol_map["route"]

        for order in orders:
            if order.status == "completed":
                continue
            rx, ry = order.restaurant_pos
            cx, cy = order.customer_pos
            cells[ry // GRID_STEP][rx // GRID_STEP] = self.symbol_map["restaurant"]
            cells[cy // GRID_STEP][cx // GRID_STEP] = self.symbol_map["customer"]

        for rider in riders:
            cells[rider.y // GRID_STEP][rider.x // GRID_STEP] = self.symbol_map["rider"]

        lines = ["   x: 0 .. 800 (1 char = 50 units)"]
        for row in range(MAP_CELLS):
            lines.append(f"{row * GRID_STEP:>4} " + "".join(cells[row]))
        lines.append(
            "Legend: @ rider  S restaurant  C customer  * route  "
            "+ intersection  - / | street"
        )
        return "\n".join(lines)

    def render_status(self, funds, tick, riders, orders):
        """Render the system snapshot shown by the ``status`` command."""
        active = [o for o in orders if o.status != "completed"]
        lines = [
            f"Funds: {funds}",
            f"Tick: {tick}",
            f"Riders: {len(riders)}",
            f"Active orders: {len(active)}",
        ]
        for rider in riders:
            order_ids = ", ".join(str(o.id) for o in rider.orders) or "-"
            lines.append(
                f"  Rider {rider.rider_id}: status={STATE_NAMES.get(rider.state, 'unknown')} "
                f"position=({rider.x}, {rider.y}) orders={len(rider.orders)} [{order_ids}]"
            )
        return "\n".join(lines)

    def render_orders(self, orders):
        """Render order details for the ``orders`` command."""
        if not orders:
            return "No orders yet."
        lines = []
        for o in orders:
            taken = "-" if o.time_taken is None else str(o.time_taken)
            rider = "-" if o.rider_id is None else str(o.rider_id)
            lines.append(
                f"Order {o.id}: start_tick={o.start_time} restaurant={o.restaurant_pos} "
                f"customer={o.customer_pos} distance={o.distance} time_taken={taken} "
                f"rider={rider} status={o.status}"
            )
        return "\n".join(lines)

    def render_rider_logs(self, riders):
        """Render every recorded status change for all riders."""
        if not riders:
            return "No riders recruited yet."
        lines = []
        for rider in riders:
            lines.append(f"Rider {rider.rider_id}:")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order = "-" if entry["order_id"] is None else str(entry["order_id"])
                lines.append(
                    f"  t={entry['time']} event={entry['event']} order={order} "
                    f"position=({entry['x']}, {entry['y']}) "
                    f"state={STATE_NAMES.get(entry['state'], 'unknown')}"
                )
        return "\n".join(lines)

"""Command parsing, input validation, and command history."""

try:  # enables arrow-key history on platforms that provide readline
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - Windows without pyreadline
    pass

MAP_MIN = 0
MAP_MAX = 800


class InputError(Exception):
    """Raised when user input is malformed or out of range."""


class InputManager:
    """Reads and validates user commands.

    Attributes:
        command_history (list[str]): Every non-empty command entered, in order.
    """

    ALIASES = {
        "s": "start",
        "st": "status",
        "o": "orders",
        "r": "riders",
        "m": "map",
        "t": "tick",
        "h": "help",
        "q": "quit",
        "x": "exit",
    }

    def __init__(self):
        """Create an input manager with an empty history."""
        self.command_history = []

    def read_line(self, prompt="dispatch> "):
        """Read one line from standard input with history shortcuts.

        Supported shortcuts:
            ``!!``  repeat the last command.
            ``!n``  repeat the n-th command in the history (1-based).

        Returns:
            The stripped (and possibly expanded) line.

        Raises:
            InputError: If a history shortcut cannot be resolved.
        """
        raw = input(prompt).strip()
        if raw == "!!":
            if not self.command_history:
                raise InputError("No command in history to repeat.")
            raw = self.command_history[-1]
        elif raw.startswith("!") and raw[1:].isdigit():
            index = int(raw[1:])
            if index < 1 or index > len(self.command_history):
                raise InputError(f"History entry {index} does not exist.")
            raw = self.command_history[index - 1]
        if raw:
            self.command_history.append(raw)
        return raw

    def parse_command(self, line):
        """Split a line into a command name and its arguments.

        Returns:
            Tuple ``(command_name, args)``; ``command_name`` is ``None`` for
            an empty line.
        """
        tokens = line.replace(",", " ").split()
        if not tokens:
            return None, []
        name = tokens[0].lower()
        name = self.ALIASES.get(name, name)
        return name, tokens[1:]

    def parse_coordinates(self, tokens):
        """Validate four integer coordinates in the range 0-800.

        Args:
            tokens: Sequence of strings ``[rx, ry, cx, cy]``.

        Returns:
            List of four ``int`` values.

        Raises:
            InputError: On wrong count, non-integer text, or out-of-range value.
        """
        if len(tokens) != 4:
            raise InputError(
                "Expected 4 integers: restaurant_x restaurant_y customer_x customer_y."
            )
        values = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise InputError(f"'{token}' is not an integer.") from None
            if value < MAP_MIN or value > MAP_MAX:
                raise InputError(f"Coordinate {value} is outside the range {MAP_MIN}-{MAP_MAX}.")
            values.append(value)
        return values

    def parse_int(self, tokens, default, low, high, name):
        """Parse an optional single integer argument within ``[low, high]``."""
        if not tokens:
            return default
        try:
            value = int(tokens[0])
        except ValueError:
            raise InputError(f"{name} must be an integer, got '{tokens[0]}'.") from None
        if value < low or value > high:
            raise InputError(f"{name} must be between {low} and {high}.")
        return value

"""Command-line entry point for the Food Delivery Order Dispatch simulator.

Run with:  python src/main.py
"""

import logging
import sys

from display import DisplayManager
from input_manager import InputError, InputManager
from order import OrderManager
from path_planner import PathPlanner
from system import SystemManager, init_system

logger = logging.getLogger("dispatch.main")

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Place an order (restaurant then customer, 0-800).
                       Shows coordinates, assigned rider, and route map.
  tick [n]             Advance the simulation by n ticks (default 1).
  status               Show funds, tick, and every rider's state and orders.
  orders               Show details of all orders.
  riders               Show the event log of every rider.
  map                  Show the ASCII map with riders and active orders.
  save [file]          Save state to a JSON file (default dispatch_state.json).
  history              Show command history.
  help                 Show this help.
  quit / exit          Leave the program.
Shortcuts: s=start, t=tick, st=status, o=orders, r=riders, m=map, q=quit,
           !! repeats the last command, !n repeats history entry n.
"""


class DispatchApp:
    """Wires the managers together and runs the interactive command loop."""

    def __init__(self):
        """Initialize global state and all manager objects."""
        init_system()
        self.manager = SystemManager()
        self.order_manager = OrderManager()
        self.planner = PathPlanner()
        self.display = DisplayManager()
        self.inputs = InputManager()
        self.commands = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self):
        """Run the main loop until the system running flag is cleared."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.manager.is_running:
            try:
                line = self.inputs.read_line()
            except (EOFError, KeyboardInterrupt):
                print()
                break
            except InputError as exc:
                print(f"Error: {exc}")
                continue
            if not line:
                continue
            name, args = self.inputs.parse_command(line)
            handler = self.commands.get(name)
            if handler is None:
                print(f"Unknown command '{name}'. Type 'help' for a list of commands.")
                continue
            try:
                handler(args)
            except InputError as exc:
                print(f"Error: {exc}")
            except Exception as exc:  # keep the simulator running after faults
                logger.exception("Command '%s' failed", name)
                print(f"Unexpected error: {exc}. System state has been preserved.")

    def active_orders(self):
        """Return orders not yet completed."""
        return self.order_manager.active_orders()

    def cmd_start(self, args):
        """Handle ``start``: create an order, assign a rider, and show the route."""
        if not args:
            args = self.inputs.read_line(
                "Enter restaurant_x restaurant_y customer_x customer_y: "
            ).replace(",", " ").split()
        coords = self.inputs.parse_coordinates(args)
        order = self.order_manager.create_order(coords)
        print(
            f"Order {order.id}: restaurant grid point {order.restaurant_pos}, "
            f"customer grid point {order.customer_pos}, distance {order.distance}"
        )

        rider = self.order_manager.dispatch(order)
        new_rider = False
        if rider is None:
            if not self.manager.can_recruit():
                self.order_manager.cancel(order)
                print(
                    f"No rider is available and funds ({self.manager.funds}) are below the "
                    f"recruitment cost ({self.manager.recruitment_cost}). Order rejected."
                )
                return
            rider = self.manager.recruit_rider()
            new_rider = True
            self.order_manager.assign(order, rider)

        queue_ahead = len(rider.orders) - 1
        route = self.planner.plan_route(rider.position, [order.restaurant_pos, order.customer_pos])
        distance = PathPlanner.path_length(route)
        ticks = PathPlanner.estimate_ticks(distance, rider.speed)

        print(f"New rider added: {'yes' if new_rider else 'no'}"
              + (f" (id {rider.rider_id}, funds now {self.manager.funds})" if new_rider else ""))
        print(f"Assigned rider: {rider.rider_id} (state: {rider.state_name})")
        print(f"Rider speed: {rider.speed} units/tick")
        print(f"Route distance: {distance} units, estimated time: {ticks} tick(s)")
        if queue_ahead > 0:
            print(f"Note: rider has {queue_ahead} order(s) ahead in queue.")
        print(self.display.render_map(self.manager.riders, self.active_orders(), route))

    def cmd_status(self, args):
        """Handle ``status``: print funds, tick, riders, and their orders."""
        print(self.display.render_status(
            self.manager.funds,
            self.manager.current_tick,
            self.manager.riders,
            self.order_manager_orders(),
        ))

    def order_manager_orders(self):
        """Return all orders, including completed ones."""
        from globalvar import get_value
        return get_value("orders", [])

    def cmd_orders(self, args):
        """Handle ``orders``: print details of every order."""
        print(self.display.render_orders(self.order_manager_orders()))

    def cmd_riders(self, args):
        """Handle ``riders``: print the status change log of every rider."""
        print(self.display.render_rider_logs(self.manager.riders))

    def cmd_map(self, args):
        """Handle ``map``: print the ASCII map of riders and active orders."""
        print(self.display.render_map(self.manager.riders, self.active_orders()))

    def cmd_tick(self, args):
        """Handle ``tick [n]``: advance the simulation."""
        steps = self.inputs.parse_int(args, 1, 1, 1000, "Tick count")
        before = len(self.order_manager.completed_orders())
        for _ in range(steps):
            self.manager.advance_tick()
        completed = len(self.order_manager.completed_orders()) - before
        print(f"Advanced {steps} tick(s). Current tick: {self.manager.current_tick}. "
              f"Orders completed: {completed}.")

    def cmd_save(self, args):
        """Handle ``save [file]``: persist state to JSON."""
        path = args[0] if args else "dispatch_state.json"
        try:
            self.manager.save_state(path)
        except OSError as exc:
            raise InputError(f"Could not write '{path}': {exc}") from None
        print(f"State saved to {path}.")

    def cmd_history(self, args):
        """Handle ``history``: list previously entered commands."""
        if not self.inputs.command_history:
            print("History is empty.")
            return
        for index, command in enumerate(self.inputs.command_history, start=1):
            print(f"{index:>4}  {command}")

    def cmd_help(self, args):
        """Handle ``help``: show the command list."""
        print(HELP_TEXT)

    def cmd_quit(self, args):
        """Handle ``quit``/``exit``: stop the main loop."""
        self.manager.is_running = False
        print("Goodbye.")


def main():
    """Configure logging and start the interactive simulator."""
    logging.basicConfig(
        level=logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        stream=sys.stderr,
    )
    app = DispatchApp()
    app.run()


if __name__ == "__main__":
    main()

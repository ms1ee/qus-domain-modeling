"""Domain models for the team building lottery system."""
from __future__ import annotations

import re
from dataclasses import dataclass


@dataclass(frozen=True)
class User:
    """A participant with a name and a department group."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize configured for a lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str]
    associated_drawer: str | None


UNRANKED_LEVEL = 99
_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3, "fourth": 4, "fifth": 5}
_LEVEL_PATTERN = re.compile(
    r"\b(first|second|third|fourth|fifth)\s+prize\b", re.IGNORECASE
)


def prize_level_rank(prize_name: str) -> int:
    """Return a numeric rank for a prize level (1 = highest). Unknown levels rank last."""
    match = _LEVEL_PATTERN.search(prize_name or "")
    if match is None:
        return UNRANKED_LEVEL
    return _LEVEL_WORDS[match.group(1).lower()]


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Sort prizes from highest level to lowest, keeping configured order within a level."""
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: (prize_level_rank(item[1].name), item[0]))
    return [prize for _, prize in indexed]

"""Participant list loading, verification and group extraction."""
from __future__ import annotations

import re
from collections import Counter

from lottery_system.default_data import DEFAULT_USER_LINES
from lottery_system.models import User

_FIELD_SPLIT = re.compile(r"[,\t]")


def _key(text: str) -> str:
    return text.strip().casefold()


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_USER_LINES.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (one 'Name,Department' per line)."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text or text.startswith("#"):
                continue
            fields = [part.strip() for part in _FIELD_SPLIT.split(text)]
            if len(fields) != 2 or not fields[0] or not fields[1]:
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department', "
                    f"got: {text!r}"
                )
                continue
            users.append(User(name=fields[0], department=fields[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = _key(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department -> number of people, in order of first appearance."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management."""
from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of configured prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Add a prize at the end of the lottery sequence. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups: list[str] = []
        seen: set[str] = set()
        for group in avoid_groups or []:
            text = str(group).strip()
            if text and text.casefold() not in seen:
                seen.add(text.casefold())
                groups.append(text)

        drawer = (associated_drawer or "").strip() or None
        self._prizes.append(Prize(clean_name, quantity, groups, drawer))

    def get_prizes(self) -> list[Prize]:
        """Return prizes in lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        count = len(self._prizes)
        if sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in positions]
        return True

"""Winner recording, grouping and CSV export."""
from __future__ import annotations

import csv
from dataclasses import dataclass
from datetime import date
from pathlib import Path

from lottery_system.models import Prize, User, sort_prizes_by_level


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    order: int


class ResultManager:
    """Stores winners in the order they were drawn."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize | str) -> Winner:
        prize_name = prize if isinstance(prize, str) else prize.name
        winner = Winner(user=user, prize_name=prize_name, order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map prize name -> winners in winning order, following the given prize order."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt."""
        path = Path.cwd() / f"awards_result_{date.today():%Y%m%d}.txt"
        ordered = sort_prizes_by_level(prizes)
        grouped = self.get_winners_by_prize(ordered)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize Name", "Winning Order", "Name", "Department"])
                for prize in ordered:
                    for position, winner in enumerate(grouped[prize.name], start=1):
                        writer.writerow(
                            [
                                prize.name,
                                position,
                                winner.user.name,
                                winner.user.department,
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""
from __future__ import annotations

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

CHENG_RU_GROUP = "Cheng Ru Group"
RELAXED_RULE_DRAWER = "Li Bin"
FALLBACK_SOURCE_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"
FALLBACK_HISTORY_SIZE = 2


def _key(text: str | None) -> str:
    return (text or "").strip().casefold()


def is_cheng_ru_member(user: User) -> bool:
    return _key(user.department) == _key(CHENG_RU_GROUP)


class LotteryManager:
    """Runs draws for a set of participants while enforcing lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Outcomes (True = winners contained Cheng Ru Group members) of the most
        # recent rounds drawn by the fallback source lottery persons.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.Random()
        self._winner_keys: set[str] = set()

    # ----- state helpers -------------------------------------------------
    def has_won(self, user: User) -> bool:
        return _key(user.name) in self._winner_keys

    def is_fallback_active(self) -> bool:
        """Fallback triggers when the last two source rounds had no Cheng Ru Group winner."""
        return (
            len(self.last_two_draw_results) == FALLBACK_HISTORY_SIZE
            and not any(self.last_two_draw_results)
        )

    def _is_relaxed(self, prize: Prize) -> bool:
        return _key(prize.associated_drawer) == _key(RELAXED_RULE_DRAWER)

    def _is_fallback_target(self, prize: Prize) -> bool:
        return _key(prize.associated_drawer) == _key(FALLBACK_TARGET_DRAWER)

    # ----- eligibility ---------------------------------------------------
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users who may win this prize right now."""
        available = [user for user in self.users if not self.has_won(user)]

        # Fallback (Cheng Ru Group protection) overrides avoidance groups.
        if self._is_fallback_target(prize) and self.is_fallback_active():
            protected = [user for user in available if is_cheng_ru_member(user)]
            if protected:
                return protected

        # Special rule for Li Bin: avoidance groups are waived; previous winners are still excluded.
        if self._is_relaxed(prize):
            return available

        avoided = {_key(group) for group in prize.avoid_groups}
        return [user for user in available if _key(user.department) not in avoided]

    def count_eligible(self, prize: Prize) -> int:
        return len(self._get_eligible_pool(prize))

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applying to a prize round."""
        rules: list[str] = []
        if prize.associated_drawer:
            rules.append(f"lottery person: {prize.associated_drawer}")
        rules.append("previous winners excluded")
        if self._is_relaxed(prize):
            rules.append("relaxed rules for Li Bin: avoidance groups waived")
        elif prize.avoid_groups:
            rules.append("avoidance groups: " + ", ".join(prize.avoid_groups))
        else:
            rules.append("no avoidance groups")
        if self._is_fallback_target(prize) and self.is_fallback_active():
            rules.append(
                f"{CHENG_RU_GROUP} protection active: only {CHENG_RU_GROUP} members are eligible"
            )
        return "; ".join(rules)

    # ----- drawing -------------------------------------------------------
    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_keys.add(_key(winner.name))
        self.result_manager.add_winner(winner, prize)
        return winner

    def record_round_outcome(self, prize: Prize, winners: list[User]) -> None:
        """Update fallback history after a round has finished."""
        drawer = _key(prize.associated_drawer)
        if drawer in {_key(name) for name in FALLBACK_SOURCE_DRAWERS}:
            hit = any(is_cheng_ru_member(winner) for winner in winners)
            self.last_two_draw_results.append(hit)
            del self.last_two_draw_results[:-FALLBACK_HISTORY_SIZE]
        elif drawer == _key(FALLBACK_TARGET_DRAWER):
            # The fallback cycle is consumed by the target lottery person's round.
            self.last_two_draw_results.clear()

"""Command-line interaction for the lottery system."""
from __future__ import annotations

from collections import Counter
from typing import Callable

from lottery_system.models import Prize, User, sort_prizes_by_level
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager

SEPARATOR = "=" * 60


def ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _ci(text: str) -> str:
    return text.strip().casefold()


class LotteryCLI:
    """Interactive console application."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    # ----- main loop -----------------------------------------------------
    def run(self) -> None:
        self._print("Welcome to the Company Team Building Smart Lottery System")
        while True:
            self._print_main_menu()
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self.load_list()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.prize_menu()
            elif choice == "4":
                self.start_lottery()
            elif choice == "5":
                self.results_menu()
            elif choice == "0":
                self._print("Goodbye!")
                return
            else:
                self._print("Invalid option, please try again.")

    def _print_main_menu(self) -> None:
        self._print(SEPARATOR)
        self._print("1. Load participant list")
        self._print("2. View department groups")
        self._print("3. Prize configuration")
        self._print("4. Start lottery")
        self._print("5. View results / export")
        self._print("0. Exit")
        self._print(SEPARATOR)

    # ----- list management -----------------------------------------------
    def load_list(self) -> None:
        self._print("1. Built-in default team building list")
        self._print("2. Upload custom txt list (format: Name,Department per line)")
        self._print("0. Cancel")
        choice = self._ask("Your choice: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip().strip('"')
            if not path:
                return
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

        resolved = self._resolve_list_issues(users, errors)
        if resolved is None:
            self._print("Upload terminated. Previous participant list is kept.")
            return
        if not resolved:
            self._print("No valid participants were loaded.")
            return
        self.users = resolved
        self.lottery = None
        self._print(f"Loaded {len(resolved)} participants.")
        self.show_groups()

    def _resolve_list_issues(
        self, users: list[User], errors: list[str]
    ) -> list[User] | None:
        """Interactively resolve format errors and duplicate names. None means cancelled."""
        while True:
            valid, duplicates = self.list_manager.validate_users(users)
            if not errors and not duplicates:
                return valid

            self._print("Data verification found issues:")
            for error in errors:
                self._print(f"  - {error}")
            for dup in duplicates:
                self._print(f"  - Duplicate name: {dup.name} ({dup.department})")
            self._print(
                "Options: [r] remove invalid and duplicate entries  "
                "[m] manually remove entries  [c] cancel upload"
            )
            choice = self._ask("Your choice: ").strip().lower()
            if choice == "r":
                return valid
            if choice == "m":
                counts = Counter(_ci(user.name) for user in users)
                self._print("Current entries:")
                for index, user in enumerate(users, start=1):
                    mark = "  [duplicate name]" if counts[_ci(user.name)] > 1 else ""
                    self._print(f"  {index}. {user.name} ({user.department}){mark}")
                raw = self._ask(
                    "Enter entry numbers to remove (comma separated, blank to skip): "
                )
                if not raw.strip():
                    continue
                tokens = raw.replace(",", " ").split()
                if not all(token.isdigit() for token in tokens) or any(
                    not 1 <= int(token) <= len(users) for token in tokens
                ):
                    self._print("Invalid entry numbers, nothing removed.")
                    continue
                to_remove = {int(token) for token in tokens}
                users = [
                    user for index, user in enumerate(users, start=1)
                    if index not in to_remove
                ]
                errors = []
                continue
            if choice == "c":
                return None
            self._print("Invalid choice.")

    def show_groups(self) -> None:
        if not self.users:
            self._print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._print("Valid department groups:")
        for index, (department, count) in enumerate(info.items(), start=1):
            self._print(f"  {index}. {department}: {count} people")

    # ----- prize configuration -------------------------------------------
    def prize_menu(self) -> None:
        while True:
            self._print(SEPARATOR)
            self._print("Prize configuration")
            self._print("1. Load preset prize library")
            self._print("2. Add custom prize")
            self._print("3. View prizes")
            self._print("4. Adjust lottery order")
            self._print("0. Back")
            self._print(SEPARATOR)
            choice = self._ask("Your choice: ").strip()
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        self._print(f"Error: {error}")
                else:
                    self._print("Preset prizes loaded.")
            elif choice == "2":
                self.add_custom_prize()
            elif choice == "3":
                self.show_prizes()
            elif choice == "4":
                self.adjust_order()
            elif choice == "0":
                return
            else:
                self._print("Invalid option, please try again.")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._print("Prizes in lottery order:")
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self._print(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoid groups: {groups} | Lottery person: {drawer}"
            )

    def add_custom_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        if not name:
            self._print("Cancelled.")
            return
        quantity = self._ask_positive_int("Quantity (blank to cancel): ")
        if quantity is None:
            self._print("Cancelled.")
            return
        drawer = self._ask("Associated lottery person (blank for none): ").strip() or None
        groups = self._select_avoid_groups()
        if groups is None:
            self._print("Cancelled.")
            return
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            self._print(f"Error: {exc}")
            return
        self._print(f"Prize '{name}' added.")

    def _ask_positive_int(self, prompt: str) -> int | None:
        while True:
            raw = self._ask(prompt).strip()
            if not raw:
                return None
            if raw.isdigit() and int(raw) >= 1:
                return int(raw)
            self._print("Please enter a positive integer (or leave blank to cancel).")

    def _select_avoid_groups(self) -> list[str] | None:
        if not self.users:
            self._print("No participant list loaded; enter group names manually.")
            raw = self._ask("Avoidance groups (comma separated, blank for none): ")
            return [token.strip() for token in raw.split(",") if token.strip()]

        groups = list(self.list_manager.get_group_info(self.users).keys())
        self._print("Available groups:")
        for index, group in enumerate(groups, start=1):
            self._print(f"  {index}. {group}")
        while True:
            raw = self._ask(
                "Select avoidance groups by number or name (comma separated, blank for none): "
            )
            tokens = [token.strip() for token in raw.split(",") if token.strip()]
            selected: list[str] = []
            unknown: str | None = None
            for token in tokens:
                if token.isdigit() and 1 <= int(token) <= len(groups):
                    match: str | None = groups[int(token) - 1]
                else:
                    match = next((g for g in groups if _ci(g) == _ci(token)), None)
                if match is None:
                    unknown = token
                    break
                if match not in selected:
                    selected.append(match)
            if unknown is not None:
                self._print(f"Unknown group: {unknown}. Please try again.")
                continue
            return selected

    def adjust_order(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self.show_prizes()
        raw = self._ask(
            "Enter new order as serial numbers, e.g. '2 1 3' (blank to cancel): "
        )
        if not raw.strip():
            return
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(token) for token in tokens]
        except ValueError:
            self._print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Lottery order updated.")
            self.show_prizes()
        else:
            self._print("Invalid order: every serial number must appear exactly once.")

    # ----- lottery -------------------------------------------------------
    def start_lottery(self) -> None:
        if not self.users:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure prizes first.")
            return
        if self.lottery is not None and self.lottery.result_manager.get_winners():
            confirm = self._ask("Existing results will be cleared. Start a new lottery? (y/n): ")
            if confirm.strip().lower() != "y":
                return

        self.lottery = LotteryManager(self.users)
        self._print(SEPARATOR)
        self._print("LOTTERY START")
        self._print(SEPARATOR)
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, len(prizes), prize)
        self._print(SEPARATOR)
        self._print("All lottery rounds completed.")
        self.show_results()

    def _run_round(self, round_no: int, total_rounds: int, prize: Prize) -> None:
        assert self.lottery is not None
        self._print(SEPARATOR)
        self._print(f"Round {round_no}/{total_rounds}: {prize.name}")
        self._print(f"Quantity: {prize.quantity}")
        self._print(f"Rule: {self.lottery.describe_rule(prize)}")
        self._print(SEPARATOR)

        round_winners: list[User] = []
        terminated = False
        for position in range(1, prize.quantity + 1):
            if self.lottery.count_eligible(prize) == 0:
                self._print("No eligible candidates remain for this prize; round ends early.")
                break
            action = self._ask_draw_action()
            if action == "terminate":
                terminated = True
                break
            self._print(f"Currently drawing the {ordinal(position)} winner...")
            winner = self.lottery.draw_one(prize)
            if winner is None:
                self._print("No eligible candidates remain for this prize; round ends early.")
                break
            round_winners.append(winner)
            self._print(
                f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}"
            )

        if terminated:
            skipped = prize.quantity - len(round_winners)
            self._print(f"Round terminated by operator; {skipped} award(s) skipped.")

        self.lottery.record_round_outcome(prize, round_winners)

        self._print(f"Current round results for {prize.name} (in winning order):")
        if not round_winners:
            self._print("  (no winners)")
        for index, winner in enumerate(round_winners, start=1):
            self._print(f"  {index}. {winner.name} ({winner.department})")

    def _ask_draw_action(self) -> str:
        """Return 'draw' or 'terminate'. Handles pause/continue commands."""
        prompt = "Press Enter to draw the next winner (or type 'pause' / 'terminate'): "
        while True:
            command = self._ask(prompt).strip().lower()
            if command in ("", "continue"):
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                self._print("Lottery paused.")
                while True:
                    resume = self._ask(
                        "Paused. Type 'continue' to resume or 'terminate' to end this round: "
                    ).strip().lower()
                    if resume == "continue":
                        self._print("Lottery resumed.")
                        return "draw"
                    if resume == "terminate":
                        return "terminate"
                    self._print("Unknown command.")
            self._print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    # ----- results -------------------------------------------------------
    def show_results(self) -> None:
        if self.lottery is None or not self.lottery.result_manager.get_winners():
            self._print("No lottery results yet.")
            return
        prizes = sort_prizes_by_level(self.prize_manager.get_prizes())
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        self._print(SEPARATOR)
        self._print("WINNER LIST (by prize level, then winning order)")
        self._print(SEPARATOR)
        for prize in prizes:
            self._print(f"[{prize.name}]")
            winners = grouped[prize.name]
            if not winners:
                self._print("  (no winners)")
            for index, winner in enumerate(winners, start=1):
                self._print(f"  {index}. {winner.user.name} ({winner.user.department})")

    def results_menu(self) -> None:
        self.show_results()
        if self.lottery is None or not self.lottery.result_manager.get_winners():
            return
        choice = self._ask("Export results to awards_result_YYYYMMDD.txt? (y/n): ")
        if choice.strip().lower() != "y":
            return
        success, message = self.lottery.result_manager.export_to_csv(
            self.prize_manager.get_prizes()
        )
        self._print(message)

    "Zhao Liu,Product Department",
    "Sun Qi,Product Department",
    "Zhou Ba,Product Department",

    "Jin Duo,Cheng Ru Group",
    "Ma Jiaqi,Cheng Ru Group",

    "Deng Hao,Cheng Ru Group",
    "Li Bin,Human Resources Department",
    "Xiao Ming,Human Resources Department",
    "Xiao Hong,Finance Department",
    "Xiao Gang,Finance Department",
    "Liu Yang,Finance Department",
    "Han Mei,Operations Department",
    "Gao Yuan,Operations Department",
    "Lin Feng,Operations Department",
]

_SEPARATOR_PATTERN = re.compile(r"[,;]")


class ListManager:
    """Loads, verifies and summarises participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'Name,Department' or 'Name;Department'."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        text = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=encoding) as handle:
                    text = handle.read()
                break
            except FileNotFoundError:
                return [], [f"File Error: file not found: {path}"]
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"File Error: cannot read '{path}': {exc}"]
        if text is None:
            return [], [f"File Error: unable to decode '{path}' as UTF-8 or GBK"]
        return self._parse_lines(text.splitlines())

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = _SEPARATOR_PATTERN.split(line, maxsplit=1)
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department' or 'Name;Department')"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). Only the first occurrence of a name is valid."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        """Return a mapping of department group name to member count."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.models import Prize

PRESET_PRIZES = [
    ("Third Prize - Movie Tickets", 3, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
]


def _norm(text: str) -> str:
    return " ".join(text.split()).casefold()


class PrizeManager:
    """Keeps the ordered prize library for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, drawer in PRESET_PRIZES:
            try:
                self.add_prize(name, quantity, None, drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        """Add a prize to the end of the lottery sequence."""
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name must not be empty.")
        name = name.strip()
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(_norm(p.name) == _norm(name) for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        prize = Prize(name, quantity, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order contains 1-based current positions in the new order."""
        n = len(self._prizes)
        try:
            ints = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(ints) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in ints]
        return True

"""Lottery rule engine: filtering, avoidance tiers, special rules and fallback."""

import random

from lottery_system.models import Prize, User

CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
MA_JIAQI = "Ma Jiaqi"
FALLBACK_TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")


def _norm(text) -> str:
    return " ".join(str(text).split()).casefold()


def _drawer_is(prize: Prize, drawer_name: str) -> bool:
    return prize.associated_drawer is not None and _norm(prize.associated_drawer) == _norm(drawer_name)


def describe_rule(prize: Prize) -> str:
    """Human readable rule description for a prize round."""
    parts: list[str] = []
    if _drawer_is(prize, LI_BIN):
        parts.append("Previous winners are excluded; avoidance groups are NOT applied.")
    else:
        parts.append("Previous winners are excluded.")
        if prize.avoid_groups:
            parts.append(
                f"Avoidance groups ({', '.join(prize.avoid_groups)}): members remain eligible "
                "but are drawn only after all other members."
            )
    if prize.associated_drawer:
        parts.append(f"Lottery person: {prize.associated_drawer}.")
    if _drawer_is(prize, MA_JIAQI):
        parts.append(
            "Fallback: if the last two draws of Chen Qixian and Jin Duo contained no "
            f"{CHENG_RU_GROUP} winners, the pool is limited to {CHENG_RU_GROUP} members."
        )
    return " ".join(parts)


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users) -> None:
        self.users: list[User] = list(users)
        self.last_two_draw_results: list[bool] = []
        self._winner_keys: set[str] = set()
        self._current_round_winners: list[User] = []

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return _norm(user.department) == _norm(CHENG_RU_GROUP)

    def _is_winner(self, user: User) -> bool:
        return _norm(user.name) in self._winner_keys

    def _protected_candidates(self) -> list[User]:
        return [u for u in self.users if not self._is_winner(u) and self._is_cheng_ru(u)]

    def is_fallback_active(self, prize: Prize) -> bool:
        """True if the Cheng Ru group protection forces this prize's pool."""
        if not _drawer_is(prize, MA_JIAQI):
            return False
        if len(self.last_two_draw_results) < 2 or any(self.last_two_draw_results):
            return False
        return bool(self._protected_candidates())

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Eligible non-winners for the prize (fallback applied if active)."""
        if self.is_fallback_active(prize):
            return self._protected_candidates()
        return [u for u in self.users if not self._is_winner(u)]

    def _split_tiers(self, prize: Prize, pool: list[User]) -> tuple[list[User], list[User]]:
        """Split pool into (preferred, avoidance-group members drawn later)."""
        if _drawer_is(prize, LI_BIN):
            return list(pool), []
        avoid = {_norm(g) for g in prize.avoid_groups}
        primary = [u for u in pool if _norm(u.department) not in avoid]
        secondary = [u for u in pool if _norm(u.department) in avoid]
        return primary, secondary

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        primary, secondary = self._split_tiers(prize, pool)
        tier = primary or secondary
        if not tier:
            return None
        winner = random.choice(tier)
        self._winner_keys.add(_norm(winner.name))
        self._current_round_winners.append(winner)
        return winner

    def end_round(self, prize: Prize) -> list[User]:
        """Finish the current round, updating fallback tracking. Returns round winners."""
        winners = self._current_round_winners
        self._current_round_winners = []
        if any(_drawer_is(prize, d) for d in FALLBACK_TRIGGER_DRAWERS):
            hit = any(self._is_cheng_ru(u) for u in winners)
            self.last_two_draw_results = (self.last_two_draw_results + [hit])[-2:]
        return winners

"""Winner recording, ordered summaries and CSV/txt export."""

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User

_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3, "fourth": 4, "fifth": 5}
_UNKNOWN_LEVEL = 99


def prize_level(prize: Prize) -> int:
    """Derive prize level from its name (First=1 is highest). Unknown levels sort last."""
    for word in re.findall(r"[a-zA-Z]+", prize.name.lower()):
        if word in _LEVEL_WORDS:
            return _LEVEL_WORDS[word]
    return _UNKNOWN_LEVEL


@dataclass
class Winner:
    user: User
    prize: Prize
    winning_time: datetime
    order_in_round: int


class ResultManager:
    """Stores winners in real time and produces ordered summaries."""

    TIME_FORMAT = "%Y-%m-%d %H:%M:%S"

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user, prize) -> Winner:
        order = sum(1 for w in self._winners if w.prize.name == prize.name) + 1
        winner = Winner(user, prize, datetime.now(), order)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low),
        and by winning time within each prize."""
        ordered = sorted(prizes, key=prize_level)
        result: dict[str, list[Winner]] = {}
        for prize in ordered:
            if prize.name in result:
                continue
            winners = [w for w in self._winners if w.prize.name == prize.name]
            winners.sort(key=lambda w: w.winning_time)
            result[prize.name] = winners
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt."""
        grouped = self.get_winners_by_prize(prizes)
        date_str = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_str}.txt")
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winners in grouped.values():
                    for w in winners:
                        writer.writerow([
                            w.user.name,
                            w.user.department,
                            w.prize.name,
                            w.winning_time.strftime(self.TIME_FORMAT),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Result successfully exported to {path}"

"""Command-line interaction for the lottery process."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager, describe_rule
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager

SEPARATOR = "-" * 70


def ordinal(n: int) -> str:
    if 11 <= n % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def ask(prompt: str) -> str:
    return input(prompt).strip()


def print_verification_report(errors, duplicates) -> None:
    if errors:
        print("Format errors found:")
        for err in errors:
            print(f"  - {err}")
    if duplicates:
        print("Duplicate names found (later occurrences ignored):")
        for user in duplicates:
            print(f"  - {user.name} ({user.department})")


def print_group_info(info: dict[str, int]) -> None:
    print(SEPARATOR)
    print("Department groups:")
    for dept, count in info.items():
        print(f"  {dept}: {count} people")


def setup_participants(list_mgr: ListManager):
    """Load and verify a participant list. Returns valid users, or None to exit."""
    while True:
        print(SEPARATOR)
        print("Participant list loading")
        print("  1. Load built-in default team building list")
        print("  2. Upload custom txt list (each line: Name,Department or Name;Department)")
        print("  0. Exit")
        choice = ask("Select: ")
        if choice == "0":
            return None
        if choice == "1":
            users, errors = list_mgr.load_default_list()
        elif choice == "2":
            path = ask("Enter txt file path: ")
            users, errors = list_mgr.load_custom_list(path)
        else:
            print("Invalid choice.")
            continue

        valid, duplicates = list_mgr.validate_users(users)
        if errors or duplicates:
            print_verification_report(errors, duplicates)
            while True:
                cmd = ask(
                    "Type 'remove' to drop invalid/duplicate data and continue, "
                    "or 'terminate' to cancel the upload: "
                ).lower()
                if cmd in ("remove", "terminate"):
                    break
                print("Unknown command.")
            if cmd == "terminate":
                print("Upload terminated.")
                continue

        if not valid:
            print("No valid participants found. Please try again.")
            continue
        print(f"Loaded {len(valid)} participants.")
        return valid


def parse_groups(text: str) -> list[str]:
    return [g.strip() for g in text.split(",") if g.strip()]


def print_prizes(prizes) -> None:
    print(SEPARATOR)
    if not prizes:
        print("No prizes configured.")
        return
    print("Prize sequence:")
    for idx, prize in enumerate(prizes, start=1):
        groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
        drawer = prize.associated_drawer or "none"
        print(f"  {idx}. {prize.name} | quantity: {prize.quantity} | "
              f"avoid groups: {groups} | lottery person: {drawer}")


def add_custom_prize(prize_mgr: PrizeManager, group_names) -> None:
    name = ask("Prize name: ")
    if not name:
        print("Prize name must not be empty.")
        return
    try:
        quantity = int(ask("Quantity: "))
    except ValueError:
        print("Quantity must be an integer.")
        return
    groups = parse_groups(ask("Avoidance groups (comma-separated, blank for none): "))
    for g in groups:
        if g not in group_names:
            print(f"Warning: group '{g}' is not in the participant list.")
    drawer = ask("Associated lottery person (blank for none): ") or None
    try:
        prize_mgr.add_prize(name, quantity, groups, drawer)
        print(f"Prize '{name}' added.")
    except ValueError as exc:
        print(f"Error: {exc}")


def set_avoid_groups(prize_mgr: PrizeManager, group_names) -> None:
    prizes = prize_mgr.get_prizes()
    print_prizes(prizes)
    if not prizes:
        return
    try:
        idx = int(ask("Prize serial number: "))
    except ValueError:
        print("Invalid serial number.")
        return
    if not 1 <= idx <= len(prizes):
        print("Serial number out of range.")
        return
    groups = parse_groups(ask("Avoidance groups (comma-separated, blank for none): "))
    for g in groups:
        if g not in group_names:
            print(f"Warning: group '{g}' is not in the participant list.")
    prizes[idx - 1].avoid_groups = groups
    print("Avoidance groups updated.")


def reorder(prize_mgr: PrizeManager) -> None:
    prizes = prize_mgr.get_prizes()
    print_prizes(prizes)
    if not prizes:
        return
    raw = ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ")
    try:
        order = [int(x) for x in raw.replace(",", " ").split()]
    except ValueError:
        print("Invalid input. Use integers only.")
        return
    if prize_mgr.reorder_prizes(order):
        print("Prize sequence updated.")
    else:
        print("Invalid order: must contain every serial number exactly once.")


def configure_prizes(prize_mgr: PrizeManager, group_info: dict[str, int]) -> None:
    group_names = set(group_info.keys())
    while True:
        print_prizes(prize_mgr.get_prizes())
        print(SEPARATOR)
        print("Prize configuration")
        print("  1. Load preset prizes")
        print("  2. Add custom prize")
        print("  3. Set avoidance groups for a prize")
        print("  4. Reorder prizes")
        print("  5. View department groups")
        print("  0. Finish configuration")
        choice = ask("Select: ")
        if choice == "1":
            for err in prize_mgr.load_default_prizes():
                print(f"Error: {err}")
        elif choice == "2":
            add_custom_prize(prize_mgr, group_names)
        elif choice == "3":
            set_avoid_groups(prize_mgr, group_names)
        elif choice == "4":
            reorder(prize_mgr)
        elif choice == "5":
            print_group_info(group_info)
        elif choice == "0":
            if prize_mgr.get_prizes():
                return
            print("Please configure at least one prize.")
        else:
            print("Invalid choice.")


def wait_while_paused() -> bool:
    """Block while paused. Returns True to continue, False to terminate the round."""
    print("Lottery paused.")
    while True:
        cmd = ask("Type 'continue' to resume or 'terminate' to end this round: ").lower()
        if cmd == "continue":
            print("Lottery resumed.")
            return True
        if cmd == "terminate":
            return False
        print("Unknown command.")


def run_round(index: int, total: int, prize, lottery: LotteryManager, result_mgr: ResultManager) -> None:
    print(SEPARATOR)
    print(f"Round {index}/{total}: {prize.name}")
    print(f"Quantity: {prize.quantity}")
    print(f"Rule: {describe_rule(prize)}")
    if lottery.is_fallback_active(prize):
        print("[Fallback] Cheng Ru Group protection active: pool limited to Cheng Ru Group members.")

    drawn = 0
    terminated = False
    while drawn < prize.quantity:
        print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
        cmd = ask("Press Enter to draw, or type 'pause' / 'terminate': ").lower()
        if cmd == "terminate":
            terminated = True
            break
        if cmd == "pause":
            if not wait_while_paused():
                terminated = True
                break
        elif cmd not in ("", "draw", "continue"):
            print("Unknown command.")
            continue

        user = lottery.draw_one(prize)
        if user is None:
            print("No eligible participants remain for this prize.")
            break
        result_mgr.add_winner(user, prize)
        drawn += 1
        print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

    round_winners = lottery.end_round(prize)
    print(f"Round finished: {len(round_winners)} winner(s)."
          + (" Round was terminated early." if terminated else ""))


def print_summary(result_mgr: ResultManager, prizes) -> None:
    print(SEPARATOR)
    print("Final winner list (by prize level, then winning order):")
    grouped = result_mgr.get_winners_by_prize(prizes)
    for prize_name, winners in grouped.items():
        print(f"\n[{prize_name}]")
        if not winners:
            print("  (no winners)")
        for w in winners:
            print(f"  {w.user.name} | {w.user.department} | {w.prize.name} | "
                  f"{w.winning_time.strftime(ResultManager.TIME_FORMAT)}")


def _run() -> None:
    list_mgr = ListManager()
    users = setup_participants(list_mgr)
    if users is None:
        print("Goodbye.")
        return
    group_info = list_mgr.get_group_info(users)
    print_group_info(group_info)

    prize_mgr = PrizeManager()
    configure_prizes(prize_mgr, group_info)
    prizes = prize_mgr.get_prizes()

    lottery = LotteryManager(users)
    result_mgr = ResultManager()

    print(SEPARATOR)
    print("Lottery starts!")
    for idx, prize in enumerate(prizes, start=1):
        run_round(idx, len(prizes), prize, lottery, result_mgr)

    print_summary(result_mgr, prizes)
    if ask("Export results to awards_result_YYYYMMDD.txt in the current directory? (y/n): ").lower() == "y":
        _, message = result_mgr.export_to_csv(prizes)
        print(message)
    print("Lottery finished. Thank you!")


def main() -> None:
    try:
        _run()
    except (KeyboardInterrupt, EOFError):
        print("\nProgram interrupted. Exiting.")

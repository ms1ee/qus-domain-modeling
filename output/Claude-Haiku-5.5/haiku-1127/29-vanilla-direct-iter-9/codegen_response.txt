"""Global variable storage for the food delivery dispatch simulator.

Shared runtime state (system funds, rider list, rider ID counter, running
flag, simulated time, order history and configuration) is kept in a single
module-level dictionary so that every module reads and updates it through the
same place.
"""

from typing import Any, Dict

_global_dict: Dict[str, Any] = {}


def _init() -> None:
    """Initialise, or reset, all global variables to their default values.

    Keys stored:
        system_funds (int): available funds, may become negative.
        riders (list[Rider]): riders currently in service.
        rider_id_counter (int): last issued rider ID.
        running (bool): control flag for the main loop.
        time (int): current simulation tick.
        orders (list[Order]): every order created so far.
        config (dict): system parameters.

    global _global_dict
    _global_dict = {
        "system_funds": 1000,
        "riders": [],
        "rider_id_counter": 0,
        "running": True,
        "time": 0,
        "orders": [],
        "config": {},
    }

    """Store a value under the given key.

    Args:
        key: Name of the global variable.
        value: Value to store.
    """
    _global_dict[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under the given key.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:

    return _global_dict.get(key, default)

"""Grid-based path planning for the 800x800 dispatch map.

The map is an 8x8 grid of 100x100 cells. Streets lie on grid lines, i.e. points
where ``x`` or ``y`` is a multiple of 100. Restaurants and customers are placed
at cell centres (50, 150, ..., 750). Routes are Manhattan routes made of at most
two axis-aligned segments (one right-angle turn). Between the two possible
corners the planner prefers the one whose segments run along streets.

Every waypoint is reached by monotone axis-aligned movement, so a rider always
makes progress towards its target and cannot deadlock.
"""

from typing import List, Sequence, Tuple

Point = Tuple[int, int]

MAP_SIZE = 800
CELL_SIZE = 100
GRID_CELLS = 8


class PathPlanner:
    """Stateless helper providing grid-based route planning and stepping."""

    @staticmethod
    def cell_center(value: int) -> int:
        """Convert a coordinate into the centre of its grid cell.

        Implements ``center = (value // 100) * 100 + 50``. Values equal to the
        map edge (800) are clamped into the last cell.

        Args:
            value: Input coordinate, expected in the range 0-800.

        Returns:
            The centre coordinate of the containing cell.
        """
        clamped = max(0, min(MAP_SIZE - 1, int(value)))
        return (clamped // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2

    @staticmethod
    def cell_of(x: int, y: int) -> Point:
        """Return the (column, row) grid index that contains a point.

        Args:
            x: X coordinate, 0-800.
            y: Y coordinate, 0-800.

        Returns:
            Tuple ``(col, row)`` with both values in the range 0-7.
        """
        col = max(0, min(GRID_CELLS - 1, int(x) // CELL_SIZE))
        row = max(0, min(GRID_CELLS - 1, int(y) // CELL_SIZE))
        return col, row

    @staticmethod
    def is_on_street(x: int, y: int) -> bool:
        """Return True when a point lies on a street (grid line)."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def is_straight(start: Point, goal: Point) -> bool:
        """Return True when the goal is directly above/below or left/right.

        This is the special case in which the rider moves along a single axis.
        """
        return start[0] == goal[0] or start[1] == goal[1]

    @classmethod
    def plan(cls, start: Point, goal: Point) -> List[Point]:
        """Plan a route from ``start`` to ``goal``.

        The result is a list of waypoints ending at ``goal``. An empty list is
        returned when the start already equals the goal. A straight route
        returns only the goal. Otherwise the route contains one corner followed
        by the goal. The corner is chosen so that the segments lie along streets
        whenever possible; ties prefer horizontal-first movement.

        Args:
            start: Current position ``(x, y)``.
            goal: Target position ``(x, y)``.

        Returns:
            Ordered list of waypoints ``[(x, y), ...]``.
        """
        sx, sy = start
        gx, gy = goal
        if (sx, sy) == (gx, gy):
            return []
        if cls.is_straight(start, goal):
            return [(gx, gy)]

        corner_h = (gx, sy)  # horizontal segment at y=sy, vertical at x=gx
        corner_v = (sx, gy)  # vertical segment at x=sx, horizontal at y=gy
        score_h = int(sy % CELL_SIZE == 0) + int(gx % CELL_SIZE == 0)
        score_v = int(sx % CELL_SIZE == 0) + int(gy % CELL_SIZE == 0)
        corner = corner_h if score_h >= score_v else corner_v
        return [corner, (gx, gy)]

    @staticmethod
    def advance(
        x: int, y: int, waypoints: Sequence[Point], speed: int
    ) -> Tuple[Point, List[Point]]:
        """Move up to ``speed`` units along the planned waypoints.

        Movement along each segment is axis-aligned; when a diagonal offset
        occurs the x axis is handled before the y axis.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            waypoints: Remaining waypoints, next target first.
            speed: Maximum distance (units) travelled this tick.

        Returns:
            A tuple ``((new_x, new_y), remaining_waypoints)``.
        """
        remaining = list(waypoints)
        budget = max(0, int(speed))
        while budget > 0 and remaining:
            tx, ty = remaining[0]
            if (x, y) == (tx, ty):
                remaining.pop(0)
                continue
            if x != tx:
                step = min(abs(tx - x), budget)
                x += step if tx > x else -step
                budget -= step
            if budget > 0 and y != ty:
                step = min(abs(ty - y), budget)
                y += step if ty > y else -step
                budget -= step
            if (x, y) == (tx, ty):
                remaining.pop(0)
        return (x, y), remaining

"""Order data model: coordinate validation, cell-centre conversion and lifecycle."""

import math
from typing import Any, Dict, Optional, Sequence, Tuple

import globalvar
from path_planner import MAP_SIZE, PathPlanner

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKED_UP = "picked_up"
STATUS_COMPLETED = "completed"


def validate_point(point: Sequence[int], label: str) -> Tuple[int, int]:
    """Validate an input ``[x, y]`` pair.

    Args:
        point: Sequence with exactly two integer coordinates.
        label: Human readable name used in error messages.

    Returns:
        The validated ``(x, y)`` tuple.

    Raises:
        ValueError: If the shape, type or range of the coordinates is invalid.
    """
    try:
        items = list(point)
    except TypeError as exc:
        raise ValueError(f"{label} location must be a pair of integers") from exc
    if len(items) != 2:
        raise ValueError(f"{label} location must contain exactly 2 values")
    for value in items:
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} coordinates must be integers, got {value!r}")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(
                f"{label} coordinates must be within 0-{MAP_SIZE}, got {value}"
            )
    return items[0], items[1]


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier (starts at 1).
        status: One of "pending", "assigned", "picked_up", "completed".
        raw_res_pos: Restaurant coordinates as entered by the user.
        raw_cus_pos: Customer coordinates as entered by the user.
        res_pos: Restaurant grid-cell centre ``(x, y)``.
        cus_pos: Customer grid-cell centre ``(x, y)``.
        rider_id: ID of the assigned rider, or None.
        start_time: Tick at which the order was created.
        pickup_time: Tick at which the rider reached the restaurant.
        complete_time: Tick at which the order was delivered.
        distance: Street (Manhattan) distance between restaurant and customer.
        time_taken: Ticks from creation to delivery.
        efficiency: Distance covered per tick over the whole order.
    """

    _id_counter = 0

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order and convert its coordinates to cell centres.

        Args:
            res_pos: Restaurant coordinates ``[x, y]`` in the range 0-800.
            cus_pos: Customer coordinates ``[x, y]`` in the range 0-800.

        Raises:
            ValueError: If a coordinate is not an integer or is out of range.
        """
        raw_res = validate_point(res_pos, "restaurant")
        raw_cus = validate_point(cus_pos, "customer")

        Order._id_counter += 1
        self.id: int = Order._id_counter
        self.raw_res_pos: Tuple[int, int] = raw_res
        self.raw_cus_pos: Tuple[int, int] = raw_cus
        self.res_pos: Tuple[int, int] = (
            PathPlanner.cell_center(raw_res[0]),
            PathPlanner.cell_center(raw_res[1]),
        )
        self.cus_pos: Tuple[int, int] = (
            PathPlanner.cell_center(raw_cus[0]),
            PathPlanner.cell_center(raw_cus[1]),
        )
        self.status: str = STATUS_PENDING
        self.rider_id: Optional[int] = None
        self.start_time: int = int(globalvar.get_value("time", 0) or 0)
        self.pickup_time: Optional[int] = None
        self.complete_time: Optional[int] = None
        self.distance: int = abs(self.res_pos[0] - self.cus_pos[0]) + abs(
            self.res_pos[1] - self.cus_pos[1]
        )
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to the rider with the given ID.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_picked_up(self, tick: int) -> None:
        """Record that the rider reached the restaurant.

        Args:
            tick: Simulation tick of the pickup.
        """
        self.pickup_time = tick
        self.status = STATUS_PICKED_UP

    def complete(self, tick: int) -> None:
        """Mark the order delivered and record completion statistics.

        Args:
            tick: Simulation tick of the delivery.
        """
        self.complete_time = tick
        self.status = STATUS_COMPLETED
        self.time_taken = max(0, tick - self.start_time)
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 3)
        else:
            self.efficiency = float(self.distance)

    def is_active(self) -> bool:
        """Return True while the order has not been completed."""
        return self.status != STATUS_COMPLETED

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "status": self.status,
            "rider_id": self.rider_id,
            "raw_res_pos": list(self.raw_res_pos),
            "raw_cus_pos": list(self.raw_cus_pos),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

"""Rider model: state machine, grid movement and event logging."""

import logging
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_SPEED = 10
SPAWN_X = 435
SPAWN_Y = 385


class Rider:
    """A delivery rider that moves across the grid and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates.
        speed: Distance covered per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders; ``orders[0]`` is the one being served.
        special_case: True while the rider is aligned with its target on one
            axis (directly above/below or beside it).
        path: Remaining waypoints towards the current target.
        distance_travelled: Total units moved since creation.
        log: Chronological event entries. Each entry is a dict with keys
            "event", "order_id", "time", "rider_id", "x", "y" and "state".
            Events: "spawn", "accepted_order", "arrived_at_merchant",
            "delivered_to_customer".
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the spawn point in the idle state.

        Args:
            rider_id: Unique rider identifier.
        """
        self.rider_id: int = int(rider_id)
        self.x: int = SPAWN_X
        self.y: int = SPAWN_Y
        self.speed: int = DEFAULT_SPEED
        self.state: int = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.path: List[Tuple[int, int]] = []
        self.distance_travelled: int = 0
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)
        logger.info("Rider #%d spawned at (%d, %d)", self.rider_id, self.x, self.y)

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _now() -> int:
        """Return the current simulation tick from the global store."""
        return int(globalvar.get_value("time", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an entry to the rider's event log."""
        self.log.append(
            {
                "event": event,
                "order_id": order_id,
                "time": self._now(),
                "rider_id": self.rider_id,
                "x": self.x,
                "y": self.y,
                "state": self.state,
            }
        )

    @property
    def state_name(self) -> str:
        """Human readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def position(self) -> Tuple[int, int]:
        """Current ``(x, y)`` position."""
        return self.x, self.y

    def is_idle(self) -> bool:
        """Return True when the rider is idle and has no queued orders."""
        return self.state == STATE_IDLE and not self.orders

    def _plan_to(self, target: Tuple[int, int]) -> None:
        """Compute the route to ``target`` and update the special-case flag."""
        self.path = PathPlanner.plan((self.x, self.y), target)
        self.special_case = self.x == target[0] or self.y == target[1]

    def _advance(self) -> None:
        """Move up to ``speed`` units along the current path."""
        if not self.path:
            return
        old_x, old_y = self.x, self.y
        (self.x, self.y), self.path = PathPlanner.advance(
            self.x, self.y, self.path, self.speed
        )
        self.distance_travelled += abs(self.x - old_x) + abs(self.y - old_y)

    def _reset_idle(self) -> None:
        """Return the rider to the idle state, releasing its resources."""
        self.state = STATE_IDLE
        self.path = []
        self.special_case = False

    def _complete_current_order(self) -> None:
        """Finish the order being delivered and update the rider state."""
        order = self.orders.pop(0)
        order.complete(self._now())
        self._record("delivered_to_customer", order.id)
        logger.info("Rider #%d delivered order #%d", self.rider_id, order.id)
        if self.orders:
            # Serve the next queued order: head back to its restaurant.
            self.state = STATE_PICKING_UP
            self._plan_to(self.orders[0].res_pos)
        else:
            self._reset_idle()

    # --------------------------------------------------------------- public API
    def move(self) -> None:
        """Advance the rider by one tick.

        Handles all state transitions and their log entries:
            Idle -> Picking Up when an order is assigned ("accepted_order").
            Picking Up -> Delivering on arrival at the restaurant
            ("arrived_at_merchant").
            Delivering -> Idle on delivery ("delivered_to_customer").
        """
        if self.state == STATE_IDLE:
            if not self.orders:
                return
            order = self.orders[0]
            self.state = STATE_PICKING_UP
            self._plan_to(order.res_pos)
            self._record("accepted_order", order.id)
            logger.debug("Rider #%d accepted order #%d", self.rider_id, order.id)

        if self.state == STATE_PICKING_UP:
            if not self.orders:
                logger.warning("Rider #%d picking up without orders; resetting", self.rider_id)
                self._reset_idle()
                return
            self._advance()
            if not self.path:
                order = self.orders[0]
                order.mark_picked_up(self._now())
                self.state = STATE_DELIVERING
                self._plan_to(order.cus_pos)
                self._record("arrived_at_merchant", order.id)
        elif self.state == STATE_DELIVERING:
            if not self.orders:
                logger.warning("Rider #%d delivering without orders; resetting", self.rider_id)
                self._reset_idle()
                return
            self._advance()
            if not self.path:
                self._complete_current_order()

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "state_name": self.state_name,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

"""Order management: creation, nearest-rider selection and assignment."""

import logging
import math
from typing import List, Optional

import globalvar
from order_model import STATUS_PENDING, Order
from rider import Rider

logger = logging.getLogger(__name__)

# Extra score added per order already queued on a rider. This balances load
# between riders while keeping the distance-first strategy dominant.
LOAD_PENALTY = 150.0


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the best rider for an order.

    The score of each rider is the Euclidean distance to the restaurant,
    ``sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2)``, plus a
    load penalty for each order already assigned. The lowest score wins, so the
    closest rider is preferred while busy riders are only chosen when they are
    clearly closer or idle riders are unavailable.

    Args:
        order: The order needing a rider.

    Returns:
        The selected rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders", []) or []
    res_x, res_y = order.res_pos
    best: Optional[Rider] = None
    best_score = math.inf
    for rider in riders:
        distance = math.hypot(rider.x - res_x, rider.y - res_y)
        score = distance + LOAD_PENALTY * len(rider.orders)
        if score < best_score:
            best = rider
            best_score = score
    return best


class OrderManager:
    """Creates, tracks, and assigns orders."""

    def create_order(self, res_pos, cus_pos) -> Order:
        """Create an order and register it in the global order list.

        Args:
            res_pos: Restaurant coordinates ``[x, y]``.
            cus_pos: Customer coordinates ``[x, y]``.

        Returns:
            The newly created order.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        order = Order(res_pos, cus_pos)
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        logger.info("Created order #%d", order.id)
        return order

    @staticmethod
    def all_orders() -> List[Order]:
        """Return every order created so far."""
        return list(globalvar.get_value("orders", []) or [])

    def pending_orders(self) -> List[Order]:
        """Return orders that have not yet been assigned to a rider."""
        return [o for o in self.all_orders() if o.status == STATUS_PENDING]

    def assign_order(self, order: Order) -> Optional[Rider]:
        """Assign an order to the best available rider.

        Args:
            order: The pending order to assign.

        Returns:
            The rider that received the order, or None if no rider exists.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            logger.warning("No rider available for order #%d", order.id)
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Assigned order #%d to rider #%d", order.id, rider.rider_id)
        return rider

"""System initialisation and the SystemManager core class.

SystemManager owns the business rules: funds, automatic rider recruitment,
order dispatch and the simulation clock. All mutable state lives in globalvar.
"""

import json
import logging
import os
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from order import OrderManager
from order_model import STATUS_COMPLETED, STATUS_PENDING, Order
from rider import STATE_IDLE, Rider

logger = logging.getLogger(__name__)

DEFAULT_CONFIG: Dict[str, int] = {
    "initial_funds": 1000,
    "recruit_cost": 300,
    "recruit_min_funds": 300,
}

CONFIG_FILE = os.path.normpath(
    os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "config.json")
)

MAX_TICKS_PER_COMMAND = 1000


def _load_config(path: str) -> Dict[str, int]:
    """Load system parameters from a JSON file, falling back to defaults.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        Configuration dictionary containing every key in DEFAULT_CONFIG.
    """
    config = dict(DEFAULT_CONFIG)
    if not os.path.isfile(path):
        return config
    try:
        with open(path, encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read config %s: %s; using defaults", path, exc)
        return config
    if not isinstance(data, dict):
        logger.warning("Config %s must contain a JSON object; using defaults", path)
        return config
    for key in DEFAULT_CONFIG:
        value = data.get(key)
        if isinstance(value, int) and not isinstance(value, bool):
            config[key] = value
    return config


def init_system() -> "SystemManager":
    """Initialise global variables and return the system manager.

    Sets system funds (from the configuration, default 1000), an empty rider
    list, rider ID counter 0, the running flag True and simulated time 0.

    Returns:
        A ready-to-use SystemManager instance.
    """
    globalvar._init()
    config = _load_config(CONFIG_FILE)
    globalvar.set_value("config", config)
    globalvar.set_value("system_funds", int(config["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    logger.info("System initialised with funds=%s", config["initial_funds"])
    return SystemManager()


class SystemManager:
    """Core manager for funds, riders, dispatching and the simulation clock."""

    def __init__(self) -> None:
        """Create the manager bound to the global state."""
        self.order_manager = OrderManager()

    # ----------------------------------------------------------- properties
    @property
    def config(self) -> Dict[str, int]:
        """System configuration parameters."""
        return globalvar.get_value("config", dict(DEFAULT_CONFIG)) or dict(DEFAULT_CONFIG)

    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return int(globalvar.get_value("system_funds", 0))

    @funds.setter
    def funds(self, value: int) -> None:
        globalvar.set_value("system_funds", int(value))

    @property
    def riders(self) -> List[Rider]:
        """Riders currently in service."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def time(self) -> int:
        """Current simulation tick."""
        return int(globalvar.get_value("time", 0) or 0)

    @property
    def running(self) -> bool:
        """Main loop control flag."""
        return bool(globalvar.get_value("running", False))

    def stop(self) -> None:
        """Clear the running flag so the main loop exits."""
        globalvar.set_value("running", False)

    # ---------------------------------------------------------- queries
    def idle_riders(self) -> List[Rider]:
        """Return riders with state idle and no assigned orders."""
        return [rider for rider in self.riders if rider.is_idle()]

    def find_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with the given ID, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    # ---------------------------------------------------- recruitment
    def should_recruit(self, pending_count: int) -> bool:
        """Decide whether a new rider must be recruited.

        Recruitment happens when funds are at least the minimum threshold
        (300) and the number of pending orders exceeds the number of idle riders.

        Args:
            pending_count: Number of orders waiting for a rider.

        Returns:
            True if a rider should be recruited.
        """
        return (
            self.funds >= int(self.config["recruit_min_funds"])
            and pending_count > len(self.idle_riders())
        )

    def recruit_rider(self) -> Rider:
        """Create a new rider and deduct the recruitment cost.

        Returns:
            The newly recruited rider.
        """
        cost = int(self.config["recruit_cost"])
        self.funds -= cost
        counter = int(globalvar.get_value("rider_id_counter", 0) or 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        logger.info("Recruited rider #%d for %d (funds now %d)", counter, cost, self.funds)
        return rider

    def _ensure_capacity(self) -> bool:
        """Recruit one rider if the recruitment rule applies."""
        pending = len(self.order_manager.pending_orders())
        if self.should_recruit(pending):
            self.recruit_rider()
            return True
        return False

    # ---------------------------------------------------- dispatching
    def submit_order(self, res_pos, cus_pos) -> Dict[str, Any]:
        """Create an order, recruit if needed and assign it to a rider.

        Args:
            res_pos: Restaurant coordinates ``[x, y]`` (0-800).
            cus_pos: Customer coordinates ``[x, y]`` (0-800).

        Returns:
            Dictionary with keys "order" (Order), "rider" (Rider or None) and
            "new_rider" (bool, True if a rider was recruited for this order).
        """
        order: Order = self.order_manager.create_order(res_pos, cus_pos)
        new_rider = self._ensure_capacity()
        rider = self.order_manager.assign_order(order)
        return {"order": order, "rider": rider, "new_rider": new_rider}

    def dispatch_pending(self) -> bool:
        """Try to assign every pending order, recruiting riders when allowed.

        Returns:
            True if at least one rider was recruited during the call.
        """
        recruited = False
        for order in self.order_manager.pending_orders():
            if self._ensure_capacity():
                recruited = True
            if self.order_manager.assign_order(order) is None:
                break
        return recruited

    # ---------------------------------------------------- simulation
    def tick(self) -> None:
        """Advance the simulation by one unit of time.

        Increments the clock, dispatches pending orders, then moves every rider.
        A failure in one rider is logged and does not stop the others.
        """
        globalvar.set_value("time", self.time + 1)
        try:
            self.dispatch_pending()
        except Exception:  # pragma: no cover - defensive recovery
            logger.exception("Dispatch failed at tick %d", self.time)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                logger.exception("Rider #%d failed at tick %d", rider.rider_id, self.time)

    def check_consistency(self) -> List[str]:
        """Verify internal invariants and return a list of problems found.

        Returns:
            Human readable problem descriptions; empty when consistent.
        """
        problems: List[str] = []
        rider_ids = {rider.rider_id for rider in self.riders}
        for rider in self.riders:
            if rider.state != STATE_IDLE and not rider.orders:
                problems.append(f"Rider #{rider.rider_id} is {rider.state_name} without orders")
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    problems.append(
                        f"Order #{order.id} is queued on rider #{rider.rider_id} "
                        f"but records rider #{order.rider_id}"
                    )
        for order in self.order_manager.all_orders():
            if order.status not in (STATUS_PENDING, STATUS_COMPLETED):
                if order.rider_id not in rider_ids:
                    problems.append(f"Order #{order.id} references missing rider {order.rider_id}")
        return problems

    # ---------------------------------------------------- persistence
    def save_state(self, path: str) -> None:
        """Write the system parameters, riders and orders to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        state = {
            "time": self.time,
            "system_funds": self.funds,
            "config": self.config,
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in self.order_manager.all_orders()],
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(state, handle, indent=2)
        logger.info("State saved to %s", path)

Point = Tuple[int, int]

SYMBOL_EMPTY = "."
SYMBOL_STREET = "#"
SYMBOL_CUSTOMER = "C"
SYMBOL_RESTAURANT = "S"
SYMBOL_RIDER = "R"

_PRIORITY = {
    SYMBOL_EMPTY: 0,
    SYMBOL_STREET: 1,
    SYMBOL_CUSTOMER: 2,
    SYMBOL_RESTAURANT: 3,
    SYMBOL_RIDER: 4,

class DisplayManager:
    """Formats system status, order tables, rider logs and the map."""

    @staticmethod
    def _cell_label(x: int, y: int) -> str:
        """Return a readable grid-cell label such as ``(4,3)``."""
        col, row = PathPlanner.cell_of(x, y)
        return f"({col},{row})"

    def render_status(self, system) -> str:
        """Render funds, time and a snapshot of every rider.

        Args:
            system: The SystemManager instance.

        Returns:
            Multi-line status text.
        """
        lines = [
            f"System time : tick {system.time}",
            f"System funds: {system.funds}",
            f"Riders      : {len(system.riders)}",
        ]
        if not system.riders:
            lines.append("  (no riders recruited yet)")
        for rider in system.riders:
            order_ids = [order.id for order in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id}: location=({rider.x}, {rider.y}) "
                f"cell={self._cell_label(rider.x, rider.y)} "
                f"state={rider.state_name} speed={rider.speed} "
                f"orders={len(rider.orders)} {order_ids}"

            f"{'Dist':>6} {'Taken':>6} {'Rider':>6} {'Status':>10}"
        )
        lines = [header, "-" * len(header)]
        if not orders:
            lines.append("(no orders yet)")
        for order in orders:
            restaurant = f"({order.res_pos[0]},{order.res_pos[1]})"
            customer = f"({order.cus_pos[0]},{order.cus_pos[1]})"
            taken = "-" if order.time_taken is None else str(order.time_taken)
            rider = "-" if order.rider_id is None else f"#{order.rider_id}"
            lines.append(
                f"{order.id:>4} {order.start_time:>6} {restaurant:>14} {customer:>14} "
                f"{order.distance:>6} {taken:>6} {rider:>6} {order.status:>10}"
            )

        Returns:

        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} (current: {rider.state_name})")

                lines.append("  (no events)")

                order = "-" if entry["order_id"] is None else f"#{entry['order_id']}"
                lines.append(

    def render_map(
        self,
        riders: Sequence[Rider],
        orders: Iterable[Order],
        trail: Optional[Iterable[Point]] = None,
    ) -> str:
        """Render the 8x8 ASCII map.

        Symbols: 'R' rider, 'S' restaurant, 'C' customer, '#' street/route,
        '.' empty. Only active (not completed) orders are drawn. When several
        symbols share a cell, the higher-priority one is shown (R > S > C > #).

        Args:
            riders: Riders to draw.
            orders: Orders whose restaurant/customer markers are drawn.
            trail: Optional sequence of points whose cells are drawn as route.

        Returns:
            Multi-line map text.
        """
        grid = [[SYMBOL_EMPTY] * GRID_CELLS for _ in range(GRID_CELLS)]

        def place(point: Point, symbol: str) -> None:
            col, row = PathPlanner.cell_of(point[0], point[1])
            if _PRIORITY[symbol] >= _PRIORITY[grid[row][col]]:
                grid[row][col] = symbol

        for x, y in trail or []:
            place((x, y), SYMBOL_STREET)
        for order in orders:
            if order.status == STATUS_COMPLETED:
                continue
            place(order.cus_pos, SYMBOL_CUSTOMER)
            place(order.res_pos, SYMBOL_RESTAURANT)
        for rider in riders:
            place((rider.x, rider.y), SYMBOL_RIDER)

        lines = ["Map (rows = y cell, columns = x cell; each cell is 100x100 units):"]
        lines.append(" " * 16 + "  ".join(str(col) for col in range(GRID_CELLS)))
        for row in reversed(range(GRID_CELLS)):
            cells = "  ".join(grid[row][col] for col in range(GRID_CELLS))
            lines.append(f"row {row} [y {row * CELL_SIZE:>3}] | {cells}")
        lines.append(
            "Legend: R rider, S restaurant, C customer, # route/street, . empty"
        )
        return "\n".join(lines)

    def render_route(
        self,
        order: Order,
        rider: Rider,
        trail: Sequence[Point],
        ticks: int,
        new_rider: bool,
    ) -> str:
        """Render the route summary and route map for an assigned order.

        Args:
            order: The order that was dispatched.
            rider: The rider that serves the order.
            trail: Positions of the rider at every tick, starting at the
                position before the first move.
            ticks: Number of simulation ticks the trip took.
            new_rider: Whether a new rider was recruited for this order.

        Returns:
            Multi-line report text.
        """
        route_cells: Set[Tuple[int, int]] = {PathPlanner.cell_of(x, y) for x, y in trail}
        res_cell = PathPlanner.cell_of(order.raw_res_pos[0], order.raw_res_pos[1])
        cus_cell = PathPlanner.cell_of(order.raw_cus_pos[0], order.raw_cus_pos[1])
        pickup = "-" if order.pickup_time is None else str(order.pickup_time)
        done = "-" if order.complete_time is None else str(order.complete_time)
        lines = [
            f"Order #{order.id} assigned to Rider #{rider.rider_id}",
            f"Restaurant: input {order.raw_res_pos} -> grid cell {res_cell} "
            f"-> centre {order.res_pos}",
            f"Customer  : input {order.raw_cus_pos} -> grid cell {cus_cell} "
            f"-> centre {order.cus_pos}",
            f"New rider added: {'yes' if new_rider else 'no'}",
            f"Rider speed: {rider.speed} units/tick",
            f"Time: started tick {order.start_time}, picked up tick {pickup}, "
            f"delivered tick {done} (trip ticks: {ticks})",
            f"Route cells visited: {len(route_cells)}",
            "",
            self.render_map([rider], [order], trail),
        ]
        return "\n".join(lines)

"""Command parsing, shortcut resolution and input validation."""

from typing import Callable, List, Optional, Sequence, Tuple

COORDINATE_NAMES = ("restaurant_x", "restaurant_y", "customer_x", "customer_y")
COORDINATE_MIN = 0
COORDINATE_MAX = 800
MAX_HISTORY = 100


class InputError(ValueError):
    """Raised when user input does not satisfy the expected format or range."""


class InputManager:
    """Reads lines, resolves shortcut keys, keeps history and validates input."""

    COMMANDS = (
        "start",
        "status",
        "orders",
        "riders",
        "map",
        "tick",
        "save",
        "history",
        "help",
        "quit",
        "exit",
    )

    SHORTCUTS = {
        "s": "start",
        "t": "status",
        "o": "orders",
        "r": "riders",
        "m": "map",
        "k": "tick",
        "v": "save",
        "y": "history",
        "h": "help",
        "?": "help",
        "q": "quit",
    }

    def __init__(self, reader: Optional[Callable[[str], str]] = None) -> None:
        """Create the input manager.

        Args:
            reader: Callable used to read a line (defaults to ``input``); a
                prompt string is passed to it.
        """
        self._reader = reader or input
        self.history: List[str] = []

    def read_line(self, prompt: str = "> ") -> str:
        """Read one stripped line and remember non-empty entries.

        Args:
            prompt: Prompt text shown to the user.

        Returns:
            The stripped input text (may be empty).
        """
        text = self._reader(prompt).strip()
        if text:
            self.history.append(text)
            if len(self.history) > MAX_HISTORY:
                del self.history[0]
        return text

    @classmethod
    def parse_command(cls, text: str) -> Tuple[str, List[str]]:
        """Split a command line into a canonical command name and arguments.

        Args:
            text: Raw command line.

        Returns:
            Tuple ``(command, arguments)``; the command is "" for blank input.
        """
        parts = text.strip().split()
        if not parts:
            return "", []
        name = parts[0].lower()
        name = cls.SHORTCUTS.get(name, name)
        return name, parts[1:]

    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> List[int]:
        """Parse and validate four integer coordinates.

        Accepts whitespace- or comma-separated values, e.g.
        ``"100 200 300 400"`` or ``"100,200,300,400"``.

        Args:
            tokens: Argument tokens or whole input strings.

        Returns:
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            InputError: If the count, type or range of the values is invalid.
        """
        raw: List[str] = []
        for token in tokens:
            raw.extend(part for part in token.replace(",", " ").split() if part)
        if len(raw) != 4:
            raise InputError(
                "expected 4 integer coordinates: restaurant_x restaurant_y "
                f"customer_x customer_y (got {len(raw)})"
            )
        values: List[int] = []
        for name, token in zip(COORDINATE_NAMES, raw):
            try:
                value = int(token)
            except ValueError as exc:
                raise InputError(f"{name} must be an integer, got '{token}'") from exc
            if not COORDINATE_MIN <= value <= COORDINATE_MAX:
                raise InputError(
                    f"{name} must be in range {COORDINATE_MIN}-{COORDINATE_MAX}, got {value}"
                )
            values.append(value)
        return values

    @staticmethod
    def parse_positive_int(tokens: Sequence[str], default: int, maximum: int) -> int:
        """Parse an optional positive integer argument.

        Args:
            tokens: Argument tokens; the first one is used when present.
            default: Value returned when no argument is given.
            maximum: Upper bound (inclusive).

        Returns:
            The parsed integer.

        Raises:
            InputError: If the value is not an integer within 1..maximum.
        """
        if not tokens:
            return default
        try:
            value = int(tokens[0])
        except ValueError as exc:
            raise InputError(f"expected an integer, got '{tokens[0]}'") from exc
        if not 1 <= value <= maximum:
            raise InputError(f"value must be between 1 and {maximum}, got {value}")
        return value

def simulate_until_delivered(
    system: SystemManager, order: Order, rider: Rider
) -> Tuple_List:
    """Advance the simulation until the order is delivered.

    Args:
        system: The SystemManager driving the simulation.
        order: Order being tracked.
        rider: Rider expected to serve the order (its positions are traced).

    Returns:
        Tuple ``(trail, ticks)`` where ``trail`` lists the rider's positions
        from the start of the trip and ``ticks`` is the number of ticks run.
    """
    trail = [(rider.x, rider.y)]
    ticks = 0
    while order.status != STATUS_COMPLETED and ticks < MAX_TRIP_TICKS:
        system.tick()
        ticks += 1
        trail.append((rider.x, rider.y))
    return trail, ticks


class App:
    """Interactive command loop tying together input, system and display."""

    def __init__(self, reader=None, writer=print) -> None:
        """Initialise the system and the UI helpers.

        Args:
            reader: Optional line reader (defaults to ``input``).
            writer: Output callable (defaults to ``print``).
        """
        self.system = init_system()
        self.display = DisplayManager()
        self.inp = InputManager(reader)
        self.out = writer
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    # ----------------------------------------------------------- main loop
    def run(self) -> None:
        """Run the interactive loop until quit, EOF or interrupt."""
        self.out("Food Delivery Order Dispatch System (8x8 grid, 800x800 units)")
        self.out("Type 'help' for the command list.")
        while self.system.running:
            try:
                line = self.inp.read_line("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self.out("")
                break
            name, args = InputManager.parse_command(line)
            if not name:
                continue
            handler = self.handlers.get(name)
            if handler is None:
                self.out(f"Unknown command '{name}'. Type 'help' for the command list.")
                continue
            try:
                handler(args)
            except (EOFError, KeyboardInterrupt):
                self.out("")
                break
            except InputError as exc:
                self.out(f"Input error: {exc}")
            except Exception:  # Error recovery: keep the simulation running.
                logger.exception("Command '%s' failed", name)
                self.out("An unexpected error occurred; the system state was kept. See log.")
        self.system.stop()
        self.out("Goodbye.")

    # ------------------------------------------------------------ helpers
    def _prompt_coordinates(self):
        """Prompt until valid coordinates or a blank line (cancel) is entered."""
        while True:
            text = self.inp.read_line(
                "Enter restaurant_x restaurant_y customer_x customer_y (blank to cancel): "
            )
            if not text:
                return None
            try:
                return InputManager.parse_coordinates([text])
            except InputError as exc:
                self.out(f"Input error: {exc}. Try again.")

    # ----------------------------------------------------------- commands
    def cmd_start(self, args) -> None:
        """Create an order, dispatch it and simulate it to completion."""
        if args:
            coords = InputManager.parse_coordinates(args)
        else:
            coords = self._prompt_coordinates()
            if coords is None:
                self.out("Order cancelled.")
                return
        res_pos = [coords[0], coords[1]]
        cus_pos = [coords[2], coords[3]]
        result = self.system.submit_order(res_pos, cus_pos)
        order = result["order"]
        rider = result["rider"]
        if rider is None:
            self.out(
                f"Order #{order.id} is pending: no rider is available "
                f"(funds {self.system.funds}, recruit needs >= "
                f"{self.system.config['recruit_min_funds']})."
            )
            return
        trail, ticks = simulate_until_delivered(self.system, order, rider)
        self.out(self.display.render_route(order, rider, trail, ticks, result["new_rider"]))
        if order.status != STATUS_COMPLETED:
            self.out(f"Order #{order.id} is not yet delivered after {ticks} ticks.")

    def cmd_status(self, args) -> None:
        """Show funds, riders and any consistency problems."""
        self.out(self.display.render_status(self.system))
        problems = self.system.check_consistency()
        for problem in problems:
            self.out(f"WARNING: {problem}")

    def cmd_orders(self, args) -> None:
        """Show the order table."""
        self.out(self.display.render_orders(self.system.order_manager.all_orders()))

    def cmd_riders(self, args) -> None:
        """Show the status-change log of all riders."""
        self.out(self.display.render_rider_log(self.system.riders))

    def cmd_map(self, args) -> None:

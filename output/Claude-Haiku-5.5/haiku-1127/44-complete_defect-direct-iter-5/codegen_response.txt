"""Data loading utilities for the Beijing Subway network.

The subway network dataset is stored in ``subway_data.json`` located in the
same directory as this module, so it can be loaded from any working directory.
"""

from __future__ import annotations

import json
import os
from typing import Any, Dict

DATA_FILE_PATH: str = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)

REQUIRED_TOP_LEVEL_KEYS = ("stations", "connections")


def load_subway_data(path: str = DATA_FILE_PATH) -> Dict[str, Any]:
    """Load and minimally validate the subway dataset.

    Args:
        path: Path to the JSON data file.

    Returns:
        The parsed JSON content as a dictionary with the keys
        ``stations`` and ``connections``.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the JSON is malformed or lacks the required structure.
    """
    with open(path, "r", encoding="utf-8") as handle:
        try:
            data = json.load(handle)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid JSON in '{path}': {exc}") from exc

    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")

    for key in REQUIRED_TOP_LEVEL_KEYS:
        if key not in data:
            raise ValueError(f"Subway data is missing the '{key}' section")
        if not isinstance(data[key], list):
            raise ValueError(f"Subway data section '{key}' must be a list")

    return data

"""Graph data structures used to model the subway network.

Stations are identified by integer station numbers assigned by the caller
(see ``subway_network.SubwayNetwork``). The graph is an undirected weighted
graph stored as adjacency lists; each undirected edge is stored once in each
endpoint's adjacency list so that traversal is direction-aware.
"""

from __future__ import annotations

from typing import Dict, Iterable, Iterator, List, Optional


class Edge:
    """A weighted connection between two stations.

    Attributes:
        source: Station number of the origin endpoint.
        dest: Station number of the destination endpoint.
        distance: Physical distance between the stations in meters.
        line: Name of the subway line operating this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source: int, dest: int, distance: float, line: str) -> None:
        if distance < 0:
            raise ValueError("Edge distance cannot be negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self) -> float:
        """Edge weight used by path algorithms (equal to the distance in meters)."""
        return self.distance

    def __repr__(self) -> str:
        return (
            f"Edge(source={self.source}, dest={self.dest}, "
            f"distance={self.distance}, line={self.line!r})"
        )


class Graph:
    """Weighted undirected graph with adjacency lists.

    Attributes:
        station_lines: Mapping from station name to the list of line names
            the station belongs to (transfer stations have several lines).
    """

    def __init__(self) -> None:
        self._adj: Dict[int, List[Edge]] = {}
        self.station_lines: Dict[str, List[str]] = {}

    # ------------------------------------------------------------------
    # Vertex management
    # ------------------------------------------------------------------
    def add_vertex(self, v: int) -> None:
        """Add a vertex if it is not already present."""
        self._adj.setdefault(v, [])

    def has_vertex(self, v: int) -> bool:
        """Return True if the vertex exists in the graph."""
        return v in self._adj

    def vertices(self) -> List[int]:
        """Return all vertex identifiers."""
        return list(self._adj.keys())

    def num_vertices(self) -> int:
        """Return the number of vertices."""
        return len(self._adj)

    def adj(self, v: int) -> List[Edge]:
        """Return the outgoing adjacency list of vertex ``v``.

        Raises:
            KeyError: If the vertex does not exist.
        """
        if v not in self._adj:
            raise KeyError(f"Unknown vertex: {v}")
        return self._adj[v]

    def set_station_lines(self, station_name: str, lines: Iterable[str]) -> None:
        """Set the line affiliations of a station, removing duplicates while keeping order."""
        self.station_lines[station_name] = list(dict.fromkeys(lines))

    # ------------------------------------------------------------------
    # Edge management
    # ------------------------------------------------------------------
    def get_edge(self, a: int, b: int) -> Optional[Edge]:
        """Return the edge from ``a`` to ``b`` if it exists, otherwise None."""
        for edge in self._adj.get(a, []):
            if edge.dest == b:
                return edge
        return None

    def add_edge(self, edge: Edge) -> None:
        """Add an undirected edge. If the pair is already connected, the
        existing connection is updated with the new distance and line.

        Raises:
            ValueError: If the edge is a self-loop.
        """
        if edge.source == edge.dest:
            raise ValueError("Self-loop edges are not allowed")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)

        existing = self.get_edge(edge.source, edge.dest)
        if existing is not None:
            existing.distance = edge.distance
            existing.line = edge.line
            reverse = self.get_edge(edge.dest, edge.source)
            if reverse is not None:
                reverse.distance = edge.distance
                reverse.line = edge.line
            return

        self._adj[edge.source].append(edge)
        self._adj[edge.dest].append(Edge(edge.dest, edge.source, edge.distance, edge.line))

    def set_edge_distance(self, a: int, b: int, distance: float) -> bool:
        """Update the distance of the connection between ``a`` and ``b``.

        Returns:
            True if the connection existed and was updated, False otherwise.

        Raises:
            ValueError: If the distance is negative.
        """
        if distance < 0:
            raise ValueError("Distance cannot be negative")
        forward = self.get_edge(a, b)
        if forward is None:
            return False
        forward.distance = distance
        backward = self.get_edge(b, a)
        if backward is not None:
            backward.distance = distance
        return True

    def edges(self) -> Iterator[Edge]:
        """Yield each undirected edge exactly once (oriented from lower to higher id)."""
        for lst in self._adj.values():
            for edge in lst:
                if edge.source < edge.dest:
                    yield edge

    def num_edges(self) -> int:
        """Return the number of undirected edges."""
        return sum(1 for _ in self.edges())

"""Path calculation for the Beijing Subway network.

Implements Dijkstra's algorithm on a binary-heap priority queue (``MinPQ``),
giving O((V+E) log V) complexity. The search runs over states of the form
``(station_number, arriving_line)`` so that line changes (transfers) can be
counted, limited, and penalised. Supported modes:

* ``shortest_distance`` - minimise total distance in meters.
* ``least_transfer``    - minimise number of transfers, then distance.
* ``comprehensive``     - minimise distance plus a penalty per transfer.

``PathfindingConfig.preferred_lines`` adds a penalty to segments on
non-preferred lines; ``PathfindingConfig.max_transfers`` prunes routes that
exceed the transfer limit.
"""

from __future__ import annotations

import heapq
import itertools
from typing import Any, Dict, List, Optional, Sequence, Set, Tuple

from models import Edge, Graph

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
MODES: Tuple[str, ...] = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance (meters) charged per transfer in comprehensive mode.
TRANSFER_PENALTY_M: float = 500.0
# Equivalent distance (meters) charged per segment on a non-preferred line.
PREFERENCE_PENALTY_M: float = 300.0

State = Tuple[int, Optional[str]]


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of ``MODES``.
        max_transfers: Maximum number of allowed transfers, or None for no limit.
        preferred_lines: Line names preferred by the passenger.
    """

    def __init__(
        self,
        mode: str = MODE_SHORTEST_DISTANCE,
        max_transfers: Optional[int] = None,
        preferred_lines: Optional[Sequence[str]] = None,
    ) -> None:
        if mode not in MODES:
            raise ValueError(f"Unsupported mode '{mode}', expected one of {MODES}")
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers cannot be negative")
        self.mode: str = mode
        self.max_transfers: Optional[int] = max_transfers
        self.preferred_lines: List[str] = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Min priority queue implemented on top of ``heapq`` (binary heap).

    Items with equal keys are dequeued in insertion order.
    """

    def __init__(self) -> None:
        self._heap: List[Tuple[Any, int, Any]] = []
        self._counter = itertools.count()

    def insert(self, key: Any, item: Any) -> None:
        """Insert an item with the given priority key."""
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self) -> Tuple[Any, Any]:
        """Remove and return ``(key, item)`` with the smallest key.

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("del_min from an empty priority queue")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min_key(self) -> Any:
        """Return the smallest key without removing it."""
        if not self._heap:
            raise IndexError("min_key from an empty priority queue")
        return self._heap[0][0]

    def is_empty(self) -> bool:
        """Return True if the queue has no items."""
        return not self._heap

    def __len__(self) -> int:
        return len(self._heap)


class _Label:
    """Best-known information for a search state."""

    __slots__ = ("key", "distance", "transfers", "penalty", "prev", "edge")

    def __init__(
        self,
        key: Tuple[Any, ...],
        distance: float,
        transfers: int,
        penalty: float,
        prev: Optional[State],
        edge: Optional[Edge],
    ) -> None:
        self.key = key
        self.distance = distance
        self.transfers = transfers
        self.penalty = penalty
        self.prev = prev
        self.edge = edge


class ShortestPath:
    """Dijkstra-based shortest path computation from a single source station.

    Attributes:
        dist_to: Mapping from station number to the travelled distance (meters)
            of the best route found from the source.
    """

    def __init__(self, graph: Graph, source: int, config: Optional[PathfindingConfig] = None) -> None:
        if not graph.has_vertex(source):
            raise ValueError(f"Unknown source station number: {source}")
        self.graph = graph
        self.source = source
        self.config: PathfindingConfig = config if config is not None else PathfindingConfig()
        self.dist_to: Dict[int, float] = {}
        self._transfers: Dict[int, int] = {}
        self._best_state: Dict[int, State] = {}
        self._labels: Dict[State, _Label] = {}
        self._run()

    def _make_key(self, distance: float, transfers: int, penalty: float) -> Tuple[Any, ...]:
        """Build the comparison key of a label according to the query mode."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, distance, penalty)
        if mode == MODE_COMPREHENSIVE:
            score = distance + TRANSFER_PENALTY_M * transfers + penalty
            return (score, distance, transfers)
        return (distance, transfers, penalty)

    def _run(self) -> None:
        """Execute Dijkstra's algorithm over (station, line) states."""
        preferred: Set[str] = set(self.config.preferred_lines)
        max_transfers = self.config.max_transfers

        start: State = (self.source, None)
        start_key = self._make_key(0.0, 0, 0.0)
        self._labels[start] = _Label(start_key, 0.0, 0, 0.0, None, None)

        pq = MinPQ()
        pq.insert(start_key, start)
        settled: Set[State] = set()

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)

            node, arrive_line = state
            label = self._labels[state]

            # The first settled state of a station is its best label.
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = label.distance
                self._transfers[node] = label.transfers

            for edge in self.graph.adj(node):
                transfers = label.transfers
                if arrive_line is not None and edge.line != arrive_line:
                    transfers += 1
                if max_transfers is not None and transfers > max_transfers:
                    continue

                next_state: State = (edge.dest, edge.line)
                if next_state in settled:
                    continue

                penalty = label.penalty
                if preferred and edge.line not in preferred:
                    penalty += PREFERENCE_PENALTY_M

                distance = label.distance + edge.distance
                key = self._make_key(distance, transfers, penalty)
                old = self._labels.get(next_state)
                if old is None or key < old.key:
                    self._labels[next_state] = _Label(key, distance, transfers, penalty, state, edge)
                    pq.insert(key, next_state)

    def has_path_to(self, target: int) -> bool:
        """Return True if ``target`` is reachable from the source under the current config."""
        return target in self._best_state

    def path_with_edges(self, target: int) -> Tuple[List[int], List[Edge]]:
        """Return the station numbers and traversed edges from source to target.

        Returns:
            A tuple ``(stations, edges)`` where ``stations`` starts with the
            source and ends with the target, and ``edges`` has one fewer
            element. If the target is unreachable, both lists are empty.
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        states: List[State] = [state]
        edges: List[Edge] = []
        while True:
            label = self._labels[state]
            if label.prev is None or label.edge is None:
                break
            edges.append(label.edge)
            state = label.prev
            states.append(state)
        states.reverse()
        edges.reverse()
        return [s[0] for s in states], edges

    def get_transfer_count(self, target: int) -> int:
        """Return the number of transfers on the best route to ``target``.

        Returns 0 if the target is unreachable.
        """
        return self._transfers.get(target, 0)

"""Fare calculation for the Beijing Subway.

Fare structure:
    0-6 km    : 3 yuan
    6-12 km   : 4 yuan
    12-22 km  : 5 yuan
    22-32 km  : 6 yuan
    over 32 km: 6 yuan + 1 yuan for every further (or partial) 20 km

Distances are measured in meters and rounded up to whole kilometers before
the tier lookup, so fractional distances are charged at the next kilometer.
"""

from __future__ import annotations

import math
from typing import List, Tuple

# Mapping of distance tiers: (upper bound in km inclusive, fare in yuan).
FARE_TIERS: List[Tuple[int, int]] = [(6, 3), (12, 4), (22, 5), (32, 6)]
EXTENSION_STEP_KM: int = 20
EXTENSION_STEP_FARE: int = 1

# Parameters for the rough travel time estimate.
AVERAGE_SPEED_KMH: float = 35.0
TRANSFER_TIME_MINUTES: float = 5.0


def _ceil_km(distance_m: float) -> int:
    """Round a distance in meters up to whole kilometers."""
    if distance_m < 0:
        raise ValueError("Distance cannot be negative")
    return math.ceil(distance_m / 1000.0)


def Sum_money(distance: float) -> int:
    """Calculate the subway fare in yuan for a distance given in meters.

    Args:
        distance: Travelled distance in meters.

    Returns:
        The fare in yuan as an integer.

    Raises:
        ValueError: If the distance is negative.
    """
    km = _ceil_km(distance)
    for upper_km, fare in FARE_TIERS:
        if km <= upper_km:
            return fare

    last_upper_km, last_fare = FARE_TIERS[-1]
    extra_km = km - last_upper_km
    steps = math.ceil(extra_km / EXTENSION_STEP_KM)
    return last_fare + steps * EXTENSION_STEP_FARE


def calculate_estimated_time(distance: float, transfer_count: int) -> float:
    """Estimate the travel time in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers on the route.

    Returns:
        Estimated travel time in minutes.

    Raises:
        ValueError: If distance is negative or transfer_count is negative.
    """
    if distance < 0:
        raise ValueError("Distance cannot be negative")
    if transfer_count < 0:
        raise ValueError("Transfer count cannot be negative")
    travel_minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    return travel_minutes + transfer_count * TRANSFER_TIME_MINUTES

"""Subway network management: station registry, connections and data maintenance.

``SubwayNetwork`` owns the weighted graph and the station name <-> station
number mappings. It supports building from the JSON dataset, name
standardisation, data updates (new lines, station modifications, distance
updates) and data integrity checks.
"""

from __future__ import annotations

from collections import defaultdict, deque
from typing import Any, Dict, Iterable, List, Optional, Set

from models import Edge, Graph

# Lines the dataset is expected to cover.
REQUIRED_LINES: List[str] = [
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
]

# Plausible range for a single inter-station distance in meters.
MIN_PLAUSIBLE_DISTANCE_M: float = 100
MAX_PLAUSIBLE_DISTANCE_M: float = 10000


class SubwayNetwork:
    """Subway network model combining the graph with station name mappings.

    Attributes:
        graph: Weighted undirected graph of stations.
        station_to_number: Mapping from station name to station number.
        number_to_station: Mapping from station number to station name.
    """

    def __init__(self) -> None:
        self.graph: Graph = Graph()
        self.station_to_number: Dict[str, int] = {}
        self.number_to_station: Dict[int, str] = {}
        self._next_number: int = 0

    # ------------------------------------------------------------------
    # Construction
    # ------------------------------------------------------------------
    @classmethod
    def from_data(cls, data: Dict[str, Any]) -> "SubwayNetwork":
        """Build a network from the parsed JSON dataset."""
        network = cls()
        for station in data.get("stations", []):
            network.add_station(station["name"], station.get("lines", []))
        for conn in data.get("connections", []):
            network.add_connection(
                conn["source"], conn["dest"], conn["distance"], conn["line"]
            )
        return network

    @property
    def station_lines(self) -> Dict[str, List[str]]:
        """Mapping from station name to its line names."""
        return self.graph.station_lines

    # ------------------------------------------------------------------
    # Station management
    # ------------------------------------------------------------------
    def add_station(self, name: str, lines: Optional[Iterable[str]] = None) -> int:
        """Register a station (or extend an existing one) and return its number."""
        name = name.strip()
        if not name:
            raise ValueError("Station name cannot be empty")
        if name in self.station_to_number:
            number = self.station_to_number[name]
        else:
            number = self._next_number
            self._next_number += 1
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
        existing = self.graph.station_lines.get(name, [])
        merged = list(existing) + list(lines or [])
        self.graph.set_station_lines(name, merged)
        return number

    def _ensure_station(self, name: str) -> int:
        """Return the number of a station, registering it if necessary."""
        if name in self.station_to_number:
            return self.station_to_number[name]
        return self.add_station(name)

    def has_station(self, name: str) -> bool:
        """Return True if a station with exactly this name exists."""
        return name in self.station_to_number

    def get_station_number(self, name: str) -> int:
        """Return the station number for a canonical station name.

        Raises:
            KeyError: If the station does not exist.
        """
        if name not in self.station_to_number:
            raise KeyError(f"Unknown station: {name}")
        return self.station_to_number[name]

    def station_name(self, number: int) -> str:
        """Return the station name for a station number."""
        return self.number_to_station[number]

    def station_count(self) -> int:
        """Return the number of registered stations."""
        return len(self.station_to_number)

    def resolve_station_name(self, query: str) -> Optional[str]:
        """Standardise a user-supplied station name to its official form.

        Accepts exact names, names with or without the trailing "站", and
        case/space-insensitive matches.

        Returns:
            The official station name, or None if no unique match exists.
        """
        q = query.strip()
        if not q:
            return None
        if q in self.station_to_number:
            return q
        if q.endswith("站") and q[:-1] in self.station_to_number:
            return q[:-1]
        if q + "站" in self.station_to_number:
            return q + "站"

        normalized = q.replace(" ", "").lower()
        matches = [
            name
            for name in self.station_to_number
            if name.replace(" ", "").lower() == normalized
        ]
        if len(matches) == 1:
            return matches[0]
        return None

    def modify_station(self, name: str, lines: Iterable[str]) -> None:
        """Replace the line affiliations of an existing station.

        Raises:
            KeyError: If the station does not exist.
        """
        if name not in self.station_to_number:
            raise KeyError(f"Unknown station: {name}")
        self.graph.set_station_lines(name, lines)

    # ------------------------------------------------------------------
    # Connection management
    # ------------------------------------------------------------------
    def add_connection(self, source_name: str, dest_name: str, distance: float, line: str) -> None:
        """Add or update a connection between two stations.

        Raises:
            ValueError: If the distance is not positive or the stations are identical.
        """
        if not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError(f"Invalid distance for {source_name}-{dest_name}: {distance}")
        if source_name == dest_name:
            raise ValueError("A station cannot be connected to itself")
        a = self._ensure_station(source_name)
        b = self._ensure_station(dest_name)
        self.graph.add_edge(Edge(a, b, distance, line))

    def update_distance(self, name_a: str, name_b: str, distance: float) -> None:
        """Update the distance of an existing connection.

        Raises:
            KeyError: If a station or the connection does not exist.
            ValueError: If the distance is not positive.
        """
        if distance <= 0:
            raise ValueError("Distance must be positive")
        a = self.get_station_number(name_a)
        b = self.get_station_number(name_b)
        if not self.graph.set_edge_distance(a, b, distance):
            raise KeyError(f"No connection between '{name_a}' and '{name_b}'")

    # ------------------------------------------------------------------
    # Line management
    # ------------------------------------------------------------------
    def line_names(self) -> Set[str]:
        """Return all line names known to the network."""
        names: Set[str] = set()
        for lines in self.graph.station_lines.values():
            names.update(lines)
        for edge in self.graph.edges():
            names.add(edge.line)
        return names

    def line_stations(self, line: str) -> List[str]:
        """Return the station names of a line in travel order."""
        adj: Dict[int, List[int]] = defaultdict(list)
        for edge in self.graph.edges():
            if edge.line == line:
                adj[edge.source].append(edge.dest)
                adj[edge.dest].append(edge.source)
        if not adj:
            return []

        endpoints = [v for v in sorted(adj) if len(adj[v]) == 1]
        start = endpoints[0] if endpoints else min(adj)
        ordered = [start]
        visited = {start}
        current = start
        while True:
            nxt = [w for w in adj[current] if w not in visited]
            if not nxt:
                break
            current = nxt[0]
            visited.add(current)
            ordered.append(current)
        return [self.number_to_station[v] for v in ordered]

    def add_line(self, line_name: str, station_names: List[str], distances: List[float]) -> None:
        """Add a new line as an ordered sequence of stations.

        Args:
            line_name: Name of the new line.
            station_names: Stations in travel order (at least two).
            distances: Distance in meters between consecutive stations.

        Raises:
            ValueError: If the line already exists or the input is inconsistent.
        """
        line_name = line_name.strip()
        if not line_name:
            raise ValueError("Line name cannot be empty")
        if line_name in self.line_names():
            raise ValueError(f"Line '{line_name}' already exists")
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        for d in distances:
            if not isinstance(d, (int, float)) or d <= 0:
                raise ValueError(f"Invalid distance: {d}")

        for name in station_names:
            self.add_station(name, [line_name])
        for i, d in enumerate(distances):
            self.add_connection(station_names[i], station_names[i + 1], d, line_name)

    # ------------------------------------------------------------------
    # Data integrity
    # ------------------------------------------------------------------
    def check_data_integrity(self) -> List[str]:
        """Run all integrity checks and return a list of issue descriptions.

        An empty list means the data passed every check.
        """
        issues: List[str] = []
        issues.extend(self._check_coverage())
        issues.extend(self._check_connections())
        issues.extend(self._check_distances())
        return issues

    def _check_coverage(self) -> List[str]:
        """Verify that the main lines are represented in the dataset."""
        issues: List[str] = []
        if not self.station_to_number:
            issues.append("Coverage: no stations are loaded")
            return issues
        present = self.line_names()
        for line in REQUIRED_LINES:
            if line not in present:
                issues.append(f"Coverage: required line '{line}' has no stations or connections")
        return issues

    def _check_connections(self) -> List[str]:
        """Verify adjacency relationships between stations."""
        issues: List[str] = []
        edge_lines: Dict[int, Set[str]] = {v: set() for v in self.graph.vertices()}
        for v in self.graph.vertices():
            for edge in self.graph.adj(v):
                edge_lines[v].add(edge.line)

        for name, number in self.station_to_number.items():
            lines = self.graph.station_lines.get(name, [])
            if not lines:
                issues.append(f"Connection: station '{name}' has no line affiliation")
            if not self.graph.adj(number):
                issues.append(f"Connection: station '{name}' has no adjacent stations")
                continue
            for line in lines:
                if line not in edge_lines[number]:
                    issues.append(
                        f"Connection: station '{name}' is on line '{line}' but has no connection on that line"
                    )

        connected = [v for v in self.graph.vertices() if self.graph.adj(v)]
        if connected:
            seen = self._reachable_from(connected[0])
            unreachable = [self.number_to_station[v] for v in connected if v not in seen]
            if unreachable:
                start_name = self.number_to_station[connected[0]]
                issues.append(
                    f"Connection: {len(unreachable)} station(s) unreachable from '{start_name}': "
                    + ", ".join(unreachable[:10])
                )
        return issues

    def _check_distances(self) -> List[str]:
        """Verify the plausibility of stored distances."""
        issues: List[str] = []
        for edge in self.graph.edges():
            a = self.number_to_station[edge.source]
            b = self.number_to_station[edge.dest]
            if not isinstance(edge.distance, (int, float)) or edge.distance <= 0:
                issues.append(f"Distance: connection {a}-{b} has invalid distance {edge.distance}")
            elif not (MIN_PLAUSIBLE_DISTANCE_M <= edge.distance <= MAX_PLAUSIBLE_DISTANCE_M):
                issues.append(
                    f"Distance: connection {a}-{b} has distance {edge.distance} m outside "
                    f"the plausible range [{MIN_PLAUSIBLE_DISTANCE_M}, {MAX_PLAUSIBLE_DISTANCE_M}]"
                )
        return issues

    def _reachable_from(self, start: int) -> Set[int]:
        """Return the set of vertices reachable from ``start`` (BFS)."""
        seen = {start}
        queue = deque([start])
        while queue:
            v = queue.popleft()
            for edge in self.graph.adj(v):
                if edge.dest not in seen:
                    seen.add(edge.dest)
                    queue.append(edge.dest)
        return seen

"""Path visualisation and export helpers.

Builds a structured summary of a route (stations, segments, transfers, distance,
fare, estimated time) and renders it as console text or JSON.
"""

from __future__ import annotations

import json
from typing import Any, Dict, List, Sequence

from fare_calculator import Sum_money, calculate_estimated_time
from models import Edge


def format_distance(distance_m: float) -> str:
    """Format a distance using meters for short and kilometers for long distances."""
    if distance_m < 1000:
        return f"{distance_m:.0f} m"
    return f"{distance_m / 1000.0:.2f} km"


def format_distance_both(distance_m: float) -> str:
    """Format a distance showing both the primary unit and its conversion."""
    if distance_m < 1000:
        return f"{distance_m:.0f} m ({distance_m / 1000.0:.2f} km)"
    return f"{distance_m / 1000.0:.2f} km ({distance_m:.0f} m)"


def find_transfers(stations: Sequence[str], edges: Sequence[Edge]) -> List[Dict[str, str]]:
    """Identify transfer stations, where the travelling line changes."""
    transfers: List[Dict[str, str]] = []
    for i in range(1, len(edges)):
        prev_line = edges[i - 1].line
        next_line = edges[i].line
        if prev_line != next_line:
            transfers.append(
                {"station": stations[i], "from_line": prev_line, "to_line": next_line}
            )
    return transfers


def build_segments(stations: Sequence[str], edges: Sequence[Edge]) -> List[Dict[str, Any]]:
    """Group consecutive edges operated by the same line into segments."""
    segments: List[Dict[str, Any]] = []
    if not edges:
        return segments
    start = 0
    for i in range(1, len(edges) + 1):
        if i == len(edges) or edges[i].line != edges[start].line:
            segments.append(
                {
                    "line": edges[start].line,
                    "from": stations[start],
                    "to": stations[i],
                    "station_count": i - start + 1,
                    "distance_m": sum(e.distance for e in edges[start:i]),
                }
            )
            start = i
    return segments


def build_summary(query: Dict[str, Any], stations: Sequence[str], edges: Sequence[Edge]) -> Dict[str, Any]:
    """Create the structured path summary.

    Args:
        query: Query information (start, end, mode, max_transfers, line_preference).
        stations: Station names from start to end.
        edges: Edges traversed between consecutive stations.

    Returns:
        A JSON-serialisable dictionary describing the route.

    Raises:
        ValueError: If the station and edge lists are inconsistent.
    """
    if len(stations) != len(edges) + 1:
        raise ValueError("Inconsistent route: stations and edges do not match")

    distance = sum(e.distance for e in edges)
    transfers = find_transfers(stations, edges)
    lines: List[str] = []
    for edge in edges:
        if edge.line not in lines:
            lines.append(edge.line)

    fare = Sum_money(distance)
    estimated = calculate_estimated_time(distance, len(transfers))

    return {
        "query": dict(query),
        "path": list(stations),
        "distance_m": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(estimated, 2),
        "transfer_count": len(transfers),
        "transfers": transfers,
        "lines": lines,
        "segments": build_segments(stations, edges),
    }


def format_text(summary: Dict[str, Any]) -> str:
    """Render a summary as console-friendly text."""
    query = summary["query"]
    out: List[str] = []
    out.append(f"Route: {query['start']} -> {query['end']}")
    out.append(f"Query mode: {query['mode']}")
    out.append("")

    out.append(f"Station sequence ({len(summary['path'])} stations):")
    for index, name in enumerate(summary["path"], start=1):
        out.append(f"  {index:>3}. {name}")
    out.append("")

    out.append("Line segments:")
    for seg in summary["segments"]:
        out.append(
            f"  - {seg['line']}: {seg['from']} -> {seg['to']} "
            f"({seg['station_count']} stations, {format_distance(seg['distance_m'])})"
        )
    out.append("")

    out.append(f"Total distance: {format_distance_both(summary['distance_m'])}")
    out.append(f"Transfers: {summary['transfer_count']}")
    if summary["transfers"]:
        out.append("Transfer stations:")
        for t in summary["transfers"]:
            out.append(f"  - {t['station']}: {t['from_line']} -> {t['to_line']}")
    out.append("Lines involved: " + ", ".join(summary["lines"]))
    out.append(f"Estimated travel time: {summary['estimated_time_min']:.1f} min")
    out.append(f"Fare: {summary['fare']} yuan")
    return "\n".join(out)


def format_json(summary: Dict[str, Any]) -> str:
    """Render a summary as indented JSON."""
    return json.dumps(summary, ensure_ascii=False, indent=2)

"""Command-line interface for the Beijing Subway fare calculation system."""

from __future__ import annotations

import argparse
import sys
from typing import Iterable, List, Optional, Sequence, Tuple

from data_loader import DATA_FILE_PATH, load_subway_data
from fare_calculator import Sum_money  # noqa: F401  (re-exported for convenience)
from path_report import build_summary, format_json, format_text
from pathfinding import MODE_SHORTEST_DISTANCE, MODES, PathfindingConfig, ShortestPath
from subway_network import SubwayNetwork

QUIT_COMMANDS = {"quit", "exit", "q"}
OUTPUT_FORMATS = ("text", "json")


class QueryFormatError(ValueError):
    """Raised when a query string does not follow the 'Start; End' format."""


def parse_query(raw: str) -> Tuple[str, str]:
    """Split a 'Starting Station; Ending Station' query into its two names.

    Raises:
        QueryFormatError: If the separator is missing or a name is empty.
    """
    message = "Input format incorrect, please use 'Starting Station; Ending Station' format"
    if ";" not in raw:
        raise QueryFormatError(message)
    parts = raw.split(";")
    if len(parts) != 2:
        raise QueryFormatError(message)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryFormatError(message)
    return start, end


def parse_line_preference(value: Optional[str]) -> List[str]:
    """Parse a comma-separated list of preferred line names."""
    if not value:
        return []
    return [item.strip() for item in value.split(",") if item.strip()]


def parse_args(argv: Optional[Sequence[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (shortest path and fare query)."
    )
    parser.add_argument("--output_file", default=None,
                        help="File to which query results are appended.")
    parser.add_argument("--format", dest="output_format", choices=OUTPUT_FORMATS, default="text",
                        help="Output format: text or json (default: text).")
    parser.add_argument("--mode", choices=MODES, default=MODE_SHORTEST_DISTANCE,
                        help="Path query mode (default: shortest_distance).")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="Maximum number of allowed transfers.")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line(s), e.g. 'Line 1' or 'Line 1,Line 2'.")
    parser.add_argument("--check_data", action="store_true",
                        help="Run the data integrity check and exit.")
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")
    return args


class QueryService:
    """Executes station queries and produces formatted results."""

    def __init__(
        self,
        network: SubwayNetwork,
        mode: str = MODE_SHORTEST_DISTANCE,
        max_transfers: Optional[int] = None,
        preferred_lines: Optional[Iterable[str]] = None,
        output_format: str = "text",
        output_file: Optional[str] = None,
    ) -> None:
        self.network = network
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines: List[str] = list(preferred_lines or [])
        self.output_format = output_format
        self.output_file = output_file
        self.config = PathfindingConfig(mode, max_transfers, self.preferred_lines)

    def run_query(self, raw: str) -> str:
        """Process one 'Start; End' query and return the rendered result or an error message."""
        try:
            start_raw, end_raw = parse_query(raw)
        except QueryFormatError as exc:
            return f"Error: {exc}"

        start = self.network.resolve_station_name(start_raw)
        if start is None:
            return (f"Error: Starting station '{start_raw}' does not exist, "
                    f"please check the station name")
        end = self.network.resolve_station_name(end_raw)
        if end is None:
            return (f"Error: Ending station '{end_raw}' does not exist, "
                    f"please check the station name")
        if start == end:
            return "Error: Starting station cannot be the same as the ending station"

        try:
            source_number = self.network.get_station_number(start)
            target_number = self.network.get_station_number(end)
            finder = ShortestPath(self.network.graph, source_number, self.config)
            nodes, edges = finder.path_with_edges(target_number)
        except Exception as exc:  # pragma: no cover - defensive
            return f"Error: path calculation failed: {exc}"

        if not nodes:
            limit = "unlimited" if self.max_transfers is None else str(self.max_transfers)
            return (f"Error: no route found from '{start}' to '{end}' "
                    f"(mode={self.mode}, max_transfers={limit})")

        stations = [self.network.station_name(n) for n in nodes]
        query = {
            "start": start,
            "end": end,
            "mode": self.mode,
            "max_transfers": self.max_transfers,
            "line_preference": self.preferred_lines,
        }
        try:
            summary = build_summary(query, stations, edges)
            output = format_json(summary) if self.output_format == "json" else format_text(summary)
        except Exception as exc:
            return f"Error: fare calculation failed: {exc}"

        self._export(output)
        return output

    def _export(self, output: str) -> None:
        """Append a rendered result to the output file, if configured."""
        if not self.output_file:
            return
        try:
            with open(self.output_file, "a", encoding="utf-8") as handle:
                handle.write(output + "\n\n")
        except OSError as exc:
            print(f"Warning: could not write to '{self.output_file}': {exc}", file=sys.stderr)


def run_interactive(service: QueryService, input_func=input) -> None:
    """Run the continuous query loop until the user quits."""
    print("Beijing Subway fare calculator. Enter 'Starting Station; Ending Station' "
          "(or 'quit' to exit).")
    while True:
        try:
            raw = input_func("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in QUIT_COMMANDS:
            break
        print(service.run_query(text))
        print()


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Entry point of the CLI. Returns a process exit code."""
    args = parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
        network = SubwayNetwork.from_data(data)
    except (OSError, ValueError, KeyError, TypeError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        issues = network.check_data_integrity()
        print(f"Stations: {network.station_count()}, connections: {network.graph.num_edges()}")
        if not issues:
            print("Data integrity check passed.")
            return 0
        print(f"Data integrity check found {len(issues)} issue(s):")
        for issue in issues:
            print(f"  - {issue}")
        return 1

    preferred = parse_line_preference(args.line_preference)
    known_lines = network.line_names()
    for line in preferred:
        if line not in known_lines:
            print(f"Warning: line preference '{line}' is not a known line", file=sys.stderr)

    service = QueryService(
        network,
        mode=args.mode,
        max_transfers=args.max_transfers,
        preferred_lines=preferred,
        output_format=args.output_format,
        output_file=args.output_file,
    )
    run_interactive(service)
    return 0

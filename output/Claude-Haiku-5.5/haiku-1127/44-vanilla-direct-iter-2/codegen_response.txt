"""Fare and travel-time calculations for the Beijing Subway.

Fare structure (distance in meters, intervals are left-closed, right-open):
    0      <= d < 6000        : 3 yuan
    6000   <= d < 12000       : 4 yuan
    12000  <= d < 22000       : 5 yuan
    22000  <= d < 32000       : 6 yuan
    d      >= 32000           : 7 yuan, plus 1 yuan for every further 20 km
                                (32-52 km: 7 yuan, 52-72 km: 8 yuan, ...)

Fractional distances are rounded up (ceiling) to whole meters before the
tier lookup, so a boundary value always enters the next higher tier.
"""

import math

# (lower bound in meters, fare in yuan) for the basic and progressive tiers.
FARE_TIERS = (
    (0, 3),
    (6000, 4),
    (12000, 5),
    (22000, 6),
)

# Extended tier: starts at 32 km with 7 yuan, +1 yuan per 20 km.
EXTENDED_TIER_START_M = 32000
EXTENDED_TIER_BASE_FARE = 7
EXTENDED_TIER_STEP_M = 20000

# Estimated travel time parameters (used for summaries and comprehensive routing).
MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def _validate_distance(distance):
    """Ensure a distance value is a finite, non-negative real number."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError(f"Distance must be a number of meters, got {type(distance).__name__}")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance):
    """Return the subway fare (integer yuan) for a travelled distance in meters.

    Fractional meters are rounded up, and boundary values are charged at the
    next higher tier (left-closed, right-open rule).
    """
    _validate_distance(distance)
    meters = math.ceil(distance)

    if meters >= EXTENDED_TIER_START_M:
        extra_steps = (meters - EXTENDED_TIER_START_M) // EXTENDED_TIER_STEP_M
        return EXTENDED_TIER_BASE_FARE + extra_steps

    fare = FARE_TIERS[0][1]
    for lower_bound, tier_fare in FARE_TIERS:
        if meters >= lower_bound:
            fare = tier_fare
    return fare


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes.

    Uses about 1.5 minutes per kilometre plus about 5 minutes per transfer.
    ``distance`` is in meters.
    """
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int):
        raise TypeError("transfer_count must be an integer")
    if transfer_count < 0:
        raise ValueError("transfer_count cannot be negative")
    riding_minutes = (distance / 1000.0) * MINUTES_PER_KM
    transfer_minutes = transfer_count * MINUTES_PER_TRANSFER
    return float(riding_minutes + transfer_minutes)

"""Core data models: weighted undirected graph and the subway network wrapper."""

import re


def natural_sort_key(text):
    """Sort key that orders embedded numbers numerically (e.g. 'Line 2' < 'Line 10')."""
    parts = re.split(r"(\d+)", text)
    return [int(part) if part.isdigit() else part.lower() for part in parts]


class Edge:
    """An undirected, weighted connection between two stations.

    ``source`` and ``dest`` are station numbers, ``distance`` is in meters,
    and ``line`` is the name of the subway line the connection belongs to.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by path algorithms (equal to the distance in meters)."""
        return self.distance

    def other(self, vertex):
        """Return the endpoint of this edge that is not ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex!r} is not an endpoint of this edge")

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"


class Graph:
    """Weighted undirected graph using adjacency lists.

    Vertices are station numbers assigned by the caller. The ``station_lines``
    dictionary maps station names to the list of line names serving them.
    """

    def __init__(self):
        self._adjacency = {}
        self._edges = []
        self.station_lines = {}

    def add_vertex(self, vertex):
        """Add a vertex if it is not already present."""
        self._adjacency.setdefault(vertex, [])

    def has_vertex(self, vertex):
        """Return True if the vertex exists in the graph."""
        return vertex in self._adjacency

    def set_station_lines(self, station_name, lines):
        """Set the line list of a station (duplicates removed, order preserved)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines))

    def add_edge(self, edge):
        """Add an undirected edge, registering both endpoints as vertices."""
        if edge.source == edge.dest:
            raise ValueError("Self-loop edges are not supported")
        if edge.distance < 0:
            raise ValueError("Edge distance cannot be negative")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adjacency[edge.source].append(edge)
        self._adjacency[edge.dest].append(edge)
        self._edges.append(edge)

    def remove_edge(self, edge):
        """Remove an edge from the graph."""
        self._adjacency[edge.source].remove(edge)
        self._adjacency[edge.dest].remove(edge)
        self._edges.remove(edge)

    def adjacent_edges(self, vertex):
        """Return the list of edges incident to ``vertex``."""
        return self._adjacency.get(vertex, [])

    def neighbors(self, vertex):
        """Return the neighbouring vertices of ``vertex``."""
        return [edge.other(vertex) for edge in self.adjacent_edges(vertex)]

    def find_edges(self, u, v):
        """Return all edges directly connecting vertices ``u`` and ``v``."""
        return [edge for edge in self.adjacent_edges(u) if edge.other(u) == v]

    def vertices(self):
        """Return a list of all vertices."""
        return list(self._adjacency)

    def edges(self):
        """Return a list of all edges."""
        return list(self._edges)

    @property
    def num_vertices(self):
        return len(self._adjacency)

    @property
    def num_edges(self):
        return len(self._edges)

    def __len__(self):
        return self.num_vertices


class Network:
    """Subway network: a Graph plus station name <-> number mappings."""

    def __init__(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_name = {}
        self.load_warnings = []
        self._next_number = 0

    def add_station(self, name, lines=()):
        """Add a station (or merge lines into an existing one) and return its number."""
        if name in self.station_to_number:
            number = self.station_to_number[name]
            merged = list(self.graph.station_lines.get(name, []))
            for line in lines:
                if line not in merged:
                    merged.append(line)
            self.graph.set_station_lines(name, merged)
            return number

        number = self._next_number
        self._next_number += 1
        self.station_to_number[name] = number
        self.number_to_name[number] = name
        self.graph.add_vertex(number)
        self.graph.set_station_lines(name, lines)
        return number

    def has_station(self, name):
        return name in self.station_to_number

    def number_of(self, name):
        """Return the station number for a station name."""
        if name not in self.station_to_number:
            raise ValueError(f"Unknown station: {name}")
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the station name for a station number."""
        return self.number_to_name[number]

    def lines_of(self, name):
        """Return the line names serving a station."""
        return list(self.graph.station_lines.get(name, []))

    def station_names(self):
        """Return all station names sorted alphabetically."""
        return sorted(self.station_to_number, key=natural_sort_key)

    def all_line_names(self):
        """Return all known line names in natural order."""
        lines = set()
        for station_lines in self.graph.station_lines.values():
            lines.update(station_lines)
        for edge in self.graph.edges():
            lines.add(edge.line)
        return sorted(lines, key=natural_sort_key)

    def add_connection(self, source_name, dest_name, distance, line):
        """Create an edge between two existing stations and return it."""
        u = self.number_of(source_name)
        v = self.number_of(dest_name)
        edge = Edge(u, v, distance, line)
        self.graph.add_edge(edge)
        return edge

    def connections_between(self, source_name, dest_name):
        """Return all edges directly connecting two stations."""
        if not (self.has_station(source_name) and self.has_station(dest_name)):
            return []
        return self.graph.find_edges(self.number_of(source_name), self.number_of(dest_name))

    def rename_station(self, old_name, new_name):
        """Rename a station; edges keep referencing the same station number."""
        if old_name not in self.station_to_number:
            raise ValueError(f"Unknown station: {old_name}")
        if new_name != old_name and new_name in self.station_to_number:
            raise ValueError(f"Station already exists: {new_name}")
        number = self.station_to_number.pop(old_name)
        self.station_to_number[new_name] = number
        self.number_to_name[number] = new_name
        lines = self.graph.station_lines.pop(old_name, [])
        self.graph.station_lines[new_name] = lines

"""Loading of the Beijing Subway dataset and construction of the network model."""

import json
import os

from models import Network

_THIS_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_FILE_PATH = os.path.join(_THIS_DIR, "subway_data.json")


class DataError(Exception):
    """Raised when the subway data file cannot be loaded or has an invalid structure."""


def load_subway_data(path):
    """Read the subway JSON file and return its content as a dictionary."""
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataError(f"Subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataError(f"Subway data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataError(f"Cannot read subway data file: {exc}") from exc

    if not isinstance(data, dict):
        raise DataError("Subway data must be a JSON object")
    for key in ("stations", "connections"):
        if key not in data or not isinstance(data[key], list):
            raise DataError(f"Subway data is missing the '{key}' list")
    return data


def build_network(data):
    """Build a Network (graph plus name/number mappings) from loaded data.

    Problems in individual records (unknown stations, invalid distances) are
    recorded in ``network.load_warnings`` instead of aborting the load, and
    bad connections are skipped.
    """
    network = Network()

    for index, station in enumerate(data["stations"]):
        if not isinstance(station, dict) or not station.get("name"):
            raise DataError(f"Station entry #{index} has no name")
        lines = station.get("lines", [])
        if not isinstance(lines, list):
            raise DataError(f"Station '{station['name']}' has an invalid 'lines' field")
        network.add_station(station["name"], lines)

    seen = set()
    for index, conn in enumerate(data["connections"]):
        if not isinstance(conn, dict):
            network.load_warnings.append(f"Connection entry #{index} is malformed; skipped")
            continue
        source = conn.get("source")
        dest = conn.get("dest")
        distance = conn.get("distance")
        line = conn.get("line", "")

        if not source or not dest or source == dest:
            network.load_warnings.append(f"Connection entry #{index} has invalid endpoints; skipped")
            continue
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            network.load_warnings.append(
                f"Connection {source} - {dest} ({line}) has an invalid distance; skipped")
            continue

        for name in (source, dest):
            if not network.has_station(name):
                network.add_station(name, [line] if line else [])
                network.load_warnings.append(
                    f"Station '{name}' used by a connection is missing from the station list")

        # Connections are bidirectional: keep one edge per (station pair, line).
        key = (frozenset((source, dest)), line)
        if key in seen:
            continue
        seen.add(key)
        network.add_connection(source, dest, int(distance), line)

    return network

<file path="src/pathfinding.py">
"""Path calculation: Dijkstra's algorithm with a binary-heap priority queue.

The search runs over (station, arriving line) states so that line changes
(transfers) can be counted and constrained. Three query modes are supported:

* shortest_distance - minimise total travelled distance (meters)
* least_transfer    - minimise the number of transfers, then the distance
* comprehensive     - minimise an estimated travel time that combines distance
                      (about 1.5 min/km) and transfers (about 5 min each)

Line preference: edges on lines not listed in ``preferred_lines`` incur a
penalty of PREFERENCE_PENALTY_METERS during the search. The reported distance
is always the real travelled distance.
"""

import heapq
import itertools

from fare_calculator import MINUTES_PER_KM, MINUTES_PER_TRANSFER

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

PREFERENCE_PENALTY_METERS = 1000.0


class NoPathError(Exception):
    """Raised when no path exists between two stations under the given constraints."""


class PathfindingConfig:
    """Configuration for path queries.

    Attributes:
        mode: one of VALID_MODES.
        max_transfers: maximum allowed number of transfers, or None for no limit.
        preferred_lines: list of line names that are preferred.
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []
        self.validate()

    def validate(self):
        """Raise ValueError if the configuration is invalid."""
        if self.mode not in VALID_MODES:
            raise ValueError(f"Unknown path mode '{self.mode}'. Valid modes: {', '.join(VALID_MODES)}")
        if self.max_transfers is not None:
            if isinstance(self.max_transfers, bool) or not isinstance(self.max_transfers, int):
                raise ValueError("max_transfers must be an integer or None")
            if self.max_transfers < 0:
                raise ValueError("max_transfers cannot be negative")


class MinPQ:
    """Min priority queue implemented on top of a binary heap (heapq)."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        """Insert an item with the given priority key."""
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self):
        """Remove and return (key, item) with the smallest key."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._heap[0][0]

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


class _Label:
    """Best known information for a search state."""

    __slots__ = ("cost", "distance", "transfers", "penalty", "prev_state", "prev_edge")

    def __init__(self, cost, distance, transfers, penalty, prev_state, prev_edge):
        self.cost = cost
        self.distance = distance
        self.transfers = transfers
        self.penalty = penalty
        self.prev_state = prev_state
        self.prev_edge = prev_edge


class ShortestPath:
    """Single-source shortest path computation.

    Attributes:
        dist_to: dict mapping reachable station numbers to the real travelled
                 distance in meters from the source.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"Source vertex {source!r} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()

        self.dist_to = {}
        self._labels = {}          # (station, arriving line) -> _Label
        self._node_state = {}      # station -> finalised best state
        self._run()

    def _cost(self, distance, transfers, penalty):
        """Return a comparable cost tuple for the configured mode."""
        total = distance + penalty
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, total)
        if mode == MODE_COMPREHENSIVE:
            score = (total / 1000.0) * MINUTES_PER_KM + transfers * MINUTES_PER_TRANSFER
            return (score, total)
        return (total, transfers)

    def _run(self):
        """Run Dijkstra's algorithm over (station, line) states."""
        start_state = (self.source, None)
        self._labels[start_state] = _Label(
            cost=self._cost(0, 0, 0.0),
            distance=0.0,
            transfers=0,
            penalty=0.0,
            prev_state=None,
            prev_edge=None,
        )

        queue = MinPQ()
        queue.insert(self._labels[start_state].cost, start_state)
        finalised = set()
        max_transfers = self.config.max_transfers
        preferred = set(self.config.preferred_lines)

        while not queue.is_empty():
            _, state = queue.del_min()
            if state in finalised:
                continue
            finalised.add(state)

            node, arriving_line = state
            label = self._labels[state]
            if node not in self._node_state:
                self._node_state[node] = state
                self.dist_to[node] = label.distance

            for edge in self.graph.adjacent_edges(node):
                next_node = edge.other(node)
                next_state = (next_node, edge.line)
                if next_state in finalised:
                    continue

                transfer = 1 if (arriving_line is not None and edge.line != arriving_line) else 0
                new_transfers = label.transfers + transfer
                if max_transfers is not None and new_transfers > max_transfers:
                    continue

                new_distance = label.distance + edge.distance
                new_penalty = label.penalty
                if preferred and edge.line not in preferred:
                    new_penalty += PREFERENCE_PENALTY_METERS

                new_cost = self._cost(new_distance, new_transfers, new_penalty)
                existing = self._labels.get(next_state)
                if existing is None or new_cost < existing.cost:
                    self._labels[next_state] = _Label(
                        cost=new_cost,
                        distance=new_distance,
                        transfers=new_transfers,
                        penalty=new_penalty,
                        prev_state=state,
                        prev_edge=edge,
                    )
                    queue.insert(new_cost, next_state)

    def has_path_to(self, target):
        """Return True if ``target`` is reachable under the current configuration."""
        return target in self._node_state

    def path_with_edges(self, target):
        """Return (station numbers, edges) from the source to ``target``.

        Raises NoPathError if the target is unreachable.
        """
        if target not in self._node_state:
            raise NoPathError(f"No path found to station number {target}")
        state = self._node_state[target]
        nodes = []
        edges = []
        while state is not None:
            label = self._labels[state]
            nodes.append(state[0])
            if label.prev_edge is not None:
                edges.append(label.prev_edge)
            state = label.prev_state
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        """Return the number of line changes on the path to ``target``."""
        if target not in self._node_state:
            raise NoPathError(f"No path found to station number {target}")
        return self._labels[self._node_state[target]].transfers


class Route:
    """Result of a path query with station names and edges."""

    def __init__(self, stations, edges, distance, transfer_count, config):
        self.stations = stations
        self.edges = edges
        self.distance = distance
        self.transfer_count = transfer_count
        self.config = config


def find_route(network, source_name, target_name, config=None):
    """Compute a route between two stations given by name."""
    config = config if config is not None else PathfindingConfig()
    source = network.number_of(source_name)
    target = network.number_of(target_name)
    sp = ShortestPath(network.graph, source, config)
    numbers, edges = sp.path_with_edges(target)
    stations = [network.name_of(number) for number in numbers]
    return Route(
        stations=stations,
        edges=edges,
        distance=sp.dist_to[target],
        transfer_count=sp.get_transfer_count(target),
        config=config,
    )

"""Path visualisation, summary generation and report export (text and JSON)."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    """Format a distance: meters for short distances, kilometres for long ones."""
    if meters < 1000:
        return f"{meters:.0f} m"
    return f"{meters / 1000:.2f} km"


def format_distance_both(meters):
    """Format a distance showing both meters and kilometres."""
    if meters < 1000:
        return f"{meters:.0f} m ({meters / 1000:.3f} km)"
    return f"{meters / 1000:.2f} km ({meters:.0f} m)"


def _build_segments(route):
    """Group consecutive edges served by the same line into segments."""
    segments = []
    current = None
    stations = route.stations
    for index, edge in enumerate(route.edges):
        start_name = stations[index]
        end_name = stations[index + 1]
        if current is None or current["line"] != edge.line:
            current = {
                "line": edge.line,
                "from": start_name,
                "to": end_name,
                "stops": 0,
                "distance_m": 0,
            }
            segments.append(current)
        current["to"] = end_name
        current["stops"] += 1
        current["distance_m"] += edge.distance
    return segments


def _build_transfers(route, network):
    """Describe every station where the traveller changes line."""
    transfers = []
    stations = route.stations
    for index in range(1, len(route.edges)):
        previous_edge = route.edges[index - 1]
        next_edge = route.edges[index]
        if previous_edge.line != next_edge.line:
            station = stations[index]
            transfers.append({
                "station": station,
                "from_line": previous_edge.line,
                "to_line": next_edge.line,
                "station_lines": network.lines_of(station),
            })
    return transfers


def build_report(route, network, query_info):
    """Build a structured summary dictionary for a computed route.

    ``query_info`` is a dict describing the query (start, end, mode, ...).
    """
    distance = route.distance
    fare = Sum_money(distance)
    estimated_minutes = calculate_estimated_time(distance, route.transfer_count)

    lines_involved = []
    for edge in route.edges:
        if edge.line not in lines_involved:
            lines_involved.append(edge.line)

    return {
        "query": dict(query_info),
        "path": list(route.stations),
        "path_length": len(route.stations),
        "distance_m": distance,
        "distance_km": round(distance / 1000.0, 3),
        "fare": fare,
        "estimated_time_min": round(estimated_minutes, 1),
        "transfer_count": route.transfer_count,
        "lines": lines_involved,
        "transfers": _build_transfers(route, network),
        "segments": _build_segments(route),
    }


def render_text(summary):
    """Render a summary as console-friendly text."""
    query = summary["query"]
    out = []
    out.append("=" * 60)
    out.append("Beijing Subway Fare Query")
    out.append("=" * 60)
    out.append(f"From: {query['start']}    To: {query['end']}")
    max_transfers = query.get("max_transfers")
    preference = query.get("line_preference") or []
    out.append(f"Mode: {query['mode']}    Max transfers: "
               f"{'unlimited' if max_transfers is None else max_transfers}")
    out.append(f"Line preference: {', '.join(preference) if preference else 'none'}")
    out.append("-" * 60)

    out.append(f"Route ({summary['path_length']} stations):")
    for index, station in enumerate(summary["path"], start=1):
        out.append(f"  {index:>3}. {station}")

    out.append("")
    out.append("Line segments:")
    for segment in summary["segments"]:
        out.append(f"  [{segment['line']}] {segment['from']} -> {segment['to']} "
                   f"({segment['stops']} stops, {format_distance(segment['distance_m'])})")

    out.append("")
    out.append("Transfers:")
    if not summary["transfers"]:
        out.append("  None (no transfers)")
    for transfer in summary["transfers"]:
        out.append(f"  {transfer['station']}: {transfer['from_line']} -> {transfer['to_line']}"
                   f"  (lines at station: {', '.join(transfer['station_lines'])})")

    out.append("")
    out.append("-" * 60)
    out.append(f"Total distance: {format_distance_both(summary['distance_m'])}")

from collections import deque

from models import natural_sort_key  # noqa: F401  (re-exported for convenience)

# Representative stations of main lines used for coverage checks.

class IntegrityReport:
    """Collects errors, warnings and informational messages from a data check."""

    def __init__(self):
        self.errors = []
        self.warnings = []
        self.info = []

    def add_error(self, message):
        self.errors.append(message)

    def add_warning(self, message):
        self.warnings.append(message)

    def add_info(self, message):
        self.info.append(message)

    def has_errors(self):
        return bool(self.errors)

    def format(self):
        """Return a human-readable text report."""
        out = ["Data integrity check report", "-" * 40]
        out.extend(f"[INFO]    {message}" for message in self.info)
        out.extend(f"[WARNING] {message}" for message in self.warnings)
        out.extend(f"[ERROR]   {message}" for message in self.errors)
        out.append("-" * 40)
        out.append(f"Errors: {len(self.errors)}, warnings: {len(self.warnings)}")
        out.append("Result: " + ("FAILED" if self.has_errors() else "PASSED"))
        return "\n".join(out)


class DataManager:
    """Integrity checks and update operations on a loaded Network."""

    def __init__(self, network):
        self.network = network

    # ------------------------------------------------------------------
    # Integrity checks
    # ------------------------------------------------------------------
    def check_integrity(self, representative_stations=None):
        """Run all integrity checks and return an IntegrityReport."""
        report = IntegrityReport()
        reps = REPRESENTATIVE_STATIONS if representative_stations is None else representative_stations
        report.add_info(
            f"Stations: {len(self.network.station_to_number)}, "
            f"connections: {self.network.graph.num_edges}, "
            f"lines: {len(self.network.all_line_names())}")
        self._check_station_coverage(report, reps)
        self._check_connections(report)
        self._check_distances(report)
        for message in self.network.load_warnings:
            report.add_warning(message)
        return report

    def _check_station_coverage(self, report, representative_stations):
        if not self.network.station_to_number:
            report.add_error("No stations are loaded")
            return
        available_lines = set(self.network.all_line_names())
        for line, stations in representative_stations.items():
            if line not in available_lines:
                report.add_warning(f"Main line '{line}' has no stations in the data")
                continue
            for station in stations:
                if not self.network.has_station(station):
                    report.add_warning(f"Representative station '{station}' of {line} is missing")
                elif line not in self.network.lines_of(station):
                    report.add_warning(f"Station '{station}' is not registered on {line}")

    def _check_connections(self, report):
        graph = self.network.graph
        for name in self.network.station_names():
            number = self.network.number_of(name)
            incident = graph.adjacent_edges(number)
            if not incident:
                report.add_warning(f"Station '{name}' has no connections")
                continue
            station_lines = set(self.network.lines_of(name))
            edge_lines = {edge.line for edge in incident}
            for line in sorted(station_lines - edge_lines):
                report.add_warning(f"Station '{name}' is listed on {line} but has no connection on it")
            for line in sorted(edge_lines - station_lines):
                report.add_warning(f"Station '{name}' has a connection on {line} but is not listed on it")

        # Overall connectivity.
        visited = set()
        component_sizes = []
        for vertex in graph.vertices():
            if vertex in visited:
                continue
            size = 0
            queue = deque([vertex])
            visited.add(vertex)
            while queue:
                current = queue.popleft()
                size += 1
                for neighbour in graph.neighbors(current):
                    if neighbour not in visited:
                        visited.add(neighbour)
                        queue.append(neighbour)
            component_sizes.append(size)
        if len(component_sizes) > 1:
            report.add_warning(f"Network is split into {len(component_sizes)} disconnected parts "
                               f"(sizes: {sorted(component_sizes, reverse=True)})")

        # Per-line continuity.
        for line in self.network.all_line_names():
            sections = self._count_line_sections(line)
            if sections > 1:
                report.add_warning(f"Line '{line}' is split into {sections} disconnected sections")

    def _count_line_sections(self, line):
        """Count connected components formed by the edges of one line."""
        graph = self.network.graph
        line_vertices = set()
        adjacency = {}
        for edge in graph.edges():
            if edge.line != line:
                continue
            line_vertices.add(edge.source)
            line_vertices.add(edge.dest)
            adjacency.setdefault(edge.source, []).append(edge.dest)
            adjacency.setdefault(edge.dest, []).append(edge.source)
        visited = set()
        sections = 0
        for vertex in line_vertices:
            if vertex in visited:
                continue
            sections += 1
            queue = deque([vertex])
            visited.add(vertex)
            while queue:
                current = queue.popleft()
                for neighbour in adjacency.get(current, []):
                    if neighbour not in visited:
                        visited.add(neighbour)
                        queue.append(neighbour)
        return sections

    def _check_distances(self, report):
        graph = self.network.graph
        for edge in graph.edges():
            source_name = self.network.name_of(edge.source)
            dest_name = self.network.name_of(edge.dest)
            label = f"{source_name} - {dest_name} ({edge.line})"
            distance = edge.distance
            if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
                report.add_error(f"Invalid distance {distance!r} on {label}")
            elif distance > MAX_NORMAL_SEGMENT_DISTANCE_M:
                report.add_warning(f"Unusually long distance {distance} m on {label}")

    # ------------------------------------------------------------------
    # Update operations
    # ------------------------------------------------------------------
    @staticmethod
    def _validate_distance(distance):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise ValueError("Distance must be a number of meters")
        if distance <= 0:
            raise ValueError("Distance must be positive")

    def add_line(self, line_name, station_names, distances):
        """Add a new line with an ordered station sequence.

        ``distances`` holds the distance in meters between consecutive stations
        and must have length ``len(station_names) - 1``.
        """
        name = (line_name or "").strip()
        if not name:
            raise ValueError("Line name cannot be empty")
        if name in self.network.all_line_names():
            raise ValueError(f"Line '{name}' already exists")
        stations = [station.strip() for station in station_names]
        if len(stations) < 2:
            raise ValueError("A line needs at least two stations")
        if len(set(stations)) != len(stations):
            raise ValueError("A line cannot contain the same station twice")
        if len(distances) != len(stations) - 1:
            raise ValueError("Number of distances must be one less than the number of stations")
        for distance in distances:
            self._validate_distance(distance)

        for station in stations:
            self.network.add_station(station, [name])
        for source, dest, distance in zip(stations, stations[1:], distances):
            self.network.add_connection(source, dest, int(distance), name)

    def modify_station(self, name, new_name=None, add_lines=(), remove_lines=()):
        """Rename a station and/or change its line membership. Returns the final name."""
        if not self.network.has_station(name):
            raise ValueError(f"Unknown station: {name}")
        target = name
        if new_name and new_name.strip() and new_name.strip() != name:
            target = new_name.strip()
            self.network.rename_station(name, target)

        lines = list(self.network.graph.station_lines.get(target, []))
        for line in add_lines:
            if line not in lines:
                lines.append(line)
        for line in remove_lines:
            if line in lines:
                lines.remove(line)
        self.network.graph.set_station_lines(target, lines)
        return target

    def update_distance(self, source_name, dest_name, distance, line=None):
        """Update the distance between two adjacent stations. Returns the number of edges changed."""
        self._validate_distance(distance)
        edges = self.network.connections_between(source_name, dest_name)
        if line is not None:
            edges = [edge for edge in edges if edge.line == line]
        if not edges:
            raise ValueError(f"No connection between '{source_name}' and '{dest_name}'"
                             + (f" on {line}" if line else ""))
        for edge in edges:
            edge.distance = int(distance)
        return len(edges)

"""Command-line interface for the Beijing Subway fare calculation system."""

PROMPT = "Enter query ('Starting Station; Ending Station', or 'quit' to exit): "
FORMAT_ERROR_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"
EXIT_WORDS = ("quit", "exit", "q")


class QueryError(Exception):
    """Base class for errors that are reported to the user without stopping the program."""


class InputFormatError(QueryError):
    """The query text does not match 'Starting Station; Ending Station'."""


class StationNotFoundError(QueryError):
    """A station name does not match any official station."""


class SameStationError(QueryError):
    """Starting and ending stations are identical."""


class PathCalculationError(QueryError):
    """No path could be computed."""


class FareCalculationError(QueryError):
    """The fare could not be calculated for the route."""


def _normalize(text):
    """Normalise a name for lenient comparison (case, spaces, hyphens, underscores)."""
    return "".join(ch for ch in text.lower() if ch not in " -_'")


def resolve_station(network, raw_name, role):
    """Map user input to an official station name, or raise StationNotFoundError."""
    name = raw_name.strip()
    if name in network.station_to_number:
        return name

    normalized = _normalize(name)
    matches = [station for station in network.station_names() if _normalize(station) == normalized]
    if len(matches) == 1:
        return matches[0]

    suggestions = difflib.get_close_matches(name, network.station_names(), n=3, cutoff=0.6)
    message = f"{role} station '{name}' does not exist, please check the station name"
    if suggestions:
        message += f". Did you mean: {', '.join(suggestions)}?"
    raise StationNotFoundError(message)


def parse_query(text):
    """Split 'Starting Station; Ending Station' into two raw names."""
    parts = text.split(";")
    if len(parts) != 2:
        raise InputFormatError(FORMAT_ERROR_MESSAGE)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise InputFormatError(FORMAT_ERROR_MESSAGE)
    return start, end


def run_query(network, text, config):
    """Execute one query and return the structured summary dictionary."""
    start_raw, end_raw = parse_query(text)
    start = resolve_station(network, start_raw, "Starting")
    end = resolve_station(network, end_raw, "Ending")
    if start == end:
        raise SameStationError("Starting station cannot be the same as the ending station")

    try:
        route = find_route(network, start, end, config)
    except NoPathError as exc:
        raise PathCalculationError(f"No route found from '{start}' to '{end}': {exc}") from exc
    except ValueError as exc:
        raise PathCalculationError(f"Path calculation failed: {exc}") from exc

    query_info = {
        "start": start,
        "end": end,
        "mode": config.mode,
        "max_transfers": config.max_transfers,
        "line_preference": list(config.preferred_lines),
    }
    try:
        return build_report(route, network, query_info)
    except (TypeError, ValueError) as exc:
        raise FareCalculationError(f"Fare calculation failed: {exc}") from exc


def _non_negative_int(value):
    try:
        number = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError("must be an integer") from exc
    if number < 0:
        raise argparse.ArgumentTypeError("must not be negative")
    return number


def build_parser():
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (CLI).")
    parser.add_argument("--output_file", default=None,
                        help="export query results to this file (overwritten with the session history)")
    parser.add_argument("--format", dest="output_format", choices=("text", "json"), default="text",
                        help="output format (default: text)")
    parser.add_argument("--mode", choices=VALID_MODES, default=MODE_SHORTEST_DISTANCE,

    return parser


def _resolve_line_preferences(raw, network):
    """Validate and canonicalise the --line_preference argument."""
    if not raw:
        return []
    known = network.all_line_names()
    lookup = {_normalize(line): line for line in known}
    result = []
    for part in raw.split(","):
        text = part.strip()
        if not text:
            continue
        key = _normalize(text)
        if key not in lookup:
            raise ValueError(f"Unknown line '{text}'. Known lines: {', '.join(known)}")
        if lookup[key] not in result:
            result.append(lookup[key])
    return result


def _write_output_file(path, summaries, output_format):
    """Write the session history to the output file."""
    if output_format == "json":
        content = json.dumps(summaries, ensure_ascii=False, indent=2)
    else:
        content = "\n\n".join(render_text(summary) for summary in summaries)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(content + "\n")


def interactive_loop(network, config, output_format, output_file=None,
                     input_func=input, output_func=print):
    """Run the continuous query loop until the user quits or input ends."""
    history = []
    output_func(f"Beijing Subway fare calculator - {len(network.station_to_number)} stations loaded.")
    output_func("Type 'quit' to exit.")

    while True:
        try:
            line = input_func(PROMPT)
        except (EOFError, KeyboardInterrupt):
            output_func("")
            break

        text = line.strip()
        if not text:
            continue
        if text.lower() in EXIT_WORDS:
            break

        try:
            summary = run_query(network, text, config)
        except QueryError as exc:
            output_func(f"Error: {exc}")
            continue
        except Exception as exc:  # defensive: never crash the interactive session
            output_func(f"Unexpected error while processing the query: {exc}")
            continue

        if output_format == "json":
            output_func(render_json(summary))
        else:
            output_func(render_text(summary))

        if output_file:
            history.append(summary)
            try:
                _write_output_file(output_file, history, output_format)
            except OSError as exc:
                output_func(f"Error: cannot write output file '{output_file}': {exc}")
    return 0


def main(argv=None):
    """Program entry point. Returns the process exit code."""
    parser = build_parser()
    args = parser.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
        network = build_network(data)
    except DataError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        report = DataManager(network).check_integrity()
        print(report.format())
        return 1 if report.has_errors() else 0

    try:
        preferences = _resolve_line_preferences(args.line_preference, network)
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=preferences,
        )
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    return interactive_loop(
        network,
        config,
        output_format=args.output_format,
        output_file=args.output_file,
    )

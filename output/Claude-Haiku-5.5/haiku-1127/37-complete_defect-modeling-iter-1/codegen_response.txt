def normalize_text(text: str) -> str:
    """Normalize a name or group for case-insensitive, whitespace-tolerant comparison."""
    return " ".join(str(text).split()).casefold()

class ListManager:
    """Loads and validates participant lists and summarises department groups."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        return self._finalize(users, errors)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (each line: 'Name,Department Group')."""
        users, errors = self._load_from_path(path)
        return self._finalize(users, errors)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it. Returns parsed users and format/file errors."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to number of members."""
        counts = Counter(user.department for user in users)
        return dict(sorted(counts.items()))

    def _finalize(self, users: list[User], errors: list[str]) -> tuple[list[User], list[str]]:
        """Remove duplicate names and report them together with the format errors."""
        valid_users, duplicates = self.validate_users(users)
        all_errors = list(errors)
        for dup in duplicates:
            all_errors.append(
                f"Duplicate Name: '{dup.name}' ({dup.department}) appears more than once; "
                "the extra entry is ignored"
            )
        return valid_users, all_errors

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines are skipped."""
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_number}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

from lottery_system.models import Prize, normalize_text

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 8, [], "Li Bin"),
    ("Second Prize - Shopping Cards", 4, ["Product Department"], "Chen Qixian"),
    ("First Prize - 1 Day Annual Leave", 2, ["Technology Department"], "Ma Jiaqi"),

class PrizeManager:
    """Manages the ordered list of prizes for a lottery session."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Load the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(normalize_text(p.name) == normalize_text(clean_name) for p in self._prizes):
            raise ValueError(f"A prize named '{clean_name}' already exists.")

        cleaned_groups: list[str] = []
        for group in avoid_groups or []:
            group_name = str(group).strip()
            if group_name and normalize_text(group_name) not in {
                normalize_text(g) for g in cleaned_groups
            }:
                cleaned_groups.append(group_name)

        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=cleaned_groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        n = len(self._prizes)
        if not isinstance(order, list) or len(order) != n:
            return False
        if not all(isinstance(x, int) and not isinstance(x, bool) for x in order):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

import random

from lottery_system.constants import (
    FALLBACK_DRAWER,
    PROTECTED_GROUP,
    SPECIAL_DRAWER,
    WATCHED_DRAWERS,
)
from lottery_system.models import Prize, User, normalize_text


def is_protected_group(department: str) -> bool:
    """True if the department is the protected 'Cheng Ru group'."""
    return normalize_text(department) == normalize_text(PROTECTED_GROUP)


class LotteryManager:
    """Runs individual draws while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        self._drawn: list[User] = []
        # Protection results (True = contained a Cheng Ru group member) of the most
        # recent draws made by the watched lottery persons (Chen Qixian, Jin Duo).
        self.last_two_draw_results: list[bool] = []

    @property
    def drawn_users(self) -> list[User]:
        """All users drawn so far, in draw order."""
        return list(self._drawn)

    def is_previous_winner(self, user: User) -> bool:
        """True if the user has already won in this lottery."""
        return normalize_text(user.name) in self._winner_names

    def _is_special_drawer(self, prize: Prize) -> bool:
        return normalize_text(prize.associated_drawer or "") == normalize_text(SPECIAL_DRAWER)

    def is_fallback_active(self, prize: Prize) -> bool:
        """Fallback applies when the drawer is Ma Jiaqi, the last two watched draws
        contained no protected members, and protected members remain available."""
        if normalize_text(prize.associated_drawer or "") != normalize_text(FALLBACK_DRAWER):
            return False
        if len(self.last_two_draw_results) != 2 or any(self.last_two_draw_results):
            return False
        return any(
            is_protected_group(u.department) and not self.is_previous_winner(u)
            for u in self._users
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return users eligible to win the given prize right now."""
        available = [u for u in self._users if not self.is_previous_winner(u)]

        # Special rule: Li Bin's prizes only filter previous winners.
        if self._is_special_drawer(prize):
            return available

        # Fallback rule: force the pool to the protected group, overriding avoidance.
        if self.is_fallback_active(prize):
            return [u for u in available if is_protected_group(u.department)]

        avoided = {normalize_text(group) for group in prize.avoid_groups}
        return [u for u in available if normalize_text(u.department) not in avoided]

    def draw_one(self, prize: Prize):
        """Draw a single winner for the prize. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None

        winner = random.choice(pool)
        self._winner_names.add(normalize_text(winner.name))
        self._drawn.append(winner)

        drawer = normalize_text(prize.associated_drawer or "")
        watched = {normalize_text(name) for name in WATCHED_DRAWERS}
        if drawer in watched:
            self.last_two_draw_results = (
                self.last_two_draw_results + [is_protected_group(winner.department)]
            )[-2:]
        elif drawer == normalize_text(FALLBACK_DRAWER):
            # Protection window consumed by Ma Jiaqi's draw.
            self.last_two_draw_results = []
        return winner

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        drawer = (prize.associated_drawer or "").strip()
        parts = ["previous winners are excluded"]
        if self._is_special_drawer(prize):
            parts.append(
                f"lottery person {SPECIAL_DRAWER}: avoidance groups do not apply"
            )
        elif self.is_fallback_active(prize):
            parts.append(
                f"{PROTECTED_GROUP} protection active: only its members are eligible"
            )
        elif prize.avoid_groups:
            parts.append("avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("no group avoidance")
        if drawer:
            parts.append(f"lottery person: {drawer}")
        return "; ".join(parts)

"""Winner recording, grouped retrieval and CSV export."""

import csv
import os
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    draw_order: int


class ResultManager:
    """Keeps the complete winner list and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    @property
    def winners(self) -> list[Winner]:
        """All winners in the order they were drawn."""
        return list(self._winners)

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for the given prize."""
        winner = Winner(user=user, prize_name=prize.name, draw_order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name. Every given prize appears as a key."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to <cwd>/awards_result_YYYYMMDD.txt in CSV format."""
        if not self._winners:
            return False, "No winners to export."

        grouped = self.get_winners_by_prize(prizes)
        ordered = [winner for prize in prizes for winner in grouped.get(prize.name, [])]

        file_name = f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
        file_path = os.path.join(os.getcwd(), file_name)
        try:
            with open(file_path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for winner in ordered:
                    writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {file_path}"

class ExitRequested(Exception):
    """Raised when standard input is closed."""


def ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        raise ExitRequested()


def ask_yes(prompt: str) -> bool:
    return ask(prompt).lower() in ("y", "yes")


def ordinal(n: int) -> str:
    if 11 <= n % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryApp:
    """Drives the full interactive lottery workflow."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ---------- top level ----------

    def run(self) -> int:
        self._banner()
        try:
            if not self._load_participants():
                print("No participant list loaded. Exiting.")
                return 0
            self._configure_prizes()
            self._run_lottery()
        except ExitRequested:
            print("\nInput closed. Exiting.")
            return 0
        print("Thank you for using the lottery system.")
        return 0

    def _banner(self) -> None:
        print("=" * 60)
        print("  Company Team Building Event - Smart Lottery System")
        print("=" * 60)

    # ---------- 2.1 list management ----------

    def _load_participants(self) -> bool:
        while True:
            print("\n--- Participant List ---")
            print("1. Load built-in default team building list")
            print("2. Upload custom txt list (format: Name,Department Group per line)")
            print("0. Exit")
            choice = ask("Select an option: ")
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
                if self._accept_users(users, errors):
                    return True
            elif choice == "2":
                path = ask("Enter the path of the txt file: ")
                users, errors = self.list_manager.load_custom_list(path)
                if self._accept_users(users, errors):
                    return True
            elif choice == "0":
                return False
            else:
                print("Invalid option.")

    def _accept_users(self, users: list[User], errors: list[str]) -> bool:
        if errors:
            print(f"Validation found {len(errors)} problem(s):")
            for error in errors:
                print(f"  - {error}")
            print("Choose 'y' to remove the invalid data and continue with valid entries,")
            print("or 'n' to terminate this upload.")
            if not ask_yes("Remove invalid data and continue? [y/N]: "):
                print("Upload terminated.")
                return False
        if not users:
            print("No valid participants found.")
            return False
        self.users = users
        print(f"Loaded {len(users)} participant(s).")
        self._print_groups()
        return True

    def _print_groups(self) -> None:
        groups = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for group, count in groups.items():
            print(f"  {group}: {count} member(s)")

    # ---------- 2.2 prize configuration ----------

    def _configure_prizes(self) -> None:
        while True:
            print("\n--- Prize Configuration ---")
            print("1. Load preset prizes")
            print("2. Add custom prize")
            print("3. View prize list")
            print("4. Reorder prizes")
            print("5. View participant groups")
            print("0. Finish configuration and continue")
            choice = ask("Select an option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        print(f"  - {error}")
                else:
                    print("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._show_prizes()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                self._print_groups()
            elif choice == "0":
                if not self.prize_manager.get_prizes():
                    print("Add at least one prize before starting the lottery.")
                    continue
                return
            else:
                print("Invalid option.")

    def _parse_groups(self, text: str) -> list[str]:
        groups = self.list_manager.get_group_info(self.users)
        names = list(groups.keys())
        selected: list[str] = []
        tokens = [token.strip() for token in text.split(",") if token.strip()]
        for token in tokens:
            if token.isdigit() and 1 <= int(token) <= len(names):
                name = names[int(token) - 1]
            else:
                matches = [n for n in names if normalize_text(n) == normalize_text(token)]
                if not matches:
                    raise ValueError(f"Unknown group: '{token}'")
                name = matches[0]
            if name not in selected:
                selected.append(name)
        return selected

    def _add_custom_prize(self) -> None:
        name = ask("Prize name: ")
        if not name:
            print("Prize name cannot be empty.")
            return
        quantity_text = ask("Quantity: ")
        if not quantity_text.isdigit() or int(quantity_text) <= 0:
            print("Quantity must be a positive integer.")
            return

        groups = self.list_manager.get_group_info(self.users)
        print("Available department groups:")
        for index, (group, count) in enumerate(groups.items(), start=1):
            print(f"  {index}. {group} ({count})")
        groups_text = ask("Avoidance groups (numbers or names, comma-separated; blank for none): ")
        try:
            avoid_groups = self._parse_groups(groups_text)

            print(exc)
            return

        print("Known lottery persons: " + ", ".join(KNOWN_DRAWERS))
        drawer = ask("Associated lottery person (blank for none): ")
        try:
            prize = self.prize_manager.add_prize(
                name, int(quantity_text), avoid_groups, drawer or None

        except ValueError as exc:
            print(f"Error: {exc}")
            return
        print(f"Prize added: {prize.name} (quantity {prize.quantity})")

    def _describe_prize_line(self, index: int, prize: Prize) -> str:
        avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
        drawer = prize.associated_drawer or "None"
        return (
            f"{index}. {prize.name} | Quantity: {prize.quantity} | "
            f"Avoid: {avoid} | Lottery person: {drawer}"

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Prize sequence:")
        for index, prize in enumerate(prizes, start=1):
            print("  " + self._describe_prize_line(index, prize))

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes to reorder.")
            return
        self._show_prizes()
        text = ask("Enter the new order as serial numbers (e.g., 2 1 3): ")
        tokens = text.replace(",", " ").split()
        if not tokens or not all(token.isdigit() for token in tokens):
            print("Invalid input: please enter serial numbers only.")
            return
        order = [int(token) for token in tokens]
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
        else:
            print("Invalid order: every serial number must appear exactly once.")

    # ---------- 2.3 / 2.4 / 2.5 lottery process ----------

    def _run_lottery(self) -> None:
        prizes = self.prize_manager.get_prizes()
        print("\n=== Lottery Overview ===")
        for index, prize in enumerate(prizes, start=1):
            print(self._describe_prize_line(index, prize))

        if not ask_yes("Start the lottery now? [y/N]: "):
            print("Lottery not started.")
            return

        self.lottery = LotteryManager(self.users)
        print(
            "Controls: press Enter to draw the next winner | 'pause' to pause | "
            "'terminate' to end the current round | 'continue' to resume."
        )
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize)

        self._show_final_results(prizes)
        if ask_yes("Export the winner list to a CSV-formatted txt file? [y/N]: "):
            _, message = self.result_manager.export_to_csv(prizes)
            print(message)

    def _run_round(self, index: int, total: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {index}/{total}: {prize.name} | Quantity: {prize.quantity} ===")
        print(f"Rule: {self.lottery.describe_rule(prize)}")

        round_winners: list[User] = []
        for position in range(1, prize.quantity + 1):
            if self._wait_for_next_draw() == "terminate":
                print("Round terminated manually; remaining draws skipped.")
                break
            print(f"Currently drawing the {ordinal(position)} winner...")
            user = self.lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize; round ends early.")
                break
            self.result_manager.add_winner(user, prize)
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        print(f"--- Round {index} results (winning order) ---")
        if round_winners:
            for order, user in enumerate(round_winners, start=1):
                print(f"  {order}. {user.name} ({user.department})")
        else:
            print("  No winners in this round.")

    def _wait_for_next_draw(self) -> str:
        """Gate before each draw. Returns 'draw' or 'terminate'."""
        while True:
            command = ask("[Enter] draw next | pause | terminate > ").lower()
            if command in ("", "draw", "continue"):
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    sub_command = ask("(paused) > ").lower()
                    if sub_command == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if sub_command == "terminate":
                        return "terminate"
                    print("Type 'continue' to resume or 'terminate' to end this round.")
            print("Unknown command. Use Enter, 'pause', or 'terminate'.")

    def _show_final_results(self, prizes: list[Prize]) -> None:
        print("\n=== Final Winner List ===")
        grouped = self.result_manager.get_winners_by_prize(prizes)
        total = 0
        for prize in prizes:
            winners = grouped.get(prize.name, [])
            print(f"{prize.name} ({len(winners)}/{prize.quantity})")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                total += 1
                print(
                    f"  {winner.draw_order}. {winner.user.name} | "
                    f"{winner.user.department} | {winner.prize_name}"
                )
        print(f"Total winners: {total}")


def main(argv: list[str] | None = None) -> int:
    app = LotteryApp()
    try:
        return app.run()
    except KeyboardInterrupt:
        print("\nLottery interrupted.")

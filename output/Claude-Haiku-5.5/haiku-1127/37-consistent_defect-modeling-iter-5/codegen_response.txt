"""Background stdin reader so that commands can be detected during a draw."""
import queue
import sys
import threading
from typing import Optional


class InputReader:
    """Reads stdin lines in a daemon thread and hands them out via a queue.

    read_line() blocks until a line is available (used for menus).
    poll() waits up to a timeout and returns None if nothing was typed
    (used to detect 'pause', 'continue' and 'terminate' during a draw).
    Both raise EOFError when the input stream is closed.
    """

    def __init__(self, stream=None):
        self._stream = stream if stream is not None else sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._thread: Optional[threading.Thread] = None
        self._eof = False

    def start(self) -> None:
        if self._thread is None:
            self._thread = threading.Thread(target=self._pump, name="stdin-reader", daemon=True)
            self._thread.start()

    def _pump(self) -> None:
        while True:
            try:
                line = self._stream.readline()
            except (OSError, ValueError):
                line = ""
            if line == "":
                self._queue.put(None)
                return
            self._queue.put(line.rstrip("\r\n"))

    def read_line(self, prompt: str = "") -> str:
        if prompt:
            print(prompt, end="", flush=True)
        if self._eof:
            raise EOFError
        self.start()
        item = self._queue.get()
        if item is None:
            self._eof = True
            raise EOFError
        return item.strip()

    def poll(self, timeout: float) -> Optional[str]:
        if self._eof:
            raise EOFError
        self.start()
        try:
            item = self._queue.get(timeout=max(timeout, 0))
        except queue.Empty:
            return None
        if item is None:
            self._eof = True
            raise EOFError
        return item.strip().lower()

"""Participant list loading, validation and department group extraction."""
import re
from lottery_system.data import DEFAULT_PARTICIPANT_LINES
from lottery_system.models import User

_SEPARATOR = re.compile(r"[,;]")

LOAD_MODE_DEFAULT = "default"
LOAD_MODE_CUSTOM = "custom"


class ListManager:
    def __init__(self):
        self.load_mode = None
        self.source_file_path = None
        self.is_valid = False

    def load_default_list(self):
        """Load the built-in team building list. Returns (users, errors)."""
        users, errors = self._parse_lines(DEFAULT_PARTICIPANT_LINES)
        self._update_state(LOAD_MODE_DEFAULT, None, users, errors)
        return users, errors

    def load_custom_list(self, path):
        """Load a custom txt list from path. Returns (users, errors)."""
        users, errors = self._load_from_path(path)
        self._update_state(LOAD_MODE_CUSTOM, path, users, errors)
        return users, errors

    def _load_from_path(self, path):
        """Read and parse a txt file. Returns (users, errors)."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                content = handle.read()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}' ({exc})"]
        return self._parse_lines(content.splitlines())

    def _parse_lines(self, lines):
        users = []
        errors = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SEPARATOR.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department Group' but got '{line}'"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def _update_state(self, mode, path, users, errors):
        self.load_mode = mode
        self.source_file_path = path
        self.is_valid = not errors and bool(users)

    def validate_users(self, users):
        """Split users into (valid_users, duplicates).

        The first occurrence of a name is valid; later occurrences are duplicates.
        """
        seen = set()
        valid_users = []
        duplicates = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users):
        """Return a mapping of department group name to number of members."""
        counts = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management: preset prizes, custom prizes, avoidance groups and ordering."""
from lottery_system.data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []
        self._default_loaded = False

    def load_default_prizes(self):
        """Add the preset prize library. Returns a list of error messages."""
        if self._default_loaded:
            return ["Preset prizes have already been loaded"]
        for name, quantity, level in DEFAULT_PRIZES:
            self._prizes.append(Prize(name=name, quantity=quantity, level=level))
        self._default_loaded = True
        return []

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        """Add a custom prize at the end of the lottery sequence and return it."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer")
        if any(prize.name == name for prize in self._prizes):
            raise ValueError(f"A prize named '{name}' already exists")
        groups = []
        for group in avoid_groups or []:
            group = (group or "").strip()
            if group and group not in groups:
                groups.append(group)
        drawer = (associated_drawer or "").strip() or None
        prize = Prize(name=name, quantity=quantity, avoid_groups=groups,
                      associated_drawer=drawer, level=0)
        self._prizes.append(prize)
        return prize

    def get_prizes(self):
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def set_avoid_groups(self, index, groups):
        """Replace the avoidance groups of the prize at 0-based index."""
        prize = self._prizes[index]
        cleaned = []
        for group in groups or []:
            group = (group or "").strip()
            if group and group not in cleaned:
                cleaned.append(group)
        prize.avoid_groups = cleaned

    def set_associated_drawer(self, index, drawer):
        """Set the associated lottery person of the prize at 0-based index (None clears)."""
        prize = self._prizes[index]
        prize.associated_drawer = (drawer or "").strip() or None

    def reorder_prizes(self, order):
        """Reorder prizes. order holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if not isinstance(order, list) or len(order) != count:
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules, fallback and anti-duplication."""
import random
from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)


def _norm(value):
    return (value or "").strip().casefold()


def _is_drawer(prize, drawer_name):
    return prize.associated_drawer is not None and _norm(prize.associated_drawer) == _norm(drawer_name)


def _is_cheng_ru(user):
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


class LotteryManager:
    def __init__(self, users, rng=None):
        self._users = list(users)
        self._rng = rng if rng is not None else random.Random()
        self._winner_keys = set()
        self._winners = []
        # True means the draw by Chen Qixian / Jin Duo produced a Cheng Ru group member.
        self.last_two_draw_results = []

    @property
    def winners(self):
        return list(self._winners)

    def is_winner(self, user):
        return _norm(user.name) in self._winner_keys

    def remaining_count(self):
        return len(self._available_users())

    def _available_users(self):
        return [user for user in self._users if not self.is_winner(user)]

    def _cheng_ru_protection_active(self):
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = self._available_users()

        # Special rule: Li Bin only filters previous winners.
        if _is_drawer(prize, LI_BIN):
            return available

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group members when the
        # two previous Chen Qixian / Jin Duo draws both had no Cheng Ru group winner.
        if _is_drawer(prize, MA_JIAQI) and self._cheng_ru_protection_active():
            protected = [user for user in available if _is_cheng_ru(user)]
            if protected:
                return protected

        # Avoidance groups: their members stay eligible but are drawn after others.
        avoid = {_norm(group) for group in prize.avoid_groups if group and group.strip()}
        preferred = [user for user in available if _norm(user.department) not in avoid]
        if preferred:
            return preferred
        return [user for user in available if _norm(user.department) in avoid]

    def _record_draw(self, prize, user):
        if _is_drawer(prize, CHEN_QIXIAN) or _is_drawer(prize, JIN_DUO):
            self.last_two_draw_results.append(_is_cheng_ru(user))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def draw_one(self, prize: Prize):
        """Draw one winner for the prize. Returns the User, or None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winner_keys.add(_norm(user.name))
        self._winners.append(user)
        self._record_draw(prize, user)
        return user


def describe_rules(prize: Prize) -> str:
    """Human readable description of the rules applied to a prize round."""
    parts = []
    if prize.associated_drawer:
        parts.append(f"Lottery person: {prize.associated_drawer}")
    parts.append("previous winners are excluded")
    if _is_drawer(prize, LI_BIN):
        parts.append("special rule for Li Bin: avoidance groups are not applied")
    else:
        if prize.avoid_groups:
            parts.append(
                "members of avoidance groups (" + ", ".join(prize.avoid_groups)
                + ") are drawn after other members"
            )
        if _is_drawer(prize, MA_JIAQI):
            parts.append(
                "Cheng Ru group protection: if the last two draws by Chen Qixian and Jin Duo "
                "both had no Cheng Ru group member, the draw is limited to Cheng Ru group members"
            )
    return "; ".join(parts)

"""Winner records, grouped results and CSV export."""
import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize: Prize
    winning_order: int = 0
    winning_time: datetime = field(default_factory=datetime.now)


def _sorted_prizes(prizes):
    """Prizes ordered from highest level to lowest (stable for equal levels)."""
    return sorted(prizes, key=lambda prize: prize.level, reverse=True)


class ResultManager:
    def __init__(self):
        self._winners = []

    def add_winner(self, user, prize):
        winner = Winner(user=user, prize=prize,
                        winning_order=len(self._winners) + 1,
                        winning_time=datetime.now())
        self._winners.append(winner)
        return winner

    def get_winners(self):
        return list(self._winners)

    def get_winners_by_prize(self, prizes):
        """Return {prize name: [Winner, ...]} ordered by prize level (high to low),
        with winners within each prize in winning order."""
        result = {}
        for prize in _sorted_prizes(prizes):
            result[prize.name] = [w for w in self._winners if w.prize.name == prize.name]
        return result

    def export_to_csv(self, prizes):
        """Export the winner list to awards_result_YYYYMMDD.txt in the current directory."""
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in self.get_winners_by_prize(prizes).items():
                    for winner in winners:
                        writer.writerow([
                            winner.user.name,
                            winner.user.department,
                            prize_name,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def ordinal(number):
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"

            print("The list contains no participants; nothing was loaded.")
            return

            return
        drawer = self.reader.read_line("Associated lottery person (Enter to clear): ").strip()
        self.prize_manager.set_associated_drawer(index, drawer or None)
        print("Associated lottery person updated.")

    def _reorder_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured yet.")
            return
        count = len(prizes)
        text = self.reader.read_line(
            f"Enter the new order as serial numbers separated by spaces (1-{count}): ")
        try:
            order = [int(token) for token in text.split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
            self._print_prizes()
        else:
            print(f"Invalid order: each serial number from 1 to {count} must appear exactly once.")

    # ------------------------------------------------------------ lottery process
    def _start_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lottery = LotteryManager(self.users)
        results = ResultManager()
        print("\nLottery starting. During the draw you may type: pause / continue / terminate")
        for round_no, prize in enumerate(prizes, start=1):
            print(f"\n=== Round {round_no}: {prize.name} ===")
            print(f"Quantity: {prize.quantity}")
            print(f"Rules: {describe_rules(prize)}")
            round_winners = self._run_round(lottery, results, prize)
            print(f"--- Round {round_no} results ---")
            if not round_winners:
                print("  No winners in this round.")
            for winner in round_winners:
                print(f"  {winner.winning_order}. {winner.user.name} ({winner.user.department})")

        self.last_results = results
        self.last_prizes = prizes
        self._print_summary(results, prizes)

    def _run_round(self, lottery, results, prize):
        winners = []
        drawn = 0
        while drawn < prize.quantity:
            command = self.reader.poll(self.draw_interval)
            if command == "pause":
                if self._wait_while_paused() == "terminate":
                    print("Round terminated. Remaining draws for this prize are skipped.")
                    break
            elif command == "terminate":
                print("Round terminated. Remaining draws for this prize are skipped.")
                break
            elif command == "continue":
                print("Lottery is not paused.")

            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize; the round ends early.")
                break
            drawn += 1
            winner = results.add_winner(user, prize)
            winners.append(winner)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")
        return winners

    def _wait_while_paused(self):
        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = self.reader.read_line("").strip().lower()
            if command == "continue":
                print("Lottery resumed.")
                return "continue"
            if command == "terminate":
                return "terminate"
            print("Type 'continue' or 'terminate'.")

    def _print_summary(self, results, prizes):
        print("\n=== Final winner list (prize level from high to low) ===")
        for prize_name, winners in results.get_winners_by_prize(prizes).items():
            print(f"{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(f"  {winner.winning_order}. {winner.user.name} ({winner.user.department})")

    # ------------------------------------------------------------ export
    def _export_results(self):
        if self.last_results is None:
            print("No lottery has been run yet.")
            return
        _, message = self.last_results.export_to_csv(self.last_prizes)
        print(message)


def main():
    app = LotteryApp()
    try:
        app.run()
    except KeyboardInterrupt:
        print("\nInterrupted. Exiting.")

"""Participant list loading, validation and group extraction."""

from __future__ import annotations

import os
import re

from lottery_system.defaults import DEFAULT_USERS
from lottery_system.models import User

# Accept ',' or ';' (ASCII and full-width) as the field separator.
_FIELD_SEPARATOR = re.compile(r"[,;\uff0c\uff1b]")


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        users = [User(name=name, department=dept) for name, dept in DEFAULT_USERS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []

        if not path or not path.strip():
            return [], ["File Error: no file path given."]

        path = os.path.expanduser(path.strip())
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        try:
            with open(path, "rb") as fh:
                raw = fh.read()
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        text = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                text = raw.decode(encoding)
                break
            except UnicodeDecodeError:
                continue
        if text is None:
            return [], [f"File Error: cannot decode '{path}' as UTF-8 or GBK."]

        # enumerate from 1 so reported line numbers match the file
        for lineno, line in enumerate(text.splitlines(), start=1):
            stripped = line.strip()
            if not stripped:
                continue
            user, error = self._parse_line(stripped, lineno)
            if error:
                errors.append(error)
            else:
                users.append(user)
        return users, errors

    @staticmethod
    def _parse_line(line: str, lineno: int) -> tuple[User | None, str | None]:
        parts = _FIELD_SEPARATOR.split(line)
        if len(parts) != 2:
            return None, (
                f"Format Error at Line {lineno}: '{line}' "
                "(expected 'Name,Department Group')"
            )
        name, dept = parts[0].strip(), parts[1].strip()
        if not name or not dept:
            return None, (
                f"Format Error at Line {lineno}: '{line}' "
                "(name and department group must not be empty)"
            )
        return User(name=name, department=dept), None

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). The first occurrence of a name is kept."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return dict(sorted(info.items()))

"""Prize library management, avoidance groups and lottery ordering."""

from __future__ import annotations

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Keeps the ordered list of prizes; list order is the lottery sequence."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(f"Prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name must not be empty.")
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        groups = list(dict.fromkeys(groups))
        drawer = None
        if associated_drawer and associated_drawer.strip():
            drawer = associated_drawer.strip()

        prize = Prize(name=name, quantity=quantity, avoid_groups=groups,
                      associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def update_avoid_groups(self, name: str, avoid_groups: list[str]) -> bool:
        """Replace the avoidance groups of the named prize."""
        groups = [g.strip() for g in avoid_groups if g and g.strip()]
        for prize in self._prizes:
            if prize.name == name:
                prize.avoid_groups = list(dict.fromkeys(groups))
                return True
        return False

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """order: 1-based current positions, listed in the new order."""
        n = len(self._prizes)
        if len(order) != n:
            return False
        if not all(isinstance(i, int) and not isinstance(i, bool) for i in order):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

"""Lottery rule engine: eligibility filtering, avoidance, fallback and draws."""

from __future__ import annotations

import random

from lottery_system.constants import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # Whether each of the two most recent draws by Chen Qixian / Jin Duo
        # produced a Cheng Ru group member (True) or not (False).
        self.last_two_draw_results: list[bool] = []

    # ----- state helpers -------------------------------------------------
    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def _non_winners(self) -> list[User]:
        return [u for u in self._users if u.name not in self._winner_names]

    def _fallback_triggered(self) -> bool:
        """True when the last two lottery-person draws had no Cheng Ru winners."""
        return len(self.last_two_draw_results) >= 2 and not any(
            self.last_two_draw_results[-2:]
        )

    # ----- rule engine ---------------------------------------------------
    def _build_tiers(self, prize: Prize) -> list[list[User]]:
        """Return candidate tiers in priority order; earlier tiers are drawn first."""
        candidates = self._non_winners()
        drawer = prize.associated_drawer

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group members
        # (ignoring avoidance groups) when the protection condition holds.
        if drawer == MA_JIAQI and self._fallback_triggered():
            cheng_ru = [u for u in candidates if u.department == CHENG_RU_GROUP]
            if cheng_ru:
                return [cheng_ru]

        # Special rule: Li Bin only filters previous winners.
        if drawer == LI_BIN:
            return [candidates]

        # Standard rule: avoidance-group members are eligible but drawn last.
        avoid = set(prize.avoid_groups)
        primary = [u for u in candidates if u.department not in avoid]
        secondary = [u for u in candidates if u.department in avoid]
        return [primary, secondary]

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the currently drawable pool (the highest-priority non-empty tier)."""
        for tier in self._build_tiers(prize):
            if tier:
                return list(tier)
        return []

    def _record_draw(self, prize: Prize, user: User) -> None:
        drawer = prize.associated_drawer
        if drawer in (CHEN_QIXIAN, JIN_DUO):
            self.last_two_draw_results.append(user.department == CHENG_RU_GROUP)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == MA_JIAQI:
            # The protection window is consumed by Ma Jiaqi's draw.
            self.last_two_draw_results = []

    # ----- public draw API -----------------------------------------------
    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize; returns None when no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self._winner_names.add(user.name)
        self._record_draw(prize, user)
        return user

"""Winner recording, grouping, prize level sorting and CSV export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User

_LEVEL_WORDS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}
_UNRANKED = 10**6


def prize_level(name: str) -> int:
    """Return the rank of a prize from its name (1 = highest). Unranked go last."""
    for word in re.findall(r"[a-z]+", name.lower()):
        if word in _LEVEL_WORDS:
            return _LEVEL_WORDS[word]
    return _UNRANKED


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Sort prizes from high to low level; ties keep their given order."""
    return sorted(prizes, key=lambda p: prize_level(p.name))


@dataclass
class Winner:
    user: User
    prize_name: str
    winning_time: datetime = field(default_factory=datetime.now)
    sequence: int = 0


class ResultManager:
    """Keeps all winners in winning order."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize | str) -> Winner:
        prize_name = prize if isinstance(prize, str) else prize.name
        winner = Winner(
            user=user,
            prize_name=prize_name,
            winning_time=datetime.now(),
            sequence=len(self._winners) + 1,
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map prize name -> winners (winning order), keyed in the given prize order."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        ordered = sort_prizes_by_level(prizes)
        grouped = self.get_winners_by_prize(ordered)

        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize in ordered:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow([
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

from __future__ import annotations

import re
from typing import Optional

class AppState:
    """Holds the session state of the running program."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: Optional[LotteryManager] = None
        self.results = ResultManager()

    def set_users(self, users: list[User]) -> None:
        self.users = list(users)
        self.reset_lottery()

    def reset_lottery(self) -> None:
        self.lottery = None
        self.results = ResultManager()


# ----- input helpers ---------------------------------------------------------
def _ask(prompt: str) -> str:
    return input(prompt).strip()


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def _group_names(state: AppState) -> list[str]:
    return list(state.list_manager.get_group_info(state.users).keys())


# ----- list menus ------------------------------------------------------------
def menu_load_default(state: AppState) -> None:
    users, errors = state.list_manager.load_default_list()
    for err in errors:
        print(err)
    state.set_users(users)
    print(f"Loaded default team building list: {len(users)} participants.")


def menu_upload_custom(state: AppState) -> None:
    path = _ask("Enter path of the txt list (blank to cancel): ")
    if not path:
        print("Upload cancelled.")
        return

    users, errors = state.list_manager.load_custom_list(path)
    valid, duplicates = state.list_manager.validate_users(users)

    if errors:
        print("Invalid entries (will be ignored):")
        for err in errors:
            print(f"  {err}")
    if duplicates:
        print("Duplicate names (only first occurrence kept):")
        for user in duplicates:
            print(f"  {user.name} ({user.department})")

    if errors or duplicates:
        choice = _ask(
            "[c] continue with valid entries / [r] remove entries manually / "
            "[q] terminate upload: "
        ).lower()
        if choice == "q":
            print("Upload terminated. Current list unchanged.")
            return
        if choice == "r":
            valid = _manual_remove(valid)
        elif choice != "c":
            print("Unknown option. Upload terminated.")
            return

    if not valid:
        print("No valid entries remain. Upload failed.")
        return

    state.set_users(valid)
    print(f"Loaded custom list: {len(valid)} participants.")
    _print_group_info(state.list_manager.get_group_info(valid))


def _manual_remove(users: list[User]) -> list[User]:
    current = list(users)
    while current:
        print("Current valid entries:")
        for i, user in enumerate(current, 1):
            print(f"  {i}. {user.name} ({user.department})")
        raw = _ask("Enter numbers to remove (comma separated), or blank to finish: ")
        if not raw:
            break
        to_remove: set[int] = set()
        for token in re.split(r"[,\s]+", raw):
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(current):
                to_remove.add(int(token) - 1)
            else:
                print(f"Ignored invalid number: {token}")
        current = [u for i, u in enumerate(current) if i not in to_remove]
    return current

def _print_prizes(prizes: list[Prize]) -> None:
    if not prizes:
        print("No prizes configured.")
        return
    print("Prizes (lottery order):")
    for i, prize in enumerate(prizes, 1):
        groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
        drawer = prize.associated_drawer or "none"
        print(f"  {i}. {prize.name} | quantity: {prize.quantity} | "
              f"avoid groups: {groups} | lottery person: {drawer}")


def _choose_groups(state: AppState, existing: Optional[list[str]] = None) -> list[str]:
    known = _group_names(state)
    if known:
        print("Available department groups:")
        for i, group in enumerate(known, 1):
            print(f"  {i}. {group}")
    raw = _ask("Avoidance groups (numbers or names, comma separated; blank for none): ")
    if not raw:
        return []
    selected: list[str] = []
    for token in re.split(r"[,\uff0c]", raw):
        token = token.strip()
        if not token:
            continue
        if token.isdigit() and known:
            idx = int(token) - 1
            if 0 <= idx < len(known):
                selected.append(known[idx])
            else:
                print(f"Ignored invalid group number: {token}")
        elif known and token not in known:
            print(f"Ignored unknown group: {token}")
        else:
            selected.append(token)
    return list(dict.fromkeys(selected))


def _choose_drawer() -> Optional[str]:
    print("Lottery person options: " + ", ".join(KNOWN_DRAWERS))
    raw = _ask("Associated lottery person (blank for none): ")
    return raw or None


def _add_custom_prize(state: AppState) -> None:
    name = _ask("Prize name: ")
    raw_qty = _ask("Quantity: ")
    if not raw_qty.isdigit() or int(raw_qty) < 1:
        print("Quantity must be a positive integer. Prize not added.")
        return
    groups = _choose_groups(state)
    drawer = _choose_drawer()
    try:
        state.prize_manager.add_prize(name, int(raw_qty), groups, drawer)
    except ValueError as exc:
        print(f"Could not add prize: {exc}")
        return
    print(f"Prize '{name.strip()}' added.")


def _reorder_prizes(state: AppState) -> None:
    prizes = state.prize_manager.get_prizes()
    if not prizes:
        print("No prizes to reorder.")
        return
    _print_prizes(prizes)
    raw = _ask("Enter new order as serial numbers separated by spaces (e.g. 3 1 2): ")
    try:
        order = [int(t) for t in raw.replace(",", " ").split()]
    except ValueError:
        print("Order must contain only numbers. Order unchanged.")
        return
    if state.prize_manager.reorder_prizes(order):
        print("Lottery order updated.")
        _print_prizes(state.prize_manager.get_prizes())
    else:
        print("Invalid order: must contain every serial number exactly once. "
              "Order unchanged.")


def _edit_prize_groups(state: AppState) -> None:
    prizes = state.prize_manager.get_prizes()
    if not prizes:
        print("No prizes configured.")
        return
    _print_prizes(prizes)
    raw = _ask("Select prize number: ")
    if not raw.isdigit() or not (1 <= int(raw) <= len(prizes)):
        print("Invalid prize number.")
        return
    prize = prizes[int(raw) - 1]
    groups = _choose_groups(state)
    state.prize_manager.update_avoid_groups(prize.name, groups)
    print(f"Avoidance groups for '{prize.name}' updated: "
          f"{', '.join(groups) if groups else 'none'}")


# ----- lottery ---------------------------------------------------------------
def describe_rule(prize: Prize) -> str:
    drawer = prize.associated_drawer
    parts: list[str] = []
    if drawer == LI_BIN:
        parts.append("only previous winners are excluded (avoidance groups not applied)")
    else:
        parts.append("previous winners are excluded")
        if prize.avoid_groups:
            parts.append(
                "members of avoidance groups (" + ", ".join(prize.avoid_groups)
                + ") are drawn only after other members"
            )
    if drawer:
        parts.append(f"lottery person: {drawer}")
    if drawer == MA_JIAQI:
        parts.append(
            "Cheng Ru group protection applies if neither of the last two draws "
            f"by {CHEN_QIXIAN} / {JIN_DUO} produced a Cheng Ru group winner"
        )
    return "; ".join(parts)


def _await_action() -> str:
    """Wait for the user's command before drawing. Returns 'draw' or 'terminate'."""
    while True:
        cmd = _ask("Press Enter to draw, or type 'pause' / 'terminate': ").lower()
        if cmd == "":
            return "draw"
        if cmd == "terminate":
            return "terminate"
        if cmd == "pause":
            print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
            while True:
                sub = _ask("> ").lower()
                if sub == "continue":
                    print("Lottery resumed.")
                    break
                if sub == "terminate":
                    return "terminate"
                print("Please type 'continue' or 'terminate'.")
            continue
        if cmd == "continue":
            print("Lottery is not paused.")
            continue
        print("Unknown command.")


def _print_round_summary(prize: Prize, winners: list[Winner]) -> None:
    print(f"--- Round results: {prize.name} ---")
    if not winners:
        print("  (no winners drawn)")
    for i, winner in enumerate(winners, 1):
        print(f"  {i}. {winner.user.name} ({winner.user.department})")


def run_round(state: AppState, lottery: LotteryManager, prize: Prize) -> list[Winner]:
    won: list[Winner] = []
    for n in range(1, prize.quantity + 1):
        print(f"Currently drawing the {ordinal(n)} winner...")
        if _await_action() == "terminate":
            print("Current round terminated; skipping remaining draws.")
            break
        user = lottery.draw_one(prize)
        if user is None:
            print("No eligible candidates remain for this prize.")
            break
        winner = state.results.add_winner(user, prize)
        won.append(winner)
        print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")
    return won


def run_lottery(state: AppState) -> None:
    if not state.users:
        print("Please load a participant list first.")
        return
    prizes = state.prize_manager.get_prizes()
    if not prizes:
        print("Please configure prizes first.")
        return
    if state.results.get_winners():
        confirm = _ask("A lottery already has results. Restart from scratch? (y/n): ").lower()
        if confirm != "y":
            print("Lottery start cancelled.")
            return
    state.reset_lottery()
    state.lottery = LotteryManager(state.users)

    print("\n=== Lottery started ===")
    total_rounds = len(prizes)
    for round_no, prize in enumerate(prizes, 1):
        print(f"\n### Round {round_no}/{total_rounds}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {describe_rule(prize)}")
        winners = run_round(state, state.lottery, prize)
        _print_round_summary(prize, winners)

    print("\n=== All rounds completed ===")
    _print_final_summary(state, prizes)


def _print_final_summary(state: AppState, prizes: list[Prize]) -> None:
    ordered = sort_prizes_by_level(prizes)
    grouped = state.results.get_winners_by_prize(ordered)
    print("=== Winner list (by prize level, then winning order) ===")
    for prize_name, winners in grouped.items():
        print(f"[{prize_name}]")
        if not winners:
            print("  (none)")
        for i, winner in enumerate(winners, 1):
            print(f"  {i}. {winner.user.name} | {winner.user.department} | "
                  f"{prize_name} | {winner.winning_time:%Y-%m-%d %H:%M:%S}")


def menu_results(state: AppState) -> None:
    prizes = state.prize_manager.get_prizes()
    if not state.results.get_winners():
        print("No results yet.")
        return
    _print_final_summary(state, prizes)
    if _ask("Export results to file? (y/n): ").lower() == "y":
        _, message = state.results.export_to_csv(prizes)
        print(message)

def _print_main_menu() -> None:

            menu_results(state)
        elif choice == "0":
            print("Goodbye.")
            return

            print("Invalid option.")

"""Participant list loading, verification and group extraction."""

from __future__ import annotations

from lottery_system.defaults import DEFAULT_LIST_TEXT
from lottery_system.models import User

FORMAT_HINT = "expected 'Name,Department Group'"


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Parse the built-in default list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list from the given path."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse each line as 'Name,Department Group'."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse raw lines. Blank lines are skipped; malformed lines are reported."""
        users: list[User] = []
        errors: list[str] = []
        for number, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = text.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {number}: Format Error - '{text}' ({FORMAT_HINT})"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group (insertion order)."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management and lottery round ordering."""

from __future__ import annotations

from typing import Iterable

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the prize list in lottery round order (addition order by default)."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, None, drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: Iterable[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the round order. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = str(group).strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer = associated_drawer.strip() if associated_drawer else None
        drawer = drawer or None

        prize = Prize(clean_name, quantity, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current round order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order contains 1-based current positions in the new order."""
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(positions) != list(range(1, len(self._prizes) + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        return True

"""Lottery rule engine: eligibility filtering, drawing and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)

PROTECTED_DRAWERS = (CHEN_QIXIAN, JIN_DUO)


def describe_rules(prize: Prize) -> str:
    """Human readable description of the rules applied to a prize round."""
    rules: list[str] = []
    drawer = prize.associated_drawer
    if drawer:
        rules.append(f"lottery person: {drawer}")
    rules.append("previous winners excluded")
    if drawer == LI_BIN:
        rules.append("special rule: avoidance groups NOT applied")
    elif prize.avoid_groups:
        rules.append("avoid groups: " + ", ".join(prize.avoid_groups))
    else:
        rules.append("no avoidance groups")
    if drawer == MA_JIAQI:
        rules.append(
            "Cheng Ru group protection: if the last two draws of Chen Qixian and "
            "Jin Duo contained no Cheng Ru group member, only Cheng Ru group "
            "members are eligible"
        )
    return "; ".join(rules)


class LotteryManager:
    """Performs draws while enforcing all lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self.users: list[User] = list(users)
        self._rng = rng if rng is not None else random.Random()
        self.winner_names: set[str] = set()
        # Results (True = Cheng Ru group member) of the most recent draws by
        # Chen Qixian / Jin Duo, keeping at most the last two.
        self.last_two_draw_results: list[bool] = []

    def _fallback_active(self) -> bool:
        """True when the last two protected draws contained no Cheng Ru group member."""
        return len(self.last_two_draw_results) == 2 and not any(
            self.last_two_draw_results
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the users eligible for the next pick of the given prize."""
        available = [user for user in self.users if user.name not in self.winner_names]
        drawer = prize.associated_drawer

        # Cheng Ru group protection overrides avoidance groups.
        if drawer == MA_JIAQI and self._fallback_active():
            protected = [user for user in available if user.department == CHENG_RU_GROUP]
            if protected:
                return protected

        # Special rule: Li Bin's prizes only filter previous winners.
        if drawer == LI_BIN:
            return available

        avoid = set(prize.avoid_groups)
        return [user for user in available if user.department not in avoid]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible user exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.winner_names.add(user.name)
        if prize.associated_drawer in PROTECTED_DRAWERS:
            self.last_two_draw_results.append(user.department == CHENG_RU_GROUP)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return user

    def finish_round(self, prize: Prize) -> None:
        """Called when a round ends. Consumes the fallback protection after Ma Jiaqi's round."""
        if prize.associated_drawer == MA_JIAQI and self._fallback_active():
            self.last_two_draw_results = []

"""Winner recording, ordering, display grouping and CSV export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User

_LEVEL_WORDS = {
    "first": 1, "1st": 1,
    "second": 2, "2nd": 2,
    "third": 3, "3rd": 3,
    "fourth": 4, "4th": 4,
    "fifth": 5, "5th": 5,
    "sixth": 6, "6th": 6,
    "seventh": 7, "7th": 7,
    "eighth": 8, "8th": 8,
    "ninth": 9, "9th": 9,
    "tenth": 10, "10th": 10,
}


def prize_level(name: str) -> int | None:
    """Extract the prize level (1 = highest) from a prize name such as 'First Prize - ...'."""
    for token in re.findall(r"[a-z0-9]+", name.lower()):
        if token in _LEVEL_WORDS:
            return _LEVEL_WORDS[token]
    return None


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Sort prizes from high to low level; unknown levels go last in addition order."""

    def key(item: tuple[int, Prize]) -> tuple[bool, int, int]:
        index, prize = item
        level = prize_level(prize.name)
        return (level is None, level or 0, index)

    return [prize for _, prize in sorted(enumerate(prizes), key=key)]


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    order: int
    win_time: str


class ResultManager:
    """Stores winners in winning order and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for the prize."""
        winner = Winner(
            user=user,
            prize_name=prize.name,
            order=len(self._winners) + 1,
            win_time=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def _winners_for(self, prize_name: str) -> list[Winner]:
        return [winner for winner in self._winners if winner.prize_name == prize_name]

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Winners grouped by prize name, prizes ordered high to low level,
        winners within a prize in winning order."""
        result: dict[str, list[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            result[prize.name] = self._winners_for(prize.name)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize in sort_prizes_by_level(prizes):
                    for winner in self._winners_for(prize.name):
                        writer.writerow(
                            [
                                winner.user.name,
                                winner.user.department,
                                winner.prize_name,
                                winner.win_time,
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Command-line interaction for the lottery process."""

from __future__ import annotations

from typing import Callable

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager, describe_rules
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(number: int) -> str:
    """Return an English ordinal such as 1st, 2nd, 3rd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive command-line controller of the whole lottery process."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output: Callable[[str], None] = print,
    ) -> None:
        self._input_func = input_func
        self._output = output
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self._groups: list[str] = []
        self.lottery: LotteryManager | None = None

    # ------------------------------------------------------------------ helpers
    def _say(self, text: str = "") -> None:
        self._output(text)

    def _ask(self, prompt: str) -> str:
        return self._input_func(prompt).strip()

    @staticmethod
    def _parse_numbers(text: str) -> list[int] | None:
        """Parse numbers separated by commas and/or spaces. None if invalid."""
        numbers: list[int] = []
        for token in text.replace(" ", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if not token.isdigit():
                return None
            numbers.append(int(token))
        return numbers

    # --------------------------------------------------------------- main flow
    def run(self) -> int:
        try:
            self._say("=" * 60)
            self._say("  Company Team Building - Smart Lottery System")
            self._say("=" * 60)
            self._setup_list()
            self._setup_prizes()
            self._run_lottery()
            self._show_summary()
            self._export_prompt()
        except (EOFError, KeyboardInterrupt):
            self._say("\nExiting the lottery system.")
        return 0

    # ------------------------------------------------------------ list setup
    def _setup_list(self) -> None:
        while True:
            self._say("\n=== Step 1: Load participant list ===")
            self._say("1) Built-in default team building list")
            self._say("2) Upload custom txt list (each line: Name,Department Group)")
            choice = self._ask("Select an option: ")
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
            elif choice == "2":
                path = self._ask("Enter txt file path: ")
                users, errors = self.list_manager.load_custom_list(path)
            else:
                self._say("Invalid option.")
                continue
            valid = self._verify_list(users, errors)
            if valid is None:
                continue
            self.users = valid
            break
        self._show_groups()

    def _verify_list(self, users: list[User], errors: list[str]) -> list[User] | None:
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            self._say(f"List verified: {len(valid)} participants loaded.")
            return valid

        if errors:
            self._say("Invalid data found:")
            for error in errors:
                self._say(f"  - {error}")
        if duplicates:
            self._say("Duplicate names found (only the first occurrence is kept):")
            for user in duplicates:
                self._say(f"  - {user.name} ({user.department})")
        if not valid:
            self._say("No valid participants available.")
            return None

        choice = self._ask(
            "Enter 'r' to remove invalid/duplicate data and continue, "
            "or 't' to terminate the upload: "
        ).lower()
        if choice == "r":
            self._say(f"Continuing with {len(valid)} valid participants.")
            return valid
        self._say("Upload terminated.")
        return None

    def _show_groups(self) -> None:
        info = self.list_manager.get_group_info(self.users)
        self._groups = list(info.keys())
        self._say("\nValid department groups:")
        for number, (department, count) in enumerate(info.items(), start=1):
            self._say(f"  {number}. {department}: {count} people")

    # ----------------------------------------------------------- prize setup
    def _setup_prizes(self) -> None:
        while True:
            self._say("\n=== Step 2: Prize configuration ===")
            self._say("1) Load preset prize library")
            self._say("2) Add custom prize")
            self._say("3) Show prize list")
            self._say("4) Set lottery round order")
            self._say("5) Show department groups")
            self._say("6) Finish configuration and start lottery")
            choice = self._ask("Select an option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        self._say(f"  - {error}")
                else:
                    self._say("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._show_prizes()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                self._show_groups()
            elif choice == "6":
                if not self.prize_manager.get_prizes():
                    self._say("Please add at least one prize first.")
                    continue
                return
            else:
                self._say("Invalid option.")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            self._say("Prize name cannot be empty.")
            return
        quantity_text = self._ask("Quantity: ")
        if not quantity_text.isdigit() or int(quantity_text) < 1:
            self._say("Quantity must be a positive integer.")
            return

        self._say("Department groups (avoid groups are excluded from this prize):")
        for number, department in enumerate(self._groups, start=1):
            self._say(f"  {number}. {department}")
        group_text = self._ask(
            "Avoid groups (serial numbers separated by commas, blank for none): "
        )
        numbers = self._parse_numbers(group_text)
        if numbers is None or any(n < 1 or n > len(self._groups) for n in numbers):
            self._say("Invalid avoid group selection.")
            return
        avoid_groups = [self._groups[n - 1] for n in numbers]

        drawer = self._ask("Associated lottery person (blank for none): ")
        try:
            self.prize_manager.add_prize(
                name, int(quantity_text), avoid_groups, drawer or None
            )
        except ValueError as exc:
            self._say(f"Failed to add prize: {exc}")
            return
        self._say("Prize added.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self._say("\nCurrent round order:")
        for number, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) or "None"
            drawer = prize.associated_drawer or "None"
            self._say(
                f"  {number}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoid groups: {avoid} | Lottery person: {drawer}"
            )

    def _reorder_prizes(self) -> None:
        self._show_prizes()
        if not self.prize_manager.get_prizes():
            return
        text = self._ask(
            "Enter new round order as serial numbers separated by spaces "
            "(blank to keep current order): "
        )
        if not text:
            return
        numbers = self._parse_numbers(text)
        if numbers is None or not self.prize_manager.reorder_prizes(numbers):
            self._say("Invalid order. Each serial number must appear exactly once.")
            return
        self._say("Round order updated.")
        self._show_prizes()

    # --------------------------------------------------------------- lottery
    def _run_lottery(self) -> None:
        self.lottery = LotteryManager(self.users)
        prizes = self.prize_manager.get_prizes()
        self._say("\n=== Step 3: Lottery ===")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize)
        self._say("\nAll lottery rounds completed.")

    def _run_round(self, index: int, total: int, prize) -> None:
        assert self.lottery is not None
        self._say(f"\n--- Round {index}/{total}: {prize.name} ---")
        self._say(f"Quantity: {prize.quantity}")
        self._say(f"Rules: {describe_rules(prize)}")
        choice = self._ask(
            "Press Enter to start this round, or type 'terminate' to skip it: "
        ).lower()
        if choice == "terminate":
            self._say("Round skipped.")
            return

        round_winners = []
        for position in range(1, prize.quantity + 1):
            self._say(f"Currently drawing the {ordinal(position)} winner...")
            user = self.lottery.draw_one(prize)
            if user is None:
                self._say("No eligible candidates remain; ending this round early.")
                break
            winner = self.result_manager.add_winner(user, prize)
            round_winners.append(winner)
            self._say(
                f"Congratulations {user.name} ({user.department}) "
                f"for winning {prize.name}!"
            )
            if position == prize.quantity:
                break
            if self._round_control() == "terminate":
                self._say("Round terminated by operator; skipping the rest.")
                break

        self.lottery.finish_round(prize)

        self._say(f"Round results for {prize.name}:")
        if not round_winners:
            self._say("  (no winners)")
        for number, winner in enumerate(round_winners, start=1):
            self._say(
                f"  {ordinal(number)}. {winner.user.name} "
                f"({winner.user.department})"
            )

    def _round_control(self) -> str:
        """Between draws: Enter=next, 'pause' suspends, 'terminate' ends the round."""
        while True:
            command = self._ask(
                "[Enter] next draw | 'pause' | 'terminate': "
            ).lower()
            if command in ("", "continue"):
                return "next"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                self._say("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("> ").lower()
                    if resume == "continue":
                        self._say("Lottery resumed.")
                        return "next"
                    if resume == "terminate":
                        return "terminate"
                    self._say("Please type 'continue' or 'terminate'.")
            self._say("Unknown command.")

    # --------------------------------------------------------------- results
    def _show_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self._say("\n=== Final Winner List ===")
        for prize_name, winners in grouped.items():
            self._say(f"\n{prize_name}")
            if not winners:
                self._say("  (no winners)")
            for number, winner in enumerate(winners, start=1):
                self._say(
                    f"  {number}. {winner.user.name} ({winner.user.department}) "
                    f"- {winner.win_time}"
                )

    def _export_prompt(self) -> None:
        choice = self._ask("\nExport the winner list to a txt file (CSV format)? (y/n): ").lower()
        if choice not in ("y", "yes"):
            return
        _, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self._say(message)


def main() -> int:
    """Program entry point."""
    return LotteryCLI().run()

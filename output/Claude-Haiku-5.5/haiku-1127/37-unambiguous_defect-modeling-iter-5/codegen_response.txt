from __future__ import annotations

import os

Chen Qixian,Technology Department

Zhang Wei,Technology Department

Zhou Jie,Finance Department
Wu Hao,Technology Department
Zheng Ting,Product Department
Feng Yu,Cheng Ru Group
Chen Lu,Marketing Department
Xu Ming,Finance Department
Hu Xin,Human Resources Department
Guo Peng,Cheng Ru Group
Lin Na,Product Department
He Jun,Technology Department
Gao Yan,Marketing Department

class ListManager:
    """Loads, validates and summarizes participant lists."""

    DELIMITER = ","

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (one 'Name,Department' per line)."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it; returns (valid users, error messages)."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: unable to read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse text lines into users, collecting format errors by line number."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [part.strip() for part in line.split(self.DELIMITER)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    f"(expected 'Name{self.DELIMITER}Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence kept) and duplicate entries."""
        valid: list[User] = []
        duplicates: list[User] = []
        seen: set[str] = set()
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department -> number of members, sorted by department name."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

from __future__ import annotations

import re

from lottery_system.constants import (
    CHEN_QIXIAN,
    JIN_DUO,
    LEVEL_CUSTOM,
    LEVEL_FIRST,
    LEVEL_SECOND,
    LEVEL_THIRD,
    MA_JIAQI,
)
from lottery_system.models import Prize

# (name, quantity, avoid_groups, associated_drawer)
DEFAULT_PRIZES: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 3, [], CHEN_QIXIAN),
    ("Second Prize - Shopping Cards", 2, [], JIN_DUO),
    ("First Prize - 1 Day Annual Leave", 1, [], MA_JIAQI),
]


class PrizeManager:
    """Manages the prize library and the lottery round order."""

    _LEVEL_PATTERN = re.compile(r"^\s*(first|second|third)\b", re.IGNORECASE)
    _LEVELS = {"first": LEVEL_FIRST, "second": LEVEL_SECOND, "third": LEVEL_THIRD}

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Load the preset prize library. Returns a list of error messages."""
        if any(prize.is_preset for prize in self._prizes):
            return ["Preset prizes have already been loaded."]
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            prize = Prize(
                name=name,
                quantity=quantity,
                avoid_groups=list(avoid_groups),
                associated_drawer=drawer,
                level=self._infer_level(name),
                is_preset=True,
            )
            self._append(prize)
        return []

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Add a custom prize to the end of the lottery sequence."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=list(avoid_groups or []),
            associated_drawer=associated_drawer or None,
            level=self._infer_level(name),
            is_preset=False,
        )
        self._append(prize)

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        try:
            positions = [int(value) for value in order]
        except (TypeError, ValueError):
            return False
        count = len(self._prizes)
        if sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        for index, prize in enumerate(self._prizes, start=1):
            prize.sequence_number = index
        return True

    def _append(self, prize: Prize) -> None:
        self._prizes.append(prize)
        prize.sequence_number = len(self._prizes)

    def _infer_level(self, name: str) -> int:
        match = self._LEVEL_PATTERN.match(name)
        if match:
            return self._LEVELS[match.group(1).lower()]
        return LEVEL_CUSTOM

"""Lottery rule engine: eligibility filtering, special rules and fallback."""

from __future__ import annotations

import random

from lottery_system.constants import (
    CHENG_RU_GROUP,
    CHEN_QIXIAN,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    normalize_text,
)
from lottery_system.models import Prize, User

FALLBACK_SOURCE_DRAWERS = (CHEN_QIXIAN, JIN_DUO)
FALLBACK_TARGET_DRAWER = MA_JIAQI
FALLBACK_WINDOW = 2


class LotteryManager:
    """Draws winners while applying the lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users = list(users)
        self._drawn_names: set[str] = set()
        self._rng = rng or random.Random()
        # Results (True = contains Cheng Ru Group member) of the most recent draws
        # by the consecutive lottery persons (Chen Qixian, Jin Duo).
        self.last_two_draw_results: list[bool] = []

    @staticmethod
    def _is_drawer(prize: Prize, drawer: str) -> bool:
        return normalize_text(prize.associated_drawer or "") == normalize_text(drawer)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return normalize_text(user.department) == normalize_text(CHENG_RU_GROUP)

    def is_fallback_active(self, prize: Prize) -> bool:
        """True when Ma Jiaqi's prize pool is forced to Cheng Ru Group members."""
        return (
            self._is_drawer(prize, FALLBACK_TARGET_DRAWER)
            and len(self.last_two_draw_results) == FALLBACK_WINDOW
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return participants eligible for the next draw of the given prize."""
        # Anti-duplication: previous winners are always excluded.
        pool = [user for user in self._users if user.name not in self._drawn_names]

        # Fallback (Cheng Ru group protection) overrides avoidance groups.
        if self.is_fallback_active(prize):
            return [user for user in pool if self._is_cheng_ru(user)]

        # Special rule for Li Bin: relaxed filtering (avoidance groups not applied).
        if self._is_drawer(prize, LI_BIN):
            return pool

        avoided = {normalize_text(group) for group in prize.avoid_groups}
        return [user for user in pool if normalize_text(user.department) not in avoided]

    def draw_winner(self, prize: Prize) -> User | None:
        """Draw one winner for the prize; returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._drawn_names.add(user.name)
        self._record_draw(prize, user)
        return user

    def _record_draw(self, prize: Prize, user: User) -> None:
        if self._is_drawer(prize, FALLBACK_TARGET_DRAWER):
            # The protection window is consumed by Ma Jiaqi's draw.
            self.last_two_draw_results.clear()
        elif any(self._is_drawer(prize, drawer) for drawer in FALLBACK_SOURCE_DRAWERS):
            self.last_two_draw_results.append(self._is_cheng_ru(user))
            del self.last_two_draw_results[:-FALLBACK_WINDOW]

    def is_drawn(self, user: User) -> bool:
        """True if the user has already won a prize in this lottery."""
        return user.name in self._drawn_names

    def describe_rules(self, prize: Prize) -> list[str]:
        """Human-readable description of the rules applying to a prize round."""
        rules = ["Previous winners are excluded (no one wins twice)."]
        drawer = prize.associated_drawer
        if drawer:
            rules.append(f"Lottery person: {drawer}")
        if self._is_drawer(prize, LI_BIN):
            rules.append(
                "Special rule (Li Bin): relaxed filtering, avoidance groups are not applied."
            )
        elif prize.avoid_groups:
            rules.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        if self.is_fallback_active(prize):
            rules.append(
                f"Fallback protection active: only {CHENG_RU_GROUP} members are eligible."
            )
        return rules

"""Winner recording, ranking and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.constants import LEVEL_CUSTOM
from lottery_system.models import Prize, User


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str = ""
    prize_level: int = LEVEL_CUSTOM
    winning_order: int = 0
    winning_time: datetime = field(default_factory=datetime.now)


class ResultManager:
    """Keeps the winner history and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for the given prize."""
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=prize.level,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map prize name to its winners (in winning order), following the prize order."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def get_ranked_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Winners sorted by prize level (high to low), then by winning order within a prize."""
        indexed = sorted(enumerate(prizes), key=lambda item: (item[1].level, item[0]))
        ranked: list[Winner] = []
        seen: set[str] = set()
        for _, prize in indexed:
            if prize.name in seen:
                continue
            seen.add(prize.name)
            ranked.extend(w for w in self._winners if w.prize_name == prize.name)
        return ranked

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the ranked winner list to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        ranked = self.get_ranked_winners(prizes)
        if not ranked:
            return False, "No winners to export."
        stamp = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{stamp}.txt")
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(
                    ["Winning Order", "Prize Name", "Prize Level", "Name", "Department", "Winning Time"]
                )
                for winner in ranked:
                    writer.writerow(
                        [
                            winner.winning_order,
                            winner.prize_name,
                            winner.prize_level,
                            winner.user.name,
                            winner.user.department,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from __future__ import annotations

from lottery_system.constants import LOTTERY_PERSONS

def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer (1st, 2nd, 11th...)."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive command-line application."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list = []
        self.lottery: LotteryManager | None = None
        self.completed_prize_names: set[str] = set()

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _confirm(self, prompt: str) -> bool:
        return self._ask(prompt).lower() in ("y", "yes")

    def _available_groups(self) -> list[str]:
        if not self.users:
            return []
        return list(self.list_manager.get_group_info(self.users).keys())

    # --------------------------------------------------------------- main loop
    def run(self) -> None:
        print("Welcome to the Smart Lottery System.")
        while True:
            print(MAIN_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                self._menu_load_list()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self._menu_prizes()
            elif choice == "4":
                self._run_lottery()
            elif choice == "5":
                self._show_results()
            elif choice == "6":
                self._export_results()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    # ------------------------------------------------------ list management
    def _menu_load_list(self) -> None:
        if self.result_manager.get_winners():
            print("Cannot reload the participant list after the lottery has started.")
            return
        print("1. Built-in default list\n2. Custom txt file\n0. Cancel")
        choice = self._ask("Select a loading mode: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

        if errors:
            print(f"Found {len(errors)} invalid line(s):")
            for error in errors:
                print(f"  {error}")
            if not self._confirm("Remove invalid data and continue? (y/n): "):
                print("Upload terminated.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  Duplicate: {user.name} ({user.department})")
            if not self._confirm("Remove duplicates and keep first occurrences? (y/n): "):
                print("Upload terminated.")
                return

        if not valid:
            print("No valid participants found. The list was not loaded.")
            return

        self.users = valid
        self.lottery = None
        print(f"Loaded {len(valid)} participant(s).")

    def _show_groups(self) -> None:
        if not self.users:
            print("No participant list loaded yet.")
            return
        info = self.list_manager.get_group_info(self.users)
        print(f"Valid groups: {len(info)}")
        for index, (department, count) in enumerate(info.items(), start=1):
            print(f"  {index}. {department}: {count} member(s)")

    # ------------------------------------------------------- prize management
    def _menu_prizes(self) -> None:
        while True:
            print(PRIZE_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        print(error)
                else:
                    print("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._reorder_prizes()
            elif choice == "4":
                self._view_prizes()
            elif choice == "0":
                return
            else:
                print("Invalid option, please try again.")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            print("Prize name cannot be empty.")
            return
        raw_quantity = self._ask("Quantity: ")
        if not raw_quantity.isdigit() or int(raw_quantity) <= 0:
            print("Quantity must be a positive integer.")
            return
        drawer = self._select_drawer()
        avoid_groups = self._select_groups()
        try:
            self.prize_manager.add_prize(name, int(raw_quantity), avoid_groups, drawer)
        except ValueError as exc:
            print(f"Failed to add prize: {exc}")
            return
        print("Prize added.")

    def _select_drawer(self) -> str | None:
        print("Lottery persons:")
        for index, person in enumerate(LOTTERY_PERSONS, start=1):
            print(f"  {index}. {person}")
        raw = self._ask("Select lottery person number (0 for none): ")
        if raw in ("", "0"):
            return None
        if raw.isdigit() and 1 <= int(raw) <= len(LOTTERY_PERSONS):
            return LOTTERY_PERSONS[int(raw) - 1]
        print("Invalid lottery person number; no lottery person assigned.")
        return None

    def _select_groups(self) -> list[str]:
        groups = self._available_groups()
        if not groups:
            print("No participant list loaded; avoidance groups cannot be selected.")
            return []
        print("Department groups:")
        for index, group in enumerate(groups, start=1):
            print(f"  {index}. {group}")
        raw = self._ask("Enter group numbers to avoid (comma-separated, blank for none): ")
        selected: list[str] = []
        for token in raw.replace(" ", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                group = groups[int(token) - 1]
                if group not in selected:
                    selected.append(group)
            else:
                print(f"Ignored invalid group number: {token}")
        return selected

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes(prizes)
        raw = self._ask("Enter new order as serial numbers (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in raw.replace(",", " ").split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self._print_prizes(self.prize_manager.get_prizes())
        else:
            print("Invalid order: each serial number must appear exactly once.")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes(prizes)

    @staticmethod
    def _print_prizes(prizes: list[Prize]) -> None:
        for prize in prizes:
            drawer = prize.associated_drawer or "None"
            groups = ", ".join(prize.avoid_groups) or "None"
            print(
                f"{prize.sequence_number}. {prize.name} | quantity: {prize.quantity} | "
                f"lottery person: {drawer} | avoid groups: {groups}"
            )

    # ------------------------------------------------------- lottery process
    def _run_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)
        pending = [prize for prize in prizes if prize.name not in self.completed_prize_names]
        if not pending:
            print("All prizes have already been drawn.")
            return
        total = len(pending)
        print(f"Lottery starts: {total} round(s) in the configured order.")
        for round_no, prize in enumerate(pending, start=1):
            self._run_round(round_no, total, prize)
        print("\nAll lottery rounds finished. Use option 5 to view the complete winner list.")

    def _run_round(self, round_no: int, total: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {round_no}/{total}: {prize.name} | Quantity: {prize.quantity} ===")
        for rule in self.lottery.describe_rules(prize):
            print(f"  - {rule}")

        round_winners: list[Winner] = []
        for position in range(1, prize.quantity + 1):
            if not self._control_gate():
                print("Round terminated; remaining winners of this prize are skipped.")
                break
            print(f"Currently drawing the {ordinal(position)} winner...")
            user = self.lottery.draw_winner(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            round_winners.append(self.result_manager.add_winner(user, prize))
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        self.completed_prize_names.add(prize.name)
        print(f"\n--- Round {round_no}/{total} results: {prize.name} ---")
        if round_winners:
            for index, winner in enumerate(round_winners, start=1):
                print(f"  {index}. {winner.user.name} ({winner.user.department})")
        else:
            print("  No winners in this round.")

    def _control_gate(self) -> bool:
        """Wait for the operator before a draw. Returns False if the round is terminated."""
        while True:
            command = self._ask("[Enter] draw next | 'pause' | 'terminate' > ").lower()
            if command == "":
                return True
            if command == "terminate":
                return False
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                return self._paused_loop()
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    def _paused_loop(self) -> bool:
        while True:
            command = self._ask("paused > ").lower()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Lottery is paused. Type 'continue' or 'terminate'.")

    # ---------------------------------------------------------------- results
    def _show_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        ranked = self.result_manager.get_ranked_winners(prizes)
        if not ranked:
            print("No winners yet.")
            return
        print("\n==================== Winner List ====================")
        current: str | None = None
        index = 0
        for winner in ranked:
            if winner.prize_name != current:
                current = winner.prize_name
                index = 0
                print(f"\n[{current}]")
            index += 1
            print(f"  {index}. {winner.user.name} ({winner.user.department})")
        print("=====================================================")

    def _export_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        _, message = self.result_manager.export_to_csv(prizes)
        print(message)


def main() -> None:
    """Application entry point."""
    try:
        LotteryCLI().run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting the lottery system.")

"""Grid-based path planning for riders on the 8x8 street map.

The map is an 800x800 coordinate system divided into 100x100 cells.
Streets are the grid lines (points whose x or y coordinate is a multiple
of the cell size). The planner prefers routes whose segments run along
streets and always moves in axis-aligned, right-angle steps.
"""

import logging
from typing import List, Tuple

LOGGER = logging.getLogger("dispatch.path")

Point = Tuple[int, int]


class GridMap:
    """Description of the 800x800 map made of 100x100 grid cells.

    Attributes:
        width: Map width in units.
        height: Map height in units.
        cell_size: Size of one grid cell in units.
    """

    def __init__(self, width: int = 800, height: int = 800,
                 cell_size: int = 100) -> None:
        """Create a grid map.

        Args:
            width: Map width (must be a multiple of ``cell_size``).
            height: Map height (must be a multiple of ``cell_size``).
            cell_size: Size of one cell.

        Raises:
            ValueError: If the dimensions are not positive multiples of
                ``cell_size``.
        """
        if cell_size <= 0 or width <= 0 or height <= 0:
            raise ValueError("Map dimensions must be positive")
        if width % cell_size or height % cell_size:
            raise ValueError("Map dimensions must be multiples of cell_size")
        self.width = width
        self.height = height
        self.cell_size = cell_size

    @property
    def cells_per_side(self) -> int:
        """Number of grid cells along one side of the map (8 by default)."""
        return self.width // self.cell_size

    def is_valid_point(self, x: int, y: int) -> bool:
        """Return True if (x, y) lies inside the map bounds."""
        return 0 <= x <= self.width and 0 <= y <= self.height

    def is_street(self, x: int, y: int) -> bool:
        """Return True if the point lies on a street (grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def cell_center(self, col: int, row: int) -> Point:
        """Return the center point of the cell at (col, row)."""
        half = self.cell_size // 2
        return col * self.cell_size + half, row * self.cell_size + half


class PathPlanner:
    """Computes right-angle, street-preferring routes on the grid.

    The planner is stateless with respect to riders; it only needs the
    current position, the target and the alternate-route flag used by
    riders to avoid conflicts.
    """

    def __init__(self, grid_map: GridMap = None) -> None:
        """Create a planner bound to a grid map.

        Args:
            grid_map: Map to plan on. A default 800x800 map is used if None.
        """
        self.grid_map = grid_map or GridMap()

    @staticmethod
    def distance(x1: int, y1: int, x2: int, y2: int) -> int:
        """Return the Manhattan distance between two points."""
        return abs(x1 - x2) + abs(y1 - y2)

    @staticmethod
    def is_directly_above_or_below(x: int, y: int, tx: int, ty: int) -> bool:
        """Return True when the rider shares the target's x coordinate.

        This is the "special case" in which the rider moves purely
        vertically toward the target.
        """
        return x == tx and y != ty

    def _on_street_line(self, value: int) -> bool:
        """Return True if a single coordinate lies on a grid line."""
        return value % self.grid_map.cell_size == 0

    def plan(self, sx: int, sy: int, tx: int, ty: int,
             alternate: bool = False) -> List[Point]:
        """Plan a route from (sx, sy) to (tx, ty) as a list of waypoints.

        If the two points share an x or y coordinate the route is a single
        straight segment. Otherwise two candidate corners are considered:
        horizontal-first and vertical-first. The corner whose segments lie
        on streets scores higher and is chosen unless ``alternate`` is True,
        in which case the other corner is used (conflict avoidance).

        Args:
            sx: Start x coordinate.
            sy: Start y coordinate.
            tx: Target x coordinate.
            ty: Target y coordinate.
            alternate: Prefer the non-default corner when True.

        Returns:
            Ordered list of waypoints beginning with the start point and
            ending with the target point.
        """
        if sx == tx or sy == ty:
            return [(sx, sy), (tx, ty)]

        corner_a = (tx, sy)
        score_a = int(self._on_street_line(sy)) + int(self._on_street_line(tx))
        corner_b = (sx, ty)
        score_b = int(self._on_street_line(sx)) + int(self._on_street_line(ty))

        prefer_a = score_a >= score_b
        if alternate:
            prefer_a = not prefer_a
        corner = corner_a if prefer_a else corner_b
        return [(sx, sy), corner, (tx, ty)]

    def next_position(self, x: int, y: int, tx: int, ty: int, speed: int,
                      alternate: bool = False) -> Point:
        """Return the position reached after moving up to ``speed`` units.

        Movement follows the planned route and never overshoots the target.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            tx: Target x coordinate.
            ty: Target y coordinate.
            speed: Maximum distance travelled in one tick.
            alternate: Use the alternate corner of the route when True.

        Returns:
            The new (x, y) position.
        """
        waypoints = self.plan(x, y, tx, ty, alternate=alternate)
        cx, cy = x, y
        remaining = speed
        for wx, wy in waypoints[1:]:
            while remaining > 0 and (cx, cy) != (wx, wy):
                if cx != wx:
                    step = min(abs(wx - cx), remaining)
                    cx += step if wx > cx else -step
                else:
                    step = min(abs(wy - cy), remaining)
                    cy += step if wy > cy else -step
                remaining -= step
            if remaining <= 0:
                break
        return cx, cy

"""Order data model: coordinate validation, grid conversion and lifecycle."""

import itertools
from typing import Any, Dict, Sequence, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
GRID_CELLS = MAP_SIZE // CELL_SIZE

_order_ids = itertools.count(1)


def validate_coordinate(value: Any, name: str = "coordinate") -> int:
    """Validate that a coordinate is an integer within 0-800.

    Args:
        value: Value to validate.
        name: Label used in error messages.

    Returns:
        The validated integer.

    Raises:
        TypeError: If the value is not an integer.
        ValueError: If the value is outside 0-800.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise TypeError(f"{name} must be an integer, got {value!r}")
    if not 0 <= value <= MAP_SIZE:
        raise ValueError(f"{name} must be within 0-{MAP_SIZE}, got {value}")
    return value


def to_grid_center(value: int) -> int:
    """Convert a coordinate to the center of its grid cell.

    Implements ``center = (value // 100) * 100 + 50``. Values at the far
    boundary (800) are clamped into the last cell so that the center stays
    inside the map.

    Args:
        value: Coordinate in units (0-800).

    Returns:
        The cell center coordinate along that axis.
    """
    index = min(max(value // CELL_SIZE, 0), GRID_CELLS - 1)
    return index * CELL_SIZE + CELL_SIZE // 2


def _parse_point(point: Sequence[int], label: str) -> Tuple[int, int]:
    """Validate a two-element (x, y) point and return it as integers."""
    if len(point) != 2:
        raise ValueError(f"{label} must contain exactly 2 coordinates")
    return (validate_coordinate(point[0], f"{label}_x"),
            validate_coordinate(point[1], f"{label}_y"))


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        restaurant_x: Grid-center x of the restaurant.
        restaurant_y: Grid-center y of the restaurant.
        customer_x: Grid-center x of the customer.
        customer_y: Grid-center y of the customer.
        status: "created", "assigned" or "completed".
        rider_id: ID of the assigned rider, or None.
        start_time: Simulated tick at which the order was created.
        completion_time: Simulated tick of delivery, or None.
        time_taken: Ticks between creation and delivery, or None.
        distance: Route length (Manhattan distance along streets).
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int],
                 start_time: int = 0) -> None:
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: Restaurant (x, y) in the range 0-800.
            cus_pos: Customer (x, y) in the range 0-800.
            start_time: Simulated tick of creation.

        Raises:
            TypeError: If a coordinate is not an integer.
            ValueError: If a coordinate is out of range or malformed.
        """
        rx, ry = _parse_point(res_pos, "restaurant")
        cx, cy = _parse_point(cus_pos, "customer")
        self.id = next(_order_ids)
        self.restaurant_x = to_grid_center(rx)
        self.restaurant_y = to_grid_center(ry)
        self.customer_x = to_grid_center(cx)
        self.customer_y = to_grid_center(cy)
        self.status = "created"
        self.rider_id = None
        self.start_time = int(start_time)
        self.completion_time = None
        self.time_taken = None
        self.distance = float(abs(self.restaurant_x - self.customer_x)
                              + abs(self.restaurant_y - self.customer_y))

    @property
    def restaurant_pos(self) -> Tuple[int, int]:
        """Restaurant grid-center coordinates as a tuple."""
        return self.restaurant_x, self.restaurant_y

    @property
    def customer_pos(self) -> Tuple[int, int]:
        """Customer grid-center coordinates as a tuple."""
        return self.customer_x, self.customer_y

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.

        Raises:
            RuntimeError: If the order is already completed.
        """
        if self.status == "completed":
            raise RuntimeError(f"Order {self.id} is already completed")
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, completion_time: int) -> None:
        """Mark the order as delivered and record efficiency data.

        Args:
            completion_time: Simulated tick at which delivery finished.
        """
        self.status = "completed"
        self.completion_time = int(completion_time)
        self.time_taken = self.completion_time - self.start_time

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
        }

"""Rider model: state machine, street-based movement and event logging.

Rider state machine:
    Idle (0) -> Picking Up (1) when an order is accepted,
    Picking Up (1) -> Delivering (2) on arrival at the restaurant,
    Delivering (2) -> Idle (0) (or Picking Up for the next order)
    on delivery to the customer.
"""

import logging
from collections import deque
from typing import Any, Deque, Dict, List, Optional, Tuple

import globalvar
from path_planner import DEFAULT_PLANNER

LOGGER = logging.getLogger("dispatch.rider")

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_LABELS = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_X = 435
DEFAULT_Y = 385
DEFAULT_SPEED = 10
MAX_LOG_ENTRIES = 1000
MAX_PATH_POINTS = 2000
MAX_CONFLICT_WAITS = 3


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x: Current x coordinate.
        y: Current y coordinate.
        speed: Movement speed in units per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders; the first element is the active order.
        special_case: True while the rider is directly above or below its
            current target.
        log: Event entries with keys "event", "order_id", "time", etc.
        path_history: Recent (time, x, y) positions for route display.
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the default start position.

        Args:
            rider_id: Unique integer identifier.

        Raises:
            TypeError: If rider_id is not an integer.
        """
        if isinstance(rider_id, bool) or not isinstance(rider_id, int):
            raise TypeError("rider_id must be an integer")
        self.rider_id = rider_id
        self.x = DEFAULT_X
        self.y = DEFAULT_Y
        self.speed = DEFAULT_SPEED
        self.state = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.log: List[Dict[str, Any]] = []
        self.path_history: Deque[Tuple[int, int, int]] = deque(
            maxlen=MAX_PATH_POINTS)
        self._wait_ticks = 0
        self._record_position()
        self._add_log("spawn", None)

    @property
    def position(self) -> Tuple[int, int]:
        """Current (x, y) position."""
        return self.x, self.y

    @property
    def state_label(self) -> str:
        """Human-readable state name."""
        return STATE_LABELS.get(self.state, "unknown")

    def is_idle(self) -> bool:
        """Return True when the rider has no state and no assigned orders."""
        return self.state == STATE_IDLE and not self.orders

    def current_order(self) -> Optional[Any]:
        """Return the active order (first in the list), or None."""
        return self.orders[0] if self.orders else None

    def accept_order(self, order: Any) -> None:
        """Append an order to this rider's list and mark it assigned.

        Args:
            order: The Order to assign.
        """
        self.orders.append(order)
        order.assign_to(self.rider_id)

    def move(self) -> None:
        """Advance the rider by one tick.

        Handles state transitions and their log entries. Movement is limited
        to ``speed`` units along street-preferring routes.
        """
        order = self.current_order()
        if order is None:
            if self.state != STATE_IDLE:
                LOGGER.warning("Rider %d had state %d without orders; "
                               "resetting to idle", self.rider_id, self.state)
                self.state = STATE_IDLE
            self.special_case = False
            self._wait_ticks = 0
            return

        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP
            self._add_log("accepted_order", order.id)

        if self.state == STATE_PICKING_UP:
            tx, ty = order.restaurant_pos
            self._travel_to(tx, ty)
            if self.position == (tx, ty):
                self._arrive_at_merchant(order)
        elif self.state == STATE_DELIVERING:
            tx, ty = order.customer_pos
            self._travel_to(tx, ty)
            if self.position == (tx, ty):
                self._deliver(order)

    def _now(self) -> int:
        """Return the current simulated tick."""
        return int(globalvar.get_value("simulated_time", 0) or 0)

    def _add_log(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log (bounded in size)."""
        entry = {
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        overflow = len(self.log) - MAX_LOG_ENTRIES
        if overflow > 0:
            del self.log[:overflow]
        LOGGER.debug("Rider %d event %s order=%s", self.rider_id, event,
                     order_id)

    def _record_position(self) -> None:
        """Record the current position in the bounded path history."""
        self.path_history.append((self._now(), self.x, self.y))

    def _arrive_at_merchant(self, order: Any) -> None:
        """Transition from picking up to delivering."""
        self.state = STATE_DELIVERING
        self.special_case = False
        self._wait_ticks = 0
        self._add_log("arrived_at_merchant", order.id)

    def _deliver(self, order: Any) -> None:
        """Complete the active order and update the rider state."""
        order.complete(self._now())
        self.orders.remove(order)
        self._add_log("delivered_to_customer", order.id)
        self.state = STATE_PICKING_UP if self.orders else STATE_IDLE
        self.special_case = False
        self._wait_ticks = 0

    def _is_occupied(self, x: int, y: int) -> bool:
        """Return True if another rider currently stands at (x, y)."""
        for other in globalvar.get_value("riders", []) or []:
            if other is not self and (other.x, other.y) == (x, y):
                return True
        return False

    def _travel_to(self, tx: int, ty: int) -> None:
        """Move one tick toward (tx, ty) with conflict avoidance.

        If the next cell is occupied by another rider, the rider waits. After
        a wait the alternate route is tried; after MAX_CONFLICT_WAITS waits
        the rider proceeds anyway, which prevents deadlock.
        """
        self.special_case = DEFAULT_PLANNER.is_directly_above_or_below(
            self.x, self.y, tx, ty)
        if self.position == (tx, ty):
            return
        alternate = self._wait_ticks > 0
        nx, ny = DEFAULT_PLANNER.next_position(
            self.x, self.y, tx, ty, self.speed, alternate=alternate)
        if (nx, ny) != (self.x, self.y) and self._is_occupied(nx, ny):
            if self._wait_ticks < MAX_CONFLICT_WAITS:
                self._wait_ticks += 1
                LOGGER.debug("Rider %d waiting for (%d,%d)", self.rider_id,
                             nx, ny)
                return
        self._wait_ticks = 0
        self.x, self.y = nx, ny
        self._record_position()

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "state_label": self.state_label,
            "special_case": self.special_case,
            "order_ids": [order.id for order in self.orders],
        }

"""Order creation, distance-first assignment and repair of order state."""

import logging
import math
from typing import Any, List, Optional, Tuple

import globalvar
from order_model import Order
from rider import Rider, STATE_IDLE

LOGGER = logging.getLogger("dispatch.order")


def _euclidean_distance(rider: Rider, order: Order) -> float:
    """Return sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2)."""
    return math.sqrt((rider.x - order.restaurant_x) ** 2
                     + (rider.y - order.restaurant_y) ** 2)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Choose the best rider for an order.

    Idle riders are preferred and the closest one (Euclidean distance to the
    restaurant) wins. If no rider is idle, the rider with the fewest assigned
    orders is chosen, with distance as the tie-breaker, to balance load.

    Args:
        order: The order needing a rider.

    Returns:
        The selected Rider, or None if there are no riders.
    """
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None
    idle = [rider for rider in riders if rider.is_idle()]
    if idle:
        return min(idle, key=lambda r: _euclidean_distance(r, order))
    return min(riders,
               key=lambda r: (len(r.orders), _euclidean_distance(r, order)))


class OrderManager:
    """Creates, assigns and repairs orders.

    All order collections live in the global store: ``"orders"`` (every
    order ever created) and ``"pending_orders"`` (orders waiting for a rider).
    """

    def submit_order(self, res_pos: Tuple[int, int],
                     cus_pos: Tuple[int, int]) -> Order:
        """Create an order at the current simulated time and queue it.

        Args:
            res_pos: Restaurant (x, y) in 0-800.
            cus_pos: Customer (x, y) in 0-800.

        Returns:
            The newly created Order (status "created").
        """
        start = int(globalvar.get_value("simulated_time", 0) or 0)
        order = Order(res_pos, cus_pos, start_time=start)
        globalvar.get_value("orders").append(order)
        globalvar.get_value("pending_orders").append(order)
        LOGGER.info("Order %d submitted", order.id)
        return order

    def assign_pending(self) -> List[Tuple[Order, Rider]]:
        """Assign every pending order that can be matched to a rider.

        Returns:
            List of (order, rider) pairs assigned during this call.
        """
        assignments: List[Tuple[Order, Rider]] = []
        pending = globalvar.get_value("pending_orders")
        for order in list(pending):
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            rider.accept_order(order)
            pending.remove(order)
            assignments.append((order, rider))
            LOGGER.info("Order %d assigned to rider %d", order.id,
                        rider.rider_id)
        return assignments

    def all_orders(self) -> List[Order]:
        """Return every order created so far."""
        return list(globalvar.get_value("orders", []) or [])

    def pending_orders(self) -> List[Order]:
        """Return orders waiting for a rider."""
        return list(globalvar.get_value("pending_orders", []) or [])

    def repair_state(self) -> int:
        """Detect and fix inconsistent order/rider state.

        Fixes performed:
            * completed orders still listed on a rider are removed;
            * riders without orders are set to idle;
            * assigned orders that no rider holds return to the pending queue.

        Returns:
            Number of repairs performed.
        """
        fixes = 0
        riders = globalvar.get_value("riders", []) or []
        pending = globalvar.get_value("pending_orders", [])
        held_ids = set()
        for rider in riders:
            for order in list(rider.orders):
                if order.status == "completed":
                    rider.orders.remove(order)
                    fixes += 1
                else:
                    held_ids.add(order.id)
            if not rider.orders and rider.state != STATE_IDLE:
                rider.state = STATE_IDLE
                fixes += 1
        for order in globalvar.get_value("orders", []) or []:
            if order.status == "assigned" and order.id not in held_ids:
                order.status = "created"
                order.rider_id = None
                if order not in pending:
                    pending.append(order)
                fixes += 1
        if fixes:
            LOGGER.warning("Repaired %d inconsistent state item(s)", fixes)
        return fixes

"""System initialization, funds management, rider recruitment and clock."""

import logging
from typing import Any, Dict, List, Optional

import globalvar
from rider import Rider, STATE_IDLE

LOGGER = logging.getLogger("dispatch.system")

DEFAULT_CONFIG: Dict[str, int] = {
    "initial_funds": 1000,
    "recruit_cost": 300,
    "recruit_threshold": 300,
}


class SystemManager:
    """Facade over the global state for funds, riders and simulated time.

    The state itself lives in :mod:`globalvar`; this class provides the
    business operations (recruitment, ticking, shutdown).
    """

    def __init__(self, config: Optional[Dict[str, Any]] = None) -> None:
        """Create the manager using defaults overridden by ``config``.

        Args:
            config: Optional mapping of known parameter names to integers.
        """
        self.config: Dict[str, int] = dict(DEFAULT_CONFIG)
        if config:
            for key, value in config.items():
                if key in self.config:
                    self.config[key] = int(value)

    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return int(globalvar.get_value("funds", 0))

    @funds.setter
    def funds(self, value: int) -> None:
        """Set the system fund balance."""
        globalvar.set_value("funds", int(value))

    @property
    def riders(self) -> List[Rider]:
        """The dynamic rider list."""
        return globalvar.get_value("riders", [])

    @property
    def simulated_time(self) -> int:
        """Current simulated tick."""
        return int(globalvar.get_value("simulated_time", 0) or 0)

    def is_running(self) -> bool:
        """Return the system running state."""
        return bool(globalvar.get_value("is_running", False))

    def shutdown(self) -> None:
        """Clear the running flag so the main loop exits."""
        globalvar.set_value("is_running", False)
        LOGGER.info("System shutdown requested")

    def idle_riders(self) -> List[Rider]:
        """Return all riders that are idle (no state and no orders)."""
        return [rider for rider in self.riders
                if rider.state == STATE_IDLE and not rider.orders]

    def should_recruit(self) -> bool:
        """Return True when funds allow recruitment and demand exceeds supply.

        Trigger: funds >= recruit_threshold and pending orders > idle riders.
        """
        pending = len(globalvar.get_value("pending_orders", []) or [])
        return (self.funds >= self.config["recruit_threshold"]
                and pending > len(self.idle_riders()))

    def create_rider(self) -> Rider:
        """Recruit a new rider, deducting the recruitment cost.

        Returns:
            The newly created Rider.
        """
        counter = int(globalvar.get_value("rider_id_counter", 0)) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        self.funds = self.funds - self.config["recruit_cost"]
        LOGGER.info("Recruited rider %d; funds now %d", counter, self.funds)
        return rider

    def maybe_recruit(self) -> Optional[Rider]:
        """Recruit a rider if the trigger conditions hold.

        Returns:
            The new Rider, or None if no recruitment happened.
        """
        if self.should_recruit():
            return self.create_rider()
        return None

    def tick(self) -> int:
        """Advance the simulated clock by one tick and move every rider.

        Exceptions raised by a single rider are logged and do not stop the
        other riders (error recovery).

        Returns:
            The new simulated time.
        """
        globalvar.set_value("simulated_time", self.simulated_time + 1)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                LOGGER.exception("Rider %d failed during tick",
                                 rider.rider_id)
        return self.simulated_time

def init_system(config: Optional[Dict[str, Any]] = None) -> SystemManager:
    """Initialise the global state and the system manager.

    Resets the global store to defaults, applies configuration and sets the
    starting funds (1000 unless configured otherwise).

    Args:
        config: Optional configuration overrides.

    Returns:
        The initialised SystemManager.
    """
    global _manager
    globalvar._init()
    _manager = SystemManager(config)
    globalvar.set_value("funds", _manager.config["initial_funds"])
    LOGGER.info("System initialised with funds %d", _manager.funds)
    return _manager


def get_system() -> SystemManager:
    """Return the active SystemManager, initialising it if necessary."""
    if _manager is None:
        return init_system()
    return _manager

from typing import Iterable, List, Optional, Sequence, Tuple

from path_planner import GridMap
from rider import STATE_LABELS

Point = Tuple[int, int]


class DisplayManager:
    """Formats system status, order tables, rider logs and the ASCII map.

    The ASCII map uses a 17x17 character canvas for the 8x8 grid: street
    lines are drawn with '#' at even indices, grid-cell centers (where
    restaurants, customers and riders stand) lie at odd indices.
    Symbols: 'R' rider, 'S' restaurant, 'C' customer, '.' empty,
    '#' street, '*' route.
    """

    def __init__(self, grid_map: Optional[GridMap] = None) -> None:
        """Create a display bound to a grid map.

        Args:
            grid_map: Map description; an 800x800 map is used if None.
        """
        self.grid_map = grid_map or GridMap()

    def to_canvas(self, x: int, y: int) -> Tuple[int, int]:
        """Convert world coordinates to (column, row) on the ASCII canvas.

        Row 0 is the top of the map (largest y).
        """
        half = self.grid_map.cell_size // 2
        max_index = self.grid_map.cells_per_side * 2
        col = min(max(x // half, 0), max_index)
        row = min(max((self.grid_map.height - y) // half, 0), max_index)
        return col, row

    def render_status(self, funds: int, sim_time: int, riders: Sequence) -> str:
        """Return a snapshot of funds, time and every rider's state.

        Args:
            funds: Current system funds.
            sim_time: Current simulated tick.
            riders: Sequence of Rider objects.

        Returns:
            Multi-line status text.
        """
        lines = [
            f"System funds: {funds}",
            f"Simulated time: {sim_time}",
            f"Riders: {len(riders)}",
        ]
        if not riders:
            lines.append("  (no riders recruited yet)")
        for rider in riders:
            order_ids = ", ".join(str(o.id) for o in rider.orders) or "-"
            lines.append(
                f"  Rider #{rider.rider_id}: location=({rider.x}, {rider.y}) "
                f"status={rider.state_label} orders={len(rider.orders)} "
                f"[{order_ids}]")
        return "\n".join(lines)

    def render_orders(self, orders: Iterable) -> str:
        """Return a table with details of every order.

        Columns: ID, start time, restaurant, customer, distance, time taken,
        assigned rider and status.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} | {'Start':>5} | {'Restaurant':>12} | "
                  f"{'Customer':>12} | {'Distance':>8} | {'Taken':>5} | "
                  f"{'Rider':>5} | Status")
        lines = [header, "-" * len(header)]
        for order in orders:
            restaurant = f"({order.restaurant_x},{order.restaurant_y})"
            customer = f"({order.customer_x},{order.customer_y})"
            taken = "-" if order.time_taken is None else str(order.time_taken)
            rider = "-" if order.rider_id is None else f"#{order.rider_id}"
            lines.append(
                f"{order.id:>4} | {order.start_time:>5} | {restaurant:>12} | "
                f"{customer:>12} | {order.distance:>8.0f} | {taken:>5} | "
                f"{rider:>5} | {order.status}")
        return "\n".join(lines)

    def render_rider_logs(self, riders: Sequence) -> str:
        """Return every recorded status change for all riders."""
        lines: List[str] = []
        for rider in riders:
            for entry in rider.log:
                order = "-" if entry["order_id"] is None else entry["order_id"]
                state = STATE_LABELS.get(entry["state"], "unknown")
                lines.append(
                    f"t={entry['time']:>4} rider#{rider.rider_id:<3} "
                    f"{entry['event']:<22} order={order:<4} "
                    f"at ({entry['x']},{entry['y']}) state={state}")
        return "\n".join(lines) if lines else "No rider events recorded."

    def _draw_segment(self, canvas: List[List[str]], start: Point,
                      end: Point) -> None:
        """Draw a route segment between two canvas cells with '*'."""
        dc = (end[0] > start[0]) - (end[0] < start[0])
        dr = (end[1] > start[1]) - (end[1] < start[1])
        col, row = start
        self._mark(canvas, col, row)
        while (col, row) != end:
            if col != end[0]:
                col += dc
            else:
                row += dr
            self._mark(canvas, col, row)

    @staticmethod
    def _mark(canvas: List[List[str]], col: int, row: int) -> None:
        """Set a route symbol unless a marker already occupies the cell."""
        if canvas[row][col] not in "SCR":
            canvas[row][col] = "*"

    def render_map(self, riders: Sequence = (), orders: Sequence = (),
                   route: Optional[Sequence[Point]] = None) -> str:
        """Return the ASCII map with riders, orders and an optional route.

        Args:
            riders: Riders to mark with 'R'.
            orders: Orders whose restaurant ('S') and customer ('C') are
                shown while they are not completed.
            route: Optional sequence of (x, y) points drawn as connected
                line segments with '*'.

        Returns:
            Multi-line string including a legend and active order list.
        """
        size = self.grid_map.cells_per_side * 2 + 1
        canvas = [["#" if (c % 2 == 0 or r % 2 == 0) else "."
                   for c in range(size)] for r in range(size)]

        if route:
            previous = None
            for x, y in route:
                current = self.to_canvas(x, y)
                if previous is not None and previous != current:
                    self._draw_segment(canvas, previous, current)
                previous = current

        active = [o for o in orders if o.status != "completed"]
        for order in active:
            col, row = self.to_canvas(order.restaurant_x, order.restaurant_y)
            canvas[row][col] = "S"
            col, row = self.to_canvas(order.customer_x, order.customer_y)
            canvas[row][col] = "C"
        for rider in riders:
            col, row = self.to_canvas(rider.x, rider.y)
            canvas[row][col] = "R"

        lines = ["".join(row) for row in canvas]
        lines.append("")
        lines.append("Legend: R=rider S=restaurant C=customer "
                     ". =empty #=street *=route")
        if active:
            lines.append("Active orders:")
            for order in active:
                lines.append(
                    f"  Order #{order.id}: status={order.status} "
                    f"restaurant=({order.restaurant_x},{order.restaurant_y}) "
                    f"customer=({order.customer_x},{order.customer_y}) "
                    f"rider={'-' if order.rider_id is None else order.rider_id}")
        else:
            lines.append("No active orders.")
        return "\n".join(lines)

"""Command parsing, shortcut resolution, history and input validation."""

from typing import List, Sequence, Tuple

COORDINATE_NAMES = ("restaurant_x", "restaurant_y",
                    "customer_x", "customer_y")
MAX_HISTORY = 100

ALIASES = {
    "start": "start", "s": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "save": "save",
    "history": "history",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Place an order (restaurant then customer, 0-800).
                       Runs the simulation until the order is delivered.
  status (st)          Funds, time, and every rider's status and orders.
  orders (o)           Details of all orders.
  riders (r)           Event log of every rider status change.
  map (m)              ASCII map with riders, restaurants and customers.
  tick [n]             Advance the simulation by n ticks (default 1).
  save [path]          Save the state to a JSON file.
  history              Show recent commands.
  help (h, ?)          Show this help.
  quit (q, exit)       Leave the program.
Shortcuts: '!!' repeats the last command, '!n' repeats history entry n.
"""


class CommandError(Exception):
    """Raised for invalid user input or unknown commands."""


class InputManager:
    """Parses command lines and validates user input.

    Attributes:
        command_history: Previously entered command lines (oldest first).
    """

    def __init__(self, max_history: int = MAX_HISTORY) -> None:
        """Create an input manager with an empty history.

        Args:
            max_history: Maximum number of history entries kept.
        """
        self.command_history: List[str] = []
        self._max_history = max_history

    def _record(self, raw: str) -> None:
        """Append a command to the history, trimming old entries."""
        self.command_history.append(raw)
        overflow = len(self.command_history) - self._max_history
        if overflow > 0:
            del self.command_history[:overflow]

    def resolve_line(self, line: str) -> Tuple[str, List[str]]:
        """Resolve a raw input line into a canonical command and arguments.

        Handles shortcut keys, '!!' and '!n' history recall.

        Args:
            line: Raw user input.

        Returns:
            Tuple of (canonical command name, argument token list).

        Raises:
            CommandError: For empty input, unknown commands or invalid
                history references.
        """
        raw = line.strip()
        if raw == "!!":
            if not self.command_history:
                raise CommandError("No previous command in history.")
            raw = self.command_history[-1]
        elif raw.startswith("!") and raw[1:].isdigit():
            index = int(raw[1:])
            if not 1 <= index <= len(self.command_history):
                raise CommandError(f"No history entry #{index}.")
            raw = self.command_history[index - 1]

        tokens = raw.split()
        if not tokens:
            raise CommandError("Empty command. Type 'help' for commands.")
        name = tokens[0].lower()
        command = ALIASES.get(name)
        if command is None:
            raise CommandError(f"Unknown command '{name}'. Type 'help'.")
        self._record(raw)
        return command, tokens[1:]

    def parse_coordinates(self, tokens: Sequence[str]) -> List[int]:
        """Validate four integer coordinates in the range 0-800.

        Commas are accepted as separators, e.g. ``100,200 400,500``.

        Args:
            tokens: Raw coordinate tokens.

        Returns:
            [restaurant_x, restaurant_y, customer_x, customer_y].

        Raises:
            CommandError: If the count, type or range is invalid.
        """
        parts: List[str] = []
        for token in tokens:
            parts.extend(token.replace(",", " ").split())
        if len(parts) != 4:
            raise CommandError(
                "Expected 4 integer coordinates: restaurant_x restaurant_y "
                f"customer_x customer_y (got {len(parts)})")
        values: List[int] = []
        for label, text in zip(COORDINATE_NAMES, parts):
            try:
                value = int(text)
            except ValueError:
                raise CommandError(
                    f"{label} must be an integer, got '{text}'") from None
            if not 0 <= value <= 800:
                raise CommandError(
                    f"{label} must be within 0-800, got {value}")
            values.append(value)
        return values

    def parse_count(self, tokens: Sequence[str], default: int = 1,
                    maximum: int = 10000) -> int:
        """Parse an optional positive tick count argument.

        Raises:
            CommandError: If the argument is not a positive integer within
                the allowed maximum.
        """
        if not tokens:
            return default
        try:
            value = int(tokens[0])
        except ValueError:
            raise CommandError(f"Tick count must be an integer, got "
                               f"'{tokens[0]}'") from None
        if not 1 <= value <= maximum:
            raise CommandError(f"Tick count must be between 1 and {maximum}")
        return value

    @staticmethod
    def help_text() -> str:
        """Return the command help text."""
        return HELP_TEXT

"""JSON persistence for system parameters, riders and order history."""

import json
import logging
import os
import tempfile
from typing import Any, Dict

import globalvar

LOGGER = logging.getLogger("dispatch.persistence")

CONFIG_KEYS = ("initial_funds", "recruit_cost", "recruit_threshold")


def load_config(path: str) -> Dict[str, int]:
    """Load system parameter overrides from a JSON file.

    Missing or malformed files produce a warning and an empty configuration
    so that the system falls back to its defaults.

    Args:
        path: Path to the JSON configuration file (may be None).

    Returns:
        Dictionary of recognised integer parameters.
    """
    if not path or not os.path.exists(path):
        return {}
    try:
        with open(path, encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        LOGGER.warning("Could not read config %s: %s", path, exc)
        return {}
    if not isinstance(data, dict):
        LOGGER.warning("Config %s must contain a JSON object", path)
        return {}

    config: Dict[str, int] = {}
    for key in CONFIG_KEYS:
        if key in data:
            try:
                config[key] = int(data[key])
            except (TypeError, ValueError):
                LOGGER.warning("Ignoring non-integer config value for %s",
                               key)
    return config


def build_snapshot() -> Dict[str, Any]:
    """Build a JSON-serialisable snapshot of the current system state."""
    riders = globalvar.get_value("riders", []) or []
    orders = globalvar.get_value("orders", []) or []
    pending = globalvar.get_value("pending_orders", []) or []
    return {
        "funds": globalvar.get_value("funds", 0),
        "simulated_time": globalvar.get_value("simulated_time", 0),
        "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
        "riders": [rider.to_dict() for rider in riders],
        "orders": [order.to_dict() for order in orders],
        "pending_order_ids": [order.id for order in pending],
    }


def save_state(path: str) -> None:
    """Write the system snapshot to ``path`` atomically.

    Args:
        path: Destination JSON file.

    Raises:
        OSError: If the file cannot be written.
    """
    directory = os.path.dirname(os.path.abspath(path))
    snapshot = build_snapshot()
    fd, tmp_path = tempfile.mkstemp(dir=directory, suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as handle:
            json.dump(snapshot, handle, indent=2)
        os.replace(tmp_path, path)
    except OSError:
        if os.path.exists(tmp_path):
            os.remove(tmp_path)
        raise
    LOGGER.info("State saved to %s", path)

"""Command-line entry point for the food delivery dispatch simulator.

Run with:  python src/main.py [--config dispatch_config.json]
"""

import argparse
import logging
import os
import sys
from typing import Callable, Dict, List, Optional

import globalvar
from display import DisplayManager
from input_manager import CommandError, InputManager
from order import OrderManager
from persistence import load_config, save_state
from system import init_system, SystemManager

LOGGER = logging.getLogger("dispatch.main")

MAX_SIMULATION_TICKS = 2000
DEFAULT_CONFIG_PATH = "dispatch_config.json"
DEFAULT_SAVE_PATH = "dispatch_state.json"


class DispatchApp:
    """Interactive command loop tying all modules together."""

    def __init__(self, config_path: Optional[str] = None,
                 input_func: Callable[[str], str] = input,
                 output: Callable[..., None] = print) -> None:
        """Initialise the system and the interactive components.

        Args:
            config_path: Optional JSON configuration file.
            input_func: Function used to read user input.
            output: Function used to print output.
        """
        self.system: SystemManager = init_system(load_config(config_path))
        self.order_manager = OrderManager()
        self.display = DisplayManager()
        self.inputs = InputManager()
        self._input = input_func
        self._output = output
        self._handlers: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def run(self) -> None:
        """Run the main loop until the system stops or input ends."""
        self._output("Food Delivery Dispatch Simulator. "
                     "Type 'help' for commands.")
        while self.system.is_running():
            try:
                line = self._input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self._output("")
                break
            if not line or not line.strip():
                continue
            try:
                self.execute(line)
            except (EOFError, KeyboardInterrupt):
                self._output("")
                break
        self._output("Goodbye.")

    def execute(self, line: str) -> None:
        """Parse and run one command line with error recovery.

        Args:
            line: Raw user input.
        """
        try:
            command, args = self.inputs.resolve_line(line)
            self._handlers[command](args)
        except CommandError as exc:
            self._output(f"Error: {exc}")
        except (EOFError, KeyboardInterrupt):
            raise
        except Exception as exc:  # error recovery path
            LOGGER.exception("Unexpected error while executing %r", line)
            self._output(f"Internal error: {exc}. Attempting state repair.")
            fixes = self.order_manager.repair_state()
            self._output(f"State check complete ({fixes} repair(s)).")

    def _tick_once(self) -> None:
        """Advance one tick, recruit if needed and assign pending orders."""
        self.system.tick()
        self.system.maybe_recruit()
        self.order_manager.assign_pending()

    def cmd_start(self, args: List[str]) -> None:
        """Place an order and simulate it until delivery."""
        if not args:
            args = self._input(
                "Enter restaurant_x restaurant_y customer_x customer_y: "
            ).split()
        coords = self.inputs.parse_coordinates(args)
        start_time = self.system.simulated_time
        order = self.order_manager.submit_order((coords[0], coords[1]),
                                                (coords[2], coords[3]))
        new_rider = self.system.maybe_recruit()
        assignments = self.order_manager.assign_pending()
        rider = next((r for o, r in assignments if o is order), None)

        if rider is None:
            self._output(f"Order #{order.id} queued: no rider available "
                         f"(funds={self.system.funds}).")
            return

        self._output(f"Order #{order.id} accepted.")
        self._output(f"  Restaurant (grid center): "
                     f"({order.restaurant_x}, {order.restaurant_y})")
        self._output(f"  Customer (grid center): "
                     f"({order.customer_x}, {order.customer_y})")
        if new_rider is not None:
            self._output(f"  New rider added: #{new_rider.rider_id} "
                         f"(funds now {self.system.funds})")
        else:
            self._output("  New rider added: no")
        self._output(f"  Assigned rider: #{rider.rider_id} "
                     f"speed={rider.speed} units/tick")

        ticks = 0
        while order.status != "completed" and ticks < MAX_SIMULATION_TICKS:
            self._tick_once()
            ticks += 1

        if order.status == "completed":
            self._output(f"  Delivered in {order.time_taken} tick(s); "
                         f"route length {order.distance:.0f} units.")
        else:
            self._output(f"  Not delivered after {ticks} ticks; "
                         f"use 'orders' to track progress.")

        route = [(x, y) for (t, x, y) in rider.path_history if t >= start_time]
        self._output(self.display.render_map(
            self.system.riders, self.order_manager.all_orders(), route))

    def cmd_status(self, args: List[str]) -> None:
        """Show funds, time and rider snapshot."""
        self._output(self.display.render_status(
            self.system.funds, self.system.simulated_time, self.system.riders))

    def cmd_orders(self, args: List[str]) -> None:
        """Show details of all orders."""
        self._output(self.display.render_orders(
            self.order_manager.all_orders()))

    def cmd_riders(self, args: List[str]) -> None:
        """Show the status change log of all riders."""
        self._output(self.display.render_rider_logs(self.system.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map with riders and orders."""
        self._output(self.display.render_map(
            self.system.riders, self.order_manager.all_orders()))

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation by one or more ticks."""
        count = self.inputs.parse_count(args)
        for _ in range(count):
            self._tick_once()
        self._output(f"Simulated time is now {self.system.simulated_time}.")

    def cmd_save(self, args: List[str]) -> None:
        """Save the system state to a JSON file."""
        path = args[0] if args else DEFAULT_SAVE_PATH
        try:
            save_state(path)
        except OSError as exc:
            raise CommandError(f"Could not save to {path}: {exc}") from None
        self._output(f"State saved to {path}.")

    def cmd_history(self, args: List[str]) -> None:
        """List recently entered commands."""
        history = self.inputs.command_history
        if not history:
            self._output("No command history yet.")
            return
        for index, entry in enumerate(history, start=1):
            self._output(f"  {index:>3}  {entry}")

    def cmd_help(self, args: List[str]) -> None:
        """Show the command help."""
        self._output(self.inputs.help_text())

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the main loop."""
        self.system.shutdown()

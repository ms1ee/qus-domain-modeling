from __future__ import annotations

import os
import re

from ..constants import DEFAULT_USER_DATA
from ..models import User, normalize_text

_SEPARATOR_RE = re.compile(r"[,;]")


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in team building list."""
        users = [User(name, department) for name, department in DEFAULT_USER_DATA]
        return users, []

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom .txt list. Each line: "Name,Department" or "Name;Department"."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        if not str(path).lower().endswith(".txt"):
            return [], [f"File Error: only .txt files are supported: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        """Parse a list file. Invalid lines are reported and excluded."""
        try:
            text = self._read_text(path)
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        users: list[User] = []
        errors: list[str] = []
        for line_no, raw_line in enumerate(text.splitlines(), start=1):
            line = raw_line.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SEPARATOR_RE.split(line, maxsplit=1)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    @staticmethod
    def _read_text(path) -> str:
        with open(path, "rb") as handle:
            raw = handle.read()
        for encoding in ("utf-8-sig", "gbk"):
            try:
                return raw.decode(encoding)
            except UnicodeDecodeError:
                continue
        raise OSError("unsupported text encoding (expected UTF-8 or GBK)")

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        valid: list[User] = []
        duplicates: list[User] = []
        seen: set[str] = set()
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        """Return a mapping of department -> number of people, in first-seen order."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from ..constants import DEFAULT_PRIZE_DATA
from ..models import Prize, normalize_text


def _clean_groups(groups) -> list[str]:
    """Strip, drop empties and de-duplicate group names (order preserved)."""
    cleaned: list[str] = []
    if not groups:
        return cleaned
    seen: set[str] = set()
    for group in groups:
        text = str(group).strip()
        key = normalize_text(text)
        if text and key not in seen:
            seen.add(key)
            cleaned.append(text)
    return cleaned


class PrizeManager:
    """Maintains the ordered list of prizes."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZE_DATA:
            try:
                self.add_prize(name, quantity, [], drawer)
            except ValueError as exc:
                errors.append(f"Cannot load preset prize '{name}': {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        """Append a new prize to the end of the lottery sequence."""
        prize_name = str(name or "").strip()
        if not prize_name:
            raise ValueError("prize name cannot be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        if any(normalize_text(p.name) == normalize_text(prize_name) for p in self._prizes):
            raise ValueError(f"prize '{prize_name}' already exists")

        drawer = None
        if associated_drawer is not None and str(associated_drawer).strip():
            drawer = str(associated_drawer).strip()

        prize = Prize(prize_name, quantity, _clean_groups(avoid_groups), drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def set_avoid_groups(self, index: int, groups) -> bool:
        """Replace the avoidance groups of the prize at 0-based index."""
        if not 0 <= index < len(self._prizes):
            return False
        self._prizes[index].avoid_groups = _clean_groups(groups)
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if not isinstance(order, (list, tuple)) or len(order) != count:
            return False
        if any(isinstance(pos, bool) or not isinstance(pos, int) for pos in order):
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in order]
        return True

"""Lottery rule engine."""

from __future__ import annotations

import random

from ..constants import CHEN_QIXIAN, CHENG_RU_GROUP, JIN_DUO, LI_BIN, MA_JIAQI
from ..models import Prize, User, normalize_text


class LotteryManager:
    """Draws winners while applying the lottery rules.

    Rules:
    - Previous winners are always excluded (anti-duplication).
    - Special person "Li Bin": only previous winners are excluded; avoidance
      groups are not applied.
    - Other prizes: members of the prize's avoidance groups remain eligible but
      are drawn only after all other candidates.
    - Cheng Ru fallback: if neither of the last two draws by Chen Qixian / Jin Duo
      produced a Cheng Ru Group member, the next draw by Ma Jiaqi is limited to
      Cheng Ru Group members (overriding avoidance groups).
    """

    def __init__(self, users):
        self.users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self.winners: list[User] = []
        self.last_two_draw_results: list[bool] = []

    @staticmethod
    def _same_person(first, second) -> bool:
        return first is not None and normalize_text(first) == normalize_text(second)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return normalize_text(user.department) == normalize_text(CHENG_RU_GROUP)

    def _is_winner(self, user: User) -> bool:
        return normalize_text(user.name) in self._winner_keys

    def is_fallback_active(self, prize: Prize) -> bool:
        """True when the Cheng Ru protection fallback applies to this prize's draw."""
        if not self._same_person(prize.associated_drawer, MA_JIAQI):
            return False
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_tiers(self, prize: Prize) -> list[list[User]]:
        """Candidate tiers in draw priority order (earlier tiers drawn first)."""
        non_winners = [user for user in self.users if not self._is_winner(user)]
        drawer = prize.associated_drawer

        if self._same_person(drawer, LI_BIN):
            return [non_winners] if non_winners else []

        if self.is_fallback_active(prize):
            forced = [user for user in non_winners if self._is_cheng_ru(user)]
            if forced:
                return [forced]

        avoid = {normalize_text(group) for group in prize.avoid_groups}
        primary = [u for u in non_winners if normalize_text(u.department) not in avoid]
        deferred = [u for u in non_winners if normalize_text(u.department) in avoid]
        return [tier for tier in (primary, deferred) if tier]

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """All currently eligible users for the prize, in draw priority order."""
        pool: list[User] = []
        for tier in self._get_tiers(prize):
            pool.extend(tier)
        return pool

    def draw_winner(self, prize: Prize):
        """Draw one winner for the prize. Returns User, or None if nobody is eligible."""
        winner = None
        for tier in self._get_tiers(prize):
            if tier:
                winner = random.choice(tier)
                break
        if winner is None:
            return None

        self._winner_keys.add(normalize_text(winner.name))
        self.winners.append(winner)

        if any(self._same_person(prize.associated_drawer, person)
               for person in (CHEN_QIXIAN, JIN_DUO)):
            self.last_two_draw_results.append(self._is_cheng_ru(winner))
            del self.last_two_draw_results[:-2]
        return winner

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize."""
        rules = ["previous winners are excluded"]
        drawer = prize.associated_drawer
        if drawer:
            rules.append(f"lottery person: {drawer}")

        if self._same_person(drawer, LI_BIN):
            rules.append("special rule for Li Bin: avoidance groups are NOT applied")
        elif self.is_fallback_active(prize):
            rules.append(
                f"Cheng Ru protection active: only {CHENG_RU_GROUP} members are eligible"
            )
        elif prize.avoid_groups:
            rules.append(
                "avoidance groups " + ", ".join(prize.avoid_groups)
                + ": members are drawn after other candidates"
            )
        return "; ".join(rules)

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from ..models import Prize, User, normalize_text


def prize_level(prize_name: str) -> int:
    """Return the level rank of a prize (1 = highest). Unknown levels rank last."""
    name = normalize_text(prize_name)
    if "first" in name:
        return 1
    if "second" in name:
        return 2
    if "third" in name:
        return 3
    return 99


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    sequence: int
    won_at: datetime


class ResultManager:
    """Keeps all winning records of a lottery session."""

    def __init__(self):
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize) -> Winner:
        """Record a winner for a prize (Prize object or prize name)."""
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        record = Winner(
            user=user,
            prize_name=prize_name,
            sequence=len(self._winners) + 1,
            won_at=datetime.now(),
        )
        self._winners.append(record)
        return record

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        """Map each prize name to its winners, in winning order."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            name = prize.name if isinstance(prize, Prize) else str(prize)
            result[name] = sorted(
                (w for w in self._winners if w.prize_name == name),
                key=lambda w: w.sequence,
            )
        return result

    def ordered_prizes(self, prizes) -> list[Prize]:
        """Prizes sorted by level from high to low (stable for equal levels)."""
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda item: (prize_level(item[1].name), item[0]))
        return [prize for _, prize in indexed]

    def get_ordered_winners(self, prizes) -> list[Winner]:
        """Winners ordered by prize level (high to low), then winning time."""
        ordered: list[Winner] = []
        winners_map = self.get_winners_by_prize(self.ordered_prizes(prizes))
        for prize in self.ordered_prizes(prizes):
            ordered.extend(winners_map.get(prize.name, []))
        return ordered

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export winners to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        date_text = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_text}.txt")
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in self.get_ordered_winners(prizes):
                    writer.writerow([
                        winner.user.name,
                        winner.user.department,
                        winner.prize_name,
                        winner.won_at.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

def ordinal(number: int) -> str:
    """Return an English ordinal such as 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def parse_int_list(text: str):
    """Parse '2, 1 3' into [2, 1, 3]. Returns None if any token is not an integer."""
    tokens = [token for token in re.split(r"[,\s]+", text.strip()) if token]
    if not tokens:
        return None
    try:
        return [int(token) for token in tokens]
    except ValueError:
        return None


class LotteryApp:

        "\n========== Team Building Smart Lottery ==========\n"

    @staticmethod

        return input(prompt).strip()

    # ------------------------------------------------------------------ main loop
    def run(self) -> None:
        print("Welcome to the Team Building Smart Lottery System.")
        while True:
            print(self.MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                self.load_participants()
            elif choice == "2":

            elif choice == "3":
                self.start_lottery()
            elif choice == "4":

            elif choice == "0":

    # ------------------------------------------------------------ list management
    def load_participants(self) -> None:
        print("\n--- Load Participant List ---")
        print("1. Built-in default team building list")
        print("2. Upload custom txt list (format per line: Name,Department)")
        print("0. Back")
        choice = self._ask("Select: ")

        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip('"')
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

        if errors:
            print(f"\nFound {len(errors)} problem(s) in the list:")
            for error in errors:
                print(f"  - {error}")
            decision = self._ask(
                "[R] Remove invalid data and continue / [T] Terminate upload: "

            if decision != "r":
                print("Upload terminated.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"\nFound {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  - {user.name} ({user.department})")
            decision = self._ask(
                "[R] Remove duplicate entries and continue / [T] Terminate upload: "

            if decision != "r":
                print("Upload terminated.")
                return

            print("No valid participants found. List not loaded.")
            return

        print(f"Loaded {len(self.users)} participant(s).")
        self._show_groups()

        for index, (group, count) in enumerate(info.items(), start=1):
            print(f"  {index}. {group}: {count} people")

            print("2. Add custom prize")
            print("3. Set avoidance groups for a prize")

                    print("Preset prizes loaded.")

                self._set_avoid_groups()
            elif choice == "4":

            else:

    @staticmethod

        groups = ", ".join(prize.avoid_groups) or "-"

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            print("Prize name cannot be empty.")
            return
        quantity_text = self._ask("Quantity: ")
        if not quantity_text.isdigit() or int(quantity_text) <= 0:

    def _choose_groups(self) -> list[str]:
        info = self.list_manager.get_group_info(self.users)
        names = list(info.keys())
        print("Department groups:")
        for index, name in enumerate(names, start=1):
            print(f"  {index}. {name} ({info[name]} people)")
        text = self._ask("Enter group numbers (comma separated), or press Enter for none: ")
        return self._parse_groups(text, names)

    @staticmethod
    def _parse_groups(text: str, names: list[str]) -> list[str]:
        selected: list[str] = []
        for token in text.split(","):
            token = token.strip()
            if not token:
                continue
            chosen = None
            if token.isdigit() and 1 <= int(token) <= len(names):
                chosen = names[int(token) - 1]
            else:
                for name in names:
                    if normalize_text(name) == normalize_text(token):
                        chosen = name
                        break
            if chosen is None:
                print(f"Ignored unknown group: {token}")
            elif chosen not in selected:
                selected.append(chosen)
        return selected

    def _choose_drawer(self):
        print("Associated lottery person:")
        print("  0. None")
        for index, person in enumerate(PRESET_DRAWERS, start=1):
            print(f"  {index}. {person}")
        other_index = len(PRESET_DRAWERS) + 1
        print(f"  {other_index}. Other (type name)")
        choice = self._ask("Select: ")

        if choice in ("", "0"):
            return None
        if choice.isdigit() and 1 <= int(choice) <= len(PRESET_DRAWERS):
            return PRESET_DRAWERS[int(choice) - 1]
        if choice == str(other_index):
            return self._ask("Enter lottery person name: ") or None
        return choice

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            print(self._format_prize(index, prize))
        choice = self._ask("Prize number: ")
        if not choice.isdigit() or not 1 <= int(choice) <= len(prizes):
            print("Invalid prize number.")
            return
        if not self.users:
            print("Load participant list first.")
            return
        groups = self._choose_groups()
        if self.prize_manager.set_avoid_groups(int(choice) - 1, groups):
            print("Avoidance groups updated.")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            print(self._format_prize(index, prize))
        order = parse_int_list(
            self._ask("Enter new order as serial numbers (e.g. 2,1,3): ")
        )
        if order is None:
            print("Invalid input. Please enter serial numbers separated by commas.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ------------------------------------------------------------- lottery flow
    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first (option 1).")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first (option 2).")
            return

        lottery = LotteryManager(self.users)
        results = ResultManager()
        print("\n========== Lottery Started ==========")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, len(prizes), prize, lottery, results)

        self.results = results
        self.lottery_prizes = prizes
        print("\n========== All Rounds Completed ==========")
        self._print_summary(results, prizes)

    def _await_action(self) -> str:
        """Return 'draw' or 'terminate' according to operator commands."""
        while True:
            command = self._ask(
                "[Enter/continue] draw next | pause | terminate: "
            ).lower()
            if command in ("", "continue"):
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("(paused) > ").lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        break
                    if resume == "terminate":
                        return "terminate"
                    print("Unknown command. Use 'continue' or 'terminate'.")
                continue
            print("Unknown command.")

    def _run_round(self, round_no, total, prize, lottery: LotteryManager,
                   results: ResultManager) -> None:
        print(f"\n===== Round {round_no}/{total}: {prize.name} =====")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {lottery.describe_rules(prize)}")

        drawn: list[User] = []
        while len(drawn) < prize.quantity:
            action = self._await_action()
            if action == "terminate":
                print("Round terminated. Remaining draws for this prize are skipped.")
                break

            print(f"Currently drawing the {ordinal(len(drawn) + 1)} winner...")
            winner = lottery.draw_winner(prize)
            if winner is None:
                print("No eligible candidates remain for this prize. Round ends.")
                break

            results.add_winner(winner, prize)
            drawn.append(winner)
            print(f"Congratulations {winner.name} ({winner.department}) "
                  f"for winning {prize.name}!")

        self._print_round_results(prize, drawn)

    @staticmethod
    def _print_round_results(prize, winners: list[User]) -> None:
        print(f"\nRound results for {prize.name} (winning order):")
        if not winners:
            print("  No winners in this round.")
            return
        for index, user in enumerate(winners, start=1):
            print(f"  {index}. {user.name} ({user.department})")

    @staticmethod
    def _print_summary(results: ResultManager, prizes) -> None:
        winners = results.get_ordered_winners(prizes)
        if not winners:
            print("No winners were recorded.")
            return
        print("\nComplete winner list (by prize level, then winning order):")
        print(f"{'No.':<5}{'Name':<16}{'Department':<30}Prize Name")
        for index, record in enumerate(winners, start=1):
            print(f"{index:<5}{record.user.name:<16}{record.user.department:<30}"
                  f"{record.prize_name}")

    # ---------------------------------------------------------------- results
    def view_results(self) -> None:
        if self.results is None:
            print("No lottery has been run yet.")
            return
        while True:
            print("\n--- Results ---")
            print("1. Show winner list")
            print("2. Export winner list to txt (CSV format)")
            print("0. Back")
            choice = self._ask("Select: ")
            if choice == "1":
                self._print_summary(self.results, self.lottery_prizes)
            elif choice == "2":
                _, message = self.results.export_to_csv(self.lottery_prizes)
                print(message)
            elif choice == "0":
                return
            else:
                print("Invalid option.")


def main() -> None:
    app = LotteryApp()

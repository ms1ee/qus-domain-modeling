"""Participant list loading, verification and group extraction."""

import os

from lottery_system.data.default_list import DEFAULT_LIST_TEXT
from lottery_system.models import User


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list from the given path."""
        if not path or not path.strip():
            return [], ["File Error: no file path provided."]
        path = path.strip()
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'."]
        if not path.lower().endswith(".txt"):
            return [], [f"File Error: only .txt files are supported: '{path}'."]
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a line-based list file ("Name,Department" per line)."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse lines into users, collecting format errors with line numbers."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [part.strip() for part in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group (first-seen order)."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    _DEFAULT_PRIZES = [
        ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
        ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
        ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
    ]

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Load the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in self._DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a new prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups: list[str] = []
        for group in avoid_groups or []:
            group_name = str(group).strip()
            if group_name and group_name not in groups:
                groups.append(group_name)

        drawer = None
        if associated_drawer and associated_drawer.strip():
            drawer = associated_drawer.strip()

        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery sequence order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine and draw execution."""

import random

from lottery_system.constants import (
    CHENG_RU_GROUP,
    FALLBACK_TRIGGER_DRAWERS,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager


def _is_drawer(prize: Prize, drawer_name: str) -> bool:
    """Return True if the prize's lottery person matches the given name."""
    if prize.associated_drawer is None:
        return False
    return prize.associated_drawer.strip().casefold() == drawer_name.casefold()


class LotteryManager:
    """Executes lottery draws while applying the lottery rule engine."""

    def __init__(self, users: list[User]):
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Cheng Ru group presence (True/False) in the two most recent rounds
        # drawn by Chen Qixian or Jin Duo.
        self.last_two_draw_results: list[bool] = []
        self._rng = random.Random()

    def _winner_names(self) -> set[str]:
        return {winner.user.name for winner in self.result_manager.get_all_winners()}

    def is_fallback_active(self) -> bool:
        """True if neither of the last two trigger-drawer rounds included Cheng Ru members."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def is_fallback_for(self, prize: Prize) -> bool:
        """True if the fallback mechanism forces this prize's pool to Cheng Ru members."""
        return _is_drawer(prize, MA_JIAQI) and self.is_fallback_active()

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the candidate pool for a prize under the lottery rules.

        - Previous winners are always excluded (anti-duplication).
        - Li Bin's prizes use relaxed rules: avoidance groups are ignored.
        - Ma Jiaqi's prizes are restricted to Cheng Ru group members when the
          fallback is active (even if that group is an avoidance group).
        - Otherwise members of the prize's avoidance groups are excluded.
        """
        winners = self._winner_names()
        available = [user for user in self.users if user.name not in winners]

        if _is_drawer(prize, LI_BIN):
            return available

        if self.is_fallback_for(prize):
            protected = [user for user in available if user.in_group(CHENG_RU_GROUP)]
            if protected:
                return protected

        return [
            user
            for user in available
            if not any(user.in_group(group) for group in prize.avoid_groups)
        ]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the eligible candidate pool of a prize."""
        return self._get_eligible_pool(prize)

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.result_manager.add_winner(user, prize)
        return user

    def record_round_result(self, prize: Prize, round_winners: list[User]) -> None:
        """Record whether a finished round of a trigger drawer included Cheng Ru members."""
        if not any(_is_drawer(prize, drawer) for drawer in FALLBACK_TRIGGER_DRAWERS):
            return
        has_cheng_ru = any(user.in_group(CHENG_RU_GROUP) for user in round_winners)
        self.last_two_draw_results = (self.last_two_draw_results + [has_cheng_ru])[-2:]

"""Winner tracking, result grouping and CSV export."""

import csv
import os
import re
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User

_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3, "fourth": 4, "fifth": 5}


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    sequence: int  # global winning order (1-based)


def prize_level(prize: Prize) -> int:
    """Return the level of a prize based on its name (1 = highest). Unlabeled prizes go last."""
    match = re.search(r"\b(first|second|third|fourth|fifth)\b", prize.name, re.IGNORECASE)
    if match:
        return _LEVEL_WORDS[match.group(1).lower()]
    return len(_LEVEL_WORDS) + 1


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Sort prizes from highest level to lowest; ties keep the lottery sequence order."""
    return sorted(prizes, key=prize_level)


class ResultManager:
    """Stores winners and produces result views and exports."""

    def __init__(self):
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(user=user, prize_name=prize.name, sequence=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, in the order of the given prizes and winning time."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export winners to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        filename = f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        sorted_prizes = sort_prizes_by_level(prizes)
        grouped = self.get_winners_by_prize(sorted_prizes)
        try:
            with open(path, "w", encoding="utf-8-sig", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize Name", "Name", "Department", "Winning Order"])
                for prize in sorted_prizes:
                    for order, winner in enumerate(grouped.get(prize.name, []), start=1):
                        writer.writerow(
                            [prize.name, winner.user.name, winner.user.department, order]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def _same_text(a: str | None, b: str) -> bool:
    return a is not None and a.strip().casefold() == b.casefold()


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _ask_positive_int(self, prompt: str) -> int:
        while True:
            raw = self._ask(prompt)
            if raw.isdigit() and int(raw) > 0:
                return int(raw)
            print("Please enter a positive integer.")

    # --------------------------------------------------------------- main loop
    def run(self) -> None:
        print("=" * 60)
        print("Company Team Building - Smart Lottery System")
        print("=" * 60)
        while True:
            self._print_main_menu()
            choice = self._ask("Select an option: ")
            if choice == "1":
                self.load_participants()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.prize_menu()
            elif choice == "4":
                self.start_lottery()
            elif choice == "5":
                self.show_results(ask_export=True)
            elif choice == "0":
                print("Goodbye!")
                break
            else:
                print("Invalid option, please try again.")

    @staticmethod
    def _print_main_menu() -> None:
        print("\n---------------- Main Menu ----------------")
        print("1. Load participant list")
        print("2. View department groups")
        print("3. Prize configuration")
        print("4. Start lottery")
        print("5. View / export results")
        print("0. Exit")

    # ------------------------------------------------------- list management
    def load_participants(self) -> None:
        print("\n-- Load participant list --")
        print("1. Built-in default team building list")
        print("2. Custom txt list (format: Name,Department per line)")
        print("0. Cancel")
        choice = self._ask("Select an option: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip('"').strip("'")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

    def _verify_and_apply_list(self, users: list[User], errors: list[str]) -> None:
        if errors:
            print("Format errors detected:")
            for error in errors:
                print(f"  {error}")
            choice = self._ask(
                "Enter 'r' to remove invalid data and continue, 't' to terminate upload: "

            if choice != "r":

        valid, duplicates = self.list_manager.validate_users(users)

            if choice != "r":

        self.lottery = None

    def _print_prizes(self, prizes: list[Prize]) -> None:

        print("\nLottery sequence:")
        for index, prize in enumerate(prizes, start=1):

    def add_prize_interactive(self) -> None:

    def _select_avoid_groups(self) -> list[str]:
        if not self.users:
            print("No participant list loaded; no avoidance groups can be selected.")
            return []
        groups = list(self.list_manager.get_group_info(self.users).keys())
        print("Available groups:")
        for index, group in enumerate(groups, start=1):
            print(f"  {index}. {group}")
        raw = self._ask("Enter group numbers to avoid, comma separated (Enter for none): ")
        selected: list[str] = []
        for token in raw.split(","):
            token = token.strip()
            if not token:
                continue
            group = None
            if token.isdigit() and 1 <= int(token) <= len(groups):
                group = groups[int(token) - 1]
            else:
                for candidate in groups:
                    if candidate.casefold() == token.casefold():
                        group = candidate
                        break
            if group is None:
                print(f"  Ignored unknown group: '{token}'")
            elif group not in selected:
                selected.append(group)
        return selected

    def reorder_prizes_interactive(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes(prizes)
        raw = self._ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in raw.split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
            self._print_prizes(self.prize_manager.get_prizes())
        else:
            print(f"Invalid order: each serial number 1-{len(prizes)} must appear exactly once.")

    # ------------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is not None and self.lottery.result_manager.get_all_winners():
            confirm = self._ask("Existing results will be cleared. Start a new lottery? (y/n): ")
            if confirm.lower() != "y":
                return

        self.lottery = LotteryManager(self.users)
        total_rounds = len(prizes)
        print(f"\nLottery started: {total_rounds} round(s).")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, total_rounds, prize)
        print("\nAll lottery rounds completed.")
        self.show_results(ask_export=False)

    def _describe_rule(self, prize: Prize) -> str:
        rules: list[str] = []
        if _same_text(prize.associated_drawer, LI_BIN):
            rules.append("Relaxed rules: previous winners excluded, avoidance groups ignored")
        else:
            rules.append("Previous winners excluded")
            if prize.avoid_groups:
                rules.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        if self.lottery is not None and self.lottery.is_fallback_for(prize):
            rules.append(f"Cheng Ru group protection: only {CHENG_RU_GROUP} members eligible")
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}")
        return "; ".join(rules)

    def _run_round(self, round_no: int, total_rounds: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {round_no}/{total_rounds}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {self._describe_rule(prize)}")
        print("Commands: Enter = draw next | pause | terminate")

        round_winners: list[User] = []
        while len(round_winners) < prize.quantity:
            command = self._ask("> ").lower()
            if command == "terminate":
                print("Round terminated; remaining winners skipped.")
                break
            if command == "pause":
                if not self._wait_while_paused():
                    print("Round terminated; remaining winners skipped.")
                    break
            elif command not in ("", "continue"):
                print("Unknown command. Use Enter, 'pause' or 'terminate'.")
                continue

            position = len(round_winners) + 1
            print(f"Currently drawing the {_ordinal(position)} winner...")
            time.sleep(DRAW_DELAY_SECONDS)
            user = self.lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize; round ends early.")
                break
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self.lottery.record_round_result(prize, round_winners)

        print(f"\nRound result for {prize.name} (winning order):")
        if not round_winners:
            print("  (no winners)")
        for order, user in enumerate(round_winners, start=1):
            print(f"  {order}. {user.name} ({user.department})")

    @staticmethod
    def _wait_while_paused() -> bool:
        """Block until 'continue' (returns True) or 'terminate' (returns False)."""
        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = input("(paused) > ").strip().lower()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Unknown command. Use 'continue' or 'terminate'.")

    # ------------------------------------------------------------- results
    def show_results(self, ask_export: bool = True) -> None:
        if self.lottery is None or not self.lottery.result_manager.get_all_winners():
            print("No lottery results yet.")
            return
        prizes = sort_prizes_by_level(self.prize_manager.get_prizes())
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)

        print("\n==================== Winner List ====================")
        for prize in prizes:
            print(f"\n[{prize.name}]")
            winners = grouped.get(prize.name, [])
            if not winners:
                print("  (no winners)")
            for order, winner in enumerate(winners, start=1):
                print(f"  {order}. {winner.user.name} ({winner.user.department})")

        if ask_export:
            choice = self._ask("\nExport results to txt (CSV format)? (y/n): ")
            if choice.lower() == "y":
                _, message = self.lottery.result_manager.export_to_csv(prizes)
                print(message)

from __future__ import annotations

import os
import re

from lottery_system import default_data
from lottery_system.models import User


class ListManager:
    """Loads and verifies participant lists."""

    _SEPARATOR_PATTERN = re.compile(r"[,;]")

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        users = [User(name.strip(), dept.strip()) for name, dept in default_data.DEFAULT_USERS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'Name,Department Group' (';' also accepted)."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file into users and a list of error messages."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.readlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        users: list[User] = []
        errors: list[str] = []
        for line_no, raw_line in enumerate(lines, start=1):
            content = raw_line.strip()
            if not content:
                continue
            parts = [part.strip() for part in self._SEPARATOR_PATTERN.split(content)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{content}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Remove duplicate names. Returns (valid_users, duplicates).

        The first occurrence of a name is kept; later occurrences are reported as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people per department, in order of first appearance."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management, avoidance configuration and ordering."""

from __future__ import annotations

from lottery_system import default_data
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in default_data.DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the lottery sequence.

        Raises ValueError for an empty name, duplicate name or invalid quantity.
        """
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = (group or "").strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer = (associated_drawer or "").strip() or None
        prize = Prize(clean_name, quantity, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order.

        Returns True if the order is a valid permutation and was applied.
        """
        count = len(self._prizes)
        if count == 0 or len(order) != count:
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    is_cheng_ru_department,
)


class LotteryManager:
    """Performs draws while enforcing all lottery rules.

    Rules:
    - Previous winners are never drawn again (for any prize).
    - Members of a prize's avoidance groups stay eligible, but are only drawn
      after all other eligible members have been drawn.
    - Lottery person Li Bin: only previous winners are filtered; avoidance groups are ignored.
    - Cheng Ru fallback: if the last two draws (by Chen Qixian / Jin Duo) both had no
      Cheng Ru Group winner, the next draw by Ma Jiaqi is limited to Cheng Ru Group members
      (even if that group is an avoidance group of the prize). The protection is reset after
      a Ma Jiaqi round.
    """

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winners: list[User] = []
        self._winner_keys: set[str] = set()
        self._round_winners: list[User] = []
        self._rng = rng or random.Random()
        # Each entry: whether a round of Chen Qixian / Jin Duo contained a Cheng Ru Group winner
        self.last_two_draw_results: list[bool] = []

    @staticmethod
    def _key(user: User) -> str:
        return user.name.strip().casefold()

    @property
    def winners(self) -> list[User]:
        return list(self._winners)

    def is_winner(self, user: User) -> bool:
        return self._key(user) in self._winner_keys

    def _available_users(self) -> list[User]:
        return [user for user in self._users if not self.is_winner(user)]

    def is_cheng_ru_protection_active(self, prize: Prize) -> bool:
        """True if the Cheng Ru fallback forces this prize's pool."""
        return (
            prize.associated_drawer == MA_JIAQI
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the current candidate pool for the prize, in priority tier."""
        available = self._available_users()
        drawer = prize.associated_drawer

        # Special rule: Li Bin only filters previous winners.
        if drawer == LI_BIN:
            return available

        # Cheng Ru fallback: restrict to Cheng Ru Group members when protection is active.
        if self.is_cheng_ru_protection_active(prize):
            cheng_ru = [user for user in available if is_cheng_ru_department(user.department)]
            if cheng_ru:
                return cheng_ru

        avoid = {group.strip().casefold() for group in prize.avoid_groups}
        primary = [user for user in available if user.department.strip().casefold() not in avoid]
        if primary:
            return primary
        # Only avoidance-group members remain: they are drawn now, after everyone else.
        return [user for user in available if user.department.strip().casefold() in avoid]

    def start_round(self, prize: Prize) -> None:
        """Mark the beginning of a lottery round."""
        self._round_winners = []

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winners.append(user)
        self._winner_keys.add(self._key(user))
        self._round_winners.append(user)
        return user

    def end_round(self, prize: Prize) -> None:
        """Finish a round and update the fallback protection state."""
        drawer = prize.associated_drawer
        if drawer in (CHEN_QIXIAN, JIN_DUO):
            has_cheng_ru = any(is_cheng_ru_department(u.department) for u in self._round_winners)
            self.last_two_draw_results.append(has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == MA_JIAQI:
            self.last_two_draw_results = []
        self._round_winners = []

"""Winner recording, ordering, display grouping and CSV-style export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import date, datetime

from lottery_system.models import Prize, User

_LEVEL_WORDS = (("first", 1), ("second", 2), ("third", 3), ("fourth", 4), ("fifth", 5))
_UNRANKED = 999


def prize_rank(name: str) -> int:
    """Return the prize level (1 = highest). Unknown levels are ranked last."""
    lowered = name.casefold()
    for word, rank in _LEVEL_WORDS:
        if re.search(rf"\b{word}\b", lowered):
            return rank
    return _UNRANKED


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    sequence: int          # global winning order (1-based)
    won_at: datetime       # real winning timestamp


class ResultManager:
    """Stores winners and produces ordered results."""

    EXPORT_HEADER = ["Name", "Department", "Prize Name", "Winning Time"]

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._counter = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for the prize."""
        self._counter += 1
        winner = Winner(user, prize.name, self._counter, datetime.now())
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    @staticmethod
    def _ordered_prizes(prizes: list[Prize]) -> list[Prize]:
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda item: (prize_rank(item[1].name), item[0]))
        return [prize for _, prize in indexed]

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map prize name -> winners, ordered by prize level (high to low).

        Winners within a prize are ordered by winning time.
        """
        result: dict[str, list[Winner]] = {}
        for prize in self._ordered_prizes(prizes):
            winners = [w for w in self._winners if w.prize_name == prize.name]
            result[prize.name] = sorted(winners, key=lambda w: w.sequence)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to awards_result_YYYYMMDD.txt in the current directory."""
        grouped = self.get_winners_by_prize(prizes)
        rows = [winner for winners in grouped.values() for winner in winners]
        if not rows:
            return False, "No winners to export."

        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(self.EXPORT_HEADER)
                for winner in rows:
                    writer.writerow([
                        winner.user.name,
                        winner.user.department,
                        winner.prize_name,
                        winner.won_at.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive command-line application."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str) -> str:
        return input(prompt).strip()

    def _choose_option(self, options: list[str]) -> int:
        """Show numbered options and return the chosen 1-based index."""
        for index, option in enumerate(options, start=1):
            print(f"  {index}. {option}")
        while True:
            choice = self._ask("Select: ")
            if choice.isdigit() and 1 <= int(choice) <= len(options):
                return int(choice)
            print("Invalid option, please try again.")

    # --------------------------------------------------------------- main loop
    def run(self) -> None:
        print("=" * 60)
        print("  Company Team Building - Smart Lottery System")
        print("=" * 60)
        while True:
            print("\n=== Main Menu ===")
            print("1. Load participant list")
            print(f"   (currently loaded: {len(self.users)} participants)")
            print("2. View department groups")
            print("3. Prize configuration")
            print("4. Start lottery")
            print("0. Exit")
            choice = self._ask("Select: ")
            if choice == "1":
                self._load_list_menu()
            elif choice == "2":
                self._view_groups()
            elif choice == "3":
                self._prize_menu()
            elif choice == "4":
                self._start_lottery()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    # ------------------------------------------------------------ list module
    def _load_list_menu(self) -> None:
        print("\n=== Load Participant List ===")
        print("1. Built-in default team building list")
        print("2. Upload custom txt list (format: Name,Department Group)")
        print("0. Back")
        choice = self._ask("Select: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
            for error in errors:
                print(error)
            self.users = users
            print(f"Default list loaded: {len(users)} participants.")
        elif choice == "2":
            self._load_custom_list()

    def _load_custom_list(self) -> None:
        path = self._ask("Enter txt file path (blank to cancel): ")
        if not path:
            return
        users, errors = self.list_manager.load_custom_list(path)
        if errors:
            print("Problems found in the file:")
            for error in errors:
                print(f"  {error}")
            if not users:
                print("No valid data found. Upload aborted.")
                return
            print("Choose how to proceed:")
            if self._choose_option(["Remove invalid entries and continue",
                                    "Terminate upload"]) == 2:
                print("Upload terminated.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print("Duplicate names found (later occurrences):")
            for user in duplicates:
                print(f"  {user.name} ({user.department})")
            print("Choose how to proceed:")
            if self._choose_option(["Remove duplicates (keep first occurrence) and continue",
                                    "Terminate upload"]) == 2:
                print("Upload terminated.")
                return

        if not valid:
            print("No valid participants. Upload aborted.")
            return
        self.users = valid
        print(f"Custom list loaded: {len(valid)} participants.")

    def _view_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("\n=== Department Groups ===")
        for group, count in info.items():
            print(f"  {group}: {count} people")
        print(f"  Total groups: {len(info)}, total people: {len(self.users)}")

    # ----------------------------------------------------------- prize module
    def _prize_menu(self) -> None:

            print("1. Add custom prize")
            print("2. Load preset prize library")
            print("3. View prize list")
            print("4. Adjust lottery order")
            print("0. Back")
            choice = self._ask("Select: ")
            if choice == "1":
                self._add_prize()
            elif choice == "2":
                errors = self.prize_manager.load_default_prizes()
                for error in errors:
                    print(f"Error: {error}")
                print("Preset prizes loaded.")
            elif choice == "3":
                self._view_prizes()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "0":
                return
            else:
                print("Invalid option, please try again.")

    def _add_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity_raw = self._ask("Quantity: ")
        if not quantity_raw.lstrip("-").isdigit():
            print("Error: quantity must be a positive integer.")
            return
        quantity = int(quantity_raw)

        print("Associated lottery person:")
        drawer_options = ["None"] + list(LOTTERY_PERSONS)
        drawer_index = self._choose_option(drawer_options)
        drawer = None if drawer_index == 1 else drawer_options[drawer_index - 1]

        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid_groups: list[str] = []
        if groups:
            print("Department groups:")
            for index, group in enumerate(groups, start=1):
                print(f"  {index}. {group}")
        raw = self._ask("Avoidance groups (numbers or names, comma separated, blank for none): ")
        for token in [t.strip() for t in raw.split(",") if t.strip()]:
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                avoid_groups.append(groups[int(token) - 1])
            else:
                avoid_groups.append(token)

        try:
            self.prize_manager.add_prize(name, quantity, avoid_groups, drawer)
            print(f"Prize '{name.strip()}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def _view_prizes(self) -> None:

        print("\n=== Prize List (lottery order) ===")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
            drawer = prize.associated_drawer or "None"
            print(f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                  f"avoid groups: {avoid} | lottery person: {drawer}")

    def _reorder_prizes(self) -> None:

        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        raw = self._ask("Enter new order as serial numbers (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in raw.replace(",", " ").split()]
        except ValueError:
            print("Error: please enter numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery order updated.")
        else:
            print("Error: order must contain each serial number exactly once.")

    # --------------------------------------------------------- lottery module
    @staticmethod
    def _describe_rules(prize: Prize, lottery: LotteryManager) -> list[str]:
        rules = ["Previous winners are excluded."]
        if prize.associated_drawer == LI_BIN:
            rules.append("Special rule (Li Bin): avoidance groups are NOT applied.")
        elif prize.avoid_groups:
            rules.append("Avoidance groups (" + ", ".join(prize.avoid_groups) + ") are drawn "
                         "only after all other eligible members.")
        if prize.associated_drawer == MA_JIAQI:
            status = " [ACTIVE this round]" if lottery.is_cheng_ru_protection_active(prize) else ""
            rules.append(f"{CHENG_RU_GROUP} protection: if the last two draws by Chen Qixian and "
                         f"Jin Duo had no {CHENG_RU_GROUP} winners, only its members are "
                         f"eligible{status}.")
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}")
        return rules

    def _wait_for_command(self, prompt: str) -> str:
        """Handle pause / continue / terminate. Returns 'continue' or 'terminate'."""
        while True:
            command = self._ask(prompt).lower()
            if command in ("", "continue"):
                return "continue"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("[paused] > ").lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        return "continue"
                    if resume == "terminate":
                        return "terminate"
                    print("Unknown command. Use 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lottery = LotteryManager(self.users)
        self.result_manager = ResultManager()
        total_rounds = len(prizes)

        for round_no, prize in enumerate(prizes, start=1):
            print("\n" + "=" * 60)
            print(f"Round {round_no}/{total_rounds}: {prize.name}")
            print(f"Quantity: {prize.quantity}")
            print("Rules:")
            for rule in self._describe_rules(prize, lottery):
                print(f"  - {rule}")
            self._run_round(lottery, prize)

        print("\n" + "=" * 60)
        self._show_final_summary(prizes)
        if self._ask("Export results to txt file? (y/n): ").lower() == "y":
            _, message = self.result_manager.export_to_csv(prizes)
            print(message)

    def _run_round(self, lottery: LotteryManager, prize: Prize) -> None:
        lottery.start_round(prize)
        round_winners: list[tuple[User, int]] = []

        if self._wait_for_command("Press Enter to start this round, 'terminate' to skip: ") == "terminate":
            print("Round skipped.")
            lottery.end_round(prize)
            return

        for position in range(1, prize.quantity + 1):
            print(f"Currently drawing the {_ordinal(position)} winner...")
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize.")
                break
            winner = self.result_manager.add_winner(user, prize)
            round_winners.append((user, winner.sequence))
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

            if position == prize.quantity:
                break
            if self._wait_for_command(
                "Press Enter to draw next, 'pause' to pause, 'terminate' to end round: "
            ) == "terminate":
                print("Round terminated early.")
                break

        lottery.end_round(prize)
        print(f"\nRound results for {prize.name}:")
        if not round_winners:
            print("  (no winners)")
        for index, (user, sequence) in enumerate(round_winners, start=1):
            print(f"  {index}. {user.name} ({user.department}) [win #{sequence}]")

    def _show_final_summary(self, prizes: list[Prize]) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("=== Final Winner List (by prize level, then winning order) ===")
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(f"  {winner.sequence}. {winner.user.name} ({winner.user.department})")


def main() -> None:
    """Application entry point."""
    try:
        LotterySystemCLI().run()
    except (KeyboardInterrupt, EOFError):
        print("\nExiting lottery system.")

# Prizes without a recognizable level word are placed after all leveled prizes.
UNLEVELED_PRIZE = 99


def normalize_text(text) -> str:
    """Normalize text for case-insensitive, whitespace-insensitive comparison."""
    return " ".join(str(text).split()).casefold()


def prize_level(name: str) -> int:
    """Derive the prize level (1 = highest) from the prize name, e.g. 'First Prize - ...'."""
    for word in re.findall(r"[a-z]+", str(name).casefold()):
        if word in _LEVEL_WORDS:
            return _LEVEL_WORDS[word]
    return UNLEVELED_PRIZE

    department: str

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    @property
    def level(self) -> int:
        return prize_level(self.name)

"""Non-blocking console input handling so commands can be typed during a lottery."""

import queue
import sys
import threading

_EOF_MARKER = "__EOF__"


class InputReader:
    """Reads stdin in a background thread and exposes blocking and polling access."""

    def __init__(self, stream=None):
        self._stream = stream if stream is not None else sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._eof = False
        self._thread = threading.Thread(target=self._pump, daemon=True)
        self._thread.start()

    def _pump(self) -> None:
        try:
            for line in self._stream:
                self._queue.put(line.rstrip("\r\n"))
        except (OSError, ValueError):
            pass
        self._queue.put(_EOF_MARKER)

    def _handle(self, item):
        if item == _EOF_MARKER:
            self._eof = True
            return None
        return item

    @property
    def closed(self) -> bool:
        return self._eof

    def poll(self):
        """Return the next line if available, otherwise None (never blocks)."""
        if self._eof:
            return None
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        return self._handle(item)

    def read_line(self, prompt: str = ""):
        """Block until a line is available. Returns None when input is closed."""
        if prompt:
            print(prompt, end="", flush=True)
        if self._eof:
            return None
        return self._handle(self._queue.get())

"""Participant list loading, validation and group extraction."""

import os

from lottery_system.default_data import DEFAULT_LIST_LINES
from lottery_system.models import User, normalize_text


class ListManager:
    def __init__(self):
        pass

    def load_default_list(self) -> tuple[list[User], list[str]]:
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [part.strip() for part in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department', got '{line}'"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items(), key=lambda item: item[0].casefold()))

"""Prize library management and prize ordering."""

from lottery_system.models import Prize, normalize_text

# (name, quantity, associated lottery person)
_PRESET_PRIZES = (
    ("Third Prize - Movie Tickets", 6, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 4, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 2, "Ma Jiaqi"),
)


class PrizeManager:
    def __init__(self):
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = normalize_text(name)
        for prize in self._prizes:
            if normalize_text(prize.name) == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, drawer in _PRESET_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize already exists, skipped: {name}")
                continue
            self._prizes.append(
                Prize(name=name, quantity=quantity, avoid_groups=[], associated_drawer=drawer)
            )
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._find(name) is not None:
            raise ValueError(f"Prize already exists: {name}")

        groups: list[str] = []
        seen: set[str] = set()
        for group in avoid_groups or []:
            group = str(group).strip()
            key = normalize_text(group)
            if group and key not in seen:
                seen.add(key)
                groups.append(group)

        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        drawer = drawer or None

        self._prizes.append(
            Prize(name=name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        )

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """order[k] is the 1-based current position of the prize placed at new position k+1."""
        count = len(self._prizes)
        try:
            values = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(values) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[value - 1] for value in values]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

import random

from lottery_system.constants import (
    CHEN_QIXIAN,
    CHENG_RU_KEYWORD,
    FALLBACK_CONSECUTIVE_DRAWS,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User, normalize_text

MONITORED_DRAWERS = frozenset({normalize_text(CHEN_QIXIAN), normalize_text(JIN_DUO)})
FALLBACK_DRAWER = normalize_text(MA_JIAQI)
RELAXED_DRAWER = normalize_text(LI_BIN)


def is_cheng_ru_member(user: User) -> bool:
    return CHENG_RU_KEYWORD in normalize_text(user.department)


class LotteryManager:
    def __init__(self, users, rng=None):
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self._rng = rng if rng is not None else random.Random()
        # Results of the most recent draws by Chen Qixian / Jin Duo:
        # True if that draw produced a Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []

    def is_winner(self, user: User) -> bool:
        return normalize_text(user.name) in self._winner_keys

    def _fallback_active(self) -> bool:
        return (
            len(self.last_two_draw_results) == FALLBACK_CONSECUTIVE_DRAWS
            and not any(self.last_two_draw_results)
        )

    def is_fallback_active_for(self, prize: Prize) -> bool:
        drawer = normalize_text(prize.associated_drawer or "")
        return drawer == FALLBACK_DRAWER and self._fallback_active()

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = [u for u in self._users if not self.is_winner(u)]
        drawer = normalize_text(prize.associated_drawer or "")

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group members,
        # even if that group is an avoidance group of the prize.
        if drawer == FALLBACK_DRAWER and self._fallback_active():
            return [u for u in available if is_cheng_ru_member(u)]

        # Special rule: Li Bin's draws use relaxed filtering (avoidance groups ignored).
        if drawer == RELAXED_DRAWER:
            return available

        avoid = {normalize_text(group) for group in prize.avoid_groups}
        return [u for u in available if normalize_text(u.department) not in avoid]

    def draw_next(self, prize: Prize) -> User | None:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_keys.add(normalize_text(winner.name))
        return winner

    def end_round(self, prize: Prize, round_winners) -> None:
        drawer = normalize_text(prize.associated_drawer or "")
        if drawer == FALLBACK_DRAWER:
            # The fallback check has been consumed by this round.
            self.last_two_draw_results.clear()
        elif drawer in MONITORED_DRAWERS:
            hit = any(is_cheng_ru_member(w) for w in round_winners)
            self.last_two_draw_results.append(hit)
            del self.last_two_draw_results[:-FALLBACK_CONSECUTIVE_DRAWS]

"""Winner recording, grouping and CSV-style export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.constants import RESULT_DATE_FORMAT, RESULT_FILE_PREFIX
from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize_name: str
    level: int
    winning_order: int
    win_time: datetime


class ResultManager:
    def __init__(self):
        self._winners: list[Winner] = []
        self._counter = 0

    def add_winner(self, user, prize):
        self._counter += 1
        winner = Winner(
            user=user,
            prize_name=prize.name,
            level=prize.level,
            winning_order=self._counter,
            win_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def _ordered_winners(self, prizes=None) -> list[Winner]:
        winners = self._winners
        if prizes is not None:
            names = {p.name for p in prizes}
            winners = [w for w in winners if w.prize_name in names]
        return sorted(winners, key=lambda w: (w.level, w.winning_order))

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        ordered_prizes = sorted(prizes, key=lambda p: p.level)
        result: dict[str, list[Winner]] = {}
        for prize in ordered_prizes:
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        date_text = datetime.now().strftime(RESULT_DATE_FORMAT)
        path = os.path.join(os.getcwd(), f"{RESULT_FILE_PREFIX}{date_text}.txt")
        header = ["Prize Level", "Prize Name", "Winning Order", "Name", "Department", "Win Time"]
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(header)
                for winner in self._ordered_winners(prizes):
                    writer.writerow([
                        winner.level,
                        winner.prize_name,
                        winner.winning_order,
                        winner.user.name,
                        winner.user.department,
                        winner.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class _RoundState:

class LotteryApp:

                return
            self.users = valid
            removed = len(errors) + len(duplicates)

            self._view_groups()

        if groups:

            selected: list[str] = []

                else:
                    match = next(
                        (g for g in groups if normalize_text(g) == normalize_text(token)), None
                    )
                    if match is None:
                        if not groups:
                            match = token
                        else:
                            print(f"Invalid group: '{token}'. Please try again.")
                            valid = False
                            break
                if match not in selected:

            if valid:

    # ------------------------------------------------------- lottery process
    @staticmethod
    def _describe_rules(prize, manager: LotteryManager) -> str:
        rules = ["exclude previous winners"]
        drawer = normalize_text(prize.associated_drawer or "")
        if drawer == normalize_text(LI_BIN):
            rules.append("relaxed rules for Li Bin (avoidance groups not applied)")
        elif prize.avoid_groups:
            rules.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if drawer == normalize_text(MA_JIAQI):
            if manager.is_fallback_active_for(prize):
                rules.append(
                    f"Cheng Ru group protection active: pool limited to {CHENG_RU_GROUP} members"
                )
            else:
                rules.append(
                    f"Cheng Ru group protection applies if the last two draws by "
                    f"{CHEN_QIXIAN} and {JIN_DUO} had no {CHENG_RU_GROUP} winner"
                )
        return "; ".join(rules)

    def _handle_command(self, command, state: _RoundState) -> None:
        if command is None:
            return
        cmd = command.strip().lower()
        if not cmd:
            return
        if cmd == "pause":
            if state.terminated:
                return
            state.paused = True
            print("Lottery paused. Type 'continue' to resume or 'terminate' to skip this round.")
        elif cmd == "continue":
            if state.paused:
                state.paused = False
                print("Lottery resumed.")
            else:
                print("Lottery is not paused.")
        elif cmd == "terminate":
            state.terminated = True
            state.paused = False
            print("Terminating current round...")
        else:
            print("Unknown command. Available commands: pause, continue, terminate")

    def _drain_commands(self, state: _RoundState) -> None:
        while True:
            command = self.reader.poll()
            if command is None:
                break
            self._handle_command(command, state)

    def _wait_while_paused(self, state: _RoundState) -> None:
        while True:
            self._drain_commands(state)
            if state.terminated or not state.paused:
                return
            command = self.reader.read_line("(paused) > ")
            if command is None:
                # Input closed while paused: resume so the lottery can complete.
                state.paused = False
                return
            self._handle_command(command, state)

    def _interval(self, state: _RoundState) -> None:
        end = time.monotonic() + self.draw_delay
        while True:
            self._drain_commands(state)
            if state.terminated or state.paused:
                return
            remaining = end - time.monotonic()
            if remaining <= 0:
                return
            time.sleep(min(0.1, remaining))

    def _run_round(self, round_no: int, prize, manager: LotteryManager) -> list:
        print(f"\n=== Round {round_no}: {prize.name} ===")
        print(f"Quantity: {prize.quantity} | Lottery person: {prize.associated_drawer or '-'}")
        print(f"Rule: {self._describe_rules(prize, manager)}")

        state = _RoundState()
        round_winners = []
        for order in range(1, prize.quantity + 1):
            self._wait_while_paused(state)
            if state.terminated:
                print("Current round terminated; skipping remaining winners.")
                break
            print(f"Currently drawing the {ordinal(order)} winner...")
            user = manager.draw_next(prize)
            if user is None:
                print("No eligible participants remain for this prize.")
                break
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            if order < prize.quantity:
                self._interval(state)

        manager.end_round(prize, round_winners)
        print(f"Round {round_no} finished: {len(round_winners)}/{prize.quantity} winners drawn.")
        if round_winners:
            print("Round results in winning order:")
            for index, user in enumerate(round_winners, start=1):
                print(f"  {index}. {user.name} ({user.department})")
        return round_winners

    def _start_lottery(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Please load or add prizes first.")
            return
        if not self.users:
            print("No participants loaded. Please load a participant list first.")
            return

        print("\nLottery starts! Commands: pause / continue / terminate")
        result = ResultManager()
        manager = LotteryManager(self.users)
        for round_no, prize in enumerate(prizes, start=1):
            if manager.is_fallback_active_for(prize):
                print(f"Notice: Cheng Ru group protection is active for {prize.name}.")
            winners = self._run_round(round_no, prize, manager)
            for user in winners:
                result.add_winner(user, prize)

        self.last_result = result
        self.last_prizes = prizes
        print("\nAll rounds completed.")
        self._print_result_summary(result, prizes)

    # ------------------------------------------------------------ results
    def _print_result_summary(self, result: ResultManager, prizes) -> None:
        print("\n=== Winner List (by prize level, then winning order) ===")
        grouped = result.get_winners_by_prize(prizes)
        for prize_name, winners in grouped.items():
            print(f"\n[{prize_name}]")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  {winner.winning_order}. {winner.user.name} ({winner.user.department}) "
                    f"- {winner.prize_name}"
                )

    def _view_results(self) -> None:
        if self.last_result is None:
            print("No lottery results yet.")
            return
        self._print_result_summary(self.last_result, self.last_prizes)

    def _export_results(self) -> None:
        if self.last_result is None:
            print("No lottery results to export. Please run the lottery first.")
            return
        success, message = self.last_result.export_to_csv(self.last_prizes)
        print(message)


def main() -> None:
    app = LotteryApp()
    app.run()

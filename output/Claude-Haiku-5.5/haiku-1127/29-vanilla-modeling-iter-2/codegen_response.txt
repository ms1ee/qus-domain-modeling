"""Configuration loading and JSON state persistence.

System parameters can be customised through a JSON file. Simulation state
(system parameters, rider information and order history) can be saved to a
JSON file at any time.
"""

import json
import logging
import os
from typing import Any, Dict

logger = logging.getLogger(__name__)

DEFAULT_CONFIG: Dict[str, int] = {
    "initial_funds": 1000,
    "recruit_cost": 300,
    "recruit_threshold": 300,
    "rider_speed": 10,
}


def load_config(path: str) -> Dict[str, int]:
    """Load system parameters from a JSON file.

    The file may either be a flat object of parameter names and integer
    values, or contain such an object under the ``system_parameters`` key.
    Unknown keys are ignored. On any error the defaults are returned.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        A dictionary with every known system parameter.
    """
    settings: Dict[str, int] = dict(DEFAULT_CONFIG)
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.error("Could not read configuration %s: %s", path, exc)
        return settings

    if not isinstance(data, dict):
        logger.error("Configuration %s must contain a JSON object", path)
        return settings

    params = data.get("system_parameters", data)
    if not isinstance(params, dict):
        logger.error("'system_parameters' in %s must be an object", path)
        return settings

    for key in DEFAULT_CONFIG:
        if key in params:
            try:
                settings[key] = int(params[key])
            except (TypeError, ValueError):
                logger.warning("Invalid value for %s in %s; using default", key, path)
    return settings


def save_state(path: str, system: Any, order_manager: Any) -> None:
    """Persist the current simulation state to a JSON file.

    The file is written atomically through a temporary file.

    Args:
        path: Destination file path.
        system: ``SystemManager`` instance providing ``snapshot()``.
        order_manager: ``OrderManager`` instance providing ``to_dicts()``.
    """
    payload = {
        "system_parameters": {
            "initial_funds": system.initial_funds,
            "recruit_cost": system.recruit_cost,
            "recruit_threshold": system.recruit_threshold,
            "rider_speed": system.rider_speed,
        },
        "system_state": system.snapshot(),
        "order_history": order_manager.to_dicts(),
    }
    tmp_path = path + ".tmp"
    with open(tmp_path, "w", encoding="utf-8") as handle:
        json.dump(payload, handle, indent=2)
    os.replace(tmp_path, path)
    logger.info("State saved to %s", path)

"""Grid-based path planning for the 800x800 delivery map.

The map is an 8x8 grid of 100x100 cells. Points whose x or y coordinate is a
multiple of 100 lie on streets. Routes are orthogonal (horizontal and vertical
segments only) and prefer corners that lie on a street. Straight movement is
used whenever the start and goal share an x or y coordinate.

Riders move independently and never block each other, so the planner cannot
create circular waits; this is the mechanism that avoids deadlocks.
"""

import math
from typing import List, Sequence, Tuple

Point = Tuple[int, int]

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = 8
STREET_SPACING = 100


def clamp_coordinate(value: int) -> int:
    """Clamp a coordinate into the valid map range 0-800.

    Args:
        value: Raw coordinate.

    Returns:
        The integer coordinate limited to ``[0, MAP_SIZE]``.
    """
    return max(0, min(MAP_SIZE, int(value)))


def cell_index(value: int) -> int:
    """Return the grid index (0-7) of the cell containing a coordinate.

    The coordinate 800 (the map edge) belongs to the last cell.

    Args:
        value: Coordinate in map units.

    Returns:
        Grid index along one axis.
    """
    value = clamp_coordinate(value)
    return min(value // CELL_SIZE, GRID_DIM - 1)


def cell_center(value: int) -> int:
    """Convert a coordinate to the centre of its grid cell.

    Implements ``center = (grid // 100) * 100 + 50``, with the map edge
    clamped into the last cell.

    Args:
        value: Coordinate in map units.

    Returns:
        Centre coordinate of the containing cell.
    """
    return cell_index(value) * CELL_SIZE + CELL_SIZE // 2


def is_street_point(point: Point) -> bool:
    """Return True if the point lies on a street grid line.

    Args:
        point: ``(x, y)`` coordinate.

    Returns:
        True when x or y is a multiple of 100.
    """
    x, y = point
    return x % STREET_SPACING == 0 or y % STREET_SPACING == 0


def euclidean(a: Point, b: Point) -> float:
    """Return the Euclidean distance between two points.

    Args:
        a: First point.
        b: Second point.

    Returns:
        ``sqrt((ax - bx)^2 + (ay - by)^2)``.
    """
    return math.hypot(a[0] - b[0], a[1] - b[1])


def manhattan(a: Point, b: Point) -> int:
    """Return the Manhattan (street) distance between two points.

    Args:
        a: First point.
        b: Second point.

    Returns:
        ``|ax - bx| + |ay - by|``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def _sign(value: int) -> int:
    """Return -1, 0 or 1 according to the sign of ``value``."""
    return (value > 0) - (value < 0)


class PathPlanner:
    """Plans orthogonal, street-preferring routes and advances riders along them.

    Attributes:
        algorithm_name: Human-readable name of the algorithm.
        map_size: Size of the square map in units.
    """

    algorithm_name = "street-first orthogonal routing"

    def __init__(self, map_size: int = MAP_SIZE) -> None:
        """Create a planner for a square map.

        Args:
            map_size: Width and height of the map in units.
        """
        self.map_size = map_size

    def plan(self, start: Point, goal: Point) -> List[Point]:
        """Plan a route from ``start`` to ``goal``.

        Straight routes are used when start and goal share a coordinate
        (directly above/below or beside each other). Otherwise a single
        right-angle corner is chosen, preferring the corner on a street.

        Args:
            start: Current position.
            goal: Destination.

        Returns:
            Waypoints to visit in order, excluding ``start`` and including
            ``goal``.
        """
        sx, sy = start
        gx, gy = goal
        if sx == gx or sy == gy:
            return [(gx, gy)]
        corner_horizontal_first = (gx, sy)
        corner_vertical_first = (sx, gy)
        if (is_street_point(corner_vertical_first)
                and not is_street_point(corner_horizontal_first)):
            corner = corner_vertical_first
        else:
            corner = corner_horizontal_first
        return [corner, (gx, gy)]

    def plan_route(self, start: Point, pickup: Point, dropoff: Point) -> List[Point]:
        """Plan a continuous route: start -> restaurant -> customer.

        Args:
            start: Rider position.
            pickup: Restaurant location.
            dropoff: Customer location.

        Returns:
            Concatenated waypoints for the whole trip.
        """
        return self.plan(start, pickup) + self.plan(pickup, dropoff)

    def path_length(self, start: Point, waypoints: Sequence[Point]) -> int:
        """Return the total travelled length along waypoints.

        Args:
            start: Starting position.
            waypoints: Waypoints returned by :meth:`plan`.

        Returns:
            Sum of segment lengths in map units.
        """
        total = 0
        current = start
        for point in waypoints:
            total += manhattan(current, point)
            current = point
        return total

    def advance(self, x: int, y: int, waypoints: Sequence[Point],
                speed: int) -> Tuple[int, int, int]:
        """Move up to ``speed`` units along the waypoints.

        Each segment is axis-aligned. When the current y equals the target y
        the rider moves horizontally, otherwise vertically.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            waypoints: Route to follow.
            speed: Movement budget for this tick.

        Returns:
            Tuple ``(new_x, new_y, distance_used)``.
        """
        pos_x, pos_y = x, y
        budget = max(0, int(speed))
        used = 0
        for target_x, target_y in waypoints:
            while budget > 0 and (pos_x, pos_y) != (target_x, target_y):
                if pos_y == target_y:
                    step = min(budget, abs(target_x - pos_x))
                    pos_x += _sign(target_x - pos_x) * step
                else:
                    step = min(budget, abs(target_y - pos_y))
                    pos_y += _sign(target_y - pos_y) * step
                budget -= step
                used += step
            if budget <= 0:
                break
        return pos_x, pos_y, used

"""Rider entity: state machine, movement and event logging.

State machine:
    IDLE (0) -> PICKING_UP (1) when an order is accepted;
    PICKING_UP (1) -> DELIVERING (2) on arrival at the restaurant;
    DELIVERING (2) -> IDLE (0) (or PICKING_UP for a queued order) on delivery.
"""

from typing import Any, Dict, List, Optional, Set, Tuple

import globalvar
from path_planner import PathPlanner

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}


class Rider:
    """A delivery rider that moves on the grid and serves assigned orders.

    Attributes:
        rider_id: Unique identifier.
        x: Current x coordinate.
        y: Current y coordinate.
        speed: Distance covered per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Assigned orders, served in order.
        special_case: True while the rider is directly above or below its target.
        log: Status change records; each entry has "event" and "order_id" keys.
        planner: Path planner used for movement.
        distance_travelled: Total distance moved.
    """

    SPAWN_X = 435
    SPAWN_Y = 385
    DEFAULT_SPEED = 10

    def __init__(self, rider_id: int, speed: int = DEFAULT_SPEED) -> None:
        """Create a rider at the spawn point in the idle state.

        Args:
            rider_id: Unique identifier.
            speed: Movement speed in units per tick.
        """
        self.rider_id = int(rider_id)
        self.x = self.SPAWN_X
        self.y = self.SPAWN_Y
        self.speed = int(speed)
        self.state = IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.log: List[Dict[str, Any]] = []
        self.planner = PathPlanner()
        self.distance_travelled = 0.0
        self._accepted_ids: Set[int] = set()
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # Queries
    # ------------------------------------------------------------------
    @property
    def position(self) -> Tuple[int, int]:
        """Return the current ``(x, y)`` position."""
        return self.x, self.y

    @property
    def state_name(self) -> str:
        """Return a readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def current_target(self) -> Optional[Tuple[int, int]]:
        """Return the point the rider is heading to, or None if idle.

        Returns:
            Restaurant position while picking up, customer position while
            delivering, otherwise None.
        """
        if not self.orders:
            return None
        if self.state == PICKING_UP:
            return self.orders[0].res_pos
        if self.state == DELIVERING:
            return self.orders[0].cus_pos
        return None

    # ------------------------------------------------------------------
    # Order handling
    # ------------------------------------------------------------------
    def add_order(self, order: Any) -> None:
        """Assign an order to this rider.

        Appends the order to ``orders``, logs an "accepted_order" event and
        moves an idle rider into the picking-up state.

        Args:
            order: ``Order`` instance to assign.
        """
        if order not in self.orders:
            self.orders.append(order)
        self._sync_orders()

    def _sync_orders(self) -> None:
        """Synchronise state with the order list.

        Logs "accepted_order" for any order not yet accepted, starts pickup
        for idle riders with orders, and forces idle when no orders remain.
        """
        for order in self.orders:
            if order.id not in self._accepted_ids:
                self._accepted_ids.add(order.id)
                self._record("accepted_order", order.id)
        if self.orders and self.state == IDLE:
            self.state = PICKING_UP
        elif not self.orders and self.state != IDLE:
            self.state = IDLE
            self.special_case = False

    def _complete_order(self, order: Any) -> None:
        """Finish delivery of ``order`` and release it from the rider.

        Args:
            order: The order that has just been delivered.
        """
        tick = globalvar.get_value("current_tick", 0)
        order.complete(tick)
        self.orders.remove(order)
        self._accepted_ids.discard(order.id)
        self._record("delivered_to_customer", order.id)
        self.special_case = False
        self.state = PICKING_UP if self.orders else IDLE

    # ------------------------------------------------------------------
    # Movement
    # ------------------------------------------------------------------
    def move(self) -> None:
        """Advance the rider by one tick.

        Moves towards the current target, handles arrival at the restaurant
        (state 1 -> 2) and at the customer (order completion).
        """
        self._sync_orders()
        if not self.orders:
            return

        order = self.orders[0]
        target = self.current_target()
        if target is None:
            return

        if self.position != target:
            self.special_case = self.x == target[0]
            waypoints = self.planner.plan(self.position, target)
            self.x, self.y, used = self.planner.advance(
                self.x, self.y, waypoints, self.speed)
            self.distance_travelled += used

        if self.position == target:
            self.special_case = False
            if self.state == PICKING_UP:
                self.state = DELIVERING
                self._record("arrived_at_merchant", order.id)
            elif self.state == DELIVERING:
                self._complete_order(order)

    # ------------------------------------------------------------------
    # Logging and serialisation
    # ------------------------------------------------------------------
    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append a status record to the rider log.

        Args:
            event: Event name such as "spawn" or "delivered_to_customer".
            order_id: Related order id, or None.
        """
        self.log.append({
            "tick": globalvar.get_value("current_tick", 0),
            "rider_id": self.rider_id,
            "event": event,
            "order_id": order_id,
            "state": self.state,
            "x": self.x,
            "y": self.y,
        })

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider.

        Returns:
            Dictionary describing the rider and its order ids.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
            "distance_travelled": self.distance_travelled,
        }

    def __repr__(self) -> str:
        return (f"Rider(id={self.rider_id}, pos=({self.x}, {self.y}), "
                f"state={self.state_name}, orders={len(self.orders)})")

"""Order data model and coordinate validation."""

import itertools
from typing import Any, Dict, List, Optional, Sequence

import globalvar
from path_planner import MAP_SIZE, cell_center, euclidean

_id_source = itertools.count(1)


def validate_order_coords(values: Sequence[Any]) -> List[int]:
    """Validate the four order coordinates.

    Args:
        values: Sequence ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

    Returns:
        The validated list of four integers.

    Raises:
        ValueError: If the count is wrong, a value is not an integer, or a
            value lies outside 0-800.
    """
    if len(values) != 4:
        raise ValueError("an order needs exactly four coordinates")
    result: List[int] = []
    for value in values:
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"coordinate {value!r} must be an integer")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"coordinate {value} is out of range 0-{MAP_SIZE}")
        result.append(value)
    return result


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order id.
        res_pos: Restaurant grid-cell centre ``(x, y)``.
        cus_pos: Customer grid-cell centre ``(x, y)``.
        status: "pending", "assigned" or "completed".
        rider_id: Id of the assigned rider, or None.
        start_time: Tick at which the order was created.
        completion_time: Tick at which delivery finished, or None.
        distance: Straight-line distance between restaurant and customer.
        time_taken: Ticks from creation to completion, or None.
        efficiency: Distance per tick after completion, or None.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order from raw input coordinates.

        Args:
            res_pos: Restaurant ``(x, y)`` in 0-800.
            cus_pos: Customer ``(x, y)`` in 0-800.

        Raises:
            ValueError: If a coordinate is invalid.
        """
        rx, ry, cx, cy = validate_order_coords(list(res_pos) + list(cus_pos))
        self.id = next(_id_source)
        self.res_pos = (cell_center(rx), cell_center(ry))
        self.cus_pos = (cell_center(cx), cell_center(cy))
        self.status = "pending"
        self.rider_id: Optional[int] = None
        self.start_time = int(globalvar.get_value("current_tick", 0))
        self.completion_time: Optional[int] = None
        self.distance = float(euclidean(self.res_pos, self.cus_pos))
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None

    def assign_to(self, rider_id: int) -> None:
        """Record the rider that serves this order.

        Args:
            rider_id: Id of the rider.
        """
        self.rider_id = int(rider_id)
        self.status = "assigned"

    def complete(self, tick: int) -> None:
        """Mark the order as delivered and compute its statistics.

        Args:
            tick: Tick at which delivery finished.
        """
        self.status = "completed"
        self.completion_time = int(tick)
        self.time_taken = max(0, self.completion_time - self.start_time)
        self.efficiency = (self.distance / self.time_taken
                           if self.time_taken > 0 else None)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "status": self.status,
            "rider_id": self.rider_id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": round(self.distance, 2),
            "time_taken": self.time_taken,
            "efficiency": (round(self.efficiency, 3)
                           if self.efficiency is not None else None),
        }

    def __repr__(self) -> str:
        return (f"Order(id={self.id}, res={self.res_pos}, cus={self.cus_pos}, "
                f"status={self.status})")

"""Order management and the closest-rider assignment algorithm."""

import logging
from typing import TYPE_CHECKING, Any, Dict, List, Optional, Sequence

import globalvar
from order_model import Order, validate_order_coords
from path_planner import euclidean

if TYPE_CHECKING:  # pragma: no cover - only for type checkers
    from rider import Rider

logger = logging.getLogger(__name__)

# Extra distance-equivalent penalty per order already queued on a rider.
# It spreads load across riders while keeping distance as the main criterion.
LOAD_PENALTY = 100.0


class OrderManager:
    """Creates, assigns and tracks orders.

    Attributes:
        pending: Orders not yet assigned to a rider.
        orders: Every order created in this session (history).
    """

    def __init__(self) -> None:
        """Create an empty order manager."""
        self.pending: List[Order] = []
        self.orders: List[Order] = []

    def create_order(self, coords: Sequence[Any]) -> Order:
        """Create an order from ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Args:
            coords: Four integer coordinates in 0-800.

        Returns:
            The newly created pending order.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        values = validate_order_coords(list(coords))
        order = Order((values[0], values[1]), (values[2], values[3]))
        self.pending.append(order)
        self.orders.append(order)
        logger.info("Created order %d", order.id)
        return order

    def pending_count(self) -> int:
        """Return the number of unassigned orders."""
        return len(self.pending)

    def assign(self, order: Order, rider: "Rider") -> None:
        """Assign ``order`` to ``rider``.

        Args:
            order: Pending order.
            rider: Rider that will serve the order.

        Raises:
            ValueError: If the order is already completed.
        """
        if order.status == "completed":
            raise ValueError(f"order {order.id} is already completed")
        rider.add_order(order)
        order.assign_to(rider.rider_id)
        if order in self.pending:
            self.pending.remove(order)
        logger.info("Assigned order %d to rider %d", order.id, rider.rider_id)

    def active_orders(self) -> List[Order]:
        """Return orders that have not been completed yet."""
        return [order for order in self.orders if order.status != "completed"]

    def completed_orders(self) -> List[Order]:
        """Return orders that have been delivered."""
        return [order for order in self.orders if order.status == "completed"]

    def to_dicts(self) -> List[Dict[str, Any]]:
        """Return all orders as JSON-serialisable dictionaries."""
        return [order.to_dict() for order in self.orders]


def find_best_rider_for_order(order: Order) -> Optional["Rider"]:
    """Select the best rider for an order.

    The score is the Euclidean distance from the rider to the restaurant,
    ``sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2)``, plus a
    load penalty per already-assigned order. Ties keep the earlier rider.

    Args:
        order: Order that needs a rider.

    Returns:
        The selected rider, or None if no rider exists.
    """
    riders = globalvar.get_value("riders", []) or []
    best: Optional["Rider"] = None
    best_score: Optional[float] = None
    for rider in riders:
        distance = euclidean((rider.x, rider.y), order.res_pos)
        score = distance + LOAD_PENALTY * len(rider.orders)
        if best_score is None or score < best_score:
            best = rider
            best_score = score
    return best

"""System manager: funds, rider pool, recruitment and simulation clock."""

import logging
from typing import Any, Dict, List, Optional

import globalvar
from config import DEFAULT_CONFIG
from path_planner import MAP_SIZE
from rider import DELIVERING, IDLE, PICKING_UP, Rider

logger = logging.getLogger(__name__)


class SystemManager:
    """Owns the system funds, the rider pool and the simulation clock.

    Attributes:
        initial_funds: Starting funds.
        recruit_cost: Cost deducted for each recruited rider.
        recruit_threshold: Minimum funds needed to recruit.
        rider_speed: Speed given to newly recruited riders.
        funds: Current funds (may be negative).
        rider_id_counter: Last issued rider id.
        is_running: Controls the main loop.
        current_tick: Simulation clock.
    """

    def __init__(self, config: Optional[Dict[str, Any]] = None) -> None:
        """Create the system manager with the given or default parameters.

        Args:
            config: Optional overrides for the keys in ``DEFAULT_CONFIG``.
        """
        settings = dict(DEFAULT_CONFIG)
        settings.update(config or {})
        self.initial_funds = int(settings["initial_funds"])
        self.recruit_cost = int(settings["recruit_cost"])
        self.recruit_threshold = int(settings["recruit_threshold"])
        self.rider_speed = int(settings["rider_speed"])
        self.funds = self.initial_funds
        self.rider_id_counter = 0
        self.is_running = True
        self.current_tick = 0
        globalvar.set_value("current_tick", 0)

    @property
    def riders(self) -> List[Rider]:
        """Return the live rider list stored in the global variables."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    def idle_riders(self) -> List[Rider]:
        """Return riders in the idle state."""
        return [rider for rider in self.riders if rider.state == IDLE]

    def create_rider(self) -> Rider:
        """Recruit a new rider, deducting the recruitment cost.

        Returns:
            The newly created rider.
        """
        self.rider_id_counter += 1
        rider = Rider(self.rider_id_counter, speed=self.rider_speed)
        self.funds -= self.recruit_cost
        self.riders.append(rider)
        logger.info("Recruited rider %d (funds now %d)", rider.rider_id, self.funds)
        return rider

    def maybe_recruit(self, pending_count: int) -> List[Rider]:
        """Recruit riders while funds allow and pending orders outnumber idle riders.

        Args:
            pending_count: Number of pending (unassigned) orders.

        Returns:
            List of riders created by this call.
        """
        created: List[Rider] = []
        while (self.funds >= self.recruit_threshold
               and pending_count > len(self.idle_riders())):
            created.append(self.create_rider())
        return created

    def remove_rider(self, rider_id: int) -> bool:
        """Remove an idle rider from the pool.

        Args:
            rider_id: Id of the rider to remove.

        Returns:
            True if the rider was removed, False otherwise.
        """
        for rider in self.riders:
            if rider.rider_id == rider_id:
                if rider.state != IDLE or rider.orders:
                    return False
                self.riders.remove(rider)
                return True
        return False

    def tick(self) -> None:
        """Advance the simulation by one unit of time.

        Increments the clock and moves every rider. Exceptions raised by a
        single rider are logged and that rider's state is repaired.
        """
        self.current_tick += 1
        globalvar.set_value("current_tick", self.current_tick)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # noqa: BLE001 - keep the simulation running
                logger.exception("Rider %d failed during tick", rider.rider_id)
                self._repair_rider(rider)

    @staticmethod
    def _repair_rider(rider: Rider) -> bool:
        """Bring a rider's state in line with its order list.

        Args:
            rider: Rider to repair.

        Returns:
            True if a change was made.
        """
        changed = False
        if rider.orders and rider.state == IDLE:
            rider.state = PICKING_UP
            changed = True
        elif not rider.orders and rider.state != IDLE:
            rider.state = IDLE
            rider.special_case = False
            changed = True
        if rider.state not in (IDLE, PICKING_UP, DELIVERING):
            rider.state = IDLE if not rider.orders else PICKING_UP
            changed = True
        return changed

    def repair_state(self) -> int:
        """Repair inconsistent rider states across the whole pool.

        Returns:
            Number of riders whose state was corrected.
        """
        return sum(1 for rider in self.riders if self._repair_rider(rider))

    def check_consistency(self) -> List[str]:
        """Check the system state and return a list of problems found.

        Returns:
            Human-readable issue descriptions; empty when consistent.
        """
        issues: List[str] = []
        for rider in self.riders:
            if rider.state == IDLE and rider.orders:
                issues.append(f"Rider #{rider.rider_id} is idle but has orders")
            if rider.state != IDLE and not rider.orders:
                issues.append(f"Rider #{rider.rider_id} is busy without orders")
            if rider.speed <= 0:
                issues.append(f"Rider #{rider.rider_id} has invalid speed")
            if not (0 <= rider.x <= MAP_SIZE and 0 <= rider.y <= MAP_SIZE):
                issues.append(f"Rider #{rider.rider_id} is outside the map")
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    issues.append(
                        f"Order #{order.id} is held by rider #{rider.rider_id} "
                        f"but recorded for rider #{order.rider_id}")
        return issues

    def snapshot(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the system state."""
        return {
            "funds": self.funds,
            "current_tick": self.current_tick,
            "rider_id_counter": self.rider_id_counter,
            "riders": [rider.to_dict() for rider in self.riders],
        }

    def shutdown(self) -> None:
        """Stop the main loop."""
        self.is_running = False


def init_system(config: Optional[Dict[str, Any]] = None) -> SystemManager:
    """Initialise global variables and create the system manager.

    Args:
        config: Optional parameter overrides.

    Returns:
        The initialised ``SystemManager``.
    """
    globalvar._init()
    system = SystemManager(config)
    globalvar.set_value("system", system)
    return system

"""Status text, order and rider reports, and ASCII map rendering."""

from typing import Any, List, Optional, Sequence, Tuple

from path_planner import MAP_SIZE
from rider import STATE_NAMES

Point = Tuple[int, int]

# One map character per 50 units. Grid lines (multiples of 100) fall on even
# indexes; cell centres (50, 150, ...) fall on odd indexes.
STEP = 50

LEGEND = "S=Restaurant  C=Customer  R=Rider  #=Street  .=Empty  *=Route"


class DisplayManager:
    """Builds textual reports and the ASCII map for the command line.

    Attributes:
        system: The ``SystemManager`` being displayed.
        order_manager: The ``OrderManager`` providing order data (optional).
        map_size: Size of the map in units.
    """

    def __init__(self, system: Any, order_manager: Any = None,
                 map_size: int = MAP_SIZE) -> None:
        """Create a display manager.

        Args:
            system: System manager instance.
            order_manager: Order manager instance, or None.
            map_size: Size of the square map in units.
        """
        self.system = system
        self.order_manager = order_manager
        self.map_size = map_size
        self.dim = map_size // STEP + 1

    # ------------------------------------------------------------------
    # Map rendering
    # ------------------------------------------------------------------
    def _index(self, value: int) -> int:
        """Convert a coordinate to a map character index."""
        return min(max(int(value), 0) // STEP, self.dim - 1)

    def _blank_grid(self) -> List[List[str]]:
        """Return an empty canvas with street lines drawn."""
        return [["#" if (row % 2 == 0 or col % 2 == 0) else "."
                 for col in range(self.dim)]
                for row in range(self.dim)]

    def _plot(self, grid: List[List[str]], point: Point, symbol: str) -> None:
        """Place a symbol at the map position of ``point``."""
        x, y = point
        grid[self._index(y)][self._index(x)] = symbol

    def _draw_route(self, grid: List[List[str]], points: Sequence[Point],
                    start: Point) -> None:
        """Draw connected route segments with '*' characters.

        Args:
            grid: Canvas to draw on.
            points: Route waypoints in order.
            start: Route starting point.
        """
        prev = start
        for point in points:
            px, py = point
            qx, qy = prev
            if px == qx:
                col = self._index(px)
                lo, hi = sorted((self._index(qy), self._index(py)))
                for row in range(lo, hi + 1):
                    grid[row][col] = "*"
            elif py == qy:
                row = self._index(py)
                lo, hi = sorted((self._index(qx), self._index(px)))
                for col in range(lo, hi + 1):
                    grid[row][col] = "*"
            prev = point

    def render(self, route: Optional[Sequence[Point]] = None,
               route_start: Optional[Point] = None) -> str:
        """Render the ASCII map.

        Args:
            route: Optional waypoints to draw as a connected line.
            route_start: Starting point of the route (defaults to the first
                riders' position when omitted).

        Returns:
            Multi-line string containing the map and the legend.
        """
        grid = self._blank_grid()
        if route:
            start = route_start if route_start is not None else route[0]
            self._draw_route(grid, route, start)
        if self.order_manager is not None:
            for order in self.order_manager.active_orders():
                self._plot(grid, order.res_pos, "S")
                self._plot(grid, order.cus_pos, "C")
        for rider in self.system.riders:
            self._plot(grid, (rider.x, rider.y), "R")

        lines = ["y\\x"]
        for row_index, row in enumerate(grid):
            lines.append(f"{row_index * STEP:>4} " + " ".join(row))
        lines.append(f"x: 0..{self.map_size} left to right, rows every {STEP} units")
        lines.append(LEGEND)
        return "\n".join(lines)

    # ------------------------------------------------------------------
    # Text reports
    # ------------------------------------------------------------------
    def status_text(self) -> str:
        """Return the system snapshot: funds, riders and their orders."""
        riders = self.system.riders
        idle = sum(1 for rider in riders if rider.state == 0)
        lines = [
            f"Simulation tick: {self.system.current_tick}",
            f"System funds: {self.system.funds}",
            f"Riders: {len(riders)} (idle {idle}, busy {len(riders) - idle})",
        ]
        for rider in riders:
            order_ids = ", ".join(f"#{o.id}" for o in rider.orders) or "-"
            lines.append(
                f"  Rider #{rider.rider_id}: position ({rider.x}, {rider.y}) | "
                f"status: {STATE_NAMES.get(rider.state, 'unknown')} | "
                f"orders: {len(rider.orders)} [{order_ids}]")
        pending = self.order_manager.pending_count() if self.order_manager else 0
        lines.append(f"Pending orders: {pending}")
        return "\n".join(lines)

    def orders_text(self) -> str:
        """Return a table of all orders with timing and assignment details."""
        orders = self.order_manager.orders if self.order_manager else []
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Distance':>9} {'Time':>6} {'Eff':>7} {'Rider':>6} {'Status':>10}")
        lines = [header, "-" * len(header)]
        for order in orders:
            restaurant = f"({order.res_pos[0]},{order.res_pos[1]})"
            customer = f"({order.cus_pos[0]},{order.cus_pos[1]})"
            time_taken = str(order.time_taken) if order.time_taken is not None else "-"
            efficiency = (f"{order.efficiency:.2f}"
                          if order.efficiency is not None else "-")
            rider = f"#{order.rider_id}" if order.rider_id is not None else "-"
            lines.append(
                f"{order.id:>4} {order.start_time:>6} {restaurant:>12} {customer:>12} "
                f"{order.distance:>9.1f} {time_taken:>6} {efficiency:>7} "
                f"{rider:>6} {order.status:>10}")
        return "\n".join(lines)

    def riders_text(self) -> str:
        """Return every recorded status change for every rider."""
        lines: List[str] = []
        for rider in self.system.riders:
            lines.append(f"Rider #{rider.rider_id} (speed {rider.speed}/tick):")
            if not rider.log:
                lines.append("  no records")
            for entry in rider.log:
                order = f"#{entry['order_id']}" if entry["order_id"] is not None else "-"
                lines.append(
                    f"  tick {entry['tick']:>4} | {entry['event']:<20} | "
                    f"order {order:>5} | pos ({entry['x']},{entry['y']}) | "
                    f"{STATE_NAMES.get(entry['state'], 'unknown')}")
        return "\n".join(lines) if lines else "No riders yet."

"""Command-line parsing, dispatch and input validation."""

import logging
import math
from typing import Any, Callable, Dict, List, Optional, Sequence, Tuple

from config import save_state
from order import find_best_rider_for_order
from order_model import validate_order_coords
from path_planner import cell_index

try:  # Command history and line editing where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    readline = None

logger = logging.getLogger(__name__)

PROMPT = "dispatch> "
MAX_TICKS_PER_COMMAND = 100000
DEFAULT_SAVE_PATH = "dispatch_state.json"

ALIASES: Dict[str, str] = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "t": "tick",
    "m": "map",
    "w": "save",
    "h": "help",
    "?": "help",
    "hist": "history",
    "q": "quit",
    "x": "exit",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant then customer, 0-800).
                       Prompts for the four integers when none are given.
  status (st)          System funds, riders, their state and orders.
  orders (o)           Details of all orders.
  riders (r)           Status change log for every rider.
  tick [n] (t)         Advance the simulation by n ticks (default 1).
  map (m)              Show the ASCII map with riders and orders.
  save [path] (w)      Save parameters, rider info and order history to JSON.
  history (hist)       Show previously entered commands.
  !!                   Repeat the last command.
  help (h, ?)          Show this help.
  quit (q) / exit (x)  Leave the program."""


class InputManager:
    """Reads commands, validates input and dispatches to handlers.

    Attributes:
        system: The ``SystemManager``.
        order_manager: The ``OrderManager``.
        display: The ``DisplayManager``.
        command_history: Every command entered, in order.
        input_func: Function used to read a line (injectable for testing).
        output_func: Function used to print messages.
    """

    def __init__(self, system: Any, order_manager: Any, display: Any,
                 input_func: Callable[[str], str] = input,
                 output_func: Callable[[str], Any] = print) -> None:
        """Create the input manager.

        Args:
            system: System manager.
            order_manager: Order manager.
            display: Display manager.
            input_func: Line reader.
            output_func: Output writer.
        """
        self.system = system
        self.order_manager = order_manager
        self.display = display
        self.input_func = input_func
        self.output_func = output_func
        self.command_history: List[str] = []
        self._handlers: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "tick": self.cmd_tick,
            "map": self.cmd_map,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def output(self, message: str = "") -> None:
        """Write a message to the user."""
        self.output_func(message)

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> None:
        """Run the interactive command loop until quit or end of input."""
        self.output("Food delivery dispatch simulator. Type 'help' for commands.")
        while self.system.is_running:
            try:
                raw = self.input_func(PROMPT)
            except (EOFError, KeyboardInterrupt):
                self.output("")
                break
            try:
                self.execute(raw)
            except Exception as exc:  # noqa: BLE001 - recover and continue
                logger.exception("Command failed: %r", raw)
                self.output(f"Error: {exc}. Checking system state...")
                repaired = self.system.repair_state()
                if repaired:
                    self.output(f"Recovered {repaired} rider state(s).")
        self.output("Goodbye.")

    def execute(self, raw: str) -> None:
        """Parse one command line and run its handler.

        Args:
            raw: The raw text entered by the user.
        """
        line = raw.strip()
        if not line:
            return
        if line == "!!":
            if not self.command_history:
                self.output("No previous command to repeat.")
                return
            line = self.command_history[-1]
            self.output(f"Repeating: {line}")
        self.command_history.append(line)

        parts = line.split()
        name = parts[0].lower()
        command = ALIASES.get(name, name)
        handler = self._handlers.get(command)
        if handler is None:
            self.output(f"Unknown command '{parts[0]}'. Type 'help' for a list of commands.")
            return
        handler(parts[1:])

    # ------------------------------------------------------------------
    # Parsing and validation
    # ------------------------------------------------------------------
    @staticmethod
    def parse_coordinates(text: str) -> List[int]:
        """Parse and validate four order coordinates.

        Args:
            text: Text containing four integers separated by spaces or commas.

        Returns:
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            ValueError: If the count, type or range is wrong.
        """
        parts = text.replace(",", " ").split()
        if len(parts) != 4:
            raise ValueError(
                f"expected 4 integers (restaurant_x restaurant_y customer_x "
                f"customer_y) but got {len(parts)}")
        values: List[int] = []
        for part in parts:
            try:
                values.append(int(part))
            except ValueError:
                raise ValueError(f"'{part}' is not an integer") from None
        return validate_order_coords(values)

    def _prompt_for_coordinates(self) -> Optional[List[int]]:
        """Ask the user for coordinates until valid input or cancellation.

        Returns:
            The validated coordinates, or None if the user cancelled.
        """
        while True:
            try:
                text = self.input_func(
                    "Enter restaurant_x restaurant_y customer_x customer_y "
                    "(blank to cancel): ").strip()
            except (EOFError, KeyboardInterrupt):
                return None
            if not text:
                return None
            try:
                return self.parse_coordinates(text)
            except ValueError as exc:
                self.output(f"Invalid input: {exc}. Please try again.")

    # ------------------------------------------------------------------
    # Handlers
    # ------------------------------------------------------------------
    def cmd_start(self, args: List[str]) -> None:
        """Create an order, assign a rider and show the route map."""
        if args:
            try:
                coords = self.parse_coordinates(" ".join(args))
            except ValueError as exc:
                self.output(f"Invalid input: {exc}")
                return
        else:
            coords = self._prompt_for_coordinates()
            if coords is None:
                self.output("Order entry cancelled.")
                return
        self._dispatch_order(coords)

    def _dispatch_order(self, coords: Sequence[int]) -> None:
        """Create, recruit for, assign and report an order."""
        riders_before = len(self.system.riders)
        order = self.order_manager.create_order(coords)
        self.system.maybe_recruit(self.order_manager.pending_count())
        new_riders = self.system.riders[riders_before:]

        rider = find_best_rider_for_order(order)
        if rider is None:
            self.output(f"Order #{order.id} stays pending: no rider available "
                        f"(funds {self.system.funds}).")
            return
        self.order_manager.assign(order, rider)

        start = rider.position
        waypoints = rider.planner.plan_route(start, order.res_pos, order.cus_pos)
        length = rider.planner.path_length(start, waypoints)
        eta = math.ceil(length / rider.speed) if rider.speed > 0 else 0

        rx, ry = order.res_pos
        cx, cy = order.cus_pos
        self.output(f"Order #{order.id} accepted.")
        self.output(f"  Restaurant: input {tuple(coords[:2])} -> grid cell "
                    f"(col {cell_index(rx)}, row {cell_index(ry)}), centre ({rx}, {ry})")
        self.output(f"  Customer:   input {tuple(coords[2:])} -> grid cell "
                    f"(col {cell_index(cx)}, row {cell_index(cy)}), centre ({cx}, {cy})")
        if new_riders:
            added = ", ".join(f"#{r.rider_id}" for r in new_riders)
            self.output(f"  New rider added: yes ({added}); funds now {self.system.funds}")
        else:
            self.output("  New rider added: no")
        self.output(f"  Assigned rider: #{rider.rider_id} at ({rider.x}, {rider.y}), "
                    f"speed {rider.speed} units/tick, status {rider.state_name}")
        self.output(f"  Route length: {length} units, estimated ticks: {eta} "
                    f"(from the rider's current position)")
        self.output("  Route map:")
        self.output(self.display.render(route=[start] + waypoints, route_start=start))

    def cmd_status(self, args: List[str]) -> None:
        """Show the system snapshot and consistency results."""
        self.output(self.display.status_text())
        issues = self.system.check_consistency()
        if issues:
            self.output("Consistency warnings:")
            for issue in issues:
                self.output(f"  - {issue}")
        else:
            self.output("Consistency check: OK")

    def cmd_orders(self, args: List[str]) -> None:
        """Show details of all orders."""
        self.output(self.display.orders_text())

    def cmd_riders(self, args: List[str]) -> None:
        """Show every recorded rider status change."""
        self.output(self.display.riders_text())

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation clock by one or more ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                self.output("Tick count must be an integer.")
                return
        if not 1 <= count <= MAX_TICKS_PER_COMMAND:
            self.output(f"Tick count must be between 1 and {MAX_TICKS_PER_COMMAND}.")
            return
        completed_before = len(self.order_manager.completed_orders())
        for _ in range(count):
            self.system.tick()
        delivered = len(self.order_manager.completed_orders()) - completed_before
        self.output(f"Advanced {count} tick(s). Current tick: {self.system.current_tick}. "
                    f"Deliveries completed: {delivered}. Funds: {self.system.funds}.")

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map."""
        self.output(self.display.render())

    def cmd_save(self, args: List[str]) -> None:
        """Save the simulation state to a JSON file."""
        path = args[0] if args else DEFAULT_SAVE_PATH
        try:
            save_state(path, self.system, self.order_manager)
        except OSError as exc:
            self.output(f"Could not save state: {exc}")
            return
        self.output(f"State saved to {path}.")

    def cmd_history(self, args: List[str]) -> None:
        """List previously entered commands."""
        if not self.command_history:
            self.output("No commands entered yet.")
            return
        for index, command in enumerate(self.command_history, start=1):
            self.output(f"  {index:>3}  {command}")

    def cmd_help(self, args: List[str]) -> None:
        """Show the command help."""
        self.output(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the simulator."""
        self.system.shutdown()

"""Entry point for the food delivery dispatch simulator.

Usage:
    python src/main.py [--config CONFIG_JSON] [--debug]
"""

import argparse
import logging
import sys
from typing import List, Optional

from config import load_config
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from system import init_system


def build_arg_parser() -> argparse.ArgumentParser:
    """Create the command-line argument parser.

    Returns:
        Configured ``ArgumentParser``.
    """
    parser = argparse.ArgumentParser(
        description="Command-line food delivery dispatch simulator.")
    parser.add_argument("--config", metavar="PATH",
                        help="JSON file with system parameters.")
    parser.add_argument("--debug", action="store_true",
                        help="Enable debug logging to stderr.")
    return parser


def main(argv: Optional[List[str]] = None) -> int:
    """Run the simulator.

    Args:
        argv: Optional argument list (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code.
    """
    args = build_arg_parser().parse_args(argv)
    logging.basicConfig(
        level=logging.DEBUG if args.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s")

    config = load_config(args.config) if args.config else None
    system = init_system(config)
    order_manager = OrderManager()
    display = DisplayManager(system, order_manager)
    shell = InputManager(system, order_manager, display)
    try:
        shell.run()
    except KeyboardInterrupt:
        print()
    return 0


if __name__ == "__main__":
    sys.exit(main())

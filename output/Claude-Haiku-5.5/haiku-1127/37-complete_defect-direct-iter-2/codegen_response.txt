class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list uploaded by the user."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file where each line is 'Name,Department Group'."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        lines = None
        for encoding in ("utf-8-sig", "gb18030"):
            try:
                with open(path, "r", encoding=encoding) as handle:
                    lines = handle.read().splitlines()
                break
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"File Error: cannot read file: {exc}"]

        if lines is None:
            return [], ["File Error: unable to decode file (expected UTF-8 or GBK)"]

        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines, collecting format errors with line numbers."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = text.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department', got '{text}'"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). The first occurrence of a name is kept as valid."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department group -> number of people, in first-appearance order."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

# Preset prize library: (name, quantity, avoid_groups, associated_drawer)
DEFAULT_PRIZES = [

    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("Technology Department Exclusive Coupon", 2, ["Product Department"], "Chen Qixian"),

class PrizeManager:
    """Manages the ordered list of prizes for a lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"Prize '{name}' not added: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the current order. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"a prize named '{clean_name}' already exists")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery round order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order.

        Returns True on success, False if `order` is not a permutation of all positions.
        """
        n = len(self._prizes)
        if n == 0 or len(order) != n:
            return False
        if set(order) != set(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)


class LotteryManager:
    """Draws winners for prizes while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.winners: list[User] = []
        self._winner_names: set[str] = set()
        # Records whether each recent round of Chen Qixian / Jin Duo contained a
        # Cheng Ru Group member. Only the last two entries are kept.
        self.last_two_draw_results: list[bool] = []

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _is_drawer(prize: Prize, drawer_name: str) -> bool:
        return (
            prize.associated_drawer is not None
            and prize.associated_drawer.strip().casefold() == drawer_name.casefold()
        )

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return user.department.strip().casefold() == CHENG_RU_GROUP.casefold()

    def has_won(self, user: User) -> bool:
        """Return True if the person has already won any prize."""
        return user.name in self._winner_names

    def _protection_active(self) -> bool:
        """Fallback is triggered when the last two Chen/Jin results both lack Cheng Ru members."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    # ------------------------------------------------------------------ rules
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the candidate pool for a prize round."""
        # Basic filtering: exclude all previous winners (any prize type).
        pool = [user for user in self.users if not self.has_won(user)]

        # Special rule: Li Bin only filters previous winners.
        if self._is_drawer(prize, LI_BIN):
            return pool

        # Fallback (Cheng Ru group protection): forced to Cheng Ru members,
        # overriding the prize's avoidance groups.
        if self._is_drawer(prize, MA_JIAQI) and self._protection_active():
            return [user for user in pool if self._is_cheng_ru(user)]

        # Group avoidance rule.
        avoid = {group.strip().casefold() for group in prize.avoid_groups if group and group.strip()}
        if avoid:
            pool = [user for user in pool if user.department.strip().casefold() not in avoid]
        return pool

    def count_eligible(self, prize: Prize) -> int:
        """Number of candidates currently eligible for the prize."""
        return len(self._get_eligible_pool(prize))

    def describe_rules(self, prize: Prize) -> str:
        """Human readable description of the rules applied to a prize round."""
        rules: list[str] = []
        if prize.associated_drawer:
            rules.append(f"lottery person: {prize.associated_drawer}")
        rules.append("previous winners are excluded")
        if self._is_drawer(prize, LI_BIN):
            rules.append("special rule for Li Bin: avoidance groups are not applied")
        elif self._is_drawer(prize, MA_JIAQI) and self._protection_active():
            rules.append(
                f"fallback active: only {CHENG_RU_GROUP} members are eligible "
                "(avoidance groups overridden)"
            )
        elif prize.avoid_groups:
            rules.append("avoid groups: " + ", ".join(prize.avoid_groups))
        return "; ".join(rules)

    # ------------------------------------------------------------------ drawing
    def draw_next(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        self._winner_names.add(user.name)
        return user

    def complete_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Finish a round and update the fallback history for Chen Qixian / Jin Duo rounds."""
        if prize.associated_drawer is None:
            return
        if self._is_drawer(prize, CHEN_QIXIAN) or self._is_drawer(prize, JIN_DUO):
            hit = any(self._is_cheng_ru(user) for user in round_winners)
            self.last_two_draw_results.append(hit)
            del self.last_two_draw_results[:-2]

"""Winner recording, grouping and CSV-style export of lottery results."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record: a user who won a prize."""

    user: User
    prize: Prize

    @property
    def prize_name(self) -> str:
        return self.prize.name


class ResultManager:
    """Keeps the winner list in winning order and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        self._winners.append(Winner(user=user, prize=prize))

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name. Every given prize has a key (possibly empty list)."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            grouped.setdefault(winner.prize.name, []).append(winner)
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)

        ordered: list[Winner] = []
        grouped = self.get_winners_by_prize(prizes)
        for prize in prizes:
            ordered.extend(grouped.get(prize.name, []))
        listed = {prize.name for prize in prizes}
        for winner in self._winners:
            if winner.prize.name not in listed:
                ordered.append(winner)

        try:
            with open(path, "w", encoding="utf-8-sig", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for winner in ordered:
                    writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"

        return True, f"Result successfully exported to {path}"

def ordinal(n: int) -> str:
    """Return the English ordinal form of n (1st, 2nd, 3rd, 11th, ...)."""
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotterySystemCLI:
    """Interactive console application."""

        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.result_manager = ResultManager()
        self.last_prizes: list[Prize] = []

    # ------------------------------------------------------------------ main loop
    def run(self) -> None:
        while True:
            print(MAIN_MENU)
            choice = self._input("Select an option: ")
            try:
                if choice == "1":
                    self._load_list_menu()
                elif choice == "2":
                    self._show_groups()
                elif choice == "3":
                    self._prize_menu()
                elif choice == "4":
                    self._start_lottery()
                elif choice == "5":
                    self._view_results()
                elif choice == "6":
                    self._export_results()
                elif choice == "0":
                    print("Goodbye.")
                    return
                else:
                    print("Invalid option, please try again.")
            except (EOFError, KeyboardInterrupt):
                print("\nExiting.")
                return

    @staticmethod
    def _input(prompt: str) -> str:
        return input(prompt).strip()

    # ------------------------------------------------------------------ list management
    def _load_list_menu(self) -> None:
        print("1. Built-in default team building list")
        print("2. Upload custom txt list (each line: Name,Department Group)")
        print("0. Cancel")
        choice = self._input("Select source: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._input("Enter the txt file path: ").strip('"')
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

    def _verify_and_apply(self, users: list[User], errors: list[str]) -> None:

            self.users = valid
            print(f"List loaded successfully: {len(valid)} participants.")
            return

        print("\nList verification found problems:")
        for error in errors:
            print(f"  - {error}")
        for dup in duplicates:
            print(f"  - Duplicate Name: '{dup.name}' ({dup.department}) appears more than once")
        print(f"Valid participants: {len(valid)}")
        print("1. Remove invalid/duplicate data and continue with valid entries")
        print("2. Terminate upload (keep previous list)")

        choice = self._input("Select an option: ")
        if choice == "1":
            if not valid:
                print("No valid participants remain. Upload terminated, previous list kept.")
                return
            self.users = valid
            print(f"List loaded with invalid/duplicate data removed: {len(valid)} participants.")
        else:
            print("Upload terminated. Previous list kept.")

    # ------------------------------------------------------------------ lottery
    def _start_lottery(self) -> None:

            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lottery = LotteryManager(self.users)
        self.result_manager = ResultManager()
        self.last_prizes = prizes
        self.has_results = False

        print("\nLottery started.")
        print("At each draw prompt: press Enter to draw the next winner, "
              "type 'pause' to pause, or 'terminate' to end the current round.")

        for index, prize in enumerate(prizes, start=1):
            self._run_round(lottery, index, prize)

        self.has_results = True
        print("\nAll lottery rounds completed.")
        self._show_results(prizes)

    def _run_round(self, lottery: LotteryManager, index: int, prize: Prize) -> None:
        print(f"\n----- Round {index}: {prize.name} -----")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {lottery.describe_rules(prize)}")

        round_winners: list[User] = []
        drawn = 0
        while drawn < prize.quantity:
            if lottery.count_eligible(prize) == 0:
                print("No eligible candidates remain for this round. Round ends.")
                break

            command = self._input("[Enter] draw next / 'pause' / 'terminate': ").lower()
            if command == "pause":
                if not self._handle_pause():
                    print("Current round terminated and skipped.")
                    break
                continue
            if command == "terminate":
                print("Current round terminated and skipped.")
                break
            if command == "continue":
                print("Lottery is not paused.")
                continue
            if command != "":
                print("Unknown command. Press Enter to draw, or type 'pause' / 'terminate'.")
                continue

            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = lottery.draw_next(prize)
            if user is None:
                print("No eligible candidates remain for this round. Round ends.")
                break
            drawn += 1
            self.result_manager.add_winner(user, prize)
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        lottery.complete_round(prize, round_winners)

        print(f"\nRound {index} results ({prize.name}):")
        if not round_winners:
            print("  No winners in this round.")
        for position, user in enumerate(round_winners, start=1):
            print(f"  {position}. {user.name} ({user.department})")

    def _handle_pause(self) -> bool:
        """Block until 'continue' (returns True) or 'terminate' (returns False)."""
        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = self._input("> ").lower()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Please type 'continue' or 'terminate'.")

    # ------------------------------------------------------------------ results
    def _show_results(self, prizes: list[Prize]) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n===== Complete Winner List =====")
        print(f"{'No.':<5}{'Name':<16}{'Department':<30}Prize")
        number = 0
        for prize in prizes:
            for winner in grouped.get(prize.name, []):
                number += 1
                print(f"{number:<5}{winner.user.name:<16}{winner.user.department:<30}{winner.prize_name}")
        if number == 0:
            print("No winners.")
        print(f"Total winners: {number}")

    def _view_results(self) -> None:
        if not self.has_results:
            print("No lottery results yet. Run the lottery first.")
            return
        self._show_results(self.last_prizes)

    def _export_results(self) -> None:
        if not self.has_results:
            print("No lottery results to export. Run the lottery first.")
            return
        success, message = self.result_manager.export_to_csv(self.last_prizes)
        print(message)

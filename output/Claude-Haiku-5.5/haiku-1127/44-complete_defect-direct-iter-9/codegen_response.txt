"""Loading and saving of the Beijing Subway dataset (subway_data.json)."""

import json
import os

# Absolute path of the dataset file, valid regardless of the current working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read, parsed or has an invalid top-level structure."""


def _validate_structure(data):
    """Check that the top-level JSON structure has the expected shape.

    Args:
        data: The parsed JSON value.

    Raises:
        DataLoadError: If the structure is not an object containing list fields
            'stations' and 'connections'.
    """
    if not isinstance(data, dict):
        raise DataLoadError("top-level JSON value must be an object")
    for key in ("stations", "connections"):
        if key not in data or not isinstance(data[key], list):
            raise DataLoadError(f"missing or invalid list field '{key}'")


def load_subway_data(path):
    """Load the subway dataset from a JSON file.

    Args:
        path: Path of the JSON file.

    Returns:
        dict: The JSON content, with 'stations' and 'connections' lists.

    Raises:
        DataLoadError: If the file is missing, unreadable, not valid JSON or malformed.
    """
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except FileNotFoundError as exc:
        raise DataLoadError(f"subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"subway data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataLoadError(f"cannot read subway data file: {exc}") from exc
    _validate_structure(data)
    return data


def save_subway_data(data, path):
    """Write the subway dataset back to a JSON file.

    Args:
        data: The dataset dictionary.
        path: Destination file path. Parent directories are created when missing.

    Raises:
        DataLoadError: If the file cannot be written.
    """
    try:
        directory = os.path.dirname(os.path.abspath(path))
        os.makedirs(directory, exist_ok=True)
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, ensure_ascii=False, indent=2)
    except OSError as exc:
        raise DataLoadError(f"cannot write subway data file: {exc}") from exc

"""Graph data structures and the subway network model.

Station numbers are assigned by the caller (here: SubwayNetwork), not by Graph.
"""


class NetworkBuildError(ValueError):
    """Raised when the raw dataset cannot be turned into a subway network."""


class Edge:
    """A weighted connection between two stations (identified by station numbers).

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance in meters.
        line: Name of the subway line operating this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by shortest path search (equal to the distance in meters)."""
        return self.distance

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``.

        Args:
            vertex: One endpoint station number.

        Returns:
            int: The other endpoint station number.

        Raises:
            ValueError: If ``vertex`` is not an endpoint of this edge.
        """
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"vertex {vertex} is not an endpoint of {self!r}")

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph with adjacency lists.

    Vertices are station numbers. Each edge is stored in the adjacency list of
    both of its endpoints. The mapping ``station_lines`` maps station names to the
    list of lines that serve them.
    """

    def __init__(self):
        self._adjacency = {}
        self._edges = []
        self.station_lines = {}

    def add_vertex(self, vertex):
        """Add a vertex if it does not already exist."""
        self._adjacency.setdefault(vertex, [])

    def has_vertex(self, vertex):
        """Return True if the vertex exists in the graph."""
        return vertex in self._adjacency

    def set_station_lines(self, station_name, lines):
        """Record the lines serving a station.

        Args:
            station_name: Official station name.
            lines: Iterable of line names.
        """
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an undirected edge, creating missing endpoint vertices.

        Args:
            edge: An Edge instance whose endpoints are station numbers.

        Raises:
            TypeError: If ``edge`` is not an Edge.
        """
        if not isinstance(edge, Edge):
            raise TypeError("add_edge expects an Edge instance")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adjacency[edge.source].append(edge)
        if edge.dest != edge.source:
            self._adjacency[edge.dest].append(edge)
        self._edges.append(edge)

    def adjacent_edges(self, vertex):
        """Return the edges incident to ``vertex``.

        The returned list is the internal adjacency list and must not be modified.

        Raises:
            KeyError: If the vertex does not exist.
        """
        return self._adjacency[vertex]

    def degree(self, vertex):
        """Return the number of edges incident to ``vertex``."""
        return len(self._adjacency[vertex])

    def vertices(self):
        """Return a list of all vertex numbers."""
        return list(self._adjacency.keys())

    def edges(self):
        """Return a list of all edges (each counted once)."""
        return list(self._edges)

    @property
    def num_vertices(self):
        """Number of vertices in the graph."""
        return len(self._adjacency)

    @property
    def num_edges(self):
        """Number of edges in the graph."""
        return len(self._edges)


class SubwayNetwork:
    """The Beijing Subway network: a graph plus station name/number mappings.

    Attributes:
        graph: The weighted undirected Graph.
        station_to_number: dict mapping station names to station numbers.
        number_to_name: dict mapping station numbers to station names.
        lines: Ordered list of all known line names.
        warnings: Non-fatal problems detected while building the network.
    """

    def __init__(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_name = {}
        self.lines = []
        self.warnings = []

    @property
    def station_lines(self):
        """dict mapping station names to lists of line names."""
        return self.graph.station_lines

    def _register_line(self, line):
        if line not in self.lines:
            self.lines.append(line)

    def add_station(self, name, lines=()):
        """Add a station if it is new and return its number.

        Args:
            name: Official station name.
            lines: Lines serving the station (used only when the station is new).

        Returns:
            int: The station number.
        """
        if name in self.station_to_number:
            return self.station_to_number[name]
        number = len(self.number_to_name)
        self.station_to_number[name] = number
        self.number_to_name[number] = name
        self.graph.add_vertex(number)
        self.graph.set_station_lines(name, lines)
        for line in lines:
            self._register_line(line)
        return number

    def number_of(self, name):
        """Return the station number of a station name (KeyError if unknown)."""
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the station name of a station number (KeyError if unknown)."""
        return self.number_to_name[number]

    def station_names(self):
        """Return the sorted list of all station names."""
        return sorted(self.station_to_number)

    @classmethod
    def from_data(cls, data):
        """Build a network from the raw dataset dictionary.

        Args:
            data: dict with 'stations' ([{name, lines}]) and
                'connections' ([{source, dest, distance, line}]) lists.

        Returns:
            SubwayNetwork: The constructed network.

        Raises:
            NetworkBuildError: If the data is malformed.
        """
        if not isinstance(data, dict):
            raise NetworkBuildError("subway data must be a JSON object")
        stations = data.get("stations", [])
        connections = data.get("connections", [])
        if not isinstance(stations, list) or not isinstance(connections, list):
            raise NetworkBuildError("'stations' and 'connections' must be lists")

        network = cls()
        for entry in stations:
            if not isinstance(entry, dict) or "name" not in entry:
                raise NetworkBuildError(f"invalid station entry: {entry!r}")
            name = str(entry["name"]).strip()
            if not name:
                raise NetworkBuildError("station name must not be empty")
            lines = [str(line) for line in entry.get("lines", [])]
            if name in network.station_to_number:
                network.warnings.append(f"station '{name}' is listed more than once; line lists merged")
                existing = network.graph.station_lines[name]
                for line in lines:
                    if line not in existing:
                        existing.append(line)
                    network._register_line(line)
            else:
                network.add_station(name, lines)

        for entry in connections:
            try:
                source = str(entry["source"]).strip()
                dest = str(entry["dest"]).strip()
                distance_raw = entry["distance"]
                line = str(entry["line"]).strip()
            except (KeyError, TypeError) as exc:
                raise NetworkBuildError(f"invalid connection entry: {entry!r}") from exc
            if isinstance(distance_raw, bool) or not isinstance(distance_raw, (int, float)) or distance_raw <= 0:
                raise NetworkBuildError(f"invalid distance in connection: {entry!r}")
            for name in (source, dest):
                if name not in network.station_to_number:
                    network.warnings.append(f"station '{name}' appears in connections but not in the station list")
                    network.add_station(name, [line])
            network._register_line(line)
            network.graph.add_edge(
                Edge(
                    network.station_to_number[source],
                    network.station_to_number[dest],
                    int(round(distance_raw)),
                    line,
                )
            )
        return network

"""Shortest path computation (Dijkstra's algorithm with a binary-heap priority queue).

The search runs over states (station number, line used to arrive). This makes
transfer counting exact and allows the maximum-transfer limit and line
preferences to be enforced during the search.

Query modes:
    shortest_distance: minimise total distance (ties broken by fewer transfers).
    least_transfer:    minimise transfers (ties broken by distance).
    comprehensive:     minimise distance + TRANSFER_PENALTY_M per transfer.
"""

import heapq
import itertools

SHORTEST_DISTANCE = "shortest_distance"
LEAST_TRANSFER = "least_transfer"
COMPREHENSIVE = "comprehensive"
VALID_MODES = (SHORTEST_DISTANCE, LEAST_TRANSFER, COMPREHENSIVE)

# Equivalent distance cost (meters) of a single transfer in comprehensive mode.
TRANSFER_PENALTY_M = 3000
# Sort-key penalty (meters) for each edge that is not on a preferred line.
PREFERENCE_PENALTY_M = 1500


class NoPathError(Exception):
    """Raised when no path exists between two stations under the given constraints."""


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of VALID_MODES.
        max_transfers: Maximum allowed number of transfers, or None for no limit.
        preferred_lines: List of line names that are preferred.
    """

    def __init__(self, mode=SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []
        self.validate()

    def validate(self):
        """Validate the configuration.

        Raises:
            ValueError: If the mode, transfer limit or preferred lines are invalid.
        """
        if self.mode not in VALID_MODES:
            raise ValueError(f"unknown mode '{self.mode}', expected one of {', '.join(VALID_MODES)}")
        if self.max_transfers is not None:
            if isinstance(self.max_transfers, bool) or not isinstance(self.max_transfers, int):
                raise ValueError("max_transfers must be an integer or None")
            if self.max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        if not all(isinstance(line, str) for line in self.preferred_lines):
            raise ValueError("preferred_lines must contain only strings")


class MinPQ:
    """Min priority queue backed by a binary heap.

    Items are stored with their keys; a monotonically increasing sequence number
    breaks ties so that items themselves never need to be comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        """Insert an item with the given key in O(log n)."""
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self):
        """Remove and return ``(key, item)`` with the smallest key in O(log n).

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("priority queue underflow")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min_key(self):
        """Return the smallest key without removing it (IndexError if empty)."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._heap[0][0]

    def is_empty(self):
        """Return True if the queue has no items."""
        return not self._heap

    def __len__(self):
        return len(self._heap)


class _Label:
    """Accumulated values for a search state."""

    __slots__ = ("dist", "transfers", "penalty")

    def __init__(self, dist, transfers, penalty):
        self.dist = dist
        self.transfers = transfers
        self.penalty = penalty


class ShortestPath:
    """Dijkstra shortest path search from a single source station.

    Attributes:
        dist_to: dict mapping reachable station numbers to the travelled distance
            (meters) of their best path.
    """

    def __init__(self, graph, source, config=None):
        """Run the search immediately.

        Args:
            graph: A models.Graph instance.
            source: Source station number.
            config: Optional PathfindingConfig; defaults to shortest distance mode.

        Raises:
            ValueError: If the source is not in the graph or the config is invalid.
        """
        if not graph.has_vertex(source):
            raise ValueError(f"unknown source station number: {source}")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()
        self.dist_to = {}
        self._transfers_to = {}
        self._settled_state = {}
        self._prev = {}
        self._run()

    def _key(self, label):
        """Return the sort key of a label according to the query mode."""
        effective = label.dist + label.penalty
        mode = self.config.mode
        if mode == LEAST_TRANSFER:
            return (label.transfers, effective)
        if mode == COMPREHENSIVE:
            return (effective + label.transfers * TRANSFER_PENALTY_M, label.dist)
        return (effective, label.transfers)

    def _run(self):
        """Execute Dijkstra's algorithm over (station, line) states."""
        start = (self.source, None)
        labels = {start: _Label(0, 0, 0)}
        pq = MinPQ()
        pq.insert(self._key(labels[start]), start)
        settled = set()
        preferred = set(self.config.preferred_lines)
        max_transfers = self.config.max_transfers

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)
            node, line = state
            label = labels[state]

            if node not in self._settled_state:
                self._settled_state[node] = state
                self.dist_to[node] = label.dist
                self._transfers_to[node] = label.transfers

            for edge in self.graph.adjacent_edges(node):
                nxt = edge.other(node)
                transfers = label.transfers
                if line is not None and line != edge.line:
                    transfers += 1
                if max_transfers is not None and transfers > max_transfers:
                    continue
                penalty = label.penalty
                if preferred and edge.line not in preferred:
                    penalty += PREFERENCE_PENALTY_M
                new_state = (nxt, edge.line)
                if new_state in settled:
                    continue
                new_label = _Label(label.dist + edge.weight, transfers, penalty)
                new_key = self._key(new_label)
                old = labels.get(new_state)
                if old is None or new_key < self._key(old):
                    labels[new_state] = new_label
                    self._prev[new_state] = (state, edge)
                    pq.insert(new_key, new_state)

    def has_path_to(self, target):
        """Return True if ``target`` is reachable under the configured constraints."""
        return target in self._settled_state

    def get_transfer_count(self, target):
        """Return the number of line changes on the best path to ``target``.

        Raises:
            NoPathError: If the target is unreachable.
        """
        if target not in self._transfers_to:
            raise NoPathError(f"station {target} is unreachable from {self.source}")
        return self._transfers_to[target]

    def path_with_edges(self, target):
        """Reconstruct the best path to ``target``.

        Returns:
            tuple: (list of station numbers from source to target, list of Edge
            objects traversed in order). The edge list is empty when the target
            equals the source.

        Raises:
            NoPathError: If the target is unreachable.
        """
        if target not in self._settled_state:
            raise NoPathError(f"station {target} is unreachable from {self.source}")
        state = self._settled_state[target]
        nodes = [state[0]]
        edges = []
        while state in self._prev:
            prev_state, edge = self._prev[state]
            edges.append(edge)
            nodes.append(prev_state[0])
            state = prev_state
        nodes.reverse()
        edges.reverse()
        return nodes, edges

"""Beijing Subway fare and travel time calculation.

Fare structure:
    0-6 km: 3 yuan; 6-12 km: 4 yuan; 12-22 km: 5 yuan; 22-32 km: 6 yuan.
    Beyond 32 km: +1 yuan for every additional (started) 20 km.
Distances are rounded up to the next whole meter before tier lookup.
"""

import math

# (upper bound in meters inclusive, fare in yuan)
FARE_TIERS = ((6000, 3), (12000, 4), (22000, 5), (32000, 6))
EXTENSION_START_M = 32000
EXTENSION_STEP_M = 20000
EXTENSION_STEP_FARE = 1

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MIN = 5.0


class FareError(ValueError):
    """Raised when fare or time calculation receives invalid input."""


def _validate_distance(distance):
    """Validate a distance in meters and return it.

    Raises:
        FareError: If the distance is not a finite non-negative number.
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise FareError(f"distance must be a number, got {type(distance).__name__}")
    if math.isnan(distance) or math.isinf(distance):
        raise FareError("distance must be finite")
    if distance < 0:
        raise FareError("distance must be non-negative")
    return distance


def Sum_money(distance):
    """Compute the subway fare for a distance.

    Args:
        distance: Travelled distance in meters. Fractional values are rounded up.

    Returns:
        int: Fare in yuan.

    Raises:
        FareError: If the distance is invalid.
    """
    value = _validate_distance(distance)
    meters = math.ceil(value)
    for upper, fare in FARE_TIERS:
        if meters <= upper:
            return fare
    extra = meters - EXTENSION_START_M
    steps = math.ceil(extra / EXTENSION_STEP_M)
    return FARE_TIERS[-1][1] + steps * EXTENSION_STEP_FARE


def fare_table():
    """Return the fare tiers as a list of (description, fare in yuan) tuples."""
    rows = []
    previous = 0
    for upper, fare in FARE_TIERS:
        rows.append((f"{previous / 1000:g}-{upper / 1000:g} km", fare))
        previous = upper
    rows.append((f"over {EXTENSION_START_M / 1000:g} km (+{EXTENSION_STEP_FARE} yuan per "
                 f"{EXTENSION_STEP_M / 1000:g} km)", FARE_TIERS[-1][1]))
    return rows


def calculate_estimated_time(distance, transfer_count):
    """Estimate the travel time in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers.

    Returns:
        float: Estimated travel time in minutes.

    Raises:
        FareError: If the inputs are invalid.
    """
    value = _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise FareError("transfer_count must be a non-negative integer")
    travel_minutes = (value / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    return float(travel_minutes + transfer_count * TRANSFER_TIME_MIN)

class DataUpdateError(ValueError):
    """Raised when a data update request is invalid."""


class IntegrityReport:
    """Result of a data integrity check.

    Attributes:
        errors: List of error messages (cause the check to fail).
        warnings: List of warning messages (informational).
        checks_run: Names of the checks that were executed.
    """

    def __init__(self):
        self.errors = []
        self.warnings = []
        self.checks_run = []

    @property
    def passed(self):
        """True if no errors were found."""
        return not self.errors

    def format(self):
        """Return a human-readable report."""
        lines = ["Data integrity check"]
        lines.append("Checks run: " + ", ".join(self.checks_run))
        if not self.errors and not self.warnings:
            lines.append("  All checks passed without issues.")
        for message in self.errors:
            lines.append(f"  [ERROR] {message}")
        for message in self.warnings:
            lines.append(f"  [WARN] {message}")
        lines.append("Result: " + ("PASSED" if self.passed else "FAILED"))
        return "\n".join(lines)


class DataManager:
    """Integrity checking and updating of the raw subway dataset."""

    def __init__(self, data):
        """Wrap a dataset dictionary.

        Args:
            data: dict with 'stations' and 'connections' lists (modified in place).
        """
        self.data = data
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ------------------------------------------------------------------ helpers
    def _stations(self):
        return self.data["stations"]

    def _connections(self):
        return self.data["connections"]

    def _find_station(self, name):
        for station in self._stations():
            if station.get("name") == name:
                return station
        return None

    def build_network(self):
        """Build a SubwayNetwork from the current dataset."""
        return SubwayNetwork.from_data(self.data)

    # ------------------------------------------------------------ integrity
    def check_integrity(self):
        """Run all integrity checks.

        Returns:
            IntegrityReport: The collected errors and warnings.
        """
        report = IntegrityReport()
        self._check_station_data(report)
        self._check_coverage(report)
        self._check_connections(report)
        self._check_distances(report)
        return report

    def _check_station_data(self, report):
        report.checks_run.append("station data")
        seen = set()
        for index, station in enumerate(self._stations()):
            name = station.get("name")
            if not isinstance(name, str) or not name.strip():
                report.errors.append(f"station entry #{index} has no valid name")
                continue
            if name in seen:
                report.errors.append(f"duplicate station name '{name}'")
            seen.add(name)
            if not station.get("lines"):
                report.warnings.append(f"station '{name}' has no line information")

    def _check_coverage(self, report):
        report.checks_run.append("station coverage")
        covered_lines = set()
        names = set()
        for station in self._stations():
            names.add(station.get("name"))
            covered_lines.update(station.get("lines", []))
        for line in MAIN_LINES:
            if line not in covered_lines:
                report.errors.append(f"main line '{line}' has no stations in the dataset")
        for line, representatives in REPRESENTATIVE_STATIONS.items():
            for name in representatives:
                if name not in names:
                    report.warnings.append(f"representative station '{name}' of {line} is missing")

    def _check_connections(self, report):
        report.checks_run.append("connection integrity")
        names = {station.get("name"): set(station.get("lines", [])) for station in self._stations()}
        adjacency = {name: set() for name in names}
        seen_pairs = {}
        for index, conn in enumerate(self._connections()):
            missing = [key for key in ("source", "dest", "distance", "line") if key not in conn]
            if missing:
                report.errors.append(f"connection #{index} is missing fields {missing}")
                continue
            source, dest, line = conn["source"], conn["dest"], conn["line"]
            if source not in names or dest not in names:
                report.errors.append(f"connection {source} -> {dest} references an unknown station")
                continue
            if source == dest:
                report.errors.append(f"connection starts and ends at the same station '{source}'")
                continue
            for endpoint in (source, dest):
                if line not in names[endpoint]:
                    report.warnings.append(f"station '{endpoint}' is not listed on line '{line}'")
            key = (frozenset((source, dest)), line)
            if key in seen_pairs:
                if seen_pairs[key] != conn.get("distance"):
                    report.errors.append(f"conflicting distances for {source} - {dest} on {line}")
                else:
                    report.warnings.append(f"duplicate connection {source} - {dest} on {line}")
            else:
                seen_pairs[key] = conn.get("distance")
            adjacency[source].add(dest)
            adjacency[dest].add(source)

        isolated = [name for name, neighbours in adjacency.items() if not neighbours]
        for name in isolated:
            report.warnings.append(f"station '{name}' has no connections")

        components = self._count_components(adjacency)
        if components > 1:
            report.warnings.append(f"the network consists of {components} disconnected components")

    @staticmethod
    def _count_components(adjacency):
        remaining = set(adjacency)
        components = 0
        while remaining:
            components += 1
            start = remaining.pop()
            queue = deque([start])
            while queue:
                node = queue.popleft()
                for neighbour in adjacency.get(node, ()):
                    if neighbour in remaining:
                        remaining.discard(neighbour)
                        queue.append(neighbour)
        return components

    def _check_distances(self, report):
        report.checks_run.append("distance accuracy")
        for index, conn in enumerate(self._connections()):
            distance = conn.get("distance")
            label = f"{conn.get('source')} - {conn.get('dest')}"
            if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
                report.errors.append(f"invalid distance for {label}: {distance!r}")
                continue
            if distance < MIN_PLAUSIBLE_DISTANCE_M or distance > MAX_PLAUSIBLE_DISTANCE_M:
                report.warnings.append(
                    f"distance {distance} m for {label} is outside the plausible range "
                    f"[{MIN_PLAUSIBLE_DISTANCE_M}, {MAX_PLAUSIBLE_DISTANCE_M}] m"
                )

    def add_line(self, line_name, station_names, distances):
        """Add a new line through an ordered list of stations.

        Existing stations are reused (and gain the new line); missing stations are created.

        Args:
            line_name: Name of the new line.
            station_names: Ordered list of station names along the line.
            distances: List of distances in meters between consecutive stations.

        Returns:
            int: Number of connections added.

        Raises:
            DataUpdateError: If the request is invalid.
        """
        line_name = (line_name or "").strip()
        if not line_name:
            raise DataUpdateError("line name must not be empty")
        for station in self._stations():
            if line_name in station.get("lines", []):
                raise DataUpdateError(f"line '{line_name}' already exists")
        names = [str(name).strip() for name in station_names]
        if len(names) < 2 or any(not name for name in names):
            raise DataUpdateError("a line needs at least two non-empty station names")
        if len(set(names)) != len(names):
            raise DataUpdateError("station names on a line must be unique")
        if len(distances) != len(names) - 1:
            raise DataUpdateError("exactly one distance per consecutive station pair is required")
        for distance in distances:
            self._validate_distance(distance)

        for name in names:
            station = self._find_station(name)
            if station is None:
                self._stations().append({"name": name, "lines": [line_name]})
            else:
                station.setdefault("lines", [])
                if line_name not in station["lines"]:
                    station["lines"].append(line_name)
        for index, distance in enumerate(distances):
            self._connections().append({
                "source": names[index],
                "dest": names[index + 1],
                "distance": int(distance),
                "line": line_name,
            })
        return len(distances)

    def modify_station(self, name, new_name=None, lines=None):
        """Modify a station's name and/or its lines.

        Renaming also updates all connections that reference the station.

        Args:

                raise DataUpdateError("lines must be a non-empty list of line names")

                raise DataUpdateError(f"station '{new_name}' already exists")
            station["name"] = new_name
            for conn in self._connections():
                if conn.get("source") == name:
                    conn["source"] = new_name
                if conn.get("dest") == name:
                    conn["dest"] = new_name
        return station

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance of the connection(s) between two adjacent stations.

        Args:
            source: One station name.
            dest: The adjacent station name.
            distance: New distance in meters.

        Returns:
            int: Number of connections updated.

            DataUpdateError: If no matching connection exists or the distance is invalid.

            raise DataUpdateError(f"no connection between '{source}' and '{dest}'")
        for conn in matches:
            conn["distance"] = int(distance)
        return len(matches)

"""Path visualization, summary generation and report export."""

import json
import os
from dataclasses import dataclass

from fare_calculator import Sum_money, calculate_estimated_time


@dataclass
class Segment:
    """A continuous run of stations served by one line."""

    line: str
    stations: list
    distance_m: int


@dataclass
class TransferInfo:
    """A transfer station where the passenger changes lines."""

    station: str
    from_line: str
    to_line: str


@dataclass
class PathSummary:
    """Key information extracted from a computed route."""

    start: str
    end: str
    mode: str
    max_transfers: object
    preferred_lines: list
    stations: list
    segments: list
    transfers: list
    lines: list
    distance_m: int
    fare_yuan: int
    transfer_count: int
    estimated_time_min: float


def format_distance(meters):
    """Format a distance using meters for short distances and kilometers for long ones.

    Examples: 850 -> "850 m (0.85 km)"; 12350 -> "12.35 km (12350 m)".
    """
    kilometers = meters / 1000.0
    if meters < 1000:
        return f"{meters} m ({kilometers:.2f} km)"
    return f"{kilometers:.2f} km ({meters} m)"


def _split_segments(names, edges):
    """Split a route into line segments and transfer stations.

    Args:
        names: Station names along the path (length = len(edges) + 1).
        edges: Edge objects along the path.

    Returns:
        tuple: (list of Segment, list of TransferInfo).
    """
    segments = []
    transfers = []
    if not edges:
        return segments, transfers
    current_line = edges[0].line
    segment_start = 0
    segment_distance = 0
    for index, edge in enumerate(edges):
        if edge.line != current_line:
            segments.append(Segment(current_line, names[segment_start:index + 1], segment_distance))
            transfers.append(TransferInfo(names[index], current_line, edge.line))
            current_line = edge.line
            segment_start = index
            segment_distance = 0
        segment_distance += edge.distance
    segments.append(Segment(current_line, names[segment_start:], segment_distance))
    return segments, transfers


def build_summary(network, start, end, station_numbers, edges, transfer_count,
                  mode, max_transfers=None, preferred_lines=None):
    """Build a PathSummary from a computed route.

    Args:
        network: The SubwayNetwork used for name lookups.
        start: Start station name.
        end: End station name.
        station_numbers: Station numbers along the path.
        edges: Edge objects along the path.
        transfer_count: Number of transfers on the path.
        mode: Query mode name.
        max_transfers: Transfer limit used for the query (or None).
        preferred_lines: Preferred lines used for the query (or None).

    Returns:
        PathSummary: The summary including total distance, fare and time.
    """
    names = [network.name_of(number) for number in station_numbers]
    distance = int(sum(edge.distance for edge in edges))
    segments, transfers = _split_segments(names, edges)
    lines = []
    for segment in segments:
        if segment.line not in lines:
            lines.append(segment.line)
    fare = Sum_money(distance)
    estimated_time = calculate_estimated_time(distance, transfer_count)
    return PathSummary(
        start=start,
        end=end,
        mode=mode,
        max_transfers=max_transfers,
        preferred_lines=list(preferred_lines or []),
        stations=names,
        segments=segments,
        transfers=transfers,
        lines=lines,
        distance_m=distance,
        fare_yuan=fare,
        transfer_count=transfer_count,
        estimated_time_min=estimated_time,
    )


def format_text(summary):
    """Render a PathSummary as console-formatted text."""
    divider = "=" * 60
    out = [divider]
    out.append(f"Route: {summary.start} -> {summary.end}")
    constraints = f"mode={summary.mode}"
    if summary.max_transfers is not None:
        constraints += f", max transfers={summary.max_transfers}"
    if summary.preferred_lines:
        constraints += f", preferred lines={', '.join(summary.preferred_lines)}"
    out.append(f"Query settings: {constraints}")
    out.append(f"Station sequence ({len(summary.stations)} stations):")
    out.append("  " + " -> ".join(summary.stations))
    out.append(f"Total distance: {format_distance(summary.distance_m)}")
    out.append(f"Number of transfers: {summary.transfer_count}")
    out.append("Lines involved: " + (", ".join(summary.lines) if summary.lines else "(none)"))
    out.append("Route segments:")
    for index, segment in enumerate(summary.segments, 1):
        sections = max(len(segment.stations) - 1, 0)
        out.append(
            f"  {index}. [{segment.line}] {segment.stations[0]} -> {segment.stations[-1]} "
            f"({sections} sections, {format_distance(segment.distance_m)})"
        )
    out.append("Transfer stations:")
    if summary.transfers:
        for transfer in summary.transfers:
            out.append(f"  - {transfer.station}: {transfer.from_line} -> {transfer.to_line}")
    else:
        out.append("  (none)")
    out.append(f"Estimated travel time: {summary.estimated_time_min:.1f} min")
    out.append(f"Fare: {summary.fare_yuan} yuan")
    out.append(divider)
    return "\n".join(out)


def summary_to_dict(summary):
    """Convert a PathSummary into a JSON-serialisable dictionary."""
    return {
        "query": {
            "start": summary.start,
            "end": summary.end,
            "mode": summary.mode,
            "max_transfers": summary.max_transfers,
            "preferred_lines": summary.preferred_lines,
        },
        "path": list(summary.stations),
        "distance_m": summary.distance_m,
        "distance_km": round(summary.distance_m / 1000.0, 2),
        "fare_yuan": summary.fare_yuan,
        "estimated_time_min": round(summary.estimated_time_min, 2),
        "transfer_count": summary.transfer_count,
        "transfers": [
            {"station": t.station, "from_line": t.from_line, "to_line": t.to_line}
            for t in summary.transfers
        ],
        "lines": list(summary.lines),
        "segments": [
            {
                "line": seg.line,
                "from": seg.stations[0],
                "to": seg.stations[-1],
                "stations": list(seg.stations),
                "distance_m": seg.distance_m,
            }
            for seg in summary.segments
        ],
    }


def format_json(summary):
    """Render a PathSummary as a JSON document string."""
    return json.dumps(summary_to_dict(summary), ensure_ascii=False, indent=2)

"""Query handling: input parsing, station name standardisation and route computation."""

from fare_calculator import FareError
from pathfinding import NoPathError, PathfindingConfig, ShortestPath
from visualizer import build_summary

INPUT_FORMAT_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class QueryError(Exception):
    """Base class for user-facing query errors."""


class InputFormatError(QueryError):
    """The query text does not follow the 'Starting Station; Ending Station' format."""


class StationNotFoundError(QueryError):
    """A station name does not match any official station."""


class LogicError(QueryError):
    """The query is logically invalid (e.g. identical start and end stations)."""


class CalculationError(QueryError):
    """Path or fare calculation failed."""


def parse_query(text):
    """Split a query of the form 'Starting Station; Ending Station'.

    Returns:
        tuple: (starting name, ending name) as raw strings.

    Raises:
        InputFormatError: If the text does not contain exactly two non-empty parts.
    """
    normalized = (text or "").replace("\uff1b", ";").strip()
    parts = normalized.split(";")
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    return parts[0].strip(), parts[1].strip()


def resolve_station(network, raw_name, role):
    """Map user input to an official station name.

    Accepts exact names, names with or without the trailing "站" and
    case-insensitive matches.

    Args:
        network: SubwayNetwork.
        raw_name: User-provided name.
        role: "Starting" or "Ending" (used in the error message).

    Returns:
        str: The official station name.

    Raises:
        StationNotFoundError: If no station matches.
    """
    name = raw_name.strip()
    mapping = network.station_to_number
    candidates = [name]
    if name.endswith("站") and len(name) > 1:
        candidates.append(name[:-1])
    else:
        candidates.append(name + "站")
    for candidate in candidates:
        if candidate in mapping:
            return candidate
    lowered = {official.lower(): official for official in mapping}
    for candidate in candidates:
        if candidate.lower() in lowered:
            return lowered[candidate.lower()]
    raise StationNotFoundError(f"{role} station '{raw_name}' does not exist, please check the station name")


def parse_line_preference(text, known_lines):
    """Parse a comma-separated list of preferred lines.

    Args:
        text: Raw argument text such as "Line 1" or "Line 1,Line 2"; may be None.
        known_lines: All line names in the network.

    Returns:
        list[str]: Canonical line names (empty list when text is empty).

    Raises:
        QueryError: If a line name is unknown.
    """
    if not text or not text.strip():
        return []
    result = []
    for item in text.split(","):
        item = item.strip()
        if not item:
            continue
        match = None
        for line in known_lines:
            if line.lower() == item.lower():
                match = line
                break
        if match is None:
            raise QueryError(f"unknown line '{item}' in line preference")
        if match not in result:
            result.append(match)
    return result


def run_query(network, start_name, end_name, mode, max_transfers=None, preferred_lines=None):
    """Compute the route between two official station names.

    Args:
        network: SubwayNetwork.
        start_name: Official starting station name.
        end_name: Official ending station name.
        mode: Query mode (see pathfinding.VALID_MODES).
        max_transfers: Optional transfer limit.
        preferred_lines: Optional list of preferred line names.

    Returns:
        PathSummary: The route summary.

    Raises:
        LogicError: If start and end are the same station.
        CalculationError: If no route is found or calculation fails.
    """
    if start_name == end_name:
        raise LogicError("Starting station cannot be the same as the ending station")
    try:
        config = PathfindingConfig(mode=mode, max_transfers=max_transfers, preferred_lines=preferred_lines)
    except ValueError as exc:
        raise CalculationError(str(exc)) from exc

    source = network.number_of(start_name)
    target = network.number_of(end_name)
    try:
        search = ShortestPath(network.graph, source, config)
        if not search.has_path_to(target):
            raise NoPathError("no route")
        numbers, edges = search.path_with_edges(target)
        transfers = search.get_transfer_count(target)
    except NoPathError as exc:
        raise CalculationError(
            f"no route found from '{start_name}' to '{end_name}' under the given constraints"
        ) from exc
    except (ValueError, KeyError) as exc:
        raise CalculationError(f"path calculation failed: {exc}") from exc

    try:
        return build_summary(
            network, start_name, end_name, numbers, edges, transfers,
            mode, max_transfers, preferred_lines,
        )
    except (FareError, ValueError, KeyError) as exc:
        raise CalculationError(f"fare calculation failed: {exc}") from exc

def _write_result(args, summary, rendered, session):
    """Write one query result to the output file.

    In text mode results are appended to the file. In JSON mode the file holds
    a JSON array containing every result of this session.

    Returns:
        bool: True if the output file has been started during this session.
    """
    if args.format == "json":
        session["json_results"].append(summary_to_dict(summary))
        content = json.dumps(session["json_results"], ensure_ascii=False, indent=2) + "\n"
        write_output_file(args.output_file, content, append=False)
        return True
    content = rendered + "\n\n"
    write_output_file(args.output_file, content, append=session["output_started"])
    return True


def main(argv=None):
    """Entry point of the CLI.

    Args:
        argv: Optional list of command-line arguments (defaults to sys.argv[1:]).

    Returns:
        int: Process exit status.
    """
    parser = build_parser()
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")

    try:
        data = load_subway_data(DATA_FILE_PATH)
        network = SubwayNetwork.from_data(data)
    except (DataLoadError, NetworkBuildError) as exc:
        print(f"Error: failed to load subway data: {exc}")

    if args.check_data:
        report = DataManager(data).check_integrity()
        print(report.format())
        return 0 if report.passed else 1

    try:
        preferred_lines = parse_line_preference(args.line_preference, network.lines)
    except QueryError as exc:
        print(f"Error: {exc}")
        return 2

    print("Beijing Subway Fare Calculation System")
    print(f"Stations: {len(network.station_to_number)}, connections: {network.graph.num_edges}")
    for warning in network.warnings:
        print(f"Data warning: {warning}")

    session = {"output_started": False, "json_results": []}
    while True:
        try:
            raw = input(PROMPT)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in EXIT_WORDS:
            break

        try:
            start_raw, end_raw = parse_query(text)
            start = resolve_station(network, start_raw, "Starting")
            end = resolve_station(network, end_raw, "Ending")
            summary = run_query(network, start, end, args.mode, args.max_transfers, preferred_lines)
            rendered = render(summary, args.format)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # unexpected failure must not terminate the session
            print(f"Error: unexpected failure while processing the query: {exc}")
            continue

        print(rendered)
        if args.output_file:
            try:
                session["output_started"] = _write_result(args, summary, rendered, session)
            except OSError as exc:
                print(f"Error: cannot write output file: {exc}")
    return 0

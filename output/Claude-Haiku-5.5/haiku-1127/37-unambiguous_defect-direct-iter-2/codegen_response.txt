def normalize(text: str) -> str:
    """Return a canonical form of a name or group for case/space-insensitive comparison."""
    return " ".join((text or "").split()).casefold()

_SPLIT_RE = re.compile(r"[,\t]")


def _parse_lines(lines: Iterable[str]) -> tuple[list[User], list[str]]:
    """Parse 'Name,Department' lines. Returns (users, format error messages)."""
    users: list[User] = []
    errors: list[str] = []
    for lineno, raw in enumerate(lines, start=1):
        line = raw.strip().lstrip("\ufeff")
        if not line or line.startswith("#"):
            continue
        parts = [p.strip() for p in _SPLIT_RE.split(line)]
        if len(parts) != 2 or not parts[0] or not parts[1]:
            errors.append(
                f"Line {lineno}: Format Error - expected 'Name,Department', "
                f"got '{line}'"
            )
            continue
        users.append(User(name=parts[0], department=parts[1]))
    return users, errors


class ListManager:
    """Loads, verifies and summarises participant lists."""

    def __init__(self) -> None:
        self.users: list[User] = []

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return _parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list (one 'Name,Department' per line)."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it. Returns (users, errors)."""
        path_str = str(path) if path is not None else ""
        if not path_str or not os.path.isfile(path_str):
            return [], [f"File Error: file not found: '{path_str}'"]
        try:
            with open(path_str, "rb") as fh:
                raw = fh.read()
        except OSError as exc:
            return [], [f"File Error: cannot read '{path_str}': {exc}"]

        text = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                text = raw.decode(encoding)
                break
            except UnicodeDecodeError:
                continue
        if text is None:
            return [], [f"File Error: unable to decode '{path_str}' (use UTF-8 or GBK)"]
        return _parse_lines(text.splitlines())

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid first occurrences, duplicate later occurrences)."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of people, sorted by name."""
        canonical: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            key = normalize(user.department)
            display = canonical.setdefault(key, user.department)
            counts[display] = counts.get(display, 0) + 1
        return dict(sorted(counts.items()))

class PrizeManager:
    """Manages the ordered list of prizes for the event."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for spec in DEFAULT_PRIZES:
            try:
                self.add_prize(
                    spec["name"],
                    spec["quantity"],
                    list(spec["avoid_groups"]),
                    spec["associated_drawer"],
                )
            except ValueError as exc:
                errors.append(f"Default prize '{spec['name']}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a new prize. Raises ValueError on invalid input."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")
        if any(normalize(p.name) == normalize(name) for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = (group or "").strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order holds 1-based current positions in the new order."""
        n = len(self._prizes)
        if not isinstance(order, list) or len(order) != n:
            return False
        if any(isinstance(i, bool) or not isinstance(i, int) for i in order):
            return False
        if set(order) != set(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

"""Winner recording, grouping, sorting and CSV export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import date
from collections.abc import Iterable

from lottery_system.models import Prize, User, normalize

_LEVEL_WORDS = (("first", 1), ("second", 2), ("third", 3), ("fourth", 4), ("fifth", 5))
UNRANKED_LEVEL = 99


def prize_level_rank(name: str) -> int:
    """Return the level rank of a prize name (1 = highest). Unknown names rank last."""
    lowered = name.lower()
    for word, rank in _LEVEL_WORDS:
        if re.search(rf"\b{word}\b", lowered):
            return rank
    return UNRANKED_LEVEL


def sort_prizes_by_level(prizes: Iterable[Prize]) -> list[Prize]:
    """Sort prizes from high to low level, keeping original order for ties."""
    return sorted(prizes, key=lambda p: prize_level_rank(p.name))


@dataclass(frozen=True)
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    prize_order: int   # 1-based winning order within the same prize
    sequence: int      # 1-based winning order across all prizes


class ResultManager:
    """Keeps the real-time winner list and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    @property
    def winners(self) -> list[Winner]:
        return list(self._winners)

    def is_winner(self, user: User) -> bool:
        key = normalize(user.name)
        return any(normalize(w.user.name) == key for w in self._winners)

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner. Raises ValueError if the person already won."""
        if self.is_winner(user):
            raise ValueError(f"{user.name} has already won a prize.")
        prize_order = sum(1 for w in self._winners if w.prize_name == prize.name) + 1
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_order=prize_order,
            sequence=len(self._winners) + 1,
        )
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes: Iterable[Prize]) -> dict[str, list[Winner]]:
        """Map prize name to its winners (winning time order), ordered high to low level."""
        result: dict[str, list[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            if prize.name in result:
                continue
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: Iterable[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Prize Level", "Prize Name", "Winning Order", "Name", "Department"])
                for prize_name, winners in grouped.items():
                    rank = prize_level_rank(prize_name)
                    level = "" if rank == UNRANKED_LEVEL else str(rank)
                    for w in winners:
                        writer.writerow([level, prize_name, w.prize_order, w.user.name, w.user.department])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random
from collections.abc import Iterable

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    normalize,
)
from lottery_system.services.result_manager import ResultManager


def _is_cheng_ru(user: User) -> bool:
    return normalize(user.department) == normalize(CHENG_RU_GROUP)


class LotteryManager:
    """Runs lottery draws according to the rule engine."""

    def __init__(
        self,
        users: Iterable[User],
        result_manager: ResultManager | None = None,
        rng: random.Random | None = None,
    ) -> None:
        self.users: list[User] = list(users)
        self.result_manager: ResultManager = result_manager or ResultManager()
        # Per-round results of the lottery persons Chen Qixian / Jin Duo:
        # True means that round's winners contained a Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng or random.Random()

    # ----- rule engine -------------------------------------------------

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users who may still be drawn for the given prize."""
        taken = {normalize(w.user.name) for w in self.result_manager.winners}
        pool = [u for u in self.users if normalize(u.name) not in taken]

        drawer = normalize(prize.associated_drawer or "")

        # Special rule: Li Bin's prizes use relaxed filtering (no avoidance groups).
        if drawer == normalize(LI_BIN):
            return pool

        # Fallback (Cheng Ru group protection) for Ma Jiaqi's prize.
        if drawer == normalize(MA_JIAQI) and self.fallback_active():
            return [u for u in pool if _is_cheng_ru(u)]

        # Basic filtering: exclude members of the prize's avoidance groups.
        avoid = {normalize(g) for g in prize.avoid_groups}
        return [u for u in pool if normalize(u.department) not in avoid]

    def eligible_candidates(self, prize: Prize) -> list[User]:
        """Public wrapper returning the current eligible pool for a prize."""
        return self._get_eligible_pool(prize)

    def fallback_active(self) -> bool:
        """True when the last two lottery-person rounds both had no Cheng Ru member."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    # ----- draw process ------------------------------------------------

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None when no candidate remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.result_manager.add_winner(user, prize)
        return user

    def complete_round(self, prize: Prize, round_winners: Iterable[User]) -> None:
        """Update fallback state after a round finishes."""
        winners = list(round_winners)
        drawer = normalize(prize.associated_drawer or "")

        if drawer in (normalize(CHEN_QIXIAN), normalize(JIN_DUO)) and winners:
            self.last_two_draw_results.append(any(_is_cheng_ru(u) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == normalize(MA_JIAQI):
            # The protection check is consumed by Ma Jiaqi's round.
            self.last_two_draw_results = []

def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Interactive console application."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ----- helpers -----------------------------------------------------

    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    def _ask_int(self, prompt: str) -> int:
        while True:
            raw = self._ask(prompt).strip()
            if raw.isdigit() and int(raw) > 0:
                return int(raw)
            self._print("Please enter a positive integer.")

    def _menu(self, title: str, options: list[str], zero_label: str = "Back") -> int:
        self._print(f"\n--- {title} ---")
        for i, option in enumerate(options, start=1):
            self._print(f"  {i}. {option}")
        self._print(f"  0. {zero_label}")
        while True:
            raw = self._ask("Select an option: ").strip()
            if raw.isdigit() and int(raw) <= len(options):
                return int(raw)
            self._print("Invalid option, please try again.")

    def _group_names(self) -> list[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    # ----- main loop ---------------------------------------------------

    def run(self) -> None:
        self._print("=" * 60)
        self._print("  Company Team Building - Smart Lottery System")
        self._print("=" * 60)
        try:
            while True:
                choice = self._menu(
                    "Main Menu",
                    [
                        "Load participant list",
                        "View group information",
                        "Configure prizes",
                        "Start lottery",
                    ],
                    zero_label="Exit",
                )
                if choice == 0:
                    break
                if choice == 1:
                    self._load_list_menu()
                elif choice == 2:
                    self._show_groups()
                elif choice == 3:
                    self._prize_menu()
                elif choice == 4:
                    self._start_lottery()
        except (EOFError, KeyboardInterrupt):
            self._print("\nExiting lottery system.")
        self._print("Goodbye!")

    # ----- participant list --------------------------------------------

    def _load_list_menu(self) -> None:
        choice = self._menu(
            "Load Participant List",
            ["Built-in default team building list", "Custom txt list"],
        )
        if choice == 1:
            users, errors = self.list_manager.load_default_list()
            self._finalize_list(users, errors)
        elif choice == 2:
            path = self._ask("Enter path of the txt file (Name,Department per line): ")
            path = path.strip().strip('"').strip("'")
            if not path:
                self._print("No path entered.")
                return
            users, errors = self.list_manager.load_custom_list(path)
            self._finalize_list(users, errors)

    def _finalize_list(self, users: list[User], errors: list[str]) -> None:
        if errors:
            self._print(f"Found {len(errors)} invalid line(s):")
            for error in errors:
                self._print(f"  {error}")

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self._print(f"Found {len(duplicates)} duplicate name(s); first occurrence is kept:")
            for user in duplicates:
                self._print(f"  Duplicate: {user.name} ({user.department})")

        if errors or duplicates:
            while True:
                choice = self._ask(
                    "[r] Remove invalid/duplicate entries and continue, "
                    "[t] Terminate upload: "
                ).strip().lower()
                if choice in ("r", "t"):
                    break
                self._print("Please enter 'r' or 't'.")
            if choice == "t":
                self._print("Upload terminated. Previous participant list is kept.")
                return

        if not valid:
            self._print("No valid participants were loaded.")
            return

        self.users = valid
        self._print(f"Loaded {len(valid)} participant(s).")
        self._show_groups()

    def _show_groups(self) -> None:

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        if not name:
            self._print("Prize name cannot be empty.")

        quantity = self._ask_int("Quantity: ")

        try:

    def _ask_avoid_groups(self) -> list[str]:
        groups = self._group_names()
        if groups:
            self._print("Available groups:")
            for index, group in enumerate(groups, start=1):
                self._print(f"  {index}. {group}")
        else:
            self._print("No participant list loaded; enter group names directly.")
        raw = self._ask("Avoided groups (numbers or names, comma-separated; Enter for none): ").strip()
        if not raw:
            return []

        selected: list[str] = []
        for token in (t.strip() for t in raw.split(",")):
            if not token:
                continue
            if groups and token.isdigit() and 1 <= int(token) <= len(groups):
                selected.append(groups[int(token) - 1])
                continue
            match = next((g for g in groups if normalize(g) == normalize(token)), token)
            if groups and match not in groups:
                self._print(f"Unknown group ignored: {token}")
                continue
            if match not in selected:
                selected.append(match)
        return selected

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._print("\nLottery order:")

    def _adjust_order(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return

        else:

    # ----- lottery process ---------------------------------------------

    def _start_lottery(self) -> None:
        if not self.users:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure at least one prize first.")
            return

        self.lottery = LotteryManager(self.users)
        self._print("\nLottery starts!")
        self._print("Commands: [Enter] draw next | pause | continue | terminate (skip this round)")

        total = len(prizes)
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)

        self._show_summary(prizes)
        self._offer_export(prizes)

    def _describe_rules(self, prize: Prize, lottery: LotteryManager) -> list[str]:
        drawer = normalize(prize.associated_drawer or "")
        rules = ["Previous winners cannot win again."]
        if drawer == normalize(LI_BIN):
            rules.append("Lottery person Li Bin: relaxed rules, avoidance groups are not applied.")
        elif prize.avoid_groups:
            rules.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}")
        if drawer == normalize(MA_JIAQI) and lottery.fallback_active():
            rules.append("Cheng Ru protection active: candidates limited to Cheng Ru Group members.")
        return rules

    def _run_round(self, index: int, total: int, prize: Prize) -> None:
        lottery = self.lottery
        assert lottery is not None

        self._print("")
        self._print("=" * 60)
        self._print(f"Round {index}/{total}: {prize.name}")
        self._print(f"Quantity: {prize.quantity}")
        self._print("Rules:")
        for rule in self._describe_rules(prize, lottery):
            self._print(f"  - {rule}")
        self._print(f"Eligible candidates now: {len(lottery.eligible_candidates(prize))}")

        drawn: list[User] = []
        paused = False
        while len(drawn) < prize.quantity:
            if paused:
                cmd = self._ask("Paused. Type 'continue' to resume or 'terminate' to skip: ")
            else:
                cmd = self._ask("[Enter]=draw / pause / terminate: ")
            cmd = cmd.strip().lower()

            if cmd == "terminate":
                self._print("Round terminated by operator.")
                break

            if paused:
                if cmd != "continue":
                    self._print("Lottery is paused. Type 'continue' or 'terminate'.")
                    continue
                paused = False
                self._print("Lottery resumed.")
            else:
                if cmd == "pause":
                    paused = True
                    self._print("Lottery paused. Type 'continue' to resume or 'terminate' to skip.")
                    continue
                if cmd not in ("", "continue"):
                    self._print("Unknown command. Use Enter, pause, continue or terminate.")
                    continue

            position = len(drawn) + 1
            self._print(f"Currently drawing the {ordinal(position)} winner...")
            user = lottery.draw_one(prize)
            if user is None:
                self._print("No eligible candidates remain for this prize. Round ends.")
                break
            drawn.append(user)
            self._print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        lottery.complete_round(prize, drawn)

        self._print(f"\nRound {index} results ({prize.name}):")
        if drawn:
            for number, user in enumerate(drawn, start=1):
                self._print(f"  {number}. {user.name} ({user.department})")
        else:
            self._print("  (no winners)")
        remaining = prize.quantity - len(drawn)
        if remaining > 0:
            self._print(f"Note: {remaining} award(s) of this prize were not drawn.")

    def _show_summary(self, prizes: list[Prize]) -> None:
        assert self.lottery is not None
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        self._print("\n" + "=" * 60)
        self._print("FINAL WINNER LIST (by prize level, high to low)")
        self._print("=" * 60)
        for prize_name, winners in grouped.items():
            self._print(f"\n[{prize_name}]")
            if not winners:
                self._print("  (no winners)")
            for winner in winners:
                self._print(
                    f"  {winner.prize_order}. {winner.user.name} "
                    f"({winner.user.department}) - {winner.prize_name}"
                )

    def _offer_export(self, prizes: list[Prize]) -> None:
        assert self.lottery is not None
        choice = self._ask("Export winner list to a txt (CSV) file? [y/N]: ").strip().lower()
        if choice in ("y", "yes"):
            _, message = self.lottery.result_manager.export_to_csv(prizes)
            self._print(message)


def main() -> int:
    LotteryCLI().run()
    return 0

"""Loading of the Beijing Subway network dataset from JSON.

The dataset file lives next to this module (``src/subway_data.json``), so the
path resolves correctly regardless of the current working directory.
"""

import json
import os

DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path):
    """Load the subway dataset JSON file and return its content as a dict.

    Args:
        path: Path to the JSON file containing ``stations`` and ``connections``.

    Returns:
        dict: The parsed JSON object.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the JSON is malformed or not a JSON object.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError("subway data file must contain a JSON object at the top level")
    return data

"""Domain models for the Beijing Subway network: edges, graph and network.

Station numbers are integers assigned by ``SubwayNetwork`` (the caller of
``Graph``); ``Graph`` itself never invents numbers.
"""

from datetime import datetime


def _clean_text(value, field):
    """Normalize a text field: convert to str, collapse whitespace, reject empties."""
    if not isinstance(value, str):
        raise ValueError(f"field '{field}' must be a string, got {type(value).__name__}")
    cleaned = " ".join(value.split())
    if not cleaned:
        raise ValueError(f"field '{field}' must not be empty")
    return cleaned


class Edge:
    """An undirected, weighted connection between two stations on one line.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance in meters (also used as the weight).
        line: Name of the subway line this connection belongs to.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = float(distance)
        self.line = line

    @property
    def weight(self):
        """Edge weight used by shortest-path algorithms (equal to distance)."""
        return self.distance

    def other(self, node):
        """Return the endpoint opposite to ``node``.

        Raises:
            ValueError: If ``node`` is not an endpoint of this edge.
        """
        if node == self.source:
            return self.dest
        if node == self.dest:
            return self.source
        raise ValueError(f"node {node} is not an endpoint of edge {self!r}")

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph with adjacency lists.

    Attributes:
        station_lines: dict mapping station name -> list of line names.
    """

    def __init__(self):
        self._adjacency = {}
        self._edges = []
        self.station_lines = {}

    @property
    def node_count(self):
        """Number of nodes (stations) in the graph."""
        return len(self._adjacency)

    @property
    def edge_count(self):
        """Number of unique undirected edges in the graph."""
        return len(self._edges)

    def add_node(self, number):
        """Register a station number as a node (no-op if already present)."""
        self._adjacency.setdefault(number, [])

    def has_node(self, number):
        """Return True if the station number is a node of the graph."""
        return number in self._adjacency

    def set_station_lines(self, station_name, lines):
        """Record the (de-duplicated, order-preserving) lines serving a station."""
        unique = []
        for line in lines:
            if line not in unique:
                unique.append(line)
        self.station_lines[station_name] = unique

    def add_edge(self, edge):
        """Add an undirected edge.

        A duplicate (same endpoints and same line, in either direction) is
        ignored so that bidirectional listings in the data do not double count.

        Returns:
            bool: True if the edge was added, False if it was a duplicate.
        """
        self.add_node(edge.source)
        self.add_node(edge.dest)
        for existing in self._adjacency[edge.source]:
            if existing.line == edge.line and existing.other(edge.source) == edge.dest:
                return False
        self._adjacency[edge.source].append(edge)
        if edge.dest != edge.source:
            self._adjacency[edge.dest].append(edge)
        self._edges.append(edge)
        return True

    def neighbors(self, node):
        """Return the list of edges incident to ``node`` (empty if unknown)."""
        return self._adjacency.get(node, [])

    def nodes(self):
        """Return a list of all station numbers in the graph."""
        return list(self._adjacency)

    def edges(self):
        """Return a list of all unique edges in the graph."""
        return list(self._edges)


class SubwayNetwork:
    """The modeled Beijing Subway network built from the raw dataset.

    Attributes:
        source_file: Where the data was loaded from.
        loaded_at: Timestamp of construction.
        declared_stations: station name -> lines as declared in the "stations" list.
        connections: normalized list of connection dicts with keys
            source, dest, distance, line.
        station_lines: effective station name -> lines (declared + connection lines).
        station_to_number: station name -> station number.
        number_to_station: station number -> station name.
        lines: line name -> list of station names serving it.
        graph: the Graph built from the network.
    """

    def __init__(self, source_file="<memory>"):
        self.source_file = source_file
        self.loaded_at = datetime.now()
        self.declared_stations = {}
        self.connections = []
        self.station_lines = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self.lines = {}
        self.graph = Graph()
        self._casefold_stations = {}
        self._casefold_lines = {}

    @classmethod
    def from_data(cls, data, source_file="<memory>"):
        """Build a network from the dataset dict.

        Args:
            data: dict with ``stations`` ([{name, lines}]) and
                ``connections`` ([{source, dest, distance, line}]).
            source_file: Label recorded as the data source.

        Raises:
            ValueError: If the data structure or any field is invalid.
        """
        if not isinstance(data, dict):
            raise ValueError("subway data must be a JSON object")
        stations_raw = data.get("stations")
        connections_raw = data.get("connections")
        if not isinstance(stations_raw, list):
            raise ValueError("subway data must contain a 'stations' list")
        if not isinstance(connections_raw, list):
            raise ValueError("subway data must contain a 'connections' list")

        network = cls(source_file)
        for item in stations_raw:
            if not isinstance(item, dict) or "name" not in item:
                raise ValueError("each station entry must be an object with a 'name'")
            name = _clean_text(item["name"], "station name")
            raw_lines = item.get("lines", [])
            if not isinstance(raw_lines, list):
                raise ValueError(f"'lines' of station '{name}' must be a list")
            bucket = network.declared_stations.setdefault(name, [])
            for raw in raw_lines:
                line = _clean_text(raw, "line name")
                if line not in bucket:
                    bucket.append(line)

        for item in connections_raw:
            if not isinstance(item, dict):
                raise ValueError("each connection entry must be an object")
            for key in ("source", "dest", "distance", "line"):
                if key not in item:
                    raise ValueError(f"connection entry is missing field '{key}'")
            distance = item["distance"]
            if isinstance(distance, bool) or not isinstance(distance, (int, float)):
                raise ValueError("connection 'distance' must be a number of meters")
            network.connections.append({
                "source": _clean_text(item["source"], "source"),
                "dest": _clean_text(item["dest"], "dest"),
                "distance": distance,
                "line": _clean_text(item["line"], "line"),
            })

        network._build()
        return network

    def _build(self):
        """Merge declared and connection data, number stations and build the graph."""
        self.station_lines = {name: list(lines) for name, lines in self.declared_stations.items()}
        for conn in self.connections:
            for end in (conn["source"], conn["dest"]):
                bucket = self.station_lines.setdefault(end, [])
                if conn["line"] not in bucket:
                    bucket.append(conn["line"])

        # Station numbers are assigned here (the caller), not by Graph.
        for number, name in enumerate(self.station_lines):
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_node(number)
            self.graph.set_station_lines(name, self.station_lines[name])

        for conn in self.connections:
            self.graph.add_edge(Edge(
                self.station_to_number[conn["source"]],
                self.station_to_number[conn["dest"]],
                conn["distance"],
                conn["line"],
            ))

        self.lines = {}
        for name, lines in self.station_lines.items():
            for line in lines:
                self.lines.setdefault(line, [])
                self.lines[line].append(name)

        self._casefold_stations = {name.casefold(): name for name in self.station_lines}
        self._casefold_lines = {line.casefold(): line for line in self.lines}

    # ----- lookup helpers -------------------------------------------------

    def has_station(self, name):
        """Return True if the exact station name exists."""
        return name in self.station_to_number

    def find_station(self, name):
        """Standardize a user-entered station name to the official name.

        Matching is whitespace-tolerant and case-insensitive; a trailing
        Chinese "站" suffix is tolerated.

        Returns:
            The canonical station name, or None if no station matches.
        """
        cleaned = " ".join(str(name).split())
        if not cleaned:
            return None
        if cleaned in self.station_to_number:
            return cleaned
        key = cleaned.casefold()
        if key in self._casefold_stations:
            return self._casefold_stations[key]
        if cleaned.endswith("站") and len(cleaned) > 1:
            return self.find_station(cleaned[:-1])
        return None

    def find_line(self, name):
        """Standardize a user-entered line name; return canonical name or None."""
        cleaned = " ".join(str(name).split())
        if not cleaned:
            return None
        if cleaned in self.lines:
            return cleaned
        return self._casefold_lines.get(cleaned.casefold())

    def get_station_number(self, name):
        """Return the station number for an exact station name.

        Raises:
            KeyError: If the station does not exist.
        """
        return self.station_to_number[name]

    def get_station_name(self, number):
        """Return the station name for a station number."""
        return self.number_to_station[number]

    def lines_of_station(self, name):
        """Return the list of lines serving a station (empty if unknown)."""
        return list(self.station_lines.get(name, []))

    def station_names(self):
        """Return all station names sorted alphabetically."""
        return sorted(self.station_lines)

    def line_names(self):
        """Return all line names sorted alphabetically."""
        return sorted(self.lines)

    def transfer_stations(self):
        """Return names of stations served by more than one line."""
        return [name for name, lines in self.station_lines.items() if len(lines) > 1]

"""Shortest-path computation over the subway graph using Dijkstra's algorithm.

Search runs over states ``(station, line, transfers)``. A transfer happens
when the incoming edge's line differs from the line of the previous edge.
Tracking the transfer count in the state makes the maximum-transfer limit
exact.

Path modes:
    shortest_distance: minimize total distance (meters).
    least_transfer:    minimize number of transfers, then distance.
    comprehensive:     minimize distance + TRANSFER_PENALTY_METERS per transfer.

If preferred lines are configured, edges on other lines receive a soft
penalty of LINE_PREFERENCE_PENALTY_METERS in every mode.
"""

import math

from models import Edge  # noqa: F401  (re-exported for convenience)

PATH_MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY_METERS = 1000.0
LEAST_TRANSFER_WEIGHT = 1.0e9
LINE_PREFERENCE_PENALTY_METERS = 500.0


class MinPQ:
    """Binary min-heap priority queue.

    Entries are stored as ``(key, sequence, item)`` so that equal keys are
    broken by insertion order and items never need to be comparable.
    """

    def __init__(self):
        self.heap = []
        self._sequence = 0

    @property
    def size(self):
        """Number of entries currently in the queue."""
        return len(self.heap)

    def is_empty(self):
        """Return True if the queue has no entries."""
        return not self.heap

    def insert(self, key, item):
        """Insert an item with the given priority key (O(log n))."""
        self._sequence += 1
        self.heap.append((key, self._sequence, item))
        self._swim(len(self.heap) - 1)

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self.heap:
            raise IndexError("priority queue is empty")
        return self.heap[0][0]

    def del_min(self):
        """Remove and return ``(key, item)`` with the smallest key (O(log n))."""
        if not self.heap:
            raise IndexError("priority queue is empty")
        top = self.heap[0]
        last = self.heap.pop()
        if self.heap:
            self.heap[0] = last
            self._sink(0)
        return top[0], top[2]

    def _less(self, i, j):
        return (self.heap[i][0], self.heap[i][1]) < (self.heap[j][0], self.heap[j][1])

    def _swap(self, i, j):
        self.heap[i], self.heap[j] = self.heap[j], self.heap[i]

    def _swim(self, k):
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._swap(k, parent)
                k = parent
            else:
                break

    def _sink(self, k):
        n = len(self.heap)
        while True:
            left = 2 * k + 1
            right = left + 1
            smallest = k
            if left < n and self._less(left, smallest):
                smallest = left
            if right < n and self._less(right, smallest):
                smallest = right
            if smallest == k:
                break
            self._swap(k, smallest)
            k = smallest


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: one of PATH_MODES.
        max_transfers: maximum allowed transfers, or None for unlimited.
        preferred_lines: list of preferred line names (soft preference).
    """

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in PATH_MODES:
            raise ValueError(f"unknown path mode '{mode}', expected one of {', '.join(PATH_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer or None")
            if max_transfers < 0:
                raise ValueError("max_transfers must not be negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class ShortestPath:
    """Dijkstra-based path search from one source station.

    Attributes:
        algorithm: name of the algorithm ("Dijkstra").
        dist_to: dict station number -> physical distance (meters) of the best path.
        edge_to: dict station number -> predecessor station number on the best path.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_node(source):
            raise ValueError(f"station number {source} is not part of the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.algorithm = "Dijkstra"
        self.dist_to = {}
        self.edge_to = {}
        self._transfers_to = {}
        self._best_state = {}
        self._state_cost = {}
        self._state_dist = {}
        self._state_transfers = {}
        self._state_prev = {}
        self._search()
        self._collect_results()

    def _edge_cost(self, edge, switched):
        """Return the search cost of traversing ``edge``."""
        cost = edge.distance
        if self.config.preferred_lines and edge.line not in self.config.preferred_lines:
            cost += LINE_PREFERENCE_PENALTY_METERS
        if self.config.mode == "least_transfer":
            cost += LEAST_TRANSFER_WEIGHT * switched
        elif self.config.mode == "comprehensive":
            cost += TRANSFER_PENALTY_METERS * switched
        return cost

    def _search(self):
        """Run Dijkstra's algorithm over (station, line, transfers) states."""
        start = (self.source, None, 0)
        self._state_cost[start] = 0.0
        self._state_dist[start] = 0.0
        self._state_transfers[start] = 0

        pq = MinPQ()
        pq.insert(0.0, start)
        settled = set()
        max_transfers = self.config.max_transfers

        while not pq.is_empty():
            cost, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)
            node, line, transfers = state
            dist = self._state_dist[state]

            for edge in self.graph.neighbors(node):
                nxt = edge.other(node)
                switched = 1 if (line is not None and edge.line != line) else 0
                new_transfers = transfers + switched
                if max_transfers is not None and new_transfers > max_transfers:
                    continue
                next_state = (nxt, edge.line, new_transfers)
                if next_state in settled:
                    continue
                new_cost = cost + self._edge_cost(edge, switched)
                if next_state not in self._state_cost or new_cost < self._state_cost[next_state]:
                    self._state_cost[next_state] = new_cost
                    self._state_dist[next_state] = dist + edge.distance
                    self._state_transfers[next_state] = new_transfers
                    self._state_prev[next_state] = (state, edge)
                    pq.insert(new_cost, next_state)

    def _collect_results(self):
        """Select the best state per station and fill the public result maps."""
        best = {}
        for state, cost in self._state_cost.items():
            node = state[0]
            if node not in best or cost < self._state_cost[best[node]]:
                best[node] = state
        self._best_state = best
        for node, state in best.items():
            self.dist_to[node] = self._state_dist[state]
            self._transfers_to[node] = self._state_transfers[state]
            if state in self._state_prev:
                self.edge_to[node] = self._state_prev[state][0][0]

    def has_path_to(self, target):
        """Return True if ``target`` is reachable from the source."""
        return target in self._best_state

    def get_distance(self, target):
        """Return the distance in meters to ``target`` (math.inf if unreachable)."""
        return self.dist_to.get(target, math.inf)

    def get_transfer_count(self, target):
        """Return the number of transfers on the best path to ``target`` (0 if unreachable)."""
        return self._transfers_to.get(target, 0)

    def path_with_edges(self, target):
        """Reconstruct the best path to ``target``.

        Returns:
            (list of station numbers from source to target, list of Edge objects
            in traversal order). Both lists are empty if the target is unreachable.
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        nodes = [state[0]]
        edges = []
        while state in self._state_prev:
            prev_state, edge = self._state_prev[state]
            edges.append(edge)
            nodes.append(prev_state[0])
            state = prev_state
        nodes.reverse()
        edges.reverse()
        return nodes, edges

"""Beijing Subway fare calculation.

Fare tiers (left-closed, right-open intervals):
    [0, 6) km      -> 3 yuan
    [6, 12) km     -> 4 yuan
    [12, 22) km    -> 5 yuan
    [22, 32) km    -> 6 yuan
    [32, ...) km   -> 7 yuan, +1 yuan for every further 20 km
Fractional meters are rounded up (ceiling) before tier lookup.
"""

import math
from dataclasses import dataclass
from typing import Optional

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


@dataclass(frozen=True)
class FareTier:
    """One fare tier.

    Attributes:
        min_distance_km: inclusive lower bound in kilometers.
        max_distance_km: exclusive upper bound, or None for the open-ended tier.
        fare_yuan: fare charged at the lower bound of the tier.
        extension_step_km: for the extended tier, km per additional yuan.
    """

    min_distance_km: float
    max_distance_km: Optional[float]
    fare_yuan: int
    extension_step_km: Optional[float] = None


FARE_TIERS = (
    FareTier(0.0, 6.0, 3),
    FareTier(6.0, 12.0, 4),
    FareTier(12.0, 22.0, 5),
    FareTier(22.0, 32.0, 6),
    FareTier(32.0, None, 7, 20.0),
)


def _km_to_meters(km):
    """Convert kilometers to integer meters without float drift."""
    return int(round(km * 1000))


def _normalize_distance(distance):
    """Validate a distance in meters and round it up to a whole meter.

    Raises:
        ValueError: If the value is not a finite non-negative number.
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("distance must be a number of meters")
    value = float(distance)
    if not math.isfinite(value):
        raise ValueError("distance must be finite")
    if value < 0:
        raise ValueError("distance must not be negative")
    return int(math.ceil(value))


def Sum_money(distance):
    """Return the subway fare in yuan for a travelled distance.

    Args:
        distance: travelled distance in meters.

    Returns:
        int: fare in yuan.

    Raises:
        ValueError: If the distance is invalid.
    """
    meters = _normalize_distance(distance)
    for tier in FARE_TIERS:
        min_m = _km_to_meters(tier.min_distance_km)
        if tier.max_distance_km is None:
            step_m = _km_to_meters(tier.extension_step_km)
            return tier.fare_yuan + (meters - min_m) // step_m
        if meters < _km_to_meters(tier.max_distance_km):
            return tier.fare_yuan
    raise ValueError("no fare tier matches the distance")  # pragma: no cover


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes.

    Args:
        distance: travelled distance in meters.
        transfer_count: number of transfers.

    Returns:
        float: estimated minutes (about 1.5 min per km plus 5 min per transfer).

    Raises:
        ValueError: If inputs are invalid.
    """
    meters = _normalize_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int):
        raise ValueError("transfer_count must be an integer")
    if transfer_count < 0:
        raise ValueError("transfer_count must not be negative")
    return (meters / 1000.0) * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""Route planning service: combines the network, path search and fare logic."""

from dataclasses import dataclass, field
from typing import List, Optional

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


class PathNotFoundError(Exception):
    """Raised when no route satisfies the query constraints."""


class FareCalculationError(Exception):
    """Raised when the fare cannot be calculated for a route."""


@dataclass
class RouteSegment:
    """A contiguous ride on one line."""

    line_name: str
    start_station: str
    end_station: str
    distance_m: float
    station_sequence: List[str] = field(default_factory=list)


@dataclass
class RouteTransfer:
    """A line change at a station."""

    station: str
    from_line: str
    to_line: str


@dataclass
class PathSummary:
    """Key figures extracted from a route."""

    total_distance_km: float
    total_fare_yuan: int
    transfer_count: int
    estimated_time_minutes: float


@dataclass
class Route:
    """Complete result of a route query."""

    start_station: str
    end_station: str
    mode: str
    max_transfers: Optional[int]
    line_preference: Optional[str]
    station_sequence: List[str]
    total_distance_m: float
    transfer_count: int
    segments: List[RouteSegment]
    transfers: List[RouteTransfer]
    lines_involved: List[str]
    fare_yuan: int
    summary: PathSummary


def plan_route(network, start_name, end_name, mode="shortest_distance",
               max_transfers=None, line_preference=None):
    """Compute the route, fare and summary between two stations.

    Args:
        network: SubwayNetwork instance.
        start_name: canonical starting station name.
        end_name: canonical ending station name.
        mode: one of pathfinding.PATH_MODES.
        max_transfers: optional maximum number of transfers.
        line_preference: optional preferred line name.

    Returns:
        Route

    Raises:
        ValueError: For invalid or identical stations / invalid configuration.
        PathNotFoundError: If no route satisfies the constraints.
        FareCalculationError: If the fare cannot be computed.
    """
    if start_name == end_name:
        raise ValueError("Starting station cannot be the same as the ending station")
    if start_name not in network.station_to_number:
        raise ValueError(f"unknown starting station '{start_name}'")
    if end_name not in network.station_to_number:
        raise ValueError(f"unknown ending station '{end_name}'")

    preferred = [line_preference] if line_preference else []
    config = PathfindingConfig(mode, max_transfers, preferred)
    source = network.get_station_number(start_name)
    target = network.get_station_number(end_name)
    search = ShortestPath(network.graph, source, config)

    if not search.has_path_to(target):
        limit = "" if max_transfers is None else f" within {max_transfers} transfer(s)"
        raise PathNotFoundError(f"No route found from '{start_name}' to '{end_name}'{limit}")

    nodes, edges = search.path_with_edges(target)
    if not edges:
        raise ValueError("Starting station cannot be the same as the ending station")

    names = [network.get_station_name(n) for n in nodes]

    # Group consecutive edges on the same line into segments.
    segments = []
    i = 0
    while i < len(edges):
        line = edges[i].line
        j = i
        distance = 0.0
        while j < len(edges) and edges[j].line == line:
            distance += edges[j].distance
            j += 1
        segments.append(RouteSegment(
            line_name=line,
            start_station=names[i],
            end_station=names[j],
            distance_m=distance,
            station_sequence=names[i:j + 1],
        ))
        i = j

    transfers = []
    for k in range(1, len(segments)):
        transfers.append(RouteTransfer(
            station=segments[k].start_station,
            from_line=segments[k - 1].line_name,
            to_line=segments[k].line_name,
        ))

    lines_involved = []
    for segment in segments:
        if segment.line_name not in lines_involved:
            lines_involved.append(segment.line_name)

    total_distance = sum(edge.distance for edge in edges)
    transfer_count = len(transfers)

    try:
        fare = Sum_money(total_distance)
    except (ValueError, TypeError) as exc:
        raise FareCalculationError(str(exc)) from exc

    try:
        estimated = calculate_estimated_time(total_distance, transfer_count)
    except (ValueError, TypeError) as exc:
        raise FareCalculationError(f"time estimation failed: {exc}") from exc

    summary = PathSummary(
        total_distance_km=total_distance / 1000.0,
        total_fare_yuan=fare,
        transfer_count=transfer_count,
        estimated_time_minutes=estimated,
    )

    return Route(
        start_station=start_name,
        end_station=end_name,
        mode=mode,
        max_transfers=max_transfers,
        line_preference=line_preference,
        station_sequence=names,
        total_distance_m=total_distance,
        transfer_count=transfer_count,
        segments=segments,
        transfers=transfers,
        lines_involved=lines_involved,
        fare_yuan=fare,
        summary=summary,
    )

        "fare_yuan": route.fare_yuan,
        "estimated_time_minutes": round(route.summary.estimated_time_minutes, 1),

def render_text(route):
    """Render a Route as human-readable console text."""
    lines = [BAR, "Beijing Subway Fare Report", BAR]
    lines.append(f"Query: {route.start_station} -> {route.end_station}")
    lines.append(f"Mode: {route.mode}")
    lines.append("Max transfers: " + ("unlimited" if route.max_transfers is None else str(route.max_transfers)))
    lines.append("Line preference: " + (route.line_preference or "none"))
    lines.append("")

    lines.append(f"Station sequence ({len(route.station_sequence)} stations):")
    for index, name in enumerate(route.station_sequence, start=1):
        lines.append(f"  {index:>3}. {name}")
    lines.append("")

    lines.append("Line segments:")
    for seg in route.segments:
        stops = len(seg.station_sequence) - 1
        lines.append(
            f"  [{seg.line_name}] {seg.start_station} -> {seg.end_station} "
            f"({format_distance(seg.distance_m)}, {stops} stop(s))"
        )
    lines.append("")

    if route.transfers:
        lines.append(f"Transfers ({len(route.transfers)}):")
        for tr in route.transfers:
            lines.append(f"  - {tr.station}: {tr.from_line} -> {tr.to_line}")
    else:
        lines.append("Transfers: none")
    lines.append("Lines involved: " + ", ".join(route.lines_involved))
    lines.append("")

    lines.append(f"Total distance: {format_distance(route.total_distance_m)}")
    lines.append(f"Transfer count: {route.transfer_count}")
    lines.append(f"Estimated time: {route.summary.estimated_time_minutes:.1f} minutes")
    lines.append(f"Fare: {route.fare_yuan} yuan")
    lines.append(BAR)
    return "\n".join(lines)


def render_json(route):
    """Render a Route as an indented JSON string."""
    return json.dumps(route_to_dict(route), ensure_ascii=False, indent=2)

        return PathReport(format="text", content=render_text(route))
    if fmt == "json":
        return PathReport(format="json", content=render_json(route))

"""Data integrity checks and data update operations for the subway dataset.

Update functions never mutate their input: they return a new dataset dict,
which can be turned into a SubwayNetwork via ``SubwayNetwork.from_data``.
"""

import copy
from dataclasses import dataclass, field
from typing import List

MAIN_LINES = (
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
)
MIN_PLAUSIBLE_DISTANCE_M = 50
MAX_PLAUSIBLE_DISTANCE_M = 10000


@dataclass
class DataIntegrityReport:
    """Result of an integrity check.

    Attributes:
        passed: True when no issues were found.
        issues: human-readable description of each problem.
    """

    passed: bool
    issues: List[str] = field(default_factory=list)


def check_data_integrity(network):
    """Run coverage, connection and distance checks on a SubwayNetwork.

    Returns:
        DataIntegrityReport
    """
    issues = []

    # 1. Coverage of main lines.
    for line in MAIN_LINES:
        if line not in network.lines:
            issues.append(f"Coverage: main line '{line}' is missing from the network")
        elif len(network.lines[line]) < 2:
            issues.append(f"Coverage: main line '{line}' has fewer than 2 stations")

    # 2. Connection integrity and distance accuracy.
    declared = network.declared_stations
    seen = {}
    connected = set()
    for conn in network.connections:
        src, dst = conn["source"], conn["dest"]
        dist, line = conn["distance"], conn["line"]
        connected.add(src)
        connected.add(dst)

        for end in (src, dst):
            if end not in declared:
                issues.append(f"Connection: station '{end}' is referenced but not declared in stations")
        if src == dst:
            issues.append(f"Connection: self-loop on station '{src}' (line '{line}')")
        if not isinstance(dist, (int, float)) or dist <= 0:
            issues.append(f"Accuracy: non-positive distance {dist} between '{src}' and '{dst}'")
        elif dist < MIN_PLAUSIBLE_DISTANCE_M or dist > MAX_PLAUSIBLE_DISTANCE_M:
            issues.append(
                f"Accuracy: implausible distance {dist} m between '{src}' and '{dst}' on '{line}'"
            )

        for end in (src, dst):
            if line not in declared.get(end, []):
                issues.append(f"Connection: line '{line}' is not listed for station '{end}'")

        key = (frozenset((src, dst)), line)
        if key in seen and seen[key] != dist:
            issues.append(
                f"Connection: conflicting distances {seen[key]} and {dist} between "
                f"'{src}' and '{dst}' on '{line}'"
            )
        seen.setdefault(key, dist)

    # 3. Stations without any connection.
    for name in declared:
        if name not in connected:
            issues.append(f"Connection: station '{name}' has no connections")

    # 4. Network connectivity.
    graph = network.graph
    if network.station_to_number:
        start = next(iter(network.station_to_number.values()))
        reached = {start}
        stack = [start]
        while stack:
            node = stack.pop()
            for edge in graph.neighbors(node):
                other = edge.other(node)
                if other not in reached:
                    reached.add(other)
                    stack.append(other)
        if len(reached) != graph.node_count:
            issues.append(
                f"Connectivity: {graph.node_count - len(reached)} station(s) unreachable "
                f"from '{network.get_station_name(start)}'"
            )

    return DataIntegrityReport(passed=not issues, issues=issues)


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("distance must be a number of meters")
    if distance <= 0:
        raise ValueError("distance must be positive")
    return int(distance)


def add_line(data, line_name, station_names, distances):
    """Return a new dataset with an additional line.

    Args:
        data: current dataset dict.
        line_name: name of the new line (must not already exist).
        station_names: ordered list of station names on the line.
        distances: distances in meters between consecutive stations
            (len(station_names) - 1 values).

    Raises:
        ValueError: On invalid input or duplicate line name.
    """
    line_name = " ".join(str(line_name).split())
    if not line_name:
        raise ValueError("line name must not be empty")
    if len(station_names) < 2:
        raise ValueError("a line needs at least two stations")
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances must have one value per consecutive station pair")
    for conn in data.get("connections", []):
        if conn.get("line") == line_name:
            raise ValueError(f"line '{line_name}' already exists")

    new_data = copy.deepcopy(data)
    stations = new_data.setdefault("stations", [])
    index = {item["name"]: item for item in stations}
    for name in station_names:
        if name not in index:
            entry = {"name": name, "lines": [line_name]}
            stations.append(entry)
            index[name] = entry
        elif line_name not in index[name].setdefault("lines", []):
            index[name]["lines"].append(line_name)

    connections = new_data.setdefault("connections", [])
    for i, dist in enumerate(distances):
        connections.append({
            "source": station_names[i],
            "dest": station_names[i + 1],
            "distance": _validate_distance(dist),
            "line": line_name,
        })
    return new_data


def modify_station(data, old_name, new_name=None, lines=None):
    """Return a new dataset with a station renamed and/or its lines replaced.

    Raises:
        KeyError: If the station does not exist.
        ValueError: If the new name is already used by another station.
    """
    new_data = copy.deepcopy(data)
    stations = new_data.get("stations", [])
    target = next((s for s in stations if s.get("name") == old_name), None)
    if target is None:
        raise KeyError(f"station '{old_name}' does not exist")

    if new_name and new_name != old_name:
        if any(s.get("name") == new_name for s in stations):
            raise ValueError(f"station '{new_name}' already exists")
        target["name"] = new_name
        for conn in new_data.get("connections", []):
            if conn.get("source") == old_name:
                conn["source"] = new_name
            if conn.get("dest") == old_name:
                conn["dest"] = new_name
    if lines is not None:
        unique = []
        for line in lines:
            if line not in unique:
                unique.append(line)
        target["lines"] = unique
    return new_data


def update_distance(data, station_a, station_b, line, distance):
    """Return a new dataset with the distance of one connection updated.

    Raises:
        KeyError: If no matching connection exists.
        ValueError: If the distance is invalid.
    """
    value = _validate_distance(distance)
    new_data = copy.deepcopy(data)
    found = False
    for conn in new_data.get("connections", []):
        if conn.get("line") == line and {conn.get("source"), conn.get("dest")} == {station_a, station_b}:
            conn["distance"] = value
            found = True
    if not found:
        raise KeyError(f"no connection on '{line}' between '{station_a}' and '{station_b}'")
    return new_data

@dataclass
class CommandLineOptions:
    """Options given on the command line.

    Attributes:
        output_file: optional export file path.
        format: "text" or "json".
        mode: path query mode.

class ResultExporter:
    """Exports query results to a file.

    Text results are appended to the file. JSON results are kept as a list
    for the session and the file is rewritten so it always holds valid JSON.
    """

    def __init__(self, path, fmt):
        self.path = path
        self.fmt = fmt
        self._json_results = []

    def export(self, route, report):
        """Write one query result to the export file."""
        if self.fmt == "json":
            self._json_results.append(route_to_dict(route))
            with open(self.path, "w", encoding="utf-8") as handle:
                json.dump(self._json_results, handle, ensure_ascii=False, indent=2)
        else:
            with open(self.path, "a", encoding="utf-8") as handle:
                handle.write(report.content + "\n\n")


def parse_args(argv=None):
    """Parse command-line arguments (exits with status 2 on invalid arguments)."""
    parser = argparse.ArgumentParser(

    )
    parser.add_argument("--output_file", default=None,

    parser.add_argument("--format", choices=["text", "json"], default="text",

    parser.add_argument("--mode", choices=list(PATH_MODES), default="shortest_distance",
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=int, default=None,

    parser.add_argument("--line_preference", default=None,
                        help="preferred line name, e.g. 'Line 1'")
    parser.add_argument("--check_data", action="store_true",
                        help="run data integrity check and exit")
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:

def parse_query_text(text):
    """Split 'Starting Station; Ending Station' into two raw names.

    Raises:
        QueryInputError: If the format is wrong.
    """
    parts = text.replace("\uff1b", ";").split(";")
    if len(parts) != 2:
        raise QueryInputError(FORMAT_ERROR_MESSAGE)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryInputError(FORMAT_ERROR_MESSAGE)
    return start, end


def validate_query(text, network):
    """Validate a query string and return canonical (start, end) station names.

    Raises:
        QueryInputError: With a user-facing message on any validation failure.
    """
    start_raw, end_raw = parse_query_text(text)
    start = network.find_station(start_raw)
    if start is None:
        raise QueryInputError(
            f"Starting station '{start_raw}' does not exist, please check the station name"
        )
    end = network.find_station(end_raw)
    if end is None:
        raise QueryInputError(
            f"Ending station '{end_raw}' does not exist, please check the station name"
        )
    if start == end:
        raise QueryInputError("Starting station cannot be the same as the ending station")
    return start, end


def run_query_loop(network, options, input_func=input, stdout=None, stderr=None):
    """Repeatedly read queries, compute routes and print reports until quit/EOF.

    Returns:
        int: exit status (0).
    """
    out = stdout if stdout is not None else sys.stdout
    err = stderr if stderr is not None else sys.stderr
    exporter = ResultExporter(options.output_file, options.format) if options.output_file else None

    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' "
          "(or 'quit' to exit).", file=out)
    while True:
        try:
            raw = input_func("Query> ")
        except (EOFError, KeyboardInterrupt):
            print(file=out)
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break

        try:
            start, end = validate_query(text, network)
        except QueryInputError as exc:
            print(f"Error: {exc}", file=err)
            continue

        try:
            route = plan_route(
                network, start, end,
                mode=options.mode,
                max_transfers=options.max_transfers,
                line_preference=options.line_preference,
            )
        except PathNotFoundError as exc:
            print(f"Error: {exc}", file=err)
            continue
        except FareCalculationError as exc:
            print(f"Fare calculation error: {exc}", file=err)
            continue
        except (ValueError, KeyError) as exc:
            print(f"Path calculation error: {exc}", file=err)
            continue
        except Exception as exc:  # last-resort guard so the session keeps running
            print(f"Unexpected error while calculating the route: {exc}", file=err)
            continue

        try:
            report = build_report(route, options.format)
        except ValueError as exc:
            print(f"Report error: {exc}", file=err)
            continue

        print(report.content, file=out)
        print(file=out)

        if exporter is not None:
            try:
                exporter.export(route, report)
                print(f"Result exported to {options.output_file}", file=out)
            except OSError as exc:
                print(f"Export error: could not write '{options.output_file}': {exc}", file=err)
    return 0


def main(argv=None, input_func=input, stdout=None, stderr=None):
    """CLI entry point. Returns the process exit status."""
    out = stdout if stdout is not None else sys.stdout
    err = stderr if stderr is not None else sys.stderr
    args = parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
        network = SubwayNetwork.from_data(data, DATA_FILE_PATH)
    except (OSError, ValueError) as exc:
        print(f"Failed to load subway data: {exc}", file=err)
        return 1

    if args.check_data:
        report = check_data_integrity(network)
        if report.passed:
            print("Data integrity check: PASSED", file=out)
        else:
            print("Data integrity check: FAILED", file=out)
            for issue in report.issues:
                print(f"  - {issue}", file=out)
        return 0 if report.passed else 1

    line_preference = None
    if args.line_preference is not None:
        line_preference = network.find_line(args.line_preference)
        if line_preference is None:
            print(f"Error: line '{args.line_preference}' does not exist. "
                  f"Available lines: {', '.join(network.line_names())}", file=err)
            return 2

    options = CommandLineOptions(
        output_file=args.output_file,
        format=args.format,
        mode=args.mode,
        max_transfers=args.max_transfers,
        line_preference=line_preference,
        check_data=False,
    )
    return run_query_loop(network, options, input_func=input_func, stdout=out, stderr=err)

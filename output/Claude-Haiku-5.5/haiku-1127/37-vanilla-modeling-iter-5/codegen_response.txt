from __future__ import annotations

from dataclasses import dataclass, field

DEFAULT_PRIZE_LEVEL = 99


def normalize_key(text) -> str:
    """Normalize a name/department/group for case-insensitive comparison."""
    if text is None:
        return ""
    return " ".join(str(text).split()).casefold()


def infer_prize_level(name: str) -> int:
    """Infer the prize level (1 = highest) from the prize name.

    Custom prizes without a recognised level keyword get a low priority
    (large number) so they are ordered after the preset levels.
    """
    lowered = normalize_key(name)
    if "first" in lowered:
        return 1
    if "second" in lowered:
        return 2
    if "third" in lowered:
        return 3
    return DEFAULT_PRIZE_LEVEL


@dataclass(frozen=True)
class User:
    """A participant with a name and department group."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that can be drawn in a lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None
    level: int = 0

    def __post_init__(self) -> None:
        self.avoid_groups = list(self.avoid_groups or [])
        if self.level <= 0:
            self.level = infer_prize_level(self.name)

import os
import re
from datetime import datetime

from ..config import DEFAULT_PARTICIPANTS_TEXT
from ..models import User

_FIELD_SEPARATOR = re.compile(r"[,，]")


class ListManager:
    """Loads participant lists from the built-in default or a txt file."""

    def __init__(self) -> None:
        self.source_type: str | None = None
        self.file_path: str | None = None
        self.loaded_at: datetime | None = None

    def load_default_list(self) -> tuple[list[User], list[str]]:
        users, errors = self._parse_lines(DEFAULT_PARTICIPANTS_TEXT.splitlines())
        self.source_type = "default"
        self.file_path = None
        self.loaded_at = datetime.now()
        return users, errors

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except UnicodeDecodeError:
            return [], [f"File Error: '{path}' is not a valid UTF-8 text file"]
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        self.source_type = "custom"
        self.file_path = path
        self.loaded_at = datetime.now()
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = _FIELD_SEPARATOR.split(text)
            if len(parts) != 2:
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department' "
                    f"but got '{text}'"
                )
                continue
            name, department = parts[0].strip(), parts[1].strip()
            if not name or not department:
                errors.append(
                    f"Line {line_no}: Format Error - name and department must not be empty"
                )
                continue
            users.append(User(name=name, department=department))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). Only the first occurrence of a name is valid."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items(), key=lambda item: item[0].casefold()))

from ..config import DEFAULT_PRIZES

class PrizeManager:
    """Manages the ordered prize library. Starts with no prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = normalize_key(name)
        for prize in self._prizes:
            if normalize_key(prize.name) == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for spec in DEFAULT_PRIZES:
            if self._find(spec["name"]) is not None:
                errors.append(f"Prize '{spec['name']}' already exists, skipped.")
                continue
            self._prizes.append(
                Prize(
                    name=spec["name"],
                    quantity=spec["quantity"],
                    avoid_groups=list(spec["avoid_groups"]),
                    associated_drawer=spec["associated_drawer"],
                )
            )
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = (name or "").strip()
        if not name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        if self._find(name) is not None:
            raise ValueError(f"prize '{name}' already exists")

        groups = []
        for group in avoid_groups or []:
            cleaned = str(group).strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer = associated_drawer.strip() if associated_drawer else None
        drawer = drawer or None

        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def update_avoid_groups(self, name, avoid_groups) -> bool:
        prize = self._find(name)
        if prize is None:
            return False
        groups = []
        for group in avoid_groups or []:
            cleaned = str(group).strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)
        prize.avoid_groups = groups
        return True

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        n = len(self._prizes)
        if not isinstance(order, (list, tuple)) or len(order) != n:
            return False
        if any(isinstance(x, bool) or not isinstance(x, int) for x in order):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

import random

from ..config import (
    FALLBACK_DRAWER,
    FALLBACK_HISTORY_SIZE,
    FALLBACK_TRIGGER_DRAWERS,
    PROTECTED_GROUP,
    SPECIAL_RULE_DRAWER,
)
from ..models import Prize, User, normalize_key


def _is_same(a, b) -> bool:
    return a is not None and b is not None and normalize_key(a) == normalize_key(b)


class LotteryManager:
    """Runs lottery draws for a fixed participant list while enforcing rules."""

    def __init__(self, users) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self.winners: list[User] = []
        # Results of the most recent draws by Chen Qixian / Jin Duo:
        # True means the round contained at least one Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []

    # ----- helpers -------------------------------------------------------
    @staticmethod
    def _is_protected(user: User) -> bool:
        return normalize_key(user.department) == normalize_key(PROTECTED_GROUP)

    def has_won(self, user: User) -> bool:
        return normalize_key(user.name) in self._winner_keys

    def is_fallback_active(self, prize: Prize) -> bool:
        """Fallback applies when the prize is drawn by Ma Jiaqi and neither of
        the two most recent Chen Qixian / Jin Duo rounds contained Cheng Ru members."""
        return (
            _is_same(prize.associated_drawer, FALLBACK_DRAWER)
            and len(self.last_two_draw_results) >= FALLBACK_HISTORY_SIZE
            and not any(self.last_two_draw_results[-FALLBACK_HISTORY_SIZE:])
        )

    # ----- rule engine ---------------------------------------------------
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        # Anti-duplication: previous winners are always excluded.
        pool = [user for user in self._users if not self.has_won(user)]

        # Special rule for Li Bin: only previous winners are filtered.
        if _is_same(prize.associated_drawer, SPECIAL_RULE_DRAWER):
            return pool

        # Fallback (Cheng Ru group protection): restrict to Cheng Ru members,
        # even if that group is an avoidance group of this prize.
        if self.is_fallback_active(prize):
            return [user for user in pool if self._is_protected(user)]

        # Basic filtering: exclude members of avoidance groups.
        avoid = {normalize_key(group) for group in prize.avoid_groups}
        return [user for user in pool if normalize_key(user.department) not in avoid]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        return self._get_eligible_pool(prize)

    def describe_rule(self, prize: Prize) -> str:
        drawer = prize.associated_drawer or "None"
        parts = [f"Lottery person: {drawer}"]
        if _is_same(prize.associated_drawer, SPECIAL_RULE_DRAWER):
            parts.append("only previous winners are excluded (special rule, avoid groups not applied)")
        elif self.is_fallback_active(prize):
            parts.append(
                f"previous winners excluded; fallback active: only '{PROTECTED_GROUP}' members "
                "eligible (avoid groups overridden)"
            )
        else:
            parts.append("previous winners excluded")
            if prize.avoid_groups:
                parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
            else:
                parts.append("no avoid groups")
        return "; ".join(parts)

    # ----- drawing -------------------------------------------------------
    def draw_one(self, prize: Prize) -> User | None:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self._winner_keys.add(normalize_key(user.name))
        self.winners.append(user)
        return user

    def finish_round(self, prize: Prize, round_winners) -> None:
        """Update fallback history after a round completes."""
        drawer = prize.associated_drawer
        if _is_same(drawer, FALLBACK_DRAWER):
            # Fallback opportunity is consumed by Ma Jiaqi's round.
            self.last_two_draw_results = []
            return
        if any(_is_same(drawer, trigger) for trigger in FALLBACK_TRIGGER_DRAWERS):
            hit = any(self._is_protected(user) for user in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-FALLBACK_HISTORY_SIZE:]

"""Winner recording, grouping and CSV export."""

import csv
import io
import os
from dataclasses import dataclass
from datetime import datetime

from ..models import Prize, User

CSV_HEADER = ["Name", "Department", "Prize Name", "Winning Time"]


@dataclass
class Winner:
    user: User
    prize: Prize
    winning_time: datetime
    winning_order: int


def _sorted_prizes(prizes) -> list[Prize]:
    """Prize level from high to low; stable so addition order is kept within a level."""
    return sorted(prizes, key=lambda prize: prize.level)


class ResultManager:
    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._counter = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        self._counter += 1
        winner = Winner(
            user=user,
            prize=prize,
            winning_time=datetime.now(),
            winning_order=self._counter,
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize.name in result:
                result[winner.prize.name].append(winner)
        return result

    def get_ordered_winners(self, prizes) -> list[Winner]:
        """Winners ordered by prize level (high to low), then winning order."""
        by_prize = self.get_winners_by_prize(_sorted_prizes(prizes))
        ordered: list[Winner] = []
        for prize in _sorted_prizes(prizes):
            ordered.extend(by_prize.get(prize.name, []))
        return ordered

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        file_name = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            buffer = io.StringIO()
            writer = csv.writer(buffer, lineterminator="\n")
            writer.writerow(CSV_HEADER)
            for winner in self.get_ordered_winners(prizes):
                writer.writerow(
                    [
                        winner.user.name,
                        winner.user.department,
                        winner.prize.name,
                        winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                    ]
                )
            with open(path, "w", encoding="utf-8", newline="") as handle:
                handle.write(buffer.getvalue())
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

class ExitProgram(Exception):

def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def _ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        print()
        raise ExitProgram()


def _read_command(prompt: str) -> str:
    try:
        return input(prompt).strip().lower()
    except EOFError:
        # Non-interactive input ended: keep the lottery going to completion.
        return "continue"

        raw_quantity = _ask("Quantity: ")
        if not raw_quantity.isdigit() or int(raw_quantity) <= 0:
            print("Quantity must be a positive integer. Cancelled.")
            return
        quantity = int(raw_quantity)

        print("Associated lottery person:")
        for index, person in enumerate(LOTTERY_PERSONS, start=1):
            print(f"  {index}. {person}")
        print("  0. None")
        answer = _ask("Select a number, type a name, or leave blank for none: ")
        if answer in ("", "0"):
            drawer = None
        elif answer.isdigit() and 1 <= int(answer) <= len(LOTTERY_PERSONS):
            drawer = LOTTERY_PERSONS[int(answer) - 1]
        else:
            drawer = answer

        groups = self._select_groups()
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            print(f"Cannot add prize: {exc}")
            return
        print(f"Prize added: {prize.name} (quantity {prize.quantity}).")

    def _select_groups(self) -> list[str]:
        available = list(self.list_manager.get_group_info(self.users).keys()) if self.users else []
        if available:
            print("Department groups:")
            for index, department in enumerate(available, start=1):
                print(f"  {index}. {department}")
        raw = _ask("Enter avoid group numbers or names, comma separated (blank for none): ")
        return self._resolve_groups(raw, available)

    @staticmethod
    def _resolve_groups(raw: str, available: list[str]) -> list[str]:
        groups: list[str] = []
        for token in (part.strip() for part in raw.split(",")):
            if not token:
                continue
            if token.isdigit() and available:
                position = int(token)
                if 1 <= position <= len(available):
                    name = available[position - 1]
                else:
                    print(f"Ignoring invalid group number: {token}")
                    continue
            else:
                name = token
            if name not in groups:
                groups.append(name)
        return groups

    def _set_avoid_groups(self) -> None:
        prizes = self._show_prizes()
        if not prizes:
            return
        answer = _ask("Select prize number: ")
        if not answer.isdigit() or not (1 <= int(answer) <= len(prizes)):
            print("Invalid prize number.")
            return
        prize = prizes[int(answer) - 1]
        groups = self._select_groups()
        if self.prize_manager.update_avoid_groups(prize.name, groups):
            shown = ", ".join(groups) or "None"
            print(f"Avoid groups for '{prize.name}' set to: {shown}")
        else:
            print("Prize not found.")

    def _reorder_prizes(self) -> None:
        prizes = self._show_prizes()
        if len(prizes) < 2:
            print("At least two prizes are needed to reorder.")
            return
        raw = _ask("Enter new order as serial numbers (e.g. 2 1 3): ")
        tokens = [token for token in re.split(r"[\s,]+", raw.strip()) if token]
        if not tokens or not all(token.isdigit() for token in tokens):
            print("Invalid input: please enter serial numbers only.")
            return
        order = [int(token) for token in tokens]
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
        else:
            print("Invalid order: each serial number must be used exactly once.")

    # ----- lottery process -----------------------------------------------
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return

        print(f"\nLottery sequence: {len(prizes)} round(s)")
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name} (quantity {prize.quantity})")
        if _ask("Start the lottery now? (y/n): ").lower() not in ("y", "yes"):
            print("Lottery not started.")
            return

        lottery = LotteryManager(self.users)
        results = ResultManager()
        print(f"\n{LINE}")
        print("Lottery started!")
        print("Commands: [Enter] draw next | pause | continue | terminate (end current round)")
        print(LINE)

        for round_index, prize in enumerate(prizes, start=1):
            round_winners = self._run_round(round_index, len(prizes), prize, lottery, results)
            lottery.finish_round(prize, round_winners)

        self.results = results
        self._last_prizes = list(prizes)
        self.lottery_completed = True
        self._print_summary(prizes, results)

        if _ask("Export results to txt now? (y/n): ").lower() in ("y", "yes"):
            self._export(prizes, results)

    def _run_round(self, round_index, total, prize, lottery, results) -> list[User]:
        print(f"\n{SUB_LINE}")
        print(f"Round {round_index}/{total}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {lottery.describe_rule(prize)}")
        print(SUB_LINE)

        drawn: list[User] = []
        for position in range(1, prize.quantity + 1):
            if not self._await_next_draw():
                print("Round terminated by user; remaining winners of this prize are skipped.")
                break
            print(f"Currently drawing the {ordinal(position)} winner...")
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible participants left for this prize; round ends early.")
                break
            results.add_winner(user, prize)
            drawn.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        print(f"\nRound {round_index} results ({len(drawn)} winner(s), winning order):")
        if not drawn:
            print("  (no winners)")
        for index, user in enumerate(drawn, start=1):
            print(f"  {index}. {user.name} ({user.department})")
        return drawn

    def _await_next_draw(self) -> bool:
        """Handle process control. Returns False when the current round is terminated."""
        while True:
            command = _read_command("Press Enter to draw next (or pause / terminate): ")
            if command in ("", "continue", "draw", "next"):
                return True
            if command == "terminate":
                return False
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    command = _read_command("[paused] > ")
                    if command in ("continue", "resume"):
                        print("Lottery resumed.")
                        return True
                    if command == "terminate":
                        return False
                    print("Unknown command. Use 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    @staticmethod
    def _print_summary(prizes, results: ResultManager) -> None:
        ordered_prizes = sorted(prizes, key=lambda prize: prize.level)
        by_prize = results.get_winners_by_prize(ordered_prizes)
        print(f"\n{LINE}")
        print("FINAL WINNER LIST (prize level high to low, then winning order)")
        print(LINE)
        for prize in ordered_prizes:
            winners = by_prize.get(prize.name, [])
            print(f"\n{prize.name} - {len(winners)} winner(s)")
            if not winners:
                print("  (no winners)")
            for index, winner in enumerate(winners, start=1):
                print(f"  {index}. {winner.user.name} ({winner.user.department})")

    # ----- export --------------------------------------------------------
    def _export_menu(self) -> None:
        if self.results is None:
            print("No lottery results yet. Run the lottery first.")
            return
        self._export(self._last_prizes, self.results)

    @staticmethod
    def _export(prizes, results: ResultManager) -> None:
        _, message = results.export_to_csv(prizes)
        print(message)

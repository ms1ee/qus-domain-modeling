"""
Data loading utilities for the Beijing Subway network.

Provides the location of the bundled ``subway_data.json`` file and functions
to read and persist the raw network data dictionary.
"""

import json
import os

# Absolute path of src/subway_data.json, valid regardless of the working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path):
    """
    Load the subway network data from a JSON file.

    Args:
        path: Path to the JSON file.

    Returns:
        dict: The JSON content, with the keys "stations" and "connections".

    Raises:
        FileNotFoundError: If the file does not exist.
        ValueError: If the file is not valid JSON or does not have the expected structure.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)

    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")
    for key in ("stations", "connections"):
        if key not in data or not isinstance(data[key], list):
            raise ValueError(f"Subway data is missing the list field '{key}'")
    return data


def save_subway_data(data, path):
    """
    Write the subway network data dictionary to a JSON file.

    Args:
        data: The network data dictionary.
        path: Destination file path.

    Raises:
        OSError: If the file cannot be written.
    """
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")

"""
Graph data structures for the subway network.

The graph is a weighted undirected graph stored as adjacency lists. Station
numbers (integers) are assigned by the caller; the graph itself only stores
edges, vertices and the mapping from station names to their lines.
"""


class Edge:
    """
    A weighted connection between two stations on a single line.

    Attributes:
        source: Station number of the starting end of the edge.
        dest: Station number of the other end of the edge.
        distance: Physical distance in meters (also used as the edge weight).
        line: Name of the subway line the edge belongs to.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        if distance < 0:
            raise ValueError("Edge distance cannot be negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by shortest-path algorithms (equal to the distance in meters)."""
        return self.distance

    def reversed(self):
        """Return a copy of this edge pointing in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """
    Weighted undirected graph represented with adjacency lists.

    Attributes:
        station_lines: Dictionary mapping station names to the list of lines serving them.
    """

    def __init__(self):
        self._adj = {}          # station number -> list[Edge]
        self._edge_count = 0
        self.station_lines = {}  # station name -> list[str]

    def set_station_lines(self, station_name, lines):
        """Record the lines that serve a station (by name)."""
        self.station_lines[station_name] = list(lines)

    def lines_of(self, station_name):
        """Return a copy of the list of lines serving the given station name."""
        return list(self.station_lines.get(station_name, []))

    def add_vertex(self, vertex):
        """Ensure a station number exists as a vertex of the graph."""
        self._adj.setdefault(vertex, [])

    def has_vertex(self, vertex):
        """Return True if the station number is part of the graph."""
        return vertex in self._adj

    def add_edge(self, edge):
        """
        Add an undirected edge. Both directions are stored in the adjacency lists.

        Args:
            edge: An Edge instance.
        """
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adj[edge.source].append(edge)
        if edge.source != edge.dest:
            self._adj[edge.dest].append(edge.reversed())
        self._edge_count += 1

    def adj(self, vertex):
        """Return the list of edges leaving the given vertex (empty if unknown)."""
        return self._adj.get(vertex, [])

    def vertices(self):
        """Return the list of all station numbers in the graph."""
        return list(self._adj.keys())

    def num_vertices(self):
        """Return the number of vertices."""
        return len(self._adj)

    def num_edges(self):
        """Return the number of undirected edges that were added."""
        return self._edge_count

"""
Subway network construction and station name resolution.

``SubwayNetwork`` converts the raw data dictionary (from subway_data.json)
into a ``Graph`` and maintains the mapping between station names and
station numbers.
"""

import re

from models import Edge, Graph

_STATION_SUFFIX_RE = re.compile(r"(站|\s*station)$", re.IGNORECASE)


def normalize_station_name(name):
    """
    Produce a normalized key for station name matching.

    Collapses whitespace, removes a trailing "站" / "Station" suffix and
    compares case-insensitively.
    """
    text = " ".join(str(name).split())
    text = _STATION_SUFFIX_RE.sub("", text).strip()
    return text.casefold()


class SubwayNetwork:
    """
    The subway network as a graph plus station name/number mappings.

    Attributes:
        station_to_number: dict mapping official station name -> station number.
        number_to_station: dict mapping station number -> official station name.
        graph: The Graph instance holding edges and station lines.
    """

    def __init__(self, data):
        """
        Build the network from the raw data dictionary.

        Args:
            data: dict with "stations" and "connections" lists.

        Raises:
            ValueError: If a connection refers to an unknown station.
        """
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self._normalized = {}
        self._lines = set()
        self._build(data)

    def _build(self, data):
        merged_lines = {}
        order = []
        for station in data.get("stations", []):
            name = station["name"]
            if name not in merged_lines:
                merged_lines[name] = []
                order.append(name)
            for line in station.get("lines", []):
                if line not in merged_lines[name]:
                    merged_lines[name].append(line)

        for name in order:
            number = len(self.station_to_number)
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self._normalized.setdefault(normalize_station_name(name), name)
            self.graph.set_station_lines(name, merged_lines[name])
            self.graph.add_vertex(number)
            self._lines.update(merged_lines[name])

        for conn in data.get("connections", []):
            source = conn["source"]
            dest = conn["dest"]
            for name in (source, dest):
                if name not in self.station_to_number:
                    raise ValueError(f"Connection refers to unknown station '{name}'")
            line = conn["line"]
            self._lines.add(line)
            self.graph.add_edge(
                Edge(
                    self.station_to_number[source],
                    self.station_to_number[dest],
                    int(conn["distance"]),
                    line,
                )
            )

    def number_of(self, name):
        """Return the station number for an official station name (KeyError if unknown)."""
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the official station name for a station number (KeyError if unknown)."""
        return self.number_to_station[number]

    def has_station(self, name):
        """Return True if the official station name exists."""
        return name in self.station_to_number

    def resolve_name(self, raw_name):
        """
        Standardize a user-entered station name against the official names.

        Returns:
            The official station name, or None if no station matches.
        """
        if raw_name in self.station_to_number:
            return raw_name
        return self._normalized.get(normalize_station_name(raw_name))

    def station_names(self):
        """Return the sorted list of official station names."""
        return sorted(self.station_to_number.keys())

    def lines(self):
        """Return the sorted list of all line names in the network."""
        return sorted(self._lines)

"""
Path computation for the subway network.

Implements a priority-queue based Dijkstra search. The search runs over
states of the form (station number, line used to arrive), which allows the
algorithm to count transfers, enforce a maximum transfer limit, and apply
line preferences while still finding the optimal route.
"""

import heapq
import itertools

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# In comprehensive mode, each transfer is treated as this many meters of extra cost.
TRANSFER_PENALTY_M = 1000.0
# When line preferences are set, each edge outside the preferred lines adds this cost.
PREFERENCE_PENALTY_M = 500.0


class PathfindingConfig:
    """
    Configuration for a path query.

    Attributes:
        mode: One of VALID_MODES.
        max_transfers: Maximum number of allowed transfers, or None for no limit.
        preferred_lines: List of preferred line names (may be empty).
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Unknown mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class MinPQ:
    """
    Min priority queue implemented on top of a binary heap (heapq).

    Items with equal priority are returned in insertion order, and items are
    never compared directly.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, priority, item):
        """Insert an item with the given priority. O(log n)."""
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def del_min(self):
        """Remove and return (priority, item) with the smallest priority. O(log n)."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def min_priority(self):
        """Return the smallest priority without removing it."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._heap[0][0]

    def is_empty(self):
        """Return True if the queue has no items."""
        return not self._heap

    def __len__(self):
        return len(self._heap)


class ShortestPath:
    """
    Dijkstra-based optimal path computation from a source station.

    Attributes:
        source: Source station number.
        config: The PathfindingConfig used for the search.
        dist_to: dict mapping reachable station number -> actual distance in meters
                 along the chosen route.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"Source station {source} is not part of the graph")
        self._graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._parent = {}         # state -> (previous state, Edge)
        self._best_state = {}     # station number -> best state
        self._best_info = {}      # station number -> (distance, transfers, penalty)
        self._run()

    def _cost(self, dist, transfers, penalty):
        """Return the comparable cost tuple of a route according to the mode."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, dist + penalty)
        if mode == MODE_COMPREHENSIVE:
            return (dist + penalty + TRANSFER_PENALTY_M * transfers,)
        return (dist + penalty,)

    def _run(self):
        """Execute Dijkstra's algorithm over (station, arrival line) states."""
        max_transfers = self.config.max_transfers
        preferred = set(self.config.preferred_lines)

        start = (self.source, None)
        infos = {start: (0, 0, 0.0)}
        costs = {start: self._cost(0, 0, 0.0)}
        pq = MinPQ()
        pq.insert(costs[start], start)
        settled = set()

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)

            vertex, arrival_line = state
            dist, transfers, penalty = infos[state]

            for edge in self._graph.adj(vertex):
                new_transfers = transfers
                if arrival_line is not None and edge.line != arrival_line:
                    new_transfers += 1
                if max_transfers is not None and new_transfers > max_transfers:
                    continue

                new_penalty = penalty
                if preferred and edge.line not in preferred:
                    new_penalty += PREFERENCE_PENALTY_M

                new_info = (dist + edge.distance, new_transfers, new_penalty)
                next_state = (edge.dest, edge.line)
                if next_state in settled:
                    continue

                new_cost = self._cost(*new_info)
                old_cost = costs.get(next_state)
                if old_cost is None or new_cost < old_cost:
                    costs[next_state] = new_cost
                    infos[next_state] = new_info
                    self._parent[next_state] = (state, edge)
                    pq.insert(new_cost, next_state)

        best_cost = {}
        for state in settled:
            vertex = state[0]
            cost = costs[state]
            if vertex not in best_cost or cost < best_cost[vertex]:
                best_cost[vertex] = cost
                self._best_state[vertex] = state
                self._best_info[vertex] = infos[state]

        for vertex, info in self._best_info.items():
            self.dist_to[vertex] = info[0]

    def has_path_to(self, target):
        """Return True if the target station is reachable within the configured limits."""
        return target in self._best_state

    def path_with_edges(self, target):
        """
        Reconstruct the route from the source to the target.

        Returns:
            (list of station numbers from source to target, list of Edge objects along the route)

        Raises:
            ValueError: If the target is unreachable.
        """
        if target not in self._best_state:
            raise ValueError(f"No path from station {self.source} to station {target}")

        state = self._best_state[target]
        vertices = [state[0]]
        edges = []
        while state in self._parent:
            previous, edge = self._parent[state]
            edges.append(edge)
            vertices.append(previous[0])
            state = previous

        vertices.reverse()
        edges.reverse()
        return vertices, edges

    def get_transfer_count(self, target):
        """
        Return the number of line changes on the optimal route to the target.

        Raises:
            ValueError: If the target is unreachable.
        """
        if target not in self._best_info:
            raise ValueError(f"No path from station {self.source} to station {target}")
        return self._best_info[target][1]

"""
Fare calculation for the Beijing Subway.

Fare structure:
    0-6 km    : 3 yuan
    6-12 km   : 4 yuan
    12-22 km  : 5 yuan
    22-32 km  : 6 yuan
    over 32 km: 6 yuan plus 1 yuan for every additional 20 km (or part thereof)

Fractional kilometers are rounded up before a tier is selected.
"""

import math

# (inclusive upper bound in km, fare in yuan) for the progressive tiers.
FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]
EXTENSION_BASE_KM = 32
EXTENSION_STEP_KM = 20

# Estimated time model parameters.
AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MIN = 5.0


def _ceil_kilometers(distance):
    """Convert meters to whole kilometers, rounding fractional values up."""
    if distance < 0:
        raise ValueError("Distance cannot be negative")
    return math.ceil(distance / 1000)


def Sum_money(distance):
    """
    Calculate the subway fare for a travel distance.

    Args:
        distance: Travel distance in meters.

    Returns:
        int: Fare in yuan.

    Raises:
        ValueError: If the distance is negative.
    """
    km = _ceil_kilometers(distance)
    for upper_km, fare in FARE_TIERS:
        if km <= upper_km:
            return fare
    extra_steps = math.ceil((km - EXTENSION_BASE_KM) / EXTENSION_STEP_KM)
    return FARE_TIERS[-1][1] + extra_steps


def calculate_estimated_time(distance, transfer_count):
    """
    Estimate the travel time of a route.

    Args:
        distance: Travel distance in meters.
        transfer_count: Number of line transfers on the route.

    Returns:
        float: Estimated travel time in minutes, rounded to two decimals.

    Raises:
        ValueError: If distance or transfer_count is negative.
    """
    if distance < 0:
        raise ValueError("Distance cannot be negative")
    if transfer_count < 0:
        raise ValueError("Transfer count cannot be negative")
    travel_minutes = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    return round(travel_minutes + transfer_count * TRANSFER_TIME_MIN, 2)

"""
Path visualization and export.

Builds a structured summary of a computed route and renders it either as
console-formatted text or as JSON.
"""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(distance_m):
    """Format a distance: meters for short distances (<1 km), kilometers otherwise."""
    if distance_m < 1000:
        return f"{distance_m:.0f} m"
    return f"{distance_m / 1000:.2f} km"


def _build_segments(station_names, edges):
    """Group consecutive edges of the same line into travel segments."""
    segments = []
    for index, edge in enumerate(edges):
        start = station_names[index]
        end = station_names[index + 1]
        if segments and segments[-1]["line"] == edge.line:
            segments[-1]["end"] = end
            segments[-1]["distance_m"] += edge.distance
        else:
            segments.append({
                "line": edge.line,
                "start": start,
                "end": end,
                "distance_m": edge.distance,
            })
    return segments


def _build_transfers(station_names, edges):
    """Identify transfer stations and the lines joined at each of them."""
    transfers = []
    for index in range(1, len(edges)):
        previous_edge = edges[index - 1]
        current_edge = edges[index]
        if previous_edge.line != current_edge.line:
            transfers.append({
                "station": station_names[index],
                "from_line": previous_edge.line,
                "to_line": current_edge.line,
            })
    return transfers


def _involved_lines(edges):
    """Return the unique line names used by the route, in travel order."""
    lines = []
    for edge in edges:
        if edge.line not in lines:
            lines.append(edge.line)
    return lines


def build_path_summary(start, end, mode, station_names, edges, transfer_count):
    """
    Build a structured summary of a route.

    Args:
        start: Official name of the starting station.
        end: Official name of the ending station.
        mode: Query mode used.
        station_names: Ordered list of station names along the route.
        edges: Ordered list of Edge objects along the route.
        transfer_count: Number of transfers on the route.

    Returns:
        dict: Summary with query, path, distance, fare, time, transfers and lines.
    """
    distance_m = sum(edge.distance for edge in edges)
    fare = Sum_money(distance_m)
    estimated_time = calculate_estimated_time(distance_m, transfer_count)
    return {
        "query": {"start": start, "end": end, "mode": mode},
        "path": list(station_names),
        "station_count": len(station_names),
        "distance": {
            "meters": distance_m,
            "kilometers": round(distance_m / 1000.0, 2),
        },
        "fare": fare,
        "estimated_time_minutes": estimated_time,
        "transfer_count": transfer_count,
        "lines": _involved_lines(edges),
        "segments": _build_segments(station_names, edges),
        "transfers": _build_transfers(station_names, edges),
    }


def format_text_report(summary):
    """Render a route summary as console-formatted text."""
    query = summary["query"]
    lines = []
    lines.append("=== Beijing Subway Route ===")
    lines.append(f"From: {query['start']}  To: {query['end']}  (mode: {query['mode']})")
    lines.append(f"Total distance: {format_distance(summary['distance']['meters'])}"
                 f" ({summary['distance']['meters']} m)")
    lines.append(f"Fare: {summary['fare']} yuan")
    lines.append(f"Estimated time: {summary['estimated_time_minutes']:.2f} min")
    lines.append(f"Transfers: {summary['transfer_count']}")
    lines.append(f"Lines involved: {', '.join(summary['lines']) if summary['lines'] else 'None'}")

    lines.append("")
    lines.append("Route details:")
    segments = summary["segments"]
    if not segments:
        lines.append("  (start and end are the same station)")
    for index, segment in enumerate(segments):
        if index > 0:
            lines.append(f"  Transfer at {segment['start']}: {segments[index - 1]['line']}"
                         f" -> {segment['line']}")
        lines.append(f"  [{segment['line']}] {segment['start']} -> {segment['end']}"
                     f" ({format_distance(segment['distance_m'])})")

    lines.append("")
    lines.append(f"Station sequence ({summary['station_count']} stations):")
    for number, name in enumerate(summary["path"], start=1):
        lines.append(f"  {number}. {name}")

    return "\n".join(lines)


def format_json_report(summary, indent=2):
    """Render a route summary as a JSON string."""
    return json.dumps(summary, ensure_ascii=False, indent=indent)

class _UnionFind:
    """Simple disjoint-set structure used for connectivity checks."""

    def __init__(self):
        self._parent = {}

    def add(self, item):
        self._parent.setdefault(item, item)

    def find(self, item):
        root = item
        while self._parent[root] != root:
            root = self._parent[root]
        while self._parent[item] != root:
            self._parent[item], item = root, self._parent[item]
        return root

    def union(self, a, b):
        self.add(a)
        self.add(b)
        root_a, root_b = self.find(a), self.find(b)
        if root_a != root_b:
            self._parent[root_b] = root_a

    def component_count(self):
        return len({self.find(item) for item in self._parent})

class DataManager:
    """
    Manage the raw subway data: integrity checking and updates.

    Attributes:
        data: The working copy of the network data dictionary.
    """

    def __init__(self, data):
        self.data = copy.deepcopy(data)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    @property
    def stations(self):
        return self.data["stations"]

    @property
    def connections(self):
        return self.data["connections"]

    def _station_index(self):
        return {station["name"]: station for station in self.stations}

    # ------------------------------------------------------------------
    # Integrity checks
    # ------------------------------------------------------------------
    def check_integrity(self):
        """
        Run all data integrity checks.

        Returns:
            dict with keys "errors" (list[str]), "warnings" (list[str]),
            "info" (dict) and "ok" (bool, True when there are no errors).
        """
        report = {"errors": [], "warnings": [], "info": {}, "ok": True}
        self._check_station_coverage(report)
        self._check_connections(report)
        self._check_distances(report)
        self._check_line_connectivity(report)
        report["info"]["station_count"] = len(self.stations)
        report["info"]["connection_count"] = len(self.connections)
        report["ok"] = not report["errors"]
        return report

    def _check_station_coverage(self, report):
        names = [station.get("name") for station in self.stations]
        name_set = set(names)

        if len(name_set) != len(names):
            report["warnings"].append("Duplicate station names found in station data")

        lines_present = set()
        for station in self.stations:
            lines_present.update(station.get("lines", []))
        for conn in self.connections:
            if "line" in conn:
                lines_present.add(conn["line"])

        for line in REQUIRED_LINES:
            if line not in lines_present:
                report["warnings"].append(f"Main line '{line}' is not covered by the data")

        for line, representatives in REPRESENTATIVE_STATIONS.items():
            for station_name in representatives:
                if station_name not in name_set:
                    report["warnings"].append(
                        f"Representative station '{station_name}' of {line} is missing")

    def _check_connections(self, report):
        station_index = self._station_index()
        seen = set()
        connected = set()

        for position, conn in enumerate(self.connections):
            missing = [key for key in ("source", "dest", "distance", "line") if key not in conn]
            if missing:
                report["errors"].append(
                    f"Connection #{position} is missing fields: {', '.join(missing)}")
                continue

            source, dest, line = conn["source"], conn["dest"], conn["line"]
            for name in (source, dest):
                if name not in station_index:
                    report["errors"].append(
                        f"Connection #{position} refers to unknown station '{name}'")
            if source == dest:
                report["errors"].append(f"Connection #{position} links '{source}' to itself")
                continue

            for name in (source, dest):
                station = station_index.get(name)
                if station is not None and line not in station.get("lines", []):
                    report["errors"].append(
                        f"Station '{name}' is not listed as served by '{line}'")

            key = (frozenset((source, dest)), line)
            if key in seen:
                report["warnings"].append(
                    f"Duplicate connection {source} - {dest} on {line}")
            seen.add(key)
            connected.update((source, dest))

        for station in self.stations:
            if station.get("name") not in connected:
                report["warnings"].append(
                    f"Station '{station.get('name')}' has no connections")

    def _check_distances(self, report):
        for position, conn in enumerate(self.connections):
            distance = conn.get("distance")
            label = f"{conn.get('source')} - {conn.get('dest')}"
            if isinstance(distance, bool) or not isinstance(distance, (int, float)):
                report["errors"].append(f"Distance for {label} is not a number")
                continue
            if distance <= 0:
                report["errors"].append(f"Distance for {label} must be positive")
            elif distance > MAX_DISTANCE_M:
                report["errors"].append(f"Distance for {label} exceeds {MAX_DISTANCE_M} m")
            elif distance > TYPICAL_MAX_DISTANCE_M:
                report["warnings"].append(
                    f"Distance for {label} ({distance} m) is unusually large")

    def _check_line_connectivity(self, report):
        per_line = {}
        for conn in self.connections:
            if "line" not in conn or "source" not in conn or "dest" not in conn:
                continue
            uf = per_line.setdefault(conn["line"], _UnionFind())
            uf.union(conn["source"], conn["dest"])

        for line, uf in sorted(per_line.items()):
            components = uf.component_count()
            if components > 1:
                report["warnings"].append(
                    f"{line} is split into {components} disconnected segments")

    # ------------------------------------------------------------------
    # Update operations
    # ------------------------------------------------------------------
    def add_line(self, line_name, station_names, distances):
        """
        Add a new line made of an ordered sequence of stations.

        Args:
            line_name: Name of the new line.
            station_names: Ordered list of station names on the line.
            distances: List of distances (meters) between consecutive stations.

        Raises:
            ValueError: On invalid input or if the line already exists.
        """
        if not line_name or not isinstance(line_name, str):
            raise ValueError("Line name must be a non-empty string")
        if line_name in {line for s in self.stations for line in s.get("lines", [])} or \
                line_name in {c.get("line") for c in self.connections}:
            raise ValueError(f"Line '{line_name}' already exists")
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(set(station_names)) != len(station_names):
            raise ValueError("Station sequence contains duplicates")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must be one less than number of stations")
        for distance in distances:
            validate_distance(distance)

        index = self._station_index()
        for name in station_names:
            if name not in index:
                station = {"name": name, "lines": []}
                self.stations.append(station)
                index[name] = station
            if line_name not in index[name]["lines"]:
                index[name]["lines"].append(line_name)

        for position, distance in enumerate(distances):
            self.connections.append({
                "source": station_names[position],
                "dest": station_names[position + 1],
                "distance": distance,
                "line": line_name,
            })

    def modify_station(self, name, new_name=None, lines=None):
        """
        Modify a station's name and/or its line affiliations.

        Args:
            name: Current official station name.
            new_name: Optional new name (renames connections too).
            lines: Optional full replacement list of line names.

        Returns:
            The final station name.

        Raises:
            KeyError: If the station does not exist.
            ValueError: If the new name is already used or the lines are invalid.
        """
        index = self._station_index()
        if name not in index:
            raise KeyError(f"Station '{name}' does not exist")
        station = index[name]

        final_name = name
        if new_name and new_name != name:
            if new_name in index:
                raise ValueError(f"Station '{new_name}' already exists")
            final_name = new_name
            station["name"] = new_name
            for conn in self.connections:
                if conn.get("source") == name:
                    conn["source"] = new_name
                if conn.get("dest") == name:
                    conn["dest"] = new_name

        if lines is not None:
            if not isinstance(lines, list) or not all(isinstance(x, str) and x for x in lines):
                raise ValueError("Lines must be a list of non-empty strings")
            station["lines"] = list(dict.fromkeys(lines))

        return final_name

    def update_distance(self, source, dest, distance):
        """
        Update the distance between two directly connected stations.

        Returns:
            int: Number of connection records updated.

        Raises:
            ValueError: If the distance is invalid.
            KeyError: If the stations are not directly connected.
        """
        validate_distance(distance)
        updated = 0
        for conn in self.connections:
            if {conn.get("source"), conn.get("dest")} == {source, dest}:
                conn["distance"] = distance
                updated += 1
        if updated == 0:
            raise KeyError(f"No direct connection between '{source}' and '{dest}'")
        return updated

    def to_dict(self):
        """Return a deep copy of the working data dictionary."""
        return copy.deepcopy(self.data)

        """Persist the working data to a JSON file."""

"""
Command-line interface for the Beijing Subway fare calculation system.

Runs an interactive loop where the user enters "Starting Station; Ending Station"
queries. Command-line arguments control output format, output file, query mode,
transfer limit, line preference and data checking.
"""

import argparse
import os
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import DataManager
from fare_calculator import Sum_money  # noqa: F401  (fare is computed inside the summary)
from network import SubwayNetwork
from path_report import build_path_summary, format_json_report, format_text_report
from pathfinding import VALID_MODES, PathfindingConfig, ShortestPath

EXIT_COMMANDS = ("quit", "exit", "q")


class QueryError(Exception):
    """Raised for any user-facing query error (format, station, logic, path, fare)."""


def parse_args(argv=None):
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (Dijkstra-based route planner).")
    parser.add_argument("--output_file", default=None,
                        help="Append query results to this file.")
    parser.add_argument("--format", choices=["text", "json"], default="text",
                        help="Output format: text or json (default: text).")
    parser.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                        help="Path query mode (default: shortest_distance).")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="Maximum number of allowed transfers.")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line name, e.g. \"Line 1\" (comma-separated for several).")
    parser.add_argument("--check_data", action="store_true",
                        help="Run the data integrity check and exit.")
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")
    return args


def parse_preferences(value):
    """Convert the --line_preference value into a list of line names."""
    if not value:
        return []
    return [item.strip() for item in value.split(",") if item.strip()]


def parse_query(text):
    """
    Split a query of the form "Starting Station; Ending Station".

    Raises:
        QueryError: If the format is wrong.
    """
    message = "Input format incorrect, please use 'Starting Station; Ending Station' format"
    if text.count(";") != 1:
        raise QueryError(message)
    start, end = (part.strip() for part in text.split(";"))
    if not start or not end:
        raise QueryError(message)
    return start, end


def answer_query(network, text, args):
    """
    Validate a query and compute the route summary.

    Args:
        network: SubwayNetwork instance.
        text: Raw query text entered by the user.
        args: Parsed command-line arguments.

    Returns:
        dict: Route summary (see path_report.build_path_summary).

    Raises:
        QueryError: For any validation, path or fare calculation problem.
    """
    start_raw, end_raw = parse_query(text)

    start = network.resolve_name(start_raw)
    if start is None:
        raise QueryError(f"Starting station '{start_raw}' does not exist, please check the station name")
    end = network.resolve_name(end_raw)
    if end is None:
        raise QueryError(f"Ending station '{end_raw}' does not exist, please check the station name")
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")

    try:
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=parse_preferences(args.line_preference),
        )
        source_number = network.number_of(start)
        target_number = network.number_of(end)
        path_search = ShortestPath(network.graph, source_number, config)
        if not path_search.has_path_to(target_number):
            raise QueryError("No route found between these stations under the current constraints")
        vertices, edges = path_search.path_with_edges(target_number)
        transfer_count = path_search.get_transfer_count(target_number)
    except (ValueError, KeyError) as exc:
        raise QueryError(f"Path calculation failed: {exc}") from exc

    station_names = [network.name_of(number) for number in vertices]
    try:
        return build_path_summary(start, end, args.mode, station_names, edges, transfer_count)
    except (ValueError, TypeError) as exc:
        raise QueryError(f"Fare calculation failed: {exc}") from exc


def render(summary, output_format):
    """Render a summary in the requested format."""
    if output_format == "json":
        return format_json_report(summary)
    return format_text_report(summary)


def export_result(content, path):
    """Append rendered content to the output file. Returns an error message or None."""
    try:
        directory = os.path.dirname(os.path.abspath(path))
        os.makedirs(directory, exist_ok=True)
        with open(path, "a", encoding="utf-8") as handle:
            handle.write(content + "\n")
        return None
    except OSError as exc:
        return f"Could not write output file '{path}': {exc}"


def format_integrity_report(report):
    """Render the data integrity report as text."""
    lines = ["=== Subway Data Integrity Check ==="]
    info = report.get("info", {})
    lines.append(f"Stations: {info.get('station_count', 0)}, "
                 f"connections: {info.get('connection_count', 0)}")
    lines.append(f"Errors: {len(report['errors'])}")
    for message in report["errors"]:
        lines.append(f"  [ERROR] {message}")
    lines.append(f"Warnings: {len(report['warnings'])}")
    for message in report["warnings"]:
        lines.append(f"  [WARN] {message}")
    lines.append("Result: PASSED" if report["ok"] else "Result: FAILED")
    return "\n".join(lines)


def run_check_data(data):
    """Run the integrity check and return the process exit code."""
    report = DataManager(data).check_integrity()
    print(format_integrity_report(report))
    return 0 if report["ok"] else 1


def run_interactive(network, args, input_func=input, output=print):
    """
    Run the continuous query loop until the user quits or input ends.

    Returns:
        int: Number of successful queries.
    """
    output("Beijing Subway fare calculator. Enter 'Starting Station; Ending Station'"
           " (or 'quit' to exit).")
    successful = 0
    while True:
        try:
            text = input_func("Query> ")
        except (EOFError, KeyboardInterrupt):
            output("")
            break
        text = text.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        try:
            summary = answer_query(network, text, args)
        except QueryError as exc:
            output(f"Error: {exc}")
            continue

        rendered = render(summary, args.format)
        output(rendered)
        successful += 1

        if args.output_file:
            error = export_result(rendered, args.output_file)
            if error:
                output(f"Error: {error}")
    return successful


def main(argv=None):
    """Entry point. Returns the process exit code."""
    args = parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except (OSError, ValueError) as exc:
        print(f"Error: could not load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        return run_check_data(data)

    try:
        network = SubwayNetwork(data)
    except (ValueError, KeyError) as exc:
        print(f"Error: subway network is invalid: {exc}", file=sys.stderr)
        return 1

    for line in parse_preferences(args.line_preference):
        if line not in network.lines():
            print(f"Warning: preferred line '{line}' does not exist in the network")

    run_interactive(network, args)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Process-wide key/value store shared by the dispatch simulator modules.

The store holds the mutable global state of the simulation, such as the
rider list (key ``"riders"``), system funds, the rider ID counter, the
running flag and the simulated tick counter.
"""

import logging
from typing import Any, Dict

logger = logging.getLogger(__name__)

_globals: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store to its default values.

    Creates an empty rider list, a zeroed rider ID counter, the running flag
    set to True and an empty order history. Funds and parameters are filled
    in later by :func:`system.init_system`.
    """
    _globals.clear()
    _globals.update(
        {
            "riders": [],
            "funds": 0,
            "rider_id_counter": 0,
            "running": True,
            "tick": 0,
            "orders": [],
            "params": {},
        }
    )
    logger.debug("Global variable store initialised")


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable, e.g. ``"riders"``.
        value: Any Python object to store.
    """
    _globals[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when ``key`` has not been set.

    Returns:
        The stored value, or ``default`` if the key is missing.
    """
    return _globals.get(key, default)

"""Grid-based path planning for the simulated 800x800 map.

The map is an 8x8 grid of 100x100 cells. Streets run along the grid lines
(coordinates that are multiples of 100). Intersections are the points where
two streets meet; restaurants and customers are attached to intersections.
Routes are computed on the intersection graph and prefer straight segments
over turns (right-angle turns are allowed but penalised).
"""

import heapq
import itertools
import logging
import math
import random
from typing import List, Optional, Sequence, Tuple

logger = logging.getLogger(__name__)

MAP_SIZE = 800
CELL_SIZE = 100
GRID_SIZE = 8
NODE_COUNT = GRID_SIZE + 1  # 9 intersections per axis

Point = Tuple[int, int]

# Direction indices: 0=east, 1=west, 2=north, 3=south.
DIRECTIONS = ((1, 0), (-1, 0), (0, 1), (0, -1))
TURN_PENALTY = 0.5
_INF = float("inf")


class PathPlanner:
    """Stateless helper providing coordinate conversion and route planning."""

    # ------------------------------------------------------------------ #
    # Coordinate helpers
    # ------------------------------------------------------------------ #
    @staticmethod
    def clamp_coordinate(value: int) -> int:
        """Clamp a coordinate into the map range ``[0, MAP_SIZE]``."""
        return max(0, min(MAP_SIZE, int(value)))

    @staticmethod
    def snap_to_road(value: int) -> int:
        """Round a coordinate to the nearest grid line (a multiple of 100).

        Half values round upwards, and the result is clamped to the map.

        Args:
            value: Raw coordinate in the range 0-800.

        Returns:
            The nearest multiple of ``CELL_SIZE`` within the map.
        """
        snapped = math.floor(int(value) / CELL_SIZE + 0.5) * CELL_SIZE
        return max(0, min(MAP_SIZE, snapped))

    @staticmethod
    def cell_index(value: int) -> int:
        """Return the grid cell index (0-7) containing ``value``."""
        return min(max(int(value), 0) // CELL_SIZE, GRID_SIZE - 1)

    @classmethod
    def grid_center(cls, value: int) -> int:
        """Return the centre coordinate of the cell containing ``value``.

        Implements ``center = (grid // 100) * 100 + 50``.
        """
        return cls.cell_index(value) * CELL_SIZE + CELL_SIZE // 2

    @staticmethod
    def is_intersection(x: int, y: int) -> bool:
        """Return True if the point lies on a street intersection."""
        return (
            0 <= x <= MAP_SIZE
            and 0 <= y <= MAP_SIZE
            and x % CELL_SIZE == 0
            and y % CELL_SIZE == 0
        )

    @staticmethod
    def is_on_road(x: int, y: int) -> bool:
        """Return True if the point lies on any street (grid line)."""
        return (
            0 <= x <= MAP_SIZE
            and 0 <= y <= MAP_SIZE
            and (x % CELL_SIZE == 0 or y % CELL_SIZE == 0)
        )

    @staticmethod
    def manhattan(a: Sequence[int], b: Sequence[int]) -> int:
        """Return the Manhattan distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def is_aligned(a: Sequence[int], b: Sequence[int]) -> bool:
        """Return True if ``b`` is directly above/below or beside ``a``.

        This is the "special case" in which a straight-line movement along a
        single street reaches the target without any turn.
        """
        return (a[0] == b[0] or a[1] == b[1]) and tuple(a) != tuple(b)

    @staticmethod
    def random_node(rng: Optional[random.Random] = None) -> Point:
        """Return a random street intersection (used to place riders)."""
        source = rng or random
        return (
            source.randrange(NODE_COUNT) * CELL_SIZE,
            source.randrange(NODE_COUNT) * CELL_SIZE,
        )

    @staticmethod
    def route_length(points: Sequence[Point]) -> int:
        """Return the total Manhattan length of a polyline of points."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    # ------------------------------------------------------------------ #
    # Route planning
    # ------------------------------------------------------------------ #
    @classmethod
    def plan(cls, start: Point, goal: Point) -> List[Point]:
        """Plan a street route between two intersections.

        Uses Dijkstra's algorithm over ``(intersection, heading)`` states so
        that straight-line movement is preferred and right-angle turns are
        only taken when they shorten the route.

        Args:
            start: Starting intersection (both coordinates multiples of 100).
            goal: Destination intersection.

        Returns:
            Ordered list of intersections visited after ``start``, ending with
            ``goal``. Empty if ``start == goal``.

        Raises:
            ValueError: If either endpoint is not an intersection.
            RuntimeError: If no route exists (should not happen on the grid).
        """
        if not cls.is_intersection(*start) or not cls.is_intersection(*goal):
            raise ValueError(f"route endpoints must be intersections: {start} -> {goal}")
        if tuple(start) == tuple(goal):
            return []

        origin = (start[0] // CELL_SIZE, start[1] // CELL_SIZE)
        target = (goal[0] // CELL_SIZE, goal[1] // CELL_SIZE)
        origin_state = (origin, -1)

        dist = {origin_state: 0.0}
        prev = {}
        counter = itertools.count()
        heap = [(0.0, next(counter), origin_state)]
        found = None

        while heap:
            cost, _, state = heapq.heappop(heap)
            if cost > dist.get(state, _INF):
                continue
            node, last_dir = state
            if node == target:
                found = state
                break
            for idx, (dx, dy) in enumerate(DIRECTIONS):
                nx, ny = node[0] + dx, node[1] + dy
                if not (0 <= nx < NODE_COUNT and 0 <= ny < NODE_COUNT):
                    continue
                step = 1.0
                if last_dir != -1 and idx != last_dir:
                    step += TURN_PENALTY
                nxt = ((nx, ny), idx)
                new_cost = cost + step
                if new_cost < dist.get(nxt, _INF):
                    dist[nxt] = new_cost
                    prev[nxt] = state
                    heapq.heappush(heap, (new_cost, next(counter), nxt))

        if found is None:
            raise RuntimeError(f"no route found from {start} to {goal}")

        nodes = []
        state = found
        while state != origin_state:
            nodes.append(state[0])
            state = prev[state]
        nodes.reverse()
        return [(x * CELL_SIZE, y * CELL_SIZE) for x, y in nodes]

    @staticmethod
    def _street_endpoints(pos: Point) -> List[Point]:
        """Return the intersections bounding the street segment holding ``pos``."""
        x, y = pos
        if x % CELL_SIZE == 0:
            lo = (y // CELL_SIZE) * CELL_SIZE
            candidates = [(x, lo), (x, lo + CELL_SIZE)]
        else:
            lo = (x // CELL_SIZE) * CELL_SIZE
            candidates = [(lo, y), (lo + CELL_SIZE, y)]
        return [
            c for c in candidates if 0 <= c[0] <= MAP_SIZE and 0 <= c[1] <= MAP_SIZE
        ]

    @classmethod
    def route_from(cls, pos: Point, target: Point) -> List[Point]:
        """Plan a route from any road point (possibly mid-street) to ``target``.

        If ``pos`` is mid-street, the route first leaves through the segment
        endpoint that gives the shortest total distance to ``target``.

        Returns:
            Ordered list of points after ``pos`` ending with ``target``.
        """
        if cls.is_intersection(*pos):
            return cls.plan(pos, target)
        best = min(
            cls._street_endpoints(pos),
            key=lambda e: cls.manhattan(pos, e) + cls.manhattan(e, target),
        )
        return [best] + cls.plan(best, target)

    @classmethod
    def full_route(cls, pos: Point, restaurant: Point, customer: Point) -> List[Point]:
        """Return the continuous route pos -> restaurant -> customer.

        The result starts with ``pos`` itself and is used for route display
        and time estimation.
        """
        route = [tuple(pos)]
        route.extend(cls.route_from(tuple(pos), tuple(restaurant)))
        route.extend(cls.plan(tuple(restaurant), tuple(customer)))
        return route

"""Order data model.

An order is created from user coordinates ``[restaurant_x, restaurant_y,
customer_x, customer_y]``, snapped to the street grid, assigned to a rider,
and finally marked completed when the rider delivers it.
"""

import itertools
import logging
from typing import Any, Dict, Optional, Sequence

import globalvar
from path_planner import PathPlanner, Point

logger = logging.getLogger(__name__)

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_COMPLETED = "completed"
STATUS_CANCELLED = "cancelled"

# Income credited to system funds for each delivered order.
ORDER_FEE = 300


def _current_tick() -> int:
    """Return the current simulated tick from the global store."""
    return int(globalvar.get_value("tick", 0) or 0)


class Order:
    """A single food delivery order.

    Attributes:
        id: Unique order identifier.
        res_pos: Restaurant location (snapped to a street intersection).
        cus_pos: Customer location (snapped to a street intersection).
        status: One of "pending", "assigned", "completed", "cancelled".
        rider_id: ID of the assigned rider, or None.
        start_tick: Tick at which the order was created.
        accepted_tick: Tick at which the order was assigned to a rider.
        completed_tick: Tick at which the order was delivered.
        distance: Planned route length from restaurant to customer (units).
        time_taken: Ticks between creation and delivery.
        efficiency: Distance covered per tick (distance / time_taken).
    """

    _id_source = itertools.count(1)

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: ``(x, y)`` of the restaurant, range 0-800.
            cus_pos: ``(x, y)`` of the customer, range 0-800.
        """
        self.id = next(Order._id_source)
        self.res_pos: Point = self._normalize(res_pos)
        self.cus_pos: Point = self._normalize(cus_pos)
        self.status = STATUS_PENDING
        self.rider_id: Optional[int] = None
        self.start_tick = _current_tick()
        self.accepted_tick: Optional[int] = None
        self.completed_tick: Optional[int] = None
        self.distance = PathPlanner.route_length(
            [self.res_pos] + PathPlanner.plan(self.res_pos, self.cus_pos)
        )
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None
        logger.debug(
            "Order %d created: restaurant=%s customer=%s distance=%d",
            self.id,
            self.res_pos,
            self.cus_pos,
            self.distance,
        )

    @staticmethod
    def _normalize(pos: Sequence[int]) -> Point:
        """Snap a raw ``(x, y)`` pair to the nearest street intersection."""
        x, y = pos
        return (PathPlanner.snap_to_road(int(x)), PathPlanner.snap_to_road(int(y)))

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to the rider with ``rider_id``."""
        self.rider_id = int(rider_id)
        self.status = STATUS_ASSIGNED
        self.accepted_tick = _current_tick()

    def mark_completed(self, tick: Optional[int] = None) -> None:
        """Mark the order as delivered and record delivery efficiency.

        Args:
            tick: Completion tick. Defaults to the current simulated tick.
        """
        completed = _current_tick() if tick is None else int(tick)
        self.completed_tick = completed
        self.status = STATUS_COMPLETED
        self.time_taken = max(0, completed - self.start_tick)
        self.efficiency = round(self.distance / max(1, self.time_taken), 2)
        logger.info(
            "Order %d completed at tick %d (time=%d, efficiency=%.2f)",
            self.id,
            completed,
            self.time_taken,
            self.efficiency,
        )

    def cancel(self) -> None:
        """Mark the order as cancelled (no rider could be assigned)."""
        self.status = STATUS_CANCELLED
        logger.warning("Order %d cancelled", self.id)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_tick": self.start_tick,
            "accepted_tick": self.accepted_tick,
            "completed_tick": self.completed_tick,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

"""Rider model: state machine, street movement and event logging.

Rider states:
    0 (idle)        - available for new orders.
    1 (picking up)  - driving to the restaurant of the current order.
    2 (delivering)  - driving to the customer of the current order.

Transitions:
    Idle -> Picking Up      on the first tick after an order is assigned.
    Picking Up -> Delivering when the restaurant is reached.
    Delivering -> Idle      when the customer is reached (order completed).
"""

import logging
from typing import Any, Dict, List, Optional, Set

import globalvar
from order_model import ORDER_FEE, Order
from path_planner import PathPlanner, Point

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}
DEFAULT_SPEED = 100  # map units per tick


def _current_tick() -> int:
    """Return the current simulated tick from the global store."""
    return int(globalvar.get_value("tick", 0) or 0)


class Rider:
    """A delivery rider moving along the street grid.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current position (integers, on a street).
        speed: Movement speed in map units per tick.
        state: Current state (0=idle, 1=picking up, 2=delivering).
        orders: Assigned orders; the first entry is the active order.
        special_case: True while moving straight towards a target that is
            directly above/below or on the same street row.
        log: Event records with keys "event", "order_id", "tick", etc.
    """

    def __init__(
        self,
        rider_id: int,
        speed: int = DEFAULT_SPEED,
        start: Optional[Point] = None,
    ) -> None:
        """Create a rider on a street intersection.

        Args:
            rider_id: Unique identifier.
            speed: Movement speed (units/tick), must be positive.
            start: Optional starting intersection. A random one is chosen if
                omitted.

        Raises:
            ValueError: If the speed is not positive or the start point is not
                an intersection.
        """
        if start is None:
            start = PathPlanner.random_node()
        x, y = int(start[0]), int(start[1])
        if not PathPlanner.is_intersection(x, y):
            raise ValueError("riders must start at a street intersection")
        if int(speed) <= 0:
            raise ValueError("rider speed must be positive")

        self.rider_id = int(rider_id)
        self.x = x
        self.y = y
        self.speed = int(speed)
        self.state = STATE_IDLE
        self.orders: List[Order] = []
        self.special_case = False
        self.log: List[Dict[str, Any]] = []
        self._route: Optional[List[Point]] = None
        self._accepted_ids: Set[int] = set()
        self._record("spawn", None)
        logger.info("Rider %d spawned at (%d, %d)", self.rider_id, self.x, self.y)

    # ------------------------------------------------------------------ #
    # Properties and bookkeeping
    # ------------------------------------------------------------------ #
    @property
    def position(self) -> Point:
        """Return the current ``(x, y)`` position."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Return a human readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event record to :attr:`log`."""
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": _current_tick(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %d event=%s order=%s", self.rider_id, event, order_id)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
        }

    def recover(self) -> None:
        """Reset transient movement state after an inconsistency or error.

        The rider keeps its assigned orders and returns to idle so that the
        active order is restarted cleanly on the next tick.
        """
        logger.warning("Recovering rider %d to idle state", self.rider_id)
        self.state = STATE_IDLE
        self._route = None
        self.special_case = False

    # ------------------------------------------------------------------ #
    # Movement
    # ------------------------------------------------------------------ #
    def move(self) -> None:
        """Advance the rider by one tick.

        Handles order acceptance logging, the idle -> picking-up transition,
        movement along the planned route, and arrival events.
        """
        self._accept_new_orders()

        if self.state == STATE_IDLE:
            if not self.orders:
                return
            self.state = STATE_PICKING_UP
            self._route = None
            self.special_case = False

        if self.state in (STATE_PICKING_UP, STATE_DELIVERING):
            self._advance_leg()

    def _accept_new_orders(self) -> None:
        """Log an ``accepted_order`` event for every newly assigned order."""
        for order in self.orders:
            if order.id not in self._accepted_ids:
                self._accepted_ids.add(order.id)
                self._record("accepted_order", order.id)

    def _advance_leg(self) -> None:
        """Move towards the target of the current leg (restaurant or customer)."""
        if not self.orders:
            logger.warning("Rider %d in state %d without orders", self.rider_id, self.state)
            self.recover()
            return

        order = self.orders[0]
        target = order.res_pos if self.state == STATE_PICKING_UP else order.cus_pos

        if self._route is None:
            self._route = PathPlanner.plan(self.position, target)
            self.special_case = PathPlanner.is_aligned(self.position, target)

        self._travel(self.speed)

        if not self._route:
            self._arrive(order)

    def _travel(self, distance: int) -> None:
        """Move along the remaining route by at most ``distance`` units.

        Movement is axis-aligned: the rider moves along the current street
        towards the next waypoint, consuming waypoints as they are reached.
        """
        remaining = distance
        while remaining > 0 and self._route:
            tx, ty = self._route[0]
            dx = tx - self.x
            dy = ty - self.y
            dist = abs(dx) + abs(dy)
            if dist <= remaining:
                self.x, self.y = tx, ty
                self._route.pop(0)
                remaining -= dist
            else:
                if dx != 0:
                    self.x += remaining if dx > 0 else -remaining
                else:
                    self.y += remaining if dy > 0 else -remaining
                remaining = 0

    def _arrive(self, order: Order) -> None:
        """Handle arrival at the target of the current leg."""
        if self.state == STATE_PICKING_UP:
            self._record("arrived_at_merchant", order.id)
            self.state = STATE_DELIVERING
            self._route = None
            self.special_case = False
            return

        order.mark_completed(_current_tick())
        self.orders.remove(order)
        self._accepted_ids.discard(order.id)
        funds = int(globalvar.get_value("funds", 0) or 0)
        globalvar.set_value("funds", funds + ORDER_FEE)
        self._record("delivered_to_customer", order.id)
        self.state = STATE_IDLE
        self._route = None
        self.special_case = False

"""Order management: input validation, creation, rider assignment and history."""

import logging
from typing import Any, List, Optional, Sequence, Tuple

import globalvar
from order_model import STATUS_ASSIGNED, Order
from path_planner import MAP_SIZE, PathPlanner
from rider import STATE_IDLE, Rider

logger = logging.getLogger(__name__)

DEFAULT_LOAD_PENALTY = 100


def validate_coordinates(values: Sequence[Any]) -> List[int]:
    """Validate ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

    Args:
        values: Four integers (or strings representing integers).

    Returns:
        The four values converted to ``int``.

    Raises:
        ValueError: If the count is wrong, a value is not an integer, or a
            value lies outside the range 0-800.
    """
    if len(values) != 4:
        raise ValueError(f"expected 4 coordinates, got {len(values)}")
    result: List[int] = []
    for raw in values:
        if isinstance(raw, bool):
            raise ValueError(f"'{raw}' is not a valid integer coordinate")
        try:
            number = raw if isinstance(raw, int) else int(str(raw).strip())
        except ValueError:
            raise ValueError(f"'{raw}' is not an integer") from None
        if not 0 <= number <= MAP_SIZE:
            raise ValueError(f"coordinate {number} is out of range 0-{MAP_SIZE}")
        result.append(number)
    return result


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the most suitable rider for ``order``.

    Candidates are idle riders with no queued orders when any exist; otherwise
    all riders are candidates. The cost is the Manhattan distance from the
    rider to the restaurant plus a load penalty for every queued order, so
    distance is the primary criterion and load balancing the secondary one.

    Args:
        order: The order to be assigned.

    Returns:
        The chosen :class:`Rider`, or None if the system has no riders at all
        (the caller is then expected to recruit one).
    """
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None

    params = globalvar.get_value("params", {}) or {}
    penalty = int(params.get("load_penalty", DEFAULT_LOAD_PENALTY))

    idle = [r for r in riders if r.state == STATE_IDLE and not r.orders]
    candidates = idle if idle else list(riders)

    best: Optional[Rider] = None
    best_key: Optional[Tuple[int, int]] = None
    for rider in candidates:
        distance = PathPlanner.manhattan(rider.position, order.res_pos)
        cost = distance + penalty * len(rider.orders)
        key = (cost, rider.rider_id)
        if best_key is None or key < best_key:
            best, best_key = rider, key
    return best


class OrderManager:
    """Creates, assigns and tracks delivery orders."""

    def _order_list(self) -> List[Order]:
        """Return the global order history list, creating it if needed."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords: Sequence[Any]) -> Order:
        """Validate coordinates and create a new order.

        Args:
            coords: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            The newly created :class:`Order`, stored in the order history.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        values = validate_coordinates(coords)
        order = Order(values[0:2], values[2:4])
        self._order_list().append(order)
        logger.info("Order %d created", order.id)
        return order

    def assign_order(self, order: Order, system: Any) -> Tuple[Optional[Rider], bool]:
        """Assign ``order`` to the best rider, recruiting one if necessary.

        Args:
            order: The order to assign.
            system: The :class:`system.SystemManager` used for recruitment.

        Returns:
            ``(rider, is_new)``: the assigned rider (None if assignment failed)
            and whether a new rider was recruited for this order.
        """

def init_system() -> "SystemManager":

    Sets the starting funds, an empty rider list, a zeroed rider ID counter,
    the running flag, the tick counter and default parameters.
    """
    globalvar._init()
    globalvar.set_value("params", dict(DEFAULT_PARAMS))
    globalvar.set_value("funds", DEFAULT_PARAMS["initial_funds"])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)

def load_config(path: str = CONFIG_FILE) -> bool:
    """Load system parameters from a JSON configuration file.

    The file may contain a ``"params"`` object whose keys override entries of
    :data:`DEFAULT_PARAMS`. Unknown or invalid entries are ignored with a
    warning.

    Returns:
        True if the file was read and applied, False otherwise.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, ValueError) as exc:
        logger.warning("Could not load configuration %s: %s", path, exc)
        return False

    params = dict(globalvar.get_value("params", DEFAULT_PARAMS) or DEFAULT_PARAMS)
    for key, value in (data.get("params") or {}).items():
        valid = isinstance(value, int) and not isinstance(value, bool)
        if key in DEFAULT_PARAMS and valid and value >= 0:
            if key == "rider_speed" and value <= 0:
                logger.warning("Ignoring non-positive rider_speed")
                continue
            params[key] = value
        else:
            logger.warning("Ignoring invalid configuration entry %r", key)
    globalvar.set_value("params", params)

    if not globalvar.get_value("riders") and int(globalvar.get_value("tick", 0) or 0) == 0:
        globalvar.set_value("funds", params["initial_funds"])
    logger.info("Configuration loaded from %s", path)
    return True


class SystemManager:
    """Manages funds, riders, the simulation clock and system state."""

    @staticmethod
    def _params() -> Dict[str, int]:
        """Return the active parameter dictionary."""
        return globalvar.get_value("params", DEFAULT_PARAMS) or DEFAULT_PARAMS

    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return int(globalvar.get_value("funds", 0) or 0)

    def _set_funds(self, value: int) -> None:
        """Store a new fund balance."""
        globalvar.set_value("funds", int(value))

    def add_funds(self, amount: int) -> None:
        """Add ``amount`` (may be negative) to the fund balance."""
        self._set_funds(self.funds + int(amount))

    @property
    def riders(self) -> List[Rider]:
        """The live rider list."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def tick_count(self) -> int:
        """Number of simulated ticks elapsed."""
        return int(globalvar.get_value("tick", 0) or 0)

    @property
    def is_running(self) -> bool:
        """Whether the main loop should keep running."""
        return bool(globalvar.get_value("running", False))

    def stop(self) -> None:
        """Request the main loop to terminate."""
        globalvar.set_value("running", False)
        logger.info("System stop requested")

    def idle_riders(self) -> List[Rider]:
        """Return riders in the idle state with no queued orders."""
        return [r for r in self.riders if r.state == STATE_IDLE and not r.orders]

    def can_recruit(self) -> bool:
        """Return True if funds and the rider limit allow a new recruit."""
        params = self._params()
        return len(self.riders) < params["max_riders"] and self.funds >= params["rider_cost"]

    def recruit_rider(self) -> Optional[Rider]:
        """Recruit a new rider, deducting the recruitment cost.

        Returns:
            The new :class:`Rider`, or None if recruitment is not possible.
        """
        if not self.can_recruit():
            logger.info("Recruitment skipped (funds=%d, riders=%d)", self.funds, len(self.riders))
            return None
        params = self._params()
        self._set_funds(self.funds - params["rider_cost"])
        rider_id = int(globalvar.get_value("rider_id_counter", 0) or 0) + 1
        globalvar.set_value("rider_id_counter", rider_id)
        rider = Rider(rider_id, speed=params["rider_speed"])
        self.riders.append(rider)
        logger.info("Recruited rider %d (funds now %d)", rider_id, self.funds)
        return rider

    def tick(self) -> int:
        """Advance the simulation by one tick.

        Moves every rider, charges running costs and runs consistency checks.
        Exceptions raised by an individual rider are logged and recovered so
        that one faulty rider cannot stop the simulation.

        Returns:
            The new tick count.
        """
        if not self.is_running:
            return self.tick_count
        globalvar.set_value("tick", self.tick_count + 1)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # noqa: BLE001 - recovery boundary
                logger.exception("Error while moving rider %d", rider.rider_id)
                rider.recover()
        cost = self._params()["running_cost"] * len(self.riders)
        if cost:
            self._set_funds(self.funds - cost)
        self.check_consistency()
        return self.tick_count

    def check_consistency(self) -> List[str]:
        """Detect and repair inconsistent rider states.

        Returns:
            Human readable descriptions of the issues found and repaired.
        """
        issues: List[str] = []
        seen_ids = set()
        for rider in self.riders:
            if rider.rider_id in seen_ids:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen_ids.add(rider.rider_id)

            if rider.state not in (0, 1, 2):
                issues.append(f"rider {rider.rider_id} has invalid state {rider.state}")
                rider.recover()
            elif rider.state != STATE_IDLE and not rider.orders:
                issues.append(f"rider {rider.rider_id} active without orders")
                rider.recover()

            finished = [o for o in rider.orders if o.status == STATUS_COMPLETED]
            if finished:
                issues.append(f"rider {rider.rider_id} still holds completed orders")
                rider.orders = [o for o in rider.orders if o.status != STATUS_COMPLETED]

            if not PathPlanner.is_on_road(rider.x, rider.y):
                issues.append(f"rider {rider.rider_id} off road at ({rider.x}, {rider.y})")
                rider.x = PathPlanner.snap_to_road(rider.x)
                rider.y = PathPlanner.snap_to_road(rider.y)
                rider.recover()

        for issue in issues:
            logger.warning("Consistency check: %s", issue)
        return issues

    def snapshot(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the full system state."""
        return {
            "params": dict(self._params()),
            "funds": self.funds,
            "tick": self.tick_count,
            "rider_id_counter": int(globalvar.get_value("rider_id_counter", 0) or 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in (globalvar.get_value("orders", []) or [])],
        }

    def save_state(self, path: str = DEFAULT_STATE_FILE) -> bool:
        """Persist the system state (parameters, riders, orders) as JSON.

        Returns:
            True on success, False if the file could not be written.
        """
        try:
            with open(path, "w", encoding="utf-8") as handle:
                json.dump(self.snapshot(), handle, indent=2)
        except OSError as exc:
            logger.error("Failed to save state to %s: %s", path, exc)
            return False
        logger.info("State saved to %s", path)
        return True

class DisplayManager:
    """Formats system status and renders the ASCII map.

    The map is drawn on a character canvas of ``2*GRID_SIZE+1`` cells square:
    intersections sit on even rows/columns, streets are drawn between them,
    and blocks are the empty positions at odd/odd coordinates.

    Symbols:
        ``+`` intersection, ``-`` / ``|`` street, `` `` empty block,
        ``*`` planned route, ``R`` restaurant, ``C`` customer,
        ``0``-``9`` rider (rider ID modulo 10).
    """

    SYMBOL_INTERSECTION = "+"
    SYMBOL_STREET_H = "-"
    SYMBOL_STREET_V = "|"
    SYMBOL_EMPTY = " "
    SYMBOL_ROUTE = "*"
    SYMBOL_RESTAURANT = "R"
    SYMBOL_CUSTOMER = "C"

    def __init__(self) -> None:
        self._size = 2 * GRID_SIZE + 1

    @staticmethod
    def to_char(x: int, y: int) -> Tuple[int, int]:
        """Convert a map point to a ``(row, col)`` canvas position.

        Row 0 is the top of the map (``y = 800``).
        """
        size = 2 * GRID_SIZE
        col = int(round(x * 2 / CELL_SIZE))
        row = int(round(size - y * 2 / CELL_SIZE))
        return max(0, min(size, row)), max(0, min(size, col))

    def render_map(
        self,
        riders: Iterable[Any],
        orders: Optional[Iterable[Any]] = None,
        route: Optional[Sequence[Point]] = None,
    ) -> str:
        """Render the street map with riders, orders and an optional route.

        Args:
            riders: Riders to display (by position and ID).
            orders: Orders whose restaurant (R) and customer (C) to display.
            route: Optional list of points drawn as a connected line.

        Returns:
            Multi-line string containing the map and a legend.
        """
        size = self._size
        canvas = [[self.SYMBOL_EMPTY] * size for _ in range(size)]

        for i in range(NODE_COUNT):
            for j in range(NODE_COUNT):
                row, col = self.to_char(i * CELL_SIZE, j * CELL_SIZE)
                canvas[row][col] = self.SYMBOL_INTERSECTION
                if i < GRID_SIZE:
                    canvas[row][col + 1] = self.SYMBOL_STREET_H
                if j < GRID_SIZE:
                    canvas[row - 1][col] = self.SYMBOL_STREET_V

        if route:
            points = [self.to_char(x, y) for x, y in route]
            for (r1, c1), (r2, c2) in zip(points, points[1:]):
                if r1 == r2:
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        canvas[r1][c] = self.SYMBOL_ROUTE
                elif c1 == c2:
                    for r in range(min(r1, r2), max(r1, r2) + 1):
                        canvas[r][c1] = self.SYMBOL_ROUTE
            for row, col in points:
                canvas[row][col] = self.SYMBOL_ROUTE

        for rider in riders:
            row, col = self.to_char(rider.x, rider.y)
            canvas[row][col] = str(rider.rider_id % 10)

        for order in orders or []:
            if order.status not in (STATUS_ASSIGNED, "pending"):
                continue
            row, col = self.to_char(*order.res_pos)
            canvas[row][col] = self.SYMBOL_RESTAURANT
            row, col = self.to_char(*order.cus_pos)
            canvas[row][col] = self.SYMBOL_CUSTOMER

        lines = []
        for r in range(size):
            if r % 2 == 0:
                label = f"{MAP_SIZE - (r // 2) * CELL_SIZE:>4} |"
            else:
                label = "     |"
            lines.append(label + "".join(canvas[r]))
        lines.append("     +" + "-" * size)
        lines.append("      x: 0 (left) to 800 (right); grid lines every 100 units")
        lines.append(self.legend())
        return "\n".join(lines)

import logging
from typing import Callable, List, Optional, Tuple

from order import validate_coordinates

logger = logging.getLogger(__name__)

COMMANDS = (
    "start",
    "status",
    "orders",
    "riders",
    "map",
    "wait",
    "save",
    "history",
    "help",
    "quit",
    "exit",
)

ALIASES = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "w": "wait",
    "f": "save",
    "hi": "history",
    "h": "help",
    "?": "help",
    "q": "quit",
    "x": "exit",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant then customer, 0-800). Prompts if omitted.
  status               Show funds, riders and their current state, plus the map.
  orders               Show all orders with timing, distance and rider.
  riders               Show the event log (spawn, accepted, arrived, delivered) per rider.
  map                  Show the ASCII street map with riders and active orders.
  wait [n]             Let n extra ticks pass (every command already advances one tick).
  save [file]          Save system state to a JSON file (default dispatch_state.json).
  history              Show recent commands.
  help                 Show this help.
  quit / exit          Leave the simulator.
Shortcuts: s=start st=status o=orders r=riders m=map w=wait f=save h=help q=quit
History: !! repeats the last command, !n repeats command number n from 'history'.
"""


class InputManager:
    """Reads, expands and validates user commands."""

    def __init__(
        self,
        reader: Callable[[str], str] = input,
        prompt: str = "dispatch> ",
        max_history: int = 100,
    ) -> None:
        """Create an input manager.

        Args:
            reader: Function used to read a line (``input`` by default).
            prompt: Prompt shown for each command.
            max_history: Number of commands kept in the history.
        """
        self._reader = reader
        self._prompt = prompt
        self._max_history = max_history
        self._history: List[str] = []
        self._readline = None
        try:
            import readline  # noqa: F401  (enables line editing and history on POSIX)

            self._readline = readline
        except ImportError:
            logger.debug("readline not available; line editing disabled")

    @property
    def history(self) -> List[str]:
        """Return a copy of the command history (oldest first)."""
        return list(self._history)

    def read_raw(self, prompt: Optional[str] = None) -> str:
        """Read one raw line of input."""
        return self._reader(prompt if prompt is not None else self._prompt)

    def read_command(self) -> Tuple[str, List[str]]:
        """Read and parse one command line.

        Returns:
            ``(command, arguments)``. ``command`` is an empty string for a blank
            line.

        Raises:
            ValueError: For unknown commands or invalid history references.
        """
        line = self._expand_history(self.read_raw().strip())
        if not line:
            return "", []
        self._remember(line)
        tokens = line.split()
        word = tokens[0].lower()
        command = ALIASES.get(word, word)
        if command not in COMMANDS:
            raise ValueError(f"unknown command '{tokens[0]}'. Type 'help' for a list of commands.")
        return command, tokens[1:]

    def _expand_history(self, raw: str) -> str:
        """Expand ``!!`` and ``!n`` history references."""
        if raw == "!!":
            if not self._history:
                raise ValueError("no previous command in history")
            return self._history[-1]
        if raw.startswith("!") and raw[1:].isdigit():
            index = int(raw[1:])
            if not 1 <= index <= len(self._history):
                raise ValueError(f"history entry {index} does not exist")
            return self._history[index - 1]
        return raw

    def _remember(self, line: str) -> None:
        """Append a command to the history, trimming old entries."""
        self._history.append(line)
        if len(self._history) > self._max_history:
            del self._history[0 : len(self._history) - self._max_history]

    def parse_order_args(self, args: List[str]) -> List[int]:
        """Return validated order coordinates from arguments or a prompt.

        Raises:
            ValueError: If the input is not exactly four integers in range.
        """
        tokens = list(args)
        if not tokens:
            tokens = self.read_raw(
                "Enter restaurant_x restaurant_y customer_x customer_y (0-800): "
            ).split()
        if len(tokens) != 4:
            raise ValueError(f"start needs exactly 4 integers, got {len(tokens)}")
        return validate_coordinates(tokens)

    @staticmethod
    def help_text() -> str:
        """Return the command help text."""
        return HELP_TEXT

Run with ``python src/main.py``. Each iteration of the main loop is one

def setup_logging(debug: bool, log_file: str = LOG_FILE) -> None:
    """Configure logging to a file, falling back to stderr."""
    level = logging.DEBUG if debug else logging.INFO
    fmt = "%(asctime)s %(levelname)s %(name)s: %(message)s"
    try:
        logging.basicConfig(filename=log_file, level=level, format=fmt)
    except OSError:
        logging.basicConfig(level=level, format=fmt)


class DispatchApp:
    """Interactive command loop tying all modules together."""

    def __init__(self, system: SystemManager) -> None:
        self.system = system
        self.orders = OrderManager()
        self.display = DisplayManager()
        self.inputs = InputManager()

    # ------------------------------------------------------------------ #
    # Main loop
    # ------------------------------------------------------------------ #
    def run(self) -> None:
        """Run the simulation loop until the user quits."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands. Each command advances the simulation by one tick.")
        while self.system.is_running:
            try:
                command, args = self.inputs.read_command()
                if command:
                    self.dispatch(command, args)
            except EOFError:
                print()
                break
            except KeyboardInterrupt:
                print()
                break
            except ValueError as exc:
                print(f"Error: {exc}")
            except Exception:  # noqa: BLE001 - keep the simulator alive
                logger.exception("Unexpected error while processing command")
                print("Internal error occurred; the system state has been checked and kept running.")
                self.system.check_consistency()

            if self.system.is_running:
                self.system.tick()
        print("Goodbye.")

    def dispatch(self, command: str, args: List[str]) -> None:
        """Route a parsed command to its handler."""
        handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "wait": self.cmd_wait,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }
        handlers[command](args)

    # ------------------------------------------------------------------ #
    # Command handlers
    # ------------------------------------------------------------------ #
    def cmd_start(self, args: List[str]) -> None:
        """Create an order, assign a rider and show the planned route."""
        coords = self.inputs.parse_order_args(args)
        order = self.orders.create_order(coords)
        rider, is_new = self.orders.assign_order(order, self.system)
        if rider is None:
            print(
                "No rider available: recruitment failed (insufficient funds or rider "
                "limit reached). Order cancelled."
            )
            return
        route = PathPlanner.full_route(rider.position, order.res_pos, order.cus_pos)
        length = PathPlanner.route_length(route)
        estimated = math.ceil(length / rider.speed) if length else 0
        print(self.display.order_report(order, rider, length, estimated, is_new))
        print(self.display.render_map(self.system.riders, orders=[order], route=route))

    def cmd_status(self, args: List[str]) -> None:
        """Show funds, riders and the current map."""
        print(
            self.display.status_report(
                self.system.funds, self.system.riders, self.system.tick_count
            )
        )
        print(
            self.display.render_map(
                self.system.riders, orders=self.orders.active_orders()
            )
        )

    def cmd_orders(self, args: List[str]) -> None:
        """Show all orders."""
        print(self.display.orders_report(self.orders.history()))

    def cmd_riders(self, args: List[str]) -> None:
        """Show the event log of all riders."""
        print(self.display.riders_report(self.system.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map with riders and active orders."""
        print(self.display.render_map(self.system.riders, orders=self.orders.active_orders()))

    def cmd_wait(self, args: List[str]) -> None:
        """Advance the simulation by extra ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise ValueError("wait expects a whole number of ticks") from None
        if not 1 <= count <= MAX_WAIT_TICKS:
            raise ValueError(f"wait count must be between 1 and {MAX_WAIT_TICKS}")
        for _ in range(count - 1):
            self.system.tick()
        print(f"Advanced {count} tick(s); simulated tick is now {self.system.tick_count + 1}"
              f" after this command.")

    def cmd_save(self, args: List[str]) -> None:
        """Save the system state to a JSON file."""
        path = args[0] if args else DEFAULT_STATE_FILE
        if self.system.save_state(path):
            print(f"State saved to {path}.")
        else:
            print(f"Could not write to {path}; see the log for details.")

    def cmd_history(self, args: List[str]) -> None:
        """Print the numbered command history."""
        history = self.inputs.history
        if not history:
            print("No commands yet.")
            return
        for index, entry in enumerate(history, start=1):
            print(f"  {index:>3}  {entry}")

    def cmd_help(self, args: List[str]) -> None:
        """Print help text."""
        print(self.inputs.help_text())

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the simulation loop."""
        self.system.stop()


def main(argv: Optional[List[str]] = None) -> int:
    """Program entry point.

    Args:
        argv: Optional command-line arguments (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=CONFIG_FILE, help="JSON configuration file")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    parser.add_argument("--log-file", default=LOG_FILE, help="log file path")
    args = parser.parse_args(argv)

    setup_logging(args.debug, args.log_file)
    system = init_system()
    if os.path.exists(args.config):
        load_config(args.config)
    app = DispatchApp(system)
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

def normalize_text(value: object) -> str:
    """Collapse whitespace and strip surrounding spaces."""
    return " ".join(str(value).split())


def normalize_key(value: object) -> str:
    """Normalized, case-insensitive key used for comparisons."""
    return normalize_text(value).casefold()

"""Participant list loading, validation and group statistics."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_USERS
from lottery_system.models import User, normalize_key, normalize_text


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in default participant list."""
        users = [User(name, dept) for name, dept in DEFAULT_USERS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            name = normalize_text(parts[0])
            dept = normalize_text(parts[1])
            if not name or not dept:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(name and department group must not be empty)"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence) and duplicates (repeats)."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_key(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department group -> number of people (case-insensitive grouping)."""
        display: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            key = normalize_key(user.department)
            if key not in display:
                display[key] = normalize_text(user.department)
            counts[key] = counts.get(key, 0) + 1
        return {display[key]: counts[key] for key in display}

"""Prize library management: presets, custom prizes, avoidance groups and ordering."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize, normalize_key, normalize_text


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = normalize_key(name)
        for prize in self._prizes:
            if normalize_key(prize.name) == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Load preset prizes. Returns a list of error messages (empty on success)."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize already exists, skipped: {name}")
                continue
            self._prizes.append(Prize(name, quantity, [], drawer))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a custom prize at the end of the lottery sequence."""
        clean_name = normalize_text(name)
        if not clean_name:
            raise ValueError("Prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize already exists: {clean_name}")

        groups: list[str] = []
        seen: set[str] = set()
        for group in avoid_groups or []:
            clean_group = normalize_text(group)
            key = normalize_key(clean_group)
            if clean_group and key not in seen:
                seen.add(key)
                groups.append(clean_group)

        drawer = normalize_text(associated_drawer) if associated_drawer else ""
        prize = Prize(clean_name, quantity, groups, drawer or None)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in lottery order (the Prize objects are shared, not copied)."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility pools, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.default_data import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User, normalize_key

_FALLBACK_TRIGGER_DRAWERS = (normalize_key(CHEN_QIXIAN), normalize_key(JIN_DUO))
_FALLBACK_TARGET_DRAWER = normalize_key(MA_JIAQI)
_SPECIAL_DRAWER = normalize_key(LI_BIN)
_CHENG_RU_KEY = normalize_key(CHENG_RU_GROUP)


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        # Whether each recent draw by Chen Qixian / Jin Duo contained a Cheng Ru member.
        self.last_two_draw_results: list[bool] = []

    # ----- helpers -------------------------------------------------------
    @staticmethod
    def _name_key(user: User) -> str:
        return normalize_key(user.name)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return normalize_key(user.department) == _CHENG_RU_KEY

    @staticmethod
    def _drawer_key(prize: Prize) -> str:
        return normalize_key(prize.associated_drawer or "")

    def is_winner(self, user: User) -> bool:
        """True if the user has already won any prize."""
        return self._name_key(user) in self._winner_keys

    def is_cheng_ru_protection_active(self, prize: Prize) -> bool:
        """Fallback: Ma Jiaqi's prize is restricted to Cheng Ru members when both
        recent Chen Qixian / Jin Duo draws contained no Cheng Ru member."""
        return (
            self._drawer_key(prize) == _FALLBACK_TARGET_DRAWER
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    # ----- rule engine ---------------------------------------------------
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return users who may be drawn for the given prize."""
        available = [user for user in self._users if not self.is_winner(user)]

        # Fallback (Cheng Ru group protection) overrides avoidance groups.
        if self.is_cheng_ru_protection_active(prize):
            protected = [user for user in available if self._is_cheng_ru(user)]
            if protected:
                return protected
            # No Cheng Ru member remains: the normal rules apply.

        # Special rule: Li Bin's prizes only exclude previous winners.
        if self._drawer_key(prize) == _SPECIAL_DRAWER:
            return available

        avoid = {normalize_key(group) for group in prize.avoid_groups}
        return [user for user in available if normalize_key(user.department) not in avoid]

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        parts = ["Previous winners are excluded"]
        if self.is_cheng_ru_protection_active(prize):
            parts.append(
                f"Cheng Ru Group protection active: only {CHENG_RU_GROUP} members "
                "are eligible (avoidance groups ignored)"
            )
        elif self._drawer_key(prize) == _SPECIAL_DRAWER:
            parts.append("special rule for Li Bin: avoidance groups do not apply")
        elif prize.avoid_groups:
            parts.append("avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("no avoidance groups")
        if prize.associated_drawer:
            parts.append(f"lottery person: {prize.associated_drawer}")
        return "; ".join(parts)

    # ----- drawing -------------------------------------------------------
    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winner_keys.add(self._name_key(winner))
        return winner

    def finish_round(self, prize: Prize, winners: list[User]) -> None:
        """Record the outcome of a finished (or terminated) round for fallback tracking."""
        if self._drawer_key(prize) in _FALLBACK_TRIGGER_DRAWERS:
            self.last_two_draw_results.append(any(self._is_cheng_ru(w) for w in winners))
            del self.last_two_draw_results[:-2]

"""Winner tracking, grouped results and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, normalize_key


@dataclass(frozen=True)
class Winner:
    """A winning participant and the prize name they won."""

    user: User
    prize_name: str


class ResultManager:
    """Keeps the winner list in winning order and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._winner_keys: set[str] = set()

    def add_winner(self, user: User, prize: Prize) -> None:
        """Record a winner. Raises ValueError if the person already won."""
        key = normalize_key(user.name)
        if key in self._winner_keys:
            raise ValueError(f"{user.name} has already won a prize")
        self._winner_keys.add(key)
        self._winners.append(Winner(user, prize.name))

    def get_winners(self) -> list[Winner]:
        """All winners in the order they were drawn."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, following the given prize order."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for prize in prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow(
                            [winner.user.name, winner.user.department, winner.prize_name]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from __future__ import annotations

from collections.abc import Callable

from lottery_system.default_data import CHEN_QIXIAN, JIN_DUO, LI_BIN, MA_JIAQI

def ordinal(n: int) -> str:
    """Return an English ordinal such as 1st, 2nd, 11th."""
    if 11 <= n % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[[str], None] = print,
    ) -> None:
        self._input = input_func
        self._output = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self._completed_prizes: set[str] = set()

    # ----- I/O helpers ---------------------------------------------------
    def _say(self, text: str = "") -> None:
        self._output(text)

    def _ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    # ----- main loop -----------------------------------------------------
    def run(self) -> None:
        self._say("=== Team Building Smart Lottery System ===")
        try:
            while True:
                self._print_menu()
                choice = self._ask("Select an option: ")
                if choice == "1":
                    self._load_participants()
                elif choice == "2":
                    self._show_group_info()
                elif choice == "3":
                    self._prize_menu()
                elif choice == "4":
                    self._reorder_prizes()
                elif choice == "5":
                    self._start_lottery()
                elif choice == "6":
                    self._print_summary()
                elif choice == "7":
                    self._export()
                elif choice == "0":
                    self._say("Goodbye!")
                    return
                else:
                    self._say("Invalid option, please try again.")
        except (EOFError, KeyboardInterrupt):
            self._say("\nExiting the lottery system.")

    def _print_menu(self) -> None:
        self._say("")
        self._say("1. Load participant list (default / custom txt)")
        self._say("2. View department groups and headcounts")
        self._say("3. Prize configuration")
        self._say("4. Reorder lottery rounds")
        self._say("5. Start lottery")
        self._say("6. View winner summary")
        self._say("7. Export winner list (CSV txt)")
        self._say("0. Exit")

    # ----- participants --------------------------------------------------
    def _load_participants(self) -> None:
        if self.result_manager.get_winners():
            self._say("Cannot reload the participant list after the lottery has started.")
            return
        self._say("1. Built-in default list")
        self._say("2. Custom txt list (each line: Name,Department Group)")
        choice = self._ask("Select source: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter the txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            self._say("Cancelled.")
            return

        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            self._say("Problems detected in the list:")
            for error in errors:
                self._say(f"  - {error}")
            for dup in duplicates:
                self._say(f"  - Duplicate name ignored: {dup.name} ({dup.department})")
            self._say(
                "Type 'remove' to delete invalid/duplicate entries and continue, "
                "or 'terminate' to cancel the upload."
            )
            decision = self._ask("Your choice: ").lower()
            if decision != "remove":
                self._say("Upload terminated. The participant list was not changed.")
                return

        if not valid:
            self._say("No valid participants loaded.")
            return
        self.users = valid
        self.lottery = None
        self._say(f"Loaded {len(valid)} participants.")

    def _show_group_info(self) -> None:
        if not self.users:
            self._say("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._say("Department groups:")
        for group, count in info.items():
            self._say(f"  {group}: {count} people")

    def _known_groups(self) -> set[str]:
        return {normalize_key(group) for group in self.list_manager.get_group_info(self.users)}

    # ----- prizes --------------------------------------------------------
    def _prize_menu(self) -> None:
        self._say("a. Load preset prizes")
        self._say("b. Add custom prize")
        self._say("c. Set avoidance groups for a prize")
        self._say("d. Show prizes")
        choice = self._ask("Select: ").lower()
        if choice == "a":
            for error in self.prize_manager.load_default_prizes():
                self._say(error)
            self._say("Preset prizes loaded.")
        elif choice == "b":
            self._add_custom_prize()
        elif choice == "c":
            self._set_avoid_groups()
        elif choice == "d":
            self._show_prizes()
        else:
            self._say("Invalid option.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) or "None"
            drawer = prize.associated_drawer or "None"
            self._say(
                f"{index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoid groups: {avoid} | Lottery person: {drawer}"
            )

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        try:
            quantity = int(self._ask("Quantity: "))
        except ValueError:
            self._say("Quantity must be an integer.")
            return
        groups_text = self._ask("Avoidance groups (comma separated, blank for none): ")
        groups = [group.strip() for group in groups_text.split(",") if group.strip()]
        drawer = self._ask("Associated lottery person (blank for none): ") or None
        known = self._known_groups()
        for group in groups:
            if known and normalize_key(group) not in known:
                self._say(f"Warning: group '{group}' is not in the current participant list.")
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            self._say(f"Failed to add prize: {exc}")
            return
        self._say(f"Prize added: {prize.name}")

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self._show_prizes()
        text = self._ask("Enter prize number: ")
        try:
            index = int(text)
        except ValueError:
            self._say("Invalid prize number.")
            return
        if not 1 <= index <= len(prizes):
            self._say("Invalid prize number.")
            return
        groups_text = self._ask("Avoidance groups (comma separated, blank for none): ")
        groups = [group.strip() for group in groups_text.split(",") if group.strip()]
        prize = prizes[index - 1]
        prize.avoid_groups = []
        for group in groups:
            if group and normalize_key(group) not in {normalize_key(g) for g in prize.avoid_groups}:
                prize.avoid_groups.append(group)
        self._say(f"Avoidance groups for '{prize.name}' set to: {', '.join(prize.avoid_groups) or 'None'}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self._show_prizes()
        text = self._ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(item) for item in text.split()]
        except ValueError:
            self._say("Invalid input: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._say("Prize order updated.")
            self._show_prizes()
        else:
            self._say("Invalid order: each prize number must appear exactly once.")

    # ----- lottery -------------------------------------------------------
    def _start_lottery(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not self.users:
            self._say("Please load the participant list first.")
            return
        if not prizes:
            self._say("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)
        pending = [prize for prize in prizes if prize.name not in self._completed_prizes]
        if not pending:
            self._say("All lottery rounds have already been completed.")
            return
        self._say("=== Lottery started ===")
        self._say("During a round: press Enter to draw the next winner, "
                  "type 'pause' to interrupt, 'terminate' to skip the round.")
        for prize in pending:
            self._run_round(prize)
            self._completed_prizes.add(prize.name)
        self._say("=== All lottery rounds completed ===")

    def _run_round(self, prize: Prize) -> None:
        assert self.lottery is not None
        self._say("")
        self._say(f"--- Round: {prize.name} ---")
        self._say(f"Quantity: {prize.quantity}")
        self._say(f"Rule: {self.lottery.describe_rule(prize)}")

        winners: list[User] = []
        while len(winners) < prize.quantity:
            command = self._ask("[Enter]=draw next | pause | terminate: ").lower()
            if command == "pause":
                if not self._wait_while_paused():
                    self._say("Round terminated by operator.")
                    break
            elif command == "terminate":
                self._say("Round terminated by operator.")
                break
            elif command != "":
                self._say("Unknown command.")
                continue

            user = self.lottery.draw_one(prize)
            if user is None:
                self._say("No eligible candidates remain. The round ends early.")
                break
            self.result_manager.add_winner(user, prize)
            winners.append(user)
            self._say(f"Currently drawing the {ordinal(len(winners))} winner...")
            self._say(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self.lottery.finish_round(prize, winners)
        self._say(f"Round results for {prize.name}:")
        if not winners:
            self._say("  (no winners)")
        for index, user in enumerate(winners, start=1):
            self._say(f"  {index}. {user.name} ({user.department})")

    def _wait_while_paused(self) -> bool:
        """Block until 'continue' (returns True) or 'terminate' (returns False)."""
        self._say("Lottery paused. Type 'continue' to resume or 'terminate' to skip this round.")
        while True:
            command = self._ask("paused> ").lower()
            if command == "continue":
                self._say("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            self._say("While paused, only 'continue' or 'terminate' is accepted.")

    # ----- results -------------------------------------------------------
    def _print_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self._say("=== Winner list ===")
        for prize in prizes:
            self._say(f"[{prize.name}]")
            winners = grouped.get(prize.name, [])
            if not winners:
                self._say("  (no winners yet)")
            for winner in winners:
                self._say(f"  {winner.user.name} | {winner.user.department} | {prize.name}")

    def _export(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        _, message = self.result_manager.export_to_csv(prizes)
        self._say(message)


def main() -> None:
    """Start the interactive command-line lottery system."""
    LotterySystemCLI().run()


if __name__ == "__main__":
    main()

"""Core data models for the lottery system."""

from __future__ import annotations

import re
from dataclasses import dataclass, field


def normalize(text: str) -> str:
    """Return a canonical form of a name or group for case-insensitive comparison."""
    return " ".join(str(text).split()).casefold()


@dataclass
class User:
    """A participant in the team building lottery."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    @property
    def level(self) -> int:
        """Prize level derived from the prize name (1 = highest).

        "First", "Second" and "Third" map to levels 1, 2 and 3.
        Custom prizes without a recognised level keyword get a large value so
        they are listed after the standard prize levels.
        """
        match = re.search(r"\b(first|second|third)\b", self.name, re.IGNORECASE)
        if not match:
            return 999
        return {"first": 1, "second": 2, "third": 3}[match.group(1).lower()]

from __future__ import annotations

import re

from lottery_system.models import User, normalize


DEFAULT_LIST_LINES = [
    "Chen Qixian;Technology Department",
    "Jin Duo;Cheng Ru Group",
    "Ma Jiaqi;Cheng Ru Group",
    "Li Bin;Product Department",
    "Zhang San;Technology Department",
    "Li Si;Product Department",
    "Wang Wu;Marketing Department",
    "Zhao Liu;Operations Department",
    "Sun Qi;Cheng Ru Group",
    "Zhou Ba;Technology Department",
    "Wu Jiu;Product Department",
    "Zheng Shi;Cheng Ru Group",
    "Feng Yi;Marketing Department",
    "Chen Er;Operations Department",
    "Chu San;Technology Department",
    "Wei Si;Product Department",
    "Shen Wu;Cheng Ru Group",
    "Han Liu;Marketing Department",
    "Yang Qi;Operations Department",
    "Zhu Ba;Technology Department",
]

_FIELD_SEPARATOR = re.compile(r"[,;]")


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list.

        Returns the valid, de-duplicated users and a list of error messages
        (format errors and duplicate names). Invalid entries are excluded from
        the returned users.
        """
        users, errors = self._load_from_path(path)
        valid_users, duplicates = self.validate_users(users)
        for dup in duplicates:
            errors.append(
                f"Duplicate Name: '{dup.name}' ({dup.department}) already appears in the list and was removed"
            )
        return valid_users, errors

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        """Read a txt file and parse each line as 'Name,Department Group' or 'Name;Department Group'."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}' ({exc})"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw_line in enumerate(lines, start=1):
            line = raw_line.strip()
            if not line:
                continue
            parts = [part.strip() for part in _FIELD_SEPARATOR.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_number}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        valid_users: list[User] = []
        duplicates: list[User] = []
        seen_names: set[str] = set()
        for user in users:
            key = normalize(user.name)
            if key in seen_names:
                duplicates.append(user)
            else:
                seen_names.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        """Return the number of people in each department group, sorted by group name."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

from __future__ import annotations

from lottery_system.models import Prize, normalize


DEFAULT_PRIZES = (
    ("Third Prize - Movie Tickets", 5, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
)


class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes to the end of the current sequence."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, None, drawer)
            except ValueError as exc:
                errors.append(f"Preset prize '{name}' skipped: {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        """Add a prize at the end of the sequence. Raises ValueError on invalid input."""
        clean_name = " ".join(str(name or "").split())
        if not clean_name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        key = normalize(clean_name)
        if any(normalize(prize.name) == key for prize in self._prizes):
            raise ValueError(f"prize '{clean_name}' already exists")
        groups = self._clean_groups(avoid_groups)
        drawer = self._clean_text(associated_drawer)
        prize = Prize(name=clean_name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def update_prize(self, index, avoid_groups=None, associated_drawer=None) -> None:
        """Replace avoidance groups and associated lottery person of the prize at a 0-based index."""
        if not 0 <= index < len(self._prizes):
            raise ValueError("prize index out of range")
        prize = self._prizes[index]
        prize.avoid_groups = self._clean_groups(avoid_groups)
        prize.associated_drawer = self._clean_text(associated_drawer)

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            valid = sorted(order) == list(range(1, count + 1))
        except TypeError:
            return False
        if not valid or len(order) != count:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

    @staticmethod
    def _clean_groups(groups) -> list[str]:
        result: list[str] = []
        seen: set[str] = set()
        for group in groups or []:
            cleaned = " ".join(str(group).split())
            if cleaned and normalize(cleaned) not in seen:
                seen.add(normalize(cleaned))
                result.append(cleaned)
        return result

    @staticmethod
    def _clean_text(value) -> str | None:
        if value is None:
            return None
        cleaned = " ".join(str(value).split())
        return cleaned or None

"""Lottery rule engine: eligible pool calculation and winner drawing."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User, normalize


CHENG_RU_GROUP = "Cheng Ru Group"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"


class LotteryManager:
    """Applies lottery rules and tracks winners across all rounds."""

    def __init__(self, users):
        self.users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self.winners: list[User] = []
        # Results of the most recent draws made by Chen Qixian / Jin Duo.
        # True means the drawn winner belonged to the Cheng Ru Group.
        self.last_two_draw_results: list[bool] = []
        self._rng = random.Random()

    def is_winner(self, user: User) -> bool:
        return normalize(user.name) in self._winner_keys

    def _is_cheng_ru(self, user: User) -> bool:
        return normalize(user.department) == normalize(CHENG_RU_GROUP)

    def _available_users(self) -> list[User]:
        return [user for user in self.users if not self.is_winner(user)]

    def _fallback_active(self) -> bool:
        """True when the last two draws by Chen Qixian / Jin Duo both had no Cheng Ru Group member."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the pool that the next winner of this prize is drawn from.

        - Previous winners are always excluded.
        - Special rule (Li Bin): avoidance groups are not applied.
        - Fallback (Ma Jiaqi): when the last two draws by Chen Qixian and Jin Duo
          had no Cheng Ru Group member, the pool is limited to Cheng Ru Group
          members (even if that group is an avoidance group). If no such member
          is available, the normal rules apply.
        - Avoidance groups: members of avoidance groups are drawn only after all
          other eligible members have been drawn.
        """
        available = self._available_users()
        drawer = normalize(prize.associated_drawer) if prize.associated_drawer else ""

        if drawer == normalize(LI_BIN):
            return available

        if drawer == normalize(MA_JIAQI) and self._fallback_active():
            protected = [user for user in available if self._is_cheng_ru(user)]
            if protected:
                return protected

        avoid = {normalize(group) for group in prize.avoid_groups}
        primary = [user for user in available if normalize(user.department) not in avoid]
        if primary:
            return primary
        return [user for user in available if normalize(user.department) in avoid]

    def draw_winner(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible participant remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_keys.add(normalize(winner.name))
        self.winners.append(winner)

        drawer = normalize(prize.associated_drawer) if prize.associated_drawer else ""
        if drawer in (normalize(CHEN_QIXIAN), normalize(JIN_DUO)):
            self.last_two_draw_results.append(self._is_cheng_ru(winner))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return winner

    def describe_rules(self, prize: Prize) -> list[str]:
        """Return human-readable descriptions of the rules applied to a prize round."""
        rules = ["Previous winners are excluded (no one can win twice)."]
        drawer = normalize(prize.associated_drawer) if prize.associated_drawer else ""
        if drawer == normalize(LI_BIN):
            rules.append("Special rule (Li Bin): avoidance groups do not apply.")
        else:
            if prize.avoid_groups:
                rules.append(
                    "Avoidance groups (members drawn only after other members): "
                    + ", ".join(prize.avoid_groups)
                    + "."
                )
            if drawer == normalize(MA_JIAQI):
                rules.append(
                    "Cheng Ru Group protection: if the last two draws by Chen Qixian and Jin Duo "
                    "contained no Cheng Ru Group member, this draw is limited to Cheng Ru Group members."
                )
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}.")
        return rules

"""Lottery process control: rounds, pause / continue / terminate and live display."""

from __future__ import annotations

import uuid


class LotteryProcess:
    """Runs the multi-round lottery and handles manual control commands."""

    IDLE = "idle"
    RUNNING = "running"
    PAUSED = "paused"
    COMPLETED = "completed"

    def __init__(self, lottery_manager, result_manager, input_func=input, output_func=print):
        self.process_id = uuid.uuid4().hex[:8]
        self.state = self.IDLE
        self.current_round_number = 0
        self.round_statuses: list[str] = []
        self._lottery_manager = lottery_manager
        self._result_manager = result_manager
        self._input = input_func
        self._output = output_func

    def run(self, prizes) -> None:
        """Run one round per prize, in the given order."""
        if not prizes:
            self._output("No prizes configured. Nothing to draw.")
            return
        self.state = self.RUNNING
        for prize in prizes:
            self.current_round_number += 1
            self._run_round(prize)
        self.state = self.COMPLETED
        self._output("All lottery rounds completed.")

    def _run_round(self, prize) -> None:
        round_number = self.current_round_number
        self._output("")
        self._output(f"=== Round {round_number}: {prize.name} | Quantity: {prize.quantity} ===")
        self._output("Rules:")
        for rule in self._lottery_manager.describe_rules(prize):
            self._output(f"  - {rule}")

        round_winners = []
        terminated = False
        for position in range(1, prize.quantity + 1):
            if self._await_control() == "terminate":
                terminated = True
                self._output("Round terminated manually; remaining winners of this round are skipped.")
                break
            self._output(f"Currently drawing the {_ordinal(position)} winner...")
            user = self._lottery_manager.draw_winner(prize)
            if user is None:
                self._output("No eligible participants remain for this prize; the round ends.")
                break
            self._result_manager.add_winner(user, prize)
            round_winners.append(user)
            self._output(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self.round_statuses.append("terminated" if terminated else "completed")
        self._output(f"Round {round_number} results ({len(round_winners)} winner(s), in winning order):")
        if not round_winners:
            self._output("  (no winners in this round)")
        for index, user in enumerate(round_winners, start=1):
            self._output(f"  {index}. {user.name} ({user.department})")

    def _await_control(self) -> str:
        """Wait for a control command before the next draw. Returns 'continue' or 'terminate'."""
        while True:
            command = self._input(
                "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
            ).strip().lower()
            if command in ("", "continue"):
                self.state = self.RUNNING
                return "continue"
            if command == "terminate":
                self.state = self.RUNNING
                return "terminate"
            if command == "pause":
                self.state = self.PAUSED
                self._output("Lottery paused.")
                return self._paused_loop()
            self._output("Unknown command. Use Enter, 'pause' or 'terminate'.")

    def _paused_loop(self) -> str:
        while True:
            command = self._input(
                "Type 'continue' to resume or 'terminate' to end this round: "
            ).strip().lower()
            if command == "continue":
                self.state = self.RUNNING
                self._output("Lottery resumed.")
                return "continue"
            if command == "terminate":
                self.state = self.RUNNING
                return "terminate"
            self._output("Unknown command. Use 'continue' or 'terminate'.")


def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"

"""Winner recording, grouping and CSV-style export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, normalize


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    prize_level: int
    winning_order: int
    winning_time: datetime


class ResultManager:
    """Stores all winners of the lottery session."""

    def __init__(self):
        self._winners: list[Winner] = []
        self._winner_keys: set[str] = set()

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner. Raises ValueError if the same person already won."""
        key = normalize(user.name)
        if key in self._winner_keys:
            raise ValueError(f"'{user.name}' has already won a prize")
        self._winner_keys.add(key)
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=prize.level,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def count(self) -> int:
        return len(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low).

        Within each prize, winners are in winning time order.
        """
        ordered = sorted(prizes, key=lambda prize: prize.level)
        result: dict[str, list[Winner]] = {}
        for prize in ordered:
            if prize.name in result:
                continue
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export winners to 'awards_result_YYYYMMDD.txt' in the current directory, in CSV format."""
        winners_map = self.get_winners_by_prize(prizes)
        file_name = datetime.now().strftime("awards_result_%Y%m%d.txt")
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in winners_map.items():
                    for winner in winners:
                        writer.writerow([
                            winner.user.name,
                            winner.user.department,
                            prize_name,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

            return
        prize = prizes[index]
        current_groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
        groups_text = self._ask(
            f"Avoidance groups (comma-separated; '-' to clear; empty to keep [{current_groups}]): "
        )
        if not groups_text:
            groups = list(prize.avoid_groups)
        elif groups_text == "-":
            groups = []
        else:
            groups = [group.strip() for group in groups_text.split(",") if group.strip()]

        current_drawer = prize.associated_drawer or "None"
        drawer_text = self._ask(
            f"Associated lottery person (empty to keep [{current_drawer}], '-' to clear): "
        )
        if not drawer_text:
            drawer = prize.associated_drawer
        elif drawer_text == "-":
            drawer = None
        else:
            drawer = drawer_text

        try:
            self.prize_manager.update_prize(index, groups, drawer)
            self._say("Prize configuration updated.")
        except ValueError as exc:
            self._say(f"Could not update prize: {exc}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured yet.")
            return
        self._show_prizes()
        text = self._ask(
            "Enter the new order as serial numbers separated by spaces (e.g. 3 1 2): "
        )
        try:
            order = [int(part) for part in text.split()]
        except ValueError:
            self._say("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._say("Lottery sequence updated.")
            self._show_prizes()
        else:
            self._say("Invalid order: every serial number must appear exactly once.")

    def _parse_index(self, text: str, count: int) -> int | None:
        try:
            number = int(text)
        except ValueError:
            self._say("Invalid number.")
            return None
        if not 1 <= number <= count:
            self._say("Number out of range.")
            return None
        return number - 1

    # ----- lottery -----

    def start_lottery(self) -> None:
        if not self.users:
            self._say("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured. Please set up prizes first.")
            return
        if self.result_manager is not None and self.result_manager.count() > 0:
            confirm = self._ask("Previous results will be cleared. Continue? [y/N]: ").lower()
            if confirm != "y":
                return

        self.lottery_prizes = prizes
        self.lottery_manager = LotteryManager(self.users)
        self.result_manager = ResultManager()
        process = LotteryProcess(
            self.lottery_manager,
            self.result_manager,
            input_func=self._input,
            output_func=self._output,
        )
        process.run(prizes)
        self._show_summary(self.lottery_prizes)
        self._offer_export(self.lottery_prizes)

    def results_menu(self) -> None:
        if self.result_manager is None or self.result_manager.count() == 0:
            self._say("No results yet.")
            return
        self._show_summary(self.lottery_prizes)
        self._offer_export(self.lottery_prizes)

    def _show_summary(self, prizes) -> None:
        winners_map = self.result_manager.get_winners_by_prize(prizes)
        self._say("")
        self._say("=== Winner list (by prize level, then winning order) ===")
        for prize_name, winners in winners_map.items():
            self._say(f"{prize_name}:")
            if not winners:
                self._say("  (no winners)")
            for index, winner in enumerate(winners, start=1):
                self._say(f"  {index}. {winner.user.name} ({winner.user.department})")

    def _offer_export(self, prizes) -> None:
        answer = self._ask("Export the winner list to a txt (CSV format) file? [y/N]: ").lower()
        if answer == "y":
            _, message = self.result_manager.export_to_csv(prizes)
            self._say(message)


def main() -> int:
    cli = LotteryCLI()
    try:
        cli.run()

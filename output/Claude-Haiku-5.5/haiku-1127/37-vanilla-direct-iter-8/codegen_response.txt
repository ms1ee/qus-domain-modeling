def normalize(text: str | None) -> str:
    """Return a whitespace-collapsed, case-insensitive comparison key."""
    if not text:

    return " ".join(str(text).split()).casefold()

from __future__ import annotations

import os

from lottery_system.defaults import DEFAULT_USER_LINES
from lottery_system.models import User, normalize

_ENCODINGS = ("utf-8-sig", "gb18030")
_FULLWIDTH_COMMA = "\uff0c"


class ListManager:
    """Loads, verifies and summarizes participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'Name,Department Group'."""
        expanded = os.path.expanduser(path.strip())
        if not os.path.isfile(expanded):
            return [], [f"File Error: file not found: {expanded}"]
        return self._load_from_path(expanded)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse its lines into users and errors."""
        try:
            lines = self._read_lines(path)
        except UnicodeDecodeError:
            return [], [
                f"File Error: '{path}' is not a readable text file "
                "(please use UTF-8 or GBK encoding)."
            ]
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _read_lines(path: str) -> list[str]:
        for encoding in _ENCODINGS[:-1]:
            try:
                with open(path, "r", encoding=encoding) as handle:
                    return handle.read().splitlines()
            except UnicodeDecodeError:
                continue
        with open(path, "r", encoding=_ENCODINGS[-1]) as handle:
            return handle.read().splitlines()

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip().replace(_FULLWIDTH_COMMA, ",")
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            name = " ".join(parts[0].split())
            department = " ".join(parts[1].split())
            users.append(User(name=name, department=department))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence) and duplicate entries."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the department groups with the number of people in each."""
        display: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            key = normalize(user.department)
            if key not in display:
                display[key] = user.department
                counts[key] = 0
            counts[key] += 1
        return {display[key]: counts[key] for key in display}

from __future__ import annotations

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize, normalize


class PrizeManager:
    """Manages the list of prizes and their lottery order."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            if self._find_index(name) is not None:
                errors.append(f"Prize '{name}' already exists and was skipped.")
                continue
            self._prizes.append(Prize(name, quantity, [], None))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Add a custom prize at the end of the lottery order."""
        clean_name = " ".join(str(name).split())
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if self._find_index(clean_name) is not None:
            raise ValueError(f"Prize '{clean_name}' already exists.")
        groups = self._clean_groups(avoid_groups)
        drawer = None
        if associated_drawer and associated_drawer.strip():
            drawer = " ".join(associated_drawer.split())
        self._prizes.append(Prize(clean_name, quantity, groups, drawer))

    def get_prizes(self) -> list[Prize]:
        """Return prizes in lottery order (the Prize objects are shared)."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        try:
            positions = [int(value) for value in order]
        except (TypeError, ValueError):
            return False
        count = len(self._prizes)
        if sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        return True

    def _find_index(self, name: str) -> int | None:
        key = normalize(name)
        for index, prize in enumerate(self._prizes):
            if normalize(prize.name) == key:
                return index
        return None

    @staticmethod
    def _clean_groups(groups: list[str] | None) -> list[str]:
        result: list[str] = []
        seen: set[str] = set()
        for group in groups or []:
            cleaned = " ".join(str(group).split())
            key = normalize(cleaned)
            if cleaned and key not in seen:
                seen.add(key)
                result.append(cleaned)
        return result

"""Lottery rule engine: eligibility filtering, special rules and fallback."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    normalize,
)

_HISTORY_DRAWERS = {normalize(CHEN_QIXIAN), normalize(JIN_DUO)}
_FALLBACK_TARGET = normalize(MA_JIAQI)
_SPECIAL_NO_AVOID = normalize(LI_BIN)
_CHENG_RU_KEY = normalize(CHENG_RU_GROUP)


class LotteryManager:
    """Runs lottery rounds for a fixed list of participants."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self._round_prize: Prize | None = None
        self._round_winners: list[User] = []
        # Round results (True = the round contained Cheng Ru Group members)
        # for the most recent rounds drawn by Chen Qixian / Jin Duo.
        self.last_two_draw_results: list[bool] = []

    @staticmethod
    def _name_key(user: User) -> str:
        return normalize(user.name)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return normalize(user.department) == _CHENG_RU_KEY

    @staticmethod
    def _drawer_key(prize: Prize) -> str:
        return normalize(prize.associated_drawer) if prize.associated_drawer else ""

    def is_winner(self, user: User) -> bool:
        """Return True if the user has already won a prize."""
        return self._name_key(user) in self._winner_keys

    def is_cheng_ru_protection_active(self) -> bool:
        """True when the last two Chen/Jin rounds both had no Cheng Ru members."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users who may win the given prize right now."""
        available = [user for user in self._users if not self.is_winner(user)]
        drawer = self._drawer_key(prize)

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru Group members,
        # overriding avoidance groups.
        if drawer == _FALLBACK_TARGET and self.is_cheng_ru_protection_active():
            return [user for user in available if self._is_cheng_ru(user)]

        # Special rule: Li Bin only filters previous winners.
        if drawer == _SPECIAL_NO_AVOID:
            return available

        avoid = {normalize(group) for group in prize.avoid_groups}
        return [user for user in available if normalize(user.department) not in avoid]

    def has_candidates(self, prize: Prize) -> bool:
        """Return True if at least one eligible candidate remains for the prize."""
        return bool(self._get_eligible_pool(prize))

    def describe_rules(self, prize: Prize) -> str:
        """Return a human-readable description of the rules for a prize round."""
        rules = ["Previous winners are excluded."]
        drawer = self._drawer_key(prize)
        if drawer == _SPECIAL_NO_AVOID:
            rules.append("Special rule for Li Bin: avoidance groups do not apply.")
        elif drawer == _FALLBACK_TARGET and self.is_cheng_ru_protection_active():
            rules.append(
                "Cheng Ru Group protection is active: only Cheng Ru Group members "
                "are eligible (avoidance groups overridden)."
            )
        elif prize.avoid_groups:
            rules.append("Avoidance groups excluded: " + ", ".join(prize.avoid_groups) + ".")
        else:
            rules.append("No avoidance groups.")
        if prize.associated_drawer:
            rules.append(f"Associated lottery person: {prize.associated_drawer}.")
        return " ".join(rules)

    def begin_round(self, prize: Prize) -> None:
        """Start a new round for the prize."""
        self._round_prize = prize
        self._round_winners = []

    def draw_next(self, prize: Prize) -> User | None:
        """Draw the next winner for the prize. Returns None if no candidate remains."""
        if self._round_prize is not prize:
            self.begin_round(prize)
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self._winner_keys.add(self._name_key(user))
        self._round_winners.append(user)
        return user

    def end_round(self, prize: Prize) -> list[User]:
        """Finish the round, update fallback history and return the round winners."""
        winners = list(self._round_winners)
        if self._drawer_key(prize) in _HISTORY_DRAWERS:
            contains_cheng_ru = any(self._is_cheng_ru(user) for user in winners)
            self.last_two_draw_results.append(contains_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        self._round_prize = None
        self._round_winners = []
        return winners

"""Winner recording, grouped result views and CSV-format export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User

_LEVEL_WORDS = {
    "first": 1, "1st": 1,
    "second": 2, "2nd": 2,
    "third": 3, "3rd": 3,
    "fourth": 4, "4th": 4,
    "fifth": 5, "5th": 5,
}
_LEVEL_PATTERN = re.compile(
    r"\b(first|second|third|fourth|fifth|1st|2nd|3rd|4th|5th)\b", re.IGNORECASE
)
_UNRANKED = 999


def prize_level_rank(name: str) -> int:
    """Return the level rank of a prize (1 = highest). Unranked prizes go last."""
    match = _LEVEL_PATTERN.search(name)
    if not match:
        return _UNRANKED
    return _LEVEL_WORDS[match.group(1).lower()]


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    sequence: int
    won_at: str


class ResultManager:
    """Stores winners and produces result views and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for the prize."""
        winner = Winner(
            user=user,
            prize_name=prize.name,
            sequence=len(self._winners) + 1,
            won_at=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level then winning time."""
        indexed = sorted(
            enumerate(prizes),
            key=lambda item: (prize_level_rank(item[1].name), item[0]),
        )
        result: dict[str, list[Winner]] = {}
        for _, prize in indexed:
            matched = [w for w in self._winners if w.prize_name == prize.name]
            result[prize.name] = sorted(matched, key=lambda w: w.sequence)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to the current directory."""
        grouped = self.get_winners_by_prize(prizes)
        rows: list[Winner] = [w for items in grouped.values() for w in items]
        known = set(grouped.keys())
        rows.extend(w for w in self._winners if w.prize_name not in known)

        date_text = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_text}.txt")
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in rows:
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.won_at,
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def ordinal(number: int) -> str:
    """Format an integer as an English ordinal (1st, 2nd, 11th...)."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive command-line controller."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.lottery_completed = False

    # ------------------------------------------------------------------ main
    def run(self) -> None:
        self._print_banner()
        while True:
            self._print_status()
            print("\nMain Menu")
            print("  1. Load participant list")
            print("  2. View department groups")
            print("  3. Configure prizes")
            print("  4. Start lottery")
            print("  5. View results")
            print("  6. Export results to file")
            print("  0. Exit")
            choice = input("Select an option: ").strip()
            if choice == "1":
                self.load_participants()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self.configure_prizes()
            elif choice == "4":
                self.start_lottery()
            elif choice == "5":
                self.show_results()
            elif choice == "6":
                self.export_results()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    @staticmethod
    def _print_banner() -> None:
        print("=" * 60)
        print("  Company Team Building Smart Lottery System")
        print("=" * 60)

    def _print_status(self) -> None:
        print(
            f"\n[Status] Participants: {len(self.users)} | "
            f"Prizes: {len(self.prize_manager.get_prizes())} | "
            f"Lottery completed: {'yes' if self.lottery_completed else 'no'}"
        )

        print("  1. Built-in default team building list")
        print("  2. Custom txt file (each line: Name,Department Group)")
        print("  0. Back")
        choice = input("Select an option: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = input("Enter the txt file path: ").strip().strip('"').strip("'")
            if not path:
                print("No path entered.")
                return
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return
        self._apply_import(users, errors)

    def _apply_import(self, users: list[User], errors: list[str]) -> None:
        if errors:
            print(f"\n{len(errors)} format error(s) found:")
            for error in errors:
                print("  - " + error)
        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"\n{len(duplicates)} duplicate name(s) found:")
            for user in duplicates:
                print(f"  - {user.name} ({user.department})")
        if errors or duplicates:
            print("\nType 'r' to remove invalid and duplicate entries and continue,")
            print("or 'q' to terminate the upload.")
            choice = normalize(input("Your choice: "))
            if choice != "r":
                print("Upload terminated. The previous participant list is unchanged.")
                return
        if not valid:
            print("No valid participants found. Upload aborted.")
            return
        self.users = valid
        self.lottery_completed = False
        print(f"Loaded {len(valid)} participant(s).")
        self._show_groups()

    def _show_groups(self) -> None:
        if not self.users:

    def _add_custom_prize(self) -> None:

        if not quantity_text.isdigit() or int(quantity_text) < 1:

    def _select_groups(self) -> list[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys()) if self.users else []
        if groups:
            print("Available department groups:")
            for index, group in enumerate(groups, start=1):
                print(f"  {index}. {group}")
        else:
            print("(No participant list loaded; group names can be typed directly.)")
        raw = input("Avoidance groups (numbers or names, comma separated, blank for none): ")
        selected: list[str] = []
        seen: set[str] = set()
        for token in raw.split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and groups:
                index = int(token)
                if not 1 <= index <= len(groups):
                    print(f"Invalid group number: {token}")

                name = groups[index - 1]
            else:
                name = next((g for g in groups if normalize(g) == normalize(token)), token)
                if groups and name not in groups:
                    print(f"Warning: group '{token}' is not in the loaded participant list.")
            key = normalize(name)
            if key not in seen:
                seen.add(key)
                selected.append(name)
        return selected

    def _set_prize_groups(self) -> None:
        index = self._select_prize_index()
        if index is None:
            return
        prize = self.prize_manager.get_prizes()[index]
        prize.avoid_groups = self._select_groups()
        print(f"Avoidance groups for '{prize.name}': {', '.join(prize.avoid_groups) or 'None'}")

    def _set_prize_drawer(self) -> None:
        index = self._select_prize_index()
        if index is None:
            return
        prize = self.prize_manager.get_prizes()[index]
        prize.associated_drawer = self._select_drawer()
        print(f"Associated lottery person for '{prize.name}': {prize.associated_drawer or 'None'}")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured yet.")
            return
        print("\nPrizes (lottery order):")
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) or "None"
            drawer = prize.associated_drawer or "None"
            print(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoid groups: {groups} | Lottery person: {drawer}"
            )

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured yet.")
            return
        self._view_prizes()
        raw = input("Enter the new order as serial numbers (e.g. 2 1 3): ")
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(token) for token in tokens]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
        else:
            print("Invalid order: each prize number must appear exactly once.")

    # ------------------------------------------------------------ lottery
    def start_lottery(self) -> None:
        if self.lottery_completed:
            print("The lottery has already been completed in this session.")
            return
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return
        total = sum(prize.quantity for prize in prizes)
        if total > len(self.users):
            print(
                f"Warning: total prize quantity ({total}) exceeds participants "
                f"({len(self.users)}); some rounds may end early."
            )
            if normalize(input("Continue? (y/n): ")) not in ("y", "yes"):
                return

        self.lottery = LotteryManager(self.users)
        print("\nLottery starting! Commands: Enter = draw, 'pause', 'terminate'.")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, len(prizes), prize)
        self.lottery_completed = True
        print("\nAll lottery rounds completed.")
        self.show_results()

    def _run_round(self, round_no: int, total_rounds: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n----- Round {round_no}/{total_rounds}: {prize.name} -----")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {self.lottery.describe_rules(prize)}")
        self.lottery.begin_round(prize)

        drawn = 0
        terminated = False
        while drawn < prize.quantity:
            if not self.lottery.has_candidates(prize):
                print("No eligible candidates remain for this prize.")
                break
            command = normalize(
                input("Press Enter to draw the next winner (or 'pause' / 'terminate'): ")
            )
            if command == "terminate":
                print("Terminating the current round.")
                terminated = True
                break
            if command == "pause":
                if not self._handle_pause():
                    print("Terminating the current round.")
                    terminated = True
                    break
                continue
            if command:
                print("Unknown command. Press Enter to draw, or type 'pause' / 'terminate'.")
                continue

            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = self.lottery.draw_next(prize)
            if user is None:
                print("No eligible candidates remain for this prize.")
                break
            drawn += 1
            self.result_manager.add_winner(user, prize)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        winners = self.lottery.end_round(prize)
        print(f"\nRound finished: {len(winners)} of {prize.quantity} winner(s) drawn.")
        if terminated:
            print("(Round ended early by terminate command.)")
        if winners:
            print("Round results (winning order):")
            for index, user in enumerate(winners, start=1):
                print(f"  {index}. {user.name} ({user.department})")

    @staticmethod
    def _handle_pause() -> bool:
        """Pause the lottery. Returns True to resume, False to terminate the round."""
        print("Lottery paused.")
        while True:
            command = normalize(
                input("Type 'continue' to resume or 'terminate' to end this round: ")
            )
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Unknown command.")

    # ------------------------------------------------------------ results
    def show_results(self) -> None:
        winners = self.result_manager.get_winners()
        if not winners:
            print("No winners recorded yet.")
            return
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n===== Winner List (prize level from high to low) =====")
        for prize_name, items in grouped.items():
            if not items:
                continue
            print(f"\n[{prize_name}]")
            for winner in items:
                print(f"  #{winner.sequence}  {winner.user.name} ({winner.user.department})")

    def export_results(self) -> None:
        if not self.result_manager.get_winners():
            print("No winners recorded yet; nothing to export.")
            return
        success, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(message)


def main() -> None:
    """Entry point for the lottery command-line program."""
    try:
        LotteryCLI().run()
    except (KeyboardInterrupt, EOFError):
        print("\nExiting the lottery system.")

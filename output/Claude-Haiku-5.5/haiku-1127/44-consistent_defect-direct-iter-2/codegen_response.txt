"""Loading of the Beijing Subway network dataset from its JSON file."""

import json
import os

# Absolute path of src/subway_data.json, valid regardless of the working directory.
DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read or has an invalid structure."""


def load_subway_data(path):
    """Load the subway dataset and return its JSON content as a dict.

    The expected structure is:
        {"stations": [{"name": str, "lines": [str]}],
         "connections": [{"source": str, "dest": str, "distance": int, "line": str}]}

    Raises:
        DataLoadError: if the file is missing, unreadable, not valid JSON,
            or does not contain the expected top-level lists.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"subway data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataLoadError(f"cannot read subway data file: {exc}") from exc

    if not isinstance(data, dict):
        raise DataLoadError("subway data must be a JSON object")
    for key in ("stations", "connections"):
        if not isinstance(data.get(key), list):
            raise DataLoadError(f"subway data is missing the list field '{key}'")
    return data

"""Graph data model for the subway network.

The graph is a weighted, directed graph. Each connection is stored as an Edge
that can only be traversed in its listed direction. Station numbers (graph
nodes) are assigned by the caller; the Graph itself only stores them.
"""


class Edge:
    """A directed, weighted connection between two stations on one line."""

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        """
        Args:
            source: station number the edge starts from.
            dest: station number the edge leads to.
            distance: actual distance between the stations in meters.
            line: name of the subway line this connection belongs to.
        """
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Base weight of the edge used by path search (distance in meters)."""
        return self.distance

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}m, {self.line!r})"


class Graph:
    """Weighted directed graph keyed by station numbers, with station line data."""

    def __init__(self):
        # station number -> list of outgoing Edge objects
        self._adj = {}
        # station name -> list of line names the station belongs to
        self.station_lines = {}

    # ----- node management -----
    def add_node(self, node):
        """Add a node (station number) if it does not already exist."""
        if node not in self._adj:
            self._adj[node] = []

    def has_node(self, node):
        """Return True if the node exists in the graph."""
        return node in self._adj

    def nodes(self):
        """Return a list of all node numbers."""
        return list(self._adj.keys())

    @property
    def num_nodes(self):
        """Number of nodes in the graph."""
        return len(self._adj)

    # ----- edge management -----
    def add_edge(self, edge):
        """Add a directed edge, creating its endpoint nodes if needed.

        Raises:
            TypeError: if edge is not an Edge.
            ValueError: if the distance is negative.
        """
        if not isinstance(edge, Edge):
            raise TypeError("edge must be an Edge instance")
        if edge.distance < 0:
            raise ValueError(f"edge distance must be non-negative: {edge!r}")
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self._adj[edge.source].append(edge)

    def edges_from(self, node):
        """Return the outgoing edges of a node (do not mutate the returned list)."""
        return self._adj.get(node, [])

    def edges(self):
        """Yield every directed edge in the graph."""
        for outgoing in self._adj.values():
            for edge in outgoing:
                yield edge

    def find_edges(self, source, dest):
        """Return all edges from source to dest (any line)."""
        return [edge for edge in self._adj.get(source, []) if edge.dest == dest]

    @property
    def num_edges(self):
        """Number of directed edges in the graph."""
        return sum(len(outgoing) for outgoing in self._adj.values())

    # ----- station line data -----
    def set_station_lines(self, station_name, lines):
        """Set the line list of a station (duplicates removed, order kept)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines))

    def get_station_lines(self, station_name):
        """Return a copy of the line list of a station (empty if unknown)."""
        return list(self.station_lines.get(station_name, []))

"""Minimum priority queue implemented as a binary min-heap."""


class MinPQ:
    """Binary min-heap of (key, item) pairs ordered by key.

    Insert and delete-min run in O(log n), which gives Dijkstra's algorithm
    its O((V+E) log V) running time.
    """

    def __init__(self):
        self._keys = []
        self._items = []

    def __len__(self):
        return len(self._keys)

    def is_empty(self):
        """Return True if the queue holds no elements."""
        return not self._keys

    def insert(self, key, item):
        """Insert an item with the given priority key."""
        self._keys.append(key)
        self._items.append(item)
        self._swim(len(self._keys) - 1)

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._keys:
            raise IndexError("priority queue is empty")
        return self._keys[0]

    def del_min(self):
        """Remove and return the (key, item) pair with the smallest key."""
        if not self._keys:
            raise IndexError("priority queue is empty")
        min_key, min_item = self._keys[0], self._items[0]
        last = len(self._keys) - 1
        self._swap(0, last)
        self._keys.pop()
        self._items.pop()
        if self._keys:
            self._sink(0)
        return min_key, min_item

    def _swap(self, i, j):
        self._keys[i], self._keys[j] = self._keys[j], self._keys[i]
        self._items[i], self._items[j] = self._items[j], self._items[i]

    def _swim(self, k):
        """Move the element at index k up until heap order holds."""
        while k > 0:
            parent = (k - 1) // 2
            if self._keys[k] < self._keys[parent]:
                self._swap(k, parent)
                k = parent
            else:
                break

    def _sink(self, k):
        """Move the element at index k down until heap order holds."""
        n = len(self._keys)
        while True:
            left = 2 * k + 1
            right = left + 1
            smallest = k
            if left < n and self._keys[left] < self._keys[smallest]:
                smallest = left
            if right < n and self._keys[right] < self._keys[smallest]:
                smallest = right
            if smallest == k:
                break
            self._swap(k, smallest)
            k = smallest

"""Dijkstra-based path search over the subway graph.

Search states are (station number, line of the arriving edge). Tracking the
arriving line lets the search count line changes (transfers) exactly and
apply transfer penalties, and enforce the maximum transfer limit.

Modes:
    shortest_distance: minimise the total distance in meters.
    least_transfer:    minimise the number of transfers, distance as tie-breaker.
    comprehensive:     minimise distance plus a time-equivalent penalty per transfer.

Line preference: edges on lines not listed in preferred_lines receive a
penalty, so preferred lines are used whenever they do not cost much more.
"""

from min_pq import MinPQ

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Roughly 5 minutes of travel time expressed in meters (1.5 min per km).
TRANSFER_PENALTY_M = 3333.0
# Dominant weight used in least_transfer mode so transfers outweigh distance.
LEAST_TRANSFER_WEIGHT = 1.0e9
# Penalty applied to an edge that is not on a preferred line.
LINE_PREFERENCE_PENALTY_M = 500.0


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: one of VALID_MODES.
        max_transfers: maximum allowed number of transfers, or None for no limit.
        preferred_lines: list of preferred line names (may be empty).
    """

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"unknown mode '{mode}', expected one of {VALID_MODES}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer or None")
            if max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []

    def __repr__(self):
        return (
            f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
            f"preferred_lines={self.preferred_lines!r})"
        )


class _Record:
    """Best known information about one search state."""

    __slots__ = ("cost", "distance", "transfers", "prev", "edge")

    def __init__(self, cost, distance, transfers, prev, edge):
        self.cost = cost            # weighted cost used for ordering
        self.distance = distance    # real travelled distance in meters
        self.transfers = transfers  # number of line changes so far
        self.prev = prev            # previous search state key
        self.edge = edge            # edge used to reach this state


class ShortestPath:
    """Single-source shortest path computation using Dijkstra's algorithm.

    Attributes:
        dist_to: dict station number -> travelled distance (meters) of the
            chosen best path to that station. Unreachable stations are absent.
    """

    def __init__(self, graph, source, config=None):
        """
        Args:
            graph: models.Graph instance.
            source: station number of the starting station.
            config: PathfindingConfig, defaults to shortest distance without limits.

        Raises:
            ValueError: if the source station is not in the graph.
        """
        if not graph.has_node(source):
            raise ValueError(f"unknown source station number: {source}")
        self._graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._records = {}   # search state -> _Record
        self._chosen = {}    # station number -> best search state reaching it
        self._run()

    # ----- core algorithm -----
    def _edge_cost(self, edge, transfer):
        """Compute the weighted cost of traversing an edge."""
        cfg = self.config
        cost = edge.weight
        if cfg.mode == "least_transfer":
            if transfer:
                cost += LEAST_TRANSFER_WEIGHT
        elif cfg.mode == "comprehensive":
            if transfer:
                cost += TRANSFER_PENALTY_M
        if cfg.preferred_lines and edge.line not in cfg.preferred_lines:
            cost += LINE_PREFERENCE_PENALTY_M
        return cost

    def _run(self):
        """Run Dijkstra's algorithm over (station, line) search states."""
        start = (self.source, None)
        records = self._records
        records[start] = _Record(0.0, 0.0, 0, None, None)

        pq = MinPQ()
        pq.insert(0.0, start)
        settled = set()
        max_transfers = self.config.max_transfers

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)
            node, line = state
            rec = records[state]

            for edge in self._graph.edges_from(node):
                transfer = line is not None and edge.line != line
                transfers = rec.transfers + (1 if transfer else 0)
                if max_transfers is not None and transfers > max_transfers:
                    continue
                new_state = (edge.dest, edge.line)
                if new_state in settled:
                    continue
                new_cost = rec.cost + self._edge_cost(edge, transfer)
                old = records.get(new_state)
                if old is None or new_cost < old.cost:
                    records[new_state] = _Record(
                        new_cost, rec.distance + edge.weight, transfers, state, edge
                    )
                    pq.insert(new_cost, new_state)

        # Choose the cheapest search state for every reachable station.
        for state, rec in records.items():
            node = state[0]
            current = self._chosen.get(node)
            if current is None or rec.cost < records[current].cost:
                self._chosen[node] = state
        for node, state in self._chosen.items():
            self.dist_to[node] = records[state].distance

    # ----- public query API -----
    def has_path_to(self, target):
        """Return True if the target station is reachable from the source."""
        return target in self._chosen

    def path_with_edges(self, target):
        """Return the path to target as (station numbers, edges).

        The station list runs from the source to the target and contains one
        more element than the edge list. Unreachable targets yield ([], []).
        """
        state = self._chosen.get(target)
        if state is None:
            return [], []
        nodes = []
        edges = []
        while state is not None:
            rec = self._records[state]
            nodes.append(state[0])
            if rec.edge is not None:
                edges.append(rec.edge)
            state = rec.prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        """Return the number of transfers on the chosen path to target (0 if unreachable)."""
        state = self._chosen.get(target)
        if state is None:
            return 0
        return self._records[state].transfers

    def get_cost(self, target):
        """Return the weighted search cost of the chosen path to target (inf if unreachable)."""
        state = self._chosen.get(target)
        if state is None:
            return float("inf")
        return self._records[state].cost

"""Fare and travel time calculation for the Beijing Subway."""

import math

# Base fare tiers as (upper bound in meters, fare in yuan).
# Boundary values follow the PRD examples: 6 km -> 3 yuan, 12 km -> 4 yuan,
# 22 km -> 6 yuan, 32 km -> 6 yuan.
EXTENDED_THRESHOLD_M = 32000   # distances above this use the extended tier
EXTENDED_BASE_FARE = 7         # fare for (32 km, 52 km]
EXTENDED_STEP_M = 20000        # +1 yuan per additional 20 km

ESTIMATED_MINUTES_PER_KM = 1.5
ESTIMATED_MINUTES_PER_TRANSFER = 5.0


def _base_fare(meters):
    """Return the base-tier fare for a distance that is at most 32 km.

    Intervals (PRD boundary rules):
        0 - 6 km (inclusive)     -> 3 yuan
        6 - 12 km (inclusive)    -> 4 yuan
        12 - 22 km (exclusive)   -> 5 yuan
        22 - 32 km (inclusive)   -> 6 yuan
    """
    if meters <= 6000:
        return 3
    if meters <= 12000:
        return 4
    if meters < 22000:
        return 5
    return 6


def Sum_money(distance):
    """Calculate the subway fare for a travelled distance.

    Args:
        distance: distance in meters (int or float). Fractional meters are
            rounded up before the fare is determined.

    Returns:
        The fare in yuan as an int.

    Raises:
        TypeError: if distance is not a number.
        ValueError: if distance is negative, NaN or infinite.
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of meters")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("distance must be a finite number")
    if distance < 0:
        raise ValueError("distance must be non-negative")

    meters = math.ceil(distance)
    if meters > EXTENDED_THRESHOLD_M:
        extra = meters - EXTENDED_THRESHOLD_M
        return EXTENDED_BASE_FARE + (extra - 1) // EXTENDED_STEP_M
    return _base_fare(meters)


def calculate_estimated_time(distance, transfer_count):
    """Estimate the travel time in minutes.

    Args:
        distance: distance in meters.
        transfer_count: number of transfers.

    Returns:
        Estimated time in minutes (about 1.5 min per km plus 5 min per transfer).
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of meters")
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int):
        raise TypeError("transfer_count must be an integer")
    if distance < 0 or transfer_count < 0:
        raise ValueError("distance and transfer_count must be non-negative")
    return (distance / 1000.0) * ESTIMATED_MINUTES_PER_KM + \
        transfer_count * ESTIMATED_MINUTES_PER_TRANSFER

"""Subway network management.

Responsibilities:
    * station registry (station names <-> station numbers)
    * construction of the weighted directed graph from the dataset
    * station name standardisation (nearest-match resolution)
    * data update mechanisms (add lines, modify stations, update distances)
    * data integrity checks
"""

import difflib
import re
from collections import deque

from models import Edge, Graph

# Main lines that must be present in the network for coverage checks.
REQUIRED_MAIN_LINES = (
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
)

MIN_REASONABLE_DISTANCE_M = 100
MAX_REASONABLE_DISTANCE_M = 5000


def _normalize(name):
    """Normalise a station name for lenient comparison."""
    return re.sub(r"[\s\-_'\u2019]", "", name).lower()


class SubwayNetwork:
    """In-memory subway network built from the subway dataset.

    Attributes:
        station_to_number: dict station name -> station number.
        number_to_name: dict station number -> station name.
        graph: models.Graph holding the directed weighted edges and station lines.
    """

    def __init__(self, data):
        """Build the network from a dataset dict (see data_loader.load_subway_data).

        Raises:
            ValueError: if the data is not a dict or contains invalid entries.
        """
        if not isinstance(data, dict):
            raise ValueError("subway data must be a dict")
        self.station_to_number = {}
        self.number_to_name = {}
        self.graph = Graph()
        self._build(data)

    # ----- construction -----
    def _build(self, data):
        """Populate stations, lines and edges from the dataset."""
        for station in data.get("stations", []):
            self._ensure_station(station["name"], station.get("lines", []))
        for conn in data.get("connections", []):
            line = str(conn["line"]).strip()
            src = self._ensure_station(conn["source"], [line])
            dst = self._ensure_station(conn["dest"], [line])
            self.graph.add_edge(Edge(src, dst, int(conn["distance"]), line))

    def _ensure_station(self, name, lines=()):
        """Return the number of a station, creating it if it does not exist."""
        name = str(name).strip()
        if not name:
            raise ValueError("station name cannot be empty")
        number = self.station_to_number.get(name)
        if number is None:
            number = len(self.number_to_name)
            self.station_to_number[name] = number
            self.number_to_name[number] = name
            self.graph.add_node(number)
            self.graph.set_station_lines(name, [])
        for line in lines:
            self._attach_line(name, str(line).strip())
        return number

    def _attach_line(self, name, line):
        """Add a line to a station's line list if it is not already present."""
        if not line:
            return
        current = self.graph.get_station_lines(name)
        if line not in current:
            current.append(line)
            self.graph.set_station_lines(name, current)

    # ----- queries -----
    def stations(self):
        """Return a sorted list of all official station names."""
        return sorted(self.station_to_number)

    @property
    def station_count(self):
        """Number of stations in the network."""
        return len(self.station_to_number)

    def station_lines(self, name):
        """Return the lines of a station by official name (empty if unknown)."""
        return self.graph.get_station_lines(name)

    def lines(self):
        """Return a sorted list of all line names known to the network."""
        result = set()
        for station_line_list in self.graph.station_lines.values():
            result.update(station_line_list)
        return sorted(result)

    def line_stations(self):
        """Return dict line name -> sorted list of station names served by that line."""
        index = {}
        for edge in self.graph.edges():
            bucket = index.setdefault(edge.line, set())
            bucket.add(self.number_to_name[edge.source])
            bucket.add(self.number_to_name[edge.dest])
        return {line: sorted(names) for line, names in index.items()}

    def transfer_stations(self):
        """Return sorted names of stations served by more than one line."""
        return [name for name in self.stations() if len(self.station_lines(name)) > 1]

    def resolve_station(self, query):
        """Standardise a user-supplied station name.

        Returns:
            (station number, official station name, exact_match flag).
            Unknown names are matched to the most similar official name.

        Raises:
            ValueError: if the query is empty or the network has no stations.
        """
        if query is None or not str(query).strip():
            raise ValueError("station name cannot be empty")
        text = str(query).strip()
        if text in self.station_to_number:
            return self.station_to_number[text], text, True

        index = {}
        for official in self.station_to_number:
            index.setdefault(_normalize(official), official)
        if not index:
            raise ValueError("the network contains no stations")

        normalized = _normalize(text)
        if normalized in index:
            official = index[normalized]
            return self.station_to_number[official], official, True

        best_key = max(
            index,
            key=lambda key: difflib.SequenceMatcher(None, normalized, key).ratio(),
        )
        official = index[best_key]
        return self.station_to_number[official], official, False

    # ----- data updates -----
    def _require_number(self, name):
        """Return the number of an official station name or raise ValueError."""
        if name not in self.station_to_number:
            raise ValueError(f"station '{name}' does not exist")
        return self.station_to_number[name]

    def add_line(self, line_name, stations, distances, bidirectional=True):
        """Add a new line described by an ordered list of stations.

        Args:
            line_name: name of the new line; must not already exist.
            stations: ordered station names along the line (at least two).
            distances: distances in meters between consecutive stations.
            bidirectional: if True, add the reverse direction as well.

        Returns:
            Number of directed edges added.

        Raises:
            ValueError: on invalid input.
        """
        line_name = str(line_name).strip()
        if not line_name:
            raise ValueError("line name cannot be empty")
        if line_name in self.lines():
            raise ValueError(f"line '{line_name}' already exists")
        names = [str(s).strip() for s in stations]
        if len(names) < 2:
            raise ValueError("a line needs at least two stations")
        if len(set(names)) != len(names):
            raise ValueError("a line cannot contain duplicate stations")
        if len(distances) != len(names) - 1:
            raise ValueError("number of distances must equal stations minus one")
        for distance in distances:
            if isinstance(distance, bool) or not isinstance(distance, int) or distance <= 0:
                raise ValueError("distances must be positive integers (meters)")

        numbers = [self._ensure_station(name, [line_name]) for name in names]
        added = 0
        for src, dst, distance in zip(numbers, numbers[1:], distances):
            self.graph.add_edge(Edge(src, dst, distance, line_name))
            added += 1
            if bidirectional:
                self.graph.add_edge(Edge(dst, src, distance, line_name))
                added += 1
        return added

    def update_station(self, name, new_name=None, lines=None):
        """Modify a station's line list and/or rename it.

        Returns:
            The (possibly new) official name of the station.

        Raises:
            ValueError: if the station does not exist or the new name is taken.
        """
        number = self._require_number(name)
        official = name
        if lines is not None:
            cleaned = [str(line).strip() for line in lines if str(line).strip()]
            self.graph.set_station_lines(official, cleaned)
        if new_name is not None and str(new_name).strip() and new_name.strip() != official:
            target = new_name.strip()
            if target in self.station_to_number:
                raise ValueError(f"station '{target}' already exists")
            self.graph.station_lines[target] = self.graph.station_lines.pop(official)
            del self.station_to_number[official]
            self.station_to_number[target] = number
            self.number_to_name[number] = target
            official = target
        return official

    def update_distance(self, source_name, dest_name, distance, bidirectional=True):
        """Update the distance of the connection between two stations.

        Returns:
            Number of directed edges updated.

        Raises:
            ValueError: on invalid distance, unknown station or missing connection.
        """
        if isinstance(distance, bool) or not isinstance(distance, int) or distance <= 0:
            raise ValueError("distance must be a positive integer (meters)")
        src = self._require_number(source_name)
        dst = self._require_number(dest_name)
        updated = 0
        for edge in self.graph.find_edges(src, dst):
            edge.distance = distance
            updated += 1
        if bidirectional:
            for edge in self.graph.find_edges(dst, src):
                edge.distance = distance
                updated += 1
        if updated == 0:
            raise ValueError(f"no connection between '{source_name}' and '{dest_name}'")
        return updated

    # ----- integrity checks -----
    def check_integrity(self):
        """Run data integrity checks.

        Returns:
            A list of human-readable issue strings prefixed with [ERROR] or [WARNING].
            An empty list means no issues were found.
        """
        issues = []
        index = self.line_stations()

        # Coverage of main lines.
        for line in REQUIRED_MAIN_LINES:
            count = len(index.get(line, []))
            if count == 0:
                issues.append(f"[ERROR] main line '{line}' is missing from the network")
            elif count < 2:
                issues.append(f"[WARNING] main line '{line}' has only {count} station")

        # Station data.
        connected = set()
        for edge in self.graph.edges():
            connected.add(edge.source)
            connected.add(edge.dest)
        for name in self.stations():
            if not self.station_lines(name):
                issues.append(f"[ERROR] station '{name}' has no line assignment")
            if self.station_to_number[name] not in connected:
                issues.append(f"[WARNING] station '{name}' has no connections")

        # Connection integrity and distance accuracy.
        seen = {}
        for edge in self.graph.edges():
            src = self.number_to_name[edge.source]
            dst = self.number_to_name[edge.dest]
            label = f"{src} -> {dst} ({edge.line})"
            if edge.line not in self.station_lines(src) or edge.line not in self.station_lines(dst):
                issues.append(f"[WARNING] {label}: line not listed for both endpoint stations")
            if edge.distance <= 0:
                issues.append(f"[ERROR] {label}: distance must be positive, got {edge.distance}")
            elif edge.distance < MIN_REASONABLE_DISTANCE_M:
                issues.append(f"[WARNING] {label}: suspiciously short distance {edge.distance} m")
            elif edge.distance > MAX_REASONABLE_DISTANCE_M:
                issues.append(f"[WARNING] {label}: suspiciously long distance {edge.distance} m")

            key = (edge.source, edge.dest, edge.line)
            if key in seen and seen[key] != edge.distance:
                issues.append(
                    f"[ERROR] {label}: conflicting duplicate distances "
                    f"{seen[key]} m and {edge.distance} m"
                )
            seen.setdefault(key, edge.distance)

            reverse = [e for e in self.graph.find_edges(edge.dest, edge.source) if e.line == edge.line]
            if not reverse:
                issues.append(f"[WARNING] {label}: reverse connection missing (one-way only)")

        # Connectivity of the network (ignoring direction).
        undirected = {node: set() for node in self.graph.nodes()}
        for edge in self.graph.edges():
            undirected[edge.source].add(edge.dest)
            undirected[edge.dest].add(edge.source)
        visited = set()
        components = 0
        for start in undirected:
            if start in visited:
                continue
            components += 1
            visited.add(start)
            queue = deque([start])
            while queue:
                node = queue.popleft()
                for neighbour in undirected[node]:
                    if neighbour not in visited:
                        visited.add(neighbour)
                        queue.append(neighbour)
        if components > 1:
            issues.append(f"[WARNING] network is split into {components} disconnected parts")

        return issues

"""Route planning service.

Combines station resolution, path search, fare calculation and segment
analysis into a single RouteResult that the report module can render.
"""

from dataclasses import dataclass, field
from typing import List, Optional

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


class RouteError(Exception):
    """Raised when a route query cannot be answered (bad input, no route, calculation failure)."""


@dataclass
class RouteResult:
    """Result of a single route query.

    Attributes:
        start / end: official station names.
        start_query / end_query: station names as typed by the user.
        start_exact / end_exact: True if the input matched an official name exactly.
        stations: station names from the start to the end (travel order).
        edge_lines: line name of every traversed edge.
        distance_m: total travelled distance in meters.
        fare: fare in yuan.
        transfer_count: number of transfers.
        estimated_minutes: estimated travel time in minutes.
        segments: list of dicts {"line", "from", "to", "stations"} in travel order.
        transfers: list of dicts {"station", "from_line", "to_line", "station_lines"}.
        involved_lines: unique line names in order of first use.
    """

    start: str
    end: str
    start_query: str
    end_query: str
    start_exact: bool
    end_exact: bool
    mode: str
    max_transfers: Optional[int]
    preferred_lines: List[str]
    stations: List[str]
    edge_lines: List[str]
    distance_m: float
    fare: int
    transfer_count: int
    estimated_minutes: float
    segments: List[dict] = field(default_factory=list)
    transfers: List[dict] = field(default_factory=list)
    involved_lines: List[str] = field(default_factory=list)


def _build_segments(stations, edge_lines):
    """Group consecutive edges of the same line into segments."""
    segments = []
    for index, line in enumerate(edge_lines):
        start_station = stations[index]
        end_station = stations[index + 1]
        if not segments or segments[-1]["line"] != line:
            segments.append({
                "line": line,
                "from": start_station,
                "to": end_station,
                "stations": [start_station, end_station],
            })
        else:
            segments[-1]["to"] = end_station
            segments[-1]["stations"].append(end_station)
    return segments


def _build_transfers(network, stations, edge_lines):
    """Identify the stations where the traveller changes line."""
    transfers = []
    for index in range(1, len(edge_lines)):
        if edge_lines[index - 1] != edge_lines[index]:
            station = stations[index]
            transfers.append({
                "station": station,
                "from_line": edge_lines[index - 1],
                "to_line": edge_lines[index],
                "station_lines": network.station_lines(station),
            })
    return transfers


def plan_route(network, start_query, end_query, config=None):
    """Plan the route between two stations and compute its fare.

    Args:
        network: subway_network.SubwayNetwork instance.
        start_query: starting station name as entered by the user.
        end_query: ending station name as entered by the user.
        config: pathfinding.PathfindingConfig (defaults to shortest distance).

    Returns:
        RouteResult describing the route.

    Raises:
        RouteError: on invalid input, unreachable destinations or calculation errors.
    """
    if config is None:
        config = PathfindingConfig()

    try:
        start_num, start_name, start_exact = network.resolve_station(start_query)
        end_num, end_name, end_exact = network.resolve_station(end_query)
    except ValueError as exc:
        raise RouteError(str(exc)) from exc

    if start_num == end_num:
        raise RouteError("Starting station cannot be the same as the ending station")

    try:
        search = ShortestPath(network.graph, start_num, config)
    except ValueError as exc:
        raise RouteError(f"path calculation failed: {exc}") from exc

    if not search.has_path_to(end_num):
        if config.max_transfers is not None:
            raise RouteError(
                f"no route from '{start_name}' to '{end_name}' within "
                f"{config.max_transfers} transfer(s)"
            )
        raise RouteError(f"no route from '{start_name}' to '{end_name}'")

    try:
        node_numbers, edges = search.path_with_edges(end_num)
        stations = [network.number_to_name[number] for number in node_numbers]
        edge_lines = [edge.line for edge in edges]
        distance = search.dist_to[end_num]
        transfer_count = search.get_transfer_count(end_num)
    except (KeyError, IndexError) as exc:
        raise RouteError(f"path calculation failed: {exc}") from exc

    try:
        fare = Sum_money(distance)
        estimated = calculate_estimated_time(distance, transfer_count)
    except (TypeError, ValueError) as exc:
        raise RouteError(f"fare calculation failed: {exc}") from exc

    involved = []
    for line in edge_lines:
        if line not in involved:
            involved.append(line)

    return RouteResult(
        start=start_name,
        end=end_name,
        start_query=str(start_query).strip(),
        end_query=str(end_query).strip(),
        start_exact=start_exact,
        end_exact=end_exact,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        stations=stations,
        edge_lines=edge_lines,
        distance_m=distance,
        fare=fare,
        transfer_count=transfer_count,
        estimated_minutes=estimated,
        segments=_build_segments(stations, edge_lines),
        transfers=_build_transfers(network, stations, edge_lines),
        involved_lines=involved,
    )

"""Path visualisation and report export.

Renders a RouteResult as console text or as JSON and writes reports to files.
"""

import json


def format_distance(meters):
    """Format a distance: meters for short distances, kilometers for long ones."""
    if meters < 1000:
        return f"{meters:.0f} m"
    return f"{meters / 1000:.2f} km ({meters:.0f} m)"


def build_summary(result):
    """Extract the key figures of a route into a dict."""
    return {
        "total_distance_m": round(result.distance_m, 1),
        "fare": result.fare,
        "transfer_count": result.transfer_count,
        "estimated_time_min": round(result.estimated_minutes, 1),
    }


def render_text(result):
    """Render a route as console-formatted text."""
    lines = []
    lines.append("=" * 60)
    lines.append(f"Query: {result.start} -> {result.end}")
    lines.append(f"Mode: {result.mode}")
    if result.max_transfers is not None:
        lines.append(f"Max transfers: {result.max_transfers}")
    if result.preferred_lines:
        lines.append("Line preference: " + ", ".join(result.preferred_lines))
    lines.append("-" * 60)

    lines.append("Station sequence (end -> start):")
    lines.append("  " + " -> ".join(reversed(result.stations)))
    lines.append("")

    lines.append("Route segments (start -> end):")
    for segment in result.segments:
        lines.append(
            f"  [{segment['line']}] " + " -> ".join(segment["stations"])
            + f" ({len(segment['stations']) - 1} stop(s))"
        )
    if result.transfers:
        lines.append("")
        lines.append("Transfers:")
        for transfer in result.transfers:
            lines.append(
                f"  Transfer at {transfer['station']}: "
                f"{transfer['from_line']} -> {transfer['to_line']} "
                f"(station lines: {', '.join(transfer['station_lines'])})"
            )
    else:
        lines.append("")
        lines.append("Transfers: none")

    lines.append("-" * 60)
    lines.append(f"Total distance: {format_distance(result.distance_m)}")
    lines.append(f"Transfers: {result.transfer_count}")
    lines.append(f"Estimated time: {result.estimated_minutes:.1f} min")
    lines.append("Involved lines: " + ", ".join(result.involved_lines))
    lines.append(f"Fare: {result.fare} yuan")
    lines.append("=" * 60)
    return "\n".join(lines)


def render_json(result):
    """Render a route as a JSON document for programmatic use."""
    payload = {
        "query": {
            "start": result.start,
            "end": result.end,
            "input_start": result.start_query,
            "input_end": result.end_query,
            "mode": result.mode,
            "max_transfers": result.max_transfers,
            "line_preference": result.preferred_lines,
        },
        "path": list(reversed(result.stations)),
        "path_order": "end_to_start",
        "distance_m": round(result.distance_m, 1),
        "distance_km": round(result.distance_m / 1000.0, 3),
        "fare": result.fare,
        "estimated_time_min": round(result.estimated_minutes, 1),
        "transfer_count": result.transfer_count,
        "involved_lines": list(result.involved_lines),
        "segments": result.segments,
        "transfers": result.transfers,
        "summary": build_summary(result),
    }
    return json.dumps(payload, ensure_ascii=False, indent=2)


def write_report(content, output_file, append=True):
    """Write a rendered report to a file (appending by default).

    Raises:
        OSError: if the file cannot be written.
    """
    mode = "a" if append else "w"
    with open(output_file, mode, encoding="utf-8") as handle:
        handle.write(content)
        handle.write("\n")

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import re
import sys

from data_loader import DATA_FILE_PATH, DataLoadError, load_subway_data
from path_report import render_json, render_text, write_report
from pathfinding import VALID_MODES, PathfindingConfig
from route_service import RouteError, plan_route
from subway_network import SubwayNetwork

EXIT_WORDS = ("quit", "exit", "q")
FORMAT_CHOICES = ("text", "json")


def build_parser():
    """Create the argument parser for the command line."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (CLI)."
    )
    parser.add_argument("--output_file", default=None,
                        help="append query results to this file")
    parser.add_argument("--format", choices=FORMAT_CHOICES, default="text",
                        help="output format: text or json (default: text)")
    parser.add_argument("--mode", choices=VALID_MODES, default="shortest_distance",
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="maximum number of transfers allowed")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line(s), comma separated, e.g. 'Line 1'")
    parser.add_argument("--check_data", action="store_true",
                        help="run the data integrity check and exit")
    return parser


def parse_line_preference(value):
    """Split a comma-separated line preference string into a list of line names."""
    if not value:
        return []
    return [item.strip() for item in value.split(",") if item.strip()]


def parse_query(text):
    """Parse 'Starting Station; Ending Station' (a comma is also accepted).

    Returns:
        (start, end) station name strings.

    Raises:
        ValueError: if the input does not contain exactly two non-empty names.
    """
    parts = re.split(r"[;,]", text)
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise ValueError(
            "Input format incorrect, please use 'Starting Station; Ending Station' format"
        )
    return parts[0].strip(), parts[1].strip()


def run_check(network):
    """Print the data integrity report and return an exit code (1 on errors)."""
    issues = network.check_integrity()
    if not issues:
        print("Data integrity check passed: no issues found.")
        return 0
    for issue in issues:
        print(issue)
    errors = sum(1 for issue in issues if issue.startswith("[ERROR]"))
    warnings = len(issues) - errors
    print(f"Integrity check finished: {errors} error(s), {warnings} warning(s).")
    return 1 if errors else 0


def describe_match(query, official, exact):
    """Return a notice when the input was matched to a nearest official name."""
    if exact:
        return ""
    return (f"Note: '{query}' is not an official station name; "
            f"matched to nearest station '{official}'.")


def run_cli(argv=None):
    """Run the CLI. Returns the process exit code."""
    args = build_parser().parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be non-negative", file=sys.stderr)
        return 2

    try:
        data = load_subway_data(DATA_FILE_PATH)
        network = SubwayNetwork(data)
    except (DataLoadError, ValueError, KeyError, TypeError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        return run_check(network)

    preferred = parse_line_preference(args.line_preference)
    known_lines = set(network.lines())
    for line in preferred:
        if line not in known_lines:
            print(f"Warning: preferred line '{line}' is not in the network",
                  file=sys.stderr)

    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=preferred)
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    print("Beijing Subway fare calculator.")
    print("Enter 'Starting Station; Ending Station' (or 'quit' to exit).")

    while True:
        try:
            raw = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in EXIT_WORDS:
            break

        try:
            start_query, end_query = parse_query(text)
        except ValueError as exc:
            print(f"Error: {exc}")
            continue

        try:
            result = plan_route(network, start_query, end_query, config)
        except RouteError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # defensive: never terminate the query loop
            print(f"Error: unexpected failure during path calculation: {exc}")
            continue

        for query, official, exact in ((start_query, result.start, result.start_exact),
                                       (end_query, result.end, result.end_exact)):
            notice = describe_match(query, official, exact)
            if notice:
                print(notice, file=sys.stderr)

        output = render_json(result) if args.format == "json" else render_text(result)
        print(output)

        if args.output_file:
            try:
                write_report(output, args.output_file, append=True)
                print(f"Result appended to {args.output_file}")
            except OSError as exc:
                print(f"Error: cannot write output file: {exc}", file=sys.stderr)

    return 0

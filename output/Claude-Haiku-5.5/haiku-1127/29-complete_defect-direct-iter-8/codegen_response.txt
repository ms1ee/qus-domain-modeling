"""Global variable store shared by the dispatch simulator modules.

The store is a plain dictionary accessed through :func:`set_value` and
:func:`get_value`. Calling :func:`_init` restores every key to its default.

Keys used by the program:
    riders           list of Rider objects (dynamic CRUD list)
    rider_id_counter int, last rider ID handed out
    funds            int, system fund balance (may be negative)
    running          bool, controls the main loop
    tick             int, simulated time (one tick per loop iteration)
    orders           list of Order objects
    config           dict of system parameters
"""

from typing import Any, Dict

_DEFAULT_CONFIG: Dict[str, int] = {
    "recruit_threshold": 300,
    "recruit_cost": 300,
    "order_fee": 150,
    "rider_speed": 50,
    "max_orders_per_rider": 3,
}

_store: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store to its default values.

    After this call the rider list is empty, the rider ID counter is 0,
    funds are 0, the running flag is True, the tick is 0, the order list
    is empty and the configuration holds the default parameters.
    """
    global _store
    _store = {
        "riders": [],
        "rider_id_counter": 0,
        "funds": 0,
        "running": True,
        "tick": 0,
        "orders": [],
        "config": dict(_DEFAULT_CONFIG),
    }


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Any Python object to store.
    """
    _store[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` when the key is missing.
    """
    return _store.get(key, default)

"""Grid-based path planning for the food delivery simulator.

The map is an 8x8 grid of 100x100 unit cells (800x800 units). Locations are
represented by cell-center points, computed as
``center = (value // 100) * 100 + 50``. Movement is always axis-aligned
(along streets) and uses right-angle turns.
"""

import logging
from typing import Iterable, List, Optional, Sequence, Set, Tuple

GRID_SIZE = 8
CELL_SIZE = 100
MAP_SIZE = GRID_SIZE * CELL_SIZE

Point = Tuple[int, int]

logger = logging.getLogger("dispatch.path")


def clamp_coordinate(value: int) -> int:
    """Clamp a coordinate into the valid map range ``[0, MAP_SIZE]``.

    Args:
        value: Raw integer coordinate.

    Returns:
        The clamped coordinate.
    """
    return max(0, min(MAP_SIZE, int(value)))


def grid_index(value: int) -> int:
    """Return the grid index (0-7) of a coordinate.

    Coordinates equal to ``MAP_SIZE`` (800) belong to the last cell.

    Args:
        value: Raw integer coordinate.

    Returns:
        Grid index in ``range(GRID_SIZE)``.
    """
    return min(clamp_coordinate(value) // CELL_SIZE, GRID_SIZE - 1)


def to_cell_center(value: int) -> int:
    """Convert one coordinate to the center of its grid cell.

    Args:
        value: Raw integer coordinate in range 0-800.

    Returns:
        ``(value // 100) * 100 + 50`` with boundary clamping.
    """
    return grid_index(value) * CELL_SIZE + CELL_SIZE // 2


def to_cell_point(x: int, y: int) -> Point:
    """Convert a raw (x, y) pair to the center point of its grid cell.

    Args:
        x: Raw x coordinate.
        y: Raw y coordinate.

    Returns:
        Tuple ``(center_x, center_y)``.
    """
    return to_cell_center(x), to_cell_center(y)


def manhattan(a: Point, b: Point) -> int:
    """Return the Manhattan (street-grid) distance between two points.

    Args:
        a: First point.
        b: Second point.

    Returns:
        ``|ax - bx| + |ay - by|``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class PathPlanner:
    """Pathfinding logic for riders on the street grid.

    All methods are stateless; the class groups the grid-based algorithm
    and its special-case handling.
    """

    @staticmethod
    def is_special_case(start: Point, goal: Point) -> bool:
        """Return True when the goal is directly above or below the start.

        Such targets are reached by a single straight vertical movement.

        Args:
            start: Current position.
            goal: Target position.

        Returns:
            True if both points share an x coordinate and differ in y.
        """
        return start[0] == goal[0] and start[1] != goal[1]

    @staticmethod
    def plan(start: Point, goal: Point,
             occupied: Optional[Iterable[Point]] = None) -> List[Point]:
        """Plan waypoints from ``start`` to ``goal`` along streets.

        Straight targets (same row or column) are reached directly. Otherwise
        a single right-angle corner is used. When both possible corners are
        available the planner prefers the corner that is not occupied by
        another rider, which reduces path conflicts.

        Args:
            start: Current position (cell-center point).
            goal: Target position (cell-center point).
            occupied: Positions currently held by other riders.

        Returns:
            List of waypoints excluding ``start`` and ending at ``goal``.
            Empty list when already at the goal.
        """
        if start == goal:
            return []
        sx, sy = start
        gx, gy = goal
        if sx == gx or sy == gy:
            return [goal]

        blocked: Set[Point] = set(occupied or ())
        horizontal_first = (gx, sy)
        vertical_first = (sx, gy)
        if horizontal_first in blocked and vertical_first not in blocked:
            corner = vertical_first
        else:
            corner = horizontal_first
        return [corner, goal]

    @staticmethod
    def full_route(start: Point, stops: Sequence[Point],
                   occupied: Optional[Iterable[Point]] = None) -> List[Point]:
        """Build a continuous route visiting ``stops`` in order.

        Args:
            start: Starting position.
            stops: Ordered destinations (e.g. restaurant then customer).
            occupied: Positions held by other riders.

        Returns:
            List of points beginning with ``start``.
        """
        occupied_set = set(occupied or ())
        points: List[Point] = [start]
        current = start
        for stop in stops:
            points.extend(PathPlanner.plan(current, stop, occupied_set))
            current = stop
        return points

    @staticmethod
    def route_length(points: Sequence[Point]) -> int:
        """Return the total street length of a route.

        Args:
            points: Consecutive route points.

        Returns:
            Sum of Manhattan distances between consecutive points.
        """
        return sum(manhattan(a, b) for a, b in zip(points, points[1:]))

    @staticmethod
    def step_towards(pos: Point, path: Sequence[Point],
                     speed: int) -> Tuple[Point, List[Point], int]:
        """Advance a position along a waypoint path by up to ``speed`` units.

        This is a pure function: the caller stores the returned position and
        remaining path.

        Args:
            pos: Current position.
            path: Remaining waypoints.
            speed: Maximum movement distance this tick.

        Returns:
            Tuple ``(new_position, remaining_path, distance_moved)``.
        """
        x, y = pos
        remaining: List[Point] = list(path)
        budget = max(0, int(speed))
        moved = 0
        while budget > 0 and remaining:
            tx, ty = remaining[0]
            dx = tx - x
            dy = ty - y
            if dx == 0 and dy == 0:
                remaining.pop(0)
                continue
            if dx != 0:
                step = min(budget, abs(dx))
                x += step if dx > 0 else -step
            else:
                step = min(budget, abs(dy))
                y += step if dy > 0 else -step
            budget -= step
            moved += step
            if (x, y) == (tx, ty):
                remaining.pop(0)
        return (x, y), remaining, moved

"""Rider model: state machine, movement and event logging.

State transitions:
    Idle (0)        -> Picking Up (1) when an assigned order is accepted.
    Picking Up (1)  -> Delivering (2) on arrival at the restaurant.
    Delivering (2)  -> Idle (0) on delivery to the customer.
"""

import logging
from typing import Any, Dict, List, Optional, Set

import globalvar
from path_planner import PathPlanner, Point

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {
    IDLE: "idle",
    PICKING_UP: "picking up",
    DELIVERING: "delivering",
}

SPAWN_POSITION: Point = (50, 50)
DEFAULT_SPEED = 50

logger = logging.getLogger("dispatch.rider")


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x: Current x coordinate (integer, 0-800).
        y: Current y coordinate (integer, 0-800).
        speed: Movement speed in units per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders, served in list order.
        special_case: True while moving straight to a target above/below.
        path: Remaining waypoints for the current leg.
        log: Chronological list of event dictionaries.
    """

    def __init__(self, rider_id: int) -> None:
        """Create an idle rider at the spawn position.

        Args:
            rider_id: Unique identifier assigned by the system.
        """
        config = globalvar.get_value("config") or {}
        self.rider_id: int = int(rider_id)
        self.x: int = SPAWN_POSITION[0]
        self.y: int = SPAWN_POSITION[1]
        self.speed: int = int(config.get("rider_speed", DEFAULT_SPEED))
        self.state: int = IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.path: List[Point] = []
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    @property
    def position(self) -> Point:
        """Return the current ``(x, y)`` position."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Return a human-readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider.

        Returns:
            Dictionary with rider attributes; orders are stored as IDs.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
        }

    def move(self) -> None:
        """Advance the rider by one tick.

        Handles order acceptance, movement along the planned path, arrival at
        the restaurant, and delivery to the customer. All state transitions
        and their log entries happen here.
        """
        if self.state == IDLE:
            if not self.orders:
                return
            self._begin_pickup()

        self._advance()

        while not self.path:
            if self.state == PICKING_UP:
                self._arrive_at_merchant()
            elif self.state == DELIVERING:
                self._deliver()
                break
            else:
                break

    def _now(self) -> int:
        """Return the current simulated tick."""
        return int(globalvar.get_value("tick", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log.

        Args:
            event: One of "spawn", "accepted_order", "arrived_at_merchant",
                "delivered_to_customer".
            order_id: Related order ID, or None.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": self._now(),
            "rider_id": self.rider_id,
            "state": self.state,
            "x": self.x,
            "y": self.y,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s (order %s)", self.rider_id, event, order_id)

    def _occupied_by_others(self) -> Set[Point]:
        """Return positions occupied by all other riders."""
        riders = globalvar.get_value("riders", []) or []
        return {rider.position for rider in riders if rider is not self}

    def _plan(self, goal: Point) -> None:
        """Plan the path for the current leg toward ``goal``."""
        self.path = PathPlanner.plan(self.position, goal,
                                     self._occupied_by_others())
        self.special_case = PathPlanner.is_special_case(self.position, goal)

    def _advance(self) -> None:
        """Move along the current path using this rider's speed."""
        if not self.path:
            return
        new_pos, self.path, _ = PathPlanner.step_towards(
            self.position, self.path, self.speed)
        self.x, self.y = new_pos

    def _begin_pickup(self) -> None:
        """Accept the next assigned order and head to the restaurant."""
        order = self.orders[0]
        self.state = PICKING_UP
        order.status = "pickup"
        self._record("accepted_order", order.id)
        self._plan(order.res_pos)

    def _arrive_at_merchant(self) -> None:
        """Handle arrival at the restaurant and head to the customer."""
        order = self.orders[0]
        self.state = DELIVERING
        order.status = "delivering"
        self._record("arrived_at_merchant", order.id)
        self._plan(order.cus_pos)

    def _deliver(self) -> None:
        """Complete the current order and return to idle."""
        order = self.orders.pop(0)
        order.mark_completed(self._now())
        self.state = IDLE
        self.special_case = False
        self.path = []
        self._record("delivered_to_customer", order.id)

"""Order data model.

An order is created from ``[restaurant_x, restaurant_y, customer_x,
customer_y]`` user input. Coordinates are validated (0-800, integers) and
converted to grid cell-center points.
"""

from typing import Any, Dict, Optional, Sequence, Tuple

import globalvar
from path_planner import MAP_SIZE, Point, manhattan, to_cell_point


def _validate_point(point: Sequence[int], label: str) -> Tuple[int, int]:
    """Validate a raw ``(x, y)`` pair.

    Args:
        point: Two-element sequence of integers.
        label: Name used in error messages.

    Returns:
        Validated ``(x, y)`` tuple.

    Raises:
        ValueError: If the pair does not have two elements or is out of range.
        TypeError: If a coordinate is not an integer.
    """
    try:
        x, y = point
    except (TypeError, ValueError) as exc:
        raise ValueError(
            f"{label} position must contain exactly two coordinates") from exc
    values = []
    for value in (x, y):
        if isinstance(value, bool) or not isinstance(value, int):
            raise TypeError(
                f"{label} coordinates must be integers, got {value!r}")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(
                f"{label} coordinate {value} is outside 0-{MAP_SIZE}")
        values.append(value)
    return values[0], values[1]


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order identifier.
        restaurant_raw: Raw restaurant coordinates as entered by the user.
        customer_raw: Raw customer coordinates as entered by the user.
        res_pos: Restaurant grid cell-center point.
        cus_pos: Customer grid cell-center point.
        status: "pending", "assigned", "pickup", "delivering" or "completed".
        rider_id: ID of the assigned rider, or None.
        start_time: Tick when the order was created.
        completed_time: Tick when delivery finished, or None.
        distance: Manhattan street distance between restaurant and customer.
        duration: Ticks taken from creation to completion, or None.
        fee: Amount charged to the customer when the order is placed.
    """

    _counter = 0

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create and validate a new order.

        Args:
            res_pos: ``(restaurant_x, restaurant_y)``, integers in 0-800.
            cus_pos: ``(customer_x, customer_y)``, integers in 0-800.

        Raises:
            ValueError: On malformed or out-of-range coordinates.
            TypeError: On non-integer coordinates.
        """
        self.restaurant_raw = _validate_point(res_pos, "Restaurant")
        self.customer_raw = _validate_point(cus_pos, "Customer")

        Order._counter += 1
        self.id: int = Order._counter

        self.res_pos: Point = to_cell_point(*self.restaurant_raw)
        self.cus_pos: Point = to_cell_point(*self.customer_raw)

        config = globalvar.get_value("config") or {}
        self.fee: int = int(config.get("order_fee", 150))
        self.status: str = "pending"
        self.rider_id: Optional[int] = None
        self.start_time: int = int(globalvar.get_value("tick", 0) or 0)
        self.completed_time: Optional[int] = None
        self.distance: int = manhattan(self.res_pos, self.cus_pos)
        self.duration: Optional[int] = None

    def assign_to(self, rider_id: int) -> None:
        """Record the rider this order is assigned to.

        The caller is responsible for appending the order to the rider's
        ``orders`` list.

        Args:
            rider_id: ID of the rider receiving the order.
        """
        self.rider_id = int(rider_id)
        self.status = "assigned"

    def mark_completed(self, tick: int) -> None:
        """Mark the order as delivered and record completion data.

        Args:
            tick: Simulated tick at which delivery finished.
        """
        self.completed_time = int(tick)
        self.duration = max(0, self.completed_time - self.start_time)
        self.status = "completed"

    @property
    def efficiency(self) -> Optional[float]:
        """Return delivery efficiency in distance units per tick.

        Returns:
            ``distance / max(duration, 1)``, or None if not yet completed.
        """
        if self.duration is None:
            return None
        return round(self.distance / max(self.duration, 1), 2)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the order.

        Returns:
            Dictionary with all order attributes.
        """
        return {
            "id": self.id,
            "restaurant": list(self.restaurant_raw),
            "customer": list(self.customer_raw),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completed_time": self.completed_time,
            "distance": self.distance,
            "duration": self.duration,
            "fee": self.fee,
            "efficiency": self.efficiency,
        }

"""Order assignment and order management.

Assignment strategy: the closest available rider is preferred. A load
penalty is added per already-assigned order so that work is balanced across
riders. Riders already at their order capacity are skipped.
"""

import logging
from typing import List, Optional

import globalvar
from order_model import Order
from path_planner import manhattan
from rider import Rider

LOAD_PENALTY = 200

logger = logging.getLogger("dispatch.order")


def _max_orders_per_rider() -> int:
    """Return the configured per-rider order capacity."""
    config = globalvar.get_value("config") or {}
    return int(config.get("max_orders_per_rider", 3))


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Choose the best rider for an order.

    Score = Manhattan distance from the rider to the restaurant plus
    ``LOAD_PENALTY`` times the number of orders the rider already holds.
    The lowest score wins; ties go to the lower rider ID.

    Args:
        order: The order needing a rider.

    Returns:
        The chosen Rider, or None when no rider has spare capacity.
    """
    riders = globalvar.get_value("riders", []) or []
    capacity = _max_orders_per_rider()
    best: Optional[Rider] = None
    best_score: Optional[int] = None
    for rider in riders:
        if len(rider.orders) >= capacity:
            continue
        distance = manhattan(rider.position, order.res_pos)
        score = distance + LOAD_PENALTY * len(rider.orders)
        if (best is None or score < best_score or
                (score == best_score and rider.rider_id < best.rider_id)):
            best = rider
            best_score = score
    return best


class OrderManager:
    """Creates, assigns and tracks orders.

    All orders are stored in the global ``"orders"`` list.
    """

    def create_order(self, restaurant_pos, customer_pos) -> Order:
        """Create an order and register it in the global order list.

        Args:
            restaurant_pos: ``(x, y)`` restaurant coordinates.
            customer_pos: ``(x, y)`` customer coordinates.

        Returns:
            The newly created Order.
        """
        order = Order(restaurant_pos, customer_pos)
        orders = globalvar.get_value("orders")
        orders.append(order)
        logger.info("Created order %s", order.id)
        return order

    def assign_order(self, order: Order) -> Optional[Rider]:
        """Assign an order to the best available rider.

        Args:
            order: Order to assign.

        Returns:
            The rider the order was assigned to, or None if none is free.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            logger.info("No rider available for order %s", order.id)
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Assigned order %s to rider %s", order.id, rider.rider_id)
        return rider

    def assign_pending_orders(self) -> List[Order]:
        """Try to assign every pending order, oldest first.

        Returns:
            Orders that were assigned during this call.
        """
        assigned: List[Order] = []
        for order in list(globalvar.get_value("orders", []) or []):
            if order.status != "pending":
                continue
            if self.assign_order(order) is not None:
                assigned.append(order)
        return assigned

    def pending_orders(self) -> List[Order]:
        """Return orders that are still waiting for a rider."""
        return [o for o in globalvar.get_value("orders", []) or []
                if o.status == "pending"]

    def completed_orders(self) -> List[Order]:
        """Return orders that have been delivered."""
        return [o for o in globalvar.get_value("orders", []) or []
                if o.status == "completed"]

"""System-level management: funds, rider recruitment, ticks and persistence."""

import json
import logging
import os
from typing import Any, Dict, List, Optional

import globalvar
from order import OrderManager
from path_planner import CELL_SIZE, GRID_SIZE, MAP_SIZE
from rider import IDLE, Rider

logger = logging.getLogger("dispatch.system")


def init_system() -> None:
    """Initialise all core parameters on system startup.

    Resets the global store: empty rider list, rider ID counter 0,
    running state True, zero funds, tick 0, empty order list, default
    configuration, and the 8x8 grid description.
    """
    globalvar._init()
    globalvar.set_value("grid", {
        "size": GRID_SIZE,
        "cell_size": CELL_SIZE,
        "map_size": MAP_SIZE,
    })
    logger.info("System initialised")


class SystemManager:
    """Manages funds, riders, orders, the simulation clock and persistence."""

    def __init__(self) -> None:
        """Create the manager; call :func:`init_system` beforehand."""
        self.order_manager = OrderManager()

    # ----- funds and configuration -------------------------------------
    @property
    def funds(self) -> int:
        """Return the current system fund balance."""
        return int(globalvar.get_value("funds", 0))

    def add_funds(self, amount: int) -> int:
        """Add ``amount`` (may be negative) to the fund balance.

        Args:
            amount: Integer amount to add.

        Returns:
            The new fund balance.
        """
        new_value = self.funds + int(amount)
        globalvar.set_value("funds", new_value)
        return new_value

    def _config(self, key: str, default: int) -> int:
        """Read an integer configuration value."""
        config = globalvar.get_value("config") or {}
        return int(config.get(key, default))

    # ----- rider CRUD ----------------------------------------------------
    @property
    def riders(self) -> List[Rider]:
        """Return the live rider list."""
        return globalvar.get_value("riders")

    def get_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with ``rider_id``, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def create_rider(self) -> Rider:
        """Create a new idle rider with the next unique ID.

        Returns:
            The newly created Rider.
        """
        counter = int(globalvar.get_value("rider_id_counter", 0)) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        logger.info("Recruited rider %s", counter)
        return rider

    def remove_rider(self, rider_id: int) -> bool:
        """Remove an idle rider with no assigned orders.

        Args:
            rider_id: ID of the rider to remove.

        Returns:
            True if the rider was removed, False otherwise.
        """
        rider = self.get_rider(rider_id)
        if rider is None or rider.orders:
            return False
        self.riders.remove(rider)
        logger.info("Removed rider %s", rider_id)
        return True

    def recruit_riders(self) -> List[Rider]:
        """Recruit riders while funds are at or above the threshold.

        Each recruit costs ``recruit_cost`` from the fund balance.

        Returns:
            List of riders created by this call.
        """
        threshold = self._config("recruit_threshold", 300)
        cost = self._config("recruit_cost", 300)
        new_riders: List[Rider] = []
        if cost <= 0:
            return new_riders
        while self.funds >= threshold:
            self.add_funds(-cost)
            new_riders.append(self.create_rider())
        return new_riders

    # ----- orders --------------------------------------------------------
    def submit_order(self, restaurant_pos, customer_pos) -> Dict[str, Any]:
        """Create an order, charge the customer, recruit and assign.

        Args:
            restaurant_pos: ``(x, y)`` restaurant coordinates.
            customer_pos: ``(x, y)`` customer coordinates.

        Returns:
            Dictionary with keys "order", "rider" (Rider or None),
            "new_riders" (list) and "new_rider" (bool).
        """
        order = self.order_manager.create_order(restaurant_pos, customer_pos)
        self.add_funds(order.fee)
        new_riders = self.recruit_riders()
        rider = self.order_manager.assign_order(order)
        return {
            "order": order,
            "rider": rider,
            "new_riders": new_riders,
            "new_rider": bool(new_riders),
        }

    # ----- simulation clock ----------------------------------------------
    @property
    def tick_count(self) -> int:
        """Return the current simulated tick."""
        return int(globalvar.get_value("tick", 0))

    def tick(self) -> int:
        """Advance the simulation by one unit of time.

        Assigns pending orders, then moves every rider. Exceptions in a
        single rider are logged and that rider is reset to a safe state so
        the rest of the system keeps running.

        Returns:
            The new tick value.
        """
        globalvar.set_value("tick", self.tick_count + 1)
        try:
            self.order_manager.assign_pending_orders()
        except Exception:  # pragma: no cover - defensive
            logger.exception("Failed to assign pending orders")
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:
                logger.exception("Rider %s failed; recovering", rider.rider_id)
                self._recover_rider(rider)
        return self.tick_count

    def _recover_rider(self, rider: Rider) -> None:
        """Reset a rider to a consistent idle state after an error."""
        for order in rider.orders:
            if order.status in ("pickup", "delivering"):
                order.status = "assigned"
        rider.state = IDLE
        rider.path = []
        rider.special_case = False

    # ----- consistency ---------------------------------------------------
    def check_consistency(self) -> List[str]:
        """Check internal state consistency.

        Returns:
            List of human-readable problems; empty when consistent.
        """
        issues: List[str] = []
        seen_ids = set()
        for rider in self.riders:
            if rider.rider_id in seen_ids:
                issues.append(f"Duplicate rider ID {rider.rider_id}")
            seen_ids.add(rider.rider_id)
            if rider.state not in (0, 1, 2):
                issues.append(f"Rider {rider.rider_id} has invalid state")
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    issues.append(
                        f"Order {order.id} not linked to rider {rider.rider_id}")
                if order.status == "completed":
                    issues.append(
                        f"Completed order {order.id} still held by rider "
                        f"{rider.rider_id}")
        if not isinstance(globalvar.get_value("funds"), int):
            issues.append("Fund balance is not an integer")
        for order in globalvar.get_value("orders", []):
            if order.rider_id is not None and self.get_rider(order.rider_id) is None:
                if order.status != "completed":
                    issues.append(
                        f"Order {order.id} references missing rider "
                        f"{order.rider_id}")
        return issues

    # ----- persistence ---------------------------------------------------
    def save_state(self, path: str) -> None:
        """Write system parameters, riders and orders to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.

        data = {
            "funds": self.funds,
            "tick": self.tick_count,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "config": dict(globalvar.get_value("config") or {}),
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict()
                       for order in globalvar.get_value("orders", [])],
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        logger.info("State saved to %s", path)

    def load_config(self, path: str) -> Dict[str, Any]:
        """Load system parameters from a JSON configuration file.

        The file may either hold the parameters at top level or inside a
        "config" object. Unknown keys are ignored and invalid values fall
        back to the current setting. A missing file keeps the defaults.

        Args:
            path: Configuration file path.

        Returns:
            The active configuration dictionary.
        """
        config = dict(globalvar.get_value("config") or {})
        if not os.path.exists(path):
            logger.info("Config file %s not found; using defaults", path)
            return config
        try:
            with open(path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
        except (OSError, json.JSONDecodeError) as exc:
            logger.warning("Could not read config %s: %s", path, exc)
            return config
        if not isinstance(data, dict):
            logger.warning("Config %s must contain a JSON object", path)
            return config

        source = data.get("config") if isinstance(data.get("config"), dict) \
            else data
        minimums = {
            "recruit_threshold": 0,
            "recruit_cost": 1,
            "order_fee": 0,
            "rider_speed": 1,
            "max_orders_per_rider": 1,
        }
        for key, minimum in minimums.items():
            value = source.get(key)
            if isinstance(value, int) and not isinstance(value, bool) \
                    and value >= minimum:
                config[key] = value
            elif value is not None:
                logger.warning("Ignoring invalid config value %s=%r", key, value)
        globalvar.set_value("config", config)
        return config

def _canvas_pos(point: Point) -> tuple:
    """Map a unit coordinate to a (row, col) position on the ASCII canvas.

    Each grid cell occupies canvas position (2*row, 2*col); the odd positions
    between cells are used for street connectors.
    """
    col = 2 * grid_index(point[0])
    row = 2 * (GRID_SIZE - 1 - grid_index(point[1]))
    return row, col

                        canvas[r][col_a] = "#"

"""Command-line interface: parsing, validation and command dispatch.

Commands:
    start [rx ry cx cy]  submit an order (prompts when no arguments given)
    status               system funds, riders and their orders
    orders               all orders with timing and assignment data
    riders               full status-change log for all riders
    map                  current ASCII map
    wait [n]             advance the simulation by n ticks (default 1)
    save [path]          save state to a JSON file
    history              list previous commands
    help                 show this help
    quit / exit          leave the simulator

Shortcuts: s, st, o, r, m, w, h/?, q, x, '!!' (repeat last), '!n' (run
history entry n). After every command the simulation advances one tick.
"""

import logging
from typing import Callable, List, Optional, Sequence

import globalvar
from display import DisplayManager
from path_planner import MAP_SIZE

try:  # readline gives arrow-key history on supported platforms
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    readline = None

PROMPT = "dispatch> "
WELCOME = ("Food Delivery Dispatch Simulator. "
           "Type 'help' for the command list.")
HELP_TEXT = """Commands:
  start [rx ry cx cy]  submit order (restaurant then customer, 0-800)
  status               funds, riders and order lists
  orders               order table with timing and riders
  riders               rider status-change log
  map                  ASCII map (R rider, S restaurant, C customer)
  wait [n]             advance n ticks (default 1)
  save [path]          save state to JSON (default dispatch_state.json)
  history              show command history
  help                 show this help
  quit | exit          leave the program
Shortcuts: s st o r m w h ? q x  |  !! repeat last  |  !n run history n
Each command advances the simulation by one tick."""

logger = logging.getLogger("dispatch.input")


class InputManager:
    """Parses user commands, validates input and runs the simulator loop."""

    def __init__(self, system_manager, display: DisplayManager,
                 input_func: Callable[[str], str] = input,
                 output_func: Callable[[str], None] = print) -> None:
        """Create the CLI.

        Args:
            system_manager: SystemManager instance managing the simulation.
            display: DisplayManager used for all output rendering.
            input_func: Function used to read a line (injectable for tests).
            output_func: Function used to print output.
        """
        self.system = system_manager
        self.display = display
        self.input_func = input_func
        self.output_func = output_func
        self.history: List[str] = []
        self.aliases = {
            "s": "start", "st": "status", "o": "orders", "r": "riders",
            "m": "map", "w": "wait", "h": "help", "?": "help",
            "q": "quit", "x": "exit",
        }

    # ----- validation ----------------------------------------------------
    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> List[int]:
        """Parse and validate four integer coordinates.

        Accepts space- or comma-separated values.

        Args:
            tokens: Raw argument strings.

        Returns:
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            ValueError: On wrong count, non-integer input or out-of-range
                values.
        """
        parts: List[str] = []
        for token in tokens:
            parts.extend(token.replace(",", " ").split())
        if len(parts) != 4:
            raise ValueError(
                "Expected 4 integers: restaurant_x restaurant_y "
                f"customer_x customer_y (got {len(parts)})")
        values: List[int] = []
        for part in parts:
            try:
                value = int(part)
            except ValueError:
                raise ValueError(f"'{part}' is not an integer") from None
            if not 0 <= value <= MAP_SIZE:
                raise ValueError(
                    f"Coordinate {value} is out of range 0-{MAP_SIZE}")
            values.append(value)
        return values

    # ----- main loop -----------------------------------------------------
    def run(self) -> None:
        """Run the main loop: one command and one tick per iteration."""
        self._print(WELCOME)
        while globalvar.get_value("running", True):
            try:
                line = self.input_func(PROMPT)
            except EOFError:
                self._print("")
                globalvar.set_value("running", False)
                break
            except KeyboardInterrupt:
                self._print("\nUse 'quit' or 'exit' to leave the simulator.")
                continue

            line = (line or "").strip()
            if not line:
                continue
            command_line = self._resolve_history(line)
            if command_line is None:
                continue
            self.history.append(command_line)

            try:
                self.dispatch(command_line)
            except Exception as exc:
                logger.exception("Command failed: %s", command_line)
                self._print(f"Error: could not run '{command_line}': {exc}")

            if globalvar.get_value("running", True):
                self._advance_tick()

    def _advance_tick(self) -> None:
        """Advance the simulation by one tick, recovering from errors."""
        try:
            self.system.tick()
        except Exception:
            logger.exception("Simulation tick failed")
            self._print("Warning: simulation tick failed; see log for details.")

    def _resolve_history(self, line: str) -> Optional[str]:
        """Expand '!!' and '!n' history references."""
        if line == "!!":
            if not self.history:
                self._print("No previous command.")
                return None
            return self.history[-1]
        if line.startswith("!") and line[1:].isdigit():
            index = int(line[1:])
            if 1 <= index <= len(self.history):
                return self.history[index - 1]
            self._print(f"No history entry {index}.")
            return None
        return line

    def dispatch(self, line: str) -> None:
        """Parse one command line and run the matching handler.

        Args:
            line: Command text entered by the user.
        """
        tokens = line.split()
        command = tokens[0].lower()
        command = self.aliases.get(command, command)
        args = tokens[1:]
        handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "wait": self.cmd_wait,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }
        handler = handlers.get(command)
        if handler is None:
            self._print(f"Unknown command '{tokens[0]}'. Type 'help'.")
            return
        handler(args)

    # ----- command handlers ----------------------------------------------
    def cmd_start(self, args: Sequence[str]) -> None:
        """Submit an order: ``start rx ry cx cy`` or prompted input."""
        if not args:
            try:
                args = [self.input_func(
                    "Enter restaurant_x restaurant_y customer_x customer_y: ")]
            except EOFError:
                globalvar.set_value("running", False)
                return
        try:
            coords = self.parse_coordinates(args)
        except ValueError as exc:
            self._print(f"Input error: {exc}")
            return
        restaurant = (coords[0], coords[1])
        customer = (coords[2], coords[3])
        try:
            result = self.system.submit_order(restaurant, customer)
        except (ValueError, TypeError) as exc:
            self._print(f"Order rejected: {exc}")
            return
        self._print(self.display.render_assignment(
            result["rider"], result["order"], result["new_rider"]))

    def cmd_status(self, args: Sequence[str]) -> None:
        """Show the status snapshot."""
        self._print(self.display.render_status())

    def cmd_orders(self, args: Sequence[str]) -> None:
        """Show the order table."""
        self._print(self.display.render_orders())

    def cmd_riders(self, args: Sequence[str]) -> None:
        """Show the rider status-change log."""
        self._print(self.display.render_rider_log())

    def cmd_map(self, args: Sequence[str]) -> None:
        """Show the ASCII map."""
        self._print(self.display.render_map())

    def cmd_wait(self, args: Sequence[str]) -> None:
        """Advance the simulation by ``n`` ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                self._print(f"'{args[0]}' is not an integer.")
                return
            if not 1 <= count <= 1000:
                self._print("Number of ticks must be between 1 and 1000.")
                return
        for _ in range(count):
            self._advance_tick()
        self._print(f"Advanced {count} tick(s); now at tick "
                    f"{globalvar.get_value('tick', 0)}.")

    def cmd_save(self, args: Sequence[str]) -> None:
        """Save the current state to a JSON file."""
        path = args[0] if args else "dispatch_state.json"
        try:
            self.system.save_state(path)
            self._print(f"State saved to {path}.")
        except OSError as exc:
            self._print(f"Could not save state: {exc}")

    def cmd_history(self, args: Sequence[str]) -> None:
        """List previously entered commands."""
        if not self.history:
            self._print("History is empty.")
            return
        for index, entry in enumerate(self.history, start=1):
            self._print(f"{index:>4}  {entry}")

    def cmd_help(self, args: Sequence[str]) -> None:
        """Show the command help."""
        self._print(HELP_TEXT)

    def cmd_quit(self, args: Sequence[str]) -> None:
        """Stop the main loop."""
        globalvar.set_value("running", False)
        self._print("Goodbye.")

    def _print(self, text: str) -> None:
        """Write text through the configured output function."""
        self.output_func(text)

Run from the project root with::

    python src/main.py [--config dispatch_config.json] [--log-level INFO]

from display import DisplayManager
from input_manager import InputManager
from system import SystemManager

def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """Parse command-line options.

    Args:

    """
    parser = argparse.ArgumentParser(

    parser.add_argument("--config", default=DEFAULT_CONFIG_PATH,
                        help="JSON file with system parameters")

                        help="logging verbosity written to stderr")

        stream=sys.stderr,

def load_parameters(path):
    """Load tunable system parameters from a JSON configuration file.

    The file must contain a JSON object whose keys are parameter names
    (case-insensitive). Unknown keys are ignored.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        list[str]: Names of the parameters that were applied.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the JSON is malformed or a value is not an integer.
    """
    with open(path, "r", encoding="utf-8") as handle:
        try:
            data = json.load(handle)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Configuration file is not valid JSON: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError("Configuration file must contain a JSON object.")

    module_globals = globals()
    applied = []
    for key, value in data.items():
        name = str(key).upper()
        if name not in _TUNABLE:
            continue
        if not isinstance(value, int) or isinstance(value, bool):
            raise ValueError(f"Parameter {name} must be an integer.")
        module_globals[name] = value
        applied.append(name)
    return applied

"""Global variable storage shared by the dispatch simulator modules.

The storage holds the core system data: system funds, the rider list, the
rider ID counter, the running flag, the simulated tick, and order lists.

from typing import Any, Dict

    """Reset the global storage to its default values.

    Defaults:
        riders: empty list.
        rider_id_counter: 0.
        funds: 0.
        running: True.

        "funds": 0,

        "orders": [],

        key: Name of the global variable.

"""Coordinate helpers for the 8x8 grid representing an 800x800 map."""

import config


def is_valid_coordinate(value):
    """Check that a value is an integer within the map range 0-800.

    Args:
        value: Candidate coordinate.

    Returns:
        bool: True if the value is a valid integer coordinate.
    """
    return (isinstance(value, int) and not isinstance(value, bool)
            and 0 <= value <= config.MAP_SIZE)


def cell_index(value):
    """Return the grid cell index (0-7) that contains a coordinate.

    Coordinates equal to the map edge (800) are clamped into the last cell.

    Args:
        value: Coordinate in units (0-800).

    Returns:
        int: Cell index along one axis.
    """
    index = value // config.CELL_SIZE
    return min(max(index, 0), config.GRID_SIZE - 1)


def to_grid_center(value):
    """Convert a coordinate to the center of its grid cell.

    Implements ``center = (value // 100) * 100 + 50`` with clamping so the
    map edge (800) maps to the center of the last cell (750).

    Args:
        value: Coordinate in units (0-800).

    Returns:
        int: Center coordinate along one axis.
    """
    return cell_index(value) * config.CELL_SIZE + config.NODE_OFFSET


def to_grid_point(x, y):
    """Convert a raw (x, y) position to grid cell center coordinates.

    Args:
        x: Raw x coordinate.
        y: Raw y coordinate.

    Returns:
        tuple[int, int]: Cell-center coordinates.
    """
    return (to_grid_center(x), to_grid_center(y))


def manhattan(a, b):
    """Return the Manhattan distance between two points.

    Args:
        a: First point as (x, y).
        b: Second point as (x, y).

    Returns:
        int: Manhattan distance, which equals the street-travel distance.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def is_street_point(x, y):
    """Return True if a point lies on a street (center line of a cell row/column).

    Args:
        x: x coordinate.
        y: y coordinate.

    Returns:
        bool: True if the point is on a street line.
    """
    return x % config.CELL_SIZE == config.NODE_OFFSET or \
        y % config.CELL_SIZE == config.NODE_OFFSET

"""Grid-based path planning for riders moving along the street network.

Streets run through the cell centers: every intersection is a cell center
(x and y each equal 50 + 100k), and the streets connect neighbouring
intersections with horizontal and vertical straight segments. Routes are
therefore made of right-angle turns and straight-line movement only.
"""

import math

import grid


class PathPlanner:
    """Compute waypoint routes between street intersections."""

    @staticmethod
    def is_straight(start, goal):
        """Return True when the goal lies directly above, below, left or right.

        This is the special case of a target on the same vertical or
        horizontal street, which is traversed as a single straight line.

        Args:
            start: Current position (x, y).
            goal: Target position (x, y).

        Returns:
            bool: True if the move is a single straight segment.
        """
        return start[0] == goal[0] or start[1] == goal[1]

    @staticmethod
    def plan(start, goal, occupied=None):
        """Plan a street route from ``start`` to ``goal``.

        Two L-shaped routes exist (horizontal-first or vertical-first). The
        planner prefers the route whose corner intersection is not occupied by
        another rider, which avoids path conflicts. Straight-line cases return
        a single waypoint.

        Args:
            start: Starting position (x, y).
            goal: Destination position (x, y).
            occupied: Optional iterable of positions occupied by other riders.

        Returns:
            list[tuple[int, int]]: Waypoints after ``start``, ending at ``goal``.
            Empty if already at the goal.
        """
        start = (int(start[0]), int(start[1]))
        goal = (int(goal[0]), int(goal[1]))
        if start == goal:
            return []
        sx, sy = start
        gx, gy = goal
        if sx == gx or sy == gy:
            return [goal]

        occupied_set = set(occupied) if occupied else set()
        horizontal_first = (gx, sy)
        vertical_first = (sx, gy)
        if horizontal_first in occupied_set and vertical_first not in occupied_set:
            corner = vertical_first
        else:
            corner = horizontal_first
        return [corner, goal]

    @staticmethod
    def plan_multi(start, stops, occupied=None):
        """Plan a continuous route through several destinations in order.

        Args:
            start: Starting position (x, y).
            stops: Ordered destinations, e.g. [restaurant, customer].
            occupied: Optional iterable of positions occupied by other riders.

        Returns:
            list[tuple[int, int]]: Concatenated waypoints for the full route.
        """
        waypoints = []
        current = (int(start[0]), int(start[1]))
        for stop in stops:
            waypoints.extend(PathPlanner.plan(current, stop, occupied))
            current = (int(stop[0]), int(stop[1]))
        return waypoints

    @staticmethod
    def path_length(start, waypoints):
        """Return the total street distance of a waypoint route.

        Args:
            start: Starting position (x, y).
            waypoints: Waypoints following the start position.

        Returns:
            int: Total travelled distance in units.
        """
        total = 0
        current = start
        for point in waypoints:
            total += grid.manhattan(current, point)
            current = point
        return total

    @staticmethod
    def estimate_ticks(start, stops, speed):
        """Estimate the number of ticks needed to visit all stops.

        Args:
            start: Starting position (x, y).
            stops: Ordered destinations.
            speed: Rider speed in units per tick.

        Returns:
            int: Estimated ticks (rounded up); 0 if speed is not positive.
        """
        if speed <= 0:
            return 0
        route = PathPlanner.plan_multi(start, stops)
        return math.ceil(PathPlanner.path_length(start, route) / speed)

"""Rider model: state machine, street movement and event logging."""

import logging

import config
import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)


class Rider:
    """A delivery rider that moves along the streets and serves orders.

    Attributes:
        rider_id: Unique integer identifier.
        x, y: Current position (cell-center coordinates).
        speed: Movement speed in units per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders; ``orders[0]`` is the active order.
        special_case: True while on a direct straight-line leg.
        path: Remaining waypoints of the current leg.
        target: Current destination (restaurant or customer).
        log: Event entries with keys "event", "order_id", "tick", "x", "y",
            "state" and "rider_id".
    """

    def __init__(self, rider_id, speed=None):
        """Create a rider at the spawn point in the idle state.

        Args:
            rider_id: Unique integer identifier.
            speed: Optional speed in units per tick (default from config).

        Raises:
            TypeError: If rider_id is not an integer.
            ValueError: If speed is not positive.
        """
        if not isinstance(rider_id, int) or isinstance(rider_id, bool):
            raise TypeError("rider_id must be an integer")
        self.rider_id = rider_id
        self.x, self.y = config.SPAWN_POINT
        self.speed = int(speed) if speed is not None else config.DEFAULT_SPEED
        if self.speed <= 0:
            raise ValueError("speed must be a positive integer")
        self.state = config.STATE_IDLE
        self.orders = []
        self.special_case = False
        self.path = []
        self.target = None
        self.wait_ticks = 0
        self.log = []
        self._record("spawn", None)

    @property
    def position(self):
        """Return the current position as an (x, y) tuple."""
        return (self.x, self.y)

    def state_name(self):
        """Return the human-readable name of the current state."""
        return config.STATE_NAMES.get(self.state, "unknown")

    def move(self):
        """Advance the rider by one tick.

        Handles the idle → picking up transition when an order is assigned,
        moves along the planned route, and performs arrival transitions
        (picking up → delivering at the restaurant, delivering → idle at the
        customer). Every status change is written to ``self.log``.
        """
        if not self.orders:
            if self.state != config.STATE_IDLE:
                logger.warning("Rider %s had no orders; resetting to idle.",
                               self.rider_id)
                self.state = config.STATE_IDLE
                self.path = []
                self.target = None
            return

        if self.state == config.STATE_IDLE:
            self._accept_order(self.orders[0])

        if self.path:
            self._advance(self.speed)

        if not self.path and self.state != config.STATE_IDLE:
            self._on_arrival()

    def to_dict(self):
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
        }

    def __repr__(self):
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state_name()}, orders={len(self.orders)})")

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _now(self):
        """Return the current simulated tick."""
        return globalvar.get_value("tick", 0)

    def _record(self, event, order_id):
        """Append a status-change entry to the rider log."""
        self.log.append({
            "event": event,
            "order_id": order_id,
            "tick": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "rider_id": self.rider_id,
        })
        logger.info("Rider %s: %s (order %s) at (%s,%s)",
                    self.rider_id, event, order_id, self.x, self.y)

    def _other_positions(self):
        """Return the set of positions currently occupied by other riders."""
        riders = globalvar.get_value("riders", []) or []
        return {rider.position for rider in riders if rider is not self}

    def _append_route(self, order):
        """Record the current position in the active order's route."""
        if not order.route or order.route[-1] != self.position:
            order.route.append(self.position)

    def _plan_leg(self, target):
        """Plan the path for a new leg and update leg bookkeeping."""
        self.target = target
        self.path = PathPlanner.plan(self.position, target,
                                     self._other_positions())
        self.special_case = PathPlanner.is_straight(self.position, target)

    def _accept_order(self, order):
        """Start working on an order: idle → picking up."""
        self.state = config.STATE_PICKING_UP
        order.mark_picking_up()
        self._append_route(order)
        self._plan_leg(order.res_grid)
        self._record("accepted_order", order.id)

    def _advance(self, budget):
        """Move along the planned path using at most ``budget`` units.

        If the next waypoint is occupied by another rider, the rider waits up
        to ``WAIT_LIMIT`` ticks (deadlock/conflict avoidance) and then proceeds.
        """
        while budget > 0 and self.path:
            target_x, target_y = self.path[0]
            if (target_x, target_y) == self.position:
                self.path.pop(0)
                continue
            if (target_x, target_y) in self._other_positions():
                if self.wait_ticks < config.WAIT_LIMIT:
                    self.wait_ticks += 1
                    return
            self.wait_ticks = 0

            if target_x != self.x:
                distance = abs(target_x - self.x)
                step = min(budget, distance)
                self.x += step if target_x > self.x else -step
            else:
                distance = abs(target_y - self.y)
                step = min(budget, distance)
                self.y += step if target_y > self.y else -step
            budget -= step

            if self.position == (target_x, target_y):
                self.path.pop(0)
            if self.orders:
                self._append_route(self.orders[0])

    def _on_arrival(self):
        """Handle arrival at the current leg destination."""
        order = self.orders[0]
        if self.state == config.STATE_PICKING_UP:
            self.state = config.STATE_DELIVERING
            order.mark_delivering()
            self._plan_leg(order.cus_grid)
            self._record("arrived_at_merchant", order.id)
        elif self.state == config.STATE_DELIVERING:
            order.complete(self._now(), self.rider_id)
            globalvar.set_value("funds",
                                globalvar.get_value("funds", 0) + config.ORDER_FEE)
            completed = globalvar.get_value("completed_orders")
            if completed is not None:
                completed.append(order)
            self.orders.pop(0)
            self.state = config.STATE_IDLE
            self.path = []
            self.target = None
            self.special_case = False
            self._record("delivered_to_customer", order.id)

"""Order data model: coordinates, lifecycle status and delivery statistics."""

import globalvar
import grid


def _validate_point(point, label):
    """Validate an [x, y] position and return it as a tuple of ints.

    Args:
        point: Sequence of two integers in the range 0-800.
        label: Name used in error messages.

    Returns:
        tuple[int, int]: Validated (x, y).

    Raises:
        ValueError: If the point is malformed or out of range.
    """
    try:
        x, y = point
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{label} position must be [x, y].") from exc
    for value in (x, y):
        if not grid.is_valid_coordinate(value):
            raise ValueError(
                f"{label} coordinate {value} is invalid; "
                "coordinates must be integers in the range 0-800.")
    return (x, y)


def _next_order_id():
    """Return the next unique order identifier."""
    counter = globalvar.get_value("order_id_counter", 0) + 1
    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order identifier.
        res_pos: Raw restaurant position as entered by the user.
        cus_pos: Raw customer position as entered by the user.
        res_grid: Restaurant position converted to grid cell center.
        cus_grid: Customer position converted to grid cell center.
        status: "created", "assigned", "picking_up", "delivering" or
            "completed".
        rider_id: Identifier of the assigned rider, or None.
        start_time: Tick at which the order was created.
        completion_time: Tick at which delivery finished, or None.
        time_taken: Ticks from creation to completion, or None.
        distance: Manhattan distance restaurant → customer in units.
        efficiency: Distance per tick (units/tick), or None.
        route: Sequence of (x, y) positions travelled for this order.
    """

    STATUS_CREATED = "created"
    STATUS_ASSIGNED = "assigned"
    STATUS_PICKING_UP = "picking_up"
    STATUS_DELIVERING = "delivering"
    STATUS_COMPLETED = "completed"

    def __init__(self, res_pos, cus_pos):
        """Create an order from restaurant and customer positions.

        Args:
            res_pos: [x, y] restaurant position (0-800).
            cus_pos: [x, y] customer position (0-800).

        Raises:
            ValueError: If any coordinate is invalid.
        """
        self.res_pos = _validate_point(res_pos, "Restaurant")
        self.cus_pos = _validate_point(cus_pos, "Customer")
        self.id = _next_order_id()
        self.res_grid = grid.to_grid_point(*self.res_pos)
        self.cus_grid = grid.to_grid_point(*self.cus_pos)
        self.status = self.STATUS_CREATED
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.distance = grid.manhattan(self.res_grid, self.cus_grid)
        self.efficiency = None
        self.route = []

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = self.STATUS_ASSIGNED

    def mark_picking_up(self):
        """Mark the rider as en route to the restaurant."""
        self.status = self.STATUS_PICKING_UP

    def mark_delivering(self):
        """Record pickup and mark the rider as en route to the customer."""
        self.pickup_time = globalvar.get_value("tick", 0)
        self.status = self.STATUS_DELIVERING

    def complete(self, tick, rider_id=None):
        """Finish the order and compute delivery statistics.

        Args:
            tick: Simulated tick at which delivery happened.
            rider_id: Optional rider identifier to store.
        """
        if rider_id is not None:
            self.rider_id = rider_id
        self.status = self.STATUS_COMPLETED
        self.completion_time = tick
        self.time_taken = max(0, tick - self.start_time)
        self.efficiency = (round(self.distance / self.time_taken, 2)
                           if self.time_taken > 0 else None)

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "efficiency": self.efficiency,
        }

    def __repr__(self):
        return f"Order(id={self.id}, status={self.status}, rider={self.rider_id})"

"""Order assignment: rider selection and order management."""

import logging

import config
import globalvar
import grid
from order_model import Order

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Find the best available rider for an order.

    Strategy: distance-first. Among riders that still have capacity
    (fewer than ``MAX_ORDERS_PER_RIDER`` orders), the rider closest to the
    restaurant is preferred. A load penalty per already-assigned order
    balances work between riders (competitive, load-aware assignment).
    Ties are broken by the lower rider ID.

    Args:
        order: The Order to be assigned.

    Returns:
        Rider: The selected rider, or None if no rider has spare capacity.
    """
    riders = globalvar.get_value("riders", []) or []
    best = None
    best_key = None
    for rider in riders:
        if len(rider.orders) >= config.MAX_ORDERS_PER_RIDER:
            continue
        distance = grid.manhattan(rider.position, order.res_grid)
        key = (distance + config.LOAD_PENALTY * len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates, assigns and tracks delivery orders."""

    def create_order(self, coords):
        """Create an order from four integer coordinates and store it.

        Args:
            coords: Sequence [restaurant_x, restaurant_y, customer_x, customer_y].

        Returns:
            Order: The newly created order.

        Raises:
            ValueError: If the sequence length or any coordinate is invalid.
        """
        if len(coords) != 4:
            raise ValueError("An order needs exactly four coordinates.")
        for value in coords:
            if not grid.is_valid_coordinate(value):
                raise ValueError(
                    f"Coordinate {value} is invalid; use integers in 0-800.")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        globalvar.get_value("orders").append(order)
        logger.info("Created order %s", order.id)
        return order

    def try_assign(self, order):
        """Assign an order to the best available rider.

        Args:
            order: The Order to assign.

        Returns:
            Rider or None: The rider that received the order, or None if no
            rider is available. The order is appended to ``rider.orders``.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Assigned order %s to rider %s", order.id, rider.rider_id)
        return rider

    def enqueue(self, order):
        """Keep an unassignable order in the pending queue for later retry.

        Args:
            order: The Order that could not be assigned.
        """
        pending = globalvar.get_value("pending_orders")
        if pending is not None and order not in pending:
            pending.append(order)

    def retry_pending(self):
        """Try to assign all pending orders.

        Returns:
            list[Rider]: Riders that received previously pending orders.
        """
        pending = globalvar.get_value("pending_orders", []) or []
        assigned = []
        for order in list(pending):
            rider = self.try_assign(order)
            if rider is not None:
                pending.remove(order)
                assigned.append(rider)
        return assigned

    def pending_orders(self):
        """Return the list of orders waiting for a rider."""
        return list(globalvar.get_value("pending_orders", []) or [])

    def all_orders(self):
        """Return all orders created in this session."""
        return list(globalvar.get_value("orders", []) or [])

class SystemManager:
    """Manages system funds, the rider list and the simulation clock.

    State lives in the ``globalvar`` store, so instances are lightweight.
    """

    def __init__(self):
        """Create a manager bound to the global system state."""
        self.order_manager = OrderManager()

    # ------------------------------------------------------------------
    # Funds
    # ------------------------------------------------------------------
    def funds(self):
        """Return the current system fund balance (may be negative)."""
        return globalvar.get_value("funds", 0)

    def add_funds(self, amount):
        """Add (or subtract, for negative amounts) funds.

        Args:
            amount: Integer amount to add.

        Returns:
            int: The new balance.
        """
        new_value = self.funds() + int(amount)
        globalvar.set_value("funds", new_value)
        return new_value

    # ------------------------------------------------------------------
    # Riders (CRUD)
    # ------------------------------------------------------------------
    def riders(self):
        """Return a copy of the rider list."""
        return list(globalvar.get_value("riders", []) or [])

    def get_rider(self, rider_id):
        """Return the rider with the given ID, or None."""
        for rider in self.riders():
            if rider.rider_id == rider_id:
                return rider
        return None

    def recruit_rider(self):
        """Create a new idle rider, assign it an ID and pay the hiring cost.

        Returns:
            Rider: The newly recruited rider.
        """
        new_id = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", new_id)
        rider = Rider(new_id)
        globalvar.get_value("riders").append(rider)
        self.add_funds(-config.RIDER_COST)
        logger.info("Recruited rider %s", new_id)
        return rider

    def check_recruitment(self):
        """Recruit a rider automatically if funds reach the threshold.

        Returns:
            Rider or None: The new rider if one was recruited, else None.
        """
        if self.funds() >= config.RECRUIT_THRESHOLD:
            return self.recruit_rider()
        return None

    def remove_rider(self, rider_id):
        """Remove an idle rider that has no assigned orders.

        Args:
            rider_id: Identifier of the rider to remove.

        Returns:
            bool: True if the rider was removed.
        """
        rider = self.get_rider(rider_id)
        if rider is None or rider.orders:
            return False
        globalvar.get_value("riders").remove(rider)
        logger.info("Removed rider %s", rider_id)
        return True

    # ------------------------------------------------------------------
    # Simulation clock
    # ------------------------------------------------------------------
    def tick(self):
        """Advance the simulation by one tick.

        Order of operations: move all riders, check recruitment, retry
        pending orders.

        Returns:
            int: The new simulated tick value.
        """
        now = globalvar.get_value("tick", 0) + 1
        globalvar.set_value("tick", now)
        for rider in self.riders():
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                logger.exception("Rider %s failed to move; resetting.",
                                 rider.rider_id)
                rider.orders = []
                rider.state = config.STATE_IDLE
                rider.path = []
        self.check_recruitment()
        self.order_manager.retry_pending()
        return now


def init_system():
    """Initialise global state and recruit the first rider if funds allow.

    Sets up the rider list (empty), rider ID counter (0), running state
    (True), funds, and the simulated tick (0).
    """
    globalvar._init()
    globalvar.set_value("funds", config.INITIAL_FUNDS)
    SystemManager().check_recruitment()
    logger.info("System initialised with funds=%s", globalvar.get_value("funds"))

"""Display helpers: status text, tables and ASCII map rendering."""

import config
import grid

ROUTE_STEP = 50  # Canvas resolution: one character per 50 units.


class DisplayManager:
    """Formats status information and renders ASCII maps.

    Symbols: 'R' rider, 'S' restaurant, 'C' customer, '.' empty,
    '#' street segment, '+' intersection, '*' route.
    """

    # ------------------------------------------------------------------
    # Maps
    # ------------------------------------------------------------------
    def render_cell_map(self, riders, orders):
        """Render the 8x8 cell map showing riders and active order locations.

        Args:
            riders: Iterable of Rider objects.
            orders: Iterable of Order objects; completed ones are skipped.

        Returns:
            str: Multi-line map text.
        """
        size = config.GRID_SIZE
        cells = [["." for _ in range(size)] for _ in range(size)]
        for order in orders:
            if order.status == "completed":
                continue
            cx, cy = order.cus_grid
            cells[grid.cell_index(cy)][grid.cell_index(cx)] = "C"
            rx, ry = order.res_grid
            cells[grid.cell_index(ry)][grid.cell_index(rx)] = "S"
        for rider in riders:
            cells[grid.cell_index(rider.y)][grid.cell_index(rider.x)] = "R"

        lines = ["Cell map (8x8, each cell 100x100 units; top row = highest y)"]
        lines.append("      " + " ".join(str(c) for c in range(size)) + "   <- x cell")
        for row in reversed(range(size)):
            lines.append(f"y{row} | " + " ".join(cells[row]) + " |")
        lines.append("Legend: R=Rider S=Restaurant C=Customer .=Empty")
        return "\n".join(lines)

    def render_street_map(self, riders, orders=(), route_orders=()):
        """Render the street-level ASCII map with routes.

        The canvas has one character per 50 units. Streets run through cell
        centers, so intersections appear at odd indices and route segments
        are drawn as connected '*' lines.

        Args:
            riders: Iterable of Rider objects.
            orders: Iterable of Order objects whose restaurant/customer
                locations should be shown.
            route_orders: Iterable of Order objects whose travelled routes
                should be drawn.

        Returns:
            str: Multi-line map text.
        """
        size = config.MAP_SIZE // ROUTE_STEP + 1
        canvas = [[self._base_char(r, c) for c in range(size)]
                  for r in range(size)]

        for order in route_orders:
            points = order.route
            for start, end in zip(points, points[1:]):
                for px, py in self._segment_points(start, end):
                    row, col = py // ROUTE_STEP, px // ROUTE_STEP
                    if canvas[row][col] not in "RSC":
                        canvas[row][col] = "*"

        for order in orders:
            if order.status == "completed":
                continue
            self._place(canvas, order.cus_grid, "C")
            self._place(canvas, order.res_grid, "S")
        for rider in riders:
            self._place(canvas, rider.position, "R")

        lines = ["Street map (ASCII, 50 units per character; top = highest y)"]
        for row in reversed(range(size)):
            label = f"{row * ROUTE_STEP:>4} "
            lines.append(label + " ".join(canvas[row]))
        lines.append("     " + " ".join(str((c * ROUTE_STEP) // 100 % 10)
                                       if (c * ROUTE_STEP) % 100 == 0 else " "
                                       for c in range(size)) + "  <- x")
        lines.append("Legend: R=Rider S=Restaurant C=Customer "
                     "*=route #=street +=intersection .=block")
        return "\n".join(lines)

    def render_route(self, order, rider=None, riders=()):
        """Render the route of one order as a connected line graph.

        Args:
            order: The Order whose route is drawn.
            rider: Optional rider to mark with 'R' at its current position.
            riders: Optional additional riders to show.

        Returns:
            str: Multi-line map text.
        """
        shown = [rider] if rider is not None else []
        shown.extend(riders)
        return self.render_street_map(shown, [order], [order])

    @staticmethod
    def _base_char(row, col):
        """Return the background character for a canvas position."""
        row_street = row % 2 == 1
        col_street = col % 2 == 1
        if row_street and col_street:
            return "+"
        if row_street or col_street:
            return "#"
        return "."

    @staticmethod
    def _place(canvas, point, symbol):
        """Place a symbol on the canvas at a map coordinate."""
        x, y = point
        canvas[y // ROUTE_STEP][x // ROUTE_STEP] = symbol

    @staticmethod
    def _segment_points(start, end):
        """Return canvas-resolution points along an axis-aligned segment."""
        (ax, ay), (bx, by) = start, end
        points = []
        if ax == bx:
            step = ROUTE_STEP if by >= ay else -ROUTE_STEP
            for y in range(ay, by + step, step):
                points.append((ax, y))
        elif ay == by:
            step = ROUTE_STEP if bx >= ax else -ROUTE_STEP
            for x in range(ax, bx + step, step):
                points.append((x, ay))
        else:
            points = [start, end]
        return points

    # ------------------------------------------------------------------
    # Text reports
    # ------------------------------------------------------------------
    def format_status(self, funds, riders):
        """Format the system snapshot shown by the 'status' command.

        Args:
            funds: Current system funds.
            riders: Iterable of Rider objects.

        Returns:
            str: Multi-line status text.
        """
        riders = list(riders)
        lines = [f"System funds: {funds}", f"Riders: {len(riders)}"]
        if not riders:
            lines.append("  (no riders recruited yet)")
        for rider in riders:
            order_ids = [order.id for order in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id}: {rider.state_name():<11} "
                f"pos=({rider.x},{rider.y}) speed={rider.speed} "
                f"orders={order_ids if order_ids else '[]'}")
        return "\n".join(lines)

    def format_orders(self, orders):
        """Format the order table shown by the 'orders' command.

        Args:
            orders: Iterable of Order objects.

        Returns:
            str: Multi-line table text.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        lines = []
        for order in orders:
            rider = f"#{order.rider_id}" if order.rider_id is not None else "-"
            taken = order.time_taken if order.time_taken is not None else "-"
            efficiency = (order.efficiency
                          if order.efficiency is not None else "-")
            lines.append(
                f"Order #{order.id}: start_tick={order.start_time} "
                f"restaurant={order.res_pos} customer={order.cus_pos} "
                f"distance={order.distance} time_taken={taken} "
                f"efficiency={efficiency} rider={rider} status={order.status}")
        return "\n".join(lines)

    def format_riders_log(self, riders):
        """Format the status-change history of all riders.

        Args:
            riders: Iterable of Rider objects.

        Returns:
            str: Multi-line log text.
        """
        lines = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} log:")
            for entry in rider.log:
                order_id = entry["order_id"] if entry["order_id"] is not None else "-"
                state = config.STATE_NAMES.get(entry["state"], "unknown")
                lines.append(
                    f"  tick {entry['tick']:>4} | ({entry['x']},{entry['y']}) | "
                    f"{entry['event']:<22} | order {order_id} | state {state}")
        return "\n".join(lines) if lines else "No riders recruited yet."

    def format_start_summary(self, order, rider, rider_added, estimated_ticks):
        """Format the summary printed after the 'start' command.

        Args:
            order: The created Order.
            rider: The assigned Rider, or None if the order is pending.
            rider_added: True if a new rider was recruited for this order.
            estimated_ticks: Estimated ticks to finish, or None.

        Returns:
            str: Multi-line summary text.
        """
        lines = [
            f"Order #{order.id} created.",
            f"  Restaurant: input {order.res_pos} -> grid center {order.res_grid}",
            f"  Customer:   input {order.cus_pos} -> grid center {order.cus_grid}",
            f"  Distance:   {order.distance} units",
        ]
        lines.append(f"  New rider added: {'yes' if rider_added else 'no'}")
        if rider is not None:
            lines.append(f"  Assigned rider: #{rider.rider_id} "
                         f"(speed {rider.speed} units/tick)")
            if estimated_ticks is not None:
                lines.append(f"  Estimated time: about {estimated_ticks} tick(s)")
        else:
            lines.append("  No rider available; order is pending until a rider "
                         "is recruited.")
        return "\n".join(lines)

"""Command parsing, input validation and command history."""

import logging

try:  # Enables arrow-key line editing where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    readline = None

logger = logging.getLogger(__name__)

MIN_COORD = 0
MAX_COORD = 800

COMMANDS = ("start", "status", "orders", "riders", "map", "tick",
            "save", "history", "help", "quit", "exit")

SHORTCUTS = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "v": "save",
    "h": "history",
    "?": "help",
    "q": "quit",
}


class InputManager:
    """Reads, parses and validates user commands.

    Supports shortcut keys ('s' for start, 'q' for quit, ...), history
    recall with '!!' (last command) and '!n' (n-th command), and optional
    readline support.
    """

    def __init__(self):
        """Create an input manager with an empty command history."""
        self.history = []

    def read_line(self, prompt="dispatch> "):
        """Read one command line from standard input.

        Args:
            prompt: Prompt string shown to the user.

        Returns:
            str: The command text (history references are expanded).
            Empty string for a blank line.

        Raises:
            EOFError: When input is closed.
            ValueError: If a history reference cannot be resolved.
        """
        raw = input(prompt).strip()
        if not raw:
            return ""
        if raw == "!!":
            if not self.history:
                raise ValueError("No command in history to repeat.")
            raw = self.history[-1]
        elif raw.startswith("!") and raw[1:].isdigit():
            index = int(raw[1:])
            if not 1 <= index <= len(self.history):
                raise ValueError(f"History entry {index} does not exist.")
            raw = self.history[index - 1]
        self.history.append(raw)
        return raw

    def parse(self, line):
        """Split a command line into a normalised command name and arguments.

        Args:
            line: Raw command text.

        Returns:
            tuple[str, list[str]]: (command, arguments).

        Raises:
            ValueError: If the command is unknown.
        """
        tokens = line.split()
        if not tokens:
            raise ValueError("Empty command.")
        command = tokens[0].lower()
        command = SHORTCUTS.get(command, command)
        if command not in COMMANDS:
            raise ValueError(f"Unknown command '{tokens[0]}'. Type 'help' for a list.")
        return command, tokens[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Parse and validate four integer coordinates.

        Accepts space- or comma-separated values, e.g. ``100 200 300 400``
        or ``100,200,300,400``.

        Args:
            tokens: Sequence of text tokens.

        Returns:
            list[int]: [restaurant_x, restaurant_y, customer_x, customer_y].

        Raises:
            ValueError: If the count is wrong, a value is not an integer,
                or a value is outside 0-800.
        """
        text = " ".join(tokens).replace(",", " ")
        parts = text.split()
        if len(parts) != 4:
            raise ValueError(
                "Expected 4 coordinates: restaurant_x restaurant_y "
                f"customer_x customer_y (got {len(parts)}).")
        values = []
        names = ("restaurant_x", "restaurant_y", "customer_x", "customer_y")
        for name, part in zip(names, parts):
            try:
                value = int(part)
            except ValueError as exc:
                raise ValueError(
                    f"{name} must be an integer, got '{part}'.") from exc
            if not MIN_COORD <= value <= MAX_COORD:
                raise ValueError(
                    f"{name} = {value} is out of range; use 0-{MAX_COORD}.")
            values.append(value)
        return values

    def prompt_coordinates(self):
        """Ask the user for the four order coordinates interactively.

        Returns:
            list[int]: Validated coordinates.

        Raises:
            ValueError: If the entered coordinates are invalid.
        """
        text = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
        return self.parse_coordinates(text.split())

"""JSON persistence: configuration loading and state snapshots."""

import json
import logging

import config
import globalvar

logger = logging.getLogger(__name__)


def build_snapshot():
    """Build a JSON-serialisable snapshot of the current system state.

    Returns:
        dict: Funds, tick, rider list, order history and pending orders.
    """
    riders = globalvar.get_value("riders", []) or []
    orders = globalvar.get_value("orders", []) or []
    pending = globalvar.get_value("pending_orders", []) or []
    return {
        "funds": globalvar.get_value("funds", 0),
        "tick": globalvar.get_value("tick", 0),
        "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
        "parameters": {
            "recruit_threshold": config.RECRUIT_THRESHOLD,
            "rider_cost": config.RIDER_COST,
            "order_fee": config.ORDER_FEE,
            "default_speed": config.DEFAULT_SPEED,
        },
        "riders": [rider.to_dict() for rider in riders],
        "orders": [order.to_dict() for order in orders],
        "pending_orders": [order.id for order in pending],
    }


def save_snapshot(path=None):
    """Write the current system state to a JSON file.

    Args:
        path: Destination file path; defaults to ``config.DEFAULT_SNAPSHOT_PATH``.

    Returns:
        str: The path that was written.

    Raises:
        OSError: If the file cannot be written.
    """
    target = path or config.DEFAULT_SNAPSHOT_PATH
    snapshot = build_snapshot()
    with open(target, "w", encoding="utf-8") as handle:
        json.dump(snapshot, handle, indent=2)
    logger.info("Snapshot saved to %s", target)
    return target


def load_configuration(path):
    """Apply system parameters from a JSON configuration file.

    Args:
        path: Path to the configuration file.

    Returns:
        list[str]: Names of the applied parameters.
    """
    applied = config.load_parameters(path)
    logger.info("Loaded configuration %s: %s", path, applied)
    return applied

"""Command-line entry point for the food delivery dispatch simulator.

Run with ``python src/main.py`` (optionally ``--config path/to/config.json``).
"""

import logging
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import config  # noqa: E402
import globalvar  # noqa: E402
from display import DisplayManager  # noqa: E402
from input_manager import InputManager  # noqa: E402
from order import OrderManager  # noqa: E402
from path_planner import PathPlanner  # noqa: E402
from persistence import load_configuration, save_snapshot  # noqa: E402
from system import SystemManager, init_system  # noqa: E402

logger = logging.getLogger("dispatch")

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant then customer, 0-800) and dispatch it
  status               System funds and rider states
  orders               All orders with time, distance and rider
  riders               Status-change log for every rider
  map                  8x8 cell map and street map of current positions
  tick [n]             Advance the simulation by n ticks (default 1)
  save [path]          Save a JSON snapshot of the system state
  history              Show command history
  help                 Show this help
  quit / exit          Leave the program
Shortcuts: s=start st=status o=orders r=riders m=map t=tick v=save h=history
           q=quit  ?=help  !!=repeat last command  !n=repeat command n
"""


class DispatchApp:
    """Connects user commands to the simulation components."""

    def __init__(self):
        """Create the application components on top of the global state."""
        self.system = SystemManager()
        self.orders = OrderManager()
        self.display = DisplayManager()
        self.inputs = InputManager()
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self):
        """Run the interactive command loop until quit or end of input."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while globalvar.get_value("running", True):
            try:
                line = self.inputs.read_line()
                if not line:
                    continue
                command, args = self.inputs.parse(line)
                self.handlers[command](args)
            except EOFError:
                print()
                break
            except KeyboardInterrupt:
                print("\nInterrupted. Type 'quit' to exit.")
            except ValueError as exc:
                print(f"Input error: {exc}")
            except Exception as exc:  # Graceful recovery for unexpected faults.
                logger.exception("Unexpected error while processing command")
                print(f"Internal error: {exc}. Attempting state recovery.")
                self.recover_state()
        print("Goodbye.")

    def recover_state(self):
        """Repair common inconsistencies after an unexpected error."""
        if not isinstance(globalvar.get_value("funds", 0), int):
            globalvar.set_value("funds", 0)
        riders = globalvar.get_value("riders", []) or []
        for rider in riders:
            if not rider.orders and rider.state != config.STATE_IDLE:
                rider.state = config.STATE_IDLE
                rider.path = []
                rider.target = None

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def cmd_start(self, args):
        """Create an order, dispatch it and simulate until it is delivered."""
        if args:
            coords = self.inputs.parse_coordinates(args)
        else:
            coords = self.inputs.prompt_coordinates()

        order = self.orders.create_order(coords)
        rider = self.orders.try_assign(order)
        rider_added = False
        if rider is None:
            new_rider = self.system.check_recruitment()
            if new_rider is not None:
                rider_added = True
                rider = self.orders.try_assign(order)
        if rider is None:
            self.orders.enqueue(order)

        estimate = None
        if rider is not None:
            estimate = PathPlanner.estimate_ticks(
                rider.position, [order.res_grid, order.cus_grid], rider.speed)
        print(self.display.format_start_summary(order, rider, rider_added, estimate))

        ticks = 0
        while order.status != "completed" and ticks < config.MAX_SIM_TICKS:
            self.system.tick()
            ticks += 1

        if order.status == "completed":
            print(f"Order #{order.id} delivered by rider #{order.rider_id} "
                  f"after {order.time_taken} tick(s); "
                  f"distance {order.distance} units.")
        else:
            print(f"Order #{order.id} not delivered within "
                  f"{config.MAX_SIM_TICKS} ticks (status: {order.status}).")

        riders = self.system.riders()
        print(self.display.render_route(order, riders=riders))

    def cmd_status(self, _args):
        """Print the system fund balance and every rider's state."""
        print(self.display.format_status(self.system.funds(), self.system.riders()))

    def cmd_orders(self, _args):
        """Print the details of all orders."""
        print(self.display.format_orders(self.orders.all_orders()))

    def cmd_riders(self, _args):
        """Print the status-change history of all riders."""
        print(self.display.format_riders_log(self.system.riders()))

    def cmd_map(self, _args):
        """Print the cell map and the street map of current positions."""
        riders = self.system.riders()
        orders = self.orders.all_orders()
        print(self.display.render_cell_map(riders, orders))
        print()
        print(self.display.render_street_map(riders, orders, []))

    def cmd_tick(self, args):
        """Advance the simulation by one or more ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError as exc:
                raise ValueError("tick count must be an integer.") from exc
        if not 1 <= count <= config.MAX_SIM_TICKS:
            raise ValueError(f"tick count must be between 1 and {config.MAX_SIM_TICKS}.")
        for _ in range(count):
            self.system.tick()
        print(f"Advanced {count} tick(s); now at tick "
              f"{globalvar.get_value('tick', 0)}. Funds: {self.system.funds()}.")

    def cmd_save(self, args):
        """Save a JSON snapshot of the system state."""
        path = args[0] if args else None
        try:
            written = save_snapshot(path)
        except OSError as exc:
            raise ValueError(f"Could not save snapshot: {exc}") from exc
        print(f"Snapshot saved to {written}.")

    def cmd_history(self, _args):
        """Print the command history."""
        if not self.inputs.history:
            print("History is empty.")
            return
        for index, entry in enumerate(self.inputs.history, start=1):
            print(f"  {index:>3}  {entry}")

    def cmd_help(self, _args):
        """Print the command help text."""
        print(HELP_TEXT)

    def cmd_quit(self, _args):
        """Stop the main loop."""
        globalvar.set_value("running", False)


def _configure_logging():
    """Send log output to the log file so the console stays clean."""
    logging.basicConfig(
        filename=config.LOG_FILE,
        level=logging.INFO,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )


def main(argv=None):
    """Program entry point.

    Args:
        argv: Optional argument list (defaults to ``sys.argv[1:]``).

    Returns:
        int: Process exit status.
    """
    argv = list(sys.argv[1:] if argv is None else argv)
    _configure_logging()

    if "--config" in argv:
        index = argv.index("--config")
        if index + 1 >= len(argv):
            print("Usage: main.py [--config CONFIG_FILE]")
            return 2
        try:
            applied = load_configuration(argv[index + 1])
            print(f"Loaded parameters: {', '.join(applied) if applied else 'none'}")
        except (OSError, ValueError) as exc:
            print(f"Configuration error: {exc}")
            return 1

    init_system()
    DispatchApp().run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

The store is a simple dictionary wrapped by ``set_value`` / ``get_value``.

"""Grid-based path planning for the 800x800 street map.

The map is an 8x8 grid of 100x100 cells. Streets are the lines whose x or y
coordinate is a multiple of 100. Restaurant and customer locations are cell
centres (for example (150, 250)), which are not on a street; they are reached
through the street segment that bounds their cell.

Routes are sequences of axis-aligned segments, so every route can be driven
by a rider moving along horizontal or vertical lines only.
"""

import heapq
import math
from typing import Dict, List, Optional, Sequence, Tuple

Point = Tuple[int, int]

MAP_SIZE = 800
CELL_SIZE = 100
HALF_CELL = 50


def manhattan(a: Point, b: Point) -> int:
    """Return the Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def path_length(path: Sequence[Point]) -> int:
    """Return the total travelled length (along streets) of a polyline."""
    return sum(manhattan(path[i], path[i + 1]) for i in range(len(path) - 1))


def is_on_street(point: Point) -> bool:
    """Return True if the point lies on a street line."""
    return point[0] % CELL_SIZE == 0 or point[1] % CELL_SIZE == 0


def is_intersection(point: Point) -> bool:
    """Return True if the point is a street intersection."""
    return point[0] % CELL_SIZE == 0 and point[1] % CELL_SIZE == 0


def _in_map(point: Point) -> bool:
    """Return True if the point lies inside the map bounds."""
    return 0 <= point[0] <= MAP_SIZE and 0 <= point[1] <= MAP_SIZE


def _append(path: List[Point], point: Point) -> None:
    """Append ``point`` to ``path`` unless it duplicates the last point."""
    if not path or path[-1] != point:
        path.append(point)


def snap_to_street(point: Point) -> Point:
    """Return the nearest street point reachable by one axis-aligned move.

    Args:
        point: Any point inside the map.

    Returns:
        ``point`` itself when it is already on a street, otherwise the
        closest point on a street that shares either its x or y coordinate.
    """
    if is_on_street(point):
        return point
    x, y = point
    row_y = min(MAP_SIZE, (y + HALF_CELL) // CELL_SIZE * CELL_SIZE)
    col_x = min(MAP_SIZE, (x + HALF_CELL) // CELL_SIZE * CELL_SIZE)
    candidates = ((x, row_y), (col_x, y))
    return min(candidates, key=lambda q: manhattan(point, q))


class PathPlanner:
    """Plans street-following routes between arbitrary map points.

    The planner runs Dijkstra's algorithm over the 9x9 grid of street
    intersections. Off-street start and target points are connected to the
    intersections through the street segments that bound them. A target that
    lies directly above, below, left or right of a street segment is reached
    through the matching access point on that segment, which covers the
    "target directly above or below" special case.
    """

    def plan(self, start: Point, target: Point) -> List[Point]:
        """Compute a street-following route from ``start`` to ``target``.

        Args:
            start: Starting point (integer coordinates, 0-800).
            target: Destination point (integer coordinates, 0-800).

        Returns:
            Waypoint list beginning with ``start`` and ending with
            ``target``. Consecutive waypoints differ along one axis only.
        """
        start = (int(start[0]), int(start[1]))
        target = (int(target[0]), int(target[1]))
        if start == target:
            return [start]

        dist: Dict[Point, int] = {}
        prev: Dict[Point, Optional[Point]] = {}
        head: Dict[Point, List[Point]] = {}
        heap: List[Tuple[int, Point]] = []

        for node, poly in self._exits(start):
            cost = path_length(poly)
            if node not in dist or cost < dist[node]:
                dist[node] = cost
                prev[node] = None
                head[node] = poly
                heapq.heappush(heap, (cost, node))

        while heap:
            d, u = heapq.heappop(heap)
            if d > dist.get(u, math.inf):
                continue
            for v in self._neighbours(u):
                nd = d + CELL_SIZE
                if nd < dist.get(v, math.inf):
                    dist[v] = nd
                    prev[v] = u
                    heapq.heappush(heap, (nd, v))

        best: Optional[Tuple[int, Point, List[Point]]] = None
        for node, tail in self._entries(target):
            if node not in dist:
                continue
            total = dist[node] + path_length(tail)
            if best is None or total < best[0]:
                best = (total, node, tail)

        if best is None:
            return [start, target]

        _, node, tail = best
        chain: List[Point] = []
        cur: Optional[Point] = node
        while cur is not None:
            chain.append(cur)
            cur = prev.get(cur)
        chain.reverse()

        path = list(head[chain[0]])
        for step in chain[1:]:
            _append(path, step)
        for point in tail:
            _append(path, point)
        return path

    def plan_route(self, start: Point, restaurant: Point,
                   customer: Point) -> List[Point]:
        """Compute the continuous route start -> restaurant -> customer.

        Args:
            start: Rider's current position.
            restaurant: Pickup point (grid cell centre).
            customer: Delivery point (grid cell centre).

        Returns:
            Single waypoint list covering both legs without duplicates.
        """
        route = list(self.plan(start, restaurant))
        for point in self.plan(restaurant, customer):
            _append(route, point)
        return route

    def _exits(self, point: Point) -> List[Tuple[Point, List[Point]]]:
        """Return intersections reachable from ``point`` along streets.

        Each result is ``(node, poly)`` where ``poly`` starts at ``point``
        and ends at ``node``.
        """
        if is_intersection(point):
            return [(point, [point])]
        if is_on_street(point):
            x, y = point
            if x % CELL_SIZE == 0:
                nodes = [(x, (y // CELL_SIZE) * CELL_SIZE),
                         (x, ((y + CELL_SIZE - 1) // CELL_SIZE) * CELL_SIZE)]
            else:
                nodes = [((x // CELL_SIZE) * CELL_SIZE, y),
                         (((x + CELL_SIZE - 1) // CELL_SIZE) * CELL_SIZE, y)]
            result: List[Tuple[Point, List[Point]]] = []
            seen = set()
            for node in nodes:
                if node in seen or not _in_map(node):
                    continue
                seen.add(node)
                result.append((node, [point, node]))
            return result
        snapped = snap_to_street(point)
        return [(node, [point] + poly)
                for node, poly in self._exits(snapped)]

    def _entries(self, target: Point) -> List[Tuple[Point, List[Point]]]:
        """Return intersections from which ``target`` can be reached.

        Each result is ``(node, tail)`` where ``tail`` starts at ``node``
        and ends at ``target``.
        """
        if is_on_street(target):
            return [(node, list(reversed(poly)))
                    for node, poly in self._exits(target)]
        x, y = target
        entries: List[Tuple[Point, List[Point]]] = []
        access_points = ((x - HALF_CELL, y), (x + HALF_CELL, y),
                         (x, y - HALF_CELL), (x, y + HALF_CELL))
        for access in access_points:
            if not _in_map(access) or not is_on_street(access):
                continue
            for node, poly in self._exits(access):
                entries.append((node, list(reversed(poly)) + [target]))
        return entries

    @staticmethod
    def _neighbours(node: Point) -> List[Point]:
        """Return the adjacent intersections of ``node`` inside the map."""
        x, y = node
        candidates = ((x + CELL_SIZE, y), (x - CELL_SIZE, y),
                      (x, y + CELL_SIZE), (x, y - CELL_SIZE))
        return [p for p in candidates if _in_map(p)]

"""Order data model.

An order is created from raw coordinates in the range 0-800. Each coordinate
pair is converted to the centre point of its grid cell.
"""

from typing import Any, Dict, Optional, Sequence, Tuple

import globalvar
from path_planner import PathPlanner, path_length

MAP_SIZE = 800
CELL_SIZE = 100
GRID_CELLS = 8

STATUS_CREATED = "created"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def validate_coordinate(value: Any, name: str = "coordinate") -> int:
    """Validate that ``value`` is an integer in the range 0-800.

    Args:
        value: Value to validate.
        name: Human-readable name used in error messages.

    Returns:
        The validated integer.

    Raises:
        TypeError: If ``value`` is not an integer.
        ValueError: If ``value`` is outside 0-800.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise TypeError(f"{name} must be an integer, got {value!r}")
    if not 0 <= value <= MAP_SIZE:
        raise ValueError(
            f"{name} must be between 0 and {MAP_SIZE}, got {value}")
    return value


def to_grid_center(x: int, y: int) -> Tuple[int, int]:
    """Convert a map coordinate to the centre of its 100x100 grid cell.

    Uses ``center = (coord // 100) * 100 + 50``. The value 800 belongs to the
    last cell, so it is clamped to that cell.

    Args:
        x: X coordinate (0-800).
        y: Y coordinate (0-800).

    Returns:
        ``(center_x, center_y)``.
    """
    x = validate_coordinate(x, "x")
    y = validate_coordinate(y, "y")
    cell_x = min(x // CELL_SIZE, GRID_CELLS - 1)
    cell_y = min(y // CELL_SIZE, GRID_CELLS - 1)
    return cell_x * CELL_SIZE + CELL_SIZE // 2, \
        cell_y * CELL_SIZE + CELL_SIZE // 2


def _split_position(pos: Sequence[Any], name: str) -> Tuple[int, int]:
    """Split a position sequence into validated ``(x, y)`` integers."""
    if len(pos) != 2:
        raise ValueError(f"{name} must contain exactly two coordinates")
    return (validate_coordinate(pos[0], f"{name} x"),
            validate_coordinate(pos[1], f"{name} y"))


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order identifier.
        raw_res: Restaurant coordinates as entered by the user.
        raw_cus: Customer coordinates as entered by the user.
        res_pos: Restaurant grid-cell centre ``(x, y)``.
        cus_pos: Customer grid-cell centre ``(x, y)``.
        status: One of "created", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: ID of the assigned rider, or None.
        start_time: Simulated tick when the order was created.
        completed_time: Simulated tick of delivery, or None.
        distance: Street route length from restaurant to customer.
        time_taken: Ticks from creation to delivery, or None.
        efficiency: Distance per tick (units/tick), or None.
    """

    _id_counter = 0

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order and convert its coordinates to grid centres.

        Args:
            res_pos: Restaurant coordinates ``(x, y)`` in range 0-800.
            cus_pos: Customer coordinates ``(x, y)`` in range 0-800.

        Raises:
            TypeError: If a coordinate is not an integer.
            ValueError: If a coordinate is outside 0-800.
        """
        raw_res = _split_position(res_pos, "restaurant")
        raw_cus = _split_position(cus_pos, "customer")

        Order._id_counter += 1
        self.id: int = Order._id_counter
        self.raw_res: Tuple[int, int] = raw_res
        self.raw_cus: Tuple[int, int] = raw_cus
        self.res_pos: Tuple[int, int] = to_grid_center(*raw_res)
        self.cus_pos: Tuple[int, int] = to_grid_center(*raw_cus)
        self.status: str = STATUS_CREATED
        self.rider_id: Optional[int] = None
        self.start_time: int = int(globalvar.get_value("time", 0) or 0)
        self.completed_time: Optional[int] = None
        self.distance: int = path_length(
            PathPlanner().plan(self.res_pos, self.cus_pos))
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None

    def assign_to(self, rider_id: int) -> None:
        """Record that the order has been assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_completed(self, time: int) -> None:
        """Mark the order as delivered and compute efficiency data.

        Args:
            time: Simulated tick at which delivery finished.
        """
        self.status = STATUS_COMPLETED
        self.completed_time = time
        self.time_taken = max(time - self.start_time, 0)
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = None

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "raw_res": list(self.raw_res),
            "raw_cus": list(self.raw_cus),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completed_time": self.completed_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

"""Rider model: state machine, movement and event log.

Rider states:
    0 = idle, 1 = picking up (en route to restaurant),
    2 = delivering (en route to customer).

Events written to ``Rider.log``:
    "spawn", "accepted_order", "arrived_at_merchant",
    "delivered_to_customer".
"""

import logging
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from path_planner import PathPlanner, is_on_street

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

DEFAULT_SPEED = 10
SPAWN_X = 435
SPAWN_Y = 385
MAX_WAIT_TICKS = 3

Point = Tuple[int, int]


class Rider:
    """A delivery rider that follows street routes one tick at a time.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates.
        speed: Movement budget in units per tick.
        state: Integer status (0 idle, 1 picking up, 2 delivering).
        orders: Assigned orders; index 0 is the active order.
        special_case: True while the rider is waiting for another rider.
        log: Chronological list of event dictionaries.
        path: Remaining waypoints of the current leg.
    """

    def __init__(self, rider_id: int, speed: int = DEFAULT_SPEED) -> None:
        """Create a rider at the spawn location in the idle state.

        Args:
            rider_id: Unique identifier for the rider.
            speed: Movement speed in units per tick.
        """
        self.rider_id: int = int(rider_id)
        self.x: int = SPAWN_X
        self.y: int = SPAWN_Y
        self.speed: int = int(speed)
        self.state: int = IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.log: List[Dict[str, Any]] = []
        self.path: List[Point] = []
        self._wait_ticks: int = 0
        self._planner = PathPlanner()
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # Properties and helpers
    # ------------------------------------------------------------------
    @property
    def position(self) -> Point:
        """Return the current ``(x, y)`` position."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Return a human-readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def current_order(self) -> Optional[Any]:
        """Return the active order, or None if the rider has none."""
        return self.orders[0] if self.orders else None

    def _now(self) -> int:
        """Return the current simulated tick."""
        return int(globalvar.get_value("time", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log."""
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "rider_id": self.rider_id,
        })

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [o.id for o in self.orders],
            "log": list(self.log),
        }

    # ------------------------------------------------------------------
    # Simulation
    # ------------------------------------------------------------------
    def move(self) -> None:
        """Advance the rider by one tick.

        Handles the state transitions:
            idle -> picking up when an order is present,
            picking up -> delivering on arrival at the restaurant,
            delivering -> idle on arrival at the customer.
        """
        if self.state == IDLE:
            if not self.orders:
                return
            self._start_pickup()

        budget = self.speed
        while True:
            budget = self._advance(budget)
            if self.path:
                return
            if self.state == PICKING_UP:
                self._arrive_merchant()
            elif self.state == DELIVERING:
                self._deliver()
                return
            else:
                return

    def _start_pickup(self) -> None:
        """Begin the pickup leg for the active order."""
        order = self.orders[0]
        self.state = PICKING_UP
        order.status = "picking_up"
        self.path = self._planner.plan((self.x, self.y), order.res_pos)[1:]
        self._record("accepted_order", order.id)

    def _arrive_merchant(self) -> None:
        """Handle arrival at the restaurant and plan the delivery leg."""
        order = self.orders[0]
        self._record("arrived_at_merchant", order.id)
        self.state = DELIVERING
        order.status = "delivering"
        self.path = self._planner.plan((self.x, self.y), order.cus_pos)[1:]

    def _deliver(self) -> None:
        """Handle delivery completion and release the rider."""
        order = self.orders.pop(0)
        order.mark_completed(self._now())
        self._record("delivered_to_customer", order.id)
        completed = globalvar.get_value("completed_orders")
        if completed is not None:
            completed.append(order)
        self.state = IDLE
        self.path = []
        self._wait_ticks = 0
        self.special_case = False

    def _advance(self, budget: int) -> int:
        """Consume movement budget along the current path.

        Returns:
            The unused budget. The rider stops early when it must wait for a
            blocking rider; the wait is limited to MAX_WAIT_TICKS to avoid
            deadlock.
        """
        while budget > 0 and self.path:
            nxt = self.path[0]
            if self._is_blocked(nxt):
                if self._wait_ticks < MAX_WAIT_TICKS:
                    self._wait_ticks += 1
                    self.special_case = True
                    return 0
                logger.info("Rider %s breaks deadlock at %s",
                            self.rider_id, nxt)
                self._wait_ticks = 0
            self.special_case = False
            budget -= self._step_towards(nxt, budget)
            if (self.x, self.y) == nxt:
                self.path.pop(0)
        return budget

    def _step_towards(self, target: Point, budget: int) -> int:
        """Move along an axis-aligned segment towards ``target``.

        Returns:
            The distance actually moved.
        """
        dx = target[0] - self.x
        dy = target[1] - self.y
        dist = abs(dx) + abs(dy)
        if dist <= budget:
            self.x, self.y = target
            return dist
        if dx != 0:
            step = min(budget, abs(dx))
            self.x += step if dx > 0 else -step
            return step
        step = min(budget, abs(dy))
        self.y += step if dy > 0 else -step
        return step

    def _is_blocked(self, point: Point) -> bool:
        """Return True if another busy rider occupies ``point``."""
        for other in globalvar.get_value("riders") or []:
            if other is self:
                continue
            if other.state != IDLE and other.position == point:
                return True
        return False

    def __repr__(self) -> str:
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state_name}, orders={len(self.orders)}, "
                f"on_street={is_on_street((self.x, self.y))})")

"""Order assignment: nearest-rider selection and order management."""

import logging
import math
from typing import TYPE_CHECKING, List, Optional, Sequence

import globalvar
from order_model import Order

if TYPE_CHECKING:
    from rider import Rider

logger = logging.getLogger(__name__)

MAX_ORDERS_PER_RIDER = 3


def find_best_rider_for_order(order: Order) -> Optional["Rider"]:
    """Select the rider closest to the order's restaurant.

    Distance is Euclidean: sqrt((rider_x - restaurant_x)^2 +
    (rider_y - restaurant_y)^2). Riders already holding
    MAX_ORDERS_PER_RIDER orders are skipped unless every rider is full,
    which provides load balancing. Ties go to the rider with fewer orders.

    Args:
        order: The order to assign.

    Returns:
        The chosen Rider, or None if no rider exists.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    rest_x, rest_y = order.res_pos
    candidates = [r for r in riders if len(r.orders) < MAX_ORDERS_PER_RIDER]
    if not candidates:
        candidates = list(riders)
    return min(
        candidates,
        key=lambda r: (math.hypot(r.x - rest_x, r.y - rest_y),
                       len(r.orders), r.rider_id),
    )


class OrderManager:
    """Creates orders, assigns them to riders and tracks order history."""

    def create_order(self, res_pos: Sequence[int],
                     cus_pos: Sequence[int]) -> Order:
        """Create an order and register it in the global order list.

        Args:
            res_pos: Restaurant coordinates ``(x, y)``.
            cus_pos: Customer coordinates ``(x, y)``.

        Returns:
            The newly created Order.

        Raises:
            TypeError, ValueError: If coordinates are invalid.
        """
        order = Order(res_pos, cus_pos)
        orders = globalvar.get_value("orders")
        if orders is not None:
            orders.append(order)
        logger.info("Order %s created: restaurant %s, customer %s",
                    order.id, order.res_pos, order.cus_pos)
        return order

    def assign_order(self, order: Order) -> Optional["Rider"]:
        """Assign an order to the best rider by appending it to rider.orders.

        Args:
            order: Order to assign.

        Returns:
            The rider that received the order, or None if no rider exists.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            logger.info("Order %s could not be assigned: no riders",
                        order.id)
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Order %s assigned to rider %s",
                    order.id, rider.rider_id)
        return rider

    @staticmethod
    def completed_orders() -> List[Order]:
        """Return the list of delivered orders."""
        return list(globalvar.get_value("completed_orders") or [])

"""System-level resource management and simulation control."""

import json
import logging
import os
from typing import Any, Dict, List, Optional, Sequence

import globalvar
from order import OrderManager
from order_model import Order
from rider import IDLE, Rider

logger = logging.getLogger(__name__)

DEFAULT_CONFIG: Dict[str, Any] = {
    "initial_funds": 1000,
    "recruit_cost": 300,
    "recruit_threshold": 300,
    "rider_speed": 10,
    "max_ticks_per_order": 5000,
}


class SystemManager:
    """Manages funds, riders, pending orders and the simulation clock.

    Recruitment rule: when funds are at least ``recruit_threshold`` and the
    number of pending orders exceeds the number of idle riders, a new rider
    is created at a cost of ``recruit_cost``.
    """

    def __init__(self, config: Dict[str, Any]) -> None:
        """Create the manager.

        Args:
            config: System parameters (see DEFAULT_CONFIG).
        """
        self.config: Dict[str, Any] = dict(config)
        self.order_manager = OrderManager()

    # ------------------------------------------------------------------
    # Accessors
    # ------------------------------------------------------------------
    @property
    def funds(self) -> int:
        """Current system fund balance."""
        return int(globalvar.get_value("funds", 0))

    @property
    def riders(self) -> List[Rider]:
        """The live rider list."""
        return globalvar.get_value("riders")

    @property
    def time(self) -> int:
        """Current simulated tick."""
        return int(globalvar.get_value("time", 0))

    @property
    def running(self) -> bool:
        """Main loop running flag."""
        return bool(globalvar.get_value("running", False))

    @property
    def orders(self) -> List[Order]:
        """All orders created in this session."""
        return globalvar.get_value("orders")

    def pending_orders(self) -> List[Order]:
        """Return the live list of orders waiting for a rider."""
        return globalvar.get_value("pending_orders")

    def active_orders(self) -> List[Order]:
        """Return orders that have not yet been completed."""
        return [o for o in self.orders if o.status != "completed"]

    def stop(self) -> None:
        """Clear the running flag so the main loop exits."""
        globalvar.set_value("running", False)

    def get_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with ``rider_id``, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def idle_rider_count(self) -> int:
        """Return the number of riders with state 0 and no orders."""
        return sum(1 for r in self.riders
                   if r.state == IDLE and not r.orders)

    # ------------------------------------------------------------------
    # Resource management
    # ------------------------------------------------------------------
    def recruit_rider(self) -> Optional[Rider]:
        """Create a new rider if funds cover the recruitment cost.

        Returns:
            The new Rider, or None if funds are insufficient.
        """
        cost = int(self.config["recruit_cost"])
        if self.funds < cost:
            logger.info("Recruitment skipped: funds %s < cost %s",
                        self.funds, cost)
            return None
        counter = int(globalvar.get_value("rider_id_counter", 0)) + 1
        globalvar.set_value("rider_id_counter", counter)
        globalvar.set_value("funds", self.funds - cost)
        rider = Rider(counter, speed=int(self.config["rider_speed"]))
        self.riders.append(rider)
        logger.info("Recruited rider %s (funds now %s)",
                    rider.rider_id, self.funds)
        return rider

    # ------------------------------------------------------------------
    # Order handling
    # ------------------------------------------------------------------
    def submit_order(self, res_pos: Sequence[int],
                     cus_pos: Sequence[int]) -> Order:
        """Create an order and try to dispatch it immediately.

        Args:
            res_pos: Restaurant coordinates ``(x, y)`` in range 0-800.
            cus_pos: Customer coordinates ``(x, y)`` in range 0-800.

        Returns:
            The created Order. Check ``order.rider_id`` to see whether it has
            been assigned; unassigned orders stay pending.
        """
        order = self.order_manager.create_order(res_pos, cus_pos)
        self.pending_orders().append(order)
        self.dispatch_pending()
        return order

    def dispatch_pending(self) -> List[Rider]:
        """Recruit riders when needed and assign pending orders.

        Returns:
            Riders recruited during this call.
        """
        recruited: List[Rider] = []
        threshold = int(self.config["recruit_threshold"])
        for order in list(self.pending_orders()):
            while (self.funds >= threshold
                   and len(self.pending_orders()) > self.idle_rider_count()):
                rider = self.recruit_rider()
                if rider is None:
                    break
                recruited.append(rider)
            if self.order_manager.assign_order(order) is not None:
                self.pending_orders().remove(order)
        return recruited

    # ------------------------------------------------------------------
    # Simulation loop
    # ------------------------------------------------------------------
    def tick(self) -> None:
        """Advance the simulation by one tick.

        Dispatches pending orders, moves every rider, then increments the
        clock. Exceptions from a single rider are logged and do not stop the
        simulation.
        """
        if not self.running:
            return
        self.dispatch_pending()
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # error recovery: keep other riders running
                logger.exception("Rider %s failed to move", rider.rider_id)
        globalvar.set_value("time", self.time + 1)

    def run_until_order_done(self, order: Order,
                             max_ticks: Optional[int] = None) -> bool:
        """Run ticks until ``order`` is completed.

        Args:
            order: Order to wait for.
            max_ticks: Tick limit; defaults to the configured value.

        Returns:
            True if the order was completed within the limit.
        """
        limit = int(max_ticks if max_ticks is not None
                    else self.config["max_ticks_per_order"])
        ticks = 0
        while order.status != "completed" and ticks < limit and self.running:
            self.tick()
            ticks += 1
        return order.status == "completed"

    def check_consistency(self) -> List[str]:
        """Return a list of detected state inconsistencies (empty if none)."""
        issues: List[str] = []
        seen: Dict[int, int] = {}
        for rider in self.riders:
            for order in rider.orders:
                if order.id in seen:
                    issues.append(f"order {order.id} held by two riders")
                seen[order.id] = rider.rider_id
                if order.status == "completed":
                    issues.append(f"completed order {order.id} still assigned")
        for order in self.pending_orders():
            if order.id in seen:
                issues.append(f"pending order {order.id} also assigned")
        return issues

    # ------------------------------------------------------------------
    # Persistence
    # ------------------------------------------------------------------
    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the system."""
        return {
            "time": self.time,
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "config": self.config,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.orders],
            "pending_order_ids": [o.id for o in self.pending_orders()],
        }

    def save_state(self, path: str) -> bool:
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination file path.

        Returns:
            True on success, False if the file could not be written.
        """
        try:
            directory = os.path.dirname(os.path.abspath(path))
            os.makedirs(directory, exist_ok=True)
            with open(path, "w", encoding="utf-8") as handle:
                json.dump(self.to_dict(), handle, indent=2)
            return True
        except (OSError, TypeError, ValueError) as exc:
            logger.error("Failed to save state to %s: %s", path, exc)
            return False


def load_config(path: str) -> Dict[str, Any]:
    """Load system parameters from a JSON configuration file.

    Unknown keys are ignored. Missing or invalid files yield an empty dict.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        A dictionary of recognised parameter overrides.
    """
    try:
        with open(path, encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read config %s: %s", path, exc)
        return {}
    if not isinstance(data, dict):
        logger.warning("Config %s must contain a JSON object", path)
        return {}
    return {k: v for k, v in data.items() if k in DEFAULT_CONFIG}


def init_system(config_path: Optional[str] = None) -> SystemManager:
    """Initialise global state and return the system manager.

    Sets funds to 1000, an empty rider list, rider ID counter 0, running
    state True and time 0, then applies optional configuration overrides.

    Args:
        config_path: Optional JSON configuration file.

    Returns:
        A ready-to-use SystemManager.
    """
    globalvar._init()
    config = dict(DEFAULT_CONFIG)
    if config_path:
        config.update(load_config(config_path))
    globalvar.set_value("config", config)
    globalvar.set_value("funds", int(config["initial_funds"]))
    return SystemManager(config)

"""Status display and ASCII map rendering.

Map layout: the 800x800 area is drawn as a 17x17 character canvas. Each
canvas index corresponds to 50 map units, so even indices are street lines
and odd-odd indices are grid-cell centres.

Symbols:
    R rider, S restaurant, C customer, . empty cell, # street, * route.
"""

from typing import Any, Iterable, List, Optional, Sequence, Tuple

from rider import STATE_NAMES

Point = Tuple[int, int]

HALF = 50
CANVAS_SIZE = 17
LEGEND = ("Legend: R=Rider  S=Restaurant  C=Customer  "
          ".=Empty  #=Street  *=Route  (x right, y up, 0-800)")


def _index(value: int) -> int:
    """Map a coordinate to a canvas index, clamped to the canvas."""
    return min(CANVAS_SIZE - 1, max(0, (value + HALF // 2) // HALF))


class DisplayManager:
    """Renders status snapshots, order tables, rider logs and the map."""

    def render_map(self, riders: Iterable[Any] = (),
                   orders: Iterable[Any] = (),
                   route: Optional[Sequence[Point]] = None) -> str:
        """Render the ASCII map with riders, orders and an optional route.

        Args:
            riders: Rider objects to plot.
            orders: Orders whose restaurant (S) and customer (C) points
                should be plotted. Completed orders are skipped.
            route: Optional waypoint list drawn with '*'.

        Returns:
            The multi-line map string.
        """
        n = CANVAS_SIZE
        canvas = [["#" if (i % 2 == 0 or j % 2 == 0) else "."
                   for i in range(n)] for j in range(n)]

        def place(x: int, y: int, symbol: str, overwrite_all: bool) -> None:
            i, j = _index(x), _index(y)
            if overwrite_all or canvas[j][i] in "#.*":
                canvas[j][i] = symbol

        if route:
            for x, y in route:
                place(x, y, "*", overwrite_all=False)
        for order in orders:
            if order.status == "completed":
                continue
            place(order.res_pos[0], order.res_pos[1], "S", True)
            place(order.cus_pos[0], order.cus_pos[1], "C", True)
        for rider in riders:
            place(rider.x, rider.y, "R", True)

        lines = []
        for j in range(n - 1, -1, -1):
            label = f"{j * HALF:>4} " if j % 2 == 0 else "     "
            lines.append(label + " ".join(canvas[j]))
        return "\n".join(lines + ["", LEGEND])

    def render_status(self, funds: int, time: int,
                      riders: Sequence[Any]) -> str:
        """Render the system status snapshot.

        Args:
            funds: Current system funds.
            time: Current simulated tick.
            riders: All riders.

        Returns:
            Multi-line status text.
        """
        lines = [
            f"System time : {time} tick(s)",
            f"System funds: {funds}",
            f"Riders      : {len(riders)}",
        ]
        for rider in riders:
            order_ids = [o.id for o in rider.orders]
            waiting = " [waiting]" if rider.special_case else ""
            lines.append(
                f"  Rider #{rider.rider_id:<3} pos=({rider.x},{rider.y})  "
                f"status={STATE_NAMES.get(rider.state, 'unknown')}{waiting}  "
                f"orders={len(order_ids)} {order_ids if order_ids else ''}"
            )
        return "\n".join(lines)

    def render_orders(self, orders: Sequence[Any]) -> str:
        """Render a table of all orders.

        Args:
            orders: Orders to display.

        Returns:
            Multi-line table text.
        """
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4}  {'Start':>5}  {'Restaurant':<11} "
                  f"{'Customer':<11} {'Dist':>6}  {'Taken':>5}  "
                  f"{'Rider':>6}  Status")
        rows: List[str] = [header, "-" * len(header)]
        for order in orders:
            rest = f"({order.res_pos[0]},{order.res_pos[1]})"
            cust = f"({order.cus_pos[0]},{order.cus_pos[1]})"
            taken = "-" if order.time_taken is None else str(order.time_taken)
            rider = "-" if order.rider_id is None else f"#{order.rider_id}"
            rows.append(
                f"{order.id:>4}  {order.start_time:>5}  {rest:<11} "
                f"{cust:<11} {order.distance:>6}  {taken:>5}  "
                f"{rider:>6}  {order.status}"
            )
        return "\n".join(rows)

    def render_rider_logs(self, riders: Sequence[Any]) -> str:
        """Render every recorded status change for all riders.

        Args:
            riders: Riders whose logs are shown.

        Returns:
            Multi-line log text.
        """
        if not riders:
            return "No riders."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} (speed {rider.speed}, "
                         f"{len(rider.log)} event(s))")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order = ("-" if entry["order_id"] is None
                         else f"#{entry['order_id']}")
                lines.append(
                    f"  t={entry['time']:>4}  {entry['event']:<22} "
                    f"order={order:<5} pos=({entry['x']},{entry['y']})  "
                    f"state={STATE_NAMES.get(entry['state'], 'unknown')}"
                )
        return "\n".join(lines)

"""Command parsing, input validation and the interactive command loop."""

import logging
import math
from typing import Callable, Dict, List, Optional, Sequence, Tuple

from display import DisplayManager
from path_planner import MAP_SIZE, PathPlanner, path_length
from rider import STATE_NAMES

logger = logging.getLogger(__name__)

BANNER = (
    "=== Food Delivery Order Dispatch Simulator ===\n"
    "Type 'help' for commands, 'quit' to exit."
)

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant x y, customer x y, 0-800).
                       Without arguments you are prompted for the four values.
                       The route is shown and the delivery is simulated.
  status  (s, st)      Show system funds, riders and their orders.
  orders  (o)          Show all orders with time, distance and rider.
  riders  (r)          Show every recorded rider status change.
  map     (m)          Show the ASCII map with riders and active orders.
  tick [n] (t)         Advance the simulation by n ticks (default 1).
  history (hist)       Show command history.
  !!                   Repeat the previous command.
  help    (h, ?)       Show this help.
  quit    (q, exit)    Exit the program.
Coordinates are integers in the range 0-800 and may be separated by spaces
or commas."""

ALIASES: Dict[str, str] = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "h": "help",
    "?": "help",
    "q": "quit",
    "x": "exit",
    "hist": "history",
}

COORD_LABELS = ("restaurant x", "restaurant y", "customer x", "customer y")


class InputManager:
    """Reads commands from the user and dispatches them to the system."""

    PROMPT = "dispatch> "

    def __init__(self, system, display: Optional[DisplayManager] = None,
                 input_func: Optional[Callable[[str], str]] = None,
                 output_func: Optional[Callable[..., None]] = None) -> None:
        """Create the input manager.

        Args:
            system: The SystemManager instance.
            display: Optional DisplayManager; a default one is created.
            input_func: Function used to read lines (defaults to ``input``).
            output_func: Function used to print lines (defaults to ``print``).
        """
        self.system = system
        self.display = display or DisplayManager()
        self._input = input_func or input
        self._output = output_func or print
        self._planner = PathPlanner()
        self.history: List[str] = []
        self._handlers: Dict[str, Callable[[List[str]], Optional[bool]]] = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
            "exit": self._cmd_quit,
        }
        self._enable_readline()

    # ------------------------------------------------------------------
    # Parsing and validation
    # ------------------------------------------------------------------
    @staticmethod
    def parse_command(line: str) -> Tuple[str, List[str]]:
        """Split a command line into a canonical command and arguments.

        Args:
            line: Raw input line.

        Returns:
            ``(command, args)``; the command is lower-case and aliases are
            expanded. An empty line yields ``("", [])``.
        """
        tokens = line.strip().split()
        if not tokens:
            return "", []
        command = tokens[0].lower()
        return ALIASES.get(command, command), tokens[1:]

    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> Tuple[int, int, int, int]:
        """Parse and validate the four order coordinates.

        Args:
            tokens: Tokens such as ``["100", "200", "300", "400"]``. Commas
                are accepted as separators.

        Returns:
            ``(restaurant_x, restaurant_y, customer_x, customer_y)``.

        Raises:
            ValueError: If the count, type or range of a value is wrong.
        """
        parts = " ".join(tokens).replace(",", " ").split()
        if len(parts) != 4:
            raise ValueError(
                "expected 4 integers (restaurant x y, customer x y), "
                f"got {len(parts)}")
        values: List[int] = []
        for label, text in zip(COORD_LABELS, parts):
            try:
                value = int(text)
            except ValueError:
                raise ValueError(
                    f"{label} must be an integer, got '{text}'") from None
            if not 0 <= value <= MAP_SIZE:
                raise ValueError(
                    f"{label} must be between 0 and {MAP_SIZE}, got {value}")
            values.append(value)
        return values[0], values[1], values[2], values[3]

    @staticmethod
    def _enable_readline() -> None:
        """Enable line editing and up-arrow history when available."""
        try:
            import readline  # noqa: F401  (side effect: line editing)
        except ImportError:
            logger.debug("readline not available; history via 'history'")

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> None:
        """Run the interactive command loop until quit or end of input."""
        self._print(BANNER)
        while self.system.running:
            try:
                line = self._input(self.PROMPT)
            except EOFError:
                self._print("")
                break
            except KeyboardInterrupt:
                self._print("\nUse 'quit' to exit.")
                continue

            stripped = line.strip()
            if not stripped:
                continue
            if stripped == "!!":
                if not self.history:
                    self._print("Error: no previous command to repeat.")
                    continue
                stripped = self.history[-1]
                self._print(f"(repeating: {stripped})")
            self.history.append(stripped)

            try:
                keep_running = self.execute(stripped)
            except Exception as exc:  # error recovery: keep the loop alive
                logger.exception("Command failed: %s", stripped)
                self._print(f"Error: unexpected failure ({exc}). "
                            "System state is preserved.")
                keep_running = True
            if keep_running is False:
                break
        self.system.stop()

    def execute(self, line: str) -> bool:
        """Execute a single command line.

        Args:
            line: Command text.

        Returns:
            False when the program should exit, True otherwise.
        """
        command, args = self.parse_command(line)
        if not command:
            return True
        handler = self._handlers.get(command)
        if handler is None:
            self._print(f"Unknown command '{command}'. Type 'help'.")
            return True
        return handler(args) is not False

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def _cmd_start(self, args: List[str]) -> None:
        """Create an order, show its route and simulate its delivery."""
        try:
            if args:
                coords = self.parse_coordinates(args)
            else:
                self._print("Enter restaurant_x restaurant_y customer_x "
                            "customer_y (0-800):")
                coords = self.parse_coordinates(self._input("  > ").split())
        except ValueError as exc:
            self._print(f"Input error: {exc}")
            return

        rx, ry, cx, cy = coords
        riders_before = len(self.system.riders)
        try:
            order = self.system.submit_order((rx, ry), (cx, cy))
        except (TypeError, ValueError) as exc:
            self._print(f"Order rejected: {exc}")
            return

        new_riders = self.system.riders[riders_before:]
        self._print(f"Order #{order.id} accepted at t={order.start_time}.")
        self._print(f"  Restaurant grid centre : {order.res_pos}")
        self._print(f"  Customer grid centre   : {order.cus_pos}")
        if new_riders:
            ids = ", ".join(f"#{r.rider_id}" for r in new_riders)
            self._print(f"  New rider added        : yes ({ids})")
        else:
            self._print("  New rider added        : no")

        if order.rider_id is None:
            self._print("  No rider available; order is pending "
                        f"(funds: {self.system.funds}).")
            return

        rider = self.system.get_rider(order.rider_id)
        route = self._planner.plan_route(rider.position, order.res_pos,
                                         order.cus_pos)
        length = path_length(route)
        ticks = math.ceil(length / rider.speed) if rider.speed else 0
        self._print(f"  Assigned to rider #{rider.rider_id} at "
                    f"({rider.x},{rider.y}), speed {rider.speed} units/tick")
        self._print(f"  Route length: {length} units (about {ticks} ticks)")
        self._print(self.display.render_map(
            riders=self.system.riders,
            orders=self.system.active_orders(),
            route=route))

        self._print("Simulating delivery ...")
        done = self.system.run_until_order_done(order)
        if done:
            self._print(
                f"Order #{order.id} delivered at t={order.completed_time}. "
                f"Time taken: {order.time_taken} tick(s), "
                f"distance: {order.distance} units, "
                f"efficiency: {order.efficiency} units/tick.")
        else:
            self._print(f"Order #{order.id} is not completed yet "
                        f"(status: {order.status}). Use 'tick' to continue.")

    def _cmd_status(self, args: List[str]) -> None:
        """Print the system status snapshot."""
        self._print(self.display.render_status(
            self.system.funds, self.system.time, self.system.riders))

    def _cmd_orders(self, args: List[str]) -> None:
        """Print the table of all orders."""
        self._print(self.display.render_orders(self.system.orders))

    def _cmd_riders(self, args: List[str]) -> None:
        """Print the event log of every rider."""
        self._print(self.display.render_rider_logs(self.system.riders))

    def _cmd_map(self, args: List[str]) -> None:
        """Print the ASCII map."""
        self._print(self.display.render_map(
            riders=self.system.riders,
            orders=self.system.active_orders()))

    def _cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation by one or more ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                self._print("Input error: tick count must be an integer.")
                return
            if not 1 <= count <= 1000:
                self._print("Input error: tick count must be 1-1000.")
                return
        for _ in range(count):
            self.system.tick()
        self._print(f"Advanced {count} tick(s); time is now "
                    f"{self.system.time}.")

    def _cmd_history(self, args: List[str]) -> None:
        """Print the command history."""
        if not self.history:
            self._print("History is empty.")
            return
        for index, entry in enumerate(self.history, start=1):
            self._print(f"  {index:>3}  {entry}")

    def _cmd_help(self, args: List[str]) -> None:
        """Print the help text."""
        self._print(HELP_TEXT)

    def _cmd_quit(self, args: List[str]) -> bool:
        """Stop the simulation and exit the loop."""
        self.system.stop()

"""Entry point for the Food Delivery Order Dispatch System simulator.

Usage:
    python src/main.py [--config config.json] [--save state.json] [--debug]
"""

import argparse
import logging
import sys
from typing import List, Optional

from input_manager import InputManager
from system import init_system


def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: Argument list (defaults to ``sys.argv[1:]``).

    Returns:
        The parsed namespace.
    """
    parser = argparse.ArgumentParser(
        description="Food delivery order dispatch simulator (CLI).")
    parser.add_argument("--config", default=None,
                        help="JSON file with system parameter overrides.")
    parser.add_argument("--save", default=None,
                        help="JSON file to write the final system state to.")
    parser.add_argument("--log-file", default=None,
                        help="Write log output to this file instead of stderr.")
    parser.add_argument("--debug", action="store_true",
                        help="Enable debug logging.")
    return parser.parse_args(argv)


def configure_logging(debug: bool, log_file: Optional[str]) -> None:
    """Configure the root logger.

    Args:
        debug: Use DEBUG level when True, otherwise WARNING.
        log_file: Optional log file path.
    """
    level = logging.DEBUG if debug else logging.WARNING
    logging.basicConfig(
        level=level,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        filename=log_file,
    )


def main(argv: Optional[List[str]] = None) -> int:
    """Run the simulator.

    Args:
        argv: Optional argument list.

    Returns:
        Process exit code.
    """
    args = parse_args(argv)
    configure_logging(args.debug, args.log_file)
    system = init_system(args.config)
    try:
        InputManager(system).run()
    finally:
        if args.save:
            system.save_state(args.save)
    return 0


if __name__ == "__main__":
    sys.exit(main())

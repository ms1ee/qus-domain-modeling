from __future__ import annotations

import os

from lottery_system.models import User, normalize_text


DEFAULT_LIST_LINES: list[str] = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Wang Wu,Product Department",
    "Zhao Liu,Product Department",
    "Sun Qi,Operations Department",
    "Zhou Ba,Marketing Department",
    "Wu Jiu,Marketing Department",
    "Zheng Shi,Sales Department",
    "Li Bin,Sales Department",
    "Chen Qixian,Operations Department",
    "Jin Duo,Technology Department",
    "Ma Jiaqi,Product Department",
    "Cheng Xu,Cheng Ru Group",
    "Cheng Hao,Cheng Ru Group",
    "Ru Lan,Cheng Ru Group",
    "Feng Yu,Cheng Ru Group",
    "Han Mei,Marketing Department",
    "Xu Ming,Sales Department",
    "Gao Yang,Operations Department",
]


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a list file, returning parsed users and error messages."""
        cleaned = str(path).strip().strip('"').strip("'")
        if not cleaned or not os.path.isfile(cleaned):
            return [], [f"File Error: file not found: {cleaned}"]
        try:
            with open(cleaned, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{cleaned}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines are skipped."""
        users: list[User] = []
        errors: list[str] = []
        for index, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {index}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            name = " ".join(parts[0].split())
            department = " ".join(parts[1].split())
            users.append(User(name=name, department=department))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique entries and duplicate-name entries."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of members."""
        display_names: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            key = normalize_text(user.department)
            display_names.setdefault(key, user.department)
            counts[key] = counts.get(key, 0) + 1
        return {display_names[key]: counts[key] for key in sorted(counts, key=lambda k: display_names[k])}

from __future__ import annotations

from lottery_system.models import Prize, normalize_text


DEFAULT_PRIZES: list[tuple[str, int, str]] = [
    ("Third Prize - Movie Tickets", 5, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = normalize_text(name)
        for prize in self._prizes:
            if normalize_text(prize.name) == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize '{name}' already exists and was skipped")
                continue
            self._prizes.append(Prize(name=name, quantity=quantity, avoid_groups=[], associated_drawer=drawer))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Add a custom prize. Raises ValueError on invalid input."""
        clean_name = " ".join(str(name or "").split())
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize '{clean_name}' already exists.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(Prize(name=clean_name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer))

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        order[k] is the 1-based current position of the prize that should be
        placed at new position k + 1. Returns False if order is not a valid permutation.
        """
        count = len(self._prizes)
        try:
            values = list(order)
        except TypeError:
            return False
        if len(values) != count:
            return False
        if any(isinstance(v, bool) or not isinstance(v, int) for v in values):
            return False
        if sorted(values) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in values]
        return True

"""Lottery rule engine: eligibility pools, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User, normalize_text


LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
CHENG_RU_GROUP = "Cheng Ru Group"


class LotteryManager:
    """Runs draws while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self.drawn_winners: list[User] = []
        # Per-round flags for draws by Chen Qixian / Jin Duo:
        # True if the round produced at least one Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []

    @staticmethod
    def _name_key(user: User) -> str:
        return normalize_text(user.name)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return normalize_text(user.department) == normalize_text(CHENG_RU_GROUP)

    @staticmethod
    def _drawer_key(prize: Prize) -> str:
        return normalize_text(prize.associated_drawer or "")

    def is_winner(self, user: User) -> bool:
        """Return True if the user has already won in any round."""
        return self._name_key(user) in self._winner_keys

    def fallback_active(self) -> bool:
        """Cheng Ru protection triggers when the last two Chen/Jin draws had no Cheng Ru member."""
        return len(self.last_two_draw_results) >= 2 and not any(self.last_two_draw_results[-2:])

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the list of users eligible for the given prize at this moment."""
        available = [u for u in self._users if not self.is_winner(u)]
        drawer = self._drawer_key(prize)

        # Special rule: Li Bin draws only filter previous winners.
        if drawer == normalize_text(LI_BIN):
            return available

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group members,
        # overriding avoidance groups.
        if drawer == normalize_text(MA_JIAQI) and self.fallback_active():
            return [u for u in available if self._is_cheng_ru(u)]

        avoid = {normalize_text(g) for g in prize.avoid_groups}
        return [u for u in available if normalize_text(u.department) not in avoid]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible participant remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winner_keys.add(self._name_key(winner))
        self.drawn_winners.append(winner)
        return winner

    def complete_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Record the outcome of a finished round for fallback tracking."""
        drawer = self._drawer_key(prize)
        if drawer in (normalize_text(CHEN_QIXIAN), normalize_text(JIN_DUO)):
            hit = any(self._is_cheng_ru(u) for u in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to this prize."""
        rules: list[str] = []
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}")
        rules.append("previous winners are excluded")
        drawer = self._drawer_key(prize)
        if drawer == normalize_text(LI_BIN):
            rules.append("special rule for Li Bin: avoidance groups are not applied")
        else:
            if prize.avoid_groups:
                rules.append("avoided groups: " + ", ".join(prize.avoid_groups))
            if drawer == normalize_text(MA_JIAQI) and self.fallback_active():
                rules.append(f"fallback active: only {CHENG_RU_GROUP} members are eligible")
        return "; ".join(rules)

"""Winner recording, grouping and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    draw_order: int


class ResultManager:
    """Stores winners and exports results."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        """Record a winner for a prize."""
        prize_name = getattr(prize, "name", prize)
        self._winners.append(
            Winner(user=user, prize_name=str(prize_name), draw_order=len(self._winners) + 1)
        )

    def get_all_winners(self) -> list[Winner]:
        """Return all winners in draw order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, following the given prize list."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list as CSV into awards_result_YYYYMMDD.txt."""
        stamp = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{stamp}.txt")
        grouped = self.get_winners_by_prize(prizes)
        known_names = {p.name for p in prizes}
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for prize in prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
                for winner in self._winners:
                    if winner.prize_name not in known_names:
                        writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: cannot write '{path}': {exc}"
        return True, f"Winner list successfully exported to {path}"

import threading

class InputReader:
    """Reads stdin in a background thread so commands can be polled during a lottery."""

    def __init__(self, stream: Optional[TextIO] = None) -> None:
        self._stream = stream or sys.stdin
        self._queue: "queue.Queue[Optional[str]]" = queue.Queue()
        self._thread = threading.Thread(target=self._run, daemon=True)
        self._thread.start()

    def _run(self) -> None:
        while True:
            try:
                line = self._stream.readline()
            except (OSError, ValueError):
                line = ""
            if line == "":
                self._queue.put(None)
                return
            self._queue.put(line.rstrip("\r\n"))

    def readline(self) -> str:
        """Blocking read of the next line. Raises EOFError at end of input."""
        item = self._queue.get()
        if item is None:
            self._queue.put(None)
            raise EOFError
        return item

    def poll(self) -> Optional[str]:
        """Non-blocking read. Returns None if no line is available."""
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        if item is None:
            self._queue.put(None)
            raise EOFError
        return item


def ordinal(number: int) -> str:
    """Format an integer as an English ordinal, e.g. 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive menu-driven controller."""

    def __init__(self, reader: Optional[InputReader] = None) -> None:
        self.reader = reader or InputReader()
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.result_manager = ResultManager()
        self.lottery_done = False
        self._menu = {
            "1": self.load_list_menu,
            "2": self.show_groups,
            "3": self.prize_menu,
            "4": self.start_lottery,
            "5": self.show_results,
            "6": self.export_results,
        }

    # ------------------------------------------------------------------ io
    def ask(self, prompt: str) -> str:
        print(prompt, end="", flush=True)
        return self.reader.readline().strip()

    def _ask_positive_int(self, prompt: str) -> Optional[int]:
        while True:
            raw = self.ask(prompt)
            if raw == "":
                return None
            try:
                value = int(raw)
            except ValueError:
                print("Please enter a whole number.")
                continue
            if value <= 0:
                print("Value must be greater than zero.")
                continue
            return value

    # ---------------------------------------------------------------- main
    def run(self) -> int:
        print("=== Company Team Building Smart Lottery System ===")
        try:
            while True:
                self._print_main_menu()
                choice = self.ask("Select an option: ")
                if choice == "0":
                    print("Goodbye!")
                    return 0
                handler = self._menu.get(choice)
                if handler is None:
                    print("Invalid option. Please try again.")
                    continue
                handler()
        except EOFError:
            print("\nInput closed. Exiting.")
            return 0

    def _print_main_menu(self) -> None:
        print("\n==== Main Menu ====")
        print("1. Load participant list")
        print("2. View department groups")
        print("3. Prize configuration")
        print("4. Start lottery")
        print("5. Show winner list")
        print("6. Export winner list")
        print("0. Exit")

    # ---------------------------------------------------------- list menu
    def load_list_menu(self) -> None:
        print("\n-- Load Participant List --")
        print("1. Built-in team building list")
        print("2. Custom txt list (each line: Name,Department Group)")
        print("0. Back")
        choice = self.ask("Select: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
            self._process_loaded(users, errors)

    def _process_loaded(self, users: list[User], errors: list[str]) -> None:
        if errors:

        if duplicates:
            print(f"Found {len(duplicates)} duplicate name(s):")

    def add_custom_prize(self) -> None:

    def reorder_prizes(self) -> None:

        if len(prizes) < 2:
            print("At least two prizes are needed to adjust the order.")
            return
        self.view_prizes()

        try:
            order = [int(token) for token in raw.replace(",", " ").split()]
        except ValueError:

            return

            print("Lottery order updated.")
        else:
            print("Invalid order: every prize serial number must appear exactly once.")

    # ------------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return
        if self.lottery_done:
            answer = self.ask("A lottery was already run. Clear previous results and restart? (y/n): ")
            if answer.lower() != "y":
                return
            self.result_manager = ResultManager()

        lottery = LotteryManager(self.users)
        total = len(prizes)
        print("\n=== Lottery started ===")
        print("Commands during the lottery: pause | continue | terminate")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(lottery, prize, round_no, total)
        self.lottery_done = True
        print("\n=== All lottery rounds completed ===")
        self.show_results()

    def _apply_command(self, raw: str, state: dict) -> None:
        command = raw.strip().lower()
        if not command:
            return
        if command == "pause":
            if not state["paused"]:
                state["paused"] = True
                print("[Lottery paused] Type 'continue' to resume or 'terminate' to end this round.")
        elif command == "continue":
            if state["paused"]:
                state["paused"] = False
                print("[Lottery resumed]")
        elif command == "terminate":
            state["terminated"] = True
            state["paused"] = False
            print("[Round terminated] Skipping to the next round.")
        else:
            print(f"Unknown command '{command}'. Available: pause, continue, terminate.")

    def _pace(self, state: dict, seconds: float = 0.5) -> None:
        """Wait for the given time while processing any commands typed by the user."""
        deadline = time.monotonic() + seconds
        while not state["terminated"]:
            command = self.reader.poll()
            if command is not None:
                self._apply_command(command, state)
                continue
            if time.monotonic() >= deadline:
                break
            time.sleep(0.05)

    def _run_round(self, lottery: LotteryManager, prize: Prize, round_no: int, total: int) -> list[User]:
        print(f"\n--- Round {round_no}/{total}: {prize.name} ---")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {lottery.describe_rules(prize)}")

        state = {"paused": False, "terminated": False}
        drawn: list[User] = []
        while len(drawn) < prize.quantity:
            print(f"Currently drawing the {ordinal(len(drawn) + 1)} winner...")
            self._pace(state)
            while state["paused"] and not state["terminated"]:
                self._apply_command(self.reader.readline(), state)
            if state["terminated"]:
                break
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize; the round ends.")
                break
            drawn.append(user)
            self.result_manager.add_winner(user, prize)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        lottery.complete_round(prize, drawn)
        print(f"--- Round {round_no} results ({len(drawn)}/{prize.quantity}) ---")
        if not drawn:
            print("  (no winners in this round)")
        for index, user in enumerate(drawn, start=1):
            print(f"  {index}. {user.name} ({user.department})")
        return drawn

    # ------------------------------------------------------------ results
    def show_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n=== Complete Winner List ===")
        for prize in prizes:
            winners = grouped.get(prize.name, [])
            print(f"{prize.name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(f"  {winner.user.name} ({winner.user.department}) - {prize.name}")

    def export_results(self) -> None:
        if not self.result_manager.get_all_winners():
            print("No lottery results to export.")
            return
        ok, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(message)

    cli = LotterySystemCLI()

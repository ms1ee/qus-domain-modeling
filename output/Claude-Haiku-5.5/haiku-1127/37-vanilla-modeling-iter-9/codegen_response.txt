"""Core domain models and shared constants."""

from __future__ import annotations

from dataclasses import dataclass, field

# Shared constants used by the rule engine.
CHENG_RU_GROUP = "Cheng Ru group"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"

_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3}
_CUSTOM_LEVEL = 99


@dataclass(frozen=True)
class User:
    """A lottery participant."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that can be drawn in a lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    def __post_init__(self) -> None:
        if isinstance(self.quantity, bool) or not isinstance(self.quantity, int):
            raise ValueError("Prize quantity must be an integer.")
        if self.quantity < 1:
            raise ValueError("Prize quantity must be at least 1.")
        self.avoid_groups = list(self.avoid_groups or [])

    @property
    def level(self) -> int:
        """Prize level derived from the name: First=1, Second=2, Third=3; others last."""
        words = self.name.strip().lower().split()
        if words and words[0] in _LEVEL_WORDS:
            return _LEVEL_WORDS[words[0]]
        return _CUSTOM_LEVEL

DEFAULT_LIST_TEXT = """\

Zhao Liu,Product Department
Sun Qi,Product Department
Zhou Ba,Product Department
Li Bin,Product Department
Wu Jiu,Cheng Ru group
Zheng Shi,Cheng Ru group
Feng Yi,Cheng Ru group
Ma Jiaqi,Cheng Ru group
Chen Er,Marketing Department
Chu San,Marketing Department
Wei Si,Human Resources Department
Shen Wu,Human Resources Department
Han Liu,Finance Department
Yang Qi,Finance Department
Chen Qixian,Administration Department
Jin Duo,Administration Department

class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (each line: 'Name,Department Group')."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File Error: '{path}' does not exist or is not a file."]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            content = raw.strip()
            if not content:
                continue
            parts = content.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_no}: Format Error - '{content}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid_users, duplicates); the first occurrence of a name is valid."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to number of members."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management and sequencing."""

from __future__ import annotations

from lottery_system.models import CHEN_QIXIAN, JIN_DUO, MA_JIAQI, Prize

DEFAULT_PRIZES: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 10, [], CHEN_QIXIAN),
    ("Second Prize - Shopping Cards", 5, [], JIN_DUO),
    ("First Prize - 1 Day Annual Leave", 2, [], MA_JIAQI),
]


class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"Preset prize '{name}' not loaded: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the sequence."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if any(existing.name == name for existing in self._prizes):
            raise ValueError(f"A prize named '{name}' already exists.")
        drawer = associated_drawer.strip() if associated_drawer else None
        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=list(avoid_groups or []),
            associated_drawer=drawer or None,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            if sorted(order) != list(range(1, count + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)


class LotteryManager:
    """Draws winners while enforcing the lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users = list(users)
        self._winner_names: set[str] = set()
        self._rng = rng or random.Random()
        # Whether each of the most recent draws by Chen Qixian / Jin Duo contained
        # at least one Cheng Ru group member (max length 2).
        self.last_two_draw_results: list[bool] = []

    def is_fallback_active(self, prize: Prize) -> bool:
        """Cheng Ru group protection applies to Ma Jiaqi when the last two results lacked Cheng Ru members."""
        return (
            prize.associated_drawer == MA_JIAQI
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = [user for user in self._users if user.name not in self._winner_names]
        if prize.associated_drawer == LI_BIN:
            # Special rule: only previous winners are filtered out.
            return available
        if self.is_fallback_active(prize):
            # Fallback: forcibly limit to Cheng Ru group members, even if avoided.
            return [user for user in available if user.department == CHENG_RU_GROUP]
        return [user for user in available if user.department not in prize.avoid_groups]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the current eligible pool."""
        return self._get_eligible_pool(prize)

    def count_eligible(self, prize: Prize) -> int:
        return len(self._get_eligible_pool(prize))

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_names.add(winner.name)
        return winner

    def end_round(self, prize: Prize, winners: list[User]) -> None:
        """Record the outcome of a finished round for fallback tracking."""
        if prize.associated_drawer in (CHEN_QIXIAN, JIN_DUO) and winners:
            has_cheng_ru = any(user.department == CHENG_RU_GROUP for user in winners)
            self.last_two_draw_results.append(has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

"""Winner recording, grouping and CSV-style export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize: Prize
    winning_order: int  # order of winning within the same prize (1-based)
    winning_time: datetime


class ResultManager:
    """Maintains the list of winners for the current lottery."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        order = sum(1 for winner in self._winners if winner.prize.name == prize.name) + 1
        winner = Winner(user=user, prize=prize, winning_order=order, winning_time=datetime.now())
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name; prizes ordered by level (high to low), then given order."""
        ordered = sorted(prizes, key=lambda prize: prize.level)  # stable sort
        result: dict[str, list[Winner]] = {}
        for prize in ordered:
            winners = [winner for winner in self._winners if winner.prize.name == prize.name]
            winners.sort(key=lambda winner: winner.winning_order)
            result[prize.name] = winners
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to awards_result_YYYYMMDD.txt in the current directory."""
        grouped = self.get_winners_by_prize(prizes)
        rows = [winner for winners in grouped.values() for winner in winners]
        if not rows:
            return False, "No winners to export."

        now = datetime.now()
        path = os.path.join(os.getcwd(), f"awards_result_{now:%Y%m%d}.txt")
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in rows:
                    writer.writerow([
                        winner.user.name,
                        winner.user.department,
                        winner.prize.name,
                        winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Command-line interaction for the team building lottery system."""

from __future__ import annotations

from lottery_system.models import CHEN_QIXIAN, JIN_DUO, LI_BIN, MA_JIAQI, CHENG_RU_GROUP, Prize, User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager

DRAWER_OPTIONS = [CHEN_QIXIAN, JIN_DUO, MA_JIAQI, LI_BIN]

MENU = """
========== Team Building Lottery ==========
 1. Load built-in default participant list
 2. Upload custom participant txt list
 3. View department groups
 4. Load preset prizes
 5. Add custom prize
 6. View prize sequence
 7. Reorder prize sequence
 8. Start lottery
 9. View winner summary
10. Export winner list
 0. Exit
"""


def _ordinal(number: int) -> str:
    if 11 <= number % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive console application."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.result_manager = ResultManager()
        self.lottery_manager: LotteryManager | None = None
        self.lottery_done = False

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str) -> str:
        return input(prompt).strip()

    def run(self) -> None:
        print("Welcome to the Team Building Smart Lottery System.")
        while True:
            print(MENU)
            choice = self._ask("Select an option: ")
            if choice == "0":
                print("Goodbye!")
                return
            handlers = {
                "1": self._load_default_list,
                "2": self._upload_custom_list,
                "3": self._view_groups,
                "4": self._load_preset_prizes,
                "5": self._add_custom_prize,
                "6": self._view_prizes,
                "7": self._reorder_prizes,
                "8": self._start_lottery,
                "9": self._show_summary,
                "10": self._export_winners,
            }
            handler = handlers.get(choice)
            if handler is None:
                print("Invalid option, please try again.")
                continue
            handler()

    # ------------------------------------------------------------ list handling
    def _load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        for error in errors:
            print(error)
        valid, duplicates = self.list_manager.validate_users(users)
        for user in duplicates:
            print(f"Duplicate ignored: {user.name} ({user.department})")
        self.users = valid
        print(f"Built-in list loaded: {len(self.users)} participants.")

    def _upload_custom_list(self) -> None:
        path = self._ask("Enter txt file path (leave empty to cancel): ")
        if not path:
            return
        users, errors = self.list_manager.load_custom_list(path)
        if errors:
            print(f"Found {len(errors)} problem(s):")
            for error in errors:
                print(f"  {error}")
            answer = self._ask("Remove invalid data and continue? (y/n): ").lower()
            if answer != "y":
                print("Upload terminated.")
                return
        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print("Duplicate names found:")
            for user in duplicates:
                print(f"  {user.name} ({user.department})")
            answer = self._ask("Remove duplicates (keep first occurrence) and continue? (y/n): ").lower()
            if answer != "y":
                print("Upload terminated.")
                return
        if not valid:
            print("No valid participants found. Upload failed.")
            return
        self.users = valid
        print(f"Custom list loaded: {len(self.users)} participants.")

    def _view_groups(self) -> None:
        if not self.users:
            print("No participant list loaded yet.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for group, count in info.items():
            print(f"  {group}: {count} people")
        print(f"Total: {len(self.users)} people in {len(info)} groups.")

    # ------------------------------------------------------------ prize handling
    def _load_preset_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for error in errors:
            print(error)
        print("Preset prizes loaded.")

    def _select_drawer(self) -> str | None:
        print("Associated lottery person:")
        print("  0. None")
        for index, drawer in enumerate(DRAWER_OPTIONS, 1):
            print(f"  {index}. {drawer}")
        raw = self._ask("Select by number or type a name (blank for none): ")
        if not raw or raw == "0":
            return None
        if raw.isdigit() and 1 <= int(raw) <= len(DRAWER_OPTIONS):
            return DRAWER_OPTIONS[int(raw) - 1]
        return raw

    def _select_avoid_groups(self) -> list[str]:
        groups = self.list_manager.get_group_info(self.users)
        if not groups:
            print("No department groups available (load a participant list first). No avoidance groups set.")
            return []
        names = list(groups.keys())
        print("Available department groups:")
        for index, name in enumerate(names, 1):
            print(f"  {index}. {name} ({groups[name]} people)")
        raw = self._ask("Select avoidance groups by number or name, comma separated (blank for none): ")
        selected: list[str] = []
        for token in [part.strip() for part in raw.split(",") if part.strip()]:
            if token.isdigit() and 1 <= int(token) <= len(names):
                group = names[int(token) - 1]
            else:
                matches = [name for name in names if name.lower() == token.lower()]
                if not matches:
                    print(f"Warning: unknown group '{token}' ignored.")
                    continue
                group = matches[0]
            if group not in selected:
                selected.append(group)
        return selected

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            print("Prize name cannot be empty.")
            return
        raw_quantity = self._ask("Quantity: ")
        if not raw_quantity.isdigit() or int(raw_quantity) < 1:
            print("Quantity must be a positive integer.")
            return
        avoid_groups = self._select_avoid_groups()
        drawer = self._select_drawer()
        try:
            self.prize_manager.add_prize(name, int(raw_quantity), avoid_groups, drawer)
        except ValueError as exc:
            print(f"Cannot add prize: {exc}")
            return
        print(f"Prize '{name}' added.")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Prize sequence:")
        for index, prize in enumerate(prizes, 1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
            drawer = prize.associated_drawer or "None"
            print(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"drawer: {drawer} | avoid groups: {avoid}"
            )

    def _reorder_prizes(self) -> None:
        self._view_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = self._ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3), blank to skip: ")
        if not raw:
            return
        try:
            order = [int(token) for token in raw.split()]
        except ValueError:
            print("Invalid input: please enter integers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
        else:
            print("Invalid order: every serial number must appear exactly once.")

    # ------------------------------------------------------------ lottery flow
    def _rule_description(self, prize: Prize) -> str:
        if prize.associated_drawer == LI_BIN:
            return "Previous winners are excluded (avoidance groups not applied)."
        if self.lottery_manager and self.lottery_manager.is_fallback_active(prize):
            return (
                f"Cheng Ru group protection active: only {CHENG_RU_GROUP} members "
                "are eligible; previous winners are excluded."
            )
        description = "Previous winners are excluded"
        if prize.avoid_groups:
            description += f"; avoided groups: {', '.join(prize.avoid_groups)}"
        return description + "."

    def _confirm_next(self, order: int) -> bool:
        """Handle pause/continue/terminate. Returns False if the current round should end."""
        while True:
            command = self._ask(f"Press Enter to draw winner #{order} (or 'pause' / 'terminate'): ").lower()
            if command == "":
                return True
            if command == "terminate":
                return False
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    control = self._ask("(paused) > ").lower()
                    if control == "continue":
                        return True
                    if control == "terminate":
                        return False
                    print("Unknown command. Use 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    def _run_round(self, index: int, total: int, prize: Prize) -> None:
        assert self.lottery_manager is not None
        print("=" * 60)
        print(f"Round {index}/{total}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {self._rule_description(prize)}")
        print(f"Eligible participants: {self.lottery_manager.count_eligible(prize)}")

        winners: list[User] = []
        for order in range(1, prize.quantity + 1):
            if not self._confirm_next(order):
                print("Round terminated; remaining winners for this prize are skipped.")
                break
            print(f"Currently drawing the {_ordinal(order)} winner...")
            user = self.lottery_manager.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize.")
                break
            self.result_manager.add_winner(user, prize)
            winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self.lottery_manager.end_round(prize, winners)
        print(f"Round finished: {len(winners)}/{prize.quantity} winner(s) drawn.")
        if winners:
            print("Round results:")
            for position, user in enumerate(winners, 1):
                print(f"  {position}. {user.name} ({user.department})")

    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Load preset prizes or add custom prizes first.")
            return
        if self.lottery_done:
            answer = self._ask("A lottery has already run. Start over and discard previous results? (y/n): ").lower()
            if answer != "y":
                return
            self.result_manager = ResultManager()
        self.lottery_manager = LotteryManager(self.users)
        for index, prize in enumerate(prizes, 1):
            self._run_round(index, len(prizes), prize)
        self.lottery_done = True
        print("=" * 60)
        print("Lottery process completed.")
        self._show_summary()

    def _show_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        total = sum(len(winners) for winners in grouped.values())
        if total == 0:
            print("No winners yet.")
            return
        print("Winner summary (by prize level, then winning order):")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            print(f"[{prize_name}]")
            for winner in winners:
                print(
                    f"  {winner.winning_order}. {winner.user.name} "
                    f"({winner.user.department}) - {winner.prize.name}"
                )
        print(f"Total winners: {total}")

    def _export_winners(self) -> None:
        prizes = self.prize_manager.get_prizes()
        success, message = self.result_manager.export_to_csv(prizes)
        print(message)


def main() -> int:
    try:
        LotterySystemCLI().run()
    except (KeyboardInterrupt, EOFError):
        print("\nExiting the lottery system.")
    return 0

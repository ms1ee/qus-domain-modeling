from __future__ import annotations

from dataclasses import dataclass, field
from typing import Optional

CHENG_RU_GROUP = "Cheng Ru Group"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"


def normalize_group(value: str) -> str:
    """Normalize a name or department for case/whitespace-insensitive comparison."""
    return " ".join(str(value).split()).casefold()


def same_text(a: Optional[str], b: Optional[str]) -> bool:
    """Return True when two optional strings match after normalization."""
    if a is None or b is None:
        return False
    return normalize_group(a) == normalize_group(b)


def is_cheng_ru_group(department: str) -> bool:
    """Return True if the department is the Cheng Ru group."""
    return same_text(department, CHENG_RU_GROUP)

    name: str
    department: str

    """

    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

"""Participant list loading, validation and group extraction."""

from __future__ import annotations

import os

from lottery_system.defaults import DEFAULT_USER_LINES
from lottery_system.models import User


class ListManager:
    """Loads participant lists from the built-in default or a custom txt file."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a txt file. Returns (valid users, error messages)."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        if not str(path).lower().endswith(".txt"):
            return [], ["File Error: only .txt files are supported."]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines are ignored."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid users, duplicates).

        The first occurrence of each name is valid; later occurrences
        with the same name are reported as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group -> number of members (in order of appearance)."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management and round ordering."""

from __future__ import annotations

from typing import Optional

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for a lottery session."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages (empty on success)."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[list[str]] = None,
        associated_drawer: Optional[str] = None,
    ) -> Prize:
        """Add a prize at the end of the current sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery round order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order.

        Returns True on success; False if `order` is not a permutation of 1..n.
        """
        n = len(self._prizes)
        if len(order) != n:
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random
from typing import Optional

from lottery_system.models import (
    CHEN_QIXIAN,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    is_cheng_ru_group,
    normalize_group,
    same_text,
)

# Lottery persons whose draw outcomes feed the Cheng Ru group protection check.
FALLBACK_TRIGGER_DRAWERS = (CHEN_QIXIAN, JIN_DUO)
# Lottery person whose prize pool is forced to Cheng Ru group when fallback is active.
FALLBACK_TARGET_DRAWER = MA_JIAQI


class LotteryManager:
    """Maintains winners across rounds and applies the lottery rules."""

    def __init__(self, users: list[User], rng: Optional[random.Random] = None) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        self._winners: list[User] = []
        self._rng = rng if rng is not None else random.Random()
        # Outcome per completed round of Chen Qixian / Jin Duo prizes:
        # True if that round produced at least one Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []

    @property
    def winners(self) -> list[User]:
        """All winners drawn so far, in drawing order."""
        return list(self._winners)

    def is_fallback_active(self) -> bool:
        """Fallback is active when the last two Chen Qixian/Jin Duo results both had no Cheng Ru member."""
        return len(self.last_two_draw_results) >= 2 and not any(self.last_two_draw_results[-2:])

    def _is_fallback_for(self, prize: Prize) -> bool:
        return same_text(prize.associated_drawer, FALLBACK_TARGET_DRAWER) and self.is_fallback_active()

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the eligible pool for a prize under the rule engine."""
        base = [user for user in self._users if user.name not in self._winner_names]

        # Special rule: Li Bin only filters previous winners.
        if same_text(prize.associated_drawer, LI_BIN):
            return base

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group (overrides avoidance groups).
        if self._is_fallback_for(prize):
            return [user for user in base if is_cheng_ru_group(user.department)]

        # Standard rule: exclude previous winners and avoidance groups.
        avoided = {normalize_group(group) for group in prize.avoid_groups}
        return [user for user in base if normalize_group(user.department) not in avoided]

    def eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rule applied to a prize round."""
        drawer = prize.associated_drawer
        if same_text(drawer, LI_BIN):
            return (
                "Only previous winners are excluded; avoidance groups are not applied "
                "(lottery person: Li Bin)."
            )
        if self._is_fallback_for(prize):
            return (
                "Cheng Ru Group protection active: only Cheng Ru Group members are eligible "
                "(avoidance groups overridden)."
            )
        parts = ["Previous winners are excluded"]
        if prize.avoid_groups:
            parts.append("avoided groups: " + ", ".join(prize.avoid_groups))
        if drawer:
            parts.append(f"lottery person: {drawer}")
        return "; ".join(parts) + "."

    def draw_one(self, prize: Prize) -> Optional[User]:
        """Draw one winner for the prize. Returns None if no eligible participant exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_names.add(winner.name)
        self._winners.append(winner)
        return winner

    def finish_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Record the outcome of a completed round for the fallback mechanism.

        Only rounds of Chen Qixian / Jin Duo that produced at least one winner are recorded.
        """
        if not round_winners:
            return
        drawer = prize.associated_drawer
        if any(same_text(drawer, name) for name in FALLBACK_TRIGGER_DRAWERS):
            hit = any(is_cheng_ru_group(user.department) for user in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

"""Winner recording, grouping and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    winning_order: int
    win_time: datetime = field(default_factory=datetime.now)


class ResultManager:
    """Stores winners and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for a prize."""
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        winner = Winner(
            user=user,
            prize_name=prize_name,
            winning_order=len(self._winners) + 1,
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, following the given prize order."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for prize in prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

from __future__ import annotations

from typing import Callable, Optional

from lottery_system.models import Prize, User

def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def _parse_ints(text: str) -> Optional[list[int]]:
    parts = [part.strip() for part in text.split(",") if part.strip()]
    try:
        return [int(part) for part in parts]
    except ValueError:
        return None


class LotteryCLI:
    """Interactive command-line controller for the lottery system."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.results = ResultManager()
        self.users: list[User] = []
        self.list_loaded = False
        self.lottery: Optional[LotteryManager] = None

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str = "") -> str:
        return self._input(prompt).strip()

    def _out(self, text: str = "") -> None:
        self._print(text)

    # ---------------------------------------------------------------- main loop
    def run(self) -> None:
        self._out("Welcome to the Team Building Smart Lottery System.")
        try:
            while True:
                self._out(MAIN_MENU)
                choice = self._ask("Select an option: ")
                if choice == "1":
                    self.load_list()
                elif choice == "2":
                    self.show_groups()
                elif choice == "3":
                    self.configure_prizes()
                elif choice == "4":
                    self.reorder_prizes()
                elif choice == "5":
                    self.start_lottery()
                elif choice == "6":
                    self.show_results_and_export()
                elif choice == "0":
                    self._out("Goodbye!")
                    return
                else:
                    self._out("Invalid option, please try again.")
        except (EOFError, KeyboardInterrupt):
            self._out("\nExiting. Goodbye!")

    # ------------------------------------------------------------ list module
    def load_list(self) -> None:
        self._out("1. Built-in default team building list")
        self._out("2. Custom txt list (each line: Name,Department Group)")
        mode = self._ask("Choose load mode: ")
        if mode == "1":
            users, errors = self.list_manager.load_default_list()
        elif mode == "2":
            path = self._ask("Enter the txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            self._out("Invalid choice.")
            return

        for error in errors:
            self._out(error)

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self._out("Duplicate names detected (later entries are duplicates):")
            for user in duplicates:
                self._out(f"  - {user.name} ({user.department})")

        if errors or duplicates:
            self._out("Invalid data detected. Options:")
            self._out("  [R] Remove invalid data and continue")
            self._out("  [T] Terminate upload")
            while True:
                action = self._ask("Choose R or T: ").lower()
                if action == "r":
                    break
                if action == "t":
                    self._out("Upload terminated. Participant list unchanged.")
                    return
                self._out("Please enter R or T.")

        if not valid:
            self._out("No valid participants loaded.")
            return

        self.users = valid
        self.list_loaded = True
        self.lottery = None
        self._out(f"Loaded {len(valid)} valid participant(s).")

    def show_groups(self) -> None:
        if not self.list_loaded:
            self._out("No participant list loaded yet.")
            return
        groups = self.list_manager.get_group_info(self.users)
        self._out("Department groups:")
        for index, (group, count) in enumerate(groups.items(), start=1):
            self._out(f"  {index}. {group}: {count} member(s)")

    # ----------------------------------------------------------- prize module
    def configure_prizes(self) -> None:
        while True:
            self._out(PRIZE_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        self._out(f"Skipped: {error}")
                else:
                    self._out("Preset prizes loaded.")
            elif choice == "2":
                self.add_custom_prize()
            elif choice == "3":
                self.show_prizes()
            elif choice == "0":
                return
            else:
                self._out("Invalid option.")

    def add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity_text = self._ask("Prize quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._out("Quantity must be an integer.")
            return

        avoid_groups = self._select_avoid_groups()
        if avoid_groups is None:
            return
        drawer = self._ask("Associated lottery person (blank for none): ")

        try:
            prize = self.prize_manager.add_prize(name, quantity, avoid_groups, drawer or None)
        except ValueError as exc:
            self._out(f"Cannot add prize: {exc}")
            return
        self._out(f"Prize added: {prize.name} (quantity {prize.quantity}).")

    def _select_avoid_groups(self) -> Optional[list[str]]:
        if not self.list_loaded:
            self._out("No participant list loaded; avoidance groups cannot be selected.")
            return []
        groups = list(self.list_manager.get_group_info(self.users).keys())
        self._out("Select avoidance groups (blank for none):")
        for index, group in enumerate(groups, start=1):
            self._out(f"  {index}. {group}")
        while True:
            text = self._ask("Enter group numbers separated by commas: ")
            if not text:
                return []
            numbers = _parse_ints(text)
            if numbers is None or any(n < 1 or n > len(groups) for n in numbers):
                self._out("Invalid selection, please try again.")
                continue
            return [groups[n - 1] for n in dict.fromkeys(numbers)]

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._out("No prizes configured.")
            return
        self._out("Prize sequence:")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self._out(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"avoid groups: {avoid} | lottery person: {drawer}"
            )

    def reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._out("No prizes configured.")
            return
        self.show_prizes()
        text = self._ask("Enter new order as serial numbers separated by commas (e.g. 2,1,3): ")
        order = _parse_ints(text)
        if order is None or not self.prize_manager.reorder_prizes(order):
            self._out("Invalid order. The sequence was not changed.")
            return
        self._out("Prize sequence updated.")
        self.show_prizes()

    # ------------------------------------------------------- lottery process
    def start_lottery(self) -> None:
        if not self.list_loaded:
            self._out("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._out("Please configure at least one prize first.")
            return

        self.lottery = LotteryManager(self.users)
        self.results = ResultManager()
        self._out("Lottery started. Commands: pause / continue / terminate")

        for number, prize in enumerate(prizes, start=1):
            self._run_round(number, prize)

        self._out("\n========== Lottery Finished ==========")
        self._show_summary(prizes)

    def _run_round(self, number: int, prize: Prize) -> None:
        lottery = self.lottery
        assert lottery is not None
        self._out(f"\n--- Round {number}: {prize.name} ---")
        self._out(f"Quantity: {prize.quantity}")
        self._out(f"Rule: {lottery.describe_rule(prize)}")

        pool_size = len(lottery.eligible_pool(prize))
        if pool_size == 0:
            self._out("No eligible participants for this prize. Round skipped.")
            lottery.finish_round(prize, [])
            return
        if pool_size < prize.quantity:
            self._out(
                f"Note: only {pool_size} eligible participant(s); fewer winners will be drawn."
            )

        round_winners: list[User] = []
        for position in range(1, prize.quantity + 1):
            if self._control() == "terminate":
                self._out("Round terminated by operator; remaining winners skipped.")
                break
            self._out(f"Currently drawing the {_ordinal(position)} winner...")
            user = lottery.draw_one(prize)
            if user is None:
                self._out("No eligible participants left. Round ends early.")
                break
            round_winners.append(user)
            self.results.add_winner(user, prize)
            self._out(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        lottery.finish_round(prize, round_winners)

        self._out(f"Round {number} results (winning order):")
        if not round_winners:
            self._out("  (no winners)")
        for index, user in enumerate(round_winners, start=1):
            self._out(f"  {index}. {user.name} ({user.department})")

    def _control(self) -> str:
        """Process operator commands before each draw. Returns 'next' or 'terminate'."""
        while True:
            command = self._ask("[Enter]=draw next | pause | terminate: ").lower()
            if command in ("", "continue"):
                return "next"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                self._out("Lottery PAUSED. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    command = self._ask("(paused) > ").lower()
                    if command == "continue":
                        self._out("Lottery resumed.")
                        return "next"
                    if command == "terminate":
                        return "terminate"
                    self._out("Type 'continue' or 'terminate'.")
            self._out("Unknown command. Use Enter, pause, or terminate.")

    # ------------------------------------------------------- results module
    def _show_summary(self, prizes: list[Prize]) -> None:
        grouped = self.results.get_winners_by_prize(prizes)
        total = 0
        self._out("Complete winner list:")
        self._out(f"  {'Name':<16}| {'Department':<24}| Prize")
        for prize in prizes:
            for winner in grouped.get(prize.name, []):
                total += 1
                self._out(
                    f"  {winner.user.name:<16}| {winner.user.department:<24}| {winner.prize_name}"
                )
        if total == 0:
            self._out("  (no winners)")
        self._out(f"Total winners: {total}")

    def show_results_and_export(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not self.results.get_all_winners():
            self._out("No lottery results yet.")
            return
        self._show_summary(prizes)
        if self._ask("Export results to txt/CSV file? (y/n): ").lower() == "y":
            _, message = self.results.export_to_csv(prizes)
            self._out(message)


def main() -> None:
    """Start the interactive lottery system."""
    LotteryCLI().run()

    to the 8x8 grid.

<file path="src/path_planner.py">
"""Grid-based path planning restricted to the street network.

Streets are the grid lines where ``x`` or ``y`` is a multiple of 100.
Restaurants and customers sit on street intersections (both coordinates are
multiples of 100). Routes are built from axis-aligned segments only, so every
movement is either straight or a right-angle turn.
"""

import math
import random
from typing import Dict, Iterable, List, Optional, Sequence, Tuple

from config import CELL_SIZE, MAP_SIZE

Point = Tuple[int, int]


class PathPlanner:
    """Compute street-constrained routes between map points.

    The planner is stateless apart from the map geometry; congestion is
    supplied per call through an ``occupancy`` mapping so that riders can
    avoid crowded corners (conflict avoidance without blocking).
    """

    def __init__(self, map_size: int = MAP_SIZE, cell_size: int = CELL_SIZE) -> None:
        self.map_size = map_size
        self.cell_size = cell_size

    # ------------------------------------------------------------------
    # Geometry helpers
    # ------------------------------------------------------------------
    def is_intersection(self, point: Point) -> bool:
        """Return True if both coordinates are multiples of the cell size."""
        x, y = point
        return x % self.cell_size == 0 and y % self.cell_size == 0

    def is_on_road(self, point: Point) -> bool:
        """Return True if the point lies on a street (grid line)."""
        x, y = point
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def in_bounds(self, point: Point) -> bool:
        """Return True if the point lies inside the map."""
        x, y = point
        return 0 <= x <= self.map_size and 0 <= y <= self.map_size

    @staticmethod
    def manhattan(a: Point, b: Point) -> int:
        """Return the Manhattan distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    def _street_neighbours(self, point: Point) -> List[Point]:
        """Return the two intersections bounding a point lying on a street."""
        x, y = point
        cs = self.cell_size
        if x % cs == 0 and y % cs != 0:
            low = (y // cs) * cs
            high = min(low + cs, self.map_size)
            return [(x, low), (x, high)]
        if y % cs == 0 and x % cs != 0:
            low = (x // cs) * cs
            high = min(low + cs, self.map_size)
            return [(low, y), (high, y)]
        return []

    @staticmethod
    def _dedupe(start: Point, points: Iterable[Point]) -> List[Point]:
        """Remove consecutive duplicate waypoints."""
        result: List[Point] = []
        previous = start
        for point in points:
            if point != previous:
                result.append(point)
                previous = point
        return result

    # ------------------------------------------------------------------
    # Route computation
    # ------------------------------------------------------------------
    def plan(
        self,
        start: Sequence[int],
        goal: Sequence[int],
        occupancy: Optional[Dict[Point, int]] = None,
    ) -> List[Point]:
        """Plan a street-constrained route from ``start`` to ``goal``.

        Special cases handled:
          * Start equals goal: empty route.
          * Goal directly above/below/left/right of the start: a single
            straight segment along the street.
          * Start not on an intersection: the rider first moves along its
            street to the nearest reachable intersection.
          * Otherwise an L-shaped route with a right-angle turn is produced;
            the corner with the lower congestion count is preferred.

        Args:
            start: Current rider position (must lie on a street).
            goal: Target position (must be a street intersection).
            occupancy: Optional mapping of point -> number of riders there.

        Returns:
            List of waypoints after ``start`` ending with ``goal``. The list
            is empty if no movement is required.

        Raises:
            ValueError: If the start is off-street or the goal is not an
                intersection.
        """
        start = (int(start[0]), int(start[1]))
        goal = (int(goal[0]), int(goal[1]))
        if not self.is_on_road(start):
            raise ValueError(f"start {start} is not on a street")
        if not self.is_intersection(goal):
            raise ValueError(f"goal {goal} is not a street intersection")
        if start == goal:
            return []
        if start[0] == goal[0] or start[1] == goal[1]:
            # Special case: directly above/below or beside the target.
            return [goal]

        waypoints: List[Point] = []
        origin = start
        if not self.is_intersection(start):
            candidates = self._street_neighbours(start)
            origin = min(candidates, key=lambda p: self.manhattan(p, goal))
            waypoints.append(origin)
            if origin[0] == goal[0] or origin[1] == goal[1]:
                waypoints.append(goal)
                return self._dedupe(start, waypoints)

        occupancy = occupancy or {}
        corner_h = (goal[0], origin[1])   # horizontal first
        corner_v = (origin[0], goal[1])   # vertical first
        if occupancy.get(corner_v, 0) < occupancy.get(corner_h, 0):
            corner = corner_v
        else:
            corner = corner_h
        waypoints.extend([corner, goal])
        return self._dedupe(start, waypoints)

    def route_length(self, start: Sequence[int], waypoints: Iterable[Sequence[int]]) -> int:
        """Return the total Manhattan length of a waypoint chain.

        Args:
            start: Starting point of the chain.
            waypoints: Ordered waypoints following the start.

        Returns:
            Sum of segment lengths in map units.
        """
        total = 0
        current = (int(start[0]), int(start[1]))
        for point in waypoints:
            point = (int(point[0]), int(point[1]))
            total += self.manhattan(current, point)
            current = point
        return total

    def dense_path(
        self,
        start: Sequence[int],
        waypoints: Iterable[Sequence[int]],
        step: int = 25,
    ) -> List[Point]:
        """Expand a waypoint chain into evenly spaced points for drawing.

        Args:
            start: Starting point.
            waypoints: Ordered waypoints.
            step: Maximum distance between consecutive sampled points.

        Returns:
            List of integer points beginning with ``start``.
        """
        current = (int(start[0]), int(start[1]))
        points: List[Point] = [current]
        for waypoint in waypoints:
            target = (int(waypoint[0]), int(waypoint[1]))
            distance = self.manhattan(current, target)
            count = max(1, math.ceil(distance / step))
            for i in range(1, count + 1):
                t = i / count
                points.append((
                    round(current[0] + (target[0] - current[0]) * t),
                    round(current[1] + (target[1] - current[1]) * t),
                ))
            current = target
        return points


def random_road_point(rng: Optional[random.Random] = None) -> Point:
    """Return a random integer point located on a street.

    Args:
        rng: Optional random generator (defaults to the ``random`` module).

    Returns:
        ``(x, y)`` where one coordinate is a street line (multiple of 100)
        and the other is a random integer within the map.
    """
    rng = rng or random
    line = rng.randrange(0, MAP_SIZE // CELL_SIZE + 1) * CELL_SIZE
    other = rng.randint(0, MAP_SIZE)
    if rng.random() < 0.5:
        return line, other
    return other, line

"""Order data model: coordinate validation, snapping and lifecycle."""

import itertools
from typing import Any, Dict, List, Optional, Tuple

from config import CELL_SIZE, MAP_SIZE, is_valid_coordinate
from path_planner import PathPlanner

Point = Tuple[int, int]

_ORDER_IDS = itertools.count(1)


def snap_coordinate(value: int) -> int:
    """Snap a single coordinate to the nearest street line (multiple of 100).

    Restaurant and customer locations are placed on street intersections,
    which matches the rule that integer hundreds denote these locations.
    The result is clamped to the map range.

    Args:
        value: Integer coordinate in ``[0, MAP_SIZE]``.

    Returns:
        The snapped coordinate.
    """
    snapped = int(round(value / CELL_SIZE)) * CELL_SIZE
    return min(max(snapped, 0), MAP_SIZE)


def snap_point(point: Point) -> Point:
    """Snap both coordinates of a point to the nearest intersection."""
    return snap_coordinate(point[0]), snap_coordinate(point[1])


def _validate_point(point: Any, label: str) -> Point:
    """Validate that ``point`` is a pair of in-range integers.

    Raises:
        ValueError: If the structure or the values are invalid.
    """
    try:
        x, y = point
    except (TypeError, ValueError):
        raise ValueError(f"{label} must be a pair of integers")
    for value in (x, y):
        if not is_valid_coordinate(value):
            raise ValueError(f"{label} coordinate {value} is outside 0-{MAP_SIZE}")
    return x, y


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique, auto-incremented order identifier.
        raw_input: The original input ``[rx, ry, cx, cy]``.
        res_pos: Snapped restaurant location ``(x, y)``.
        cus_pos: Snapped customer location ``(x, y)``.
        status: One of ``created``, ``assigned``, ``picking_up``,
            ``delivering``, ``completed`` or ``rejected``.
        rider_id: Identifier of the assigned rider, or None.
        created_tick: Simulated time when the order was created.
        completed_tick: Simulated time of delivery, or None.
        distance: Route length from restaurant to customer.
        time_taken: Ticks from creation to completion, or None.
        efficiency: Distance per tick for the completed order, or None.
    """

    def __init__(self, res_pos: Any, cus_pos: Any, created_tick: int = 0) -> None:
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: ``(x, y)`` restaurant coordinates in 0-800.
            cus_pos: ``(x, y)`` customer coordinates in 0-800.
            created_tick: Simulated time at creation.

        Raises:
            ValueError: If coordinates are malformed or out of range.
        """
        raw_res = _validate_point(res_pos, "restaurant")
        raw_cus = _validate_point(cus_pos, "customer")
        self.id = next(_ORDER_IDS)
        self.raw_input: List[int] = [raw_res[0], raw_res[1], raw_cus[0], raw_cus[1]]
        self.res_pos: Point = snap_point(raw_res)
        self.cus_pos: Point = snap_point(raw_cus)
        self.status = "created"
        self.rider_id: Optional[int] = None
        self.created_tick = created_tick
        self.pickup_started_tick: Optional[int] = None
        self.delivery_started_tick: Optional[int] = None
        self.completed_tick: Optional[int] = None
        self.distance = PathPlanner().route_length(
            self.res_pos,
            PathPlanner().plan(self.res_pos, self.cus_pos),
        )
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None

    def as_list(self) -> List[int]:
        """Return the snapped order as ``[rx, ry, cx, cy]``."""
        return [self.res_pos[0], self.res_pos[1], self.cus_pos[0], self.cus_pos[1]]

    def assign_to(self, rider_id: int) -> None:
        """Assign this order to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = "assigned"

    def mark_picking_up(self, tick: int) -> None:
        """Record that the rider has started travelling to the restaurant."""
        self.status = "picking_up"
        self.pickup_started_tick = tick

    def mark_delivering(self, tick: int) -> None:
        """Record that the food has been collected and delivery has started."""
        self.status = "delivering"
        self.delivery_started_tick = tick

    def complete(self, tick: int) -> None:
        """Mark the order as delivered and compute efficiency metrics.

        Args:
            tick: Simulated time of delivery.
        """
        self.status = "completed"
        self.completed_tick = tick
        self.time_taken = max(tick - self.created_tick, 0)
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = float(self.distance)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_tick": self.created_tick,
            "completed_tick": self.completed_tick,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

    def __repr__(self) -> str:
        return (f"Order(id={self.id}, res={self.res_pos}, cus={self.cus_pos}, "
                f"status={self.status})")

"""Rider model: state machine, street-constrained movement and event log."""

import logging
import random
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from config import (
    STATE_DELIVERING,
    STATE_IDLE,
    STATE_NAMES,
    STATE_PICKING_UP,
    param,
)
from path_planner import PathPlanner, random_road_point

logger = logging.getLogger("dispatch.rider")

EVENT_SPAWN = "spawn"
EVENT_ACCEPTED = "accepted_order"
EVENT_ARRIVED_MERCHANT = "arrived_at_merchant"
EVENT_DELIVERED = "delivered_to_customer"

Point = Tuple[int, int]


def _current_tick() -> int:
    """Return the simulated time stored in the global variable store."""
    return globalvar.get_value("tick", 0)


def occupancy_map(exclude: Optional["Rider"] = None) -> Dict[Point, int]:
    """Count riders at each map point, optionally excluding one rider.

    Args:
        exclude: Rider to ignore (typically the rider being planned).

    Returns:
        Mapping of ``(x, y)`` to the number of riders located there.
    """
    counts: Dict[Point, int] = {}
    for rider in globalvar.get_value("riders", []) or []:
        if rider is exclude:
            continue
        key = (rider.x, rider.y)
        counts[key] = counts.get(key, 0) + 1
    return counts


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current integer coordinates on the street network.
        speed: Movement budget per tick, in map units.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders; the first element is the active order.
        special_case: True when the current target lies directly above,
            below, or beside the rider on the same street.
        path: Remaining waypoints for the current leg.
        log: Chronological list of event dictionaries. Each entry contains
            at least ``event`` and ``order_id``.
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at a random street position in idle state.

        Args:
            rider_id: Unique identifier assigned by the system manager.
        """
        self.rider_id = rider_id
        self.x, self.y = random_road_point()
        self.speed = random.randint(param("speed_min"), param("speed_max"))
        self.state = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.path: List[Point] = []
        self.log: List[Dict[str, Any]] = []
        self._planner = PathPlanner()
        self._record(EVENT_SPAWN, None)
        logger.info("Rider %s spawned at (%s, %s) speed %s",
                    rider_id, self.x, self.y, self.speed)

    # ------------------------------------------------------------------
    # Properties
    # ------------------------------------------------------------------
    @property
    def position(self) -> Point:
        """Current ``(x, y)`` position."""
        return self.x, self.y

    @property
    def state_name(self) -> str:
        """Human-readable state label."""
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def current_order(self):
        """The order currently being served, or None."""
        return self.orders[0] if self.orders else None

    # ------------------------------------------------------------------
    # Logging
    # ------------------------------------------------------------------
    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log."""
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": _current_tick(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.info("t=%s rider=%s event=%s order=%s pos=(%s,%s)",
                    entry["tick"], self.rider_id, event, order_id, self.x, self.y)

    # ------------------------------------------------------------------
    # Public behaviour
    # ------------------------------------------------------------------
    def move(self) -> None:
        """Advance the rider by one tick.

        Handles state transitions:
          * Idle -> Picking up when an assigned order exists (``accepted_order``).
          * Picking up -> Delivering on reaching the restaurant
            (``arrived_at_merchant``).
          * Delivering -> Idle on reaching the customer
            (``delivered_to_customer``).
        """
        if self.state == STATE_IDLE:
            if not self.orders:
                return
            self._start_pickup(self.orders[0])
        elif not self.orders:
            logger.warning("Rider %s was active without orders; resetting",
                           self.rider_id)
            self._reset_idle()
            return

        if self.path:
            self._advance()
        if not self.path:
            self._handle_arrival()

    def remaining_distance(self) -> int:
        """Estimate the distance still to travel for all assigned orders.

        Returns:
            Total Manhattan distance in map units.
        """
        if not self.orders:
            return 0
        total = 0
        current = self.position
        first = self.orders[0]
        if self.state == STATE_PICKING_UP:
            total += self._planner.route_length(current, self.path)
            current = first.res_pos
            stops = [first.cus_pos]
        elif self.state == STATE_DELIVERING:
            total += self._planner.route_length(current, self.path)
            current = first.cus_pos
            stops = []
        else:
            stops = [first.res_pos, first.cus_pos]
        for order in self.orders[1:]:
            stops.extend([order.res_pos, order.cus_pos])
        for stop in stops:
            total += self._planner.manhattan(current, stop)
            current = stop
        return total

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [order.id for order in self.orders],
        }

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _start_pickup(self, order) -> None:
        """Transition from idle to picking up for ``order``."""
        self.state = STATE_PICKING_UP
        order.mark_picking_up(_current_tick())
        self._record(EVENT_ACCEPTED, order.id)
        self._set_path(order.res_pos)

    def _handle_arrival(self) -> None:
        """Process arrival at the current target and transition state."""
        if not self.orders:
            self._reset_idle()
            return
        order = self.orders[0]
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.mark_delivering(_current_tick())
            self._record(EVENT_ARRIVED_MERCHANT, order.id)
            self._set_path(order.cus_pos)
        elif self.state == STATE_DELIVERING:
            order.complete(_current_tick())
            self.orders.pop(0)
            self._reset_idle()
            globalvar.set_value(
                "funds", globalvar.get_value("funds", 0) + param("order_revenue"))
            self._record(EVENT_DELIVERED, order.id)
            logger.info("Order %s completed by rider %s", order.id, self.rider_id)

    def _set_path(self, target: Point) -> None:
        """Plan a route to ``target`` avoiding congested corners."""
        self.path = self._planner.plan(self.position, target, occupancy_map(exclude=self))
        self.special_case = self.x == target[0] or self.y == target[1]

    def _reset_idle(self) -> None:
        """Return the rider to the idle state and clear the route."""
        self.state = STATE_IDLE
        self.path = []
        self.special_case = False

    def _advance(self) -> None:
        """Move along the planned waypoints using up to ``speed`` units."""
        budget = self.speed
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            dx = tx - self.x
            dy = ty - self.y
            if dx == 0 and dy == 0:
                self.path.pop(0)
                continue
            distance = abs(dx) + abs(dy)
            if distance <= budget:
                self.x, self.y = tx, ty
                budget -= distance
                self.path.pop(0)
            else:
                step_x = min(abs(dx), budget)
                self.x += step_x if dx > 0 else -step_x
                budget -= step_x
                step_y = min(abs(dy), budget)
                self.y += step_y if dy > 0 else -step_y
                budget -= step_y

"""Order management: creation, rider selection, recruitment and assignment."""

def find_best_rider_for_order(order: Order):
    """Select the most suitable rider for ``order``.

    The score combines the street distance from the rider to the restaurant
    with a load penalty for each order the rider already holds. This favours
    nearby riders (distance-first) while balancing the load across riders.
    Ties are resolved by the lower rider ID. If no riders exist, one is
    recruited when funds allow.

    Args:
        order: The order to be assigned.

    Returns:
        The chosen ``Rider``.

    Raises:
        RuntimeError: If no rider exists and none can be recruited.
    """
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        recruited = get_system().recruit_rider()
        if recruited is None:
            raise RuntimeError("no rider available and recruitment is not possible")
        riders = globalvar.get_value("riders", [])

    planner = PathPlanner()
    best = None
    best_score = None
    for rider in riders:
        to_restaurant = planner.plan(rider.position, order.res_pos,
                                     occupancy_map(exclude=rider))
        pickup_distance = planner.route_length(rider.position, to_restaurant)
        score = pickup_distance + param("load_penalty") * len(rider.orders)
        if (best is None or score < best_score
                or (score == best_score and rider.rider_id < best.rider_id)):
            best, best_score = rider, score
    logger.info("Order %s best rider %s (score %s)", order.id, best.rider_id, best_score)
    return best

"""System manager: funds, rider roster, simulated clock and consistency checks."""

import logging
from typing import Any, Dict, List, Optional

import globalvar
from config import (
    CELL_SIZE,
    DEFAULT_PARAMS,
    GRID_DIM,
    STATE_IDLE,
    cell_center,
    param,
)
from rider import Rider

logger = logging.getLogger("dispatch.system")


def build_grid() -> List[List[tuple]]:
    """Build the 8x8 grid map with each cell's centre point.

    Returns:
        A list of 8 rows; each row contains 8 ``(center_x, center_y)`` tuples.
    """
    return [[cell_center(col * CELL_SIZE, row * CELL_SIZE)
             for col in range(GRID_DIM)] for row in range(GRID_DIM)]


class SystemManager:
    """Owns system-wide state: funds, riders, clock and run flag.

    All state lives in the ``globalvar`` store under the keys ``funds``,
    ``riders``, ``rider_counter``, ``running``, ``tick``, ``orders`` and
    ``grid``.
    """

    def __init__(self, params: Optional[Dict[str, int]] = None) -> None:
        """Initialise the global state.

        Args:
            params: Optional parameter overrides merged over the defaults.
        """
        if params:
            globalvar.set_value("params", dict(params))
        globalvar.set_value("funds", param("initial_funds"))
        globalvar.set_value("riders", [])
        globalvar.set_value("rider_counter", 0)
        globalvar.set_value("running", True)
        globalvar.set_value("tick", 0)
        globalvar.set_value("orders", [])
        globalvar.set_value("grid", build_grid())
        logger.info("System initialised with funds %s", self.funds)

    # ------------------------------------------------------------------
    # Accessors
    # ------------------------------------------------------------------
    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return globalvar.get_value("funds", 0)

    @property
    def riders(self) -> List[Rider]:
        """The rider roster (live list)."""
        return globalvar.get_value("riders", [])

    @property
    def tick_count(self) -> int:
        """Current simulated time."""
        return globalvar.get_value("tick", 0)

    def add_funds(self, amount: int) -> None:
        """Adjust the fund balance by ``amount`` (negative to spend)."""
        globalvar.set_value("funds", self.funds + int(amount))

    def get_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def is_running(self) -> bool:
        """Return the main loop run flag."""
        return bool(globalvar.get_value("running", False))

    def stop(self) -> None:
        """Request the main loop to terminate."""
        globalvar.set_value("running", False)

    # ------------------------------------------------------------------
    # Rider lifecycle
    # ------------------------------------------------------------------
    def recruit_rider(self) -> Optional[Rider]:
        """Recruit a new rider if the roster has room and funds cover the cost.

        Returns:
            The new ``Rider`` or None if recruitment was not possible.
        """
        if len(self.riders) >= param("max_riders"):
            logger.info("Recruitment skipped: roster is full")
            return None
        cost = param("recruit_cost")
        if self.funds < cost:
            logger.info("Recruitment skipped: insufficient funds (%s < %s)",
                        self.funds, cost)
            return None
        self.add_funds(-cost)
        rider_id = globalvar.get_value("rider_counter", 0) + 1
        globalvar.set_value("rider_counter", rider_id)
        rider = Rider(rider_id)
        self.riders.append(rider)
        logger.info("Recruited rider %s for %s", rider_id, cost)
        return rider

    def remove_rider(self, rider_id: int) -> bool:
        """Remove an idle rider from the roster.

        Args:
            rider_id: Identifier of the rider to remove.

        Returns:
            True if removed, False if the rider does not exist.

        Raises:
            ValueError: If the rider still has assigned orders.
        """
        rider = self.get_rider(rider_id)
        if rider is None:
            return False
        if rider.orders:
            raise ValueError(f"rider {rider_id} still has {len(rider.orders)} order(s)")
        self.riders.remove(rider)
        return True

    # ------------------------------------------------------------------
    # Simulation clock
    # ------------------------------------------------------------------
    def tick(self) -> List[Dict[str, Any]]:
        """Advance the simulation by one time unit.

        Every rider moves once, upkeep is charged, and the clock increments.

        Returns:
            Event log entries produced during this tick.
        """
        events: List[Dict[str, Any]] = []
        for rider in list(self.riders):
            before = len(rider.log)
            rider.move()
            events.extend(rider.log[before:])
        globalvar.set_value("tick", self.tick_count + 1)
        self.add_funds(-param("upkeep_per_rider") * len(self.riders))
        return events

    # ------------------------------------------------------------------
    # Consistency and recovery
    # ------------------------------------------------------------------
    def consistency_check(self) -> List[str]:
        """Verify and repair the rider/order state.

        Repairs performed:
          * Riders that are active without orders are reset to idle.
          * Completed or rejected orders are removed from rider queues.
          * Duplicate rider IDs are reported.

        Returns:
            Human-readable list of issues found (empty when consistent).
        """
        issues: List[str] = []
        seen = set()
        for rider in self.riders:
            if rider.rider_id in seen:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen.add(rider.rider_id)
            if rider.state != STATE_IDLE and not rider.orders:
                issues.append(f"rider {rider.rider_id} active without orders; reset")
                rider.state = STATE_IDLE
                rider.path = []
                rider.special_case = False
            stale = [o for o in rider.orders if o.status in ("completed", "rejected")]
            if stale:
                issues.append(f"rider {rider.rider_id} held {len(stale)} finished order(s); removed")
                rider.orders = [o for o in rider.orders if o not in stale]
        if issues:
            logger.warning("Consistency check: %s", "; ".join(issues))
        return issues

    # ------------------------------------------------------------------
    # Serialisation
    # ------------------------------------------------------------------
    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the whole system."""
        return {
            "params": {key: param(key) for key in DEFAULT_PARAMS},
            "funds": self.funds,
            "tick": self.tick_count,
            "rider_counter": globalvar.get_value("rider_counter", 0),
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in globalvar.get_value("orders", [])],
        }


def init_system(params: Optional[Dict[str, int]] = None) -> SystemManager:
    """Reset the global store and create the system manager.

    Args:
        params: Optional parameter overrides.

    Returns:
        The initialised ``SystemManager``.
    """
    globalvar._init()
    manager = SystemManager(params)
    globalvar.set_value("system", manager)
    return manager


def get_system() -> SystemManager:
    """Return the active system manager, initialising one if needed."""
    manager = globalvar.get_value("system")
    if manager is None:
        manager = init_system()
    return manager

from typing import Iterable, List, Optional, Sequence

from config import CELL_SIZE, MAP_SIZE, STATE_NAMES
from path_planner import PathPlanner

SYMBOL_ROAD_H = "-"
SYMBOL_ROAD_V = "|"
SYMBOL_NODE = "+"
SYMBOL_EMPTY = " "
SYMBOL_RESTAURANT = "M"
SYMBOL_CUSTOMER = "C"
SYMBOL_ROUTE = "*"


class DisplayManager:
    """Produce text output for the command-line interface.

    The ASCII map uses a character canvas with two characters per 100-unit
    cell, so every street intersection sits on an even row and column.
    Rows are printed top (y = 800) to bottom (y = 0); columns are left (x = 0)
    to right (x = 800).
    """

    CANVAS_SIZE = 2 * (MAP_SIZE // CELL_SIZE) + 1

    def __init__(self, planner: Optional[PathPlanner] = None) -> None:
        self.planner = planner or PathPlanner()

    # ------------------------------------------------------------------
    # Map
    # ------------------------------------------------------------------
    def _cell(self, point: Sequence[int]):
        """Convert a map point to canvas ``(row, col)``."""
        scale = 2
        col = round(point[0] * scale / CELL_SIZE)
        row = self.CANVAS_SIZE - 1 - round(point[1] * scale / CELL_SIZE)
        return row, col

    def _put(self, canvas: List[List[str]], point: Sequence[int], symbol: str) -> None:
        """Place ``symbol`` on the canvas at the position of ``point``."""
        row, col = self._cell(point)
        if 0 <= row < self.CANVAS_SIZE and 0 <= col < self.CANVAS_SIZE:
            canvas[row][col] = symbol

    def _base_symbol(self, row: int, col: int) -> str:
        """Return the background symbol (intersection, street or empty)."""
        if row % 2 == 0 and col % 2 == 0:
            return SYMBOL_NODE
        if row % 2 == 0:
            return SYMBOL_ROAD_H
        if col % 2 == 0:
            return SYMBOL_ROAD_V
        return SYMBOL_EMPTY

    def render_map(
        self,
        riders: Iterable = (),
        orders: Iterable = (),
        route: Optional[Sequence[Sequence[int]]] = None,
    ) -> str:
        """Render the 8x8 cell map with riders, orders and an optional route.

        Symbols:
            ``+`` intersection, ``-``/``|`` street, ``M`` restaurant,
            ``C`` customer, ``0``-``9`` rider (last digit of the rider ID),
            ``*`` planned route.

        Args:
            riders: Riders to draw.
            orders: Orders whose restaurant and customer should be drawn.
            route: Optional list of points beginning with the rider position.

        Returns:
            Multi-line string containing the map and legend.
        """
        size = self.CANVAS_SIZE
        canvas = [[self._base_symbol(r, c) for c in range(size)] for r in range(size)]

        if route:
            for point in self.planner.dense_path(route[0], route[1:]):
                self._put(canvas, point, SYMBOL_ROUTE)

        for order in orders:
            if order.status in ("completed", "rejected"):
                continue
            self._put(canvas, order.res_pos, SYMBOL_RESTAURANT)
            self._put(canvas, order.cus_pos, SYMBOL_CUSTOMER)

        for rider in riders:
            self._put(canvas, (rider.x, rider.y), str(rider.rider_id % 10))

        lines = []
        for r, row in enumerate(canvas):
            label = f"{(size - 1 - r) * CELL_SIZE // 2:>4} " if r % 2 == 0 else "     "
            lines.append(label + "".join(row))
        lines.append("     x: 0 (left) -> 800 (right); y: 0 (bottom) -> 800 (top)")
        lines.append("     Legend: + node  - | street  M restaurant  C customer  "
                     "0-9 rider id  * route")
        return "\n".join(lines)

    # ------------------------------------------------------------------
    # Reports
    # ------------------------------------------------------------------
    def render_status(self, system) -> str:
        """Render the system snapshot: funds, tick and rider states."""
        riders = system.riders
        lines = [
            f"Simulated tick : {system.tick_count}",
            f"System funds   : {system.funds}",
            f"Riders         : {len(riders)}",
        ]
        if not riders:
            lines.append("  (no riders recruited yet)")
        for rider in riders:
            order_ids = [o.id for o in rider.orders]
            lines.append(
                f"  Rider {rider.rider_id:>2}: pos=({rider.x},{rider.y}) "
                f"state={STATE_NAMES.get(rider.state, 'unknown'):<10} "
                f"speed={rider.speed:>2} orders={len(rider.orders)} {order_ids}"
            )
        return "\n".join(lines)

    def render_orders(self, orders: Iterable) -> str:
        """Render a table of all orders with timing and assignment data."""
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Dist':>6} {'Taken':>6} {'Rider':>6} {'Status':>11}")
        lines = [header, "-" * len(header)]
        count = 0
        for order in orders:
            count += 1
            taken = "-" if order.time_taken is None else str(order.time_taken)
            rider = "-" if order.rider_id is None else str(order.rider_id)
            lines.append(
                f"{order.id:>4} {order.created_tick:>6} "
                f"{str(order.res_pos):>12} {str(order.cus_pos):>12} "
                f"{order.distance:>6} {taken:>6} {rider:>6} {order.status:>11}"
            )
        if count == 0:
            lines.append("  (no orders yet)")
        return "\n".join(lines)

    def render_rider_log(self, riders: Iterable) -> str:
        """Render every recorded status change of every rider."""
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider {rider.rider_id} (now {STATE_NAMES.get(rider.state, '?')}, "
                         f"at ({rider.x},{rider.y}))")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:

                lines.append(
                    f"  t={entry['tick']:>4} {entry['event']:<20} order={order:>4} "
                    f"at ({entry['x']},{entry['y']}) state={state}"
                )
        if not lines:
            return "No riders have been recruited yet."
        return "\n".join(lines)

    def render_event(self, entry) -> str:
        """Render a single event entry as a one-line message."""
        order = "" if entry["order_id"] is None else f" (order {entry['order_id']})"
        return (f"  [t={entry['tick']}] Rider {entry['rider_id']} {entry['event']}"
                f"{order} at ({entry['x']},{entry['y']})")

def load_config(path: str) -> Dict[str, int]:
    """Load system parameter overrides from a JSON file.

    The file may either contain the parameters at top level or nested under
    a ``"params"`` key. Only known, non-negative integer parameters are kept.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        Dictionary of valid overrides; empty if the file is missing or invalid.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError:
        logger.info("No configuration file at %s; using defaults", path)
        return {}
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read configuration %s: %s", path, exc)
        return {}

    if not isinstance(data, dict):
        return {}
    source = data.get("params", data)
    if not isinstance(source, dict):
        return {}

    overrides: Dict[str, int] = {}
    for key in DEFAULT_PARAMS:
        value = source.get(key)
        if isinstance(value, int) and not isinstance(value, bool) and value >= 0:
            overrides[key] = value
    if ("speed_min" in overrides and "speed_max" in overrides
            and overrides["speed_min"] > overrides["speed_max"]):
        logger.warning("speed_min greater than speed_max; ignoring speed overrides")
        overrides.pop("speed_min")
        overrides.pop("speed_max")
    return overrides


def save_state(system, path: str) -> bool:

    except OSError as exc:
        logger.error("Could not save state to %s: %s", path, exc)
        return False

"""Command parsing, shortcut resolution and input validation."""

from typing import Callable, List, Optional, Sequence, Tuple

from config import MAP_SIZE, is_valid_coordinate

COMMANDS = ("start", "status", "orders", "riders", "map", "tick",
            "history", "help", "quit")

COMMAND_ALIASES = {
    "n": "start",
    "s": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "h": "help",
    "?": "help",
    "q": "quit",
    "x": "quit",
    "exit": "quit",
}

MAX_TICKS_PER_COMMAND = 1000


class InputError(ValueError):
    """Raised for malformed user input or unknown commands."""


class InputManager:
    """Read and validate user commands and order coordinates.

    Attributes:
        history: Every raw line entered by the user, in order.
    """

    def __init__(self, input_func: Callable[[str], str] = input) -> None:
        """Create the manager.

        Args:
            input_func: Function used to read a line (injectable for tests).
        """
        self._input = input_func
        self.history: List[str] = []

    def read_line(self, prompt: str = "dispatch> ") -> str:
        """Read one line from the user and record it in the history."""
        line = self._input(prompt)
        self.history.append(line.strip())
        return line.strip()

    def parse_command(self, line: str) -> Tuple[Optional[str], List[str]]:
        """Split a raw line into a canonical command and its arguments.

        Args:
            line: Raw user input.

        Returns:
            ``(command, args)``; ``command`` is None for a blank line.

        Raises:
            InputError: If the command is unknown.
        """
        tokens = line.strip().split()
        if not tokens:
            return None, []
        word = tokens[0].lower()
        command = COMMAND_ALIASES.get(word, word)
        if command not in COMMANDS:
            raise InputError(f"unknown command '{tokens[0]}'. Type 'help' for the command list.")
        return command, tokens[1:]

    @staticmethod
    def parse_coordinates(text: str):
        """Parse ``restaurant_x restaurant_y customer_x customer_y``.

        Commas or spaces may separate the values.

        Args:
            text: Raw coordinate text.

        Returns:
            ``((rx, ry), (cx, cy))``.

        Raises:
            InputError: On wrong count, non-integer or out-of-range values.
        """
        tokens = text.replace(",", " ").split()
        if len(tokens) != 4:
            raise InputError(f"expected four integers, got {len(tokens)}. "
                             "Format: restaurant_x restaurant_y customer_x customer_y")
        values: List[int] = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise InputError(f"'{token}' is not an integer")
            if not is_valid_coordinate(value):
                raise InputError(f"coordinate {value} is outside the range 0-{MAP_SIZE}")
            values.append(value)
        return (values[0], values[1]), (values[2], values[3])

    def prompt_order(self):
        """Prompt repeatedly for order coordinates.

        Returns:
            ``((rx, ry), (cx, cy))`` or None if the user cancels.
        """
        while True:
            text = self._input("Enter restaurant x y and customer x y (or 'cancel'): ")
            self.history.append(text.strip())
            if text.strip().lower() in ("", "cancel", "c"):
                return None
            try:
                return self.parse_coordinates(text)
            except InputError as exc:
                print(f"Invalid input: {exc}. Please try again.")

    @staticmethod
    def parse_tick_count(args: Sequence[str]) -> int:
        """Parse the optional step count for the ``tick`` command.

        Raises:
            InputError: If the value is not an integer between 1 and 1000.
        """
        if not args:
            return 1
        try:
            steps = int(args[0])
        except ValueError:
            raise InputError(f"'{args[0]}' is not an integer tick count")
        if not 1 <= steps <= MAX_TICKS_PER_COMMAND:
            raise InputError(f"tick count must be between 1 and {MAX_TICKS_PER_COMMAND}")
        return steps

  map     [m]  Show the 8x8 ASCII map with riders and active orders.
  tick    [t]  Advance the simulation by N ticks (default 1, max 1000).
  history [ ]  Show recently entered commands.
  help    [h]  Show this help text.
  quit    [q]  Save state and exit (also: exit, x)."""


class DispatchApp:
    """Interactive application wiring input, system state and display."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
        config_path: str = CONFIG_FILE,
        state_path: str = STATE_FILE,
    ) -> None:
        """Initialise the system from configuration and build command handlers.

        Args:
            input_func: Function used to read user input.
            output_func: Function used to print output.
            config_path: JSON configuration file path.
            state_path: JSON file where state is saved on exit.
        """
        self.out = output_func
        self.inputs = InputManager(input_func)
        self.display = DisplayManager()
        self.orders = OrderManager()
        self.state_path = state_path
        self.system = init_system(load_config(config_path))
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> int:
        """Run the command loop until the user quits.

        Returns:
            Process exit code (0 on normal exit).
        """
        self.out("Food delivery dispatch simulator. Type 'help' for commands.")
        while self.system.is_running():
            try:
                line = self.inputs.read_line("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self.out("")
                break
            self.execute(line)
        self.shutdown()
        return 0

    def execute(self, line: str) -> None:
        """Parse and run a single command line with error recovery."""
        try:
            command, args = self.inputs.parse_command(line)
        except InputError as exc:
            self.out(f"Error: {exc}")
            return
        if command is None:
            return
        try:
            self.handlers[command](args)
        except InputError as exc:
            self.out(f"Error: {exc}")
        except Exception as exc:  # Recovery: keep the simulator running.
            logger.exception("Command '%s' failed", command)
            self.out(f"Unexpected error while running '{command}': {exc}")
            for issue in self.system.consistency_check():
                self.out(f"  Recovered: {issue}")

    def shutdown(self) -> None:
        """Save the state file and report the outcome."""
        if save_state(self.system, self.state_path):
            self.out(f"State saved to {self.state_path}.")
        else:
            self.out("Warning: state could not be saved.")
        self.out("Goodbye.")

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def cmd_start(self, args: List[str]) -> None:
        """Create and assign an order, then show the route map."""
        if args:
            restaurant, customer = self.inputs.parse_coordinates(" ".join(args))
        else:
            pair = self.inputs.prompt_order()
            if pair is None:
                self.out("Order entry cancelled.")
                return
            restaurant, customer = pair

        order = self.orders.create_order(restaurant, customer)
        try:
            result = self.orders.assign_order(order)
        except RuntimeError as exc:
            self.out(f"Order {order.id} could not be assigned: {exc}.")
            return

        rider = result["rider"]
        self.out(f"Order {order.id} accepted.")
        self.out(f"  Restaurant grid position : {order.res_pos} "
                 f"(entered {restaurant})")
        self.out(f"  Customer grid position   : {order.cus_pos} "
                 f"(entered {customer})")
        self.out(f"  New rider added          : {'yes' if result['new_rider'] else 'no'}")
        self.out(f"  Assigned rider           : {rider.rider_id} "
                 f"(speed {rider.speed} units/tick)")
        self.out(f"  Route distance           : {result['distance']} units")
        self.out(f"  Estimated time           : {result['eta_ticks']} tick(s)")
        self.out("Route map (planned from the rider's current position):")
        self.out(self.display.render_map(
            riders=self.system.riders,
            orders=[order],
            route=result["route"],
        ))

    def cmd_status(self, args: List[str]) -> None:
        """Show the system snapshot and any consistency issues."""
        issues = self.system.consistency_check()
        self.out(self.display.render_status(self.system))
        for issue in issues:
            self.out(f"  Recovered: {issue}")

    def cmd_orders(self, args: List[str]) -> None:
        """Show the order table."""
        self.out(self.display.render_orders(self.orders.all_orders()))

    def cmd_riders(self, args: List[str]) -> None:
        """Show every rider's recorded status changes."""
        self.out(self.display.render_rider_log(self.system.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map with riders and active orders."""
        self.out(self.display.render_map(
            riders=self.system.riders,
            orders=self.orders.active_orders(),
        ))

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation and report arrival and delivery events."""
        steps = self.inputs.parse_tick_count(args)
        for _ in range(steps):
            for event in self.system.tick():
                self.out(self.display.render_event(event))
        self.out(f"Advanced {steps} tick(s). Current tick: {self.system.tick_count}. "
                 f"Funds: {self.system.funds}")

    def cmd_history(self, args: List[str]) -> None:
        """Show the most recent entered lines."""
        recent = self.inputs.history[-20:]
        if not recent:
            self.out("No history yet.")
            return
        for index, line in enumerate(recent, start=max(len(self.inputs.history) - 20, 0) + 1):
            self.out(f"  {index:>4}  {line}")

    def cmd_help(self, args: List[str]) -> None:
        """Show the command reference."""
        self.out(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the main loop."""
        self.system.stop()


def main(argv: Optional[List[str]] = None) -> int:
    """Configure logging and run the interactive simulator.

    Args:
        argv: Command-line arguments (excluding the program name).

    Returns:
        Process exit code.
    """
    argv = list(sys.argv[1:] if argv is None else argv)
    level = logging.DEBUG if "--debug" in argv else logging.INFO
    logging.basicConfig(
        filename=LOG_FILE,
        level=level,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    app = DispatchApp()
    return app.run()


if __name__ == "__main__":
    sys.exit(main())

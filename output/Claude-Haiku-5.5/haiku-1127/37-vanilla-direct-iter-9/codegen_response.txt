def normalize_text(value: str) -> str:
    """Normalize text for case-insensitive, whitespace-insensitive comparison."""
    return " ".join((value or "").split()).casefold()


def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer, e.g. 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"

"""Background stdin reader so the lottery can accept control commands while running."""
import queue
import sys
import threading


class InputReader:
    """Reads stdin lines in a daemon thread and serves them through a queue.

    All interactive input goes through this reader so that control commands
    (pause / continue / terminate) can be polled during the lottery without
    blocking the drawing process.
    """

    def __init__(self, stream=None):
        self._stream = stream
        self._queue: queue.Queue = queue.Queue()
        self._thread: threading.Thread | None = None
        self.eof = False

    def _start(self) -> None:
        if self._thread is None:
            stream = self._stream if self._stream is not None else sys.stdin
            self._thread = threading.Thread(
                target=self._pump, args=(stream,), daemon=True, name="stdin-reader"
            )
            self._thread.start()

    def _pump(self, stream) -> None:
        try:
            for line in stream:
                self._queue.put(line.rstrip("\r\n"))
        except (OSError, ValueError):
            pass
        finally:
            self._queue.put(None)  # EOF sentinel

    def prompt(self, text: str) -> str | None:
        """Print a prompt and wait for one line. Returns None when input is closed."""
        if self.eof:
            print(text)
            return None
        print(text, end="", flush=True)
        self._start()
        item = self._queue.get()
        if item is None:
            self.eof = True
            print()
            return None
        return item.strip()

    def poll(self, timeout: float | None = 0.0) -> tuple[bool, str | None]:
        """Get a line without blocking forever.

        timeout None  -> block until a line (or EOF) arrives
        timeout > 0   -> wait up to timeout seconds
        timeout <= 0  -> return immediately

        Returns (True, line) when a line was received, otherwise (False, None).
        """
        if self.eof:
            return False, None
        self._start()
        try:
            if timeout is None:
                item = self._queue.get()
            elif timeout > 0:
                item = self._queue.get(timeout=timeout)
            else:
                item = self._queue.get_nowait()
        except queue.Empty:
            return False, None
        if item is None:
            self.eof = True
            return False, None
        return True, item

BANNER = "=" * 60 + "\n   Company Team Building - Smart Lottery System\n" + "=" * 60


class LotteryApp:
    """Interactive controller for the whole application."""

    def __init__(self, reader: InputReader | None = None, draw_delay: float = DEFAULT_DRAW_DELAY):
        self.reader = reader or InputReader()
        self.draw_delay = draw_delay
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.last_results: ResultManager | None = None
        self.last_prizes: list[Prize] = []

    # ------------------------------------------------------------------ utils
    def _prompt(self, text: str) -> str | None:
        return self.reader.prompt(text)

    @staticmethod
    def _print_main_menu() -> None:
        print()
        print("---------------- Main Menu ----------------")
        print("1. Load default team building list")
        print("2. Upload custom txt list")
        print("3. View department groups")
        print("4. Manage prizes")
        print("5. Reorder prizes")
        print("6. Start lottery")
        print("7. Export results")
        print("0. Exit")

    # -------------------------------------------------------------- main loop
    def run(self) -> None:
        print(BANNER)
        while True:
            self._print_main_menu()
            choice = self._prompt("Select an option: ")
            if choice is None:
                print("Input closed. Exiting.")
                break
            if choice == "1":
                self._load_default_list()
            elif choice == "2":
                self._upload_custom_list()
            elif choice == "3":
                self._show_groups()
            elif choice == "4":
                self._manage_prizes()
            elif choice == "5":
                self._reorder_prizes()
            elif choice == "6":
                self._start_lottery()
            elif choice == "7":
                self._export_results()
            elif choice == "0":
                print("Goodbye!")
                break
            else:
                print("Invalid option. Please try again.")

    # ------------------------------------------------------------ list module
    def _load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._apply_loaded_list(users, errors)

    def _upload_custom_list(self) -> None:
        path = self._prompt("Enter the path of the txt file (each line: Name,Department Group): ")
        if path is None:
            return
        path = path.strip().strip('"').strip("'")
        if not path:
            print("No file path given. Upload cancelled.")
            return
        users, errors = self.list_manager.load_custom_list(path)
        self._apply_loaded_list(users, errors)

    def _apply_loaded_list(self, users: list[User], errors: list[str]) -> None:
        if errors:
            print("Problems found in the list:")
            for error in errors:
                print(f"  - {error}")
        valid, duplicates = self.list_manager.validate_users(users)
        for dup in duplicates:
            print(f"  - Duplicate Name: '{dup.name}' ({dup.department}) already exists in the list.")

        if errors or duplicates:
            print(f"Valid records: {len(valid)} | invalid or duplicate records: "
                  f"{len(errors) + len(duplicates)}")
            choice = self._prompt(
                "Enter 'r' to remove invalid data and continue, or 't' to terminate the upload: "
            )
            if choice is None or choice.lower() != "r":
                print("Upload terminated. Current list unchanged.")
                return

        if not valid:
            print("No valid users found. Current list unchanged.")
            return
        self.users = valid
        print(f"Loaded {len(valid)} users.")

    def _show_groups(self) -> None:
        if not self.users:
            print("No user list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for department, count in info.items():
            print(f"  {department}: {count} people")
        print(f"Total: {len(self.users)} people in {len(info)} groups.")

    def _manage_prizes(self) -> None:

            print("---------------- Prize Library ----------------")
            print("1. Load preset prize library")
            print("2. Add custom prize")

        name = self._prompt("Prize name (blank to cancel): ")
        if not name:
            print("Cancelled.")
            return

        quantity = None
        while quantity is None:
            text = self._prompt("Quantity (positive integer, blank to cancel): ")
            if not text:
                print("Cancelled.")
                return
            try:
                value = int(text)
                if value < 1:
                    raise ValueError
                quantity = value
            except ValueError:
                print("Please enter a positive integer.")

        if drawer is False:
            print("Cancelled.")

        avoid_groups = self._choose_avoid_groups()
        if avoid_groups is None:
            print("Cancelled.")
            return

        try:
            self.prize_manager.add_prize(name, quantity, avoid_groups, drawer)
        except ValueError as exc:
            print(f"Failed to add prize: {exc}")

    def _choose_drawer(self):
        """Return drawer name, None for no drawer, or False if cancelled."""
        print("Associated lottery person:")
        print("  0. None")
        for index, drawer in enumerate(DRAWERS, start=1):
            print(f"  {index}. {drawer}")
        print(f"  {len(DRAWERS) + 1}. Other (type a name)")
        text = self._prompt("Choose a number or type a name (blank for none): ")
        if text is None:
            return False
        if text == "" or text == "0":
            return None
        if text.isdigit():
            number = int(text)
            if 1 <= number <= len(DRAWERS):
                return DRAWERS[number - 1]
            if number == len(DRAWERS) + 1:
                other = self._prompt("Enter lottery person name (blank for none): ")
                if other is None:
                    return False
                return other.strip() or None
            print("Invalid choice; no lottery person set.")
            return None
        return text.strip() or None

    def _choose_avoid_groups(self) -> list[str] | None:
        options: list[str] = []
        if self.users:
            options = list(self.list_manager.get_group_info(self.users).keys())
            print("Available department groups:")
            for index, group in enumerate(options, start=1):
                print(f"  {index}. {group}")
        text = self._prompt(
            "Avoid groups (comma-separated numbers or names, blank for none): "
        )
        if text is None:
            return None
        groups: list[str] = []
        for token in text.split(","):
            item = token.strip()
            if not item:
                continue
            if item.isdigit() and options and 1 <= int(item) <= len(options):
                group = options[int(item) - 1]
            else:
                group = item
            if group not in groups:
                groups.append(group)
        return groups

        if not prizes:
            print("No prizes configured.")
            return
        self._show_prizes()
        text = self._prompt(
            "Enter the new order as serial numbers separated by spaces (e.g. 2 1 3): "
        )
        if text is None or not text.strip():
            print("Reorder cancelled.")
            return
        try:
            order = [int(token) for token in text.split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self._show_prizes()
        else:
            print("Invalid order: every prize serial number must appear exactly once.")

    # ------------------------------------------------------- lottery process
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a user list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return

        lottery = LotteryManager(self.users)
        results = ResultManager()
        print()
        print("=== Lottery started ===")
        print("Commands during the lottery: 'pause', 'continue', 'terminate'")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, prize, lottery, results)

        self.last_results = results
        self.last_prizes = prizes
        self._print_final_summary(results, prizes)

    def _run_round(self, round_no: int, prize: Prize, lottery: LotteryManager,
                   results: ResultManager) -> None:
        print()
        print(f"=== Round {round_no}: {prize.name} ===")
        print(f"Quantity: {prize.quantity} | Lottery person: {prize.associated_drawer or 'None'}")
        print(f"Rule: {lottery.describe_rule(prize)}")

        if lottery.eligible_count(prize) == 0:
            print("No eligible candidates for this round. Round skipped.")
            return

        round_winners: list[User] = []
        for position in range(1, prize.quantity + 1):
            if self._process_controls(wait=self.draw_delay):
                print(f"Round ended early. Winners drawn: {len(round_winners)}/{prize.quantity}")
                break
            print(f"Currently drawing the {ordinal(position)} winner...")
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this round.")
                break
            results.add_winner(user, prize)
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        print(f"--- Round {round_no} results ---")
        if not round_winners:
            print("  (no winners)")
        for index, user in enumerate(round_winners, start=1):
            print(f"  {index}. {user.name} ({user.department})")

    def _process_controls(self, wait: float) -> bool:
        """Wait up to `wait` seconds, processing control commands.

        Returns True if the current round must be terminated.
        """
        ok, line = self.reader.poll(wait)
        while ok:
            command = (line or "").strip().lower()
            if command == "pause":
                if self._wait_while_paused():
                    return True
            elif command == "terminate":
                print("Terminate command received: ending the current round.")
                return True
            elif command == "continue":
                print("Lottery is not paused.")
            elif command:
                print(f"Unknown command '{line}'. Available: pause, continue, terminate.")
            ok, line = self.reader.poll(0.0)
        return False

    def _wait_while_paused(self) -> bool:
        """Block while paused. Returns True if the round should be terminated."""
        print("Lottery PAUSED. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            ok, line = self.reader.poll(None)
            if not ok:
                print("Input closed: ending the current round.")
                return True
            command = (line or "").strip().lower()
            if command == "continue":
                print("Lottery resumed.")
                return False
            if command == "terminate":
                print("Current round terminated.")
                return True
            if command == "pause":
                print("Lottery is already paused.")
            elif command:
                print("While paused, only 'continue' or 'terminate' is accepted.")

    def _print_final_summary(self, results: ResultManager, prizes: list[Prize]) -> None:
        print()
        print("=== Final Winner List (by prize level, then winning order) ===")
        grouped = results.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            print("No winners were drawn.")
            return
        for prize_name, winners in grouped.items():
            print(f"[{prize_name}] ({len(winners)})")
            for index, winner in enumerate(winners, start=1):
                print(f"  {index}. {winner.user.name} ({winner.user.department})")
        print("=== Lottery finished ===")

    # ----------------------------------------------------------- export
    def _export_results(self) -> None:
        if self.last_results is None:
            print("No lottery results to export. Please run the lottery first.")
            return
        _, message = self.last_results.export_to_csv(self.last_prizes)
        print(message)


def main() -> int:
    app = LotteryApp()
    app.run()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""List loading, verification and group extraction."""
from lottery_system.default_data import DEFAULT_USER_RECORDS
from lottery_system.models import User
from lottery_system.utils import normalize_text


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        users = [User(name, department) for name, department in DEFAULT_USER_RECORDS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        for line_no, raw in enumerate(lines, start=1):
            if not raw.strip():
                continue
            parts = raw.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{raw.strip()}' "
                    f"(expected format: Name,Department Group)"
                )
                continue
            name = " ".join(parts[0].split())
            department = " ".join(parts[1].split())
            users.append(User(name, department))

        if not users and not errors:
            errors.append("File Error: the file contains no user records.")
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates).

        The first occurrence of each name is kept as valid; later occurrences
        of the same name are reported as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group -> number of people."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management and ordering."""
from lottery_system.default_data import DEFAULT_PRIZE_RECORDS
from lottery_system.models import Prize
from lottery_system.utils import normalize_text


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZE_RECORDS:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"Prize '{name}' not loaded: {exc}")
        return errors

    def add_prize(self, name: str, quantity: int, avoid_groups: list[str] | None = None,
                  associated_drawer: str | None = None) -> None:
        """Append a prize. Raises ValueError on invalid input."""
        clean_name = " ".join((name or "").split())
        if not clean_name:
            raise ValueError("prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("quantity must be a positive integer.")
        key = normalize_text(clean_name)
        if any(normalize_text(prize.name) == key for prize in self._prizes):
            raise ValueError(f"a prize named '{clean_name}' already exists.")

        groups: list[str] = []
        seen: set[str] = set()
        for group in avoid_groups or []:
            cleaned = " ".join(str(group).split())
            if cleaned and normalize_text(cleaned) not in seen:
                seen.add(normalize_text(cleaned))
                groups.append(cleaned)

        drawer = " ".join(associated_drawer.split()) if associated_drawer else None
        self._prizes.append(Prize(clean_name, quantity, groups, drawer or None))

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if len(positions) != count or sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        return True

def _drawer_key(prize: Prize) -> str:
    return normalize_text(prize.associated_drawer or "")


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        # Results of the most recent draws by Chen Qixian / Jin Duo:
        # True means the winner belonged to the Cheng Ru group.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng or random.Random()

    # ----------------------------------------------------------- helpers
    def has_won(self, user: User) -> bool:
        return normalize_text(user.name) in self._winner_keys

    def remaining_users(self) -> list[User]:
        return [user for user in self._users if not self.has_won(user)]

    @staticmethod
    def is_cheng_ru_member(user: User) -> bool:
        return normalize_text(user.department) == normalize_text(CHENG_RU_GROUP)

    def is_fallback_active(self) -> bool:
        """True when the last two draws by Chen Qixian / Jin Duo had no Cheng Ru member."""
        return len(self.last_two_draw_results) >= 2 and not any(self.last_two_draw_results[-2:])

    def _cheng_ru_remaining(self) -> list[User]:
        return [user for user in self.remaining_users() if self.is_cheng_ru_member(user)]

    # ------------------------------------------------------------- rules
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users who may win the given prize right now."""
        available = self.remaining_users()
        drawer = _drawer_key(prize)

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru members, ignoring avoid groups.
        if drawer == normalize_text(MA_JIAQI) and self.is_fallback_active():
            forced = [user for user in available if self.is_cheng_ru_member(user)]
            if forced:
                return forced
            # No Cheng Ru members remain: fall back to the ordinary rules.

        # Special rule: Li Bin only filters previous winners.
        if drawer == normalize_text(LI_BIN):
            return available

        avoid = {normalize_text(group) for group in prize.avoid_groups}
        return [user for user in available if normalize_text(user.department) not in avoid]

    def eligible_count(self, prize: Prize) -> int:
        return len(self._get_eligible_pool(prize))

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to this prize."""
        drawer = _drawer_key(prize)
        if (drawer == normalize_text(MA_JIAQI) and self.is_fallback_active()
                and self._cheng_ru_remaining()):
            return (f"{CHENG_RU_GROUP} protection is active: only unawarded {CHENG_RU_GROUP} "
                    f"members can win this prize (avoid groups are ignored).")
        base = "Previous winners are excluded."
        if drawer == normalize_text(LI_BIN):
            return base + " Avoid groups do not apply to this lottery person."
        if prize.avoid_groups:
            return base + " Avoid groups: " + ", ".join(prize.avoid_groups) + "."
        return base + " No avoid groups."

    # ------------------------------------------------------------- drawing
    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_keys.add(normalize_text(winner.name))

        drawer = _drawer_key(prize)
        if drawer in (normalize_text(CHEN_QIXIAN), normalize_text(JIN_DUO)):
            self.last_two_draw_results.append(self.is_cheng_ru_member(winner))
            del self.last_two_draw_results[:-2]
        return winner

"""Winner tracking, summary grouping and CSV export."""
import csv
import os
import re
from dataclasses import dataclass
from datetime import date, datetime

from lottery_system.config import CSV_HEADER, RESULT_FILE_PREFIX
from lottery_system.models import Prize, User

_LEVEL_WORDS = {
    "first": 1, "second": 2, "third": 3, "fourth": 4, "fifth": 5,
    "sixth": 6, "seventh": 7, "eighth": 8, "ninth": 9, "tenth": 10,
}
_UNKNOWN_LEVEL = 1000


def prize_level_rank(name: str) -> int:
    """Lower rank means higher prize level. Unknown levels rank last."""
    match = re.match(r"\s*([A-Za-z]+)\b", name or "")
    if match:
        return _LEVEL_WORDS.get(match.group(1).lower(), _UNKNOWN_LEVEL)
    return _UNKNOWN_LEVEL


def _sort_prizes(prizes: list[Prize]) -> list[Prize]:
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: (prize_level_rank(item[1].name), item[0]))
    return [prize for _, prize in indexed]


@dataclass(frozen=True)
class Winner:
    user: User
    prize_name: str
    sequence: int
    timestamp: datetime


class ResultManager:
    """Stores winners in winning order and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        winner = Winner(user, prize_name, len(self._winners) + 1, datetime.now())
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low),
        and by winning order within each prize."""
        result: dict[str, list[Winner]] = {}
        for prize in _sort_prizes(prizes):
            winners = [w for w in self._winners if w.prize_name == prize.name]
            winners.sort(key=lambda w: w.sequence)
            result[prize.name] = winners
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        path = os.path.join(os.getcwd(), f"{RESULT_FILE_PREFIX}{date.today():%Y%m%d}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(CSV_HEADER)
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        writer.writerow([
                            winner.user.name,
                            winner.user.department,
                            prize_name,
                            winner.timestamp.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

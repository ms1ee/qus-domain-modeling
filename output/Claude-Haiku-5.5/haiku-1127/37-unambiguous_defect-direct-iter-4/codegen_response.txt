from __future__ import annotations

import os

from lottery_system.defaults import DEFAULT_LIST_TEXT
from lottery_system.models import User


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list uploaded by the user."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it into users.

        Each non-empty, non-comment line must be 'Name,Department'
        (a tab may also be used as the separator).
        """
        if not path or not os.path.isfile(path):
            return [], [f"File not found: '{path}'"]
        try:
            lines = self._read_lines(path)
        except (OSError, ValueError) as exc:
            return [], [f"Cannot read file '{path}': {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _read_lines(path: str) -> list[str]:
        last_error: Exception | None = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, encoding=encoding) as handle:
                    return handle.read().splitlines()
            except UnicodeDecodeError as exc:
                last_error = exc
                continue
        raise ValueError(f"unsupported text encoding ({last_error})")

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = line.split("\t") if "\t" in line else line.split(",")
            parts = [part.strip() for part in parts]
            if len(parts) != 2 or not all(parts):
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    f"(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Detect duplicate names.

        Returns (valid_users, duplicates). The first occurrence of each name is
        kept in valid_users; later occurrences are reported in duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of people."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management: presets, custom prizes, avoidance groups, ordering."""

from __future__ import annotations

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the list of prizes and their lottery sequence."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the lottery sequence."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if any(prize.name == name for prize in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = group.strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer = associated_drawer.strip() if associated_drawer else ""
        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer or None,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        ``order`` contains 1-based current positions in the new order, e.g.
        [3, 1, 2] means the prize at position 3 becomes first.
        Returns False if the order is not a valid permutation.
        """
        count = len(self._prizes)
        if len(order) != count:
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

from __future__ import annotations

import random
from typing import Callable

from lottery_system.models import (
    LI_BIN,
    MA_JIAQI,
    PROTECTION_DRAWERS,
    Prize,
    User,
    is_cheng_ru,
)
from lottery_system.services.result_manager import ResultManager


def _norm(text: str | None) -> str:
    return (text or "").strip().lower()


class LotteryManager:
    """Runs lottery rounds while applying the lottery rule engine."""

    def __init__(self, users: list[User], result_manager: ResultManager | None = None) -> None:
        self.users: list[User] = list(users)
        self.result_manager: ResultManager = (
            result_manager if result_manager is not None else ResultManager()
        )
        # Results of the most recent rounds run by Chen Qixian / Jin Duo.
        # True means that round produced at least one Cheng Ru group winner.
        self.last_two_draw_results: list[bool] = []
        # Anti-duplication: names of everyone who has already won.
        self._winner_names: set[str] = {
            winner.user.name for winner in self.result_manager.get_all_winners()
        }

    # ------------------------------------------------------------------
    # Public helpers
    # ------------------------------------------------------------------
    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def get_winner_names(self) -> set[str]:
        return set(self._winner_names)

    def run_round(
        self,
        prize: Prize,
        on_winner: Callable[[int, int, User], None] | None = None,
        before_draw: Callable[[int, int], bool] | None = None,
    ) -> list[User]:
        """Run one lottery round for ``prize``.

        ``before_draw(index, total)`` is called before each draw; returning
        False ends the round early (used for the 'terminate' command).
        ``on_winner(index, total, user)`` is called after each successful draw.
        """
        winners: list[User] = []
        total = prize.quantity
        for index in range(1, total + 1):
            pool = self._get_eligible_pool(prize)
            if not pool:
                break
            if before_draw is not None and not before_draw(index, total):
                break
            chosen = random.choice(pool)
            self._record_winner(chosen, prize)
            winners.append(chosen)
            if on_winner is not None:
                on_winner(index, total, chosen)
        self._update_draw_history(prize, winners)
        return winners

    # ------------------------------------------------------------------
    # Rule engine
    # ------------------------------------------------------------------
    def _available_users(self) -> list[User]:
        return [user for user in self.users if user.name not in self._winner_names]

    def _protection_active(self) -> bool:
        """Cheng Ru protection triggers when the last two results both lack Cheng Ru winners."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = self._available_users()
        drawer = _norm(prize.associated_drawer)

        # Fallback mechanism: Cheng Ru group protection for Ma Jiaqi's prize.
        if drawer == _norm(MA_JIAQI) and self._protection_active():
            protected = [user for user in available if is_cheng_ru(user)]
            if protected:
                return protected
            # No Cheng Ru members left: fall back to the normal rules.

        # Special rule: relaxed filtering for Li Bin (avoidance groups ignored).
        if drawer == _norm(LI_BIN):
            return available

        # Basic filtering: exclude members of the prize's avoidance groups.
        avoid = {_norm(group) for group in prize.avoid_groups}
        return [user for user in available if _norm(user.department) not in avoid]

    def _record_winner(self, user: User, prize: Prize) -> None:
        self._winner_names.add(user.name)
        self.result_manager.add_winner(user, prize)

    def _update_draw_history(self, prize: Prize, winners: list[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer == _norm(MA_JIAQI):
            # Protection has been applied/consumed; reset the history.
            self.last_two_draw_results.clear()
            return
        protection_drawers = {_norm(name) for name in PROTECTION_DRAWERS}
        if drawer in protection_drawers and winners:
            has_cheng_ru = any(is_cheng_ru(user) for user in winners)
            self.last_two_draw_results.append(has_cheng_ru)
            del self.last_two_draw_results[:-2]

"""Winner recording, grouping, sorting and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User

ORDINAL_LEVELS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
}


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    order: int  # global winning time order (1-based)


def prize_level_key(prize: Prize, index: int) -> tuple[int, int]:
    """Sort key: higher prize level (First > Second > Third) first, then configured order."""
    words = prize.name.strip().split()
    first_word = words[0].lower() if words else ""
    level = ORDINAL_LEVELS.get(first_word, len(ORDINAL_LEVELS) + 1)
    return level, index


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Return prizes ordered from highest prize level to lowest."""
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: prize_level_key(item[1], item[0]))
    return [prize for _, prize in indexed]


class ResultManager:
    """Keeps the winner list and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize | str) -> Winner:
        prize_name = prize if isinstance(prize, str) else prize.name
        winner = Winner(user=user, prize_name=prize_name, order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """All winners in winning time order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map prize name to its winners (winning time order).

        The dictionary is ordered from highest prize level to lowest.
        """
        result: dict[str, list[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt."""
        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize", "Name", "Department", "Winning Order"])
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        writer.writerow(
                            [
                                prize_name,
                                winner.user.name,
                                winner.user.department,
                                winner.order,
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

from __future__ import annotations

def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer, e.g. 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ------------------------------------------------------------------
    # Input helpers
    # ------------------------------------------------------------------
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _ask_int(self, prompt: str, minimum: int = 1) -> int:
        while True:
            text = self._ask(prompt)
            try:
                value = int(text)
            except ValueError:
                print("Please enter a valid integer.")
                continue
            if value < minimum:
                print(f"Please enter a number >= {minimum}.")
                continue
            return value

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> None:
        print("Welcome to the Team Building Smart Lottery System.")
        while True:
            print(MAIN_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                self.load_participants()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.prize_menu()
            elif choice == "4":
                self.start_lottery()
            elif choice == "5":
                self.results_menu()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    # ------------------------------------------------------------------
    # List management
    # ------------------------------------------------------------------
    def load_participants(self) -> None:
        print("1. Built-in default team building list")
        print("2. Upload custom txt list (one 'Name,Department' per line)")
        choice = self._ask("Select a loading mode: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter the txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Invalid choice.")
            return

        valid = self._verify_list(users, errors)
        if valid is None:
            print("Upload terminated. The current list is unchanged.")
            return
        self._set_users(valid)
        print(f"Loaded {len(valid)} valid participants.")

    def _verify_list(self, users: list[User], errors: list[str]) -> list[User] | None:
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            return valid

        print("\nData verification found problems:")
        for error in errors:
            print(f"  - {error}")
        for duplicate in duplicates:
            print(f"  - Duplicate name: {duplicate.name} ({duplicate.department})")

        if not valid:
            print("No valid participants remain.")
            return None

        while True:
            choice = self._ask(
                "Type 'r' to remove invalid/duplicate data and continue, "
                "or 't' to terminate the upload: "
            ).lower()
            if choice == "r":
                return valid
            if choice == "t":
                return None
            print("Please enter 'r' or 't'.")

    def _set_users(self, users: list[User]) -> None:
        self.users = list(users)
        self.lottery = LotteryManager(self.users, self.result_manager)

    def show_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("\nValid department groups:")
        for number, (department, count) in enumerate(info.items(), start=1):
            print(f"  {number}. {department}: {count} people")

    # ------------------------------------------------------------------
    # Prize configuration
    # ------------------------------------------------------------------
    def prize_menu(self) -> None:
        while True:
            print(PRIZE_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                self._load_preset_prizes()
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._set_avoid_groups()
            elif choice == "4":
                self._view_prizes()
            elif choice == "5":
                self._reorder_prizes()
            elif choice == "0":
                return
            else:
                print("Invalid option, please try again.")

    def _load_preset_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            for error in errors:
                print(f"  - {error}")
        else:
            print("Preset prizes loaded.")

    def _group_names(self) -> list[str]:
        if not self.users:
            return []
        return list(self.list_manager.get_group_info(self.users).keys())

    def _show_groups_numbered(self, groups: list[str]) -> None:
        if not groups:
            print("(No department groups available; load a participant list first.)")
            return
        print("Available department groups:")
        for number, group in enumerate(groups, start=1):
            print(f"  {number}. {group}")

    def _parse_groups(self, text: str, groups: list[str]) -> list[str]:
        """Parse comma-separated group numbers or names into a list of group names."""
        result: list[str] = []
        normalized = text.replace("\uff0c", ",")
        for token in (part.strip() for part in normalized.split(",")):
            if not token:
                continue
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                name = groups[int(token) - 1]
            else:
                name = next((g for g in groups if g.lower() == token.lower()), token)
            if name not in result:
                result.append(name)
        return result

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            print("Prize name cannot be empty.")
            return
        quantity = self._ask_int("Prize quantity: ", 1)
        drawer = self._ask("Associated lottery person (blank for none): ") or None
        groups = self._group_names()
        self._show_groups_numbered(groups)
        avoid_text = self._ask("Avoidance groups (numbers or names, comma separated; blank for none): ")
        avoid_groups = self._parse_groups(avoid_text, groups)
        try:
            prize = self.prize_manager.add_prize(name, quantity, avoid_groups, drawer)
        except ValueError as exc:
            print(f"Error: {exc}")
            return
        print(f"Prize added: {self._format_prize(prize)}")

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        number = self._ask_int("Select prize number: ", 1)
        if number > len(prizes):
            print("Invalid prize number.")
            return
        prize = prizes[number - 1]
        groups = self._group_names()
        self._show_groups_numbered(groups)
        print(f"Current avoidance groups: {', '.join(prize.avoid_groups) or 'None'}")
        text = self._ask("New avoidance groups (numbers or names, blank to clear): ")
        prize.avoid_groups = self._parse_groups(text, groups)
        print(f"Updated avoidance groups: {', '.join(prize.avoid_groups) or 'None'}")

    @staticmethod
    def _format_prize(prize: Prize) -> str:
        parts = [f"{prize.name} x{prize.quantity}"]
        if prize.associated_drawer:
            parts.append(f"drawer: {prize.associated_drawer}")
        if prize.avoid_groups:
            parts.append(f"avoid: {', '.join(prize.avoid_groups)}")
        return " | ".join(parts)

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nLottery sequence:")
        for number, prize in enumerate(prizes, start=1):
            print(f"  {number}. {self._format_prize(prize)}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        text = self._ask("Enter the new order as serial numbers (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in text.replace(",", " ").split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
            self._view_prizes()
        else:
            print("Invalid order: each serial number from 1 to "
                  f"{len(prizes)} must appear exactly once.")

    # ------------------------------------------------------------------
    # Lottery process
    # ------------------------------------------------------------------
    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users, self.result_manager)

        print("\nLottery starts! Rounds in order:")
        for number, prize in enumerate(prizes, start=1):
            print(f"  {number}. {self._format_prize(prize)}")

        for round_no, prize in enumerate(prizes, start=1):
            print(f"\n===== Round {round_no}/{len(prizes)}: {prize.name} =====")
            print(f"Quantity: {prize.quantity}")
            print(f"Rules: {self._describe_rules(prize)}")

            def announce(index: int, total: int, user: User, current: Prize = prize) -> None:
                print(f"Congratulations {user.name} ({user.department}) "
                      f"for winning {current.name}!")

            winners = self.lottery.run_round(
                prize,
                on_winner=announce,
                before_draw=self._before_draw,
            )
            self._print_round_results(prize, winners)

        print("\n========== Final Winner List ==========")
        self._print_summary()

    def _before_draw(self, index: int, total: int) -> bool:
        """Prompt before each draw. Returns False if the round should be terminated."""
        print(f"\nCurrently drawing the {ordinal(index)} winner ({index}/{total})")
        while True:
            command = self._ask("Press Enter to draw, or type 'pause' / 'terminate': ").lower()
            if command == "":
                return True
            if command == "terminate":
                print("Current round terminated and skipped.")
                return False
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("(paused) > ").lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        return True
                    if resume == "terminate":
                        print("Current round terminated and skipped.")
                        return False
                    print("Please type 'continue' or 'terminate'.")
            print("Unknown command.")

    def _describe_rules(self, prize: Prize) -> str:
        rules = ["previous winners are excluded"]
        drawer = (prize.associated_drawer or "").strip()
        if drawer.lower() == LI_BIN.lower():
            rules.append("relaxed rules for Li Bin (avoidance groups ignored)")
        elif prize.avoid_groups:
            rules.append(f"avoid groups: {', '.join(prize.avoid_groups)}")
        if drawer.lower() == MA_JIAQI.lower():
            rules.append(
                "Cheng Ru Group protection: if the last two Chen Qixian/Jin Duo rounds "
                "had no Cheng Ru Group winners, only Cheng Ru Group members are eligible"
            )
        if drawer:
            rules.insert(0, f"lottery person: {drawer}")
        return "; ".join(rules)

    def _print_round_results(self, prize: Prize, winners: list[User]) -> None:
        if not winners:
            print(f"No winners drawn for {prize.name} (no eligible candidates or round terminated).")
            return
        print(f"\nRound results for {prize.name} (in winning order):")
        for number, user in enumerate(winners, start=1):
            print(f"  {number}. {user.name} ({user.department})")
        if len(winners) < prize.quantity:
            print(f"Note: {len(winners)} of {prize.quantity} winners drawn "
                  f"(round terminated or not enough eligible candidates).")

    def _print_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        has_winners = False
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            has_winners = True
            print(f"\n[{prize_name}]")
            for winner in winners:
                print(f"  #{winner.order:<3} {winner.user.name} ({winner.user.department}) "
                      f"- {prize_name}")
        if not has_winners:
            print("No winners yet.")

    # ------------------------------------------------------------------
    # Results
    # ------------------------------------------------------------------
    def results_menu(self) -> None:
        while True:
            print(RESULT_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                self._print_summary()
            elif choice == "2":
                ok, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
                print(message)
            elif choice == "0":
                return
            else:
                print("Invalid option, please try again.")

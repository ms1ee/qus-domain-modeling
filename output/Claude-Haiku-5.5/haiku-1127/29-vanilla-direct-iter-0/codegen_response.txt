"""Grid geometry constants and coordinate helpers.

The simulated map is 800x800 units, divided into an 8x8 grid of 100x100
cells. Lines at multiples of 100 are streets. Restaurant and customer
locations are converted to the centre point of their grid cell.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = 8

RIDER_SPAWN = (435, 385)
DEFAULT_RIDER_SPEED = 10
RECRUIT_COST = 300
INITIAL_FUNDS = 1000
MAX_ROUTE_TICKS = 2000


def validate_coordinate(value) -> int:
    """Validate that a coordinate is an integer within the map range.

    Args:
        value: The raw value to validate.

    Returns:
        The coordinate as an int.

    Raises:
        ValueError: If the value is not an integer or lies outside 0-800.
    """
    if isinstance(value, bool):
        raise ValueError(f"'{value}' is not an integer coordinate")
    try:
        number = int(value)
    except (TypeError, ValueError):
        raise ValueError(f"'{value}' is not an integer coordinate") from None
    if str(value).strip() != str(number) and not isinstance(value, int):
        raise ValueError(f"'{value}' is not an integer coordinate")
    if number < 0 or number > MAP_SIZE:
        raise ValueError(
            f"coordinate {number} is out of range (0-{MAP_SIZE})"
        )
    return number


def to_cell_center(x, y) -> tuple:
    """Convert a raw coordinate pair to the centre of its grid cell.

    Values at or beyond the map edge (800) are clamped into the last cell.

    Args:
        x: Raw x coordinate.
        y: Raw y coordinate.

    Returns:
        Tuple (center_x, center_y) with center = (v // 100) * 100 + 50.
    """
    cx = min(max(int(x), 0), MAP_SIZE - 1)
    cy = min(max(int(y), 0), MAP_SIZE - 1)
    center_x = (cx // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2
    center_y = (cy // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2
    return center_x, center_y


def cell_index(x, y) -> tuple:
    """Return the (column, row) grid index of the cell containing a point.

    Args:
        x: x coordinate.
        y: y coordinate.

    Returns:
        Tuple (column, row), each in range 0-7.
    """
    last = GRID_DIM - 1
    col = min(max(int(x) // CELL_SIZE, 0), last)
    row = min(max(int(y) // CELL_SIZE, 0), last)
    return col, row


def cell_label(x, y) -> str:
    """Return a human readable grid cell label such as '(col 1, row 2)'."""
    col, row = cell_index(x, y)
    return f"(col {col}, row {row})"


def is_street(x, y) -> bool:
    """Return True if the point lies on a street (grid line)."""
    return int(x) % CELL_SIZE == 0 or int(y) % CELL_SIZE == 0

"""Process-wide key/value store holding the simulator's global variables.

Keys used by the program:
    system_funds (int), riders (list[Rider]), rider_id_counter (int),

    """Reset every global variable to its startup default value."""
    global _store
    _store = {
        "system_funds": INITIAL_FUNDS,
        "riders": [],
        "rider_id_counter": 0,

    """Store a value under the given key.

    Args:
        key: Variable name.

    """

    Args:
        key: Variable name.

    Returns:

"""Grid-based path planning for riders.

Routes are axis-aligned polylines. Between a start and a goal the planner
chooses one of the two L-shaped routes, preferring the one whose legs run
along streets (grid lines at multiples of 100). Points that share an x or y
coordinate with the goal are reached by a straight line (the "directly
above or below" special case).
"""

from grid import CELL_SIZE


class PathPlanner:
    """Plans street-preferring, right-angle routes on the 800x800 map."""

    def plan(self, start, goal) -> list:
        """Compute the waypoints from start to goal.

        Args:
            start: Tuple (x, y) of the current position.
            goal: Tuple (x, y) of the destination.

        Returns:
            List of waypoints (tuples) ending at goal. Empty if the start
            already equals the goal. Consecutive points are axis-aligned.
        """
        sx, sy = int(start[0]), int(start[1])
        gx, gy = int(goal[0]), int(goal[1])
        if (sx, sy) == (gx, gy):
            return []
        if sx == gx or sy == gy:
            # Directly above/below or beside the target: straight line.
            return [(gx, gy)]

        horizontal_first = (gx, sy)
        vertical_first = (sx, gy)
        h_score = (1 if sy % CELL_SIZE == 0 else 0) + \
                  (1 if gx % CELL_SIZE == 0 else 0)
        v_score = (1 if sx % CELL_SIZE == 0 else 0) + \
                  (1 if gy % CELL_SIZE == 0 else 0)
        corner = horizontal_first if h_score >= v_score else vertical_first
        return [corner, (gx, gy)]


def path_length(start, path) -> int:
    """Return the Manhattan length of a waypoint path starting at start."""
    total = 0
    cur = (int(start[0]), int(start[1]))
    for point in path:
        total += abs(point[0] - cur[0]) + abs(point[1] - cur[1])
        cur = point
    return total

"""Rider model: state machine, movement and delivery event log.

Rider states:
    0 = Idle, 1 = Picking Up (heading to restaurant),
    2 = Delivering (heading to customer).

Events written to ``Rider.log``:
    "spawn", "accepted_order", "arrived_at_merchant", "delivered_to_customer".
"""

import logging

from globalvar import get_value
from grid import DEFAULT_RIDER_SPEED, RIDER_SPAWN
from path_planner import PathPlanner

logger = logging.getLogger("dispatch.rider")

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {
    IDLE: "Idle",
    PICKING_UP: "Picking Up",
    DELIVERING: "Delivering",
}

_planner = PathPlanner()


class Rider:
    """A delivery rider that moves on the grid and serves assigned orders."""

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the spawn point, idle, with a spawn log entry.

        Args:
            rider_id: Unique integer identifier of the rider.
        """
        self.rider_id = rider_id
        self.x, self.y = RIDER_SPAWN
        self.speed = int(get_value("rider_speed", DEFAULT_RIDER_SPEED))
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.path = []
        self.distance_travelled = 0
        self._target = None
        self._record("spawn", None)

    @property
    def position(self) -> tuple:
        """Current (x, y) position."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Human readable state name."""
        return STATE_NAMES.get(self.state, "Unknown")

    def is_idle(self) -> bool:
        """Return True if the rider is free to take a new order."""
        return self.state == IDLE and not self.orders

    def reset_movement(self) -> None:
        """Clear the planned path (used for error recovery)."""
        self.path = []
        self._target = None

    def move(self) -> None:
        """Advance the rider by one tick (one call = one unit of time).

        Handles the idle->picking-up transition when an order is present,
        moves along the planned route by up to ``speed`` units, and applies
        the arrival transitions at the restaurant and the customer.
        """
        if self.state == IDLE:
            if not self.orders:
                return
            self.state = PICKING_UP
            self.reset_movement()
            order = self.orders[0]
            order.status = "picking_up"
            self._record("accepted_order", order.id)

        target = self._current_target()
        if target is None:
            logger.warning("Rider %s has no target; resetting to idle",
                           self.rider_id)
            self.state = IDLE
            self.reset_movement()
            return

        if self._target != target:
            self._target = target
            self.path = _planner.plan(self.position, target)
            self.special_case = self.x == target[0] or self.y == target[1]

        self._advance()
        if self.position == target:
            self._on_arrival()

    def _current_order(self):
        """Return the order currently being served, or None."""
        return self.orders[0] if self.orders else None

    def _current_target(self):
        """Return the coordinate the rider is heading to, or None."""
        order = self._current_order()
        if order is None:
            return None
        if self.state == PICKING_UP:
            return order.res_pos
        if self.state == DELIVERING:
            return order.cus_pos
        return None

    def _advance(self) -> None:
        """Move along the planned waypoints by at most ``speed`` units."""
        remaining = self.speed
        while remaining > 0 and self.path:
            wx, wy = self.path[0]
            dx = wx - self.x
            dy = wy - self.y
            if dx == 0 and dy == 0:
                self.path.pop(0)
                continue
            if dx != 0:
                step = min(remaining, abs(dx))
                self.x += step if dx > 0 else -step
            else:
                step = min(remaining, abs(dy))
                self.y += step if dy > 0 else -step
            remaining -= step
            self.distance_travelled += step
            if (self.x, self.y) == (wx, wy):
                self.path.pop(0)

    def _on_arrival(self) -> None:
        """Apply state transitions when the rider reaches its target."""
        order = self._current_order()
        if order is None:
            self.state = IDLE
            self.reset_movement()
            return
        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.status = "delivering"
            self.reset_movement()
            self._record("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            self.orders.pop(0)
            order.complete(self._now())
            self.state = IDLE
            self.reset_movement()
            self._record("delivered_to_customer", order.id)

    def _now(self) -> int:
        """Return the current simulated tick."""
        return int(get_value("tick", 0))

    def _record(self, event: str, order_id) -> None:
        """Append a status-change entry to the rider's log."""
        self.log.append({
            "time": self._now(),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

"""Order data model with coordinate normalisation and lifecycle tracking."""

import math

from globalvar import get_value
from grid import to_cell_center


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order number.
        res_pos: Restaurant grid-cell centre (x, y).
        cus_pos: Customer grid-cell centre (x, y).
        status: "pending", "assigned", "picking_up", "delivering" or
            "completed".
        rider_id: Identifier of the assigned rider, or None.
        start_time: Tick at which the order was created.
        completion_time: Tick at which delivery finished, or None.
        time_taken: Ticks between creation and completion, or None.
        distance: Euclidean distance between restaurant and customer.
        efficiency: distance / time_taken (units per tick), or None.
    """

    _id_counter = 0

    def __init__(self, res_pos, cus_pos) -> None:
        """Create an order and convert its coordinates to cell centres.

        Args:
            res_pos: Raw restaurant coordinate (x, y), range 0-800.
            cus_pos: Raw customer coordinate (x, y), range 0-800.
        """
        Order._id_counter += 1
        self.id = Order._id_counter
        self.res_pos = to_cell_center(res_pos[0], res_pos[1])
        self.cus_pos = to_cell_center(cus_pos[0], cus_pos[1])
        self.status = "pending"
        self.rider_id = None
        self.start_time = int(get_value("tick", 0))
        self.completion_time = None
        self.time_taken = None
        self.distance = round(math.hypot(
            self.cus_pos[0] - self.res_pos[0],
            self.cus_pos[1] - self.res_pos[1]), 2)
        self.efficiency = None

    def assign_to(self, rider_id: int) -> None:
        """Record that the order has been assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick: int) -> None:
        """Mark the order as delivered and compute delivery efficiency.

        Args:
            tick: The simulated tick at which delivery finished.
        """
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = max(0, tick - self.start_time)
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = round(self.distance, 2)

    def to_dict(self) -> dict:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "efficiency": self.efficiency,
        }

"""Order management: creation, assignment and nearest-rider selection."""

import logging
import math

from globalvar import get_value
from grid import validate_coordinate
from order_model import Order
from rider import IDLE

logger = logging.getLogger("dispatch.order")


def find_best_rider_for_order(order) -> "Rider":
    """Select the rider best suited to serve an order.

    Strategy: closest rider first by Euclidean distance from the rider to
    the restaurant. Idle riders are preferred. If no rider is idle, the
    rider with the fewest assigned orders (then closest) is chosen to
    balance load.

    Args:
        order: The Order to be assigned.

    Returns:
        The selected Rider, or None if no riders exist.
    """
    riders = get_value("riders", []) or []
    if not riders:
        return None

    rest_x, rest_y = order.res_pos

    def distance(rider):
        return math.sqrt((rider.x - rest_x) ** 2 + (rider.y - rest_y) ** 2)

    idle = [r for r in riders if r.state == IDLE and not r.orders]
    if idle:
        return min(idle, key=distance)
    return min(riders, key=lambda r: (len(r.orders), distance(r)))


class OrderManager:
    """Creates, assigns and tracks delivery orders."""

    def create_order(self, rest_x, rest_y, cus_x, cus_y) -> Order:
        """Validate raw coordinates and create a pending order.

        Args:
            rest_x: Restaurant x coordinate (0-800).
            rest_y: Restaurant y coordinate (0-800).
            cus_x: Customer x coordinate (0-800).
            cus_y: Customer y coordinate (0-800).

        Returns:
            The newly created Order, also registered as pending.

        Raises:
            ValueError: If any coordinate is invalid.
        """
        values = [validate_coordinate(v)
                  for v in (rest_x, rest_y, cus_x, cus_y)]
        order = Order((values[0], values[1]), (values[2], values[3]))
        get_value("orders").append(order)
        get_value("pending_orders").append(order)
        logger.info("Created order %s", order.id)
        return order

    def pending_orders(self) -> list:
        """Return the list of orders not yet assigned to a rider."""
        return list(get_value("pending_orders", []))

    def assign(self, order):
        """Assign an order to the best rider and queue it on that rider.

        Args:
            order: The Order to assign.

        Returns:
            The Rider that received the order, or None if no rider exists
            (the order then stays pending).
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            logger.warning("No rider available for order %s", order.id)
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        pending = get_value("pending_orders")
        if order in pending:
            pending.remove(order)
        return rider

    def assign_all_pending(self) -> list:
        """Try to assign every pending order; return the riders used."""
        used = []
        for order in self.pending_orders():
            rider = self.assign(order)
            if rider is not None:
                used.append(rider)
        return used

    def completed_orders(self) -> list:
        """Return all orders that have been delivered."""
        return [o for o in get_value("orders", []) if o.status == "completed"]

"""System-level management: funds, riders, ticks, recruitment and persistence."""

import json
import logging
import os

import globalvar
from globalvar import get_value, set_value
from rider import IDLE, Rider

logger = logging.getLogger("dispatch.system")


def init_system() -> None:
    """Initialise all global variables to their startup defaults."""
    globalvar._init()
    logger.info("System initialised")


class SystemManager:
    """Manages funds, the rider roster, the simulated clock and recruitment."""

    @property
    def is_running(self) -> bool:
        """True while the main loop should keep running."""
        return bool(get_value("system_running", False))

    def stop(self) -> None:
        """Signal the main loop to terminate."""
        set_value("system_running", False)

    @property
    def funds(self) -> int:
        """Current system fund balance."""
        return int(get_value("system_funds", 0))

    def riders(self) -> list:
        """Return the rider list."""
        return get_value("riders", [])

    def idle_rider_count(self) -> int:
        """Return the number of riders free to accept new orders."""
        return sum(1 for r in self.riders() if r.is_idle())

    def create_rider(self) -> Rider:
        """Create a new rider with an auto-incremented ID and add it to the list.

        Returns:
            The newly created Rider.
        """
        counter = int(get_value("rider_id_counter", 0)) + 1
        set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders().append(rider)
        logger.info("Recruited rider %s", counter)
        return rider

    def recruit_if_needed(self, pending_count: int) -> list:
        """Recruit riders while funds allow and pending orders exceed idle riders.

        Each recruit costs ``recruit_cost`` from system funds.

        Args:
            pending_count: Number of orders waiting for a rider.

        Returns:
            List of newly created Rider objects (possibly empty).
        """
        created = []
        cost = int(get_value("recruit_cost", 300))
        while self.funds >= cost and pending_count > self.idle_rider_count():
            set_value("system_funds", self.funds - cost)
            created.append(self.create_rider())
        return created

    def tick(self) -> None:
        """Advance the simulation by one unit of time."""
        set_value("tick", int(get_value("tick", 0)) + 1)
        for rider in list(self.riders()):
            try:
                rider.move()
            except Exception:  # error recovery: keep the simulation alive
                logger.exception("Rider %s failed; recovering", rider.rider_id)
                rider.reset_movement()
                rider.state = IDLE if not rider.orders else rider.state
        for issue in self.check_consistency():
            logger.warning(issue)

    def check_consistency(self) -> list:
        """Check rider/order state consistency and repair simple faults.

        Returns:
            List of human readable descriptions of issues found.
        """
        issues = []
        for rider in self.riders():
            if rider.state not in (0, 1, 2):
                issues.append(f"Rider {rider.rider_id} has invalid state "
                              f"{rider.state}; reset to idle")
                rider.state = IDLE
                rider.reset_movement()
            if rider.state != IDLE and not rider.orders:
                issues.append(f"Rider {rider.rider_id} active without orders; "
                              "reset to idle")
                rider.state = IDLE
                rider.reset_movement()
            for order in list(rider.orders):
                if order.status == "completed":
                    issues.append(f"Completed order {order.id} still on "
                                  f"rider {rider.rider_id}; removed")
                    rider.orders.remove(order)
        return issues


def save_snapshot(path: str) -> str:
    """Write the current system state to a JSON file.

    Args:
        path: Destination file path.

    Returns:
        The path written.
    """
    data = {
        "system_funds": get_value("system_funds"),
        "tick": get_value("tick"),
        "rider_id_counter": get_value("rider_id_counter"),
        "riders": [{
            "rider_id": r.rider_id,
            "x": r.x,
            "y": r.y,
            "speed": r.speed,
            "state": r.state,
            "orders": [o.id for o in r.orders],
            "special_case": r.special_case,
        } for r in get_value("riders", [])],
        "orders": [o.to_dict() for o in get_value("orders", [])],
    }
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2)
    logger.info("Snapshot saved to %s", path)
    return path


def load_config(path: str) -> dict:
    """Load system parameters from a JSON configuration file.

    Recognised keys: "initial_funds", "recruit_cost", "rider_speed".
    Missing files are ignored and default values are kept.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        The parsed configuration dictionary (empty if none was loaded).
    """
    if not os.path.exists(path):
        return {}
    try:
        with open(path, "r", encoding="utf-8") as handle:
            config = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.error("Could not read config %s: %s", path, exc)
        return {}
    if "initial_funds" in config:
        set_value("system_funds", int(config["initial_funds"]))
    if "recruit_cost" in config:
        set_value("recruit_cost", int(config["recruit_cost"]))
    if "rider_speed" in config:
        set_value("rider_speed", int(config["rider_speed"]))
    return config

"""Status displays and ASCII map visualisation."""

from globalvar import get_value
from grid import CELL_SIZE, MAP_SIZE, cell_label
from rider import STATE_NAMES

# Map resolution: one character per 50 units. Streets (multiples of 100)
# fall on even indices and cell centres (x % 100 == 50) on odd indices, so
# the 8x8 cell grid is drawn with '#' street lines between the blocks.
STEP = CELL_SIZE // 2
CANVAS = MAP_SIZE // STEP + 1


def _idx(value) -> int:
    """Map a coordinate to a canvas index."""
    return min(CANVAS - 1, max(0, round(value / STEP)))


def _line_cells(a, b) -> list:
    """Return canvas cells on the line between two canvas indices."""
    dx = b[0] - a[0]
    dy = b[1] - a[1]
    steps = max(abs(dx), abs(dy))
    if steps == 0:
        return [a]
    cells = []
    for k in range(steps + 1):
        cells.append((round(a[0] + dx * k / steps),
                      round(a[1] + dy * k / steps)))
    return cells


class DisplayManager:
    """Produces text output for status, orders, logs and the map."""

    def render_map(self, riders=None, orders=None, route=None) -> str:
        """Render the ASCII city map.

        Symbols: 'R' rider, 'S' restaurant, 'C' customer, '#' street,
        '.' empty block, '*' route line.

        Args:
            riders: Riders to draw (defaults to all riders).
            orders: Orders whose restaurant/customer to draw (defaults to
                all uncompleted orders).
            route: Optional list of (x, y) points drawn as connected lines.

        Returns:
            Multi-line string containing the map and a legend.
        """
        if riders is None:
            riders = get_value("riders", []) or []
        if orders is None:
            orders = [o for o in get_value("orders", []) or []
                      if o.status != "completed"]

        n = CANVAS
        canvas = [["#" if (r % 2 == 0 or c % 2 == 0) else "."
                   for c in range(n)] for r in range(n)]

        def put(ci, ri, symbol):
            canvas[n - 1 - ri][ci] = symbol

        if route:
            indices = [(_idx(x), _idx(y)) for x, y in route]
            for a, b in zip(indices, indices[1:]):
                for cell in _line_cells(a, b):
                    put(cell[0], cell[1], "*")
            if len(indices) == 1:

            f"Riders: {len(riders)}",

        orders = get_value("orders", []) or []
        if not orders:

                    f"  [t={entry['time']:>4}] {entry['event']:<22} "
                    f"{order:<10} pos=({entry['x']}, {entry['y']}) "

            f"  Ticks simulated: {ticks}",
            f"  Order status: {order.status}",

            lines.append(f"  Time taken: {order.time_taken} ticks, "

"""Command parsing, shortcut handling, history and input validation."""

import logging

from grid import MAP_SIZE, validate_coordinate

try:  # enables arrow-key line editing where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - Windows without pyreadline
    readline = None

logger = logging.getLogger("dispatch.input")

COMMAND_NAMES = ("start", "status", "orders", "riders", "wait", "history",
                 "save", "help", "quit", "exit")

ALIASES = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "t": "wait",
    "hist": "history",
    "w": "save",
    "h": "help",
    "?": "help",
    "q": "quit",
    "e": "exit",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  (s)  Create an order: restaurant then customer
                            coordinates (0-800). Prompts if omitted.
  status               (st) Show system funds, riders and map.
  orders               (o)  Show all orders.
  riders               (r)  Show every rider status-change record.
  wait [n]             (t)  Advance simulated time by n ticks (default 1).
  history              (hist) Show command history.
  save [path]          (w)  Save a JSON snapshot of the system state.
  help                 (h)  Show this help.
  quit / exit          (q)  Leave the program.
  !!  repeat last command, !n repeat history entry n.
"""


class InputManager:
    """Reads, expands and parses user commands and coordinates."""

    def __init__(self, reader=None) -> None:
        """Create an input manager.

        Args:
            reader: Callable used to read a line (defaults to ``input``).
        """
        self._reader = reader or input
        self.history = []

    def read_line(self, prompt: str = "> ") -> str:
        """Read one raw line from the user."""
        return self._reader(prompt)

    def read_command(self, prompt: str = "> "):
        """Read a command line, expanding history shortcuts.

        Returns:
            The command string, or None for an empty line.

        Raises:
            ValueError: If a history shortcut cannot be resolved.
        """
        raw = self.read_line(prompt).strip()
        if not raw:
            return None
        if raw == "!!":
            if not self.history:
                raise ValueError("No previous command to repeat.")
            raw = self.history[-1]
        elif raw.startswith("!") and raw[1:].isdigit():
            index = int(raw[1:]) - 1
            if index < 0 or index >= len(self.history):
                raise ValueError(f"No history entry {raw[1:]}.")
            raw = self.history[index]
        self.history.append(raw)
        return raw

    @staticmethod
    def parse_command(line: str) -> tuple:
        """Split a command line into a canonical command and its arguments.

        Args:
            line: Raw command text.

        Returns:
            Tuple (command_name, argument_list).
        """
        parts = line.split()
        if not parts:
            return "", []
        name = parts[0].lower()
        return ALIASES.get(name, name), parts[1:]

    @staticmethod
    def validate_coordinates(values) -> list:
        """Validate that every value is an integer in the range 0-800.

        Args:
            values: Iterable of raw values.

        Returns:
            List of validated ints.

        Raises:
            ValueError: On the first invalid value.
        """
        return [validate_coordinate(v) for v in values]

    def parse_coordinates(self, tokens) -> list:
        """Parse four command-line tokens into validated coordinates.

        Raises:
            ValueError: If the count is wrong, a token is not an integer, or
                a value is outside 0-800.
        """
        if len(tokens) != 4:
            raise ValueError("expected 4 integer coordinates: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for token in tokens:
            try:
                values.append(int(token))
            except ValueError:
                raise ValueError(f"'{token}' is not an integer") from None
        return self.validate_coordinates(values)

    def prompt_coordinates(self):
        """Interactively ask for the four coordinates, re-prompting on errors.

        Returns:
            List of four validated ints, or None if the user cancelled by
            entering an empty line.
        """
        labels = ["Restaurant X", "Restaurant Y", "Customer X", "Customer Y"]
        values = []
        for label in labels:
            while True:
                raw = self.read_line(f"{label} (0-{MAP_SIZE}, blank to "
                                     "cancel): ").strip()
                if raw == "":
                    return None
                try:
                    values.append(self.parse_coordinates([raw])[0])
                    break
                except ValueError as exc:
                    print(f"Invalid value: {exc}. Please try again.")
        return values

"""Command-line entry point for the food delivery dispatch simulator."""

import logging
import os
import sys

from display import DisplayManager
from globalvar import get_value
from grid import MAX_ROUTE_TICKS
from input_manager import HELP_TEXT, InputManager
from order import OrderManager
from system import SystemManager, init_system, load_config, save_snapshot

logger = logging.getLogger("dispatch")

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
CONFIG_PATH = os.path.join(BASE_DIR, "config.json")
SNAPSHOT_PATH = os.path.join(BASE_DIR, "snapshot.json")
MAX_WAIT_TICKS = 1000

BANNER = """==============================================
 Food Delivery Order Dispatch Simulator (8x8 grid)
 Type 'help' for commands.
=============================================="""


class DispatchApp:
    """Ties together the managers and runs the interactive command loop."""

    def __init__(self, reader=None, writer=print) -> None:
        """Initialise system state and collaborators.

        Args:
            reader: Optional callable used to read input lines.
            writer: Callable used to output text.
        """
        init_system()
        load_config(CONFIG_PATH)
        self.system = SystemManager()
        self.orders = OrderManager()
        self.display = DisplayManager()
        self.inputs = InputManager(reader)
        self.out = writer
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "wait": self.cmd_wait,
            "history": self.cmd_history,
            "save": self.cmd_save,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self) -> None:
        """Run the main loop until the user quits or input ends."""
        self.out(BANNER)
        while self.system.is_running:
            try:
                line = self.inputs.read_command("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self.out("")
                break
            except ValueError as exc:
                self.out(f"Error: {exc}")
                continue
            if line is None:
                continue
            name, args = self.inputs.parse_command(line)
            handler = self.handlers.get(name)
            if handler is None:
                self.out(f"Unknown command '{name}'. Type 'help' for a list.")
                continue
            try:
                handler(args)
            except ValueError as exc:
                self.out(f"Input error: {exc}")
            except Exception:  # keep the system running on unexpected errors
                logger.exception("Unexpected error in command %s", name)
                self.out("An unexpected error occurred; system state kept.")
        self.out("Goodbye.")

    def cmd_start(self, args) -> None:
        """Create an order, recruit if needed, assign it and simulate delivery."""
        if args:
            coords = self.inputs.parse_coordinates(args)
        else:
            coords = self.inputs.prompt_coordinates()
        if coords is None:
            self.out("Order cancelled.")
            return

        order = self.orders.create_order(*coords)
        created = self.system.recruit_if_needed(
            len(self.orders.pending_orders()))
        for rider in created:
            self.out(f"New rider #{rider.rider_id} recruited "
                     f"(funds now {self.system.funds}).")

        rider = self.orders.assign(order)
        if rider is None:
            self.out("No rider available; order remains pending.")
            return

        route = [rider.position]
        ticks = 0
        while order.status != "completed" and ticks < MAX_ROUTE_TICKS:
            self.system.tick()
            ticks += 1
            route.append(rider.position)
        if order.status != "completed":
            self.out("Warning: delivery did not finish within the tick limit.")

        self.out(self.display.format_assignment(
            order, rider, bool(created), route, ticks))
        self.out(self.display.render_map(
            riders=self.system.riders(), orders=[order], route=route))

    def cmd_status(self, _args) -> None:
        """Show funds, riders and the current map."""
        self.out(self.display.format_status())
        self.out(self.display.render_map())

    def cmd_orders(self, _args) -> None:
        """Show every order."""
        self.out(self.display.format_orders())

    def cmd_riders(self, _args) -> None:
        """Show the status-change log for every rider."""
        self.out(self.display.format_rider_logs())

    def cmd_wait(self, args) -> None:
        """Advance simulated time by a number of ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise ValueError(f"'{args[0]}' is not an integer") from None
        if count < 1 or count > MAX_WAIT_TICKS:
            raise ValueError(f"tick count must be between 1 and "
                             f"{MAX_WAIT_TICKS}")
        for _ in range(count):
            self.system.tick()
        self.out(f"Advanced {count} tick(s); now at tick "
                 f"{get_value('tick', 0)}.")

    def cmd_history(self, _args) -> None:
        """Print the command history."""
        if not self.inputs.history:
            self.out("No commands yet.")
            return
        for index, command in enumerate(self.inputs.history, start=1):
            self.out(f"{index:>4}  {command}")

    def cmd_save(self, args) -> None:
        """Save a JSON snapshot of the system state."""
        path = args[0] if args else SNAPSHOT_PATH
        try:
            save_snapshot(path)
            self.out(f"Snapshot saved to {path}")
        except OSError as exc:
            self.out(f"Could not save snapshot: {exc}")

    def cmd_help(self, _args) -> None:
        """Print the command help."""
        self.out(HELP_TEXT)

    def cmd_quit(self, _args) -> None:
        """Stop the main loop."""
        self.system.stop()


def main() -> int:
    """Program entry point.

    Returns:
        Process exit code.
    """
    logging.basicConfig(
        level=logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    app = DispatchApp()
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

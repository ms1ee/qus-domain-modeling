"""Command-line interface for the Beijing Subway fare calculation system.

Handles argument parsing, real-time input validation, continuous query
processing and multi-level error reporting.
"""

import argparse
import re
import sys

from data_loader import DATA_FILE_PATH, DataLoadError, load_subway_data
from data_management import check_data_integrity, print_integrity_report
from pathfinding import (
    MODE_SHORTEST_DISTANCE,
    VALID_MODES,
    NoPathError,
    PathfindingConfig,
    plan_route,
)
from route_report import export_text, render_route
from subway_network import SubwayNetwork


# Accepted separators between the starting and ending station.
QUERY_SEPARATOR_PATTERN = re.compile(r"\s*[;；,，]\s*")

FORMAT_ERROR_MESSAGE = (
    "Input format incorrect, please use 'Starting Station; Ending Station' format"
)

HELP_TEXT = (
    "Commands:\n"
    "  <start>; <end>   query the route and fare between two stations\n"
    "                   (separators: ';' '；' ',' '，')\n"
    "  help             show this help message\n"
    "  quit / exit      leave the program"
)


class QueryFormatError(Exception):
    """Raised when a query string does not follow the 'start; end' format."""


def parse_query(text):
    """Split a query string into (starting station, ending station).

    Raises QueryFormatError when the input does not contain exactly two
    non-empty station names separated by an accepted separator.
    """
    parts = QUERY_SEPARATOR_PATTERN.split(text.strip())
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise QueryFormatError(FORMAT_ERROR_MESSAGE)
    return parts[0].strip(), parts[1].strip()


def parse_line_preference(value):
    """Convert the --line_preference argument into a list of line names."""
    if not value:
        return []
    raw = re.split(r"[,，]", value)
    return [item.strip() for item in raw if item.strip()]


def build_parser():
    """Create the argparse parser with all supported command-line options."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (CLI)."
    )
    parser.add_argument(
        "--output_file",
        default=None,
        help="append query results to this file",
    )
    parser.add_argument(
        "--format",
        choices=["text", "json"],
        default="text",
        help="output format for query results (default: text)",
    )
    parser.add_argument(
        "--mode",
        choices=list(VALID_MODES),
        default=MODE_SHORTEST_DISTANCE,
        help="path query mode (default: shortest_distance)",
    )
    parser.add_argument(
        "--max_transfers",
        type=int,
        default=None,
        help="maximum number of allowed transfers",
    )
    parser.add_argument(
        "--line_preference",
        default=None,
        help="preferred line(s), e.g. \"Line 1\" or \"Line 1,Line 2\"",
    )
    parser.add_argument(
        "--check_data",
        action="store_true",
        help="run the data integrity check and exit",
    )
    return parser


def answer_query(network, start_raw, end_raw, config, fmt, output_file=None,
                 out=print):
    """Validate one query, compute the route and print/export the report.

    Returns True when a route was successfully produced, False otherwise.
    """
    start = network.resolve_station(start_raw)
    if start is None:
        out(f"Starting station '{start_raw}' does not exist, please check the station name")
        return False
    end = network.resolve_station(end_raw)
    if end is None:
        out(f"Ending station '{end_raw}' does not exist, please check the station name")
        return False
    if start == end:
        out("Starting station cannot be the same as the ending station")
        return False

    try:
        route = plan_route(network, start, end, config)
    except NoPathError as exc:
        out(f"No route found: {exc}")
        return False
    except ValueError as exc:
        out(f"Fare or path calculation error: {exc}")
        return False
    except Exception as exc:  # defensive: never let one query crash the session
        out(f"Unexpected path calculation error: {exc}")
        return False

    try:
        report = render_route(route, fmt)
    except Exception as exc:
        out(f"Report generation error: {exc}")
        return False

    out(report)

    if output_file:
        try:
            export_text(report, output_file)
            out(f"Result appended to {output_file}")
        except OSError as exc:
            out(f"Could not write output file '{output_file}': {exc}")
            return False
    return True


def run_interactive(network, config, fmt, output_file=None,
                    input_func=input, out=print):
    """Run the continuous query loop until the user quits or input ends."""
    out("Beijing Subway Fare Calculator")
    out(HELP_TEXT)
    while True:
        try:
            raw = input_func("\nEnter query (Starting Station; Ending Station): ")
        except (EOFError, KeyboardInterrupt):
            out("")
            break

        text = (raw or "").strip()
        if not text:
            continue
        lowered = text.lower()
        if lowered in ("quit", "exit", "q"):
            break
        if lowered in ("help", "?"):
            out(HELP_TEXT)
            continue

        try:
            start_raw, end_raw = parse_query(text)
        except QueryFormatError as exc:
            out(str(exc))
            continue

        try:
            answer_query(network, start_raw, end_raw, config, fmt,
                         output_file, out)
        except Exception as exc:  # keep the session alive on unexpected errors
            out(f"Unexpected error while processing query: {exc}")
    out("Goodbye.")


def main(argv=None):
    """Program entry point. Returns the process exit code."""
    parser = build_parser()
    args = parser.parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except DataLoadError as exc:
        print(f"Error: could not load subway data: {exc}", file=sys.stderr)
        return 1

    try:
        network = SubwayNetwork(data)
    except Exception as exc:
        print(f"Error: could not build subway network: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        result = check_data_integrity(network)
        print_integrity_report(result)
        return 1 if result["errors"] else 0

    preferred = parse_line_preference(args.line_preference)
    known_lines = set(network.line_names())
    for line in preferred:
        if line not in known_lines:
            print(f"Warning: line '{line}' is not a known subway line")

    try:
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=preferred,
        )
    except ValueError as exc:
        parser.error(str(exc))
        return 2

    try:
        run_interactive(network, config, args.format, args.output_file)
    except KeyboardInterrupt:
        print("")
    return 0

"""Data loading utilities for the Beijing Subway network dataset.

The dataset lives in ``subway_data.json`` next to this module, so the
location is resolved relative to this file and works from any working
directory.
"""

import json
import os


_THIS_DIR = os.path.dirname(os.path.abspath(__file__))

# Absolute path of src/subway_data.json, valid from any working directory.
DATA_FILE_PATH = os.path.join(_THIS_DIR, "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read or parsed."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway dataset file.

    The returned dictionary has the structure::

        {"stations": [{"name": str, "lines": [str]}],
         "connections": [{"source": str, "dest": str,
                          "distance": int, "line": str}]}

    Missing top-level keys are replaced by empty lists.
    """
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except FileNotFoundError as exc:
        raise DataLoadError(f"data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataLoadError(f"cannot read data file: {exc}") from exc

    if not isinstance(data, dict):
        raise DataLoadError("data file must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data

"""Graph data structures used to model the subway network.

The graph is an undirected, weighted graph stored as adjacency lists.
Vertices are station numbers assigned by the caller (see
``subway_network.SubwayNetwork``); edges carry the inter-station distance
in meters as their weight and the line on which the connection runs.
"""


class Edge:
    """An undirected weighted edge between two stations.

    Attributes:
        source: station number of one endpoint.
        dest: station number of the other endpoint.
        distance: inter-station distance in meters (also exposed as ``weight``).
        line: name of the subway line operating this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by path algorithms (equal to the distance)."""
        return self.distance

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"vertex {vertex!r} is not an endpoint of {self!r}")

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"


class Graph:
    """Weighted undirected graph with adjacency lists.

    Attributes:
        adj: dict mapping station number -> list of incident Edge objects.
        station_lines: dict mapping station name -> list of line names.
    """

    def __init__(self):
        self.adj = {}
        self.station_lines = {}
        self._edges = []

    # ----- vertex management -------------------------------------------
    def add_node(self, vertex):
        """Add a vertex if it does not exist yet."""
        if vertex not in self.adj:
            self.adj[vertex] = []

    def has_vertex(self, vertex):
        """Return True if the vertex exists in the graph."""
        return vertex in self.adj

    def vertices(self):
        """Return the list of all vertices."""
        return list(self.adj.keys())

    # ----- station metadata --------------------------------------------
    def set_station_lines(self, station_name, lines):
        """Store the list of lines that serve a station (keyed by name)."""
        self.station_lines[station_name] = list(lines)

    def get_station_lines(self, station_name):
        """Return a copy of the line list for a station name."""
        return list(self.station_lines.get(station_name, []))

    # ----- edge management ---------------------------------------------
    def add_edge(self, edge):
        """Insert an undirected edge, creating missing endpoints."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self.adj[edge.dest].append(edge)
        self._edges.append(edge)

    def remove_edge(self, edge):
        """Remove an edge object from the graph."""
        self.adj[edge.source].remove(edge)
        if edge.dest != edge.source:
            self.adj[edge.dest].remove(edge)
        self._edges.remove(edge)

    def adjacent(self, vertex):
        """Return a copy of the list of edges incident to ``vertex``."""
        return list(self.adj.get(vertex, []))

    def edges_between(self, u, v):
        """Return all edges joining vertices ``u`` and ``v``."""
        result = []
        for edge in self.adj.get(u, []):
            if (edge.source == u and edge.dest == v) or (edge.source == v and edge.dest == u):
                result.append(edge)
        return result

    def edges(self):
        """Return a copy of all edges (each undirected edge listed once)."""
        return list(self._edges)

    # ----- statistics --------------------------------------------------
    @property
    def num_vertices(self):
        return len(self.adj)

    @property
    def num_edges(self):
        return len(self._edges)

"""Binary min-heap priority queue used by Dijkstra's algorithm.

Each entry is stored as ``(priority, sequence, item)``. The sequence number
keeps insertion order for equal priorities and guarantees that items are
never compared directly. Insert and delete-min run in O(log n).
"""


class MinPQ:
    """A minimum priority queue implemented as a binary heap."""

    def __init__(self):
        self._heap = []
        self._seq = 0

    def __len__(self):
        return len(self._heap)

    def is_empty(self):
        """Return True when the queue holds no entries."""
        return not self._heap

    def min(self):
        """Return ``(priority, item)`` of the smallest entry without removing it."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        top = self._heap[0]
        return top[0], top[2]

    def insert(self, priority, item):
        """Insert an item with the given priority."""
        self._heap.append((priority, self._seq, item))
        self._seq += 1
        self._swim(len(self._heap) - 1)

    def del_min(self):
        """Remove and return ``(priority, item)`` of the smallest entry."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        heap = self._heap
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sink(0)
        return top[0], top[2]

    # ----- heap helpers -------------------------------------------------
    def _less(self, i, j):
        a = self._heap[i]
        b = self._heap[j]
        return (a[0], a[1]) < (b[0], b[1])

    def _swap(self, i, j):
        self._heap[i], self._heap[j] = self._heap[j], self._heap[i]

    def _swim(self, k):
        """Move the entry at index k up until the heap order holds."""
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._swap(k, parent)
                k = parent
            else:
                break

    def _sink(self, k):
        """Move the entry at index k down until the heap order holds."""
        n = len(self._heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and self._less(right, left):
                child = right
            if not self._less(child, k):
                break
            self._swap(k, child)
            k = child

"""Subway network model.

``SubwayNetwork`` builds the weighted undirected ``Graph`` from the dataset,
assigns station numbers, keeps the station-name/number mappings and the
ordered station sequence of every line, and offers data update operations
(adding lines, modifying stations, updating distances).
"""

from data_loader import load_subway_data
from models import Edge, Graph


def _clean_name(value):
    """Collapse whitespace and convert the value to a clean string."""
    if value is None:
        return ""
    return " ".join(str(value).split())


def _unique_clean(values):
    """Clean a list of strings and drop empty and duplicate entries."""
    result = []
    for value in values:
        cleaned = _clean_name(value)
        if cleaned and cleaned not in result:
            result.append(cleaned)
    return result


def _normalize_key(name):
    """Normalization used for station name matching (case/space insensitive)."""
    return "".join(str(name).split()).lower()


class SubwayNetwork:
    """Graph-based model of the Beijing Subway network.

    Attributes:
        graph: the ``Graph`` instance (vertices are station numbers).
        station_to_number: dict station name -> station number.
        number_to_station: dict station number -> station name.
        line_stations: dict line name -> ordered list of station names.
        build_warnings: list of non-fatal problems found while loading data.
    """

    def __init__(self, data=None):
        if data is None:
            data = load_subway_data()
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        self.line_stations = {}
        self.build_warnings = []
        self._next_number = 0
        self._build(data)

    # ----- construction -------------------------------------------------
    def _build(self, data):
        """Populate stations, lines and edges from the raw dataset dict."""
        for record in data.get("stations") or []:
            if not isinstance(record, dict):
                self.build_warnings.append(f"malformed station record skipped: {record!r}")
                continue
            name = _clean_name(record.get("name"))
            if not name:
                self.build_warnings.append("station record without a name was skipped")
                continue
            lines = _unique_clean(record.get("lines") or [])
            if name in self.station_to_number:
                existing = self.graph.get_station_lines(name)
                self.build_warnings.append(
                    f"duplicate station record '{name}', line lists were merged"
                )
                lines = existing + [line for line in lines if line not in existing]
            self._ensure_station(name)
            self.graph.set_station_lines(name, lines)

        for conn in data.get("connections") or []:
            try:
                src = _clean_name(conn["source"])
                dst = _clean_name(conn["dest"])
                line = _clean_name(conn["line"])
                distance = int(conn["distance"])
            except (KeyError, TypeError, ValueError):
                self.build_warnings.append(f"malformed connection skipped: {conn!r}")
                continue
            if not src or not dst or src == dst:
                self.build_warnings.append(f"invalid connection skipped: {conn!r}")
                continue

            for name in (src, dst):
                if name not in self.station_to_number:
                    self.build_warnings.append(
                        f"connection references unlisted station '{name}'; "
                        "it was added without line information"
                    )
                    self._ensure_station(name)

            src_num = self.station_to_number[src]
            dst_num = self.station_to_number[dst]
            self.graph.add_edge(Edge(src_num, dst_num, distance, line))
            self._register_line_station(line, src)
            self._register_line_station(line, dst)

    def _ensure_station(self, name):
        """Return the number of a station, creating it when needed."""
        number = self.station_to_number.get(name)
        if number is not None:
            return number
        number = self._next_number
        self._next_number += 1
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.graph.add_node(number)
        self.graph.set_station_lines(name, [])
        return number

    def _register_line_station(self, line, name):
        """Record that a station is served by a line, in line order of appearance."""
        members = self.line_stations.setdefault(line, [])
        if name not in members:
            members.append(name)
        if line not in self.graph.get_station_lines(name):
            self.build_warnings.append(
                f"station '{name}' is on line '{line}' in connections but its "
                "line list did not include it; it was added"
            )
            current = self.graph.get_station_lines(name)
            current.append(line)
            self.graph.set_station_lines(name, current)

    # ----- lookup helpers -----------------------------------------------
    def resolve_station(self, query):
        """Standardize a user-supplied station name.

        Matching ignores case and whitespace, and accepts names with or
        without the trailing '站' (station) character. Returns the official
        station name, or None if no station matches.
        """
        if query is None:
            return None
        key = _normalize_key(query)
        if not key:
            return None

        index = {}
        for name in self.station_to_number:
            index.setdefault(_normalize_key(name), name)

        if key in index:
            return index[key]
        if key.endswith("站"):
            alternative = key[:-1]
        else:
            alternative = key + "站"
        if alternative in index:
            return index[alternative]
        return None

    def get_station_number(self, name):
        """Return the station number of an official station name."""
        if name not in self.station_to_number:
            raise ValueError(f"unknown station '{name}'")
        return self.station_to_number[name]

    def get_station_name(self, number):
        """Return the official station name of a station number."""
        return self.number_to_station[number]

    def station_names(self):
        """Return all station names sorted alphabetically."""
        return sorted(self.station_to_number)

    def line_names(self):
        """Return all known line names sorted alphabetically."""
        names = set(self.line_stations)
        for lines in self.graph.station_lines.values():
            names.update(lines)
        return sorted(names)

    def get_line_stations(self, line):
        """Return the ordered list of station names served by a line."""
        return list(self.line_stations.get(line, []))

    def transfer_stations(self):
        """Return sorted names of multi-line interchange stations."""
        return sorted(
            name for name, lines in self.graph.station_lines.items() if len(lines) > 1
        )

    # ----- data update operations --------------------------------------
    def add_line(self, line_name, station_names, distances):
        """Add a new line.

        Args:
            line_name: name of the new line; must not already exist.
            station_names: ordered list of station names on the line.
            distances: distances in meters between consecutive stations
                (length must be ``len(station_names) - 1``).
        Unknown stations are created; existing stations keep their numbers.
        """
        line = _clean_name(line_name)
        if not line:
            raise ValueError("line name must not be empty")
        if line in self.line_stations:
            raise ValueError(f"line '{line}' already exists")
        names = [_clean_name(name) for name in station_names]
        if len(names) < 2:
            raise ValueError("a line needs at least two stations")
        if any(not name for name in names):
            raise ValueError("station names must not be empty")
        if len(set(names)) != len(names):
            raise ValueError("station names on a line must be unique")
        if len(distances) != len(names) - 1:
            raise ValueError("expected one distance per pair of consecutive stations")
        dists = []
        for value in distances:
            distance = int(value)
            if distance <= 0:
                raise ValueError("distances must be positive")
            dists.append(distance)

        numbers = [self._ensure_station(name) for name in names]
        for name in names:
            current = self.graph.get_station_lines(name)
            if line not in current:
                current.append(line)
                self.graph.set_station_lines(name, current)
        for i, distance in enumerate(dists):
            self.graph.add_edge(Edge(numbers[i], numbers[i + 1], distance, line))
        self.line_stations[line] = list(names)

    def modify_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its list of lines.

        Returns the (possibly new) official name of the station.
        """
        canonical = self.resolve_station(name)
        if canonical is None:
            raise ValueError(f"unknown station '{name}'")
        number = self.station_to_number[canonical]
        target = canonical

        if new_name is not None:
            target = _clean_name(new_name)
            if not target:
                raise ValueError("new station name must not be empty")
            if target != canonical:
                if target in self.station_to_number:
                    raise ValueError(f"station '{target}' already exists")
                del self.station_to_number[canonical]
                self.station_to_number[target] = number
                self.number_to_station[number] = target
                self.graph.station_lines[target] = self.graph.station_lines.pop(canonical)
                for line, members in self.line_stations.items():
                    self.line_stations[line] = [
                        target if member == canonical else member for member in members
                    ]

        if lines is not None:
            self.graph.set_station_lines(target, _unique_clean(lines))
        return target

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance between two adjacent stations.

        Args:
            station_a, station_b: official station names.
            distance: new distance in meters (positive integer).
            line: optionally restrict the update to connections of one line.
        Returns the number of edges updated.
        """
        name_a = self.resolve_station(station_a)
        name_b = self.resolve_station(station_b)
        if name_a is None or name_b is None:
            raise ValueError("both stations must exist")
        value = int(distance)
        if value <= 0:
            raise ValueError("distance must be a positive number of meters")
        edges = self.graph.edges_between(
            self.station_to_number[name_a], self.station_to_number[name_b]
        )
        if line:
            edges = [edge for edge in edges if edge.line == line]
        if not edges:
            raise ValueError(f"no connection between '{name_a}' and '{name_b}'")
        for edge in edges:
            edge.distance = value
        return len(edges)

<file path="src/pathfinding.py">
"""Path calculation: Dijkstra-based shortest path search over the subway graph.

The search runs over states ``(station, line of last edge, transfers so far)``
so that transfers can be counted, limited and optimized. Each state is
ordered in a ``MinPQ`` (binary heap), which gives the O((V+E) log V) bound
for the usual case.

Query modes:
    shortest_distance  minimize the total distance in meters
    least_transfer     minimize the number of transfers, then the distance
    comprehensive      minimize distance plus a penalty per transfer
"""

from fare_calculator import Sum_money, calculate_estimated_time
from min_pq import MinPQ


MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance (meters) charged for each transfer in comprehensive mode.
TRANSFER_PENALTY_M = 1500.0
# Extra cost (meters) charged for riding a line outside the preferred lines.
LINE_PREFERENCE_PENALTY_M = 1000.0
# Safety limit on transfers when no maximum is configured.
MAX_TRANSFER_CAP = 20


class NoPathError(Exception):
    """Raised when no route exists under the given constraints."""


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: one of VALID_MODES.
        max_transfers: maximum number of transfers, or None for no limit.
        preferred_lines: list of preferred line names (may be empty).
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"unsupported mode '{mode}'")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer")
            if max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = [
            str(line).strip() for line in (preferred_lines or []) if str(line).strip()
        ]

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, "
                f"max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class ShortestPath:
    """Dijkstra's algorithm from a single source station.

    Attributes:
        dist_to: dict station number -> physical distance (meters) along the
            selected path from the source.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"source {source!r} is not a vertex of the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._vertex_state = {}   # station number -> best finalized state
        self._parent = {}         # finalized state -> (parent state, edge)
        self._run()

    # ----- cost model ---------------------------------------------------
    def _key(self, cost, transfers):
        """Return the priority key for a state according to the query mode."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, cost)
        if mode == MODE_COMPREHENSIVE:
            return (cost + TRANSFER_PENALTY_M * transfers,)
        return (cost,)

    # ----- Dijkstra -----------------------------------------------------
    def _run(self):
        cfg = self.config
        pq = MinPQ()
        start = (self.source, None, 0)
        start_key = self._key(0.0, 0)
        # best[state] = (key, distance, cost, parent state, edge used)
        best = {start: (start_key, 0.0, 0.0, None, None)}
        pq.insert(start_key, start)
        done = set()

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in done:
                continue
            key, dist, cost, parent, edge = best[state]
            done.add(state)
            self._parent[state] = (parent, edge)

            vertex, line, transfers = state
            if vertex not in self._vertex_state:
                self._vertex_state[vertex] = state
                self.dist_to[vertex] = dist

            for e in self.graph.adj.get(vertex, ()):
                nxt = e.other(vertex)
                new_transfers = transfers
                if line is not None and e.line != line:
                    new_transfers = transfers + 1
                if cfg.max_transfers is not None and new_transfers > cfg.max_transfers:
                    continue
                if new_transfers > MAX_TRANSFER_CAP:
                    continue

                nstate = (nxt, e.line, new_transfers)
                if nstate in done:
                    continue

                penalty = 0.0
                if cfg.preferred_lines and e.line not in cfg.preferred_lines:
                    penalty = LINE_PREFERENCE_PENALTY_M
                ncost = cost + e.distance + penalty
                ndist = dist + e.distance
                nkey = self._key(ncost, new_transfers)

                old = best.get(nstate)
                if old is None or nkey < old[0]:
                    best[nstate] = (nkey, ndist, ncost, state, e)
                    pq.insert(nkey, nstate)

    # ----- queries ------------------------------------------------------
    def has_path_to(self, target):
        """Return True if ``target`` is reachable from the source."""
        return target in self._vertex_state

    def path_with_edges(self, target):
        """Return ``(stations, edges)`` of the best path from source to target.

        ``stations`` is the list of station numbers from source to target and
        ``edges`` the list of Edge objects traversed between them.
        Raises NoPathError if the target is unreachable under the constraints.
        """
        if target not in self._vertex_state:
            raise NoPathError(f"no path to station {target!r}")
        state = self._vertex_state[target]
        vertices = []
        edges = []
        while True:
            vertices.append(state[0])
            parent, edge = self._parent[state]
            if parent is None:
                break
            edges.append(edge)
            state = parent
        vertices.reverse()
        edges.reverse()
        return vertices, edges

    def get_transfer_count(self, target):
        """Return the number of line changes on the best path to target."""
        if target not in self._vertex_state:
            raise NoPathError(f"no path to station {target!r}")
        return self._vertex_state[target][2]


class Route:
    """A computed route with all information needed for reporting.

    Attributes:
        start, end: official station names.
        config: the PathfindingConfig used.
        station_names: ordered list of station names from start to end.
        edges: list of Edge objects along the route.
        segments: list of dicts with keys line, from, to, stations, distance_m.
        transfer_stations: list of dicts with keys station, from_line, to_line.
        distance_m: total distance in meters.
        fare: fare in yuan.
        estimated_minutes: estimated travel time in minutes.
    """

    def __init__(self, start, end, config, station_names, edges, segments,
                 transfer_stations, distance_m, fare, estimated_minutes):
        self.start = start
        self.end = end
        self.config = config
        self.station_names = list(station_names)
        self.edges = list(edges)
        self.segments = list(segments)
        self.transfer_stations = list(transfer_stations)
        self.distance_m = distance_m
        self.fare = fare
        self.estimated_minutes = estimated_minutes

    @property
    def distance_km(self):
        return self.distance_m / 1000.0

    @property
    def transfer_count(self):
        return len(self.transfer_stations)

    @property
    def lines(self):
        """Unique line names involved, in the order they are ridden."""
        result = []
        for segment in self.segments:
            if segment["line"] not in result:
                result.append(segment["line"])
        return result


def plan_route(network, start_name, end_name, config=None):
    """Compute the route, fare and time estimate between two official stations.

    Raises ValueError for invalid stations or identical endpoints and
    NoPathError when no route satisfies the constraints.
    """
    cfg = config if config is not None else PathfindingConfig()
    if start_name == end_name:
        raise ValueError("starting station cannot be the same as the ending station")

    src = network.get_station_number(start_name)
    dst = network.get_station_number(end_name)
    sp = ShortestPath(network.graph, src, cfg)
    if not sp.has_path_to(dst):
        if cfg.max_transfers is not None:
            raise NoPathError(
                f"no route from '{start_name}' to '{end_name}' within "
                f"{cfg.max_transfers} transfer(s)"
            )
        raise NoPathError(f"no route from '{start_name}' to '{end_name}'")

    numbers, edges = sp.path_with_edges(dst)
    names = [network.get_station_name(num) for num in numbers]

    segments = []
    transfer_stations = []
    for i, edge in enumerate(edges):
        if i == 0 or edges[i - 1].line != edge.line:
            if i > 0:
                transfer_stations.append({
                    "station": names[i],
                    "from_line": edges[i - 1].line,
                    "to_line": edge.line,
                })
            segments.append({
                "line": edge.line,
                "from": names[i],
                "to": names[i + 1],
                "stations": 2,
                "distance_m": edge.distance,
            })
        else:
            segment = segments[-1]
            segment["to"] = names[i + 1]
            segment["stations"] += 1
            segment["distance_m"] += edge.distance

    distance_m = sum(edge.distance for edge in edges)
    transfer_count = sp.get_transfer_count(dst)
    fare = Sum_money(distance_m)
    estimated = calculate_estimated_time(distance_m, transfer_count)

    return Route(
        start=start_name,
        end=end_name,
        config=cfg,
        station_names=names,
        edges=edges,
        segments=segments,
        transfer_stations=transfer_stations,
        distance_m=distance_m,
        fare=fare,
        estimated_minutes=estimated,
    )

"""Fare calculation based on the Beijing Subway fare structure.

Fare tiers (distance in meters, lower bound inclusive):
    0 - 6 km        3 yuan
    6 - 12 km       4 yuan
    12 - 22 km      5 yuan
    22 - 32 km      6 yuan
    over 32 km      7 yuan, plus 1 yuan for every further 20 km
Fractional meters are rounded up to the next whole meter before lookup.
"""

import math


# (lower bound in meters, fare in yuan) for the basic and progressive tiers.
FARE_TIERS = (
    (0, 3),
    (6000, 4),
    (12000, 5),
    (22000, 6),
)
EXTENDED_THRESHOLD_M = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP_M = 20000

# Time model used for the estimated travel time.
AVERAGE_SPEED_KMH = 35.0
BOARDING_TIME_MIN = 2.0
TRANSFER_TIME_MIN = 4.0


def Sum_money(distance):
    """Return the subway fare in yuan for a travel distance in meters.

    Raises ValueError for negative or non-numeric distances.
    """
    try:
        meters = float(distance)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"distance must be a number of meters: {distance!r}") from exc
    if math.isnan(meters) or math.isinf(meters):
        raise ValueError("distance must be a finite number")
    if meters < 0:
        raise ValueError("distance must not be negative")

    whole_meters = math.ceil(meters)

    fare = FARE_TIERS[0][1]
    for lower_bound, price in FARE_TIERS:
        if whole_meters >= lower_bound:
            fare = price

    if whole_meters >= EXTENDED_THRESHOLD_M:
        extra_steps = (whole_meters - EXTENDED_THRESHOLD_M) // EXTENDED_STEP_M
        fare = EXTENDED_BASE_FARE + extra_steps
    return int(fare)


def calculate_estimated_time(distance, transfer_count):
    """Estimate the travel time in minutes.

    The estimate combines running time at an average speed, a boarding
    allowance, and a fixed waiting/walking time for every transfer.
    Returns a float rounded to one decimal place.
    """
    try:
        meters = float(distance)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"distance must be a number of meters: {distance!r}") from exc
    if meters < 0:
        raise ValueError("distance must not be negative")
    if transfer_count < 0:
        raise ValueError("transfer count must not be negative")

    running = (meters / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    total = running + BOARDING_TIME_MIN + transfer_count * TRANSFER_TIME_MIN
    return round(total, 1)

import json


def summarize_route(route):

        "start": route.start,

        "station_count": len(route.station_names),

        "transfer_count": route.transfer_count,

    return {

            "end": route.end,
            "mode": cfg.mode,

        "fare": route.fare,
        "estimated_time_min": route.estimated_minutes,

        "transfers": [dict(item) for item in route.transfer_stations],
        "lines": list(route.lines),
        "segments": [dict(item) for item in route.segments],
    }


def format_text(route):
    """Return a human-readable console report for a route."""
    cfg = route.config
    max_t = "unlimited" if cfg.max_transfers is None else str(cfg.max_transfers)
    preference = ", ".join(cfg.preferred_lines) if cfg.preferred_lines else "none"

    out = []
    out.append("=" * 60)
    out.append("Beijing Subway Route Report")
    out.append("=" * 60)
    out.append(f"From: {route.start}")
    out.append(f"To: {route.end}")
    out.append(f"Mode: {cfg.mode} | Max transfers: {max_t} | Line preference: {preference}")
    out.append("")

    out.append("Line segments:")
    for index, segment in enumerate(route.segments, 1):
        out.append(
            f"  {index}. [{segment['line']}] {segment['from']} -> {segment['to']} "
            f"({segment['stations']} stations, {segment['distance_m']} m)"
        )
    out.append("")

    out.append(f"Station sequence ({len(route.station_names)} stations):")
    for index, name in enumerate(route.station_names, 1):
        out.append(f"  {index:>3}. {name}")
    out.append("")

    out.append(f"Total distance: {route.distance_m} m ({route.distance_km:.2f} km)")
    out.append(f"Transfers: {route.transfer_count}")
    if route.transfer_stations:
        for item in route.transfer_stations:
            out.append(
                f"  - Transfer at {item['station']}: {item['from_line']} -> {item['to_line']}"
            )
    else:
        out.append("  No transfer required")
    out.append(f"Lines involved: {', '.join(route.lines)}")
    out.append(f"Estimated travel time: {route.estimated_minutes:.1f} min")
    out.append(f"Fare: {route.fare} yuan")
    out.append("=" * 60)
    return "\n".join(out)


def format_json(route):
    """Return the route report as a pretty-printed JSON string."""
    return json.dumps(route_to_dict(route), ensure_ascii=False, indent=2)


def render_route(route, fmt="text"):
    """Render a route in the requested format ('text' or 'json')."""

        return format_text(route)
    if fmt == "json":
        return format_json(route)
    raise ValueError(f"unsupported output format '{fmt}'")


def export_text(content, path):
    """Append a rendered report to a file, creating it when necessary."""
    with open(path, "a", encoding="utf-8") as fh:
        fh.write(content)
        fh.write("\n\n")

"""Data integrity checks for the subway network.

Checks coverage of the main lines, connection integrity and distance
accuracy. Results are returned as a dictionary with ``errors``,
``warnings`` and ``stats`` entries.
"""


# Main lines the dataset is expected to cover.
REQUIRED_MAIN_LINES = (
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
)

# Representative hub stations that should be present in the network.
REPRESENTATIVE_STATIONS = (
    "西单",
    "国贸",
    "西直门",
    "东直门",
    "天安门东",
)

# Plausible range for a single inter-station distance in meters.
MIN_PLAUSIBLE_DISTANCE_M = 1
MAX_PLAUSIBLE_DISTANCE_M = 10000


def _connected_components(graph):
    """Return a list of connected components, each a list of station numbers."""
    seen = set()
    components = []
    for start in graph.vertices():
        if start in seen:
            continue
        component = []
        stack = [start]
        seen.add(start)
        while stack:
            current = stack.pop()
            component.append(current)
            for edge in graph.adj.get(current, []):
                nxt = edge.other(current)
                if nxt not in seen:
                    seen.add(nxt)
                    stack.append(nxt)
        components.append(component)
    return components


def check_data_integrity(network):
    """Run all integrity checks on a SubwayNetwork.

    Returns a dict::

        {"errors": [str], "warnings": [str],
         "stats": {"stations": int, "connections": int,
                   "lines": int, "transfer_stations": int}}
    """
    errors = []
    warnings = []
    graph = network.graph
    lines_present = set(network.line_names())

    # 1. Coverage of main lines and representative stations.
    for line in REQUIRED_MAIN_LINES:
        if line not in lines_present:
            warnings.append(f"coverage: main line '{line}' is not present in the data")
        elif len(network.get_line_stations(line)) < 2:
            warnings.append(f"coverage: main line '{line}' has fewer than two stations")
    for station in REPRESENTATIVE_STATIONS:
        if network.resolve_station(station) is None:
            warnings.append(f"coverage: representative station '{station}' is missing")

    # 2. Station-level checks.
    for name in network.station_names():
        if not graph.get_station_lines(name):
            warnings.append(f"station '{name}' has no line information")

    # 3. Connection and distance checks.
    seen_pairs = {}
    for edge in graph.edges():
        src_name = network.get_station_name(edge.source)
        dst_name = network.get_station_name(edge.dest)
        label = f"connection '{src_name}' - '{dst_name}' ({edge.line})"

        if not edge.line:
            errors.append(f"{label}: missing line name")
        if not isinstance(edge.distance, int) or edge.distance <= 0:
            errors.append(f"{label}: distance must be a positive integer, got {edge.distance!r}")
        elif edge.distance < MIN_PLAUSIBLE_DISTANCE_M or edge.distance > MAX_PLAUSIBLE_DISTANCE_M:
            warnings.append(
                f"{label}: distance {edge.distance} m is outside the plausible range "
                f"({MIN_PLAUSIBLE_DISTANCE_M}-{MAX_PLAUSIBLE_DISTANCE_M} m)"
            )

        for name in (src_name, dst_name):
            if edge.line not in graph.get_station_lines(name):
                warnings.append(
                    f"connection consistency: '{name}' is not listed as serving '{edge.line}'"
                )

        key = (min(edge.source, edge.dest), max(edge.source, edge.dest), edge.line)
        if key in seen_pairs:
            warnings.append(f"duplicate connection: {label}")
        else:
            seen_pairs[key] = edge

    # 4. Connectivity of the network.
    components = _connected_components(graph)
    if len(components) > 1:
        for component in components:
            names = sorted(network.get_station_name(num) for num in component)
            preview = ", ".join(names[:5])
            warnings.append(
                f"connectivity: a group of {len(names)} station(s) is not connected "
                f"to the rest of the network ({preview})"
            )

    # 5. Problems detected while loading the raw data.
    warnings.extend(f"loading: {message}" for message in network.build_warnings)

    stats = {
        "stations": graph.num_vertices,
        "connections": graph.num_edges,
        "lines": len(lines_present),
        "transfer_stations": len(network.transfer_stations()),
    }
    return {"errors": errors, "warnings": warnings, "stats": stats}


def print_integrity_report(result, out=print):
    """Print the result of check_data_integrity in a readable form."""
    stats = result["stats"]
    out("Data integrity check")
    out(f"  Stations: {stats['stations']}")
    out(f"  Connections: {stats['connections']}")
    out(f"  Lines: {stats['lines']}")
    out(f"  Transfer stations: {stats['transfer_stations']}")
    for message in result["errors"]:
        out(f"ERROR: {message}")
    for message in result["warnings"]:
        out(f"WARNING: {message}")
    if result["errors"]:
        out(f"Result: FAILED ({len(result['errors'])} error(s), "
            f"{len(result['warnings'])} warning(s))")
    else:
        out(f"Result: PASSED ({len(result['warnings'])} warning(s))")

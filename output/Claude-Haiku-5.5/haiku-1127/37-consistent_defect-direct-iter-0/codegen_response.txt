# Special lottery persons referenced by the lottery rule engine.

    "first",
    "second",
    "third",
    "fourth",
    "fifth",
    "sixth",
    "seventh",
    "eighth",
    "ninth",
    "tenth",

def normalize(text: str | None) -> str:
    """Return a comparison key: whitespace collapsed, case-insensitive."""
    return " ".join((text or "").split()).casefold()


def same_text(first: str | None, second: str | None) -> bool:
    """Case-insensitive, whitespace-tolerant equality check."""
    return normalize(first) == normalize(second)


@dataclass(frozen=True)
class User:
    """A participant in the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None


def prize_level_rank(name: str | None) -> int:
    """Return the level rank of a prize from its name (1 = highest).

    Names containing an ordinal word such as "First", "Second" or "Third"
    are ranked by that ordinal. Other names rank after all ordinal prizes.
    """
    lowered = (name or "").lower()
    for rank, word in enumerate(_ORDINAL_WORDS, start=1):
        if re.search(r"\b" + word + r"\b", lowered):
            return rank
    return len(_ORDINAL_WORDS) + 1

from __future__ import annotations

import os
import re

from lottery_system.defaults import DEFAULT_USERS
from lottery_system.models import User, normalize

# Accepts ASCII and full-width comma / semicolon as the separator.
_SEPARATOR_PATTERN = re.compile(r"[,;\uff0c\uff1b]")
_ENCODINGS = ("utf-8-sig", "gbk")


class ListManager:
    """Loads, validates and summarizes participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in default participant list and any errors."""
        users = [User(name, department) for name, department in DEFAULT_USERS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is "Name,Department Group"."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file into users and a list of error messages.

        Lines that cannot be parsed produce a "Format Error" message naming
        the line number and are excluded from the returned users.
        """
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        lines, error = self._read_lines(path)
        if error is not None:
            return [], [error]

        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = _SEPARATOR_PATTERN.split(text, maxsplit=1)
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{text}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    @staticmethod
    def _read_lines(path: str) -> tuple[list[str], str | None]:
        """Read file lines trying several encodings. Returns (lines, error)."""
        last_error: Exception | None = None
        for encoding in _ENCODINGS:
            try:
                with open(path, "r", encoding=encoding) as handle:
                    return handle.read().splitlines(), None
            except UnicodeDecodeError as exc:
                last_error = exc
            except OSError as exc:
                return [], f"File Error: cannot read '{path}': {exc}"
        return [], f"File Error: unsupported encoding in '{path}': {last_error}"

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid users, duplicate entries).

        The first occurrence of each name is kept as valid; later occurrences
        with the same name are reported as duplicates.
        """
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to number of people."""
        display_names: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            key = normalize(user.department)
            display = display_names.setdefault(key, user.department.strip())
            counts[display] = counts.get(display, 0) + 1
        return counts

"""Prize library management and lottery sequence ordering."""

from __future__ import annotations

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize, normalize


class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize already exists and was skipped: {name}")
                continue
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(f"Cannot add preset prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize to the end of the lottery sequence.

        Raises ValueError for an empty name, a non-positive quantity or a
        duplicate prize name.
        """
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name cannot be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        if self._find(clean_name) is not None:
            raise ValueError(f"a prize named '{clean_name}' already exists")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = (group or "").strip()
            if cleaned and normalize(cleaned) not in {normalize(g) for g in groups}:
                groups.append(cleaned)

        drawer = (associated_drawer or "").strip() or None
        self._prizes.append(Prize(clean_name, quantity, groups, drawer))

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        ``order`` contains 1-based current positions in the desired new order.
        It must be a permutation of 1..N; otherwise nothing changes and False
        is returned.
        """
        try:
            valid = sorted(order) == list(range(1, len(self._prizes) + 1))
        except TypeError:
            return False
        if not valid:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

    def _find(self, name: str) -> Prize | None:
        key = normalize(name)
        for prize in self._prizes:
            if normalize(prize.name) == key:
                return prize
        return None

"""Winner recording, ordered result summaries and CSV/txt export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import date, datetime

from lottery_system.models import Prize, User, prize_level_rank


@dataclass(frozen=True)
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    sequence: int
    winning_time: str


def _prize_name(prize: Prize | str) -> str:
    return prize if isinstance(prize, str) else prize.name


def _ordered_prize_names(prizes: list[Prize] | list[str]) -> list[str]:
    """Prize names ordered by level (high to low), then by addition order."""
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: (prize_level_rank(_prize_name(item[1])), item[0]))
    names: list[str] = []
    for _, prize in indexed:
        name = _prize_name(prize)
        if name not in names:
            names.append(name)
    return names


class ResultManager:
    """Maintains the real-time winner list."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize | str) -> Winner:
        """Record a winner for a prize and return the winner record."""
        winner = Winner(
            user=user,
            prize_name=_prize_name(prize),
            sequence=len(self._winners) + 1,
            winning_time=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name.

        Keys are ordered by prize level from high to low; each list is in
        winning time order.
        """
        names = _ordered_prize_names(prizes)
        grouped: dict[str, list[Winner]] = {name: [] for name in names}
        for winner in sorted(self._winners, key=lambda w: w.sequence):
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def _ordered_winners(self, prizes: list[Prize]) -> list[Winner]:
        grouped = self.get_winners_by_prize(prizes)
        ordered = [winner for winners in grouped.values() for winner in winners]
        known = set(grouped)
        leftovers = sorted(
            (w for w in self._winners if w.prize_name not in known),
            key=lambda w: w.sequence,
        )
        return ordered + leftovers

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt.

        The file is CSV with the header Name,Department,Prize Name,Winning Time.
        Returns (success, message).
        """
        filename = f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in self._ordered_winners(prizes):
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.winning_time,
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    normalize,
    same_text,
)
from lottery_system.services.result_manager import ResultManager, Winner


class LotteryManager:
    """Runs lottery draws for a fixed participant list while applying all rules.

    Rules:
    - Previous winners are never drawn again (regardless of prize type).
    - Members of a prize's avoid groups stay eligible but are drawn only
      after all other eligible members.
    - Lottery person "Li Bin": only previous winners are excluded; avoid
      groups are not applied.
    - Cheng Ru fallback: if the last two draws made by Chen Qixian / Jin Duo
      contained no Cheng Ru Group member, the next round drawn by Ma Jiaqi
      is limited to Cheng Ru Group members.
    """

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Cheng Ru membership flags of the most recent draws by Chen Qixian / Jin Duo.
        self.last_two_draw_results: list[bool] = []
        self._winner_keys: set[str] = set()
        self._forced_round: str | None = None
        self._rng = random.SystemRandom()

    # ----- helpers -------------------------------------------------------

    @staticmethod
    def _user_key(user: User) -> str:
        return normalize(user.name)

    def is_winner(self, user: User) -> bool:
        return self._user_key(user) in self._winner_keys

    @staticmethod
    def _is_li_bin(prize: Prize) -> bool:
        return same_text(prize.associated_drawer or "", LI_BIN)

    @staticmethod
    def _is_ma_jiaqi(prize: Prize) -> bool:
        return same_text(prize.associated_drawer or "", MA_JIAQI)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return same_text(user.department, CHENG_RU_GROUP)

    def _is_avoided(self, user: User, prize: Prize) -> bool:
        if self._is_li_bin(prize):
            return False
        return any(same_text(user.department, group) for group in prize.avoid_groups)

    def _fallback_active(self, prize: Prize) -> bool:
        """True if the Cheng Ru fallback limits this prize's pool."""
        if not self._is_ma_jiaqi(prize):
            return False
        if self._forced_round == prize.name:
            return True
        return len(self.last_two_draw_results) == 2 and not any(
            self.last_two_draw_results
        )

    # ----- round control -------------------------------------------------

    def start_round(self, prize: Prize) -> None:
        """Prepare a new round; decides whether the fallback applies to it."""
        self._forced_round = None
        if self._is_ma_jiaqi(prize) and self._fallback_active(prize):
            self._forced_round = prize.name
            # The protection is consumed by this round.
            self.last_two_draw_results = []

    def describe_rule(self, prize: Prize) -> str:
        """Human readable description of the rules applied to a prize round."""
        if self._is_li_bin(prize):
            return (
                "Only previous winners are excluded; avoid groups are not applied "
                "(lottery person: Li Bin)."
            )
        if self._fallback_active(prize):
            return (
                "Cheng Ru fallback active: only "
                f"{CHENG_RU_GROUP} members are eligible in this round."
            )
        parts = ["Previous winners are excluded."]
        if prize.avoid_groups:
            parts.append(
                "Members of avoid groups ("
                + ", ".join(prize.avoid_groups)
                + ") are drawn after other members."
            )
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}.")
        return " ".join(parts)

    # ----- eligibility and drawing --------------------------------------

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return all eligible users for a prize.

        Non-avoided members come first, followed by avoided members (who
        remain eligible but are drawn last).
        """
        available = [u for u in self.users if not self.is_winner(u)]

        if self._is_li_bin(prize):
            return available

        if self._fallback_active(prize):
            return [u for u in available if self._is_cheng_ru(u)]

        normal = [u for u in available if not self._is_avoided(u, prize)]
        avoided = [u for u in available if self._is_avoided(u, prize)]
        return normal + avoided

    def draw_one(self, prize: Prize) -> Winner | None:
        """Draw one winner for a prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None

        preferred = [u for u in pool if not self._is_avoided(u, prize)]
        candidates = preferred or pool
        user = self._rng.choice(candidates)

        winner = self.result_manager.add_winner(user, prize)
        self._winner_keys.add(self._user_key(user))
        self._record_draw_result(prize, user)
        return winner

    def _record_draw_result(self, prize: Prize, user: User) -> None:
        drawer = normalize(prize.associated_drawer or "")
        if drawer in (normalize(CHEN_QIXIAN), normalize(JIN_DUO)):
            self.last_two_draw_results.append(self._is_cheng_ru(user))
            del self.last_two_draw_results[:-2]

"""Command-line interaction for the team building lottery system."""

from __future__ import annotations

import re

from lottery_system.models import DRAWER_CHOICES, Prize, User, normalize
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager


def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive controller for the whole lottery process."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ----- basic helpers -------------------------------------------------

    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def run(self) -> int:
        self._banner()

        users = self._load_participants()
        if not users:
            print("No participants loaded. Exiting.")
            return 0
        self.users = users

        if not self._configure_prizes():
            print("Exiting without starting the lottery.")
            return 0

        self.lottery = LotteryManager(self.users)
        self._run_lottery()
        self._show_summary()
        self._offer_export()
        print("\nThank you for using the Smart Lottery System.")
        return 0

    @staticmethod
    def _banner() -> None:
        print("=" * 60)
        print("  Company Team Building Event - Smart Lottery System")
        print("=" * 60)

    # ----- step 1: participant list -------------------------------------

    def _load_participants(self) -> list[User] | None:
        while True:
            print("\n=== Step 1: Load Participant List ===")
            print("1. Load built-in default list")
            print("2. Upload custom txt list (format: Name,Department Group)")
            print("0. Quit")
            choice = self._ask("Select an option: ")

            if choice == "1":
                users, errors = self.list_manager.load_default_list()
            elif choice == "2":
                path = self._ask("Enter the txt file path: ")
                users, errors = self.list_manager.load_custom_list(path)
            elif choice == "0":
                return None
            else:
                print("Invalid option, please try again.")
                continue

            if not users:
                self._print_problems(errors, [])
                print("No valid participants found, please try again.")
                continue

            verified = self._verify_users(users, errors)
            if not verified:
                continue

            self._print_groups(verified)
            return verified

    def _verify_users(self, users: list[User], errors: list[str]) -> list[User] | None:
        """Detect duplicates and format errors; let the operator decide."""
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            return valid

        self._print_problems(errors, duplicates)
        while True:
            print("\nOptions:")
            print("  r - remove all invalid data and continue")
            print("  n - remove specific names manually and continue")
            print("  t - terminate the upload")
            action = self._ask("Select an action: ").lower()

            if action == "r":
                return valid
            if action == "n":
                raw = self._ask("Enter names to remove (comma separated): ")
                remove_keys = {normalize(n) for n in raw.split(",") if n.strip()}
                remaining = [u for u in valid if normalize(u.name) not in remove_keys]
                removed = len(valid) - len(remaining)
                print(f"Removed {removed} participant(s).")
                return remaining
            if action == "t":
                print("Upload terminated.")
                return None
            print("Unknown action, please choose r, n or t.")

    @staticmethod
    def _print_problems(errors: list[str], duplicates: list[User]) -> None:
        if errors:
            print(f"\nFound {len(errors)} format error(s):")
            for error in errors:
                print(f"  - {error}")
        if duplicates:
            print(f"\nFound {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  - Duplicate Name: {user.name} ({user.department})")

    def _print_groups(self, users: list[User]) -> None:
        info = self.list_manager.get_group_info(users)
        print(f"\nValid department groups ({len(info)}):")
        for department, count in info.items():
            print(f"  - {department}: {count} people")
        print(f"Total participants: {len(users)}")

    # ----- step 2: prize configuration ----------------------------------

    def _configure_prizes(self) -> bool:
        """Prize configuration menu. Returns True when the lottery should start."""
        while True:
            print("\n=== Step 2: Prize Configuration ===")
            print("1. Load preset prize library")
            print("2. Add custom prize")
            print("3. View prize list and rule settings")
            print("4. Adjust lottery sequence")
            print("5. View participant groups")
            print("6. Start lottery")
            print("0. Quit")
            choice = self._ask("Select an option: ")

            if choice == "1":
                for error in self.prize_manager.load_default_prizes():
                    print(error)
                print("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._view_prizes()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                self._print_groups(self.users)
            elif choice == "6":
                if not self.prize_manager.get_prizes():
                    print("Please configure at least one prize first.")
                    continue
                return True
            elif choice == "0":
                return False
            else:
                print("Invalid option, please try again.")

    def _group_names(self) -> list[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            print("Prize name cannot be empty.")
            return
        quantity = self._ask_positive_int("Quantity: ")
        drawer = self._ask_drawer()
        groups = self._ask_avoid_groups()
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Cannot add prize: {exc}")

    def _ask_positive_int(self, prompt: str) -> int:
        while True:
            text = self._ask(prompt)
            if text.isdigit() and int(text) > 0:
                return int(text)
            print("Please enter a positive whole number.")

    def _ask_drawer(self) -> str | None:
        print("Associated lottery person (optional):")
        for index, person in enumerate(DRAWER_CHOICES, start=1):
            print(f"  {index}. {person}")
        print("  Enter a number, type another name, or press Enter for none.")
        text = self._ask("Choice: ")
        if not text:
            return None
        if text.isdigit() and 1 <= int(text) <= len(DRAWER_CHOICES):
            return DRAWER_CHOICES[int(text) - 1]
        return text

    def _ask_avoid_groups(self) -> list[str]:
        groups = self._group_names()
        if not groups:
            return []
        print("Department groups:")
        for index, group in enumerate(groups, start=1):
            print(f"  {index}. {group}")
        text = self._ask(
            "Avoid group numbers or names (comma separated, Enter for none): "
        )
        selected: list[str] = []
        for token in (t.strip() for t in text.split(",")):
            if not token:
                continue
            match: str | None
            if token.isdigit() and 1 <= int(token) <= len(groups):
                match = groups[int(token) - 1]
            else:
                match = next((g for g in groups if normalize(g) == normalize(token)), None)
            if match is None:
                print(f"Unknown group ignored: {token}")
            elif match not in selected:
                selected.append(match)
        return selected

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nLottery sequence:")
        for index, prize in enumerate(prizes, start=1):
            self._print_prize_line(index, prize)

    @staticmethod
    def _print_prize_line(index: int, prize: Prize) -> None:
        drawer = prize.associated_drawer or "None"
        groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
        print(
            f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
            f"Lottery person: {drawer} | Avoid groups: {groups}"
        )

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes to reorder.")
            return
        self._view_prizes()
        print("Enter the new order as serial numbers (e.g. '2 1 3' or '2,1,3').")
        text = self._ask("New order (Enter to cancel): ")
        if not text:
            print("Reorder cancelled.")
            return
        tokens = [t for t in re.split(r"[\s,]+", text) if t]
        try:
            order = [int(t) for t in tokens]
        except ValueError:
            print("Invalid input: please enter numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ----- step 3: lottery ----------------------------------------------

    def _run_lottery(self) -> None:
        assert self.lottery is not None
        prizes = self.prize_manager.get_prizes()
        print("\n=== Step 3: Lottery ===")
        print("Before each draw: press Enter to draw, 'pause' to interrupt,")
        print("or 'terminate' to end the current round and skip it.")

        for round_no, prize in enumerate(prizes, start=1):
            print(f"\n===== Round {round_no}: {prize.name} =====")
            print(f"Quantity: {prize.quantity}")
            print(f"Rule: {self.lottery.describe_rule(prize)}")
            self.lottery.start_round(prize)

            round_winners = []
            for position in range(1, prize.quantity + 1):
                command = self._wait_for_command()
                if command == "terminate":
                    print("Round terminated; remaining draws are skipped.")
                    break
                print(f"Currently drawing the {_ordinal(position)} winner...")
                winner = self.lottery.draw_one(prize)
                if winner is None:
                    print("No eligible candidates remain for this prize.")
                    break
                round_winners.append(winner)
                user = winner.user
                print(
                    f"Congratulations {user.name} ({user.department}) "
                    f"for winning {prize.name}!"
                )

            print(f"\n--- Round {round_no} results ({len(round_winners)} winner(s)) ---")
            for winner in round_winners:
                print(
                    f"  {_ordinal(winner.sequence)} overall: {winner.user.name} "
                    f"({winner.user.department})"
                )

    def _wait_for_command(self) -> str:
        """Return 'draw' or 'terminate'. Handles 'pause' / 'continue'."""
        while True:
            command = self._ask(
                "Press Enter to draw next winner (pause / terminate): "
            ).lower()
            if command == "":
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to skip.")
                while True:
                    resume = self._ask("> ").lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        break
                    if resume == "terminate":
                        return "terminate"
                    print("Please type 'continue' or 'terminate'.")
                continue
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    # ----- step 4: results ----------------------------------------------

    def _show_summary(self) -> None:
        assert self.lottery is not None
        prizes = self.prize_manager.get_prizes()
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        print("\n=== Final Winner List (by prize level, high to low) ===")
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  {winner.sequence}. {winner.user.name} "
                    f"({winner.user.department}) - {winner.winning_time}"
                )

    def _offer_export(self) -> None:
        assert self.lottery is not None
        answer = self._ask("\nExport the winner list to a file? (y/n): ").lower()
        if answer in ("y", "yes"):
            _, message = self.lottery.result_manager.export_to_csv(
                self.prize_manager.get_prizes()
            )
            print(message)


def main() -> int:
    """Run the interactive lottery system."""
    try:
        return LotterySystemCLI().run()
    except (EOFError, KeyboardInterrupt):
        print("\nSession ended.")
        return 1

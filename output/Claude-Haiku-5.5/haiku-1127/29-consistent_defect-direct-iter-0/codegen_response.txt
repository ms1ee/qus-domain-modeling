"""Grid-based path planning for the 800x800 street map.

The map is divided into 8x8 cells of 100x100 units. Streets are the grid
lines (points where x or y is a multiple of 100). Restaurants and customers
sit at integer-hundred intersections. Riders always move along streets, so
their routes are right-angle paths made of straight segments.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_CELLS = MAP_SIZE // CELL_SIZE  # 8 cells per axis


class PathPlanner:
    """Stateless helper providing coordinate conversion and path logic."""

    @staticmethod
    def snap_to_grid(value):
        """Convert a raw coordinate to the top-left corner of its grid cell.

        Values are clamped to the map range first (0-800).

        Args:
            value: Raw coordinate (int-convertible).

        Returns:
            int: Coordinate that is a multiple of 100 within 0-800.
        """
        value = max(0, min(MAP_SIZE, int(value)))
        return min(MAP_SIZE, (value // CELL_SIZE) * CELL_SIZE)

    @classmethod
    def snap_point(cls, point):
        """Snap an ``(x, y)`` pair to grid cell corners.

        Args:
            point: Sequence of two coordinates.

        Returns:
            tuple[int, int]: Snapped point.
        """
        x, y = point
        return (cls.snap_to_grid(x), cls.snap_to_grid(y))

    @staticmethod
    def grid_cell(x, y):
        """Return the (column, row) grid cell containing a point.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            tuple[int, int]: Cell indices in the range 0-7.
        """
        col = min(max(int(x), 0), MAP_SIZE - 1) // CELL_SIZE
        row = min(max(int(y), 0), MAP_SIZE - 1) // CELL_SIZE
        return (col, row)

    @classmethod
    def grid_center(cls, x, y):
        """Return the center point of the cell containing ``(x, y)``.

        Uses ``center = (grid // 100) * 100 + 50``.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            tuple[int, int]: Center coordinates of the cell.
        """
        col, row = cls.grid_cell(x, y)
        return (col * CELL_SIZE + 50, row * CELL_SIZE + 50)

    @staticmethod
    def is_on_street(x, y):
        """Return True when the point lies on a street (grid line).

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            bool: True if x or y is a multiple of 100.
        """
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def manhattan(a, b):
        """Manhattan distance between two points.

        Args:
            a: ``(x, y)`` tuple.
            b: ``(x, y)`` tuple.

        Returns:
            int: ``|ax - bx| + |ay - by|``.
        """
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def euclidean(a, b):
        """Euclidean distance between two points.

        Args:
            a: ``(x, y)`` tuple.
            b: ``(x, y)`` tuple.

        Returns:
            float: ``sqrt((ax - bx)^2 + (ay - by)^2)``.
        """
        return ((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2) ** 0.5

    @staticmethod
    def next_axis(x, y, tx, ty):
        """Decide which axis to move along next to reach the target.

        Special cases:
            * Target directly above/below (same x): move vertically only.
            * Target directly left/right (same y): move horizontally only.
            * Otherwise, move along the axis whose street the rider is on,
              which keeps every step on a street.

        Args:
            x, y: Current position.
            tx, ty: Target position.

        Returns:
            str | None: ``"x"``, ``"y"``, or None when already at target.
        """
        if (x, y) == (tx, ty):
            return None
        if x == tx:
            return "y"
        if y == ty:
            return "x"
        return "x" if y % CELL_SIZE == 0 else "y"

    @classmethod
    def step_towards(cls, x, y, tx, ty, speed):
        """Advance a position toward a target by up to ``speed`` units.

        The movement follows street lines only and may turn a right angle
        within the same tick if budget remains.

        Args:
            x, y: Current position.
            tx, ty: Target position.
            speed: Maximum distance to travel this tick.

        Returns:
            tuple[int, int]: New position.
        """
        budget = int(speed)
        while budget > 0 and (x, y) != (tx, ty):
            axis = cls.next_axis(x, y, tx, ty)
            if axis == "x":
                step = min(budget, abs(tx - x))
                x += step if tx > x else -step
            else:
                step = min(budget, abs(ty - y))
                y += step if ty > y else -step
            budget -= step
        return (x, y)

    @classmethod
    def plan_route(cls, start, targets):
        """Build the right-angle corner路径 from ``start`` through ``targets``.

        Riders do not block each other, so route planning is deterministic
        and free of deadlocks.

        Args:
            start: Starting ``(x, y)`` point.
            targets: Ordered list of ``(x, y)`` waypoints.

        Returns:
            list[tuple[int, int]]: Corner points including the start.
        """
        cx, cy = int(start[0]), int(start[1])
        points = [(cx, cy)]
        for tx, ty in targets:
            while (cx, cy) != (tx, ty):
                axis = cls.next_axis(cx, cy, tx, ty)
                if axis == "x":
                    cx = tx
                else:
                    cy = ty
                points.append((cx, cy))
        return points

    @classmethod
    def route_length(cls, points):
        """Total Manhattan length of a polyline.

        Args:
            points: List of ``(x, y)`` points.

        Returns:
            int: Sum of consecutive Manhattan distances.
        """
        return sum(cls.manhattan(a, b) for a, b in zip(points, points[1:]))

An order travels through the statuses:

def _next_order_id():
    """Issue the next unique order id.

    Returns:
        int: Monotonically increasing order id starting at 1.

    counter = globalvar.get_value("order_id_counter", 0) + 1
    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A single food delivery order.

    Attributes:
        id (int): Unique order identifier.
        res_pos (tuple[int, int]): Restaurant grid point (snapped).
        cus_pos (tuple[int, int]): Customer grid point (snapped).
        status (str): Lifecycle status (see module constants).
        rider_id (int | None): Assigned rider id.
        start_time (int): Tick when the order was created.
        completion_time (int | None): Tick when delivery finished.

        self.res_pos = PathPlanner.snap_point(res_pos)
        self.cus_pos = PathPlanner.snap_point(cus_pos)
        self.status = STATUS_PENDING

"""Rider model: state machine, movement along streets and event logging.

Rider states:
    0 = idle, 1 = picking up (heading to restaurant), 2 = delivering.

Log events recorded in ``Rider.log``:
    "spawn", "accepted_order", "arrived_at_merchant", "delivered_to_customer"
"""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

DEFAULT_SPEED = 20
SPAWN_X = 400
SPAWN_Y = 400

ACTIVE_STATUSES = ("assigned", "picking_up", "delivering")
IN_PROGRESS_STATUSES = ("picking_up", "delivering")


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    Attributes:
        rider_id (int): Unique identifier.
        x, y (int): Current location.
        speed (int): Units moved per tick.
        state (int): 0 idle, 1 picking up, 2 delivering.
        orders (list[Order]): Assigned orders, including completed history.
        special_case (bool): Flag set when an unusual condition was repaired.
        log (list[dict]): Status change records.
    """

    def __init__(self, rider_id, x=SPAWN_X, y=SPAWN_Y, speed=DEFAULT_SPEED):
        """Create a rider and log its spawn event.

        Args:
            rider_id: Unique rider identifier.
            x: Initial x coordinate (default 400).
            y: Initial y coordinate (default 400).
            speed: Movement speed in units per tick (default 20).
        """
        self.rider_id = int(rider_id)
        self.x = int(x)
        self.y = int(y)
        self.speed = int(speed)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # Properties and queries
    # ------------------------------------------------------------------
    @property
    def position(self):
        """Current ``(x, y)`` location."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human-readable state label."""
        return STATE_NAMES.get(self.state, "unknown")

    def active_orders(self):
        """Orders that are not yet completed, in queue order.

        Returns:
            list[Order]: Active orders.
        """
        return [o for o in self.orders if o.status in ACTIVE_STATUSES]

    def active_order_count(self):
        """Number of active (not completed) orders.

        Returns:
            int: Count of active orders.
        """
        return len(self.active_orders())

    def is_idle(self):
        """Whether the rider is free to receive work.

        Returns:
            bool: True when state is idle and no active orders exist.
        """
        return self.state == IDLE and self.active_order_count() == 0

    def current_order(self):
        """The order currently being picked up or delivered.

        Returns:
            Order | None: The in-progress order, if any.
        """
        for order in self.orders:
            if order.status in IN_PROGRESS_STATUSES:
                return order
        return None

    def remaining_waypoints(self):
        """Ordered waypoints still to visit for all active orders.

        Returns:
            list[tuple[int, int]]: Restaurant/customer points to visit.
        """
        points = []
        for order in self.active_orders():
            if order.status == "delivering":
                points.append(order.cus_pos)
            else:
                points.extend([order.res_pos, order.cus_pos])
        return points

    # ------------------------------------------------------------------
    # State machine and movement
    # ------------------------------------------------------------------
    def move(self):
        """Advance the rider by one tick.

        Handles state transitions:
            idle -> picking up when an assigned order exists,
            picking up -> delivering on reaching the restaurant,
            delivering -> idle (then picking up the next queued order)
            on reaching the customer.
        """
        tick = self._tick()
        if self.state == IDLE and not self._accept_next(tick):
            return

        order = self.current_order()
        if order is None:
            self._repair(tick)
            return

        self._sync_state(order)
        target = self._target(order)
        if self.position != target:
            self.x, self.y = PathPlanner.step_towards(
                self.x, self.y, target[0], target[1], self.speed
            )
        if self.position == target:
            self._on_arrival(order, tick)

    def to_dict(self):
        """Serialize the rider to a JSON-compatible dict.

        Returns:
            dict: Rider attributes including order history.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [o.to_dict() for o in self.orders],
        }

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    @staticmethod
    def _tick():
        """Current simulation tick from the global store."""
        return globalvar.get_value("tick", 0)

    def _record(self, event, order_id):
        """Append a status-change record to the rider log.

        Args:
            event: Event name.
            order_id: Related order id or None.
        """
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": self._tick(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def _accept_next(self, tick):
        """Accept the first queued order if the rider is able to start it.

        Args:
            tick: Current tick (used for logging consistency).

        Returns:
            bool: True when an order was accepted.
        """
        for order in self.orders:
            if order.status == "assigned":
                order.status = "picking_up"
                self.state = PICKING_UP
                self._record("accepted_order", order.id)
                return True
        return False

    def _sync_state(self, order):
        """Make the rider state agree with its in-progress order.

        Args:
            order: The in-progress order.
        """
        if order.status == "picking_up":
            self.state = PICKING_UP
        elif order.status == "delivering":
            self.state = DELIVERING

    def _target(self, order):
        """Destination for the given in-progress order.

        Args:
            order: The in-progress order.

        Returns:
            tuple[int, int]: Restaurant point when picking up, else customer.
        """
        return order.res_pos if order.status == "picking_up" else order.cus_pos

    def _on_arrival(self, order, tick):
        """Handle arriving at the restaurant or the customer.

        Args:
            order: The in-progress order.
            tick: Current tick.
        """
        if order.status == "picking_up":
            order.status = "delivering"
            self.state = DELIVERING
            self._record("arrived_at_merchant", order.id)
        else:
            order.complete(tick)
            self.state = IDLE
            self._record("delivered_to_customer", order.id)
            self._accept_next(tick)

    def _repair(self, tick):
        """Recover from an inconsistent state (state without an order).

        Args:
            tick: Current tick.
        """
        logger.warning("Rider %s inconsistent state; repairing", self.rider_id)
        self.special_case = True
        self.state = IDLE
        self._accept_next(tick)

"""Order management: creation, distance-first assignment and dispatch."""

import logging

import globalvar
from order_model import Order, STATUS_PENDING
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

MIN_COORD = 0
MAX_COORD = 1000


def find_best_rider_for_order(order):
    """Choose the best rider for an order.

    The rider with the fewest active orders is preferred. Ties are broken by
    the Euclidean distance from the rider to the restaurant.

    Args:
        order: Order whose restaurant location drives the distance check.

    Returns:
        Rider | None: Selected rider, or None when no riders exist.
    """
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None

    def sort_key(rider):
        return (
            rider.active_order_count(),
            PathPlanner.euclidean(rider.position, order.res_pos),
        )

    return min(riders, key=sort_key)


class OrderManager:
    """Creates, dispatches and tracks orders.

    Attributes:
        system: SystemManager used for rider recruitment.
        pending (list[Order]): Orders waiting for a rider.
    """

    def __init__(self, system):
        """Create an order manager bound to a system manager.

        Args:
            system: SystemManager instance.
        """
        self.system = system
        self.pending = []

    @staticmethod
    def validate_coordinates(coords):
        """Validate a four-element coordinate list.

        Args:
            coords: Sequence ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            list[int]: Validated integers.

        Raises:
            ValueError: If the format or range is invalid.
        """
        if len(coords) != 4:
            raise ValueError("Exactly 4 coordinates are required.")
        values = []
        for value in coords:
            if isinstance(value, bool) or not isinstance(value, int):
                raise ValueError("Coordinates must be integers.")
            if not MIN_COORD <= value <= MAX_COORD:
                raise ValueError(
                    "Coordinates must be within %d-%d." % (MIN_COORD, MAX_COORD)
                )
            values.append(value)
        return values

    def create_order(self, coords):
        """Create an order from raw coordinates and register it globally.

        Args:
            coords: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            Order: The newly created order (status "pending").

        Raises:
            ValueError: If the coordinates are invalid.
        """
        rx, ry, cx, cy = self.validate_coordinates(list(coords))
        order = Order((rx, ry), (cx, cy))
        globalvar.get_value("orders").append(order)
        logger.info("Created order %s", order.id)
        return order

    def dispatch(self, order):
        """Queue an order, recruit if needed, and assign it to a rider.

        Args:
            order: Order to dispatch.

        Returns:
            tuple[Rider | None, bool]: (assigned rider or None, whether a new
            rider was recruited during this dispatch).
        """
        if order not in self.pending:
            self.pending.append(order)
        recruited = self.system.try_recruit(len(self.pending)) is not None
        rider = self._assign(order)
        return rider, recruited

    def retry_pending(self):
        """Attempt to assign all pending orders (called each tick).

        Returns:
            list[Order]: Orders assigned during this call.
        """
        assigned = []
        for order in list(self.pending):
            self.system.try_recruit(len(self.pending))
            if self._assign(order) is not None:
                assigned.append(order)
        return assigned

    def _assign(self, order):
        """Assign an order to the best rider if one exists.

        Args:
            order: Order to assign.

        Returns:
            Rider | None: The rider that received the order.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        if order in self.pending:
            self.pending.remove(order)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

    @staticmethod
    def all_orders():
        """All orders created in this session.

        Returns:
            list[Order]: Order list from the global store.
        """
        return globalvar.get_value("orders", [])

    def completed_orders(self):
        """Orders whose delivery has finished.

        Returns:
            list[Order]: Completed orders.
        """
        return [o for o in self.all_orders() if o.status == "completed"]

    def unassigned_count(self):
        """Number of orders still waiting for a rider.

        Returns:
            int: Count of pending orders.
        """
        return sum(1 for o in self.pending if o.status == STATUS_PENDING)

        self.riders.append(rider)
        logger.info("Rider %s recruited", rider_id)
        return rider

    def try_recruit(self, pending_count):
        """Recruit a rider when funds allow and demand exceeds idle riders.

        Trigger: funds >= recruit_min_funds AND pending orders > idle riders.
        Cost: recruit_cost deducted from funds.

        Args:
            pending_count: Number of orders waiting for a rider.

        Returns:
            Rider | None: The new rider, or None if no recruitment happened.
        """
        if self.funds < self.config["recruit_min_funds"]:
            return None
        if pending_count <= self.idle_rider_count():
            return None
        globalvar.set_value(
            "system_funds", self.funds - self.config["recruit_cost"]
        )
        return self.create_rider()

    def get_rider(self, rider_id):
        """Look up a rider by id.

        Args:
            rider_id: Rider identifier.

        Returns:
            Rider | None: Matching rider or None.
        """
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def tick(self):
        """Advance the simulation by one tick and move every rider.

        Returns:
            int: The new tick value.
        """
        now = self.tick_count + 1
        globalvar.set_value("tick", now)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # keep the simulation alive on rider errors
                logger.exception("Rider %s failed to move", rider.rider_id)
        return now

    def run_ticks(self, count):
        """Run several ticks, retrying pending orders each tick.

        Args:
            count: Number of ticks to simulate.

        Returns:
            int: The final tick value.
        """
        from order import OrderManager  # local import avoids import cycle
        manager = OrderManager(self)
        for _ in range(count):
            manager.pending = globalvar.get_value("pending_orders", [])
            manager.retry_pending()
            globalvar.set_value("pending_orders", manager.pending)
            self.tick()
        return self.tick_count

    def load_config(self, path):
        """Load system parameters from a JSON file.

        Unknown keys are ignored. Numeric values must be non-negative ints.

        Args:
            path: JSON file path.

        Raises:
            ValueError: If a known parameter has an invalid value.
        """
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
        config = globalvar.get_value("config")
        for key, value in data.items():
            if key not in config:
                continue
            if key == "rider_spawn":
                if (not isinstance(value, list) or len(value) != 2
                        or not all(isinstance(v, int) for v in value)):
                    raise ValueError("rider_spawn must be [x, y] integers")
            elif not isinstance(value, int) or value < 0:
                raise ValueError("%s must be a non-negative integer" % key)
            config[key] = value
        globalvar.set_value("config", config)

    def save_state(self, path):
        """Persist configuration, funds, riders and orders to a JSON file.

        Args:
            path: Destination file path.
        """
        data = {
            "config": self.config,
            "system_funds": self.funds,
            "tick": self.tick_count,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in globalvar.get_value("orders", [])],
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        logger.info("State saved to %s", path)

"""Status displays, rider/order reports and ASCII map rendering."""

from path_planner import MAP_SIZE, CELL_SIZE, PathPlanner
from rider import STATE_NAMES

STEP = 50                       # map character resolution (units per char)
MAP_CHARS = MAP_SIZE // STEP + 1  # 17 characters per axis (9x9 intersections)


def _idx(value):
    """Convert a coordinate to a map character index.

    Args:
        value: Coordinate in units.

    Returns:
        int: Index clamped to the map range.
    """
    return min(MAP_CHARS - 1, max(0, round(value / STEP)))


class DisplayManager:
    """Builds text output for status, orders, rider logs and the map."""

    def render_status(self, system, riders):
        """Snapshot of funds, riders, their states and order lists.

        Args:
            system: SystemManager instance.
            riders: Iterable of Rider objects.

        Returns:
            str: Multi-line report.
        """
        riders = list(riders)
        lines = [
            "=== System Status ===",
            "Tick: %d" % system.tick_count,
            "System funds: %d" % system.funds,
            "Riders: %d" % len(riders),
        ]
        for rider in riders:
            col, row = PathPlanner.grid_cell(rider.x, rider.y)
            order_text = ", ".join(
                "#%d:%s" % (o.id, o.status) for o in rider.orders
            ) or "none"
            lines.append(
                "  Rider %d | pos=(%d,%d) cell=(%d,%d) | state=%s | "
                "active=%d | orders: %s"
                % (rider.rider_id, rider.x, rider.y, col, row,
                   STATE_NAMES.get(rider.state, "unknown"),
                   rider.active_order_count(), order_text)
            )
        return "\n".join(lines)

    def render_orders(self, orders):
        """Detailed table of all orders.

        Args:
            orders: Iterable of Order objects.

        Returns:
            str: Multi-line report.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        lines = ["=== Orders ==="]
        for o in orders:
            taken = "-" if o.time_taken is None else str(o.time_taken)
            eff = "-" if o.efficiency is None else str(o.efficiency)
            rider = "-" if o.rider_id is None else str(o.rider_id)
            lines.append(
                "  #%d start=t%d restaurant=%s customer=%s distance=%d "
                "time_taken=%s efficiency=%s rider=%s status=%s"
                % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                   taken, eff, rider, o.status)
            )
        return "\n".join(lines)

    def render_rider_log(self, riders):
        """Every recorded status change for every rider.

        Args:
            riders: Iterable of Rider objects.

        Returns:
            str: Multi-line report.
        """
        lines = ["=== Rider Status Log ==="]
        any_entry = False
        for rider in riders:
            for entry in rider.log:
                any_entry = True
                order = "-" if entry["order_id"] is None else entry["order_id"]
                lines.append(
                    "  t=%d rider=%d event=%s order=%s pos=(%d,%d) state=%s"
                    % (entry["time"], entry["rider_id"], entry["event"], order,
                       entry["x"], entry["y"],
                       STATE_NAMES.get(entry["state"], "unknown"))
                )
        if not any_entry:
            lines.append("  No rider events yet.")
        return "\n".join(lines)

    @staticmethod
    def _segment_indices(a, b):
        """Map character indices along an axis-aligned segment.

        Args:
            a: Start point ``(x, y)``.
            b: End point ``(x, y)``.

        Returns:
            list[tuple[int, int]]: (row, col) indices along the segment.
        """
        r1, c1 = _idx(a[1]), _idx(a[0])
        r2, c2 = _idx(b[1]), _idx(b[0])
        cells = []
        if c1 == c2:
            step = 1 if r2 >= r1 else -1
            for r in range(r1, r2 + step, step):
                cells.append((r, c1))
        else:
            step = 1 if c2 >= c1 else -1
            for c in range(c1, c2 + step, step):
                cells.append((r1, c))
        return cells

    def render_map(self, riders=None, orders=None, route=None):
        """Render the ASCII map.

        Symbols: 'R' rider, 'S' restaurant, 'C' customer, '#' street,
        '.' empty space, '*' planned route. Riders are drawn on top.

        Args:
            riders: Optional iterable of Rider objects to show.
            orders: Optional iterable of Order objects whose restaurant and
                customer points should be shown (completed orders are skipped).
            route: Optional list of route corner points to draw.

        Returns:
            str: Rendered map with a legend.
        """
        canvas = [["." for _ in range(MAP_CHARS)] for _ in range(MAP_CHARS)]
        for row in range(MAP_CHARS):
            for col in range(MAP_CHARS):
                x, y = col * STEP, row * STEP
                if PathPlanner.is_on_street(x, y):
                    canvas[row][col] = "#"

        if route:
            for a, b in zip(route, route[1:]):
                for row, col in self._segment_indices(a, b):
                    canvas[row][col] = "*"
            if len(route) == 1:
                canvas[_idx(route[0][1])][_idx(route[0][0])] = "*"

        for order in orders or []:
            if order.status == "completed":
                continue
            rx, ry = order.res_pos
            canvas[_idx(ry)][_idx(rx)] = "S"
            cx, cy = order.cus_pos
            canvas[_idx(cy)][_idx(cx)] = "C"

        for rider in riders or []:
            canvas[_idx(rider.y)][_idx(rider.x)] = "R"

        lines = ["     " + "".join(str((c * STEP // 100) % 10)
                                    for c in range(MAP_CHARS))]
        for row in range(MAP_CHARS):
            lines.append("%4d " % (row * STEP) + "".join(canvas[row]))
        lines.append("Legend: R=Rider S=Restaurant C=Customer "
                     "#=Street .=Empty *=Route (1 char = %d units)" % STEP)
        return "\n".join(lines)

"""Command parsing, shortcut keys, command history and input validation."""

COMMANDS = (
    "start", "status", "orders", "riders", "tick", "map",
    "save", "help", "quit", "exit",
)

ALIASES = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "t": "tick",
    "m": "map",
    "h": "help",
    "?": "help",
    "q": "quit",
}

COORD_MIN = 0
COORD_MAX = 1000


class InputError(Exception):
    """Raised for malformed or invalid user input."""


class InputManager:
    """Reads and validates user commands.

    Attributes:
        history (list[str]): Executed command lines, oldest first.
    """

    def __init__(self, reader=input):
        """Create the manager.

        Args:
            reader: Callable used to read a line (defaults to ``input``).
        """
        self._reader = reader
        self.history = []

    def read_line(self, prompt="sim> "):
        """Read one line from the user.

        Args:
            prompt: Prompt text.

        Returns:
            str: Stripped input line.

        Raises:
            EOFError: When input is closed.
        """
        return self._reader(prompt).strip()

    def parse(self, line):
        """Parse a command line, handling shortcuts and history recall.

        Supports ``!!`` (repeat last command) and ``!n`` (repeat history entry n).

        Args:
            line: Raw input text.

        Returns:
            tuple[str | None, list[str]]: Command name (None for blank input)
            and its argument tokens.

        Raises:
            InputError: For unknown commands or invalid history references.
        """
        text = line.strip()
        if not text:
            return None, []
        if text == "!!":
            if not self.history:
                raise InputError("No command in history yet.")
            text = self.history[-1]
        elif text.startswith("!") and text[1:].isdigit():
            index = int(text[1:])
            if not 1 <= index <= len(self.history):
                raise InputError("History entry %d does not exist." % index)
            text = self.history[index - 1]

        tokens = text.split()
        name = tokens[0].lower()
        command = ALIASES.get(name, name)
        if command not in COMMANDS:
            raise InputError(
                "Unknown command '%s'. Type 'help' for a list." % tokens[0]
            )
        self.history.append(text)
        return command, tokens[1:]

    @staticmethod
    def parse_int(text, name="value"):
        """Convert text to an integer with a helpful error message.

        Args:
            text: Token to convert.
            name: Field name used in the error message.

        Returns:
            int: Parsed integer.

        Raises:
            InputError: If the text is not an integer.
        """
        try:
            return int(text)
        except (TypeError, ValueError):
            raise InputError("%s must be an integer, got '%s'." % (name, text))

    @classmethod
    def parse_coordinates(cls, tokens):
        """Parse four coordinate tokens.

        Args:
            tokens: Iterable of four strings.

        Returns:
            list[int]: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Raises:
            InputError: On wrong count, non-integer or out-of-range values.
        """
        tokens = list(tokens)
        if len(tokens) != 4:
            raise InputError(
                "Expected 4 integers: restaurant_x restaurant_y "
                "customer_x customer_y."
            )
        names = ("restaurant_x", "restaurant_y", "customer_x", "customer_y")
        values = [cls.parse_int(t, n) for t, n in zip(tokens, names)]
        for name, value in zip(names, values):
            if not COORD_MIN <= value <= COORD_MAX:
                raise InputError(
                    "%s=%d is out of range (%d-%d)."
                    % (name, value, COORD_MIN, COORD_MAX)
                )
        return values

    def prompt_coordinates(self):
        """Ask the user for four coordinates interactively.

        Returns:
            list[int]: Validated coordinates.

        Raises:
            InputError: On invalid input.
        """
        line = self.read_line(
            "Enter restaurant_x restaurant_y customer_x customer_y: "
        )
        tokens = line.replace(",", " ").split()
        return self.parse_coordinates(tokens)

  start [rx ry cx cy]  Create an order (restaurant then customer, 0-1000).
                       Without arguments you will be prompted.
  status (s)           System funds, riders and their order lists.
  orders (o)           Details for all orders.
  riders (r)           Status change log for every rider.
  tick [n] (t)         Advance simulated time by n ticks (default 1).

  save [path]          Save state to JSON (default simulation_state.json).
  help (h, ?)          Show this help.
  quit (q) / exit (e)  Leave the program.
  !!  and  !n          Repeat the last command / history entry n."""


class SimulatorApp:
    """Interactive REPL tying together all simulator modules."""

    def __init__(self, system, inputs=None):
        """Create the application.

        Args:
            system: SystemManager instance.
            inputs: Optional InputManager (defaults to a new one).
        """
        self.system = system
        self.orders = OrderManager(system)
        self.inputs = inputs or InputManager()
        self.display = DisplayManager()
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "tick": self.cmd_tick,
            "map": self.cmd_map,
            "save": self.cmd_save,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self):
        """Run the main command loop until the user quits."""
        print(BANNER)
        while self.system.running:
            try:
                line = self.inputs.read_line()
            except (EOFError, KeyboardInterrupt):
                print()
                self.system.stop()
                break
            try:
                command, args = self.inputs.parse(line)
            except InputError as exc:
                print("Error: %s" % exc)
                continue
            if command is None:
                continue
            try:
                self._handlers[command](args)
            except EOFError:
                self.system.stop()
                break
            except (InputError, ValueError) as exc:
                print("Error: %s" % exc)
            except Exception:  # error recovery: keep the session alive
                logger.exception("Unexpected failure in command '%s'", command)
                print("Unexpected internal error; state preserved. "
                      "See log for details.")
        print("Goodbye.")

    def cmd_start(self, args):
        """Create an order, dispatch it and show the rider route map.

        Args:
            args: Optional four coordinate tokens.
        """
        coords = (self.inputs.parse_coordinates(args) if args
                  else self.inputs.prompt_coordinates())
        order = self.orders.create_order(coords)
        rider, recruited = self.orders.dispatch(order)

        print("Order #%d created at tick %d." % (order.id, order.start_time))
        print("Restaurant grid point: %s (cell %s)"
              % (order.res_pos, PathPlanner.grid_cell(*order.res_pos)))
        print("Customer grid point:   %s (cell %s)"
              % (order.cus_pos, PathPlanner.grid_cell(*order.cus_pos)))
        if recruited:
            print("New rider added: yes (rider %d)"
                  % self.system.riders[-1].rider_id)
        else:
            print("New rider added: no")

        if rider is None:
            print("No rider available. The order stays pending and will be "
                  "assigned when a rider frees up.")
            return

        print("Assigned to rider %d (state: %s, active orders: %d)"
              % (rider.rider_id, rider.state_name, rider.active_order_count()))
        route = PathPlanner.plan_route(rider.position,
                                       rider.remaining_waypoints())
        distance = PathPlanner.route_length(route)
        ticks = math.ceil(distance / rider.speed) if rider.speed else 0
        print("Speed: %d units/tick | Route distance: %d units | "

    def cmd_orders(self, _args):

def normalize(text: str) -> str:
    """Normalize text for case-insensitive and whitespace-insensitive comparison."""
    return " ".join(str(text).split()).casefold()

import os

from lottery_system.config import DEFAULT_USERS
from lottery_system.models import User, normalize


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in team building participant list."""
        users = [User(name, department) for name, department in DEFAULT_USERS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        if not path or not os.path.isfile(path):
            return [], [f"File not found: '{path}'"]
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file into users and a list of error messages."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"Read Error: cannot read '{path}': {exc}"]

        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            stripped = raw.strip()
            if not stripped:
                continue
            fields = stripped.replace("\uff0c", ",").split(",")
            if len(fields) != 2 or not fields[0].strip() or not fields[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{stripped}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(fields[0].strip(), fields[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into unique valid users and duplicate entries.

        The first occurrence of a name is kept as valid; later occurrences are
        returned as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group to number of people."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from lottery_system.config import DEFAULT_PRIZES
from lottery_system.models import Prize, normalize


class PrizeManager:
    """Manages the prize library and the lottery round sequence."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = normalize(name)
        for prize in self._prizes:
            if normalize(prize.name) == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns error messages for skipped items."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize already exists and was skipped: {name}")
                continue
            self._prizes.append(Prize(name, quantity, [], drawer))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a custom prize. Raises ValueError on invalid input."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._find(name) is not None:
            raise ValueError(f"A prize named '{name}' already exists.")
        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = str(group).strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(Prize(name, quantity, groups, drawer))

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            if len(order) != count or sorted(order) != list(range(1, count + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine and draw execution."""

import random

from lottery_system.config import (
    CHENG_RU_GROUP,
    DRAWER_CHEN_QIXIAN,
    DRAWER_JIN_DUO,
    DRAWER_LI_BIN,
    DRAWER_MA_JIAQI,
)
from lottery_system.models import Prize, User, normalize
from lottery_system.services.result_manager import ResultManager


def is_cheng_ru_member(user: User) -> bool:
    """Return True if the user belongs to the Cheng Ru group."""
    return normalize(user.department) == normalize(CHENG_RU_GROUP)


class LotteryManager:
    """Executes lottery rounds while enforcing all eligibility rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self._round_winners: list[User] = []
        # Results of the last draws by Chen Qixian / Jin Duo: True if the round
        # contained at least one Cheng Ru group member. Keeps at most two entries.
        self.last_two_draw_results: list[bool] = []
        self.result_manager = ResultManager()

    # ----- rule helpers -------------------------------------------------

    @staticmethod
    def _is_drawer(prize: Prize, drawer_name: str) -> bool:
        return prize.associated_drawer is not None and normalize(prize.associated_drawer) == normalize(
            drawer_name
        )

    def _fallback_active(self) -> bool:
        """Cheng Ru protection applies when the last two draw results both lack Cheng Ru members."""
        return len(self.last_two_draw_results) >= 2 and self.last_two_draw_results[-2:] == [False, False]

    def _available_users(self) -> list[User]:
        return [user for user in self._users if normalize(user.name) not in self._winner_keys]

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users eligible to win the given prize in the current state."""
        available = self._available_users()

        # Special rule: Li Bin's prizes only filter previous winners.
        if self._is_drawer(prize, DRAWER_LI_BIN):
            return available

        # Fallback rule: Ma Jiaqi's prize is restricted to Cheng Ru group members,
        # overriding avoidance groups.
        if self._is_drawer(prize, DRAWER_MA_JIAQI) and self._fallback_active():
            return [user for user in available if is_cheng_ru_member(user)]

        # Basic rule: exclude previous winners and avoidance groups.
        avoided = {normalize(group) for group in prize.avoid_groups}
        return [user for user in available if normalize(user.department) not in avoided]

    def eligible_candidates(self, prize: Prize) -> list[User]:
        """Public accessor for the current eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def describe_rule(self, prize: Prize) -> str:
        """Return a human-readable description of the rules applied to a prize."""
        if self._is_drawer(prize, DRAWER_LI_BIN):
            return (
                "Only previous winners are excluded (special rule for Li Bin: "
                "avoidance groups are not applied)."
            )
        if self._is_drawer(prize, DRAWER_MA_JIAQI) and self._fallback_active():
            return (
                f"{CHENG_RU_GROUP} protection active: only {CHENG_RU_GROUP} members are eligible "
                "(avoidance groups overridden)."
            )
        parts = ["Previous winners are excluded"]
        if prize.avoid_groups:
            parts.append("avoided groups: " + ", ".join(prize.avoid_groups))
        return "; ".join(parts) + "."

    # ----- round execution ----------------------------------------------

    def start_round(self, prize: Prize) -> None:
        """Begin a round for the given prize."""
        self._round_winners = []

    def draw_winner(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self._winner_keys.add(normalize(user.name))
        self._round_winners.append(user)
        self.result_manager.add_winner(user, prize)
        return user

    def finish_round(self, prize: Prize) -> list[User]:
        """Finish the round, update fallback history, and return the round winners."""
        winners = list(self._round_winners)
        if self._is_drawer(prize, DRAWER_CHEN_QIXIAN) or self._is_drawer(prize, DRAWER_JIN_DUO):
            contains_cheng_ru = any(is_cheng_ru_member(user) for user in winners)
            self.last_two_draw_results.append(contains_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self._is_drawer(prize, DRAWER_MA_JIAQI):
            # Protection (if any) has been applied; reset the history.
            self.last_two_draw_results = []
        self._round_winners = []
        return winners

"""Winner recording, ordering, and CSV export."""

import csv
import os
import re
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User

_LEVEL_PATTERN = re.compile(r"\b(first|second|third)\s+prize\b", re.IGNORECASE)
_LEVELS = {"first": 1, "second": 2, "third": 3}
_UNRANKED_LEVEL = 99


def prize_level(prize_name: str) -> int:
    """Return the numeric level of a prize (1 = highest). Unranked prizes return 99."""
    match = _LEVEL_PATTERN.search(prize_name or "")
    if not match:
        return _UNRANKED_LEVEL
    return _LEVELS[match.group(1).lower()]


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    sequence: int  # global winning order (1-based)


class ResultManager:
    """Stores winners and produces ordered results and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        winner = Winner(user=user, prize_name=prize_name, sequence=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, each list in winning order."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def get_sorted_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Winners sorted by prize level (high to low), then prize order, then winning order."""
        order = {prize.name: index for index, prize in enumerate(prizes)}
        fallback_index = len(order)
        return sorted(
            self._winners,
            key=lambda w: (
                prize_level(w.prize_name),
                order.get(w.prize_name, fallback_index),
                w.sequence,
            ),
        )

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        winners = self.get_sorted_winners(prizes)
        if not winners:
            return False, "No winners to export."

        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in winners:
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.sequence,
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive menu-driven controller."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.lottery_prizes: list[Prize] = []

    # ----- main loop ----------------------------------------------------

    MENU = [
        ("1", "Load default participant list"),
        ("2", "Load custom participant list (txt)"),
        ("3", "View department groups and counts"),
        ("4", "Load default prize library"),
        ("5", "Add custom prize"),
        ("6", "View prize list"),
        ("7", "Reorder prizes"),
        ("8", "Start lottery"),
        ("9", "View and export results"),
        ("0", "Exit"),
    ]

    def run(self) -> None:
        print("=" * 60)
        print("  Company Team Building - Smart Lottery System")
        print("=" * 60)
        handlers = {
            "1": self._load_default_list,
            "2": self._load_custom_list,
            "3": self._view_groups,
            "4": self._load_default_prizes,
            "5": self._add_prize,
            "6": self._view_prizes,
            "7": self._reorder_prizes,
            "8": self._start_lottery,
            "9": self._export_results,
        }
        while True:
            print()
            for key, label in self.MENU:
                print(f"  [{key}] {label}")
            choice = input("Select an option: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            handler = handlers.get(choice)
            if handler is None:
                print("Invalid option, please try again.")
                continue
            handler()

    # ----- list management ----------------------------------------------

    def _load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        valid, duplicates = self.list_manager.validate_users(users)
        for error in errors:
            print(error)
        if duplicates:
            print(f"Ignored {len(duplicates)} duplicate entries in the default list.")
        self.users = valid
        print(f"Default participant list loaded: {len(self.users)} people.")

    def _load_custom_list(self) -> None:
        path = input("Enter the path of the txt list file: ").strip().strip('"')
        users, errors = self.list_manager.load_custom_list(path)

        if errors:
            print("Problems found in the uploaded list:")
            for error in errors:
                print(f"  - {error}")
            if not self._confirm_removal("invalid data"):
                print("Upload terminated. Current participant list unchanged.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print("Duplicate names detected:")
            for user in duplicates:
                print(f"  - {user.name} ({user.department})")
            if not self._confirm_removal("duplicate entries"):
                print("Upload terminated. Current participant list unchanged.")
                return

        if not valid:
            print("No valid participants found. Upload cancelled.")
            return

        self.users = valid
        print(f"Custom participant list loaded: {len(self.users)} people.")

    @staticmethod

        while True:

                f"Enter 'r' to remove {label} and continue, or 'a' to abort the upload: "

            if answer == "a":
                return False
            print("Please enter 'r' or 'a'.")

        except ValueError:

    # ----- lottery process ----------------------------------------------

    def _start_lottery(self) -> None:

            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return
        if self.lottery is not None and self.lottery.result_manager.get_winners():
            answer = input("Previous results exist. Restart and clear them? (y/n): ").strip().lower()
            if answer != "y":
                print("Lottery cancelled.")
                return

        lottery = LotteryManager(self.users)
        self.lottery = lottery
        self.lottery_prizes = prizes

        total_rounds = len(prizes)
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(lottery, prize, round_no, total_rounds)

        print()
        print("=" * 60)
        print("  Final Winner List")
        print("=" * 60)
        self._print_sorted_results(lottery, prizes)

    def _run_round(self, lottery: LotteryManager, prize: Prize, round_no: int, total_rounds: int) -> None:
        print()
        print("-" * 60)
        print(f"Round {round_no}/{total_rounds}")
        print(f"Prize: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Lottery person: {prize.associated_drawer or 'None'}")
        print(f"Rule: {lottery.describe_rule(prize)}")
        print("-" * 60)

        lottery.start_round(prize)
        drawn = 0
        terminated = False

        while drawn < prize.quantity:
            if not lottery.eligible_candidates(prize):
                print("No eligible candidates remain for this prize. Round ends early.")
                break

            command = input(
                "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
            ).strip().lower()
            if command == "terminate":
                terminated = True
                break
            if command == "pause":
                print("Lottery paused.")
                while True:
                    resume = input("Type 'continue' to resume or 'terminate' to end this round: ").strip().lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        break
                    if resume == "terminate":
                        terminated = True
                        break
                    print("Unknown command.")
                if terminated:
                    break
            elif command not in ("", "continue"):
                print("Unknown command. Press Enter to draw, or type 'pause' / 'terminate'.")
                continue

            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = lottery.draw_winner(prize)
            if user is None:
                print("No eligible candidates remain for this prize. Round ends early.")
                break
            drawn += 1
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        winners = lottery.finish_round(prize)
        print()
        if terminated:
            print("Round terminated early.")
        print(f"Round results for {prize.name} ({len(winners)} winner(s)):")
        if not winners:
            print("  (no winners)")
        for index, user in enumerate(winners, start=1):
            print(f"  {index}. {user.name} ({user.department})")

    def _print_sorted_results(self, lottery: LotteryManager, prizes: list[Prize]) -> None:
        winners = lottery.result_manager.get_sorted_winners(prizes)
        if not winners:
            print("No winners recorded.")
            return
        current_prize = None
        for winner in winners:
            if winner.prize_name != current_prize:
                current_prize = winner.prize_name
                print(f"\n{current_prize}:")
            print(f"  #{winner.sequence} {winner.user.name} ({winner.user.department})")

    # ----- results ------------------------------------------------------

    def _export_results(self) -> None:
        if self.lottery is None or not self.lottery.result_manager.get_winners():
            print("No lottery results to display or export yet.")
            return
        self._print_sorted_results(self.lottery, self.lottery_prizes)
        success, message = self.lottery.result_manager.export_to_csv(self.lottery_prizes)
        print(message)

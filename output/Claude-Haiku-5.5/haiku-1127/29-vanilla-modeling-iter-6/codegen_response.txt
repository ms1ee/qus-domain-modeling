"""Process-wide global variable store used by the dispatch simulator.

The store is a simple dictionary wrapped by ``set_value`` and ``get_value``.
Well-known keys used by the program:

* ``"riders"``: ``list[Rider]`` of all currently active riders.
* ``"current_tick"``: ``int`` simulation tick counter.
"""

import logging

logger = logging.getLogger(__name__)

_global_dict = {}


def _init():
    """Reset the global variable store to its default values.

    Creates an empty rider list and sets the simulation tick to zero.
    This function is safe to call multiple times; each call resets state.
    """
    global _global_dict
    _global_dict = {
        "riders": [],
        "current_tick": 0,
    }
    logger.debug("Global variable store initialised.")


def set_value(key, value):
    """Store ``value`` under ``key`` in the global store.

    Args:
        key: Hashable identifier, e.g. ``"riders"``.
        value: Any object to store.
    """
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key``.

    Args:
        key: Identifier used with :func:`set_value`.
        default: Value returned when ``key`` does not exist.

    Returns:
        The stored value, or ``default`` when the key is missing.
    """
    return _global_dict.get(key, default)

"""Grid coordinate system for the 800x800 dispatch map.

The map is divided into an 8x8 grid of 100x100 unit cells. Grid lines
(coordinates that are multiples of 100) form the street network. Cell
centers (coordinates such as 50, 150, ..., 750) are the restaurant and
customer locations after coordinate conversion.
"""

from dataclasses import dataclass

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = 8


@dataclass(frozen=True)
class Point:
    """An immutable integer coordinate on the map.

    Attributes:
        x: Horizontal coordinate (0-800).
        y: Vertical coordinate (0-800).
    """

    x: int
    y: int


@dataclass(frozen=True)
class GridCell:
    """One cell of the 8x8 grid.

    Attributes:
        grid_x: Column index of the cell (0-7).
        grid_y: Row index of the cell (0-7).
        center_x: Horizontal coordinate of the cell center.
        center_y: Vertical coordinate of the cell center.
    """

    grid_x: int
    grid_y: int
    center_x: int
    center_y: int


def is_valid_coordinate(value):
    """Check whether ``value`` is an integer within the map range.

    Args:
        value: Candidate coordinate.

    Returns:
        True if ``value`` is an int (not bool) between 0 and 800 inclusive.
    """
    return (
        isinstance(value, int)
        and not isinstance(value, bool)
        and 0 <= value <= MAP_SIZE
    )


def clamp_cell_index(value):
    """Return the grid cell index for a coordinate, clamped to the map.

    Args:
        value: Coordinate in map units.

    Returns:
        Cell index in the range 0..GRID_DIMENSION-1.
    """
    index = value // CELL_SIZE
    return max(0, min(GRID_DIMENSION - 1, index))


def to_cell_center(value):
    """Convert a coordinate to the center of its grid cell along one axis.

    Implements ``center = (value // 100) * 100 + 50`` with clamping so that
    the boundary value 800 maps to the center of the last cell (750).

    Args:
        value: Coordinate in map units (0-800).

    Returns:
        The center coordinate of the containing cell.
    """
    return clamp_cell_index(value) * CELL_SIZE + CELL_SIZE // 2


def to_cell_center_point(x, y):
    """Convert a coordinate pair to the center point of its grid cell.

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.

    Returns:
        A Point located at the center of the containing cell.
    """
    return Point(to_cell_center(x), to_cell_center(y))


def is_street_point(x, y):
    """Return True if the point lies on a street (a grid line).

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.

    Returns:
        True when x or y is a multiple of the cell size.
    """
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


class GridMap:
    """Represents the 8x8 grid of cells and the street network.

    Attributes:
        map_size: Side length of the map in units.
        cell_size: Side length of one cell in units.
        dimension: Number of cells per row and column.
        cells: List of all GridCell objects in row-major order.
    """

    def __init__(self, map_size=MAP_SIZE, cell_size=CELL_SIZE,
                 dimension=GRID_DIMENSION):
        """Build the grid and pre-compute all cell centers.

        Args:
            map_size: Side length of the map in units.
            cell_size: Side length of one cell in units.
            dimension: Number of cells per row and column.
        """
        self.map_size = map_size
        self.cell_size = cell_size
        self.dimension = dimension
        self.cells = []
        for grid_y in range(dimension):
            for grid_x in range(dimension):
                self.cells.append(
                    GridCell(
                        grid_x=grid_x,
                        grid_y=grid_y,
                        center_x=grid_x * cell_size + cell_size // 2,
                        center_y=grid_y * cell_size + cell_size // 2,
                    )
                )

    def cell_at(self, x, y):
        """Return the cell containing the point (x, y).

        Args:
            x: Horizontal coordinate.
            y: Vertical coordinate.

        Returns:
            The GridCell that contains the point (boundaries are clamped).
        """
        grid_x = max(0, min(self.dimension - 1, x // self.cell_size))
        grid_y = max(0, min(self.dimension - 1, y // self.cell_size))
        return self.cells[grid_y * self.dimension + grid_x]

    def is_valid(self, x, y):
        """Return True if (x, y) lies inside the map bounds.

        Args:
            x: Horizontal coordinate.
            y: Vertical coordinate.
        """
        return (is_valid_coordinate(x) and is_valid_coordinate(y)
                and x <= self.map_size and y <= self.map_size)

    def is_street(self, x, y):
        """Return True if (x, y) lies on the street network.

        Args:
            x: Horizontal coordinate.
            y: Vertical coordinate.
        """
        return is_street_point(x, y)

"""System configuration loaded from an optional JSON file.

The JSON file may either be a flat object or contain a ``"system"`` object
holding the parameters. Unknown keys are ignored and invalid values fall
back to the defaults.
"""

import json
import logging
import os
from dataclasses import asdict, dataclass, fields

logger = logging.getLogger(__name__)


@dataclass
class Configuration:
    """Tunable simulation parameters.

    Attributes:
        initial_funds: Starting system funds.
        rider_recruitment_cost: Funds deducted per recruited rider.
        recruitment_fund_threshold: Minimum funds required to recruit.
        initial_rider_speed: Speed of new riders in units per tick.
        initial_rider_x: Spawn x coordinate for new riders.
        initial_rider_y: Spawn y coordinate for new riders.
        max_ticks_per_order: Safety limit when running an order to completion.
    """

    initial_funds: int = 1000
    rider_recruitment_cost: int = 300
    recruitment_fund_threshold: int = 300
    initial_rider_speed: int = 10
    initial_rider_x: int = 435
    initial_rider_y: int = 385
    max_ticks_per_order: int = 1000

    @classmethod
    def load(cls, path):
        """Load configuration from a JSON file.

        Args:
            path: Path to a JSON configuration file.

        Returns:
            A Configuration instance. Defaults are used for any missing or
            invalid entries, and when the file cannot be read.
        """
        config = cls()
        if not path or not os.path.exists(path):
            logger.warning("Config file %s not found; using defaults.", path)
            return config
        try:
            with open(path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
        except (OSError, json.JSONDecodeError) as exc:
            logger.error("Failed to read config %s: %s", path, exc)
            return config

        params = data.get("system", data) if isinstance(data, dict) else {}
        if not isinstance(params, dict):
            logger.error("Config %s has an invalid structure.", path)
            return config

        for field_info in fields(cls):
            if field_info.name not in params:
                continue
            try:
                setattr(config, field_info.name, int(params[field_info.name]))
            except (TypeError, ValueError):
                logger.error("Invalid value for %s; using default.",
                             field_info.name)
        return config

    def save(self, path):
        """Write the configuration to a JSON file.

        Args:
            path: Destination file path.

        Returns:
            True on success, False when the file cannot be written.
        """
        try:
            with open(path, "w", encoding="utf-8") as handle:
                json.dump({"system": asdict(self)}, handle, indent=2)
            return True
        except OSError as exc:
            logger.error("Failed to write config %s: %s", path, exc)
            return False

"""Grid-based path planning for riders.

Routes are sequences of axis-aligned waypoints. For every pair of points the
planner evaluates the two right-angle (L-shaped) candidates and prefers the
one whose segments run along streets (grid lines). Straight-line movement is
used when start and goal share an x or y coordinate, which covers the
"directly above or below" special case.
"""

import logging
from dataclasses import dataclass, field

from grid_map import CELL_SIZE, GridMap, Point

logger = logging.getLogger(__name__)


@dataclass
class Route:
    """A planned path.

    Attributes:
        waypoints: Ordered list of Points, starting with the origin.
        total_length: Total Manhattan length of the path in map units.
    """

    waypoints: list = field(default_factory=list)
    total_length: float = 0.0


def _segment_length(a, b):
    """Return the Manhattan length of an axis-aligned segment."""
    return abs(a.x - b.x) + abs(a.y - b.y)


def _segment_on_street(a, b):
    """Return True if the axis-aligned segment a-b lies on a street."""
    if a.x == b.x:
        return a.x % CELL_SIZE == 0
    if a.y == b.y:
        return a.y % CELL_SIZE == 0
    return False


def _pairs(points):
    """Yield consecutive point pairs of a list."""
    return zip(points, points[1:])


class PathPlanner:
    """Plans routes on the grid, favouring street segments."""

    def __init__(self, grid_map=None):
        """Create a planner.

        Args:
            grid_map: Optional GridMap used for bounds checks.
        """
        self.grid_map = grid_map or GridMap()

    def plan(self, start, goal):
        """Plan a route from ``start`` to ``goal``.

        Args:
            start: Origin Point.
            goal: Destination Point.

        Returns:
            A Route. If the points coincide the route has one waypoint and
            zero length.

        Raises:
            ValueError: If either point lies outside the map.
        """
        for point in (start, goal):
            if not self.grid_map.is_valid(point.x, point.y):
                raise ValueError(
                    f"point ({point.x}, {point.y}) is outside the map")

        if start == goal:
            return Route(waypoints=[start], total_length=0.0)

        if start.x == goal.x or start.y == goal.y:
            candidates = [[start, goal]]
        else:
            candidates = [
                [start, Point(goal.x, start.y), goal],
                [start, Point(start.x, goal.y), goal],
            ]

        best = max(candidates, key=self._route_score)
        total = sum(_segment_length(a, b) for a, b in _pairs(best))
        logger.debug("Planned route %s -> %s with length %s",
                     start, goal, total)
        return Route(waypoints=list(best), total_length=float(total))

    def plan_multi(self, start, destinations):
        """Plan a continuous route through several destinations in order.

        Args:
            start: Origin Point.
            destinations: Ordered list of Points to visit.

        Returns:
            A Route covering all legs.
        """
        waypoints = [start]
        total = 0.0
        current = start
        for destination in destinations:
            leg = self.plan(current, destination)
            waypoints.extend(leg.waypoints[1:])
            total += leg.total_length
            current = destination
        return Route(waypoints=waypoints, total_length=total)

    @staticmethod
    def _route_score(points):
        """Score a candidate route: more street length is better.

        Ties are resolved in favour of the shorter total length, which is
        identical for all L-shaped candidates, so the first candidate wins.
        """
        street = sum(_segment_length(a, b) for a, b in _pairs(points)
                     if _segment_on_street(a, b))
        total = sum(_segment_length(a, b) for a, b in _pairs(points))
        return (street, -total)

"""Rider model: state machine, movement along planned routes, and logging.

State transitions:
    Idle (0) -> Picking up (1) when an order is accepted.
    Picking up (1) -> Delivering (2) on arrival at the restaurant.
    Delivering (2) -> Idle (0) on delivery to the customer.

Riders do not reserve map cells, so they never block one another; this
design avoids path-conflict deadlocks by construction.
"""

import logging

from globalvar import get_value
from grid_map import Point
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_X = 435
DEFAULT_Y = 385
DEFAULT_SPEED = 10

_PLANNER = PathPlanner()


def _current_tick():
    """Return the current simulation tick from the global store."""
    return get_value("current_tick", 0)


class Rider:
    """A delivery rider that moves on the grid and serves orders.

    Attributes:
        rider_id: Unique integer identifier.
        x: Current x coordinate.
        y: Current y coordinate.
        speed: Movement speed in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: List of assigned Order objects (the first is active).
        special_case: True while moving in a straight line toward a target
            located directly above, below, left or right of the rider.
        log: List of dict entries describing status changes.
        route: The currently planned Route, or None.
    """

    def __init__(self, rider_id, x=DEFAULT_X, y=DEFAULT_Y,
                 speed=DEFAULT_SPEED):
        """Create a rider and record its spawn event.

        Args:
            rider_id: Unique identifier.
            x: Initial x coordinate.
            y: Initial y coordinate.
            speed: Movement speed in units per tick.
        """
        self.rider_id = int(rider_id)
        self.x = int(x)
        self.y = int(y)
        self.speed = int(speed)
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.route = None
        self._waypoints = []
        self._record("spawn", None)

    def position(self):
        """Return the current position as a Point."""
        return Point(self.x, self.y)

    def current_target(self):
        """Return the Point the rider is currently heading to.

        Returns:
            Restaurant point while picking up, customer point while
            delivering, or None when the rider has no orders.
        """
        if not self.orders:
            return None
        order = self.orders[0]
        if self.state == STATE_DELIVERING:
            return Point(order.customer_x, order.customer_y)
        return Point(order.restaurant_x, order.restaurant_y)

    def move(self):
        """Advance the rider by one tick.

        Handles state transitions, movement by ``speed`` units along the
        planned route, arrival detection and order completion. All status
        changes are recorded in ``self.log``.
        """
        if self.orders and self.orders[0].status == "completed":
            self.orders.pop(0)

        if not self.orders:
            if self.state != STATE_IDLE:
                logger.warning("Rider %d had no orders while state=%d; "
                               "resetting to idle.", self.rider_id,
                               self.state)
                self.state = STATE_IDLE
                self.reset_route()
            return

        order = self.orders[0]
        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP
            self._record("accepted_order", order.id)
            self._plan_to(self.current_target())

        self._advance(self.speed)

        if self.state == STATE_PICKING_UP:
            if self._at_target():
                self.state = STATE_DELIVERING
                self._record("arrived_at_merchant", order.id)
                self._plan_to(self.current_target())
        elif self.state == STATE_DELIVERING and self._at_target():
            self._complete_order()

    def resync(self):
        """Repair an inconsistent movement state.

        Returns:
            True if the rider state or route was changed.
        """
        if not self.orders:
            if self.state != STATE_IDLE:
                self.state = STATE_IDLE
                self.reset_route()
                return True
            return False
        if (self.state != STATE_IDLE and not self._waypoints
                and not self._at_target()):
            self._plan_to(self.current_target())
            return True
        return False

    def reset_route(self):
        """Clear the planned route and the straight-line flag."""
        self.route = None
        self._waypoints = []
        self.special_case = False

    def to_dict(self):
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
        }

    def _record(self, event, order_id):
        """Append a status-change entry to the rider log."""
        entry = {
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "tick": _current_tick(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.info("Rider %d %s (order %s) at (%d, %d) tick %d",
                    self.rider_id, event, order_id, self.x, self.y,
                    entry["tick"])

    def _plan_to(self, goal):
        """Plan a route from the current position to ``goal``."""
        if goal is None:
            self.reset_route()
            return
        start = self.position()
        self.route = _PLANNER.plan(start, goal)
        self._waypoints = list(self.route.waypoints[1:])
        self.special_case = (start != goal
                             and (start.x == goal.x or start.y == goal.y))

    def _at_target(self):
        """Return True if the rider stands on its current target."""
        target = self.current_target()
        if target is None:
            return False
        return not self._waypoints and (self.x, self.y) == (target.x,
                                                            target.y)

    def _advance(self, budget):
        """Move along the waypoints using up to ``budget`` units.

        Segments are axis-aligned, so each step changes only one coordinate.
        """
        budget = int(budget)
        while budget > 0 and self._waypoints:
            nxt = self._waypoints[0]
            dx = nxt.x - self.x
            dy = nxt.y - self.y
            if dx == 0 and dy == 0:
                self._waypoints.pop(0)
                continue
            if dx != 0:
                step = min(budget, abs(dx))
                self.x += step if dx > 0 else -step
            else:
                step = min(budget, abs(dy))
                self.y += step if dy > 0 else -step
            budget -= step
            if (self.x, self.y) == (nxt.x, nxt.y):
                self._waypoints.pop(0)

    def _complete_order(self):
        """Finish the active order and return the rider to idle."""
        order = self.orders.pop(0)
        order.complete(_current_tick())
        self._record("delivered_to_customer", order.id)
        self.state = STATE_IDLE
        self.reset_route()

"""Order data model with coordinate conversion and lifecycle tracking."""

import itertools
import logging
import math

from globalvar import get_value
from grid_map import is_valid_coordinate, to_cell_center

logger = logging.getLogger(__name__)

_ORDER_IDS = itertools.count(1)


def _parse_position(pos, label):
    """Validate a (x, y) pair and return it as integers.

    Args:
        pos: Sequence of two integers in the range 0-800.
        label: Human-readable name used in error messages.

    Raises:
        ValueError: If the pair is malformed or out of range.
    """
    try:
        x, y = pos
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{label} must be a pair of coordinates") from exc
    for value in (x, y):
        if not is_valid_coordinate(value):
            raise ValueError(
                f"{label} coordinate {value} is out of range 0-800")
    return int(x), int(y)


class Order:
    """A food delivery order.

    Attributes:
        id: Unique integer order number.
        raw_restaurant: Restaurant coordinates as entered by the user.
        raw_customer: Customer coordinates as entered by the user.
        restaurant_x, restaurant_y: Grid-center restaurant location.
        customer_x, customer_y: Grid-center customer location.
        start_time: Tick at which the order was created.
        completion_time: Tick of delivery, or None.
        distance: Euclidean distance between restaurant and customer.
        time_taken: Ticks from creation to delivery, or None.
        rider_id: Assigned rider id, or None.
        status: "pending", "assigned" or "completed".
    """

    def __init__(self, res_pos, cus_pos):
        """Create an order and convert coordinates to grid centers.

        Args:
            res_pos: (x, y) restaurant coordinates (0-800).
            cus_pos: (x, y) customer coordinates (0-800).

        Raises:
            ValueError: If a coordinate is malformed or out of range.
        """
        raw_rx, raw_ry = _parse_position(res_pos, "restaurant")
        raw_cx, raw_cy = _parse_position(cus_pos, "customer")
        self.id = next(_ORDER_IDS)
        self.raw_restaurant = (raw_rx, raw_ry)
        self.raw_customer = (raw_cx, raw_cy)
        self.restaurant_x = to_cell_center(raw_rx)
        self.restaurant_y = to_cell_center(raw_ry)
        self.customer_x = to_cell_center(raw_cx)
        self.customer_y = to_cell_center(raw_cy)
        self.start_time = get_value("current_tick", 0)
        self.completion_time = None
        self.distance = math.hypot(self.customer_x - self.restaurant_x,
                                   self.customer_y - self.restaurant_y)
        self.time_taken = None
        self.rider_id = None
        self.status = "pending"
        logger.info("Order %d created: restaurant (%d, %d) customer (%d, %d)",
                    self.id, self.restaurant_x, self.restaurant_y,
                    self.customer_x, self.customer_y)

    def assign_to(self, rider_id):
        """Mark the order as assigned to the rider with ``rider_id``.

        Args:
            rider_id: Identifier of the rider receiving the order.

        Raises:
            ValueError: If the order is already completed.
        """
        if self.status == "completed":
            raise ValueError(f"order {self.id} is already completed")
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick):
        """Mark the order delivered and record completion data.

        Args:
            tick: Simulation tick at which delivery happened.
        """
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time
        logger.info("Order %d completed at tick %d", self.id, tick)

    @property
    def efficiency(self):
        """Distance covered per tick for a completed order, else None."""
        if self.status != "completed" or not self.time_taken:
            return None
        return self.distance / self.time_taken

    def to_dict(self):
        """Return a JSON-serialisable snapshot of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "rider_id": self.rider_id,
            "status": self.status,
        }

"""Order management and rider assignment.

Assignment strategy: the rider closest (Euclidean distance) to the restaurant
receives the order. Ties are broken by the smaller current workload so that
load is spread across riders.
"""

import logging
import math

from globalvar import get_value
from order_model import Order

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Find the best rider for an order.

    The distance is ``sqrt((rider_x - restaurant_x)^2 + (rider_y -
    restaurant_y)^2)``. Among riders, the one with the smallest distance wins;
    ties are resolved by fewer assigned orders, then by rider id.

    Args:
        order: The Order that needs a rider.

    Returns:
        The selected Rider, or None when no riders exist.
    """
    riders = get_value("riders") or []
    best = None
    best_key = None
    for rider in riders:
        distance = math.hypot(rider.x - order.restaurant_x,
                              rider.y - order.restaurant_y)
        key = (round(distance, 6), len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates, assigns and tracks orders.

    Attributes:

"""Status text and ASCII map rendering for the dispatch simulator.

The map is drawn on a 17x17 character canvas: one character represents 50
map units, so grid lines (multiples of 100) fall on even indices and cell
centers fall on odd indices. Symbols:

* ``R`` rider, ``S`` restaurant, ``C`` customer, ``.`` empty cell center,
  ``#`` street, ``*`` planned route.
"""

import logging

from grid_map import CELL_SIZE, MAP_SIZE
from rider import STATE_NAMES

logger = logging.getLogger(__name__)

SCALE = 50
CANVAS_SIZE = MAP_SIZE // SCALE + 1


class DisplayManager:
    """Formats monitoring output and renders the command-line map.

    Attributes:
        map_symbol_rider: Character for riders.
        map_symbol_restaurant: Character for restaurants.
        map_symbol_customer: Character for customers.
        map_symbol_empty: Character for empty cell centers.
        map_symbol_street: Character for streets.
        map_symbol_route: Character for planned routes.
    """

    def __init__(self):
        """Create a display manager with the standard symbols."""
        self.map_symbol_rider = "R"
        self.map_symbol_restaurant = "S"
        self.map_symbol_customer = "C"
        self.map_symbol_empty = "."
        self.map_symbol_street = "#"
        self.map_symbol_route = "*"

    @staticmethod
    def _index(value):
        """Convert a map coordinate to a canvas index."""
        return max(0, min(CANVAS_SIZE - 1, value // SCALE))

    def render_map(self, riders, orders, route_waypoints=None):
        """Render the 8x8 grid with riders, active orders and an optional route.

        Args:
            riders: Iterable of Rider objects.
            orders: Iterable of Order objects; completed orders are skipped.
            route_waypoints: Optional list of Points to draw as a route.

        Returns:
            Multi-line string containing the map and a legend.
        """
        canvas = [[" "] * CANVAS_SIZE for _ in range(CANVAS_SIZE)]

        for row in range(CANVAS_SIZE):
            for col in range(CANVAS_SIZE):
                if row % 2 == 1 and col % 2 == 1:
                    canvas[row][col] = self.map_symbol_empty

        for value in range(0, MAP_SIZE + 1, CELL_SIZE):
            index = self._index(value)
            for other in range(CANVAS_SIZE):
                canvas[index][other] = self.map_symbol_street
                canvas[other][index] = self.map_symbol_street

        if route_waypoints:
            self._draw_route(canvas, route_waypoints)

        for order in orders:
            if order.status == "completed":
                continue
            canvas[self._index(order.restaurant_y)][
                self._index(order.restaurant_x)] = self.map_symbol_restaurant
            canvas[self._index(order.customer_y)][
                self._index(order.customer_x)] = self.map_symbol_customer

        for rider in riders:
            canvas[self._index(rider.y)][
                self._index(rider.x)] = self.map_symbol_rider

        lines = ["".join(row) for row in canvas]
        legend = ("Legend: R rider  S restaurant  C customer  "
                  "# street  . cell center  * route")
        return "\n".join(lines) + "\n" + legend

    def _draw_route(self, canvas, waypoints):
        """Draw axis-aligned route segments onto the canvas."""
        for start, end in zip(waypoints, waypoints[1:]):
            ax, ay = self._index(start.x), self._index(start.y)
            bx, by = self._index(end.x), self._index(end.y)
            if ax == bx:
                for row in range(min(ay, by), max(ay, by) + 1):
                    canvas[row][ax] = self.map_symbol_route
            elif ay == by:
                for col in range(min(ax, bx), max(ax, bx) + 1):
                    canvas[ay][col] = self.map_symbol_route
        if len(waypoints) == 1:
            canvas[self._index(waypoints[0].y)][
                self._index(waypoints[0].x)] = self.map_symbol_route

    def format_status(self, system):
        """Return a snapshot of funds, riders and their orders.

        Args:
            system: The SystemManager instance.
        """
        lines = [
            f"=== System Status (tick {system.current_tick}) ===",
            f"Funds: {system.funds}",
            f"Riders: {len(system.riders)}",
            f"Pending orders: {system.order_manager.pending_count()}",
        ]
        if not system.riders:
            lines.append("No riders on duty.")
        for rider in system.riders:
            order_ids = [order.id for order in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id} | position ({rider.x}, {rider.y})"
                f" | state: {STATE_NAMES.get(rider.state, 'unknown')}"
                f" | orders: {len(rider.orders)} {order_ids}")
        return "\n".join(lines)

    def format_orders(self, orders):
        """Return a table of order details.

        Args:
            orders: Iterable of Order objects.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} | {'start':>6} | {'restaurant':>12} | "
                  f"{'customer':>12} | {'distance':>8} | {'taken':>6} | "
                  f"{'rider':>5} | status")
        lines = [header, "-" * len(header)]
        for order in orders:
            taken = "-" if order.time_taken is None else str(order.time_taken)
            rider = "-" if order.rider_id is None else str(order.rider_id)
            lines.append(
                f"{order.id:>4} | {order.start_time:>6} | "
                f"{str((order.restaurant_x, order.restaurant_y)):>12} | "
                f"{str((order.customer_x, order.customer_y)):>12} | "
                f"{order.distance:>8.1f} | {taken:>6} | {rider:>5} | "
                f"{order.status}")
        return "\n".join(lines)

    def format_rider_log(self, riders):
        """Return every recorded status change for all riders.

        Args:
            riders: Iterable of Rider objects.
        """
        lines = []
        for rider in riders:

            for entry in rider.log:

                lines.append(
                    f"  tick {entry['tick']:>4} | {entry['event']:<22} | "
                    f"order {order:>4} | at ({entry['x']}, {entry['y']})")
        if not lines:
            return "No riders on duty."
        return "\n".join(lines)

"""System manager: funds, riders, recruitment, ticks and state persistence."""

import json
import logging
from dataclasses import asdict

import globalvar
from config import Configuration
from display import DisplayManager
from grid_map import GridMap
from order import OrderManager
from path_planner import PathPlanner
from rider import STATE_IDLE, STATE_PICKING_UP, Rider

logger = logging.getLogger(__name__)


class SystemManager:
    """Coordinates riders, orders, funds and the simulation clock.

    Attributes:
        config: Configuration parameters.
        funds: Current system funds (may become negative).
        rider_id_counter: Last issued rider id.
        is_running: Main loop control flag.
        current_tick: Number of simulated ticks so far.
        grid_map: The 8x8 grid model.
        planner: Shared PathPlanner.
        order_manager: OrderManager instance.
        display: DisplayManager instance.
    """

    def __init__(self, config=None):
        """Create the system with the given configuration.

        Args:
            config: Optional Configuration; defaults are used when omitted.
        """
        self.config = config or Configuration()
        self.funds = int(self.config.initial_funds)
        self.rider_id_counter = 0
        self.is_running = True
        self.current_tick = 0
        self.grid_map = GridMap()
        self.planner = PathPlanner(self.grid_map)
        self.order_manager = OrderManager()
        self.display = DisplayManager()
        if globalvar.get_value("riders") is None:
            globalvar.set_value("riders", [])
        globalvar.set_value("current_tick", self.current_tick)

    @property
    def riders(self):
        """The live list of riders stored in the global variable store."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    def idle_riders(self):
        """Return riders that are idle and have no assigned orders."""
        return [rider for rider in self.riders
                if rider.state == STATE_IDLE and not rider.orders]

    def find_rider(self, rider_id):
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def recruit_rider(self):
        """Create a new rider, deducting the recruitment cost.

        Returns:
            The newly created Rider.
        """
        self.rider_id_counter += 1
        rider = Rider(self.rider_id_counter,
                      self.config.initial_rider_x,
                      self.config.initial_rider_y,
                      self.config.initial_rider_speed)
        self.funds -= self.config.rider_recruitment_cost
        self.riders.append(rider)
        logger.info("Recruited rider %d; funds now %d", rider.rider_id,
                    self.funds)
        return rider

    def maybe_recruit(self):
        """Recruit riders while pending orders exceed idle riders.

        Recruitment requires funds of at least the configured threshold.

        Returns:
            List of riders recruited during this call.
        """
        new_riders = []
        idle = len(self.idle_riders())
        while (self.order_manager.pending_count() > idle
               and self.funds >= self.config.recruitment_fund_threshold):
            new_riders.append(self.recruit_rider())
            idle += 1
        return new_riders

    def submit_order(self, restaurant_pos, customer_pos):
        """Create an order, recruit if needed and assign it.

        Args:
            restaurant_pos: (x, y) restaurant coordinates.
            customer_pos: (x, y) customer coordinates.

        Returns:
            Tuple (order, added) where ``added`` is True when a new rider
            was recruited for this order.
        """
        order = self.order_manager.create_order(restaurant_pos, customer_pos)
        added = bool(self.maybe_recruit())
        self.order_manager.assign_pending()
        return order, added

    def tick(self):
        """Advance the whole simulation by one tick.

        Order: update the clock, recruit and assign pending orders, then move
        every rider. Exceptions in one rider are logged and do not stop the
        others.
        """
        self.current_tick += 1
        globalvar.set_value("current_tick", self.current_tick)
        try:
            self.maybe_recruit()
            self.order_manager.assign_pending()
        except Exception:
            logger.exception("Error while dispatching pending orders")
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:
                logger.exception("Error while moving rider %d",
                                 rider.rider_id)
                rider.resync()

    def run_until_order_done(self, order, max_ticks=None):
        """Run ticks until ``order`` is delivered or the limit is reached.

        Args:
            order: The Order to monitor.
            max_ticks: Maximum ticks to simulate; defaults to configuration.

        Returns:
            True if the order was completed.
        """
        limit = max_ticks or self.config.max_ticks_per_order
        for _ in range(limit):
            if order.status == "completed":
                return True
            self.tick()
        return order.status == "completed"

    def check_consistency(self):
        """Return a list of human-readable consistency problems.

        Problems are detected without modifying state.
        """
        problems = []
        for rider in self.riders:
            if rider.state != STATE_IDLE and not rider.orders:
                problems.append(f"rider {rider.rider_id} is busy without "
                                "orders")
            if rider.state == STATE_PICKING_UP and rider.orders:
                if rider.orders[0].status == "completed":
                    problems.append(f"rider {rider.rider_id} holds a "
                                    "completed order")
        return problems

    def repair_state(self):
        """Repair inconsistent rider states and return the number of fixes."""
        fixes = 0
        for rider in self.riders:
            before = len(rider.orders)
            rider.orders = [order for order in rider.orders
                            if order.status != "completed"]
            if len(rider.orders) != before:
                fixes += 1
            if rider.resync():
                fixes += 1
        if fixes:
            logger.warning("Repaired %d inconsistent rider state(s)", fixes)
        return fixes

    def to_dict(self):
        """Return a JSON-serialisable snapshot of the whole system."""
        return {
            "tick": self.current_tick,
            "funds": self.funds,
            "rider_id_counter": self.rider_id_counter,
            "config": asdict(self.config),
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict()
                       for order in self.order_manager.orders],
        }

    def save_state(self, path):
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination file path.

        Returns:
            True on success, False if the file could not be written.
        """
        try:
            with open(path, "w", encoding="utf-8") as handle:
                json.dump(self.to_dict(), handle, indent=2)
            logger.info("State saved to %s", path)
            return True
        except OSError as exc:
            logger.error("Could not save state to %s: %s", path, exc)
            return False


def init_system(config_path=None):
    """Initialise the global store and create the system manager.

    Sets funds to the configured initial value (default 1000), an empty rider
    list, rider id counter 0, running state True and tick 0.

    Args:
        config_path: Optional path to a JSON configuration file.

    Returns:
        The initialised SystemManager.
    """
    globalvar._init()
    config = Configuration.load(config_path) if config_path else Configuration()
    system = SystemManager(config)
    logger.info("System initialised with funds %d", system.funds)
    return system

class InputError(ValueError):
    """Raised when user input fails validation."""


class InputManager:
    """Parses commands and dispatches them to the system.

    Attributes:
        system: The SystemManager being controlled.
        command_history: Executed command lines, oldest first.
    """

    def __init__(self, system, input_func=None, output_func=None):
        """Create an input manager.

        Args:
            system: The SystemManager instance.
            input_func: Optional replacement for ``input``.
            output_func: Optional replacement for ``print``.
        """
        self.system = system
        self.command_history = []
        self._input = input_func or input
        self._output = output_func or print
        self._handlers = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "save": self._cmd_save,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
            "exit": self._cmd_quit,
        }

    def run(self):
        """Run the interactive loop until the system stops or input ends."""
        self._output(WELCOME_TEXT)
        while self.system.is_running:
            try:
                line = self._input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self._output("\nExiting.")
                self.system.is_running = False
                break
            self.handle_line(line)

    def handle_line(self, line):
        """Process one command line.

        Args:
            line: Raw text entered by the user.
        """
        text = (line or "").strip()
        if not text:
            return
        if text == "!!":
            if not self.command_history:
                self._output("No command in history.")
                return
            text = self.command_history[-1]
        elif text.startswith("!") and text[1:].isdigit():
            index = int(text[1:])
            if not 1 <= index <= len(self.command_history):
                self._output(f"No history entry {index}.")
                return
            text = self.command_history[index - 1]

        self.command_history.append(text)
        parts = text.split()
        command = COMMAND_ALIASES.get(parts[0].lower(), parts[0].lower())
        handler = self._handlers.get(command)
        if handler is None:
            self._output(f"Unknown command '{parts[0]}'. Type 'help'.")
            return
        try:
            handler(parts[1:])
        except InputError as exc:
            self._output(f"Input error: {exc}")
        except Exception as exc:
            logger.exception("Command '%s' failed", command)
            self._output(f"Error while running '{command}': {exc}")
            fixes = self.system.repair_state()
            if fixes:
                self._output(f"System state recovered ({fixes} fix(es)).")

    @staticmethod
    def parse_coordinates(tokens):
        """Validate four coordinate tokens.

        Args:
            tokens: Sequence of strings: restaurant_x, restaurant_y,
                customer_x, customer_y.

        Returns:
            List of four integers.

        Raises:
            InputError: On wrong count, non-integer or out-of-range values.
        """
        if len(tokens) != 4:
            raise InputError(
                f"expected 4 integers (restaurant x y, customer x y), "
                f"got {len(tokens)}")
        values = []
        names = ["restaurant x", "restaurant y", "customer x", "customer y"]
        for name, token in zip(names, tokens):
            try:
                value = int(token)
            except ValueError as exc:
                raise InputError(
                    f"{name} '{token}' is not an integer") from exc
            if not is_valid_coordinate(value):
                raise InputError(
                    f"{name} {value} is out of range 0-{MAP_SIZE}")
            values.append(value)
        return values

    def _cmd_start(self, args):
        """Handle ``start``: place an order and run it to completion."""
        if args:
            tokens = args
        else:
            tokens = self._input(
                f"Enter restaurant_x restaurant_y customer_x customer_y "

        rx, ry, cx, cy = self.parse_coordinates(tokens)
        order, added = self.system.submit_order((rx, ry), (cx, cy))

        self._output(f"Order #{order.id} created.")
        self._output(f"Restaurant grid point : ({order.restaurant_x}, "
                     f"{order.restaurant_y})")
        self._output(f"Customer grid point   : ({order.customer_x}, "
                     f"{order.customer_y})")
        self._output(f"New rider added       : {'yes' if added else 'no'}")

        if order.rider_id is None:
            self._output("No rider is available yet; the order is pending.")
            return

        rider = self.system.find_rider(order.rider_id)
        route = self.system.planner.plan_multi(
            rider.position(),
            [Point(order.restaurant_x, order.restaurant_y),
             Point(order.customer_x, order.customer_y)])
        eta = math.ceil(route.total_length / rider.speed) if rider.speed else 0
        self._output(
            f"Assigned rider #{rider.rider_id} | speed {rider.speed} "
            f"units/tick | route {route.total_length:.0f} units | "
            f"estimated {eta} ticks")
        self._output(self.system.display.render_map(
            self.system.riders, self.system.order_manager.orders,

        done = self.system.run_until_order_done(order)
        if done:
            self._output(
                f"Order #{order.id} delivered at tick {order.completion_time}"
                f" (time taken {order.time_taken} ticks, distance "

                f"{order.efficiency:.2f} units/tick).")
        else:
            self._output(
                f"Order #{order.id} not delivered within "

            return
        for index, entry in enumerate(self.command_history, start=1):
            self._output(f"{index:>4}  {entry}")

    def _cmd_help(self, args):
        """Handle ``help``."""
        self._output(HELP_TEXT)

    def _cmd_quit(self, args):
        """Handle ``quit`` and ``exit``."""
        self.system.is_running = False
        self._output("Goodbye.")

def parse_args(argv=None):
    """Parse command-line arguments.

    Args:
        argv: Optional argument list; defaults to ``sys.argv[1:]``.

    """

def main(argv=None):
    """Configure logging, build the system and run the command loop.

    Args:
        argv: Optional argument list for testing.

    Returns:
        Process exit code.
    """
    args = parse_args(argv)
    level = logging.DEBUG if args.debug else logging.WARNING
    logging.basicConfig(level=level,
                        format="%(asctime)s %(levelname)s %(name)s: "
                               "%(message)s")
    try:
        system = init_system(args.config)
        InputManager(system).run()

class ListManager:
    """Loads, validates and analyses participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it. Returns (users, errors)."""
        path = str(path)
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        raw = None
        for encoding in ("utf-8-sig", "gb18030"):
            try:
                with open(path, "r", encoding=encoding) as handle:
                    raw = handle.read()
                break
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"File Error: cannot read '{path}': {exc}"]

        if raw is None:
            return [], [f"File Error: cannot decode '{path}'"]
        return self._parse_lines(raw.splitlines())

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines are ignored."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = [part.strip() for part in text.split(",")]
            if len(parts) != 2 or not all(parts):
                errors.append(
                    f"Line {line_no}: Format Error - '{text}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Remove duplicate names. Returns (valid_users, duplicates).

        The first occurrence of a name is kept; later occurrences are
        reported as duplicates.
        """
        valid: list[User] = []
        duplicates: list[User] = []
        seen: set[str] = set()
        for user in users:
            key = user.name.casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        """Return an ordered mapping of department group -> number of people."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.models import Prize

DEFAULT_PRIZES: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 6, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),

class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        """Append a prize to the end of the lottery order.

        Raises ValueError for an empty name, a non-positive quantity or a
        duplicate prize name.
        """
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        groups = list(dict.fromkeys(groups))
        drawer = None
        if associated_drawer and associated_drawer.strip():
            drawer = associated_drawer.strip()

        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order.

        Returns False if `order` is not a permutation of all current positions.
        """
        n = len(self._prizes)
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if len(positions) != n or sorted(positions) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in positions]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
MA_JIAQI = "Ma Jiaqi"
RESULT_DRAWERS = ("Chen Qixian", "Jin Duo")


def _norm(text: str) -> str:
    return (text or "").strip().casefold()


class LotteryManager:
    """Applies lottery rules and draws winners for each prize."""

    def __init__(self, users):
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Results of the last two draws by Chen Qixian / Jin Duo.
        # True means that round produced at least one Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []
        self._winner_names: set[str] = set()

    # ----- rule helpers -------------------------------------------------
    @staticmethod
    def _is_drawer(prize: Prize, drawer_name: str) -> bool:
        return _norm(prize.associated_drawer) == _norm(drawer_name)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return _norm(user.department) == _norm(CHENG_RU_GROUP)

    def _fallback_active(self, prize: Prize) -> bool:
        """Cheng Ru protection: Ma Jiaqi's pool is forced to Cheng Ru members
        when neither of the last two Chen Qixian / Jin Duo results contained
        a Cheng Ru member."""
        return (
            self._is_drawer(prize, MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def is_winner(self, user: User) -> bool:
        return _norm(user.name) in self._winner_names

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return users who may still win the given prize."""
        candidates = [u for u in self.users if not self.is_winner(u)]

        # Special rule: Li Bin only filters previous winners.
        if self._is_drawer(prize, LI_BIN):
            return candidates

        # Fallback rule: forced to Cheng Ru members, overriding avoidance groups.
        if self._fallback_active(prize):
            return [u for u in candidates if self._is_cheng_ru(u)]

        # Basic rule: exclude previous winners and avoidance groups.
        avoided = {_norm(group) for group in prize.avoid_groups}
        return [u for u in candidates if _norm(u.department) not in avoided]

    def describe_rule(self, prize: Prize) -> str:
        """Human readable description of the rules applied to a prize round."""
        parts: list[str] = []
        if self._is_drawer(prize, LI_BIN):
            parts.append(
                "Special rule (Li Bin): only previous winners are excluded; "
                "avoidance groups do not apply."
            )
        elif self._fallback_active(prize):
            parts.append(
                "Fallback protection active: only Cheng Ru Group members are "
                "eligible (avoidance groups overridden)."
            )
        else:
            parts.append("Previous winners are excluded.")
            if prize.avoid_groups:
                parts.append("Avoided groups: " + ", ".join(prize.avoid_groups) + ".")
            else:
                parts.append("No avoidance groups.")
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}.")
        return " ".join(parts)

    # ----- drawing ------------------------------------------------------
    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winner_names.add(_norm(winner.name))
        self.result_manager.add_winner(winner, prize)
        return winner

    def record_round(self, prize: Prize, winners: list[User]) -> None:
        """Update fallback state after a round has finished."""
        if self._is_drawer(prize, MA_JIAQI):
            # The fallback window is consumed by this round.
            self.last_two_draw_results = []
            return
        if any(self._is_drawer(prize, drawer) for drawer in RESULT_DRAWERS):
            has_cheng_ru = any(self._is_cheng_ru(u) for u in winners)
            self.last_two_draw_results = (self.last_two_draw_results + [has_cheng_ru])[-2:]

"""Winner recording, grouping and CSV export."""

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User

_ORDINALS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}
_LEVEL_RE = re.compile(
    r"^\s*(first|second|third|fourth|fifth|sixth|seventh|eighth|ninth|tenth)\s+prize\b",
    re.IGNORECASE,
)


def prize_level(name: str) -> int:
    """Smaller number means higher prize level. Unranked prizes come last."""
    match = _LEVEL_RE.match(name or "")
    if match:
        return _ORDINALS[match.group(1).lower()]
    return len(_ORDINALS) + 1


def sorted_prizes(prizes) -> list[Prize]:
    """Sort prizes by level from high to low, keeping addition order within a level."""
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: (prize_level(item[1].name), item[0]))
    return [prize for _, prize in indexed]


@dataclass(frozen=True)
class Winner:
    """A winning record.

    order: winning time sequence number (1-based, across all rounds).
    """

    user: User
    prize_name: str
    order: int


class ResultManager:
    """Keeps all winners in winning order."""

    def __init__(self):
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(user=user, prize_name=prize.name, order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        """Winners grouped by prize name, prizes ordered from high to low level."""
        grouped: dict[str, list[Winner]] = {}
        for prize in sorted_prizes(prizes):
            grouped[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return grouped

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export winners to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        grouped = self.get_winners_by_prize(prizes)
        rows = [winner for winners in grouped.values() for winner in winners]
        if not rows:
            return False, "No winners to export."

        filename = f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in rows:
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.order,
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

class InputHub:
    """Reads stdin in a background thread so commands can be checked while drawing."""

    _EOF = object()

    def __init__(self, stream=None):
        self._stream = stream or sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._eof = False
        self._thread = threading.Thread(target=self._reader, daemon=True)
        self._thread.start()

    def _reader(self):
        while True:
            try:
                line = self._stream.readline()
            except (OSError, ValueError):
                line = ""
            if not line:
                self._queue.put(self._EOF)
                return
            self._queue.put(line.strip())

    def _convert(self, item):
        if item is self._EOF:
            self._eof = True
            raise EOFError
        return item

    def read_line(self) -> str:
        """Blocking read of the next input line."""
        if self._eof:
            raise EOFError
        return self._convert(self._queue.get())

    def poll(self) -> str | None:
        """Non-blocking read. Returns None when no line is waiting."""
        if self._eof:
            raise EOFError
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        return self._convert(item)


def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"

    def _parse_groups(self, text: str, names: list[str]) -> list[str]:
        """Parse avoidance group selection by serial number or name."""
        if not text.strip():
            return []
        result: list[str] = []
        for token in text.split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit():
                index = int(token)
                if not names or not 1 <= index <= len(names):
                    raise ValueError(f"Invalid group number: {token}")
                result.append(names[index - 1])
            elif names:
                match = next((g for g in names if g.casefold() == token.casefold()), None)
                if match is None:
                    raise ValueError(f"Unknown group: {token}")
                result.append(match)
            else:
                result.append(token)
        return list(dict.fromkeys(result))

            print("Please load a participant list first.")
            return

            print("Please configure prizes first.")
            return

    def _run_round(self, prize):
        lottery = self.lottery
        print("\n" + "=" * 60)
        print(f"Round: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {lottery.describe_rule(prize)}")
        print("=" * 60)

        round_winners = []
        for position in range(1, prize.quantity + 1):
            print(f"Currently drawing the {_ordinal(position)} winner...", flush=True)
            if not self._checkpoint(DRAW_DELAY_SECONDS):
                print("Round terminated. Remaining draws skipped.")
                break
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize; the round ends early.")
                break
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        lottery.record_round(prize, round_winners)
        self.completed_prizes.add(prize.name)
        print(f"Round finished: {len(round_winners)}/{prize.quantity} winner(s) drawn.")
        for index, user in enumerate(round_winners, start=1):
            print(f"  {index}. {user.name} ({user.department})")

    def _checkpoint(self, duration: float) -> bool:
        """Wait for the draw interval while handling control commands.

        Returns False if the current round must be terminated.
        """
        deadline = time.monotonic() + duration
        while True:
            command = self.hub.poll()
            if command is None:
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    return True
                time.sleep(min(0.05, remaining))
                continue

            command = command.lower()
            if command == "pause":
                print("[PAUSED] Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    answer = self.hub.read_line().lower()
                    if answer == "continue":
                        print("[RESUMED]")
                        return True
                    if answer == "terminate":
                        print("[TERMINATED]")
                        return False
                    print("Type 'continue' or 'terminate'.")
            elif command == "terminate":
                print("[TERMINATED]")
                return False
            elif command == "continue":
                print("Lottery is not paused.")
            elif command == "":
                continue
            else:
                print("Unknown command. Available during lottery: pause, continue, terminate")

    # ----- results ------------------------------------------------------
    def _show_summary(self):
        if self.lottery is None:
            print("No lottery results yet.")
            return
        prizes = self.prize_manager.get_prizes()
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        print("\n=== Winner List (by prize level, then winning order) ===")
        if not any(grouped.values()):
            print("No winners yet.")
            return
        for prize_name, winners in grouped.items():
            print(f"{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  {winner.order}. {winner.user.name} | "
                    f"{winner.user.department} | {prize_name}"
                )

    def _export_results(self):
        if self.lottery is None:
            print("No lottery results to export.")
            return
        success, message = self.lottery.result_manager.export_to_csv(
            self.prize_manager.get_prizes()
        )
        print(message)

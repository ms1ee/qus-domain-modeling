"""Loading and saving of the Beijing Subway network data file.

The data file has the following structure:
    {"stations":    [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str, "distance": int, "line": str}]}
Distances are in meters.
"""

import json
import os


class DataLoadError(Exception):
    """Raised when the subway data file cannot be read or is malformed."""


# Absolute path of src/subway_data.json, valid regardless of the working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def _is_number(value):
    """Return True for int/float values that are not booleans."""
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def validate_subway_data(data):
    """Check the structure of the loaded subway data.

    Raises DataLoadError if the structure does not match the expected schema.
    """
    if not isinstance(data, dict):
        raise DataLoadError("Subway data must be a JSON object")

    stations = data.get("stations")
    connections = data.get("connections")
    if not isinstance(stations, list):
        raise DataLoadError("Subway data must contain a 'stations' list")
    if not isinstance(connections, list):
        raise DataLoadError("Subway data must contain a 'connections' list")

    for index, station in enumerate(stations):
        if not isinstance(station, dict) or not isinstance(station.get("name"), str):
            raise DataLoadError(f"Malformed station entry at index {index}")
        lines = station.get("lines", [])
        if not isinstance(lines, list) or not all(isinstance(line, str) for line in lines):
            raise DataLoadError(f"Station '{station.get('name')}' has malformed 'lines'")

    for index, conn in enumerate(connections):
        if not isinstance(conn, dict):
            raise DataLoadError(f"Malformed connection entry at index {index}")
        if not isinstance(conn.get("source"), str) or not isinstance(conn.get("dest"), str):
            raise DataLoadError(f"Connection at index {index} needs string 'source' and 'dest'")
        if not isinstance(conn.get("line"), str):
            raise DataLoadError(f"Connection at index {index} needs a string 'line'")
        if not _is_number(conn.get("distance")):
            raise DataLoadError(f"Connection at index {index} needs a numeric 'distance'")


def load_subway_data(path):
    """Load the subway data JSON file and return its content as a dict."""
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Subway data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataLoadError(f"Cannot read subway data file: {exc}") from exc

    validate_subway_data(data)
    return data


def save_subway_data(data, path):
    """Write the subway data to a JSON file atomically."""
    temp_path = path + ".tmp"
    try:
        with open(temp_path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, ensure_ascii=False, indent=2)
        os.replace(temp_path, path)
    except OSError as exc:
        raise DataLoadError(f"Cannot write subway data file: {exc}") from exc

"""Binary min-heap priority queue used by the Dijkstra implementation."""


class MinPQ:
    """A min priority queue implemented as an explicit binary heap.

    Each entry is stored as (priority, sequence_number, item). The sequence
    number makes ordering deterministic when priorities are equal (FIFO for ties).
    Insert and delete-min run in O(log n) time.
    """

    def __init__(self):
        self._heap = []
        self._seq = 0

    def insert(self, priority, item):
        """Insert an item with the given priority."""
        self._heap.append((priority, self._seq, item))
        self._seq += 1
        self._sift_up(len(self._heap) - 1)

    def del_min(self):
        """Remove and return the (priority, item) pair with the smallest priority."""
        if not self._heap:
            raise IndexError("delete-min from an empty priority queue")
        top = self._heap[0]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._sift_down(0)
        return top[0], top[2]

    def peek_min(self):
        """Return the (priority, item) pair with the smallest priority without removing it."""
        if not self._heap:
            raise IndexError("peek on an empty priority queue")
        return self._heap[0][0], self._heap[0][2]

    def is_empty(self):
        """Return True if the queue has no elements."""
        return not self._heap

    def size(self):
        """Return the number of elements in the queue."""
        return len(self._heap)

    def _less(self, i, j):
        a = self._heap[i]
        b = self._heap[j]
        return (a[0], a[1]) < (b[0], b[1])

    def _swap(self, i, j):
        self._heap[i], self._heap[j] = self._heap[j], self._heap[i]

    def _sift_up(self, k):
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._swap(k, parent)
                k = parent
            else:
                break

    def _sift_down(self, k):
        n = len(self._heap)
        while True:
            left = 2 * k + 1
            right = left + 1
            smallest = k
            if left < n and self._less(left, smallest):
                smallest = left
            if right < n and self._less(right, smallest):
                smallest = right
            if smallest == k:
                break
            self._swap(k, smallest)
            k = smallest

"""Graph data structures for the subway network.

Station numbers (integers) are assigned by the caller (see subway_network.py).
The Graph stores an undirected weighted adjacency list.
"""


class Edge:
    """An undirected weighted edge between two stations.

    Attributes:
        source: station number of one endpoint
        dest:   station number of the other endpoint
        distance: physical distance in meters
        line:   name of the subway line the edge belongs to
    The ``weight`` attribute is an alias of ``distance``.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        if distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Weight used by shortest path algorithms (equal to distance in meters)."""
        return self.distance

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex} is not an endpoint of this edge")

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph represented by adjacency lists.

    Attributes:
        station_lines: dict mapping station name -> list of line names
    """

    def __init__(self):
        self._adj = {}
        self.station_lines = {}
        self._edge_count = 0

    def add_node(self, v):
        """Ensure that vertex ``v`` exists in the graph."""
        self._adj.setdefault(v, [])

    def has_node(self, v):
        """Return True if vertex ``v`` exists."""
        return v in self._adj

    def nodes(self):
        """Return a sorted list of all vertex numbers."""
        return sorted(self._adj)

    def set_station_lines(self, station_name, lines):
        """Record the (de-duplicated, ordered) lines serving a station."""
        unique = []
        for line in lines:
            if line not in unique:
                unique.append(line)
        self.station_lines[station_name] = unique

    def add_edge(self, edge):
        """Add an undirected edge; both endpoints are created if necessary."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self._adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self._adj[edge.dest].append(edge)
        self._edge_count += 1

    def adjacent(self, v):
        """Return a list of edges incident to vertex ``v``."""
        if v not in self._adj:
            raise KeyError(f"Unknown station number: {v}")
        return list(self._adj[v])

    def degree(self, v):
        """Return the number of edges incident to vertex ``v``."""
        return len(self.adjacent(v))

    def edges(self):
        """Return every distinct edge exactly once."""
        seen = set()
        result = []
        for incident in self._adj.values():
            for edge in incident:
                if id(edge) not in seen:
                    seen.add(id(edge))
                    result.append(edge)
        return result

    def remove_edge(self, source, dest, line=None):
        """Remove edges between two vertices (optionally only on ``line``).

        Returns True if at least one edge was removed.
        """
        if source not in self._adj:
            return False
        to_remove = []
        for edge in self._adj[source]:
            same_pair = {edge.source, edge.dest} == {source, dest}
            if same_pair and (line is None or edge.line == line):
                to_remove.append(edge)
        if not to_remove:
            return False
        removed_ids = {id(edge) for edge in to_remove}
        for vertex in {source, dest}:
            if vertex in self._adj:
                self._adj[vertex] = [e for e in self._adj[vertex] if id(e) not in removed_ids]
        self._edge_count -= len(to_remove)
        return True

    def vertex_count(self):
        """Return the number of vertices."""
        return len(self._adj)

    def edge_count(self):
        """Return the number of edges."""
        return self._edge_count

"""Shortest path computation using Dijkstra's algorithm with a binary-heap priority queue.

The search runs over states (station, line of arrival edge, transfers so far), which
allows enforcing a maximum transfer limit and computing the number of line changes
along the chosen path. Complexity is O((V+E) log V) for an unconstrained search,
multiplied by the (small) number of transfer levels when a transfer limit is applied.
"""

from min_pq import MinPQ

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Transfer cap used by the search when no maximum transfer limit is configured.
DEFAULT_TRANSFER_CAP = 10
# Extra search cost (in meters-equivalent) for riding a line not in the preference list.
PREFERENCE_PENALTY = 1000.0
# Extra search cost per transfer in comprehensive mode (meters-equivalent).
COMPREHENSIVE_TRANSFER_PENALTY = 500.0
# Very large cost per transfer in least_transfer mode, so transfers dominate distance.
LEAST_TRANSFER_WEIGHT = 1e9


class PathfindingConfig:
    """Configuration for a path query.

    Attributes:
        mode: one of MODES
        max_transfers: maximum allowed number of transfers, or None for no explicit limit
        preferred_lines: list of preferred line names (may be empty)
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in MODES:
            raise ValueError(f"Unknown mode '{mode}', expected one of {', '.join(MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        preferred = list(preferred_lines) if preferred_lines else []
        if not all(isinstance(line, str) for line in preferred):
            raise ValueError("preferred_lines must be a list of strings")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = preferred

    def transfer_limit(self):
        """Return the effective transfer cap used during the search."""
        if self.max_transfers is None:
            return DEFAULT_TRANSFER_CAP
        return self.max_transfers

    def step_cost(self, edge, is_transfer):
        """Return the search cost of traversing ``edge``.

        The search cost is only used to order the search; reported distances are
        always the real inter-station distances in meters.
        """
        cost = float(edge.weight)
        if self.preferred_lines and edge.line not in self.preferred_lines:
            cost += PREFERENCE_PENALTY
        if is_transfer:
            if self.mode == MODE_COMPREHENSIVE:
                cost += COMPREHENSIVE_TRANSFER_PENALTY
            elif self.mode == MODE_LEAST_TRANSFER:
                cost += LEAST_TRANSFER_WEIGHT
        return cost

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class ShortestPath:
    """Single-source shortest path computation.

    Attributes:
        dist_to: dict station number -> real distance (meters) of the chosen path
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_node(source):
            raise ValueError(f"Source station number {source} is not in the graph")
        self._graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self.cost_to = {}
        self._best_state = {}   # station number -> best state key reaching it
        self._state_info = {}   # state -> (search cost, real distance, parent state, edge)
        self._run()

    def _run(self):
        limit = self.config.transfer_limit()
        start = (self.source, None, 0)
        self._state_info[start] = (0.0, 0, None, None)

        pq = MinPQ()
        pq.insert(0.0, start)
        settled = set()

        while not pq.is_empty():
            cost, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)

            node, line, transfers = state
            _, distance, _, _ = self._state_info[state]

            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = distance
                self.cost_to[node] = cost

            for edge in self._graph.adjacent(node):
                nxt = edge.other(node)
                is_transfer = line is not None and edge.line != line
                new_transfers = transfers + (1 if is_transfer else 0)
                if new_transfers > limit:
                    continue
                new_state = (nxt, edge.line, new_transfers)
                if new_state in settled:
                    continue
                new_cost = cost + self.config.step_cost(edge, is_transfer)
                new_distance = distance + edge.distance
                old = self._state_info.get(new_state)
                if old is None or new_cost < old[0]:
                    self._state_info[new_state] = (new_cost, new_distance, state, edge)
                    pq.insert(new_cost, new_state)

    def has_path_to(self, target):
        """Return True if ``target`` is reachable under the current configuration."""
        return target in self._best_state

    def path_with_edges(self, target):
        """Return (stations, edges) of the shortest path from source to target.

        ``stations`` is the ordered list of station numbers from source to target;
        ``edges`` is the ordered list of Edge objects between consecutive stations.
        If the target is unreachable, ``([], [])`` is returned.
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        stations = []
        edges = []
        while state is not None:
            _, _, parent, edge = self._state_info[state]
            stations.append(state[0])
            if edge is not None:
                edges.append(edge)
            state = parent
        stations.reverse()
        edges.reverse()
        return stations, edges

    def get_transfer_count(self, target):
        """Return the number of line changes along the path to ``target``."""
        _, edges = self.path_with_edges(target)
        count = 0
        for previous, current in zip(edges, edges[1:]):
            if previous.line != current.line:
                count += 1
        return count

"""Beijing Subway fare calculation and travel time estimation.

Fare structure (distance is rounded up to whole kilometers):
    0-6 km    : 3 yuan
    6-12 km   : 4 yuan
    12-22 km  : 5 yuan
    22-32 km  : 6 yuan
    beyond 32 km: +1 yuan for every additional 20 km (or part thereof)
"""

import math

BASE_FARE_YUAN = 3
# (upper bound in kilometers, inclusive; fare in yuan)
FARE_TIERS = ((6, 3), (12, 4), (22, 5), (32, 6))
EXTENSION_THRESHOLD_KM = 32
EXTENSION_STEP_KM = 20
EXTENSION_STEP_FARE = 1

# Average operating speed used for time estimation (km/h).
AVERAGE_SPEED_KMH = 35.0
# Average extra time spent per transfer (minutes).
TRANSFER_TIME_MIN = 5.0


def _validate_distance(distance):
    """Validate a distance in meters and return it as a float."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of meters")
    if not math.isfinite(distance):
        raise ValueError("distance must be a finite number")
    if distance < 0:
        raise ValueError("distance must be non-negative")
    return float(distance)


def Sum_money(distance):
    """Return the subway fare (integer yuan) for a trip of ``distance`` meters.

    Fractional kilometers are rounded up (ceiling), so 6.2 km is charged as 7 km.
    A distance of 0 meters costs nothing.
    """
    distance = _validate_distance(distance)
    if distance == 0:
        return 0

    km = math.ceil(distance / 1000.0)
    for upper_km, fare in FARE_TIERS:
        if km <= upper_km:
            return fare

    extra_km = km - EXTENSION_THRESHOLD_KM
    steps = math.ceil(extra_km / EXTENSION_STEP_KM)
    return FARE_TIERS[-1][1] + steps * EXTENSION_STEP_FARE


def calculate_estimated_time(distance, transfer_count):
    """Return the estimated travel time in minutes.

    Travel time is derived from the distance at AVERAGE_SPEED_KMH, plus
    TRANSFER_TIME_MIN for every transfer.
    """
    distance = _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise ValueError("transfer_count must be a non-negative integer")
    travel_minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    return float(travel_minutes + transfer_count * TRANSFER_TIME_MIN)

"""Builds the weighted graph model of the subway network from the data dictionary.

Each unique station name receives one station number. Transfer stations are
represented by a single node whose incident edges belong to several lines; a line
change along a path is counted as a transfer.
"""

from models import Edge, Graph


def _clean_name(value):
    """Normalize a station or line name (strip and collapse whitespace)."""
    if not isinstance(value, str):
        raise ValueError(f"Expected a string name, got {type(value).__name__}")
    cleaned = " ".join(value.split())
    if not cleaned:
        raise ValueError("Names must not be empty")
    return cleaned


class SubwayNetwork:
    """Container for the subway graph and name lookup tables.

    Attributes:
        station_to_number: dict station name -> station number
        number_to_station: list where index is station number and value is name
        graph: the Graph instance holding edges between station numbers
    """

    def __init__(self):
        self.station_to_number = {}
        self.number_to_station = []
        self.graph = Graph()
        self._casefold_index = {}
        self._lines = {}

    def _register_station(self, name):
        """Register a station name (if new) and return its number."""
        if name in self.station_to_number:
            return self.station_to_number[name]
        number = len(self.number_to_station)
        self.station_to_number[name] = number
        self.number_to_station.append(name)
        self._casefold_index.setdefault(name.casefold(), name)
        self.graph.add_node(number)
        return number

    def number_of(self, name):
        """Return the station number of an official station name."""
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the official station name of a station number."""
        return self.number_to_station[number]

    def has_station(self, name):
        """Return True if ``name`` is an official station name."""
        return name in self.station_to_number

    def standardize_station(self, raw):
        """Map user input to an official station name, or None if it does not exist.

        Whitespace is normalized and matching is case-insensitive.
        """
        if raw is None:
            return None
        normalized = " ".join(str(raw).split())
        if not normalized:
            return None
        if normalized in self.station_to_number:
            return normalized
        return self._casefold_index.get(normalized.casefold())

    def standardize_line(self, raw):
        """Map user input to an official line name, or None if it does not exist."""
        if raw is None:
            return None
        normalized = " ".join(str(raw).split())
        for line in self._lines:
            if line == normalized or line.casefold() == normalized.casefold():
                return line
        return None

    def station_names(self):
        """Return a sorted list of all official station names."""
        return sorted(self.station_to_number)

    def lines_of(self, name):
        """Return the lines serving a station."""
        return list(self.graph.station_lines.get(name, []))

    def line_names(self):
        """Return a sorted list of all line names."""
        return sorted(self._lines)

    def stations_on_line(self, line):
        """Return the stations served by ``line`` in data order."""
        return list(self._lines.get(line, []))

    def is_transfer_station(self, name):
        """Return True if the station is served by two or more lines."""
        return len(self.lines_of(name)) >= 2

    def transfer_stations(self):
        """Return a sorted list of transfer station names."""
        return sorted(name for name in self.station_to_number if self.is_transfer_station(name))


def build_network(data):
    """Build a SubwayNetwork from the dict loaded from subway_data.json."""
    network = SubwayNetwork()
    station_lines = {}

    def add_station_lines(name, lines):
        bucket = station_lines.setdefault(name, [])
        for line in lines:
            if line not in bucket:
                bucket.append(line)

    # Declared stations first, so station numbers follow the data file order.
    for station in data.get("stations", []):
        name = _clean_name(station["name"])
        network._register_station(name)
        add_station_lines(name, [_clean_name(line) for line in station.get("lines", [])])

    # Connections: register any undeclared station and collect the line information.
    parsed_connections = []
    for conn in data.get("connections", []):
        source = _clean_name(conn["source"])
        dest = _clean_name(conn["dest"])
        line = _clean_name(conn["line"])
        distance = int(conn["distance"])
        for name in (source, dest):
            network._register_station(name)
            add_station_lines(name, [line])
        parsed_connections.append((source, dest, distance, line))

    # Finalize station-to-line information.
    for name in network.number_to_station:
        network.graph.set_station_lines(name, station_lines.get(name, []))
        for line in network.graph.station_lines[name]:
            network._lines.setdefault(line, []).append(name)

    # Build undirected weighted edges between station numbers.
    for source, dest, distance, line in parsed_connections:
        network.graph.add_edge(
            Edge(network.number_of(source), network.number_of(dest), distance, line)
        )

    return network

"""Data update operations: adding lines, modifying stations and updating distances.

All operations work on a private deep copy of the data dictionary, so a failed
update never leaves the data in a partially modified state. Use ``build_network``
on the result to obtain an updated network model.
"""

import copy

from data_loader import save_subway_data
from subway_network import build_network


class DataUpdateError(ValueError):
    """Raised when a requested data update is invalid."""


def _valid_distance(distance):
    """Validate a distance in meters (positive integer)."""
    if isinstance(distance, bool) or not isinstance(distance, int) or distance <= 0:
        raise DataUpdateError("distance must be a positive integer number of meters")
    return distance


class DataManager:
    """Manages updates to the subway network data dictionary."""

    def __init__(self, data):
        self._data = copy.deepcopy(data)
        self._data.setdefault("stations", [])
        self._data.setdefault("connections", [])

    @property
    def data(self):
        """Return a deep copy of the current data dictionary."""
        return copy.deepcopy(self._data)

    def _find_station(self, name):
        for station in self._data["stations"]:
            if station["name"] == name:
                return station
        return None

    def _existing_lines(self):
        lines = set()
        for station in self._data["stations"]:
            lines.update(station.get("lines", []))
        for conn in self._data["connections"]:
            lines.add(conn["line"])
        return lines

    def add_line(self, line_name, station_names, distances):
        """Add a new line.

        ``station_names`` is the ordered list of stations on the line, and
        ``distances`` contains the distance (meters) between consecutive stations.
        Existing stations are linked to the new line; new stations are created.
        """
        line_name = " ".join(str(line_name).split())
        if not line_name:
            raise DataUpdateError("line name must not be empty")
        if line_name in self._existing_lines():
            raise DataUpdateError(f"line '{line_name}' already exists")
        if len(station_names) < 2:
            raise DataUpdateError("a line needs at least two stations")
        if len(set(station_names)) != len(station_names):
            raise DataUpdateError("a line cannot contain duplicate stations")
        if len(distances) != len(station_names) - 1:
            raise DataUpdateError("number of distances must be one less than number of stations")
        for distance in distances:
            _valid_distance(distance)

        for name in station_names:
            station = self._find_station(name)
            if station is None:
                self._data["stations"].append({"name": name, "lines": [line_name]})
            elif line_name not in station.get("lines", []):
                station.setdefault("lines", []).append(line_name)

        for index, distance in enumerate(distances):
            self._data["connections"].append({
                "source": station_names[index],
                "dest": station_names[index + 1],
                "distance": distance,
                "line": line_name,
            })

    def modify_station(self, old_name, new_name=None, add_lines=None, remove_lines=None):
        """Rename a station and/or add or remove lines serving it."""
        station = self._find_station(old_name)
        if station is None:
            raise DataUpdateError(f"station '{old_name}' does not exist")

        if new_name and new_name != old_name:
            new_name = " ".join(new_name.split())
            if self._find_station(new_name) is not None:
                raise DataUpdateError(f"station '{new_name}' already exists")
            station["name"] = new_name
            for conn in self._data["connections"]:
                if conn["source"] == old_name:
                    conn["source"] = new_name
                if conn["dest"] == old_name:
                    conn["dest"] = new_name

        lines = station.setdefault("lines", [])
        for line in add_lines or []:
            if line not in lines:
                lines.append(line)
        for line in remove_lines or []:
            if line in lines:
                lines.remove(line)

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance of connections between two stations.

        If ``line`` is given, only the connection on that line is updated.
        Returns the number of connections updated.
        """
        _valid_distance(distance)
        updated = 0
        for conn in self._data["connections"]:
            same_pair = {conn["source"], conn["dest"]} == {source, dest}
            if same_pair and (line is None or conn["line"] == line):
                conn["distance"] = distance
                updated += 1
        if updated == 0:
            raise DataUpdateError(f"no connection between '{source}' and '{dest}'")
        return updated

    def build_network(self):
        """Return a SubwayNetwork built from the current data."""
        return build_network(self._data)

    def save(self, path):
        """Persist the current data to a JSON file."""
        save_subway_data(self._data, path)

"""Data integrity checks for the subway network (triggered by --check_data)."""

from dataclasses import dataclass, field

from subway_network import build_network

# Main lines expected to be covered by the data set.
REQUIRED_MAIN_LINES = (
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
)
MIN_STATIONS_PER_LINE = 2
# Plausible range for distances between adjacent stations (meters).
MIN_REASONABLE_DISTANCE_M = 100
MAX_REASONABLE_DISTANCE_M = 10000


@dataclass
class IntegrityReport:
    """Result of a data integrity check."""

    errors: list = field(default_factory=list)
    warnings: list = field(default_factory=list)
    summary: list = field(default_factory=list)

    @property
    def ok(self):
        """True when no errors were found (warnings are allowed)."""
        return not self.errors

    def format_report(self):
        """Return a human-readable text report."""
        out = ["Data integrity check"]
        out.extend(self.summary)
        for message in self.errors:
            out.append(f"  [ERROR] {message}")
        for message in self.warnings:
            out.append(f"  [WARNING] {message}")
        if not self.errors and not self.warnings:
            out.append("Result: PASSED")
        elif self.errors:
            out.append(f"Result: FAILED ({len(self.errors)} error(s), {len(self.warnings)} warning(s))")
        else:
            out.append(f"Result: PASSED with {len(self.warnings)} warning(s)")
        return "\n".join(out)


def _check_station_coverage(network, report):
    """Verify that representative main lines are present and have stations."""
    if not network.station_names():
        report.errors.append("No stations found in the data set")
        return
    present = set(network.line_names())
    for line in REQUIRED_MAIN_LINES:
        if line not in present:
            report.warnings.append(f"Main line '{line}' is not covered by the data set")
    for line in sorted(present):
        count = len(network.stations_on_line(line))
        if count < MIN_STATIONS_PER_LINE:
            report.warnings.append(f"Line '{line}' has only {count} station(s)")


def _check_connections(stations, connections, network, report):
    """Verify connection relationships between declared stations."""
    declared = {station.get("name") for station in stations}
    seen_pairs = set()
    for index, conn in enumerate(connections):
        source = conn.get("source")
        dest = conn.get("dest")
        line = conn.get("line")
        if source not in declared:
            report.errors.append(f"Connection {index}: station '{source}' is not declared")
        if dest not in declared:
            report.errors.append(f"Connection {index}: station '{dest}' is not declared")
        if source == dest:
            report.errors.append(f"Connection {index}: station '{source}' connects to itself")
        key = (frozenset((source, dest)), line)
        if key in seen_pairs:
            report.warnings.append(f"Duplicate connection {source} - {dest} on '{line}'")
        seen_pairs.add(key)

    isolated = [name for name in network.station_names()
                if network.graph.degree(network.number_of(name)) == 0]
    for name in isolated:
        report.warnings.append(f"Station '{name}' has no connections")

    connected_nodes = [v for v in network.graph.nodes() if network.graph.degree(v) > 0]
    if connected_nodes:
        seen = set()
        components = 0
        for start in connected_nodes:
            if start in seen:
                continue
            components += 1
            seen.add(start)
            stack = [start]
            while stack:
                vertex = stack.pop()
                for edge in network.graph.adjacent(vertex):
                    other = edge.other(vertex)
                    if other not in seen:
                        seen.add(other)
                        stack.append(other)
        if components > 1:
            report.warnings.append(f"Network is split into {components} disconnected components")


def _check_distances(connections, report):
    """Verify that distance data is numeric, positive and plausible."""
    for index, conn in enumerate(connections):
        distance = conn.get("distance")
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            report.errors.append(f"Connection {index}: distance is not numeric")
            continue
        if distance <= 0:
            report.errors.append(f"Connection {index}: distance {distance} must be positive")
        elif not (MIN_REASONABLE_DISTANCE_M <= distance <= MAX_REASONABLE_DISTANCE_M):
            report.warnings.append(
                f"Connection {index} ({conn.get('source')} - {conn.get('dest')}): "
                f"distance {distance} m is outside the plausible range"
            )


def check_data_integrity(data, network=None):
    """Run all integrity checks and return an IntegrityReport."""
    report = IntegrityReport()
    try:
        if network is None:
            network = build_network(data)
    except (KeyError, TypeError, ValueError) as exc:
        report.errors.append(f"Network cannot be built from the data: {exc}")
        return report

    stations = data.get("stations", [])
    connections = data.get("connections", [])
    report.summary = [
        f"Stations: {len(network.station_names())}",
        f"Lines: {len(network.line_names())}",
        f"Connections: {len(connections)}",
    ]
    _check_station_coverage(network, report)
    _check_connections(stations, connections, network, report)
    _check_distances(connections, report)
    return report

"""Route computation, path summary and report rendering (console text and JSON)."""

import json
from dataclasses import dataclass, field

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


class RouteError(Exception):
    """Base class for route query errors."""


class RouteNotFoundError(RouteError):
    """Raised when no path exists under the given configuration."""


@dataclass
class RouteSegment:
    """A continuous ride on a single line."""

    line: str
    start: str
    end: str
    stations: list = field(default_factory=list)
    distance_m: int = 0

    @property
    def stop_count(self):
        """Number of inter-station hops within the segment."""
        return len(self.stations) - 1


@dataclass
class TransferInfo:
    """A line change at a transfer station."""

    station: str
    from_line: str
    to_line: str


@dataclass
class RouteResult:
    """Complete result of a route query."""

    start: str
    end: str
    mode: str
    max_transfers: object
    preferred_lines: list
    stations: list
    segments: list
    transfers: list
    distance_m: int
    fare: int
    transfer_count: int
    estimated_time_min: float

    @property
    def lines(self):
        """Ordered list of unique lines used by the route."""
        result = []
        for segment in self.segments:
            if segment.line not in result:
                result.append(segment.line)
        return result

    @property
    def distance_km(self):
        """Total distance in kilometers."""
        return self.distance_m / 1000.0


def _build_segments(names, edges):
    """Group consecutive edges on the same line into segments."""
    segments = []
    for index, edge in enumerate(edges):
        start = names[index]
        end = names[index + 1]
        if segments and segments[-1].line == edge.line:
            segment = segments[-1]
            segment.end = end
            segment.stations.append(end)
            segment.distance_m += edge.distance
        else:
            segments.append(RouteSegment(
                line=edge.line, start=start, end=end,
                stations=[start, end], distance_m=edge.distance,
            ))
    return segments


def _build_transfers(segments):
    """Derive transfer details from consecutive segments."""
    transfers = []
    for previous, current in zip(segments, segments[1:]):
        transfers.append(TransferInfo(
            station=current.start, from_line=previous.line, to_line=current.line,
        ))
    return transfers


def compute_route(network, start_name, end_name, config=None):
    """Compute the route, fare and summary between two official station names."""
    config = config or PathfindingConfig()
    try:
        source = network.number_of(start_name)
        target = network.number_of(end_name)
    except KeyError as exc:
        raise RouteError(f"Unknown station: {exc}") from exc

    try:
        sp = ShortestPath(network.graph, source, config)
    except (ValueError, KeyError) as exc:
        raise RouteError(f"Path calculation failed: {exc}") from exc

    stations, edges = sp.path_with_edges(target)
    if not stations or stations[-1] != target:
        raise RouteNotFoundError(
            f"No route found from '{start_name}' to '{end_name}' with the given options"
        )

    names = [network.name_of(number) for number in stations]
    segments = _build_segments(names, edges)
    transfers = _build_transfers(segments)
    distance_m = sum(edge.distance for edge in edges)
    transfer_count = sp.get_transfer_count(target)

    try:
        fare = Sum_money(distance_m)
        estimated_time = calculate_estimated_time(distance_m, transfer_count)
    except (TypeError, ValueError) as exc:
        raise RouteError(f"Fare calculation failed: {exc}") from exc

    return RouteResult(
        start=start_name,
        end=end_name,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        stations=names,
        segments=segments,
        transfers=transfers,
        distance_m=distance_m,
        fare=fare,
        transfer_count=transfer_count,
        estimated_time_min=estimated_time,
    )


def format_distance(meters):
    """Show distance in meters for short trips and kilometers for long trips, with both units."""
    if meters < 1000:
        return f"{meters} m ({meters / 1000:.2f} km)"
    return f"{meters / 1000:.2f} km ({meters} m)"


def build_summary(result):
    """Return the key figures of a route as a dictionary."""
    return {
        "total_distance_km": round(result.distance_km, 3),
        "total_fare_yuan": result.fare,
        "transfer_count": result.transfer_count,
        "estimated_time_minutes": round(result.estimated_time_min, 1),
    }


def to_dict(result):
    """Convert a RouteResult into a JSON-serializable dictionary."""
    return {
        "query": {
            "start": result.start,
            "end": result.end,
            "mode": result.mode,
            "max_transfers": result.max_transfers,
            "line_preference": result.preferred_lines,
        },
        "path": list(result.stations),
        "distance": {
            "meters": result.distance_m,
            "kilometers": round(result.distance_km, 3),
        },
        "fare": result.fare,
        "estimated_time_minutes": round(result.estimated_time_min, 1),
        "transfer_count": result.transfer_count,
        "transfers": [
            {"station": t.station, "from_line": t.from_line, "to_line": t.to_line}
            for t in result.transfers
        ],
        "lines": result.lines,
        "segments": [
            {
                "line": s.line,
                "start": s.start,
                "end": s.end,
                "stations": list(s.stations),
                "distance_m": s.distance_m,
            }
            for s in result.segments
        ],
        "summary": build_summary(result),
    }


def format_text(result):
    """Render the route as console-friendly text."""
    out = []
    out.append("=" * 60)
    out.append(f"Route: {result.start} -> {result.end}")
    out.append(f"Mode: {result.mode}")
    if result.max_transfers is not None:
        out.append(f"Max transfers: {result.max_transfers}")
    if result.preferred_lines:
        out.append(f"Line preference: {', '.join(result.preferred_lines)}")
    out.append("-" * 60)
    out.append(f"Station sequence ({len(result.stations)} stations):")
    for index, name in enumerate(result.stations, 1):
        out.append(f"  {index:>3}. {name}")
    out.append("-" * 60)
    out.append("Line segments:")
    for segment in result.segments:
        out.append(
            f"  [{segment.line}] {segment.start} -> {segment.end} "
            f"({segment.stop_count} stops, {format_distance(segment.distance_m)})"
        )
    out.append("Transfer details:")
    if result.transfers:
        for transfer in result.transfers:
            out.append(f"  Transfer at {transfer.station}: {transfer.from_line} -> {transfer.to_line}")
    else:
        out.append("  No transfers required")
    out.append(f"Lines involved: {', '.join(result.lines)}")
    out.append("-" * 60)
    out.append(f"Total distance: {format_distance(result.distance_m)}")
    out.append(f"Transfers: {result.transfer_count}")
    out.append(f"Estimated time: {result.estimated_time_min:.1f} minutes")
    out.append(f"Fare: {result.fare} yuan")
    out.append("=" * 60)
    return "\n".join(out)


def render_result(result, fmt, compact=False):
    """Render a route in the requested format ('text' or 'json').

    With ``compact=True`` JSON is written on a single line (JSON Lines style),
    which is used when appending results to an output file.
    """
    if fmt == "json":
        return json.dumps(to_dict(result), ensure_ascii=False, indent=None if compact else 2)
    if fmt == "text":
        return format_text(result)
    raise ValueError(f"Unsupported output format: {fmt}")


def append_to_file(path, content):
    """Append rendered content to a file (UTF-8)."""
    with open(path, "a", encoding="utf-8") as handle:
        handle.write(content.rstrip("\n") + "\n\n")

"""Command-line interface for the Beijing Subway fare calculation system.

Usage examples (run from any directory):
    python src/cli.py
    python src/cli.py --format json --mode least_transfer
    python src/cli.py --line_preference "Line 1" --max_transfers 2

import argparse
import sys

from data_integrity import check_data_integrity
from data_loader import DATA_FILE_PATH, DataLoadError, load_subway_data
from pathfinding import MODES, MODE_SHORTEST_DISTANCE, PathfindingConfig
from route_report import RouteError, append_to_file, compute_route, render_result
from subway_network import build_network

FORMAT_HINT = "Input format incorrect, please use 'Starting Station; Ending Station' format"
PROMPT = "Enter query as 'Starting Station; Ending Station' (or 'quit' to exit): "
QUIT_COMMANDS = {"quit", "exit", "q"}


class QueryFormatError(ValueError):
    """Raised when the user input does not follow the 'start; end' format."""


class StationValidationError(ValueError):
    """Raised when a station name or query logic is invalid."""


def _non_negative_int(value):
    """argparse type: non-negative integer."""
    try:
        number = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError(f"invalid integer value: '{value}'") from exc
    if number < 0:
        raise argparse.ArgumentTypeError("value must be a non-negative integer")
    return number


def build_parser():
    """Create the argument parser."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (shortest path and fare query)."
    )
    parser.add_argument("--output_file", default=None,
                        help="append query results to this file")
    parser.add_argument("--format", choices=("text", "json"), default="text",
                        help="output format (default: text)")
    parser.add_argument("--mode", choices=MODES, default=MODE_SHORTEST_DISTANCE,
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=_non_negative_int, default=None,
                        help="maximum number of transfers allowed")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line name, e.g. 'Line 1'")
    parser.add_argument("--check_data", action="store_true",
                        help="run the data integrity check and exit")
    return parser


def parse_query(text):
    """Split 'Starting Station; Ending Station' into two stripped names."""
    parts = text.split(";")
    if len(parts) != 2:
        raise QueryFormatError(FORMAT_HINT)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryFormatError(FORMAT_HINT)
    return start, end


def resolve_station(network, raw, role):
    """Map user input to an official station name, raising if it does not exist."""
    standard = network.standardize_station(raw)
    if standard is None:
        raise StationValidationError(
            f"{role} station '{raw}' does not exist, please check the station name"
        )
    return standard


def validate_query(network, text):
    """Parse and validate a query, returning (start, end) official station names."""
    raw_start, raw_end = parse_query(text)
    start = resolve_station(network, raw_start, "Starting")
    end = resolve_station(network, raw_end, "Ending")
    if start == end:
        raise StationValidationError("Starting station cannot be the same as the ending station")
    return start, end


def build_config(args, network):
    """Create a PathfindingConfig from the command-line arguments."""
    preferred = []
    if args.line_preference:
        line = network.standardize_line(args.line_preference)
        if line is None:
            available = ", ".join(network.line_names())
            raise ValueError(
                f"line '{args.line_preference}' does not exist. Available lines: {available}"
            )
        preferred = [line]
    return PathfindingConfig(
        mode=args.mode,
        max_transfers=args.max_transfers,
        preferred_lines=preferred,
    )


def handle_query(network, text, config, args, out, err):
    """Process one query line. Returns True on success."""
    try:
        start, end = validate_query(network, text)
    except (QueryFormatError, StationValidationError) as exc:
        print(f"Error: {exc}", file=err)
        return False

    try:
        result = compute_route(network, start, end, config)
    except RouteError as exc:
        print(f"Error: {exc}", file=err)
        return False
    except Exception as exc:  # defensive: never let one query terminate the session
        print(f"Error: unexpected failure while calculating the route: {exc}", file=err)
        return False

    print(render_result(result, args.format), file=out)

    if args.output_file:
        try:
            append_to_file(args.output_file, render_result(result, args.format, compact=True))
        except OSError as exc:
            print(f"Error: cannot write output file '{args.output_file}': {exc}", file=err)
            return False
    return True


def run_interactive(network, args, config, input_func=input, out=None, err=None):
    """Run the continuous query loop until the user quits or input ends."""
    out = out or sys.stdout
    err = err or sys.stderr
    print("Beijing Subway fare calculator. Type 'quit' to exit.", file=out)
    while True:
        try:
            text = input_func(PROMPT)
        except (EOFError, KeyboardInterrupt):
            print(file=out)
            break
        stripped = text.strip()
        if not stripped:
            continue
        if stripped.lower() in QUIT_COMMANDS:
            break
        handle_query(network, stripped, config, args, out, err)
    return 0


def main(argv=None, input_func=input, out=None, err=None):
    """Program entry point. Returns the process exit code."""
    out = out or sys.stdout
    err = err or sys.stderr
    parser = build_parser()
    args = parser.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except DataLoadError as exc:
        print(f"Error: {exc}", file=err)
        return 1

    try:
        network = build_network(data)
    except (KeyError, TypeError, ValueError) as exc:
        print(f"Error: cannot build the subway network: {exc}", file=err)
        return 1

    if args.check_data:
        report = check_data_integrity(data, network)
        print(report.format_report(), file=out)
        return 0 if report.ok else 1

    try:
        config = build_config(args, network)
    except ValueError as exc:
        print(f"Error: {exc}", file=err)
        return 2

    return run_interactive(network, args, config, input_func=input_func, out=out, err=err)


if __name__ == "__main__":
    sys.exit(main())

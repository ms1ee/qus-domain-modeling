"""Configuration constants and runtime-overridable settings.

Map geometry is fixed by the PRD (800x800 units, 100-unit cells, 8x8 grid).
Business parameters live in SETTINGS and may be overridden by a JSON file.
"""

import json
import os

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE  # 8 cells per axis

SETTINGS = {
    "initial_funds": 10000,
    "recruitment_cost": 500,
    "max_riders": 12,
    "order_load_limit": 2,
    "load_penalty": 150,
    "delivery_fee": 80,
    "base_rider_speed": 40,
    "save_file": "dispatch_state.json",
    "log_file": "dispatch.log",
}

DEFAULT_CONFIG_FILE = "dispatch_config.json"


def load_config(path: str = DEFAULT_CONFIG_FILE) -> bool:
    """Load overrides for SETTINGS from a JSON file.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        True if a file was found and applied, False if it does not exist.

    Raises:
        ValueError: If the file content is not a JSON object or a value has
            the wrong type.
    """
    if not os.path.exists(path):
        return False
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError("Configuration file must contain a JSON object.")
    for key, value in data.items():
        if key not in SETTINGS:
            continue
        expected = type(SETTINGS[key])
        if not isinstance(value, expected):
            raise ValueError(
                f"Setting '{key}' must be of type {expected.__name__}."
            )
        SETTINGS[key] = value
    return True

"""Coordinate and grid helpers for the 800x800 map.

The map is an 8x8 grid of 100x100 cells. Points whose coordinates are both
multiples of 100 are street intersections (restaurant and customer
locations). Streets run along every line where x or y is a multiple of 100.
"""

import config


def clamp(value, low, high):
    """Clamp value into the closed interval [low, high].

    Args:
        value: Number to clamp.
        low: Lower bound.
        high: Upper bound.

    Returns:
        The clamped value.
    """
    return max(low, min(high, value))


def snap_to_grid(value):
    """Snap a raw coordinate to the nearest intersection line.

    The input is first clamped to the map range, then rounded to the nearest
    multiple of CELL_SIZE.

    Args:
        value: Raw integer coordinate (0-800 expected; out-of-range values
            are clamped).

    Returns:
        The snapped coordinate, a multiple of CELL_SIZE within [0, MAP_SIZE].
    """
    value = clamp(int(value), 0, config.MAP_SIZE)
    half = config.CELL_SIZE // 2
    snapped = ((value + half) // config.CELL_SIZE) * config.CELL_SIZE
    return clamp(snapped, 0, config.MAP_SIZE)


def cell_index(value):
    """Return the grid cell index (0..GRID_DIM-1) containing a coordinate.

    Args:
        value: Coordinate in map units.

    Returns:
        Integer cell index along that axis.
    """
    value = clamp(int(value), 0, config.MAP_SIZE)
    return min(value // config.CELL_SIZE, config.GRID_DIM - 1)


def cell_center(value):
    """Return the grid cell center for a coordinate.

    Implements center = (grid // 100) * 100 + 50.

    Args:
        value: Coordinate in map units.

    Returns:
        Center coordinate of the containing cell.
    """
    return cell_index(value) * config.CELL_SIZE + config.CELL_SIZE // 2


def is_road(x, y):
    """Check whether a point lies on a street.

    Args:
        x: X coordinate.
        y: Y coordinate.

    Returns:
        True if the point is inside the map and on a grid line.
    """
    if not (0 <= x <= config.MAP_SIZE and 0 <= y <= config.MAP_SIZE):
        return False
    return x % config.CELL_SIZE == 0 or y % config.CELL_SIZE == 0


def is_intersection(x, y):
    """Check whether a point is a street intersection (restaurant/customer spot).

    Args:
        x: X coordinate.
        y: Y coordinate.

    Returns:
        True if both coordinates are multiples of CELL_SIZE within the map.
    """
    return (
        0 <= x <= config.MAP_SIZE
        and 0 <= y <= config.MAP_SIZE
        and x % config.CELL_SIZE == 0
        and y % config.CELL_SIZE == 0
    )


def manhattan(a, b):
    """Return the Manhattan distance between two points.

    Args:
        a: (x, y) tuple.
        b: (x, y) tuple.

    Returns:
        Integer distance |dx| + |dy|.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def all_intersections():
    """Return every intersection point of the map.

    Returns:
        List of (x, y) tuples for the 9x9 intersection lattice.
    """
    count = config.GRID_DIM + 1
    return [
        (i * config.CELL_SIZE, j * config.CELL_SIZE)
        for i in range(count)
        for j in range(count)
    ]


class MapGrid:
    """Description of the street grid used by the path planner and display."""

    def __init__(self, size=config.MAP_SIZE, cell_size=config.CELL_SIZE):
        """Create a grid description.

        Args:
            size: Map side length in units.
            cell_size: Side length of one grid cell in units.
        """
        self.size = size
        self.cell_size = cell_size

    @property
    def grid_dim(self):
        """Number of cells along one axis."""
        return self.size // self.cell_size

    def contains(self, x, y):
        """Return True if (x, y) lies inside the map bounds."""
        return 0 <= x <= self.size and 0 <= y <= self.size

    def is_road(self, x, y):
        """Return True if (x, y) lies on a street of this grid."""
        if not self.contains(x, y):
            return False
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def intersections(self):
        """Return all intersection points of this grid."""
        count = self.grid_dim + 1
        return [
            (i * self.cell_size, j * self.cell_size)
            for i in range(count)
            for j in range(count)
        ]

    def cell_of(self, x, y):
        """Return the (column, row) cell index containing (x, y)."""
        return cell_index(x), cell_index(y)

"""Grid-based path planning.

Routes always follow streets: movement is axis-aligned, and when a rider must
change direction the corner is chosen so that the whole path lies on a
street. Straight segments (rider directly above/below or beside the target)
are handled as a special case.
"""

import logging

from grid_map import MapGrid

logger = logging.getLogger(__name__)


class PathPlanner:
    """Plans street-following routes and computes per-tick movement."""

    def __init__(self, grid=None):
        """Create a planner.

        Args:
            grid: Optional MapGrid describing the street layout.
        """
        self.grid = grid if grid is not None else MapGrid()

    def plan(self, start, target):
        """Plan a street-following path from start to target.

        Args:
            start: (x, y) current position, which must lie on a street.
            target: (x, y) destination intersection.

        Returns:
            List of corner points ending with target (start excluded). Empty
            if start equals target.
        """
        sx, sy = start
        tx, ty = target
        if (sx, sy) == (tx, ty):
            return []
        # Special case: directly above/below or beside the target.
        if sx == tx or sy == ty:
            return [(tx, ty)]
        cell = self.grid.cell_size
        horizontal_first = (tx, sy)
        vertical_first = (sx, ty)
        if sy % cell == 0 and self.grid.is_road(tx, sy):
            corner = horizontal_first
        elif sx % cell == 0 and self.grid.is_road(sx, ty):
            corner = vertical_first
        else:
            logger.warning("No street-only corner from %s to %s", start, target)
            corner = horizontal_first
        return [corner, (tx, ty)]

    def plan_route(self, start, stops):
        """Plan a route visiting several stops in order.

        Args:
            start: (x, y) starting position.
            stops: Ordered list of (x, y) destinations.

        Returns:
            List of points beginning with start, containing every corner and
            stop of the route.
        """
        points = [tuple(start)]
        current = tuple(start)
        for stop in stops:
            stop = tuple(stop)
            points.extend(self.plan(current, stop))
            current = stop
        return points

    def step_toward(self, x, y, tx, ty, speed):
        """Advance a position along the planned route by at most speed units.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            tx: Target x coordinate.
            ty: Target y coordinate.
            speed: Maximum distance to travel this tick.

        Returns:
            New (x, y) position after the movement.
        """
        if speed <= 0:
            return x, y
        budget = speed
        cur_x, cur_y = x, y
        for corner_x, corner_y in self.plan((x, y), (tx, ty)):
            while budget > 0 and (cur_x, cur_y) != (corner_x, corner_y):
                if cur_x != corner_x:
                    step = min(abs(corner_x - cur_x), budget)
                    cur_x += step if corner_x > cur_x else -step
                else:
                    step = min(abs(corner_y - cur_y), budget)
                    cur_y += step if corner_y > cur_y else -step
                budget -= step
            if budget <= 0:
                break
        return cur_x, cur_y

    @staticmethod
    def route_length(points):
        """Return the total Manhattan length of a point sequence.

        Args:
            points: List of (x, y) points.

        Returns:
            Sum of segment lengths as an integer.
        """
        total = 0
        for (ax, ay), (bx, by) in zip(points, points[1:]):
            total += abs(bx - ax) + abs(by - ay)
        return total

"""Rider model: state machine, movement and event logging.

Rider states: 0 idle, 1 picking up (heading to restaurant),
2 delivering (heading to customer).
"""

import logging

import config
import globalvar
import grid_map
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

_planner = PathPlanner()
_SPAWN_POINTS = grid_map.all_intersections()


class Rider:
    """A delivery rider that serves a queue of orders."""

    def __init__(self, rider_id):
        """Create a rider on a street intersection.

        Args:
            rider_id: Unique integer identifier.
        """
        self.rider_id = int(rider_id)
        self.x, self.y = _SPAWN_POINTS[(self.rider_id * 13) % len(_SPAWN_POINTS)]
        self.speed = config.SETTINGS["base_rider_speed"] + 10 * (self.rider_id % 3)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    @property
    def position(self):
        """Current (x, y) position."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human-readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    @staticmethod
    def _current_tick():
        """Return the current simulated tick."""
        return int(globalvar.get_value("current_tick") or 0)

    def _record(self, event, order_id):
        """Append a status-change entry to the rider log.

        Args:
            event: Event name (spawn, accepted_order, arrived_at_merchant,
                delivered_to_customer).
            order_id: Related order ID, or None.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": self._current_tick(),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.info(
            "tick=%s rider=%s event=%s order=%s pos=(%s,%s) state=%s",
            entry["tick"], self.rider_id, event, order_id, self.x, self.y,
            self.state_name,
        )

    def target(self):
        """Return the current destination, or None if the rider has no orders."""
        if not self.orders:
            return None
        order = self.orders[0]
        if self.state == DELIVERING:
            return order.cus_pos
        return order.res_pos

    def _activate(self, order):
        """Start working on an order (idle -> picking up, or next order)."""
        if order.picked_up:
            self.state = DELIVERING
            order.set_status("delivering")
        else:
            self.state = PICKING_UP
            order.set_status("picking_up")
        self._record("accepted_order", order.order_id)

    def _arrive(self, order):
        """Handle arrival at the current destination."""
        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.picked_up = True
            order.set_status("delivering")
            self._record("arrived_at_merchant", order.order_id)
        elif self.state == DELIVERING:
            self._deliver(order)

    def _deliver(self, order):
        """Complete the active order, pay the fee and release the rider."""
        self._record("delivered_to_customer", order.order_id)
        order.complete(self._current_tick())
        funds = int(globalvar.get_value("funds") or 0)
        globalvar.set_value("funds", funds + config.SETTINGS["delivery_fee"])
        self.orders.remove(order)
        if self.orders:
            self._activate(self.orders[0])
        else:
            self.state = IDLE
            self.special_case = False

    def _is_blocked(self, nx, ny):
        """Return True if a higher-priority rider occupies the next cell.

        Lower IDs always have priority, so the lowest-ID rider on any shared
        cell never waits and deadlocks are impossible.
        """
        riders = globalvar.get_value("riders") or []
        for other in riders:
            if other is self:
                continue
            if (
                other.rider_id < self.rider_id
                and other.state != IDLE
                and other.position == (nx, ny)
            ):
                return True
        return False

    def move(self):
        """Advance the rider by one tick.

        Performs the state transitions: idle -> picking up when an order is
        accepted, picking up -> delivering on arrival at the restaurant, and
        delivering -> idle (or next order) on delivery.
        """
        if not self.orders:
            if self.state != IDLE:
                self.state = IDLE
                self.special_case = False
            return
        order = self.orders[0]
        if self.state == IDLE:
            self._activate(order)
        target = self.target()
        if target is None:
            return
        if self.position == target:
            self._arrive(order)
            return
        tx, ty = target
        self.special_case = self.x == tx or self.y == ty
        nx, ny = _planner.step_toward(self.x, self.y, tx, ty, self.speed)
        if self._is_blocked(nx, ny):
            self.special_case = True
            logger.debug("rider %s yields at %s", self.rider_id, (nx, ny))
            return
        self.x, self.y = nx, ny
        if self.position == target:
            self._arrive(order)

    def __repr__(self):
        return (
            f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
            f"state={self.state_name}, orders={len(self.orders)})"
        )

"""Order data model with coordinate normalisation and lifecycle tracking."""

import globalvar
import grid_map

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def _current_tick():
    """Return the current simulated tick."""
    return int(globalvar.get_value("current_tick") or 0)


def _next_order_id():
    """Allocate the next unique order ID from the global counter."""
    counter = int(globalvar.get_value("order_id_counter") or 0) + 1
    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A food delivery order: restaurant pickup followed by customer delivery."""

    def __init__(self, res_pos, cus_pos):
        """Create an order and normalise its coordinates.

        Args:
            res_pos: (x, y) restaurant location as entered by the user (0-800).
            cus_pos: (x, y) customer location as entered by the user (0-800).
        """
        rx, ry = res_pos
        cx, cy = cus_pos
        self.res_pos = (grid_map.snap_to_grid(rx), grid_map.snap_to_grid(ry))
        self.cus_pos = (grid_map.snap_to_grid(cx), grid_map.snap_to_grid(cy))
        self.order_id = _next_order_id()
        self.id = self.order_id
        self.status = STATUS_PENDING
        self.rider_id = None
        self.start_time = _current_tick()
        self.completion_time = None
        self.distance = grid_map.manhattan(self.res_pos, self.cus_pos)
        self.time_taken = None
        self.picked_up = False

    def assign_to(self, rider_id):
        """Record the rider this order is assigned to.

        Args:
            rider_id: ID of the assigned rider.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def set_status(self, status):
        """Set the lifecycle status.

        Args:
            status: One of the STATUS_* constants.
        """
        self.status = status

    def complete(self, tick):
        """Mark the order as delivered and record completion data.

        Args:
            tick: Simulated tick at which delivery finished.
        """
        self.status = STATUS_COMPLETED
        self.completion_time = tick
        self.time_taken = max(0, tick - self.start_time)

    @property
    def efficiency(self):
        """Delivery efficiency: distance units per tick (0.0 if unfinished)."""
        if self.time_taken is None:
            return 0.0
        if self.time_taken == 0:
            return float(self.distance)
        return self.distance / self.time_taken

    def to_dict(self):
        """Serialise the order to a JSON-compatible dict."""
        return {
            "order_id": self.order_id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "picked_up": self.picked_up,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from a dict produced by to_dict without side effects.

        Args:
            data: Dict as returned by to_dict.

        Returns:
            Restored Order instance.
        """
        obj = cls.__new__(cls)
        obj.res_pos = tuple(int(v) for v in data["res_pos"])
        obj.cus_pos = tuple(int(v) for v in data["cus_pos"])
        obj.order_id = int(data["order_id"])
        obj.id = obj.order_id
        obj.status = data.get("status", STATUS_PENDING)
        obj.rider_id = data.get("rider_id")
        obj.start_time = int(data.get("start_time", 0))
        obj.completion_time = data.get("completion_time")
        obj.distance = int(data.get("distance", 0))
        obj.time_taken = data.get("time_taken")
        obj.picked_up = bool(data.get("picked_up", False))
        return obj

    def __repr__(self):
        return (
            f"Order(id={self.order_id}, res={self.res_pos}, "
            f"cus={self.cus_pos}, status={self.status})"
        )

"""Order assignment: distance-first rider selection with load balancing.

Also provides OrderManager, which creates orders, assigns them and tracks
the global order list.
"""

import logging

import config
import globalvar
import grid_map
from order_model import Order
from rider import Rider
from system import SystemManager

logger = logging.getLogger(__name__)


def _orders_list():
    """Return the global order list, creating it if missing."""
    orders = globalvar.get_value("orders")
    if orders is None:
        orders = []
        globalvar.set_value("orders", orders)
    return orders


def _score(rider, order):
    """Score a rider for an order; lower is better.

    The score is the Manhattan distance from the rider to the restaurant plus
    a penalty per already-assigned order (load balancing).
    """
    distance = grid_map.manhattan(rider.position, order.res_pos)
    return distance + config.SETTINGS["load_penalty"] * len(rider.orders)


def find_best_rider_for_order(order) -> Rider:
    """Select the best rider for an order, recruiting one if needed.

    A new rider is recruited when there are no riders, or when every rider
    has already reached the order load limit and recruitment is affordable.

    Args:
        order: The Order to be assigned.

    Returns:
        The selected Rider. The caller appends the order to rider.orders.

    Raises:
        RuntimeError: If there is no rider and none can be recruited.
    """
    manager = SystemManager()
    limit = config.SETTINGS["order_load_limit"]
    riders = list(globalvar.get_value("riders") or [])
    if not riders or all(len(r.orders) >= limit for r in riders):
        recruited = manager.recruit_rider()
        if recruited is not None:
            logger.info("Recruited rider %s for order %s", recruited.rider_id, order.order_id)
        riders = list(globalvar.get_value("riders") or [])
    if not riders:
        raise RuntimeError(
            "No rider available and recruitment is not possible "
            "(insufficient funds or capacity)."
        )
    return min(riders, key=lambda rider: _score(rider, order))


class OrderManager:
    """Creates, assigns and tracks orders."""

    def __init__(self, system=None):
        """Create an order manager.

        Args:
            system: Optional SystemManager; a new facade is used if omitted.
        """
        self.system = system if system is not None else SystemManager()

    def create_order(self, restaurant, customer) -> Order:
        """Create an order from raw restaurant and customer coordinates.

        Args:
            restaurant: (x, y) restaurant coordinates.
            customer: (x, y) customer coordinates.

        Returns:
            The newly created Order (coordinates already snapped to the grid).
        """
        order = Order(restaurant, customer)
        _orders_list().append(order)
        logger.info("Created order %s", order.order_id)
        return order

    def assign_order(self, order):
        """Assign an order to the best rider.

        Args:
            order: Order to assign.

        Returns:
            Tuple (rider, new_rider_added).

        Raises:
            RuntimeError: If no rider can serve the order. The order is
                removed from the order list in that case.
        """
        riders_before = len(self.system.riders)
        try:
            rider = find_best_rider_for_order(order)
        except RuntimeError:
            self.cancel_order(order)
            raise
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        new_rider = len(self.system.riders) > riders_before
        logger.info("Order %s assigned to rider %s", order.order_id, rider.rider_id)
        return rider, new_rider

    def cancel_order(self, order):
        """Remove an order from the global order list.

        Args:
            order: Order to remove.
        """
        orders = _orders_list()
        if order in orders:
            orders.remove(order)

    def all_orders(self):
        """Return a copy of all orders, including completed ones."""
        return list(_orders_list())

    def active_orders(self):
        """Return orders that are not yet completed."""
        return [o for o in _orders_list() if o.status != "completed"]

"""System-level state management: funds, riders, clock and running flag."""

import logging

import config
import globalvar
from rider import Rider

logger = logging.getLogger(__name__)


def init_system():
    """Initialise global system state on startup.

    Sets funds, an empty rider list, the rider ID counter (0), the running
    flag (True), the simulated tick (0) and an empty order list.
    """
    globalvar._init()
    globalvar.set_value("funds", config.SETTINGS["initial_funds"])
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("is_running", True)
    globalvar.set_value("current_tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)

class SystemManager:

        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def can_recruit(self):
        """Return True if funds and capacity allow recruiting a rider."""
        return (
            len(self.riders) < config.SETTINGS["max_riders"]
            and self.funds >= self.recruitment_cost
        )

    def recruit_rider(self):
        """Recruit a new rider if funds and capacity allow.

        Deducts the recruitment cost, increments the ID counter and appends
        the new rider to the rider list.

        Returns:
            The new Rider, or None if recruitment is not possible.
        """
        if len(self.riders) >= config.SETTINGS["max_riders"]:
            logger.warning("Rider capacity reached")
            return None
        if self.funds < self.recruitment_cost:
            logger.warning("Insufficient funds to recruit a rider")
            return None
        self.funds = self.funds - self.recruitment_cost
        self.rider_id_counter = self.rider_id_counter + 1
        rider = Rider(self.rider_id_counter)
        self.riders.append(rider)
        logger.info("Recruited rider %s", rider.rider_id)
        return rider

    def advance_tick(self):
        """Advance the simulation by one tick and move every rider.

        Errors raised by an individual rider are logged and do not stop the
        simulation, which keeps the system recoverable.

        Returns:
            The new tick value.
        """
        self.current_tick = self.current_tick + 1
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                logger.exception("Rider %s failed during tick; resetting", rider.rider_id)
                rider.state = 0
                rider.special_case = False
        return self.current_tick

import math

import config
from path_planner import PathPlanner
from rider import STATE_NAMES

SYMBOLS = {
    "empty": " ",
    "intersection": "+",
    "street_h": "-",
    "street_v": "|",
    "restaurant": "R",
    "customer": "C",
    "route": "*",
}


class DisplayManager:
    """Renders status tables, rider logs, routes and the ASCII map."""

    def __init__(self, planner=None):
        """Create a display manager.

        Args:
            planner: Optional PathPlanner used to draw routes.
        """
        self.symbol_map = dict(SYMBOLS)
        self.planner = planner if planner is not None else PathPlanner()

    @staticmethod
    def _cell(x, y):
        """Map a map coordinate to (row, col) on the ASCII canvas.

        Each 100-unit street line is one canvas row/column pair, so the canvas
        has 2*GRID_DIM+1 rows and columns. Row 0 is the top (y = 800).
        """
        col = (int(x) * 2 + config.CELL_SIZE // 2) // config.CELL_SIZE
        row = ((config.MAP_SIZE - int(y)) * 2 + config.CELL_SIZE // 2) // config.CELL_SIZE
        return row, col

    @staticmethod
    def _sample_segment(a, b, step=10):
        """Sample points along an axis-aligned segment every `step` units."""
        (ax, ay), (bx, by) = a, b
        dist = abs(bx - ax) + abs(by - ay)
        count = max(1, dist // step)
        points = []
        for k in range(count + 1):
            t = k / count
            points.append((round(ax + (bx - ax) * t), round(ay + (by - ay) * t)))
        return points

    def render_map(self, riders=(), orders=(), route_points=None):
        """Render the ASCII map.

        Legend: '+' intersection, '-' and '|' streets, 'R' restaurant,
        'C' customer, '*' planned route, digits rider IDs (mod 10).

        Args:
            riders: Riders to draw at their current positions.
            orders: Active orders whose restaurant/customer points are drawn.
            route_points: Optional list of route points to highlight.

        Returns:
            Multi-line string.
        """
        dim = config.GRID_DIM * 2 + 1
        canvas = [[" "] * dim for _ in range(dim)]
        street_chars = (
            self.symbol_map["intersection"],
            self.symbol_map["street_h"],
            self.symbol_map["street_v"],
        )
        for r in range(dim):
            for c in range(dim):
                if r % 2 == 0 and c % 2 == 0:
                    canvas[r][c] = self.symbol_map["intersection"]
                elif r % 2 == 0:
                    canvas[r][c] = self.symbol_map["street_h"]
                elif c % 2 == 0:
                    canvas[r][c] = self.symbol_map["street_v"]
        if route_points:
            for a, b in zip(route_points, route_points[1:]):
                for px, py in self._sample_segment(a, b):
                    row, col = self._cell(px, py)
                    if canvas[row][col] in street_chars:
                        canvas[row][col] = self.symbol_map["route"]
        for order in orders:
            if order.status == "completed":
                continue
            row, col = self._cell(*order.res_pos)
            canvas[row][col] = self.symbol_map["restaurant"]
            row, col = self._cell(*order.cus_pos)
            canvas[row][col] = self.symbol_map["customer"]
        for rider in riders:
            row, col = self._cell(rider.x, rider.y)
            canvas[row][col] = str(rider.rider_id % 10)

        lines = []
        for r in range(dim):
            if r % 2 == 0:
                y_value = config.MAP_SIZE - (r // 2) * config.CELL_SIZE
                label = f"{y_value:>4} "
            else:
                label = "     "
            lines.append(label + "".join(canvas[r]))
        lines.append("")
        lines.append("Legend: + intersection, - |streets, R restaurant, "
                     "C customer, * route, 0-9 rider ID")
        lines.append("Grid lines every 100 units (x and y: 0 to 800).")
        return "\n".join(lines)

    def render_route(self, rider, order):
        """Render the route map for a rider including speed and time estimate.

        Args:
            rider: Assigned rider.
            order: Newly assigned order.

        Returns:
            Multi-line string describing the route and drawing it on the map.
        """
        stops = []
        for queued in rider.orders:
            if not queued.picked_up:
                stops.append(queued.res_pos)
            stops.append(queued.cus_pos)
        if order not in rider.orders:
            stops.extend([order.res_pos, order.cus_pos])
        points = self.planner.plan_route(rider.position, stops)
        length = self.planner.route_length(points)
        speed = max(1, rider.speed)
        eta = math.ceil(length / speed) if length else 0
        header = (
            f"Route for rider {rider.rider_id}: speed={rider.speed} units/tick, "
            f"path length={length} units, estimated time={eta} tick(s)"
        )
        return header + "\n" + self.render_map(
            riders=[rider], orders=[order], route_points=points
        )

    def render_status(self, funds, tick, riders):
        """Render the system snapshot: funds, tick and every rider.

        Args:
            funds: Current fund balance.
            tick: Current simulated tick.
            riders: Iterable of riders.

        Returns:
            Multi-line string.

        riders = list(riders)
        lines = [
            f"Tick: {tick}",
            f"System funds: {funds}",
            f"Riders: {len(riders)}",
        ]
        for rider in riders:
            order_ids = [o.order_id for o in rider.orders]
            lines.append(
                f"  Rider {rider.rider_id}: position=({rider.x},{rider.y}) "
                f"status={STATE_NAMES.get(rider.state, 'unknown')} "
                f"speed={rider.speed} orders={len(order_ids)} {order_ids} "
                f"special_case={rider.special_case}"
            )
        if not riders:
            lines.append("  No riders recruited yet.")
        return "\n".join(lines)

    def render_orders(self, orders):
        """Render a table of orders.

        Args:
            orders: Iterable of Order objects.

        Returns:
            Multi-line string.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        header = (
            f"{'ID':>4} | {'Start':>5} | {'Restaurant':>12} | {'Customer':>12} | "
            f"{'Dist':>5} | {'Time':>5} | {'Rider':>5} | {'Status':<12} | {'Eff':>6}"
        )
        lines = [header, "-" * len(header)]
        for o in orders:
            rider = "-" if o.rider_id is None else str(o.rider_id)
            time_taken = "-" if o.time_taken is None else str(o.time_taken)
            lines.append(
                f"{o.order_id:>4} | {o.start_time:>5} | {str(o.res_pos):>12} | "
                f"{str(o.cus_pos):>12} | {o.distance:>5} | {time_taken:>5} | "
                f"{rider:>5} | {o.status:<12} | {o.efficiency:>6.2f}"
            )
        return "\n".join(lines)

    def render_riders(self, riders):
        """Render the status-change log of every given rider.

        Args:
            riders: Iterable of riders.

        Returns:
            Multi-line string.
        """
        lines = []
        for rider in riders:
            lines.append(f"Rider {rider.rider_id} log:")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order = "-" if entry["order_id"] is None else entry["order_id"]
                lines.append(
                    f"  [tick {entry['tick']}] {entry['event']:<22} "
                    f"order={order} pos=({entry['x']},{entry['y']}) "
                    f"state={STATE_NAMES.get(entry['state'], 'unknown')}"
                )
        return "\n".join(lines) if lines else "No riders recruited yet."

def load_state(path=None):
    """Load a saved state, replacing the current in-memory state.

    The new state is fully built before any global is replaced, so a corrupt
    file leaves the current state untouched.

    Args:
        path: Source file path; defaults to SETTINGS['save_file'].

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the file content is invalid.
    """
    path = path or config.SETTINGS["save_file"]
    with open(path, "r", encoding="utf-8") as handle:
        try:
            data = json.load(handle)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid save file: {exc}") from exc
    try:
        orders = [Order.from_dict(od) for od in data.get("orders", [])]
        orders_by_id = {o.order_id: o for o in orders}
        riders = []
        for rd in data.get("riders", []):
            rider = Rider(int(rd["rider_id"]))
            rider.x = int(rd["x"])
            rider.y = int(rd["y"])
            rider.speed = int(rd["speed"])
            rider.state = int(rd["state"])

            rider.orders = [orders_by_id[oid] for oid in rd.get("order_ids", [])
                            if oid in orders_by_id]

    globalvar.set_value("order_id_counter", order_counter)

    logger.info("State loaded from %s", path)

class InputManager:
    """Parses user commands and dispatches them to the managers."""

    def __init__(self, system, order_manager, display, input_func=input, output_func=print):
        """Create the input manager.

        Args:
            system: SystemManager facade.
            order_manager: OrderManager used for order creation and assignment.
            display: DisplayManager used for output rendering.
            input_func: Callable used to read a line (injectable for tests).
            output_func: Callable used to print output.
        """
        self.system = system
        self.order_manager = order_manager
        self.display = display
        self.input_func = input_func
        self.output_func = output_func
        self.command_history = []
        self._handlers = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "step": self._cmd_step,
            "save": self._cmd_save,
            "load": self._cmd_load,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
            "exit": self._cmd_quit,
        }

    def _print(self, text=""):
        """Print text through the output callable."""
        self.output_func(text)

    @staticmethod
    def parse_coordinates(tokens):
        """Validate and convert four coordinate tokens.

        Args:
            tokens: Sequence of four string tokens.

        Returns:
            List of four integers in the range 0-800.

        Raises:
            ValueError: On wrong count, non-integer or out-of-range values.
        """
        if len(tokens) != 4:
            raise ValueError(
                "Expected 4 coordinates: restaurant_x restaurant_y "
                "customer_x customer_y."
            )
        values = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise ValueError(f"'{token}' is not an integer.") from None
            if not 0 <= value <= config.MAP_SIZE:
                raise ValueError(
                    f"Coordinate {value} is out of range 0-{config.MAP_SIZE}."
                )
            values.append(value)
        return values

    def handle_command(self, line):
        """Parse and execute a single command line.

        Args:
            line: Raw user input.

        Returns:
            True if the program should keep running.
        """
        line = line.strip()
        if not line:
            return self.system.is_running
        self.command_history.append(line)
        parts = line.split()
        name = parts[0].lower()
        name = COMMAND_ALIASES.get(name, name)
        handler = self._handlers.get(name)
        if handler is None:
            self._print(f"Unknown command '{parts[0]}'. Type 'help' for a list of commands.")
            return self.system.is_running
        try:
            handler(parts[1:])
        except ValueError as exc:
            self._print(f"Error: {exc}")
        except RuntimeError as exc:
            self._print(f"Error: {exc}")
        except OSError as exc:
            self._print(f"File error: {exc}")
        except Exception:  # Error recovery: keep the session alive.
            logger.exception("Unexpected error handling command '%s'", line)
            self._print("An unexpected error occurred; the system state is preserved.")
        return self.system.is_running

    def run(self):
        """Run the interactive loop until the system stops."""
        self._print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.is_running:
            try:
                line = self.input_func(PROMPT)
            except (EOFError, KeyboardInterrupt):
                self._print("")
                self.system.stop()
                break
            self.handle_command(line)

    def _cmd_start(self, args):
        """Create and assign an order from inline or prompted coordinates."""
        if args:
            tokens = COORD_SPLIT.split(" ".join(args).strip())
        else:
            raw = self.input_func(
                "Enter restaurant_x restaurant_y customer_x customer_y: "
            )
            tokens = COORD_SPLIT.split(raw.strip())
        rx, ry, cx, cy = self.parse_coordinates([t for t in tokens if t])
        order = self.order_manager.create_order((rx, ry), (cx, cy))
        rider, new_rider = self.order_manager.assign_order(order)

        self._print(f"Order #{order.order_id} created.")
        self._print(
            f"Restaurant grid position: {order.res_pos} (input {(rx, ry)})"
        )
        self._print(
            f"Customer grid position:   {order.cus_pos} (input {(cx, cy)})"
        )
        if new_rider:
            self._print(f"A new rider was recruited: rider {rider.rider_id}.")
        else:
            self._print(f"No new rider added; assigned to rider {rider.rider_id}.")
        self._print(f"Rider {rider.rider_id} has {len(rider.orders)} order(s) queued.")
        self._print(self.display.render_route(rider, order))

    def _cmd_status(self, args):
        """Show the system snapshot."""
        system = self.system
        self._print(
            self.display.render_status(system.funds, system.current_tick, system.riders)

    def _cmd_orders(self, args):

    def _cmd_riders(self, args):

    def _cmd_step(self, args):

                raise ValueError(f"'{args[0]}' is not an integer.") from None

            self._print(f"{index:>4}  {cmd}")

    def _cmd_help(self, args):
        """Show the help text."""
        self._print(HELP_TEXT)

    def _cmd_quit(self, args):
        """Stop the main loop."""
        self.system.stop()
        self._print("Goodbye.")

if _SRC_DIR not in sys.path:
    sys.path.insert(0, _SRC_DIR)

import config  # noqa: E402
from input_manager import build_input_manager  # noqa: E402
from system import init_system  # noqa: E402


def setup_logging():
    """Configure file logging for debugging and diagnostics."""
    try:
        logging.basicConfig(
            filename=config.SETTINGS["log_file"],
            level=logging.INFO,
            format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        )
    except OSError:
        logging.basicConfig(level=logging.WARNING)


def main(argv=None):
    """Run the interactive dispatch simulator.

    Args:
        argv: Unused placeholder for command-line arguments.

    Returns:
        Process exit code.
    """
    setup_logging()
    try:
        if config.load_config():
            logging.getLogger(__name__).info("Loaded configuration overrides")
    except (OSError, ValueError, json.JSONDecodeError) as exc:
        print(f"Warning: could not read configuration file, using defaults ({exc}).")
    init_system()
    manager = build_input_manager()
    try:
        manager.run()
    except Exception:  # Last-resort recovery: report and exit cleanly.
        logging.getLogger(__name__).exception("Fatal error in main loop")
        print("A fatal error occurred. See the log file for details.")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

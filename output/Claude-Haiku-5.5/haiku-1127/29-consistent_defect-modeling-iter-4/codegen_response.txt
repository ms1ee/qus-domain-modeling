"""Order data model and coordinate handling.

An order is described by a restaurant point and a customer point. Input
coordinates are snapped to the top-left corner of their 100x100 grid cell so
that every location lies on a street intersection of the 8x8 grid.
"""

import itertools
from typing import Any, Dict, Optional, Sequence, Tuple

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = 8
INPUT_MIN = 0
INPUT_MAX = 1000

STATUS_CREATED = "created"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"

_order_ids = itertools.count(1)


def to_grid_corner(value: int) -> int:
    """Snap a coordinate to the top-left corner of its grid cell.

    Values beyond the last cell are clamped to the last cell's corner.

    Args:
        value: Raw coordinate in the range 0-1000.

    Returns:
        The corner coordinate (a multiple of CELL_SIZE, at most 700).
    """
    cell = min(max(int(value), 0) // CELL_SIZE, GRID_DIMENSION - 1)
    return cell * CELL_SIZE


def validate_point(point: Sequence[Any], label: str = "point") -> Tuple[int, int]:
    """Validate a 2-element coordinate pair.

    Args:
        point: Sequence containing x and y.
        label: Human readable name used in error messages.

    Returns:
        A tuple ``(x, y)`` of integers.

    Raises:
        ValueError: If the point is malformed, not integer or out of range.
    """
    if len(point) != 2:
        raise ValueError(f"{label} must contain exactly 2 coordinates")
    result = []
    for value in point:
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} coordinates must be integers")
        if not INPUT_MIN <= value <= INPUT_MAX:
            raise ValueError(
                f"{label} coordinate {value} out of range "
                f"{INPUT_MIN}-{INPUT_MAX}"
            )
        result.append(value)
    return result[0], result[1]


def manhattan(a: Tuple[int, int], b: Tuple[int, int]) -> int:
    """Return the Manhattan distance between two points.

    Args:
        a: First point ``(x, y)``.
        b: Second point ``(x, y)``.

    Returns:
        ``|ax - bx| + |ay - by|``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id (int): Unique order identifier.
        restaurant_x, restaurant_y (int): Snapped restaurant grid point.
        customer_x, customer_y (int): Snapped customer grid point.
        start_time (int): Tick at which the order was created.
        completion_time (int | None): Tick at which delivery completed.
        distance (int): Manhattan distance between restaurant and customer.
        time_taken (int | None): Ticks from creation to completion.
        rider_id (int | None): ID of the assigned rider.
        status (str): One of created, assigned, picking_up, delivering,
            completed.
        picked_up_time (int | None): Tick at which the rider reached the
            restaurant.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int],
                 start_time: int = 0) -> None:
        """Create an order from raw input coordinates.

        Args:
            res_pos: Restaurant point ``(x, y)`` in range 0-1000.
            cus_pos: Customer point ``(x, y)`` in range 0-1000.
            start_time: Creation tick.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        res_x, res_y = validate_point(res_pos, "restaurant")
        cus_x, cus_y = validate_point(cus_pos, "customer")
        self.id: int = next(_order_ids)
        self.restaurant_x = to_grid_corner(res_x)
        self.restaurant_y = to_grid_corner(res_y)
        self.customer_x = to_grid_corner(cus_x)
        self.customer_y = to_grid_corner(cus_y)
        self.start_time = int(start_time)
        self.completion_time: Optional[int] = None
        self.distance = manhattan(self.restaurant_pos, self.customer_pos)
        self.time_taken: Optional[int] = None
        self.rider_id: Optional[int] = None
        self.picked_up_time: Optional[int] = None
        self.status = STATUS_CREATED

    @property
    def restaurant_pos(self) -> Tuple[int, int]:
        """Return the restaurant grid point as ``(x, y)``."""
        return self.restaurant_x, self.restaurant_y

    @property
    def customer_pos(self) -> Tuple[int, int]:
        """Return the customer grid point as ``(x, y)``."""
        return self.customer_x, self.customer_y

    @property
    def efficiency(self) -> Optional[float]:
        """Return delivery efficiency (distance units per tick), if completed."""
        if self.time_taken is None:
            return None
        return self.distance / max(self.time_taken, 1)

    def assign_to(self, rider_id: int) -> None:
        """Record the assigned rider and mark the order as assigned.

        Args:
            rider_id: ID of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_accepted(self) -> None:
        """Mark the order as accepted by its rider (pickup in progress)."""
        self.status = STATUS_PICKING_UP

    def mark_picked_up(self, now: int) -> None:
        """Mark arrival at the restaurant.

        Args:
            now: Current simulated tick.
        """
        self.picked_up_time = now
        self.status = STATUS_DELIVERING

    def mark_completed(self, now: int) -> None:
        """Mark the order as delivered and compute timing data.

        Args:
            now: Current simulated tick.
        """
        self.completion_time = now
        self.time_taken = now - self.start_time
        self.status = STATUS_COMPLETED

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "rider_id": self.rider_id,
            "status": self.status,
            "efficiency": self.efficiency,
        }

"""Grid-based path planning on the 8x8 street network.

Streets are the grid lines where x or y is a multiple of the cell size.
Riders travel along streets using right-angle turns: they first move
horizontally to the target column, then vertically to the target row.
"""

import math
from typing import List, Tuple

Point = Tuple[int, int]


class PathPlanner:
    """Pathfinding and movement logic for riders on the street grid.

    Attributes:
        grid_dimension (int): Number of cells per side (8).
        cell_size (int): Size of one cell in map units (100).
        map_size (int): Total map size in units (800).
    """

    def __init__(self, grid_dimension: int = 8, cell_size: int = 100) -> None:
        """Create a planner for the given grid.

        Args:
            grid_dimension: Number of grid cells per side.
            cell_size: Size of each grid cell in units.
        """
        self.grid_dimension = grid_dimension
        self.cell_size = cell_size
        self.map_size = grid_dimension * cell_size

    def is_on_street(self, x: int, y: int) -> bool:
        """Return True if the point lies on a street (grid line).

        Args:
            x: X coordinate.
            y: Y coordinate.
        """
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def in_bounds(self, x: int, y: int) -> bool:
        """Return True if the point lies inside the map.

        Args:
            x: X coordinate.
            y: Y coordinate.
        """
        return 0 <= x <= self.map_size and 0 <= y <= self.map_size

    def next_step(self, x: int, y: int, tx: int, ty: int,
                  speed: int) -> Tuple[int, int, bool]:
        """Advance a position by at most ``speed`` units toward a target.

        Movement is horizontal first, then vertical, so the rider keeps
        travelling along streets. When the target lies directly above or
        below (or beside) the current position the movement is a straight
        line and ``special_case`` is True.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            tx: Target x coordinate.
            ty: Target y coordinate.
            speed: Maximum distance to travel this tick.

        Returns:
            A tuple ``(new_x, new_y, special_case)``.
        """
        if (x, y) == (tx, ty):
            return x, y, False
        special_case = x == tx or y == ty
        remaining = max(int(speed), 0)
        cx, cy = x, y
        if cx != tx and remaining > 0:
            step = min(remaining, abs(tx - cx))
            cx += step if tx > cx else -step
            remaining -= step
        if cx == tx and cy != ty and remaining > 0:
            step = min(remaining, abs(ty - cy))
            cy += step if ty > cy else -step
        return cx, cy, special_case

    def route_points(self, start: Point, end: Point, step: int = 50) -> List[Point]:
        """Return the street route from ``start`` to ``end`` as points.

        The route goes horizontally first, then vertically. Points are
        emitted every ``step`` units so the route can be drawn on a map.

        Args:
            start: Starting point.
            end: Destination point.
            step: Spacing between emitted points.

        Returns:
            List of points including both endpoints.
        """
        points: List[Point] = [start]
        x, y = start
        tx, ty = end
        while x != tx:
            delta = min(step, abs(tx - x))
            x += delta if tx > x else -delta
            points.append((x, y))
        while y != ty:
            delta = min(step, abs(ty - y))
            y += delta if ty > y else -delta
            points.append((x, y))
        return points

    def route_through(self, stops: List[Point], step: int = 50) -> List[Point]:
        """Build one continuous route visiting ``stops`` in order.

        Args:
            stops: Ordered list of points (for example rider, restaurant,
                customer).
            step: Spacing between emitted points.

        Returns:
            Combined list of route points without duplicated joints.
        """
        if not stops:
            return []
        route: List[Point] = [stops[0]]
        for previous, current in zip(stops, stops[1:]):
            segment = self.route_points(previous, current, step)
            route.extend(segment[1:])
        return route

    @staticmethod
    def ticks_needed(distance: int, speed: int) -> int:
        """Return the number of ticks needed to cover a distance.

        Args:
            distance: Distance in map units.
            speed: Speed in units per tick.

        Returns:
            Ceiling of distance / speed (0 if distance is 0).
        """
        if distance <= 0:
            return 0
        if speed <= 0:
            raise ValueError("speed must be positive")
        return math.ceil(distance / speed)

    @staticmethod
    def manhattan(a: Point, b: Point) -> int:
        """Return the Manhattan distance between two points.

        Args:
            a: First point.
            b: Second point.
        """
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Rider model: state machine, movement along streets and event log.

Rider states:
    0 (idle): available for new orders.
    1 (picking up): travelling to the restaurant of the current order.
    2 (delivering): travelling to the customer of the current order.

Every state change is recorded in ``Rider.log`` as a dictionary with at
least the keys ``event`` and ``order_id``.
"""

import logging
from typing import Any, Dict, List, Optional

import globalvar
from order_model import (
    STATUS_ASSIGNED,
    STATUS_COMPLETED,
    Order,
)
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

# The PRD lists both 10 and 20 units/tick; the rider data model fixes it at 20.
DEFAULT_RIDER_SPEED = 20
START_X = 400
START_Y = 400

EVENT_SPAWN = "spawn"
EVENT_ACCEPTED = "accepted_order"
EVENT_ARRIVED = "arrived_at_merchant"
EVENT_DELIVERED = "delivered_to_customer"


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    Attributes:
        rider_id (int): Unique identifier.
        x, y (int): Current position.
        speed (int): Units travelled per tick.
        state (int): 0 idle, 1 picking up, 2 delivering.
        orders (list[Order]): Assigned orders, including delivery history.
        special_case (bool): True while moving in a straight line toward a
            target directly above, below or beside the rider.
        log (list[dict]): Status change history.
        planner (PathPlanner): Pathfinding helper.
    """

    def __init__(self, rider_id: int, x: int = START_X, y: int = START_Y,
                 speed: int = DEFAULT_RIDER_SPEED) -> None:
        """Create a rider on a road point.

        Args:
            rider_id: Unique rider identifier.
            x: Initial x coordinate.
            y: Initial y coordinate.
            speed: Units per tick.
        """
        self.rider_id = rider_id
        self.x = x
        self.y = y
        self.speed = speed
        self.state = STATE_IDLE
        self.orders: List[Order] = []
        self.special_case = False
        self.log: List[Dict[str, Any]] = []
        self.planner = PathPlanner()
        self._record(EVENT_SPAWN, None)
        logger.info("Rider #%d spawned at (%d, %d)", rider_id, x, y)

    def _now(self) -> int:
        """Return the current simulated tick from the global store."""
        return int(globalvar.get_value("simulated_time", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> Dict[str, Any]:
        """Append a status change entry to the rider log.

        Args:
            event: Event name.
            order_id: Related order ID, or None.

        Returns:
            The created log entry.
        """
        entry = {
            "time": self._now(),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider #%d event=%s order=%s", self.rider_id, event, order_id)
        return entry

    def position(self) -> tuple:
        """Return the current position as ``(x, y)``."""
        return self.x, self.y

    def active_orders(self) -> List[Order]:
        """Return orders that are not yet completed, in service order."""
        return [order for order in self.orders if order.status != STATUS_COMPLETED]

    def current_order(self) -> Optional[Order]:
        """Return the order currently being served, or None."""
        active = self.active_orders()
        return active[0] if active else None

    def refresh_state(self) -> None:
        """Acknowledge newly assigned orders and update idle/picking-up state.

        If the rider is idle and receives a new order it switches to the
        picking-up state and logs ``accepted_order``. This method is also
        called by the order manager right after an assignment.
        """
        order = self.current_order()
        if order is None:
            if self.state != STATE_IDLE:
                self.state = STATE_IDLE
            return
        if order.status == STATUS_ASSIGNED:
            order.mark_accepted()
            if self.state == STATE_IDLE:
                self.state = STATE_PICKING_UP
            self._record(EVENT_ACCEPTED, order.id)

    def move(self) -> None:
        """Advance the rider by one tick.

        Handles the state transitions:
            idle -> picking up when an order is accepted,
            picking up -> delivering on arrival at the restaurant,
            delivering -> picking up (next order) or idle on delivery.
        """
        self.refresh_state()
        order = self.current_order()
        if order is None:
            self.state = STATE_IDLE
            self.special_case = False
            return
        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP

        if self.state == STATE_PICKING_UP:
            if self.position() == order.restaurant_pos:
                self._arrive_at_merchant(order)
                return
            target = order.restaurant_pos
        else:
            if self.position() == order.customer_pos:
                self._deliver(order)
                return
            target = order.customer_pos

        self.x, self.y, self.special_case = self.planner.next_step(
            self.x, self.y, target[0], target[1], self.speed
        )

    def _arrive_at_merchant(self, order: Order) -> None:
        """Handle arrival at the restaurant: pick up the food.

        Args:
            order: The order being picked up.
        """
        order.mark_picked_up(self._now())
        self.state = STATE_DELIVERING
        self.special_case = False
        self._record(EVENT_ARRIVED, order.id)

    def _deliver(self, order: Order) -> None:
        """Handle delivery to the customer and release the rider if free.

        Args:
            order: The order that has just been delivered.
        """
        order.mark_completed(self._now())
        self._record(EVENT_DELIVERED, order.id)
        self.special_case = False
        if self.active_orders():
            self.state = STATE_PICKING_UP
            self.refresh_state()
        else:
            self.state = STATE_IDLE
        logger.info("Order #%d delivered by rider #%d", order.id, self.rider_id)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [order.id for order in self.orders],
            "log": self.log,
        }

"""Order assignment, rider selection and order lifecycle management."""

import logging
import math
from typing import List, Optional

import globalvar
from order_model import STATUS_CREATED, Order
from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the rider that should receive ``order``.

    The rider with the fewest current (uncompleted) orders is preferred.
    Ties are broken by the Euclidean distance between the rider and the
    restaurant, then by rider ID.

    Args:
        order: The order that needs a rider.

    Returns:
        The selected Rider, or None if no rider exists.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    rest_x, rest_y = order.restaurant_pos

    def ranking(rider: Rider):
        distance = math.hypot(rider.x - rest_x, rider.y - rest_y)
        return len(rider.active_orders()), distance, rider.rider_id

    return min(riders, key=ranking)


class OrderManager:
    """Creates, tracks and assigns orders.

    Attributes:
        orders (list[Order]): All orders created in this session.
    """

    def __init__(self) -> None:
        """Create an empty order manager."""
        self.orders: List[Order] = []

    def create_order(self, res_pos, cus_pos, start_time: int) -> Order:
        """Create and register a new order.

        Args:
            res_pos: Restaurant point ``(x, y)``.
            cus_pos: Customer point ``(x, y)``.
            start_time: Creation tick.

        Returns:
            The new Order.

        Raises:
            ValueError: If coordinates are invalid.
        """
        order = Order(res_pos, cus_pos, start_time)
        self.orders.append(order)
        logger.info("Order #%d created", order.id)
        return order

    def pending_orders(self) -> List[Order]:
        """Return orders that have not yet been assigned to a rider."""
        return [order for order in self.orders if order.status == STATUS_CREATED]

    def completed_orders(self) -> List[Order]:
        """Return delivered orders."""
        return [order for order in self.orders if order.completion_time is not None]

    def get_order(self, order_id: int) -> Optional[Order]:
        """Return the order with ``order_id``, or None."""
        for order in self.orders:
            if order.id == order_id:
                return order
        return None

    def assign(self, order: Order) -> Optional[Rider]:
        """Assign ``order`` to the best rider.

        Args:
            order: Pending order to assign.

        Returns:
            The rider that received the order, or None if no rider exists.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            logger.warning("No rider available for order #%d", order.id)
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        rider.refresh_state()
        logger.info("Order #%d assigned to rider #%d", order.id, rider.rider_id)
        return rider

        self.rider_id_counter = self.rider_id_counter + 1
        rider = Rider(self.rider_id_counter)
        self.riders.append(rider)
        return rider

    def maybe_recruit(self, pending_count: int) -> Optional[Rider]:
        """Recruit a rider if funds allow and pending orders exceed idle riders.

        Args:
            pending_count: Number of orders waiting for a rider.

        Returns:
            The new Rider, or None if no recruitment happened.
        """
        min_funds = int(self.config["recruit_min_funds"])
        cost = int(self.config["recruit_cost"])
        if self.funds >= min_funds and pending_count > len(self.idle_riders()):
            rider = self.create_rider()
            self.funds = self.funds - cost
            logger.info("Recruited rider #%d (funds now %d)", rider.rider_id, self.funds)
            return rider
        return None

    def dispatch_pending(self) -> List[Order]:
        """Assign every pending order, recruiting riders when required.

        Returns:
            Orders that were assigned during this call.
        """
        manager = self.order_manager
        assigned: List[Order] = []
        for order in manager.pending_orders():
            self.maybe_recruit(len(manager.pending_orders()))
            if manager.assign(order) is not None:
                assigned.append(order)
        return assigned

    def submit_order(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> Dict[str, Any]:
        """Create an order and dispatch it.

        Args:
            res_pos: Restaurant point ``(x, y)``.
            cus_pos: Customer point ``(x, y)``.

        Returns:
            Dictionary with keys ``order``, ``rider`` (may be None) and
            ``new_rider`` (bool).

        Raises:
            ValueError: If coordinates are invalid.
        """
        riders_before = len(self.riders)
        order = self.order_manager.create_order(res_pos, cus_pos, self.simulated_time)
        self.dispatch_pending()
        rider = self.get_rider(order.rider_id) if order.rider_id is not None else None
        return {
            "order": order,
            "rider": rider,
            "new_rider": len(self.riders) > riders_before,
        }

    def tick(self) -> List[Dict[str, Any]]:
        """Advance the simulation by one tick.

        Pending orders are dispatched first, then every rider moves.

        Returns:
            Log entries produced during this tick (each contains ``rider_id``).
        """
        self.simulated_time = self.simulated_time + 1
        self.dispatch_pending()
        events: List[Dict[str, Any]] = []
        for rider in list(self.riders):
            start = len(rider.log)
            rider.move()
            events.extend(rider.log[start:])
        return events

    def run_ticks(self, count: int) -> List[Dict[str, Any]]:
        """Advance the simulation by ``count`` ticks.

        Args:
            count: Number of ticks to run.

        Returns:
            All events produced during the run.

        Raises:
            ValueError: If ``count`` is outside the allowed range.
        """
        limit = int(self.config.get("max_ticks_per_command", 10000))
        if not 1 <= count <= limit:
            raise ValueError(f"tick count must be between 1 and {limit}")
        events: List[Dict[str, Any]] = []
        for _ in range(count):
            events.extend(self.tick())
        for problem in self.check_consistency():
            logger.warning("Consistency check: %s", problem)
        return events

    def check_consistency(self) -> List[str]:
        """Check internal invariants and return a list of problems found.

        Returns:
            Human readable problem descriptions (empty when consistent).
        """
        problems: List[str] = []
        rider_ids = [rider.rider_id for rider in self.riders]
        if len(set(rider_ids)) != len(rider_ids):
            problems.append("duplicate rider IDs detected")
        for rider in self.riders:
            has_orders = bool(rider.active_orders())
            if rider.state == STATE_IDLE and has_orders:
                problems.append(f"rider #{rider.rider_id} is idle but has active orders")
            if rider.state != STATE_IDLE and not has_orders:
                problems.append(f"rider #{rider.rider_id} is busy without orders")
        known = set(rider_ids)
        for order in self.order_manager.orders:
            if order.rider_id is not None and order.rider_id not in known:
                problems.append(f"order #{order.id} references unknown rider")
        return problems

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the whole system."""
        return {
            "funds": self.funds,
            "simulated_time": self.simulated_time,
            "rider_id_counter": self.rider_id_counter,
            "config": self.config,
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in self.order_manager.orders],
        }

    def save_state(self, path: str) -> None:
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(self.to_dict(), handle, indent=2)
        logger.info("State saved to %s", path)

    """Load configuration overrides from a JSON file.

    Missing or malformed files fall back to the defaults.

    Args:
        config_path: Path to the JSON configuration file, or None.

    Returns:
        The effective configuration dictionary.
    """
    config = dict(DEFAULT_CONFIG)
    if not config_path:
        return config
    if not os.path.exists(config_path):

        return config

    except (OSError, ValueError, TypeError) as exc:
        logger.error("Could not load config %s: %s; using defaults", config_path, exc)
        config = dict(DEFAULT_CONFIG)

"""Status displays and ASCII map visualisation.

The map is a 16x16 character grid covering the 800x800 unit area, so each
character cell represents 50x50 units. Symbols:
    R  rider          S  restaurant      C  customer
    #  street         .  empty space     *  planned route
"""

from typing import Iterable, List, Sequence, Tuple

from order_model import Order, manhattan
from path_planner import PathPlanner
from rider import STATE_NAMES, Rider

Point = Tuple[int, int]


class DisplayManager:
    """Formats status tables, order/rider logs and the ASCII map.

    Attributes:
        map_dimension (int): Characters per side of the map (16).
        map_size (int): Size of the represented area in units (800).
        unit (int): Units represented by one map character.
        street_spacing (int): Distance between street lines (100).
    """

    def __init__(self, map_dimension: int = 16, map_size: int = 800,
                 street_spacing: int = 100) -> None:
        """Create a display manager.

        Args:
            map_dimension: Number of characters per map side.
            map_size: Size of the map area in units.
            street_spacing: Distance between streets in units.
        """
        self.map_dimension = map_dimension
        self.map_size = map_size
        self.unit = map_size // map_dimension
        self.street_spacing = street_spacing

    def _cell(self, value: int) -> int:
        """Convert a coordinate to a map index, clamped to the map."""
        return min(max(int(value), 0) // self.unit, self.map_dimension - 1)

    def render_map(self, riders: Iterable[Rider] = (),
                   orders: Iterable[Order] = (),
                   route: Sequence[Point] = ()) -> str:
        """Render the ASCII map with riders, orders and an optional route.

        Args:
            riders: Riders to draw as 'R'.
            orders: Orders whose restaurant ('S') and customer ('C') are drawn.
            route: Points of a planned route drawn as '*'.

        Returns:
            Multi-line string containing the map and a legend.
        """
        size = self.map_dimension
        grid: List[List[str]] = []
        for row in range(size):
            y = row * self.unit
            line = []
            for col in range(size):
                x = col * self.unit
                on_street = x % self.street_spacing == 0 or y % self.street_spacing == 0
                line.append("#" if on_street else ".")
            grid.append(line)

        for x, y in route:
            row, col = self._cell(y), self._cell(x)
            if grid[row][col] in "#.":
                grid[row][col] = "*"
        for order in orders:
            grid[self._cell(order.restaurant_y)][self._cell(order.restaurant_x)] = "S"
            grid[self._cell(order.customer_y)][self._cell(order.customer_x)] = "C"
        for rider in riders:
            grid[self._cell(rider.y)][self._cell(rider.x)] = "R"

        lines = ["    " + " ".join(str(col % 10) for col in range(size))]
        for row in range(size):
            lines.append(f"{row * self.unit:>4} " + " ".join(grid[row]))
        lines.append("Legend: R=rider S=restaurant C=customer #=street .=empty *=route")
        return "\n".join(lines)

    def format_status(self, system) -> str:
        """Return a snapshot of funds, riders and their orders.

        Args:
            system: The SystemManager instance.
        """
        riders = system.riders
        pending = len(system.order_manager.pending_orders())
        lines = [
            f"Simulated time: {system.simulated_time} tick(s)",
            f"System funds: {system.funds}",
            f"Riders: {len(riders)} (idle: {len(system.idle_riders())})",
            f"Pending orders: {pending}",
        ]
        if not riders:
            lines.append("  (no riders yet)")
        for rider in riders:
            order_ids = [order.id for order in rider.active_orders()]
            lines.append(
                f"  Rider #{rider.rider_id}: position=({rider.x}, {rider.y}) "
                f"state={STATE_NAMES[rider.state]} speed={rider.speed} "
                f"active_orders={len(order_ids)} order_ids={order_ids}"
            )
        return "\n".join(lines)

    def format_orders(self, orders: Sequence[Order]) -> str:
        """Return a table with details for all orders.

        Args:
            orders: Orders to display.
        """
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Distance':>8} {'Taken':>6} {'Efficiency':>10} "
                  f"{'Rider':>6} {'Status':>12}")
        lines = [header, "-" * len(header)]
        for order in orders:
            taken = "-" if order.time_taken is None else str(order.time_taken)
            eff = "-" if order.efficiency is None else f"{order.efficiency:.2f}"
            rider = "-" if order.rider_id is None else str(order.rider_id)
            restaurant = f"({order.restaurant_x},{order.restaurant_y})"
            customer = f"({order.customer_x},{order.customer_y})"
            lines.append(
                f"{order.id:>4} {order.start_time:>6} {restaurant:>12} {customer:>12} "
                f"{order.distance:>8} {taken:>6} {eff:>10} {rider:>6} {order.status:>12}"
            )
        return "\n".join(lines)

    def format_rider_logs(self, riders: Sequence[Rider]) -> str:
        """Return the status change history of every rider.

        Args:
            riders: Riders whose logs are displayed.
        """
        if not riders:
            return "No riders yet."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} (current state: {STATE_NAMES[rider.state]})")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order_text = "-" if entry["order_id"] is None else str(entry["order_id"])
                lines.append(
                    f"  t={entry['time']:>4} {entry['event']:<20} order={order_text:<5} "
                    f"at ({entry['x']},{entry['y']}) state={STATE_NAMES[entry['state']]}"
                )
        return "\n".join(lines)

    def format_route_summary(self, order: Order, rider: Rider,
                             planner: PathPlanner) -> str:
        """Return speed and time estimates for serving an order.

        Args:
            order: The assigned order.
            rider: The rider serving it.
            planner: Path planner used for tick estimates.
        """
        pickup = manhattan(rider.position(), order.restaurant_pos)
        delivery = order.distance
        pickup_ticks = planner.ticks_needed(pickup, rider.speed)
        delivery_ticks = planner.ticks_needed(delivery, rider.speed)
        return (
            f"Speed: {rider.speed} units/tick | pickup: {pickup} units "
            f"(~{pickup_ticks} ticks) | delivery: {delivery} units "
            f"(~{delivery_ticks} ticks) | total ~{pickup_ticks + delivery_ticks} ticks"
        )

    def format_event(self, entry: dict) -> str:
        """Return a one-line description of a rider log entry.

        Args:
            entry: A rider log dictionary.
        """
        order_text = "-" if entry["order_id"] is None else str(entry["order_id"])
        return (f"t={entry['time']} Rider #{entry['rider_id']} {entry['event']} "
                f"order={order_text} at ({entry['x']},{entry['y']})")

"""Command parsing, input validation and command dispatch for the CLI."""

import logging
from typing import Callable, Dict, List, Optional, Sequence, Tuple

from display_manager import DisplayManager
from order_model import INPUT_MAX, INPUT_MIN
from path_planner import PathPlanner
from system import SystemManager

logger = logging.getLogger(__name__)

SHORTCUTS: Dict[str, str] = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "t": "tick",
    "m": "map",
    "h": "help",
    "?": "help",
    "sv": "save",
    "hist": "history",
    "q": "quit",
    "x": "exit",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant x y, customer x y; 0-1000)
                       and show the assigned rider's route.
  status               System funds, riders and their orders.
  orders               Details for all orders.
  riders               Status change log of all riders.
  tick [n]             Advance the simulation by n ticks (default 1).
  map                  Show the ASCII map with riders and orders.
  save [file]          Save the system state as JSON (default dispatch_state.json).
  history              Show the command history.
  !! / !n              Repeat the last command / the n-th history entry.
  help (h, ?)          Show this help.
  quit (q) / exit (x)  Leave the program.
Shortcuts: s=start, st=status, o=orders, r=riders, t=tick, m=map, sv=save"""


class InputManager:
    """Parses and executes user commands.

    Attributes:
        system (SystemManager): The simulation being controlled.
        display (DisplayManager): Formatter for output text.
        reader (Callable[[str], str]): Function used to prompt for input.
        command_history (list[str]): Previously entered commands.
    """

    def __init__(self, system: SystemManager,
                 display: Optional[DisplayManager] = None,
                 reader: Callable[[str], str] = input) -> None:
        """Create the input manager.

        Args:
            system: The SystemManager to control.
            display: Optional display manager; a default one is created if None.
            reader: Function used to read additional input lines.
        """
        self.system = system
        self.display = display or DisplayManager()
        self.planner = PathPlanner()
        self.reader = reader
        self.command_history: List[str] = []
        self._commands: Dict[str, Callable[[List[str]], str]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "tick": self.cmd_tick,
            "map": self.cmd_map,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    @staticmethod
    def parse_command(line: str) -> Tuple[str, List[str]]:
        """Split a line into a command name and arguments.

        Shortcut keys are expanded to full command names.

        Args:
            line: Raw input line.

        Returns:
            Tuple ``(command, arguments)``; command is '' for blank input.
        """
        tokens = line.strip().split()
        if not tokens:
            return "", []
        name = tokens[0].lower()
        return SHORTCUTS.get(name, name), tokens[1:]

    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> List[int]:
        """Validate and convert four coordinate tokens.

        Args:
            tokens: Four strings: restaurant x, restaurant y, customer x,
                customer y.

        Returns:
            List of four integers.

        Raises:
            ValueError: On wrong count, non-integer or out-of-range values.
        """
        if len(tokens) != 4:
            raise ValueError(
                "expected 4 integer coordinates: restaurant_x restaurant_y "
                "customer_x customer_y"
            )
        values: List[int] = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise ValueError(f"'{token}' is not an integer") from None
            if not INPUT_MIN <= value <= INPUT_MAX:
                raise ValueError(
                    f"coordinate {value} is out of range {INPUT_MIN}-{INPUT_MAX}"
                )
            values.append(value)
        return values

    def _expand_history(self, line: str) -> Optional[str]:
        """Resolve '!!' and '!n' history references.

        Args:
            line: Stripped input line.

        Returns:
            The expanded command, or None when the reference is invalid.
        """
        if line == "!!":
            return self.command_history[-1] if self.command_history else None
        if line.startswith("!") and line[1:].isdigit():
            index = int(line[1:])
            if 1 <= index <= len(self.command_history):
                return self.command_history[index - 1]
            return None
        return line

    def handle(self, line: str) -> str:
        """Execute one input line and return the text to print.

        Args:
            line: Raw input line.

        Returns:
            Output text (may be empty). Errors are reported as text.
        """
        stripped = line.strip()
        if not stripped:
            return ""
        expanded = self._expand_history(stripped)
        if expanded is None:
            return "No matching command in history."
        self.command_history.append(expanded)
        command, args = self.parse_command(expanded)
        handler = self._commands.get(command)
        if handler is None:
            return f"Unknown command '{command}'. Type 'help' for the command list."
        try:
            return handler(args)
        except ValueError as exc:
            return f"Input error: {exc}"
        except Exception as exc:  # error recovery: keep the system running
            logger.exception("Command '%s' failed", command)
            return f"Internal error: {exc}. System state was preserved."

    def cmd_start(self, args: List[str]) -> str:
        """Create an order and show the assigned rider's route.

        Args:
            args: Four coordinates, or empty to be prompted for them.
        """
        if not args:
            args = self.reader(
                "Enter restaurant_x restaurant_y customer_x customer_y: "
            ).split()
        values = self.parse_coordinates(args)
        res_pos = (values[0], values[1])
        cus_pos = (values[2], values[3])
        result = self.system.submit_order(res_pos, cus_pos)
        order = result["order"]
        rider = result["rider"]

        lines = [
            f"Order #{order.id} created at t={order.start_time}.",
            f"Restaurant grid point: ({order.restaurant_x}, {order.restaurant_y})",
            f"Customer grid point: ({order.customer_x}, {order.customer_y})",
            f"New rider added: {'yes' if result['new_rider'] else 'no'}",
        ]
        if rider is None:
            lines.append(
                f"No rider available; order is pending (funds: {self.system.funds})."
            )
            return "\n".join(lines)

        lines.append(f"Assigned to Rider #{rider.rider_id} "
                     f"(state: {self._state_name(rider)}, "
                     f"active orders: {len(rider.active_orders())}).")
        route = self.planner.route_through(
            [rider.position(), order.restaurant_pos, order.customer_pos]
        )
        lines.append(self.display.render_map(
            riders=self.system.riders, orders=[order], route=route))
        lines.append(self.display.format_route_summary(order, rider, self.planner))
        return "\n".join(lines)

    @staticmethod
    def _state_name(rider) -> str:
        """Return a readable state name for a rider."""
        from rider import STATE_NAMES
        return STATE_NAMES[rider.state]

    def cmd_status(self, args: List[str]) -> str:
        """Show the system status snapshot."""
        return self.display.format_status(self.system)

    def cmd_orders(self, args: List[str]) -> str:
        """Show the details of all orders."""
        return self.display.format_orders(self.system.order_manager.orders)

    def cmd_riders(self, args: List[str]) -> str:
        """Show the status change log of all riders."""
        return self.display.format_rider_logs(self.system.riders)

    def cmd_tick(self, args: List[str]) -> str:
        """Advance the simulation.

        Args:
            args: Optional number of ticks.
        """
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise ValueError(f"'{args[0]}' is not an integer tick count") from None
        events = self.system.run_ticks(count)
        lines = [f"Advanced {count} tick(s). Current time: t={self.system.simulated_time}."]
        if events:
            lines.extend(self.display.format_event(entry) for entry in events)
        else:
            lines.append("No rider status changes.")
        lines.append(f"System funds: {self.system.funds}")
        return "\n".join(lines)

    def cmd_map(self, args: List[str]) -> str:
        """Show the ASCII map with riders and active orders."""
        active = [o for o in self.system.order_manager.orders if o.status != "completed"]
        return self.display.render_map(riders=self.system.riders, orders=active)

    def cmd_save(self, args: List[str]) -> str:
        """Save the system state to a JSON file.

        Args:
            args: Optional file path.
        """
        path = args[0] if args else "dispatch_state.json"
        try:
            self.system.save_state(path)
        except OSError as exc:
            return f"Could not save state to {path}: {exc}"
        return f"State saved to {path}."

    def cmd_history(self, args: List[str]) -> str:
        """Show the command history."""
        if not self.command_history:
            return "No commands entered yet."
        return "\n".join(f"{i:>3}  {cmd}" for i, cmd in enumerate(self.command_history, 1))

    def cmd_help(self, args: List[str]) -> str:
        """Show the command help text."""
        return HELP_TEXT

    def cmd_quit(self, args: List[str]) -> str:
        """Stop the main loop."""
        self.system.is_running = False
        return "Goodbye."

def setup_logging(debug: bool, log_file: Optional[str]) -> None:
    """Configure logging handlers.

        log_file: Optional file that receives log output.
    """
    level = logging.DEBUG if debug else logging.WARNING
    handlers: List[logging.Handler] = []
    if log_file:
        handlers.append(logging.FileHandler(log_file, encoding="utf-8"))
    else:
        handlers.append(logging.StreamHandler(sys.stderr))
    logging.basicConfig(
        level=level,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        handlers=handlers,
    )


def main(argv: Optional[List[str]] = None) -> int:
    """Run the interactive command loop.

    setup_logging(args.debug, args.log_file)
    try:
        system = init_system(args.config)
    except Exception:  # startup failure is reported and aborts cleanly
        logger.exception("System initialisation failed")

        return 1

    display = DisplayManager()
    manager = InputManager(system, display, reader=input)
    print(BANNER)

    while system.is_running:
        try:
            line = input("dispatch> ")
        except (EOFError, KeyboardInterrupt):

            break
        try:
            output = manager.handle(line)
        except Exception:  # keep the loop alive on unexpected failures

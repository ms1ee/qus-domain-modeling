"""Command-line interface for the Beijing Subway fare calculation system.

Responsibilities:
    * parse command-line arguments
    * validate user input ("Starting Station; Ending Station")
    * run continuous query loops
    * print or export query results
    * run the data integrity check when ``--check_data`` is given
"""

import argparse
from typing import List, Optional, Tuple

from data_manager import run_integrity_check
from data_loader import DATA_FILE_PATH  # noqa: F401  (re-exported for convenience)
from path_visualizer import render
from pathfinding import PathfindingConfig, VALID_MODES
from route_service import (
    InvalidQueryError,
    RouteError,
    StationNotFoundError,
    compute_route,
)
from subway_network import SubwayNetwork


EXIT_COMMANDS = ("quit", "exit", "q")
PROMPT_TEXT = "Enter query as 'Starting Station; Ending Station' (or 'quit' to exit): "


class InputFormatError(ValueError):
    """Raised when the user input does not match 'Starting Station; Ending Station'."""


def build_parser() -> argparse.ArgumentParser:
    """Create the argument parser for the CLI."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (command-line interface)."
    )
    parser.add_argument(
        "--output_file",
        default=None,
        help="Export query results to this file (results are appended).",
    )
    parser.add_argument(
        "--format",
        choices=("text", "json"),
        default="text",
        help="Output format: 'text' (default) or 'json'.",
    )
    parser.add_argument(
        "--mode",
        choices=VALID_MODES,
        default="shortest_distance",
        help="Path query mode (default: shortest_distance).",
    )
    parser.add_argument(
        "--max_transfers",
        type=int,
        default=None,
        help="Maximum number of allowed transfers (default: unlimited).",
    )
    parser.add_argument(
        "--line_preference",
        default=None,
        help="Preferred subway line, e.g. \"Line 1\". Other lines are penalised softly.",
    )
    parser.add_argument(
        "--check_data",
        action="store_true",
        help="Run the data integrity check and exit.",
    )
    return parser


def parse_query(text: str) -> Tuple[str, str]:
    """Split a raw query string into (starting station, ending station).

    Accepts the ASCII semicolon ';' as well as the full-width '；' separator.

    Raises:
        InputFormatError: if the format is not 'Starting Station; Ending Station'.
    """
    if not isinstance(text, str):
        raise InputFormatError(
            "Input format incorrect, please use 'Starting Station; Ending Station' format"
        )
    normalized = text.replace("\uff1b", ";")
    if normalized.count(";") != 1:
        raise InputFormatError(
            "Input format incorrect, please use 'Starting Station; Ending Station' format"
        )
    start, end = [part.strip() for part in normalized.split(";")]
    if not start or not end:
        raise InputFormatError(
            "Input format incorrect, please use 'Starting Station; Ending Station' format"
        )
    return start, end


def resolve_query(network: SubwayNetwork, text: str) -> Tuple[str, str]:
    """Parse the query and standardise both station names against the network.

    Returns:
        (official starting station name, official ending station name)

    Raises:
        InputFormatError, StationNotFoundError, InvalidQueryError
    """
    start_input, end_input = parse_query(text)

    start = network.resolve_station(start_input)
    if start is None:
        raise StationNotFoundError(
            f"Starting station '{start_input}' does not exist, please check the station name"
        )
    end = network.resolve_station(end_input)
    if end is None:
        raise StationNotFoundError(
            f"Ending station '{end_input}' does not exist, please check the station name"
        )
    if start == end:
        raise InvalidQueryError("Starting station cannot be the same as the ending station")
    return start, end


def resolve_line(network: SubwayNetwork, text: Optional[str]) -> Optional[str]:
    """Match a user supplied line name (case-insensitive, '1' accepted for 'Line 1')."""
    if text is None:
        return None
    key = text.strip().casefold()
    if not key:
        return None
    lines = network.lines()
    for line in lines:
        if line.casefold() == key:
            return line
    for line in lines:
        if line.casefold() == "line " + key:
            return line
    return None


def run_query(
    network: SubwayNetwork,
    text: str,
    config: PathfindingConfig,
    fmt: str,
    output_file: Optional[str],
) -> bool:
    """Execute one query, print the result and optionally export it.

    Returns:
        True if the query succeeded, False otherwise.
    """
    try:
        start, end = resolve_query(network, text)
    except RouteError as exc:
        print(f"Error: {exc}")
        return False
    except InputFormatError as exc:
        print(f"Error: {exc}")
        return False

    try:
        route = compute_route(network, start, end, config)
        output = render(route, fmt)
    except RouteError as exc:
        print(f"Error: {exc}")
        return False
    except Exception as exc:  # Path or fare calculation failure must not stop the CLI.
        print(f"Error: path or fare calculation failed: {exc}")
        return False

    print(output)

    if output_file:
        try:
            with open(output_file, "a", encoding="utf-8") as handle:
                handle.write(output)
                handle.write("\n\n")
            print(f"Result exported to: {output_file}")
        except OSError as exc:
            print(f"Error: could not write output file '{output_file}': {exc}")
            return False
    return True


def interactive_loop(
    network: SubwayNetwork,
    config: PathfindingConfig,
    fmt: str,
    output_file: Optional[str],
) -> None:
    """Continuously read queries until the user quits or input ends."""
    print("Beijing Subway fare calculator. Type 'quit' to exit.")
    print("Input format: Starting Station; Ending Station")
    while True:
        try:
            raw = input(PROMPT_TEXT)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.casefold() in EXIT_COMMANDS:
            break
        run_query(network, text, config, fmt, output_file)
        print()
    print("Goodbye.")


def main(argv: Optional[List[str]] = None) -> int:
    """CLI entry point. Returns the process exit code."""
    parser = build_parser()
    args = parser.parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")

    try:
        network = SubwayNetwork.from_file()
    except (OSError, ValueError) as exc:
        print(f"Error: could not load subway data: {exc}")
        return 1

    if args.check_data:
        report = run_integrity_check(network)
        print(report.render())
        return 0 if report.ok else 1

    preferred: List[str] = []
    if args.line_preference:
        line = resolve_line(network, args.line_preference)
        if line is None:
            print(
                f"Warning: line '{args.line_preference}' not found in the network; "
                "line preference ignored."
            )
        else:
            preferred = [line]

    try:
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=preferred,
        )
    except ValueError as exc:
        parser.error(str(exc))
        return 2

    interactive_loop(network, config, args.format, args.output_file)
    return 0

"""Loading, validation and saving of the Beijing Subway dataset (subway_data.json)."""

import json
import os
from numbers import Number


DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


class DataFormatError(ValueError):
    """Raised when the subway dataset does not follow the expected structure."""


def _is_number(value) -> bool:
    """Return True for int/float values, excluding booleans."""
    return isinstance(value, Number) and not isinstance(value, bool)


def validate_subway_data(data) -> None:
    """Check the structure of a subway dataset.

    Expected structure:
        {"stations": [{"name": str, "lines": [str]}],
         "connections": [{"source": str, "dest": str, "distance": int, "line": str}]}

    Raises:
        DataFormatError: if the structure is invalid.
    """
    if not isinstance(data, dict):
        raise DataFormatError("Subway data must be a JSON object")

    stations = data.get("stations")
    connections = data.get("connections")
    if not isinstance(stations, list):
        raise DataFormatError("'stations' must be a list")
    if not isinstance(connections, list):
        raise DataFormatError("'connections' must be a list")

    for index, station in enumerate(stations):
        if not isinstance(station, dict):
            raise DataFormatError(f"Station #{index} must be an object")
        name = station.get("name")
        if not isinstance(name, str) or not name.strip():
            raise DataFormatError(f"Station #{index} has an invalid 'name'")
        lines = station.get("lines", [])
        if not isinstance(lines, list) or not all(isinstance(line, str) for line in lines):
            raise DataFormatError(f"Station '{name}' has invalid 'lines'")

    for index, conn in enumerate(connections):
        if not isinstance(conn, dict):
            raise DataFormatError(f"Connection #{index} must be an object")
        for key in ("source", "dest", "line"):
            if not isinstance(conn.get(key), str) or not conn.get(key).strip():
                raise DataFormatError(f"Connection #{index} has an invalid '{key}'")
        distance = conn.get("distance")
        if not _is_number(distance) or distance <= 0:
            raise DataFormatError(f"Connection #{index} has an invalid 'distance'")


def load_subway_data(path=DATA_FILE_PATH) -> dict:
    """Load and validate the subway dataset from a JSON file.

    Returns:
        The JSON content of the file as a dict.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    validate_subway_data(data)
    return data


def save_subway_data(data: dict, path=DATA_FILE_PATH) -> None:
    """Validate and write a subway dataset to a JSON file."""
    validate_subway_data(data)
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")

"""Core graph data structures: Edge and Graph (weighted undirected graph)."""

from numbers import Number


class Edge:
    """A weighted, line-tagged connection between two stations.

    Attributes:
        source: station number of one endpoint.
        dest: station number of the other endpoint.
        distance: actual distance between the stations in meters.
        line: name of the subway line the connection belongs to.
    """

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, Number):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by shortest-path algorithms (equal to the distance)."""
        return self.distance

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex} is not an endpoint of this edge")

    def __repr__(self):
        return (
            f"Edge(source={self.source}, dest={self.dest}, "
            f"distance={self.distance}, line={self.line!r})"
        )


class Graph:
    """Weighted undirected graph with adjacency lists.

    Station numbers are assigned by the caller. The graph also keeps the
    mapping ``station_lines`` from station name to the list of lines it belongs to.
    Each undirected edge object is stored in the adjacency lists of both endpoints,
    so updating an edge object updates both directions at once.
    """

    def __init__(self):
        self.adj = {}  # station number -> list[Edge]
        self.station_lines = {}  # station name -> list[str]

    # ----- vertices -----
    def add_node(self, vertex):
        """Register a station number as a vertex (no-op if already present)."""
        self.adj.setdefault(vertex, [])

    def has_node(self, vertex):
        """Return True if the vertex exists in the graph."""
        return vertex in self.adj

    def vertices(self):
        """Return the list of station numbers in the graph."""
        return list(self.adj.keys())

    def V(self):
        """Return the number of vertices."""
        return len(self.adj)

    # ----- edges -----
    def add_edge(self, edge):
        """Add an undirected edge to the graph."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self.adj[edge.dest].append(edge)

    def neighbors(self, vertex):
        """Return the list of edges incident to ``vertex``."""
        return self.adj.get(vertex, [])

    def edges(self):
        """Return every distinct edge object in the graph once."""
        seen = set()
        result = []
        for incident in self.adj.values():
            for edge in incident:
                if id(edge) not in seen:
                    seen.add(id(edge))
                    result.append(edge)
        return result

    def E(self):
        """Return the number of distinct edges."""
        return len(self.edges())

    # ----- station line information -----
    def set_station_lines(self, station_name, lines):
        """Set the lines of a station (duplicates removed, order preserved)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines))

    def get_station_lines(self, station_name):
        """Return the lines of a station (empty list if unknown)."""
        return list(self.station_lines.get(station_name, []))

"""Path calculation: priority queue, path configuration and Dijkstra's algorithm.

Dijkstra's algorithm is run over states (station number, arrival line). This makes
transfer counting exact: a transfer happens whenever consecutive edges belong to
different lines. Priority queue operations are O(log V), giving an overall
complexity of O((V + E) log V) for the search.
"""

import heapq
import itertools


VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Equivalent distance (meters) added per transfer in comprehensive mode.
TRANSFER_PENALTY_M = 3000
# Equivalent distance (meters) added for each edge that is not on a preferred line.
PREFERRED_LINE_PENALTY_M = 1000


class MinPQ:
    """Min priority queue implemented on top of a binary heap (heapq)."""

    def __init__(self):
        self._heap = []
        self._seq = itertools.count()  # tie-breaker so items are never compared

    def insert(self, key, item):
        """Insert an item with the given priority key."""
        heapq.heappush(self._heap, (key, next(self._seq), item))

    def del_min(self):
        """Remove and return the item with the smallest key."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return heapq.heappop(self._heap)[2]

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._heap[0][0]

    def is_empty(self):
        """Return True if the queue has no items."""
        return not self._heap

    def __len__(self):
        return len(self._heap)


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: one of VALID_MODES.
            - "shortest_distance": minimise total distance.
            - "least_transfer": minimise transfers, then distance.
            - "comprehensive": minimise distance + TRANSFER_PENALTY_M * transfers.
        max_transfers: maximum allowed transfers, or None for no limit.
        preferred_lines: list of preferred line names; edges on other lines
            receive PREFERRED_LINE_PENALTY_M as an extra cost (soft preference).
    """

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Unsupported mode '{mode}'. Choose from {VALID_MODES}")
        if max_transfers is not None and (not isinstance(max_transfers, int) or max_transfers < 0):
            raise ValueError("max_transfers must be a non-negative integer or None")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class _Label:
    """Best known label for a search state."""

    __slots__ = ("key", "distance", "penalty", "transfers", "prev", "edge")

    def __init__(self, key, distance, penalty, transfers, prev, edge):
        self.key = key  # comparable score used by the priority queue
        self.distance = distance  # real travelled distance (meters)
        self.penalty = penalty  # accumulated preference penalty
        self.transfers = transfers  # number of line changes so far
        self.prev = prev  # previous state, or None for the source
        self.edge = edge  # edge used to reach this state


class ShortestPath:
    """Dijkstra-based optimal path computation from a single source station.

    Attributes:
        dist_to: station number -> real distance (meters) of the chosen path.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_node(source):
            raise ValueError(f"Source station {source} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._labels = {}  # state (node, line) -> _Label
        self._best = {}  # node -> first (optimal) settled state
        self._run()

    def _score(self, distance, penalty, transfers):
        """Compute the comparable priority of a label according to the mode."""
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, distance + penalty)
        if mode == "comprehensive":
            return (distance + penalty + TRANSFER_PENALTY_M * transfers, transfers)
        return (distance + penalty, transfers)

    def _preference_penalty(self, line):
        """Return the soft penalty for travelling on ``line``."""
        preferred = self.config.preferred_lines
        if preferred and line not in preferred:
            return PREFERRED_LINE_PENALTY_M
        return 0

    def _run(self):
        """Run Dijkstra's algorithm over (station, line) states."""
        max_transfers = self.config.max_transfers
        start_state = (self.source, None)
        start_label = _Label(self._score(0, 0, 0), 0, 0, 0, None, None)
        self._labels[start_state] = start_label

        pq = MinPQ()
        pq.insert(start_label.key, start_state)
        settled = set()

        while not pq.is_empty():
            state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)

            node, line = state
            label = self._labels[state]
            if node not in self._best:
                self._best[node] = state
                self.dist_to[node] = label.distance

            for edge in self.graph.neighbors(node):
                nxt = edge.other(node)
                step = 1 if (line is not None and edge.line != line) else 0
                transfers = label.transfers + step
                if max_transfers is not None and transfers > max_transfers:
                    continue
                distance = label.distance + edge.weight
                penalty = label.penalty + self._preference_penalty(edge.line)
                key = self._score(distance, penalty, transfers)

                next_state = (nxt, edge.line)
                if next_state in settled:
                    continue
                existing = self._labels.get(next_state)
                if existing is not None and existing.key <= key:
                    continue
                self._labels[next_state] = _Label(key, distance, penalty, transfers, state, edge)
                pq.insert(key, next_state)

    def has_path_to(self, target):
        """Return True if ``target`` is reachable under the current configuration."""
        return target in self._best

    def path_with_edges(self, target):
        """Return the path to ``target`` as (station numbers, edges).

        The station list starts at the source and ends at the target. The edge
        list has one fewer element. Returns ([], []) if the target is unreachable.
        """
        state = self._best.get(target)
        if state is None:
            return [], []
        stations = []
        edges = []
        current = state
        while True:
            label = self._labels[current]
            stations.append(current[0])
            if label.prev is None:
                break
            edges.append(label.edge)
            current = label.prev
        stations.reverse()
        edges.reverse()
        return stations, edges

    def get_transfer_count(self, target):
        """Return the number of line changes on the path to ``target`` (0 if unreachable)."""
        state = self._best.get(target)
        if state is None:
            return 0
        return self._labels[state].transfers

"""Beijing Subway fare calculation.

Fare structure (distance in meters, intervals are left-closed and right-open):
    [0, 6000)            -> 3 yuan
    [6000, 12000)        -> 4 yuan
    [12000, 22000)       -> 5 yuan
    [22000, 32000)       -> 6 yuan
    [32000, 52000)       -> 7 yuan
    then +1 yuan for every additional 20 km (52000-72000 -> 8, ...)
Fractional distances are rounded up to the next whole meter before lookup.
"""

import math
from numbers import Number


# (upper bound in meters, fare in yuan) for the fixed tiers below 32 km.
_FIXED_TIERS = (
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
)
_EXTENDED_BASE_FARE = 7  # fare for [32000, 52000)
_EXTENDED_STEP_M = 20000  # every 20 km
_EXTENDED_STEP_FARE = 1


def _validate_distance(distance):
    """Validate that the distance is a finite, non-negative number."""
    if isinstance(distance, bool) or not isinstance(distance, Number):
        raise TypeError("distance must be a number of meters")
    if isinstance(distance, float) and (math.isnan(distance) or math.isinf(distance)):
        raise ValueError("distance must be a finite number")
    if distance < 0:
        raise ValueError("distance must be non-negative")


def Sum_money(distance) -> int:
    """Return the subway fare in yuan for a travel distance given in meters.

    Fractional distances are ceiling-rounded to whole meters, and boundary
    values belong to the next higher tier (e.g. 6000 m costs 4 yuan).
    """
    _validate_distance(distance)
    meters = math.ceil(distance)
    for upper, fare in _FIXED_TIERS:
        if meters < upper:
            return fare
    return _EXTENDED_BASE_FARE + (meters - 32000) // _EXTENDED_STEP_M * _EXTENDED_STEP_FARE


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimate travel time in minutes.

    Uses about 1.5 minutes per kilometer plus about 5 minutes per transfer.

    Args:
        distance: travel distance in meters.
        transfer_count: number of transfers.

    Returns:
        Estimated time in minutes (float).
    """
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise ValueError("transfer_count must be a non-negative integer")
    return float(distance) / 1000.0 * 1.5 + transfer_count * 5.0

"""Subway network modelling and data management.

SubwayNetwork builds the weighted undirected graph from the dataset, assigns
station numbers (station_to_number), keeps line information, and provides
data update operations: adding lines, modifying stations, and updating distances.
"""

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph


def _normalize_name(name):
    """Collapse internal whitespace and strip a station or line name."""
    if not isinstance(name, str):
        raise ValueError("name must be a string")
    cleaned = " ".join(name.split())
    if not cleaned:
        raise ValueError("name must not be empty")
    return cleaned


def _normalize_distance(distance):
    """Validate a distance in meters and return it as a positive integer."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("distance must be a number of meters")
    if distance <= 0:
        raise ValueError("distance must be positive")
    return int(round(distance))


class SubwayNetwork:
    """Beijing Subway network: stations, lines, connections and graph.

    Attributes:
        graph: the Graph instance holding adjacency lists and station lines.
        station_to_number: station name -> station number.
        number_to_station: station number -> station name.
    """

    def __init__(self, data=None):
        if data is None:
            data = load_subway_data()
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        self._station_order = []
        self._connections = []  # list of dicts: source, dest, distance, line
        self.undeclared_stations = []  # stations used by connections but not declared
        self._build(data)

    @classmethod
    def from_file(cls, path=DATA_FILE_PATH):
        """Create a network from a subway_data.json file."""
        return cls(load_subway_data(path))

    # ----- construction -----
    def _build(self, data):
        """Populate stations, lines and connections from the dataset."""
        declared = set()
        for station in data.get("stations", []):
            name = _normalize_name(station["name"])
            declared.add(name)
            lines = [str(line).strip() for line in station.get("lines", []) if str(line).strip()]
            self._register_station(name, lines)

        for conn in data.get("connections", []):
            source = _normalize_name(conn["source"])
            dest = _normalize_name(conn["dest"])
            for name in (source, dest):
                if name not in declared and name not in self.undeclared_stations:
                    self.undeclared_stations.append(name)
            self._add_connection_record(source, dest, conn["distance"], conn["line"].strip())

    def _register_station(self, name, lines=()):
        """Add a station if needed and merge the given lines into its line list."""
        if name not in self.station_to_number:
            number = len(self.station_to_number)
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self._station_order.append(name)
            self.graph.add_node(number)
            self.graph.set_station_lines(name, [])
        current = self.graph.get_station_lines(name)
        self.graph.set_station_lines(name, current + [line for line in lines if line])
        return self.station_to_number[name]

    def _add_connection_record(self, source, dest, distance, line):
        """Create a bidirectional connection between two stations on a line."""
        if source == dest:
            raise ValueError(f"Connection from '{source}' to itself is not allowed")
        if not line:
            raise ValueError("connection line must not be empty")
        distance = _normalize_distance(distance)
        a = self._register_station(source, [line])
        b = self._register_station(dest, [line])
        self.graph.add_edge(Edge(a, b, distance, line))
        self._connections.append(
            {"source": source, "dest": dest, "distance": distance, "line": line}
        )

    # ----- queries -----
    def resolve_station(self, name):
        """Return the official station name matching ``name``, or None.

        Matching ignores case, surrounding and repeated whitespace, and an optional
        trailing '站' (station) suffix.
        """
        if not isinstance(name, str):
            return None
        cleaned = " ".join(name.split())
        if not cleaned:
            return None
        if cleaned in self.station_to_number:
            return cleaned
        candidates = [cleaned]
        if cleaned.endswith("站"):
            candidates.append(cleaned[:-1].strip())
        keys = {official.casefold().replace(" ", ""): official for official in self._station_order}
        for candidate in candidates:
            key = candidate.casefold().replace(" ", "")
            if key in keys:
                return keys[key]
        return None

    def has_station(self, name):
        """Return True if the station exists."""
        return name in self.station_to_number

    def station_number(self, name):
        """Return the station number of an official station name."""
        if name not in self.station_to_number:
            raise KeyError(f"Unknown station '{name}'")
        return self.station_to_number[name]

    def station_name(self, number):
        """Return the official station name for a station number."""
        return self.number_to_station[number]

    def station_names(self):
        """Return all station names in definition order."""
        return list(self._station_order)

    def lines(self):
        """Return the sorted list of all line names in the network."""
        found = set()
        for lines in self.graph.station_lines.values():
            found.update(lines)
        return sorted(found)

    def stations_on_line(self, line):
        """Return the names of the stations that belong to ``line``."""
        return [name for name in self._station_order if line in self.graph.get_station_lines(name)]

    def transfer_stations(self):
        """Return the names of stations served by more than one line."""
        return [
            name for name in self._station_order if len(self.graph.get_station_lines(name)) > 1
        ]

    def connection_records(self):
        """Return a copy of all connection records."""
        return [dict(record) for record in self._connections]

    # ----- updates -----
    def add_station(self, name, lines=()):
        """Add a new station that is not yet connected to the network."""
        cleaned = _normalize_name(name)
        if self.resolve_station(cleaned) is not None:
            raise ValueError(f"Station '{cleaned}' already exists")
        clean_lines = [str(line).strip() for line in lines if str(line).strip()]
        self._register_station(cleaned, clean_lines)
        return cleaned

    def add_line(self, line_name, station_names, distances):
        """Add a new line through an ordered list of existing stations.

        Args:
            line_name: name of the new line (must not exist yet).
            station_names: ordered station names of the line (at least two).
            distances: distances in meters between consecutive stations.

        Returns:
            The number of connections added.
        """
        line = _normalize_name(line_name)
        if line in self.lines():
            raise ValueError(f"Line '{line}' already exists")
        resolved = []
        for name in station_names:
            official = self.resolve_station(name)
            if official is None:
                raise ValueError(f"Station '{name}' does not exist")
            resolved.append(official)
        if len(resolved) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(resolved) - 1:
            raise ValueError("distances must contain one value per consecutive station pair")
        normalized = [_normalize_distance(d) for d in distances]
        for index, distance in enumerate(normalized):
            self._add_connection_record(resolved[index], resolved[index + 1], distance, line)
        return len(normalized)

    def update_station(self, old_name, new_name=None, lines=None):
        """Rename a station and/or replace its line list.

        Returns:
            The (possibly new) official name of the station.
        """
        official = self.resolve_station(old_name)
        if official is None:
            raise ValueError(f"Station '{old_name}' does not exist")

        if new_name is not None:
            target = _normalize_name(new_name)
            existing = self.resolve_station(target)
            if existing is not None and existing != official:
                raise ValueError(f"Station '{target}' already exists")
            if target != official:
                number = self.station_to_number.pop(official)
                self.station_to_number[target] = number
                self.number_to_station[number] = target
                self._station_order[self._station_order.index(official)] = target
                self.graph.station_lines[target] = self.graph.station_lines.pop(official)
                for record in self._connections:
                    if record["source"] == official:
                        record["source"] = target
                    if record["dest"] == official:
                        record["dest"] = target
                official = target

        if lines is not None:
            clean_lines = [str(line).strip() for line in lines if str(line).strip()]
            self.graph.set_station_lines(official, clean_lines)
        return official

    def update_distance(self, a, b, distance, line=None):
        """Update the distance between two adjacent stations.

        Args:
            a, b: station names (any accepted spelling).
            distance: new distance in meters.
            line: restrict the update to this line (None updates all connections).

        Returns:
            The number of connections updated.
        """
        name_a = self.resolve_station(a)
        name_b = self.resolve_station(b)
        if name_a is None or name_b is None:
            raise ValueError("Both stations must exist")
        distance = _normalize_distance(distance)
        num_a = self.station_to_number[name_a]
        num_b = self.station_to_number[name_b]

        updated = 0
        for edge in self.graph.neighbors(num_a):
            if edge.other(num_a) == num_b and (line is None or edge.line == line):
                edge.distance = distance
                updated += 1
        if updated == 0:
            raise ValueError(f"No connection between '{name_a}' and '{name_b}'")

        for record in self._connections:
            pair = {record["source"], record["dest"]}
            if pair == {name_a, name_b} and (line is None or record["line"] == line):
                record["distance"] = distance
        return updated

    def to_data(self):
        """Export the network in the subway_data.json structure."""
        stations = [
            {"name": name, "lines": self.graph.get_station_lines(name)}
            for name in self._station_order
        ]
        return {"stations": stations, "connections": self.connection_records()}

"""Route computation service: combines the network, path finding and fare modules."""

from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


class RouteError(Exception):
    """Base class for route query errors."""


class InvalidQueryError(RouteError):
    """Raised for logically invalid queries (e.g. identical start and end)."""


class StationNotFoundError(RouteError):
    """Raised when a station does not exist in the network."""


class NoRouteError(RouteError):
    """Raised when no path satisfies the query constraints."""


@dataclass
class Route:
    """Result of a route query.

    Attributes:
        start, end: official station names.
        mode: path query mode used.
        max_transfers: transfer limit used (None = unlimited).
        preferred_lines: preferred lines used.
        station_names: ordered station names from start to end.
        segments: list of {"line": str, "stations": [str, ...]} line segments.
        transfer_details: list of {"station", "from_line", "to_line", "all_lines"}.
        lines: ordered list of distinct lines used.
        distance_m: total travelled distance in meters.
        transfer_count: number of transfers.
        fare: fare in yuan.
        estimated_time_min: estimated travel time in minutes.
    """

    start: str
    end: str
    mode: str
    max_transfers: Optional[int]
    preferred_lines: List[str]
    station_names: List[str]
    segments: List[Dict[str, Any]] = field(default_factory=list)
    transfer_details: List[Dict[str, Any]] = field(default_factory=list)
    lines: List[str] = field(default_factory=list)
    distance_m: int = 0
    transfer_count: int = 0
    fare: int = 0
    estimated_time_min: float = 0.0


def _build_segments(names, edges):
    """Group the path into consecutive same-line segments."""
    if not edges:
        return []
    segments = []
    current_line = edges[0].line
    current_stations = [names[0]]
    for index, edge in enumerate(edges):
        if edge.line != current_line:
            segments.append({"line": current_line, "stations": current_stations})
            current_line = edge.line
            current_stations = [names[index]]
        current_stations.append(names[index + 1])
    segments.append({"line": current_line, "stations": current_stations})
    return segments


def _build_transfers(names, edges, network):
    """Describe every station where the path changes line."""
    details = []
    for index in range(1, len(edges)):
        previous_line = edges[index - 1].line
        next_line = edges[index].line
        if previous_line != next_line:
            station = names[index]
            details.append(
                {
                    "station": station,
                    "from_line": previous_line,
                    "to_line": next_line,
                    "all_lines": network.graph.get_station_lines(station),
                }
            )
    return details


def compute_route(network, start, end, config=None) -> Route:
    """Compute the route between two stations and its summary information.

    Args:
        network: a SubwayNetwork instance.
        start: starting station name.
        end: ending station name.
        config: PathfindingConfig (defaults to shortest-distance mode).

    Raises:
        StationNotFoundError, InvalidQueryError, NoRouteError
    """
    config = config if config is not None else PathfindingConfig()

    start_name = network.resolve_station(start)
    if start_name is None:
        raise StationNotFoundError(
            f"Starting station '{start}' does not exist, please check the station name"
        )
    end_name = network.resolve_station(end)
    if end_name is None:
        raise StationNotFoundError(
            f"Ending station '{end}' does not exist, please check the station name"
        )
    if start_name == end_name:
        raise InvalidQueryError("Starting station cannot be the same as the ending station")

    source = network.station_number(start_name)
    target = network.station_number(end_name)
    shortest = ShortestPath(network.graph, source, config)
    if not shortest.has_path_to(target):
        raise NoRouteError(
            f"No route found from '{start_name}' to '{end_name}' with the current settings "
            f"(mode={config.mode}, max_transfers={config.max_transfers})"
        )

    numbers, edges = shortest.path_with_edges(target)
    names = [network.station_name(number) for number in numbers]
    distance = sum(edge.weight for edge in edges)
    transfers = shortest.get_transfer_count(target)

    segments = _build_segments(names, edges)
    transfer_details = _build_transfers(names, edges, network)
    lines = []
    for segment in segments:
        if segment["line"] not in lines:
            lines.append(segment["line"])

    fare = Sum_money(distance)
    estimated = calculate_estimated_time(distance, transfers)

    return Route(
        start=start_name,
        end=end_name,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        station_names=names,
        segments=segments,
        transfer_details=transfer_details,
        lines=lines,
        distance_m=distance,
        transfer_count=transfers,
        fare=fare,
        estimated_time_min=estimated,
    )

import json


MODE_LABELS = {
    "shortest_distance": "Shortest distance",
    "least_transfer": "Minimum transfers",
    "comprehensive": "Comprehensive optimal",
}


def format_distance(distance_m) -> str:
    """Format a distance in meters and kilometers.

    Short distances (< 1 km) are shown primarily in meters, longer distances
    primarily in kilometers. Both units are always shown.
    """
    if distance_m < 1000:
        return f"{distance_m:.0f} m ({distance_m / 1000:.2f} km)"
    return f"{distance_m / 1000:.2f} km ({distance_m:.0f} m)"


def build_summary(route) -> dict:
    """Extract key figures from a route: distance, fare, transfers and time."""
    return {
        "total_distance_m": route.distance_m,
        "total_distance_km": round(route.distance_m / 1000.0, 2),
        "total_fare": route.fare,
        "transfer_count": route.transfer_count,
        "estimated_time_min": round(route.estimated_time_min, 1),
    }


def format_text(route) -> str:
    """Render a route as a console-formatted text report."""
    out = []
    rule = "=" * 64
    out.append(rule)
    out.append(f"Route: {route.start} -> {route.end}")
    out.append(f"Query mode: {MODE_LABELS.get(route.mode, route.mode)}")
    if route.max_transfers is not None:
        out.append(f"Max transfers: {route.max_transfers}")
    if route.preferred_lines:
        out.append(f"Line preference: {', '.join(route.preferred_lines)}")
    out.append("-" * 64)

    out.append(f"Station sequence ({len(route.station_names)} stations):")
    for index, name in enumerate(route.station_names, start=1):
        out.append(f"  {index:>3}. {name}")
    out.append("")

    out.append("Line segments:")
    for segment in route.segments:
        stations = segment["stations"]
        stops = len(stations) - 1
        out.append(
            f"  [{segment['line']}] {stations[0]} -> {stations[-1]} ({stops} stops)"
        )
    out.append("")

    out.append("Transfer stations:")
    if route.transfer_details:
        for transfer in route.transfer_details:
            all_lines = ", ".join(transfer["all_lines"])
            out.append(
                f"  {transfer['station']}: {transfer['from_line']} -> {transfer['to_line']}"
                f"  (interchange lines: {all_lines})"
            )
    else:
        out.append("  None (no transfer required)")
    out.append("")

    out.append(f"Lines involved: {', '.join(route.lines)}")
    out.append("-" * 64)
    out.append(f"Total distance: {format_distance(route.distance_m)}")
    out.append(f"Transfers: {route.transfer_count}")
    out.append(f"Estimated time: {route.estimated_time_min:.1f} min")
    out.append(f"Fare: {route.fare} yuan")
    out.append(rule)
    return "\n".join(out)


def to_json_dict(route) -> dict:
    """Build the structured (JSON-serialisable) representation of a route."""
    summary = build_summary(route)
    return {
        "query": {
            "start": route.start,
            "end": route.end,
            "mode": route.mode,
            "max_transfers": route.max_transfers,
            "line_preference": route.preferred_lines,
        },
        "path": route.station_names,
        "segments": [
            {"line": segment["line"], "stations": segment["stations"]}
            for segment in route.segments
        ],
        "distance": {
            "meters": route.distance_m,
            "kilometers": summary["total_distance_km"],
        },
        "fare": route.fare,
        "estimated_time_min": summary["estimated_time_min"],
        "transfer_count": route.transfer_count,
        "transfer_stations": route.transfer_details,
        "lines": route.lines,
        "summary": summary,
    }


def to_json(route) -> str:
    """Render a route as a JSON string."""
    return json.dumps(to_json_dict(route), ensure_ascii=False, indent=2)


def render(route, fmt="text") -> str:
    """Render a route in the requested format ('text' or 'json')."""
    if fmt == "json":
        return to_json(route)
    if fmt == "text":
        return format_text(route)
    raise ValueError(f"Unsupported output format '{fmt}'")

"""Data integrity checks for the subway network.

Checks performed:
    * station coverage (main lines and representative stations are present)
    * connection integrity (valid endpoints, no duplicates, symmetric adjacency,
      no isolated stations, network connectivity)
    * distance accuracy (positive values within plausible ranges)
"""

from collections import deque


REQUIRED_LINES = [
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
]

REPRESENTATIVE_STATIONS = {
    "Line 1": ["Pingguoyuan", "Sihui East"],
    "Line 2": ["Xizhimen", "Chegongzhuang"],
    "Line 4": ["Anheqiao North"],
    "Line 5": ["Songjiazhuang"],
    "Line 6": ["Hujialou"],
    "Line 10": ["Bagou"],
    "Line 13": ["Xi'erqi"],
}

MIN_PLAUSIBLE_DISTANCE_M = 100
MAX_PLAUSIBLE_DISTANCE_M = 5000


class IntegrityReport:
    """Collects errors, warnings and informational messages from the checks."""

    def __init__(self):
        self.errors = []
        self.warnings = []
        self.info = []

    @property
    def ok(self):
        """True when no errors were found (warnings do not fail the check)."""
        return not self.errors

    def render(self):
        """Return a human readable report."""
        lines = ["Data integrity check", "=" * 40]
        lines.extend(f"[INFO]    {message}" for message in self.info)
        lines.extend(f"[WARNING] {message}" for message in self.warnings)
        lines.extend(f"[ERROR]   {message}" for message in self.errors)
        lines.append("=" * 40)
        status = "PASSED" if self.ok else "FAILED"
        lines.append(
            f"Result: {status} ({len(self.errors)} errors, {len(self.warnings)} warnings)"
        )
        return "\n".join(lines)


def check_station_coverage(network, report):
    """Verify that main lines and representative stations are present."""
    stations = network.station_names()
    report.info.append(f"Stations loaded: {len(stations)}")
    if not stations:
        report.errors.append("No stations are defined in the network")
        return

    available = set(network.lines())
    for line in REQUIRED_LINES:
        if line not in available:
            report.warnings.append(f"Main line '{line}' is not present in the data")

    for line, names in REPRESENTATIVE_STATIONS.items():
        for name in names:
            if not network.has_station(name):
                report.warnings.append(
                    f"Representative station '{name}' of {line} is not in the data"
                )
            elif line not in network.graph.get_station_lines(name):
                report.warnings.append(
                    f"Station '{name}' is not recorded as belonging to {line}"
                )

    for name in stations:
        if not network.graph.get_station_lines(name):
            report.warnings.append(f"Station '{name}' has no line information")

    for name in network.undeclared_stations:
        report.warnings.append(f"Station '{name}' is used by connections but not declared")


def check_connections(network, report):
    """Verify endpoints, duplicates, adjacency symmetry and connectivity."""
    records = network.connection_records()
    report.info.append(f"Connections loaded: {len(records)}")

    seen = set()
    for record in records:
        source, dest, line = record["source"], record["dest"], record["line"]
        if source == dest:
            report.errors.append(f"Connection on {line} links '{source}' to itself")
        for name in (source, dest):
            if not network.has_station(name):
                report.errors.append(f"Connection on {line} references unknown station '{name}'")
        key = (frozenset((source, dest)), line)
        if key in seen:
            report.errors.append(
                f"Duplicate connection between '{source}' and '{dest}' on {line}"
            )
        seen.add(key)

    graph = network.graph
    for vertex, incident in graph.adj.items():
        for edge in incident:
            other = edge.other(vertex)
            if not any(candidate is edge for candidate in graph.adj.get(other, [])):
                report.errors.append(
                    f"Adjacency is not bidirectional for station number {vertex}"
                )

    stations = network.station_names()
    isolated = [name for name in stations if not graph.neighbors(network.station_number(name))]
    for name in isolated:
        report.warnings.append(f"Station '{name}' has no connections")

    if stations:
        start = network.station_number(stations[0])
        visited = {start}
        queue = deque([start])
        while queue:
            current = queue.popleft()
            for edge in graph.neighbors(current):
                nxt = edge.other(current)
                if nxt not in visited:
                    visited.add(nxt)
                    queue.append(nxt)
        unreachable = [name for name in stations if network.station_number(name) not in visited]
        if unreachable:
            report.warnings.append(
                f"{len(unreachable)} station(s) are not connected to the main network: "
                + ", ".join(unreachable[:10])
            )


def check_distances(network, report):
    """Verify that every connection distance is positive and plausible."""
    for record in network.connection_records():
        distance = record["distance"]
        label = f"{record['source']} - {record['dest']} ({record['line']})"
        if not isinstance(distance, (int, float)) or distance <= 0:
            report.errors.append(f"Invalid distance {distance!r} for {label}")
        elif distance < MIN_PLAUSIBLE_DISTANCE_M or distance > MAX_PLAUSIBLE_DISTANCE_M:
            report.warnings.append(f"Unusual distance {distance} m for {label}")


def run_integrity_check(network) -> IntegrityReport:
    """Run all integrity checks and return the collected report."""
    report = IntegrityReport()
    check_station_coverage(network, report)
    check_connections(network, report)
    check_distances(network, report)
    return report

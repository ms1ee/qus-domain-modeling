import os

from lottery_system.default_data import DEFAULT_USER_LINES
from lottery_system.models import User


def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
    """Parse "Name,Department Group" lines into users and format error messages."""
    users: list[User] = []
    errors: list[str] = []
    for line_no, raw in enumerate(lines, start=1):
        stripped = raw.strip()
        if not stripped:
            continue
        parts = stripped.split(",")
        if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
            errors.append(
                f"Format Error at Line {line_no}: '{stripped}' "
                f"(expected 'Name,Department Group')"
            )
            continue
        users.append(User(name=parts[0].strip(), department=parts[1].strip()))
    return users, errors


class ListManager:
    """Loads, validates and summarizes participant lists."""

    def __init__(self):
        pass

    def load_default_list(self) -> tuple[list[User], list[str]]:
        return _parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return _parse_lines(lines)

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Return (valid users with first occurrence kept, duplicate entries removed)."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict:
        """Return a mapping of department group -> number of people (insertion order)."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.default_data import DEFAULT_PRIZE_SPECS
from lottery_system.models import Prize


class PrizeManager:
    """Manages the prize library and the lottery round sequence."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, avoid, drawer in DEFAULT_PRIZE_SPECS:
            try:
                self.add_prize(name, quantity, avoid, drawer)
            except ValueError as exc:
                errors.append(f"Default prize '{name}': {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if any(p.name == clean_name for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        unique_groups = list(dict.fromkeys(groups))
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=unique_groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes; `order` contains 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            if len(order) != count or sorted(order) != list(range(1, count + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[index - 1] for index in order]
        return True

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    sequence: int
    timestamp: str


_ORDINAL_RANKS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}

_UNRANKED = 10_000


def prize_level_rank(prize_name: str) -> int:
    """Lower rank means higher prize level. Unranked prizes come last."""
    match = re.search(
        r"\b(first|second|third|fourth|fifth|sixth|seventh|eighth|ninth|tenth)\b",
        prize_name,
        re.IGNORECASE,
    )
    if match:
        return _ORDINAL_RANKS[match.group(1).lower()]
    match = re.search(r"\b(\d+)(st|nd|rd|th)\b", prize_name, re.IGNORECASE)
    if match:
        return int(match.group(1))
    return _UNRANKED


class ResultManager:
    """Stores winners and produces ordered results and exports."""

    def __init__(self):
        self._winners: list[Winner] = []
        self._sequence = 0

    def add_winner(self, user, prize) -> Winner:
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        self._sequence += 1
        winner = Winner(
            user=user,
            prize_name=prize_name,
            sequence=self._sequence,
            timestamp=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            name = prize.name if isinstance(prize, Prize) else str(prize)
            result.setdefault(name, [])
        for winner in self._winners:
            result.setdefault(winner.prize_name, []).append(winner)
        return result

    def get_sorted_winners(self, prizes) -> list[Winner]:
        """Winners sorted by prize level (high to low), then by winning order."""
        names = [p.name if isinstance(p, Prize) else str(p) for p in prizes]
        index_map = {name: idx for idx, name in enumerate(names)}

        def key(winner: Winner):
            return (
                prize_level_rank(winner.prize_name),
                index_map.get(winner.prize_name, len(names)),
                winner.sequence,
            )

        return sorted(self._winners, key=key)

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in self.get_sorted_winners(prizes):
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.timestamp,
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryManager:
    """Runs lottery rounds and applies the lottery rule engine."""

    CHENG_RU_GROUP = "Cheng Ru Group"
    NO_AVOIDANCE_DRAWER = "Li Bin"
    FALLBACK_DRAWER = "Ma Jiaqi"
    HISTORY_DRAWERS = ("Chen Qixian", "Jin Duo")

    def __init__(self, users):
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Each entry: whether that lottery person's round produced a Cheng Ru Group winner.
        self.last_two_draw_results: list[bool] = []
        self._winner_names: set[str] = set()
        self._rng = random.SystemRandom()

    def has_won(self, user: User) -> bool:
        return user.name in self._winner_names

    def _is_fallback_active(self, prize: Prize) -> bool:
        return (
            prize.associated_drawer == self.FALLBACK_DRAWER
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        candidates = [u for u in self.users if u.name not in self._winner_names]
        if self._is_fallback_active(prize):
            # Fallback protection overrides avoidance groups.
            return [u for u in candidates if u.department == self.CHENG_RU_GROUP]
        if prize.associated_drawer == self.NO_AVOIDANCE_DRAWER:
            return candidates
        avoid = set(prize.avoid_groups)
        return [u for u in candidates if u.department not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts: list[str] = []
        if prize.associated_drawer == self.NO_AVOIDANCE_DRAWER:
            parts.append("previous winners are excluded; avoidance groups are not applied")
        elif self._is_fallback_active(prize):
            parts.append("Cheng Ru Group protection active: only Cheng Ru Group members are eligible")
        else:
            parts.append("previous winners are excluded")
            if prize.avoid_groups:
                parts.append("avoidance groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer:
            parts.append(f"lottery person: {prize.associated_drawer}")
        return "; ".join(parts)

    def draw_one(self, prize: Prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winner_names.add(user.name)
        self.result_manager.add_winner(user, prize)
        return user

    def record_round_completion(self, prize: Prize, round_winners: list[User]) -> None:
        if prize.associated_drawer in self.HISTORY_DRAWERS:
            got_cheng_ru = any(u.department == self.CHENG_RU_GROUP for u in round_winners)
            self.last_two_draw_results = (self.last_two_draw_results + [got_cheng_ru])[-2:]
        elif prize.associated_drawer == self.FALLBACK_DRAWER and self._is_fallback_active(prize):
            # The protection has been applied; start a fresh history.
            self.last_two_draw_results = []

    def _wait_while_paused(self, input_func, output_func) -> str:
        output_func("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = input_func("(paused) > ").strip().lower()
            if command in ("continue", "terminate"):
                return command
            output_func("Unknown command. Use 'continue' or 'terminate'.")

    def run_round(self, prize: Prize, input_func=input, output_func=print) -> list[User]:
        output_func("")
        output_func(
            f"=== Current round: {prize.name} | Quantity: {prize.quantity} | "
            f"Lottery person: {prize.associated_drawer or 'None'}"
        )
        output_func(f"Rule: {self.describe_rules(prize)}")

        round_winners: list[User] = []
        terminated = False
        while len(round_winners) < prize.quantity:
            if not self._get_eligible_pool(prize):
                output_func("No eligible candidates remain; this round ends early.")
                break

            command = input_func(
                f"[Enter] draw the {_ordinal(len(round_winners) + 1)} winner | "
                f"'pause' | 'terminate' > "
            ).strip().lower()

            if command == "pause":
                command = self._wait_while_paused(input_func, output_func)
                if command == "continue":
                    command = ""

            if command == "terminate":
                terminated = True
                output_func("Round terminated by operator; skipping remaining draws.")
                break
            if command not in ("", "continue"):
                output_func(f"Unknown command '{command}'.")
                continue

            position = len(round_winners) + 1
            output_func(f"Currently drawing the {_ordinal(position)} winner...")
            user = self.draw_one(prize)
            if user is None:
                output_func("No eligible candidates remain; this round ends early.")
                break
            round_winners.append(user)
            output_func(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        self.record_round_completion(prize, round_winners)

        output_func(f"Round finished: {len(round_winners)}/{prize.quantity} winner(s) for {prize.name}.")
        if round_winners:
            output_func("Round results:")
            for idx, user in enumerate(round_winners, start=1):
                output_func(f"  {idx}. {user.name} ({user.department})")
        if terminated:
            output_func("This round was terminated early.")
        return round_winners

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager


class LotterySystemCLI:
    """Command-line interface for the team building lottery."""

    def __init__(self, input_func=input, output_func=print):
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery_manager = None

    def ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    def out(self, message: str = "") -> None:
        self._print(message)

    def run(self) -> None:
        self.out("=== Company Team Building Smart Lottery System ===")
        while True:
            self.out("\n--- Main Menu ---")
            self.out("1. Load participant list")
            self.out("2. View department groups")
            self.out("3. Prize configuration")
            self.out("4. Start lottery")
            self.out("5. Results (view / export)")
            self.out("0. Exit")
            choice = self.ask("Select an option: ")
            if choice == "1":
                self.load_list()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.prize_menu()
            elif choice == "4":
                self.start_lottery()
            elif choice == "5":
                self.results_menu()
            elif choice == "0":
                self.out("Goodbye!")
                return
            else:
                self.out("Invalid option, please try again.")

    # ----- participant list -----

    def load_list(self) -> None:
        self.out("Select list source:")
        self.out("1. Built-in default team building list")
        self.out("2. Upload custom txt list (format per line: Name,Department Group)")
        self.out("0. Cancel")
        source = self.ask("Option: ")
        if source == "1":
            users, errors = self.list_manager.load_default_list()
        elif source == "2":
            path = self.ask("Path to txt file: ")
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

        if errors:
            self.out(f"Found {len(errors)} invalid entr{'y' if len(errors) == 1 else 'ies'}:")
            for error in errors:
                self.out(f"  {error}")
            answer = self.ask("Enter 'r' to remove invalid data and continue, 'q' to terminate upload: ").lower()
            if answer != "r":
                self.out("Upload terminated. Current list unchanged.")
                return

        valid_users, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self.out(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                self.out(f"  {user.name} ({user.department})")
            answer = self.ask(
                "Enter 'r' to remove duplicates (keep first occurrence), 'q' to terminate upload: "
            ).lower()
            if answer != "r":
                self.out("Upload terminated. Current list unchanged.")
                return

        if not valid_users:
            self.out("No valid participants found. Current list unchanged.")
            return

        self.users = valid_users
        self.out(f"Loaded {len(self.users)} participants.")

    def show_groups(self) -> None:
        if not self.users:
            self.out("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self.out("Department groups:")
        for department, count in info.items():
            self.out(f"  {department}: {count} people")

    # ----- prizes -----

    def prize_menu(self) -> None:
        while True:
            self.out("\n--- Prize Configuration ---")
            self.out("1. Load default prize library")
            self.out("2. Add custom prize")
            self.out("3. Show prizes")
            self.out("4. Reorder lottery sequence")
            self.out("0. Back")
            choice = self.ask("Option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        self.out(f"  {error}")
                else:
                    self.out("Default prize library loaded.")
            elif choice == "2":
                self.add_custom_prize()
            elif choice == "3":
                self.show_prizes()
            elif choice == "4":
                self.reorder_prizes()
            elif choice == "0":
                return
            else:
                self.out("Invalid option.")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("No prizes configured.")
            return
        self.out("Lottery sequence:")
        for idx, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
            self.out(
                f"  {idx}. {prize.name} | Quantity: {prize.quantity} | "
                f"Lottery person: {prize.associated_drawer or 'None'} | Avoid groups: {avoid}"
            )

    def add_custom_prize(self) -> None:
        name = self.ask("Prize name: ")
        quantity_text = self.ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            self.out("Error: quantity must be an integer.")
            return
        if self.users:
            known = ", ".join(self.list_manager.get_group_info(self.users).keys())
            self.out(f"Known groups: {known}")
        avoid_text = self.ask("Avoidance groups (comma-separated, blank for none): ")
        avoid = [g.strip() for g in avoid_text.split(",") if g.strip()]
        drawer = self.ask("Associated lottery person (blank for none): ") or None
        try:
            self.prize_manager.add_prize(name, quantity, avoid, drawer)
        except ValueError as exc:
            self.out(f"Error: {exc}")
            return
        self.out(f"Prize '{name.strip()}' added.")

    def reorder_prizes(self) -> None:
        if not self.prize_manager.get_prizes():
            self.out("No prizes configured.")
            return
        self.show_prizes()
        text = self.ask("Enter new order as serial numbers (e.g. 2 1 3), blank to cancel: ")
        if not text:
            return
        try:
            order = [int(token) for token in text.replace(",", " ").split()]
        except ValueError:
            self.out("Error: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.out("Lottery sequence updated.")
        else:
            self.out("Error: invalid order. Each position must be used exactly once.")

    # ----- lottery -----

    def start_lottery(self) -> None:
        if not self.users:
            self.out("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("Please configure prizes first.")
            return
        if self.lottery_manager and self.lottery_manager.result_manager.get_winners():
            answer = self.ask("A previous session has results. Starting a new session clears them. Continue? (y/n): ")
            if answer.lower() != "y":
                return

        self.lottery_manager = LotteryManager(self.users)
        self.out(
            f"Lottery started with {len(self.users)} participants and {len(prizes)} round(s)."
        )
        self.out("Commands during a draw: [Enter] draw, 'pause', 'terminate'.")
        for prize in prizes:
            self.lottery_manager.run_round(prize, self._input, self._print)

        self.out("\n=== Lottery finished: final winner list ===")
        self.show_winners(prizes)

    def show_winners(self, prizes) -> None:
        if not self.lottery_manager:
            self.out("No lottery results yet.")
            return
        winners = self.lottery_manager.result_manager.get_sorted_winners(prizes)
        if not winners:
            self.out("No winners recorded.")
            return
        for winner in winners:
            self.out(
                f"{winner.prize_name}: {winner.user.name} ({winner.user.department}) "
                f"[{winner.timestamp}]"
            )

    def results_menu(self) -> None:
        if not self.lottery_manager:
            self.out("No lottery results yet.")
            return
        while True:
            self.out("\n--- Results ---")
            self.out("1. Show winner list")
            self.out("2. Export winner list to txt (CSV format)")
            self.out("0. Back")
            choice = self.ask("Option: ")
            prizes = self.prize_manager.get_prizes()
            if choice == "1":
                self.show_winners(prizes)
            elif choice == "2":
                _, message = self.lottery_manager.result_manager.export_to_csv(prizes)
                self.out(message)
            elif choice == "0":
                return
            else:
                self.out("Invalid option.")


def main() -> None:
    cli = LotterySystemCLI()
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting.")

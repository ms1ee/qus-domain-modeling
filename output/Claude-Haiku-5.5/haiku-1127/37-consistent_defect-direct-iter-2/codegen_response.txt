from __future__ import annotations

import os
import re

from lottery_system.models import User


DEFAULT_LIST_TEXT = """\
Zhang San,Technology Department
Li Si,Product Department
Wang Wu,Technology Department
Zhao Liu,Marketing Department
Li Bin,Finance Department
Chen Qixian,Cheng Ru Group
Jin Duo,Product Department
Ma Jiaqi,Cheng Ru Group
Sun Qi,Cheng Ru Group
Zhou Ba,Marketing Department
Wu Jiu,Technology Department
Zheng Shi,Finance Department
"""

_SEPARATOR_RE = re.compile(r"[,;]")
_ENCODINGS = ("utf-8-sig", "gbk")


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list ("Name,Department Group" per line)."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it into users and error messages."""
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        content = None
        last_error = None
        for encoding in _ENCODINGS:
            try:
                with open(path, "r", encoding=encoding) as handle:
                    content = handle.read()
                break
            except UnicodeDecodeError as exc:
                last_error = exc
            except OSError as exc:
                return [], [f"File Error: cannot read '{path}': {exc}"]

        if content is None:
            return [], [f"File Error: cannot decode '{path}': {last_error}"]

        return self._parse_lines(content.splitlines())

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse raw lines into User objects, collecting format errors."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SEPARATOR_RE.split(line, maxsplit=1)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected format: Name,Department Group)"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Remove duplicate names; keep the first occurrence of each name.

        Returns (valid_users, duplicates) where duplicates are the later
        entries that were rejected.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management: presets, custom prizes, avoidance rules and ordering."""
from __future__ import annotations

import re

from lottery_system.models import Prize


DEFAULT_PRIZES: list[tuple[str, int]] = [
    ("Third Prize - Movie Tickets", 5),
    ("Second Prize - Shopping Cards", 3),
    ("First Prize - 1 Day Annual Leave", 1),
]

_SPLIT_RE = re.compile(r"[,;]")


def _clean_list(values) -> list[str]:
    """Strip, drop empties and remove duplicates while keeping order."""
    cleaned: list[str] = []
    for value in values or []:
        text = str(value).strip()
        if text and text not in cleaned:
            cleaned.append(text)
    return cleaned


class PrizeManager:
    """Manages the ordered prize library."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a new prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=_clean_list(avoid_groups),
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def set_avoid_groups(self, index: int, groups: list[str]) -> bool:
        """Set avoidance groups of the prize at 0-based index."""
        if not 0 <= index < len(self._prizes):
            return False
        self._prizes[index].avoid_groups = _clean_list(groups)
        return True

    def set_associated_drawer(self, index: int, drawer: str | None) -> bool:
        """Set the lottery person of the prize at 0-based index."""
        if not 0 <= index < len(self._prizes):
            return False
        self._prizes[index].associated_drawer = (drawer or "").strip() or None
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if not isinstance(order, list) or len(order) != count:
            return False
        if any(isinstance(p, bool) or not isinstance(p, int) for p in order):
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

    @staticmethod
    def parse_group_text(text: str) -> list[str]:
        """Parse a comma/semicolon separated text into a clean list."""
        return _clean_list(_SPLIT_RE.split(text or ""))

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""
from __future__ import annotations

import random

from lottery_system.models import Prize, User


LI_BIN = "Li Bin"
MA_JIAQI = "Ma Jiaqi"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
CHENG_RU_GROUP = "Cheng Ru Group"


def _norm(text: str | None) -> str:
    """Normalize text for case- and whitespace-insensitive comparison."""
    return " ".join((text or "").split()).casefold()


def is_cheng_ru_member(user: User) -> bool:
    """Return True if the user belongs to the Cheng Ru group."""
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


def describe_rules(prize: Prize) -> str:
    """Build a human-readable description of the rules of a prize round."""
    parts = ["Previous winners are excluded."]
    drawer = _norm(prize.associated_drawer)
    if prize.avoid_groups:
        parts.append(
            "Avoidance groups (drawn after other members): "
            + ", ".join(prize.avoid_groups)
            + "."
        )
    else:
        parts.append("No avoidance groups.")
    if prize.associated_drawer:
        parts.append(f"Lottery person: {prize.associated_drawer}.")
    if drawer == _norm(LI_BIN):
        parts.append("Special rule: avoidance groups are NOT applied for this lottery person.")
    if drawer == _norm(MA_JIAQI):
        parts.append(
            "Cheng Ru Group protection: if the last two lottery persons "
            "had no Cheng Ru Group winner, only Cheng Ru Group members are eligible."
        )
    return " ".join(parts)


class LotteryManager:
    """Performs draws while enforcing all lottery rules."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winners: list[User] = []
        self._winner_names: set[str] = set()
        self._rng = rng or random.Random()
        # Results of the most recent lottery persons' rounds:
        # True means the round contained a Cheng Ru Group winner.
        self.last_two_draw_results: list[bool] = []

    @property
    def winners(self) -> list[User]:
        """All winners drawn so far, in winning order."""
        return list(self._winners)

    def is_winner(self, user: User) -> bool:
        """Return True if the user has already won any prize."""
        return user.name.strip() in self._winner_names

    def is_cheng_ru_protection_active(self, prize: Prize) -> bool:
        """Return True if the Cheng Ru group fallback restricts this prize's pool."""
        return (
            _norm(prize.associated_drawer) == _norm(MA_JIAQI)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users that may be drawn for the given prize right now."""
        available = [user for user in self._users if not self.is_winner(user)]
        drawer = _norm(prize.associated_drawer)

        # Special rule: Li Bin only filters previous winners.
        if drawer == _norm(LI_BIN):
            return available

        # Fallback: force the pool to Cheng Ru Group members, ignoring avoidance groups.
        if self.is_cheng_ru_protection_active(prize):
            cheng_ru_pool = [user for user in available if is_cheng_ru_member(user)]
            if cheng_ru_pool:
                return cheng_ru_pool

        avoid = {_norm(group) for group in prize.avoid_groups}
        primary = [user for user in available if _norm(user.department) not in avoid]
        if primary:
            return primary

        # Avoidance-group members remain eligible but are drawn only after everyone else.
        return [user for user in available if _norm(user.department) in avoid]

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winners.append(winner)
        self._winner_names.add(winner.name.strip())
        return winner

    def finish_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Record the outcome of a completed round for the fallback mechanism."""
        if not prize.associated_drawer:
            return
        had_cheng_ru = any(is_cheng_ru_member(user) for user in round_winners)
        self.last_two_draw_results.append(had_cheng_ru)
        self.last_two_draw_results = self.last_two_draw_results[-2:]

"""Winner recording, grouping by prize level, and CSV export."""
from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User


_LEVEL_WORDS = ("first", "second", "third")
_TIME_FORMAT = "%Y-%m-%d %H:%M:%S"


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    sequence: int = 0
    won_at: datetime = field(default_factory=datetime.now)


def _prize_rank(name: str) -> int:
    """Return the level rank of a prize (1 = highest). Unknown levels rank last."""
    lowered = name.casefold()
    for index, word in enumerate(_LEVEL_WORDS, start=1):
        if re.search(rf"\b{word}\b", lowered):
            return index
    return len(_LEVEL_WORDS) + 1


class ResultManager:
    """Stores winners and produces ordered summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for the given prize."""
        winner = Winner(
            user=user,
            prize_name=prize.name,
            sequence=len(self._winners) + 1,
            won_at=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered from highest prize level to lowest.

        Within the same prize, winners are in winning time order.
        """
        ordered = sorted(
            enumerate(prizes),
            key=lambda item: (_prize_rank(item[1].name), item[0]),
        )
        result: dict[str, list[Winner]] = {prize.name: [] for _, prize in ordered}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt."""
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winners in grouped.values():
                    for winner in winners:
                        writer.writerow([
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.won_at.strftime(_TIME_FORMAT),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def _ordinal(number: int) -> str:
    """Return an English ordinal such as 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery_manager: LotteryManager | None = None

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _ask_index(self, prompt: str, count: int) -> int | None:
        """Ask for a 1-based number and return a 0-based index, or None if invalid."""
        text = self._ask(prompt)
        if not text.isdigit() or not 1 <= int(text) <= count:
            print("Invalid selection.")
            return None
        return int(text) - 1

    def _print_groups(self) -> None:
        if not self.users:
            print("No participants loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print(f"Valid department groups ({len(info)}):")
        for department, count in info.items():
            print(f"  - {department}: {count} people")

    # ------------------------------------------------------------ list loading
    def _setup_user_list(self) -> bool:
        while True:
            print("\n========== Participant List ==========")
            print("1. Load built-in team building list")
            print("2. Upload custom txt list (Name,Department Group per line)")
            print("0. Exit")
            choice = self._ask("Select an option: ")
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
                for error in errors:
                    print(error)
                if not users:
                    print("Built-in list is empty.")
                    continue
                self.users = users
                print(f"Loaded {len(users)} participants from the built-in list.")
                self._print_groups()
                return True
            if choice == "2":
                if self._load_custom_list():
                    return True
            elif choice == "0":
                return False
            else:
                print("Invalid option.")

    def _load_custom_list(self) -> bool:
        path = self._ask("Enter the path of the custom txt file: ").strip('"')
        if not path:
            print("No path entered.")
            return False

        users, errors = self.list_manager.load_custom_list(path)
        for error in errors:
            print(error)
        if not users and errors:
            print("Upload failed: no readable data.")
            return False

        valid, duplicates = self.list_manager.validate_users(users)
        for dup in duplicates:
            print(f"Duplicate Name: '{dup.name}' ({dup.department}) appears more than once and is removed.")

        if errors or duplicates:
            print(f"Found {len(errors)} format error(s) and {len(duplicates)} duplicate(s).")
            while True:
                choice = self._ask(
                    "Type 'r' to remove invalid data and continue, or 't' to terminate the upload: "
                ).lower()
                if choice == "t":
                    print("Upload terminated.")
                    return False
                if choice == "r":
                    break
                print("Invalid choice.")

        if not valid:
            print("No valid participants found.")
            return False

        self.users = valid
        print(f"Loaded {len(valid)} participants from '{path}'.")
        self._print_groups()
        return True

    # ---------------------------------------------------------- prize config
    def _print_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            print(f"{index}. {prize.name} | Quantity: {prize.quantity}")
            print(f"   Rules: {describe_rules(prize)}")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity_text = self._ask("Prize quantity: ")
        if not quantity_text.isdigit():
            print("Quantity must be a positive integer.")
            return
        groups_text = self._ask("Avoidance groups (comma separated, empty for none): ")
        groups = PrizeManager.parse_group_text(groups_text)
        drawer = self._choose_drawer()
        if drawer is False:
            return
        try:
            prize = self.prize_manager.add_prize(name, int(quantity_text), groups, drawer)
            print(f"Prize added: {prize.name}")
        except ValueError as exc:
            print(f"Error: {exc}")

    def _choose_drawer(self) -> str | None | bool:
        """Return a valid drawer name, None for no drawer, or False on invalid input."""
        names = []
        for user in self.users:
            if user.name not in names:
                names.append(user.name)
        if names:
            print("Available lottery persons: " + ", ".join(names))
        text = self._ask("Associated lottery person (empty for none): ")
        if not text:
            return None
        if text not in names:
            print(f"'{text}' is not a participant in the loaded list.")
            return False
        return text

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes()
        index = self._ask_index("Select prize number: ", len(prizes))
        if index is None:
            return
        info = self.list_manager.get_group_info(self.users) if self.users else {}
        if info:
            print("Available groups: " + ", ".join(info.keys()))
        groups = PrizeManager.parse_group_text(
            self._ask("Avoidance groups (comma separated, empty to clear): ")
        )
        if self.prize_manager.set_avoid_groups(index, groups):
            print("Avoidance groups updated.")

    def _set_drawer(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes()
        index = self._ask_index("Select prize number: ", len(prizes))
        if index is None:
            return
        drawer = self._choose_drawer()
        if drawer is False:
            return
        if self.prize_manager.set_associated_drawer(index, drawer):
            print("Lottery person updated.")

    def _reorder(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes()
        text = self._ask(
            "Enter new order as serial numbers separated by spaces (e.g. 3 1 2): "
        )
        try:
            order = [int(token) for token in text.replace(",", " ").split()]
        except ValueError:
            print("Order must contain only numbers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
            self._print_prizes()
        else:
            print("Invalid order: each serial number must appear exactly once.")

    def _run_prize_menu(self) -> bool:
        """Run the prize configuration menu. Returns True if the lottery was run."""
        while True:
            print(PRIZE_MENU)
            choice = self._ask("Select an option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                for error in errors:
                    print(f"Error: {error}")
                if not errors:
                    print("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._set_avoid_groups()
            elif choice == "4":
                self._set_drawer()
            elif choice == "5":
                self._print_prizes()
            elif choice == "6":
                self._reorder()
            elif choice == "7":
                self._print_groups()
            elif choice == "8":
                if not self.prize_manager.get_prizes():
                    print("Please configure at least one prize first.")
                    continue
                self._run_lottery()
                return True
            elif choice == "0":
                return False
            else:
                print("Invalid option.")

    # ------------------------------------------------------------- lottery
    def _run_lottery(self) -> None:
        self.lottery_manager = LotteryManager(self.users)
        prizes = self.prize_manager.get_prizes()
        print("\n========== Lottery Start ==========")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, prize)
        self._show_summary(prizes)
        self._offer_export(prizes)

    def _run_round(self, round_no: int, prize: Prize) -> None:
        assert self.lottery_manager is not None
        lm = self.lottery_manager
        print(f"\n----- Round {round_no}: {prize.name} | Quantity: {prize.quantity} -----")
        print(f"Rules: {describe_rules(prize)}")
        if lm.is_cheng_ru_protection_active(prize):
            print("Notice: Cheng Ru Group protection is active for this round.")

        round_winners: list[User] = []
        terminated = False
        draw_no = 1
        while draw_no <= prize.quantity:
            command = self._ask(
                "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
            ).lower()
            if command == "terminate":
                terminated = True
                break
            if command == "pause":
                print("Lottery paused.")
                if not self._wait_for_resume():
                    terminated = True
                    break
                print("Lottery resumed.")
                continue
            if command:
                print("Unknown command.")
                continue

            print(f"Currently drawing the {_ordinal(draw_no)} winner...")
            winner = lm.draw_one(prize)
            if winner is None:
                print("No eligible candidates remain for this prize. Round ends.")
                break
            self.result_manager.add_winner(winner, prize)
            round_winners.append(winner)
            print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}!")
            draw_no += 1

        lm.finish_round(prize, round_winners)
        if terminated:
            print("Current round terminated and skipped.")

        print(f"Round results for {prize.name}:")
        if not round_winners:
            print("  (no winners)")
        for index, winner in enumerate(round_winners, start=1):
            print(f"  {index}. {winner.name} ({winner.department})")

    def _wait_for_resume(self) -> bool:
        """Block until 'continue' (True) or 'terminate' (False)."""
        while True:
            command = self._ask("Type 'continue' to resume or 'terminate' to end this round: ").lower()
            if command == "continue":
                return True
            if command == "terminate":
                return False
            print("Unknown command.")

    def _show_summary(self, prizes: list[Prize]) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n========== Final Winner List ==========")
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  {winner.sequence}. Name: {winner.user.name} | "
                    f"Department: {winner.user.department} | Prize: {prize_name}"
                )

    def _offer_export(self, prizes: list[Prize]) -> None:
        choice = self._ask("\nExport the winner list to file? (y/n): ").lower()
        if choice in ("y", "yes"):
            _, message = self.result_manager.export_to_csv(prizes)
            print(message)

    # ---------------------------------------------------------------- run
    def run(self) -> None:
        print("=====================================================")
        print("  Company Team Building Smart Lottery System")
        print("=====================================================")
        if not self._setup_user_list():
            print("Program exited.")
            return
        self._run_prize_menu()
        print("Thank you for using the lottery system. Goodbye!")


def main() -> int:
    """Program entry point. Returns a process exit code."""
    try:
        LotterySystemCLI().run()
    except (KeyboardInterrupt, EOFError):
        print("\nProgram interrupted. Exiting.")
        return 1
    return 0

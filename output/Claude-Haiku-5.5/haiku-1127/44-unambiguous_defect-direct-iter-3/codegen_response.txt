    "Input format incorrect, please use 'Starting Station; Ending Station' format"
)


class SubwayError(Exception):

class PathNotFoundError(SubwayError):
    """Raised when no route exists between two stations under the given settings."""

        super().__init__(message)

"""Loading and saving of the Beijing Subway data file."""

from __future__ import annotations

import json
import os
from typing import Any, Dict

from errors import DataError

DATA_FILE_PATH: str = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


def load_subway_data(path: str) -> Dict[str, Any]:
    """Load the subway data JSON file and return its content as a dictionary.

    Raises:
        OSError: if the file cannot be read.
        DataError: if the JSON structure is not the expected one.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)

    if not isinstance(data, dict):
        raise DataError("subway data root must be a JSON object")
    if not isinstance(data.get("stations"), list):
        raise DataError("subway data must contain a 'stations' list")
    if not isinstance(data.get("connections"), list):
        raise DataError("subway data must contain a 'connections' list")
    return data


def save_subway_data(data: Dict[str, Any], path: str = DATA_FILE_PATH) -> None:
    """Write subway data back to a JSON file (UTF-8, pretty-printed)."""
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")

"""Graph data structures used to model the subway network.

The graph is a weighted, undirected multigraph. Vertices are station numbers
(integers assigned by the caller, e.g. ``SubwayNetwork``), and every edge
stores the inter-station distance in metres as its weight.
"""

from __future__ import annotations

from typing import Dict, List, Optional


class Edge:
    """An undirected, weighted connection between two stations on a line."""

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source: int, dest: int, distance: int, line: str) -> None:
        if source == dest:
            raise ValueError("an edge cannot connect a station to itself")
        if distance < 0:
            raise ValueError("edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self) -> int:
        """Edge weight used by path algorithms (equal to the distance in metres)."""
        return self.distance

    def other(self, vertex: int) -> int:
        """Return the endpoint of this edge that is not ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"vertex {vertex} is not an endpoint of this edge")

    def __repr__(self) -> str:
        return (
            f"Edge({self.source}, {self.dest}, distance={self.distance}, "
            f"line={self.line!r})"
        )


class Graph:
    """Weighted undirected graph with adjacency lists.

    Station numbers are assigned by the caller. The graph additionally keeps
    ``station_lines``, mapping each station name to the lines serving it.
    """

    def __init__(self) -> None:
        self._adj: Dict[int, List[Edge]] = {}
        self._edges: List[Edge] = []
        self.station_lines: Dict[str, List[str]] = {}

    # ------------------------------------------------------------------ vertices
    def add_vertex(self, vertex: int) -> None:
        """Add a vertex if it does not exist yet."""
        if vertex not in self._adj:
            self._adj[vertex] = []

    def has_vertex(self, vertex: int) -> bool:
        return vertex in self._adj

    def vertices(self) -> List[int]:
        return list(self._adj)

    @property
    def num_vertices(self) -> int:
        return len(self._adj)

    # --------------------------------------------------------------------- edges
    def add_edge(self, edge: Edge) -> None:
        """Add an undirected edge, creating its endpoints when necessary."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adj[edge.source].append(edge)
        self._adj[edge.dest].append(edge)
        self._edges.append(edge)

    def adjacent(self, vertex: int) -> List[Edge]:
        """Edges incident to ``vertex`` (treat the returned list as read-only)."""
        return self._adj.get(vertex, [])

    def edges(self) -> List[Edge]:
        """A copy of the list of all edges, each listed once."""
        return list(self._edges)

    @property
    def num_edges(self) -> int:
        return len(self._edges)

    def edges_between(
        self, a: int, b: int, line: Optional[str] = None
    ) -> List[Edge]:
        """Return edges joining ``a`` and ``b`` (optionally restricted to a line)."""
        return [
            e
            for e in self._adj.get(a, [])
            if e.other(a) == b and (line is None or e.line == line)
        ]

    def remove_edges(self, a: int, b: int, line: Optional[str] = None) -> int:
        """Remove edges between ``a`` and ``b``; return how many were removed."""
        removed = self.edges_between(a, b, line)
        if not removed:
            return 0
        removed_ids = {id(e) for e in removed}
        self._edges = [e for e in self._edges if id(e) not in removed_ids]
        for vertex in (a, b):
            self._adj[vertex] = [
                e for e in self._adj[vertex] if id(e) not in removed_ids
            ]
        return len(removed)

    # ------------------------------------------------------------ station lines
    def set_station_lines(self, station_name: str, lines: List[str]) -> None:
        """Set the list of lines serving a station (by name)."""
        self.station_lines[station_name] = list(lines)

    def get_station_lines(self, station_name: str) -> List[str]:
        return list(self.station_lines.get(station_name, []))

<file path="src/min_pq.py">
"""Binary-heap based minimum priority queue."""

from __future__ import annotations

import heapq
import itertools
from typing import Any, Tuple


class MinPQ:
    """Min priority queue backed by ``heapq``.

    Items with equal keys are returned in insertion order thanks to a
    monotonically increasing sequence number, so items themselves are never
    compared. Insert/delete-min run in O(log n).
    """

    def __init__(self) -> None:
        self._heap: list = []
        self._seq = itertools.count()

    def insert(self, key: Any, item: Any) -> None:
        heapq.heappush(self._heap, (key, next(self._seq), item))

    def del_min(self) -> Tuple[Any, Any]:
        """Remove and return ``(key, item)`` with the smallest key."""
        if not self._heap:
            raise IndexError("del_min from an empty priority queue")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min_key(self) -> Any:
        if not self._heap:
            raise IndexError("min_key from an empty priority queue")
        return self._heap[0][0]

    def is_empty(self) -> bool:
        return not self._heap

    def __len__(self) -> int:
        return len(self._heap)

"""Shortest path computation on the subway graph using Dijkstra's algorithm.

Supported query modes:
  * ``shortest_distance`` - minimise the total distance in metres.
  * ``least_transfer``    - minimise the number of line changes, then distance.
  * ``comprehensive``     - minimise distance plus a penalty per transfer.

``max_transfers`` limits the number of line changes allowed on a route, and
``preferred_lines`` makes segments on other lines slightly more expensive so
that routes along preferred lines are favoured.
"""

from __future__ import annotations

from typing import Dict, List, Optional, Sequence, Tuple

from errors import PathNotFoundError
from min_pq import MinPQ
from models import Edge, Graph

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES: Tuple[str, ...] = (
    MODE_SHORTEST_DISTANCE,
    MODE_LEAST_TRANSFER,
    MODE_COMPREHENSIVE,
)

# Equivalent distance (metres) added for every transfer in comprehensive mode.
TRANSFER_PENALTY_M: float = 500.0
# Equivalent distance (metres) added per metre-segment on non-preferred lines.
NON_PREFERRED_LINE_PENALTY_M: float = 300.0


def normalize_line_name(name: str) -> str:
    """Normalise a line name for case- and whitespace-insensitive comparison."""
    return " ".join(str(name).split()).casefold()


class PathfindingConfig:
    """Settings that control how a route is calculated."""

    def __init__(
        self,
        mode: str = MODE_SHORTEST_DISTANCE,
        max_transfers: Optional[int] = None,
        preferred_lines: Optional[Sequence[str]] = None,
    ) -> None:
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines: List[str] = list(preferred_lines or [])

    def validate(self) -> None:
        """Raise ``ValueError`` if the configuration is invalid."""
        if self.mode not in VALID_MODES:
            raise ValueError(
                f"unknown mode '{self.mode}', expected one of {', '.join(VALID_MODES)}"
            )
        if self.max_transfers is not None:
            if isinstance(self.max_transfers, bool) or not isinstance(
                self.max_transfers, int
            ):
                raise ValueError("max_transfers must be an integer")
            if self.max_transfers < 0:
                raise ValueError("max_transfers must be >= 0")

    def is_preferred_line(self, line: str) -> bool:
        """True if ``line`` is preferred (always True when no preference is set)."""
        if not self.preferred_lines:
            return True
        target = normalize_line_name(line)
        return any(normalize_line_name(p) == target for p in self.preferred_lines)


class ShortestPath:
    """Single-source shortest paths computed with Dijkstra's algorithm.

    The priority queue (``MinPQ``, a binary heap) gives O((V + E) log V) time.
    Path costs are compared as tuples, so the chosen criterion depends on the
    configured mode. ``dist_to`` always contains the real distance in metres
    of the chosen route, independent of penalties.
    """

    def __init__(
        self, graph: Graph, source: int, config: Optional[PathfindingConfig] = None
    ) -> None:
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()
        if not graph.has_vertex(source):
            raise ValueError(f"unknown source station number: {source}")

        self.dist_to: Dict[int, int] = {}          # real distance in metres
        self.edge_to: Dict[int, Edge] = {}         # predecessor edge
        self.transfers: Dict[int, int] = {}        # transfers along chosen route
        self._key_to: Dict[int, tuple] = {}        # comparison key per vertex
        self._effective: Dict[int, float] = {}     # distance incl. penalties

        self._compute()

    # ----------------------------------------------------------------- helpers
    def _make_key(self, effective: float, transfers: int) -> tuple:
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, effective)
        if mode == MODE_COMPREHENSIVE:
            return (effective + TRANSFER_PENALTY_M * transfers, transfers)
        return (effective, transfers)

    def _line_penalty(self, line: str) -> float:
        if not self.config.preferred_lines or self.config.is_preferred_line(line):
            return 0.0
        return NON_PREFERRED_LINE_PENALTY_M

    # --------------------------------------------------------------- algorithm
    def _compute(self) -> None:
        source = self.source
        self.dist_to[source] = 0
        self.transfers[source] = 0
        self._effective[source] = 0.0
        self._key_to[source] = self._make_key(0.0, 0)

        pq = MinPQ()
        pq.insert(self._key_to[source], source)
        settled = set()
        max_transfers = self.config.max_transfers

        while not pq.is_empty():
            _, u = pq.del_min()
            if u in settled:
                continue
            settled.add(u)

            incoming_line = self.edge_to[u].line if u != source else None

            for edge in self.graph.adjacent(u):
                v = edge.other(u)
                if v in settled:
                    continue

                # A transfer happens when the line changes at an interior station.
                is_transfer = 1 if (
                    incoming_line is not None and incoming_line != edge.line
                ) else 0
                transfers = self.transfers[u] + is_transfer
                if max_transfers is not None and transfers > max_transfers:
                    continue

                effective = (
                    self._effective[u]
                    + edge.distance
                    + self._line_penalty(edge.line)
                )
                key = self._make_key(effective, transfers)
                if v not in self._key_to or key < self._key_to[v]:
                    self._key_to[v] = key
                    self._effective[v] = effective
                    self.transfers[v] = transfers
                    self.dist_to[v] = self.dist_to[u] + edge.distance
                    self.edge_to[v] = edge
                    pq.insert(key, v)

    # ------------------------------------------------------------------ queries
    def has_path_to(self, target: int) -> bool:
        return target in self.dist_to

    def distance_to(self, target: int) -> int:
        if not self.has_path_to(target):
            raise PathNotFoundError(f"no path to station number {target}")
        return self.dist_to[target]

    def get_transfer_count(self, target: int) -> int:
        """Number of line changes on the route from the source to ``target``."""
        if not self.has_path_to(target):
            raise PathNotFoundError(f"no path to station number {target}")
        return self.transfers[target]

    def path_with_edges(self, target: int) -> Tuple[List[int], List[Edge]]:
        """Return ``(station numbers, edges)`` from the source to ``target``.

        The station list has one more element than the edge list.
        """
        if not self.has_path_to(target):
            raise PathNotFoundError(f"no path to station number {target}")

        vertices = [target]
        edges: List[Edge] = []
        v = target
        while v != self.source:
            edge = self.edge_to[v]
            edges.append(edge)
            v = edge.other(v)
            vertices.append(v)
        vertices.reverse()
        edges.reverse()
        return vertices, edges

"""Beijing Subway fare and travel-time estimation.

Fare structure:
  * 0 - 6 km      : 3 yuan
  * 6 - 12 km     : 4 yuan
  * 12 - 22 km    : 5 yuan
  * 22 - 32 km    : 6 yuan
  * over 32 km    : 7 yuan, plus 1 yuan for every further 20 km

Boundary values belong to the higher tier (6000 m -> 4 yuan, 32000 m -> 7 yuan).
Fractional distances are rounded up to the next whole metre before lookup.
"""

from __future__ import annotations

import math
from typing import Tuple

BASIC_FARE_TIERS: Tuple[Tuple[int, int], ...] = (
    (6000, 3),    # distances below 6000 m
    (12000, 4),   # 6000 m up to 12000 m
    (22000, 5),   # 12000 m up to 22000 m
    (32000, 6),   # 22000 m up to 32000 m
)
EXTENDED_BASE_DISTANCE_M: int = 32000
EXTENDED_BASE_FARE: int = 7
EXTENDED_STEP_DISTANCE_M: int = 20000

AVERAGE_SPEED_KMH: float = 35.0   # average operating speed of urban subway trains
BOARDING_TIME_MIN: float = 2.0    # fixed time for entering the system
TRANSFER_TIME_MIN: float = 4.0    # walking and waiting time per transfer


def _validate_distance(distance: float) -> float:
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of metres")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("distance must be a finite number")
    if distance < 0:
        raise ValueError("distance must be non-negative")
    return float(distance)


def Sum_money(distance: float) -> int:
    """Return the subway fare in yuan for a travelled ``distance`` in metres."""
    metres = math.ceil(_validate_distance(distance))
    for upper_bound, fare in BASIC_FARE_TIERS:
        if metres < upper_bound:
            return fare
    extra_steps = (metres - EXTENDED_BASE_DISTANCE_M) // EXTENDED_STEP_DISTANCE_M
    return EXTENDED_BASE_FARE + extra_steps


def calculate_estimated_time(distance: float, transfer_count: int) -> float:
    """Estimate travel time in minutes from distance (metres) and transfers."""
    metres = _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int):
        raise TypeError("transfer_count must be an integer")
    if transfer_count < 0:
        raise ValueError("transfer_count must be non-negative")
    travel_min = (metres / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    total = BOARDING_TIME_MIN + travel_min + transfer_count * TRANSFER_TIME_MIN
    return round(total, 1)

"""Subway network model: stations, lines and connections.

``SubwayNetwork`` owns the station-number mapping (``station_to_number``),
the weighted undirected graph (``graph``) and provides lookup, validation and
data-update operations.
"""

from __future__ import annotations

from typing import Any, Dict, Iterable, List, Optional, Sequence

from errors import DataError, StationNotFoundError
from models import Edge, Graph


def normalize_station_name(name: str) -> str:
    """Normalise a station name for case- and whitespace-insensitive matching."""
    return " ".join(str(name).split()).casefold()


def _clean_name(name: str, what: str) -> str:
    cleaned = " ".join(str(name).split())
    if not cleaned:
        raise DataError(f"{what} must not be empty")
    return cleaned


def _validate_distance(distance: Any) -> int:
    if isinstance(distance, bool):
        raise DataError("distance must be a positive integer number of metres")
    if isinstance(distance, float) and distance.is_integer():
        distance = int(distance)
    if not isinstance(distance, int) or distance <= 0:
        raise DataError("distance must be a positive integer number of metres")
    return distance


class SubwayNetwork:
    """Container for the Beijing Subway network."""

    def __init__(self) -> None:
        self.station_to_number: Dict[str, int] = {}
        self.number_to_station: Dict[int, str] = {}
        self.graph = Graph()
        self.issues: List[str] = []  # problems found while building the network
        self._normalized: Dict[str, str] = {}
        self._next_number = 0

    # ------------------------------------------------------------------ lookup
    def station_names(self) -> List[str]:
        return list(self.station_to_number)

    def standardize_name(self, raw: str) -> Optional[str]:
        """Return the official station name matching ``raw``, or None."""
        return self._normalized.get(normalize_station_name(raw))

    def station_exists(self, raw: str) -> bool:
        return self.standardize_name(raw) is not None

    def require_station(self, raw: str, role: str = "Target") -> str:
        """Return the official name for ``raw`` or raise ``StationNotFoundError``."""
        name = self.standardize_name(raw)
        if name is None:
            raise StationNotFoundError(str(raw).strip(), role)
        return name

    def number_of(self, name: str) -> int:
        return self.station_to_number[name]

    def name_of(self, number: int) -> str:
        return self.number_to_station[number]

    def lines_of(self, name: str) -> List[str]:
        return self.graph.get_station_lines(name)

    def connection_line_names(self) -> List[str]:
        """Names of all lines that have at least one connection."""
        return list(dict.fromkeys(e.line for e in self.graph.edges()))

    def line_names(self) -> List[str]:
        """Names of all lines known from station data or connections."""
        names: Dict[str, None] = {}
        for lines in self.graph.station_lines.values():
            for line in lines:
                names.setdefault(line, None)
        for edge in self.graph.edges():
            names.setdefault(edge.line, None)
        return list(names)

    def stations_on_line(self, line: str) -> List[str]:
        """Stations served by ``line`` (from connections or station data)."""
        names: Dict[str, None] = {}
        for edge in self.graph.edges():
            if edge.line == line:
                names.setdefault(self.name_of(edge.source), None)
                names.setdefault(self.name_of(edge.dest), None)
        for name, lines in self.graph.station_lines.items():
            if line in lines:
                names.setdefault(name, None)
        return list(names)

    def connections(self) -> List[Edge]:
        return self.graph.edges()

    # ---------------------------------------------------------------- updates
    def _merge_lines(self, name: str, lines: Iterable[str]) -> None:
        current = self.graph.get_station_lines(name)
        for line in lines:
            if line and line not in current:
                current.append(line)
        self.graph.set_station_lines(name, current)

    def add_station(self, name: str, lines: Iterable[str] = ()) -> int:
        """Add a station (or merge lines into an existing one); return its number."""
        clean = _clean_name(name, "station name")
        key = normalize_station_name(clean)
        if key in self._normalized:
            existing = self._normalized[key]
            self._merge_lines(existing, lines)
            return self.station_to_number[existing]

        number = self._next_number
        self._next_number += 1
        self.station_to_number[clean] = number
        self.number_to_station[number] = clean
        self._normalized[key] = clean
        self.graph.add_vertex(number)
        self.graph.set_station_lines(clean, [])
        self._merge_lines(clean, lines)
        return number

    def add_connection(
        self, source_name: str, dest_name: str, distance: Any, line: str
    ) -> Edge:
        """Add an undirected connection between two existing stations."""
        source = self.require_station(source_name, "Starting")
        dest = self.require_station(dest_name, "Ending")
        if source == dest:
            raise DataError(f"station '{source}' cannot connect to itself")
        metres = _validate_distance(distance)
        if not str(line).strip():
            raise DataError("connection line must not be empty")
        a, b = self.number_of(source), self.number_of(dest)
        if self.graph.edges_between(a, b, line):
            raise DataError(
                f"duplicate connection between '{source}' and '{dest}' on '{line}'"
            )
        edge = Edge(a, b, metres, line)
        self.graph.add_edge(edge)
        return edge

    def add_line(
        self, line_name: str, station_names: Sequence[str], distances: Sequence[Any]
    ) -> List[Edge]:
        """Add a new line from an ordered list of stations and segment distances."""
        line = _clean_name(line_name, "line name")
        if line in self.line_names():
            raise DataError(f"line '{line}' already exists")
        if len(station_names) < 2:
            raise DataError("a line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise DataError("number of distances must equal number of stations - 1")
        checked = [_validate_distance(d) for d in distances]

        for name in station_names:
            self.add_station(name, [line])
        edges = []
        for i, metres in enumerate(checked):
            edges.append(
                self.add_connection(station_names[i], station_names[i + 1], metres, line)
            )
        return edges

    def update_station(
        self,
        name: str,
        new_name: Optional[str] = None,
        lines: Optional[Sequence[str]] = None,
    ) -> str:
        """Rename a station and/or replace its line list. Returns the final name."""
        current = self.require_station(name)
        if new_name is not None:
            clean = _clean_name(new_name, "station name")
            new_key = normalize_station_name(clean)
            old_key = normalize_station_name(current)
            if new_key != old_key and new_key in self._normalized:
                raise DataError(f"station '{clean}' already exists")
            number = self.station_to_number.pop(current)
            self._normalized.pop(old_key, None)
            station_lines = self.graph.station_lines.pop(current)
            self.station_to_number[clean] = number
            self.number_to_station[number] = clean
            self._normalized[new_key] = clean
            self.graph.station_lines[clean] = station_lines
            current = clean
        if lines is not None:
            cleaned = [str(x).strip() for x in lines if str(x).strip()]
            self.graph.set_station_lines(current, list(dict.fromkeys(cleaned)))
        return current

    def update_distance(
        self, source_name: str, dest_name: str, distance: Any, line: Optional[str] = None
    ) -> int:
        """Update the distance of connection(s) between two stations.

        Returns the number of connections updated.
        """
        a = self.number_of(self.require_station(source_name, "Starting"))
        b = self.number_of(self.require_station(dest_name, "Ending"))
        metres = _validate_distance(distance)
        edges = self.graph.edges_between(a, b, line)
        if not edges:
            raise DataError(
                f"no connection between '{source_name}' and '{dest_name}'"
                + (f" on line '{line}'" if line else "")
            )
        for edge in edges:
            edge.distance = metres
        return len(edges)


def build_network(data: Dict[str, Any]) -> SubwayNetwork:
    """Build a ``SubwayNetwork`` from the parsed subway data dictionary.

    Malformed or inconsistent entries are skipped and recorded in
    ``network.issues`` so that the integrity check can report them.
    """
    if not isinstance(data, dict):
        raise DataError("subway data must be a dictionary")

    network = SubwayNetwork()
    stations = data.get("stations", [])
    connections = data.get("connections", [])
    if not isinstance(stations, list) or not isinstance(connections, list):
        raise DataError("'stations' and 'connections' must be lists")

    for index, entry in enumerate(stations):
        if not isinstance(entry, dict) or "name" not in entry:
            network.issues.append(f"station entry #{index} is malformed and was skipped")
            continue
        raw_lines = entry.get("lines", [])
        if not isinstance(raw_lines, list):
            network.issues.append(f"station '{entry['name']}' has invalid line list")
            raw_lines = []
        try:
            network.add_station(str(entry["name"]), [str(x) for x in raw_lines])
        except DataError as exc:
            network.issues.append(f"station entry #{index} skipped: {exc}")

    for index, entry in enumerate(connections):
        if not isinstance(entry, dict):
            network.issues.append(f"connection #{index} is malformed and was skipped")
            continue
        try:
            source = str(entry["source"])
            dest = str(entry["dest"])
            line = str(entry["line"])
            distance = _validate_distance(entry["distance"])
        except (KeyError, DataError) as exc:
            network.issues.append(f"connection #{index} skipped: {exc}")
            continue

        for name in (source, dest):
            if not network.station_exists(name):
                network.issues.append(
                    f"connection #{index}: station '{name}' is not declared in stations"
                )
                try:
                    network.add_station(name, [])
                except DataError as exc:
                    network.issues.append(f"connection #{index} skipped: {exc}")
                    break
        else:
            try:
                network.add_connection(source, dest, distance, line)
            except DataError as exc:
                network.issues.append(f"connection #{index} skipped: {exc}")

    return network


def network_to_data(network: SubwayNetwork) -> Dict[str, Any]:
    """Export the network back into the ``subway_data.json`` schema."""
    stations = [
        {"name": name, "lines": network.lines_of(name)}
        for name in network.station_to_number
    ]
    connections = [
        {
            "source": network.name_of(e.source),
            "dest": network.name_of(e.dest),
            "distance": e.distance,
            "line": e.line,
        }
        for e in network.graph.edges()
    ]
    return {"stations": stations, "connections": connections}

from __future__ import annotations

from collections import deque
from typing import Dict, List, Sequence

from network import SubwayNetwork

REQUIRED_MAIN_LINES: tuple = (

    "Line 2",
    "Line 4",

    "Line 10",
    "Line 13",

    "Line 15",
    "Changping Line",

    "Airport Line",
)
MAX_REASONABLE_DISTANCE_M: int = 6000
MIN_STATIONS_PER_LINE: int = 2


class IntegrityReport:
    """Collects errors, warnings and informational messages."""

    def __init__(self) -> None:
        self.errors: List[str] = []
        self.warnings: List[str] = []
        self.info: List[str] = []

    @property
    def ok(self) -> bool:
        return not self.errors

    def add_error(self, message: str) -> None:
        self.errors.append(message)

    def add_warning(self, message: str) -> None:
        self.warnings.append(message)

    def add_info(self, message: str) -> None:
        self.info.append(message)

    def format(self) -> str:
        out = ["Data integrity check", "=" * 40]
        out.append(f"Errors: {len(self.errors)}, warnings: {len(self.warnings)}")
        for title, items in (
            ("ERRORS", self.errors),
            ("WARNINGS", self.warnings),
            ("INFO", self.info),
        ):
            if items:
                out.append(f"{title}:")
                out.extend(f"  - {item}" for item in items)
        out.append("Result: PASSED" if self.ok else "Result: FAILED")
        return "\n".join(out)


def check_network(
    network: SubwayNetwork,
    required_lines: Sequence[str] = REQUIRED_MAIN_LINES,
    max_distance: int = MAX_REASONABLE_DISTANCE_M,
) -> IntegrityReport:
    """Run all integrity checks and return the resulting report."""
    report = IntegrityReport()
    for issue in network.issues:
        report.add_warning(issue)

    _check_coverage(network, required_lines, report)
    _check_stations(network, report)
    _check_connections(network, report)
    _check_line_continuity(network, report)
    _check_distances(network, max_distance, report)

    report.add_info(
        f"stations={len(network.station_to_number)}, "
        f"connections={network.graph.num_edges}, "
        f"lines={len(network.line_names())}"
    )
    transfer_stations = sum(
        1 for name in network.station_to_number if len(network.lines_of(name)) >= 2
    )
    report.add_info(f"interchange (multi-line) stations: {transfer_stations}")
    return report


def _check_coverage(
    network: SubwayNetwork, required_lines: Sequence[str], report: IntegrityReport
) -> None:
    present = set(network.connection_line_names())
    for line in required_lines:
        if line not in present:
            report.add_error(f"required main line '{line}' has no connections in the data")
            continue
        count = len(network.stations_on_line(line))
        if count < MIN_STATIONS_PER_LINE:
            report.add_error(
                f"line '{line}' covers only {count} station(s); expected at least "
                f"{MIN_STATIONS_PER_LINE}"
            )


def _check_stations(network: SubwayNetwork, report: IntegrityReport) -> None:
    for name, number in network.station_to_number.items():
        if not network.lines_of(name):
            report.add_warning(f"station '{name}' has no line assignment")
        if not network.graph.adjacent(number):
            report.add_warning(f"station '{name}' has no connections")


def _check_connections(network: SubwayNetwork, report: IntegrityReport) -> None:
    for edge in network.graph.edges():
        source = network.name_of(edge.source)
        dest = network.name_of(edge.dest)
        if not isinstance(edge.distance, int) or edge.distance <= 0:
            report.add_error(
                f"connection {source} - {dest} on '{edge.line}' has invalid distance"
            )
        for name in (source, dest):
            if edge.line not in network.lines_of(name):
                report.add_warning(
                    f"station '{name}' is connected on '{edge.line}' but its line "
                    f"list does not include it"
                )


def _check_line_continuity(network: SubwayNetwork, report: IntegrityReport) -> None:
    adjacency: Dict[str, Dict[int, List[int]]] = {}
    for edge in network.graph.edges():
        per_line = adjacency.setdefault(edge.line, {})
        per_line.setdefault(edge.source, []).append(edge.dest)
        per_line.setdefault(edge.dest, []).append(edge.source)

    for line, graph in adjacency.items():
        seen = set()
        components = 0
        for start in graph:
            if start in seen:
                continue
            components += 1
            seen.add(start)
            queue = deque([start])
            while queue:
                current = queue.popleft()
                for nxt in graph[current]:
                    if nxt not in seen:
                        seen.add(nxt)
                        queue.append(nxt)
        if components > 1:
            report.add_warning(
                f"line '{line}' is split into {components} disconnected segments"
            )


def _check_distances(
    network: SubwayNetwork, max_distance: int, report: IntegrityReport
) -> None:
    for edge in network.graph.edges():
        if edge.distance > max_distance:
            report.add_warning(
                f"distance {network.name_of(edge.source)} - {network.name_of(edge.dest)} "
                f"on '{edge.line}' is {edge.distance} m (above {max_distance} m); "
                f"please verify"
            )

from __future__ import annotations

import json

    ShortestPath,

MODE_LABELS: Dict[str, str] = {
    MODE_SHORTEST_DISTANCE: "Shortest distance",
    MODE_LEAST_TRANSFER: "Minimum transfers",
    MODE_COMPREHENSIVE: "Comprehensive optimal",
}


@dataclass
class PathSegment:
    """A continuous ride on a single line."""

    line: str
    stations: List[str]
    distance_m: int

    @property
    def stop_count(self) -> int:
        return len(self.stations) - 1


@dataclass
class TransferInfo:
    """A station where the passenger changes from one line to another."""

    station: str
    from_line: str
    to_line: str
    lines: List[str]


@dataclass
class PathResult:
    """Complete result of a route query."""

    start: str
    end: str
    mode: str
    max_transfers: Optional[int]
    preferred_lines: List[str]
    stations: List[str]
    segments: List[PathSegment]
    transfers: List[TransferInfo]
    lines_used: List[str]
    total_distance_m: int
    fare: int
    transfer_count: int
    estimated_minutes: float

    @property
    def total_distance_km(self) -> float:
        return round(self.total_distance_m / 1000.0, 2)


def _unique(items: Sequence[str]) -> List[str]:
    return list(dict.fromkeys(items))


def _build_segments(names: List[str], edges: List[Edge]) -> List[PathSegment]:
    segments: List[PathSegment] = []
    if not edges:
        return segments
    start = 0
    for i in range(1, len(edges) + 1):
        if i == len(edges) or edges[i].line != edges[start].line:
            distance = sum(e.distance for e in edges[start:i])
            segments.append(
                PathSegment(
                    line=edges[start].line,
                    stations=names[start : i + 1],
                    distance_m=distance,
                )
            )
            start = i
    return segments


def build_path_result(
    network: SubwayNetwork,
    start_name: str,
    end_name: str,
    config: Optional[PathfindingConfig] = None,
) -> PathResult:
    """Compute the route between two official station names and summarise it."""
    config = config if config is not None else PathfindingConfig()
    config.validate()
    if start_name == end_name:
        raise QueryError("Starting station cannot be the same as the ending station")

    source = network.number_of(start_name)
    target = network.number_of(end_name)
    shortest = ShortestPath(network.graph, source, config)
    if not shortest.has_path_to(target):
        raise PathNotFoundError(
            f"No route found from '{start_name}' to '{end_name}' with the current settings"
        )

    numbers, edges = shortest.path_with_edges(target)
    names = [network.name_of(n) for n in numbers]
    total = sum(e.distance for e in edges)

    transfers: List[TransferInfo] = []
    for k in range(1, len(edges)):
        if edges[k - 1].line != edges[k].line:
            station = names[k]
            transfers.append(
                TransferInfo(
                    station=station,
                    from_line=edges[k - 1].line,
                    to_line=edges[k].line,
                    lines=network.lines_of(station),
                )
            )

    transfer_count = len(transfers)
    return PathResult(
        start=start_name,
        end=end_name,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        stations=names,
        segments=_build_segments(names, edges),
        transfers=transfers,
        lines_used=_unique([e.line for e in edges]),
        total_distance_m=total,
        fare=Sum_money(total),
        transfer_count=transfer_count,
        estimated_minutes=calculate_estimated_time(total, transfer_count),
    )


def build_summary(result: PathResult) -> Dict[str, Any]:
    """Key figures of a path: distance, fare, transfers and estimated time."""
    return {
        "total_distance_km": result.total_distance_km,
        "total_fare_yuan": result.fare,
        "transfer_count": result.transfer_count,
        "estimated_time_minutes": result.estimated_minutes,
    }


def render_text(result: PathResult) -> str:
    """Human-readable console report."""
    out: List[str] = []
    out.append("=" * 60)
    out.append(f"Route: {result.start} -> {result.end}")
    out.append(f"Query mode: {MODE_LABELS.get(result.mode, result.mode)}")
    if result.max_transfers is not None:
        out.append(f"Max transfers: {result.max_transfers}")
    if result.preferred_lines:
        out.append(f"Preferred lines: {', '.join(result.preferred_lines)}")
    out.append("-" * 60)
    out.append(f"Station sequence ({len(result.stations)} stations):")
    out.append("  " + " -> ".join(result.stations))
    out.append(
        f"Total distance: {result.total_distance_m} m "
        f"({result.total_distance_km:.2f} km)"
    )
    out.append(f"Transfers: {result.transfer_count}")
    out.append(f"Estimated travel time: {result.estimated_minutes:.1f} min")
    out.append("Line segments:")
    for index, segment in enumerate(result.segments, start=1):
        out.append(
            f"  {index}. {segment.line}: {segment.stations[0]} -> "
            f"{segment.stations[-1]} ({segment.stop_count} stops, "
            f"{segment.distance_m} m)"
        )
    out.append("Transfer stations:")
    if not result.transfers:
        out.append("  None (no transfer needed)")
    for transfer in result.transfers:
        out.append(
            f"  - {transfer.station}: {transfer.from_line} -> {transfer.to_line} "
            f"(lines at station: {', '.join(transfer.lines)})"
        )
    out.append(f"Lines involved: {', '.join(result.lines_used)}")
    out.append("-" * 60)
    out.append(f"Fare: {result.fare} yuan")
    out.append("=" * 60)
    return "\n".join(out)


def render_json(result: PathResult) -> str:
    """Structured JSON report for programmatic integration."""
    payload = {
        "query": {
            "start": result.start,
            "end": result.end,
            "mode": result.mode,
            "max_transfers": result.max_transfers,
            "preferred_lines": result.preferred_lines,
        },
        "path": {
            "stations": result.stations,
            "station_count": len(result.stations),
        },
        "distance": {
            "meters": result.total_distance_m,
            "kilometers": result.total_distance_km,
        },
        "fare": result.fare,
        "estimated_time_minutes": result.estimated_minutes,
        "transfer_count": result.transfer_count,
        "lines": result.lines_used,
        "transfers": [
            {
                "station": t.station,
                "from_line": t.from_line,
                "to_line": t.to_line,
                "lines": t.lines,
            }
            for t in result.transfers
        ],
        "segments": [
            {
                "line": s.line,
                "start": s.stations[0],
                "end": s.stations[-1],
                "stations": s.stations,
                "distance_m": s.distance_m,
            }
            for s in result.segments
        ],
        "summary": build_summary(result),
    }
    return json.dumps(payload, ensure_ascii=False, indent=2)

def parse_query(text: str) -> Tuple[str, str]:
    """Split a query of the form 'Starting Station; Ending Station'."""
    positions = [i for i, ch in enumerate(text) if ch in QUERY_SEPARATORS]
    if len(positions) != 1:
        raise InputFormatError()
    pos = positions[0]
    start = text[:pos].strip()
    end = text[pos + 1 :].strip()
    if not start or not end:
        raise InputFormatError()
    return start, end


def resolve_stations(
    network: SubwayNetwork, start_raw: str, end_raw: str
) -> Tuple[str, str]:
    """Standardise and validate both station names."""
    start = network.require_station(start_raw, "Starting")
    end = network.require_station(end_raw, "Ending")
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")
    return start, end


def run_query(
    network: SubwayNetwork, config: PathfindingConfig, text: str, fmt: str
) -> str:
    """Process one query string and return the rendered report."""
    start_raw, end_raw = parse_query(text)
    start, end = resolve_stations(network, start_raw, end_raw)
    result = build_path_result(network, start, end, config)
    return render_json(result) if fmt == "json" else render_text(result)

def build_config(
    args: argparse.Namespace, network: SubwayNetwork, out: TextIO
) -> PathfindingConfig:
    """Create a validated ``PathfindingConfig`` from command-line arguments."""
    known = {normalize_line_name(name): name for name in network.line_names()}
    preferred = []
    if args.line_preference:
        for raw in args.line_preference.split(","):
            raw = raw.strip()
            if not raw:
                continue
            key = normalize_line_name(raw)
            if key in known:
                preferred.append(known[key])
            else:
                print(f"Warning: line '{raw}' not found; preference ignored", file=out)
    config = PathfindingConfig(
        mode=args.mode,
        max_transfers=args.max_transfers,
        preferred_lines=preferred,
    )
    config.validate()
    return config


def write_output(path: str, content: str, append: bool) -> None:
    with open(path, "a" if append else "w", encoding="utf-8") as handle:
        handle.write(content + "\n\n")


def main(
    argv: Optional[Sequence[str]] = None,
    stdin: Optional[TextIO] = None,
    stdout: Optional[TextIO] = None,
) -> int:
    """Run the CLI. Returns a process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)
    inp = stdin if stdin is not None else sys.stdin
    out = stdout if stdout is not None else sys.stdout

    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be >= 0")

    try:
        network = build_network(load_subway_data(DATA_FILE_PATH))
    except (OSError, ValueError, DataError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=out)
        return 1

    if args.check_data:
        report = check_network(network)
        print(report.format(), file=out)
        return 0 if report.ok else 1

    try:
        config = build_config(args, network, out)
    except ValueError as exc:
        print(f"Error: invalid settings: {exc}", file=out)
        return 2

    print(BANNER, file=out)
    first_write = True
    while True:
        out.write("Query> ")
        out.flush()
        line = inp.readline()
        if not line:  # end of input
            break
        text = line.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break

        try:
            rendered = run_query(network, config, text, args.format)
        except SubwayError as exc:
            print(f"Error: {exc}", file=out)
            continue
        except ValueError as exc:
            print(f"Error: route or fare calculation failed: {exc}", file=out)
            continue
        except Exception as exc:  # keep the session alive on unexpected failures
            print(f"Error: unexpected failure while querying: {exc}", file=out)
            continue

        print(rendered, file=out)
        if args.output_file:
            try:
                write_output(args.output_file, rendered, append=not first_write)
                first_write = False
            except OSError as exc:
                print(f"Error: cannot write output file: {exc}", file=out)

    return 0

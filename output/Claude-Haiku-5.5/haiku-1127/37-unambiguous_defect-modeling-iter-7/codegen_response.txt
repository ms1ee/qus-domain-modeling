"""Employee list loading, validation and group extraction."""

import os

from lottery_system.constants import DEFAULT_TEAM_LIST
from lottery_system.models import User


class ListManager:
    """Loads and validates employee lists.

    Line format for custom txt files: ``Name,Department`` (one person per line).
    Blank lines are ignored.
    """

    def __init__(self):
        self.load_mode = None
        self.source_file_path = None
        self.is_validated = False

    def load_default_list(self):
        """Load the built-in team building list.

        Returns (valid users, error messages).
        """
        users, errors = self._parse_lines(DEFAULT_TEAM_LIST)
        valid_users, duplicates = self.validate_users(users)
        errors.extend(self._duplicate_messages(duplicates))
        self.load_mode = "default"
        self.source_file_path = None
        self.is_validated = True
        return valid_users, errors

    def load_custom_list(self, path):
        """Load a custom txt list and validate it.

        Returns (valid users, error messages). Invalid lines and duplicate
        names are reported in the error messages and excluded from users.
        """
        users, errors = self._load_from_path(path)
        valid_users, duplicates = self.validate_users(users)
        errors.extend(self._duplicate_messages(duplicates))
        self.load_mode = "custom"
        self.source_file_path = path
        self.is_validated = True
        return valid_users, errors

    def _load_from_path(self, path):
        """Parse a txt file into users, collecting format errors.

        Returns (users, error messages).
        """
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except UnicodeDecodeError:
            return [], ["File Error: file must be UTF-8 encoded text"]
        except OSError as exc:
            return [], [f"File Error: cannot read file ({exc})"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines):
        users = []
        errors = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department' "
                    f"but got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        """Split users into (valid_users, duplicates).

        The first occurrence of each name is kept as valid; later occurrences
        are returned as duplicates.
        """
        seen = set()
        valid_users = []
        duplicates = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users):
        """Return a mapping of department name to member count (sorted by name)."""
        counts = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

    @staticmethod
    def _duplicate_messages(duplicates):
        return [
            f"Duplicate Name: '{user.name}' ({user.department}) appears more than "
            f"once and was removed"
            for user in duplicates
        ]

"""Prize library management and round ordering."""

import re

from lottery_system.constants import DEFAULT_PRIZES
from lottery_system.models import Prize

_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3}
_CUSTOM_LEVEL_BASE = 100


def _level_from_name(name):
    match = re.match(r"\s*(first|second|third)\s+prize", name, re.IGNORECASE)
    if match:
        return _LEVEL_WORDS[match.group(1).lower()]
    return None


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        """Load the preset prize library. Returns a list of error messages."""
        errors = []
        for config in DEFAULT_PRIZES:
            try:
                self.add_prize(
                    config["name"],
                    config["quantity"],
                    list(config["avoid_groups"]),
                    config["associated_drawer"],
                )
                self._prizes[-1].is_preset = True
            except ValueError as exc:
                errors.append(f"Preset prize not loaded: {config['name']} ({exc})")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        """Add a prize at the end of the round order.

        Raises ValueError for an empty name, invalid quantity or duplicate name.
        """
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(existing.name == name for existing in self._prizes):
            raise ValueError(f"A prize named '{name}' already exists.")

        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None

        prize = Prize(name, quantity, groups, drawer)
        level = _level_from_name(name)
        prize.level = level if level is not None else _CUSTOM_LEVEL_BASE + len(self._prizes)
        prize.is_preset = False
        self._prizes.append(prize)
        self._refresh_sequence_numbers()

    def get_prizes(self):
        """Return prizes in current round order."""
        return list(self._prizes)

    def reorder_prizes(self, order):
        """Reorder prizes using 1-based current positions.

        ``order`` must contain every current position exactly once.
        Returns True on success, False otherwise.
        """
        try:
            normalized = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(normalized) != list(range(1, len(self._prizes) + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in normalized]
        self._refresh_sequence_numbers()
        return True

    def _refresh_sequence_numbers(self):
        for index, prize in enumerate(self._prizes, start=1):
            prize.sequence_number = index

"""Lottery rule engine: eligibility filtering and winner drawing."""

import random

from lottery_system.constants import (
    CHENG_RU_GROUP,
    DRAWER_CHEN_QIXIAN,
    DRAWER_JIN_DUO,
    DRAWER_LI_BIN,
    DRAWER_MA_JIAQI,
)


def is_cheng_ru_member(user):
    """Return True if the user belongs to the Cheng Ru group."""
    return user.department.strip().lower() == CHENG_RU_GROUP.lower()


def _is_drawer(prize, drawer_name):
    drawer = (prize.associated_drawer or "").strip().lower()
    return drawer == drawer_name.lower()


class LotteryManager:
    """Runs lottery rounds over a fixed list of users."""

    def __init__(self, users):
        self.users = list(users)
        self.winner_names = set()
        # Results (True = contained a Cheng Ru member) of the most recent two
        # rounds drawn by Chen Qixian or Jin Duo.
        self.last_two_draw_results = []
        self._rng = random.Random()

    def is_fallback_active(self):
        """True when the Cheng Ru fallback protection is triggered."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize):
        """Return users eligible to win the given prize in the current state."""
        available = [user for user in self.users if user.name not in self.winner_names]

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru members, overriding
        # avoidance groups.
        if _is_drawer(prize, DRAWER_MA_JIAQI) and self.is_fallback_active():
            return [user for user in available if is_cheng_ru_member(user)]

        # Special rule: Li Bin relaxes filtering (only anti-duplication applies).
        if _is_drawer(prize, DRAWER_LI_BIN):
            return available

        avoid = {group.strip().lower() for group in prize.avoid_groups}
        return [user for user in available if user.department.strip().lower() not in avoid]

    def get_eligible_pool(self, prize):
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def draw_one(self, prize):
        """Draw a single winner for the prize. Returns the User or None."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self.winner_names.add(winner.name)
        return winner

    def finish_round(self, prize, round_winners):
        """Record the outcome of a finished round for the fallback rule."""
        if _is_drawer(prize, DRAWER_CHEN_QIXIAN) or _is_drawer(prize, DRAWER_JIN_DUO):
            contains_cheng_ru = any(is_cheng_ru_member(user) for user in round_winners)
            self.last_two_draw_results.append(contains_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def describe_rules(self, prize):
        """Return human-readable rule descriptions for the prize's round."""
        rules = [
            "Previous winners are excluded (anti-duplication).",
            f"Lottery person: {prize.associated_drawer or 'None'}.",
        ]
        if _is_drawer(prize, DRAWER_LI_BIN):
            rules.append("Relaxed rule (Li Bin): avoidance groups are not applied.")
        elif prize.avoid_groups:
            rules.append(f"Avoidance groups excluded: {', '.join(prize.avoid_groups)}.")
        else:
            rules.append("No avoidance groups.")

        if _is_drawer(prize, DRAWER_MA_JIAQI):
            if self.is_fallback_active():
                rules.append(
                    "Cheng Ru fallback ACTIVE: only Cheng Ru Group members are eligible "
                    "(avoidance groups overridden)."
                )
            else:
                rules.append(
                    "Cheng Ru fallback: if neither of the last two Chen Qixian/Jin Duo "
                    "rounds had a Cheng Ru Group winner, only Cheng Ru Group members "
                    "can win."
                )
        return rules

"""Winner recording, summarizing and CSV export."""

import csv
import os
from dataclasses import dataclass
from datetime import date, datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_order: int
    winning_time: datetime

    @property
    def prize_name(self):
        return self.prize.name

    @property
    def prize_level(self):
        return self.prize.level


class ResultManager:
    """Keeps the winner list and produces summaries and exports."""

    def __init__(self):
        self._winners = []
        self._counter = 0

    def add_winner(self, user, prize):
        """Record a winner for a prize and return the Winner record."""
        self._counter += 1
        winner = Winner(user, prize, self._counter, datetime.now())
        self._winners.append(winner)
        return winner

    def get_all_winners(self):
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes):
        """Return {prize name: [Winner, ...]} sorted by winning order."""
        result = {}
        for prize in prizes:
            matched = [winner for winner in self._winners if winner.prize.name == prize.name]
            result[prize.name] = sorted(matched, key=lambda winner: winner.winning_order)
        return result

    def get_ordered_winners(self, prizes):
        """Return winners ordered by prize level (high to low), then winning order."""
        grouped = self.get_winners_by_prize(prizes)
        ordered = []
        for prize in sorted(prizes, key=lambda item: item.level):
            ordered.extend(grouped.get(prize.name, []))
        return ordered

    def export_to_csv(self, prizes):
        """Export the ordered winner list to current_dir/awards_result_YYYYMMDD.txt.

        Returns (success, message).
        """
        try:
            filename = f"awards_result_{date.today().strftime('%Y%m%d')}.txt"
            path = os.path.join(os.getcwd(), filename)
            rows = self.get_ordered_winners(prizes)
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(
                    ["Prize Level", "Prize Name", "Winning Order", "Name", "Department", "Winning Time"]
                )
                for winner in rows:
                    writer.writerow(
                        [
                            winner.prize_level,
                            winner.prize_name,
                            winner.winning_order,
                            winner.user.name,
                            winner.user.department,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from lottery_system.services.list_manager import ListManager

def _ordinal(number):
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive command-line controller."""

    MENU = """
========== Team Building Smart Lottery ==========
1. Load built-in default team list
2. Upload custom txt list
3. View department groups
4. Prize configuration
5. Start lottery
6. View winner list
7. Export winner list (txt/CSV)
0. Exit
"""

    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.list_loaded = False
        self.result_manager = None
        self.last_prizes = []

    # ------------------------------------------------------------------ main
    def run(self):
        print("Welcome to the Team Building Smart Lottery System.")
        while True:
            print(self.MENU)
            choice = input("Select an option: ").strip()
            if choice == "1":
                self.load_default_list()
            elif choice == "2":
                self.load_custom_list()
            elif choice == "3":
                self.view_groups()
            elif choice == "4":
                self.prize_menu()
            elif choice == "5":
                self.start_lottery()
            elif choice == "6":
                self.view_results()
            elif choice == "7":
                self.export_results()
            elif choice == "0":
                print("Goodbye!")
                break
            else:
                print("Invalid option, please try again.")

    # ------------------------------------------------------------ list menu
    def load_default_list(self):
        users, errors = self.list_manager.load_default_list()
        self._show_errors(errors)
        self._apply_users(users)

    def load_custom_list(self):
        path = input("Enter the path of the txt file: ").strip().strip('"')
        users, errors = self.list_manager.load_custom_list(path)
        if errors:
            print("Validation found the following issues:")
            self._show_errors(errors)
            if not users:
                print("No valid entries found. Upload terminated.")
                return
            answer = input(
                f"Remove invalid entries and continue with {len(users)} valid entries? "
                "(y = continue / n = terminate upload): "
            ).strip().lower()
            if answer != "y":
                print("Upload terminated. Current list unchanged.")
                return
        self._apply_users(users)

    def _apply_users(self, users):
        self.users = list(users)
        self.list_loaded = True
        print(f"List loaded successfully with {len(self.users)} valid entries.")

    def view_groups(self):
        if not self.list_loaded:
            print("Please load a list first.")
            return
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No department groups found.")
            return
        print("Department groups:")
        for department, count in info.items():
            print(f"  {department}: {count} people")

    # ----------------------------------------------------------- prize menu
    def prize_menu(self):

    def _print_prizes(self):

        for index, prize in enumerate(prizes, start=1):

    def _add_prize_interactive(self):
        name = input("Prize name: ").strip()
        quantity_text = input("Quantity: ").strip()
        try:
            quantity = int(quantity_text)
        except ValueError:
            print("Error: quantity must be an integer.")
            return
        groups_text = input("Avoidance groups (comma-separated, empty for none): ")
        groups = [group.strip() for group in groups_text.split(",") if group.strip()]

                print(f"Warning: these groups are not in the current list: {', '.join(unknown)}")
        drawer = input("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:

    def _reorder_interactive(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes to reorder.")
            return
        self._print_prizes()
        raw = input("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ").split()
        try:
            order = [int(item) for item in raw]
        except ValueError:
            print("Invalid input: please enter integers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery round order updated.")
            self._print_prizes()
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # -------------------------------------------------------------- lottery
    def start_lottery(self):
        if not self.list_loaded or not self.users:
            print("Please load a valid employee list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Please configure prizes first.")
            return
        if self.result_manager and self.result_manager.get_all_winners():
            answer = input("Previous results will be replaced. Continue? (y/n): ").strip().lower()
            if answer != "y":
                return

        lottery = LotteryManager(self.users)
        results = ResultManager()
        self.result_manager = results
        self.last_prizes = prizes

        print("\n========== Lottery Started ==========")
        print("Commands: [Enter] draw next | 'pause' | 'terminate' (skip current round)")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, prize, lottery, results)
        print("\n========== All rounds completed ==========")
        self.show_summary(prizes)

    def _run_round(self, index, prize, lottery, results):
        print(f"\n----- Round {index}: {prize.name} -----")
        print(f"Quantity: {prize.quantity}")
        print("Rules:")
        for rule in lottery.describe_rules(prize):
            print(f"  - {rule}")

        round_winners = []
        for position in range(1, prize.quantity + 1):
            if not lottery.get_eligible_pool(prize):
                print("No eligible candidates remain for this prize. Round ends early.")
                break
            print(f"Currently drawing the {_ordinal(position)} winner...")
            command = self._await_command()
            if command == "terminate":
                print("Current round terminated by operator. Skipping to next round.")
                break
            winner = lottery.draw_one(prize)
            if winner is None:
                print("No eligible candidates remain for this prize. Round ends early.")
                break
            results.add_winner(winner, prize)
            round_winners.append(winner)
            print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}")

        lottery.finish_round(prize, round_winners)

        if round_winners:
            print(f"Round {index} results:")
            for order, winner in enumerate(round_winners, start=1):
                print(f"  {order}. {winner.name} ({winner.department})")
        else:
            print(f"Round {index} produced no winners.")

    def _await_command(self):
        """Wait for operator input. Returns 'draw' or 'terminate'."""
        while True:
            command = input("[Enter] draw next | 'pause' | 'terminate': ").strip().lower()
            if command == "":
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = input("(paused) > ").strip().lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if resume == "terminate":
                        return "terminate"
                    print("Please type 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    # -------------------------------------------------------------- results
    def show_summary(self, prizes):
        if not self.result_manager:
            print("No lottery results yet.")
            return
        winners = self.result_manager.get_ordered_winners(prizes)
        print("\n========== Winner List ==========")
        if not winners:
            print("No winners.")
            return
        for winner in winners:
            print(
                f"[{winner.prize_name}] {winner.user.name} ({winner.user.department}) "
                f"- winning order #{winner.winning_order}"
            )

    def view_results(self):
        if not self.result_manager or not self.result_manager.get_all_winners():
            print("No results yet. Run a lottery first.")
            return
        self.show_summary(self.last_prizes)

    def export_results(self):
        if not self.result_manager or not self.result_manager.get_all_winners():
            print("No results to export. Run a lottery first.")
            return
        _, message = self.result_manager.export_to_csv(self.last_prizes)
        print(message)

    # ---------------------------------------------------------------- utils
    @staticmethod
    def _show_errors(errors):
        if errors:
            print("Issues detected:")
            for error in errors:
                print(f"  - {error}")


def main():

"""Core data models."""

from __future__ import annotations

import re
from dataclasses import dataclass, field

_LEVEL_PATTERN = re.compile(r"^\s*(First|Second|Third)\b", re.IGNORECASE)
_LEVEL_VALUES = {"first": 1, "second": 2, "third": 3}
_UNRANKED_LEVEL = 4


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize in the lottery library.

    avoid_groups: department groups whose members are excluded for this prize.
    associated_drawer: the lottery person who draws this prize (or None).
    """

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    @property
    def level(self) -> int:
        """Prize level derived from the name: First=1 (highest) ... others=4."""
        match = _LEVEL_PATTERN.match(self.name)
        if match:
            return _LEVEL_VALUES[match.group(1).lower()]
        return _UNRANKED_LEVEL

from __future__ import annotations

import os

from lottery_system.config import DEFAULT_PARTICIPANTS
from lottery_system.models import User


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in team building list and no errors."""
        users = [User(name, department) for name, department in DEFAULT_PARTICIPANTS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse the file at path. Returns (valid users, format error messages)."""
        users: list[User] = []
        errors: list[str] = []
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.readlines()
        except UnicodeDecodeError:
            return [], ["Format Error: file is not valid UTF-8 text"]
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        for line_number, raw in enumerate(lines, start=1):
            content = raw.strip()
            if not content:
                continue
            parts = content.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_number}: '{content}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid users keeping first occurrence, duplicate entries)."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = " ".join(user.name.split()).casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to number of people."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.config import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered prize library. Starts empty."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        for prize in self._prizes:
            if prize.name == name:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns error messages (empty on success)."""
        errors: list[str] = []
        for spec in DEFAULT_PRIZES:
            if self._find(spec["name"]) is not None:
                errors.append(f"Prize already exists, skipped: {spec['name']}")
                continue
            self._prizes.append(
                Prize(
                    name=spec["name"],
                    quantity=spec["quantity"],
                    avoid_groups=list(spec["avoid_groups"]),
                    associated_drawer=spec["associated_drawer"],
                )
            )
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a custom prize. Raises ValueError on invalid input."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name must not be empty.")
        if self._find(name) is not None:
            raise ValueError(f"Prize already exists: {name}")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(
            Prize(name=name, quantity=quantity, avoid_groups=groups, associated_drawer=drawer)
        )

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order holds 1-based current positions in the new order.

        Returns False (and leaves the library unchanged) if order is not a
        permutation of 1..n.
        """
        count = len(self._prizes)
        try:
            normalized = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(normalized) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in normalized]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and winner tracking."""

from __future__ import annotations

import random

from lottery_system.config import CHEN_QIXIAN, CHENG_RU_GROUP, JIN_DUO, LI_BIN, MA_JIAQI
from lottery_system.models import Prize, User


def normalize_text(text: str) -> str:
    """Normalize names/groups for comparison (whitespace and case insensitive)."""
    return " ".join(text.split()).casefold()


def is_cheng_ru_member(user: User) -> bool:
    """True if the user belongs to the Cheng Ru group."""
    return normalize_text(user.department) == normalize_text(CHENG_RU_GROUP)


class LotteryManager:
    """Runs the eligibility rules and records winners in real time."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # Whether each recent Chen Qixian / Jin Duo round contained a Cheng Ru member.
        # True = contained a Cheng Ru member; only the last two results are kept.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.Random()

    def has_won(self, user: User) -> bool:
        """True if the person has already won any prize."""
        return user.name in self._winner_names

    def is_fallback_active(self) -> bool:
        """Fallback applies when the last two draw results both lack Cheng Ru members."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the users eligible to win the given prize right now."""
        pool = [user for user in self._users if not self.has_won(user)]
        drawer = prize.associated_drawer

        if drawer == LI_BIN:
            # Special rule: only previous winners are filtered out.
            return pool

        if drawer == MA_JIAQI and self.is_fallback_active():
            # Fallback: forcibly limit to Cheng Ru group members, overriding avoidance groups.
            return [user for user in pool if is_cheng_ru_member(user)]

        avoided = {normalize_text(group) for group in prize.avoid_groups}
        return [user for user in pool if normalize_text(user.department) not in avoided]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_names.add(winner.name)
        return winner

    def complete_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Update fallback state after a round has finished."""
        drawer = prize.associated_drawer
        if drawer in (CHEN_QIXIAN, JIN_DUO):
            contains_cheng_ru = any(is_cheng_ru_member(user) for user in round_winners)
            self.last_two_draw_results.append(contains_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == MA_JIAQI and self.is_fallback_active():
            # The fallback protection has been consumed by this round.
            self.last_two_draw_results = []

"""Winner recording, grouping and CSV-style export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.config import EXPORT_FILE_PREFIX
from lottery_system.models import Prize, User

TIME_FORMAT = "%Y-%m-%d %H:%M:%S"


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    prize_level: int
    winning_order: int
    winning_time: datetime


class ResultManager:
    """Stores all winners in winning order."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for a prize. Returns the created record."""
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=prize.level,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map each prize name to its winners, ordered by winning order."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt.

        Rows are ordered by prize level (high to low), then winning order.
        """
        now = datetime.now()
        file_path = os.path.join(os.getcwd(), f"{EXPORT_FILE_PREFIX}{now:%Y%m%d}.txt")
        grouped = self.get_winners_by_prize(prizes)
        ordered_prizes = sorted(prizes, key=lambda prize: prize.level)
        try:
            with open(file_path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize in ordered_prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow(
                            [
                                winner.user.name,
                                winner.user.department,
                                winner.prize_name,
                                winner.winning_time.strftime(TIME_FORMAT),
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {file_path}"

def ordinal(number: int) -> str:
    """Return an English ordinal, e.g. 1 -> '1st'."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def parse_numbers(text: str) -> list[int] | None:
    """Parse whitespace/comma separated integers. Returns None if invalid."""
    tokens = text.replace(",", " ").split()
    if not tokens:
        return None
    try:
        return [int(token) for token in tokens]
    except ValueError:
        return None


def describe_rule(prize: Prize) -> str:
    """Human readable rule description for a prize round."""
    parts: list[str] = []
    drawer = prize.associated_drawer
    if drawer == LI_BIN:
        parts.append(
            "Only previous winners are excluded; avoidance groups are NOT applied (Li Bin rule)."
        )
    else:
        parts.append("Previous winners are excluded.")
        if prize.avoid_groups:
            parts.append(f"Members of avoided groups are excluded: {', '.join(prize.avoid_groups)}.")
    if drawer == MA_JIAQI:
        parts.append(
            "Fallback rule: if the last two Chen Qixian/Jin Duo draws both had no Cheng Ru Group "
            "member, only Cheng Ru Group members are eligible (overrides avoidance groups)."
        )
    return " ".join(parts)

            elif choice == "3":

            elif choice == "4":

                return

    # ----- list management -----------------------------------------------
    def _menu_load_list(self) -> None:
        print("\nLoad participant list:")
        print("  1. Built-in default team building list")
        print("  2. Upload custom txt list (format: Name,Department Group per line)")
        print("  0. Back")

        if choice == "1":
            users, errors = self.list_manager.load_default_list()

            f"Duplicate Name: '{user.name}' ({user.department}) - duplicate entry"

            return valid

        print("\nData validation found problems:")
        for problem in problems:
            print(f"  - {problem}")

        while True:
            choice = self._ask(
                f"Enter 'r' to remove invalid data and keep {len(valid)} valid entries, "
                "or 't' to terminate upload: "
            ).lower()
            if choice == "r":
                if not valid:
                    print("No valid entries remain.")

    def _run_round(
        self,
        round_number: int,
        prize: Prize,
        lottery: LotteryManager,
        results: ResultManager,
    ) -> list[User]:
        drawer = prize.associated_drawer or "None"
        print(f"\n=== Round {round_number}: {prize.name} ===")
        print(f"Quantity: {prize.quantity} | Lottery person: {drawer}")
        print(f"Rule: {describe_rule(prize)}")
        if prize.associated_drawer == MA_JIAQI and lottery.is_fallback_active():
            print("Fallback mechanism ACTIVE: only Cheng Ru Group members are eligible this round.")

        round_winners: list[User] = []
        paused = False
        while len(round_winners) < prize.quantity:
            if paused:
                prompt = "[PAUSED] Type 'continue' to resume or 'terminate' to end this round: "
            else:
                prompt = "Press Enter to draw, or type 'pause' / 'terminate': "
            command = self._ask(prompt).lower()

            if command == "terminate":
                print("Current round terminated and skipped.")
                break
            if command == "pause":
                if paused:
                    print("Lottery is already paused.")
                else:
                    paused = True
                    print("Lottery paused. Type 'continue' to resume.")
                continue
            if command == "continue":
                if paused:
                    paused = False
                    print("Lottery resumed.")
                else:
                    print("Lottery is not paused.")
                continue
            if paused:
                print("Lottery is paused. Type 'continue' to resume or 'terminate' to skip.")
                continue
            if command not in ("", "draw"):
                print("Unknown command.")
                continue

            position = len(round_winners) + 1
            print(f"Currently drawing the {ordinal(position)} winner...")
            winner = lottery.draw_one(prize)
            if winner is None:
                print("No eligible candidates remain for this prize. Round ends early.")
                break
            results.add_winner(winner, prize)
            round_winners.append(winner)
            print(
                f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}!"
            )

        lottery.complete_round(prize, round_winners)
        print(f"Round {round_number} finished: {len(round_winners)} winner(s).")
        for index, winner in enumerate(round_winners, start=1):
            print(f"  {index}. {winner.name} ({winner.department})")
        return round_winners

    def _show_summary(self, prizes: list[Prize]) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n=== Complete Winner List (by prize level, then winning order) ===")
        for prize in sorted(prizes, key=lambda item: item.level):
            print(f"\n{prize.name}:")
            winners = grouped.get(prize.name, [])
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  {winner.winning_order}. {winner.user.name} ({winner.user.department})"
                )

        answer = self._ask("\nExport winner list to txt file? (y/n): ").lower()
        if answer in ("y", "yes"):
            _, message = self.result_manager.export_to_csv(prizes)
            print(message)

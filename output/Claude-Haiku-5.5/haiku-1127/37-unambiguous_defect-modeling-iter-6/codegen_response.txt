# Accepted separators between name and department in a custom txt list.
DELIMITERS: tuple[str, ...] = (",", "\t", "|")

# Encodings tried, in order, when reading a custom txt list.
ENCODINGS: tuple[str, ...] = ("utf-8-sig", "gbk")


def _name_key(name: str) -> str:
    return name.strip().casefold()


class ListManager:
    """Loads, verifies and summarises participant lists."""

    def __init__(self) -> None:
        self.load_mode: str | None = None
        self.source_file_path: str | None = None

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        self.load_mode = "default"
        self.source_file_path = None
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list uploaded by the user."""
        self.load_mode = "custom"
        self.source_file_path = str(path)
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        """Read and parse a txt file. Returns (users, errors)."""
        clean_path = os.path.expanduser(str(path).strip().strip('"').strip("'"))
        if not os.path.isfile(clean_path):
            return [], [f"File Error: file not found: {clean_path}"]

        raw_text = None
        for encoding in ENCODINGS:
            try:
                with open(clean_path, "r", encoding=encoding) as handle:
                    raw_text = handle.read()
                break
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"File Error: cannot read {clean_path}: {exc}"]

        if raw_text is None:
            return [], [f"File Error: unsupported text encoding in {clean_path}"]

        return self._parse_lines(raw_text.splitlines())

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw_line in enumerate(lines, start=1):
            text = raw_line.strip()
            if not text or text.startswith("#"):
                continue
            user = self._parse_line(text)
            if user is None:
                errors.append(
                    f"Line {line_no}: Format Error - '{text}' "
                    f"(expected 'Name,Department')"
                )
            else:
                users.append(user)
        return users, errors

    @staticmethod
    def _parse_line(text: str) -> User | None:
        for delimiter in DELIMITERS:
            if delimiter in text:
                name, _, department = text.partition(delimiter)
                name = name.strip()
                department = department.strip()
                if name and department and delimiter not in department:
                    return User(name=name, department=department)
                return None
        return None

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Split users into (valid_users, duplicates).

        The first occurrence of a name is kept as valid; later occurrences
        with the same name are reported as duplicates.
        """
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = _name_key(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users) -> dict:
        """Return a mapping of department name to number of members."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

def _clean_groups(groups) -> list[str]:
    """Strip, drop empties and remove duplicates (case-insensitive) keeping order."""
    if not groups:
        return []
    if isinstance(groups, str):
        groups = groups.split(",")
    cleaned: list[str] = []
    seen: set[str] = set()
    for group in groups:
        text = str(group).strip()
        if not text:
            continue
        key = text.casefold()
        if key not in seen:
            seen.add(key)
            cleaned.append(text)
    return cleaned


class PrizeManager:
    """Keeps the ordered list of prizes configured for the event."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for preset in PRESET_PRIZES:
            if self._find(preset["name"]) is not None:
                errors.append(f"Prize already exists, skipped: {preset['name']}")
                continue
            self.add_prize(
                preset["name"],
                preset["quantity"],
                preset["avoid_groups"],
                preset["associated_drawer"],
            )
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        """Add a custom or preset prize at the end of the sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize already exists: {clean_name}")

        drawer = None
        if associated_drawer and str(associated_drawer).strip():
            drawer = str(associated_drawer).strip()

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=_clean_groups(avoid_groups),
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def set_avoid_groups(self, name: str, groups) -> bool:
        """Replace the avoidance groups of the named prize."""
        prize = self._find(name)
        if prize is None:
            return False
        prize.avoid_groups = _clean_groups(groups)
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. ``order`` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            normalized = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(normalized) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in normalized]
        return True

    def _find(self, name: str) -> Prize | None:
        key = name.strip().casefold()
        for prize in self._prizes:
            if prize.name.casefold() == key:
                return prize
        return None

_LEVEL_KEYWORDS: tuple[tuple[str, int], ...] = (
    ("first", 1),
    ("second", 2),
    ("third", 3),
)
_LEVEL_LABELS: dict[int, str] = {1: "First", 2: "Second", 3: "Third"}
_OTHER_LEVEL = len(_LEVEL_KEYWORDS) + 1

EXPORT_HEADER: list[str] = [
    "Prize Level",
    "Prize Name",
    "Name",
    "Department",
    "Winning Order",
    "Win Time",
]


def prize_level(prize_name: str) -> int:
    """Return the level rank of a prize (1 = highest). Unknown levels rank last."""
    lowered = prize_name.casefold()
    for keyword, rank in _LEVEL_KEYWORDS:
        if re.search(rf"\b{keyword}\b", lowered):
            return rank
    return _OTHER_LEVEL


def sort_prizes_by_level(prizes) -> list[Prize]:
    """Sort prizes from high level to low level, keeping sequence for ties."""
    return sorted(prizes, key=lambda prize: prize_level(prize.name))


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize: Prize
    winning_order: int
    win_time: datetime = field(default_factory=datetime.now)

    @property
    def prize_name(self) -> str:
        return self.prize.name


class ResultManager:
    """Maintains the winner list and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(user=user, prize=prize, winning_order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict:
        """Map prize name to its winners (winning order), prizes sorted high to low."""
        ordered = sort_prizes_by_level(prizes)
        result: dict[str, list[Winner]] = {prize.name: [] for prize in ordered}
        for winner in self._winners:
            if winner.prize.name in result:
                result[winner.prize.name].append(winner)
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export all winners to ``awards_result_YYYYMMDD.txt`` in the current directory."""
        file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(EXPORT_HEADER)
                for prize_name, winners in grouped.items():
                    level_label = _LEVEL_LABELS.get(prize_level(prize_name), "Other")
                    for winner in winners:
                        writer.writerow([
                            level_label,
                            prize_name,
                            winner.user.name,
                            winner.user.department,
                            winner.winning_order,
                            winner.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""
from __future__ import annotations

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager, Winner

SPECIAL_DRAWER = "Li Bin"
FALLBACK_WATCH_DRAWERS: tuple[str, ...] = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def _same(a: str | None, b: str | None) -> bool:
    if a is None or b is None:
        return False
    return a.strip().casefold() == b.strip().casefold()


def _key(name: str) -> str:
    return name.strip().casefold()


class LotteryManager:
    """Runs lottery draws for a list of users while enforcing all rules."""

    def __init__(self, users) -> None:
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Results (True = contained Cheng Ru group member) of the most recent
        # draws by Chen Qixian / Jin Duo, oldest first, at most two entries.
        self.last_two_draw_results: list[bool] = []
        self._winner_keys: set[str] = set()
        self._rng = random.Random()

    # ----- winner bookkeeping -------------------------------------------------
    def is_winner(self, user: User) -> bool:
        return _key(user.name) in self._winner_keys

    def _available_users(self) -> list[User]:
        return [user for user in self.users if not self.is_winner(user)]

    # ----- fallback protection ------------------------------------------------
    def is_fallback_active(self) -> bool:
        """True when the last two Chen/Jin draws both had no Cheng Ru group member."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _fallback_applies(self, prize: Prize) -> bool:
        return _same(prize.associated_drawer, FALLBACK_TARGET_DRAWER) and self.is_fallback_active()

    def _protected_pool(self) -> list[User]:
        return [
            user
            for user in self._available_users()
            if _same(user.department, PROTECTED_GROUP)
        ]

    # ----- rule engine --------------------------------------------------------
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = self._available_users()

        # Special lottery person: relaxed rules (avoidance groups not applied).
        if _same(prize.associated_drawer, SPECIAL_DRAWER):
            return available

        # Fallback: Ma Jiaqi's prize is limited to Cheng Ru group members,
        # overriding avoidance groups.
        if self._fallback_applies(prize):
            protected = self._protected_pool()
            if protected:
                return protected

        # Basic filtering: exclude avoidance groups.
        avoid = prize.avoid_groups or []
        return [
            user
            for user in available
            if not any(_same(user.department, group) for group in avoid)
        ]

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        rules = ["previous winners are excluded"]
        if _same(prize.associated_drawer, SPECIAL_DRAWER):
            rules.append("relaxed rules for Li Bin: avoidance groups are not applied")
        elif self._fallback_applies(prize) and self._protected_pool():
            rules.append(
                "Cheng Ru group protection: only Cheng Ru Group members are eligible "
                "(avoidance groups overridden)"
            )
        elif prize.avoid_groups:
            rules.append("avoided groups: " + ", ".join(prize.avoid_groups))
        return "; ".join(rules)

    # ----- drawing ------------------------------------------------------------
    def draw_next(self, prize: Prize) -> Winner | None:
        """Draw one winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        winner = self.result_manager.add_winner(user, prize)
        self._winner_keys.add(_key(user.name))
        return winner

    def record_round_result(self, prize: Prize, winners: list[Winner]) -> None:
        """Update fallback state after a lottery round finishes."""
        drawer = prize.associated_drawer
        if _same(drawer, FALLBACK_TARGET_DRAWER):
            # The protection has been consumed by this round; restart the sequence.
            self.last_two_draw_results = []
            return
        if any(_same(drawer, watched) for watched in FALLBACK_WATCH_DRAWERS):
            has_protected = any(
                _same(winner.user.department, PROTECTED_GROUP) for winner in winners
            )
            self.last_two_draw_results = (self.last_two_draw_results + [has_protected])[-2:]

def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive command-line application."""

    # ----- input helpers ------------------------------------------------------
    @staticmethod
    def _ask(prompt: str = "") -> str:
        try:
            return input(prompt).strip()
        except (EOFError, KeyboardInterrupt):
            print()
            raise SystemExit(0)

    def _ask_int(self, prompt: str, minimum: int = 1) -> int:
        while True:
            raw = self._ask(prompt)
            try:
                value = int(raw)
            except ValueError:
                print("Please enter a valid integer.")
                continue
            if value < minimum:
                print(f"Please enter a number >= {minimum}.")
                continue
            return value

    def _ask_groups(self, prompt: str) -> list[str]:
        raw = self._ask(prompt)
        groups = [item.strip() for item in raw.split(",") if item.strip()]
        known = {user.department.casefold() for user in self.users}
        if self.users:
            for group in groups:
                if group.casefold() not in known:
                    print(f"Warning: group '{group}' does not exist in the current list.")
        return groups

        print("=" * 60)
        print("Company Team Building - Smart Lottery System")
        print("=" * 60)

            for error in self.prize_manager.load_default_prizes():
                print(error)
            print("Preset prizes loaded.")

        while True:
            self._print_main_menu()
            choice = self._ask("Select an option: ")
            if choice == "1":
                self.load_participants()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.manage_prizes()
            elif choice == "4":
                self.adjust_sequence()
            elif choice == "5":
                self.start_lottery()
            elif choice == "6":
                self.show_results()
            elif choice == "7":
                self.export_results()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes()
        index = self._ask_int("Prize number: ", 1)
        if index > len(prizes):
            print("Prize number out of range.")
            return
        groups = self._ask_groups("Avoidance groups (comma separated, blank for none): ")
        prize = prizes[index - 1]
        self.prize_manager.set_avoid_groups(prize.name, groups)
        print(f"Avoidance groups updated for '{prize.name}': {', '.join(groups) or 'None'}")

    def adjust_sequence(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            print("At least two prizes are needed to adjust the sequence.")
            return
        self._print_prizes()
        raw = self._ask("Enter new order as serial numbers, e.g. '2 1 3' (blank to cancel): ")
        if not raw:
            return
        try:
            order = [int(item) for item in raw.replace(",", " ").split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
            self._print_prizes()
        else:
            print(f"Invalid order: must contain each number 1-{len(prizes)} exactly once.")

    # ----- lottery process ----------------------------------------------------
    def start_lottery(self) -> None:
        if not self.users:
            print("Please load the participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is not None:
            print("A lottery has already been run in this session. Showing results instead.")
            self.show_results()
            return

        self.lottery = LotteryManager(self.users)
        print()
        print("Lottery started. Commands: Enter = draw next, 'pause', 'terminate' (end round).")
        total = len(prizes)
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)
        print()
        print("All lottery rounds completed!")
        self.show_results()

    def _run_round(self, index: int, total: int, prize: Prize) -> None:
        assert self.lottery is not None
        print()
        print(f"=== Round {index}/{total}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")
        print(f"Lottery person: {prize.associated_drawer or 'Not specified'}")
        print(f"Rule: {self.lottery.describe_rules(prize)}")

        round_winners: list[Winner] = []
        drawn = 0
        while drawn < prize.quantity:
            command = self._ask("Press Enter to draw next winner (pause / terminate): ").casefold()
            if command == "pause":
                if not self._wait_for_resume():
                    print("Round terminated, skipping to the next round.")
                    break
            elif command == "terminate":
                print("Round terminated, skipping to the next round.")
                break
            elif command not in ("", "continue"):
                print("Unknown command. Use Enter, 'pause' or 'terminate'.")
                continue

            print(f"Currently drawing the {_ordinal(drawn + 1)} winner...")
            winner = self.lottery.draw_next(prize)
            if winner is None:
                print("No eligible participants remain for this prize; round ends.")
                break
            round_winners.append(winner)
            drawn += 1
            user = winner.user
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        self.lottery.record_round_result(prize, round_winners)
        print(f"Round {index} finished: {len(round_winners)} of {prize.quantity} winner(s) drawn.")

    @staticmethod
    def _wait_for_resume() -> bool:
        """Block while paused. Returns True to resume, False to terminate the round."""
        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = LotteryCLI._ask("> ").casefold()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Lottery is paused. Type 'continue' or 'terminate'.")

    # ----- results ------------------------------------------------------------
    def show_results(self) -> None:
        if self.lottery is None:
            print("No lottery results yet.")
            return
        prizes = sort_prizes_by_level(self.prize_manager.get_prizes())
        grouped = self.lottery.result_manager.get_winners_by_prize(prizes)
        print()
        print("==================== Winner List ====================")
        for prize_name, winners in grouped.items():
            print(f"[{prize_name}]")
            if not winners:
                print("  (no winners)")
                continue
            for position, winner in enumerate(winners, start=1):
                user = winner.user
                print(f"  {position}. {user.name} ({user.department})")
        print("=====================================================")

    def export_results(self) -> None:
        if self.lottery is None:
            print("No lottery results to export.")
            return
        _, message = self.lottery.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(message)


def main() -> None:
    """Application entry point."""
    LotteryCLI().run()

"""Global variable store shared by all simulator modules.

The store is a plain dictionary wrapped by ``set_value`` / ``get_value`` so
that every module reads and writes the same system state (funds, rider list,
rider ID counter, running flag, simulated clock, configuration parameters).
"""

from typing import Any, Dict

_store: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store to the default system parameters.

    Defaults:
        funds: 0 (may become negative).
        riders: empty list of Rider objects.
        rider_id_counter: 0.
        running_state: True.
        simulated_time: 0 ticks.
        orders: empty list of Order objects.
        next_order_id: 1.
        order_fee: 100 funds charged per accepted order.
        recruit_threshold: 300 funds needed to trigger recruitment.
        recruit_cost: 300 funds deducted for each recruited rider.
        rider_speed: 50 units per tick.
        max_ticks: 10000 safety cap for ticks spent on one order.
    """
    _store.clear()
    _store.update(
        {
            "funds": 0,
            "riders": [],
            "rider_id_counter": 0,
            "running_state": True,
            "simulated_time": 0,
            "orders": [],
            "next_order_id": 1,
            "order_fee": 100,
            "recruit_threshold": 300,
            "recruit_cost": 300,
            "rider_speed": 50,
            "max_ticks": 10000,
        }
    )


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Any Python object to store.
    """
    _store[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key`` or ``default`` when missing.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default``.
    """
    return _store.get(key, default)

"""Grid-based path planning for the 800x800 map (8x8 cells of 100 units).

Cell centers sit at ``50 + 100 * k``. The street network runs through these
center lines, so every grid node lies on a street and every movement is an
axis-aligned (horizontal or vertical) step with right-angle turns.
"""

from typing import Iterable, List, Optional, Sequence, Set, Tuple

Point = Tuple[int, int]

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = 8
CELL_HALF = CELL_SIZE // 2


def cell_center(value: int) -> int:
    """Convert a 0-800 coordinate into the center of its grid cell.

    Implements ``center = (value // 100) * 100 + 50``. Values at or beyond
    the map edge (e.g. 800) are clamped into the last cell.

    Args:
        value: Integer coordinate in the range 0-800.

    Returns:
        The cell-center coordinate along that axis.
    """
    index = int(value) // CELL_SIZE
    index = min(max(index, 0), GRID_DIM - 1)
    return index * CELL_SIZE + CELL_HALF


def snap_to_grid(x: int, y: int) -> Point:
    """Snap an (x, y) map coordinate to its grid cell center.

    Args:
        x: Horizontal coordinate (0-800).
        y: Vertical coordinate (0-800).

    Returns:
        Tuple ``(center_x, center_y)``.
    """
    return cell_center(x), cell_center(y)


def manhattan(a: Point, b: Point) -> int:
    """Return the Manhattan (street-network) distance between two points.

    Args:
        a: First point.
        b: Second point.

    Returns:
        ``|ax - bx| + |ay - by|``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def is_street_point(x: int, y: int) -> bool:
    """Return True when the point lies on a street (grid line).

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.

    Returns:
        True if x or y aligns with a street center line.
    """
    return (x - CELL_HALF) % CELL_SIZE == 0 or (y - CELL_HALF) % CELL_SIZE == 0


def path_length(start: Point, waypoints: Sequence[Point]) -> int:
    """Compute the total Manhattan length of a waypoint sequence.

    Args:
        start: Starting point.
        waypoints: Ordered waypoints following ``start``.

    Returns:
        Total length in units.
    """
    total = 0
    current = start
    for point in waypoints:
        total += manhattan(current, point)
        current = point
    return total


class PathPlanner:
    """Plans street-following routes between grid points.

    Rules:
        * Same row or column (directly above/below/left/right of the target):
          a straight-line special case is used.
        * Otherwise a single right-angle turn is used. The planner prefers the
          horizontal-first corner, but when ``avoid_conflicts`` is enabled and
          that corner is occupied by another rider, it switches to the
          vertical-first corner to avoid path conflicts and deadlocks.
    """

    def __init__(self, avoid_conflicts: bool = True) -> None:
        """Create a planner.

        Args:
            avoid_conflicts: Whether to reroute around occupied corners.
        """
        self.avoid_conflicts = avoid_conflicts

    @staticmethod
    def is_special_case(start: Point, goal: Point) -> bool:
        """Return True when the goal is directly above/below/left/right.

        Args:
            start: Current position.
            goal: Target position.

        Returns:
            True for a straight-line (special) movement.
        """
        return start != goal and (start[0] == goal[0] or start[1] == goal[1])

    def plan(
        self,
        start: Point,
        goal: Point,
        occupied: Optional[Iterable[Point]] = None,
    ) -> List[Point]:
        """Plan a route from ``start`` to ``goal``.

        Args:
            start: Current position.
            goal: Destination grid point.
            occupied: Positions currently held by other riders.

        Returns:
            List of waypoints to visit in order, excluding ``start`` and
            ending at ``goal``. Empty when already at the goal.
        """
        if start == goal:
            return []
        sx, sy = start
        gx, gy = goal
        if sx == gx or sy == gy:
            return [goal]

        occupied_set: Set[Point] = set(occupied) if occupied else set()
        horizontal_corner = (gx, sy)
        vertical_corner = (sx, gy)
        corner = horizontal_corner
        if (
            self.avoid_conflicts
            and horizontal_corner in occupied_set
            and vertical_corner not in occupied_set
        ):
            corner = vertical_corner
        return [corner, goal]

    def plan_route(
        self,
        start: Point,
        stops: Sequence[Point],
        occupied: Optional[Iterable[Point]] = None,
    ) -> List[Point]:
        """Plan a continuous multi-destination route through ``stops``.

        Args:
            start: Starting position.
            stops: Ordered destinations (e.g. restaurant then customer).
            occupied: Positions currently held by other riders.

        Returns:
            Concatenated waypoint list for the whole trip.
        """
        occupied_list = list(occupied) if occupied else []
        route: List[Point] = []
        current = start
        for stop in stops:
            leg = self.plan(current, stop, occupied_list)
            route.extend(leg)
            current = stop
        return route

"""Rider model: state machine, grid movement and event logging."""

import logging
from typing import Any, Dict, List, Set

import globalvar
from path_planner import Point, PathPlanner

LOGGER = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}
SPAWN_POINT: Point = (50, 50)


def _clamp_step(delta: int, speed: int) -> int:
    """Limit a signed distance to at most ``speed`` units in magnitude."""
    if delta > 0:
        return min(delta, speed)
    return max(delta, -speed)


class Rider:
    """A delivery rider that moves along streets and serves orders in FIFO order.

    The state machine is:
        idle (0) -> picking up (1) when an order is accepted;
        picking up (1) -> delivering (2) on arrival at the restaurant;
        delivering (2) -> idle (0) on delivery to the customer.

    Every transition and the arrival/acceptance moments are recorded in
    ``self.log`` as dictionaries containing at least the keys ``"event"`` and
    ``"order_id"``.
    """

    def __init__(self, rider_id: int) -> None:
        """Create an idle rider at the spawn point and log the spawn event.

        Args:
            rider_id: Unique rider identifier.
        """
        self.rider_id: int = int(rider_id)
        self.x, self.y = SPAWN_POINT
        self.speed: int = max(1, int(globalvar.get_value("rider_speed", 50)))
        self.state: int = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.path: List[Point] = []
        self.planner = PathPlanner()
        self.distance_traveled: int = 0
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    @property
    def position(self) -> Point:
        """Current ``(x, y)`` position."""
        return (self.x, self.y)

    def _now(self) -> int:
        """Return the current simulated time in ticks."""
        return int(globalvar.get_value("simulated_time", 0))

    def _record(self, event: str, order_id: Any) -> None:
        """Append a status-change entry to the rider log.

        Args:
            event: Event name such as ``"accepted_order"``.
            order_id: Related order ID, or None.
        """
        entry = {
            "time": self._now(),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "state": self.state,
            "x": self.x,
            "y": self.y,
        }
        self.log.append(entry)
        LOGGER.info(
            "rider=%s t=%s event=%s order=%s pos=(%s,%s)",
            self.rider_id,
            entry["time"],
            event,
            order_id,
            self.x,
            self.y,
        )

    def _other_positions(self) -> Set[Point]:
        """Return positions of all other riders, used for conflict avoidance."""
        return {
            rider.position
            for rider in globalvar.get_value("riders", [])
            if rider is not self
        }

    def _target_for_state(self) -> Point:
        """Return the grid point the rider should currently be heading to."""
        order = self.orders[0]
        if self.state == STATE_PICKING_UP:
            return order.res_pos
        return order.cus_pos

    def move(self) -> None:
        """Advance the rider by one simulation tick.

        Handles order acceptance (idle -> picking up), movement along the
        planned route, arrival at the restaurant (picking up -> delivering)
        and delivery to the customer (delivering -> idle).
        """
        if self.state == STATE_IDLE:
            if not self.orders:
                return
            self._accept_order(self.orders[0])
        elif not self.path and self.orders:
            # Recovery path: an active order without a planned leg.
            self._begin_leg(self._target_for_state())
        if self.path:
            self._step()

    def _accept_order(self, order: Any) -> None:
        """Transition to picking up and plan the leg to the restaurant.

        Args:
            order: The order being accepted (first in the rider's queue).
        """
        self.state = STATE_PICKING_UP
        order.status = "picking_up"
        if not order.route:
            order.route.append(self.position)
        self._record("accepted_order", order.id)
        self._begin_leg(order.res_pos)

    def _begin_leg(self, goal: Point) -> None:
        """Plan the route from the current position to ``goal``.

        If the rider is already at the goal, the arrival is processed
        immediately.

        Args:
            goal: Destination grid point.
        """
        here = self.position
        self.special_case = PathPlanner.is_special_case(here, goal)
        self.path = self.planner.plan(here, goal, self._other_positions())
        if not self.path:
            self._on_arrival()

    def _step(self) -> None:
        """Move one tick toward the next waypoint along a street."""
        tx, ty = self.path[0]
        moved = 0
        if self.x != tx:
            delta = _clamp_step(tx - self.x, self.speed)
            self.x += delta
            moved = abs(delta)
        elif self.y != ty:
            delta = _clamp_step(ty - self.y, self.speed)
            self.y += delta
            moved = abs(delta)

        if moved:
            self.distance_traveled += moved
            if self.orders:
                order = self.orders[0]
                order.distance += moved
                order.route.append(self.position)

        if self.position == (tx, ty):
            self.path.pop(0)
            if not self.path:
                self._on_arrival()

    def _on_arrival(self) -> None:
        """Handle reaching the end of the current leg."""
        if not self.orders:
            self.state = STATE_IDLE
            self.special_case = False
            return

        order = self.orders[0]
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.status = "delivering"
            self._record("arrived_at_merchant", order.id)
            self._begin_leg(order.cus_pos)
        elif self.state == STATE_DELIVERING:
            order.mark_completed(self._now())
            self.orders.pop(0)
            self.state = STATE_IDLE
            self.special_case = False
            self._record("delivered_to_customer", order.id)
            if self.orders:
                self._accept_order(self.orders[0])

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serializable snapshot of this rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
            "distance_traveled": self.distance_traveled,
            "log": list(self.log),
        }

"""Order data model and coordinate validation."""

from typing import Any, Dict, List, Optional, Sequence

import globalvar
from path_planner import MAP_SIZE, Point, manhattan, snap_to_grid

MIN_COORD = 0
MAX_COORD = MAP_SIZE


def validate_coordinate(value: Any, name: str = "coordinate") -> int:
    """Validate that a value is an integer within 0-800.

    Args:
        value: Candidate value.
        name: Label used in error messages.

    Returns:
        The validated integer.

    Raises:
        ValueError: If the value is not an integer or is out of range.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError(f"{name} must be an integer, got {value!r}")
    if not MIN_COORD <= value <= MAX_COORD:
        raise ValueError(
            f"{name} {value} is out of range ({MIN_COORD}-{MAX_COORD})"
        )
    return value


def validate_point(point: Sequence[Any], label: str) -> Point:
    """Validate an ``(x, y)`` pair of integers in the 0-800 range.

    Args:
        point: Sequence of exactly two values.
        label: Name of the point used in error messages.

    Returns:
        Tuple ``(x, y)`` of validated integers.

    Raises:
        ValueError: If the sequence is malformed or a value is invalid.
    """
    if len(point) != 2:
        raise ValueError(f"{label} point must have exactly 2 coordinates")
    x = validate_coordinate(point[0], f"{label} x")
    y = validate_coordinate(point[1], f"{label} y")
    return x, y


class Order:
    """A food delivery order from a restaurant to a customer.

    Input coordinates are validated and snapped to grid cell centers. The
    order moves through the statuses: created -> assigned -> picking_up ->
    delivering -> completed.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order and assign it a unique ID.

        Args:
            res_pos: Restaurant ``(x, y)`` in map units (0-800).
            cus_pos: Customer ``(x, y)`` in map units (0-800).

        Raises:
            ValueError: If coordinates are not integers or out of range.
        """
        self.res_input: Point = validate_point(res_pos, "restaurant")
        self.cus_input: Point = validate_point(cus_pos, "customer")
        self.res_pos: Point = snap_to_grid(*self.res_input)
        self.cus_pos: Point = snap_to_grid(*self.cus_input)

        self.id: int = int(globalvar.get_value("next_order_id", 1))
        globalvar.set_value("next_order_id", self.id + 1)
        self.start_time: int = int(globalvar.get_value("simulated_time", 0))
        self.completion_time: Optional[int] = None
        self.distance: float = 0.0
        self.time_taken: Optional[int] = None
        self.efficiency: Optional[float] = None
        self.status: str = "created"
        self.rider_id: Optional[int] = None
        self.route: List[Point] = []

    def assign_to(self, rider_id: int) -> None:
        """Assign this order to a rider by appending it to ``rider.orders``.

        Args:
            rider_id: ID of the rider receiving the order.

        Raises:
            ValueError: If the order is not in the created state or the rider
                does not exist.
        """
        if self.status != "created":
            raise ValueError(f"Order {self.id} is already {self.status}")
        target = None
        for candidate in globalvar.get_value("riders", []):
            if candidate.rider_id == rider_id:
                target = candidate
                break
        if target is None:
            raise ValueError(f"Rider {rider_id} does not exist")
        self.rider_id = rider_id
        self.status = "assigned"
        target.orders.append(self)

    def mark_completed(self, completion_time: int) -> None:
        """Record completion time, time taken and delivery efficiency.

        Efficiency is the direct restaurant-to-customer Manhattan distance
        divided by the total distance the rider travelled for this order.

        Args:
            completion_time: Simulated tick at which delivery finished.
        """
        self.completion_time = completion_time
        self.time_taken = completion_time - self.start_time
        self.status = "completed"
        direct = manhattan(self.res_pos, self.cus_pos)
        self.efficiency = round(direct / self.distance, 3) if self.distance > 0 else 1.0

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serializable snapshot of this order."""
        return {
            "id": self.id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
            "status": self.status,
            "rider_id": self.rider_id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
        }

"""Order creation, tracking and rider assignment logic."""

import logging
from typing import List, Optional

import globalvar
from order_model import Order
from path_planner import manhattan

LOGGER = logging.getLogger(__name__)

LOAD_PENALTY = 200  # Extra distance units charged per order already held.


class OrderManager:
    """Creates orders and keeps track of every order in the system."""

    def create_order(self, res_x: int, res_y: int, cus_x: int, cus_y: int) -> Order:
        """Validate coordinates, create an order and store it globally.

        Args:
            res_x: Restaurant x coordinate (0-800).
            res_y: Restaurant y coordinate (0-800).
            cus_x: Customer x coordinate (0-800).
            cus_y: Customer y coordinate (0-800).

        Returns:
            The newly created Order.

        Raises:
            ValueError: If any coordinate is invalid.
        """
        order = Order((res_x, res_y), (cus_x, cus_y))
        globalvar.get_value("orders").append(order)
        LOGGER.info("created order %s", order.id)
        return order

    def all_orders(self) -> List[Order]:
        """Return every order ever created, in creation order."""
        return list(globalvar.get_value("orders", []))

    def active_orders(self) -> List[Order]:
        """Return orders that are not yet completed."""
        return [o for o in self.all_orders() if o.status != "completed"]

    def completed_orders(self) -> List[Order]:
        """Return completed orders."""
        return [o for o in self.all_orders() if o.status == "completed"]


def find_best_rider_for_order(order: Order) -> Optional[Order]:
    """Choose the best rider for an order (closest rider first, load-balanced).

    Each rider is scored by its Manhattan distance to the restaurant plus a
    penalty of ``LOAD_PENALTY`` per order it already holds. The lowest score
    wins; ties are broken by the lower rider ID.

    Args:
        order: The order needing a rider.

    Returns:
        The selected Rider, or None when no riders exist.
    """
    riders = globalvar.get_value("riders", [])
    best = None
    best_key = None
    for rider in riders:
        distance = manhattan(rider.position, order.res_pos)
        score = distance + LOAD_PENALTY * len(rider.orders)
        key = (score, rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    if best is None:
        LOGGER.warning("no rider available for order %s", order.id)
    return best

"""System-level management: initialization, funds, recruitment, clock and persistence."""

import json
import logging
from typing import Any, Dict, List, Optional

import globalvar
from rider import STATE_IDLE, STATE_PICKING_UP, Rider

LOGGER = logging.getLogger(__name__)

CONFIG_KEYS = (
    "order_fee",
    "recruit_threshold",
    "recruit_cost",
    "rider_speed",
    "max_ticks",
)


def init_system() -> None:
    """Initialize (or reset) all global system parameters to defaults."""
    globalvar._init()
    LOGGER.info("system initialized")


def load_config(path: str) -> Dict[str, int]:
    """Load system parameters from a JSON configuration file.

    Args:
        path: Path to the JSON file.

    Returns:
        Dictionary of parameters that were applied.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the JSON is malformed or a parameter is invalid.
    """
    with open(path, "r", encoding="utf-8") as handle:
        try:
            data = json.load(handle)
        except json.JSONDecodeError as exc:
            raise ValueError(f"invalid JSON in {path}: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError("configuration root must be a JSON object")

    applied: Dict[str, int] = {}
    for key in CONFIG_KEYS:
        if key not in data:
            continue
        value = data[key]
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"config '{key}' must be an integer")
        if key in ("rider_speed", "max_ticks", "recruit_threshold") and value <= 0:
            raise ValueError(f"config '{key}' must be positive")
        globalvar.set_value(key, value)
        applied[key] = value
    LOGGER.info("loaded configuration from %s: %s", path, applied)
    return applied


class SystemManager:
    """Manages system funds, rider recruitment, the simulated clock and persistence.

    All state lives in the global store (see ``globalvar``).
    """

    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return int(globalvar.get_value("funds", 0))

    @property
    def riders(self) -> List[Rider]:
        """The live rider list."""
        return globalvar.get_value("riders", [])

    @property
    def running(self) -> bool:
        """Whether the main loop should keep running."""
        return bool(globalvar.get_value("running_state", True))

    def add_funds(self, amount: int) -> int:
        """Add ``amount`` to the fund balance.

        Args:
            amount: Signed amount to add.

        Returns:
            The new fund balance.
        """
        new_value = self.funds + int(amount)
        globalvar.set_value("funds", new_value)
        return new_value

    def find_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def recruit_rider(self) -> Rider:
        """Create a new idle rider with an auto-incremented ID.

        The recruitment cost is deducted from the funds.

        Returns:
            The newly created Rider.
        """
        counter = int(globalvar.get_value("rider_id_counter", 0)) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        self.add_funds(-int(globalvar.get_value("recruit_cost", 300)))
        LOGGER.info("recruited rider %s (state=%s)", counter, STATE_IDLE)
        return rider

    def check_recruitment(self) -> Optional[Rider]:
        """Recruit a rider automatically when funds reach the threshold.

        Returns:
            The new Rider if one was created, otherwise None.
        """
        threshold = int(globalvar.get_value("recruit_threshold", 300))
        if self.funds >= threshold:
            return self.recruit_rider()
        return None

    def tick(self) -> Optional[Rider]:
        """Advance the simulation by one tick.

        Increments the simulated clock, moves every rider and checks the
        recruitment trigger. A failing rider is recovered rather than
        crashing the whole simulation.

        Returns:
            A newly recruited Rider, or None.
        """
        globalvar.set_value(
            "simulated_time", int(globalvar.get_value("simulated_time", 0)) + 1
        )
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pylint: disable=broad-except

                self._recover_rider(rider)
        return self.check_recruitment()

    @staticmethod
    def _recover_rider(rider: Rider) -> None:
        """Reset a rider's transient movement state after an exception.

        Args:
            rider: The rider to repair. Orders are preserved.
        """
        rider.path = []
        rider.special_case = False
        rider.state = STATE_PICKING_UP if rider.orders else STATE_IDLE

    def stop(self) -> None:
        """Signal the main loop to terminate."""
        globalvar.set_value("running_state", False)

    def snapshot(self) -> Dict[str, Any]:
        """Return a JSON-serializable snapshot of the whole system."""
        return {
            "funds": self.funds,
            "simulated_time": int(globalvar.get_value("simulated_time", 0)),
            "rider_id_counter": int(globalvar.get_value("rider_id_counter", 0)),
            "config": {key: globalvar.get_value(key) for key in CONFIG_KEYS},
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in globalvar.get_value("orders", [])],
        }

            json.dump(self.snapshot(), handle, indent=2)
        LOGGER.info("state saved to %s", path)

"""Status text and ASCII map rendering for the dispatch simulator."""

from typing import Iterable, List, Optional, Sequence

from path_planner import CELL_HALF, CELL_SIZE, GRID_DIM, Point
from rider import STATE_NAMES, Rider


def _round_half_up(value: float) -> int:
    """Round to the nearest integer, halves rounding up."""
    return int(value + 0.5) if value >= 0 else -int(-value + 0.5)


class DisplayManager:
    """Renders the 8x8 map and textual status reports.

    Map layout: every grid cell center is a node drawn at an even
    row/column of a 15x15 character canvas, and streets connecting adjacent
    nodes are drawn at the odd positions in between.

    Symbols:
        R rider, S restaurant, C customer, . empty node, # street,
        * route taken by a rider.
    """

    def __init__(self) -> None:
        """Create the display manager with its symbol legend."""
        self.map_symbols: str = "R=rider S=restaurant C=customer .=empty #=street *=route"
        self._size: int = 2 * GRID_DIM - 1

    def _to_canvas(self, point: Point):
        """Convert a map coordinate into canvas (row, column) indexes."""
        x, y = point
        half = CELL_SIZE // 2
        col = _round_half_up((x - CELL_HALF) / half)
        node_row = _round_half_up((y - CELL_HALF) / half)
        col = min(max(col, 0), self._size - 1)
        row = min(max(self._size - 1 - node_row, 0), self._size - 1)
        return row, col

    def _put(self, canvas: List[List[str]], point: Point, symbol: str) -> None:
        """Place a symbol on the canvas at a map coordinate."""
        row, col = self._to_canvas(point)
        canvas[row][col] = symbol

    def render_map(
        self,
        riders: Optional[Iterable[Rider]] = None,
        orders: Optional[Iterable] = None,
        routes: Optional[Iterable[Sequence[Point]]] = None,
    ) -> str:
        """Render the ASCII map with riders, orders and routes.

        Args:
            riders: Riders whose positions are marked with 'R'.
            orders: Orders whose restaurants ('S') and customers ('C') are marked.
            routes: Sequences of visited points drawn with '*'.

        Returns:
            Multi-line string containing the map and a legend.
        """
        size = self._size
        canvas: List[List[str]] = []
        for r in range(size):
            row_cells = []
            for c in range(size):
                if r % 2 == 0 and c % 2 == 0:
                    row_cells.append(".")
                elif r % 2 == 1 and c % 2 == 1:
                    row_cells.append(" ")
                else:
                    row_cells.append("#")
            canvas.append(row_cells)

        for route in routes or []:
            for point in route:
                row, col = self._to_canvas(point)
                if canvas[row][col] in ".#":
                    canvas[row][col] = "*"

        for order in orders or []:
            self._put(canvas, order.res_pos, "S")
            self._put(canvas, order.cus_pos, "C")

        for rider in riders or []:
            self._put(canvas, rider.position, "R")

        header = [" "] * (2 + 2 * size)
        for i in range(GRID_DIM):
            header[2 + 4 * i] = str(i)
        lines = ["".join(header).rstrip()]
        for r, row_cells in enumerate(canvas):
            label = str(GRID_DIM - 1 - r // 2) if r % 2 == 0 else " "
            lines.append(f"{label} " + " ".join(row_cells))
        lines.append("")
        lines.append("Legend: " + self.map_symbols)
        lines.append("Columns: x grid index (0-7), rows: y grid index (0-7, top = 7).")
        return "\n".join(lines)

    def status_text(self, funds: int, simulated_time: int, riders: Sequence[Rider], active_orders: Sequence) -> str:
        """Build the snapshot report shown by the ``status`` command.

        Args:
            funds: Current system funds.
            simulated_time: Current tick.
            riders: All riders.
            active_orders: Orders not yet completed.

        Returns:
            Multi-line status text.
        """
        lines = [
            f"Simulated time : {simulated_time} ticks",
            f"System funds   : {funds}",
            f"Riders         : {len(riders)}",
            f"Active orders  : {len(active_orders)}",
        ]
        for rider in riders:
            order_ids = [order.id for order in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id:<3} {STATE_NAMES[rider.state]:<11} "
                f"pos=({rider.x},{rider.y}) orders={order_ids}"
            )
        return "\n".join(lines)

    def riders_text(self, riders: Sequence[Rider]) -> str:
        """Build the detailed rider report with every logged status change.

        Args:
            riders: Riders to report on.

        Returns:
            Multi-line text.
        """
        if not riders:
            return "No riders recruited yet."
        lines: List[str] = []
        for rider in riders:
            lines.append(
                f"Rider #{rider.rider_id} | state={STATE_NAMES[rider.state]} | "
                f"pos=({rider.x},{rider.y}) | speed={rider.speed} | "
                f"special_case={rider.special_case} | distance={rider.distance_traveled}"
            )
            if not rider.log:
                lines.append("    (no events)")
            for entry in rider.log:
                order_text = entry["order_id"] if entry["order_id"] is not None else "-"
                lines.append(
                    f"    t={entry['time']:<6} {entry['event']:<22} order={order_text:<5} "
                    f"state={STATE_NAMES.get(entry['state'], '?'):<11} "
                    f"pos=({entry['x']},{entry['y']})"
                )
        return "\n".join(lines)

    def orders_text(self, orders: Sequence) -> str:
        """Build the order table shown by the ``orders`` command.

        Args:
            orders: Orders to display.

        Returns:
            Multi-line table text.
        """
        if not orders:
            return "No orders yet."
        header = (
            f"{'ID':<4}{'Start':<7}{'Restaurant':<13}{'Customer':<13}"
            f"{'Distance':<10}{'Time':<7}{'Rider':<7}{'Status':<12}{'Eff.':<6}"
        )
        lines = [header, "-" * len(header)]
        for order in orders:
            rider_text = f"#{order.rider_id}" if order.rider_id is not None else "-"
            time_text = str(order.time_taken) if order.time_taken is not None else "-"
            eff_text = f"{order.efficiency:.2f}" if order.efficiency is not None else "-"
            lines.append(
                f"{order.id:<4}{order.start_time:<7}"
                f"{str(order.res_pos):<13}{str(order.cus_pos):<13}"
                f"{order.distance:<10.0f}{time_text:<7}{rider_text:<7}"
                f"{order.status:<12}{eff_text:<6}"
            )
        return "\n".join(lines)

from typing import List, Tuple

from order_model import validate_coordinate

try:  # readline gives arrow-key editing and history where available.
    import readline  # noqa: F401  pylint: disable=unused-import
except ImportError:  # pragma: no cover - platform dependent
    readline = None

COMMAND_ALIASES = {
    "start": "start",
    "new": "start",
    "n": "start",
    "status": "status",
    "s": "status",
    "orders": "orders",
    "o": "orders",
    "riders": "riders",
    "r": "riders",
    "map": "map",
    "m": "map",
    "tick": "tick",
    "t": "tick",
    "save": "save",
    "history": "history",
    "help": "help",
    "h": "help",
    "?": "help",
    "quit": "quit",
    "exit": "quit",
    "q": "quit",
}

MAX_HISTORY = 200


class InputManager:
    """Parses user input lines into validated commands.

    Attributes:
        command_history: Previously entered command lines (most recent last).
    """

    def __init__(self) -> None:
        """Create an input manager with an empty history."""
        self.command_history: List[str] = []

    def expand(self, line: str) -> str:
        """Expand shortcut keys such as ``!!`` (repeat last command).

        Args:
            line: Raw input line.

        Returns:
            The expanded line.

        Raises:
            ValueError: If ``!!`` is used with an empty history.
        """
        stripped = line.strip()
        if stripped == "!!":
            if not self.command_history:
                raise ValueError("no previous command to repeat")
            return self.command_history[-1]
        return stripped

    def parse(self, line: str) -> Tuple[str, List[str]]:
        """Split a line into a canonical command name and its arguments.

        Args:
            line: Expanded input line.

        Returns:
            Tuple ``(command, args)``. ``command`` is an empty string for blank input.

        Raises:
            ValueError: If the command is unknown.
        """
        tokens = line.replace(",", " ").split()
        if not tokens:
            return "", []
        name = tokens[0].lower()
        if name not in COMMAND_ALIASES:
            raise ValueError(f"unknown command '{name}'. Type 'help' for the command list")
        return COMMAND_ALIASES[name], tokens[1:]

    def record(self, line: str) -> None:
        """Append a command line to the history (bounded in size).

        Args:
            line: Command line to store.
        """
        if not line.strip():
            return
        self.command_history.append(line.strip())
        if len(self.command_history) > MAX_HISTORY:
            del self.command_history[0]

    @staticmethod
    def parse_coordinates(tokens: List[str]) -> List[int]:
        """Validate four coordinate tokens.

        Args:
            tokens: Strings expected to hold restaurant_x, restaurant_y,
                customer_x and customer_y.

        Returns:
            List of four validated integers.

        Raises:
            ValueError: On wrong token count, non-integer or out-of-range values.
        """
        if len(tokens) != 4:
            raise ValueError(
                "expected 4 integers: restaurant_x restaurant_y customer_x customer_y "
                f"(got {len(tokens)})"
            )
        labels = ["restaurant x", "restaurant y", "customer x", "customer y"]
        values: List[int] = []
        for token, label in zip(tokens, labels):
            try:
                number = int(token)
            except ValueError as exc:
                raise ValueError(f"{label} '{token}' is not an integer") from exc
            values.append(validate_coordinate(number, label))
        return values

    @staticmethod
    def read_line(prompt: str) -> str:
        """Prompt the user for a line of input.

        Args:
            prompt: Prompt text.

        Returns:
            The entered line.
        """
        return input(prompt)

"""Command-line entry point for the food delivery dispatch simulator."""

import logging
import os
import sys
from typing import List

import globalvar
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager, find_best_rider_for_order
from system import SystemManager, init_system, load_config

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
CONFIG_PATH = os.path.join(BASE_DIR, "config.json")
DEFAULT_SAVE_PATH = "dispatch_state.json"
LOG_PATH = "dispatch.log"
LOGGER = logging.getLogger("dispatch")

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy]  (n)  Place an order: restaurant then customer grid coords (0-800).
                            Runs the simulation until the order is delivered.
  status               (s)  System funds, time, and every rider's state and orders.
  orders               (o)  Table of all orders with timing, distance and rider.
  riders               (r)  Detailed rider info and every logged status change.
  map                  (m)  ASCII map with riders and active orders.
  tick [n]             (t)  Advance the simulation by n ticks (default 1).
  save [path]               Save system state to JSON (default dispatch_state.json).
  history              List previously entered commands.
  !!                        Repeat the last command.
  help                 (h)  Show this help.
  quit / exit          (q)  Leave the simulator.

class Simulator:
    """Wires the managers together and runs the interactive command loop."""

    def __init__(self, config_path: str = CONFIG_PATH) -> None:
        """Initialize the system state, load configuration and build managers.

        Args:
            config_path: Optional JSON configuration file path.
        """
        init_system()
        self.applied_config = {}
        if os.path.exists(config_path):
            try:
                self.applied_config = load_config(config_path)
            except (OSError, ValueError) as exc:
                print(f"Warning: could not load configuration ({exc}); using defaults.")
        self.system = SystemManager()
        self.orders = OrderManager()
        self.display = DisplayManager()
        self.inputs = InputManager()
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def run(self) -> None:
        """Run the interactive loop until quit or end of input."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands.")
        while self.system.running:
            try:
                raw = self.inputs.read_line("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            try:
                self.execute(raw)
            except ValueError as exc:
                print(f"Error: {exc}")
            except Exception:  # pylint: disable=broad-except
                LOGGER.exception("unexpected failure while executing %r", raw)
                print("Unexpected internal error; system state preserved. See dispatch.log.")
        self.system.stop()

    def execute(self, raw: str) -> None:
        """Expand, parse and dispatch one command line.

        Args:
            raw: Raw user input.

        Raises:
            ValueError: For invalid input that the user must correct.
        """
        line = self.inputs.expand(raw)
        command, args = self.inputs.parse(line)
        if not command:
            return
        self.inputs.record(line)
        self.handlers[command](args)

    def _riders(self):
        """Return the current rider list."""
        return self.system.riders

    def cmd_start(self, args: List[str]) -> None:
        """Place an order, assign the nearest rider and simulate until delivery.

        Args:
            args: Optional four coordinate tokens; prompted when absent.
        """
        if not args:
            args = self.inputs.read_line(
                "Enter restaurant_x restaurant_y customer_x customer_y: "
            ).split()
        res_x, res_y, cus_x, cus_y = self.inputs.parse_coordinates(args)

        order = self.orders.create_order(res_x, res_y, cus_x, cus_y)
        self.system.add_funds(int(globalvar.get_value("order_fee", 100)))

        riders_before = len(self._riders())
        self.system.check_recruitment()
        if not self._riders():
            self.system.recruit_rider()

        rider = find_best_rider_for_order(order)
        if rider is None:
            raise ValueError("no rider is available to take this order")
        order.assign_to(rider.rider_id)

        print(f"Order #{order.id} accepted.")
        print(f"Restaurant grid point : {order.res_pos}")
        print(f"Customer grid point   : {order.cus_pos}")

        max_ticks = int(globalvar.get_value("max_ticks", 10000))
        ticks = 0
        while order.status != "completed":
            if ticks >= max_ticks:
                print(f"Error: order #{order.id} not delivered within {max_ticks} ticks.")
                break
            self.system.tick()
            ticks += 1

        added = len(self._riders()) - riders_before
        if added > 0:
            new_ids = ", ".join(f"#{r.rider_id}" for r in self._riders()[riders_before:])
            print(f"New rider added       : yes ({new_ids})")
        else:
            print("New rider added       : no")
        print(f"Assigned rider        : #{rider.rider_id} (speed {rider.speed} units/tick)")
        print(f"Distance travelled    : {order.distance:.0f} units")
        print(f"Time taken            : {order.time_taken} ticks")
        print("Route map:")
        print(
            self.display.render_map(
                riders=[rider], orders=[order], routes=[order.route]
            )
        )

    def cmd_status(self, args: List[str]) -> None:
        """Print the system snapshot.

        Args:
            args: Ignored.
        """
        print(
            self.display.status_text(
                self.system.funds,
                int(globalvar.get_value("simulated_time", 0)),
                self._riders(),
                self.orders.active_orders(),
            )
        )

    def cmd_orders(self, args: List[str]) -> None:
        """Print the table of all orders.

        Args:
            args: Ignored.

    def cmd_riders(self, args: List[str]) -> None:
        """Print every rider with its full status-change log.

        Args:
            args: Ignored.
        """
        print(self.display.riders_text(self._riders()))

    def cmd_map(self, args: List[str]) -> None:
        """Print the ASCII map of riders and active orders.

        Args:
            args: Ignored.

        active = self.orders.active_orders()
        print(
            self.display.render_map(
                riders=self._riders(),
                orders=active,
                routes=[order.route for order in active],
            )
        )

    def cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation by one or more ticks.

        Args:
            args: Optional tick count (1-1000).

        Raises:
            ValueError: If the count is not an integer in range.
        """
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError as exc:
                raise ValueError(f"tick count '{args[0]}' is not an integer") from exc
            if not 1 <= count <= 1000:
                raise ValueError("tick count must be between 1 and 1000")
        recruited = []
        for _ in range(count):
            new_rider = self.system.tick()
            if new_rider is not None:
                recruited.append(new_rider.rider_id)
        print(f"Advanced {count} tick(s). Simulated time: {globalvar.get_value('simulated_time')}.")
        if recruited:
            print("Recruited rider(s): " + ", ".join(f"#{rid}" for rid in recruited))

    def cmd_save(self, args: List[str]) -> None:
        """Save the system state to a JSON file.

        Args:
            args: Optional output path.

        Raises:
            ValueError: If the file cannot be written.
        """
        path = args[0] if args else DEFAULT_SAVE_PATH
        try:
            self.system.save_state(path)
        except OSError as exc:
            raise ValueError(f"could not save state to {path}: {exc}") from exc
        print(f"State saved to {path}.")

    def cmd_history(self, args: List[str]) -> None:
        """Print the command history.

        Args:
            args: Ignored.
        """
        if not self.inputs.command_history:
            print("No commands entered yet.")
            return
        for index, line in enumerate(self.inputs.command_history, start=1):
            print(f"{index:>4}  {line}")

    def cmd_help(self, args: List[str]) -> None:
        """Print the help text.

        Args:
            args: Ignored.
        """
        print(HELP_TEXT)

    def cmd_quit(self, args: List[str]) -> None:
        """Stop the main loop.

        Args:
            args: Ignored.
        """
        self.system.stop()
        print("Goodbye.")


def main() -> int:
    """Configure logging and run the simulator.

    Returns:
        Process exit code.
    """
    try:
        logging.basicConfig(
            filename=LOG_PATH,
            level=logging.INFO,
            format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        )
    except OSError:
        logging.basicConfig(level=logging.WARNING)
    simulator = Simulator()
    simulator.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Participant list loading, validation and group extraction."""

from __future__ import annotations

from lottery_system.config import LIST_SEPARATOR
from lottery_system.default_data import DEFAULT_USER_LINES
from lottery_system.models import User


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (one "Name,Department" per line)."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [part.strip() for part in line.split(LIST_SEPARATOR)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    f"(expected 'Name{LIST_SEPARATOR}Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence kept) and duplicate entries."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management and lottery ordering."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZE_SPECS
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZE_SPECS:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"Cannot load preset prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize to the lottery sequence (addition order)."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name cannot be empty")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"prize '{clean_name}' already exists")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        groups = list(dict.fromkeys(avoid_groups or []))
        drawer = associated_drawer.strip() if associated_drawer else None
        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer or None,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. ``order`` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if len(order) != count:
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, drawing and fallback protection."""

from __future__ import annotations

import random

from lottery_system.config import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # Results of the last two rounds run by Chen Qixian / Jin Duo.
        # True means that round produced at least one Cheng Ru group winner.
        self.last_two_draw_results: list[bool] = []
        self._rng = random.Random()

    # ----- winner tracking -------------------------------------------------
    def mark_winner(self, user: User) -> None:
        self._winner_names.add(user.name)

    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def remaining_users(self) -> list[User]:
        """Users who have not won any prize yet."""
        return [user for user in self._users if user.name not in self._winner_names]

    # ----- rule engine -----------------------------------------------------
    def is_cheng_ru_protection_active(self, prize: Prize) -> bool:
        """Fallback: Ma Jiaqi's round is restricted to Cheng Ru group members when
        neither of the last two Chen Qixian / Jin Duo rounds had a Cheng Ru winner."""
        return (
            prize.associated_drawer == MA_JIAQI
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        candidates = self.remaining_users()

        # Fallback mechanism overrides all other rules, including avoidance groups.
        if self.is_cheng_ru_protection_active(prize):
            return [user for user in candidates if user.department == CHENG_RU_GROUP]

        # Special rule: Li Bin's rounds use relaxed filtering (avoidance groups ignored).
        if prize.associated_drawer == LI_BIN:
            return candidates

        # Basic filtering: exclude avoidance groups.
        avoid = set(prize.avoid_groups)
        return [user for user in candidates if user.department not in avoid]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the current eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for ``prize``. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.mark_winner(user)
        return user

    def finish_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Update fallback state after a round has ended."""
        drawer = prize.associated_drawer
        if drawer in (CHEN_QIXIAN, JIN_DUO):
            hit = any(user.department == CHENG_RU_GROUP for user in round_winners)
            self.last_two_draw_results.append(hit)
            del self.last_two_draw_results[:-2]
        elif drawer == MA_JIAQI:
            # A Ma Jiaqi round starts a new cycle of observation.
            self.last_two_draw_results.clear()

"""Winner recording, ordering, display grouping and CSV export."""

from __future__ import annotations

import csv
import datetime
import os
import re
from dataclasses import dataclass

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    sequence: int = 0


_LEVEL_PATTERNS: list[tuple[int, re.Pattern[str]]] = [
    (1, re.compile(r"\bfirst\b", re.IGNORECASE)),
    (2, re.compile(r"\bsecond\b", re.IGNORECASE)),
    (3, re.compile(r"\bthird\b", re.IGNORECASE)),
]


def _prize_rank(prize: Prize) -> int:
    for rank, pattern in _LEVEL_PATTERNS:
        if pattern.search(prize.name):
            return rank
    return 4  # Unlabelled prizes come after the leveled ones.


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Sort prizes from high level (First) to low level; ties keep addition order."""
    return sorted(prizes, key=_prize_rank)


class ResultManager:
    """Keeps the complete winner history of the event."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(user=user, prize=prize, sequence=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low).
        Within a prize, winners are in winning time order."""
        result: dict[str, list[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            result[prize.name] = [
                winner for winner in self._winners if winner.prize.name == prize.name
            ]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        date_text = datetime.datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_text}.txt")
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize Name", "Winning Order", "Name", "Department"])
                winners_map = self.get_winners_by_prize(prizes)
                for prize_name, winners in winners_map.items():
                    for winner in winners:
                        writer.writerow(
                            [
                                prize_name,
                                winner.sequence,
                                winner.user.name,
                                winner.user.department,
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

# ----- helpers --------------------------------------------------------------
def ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _print_header(text: str) -> None:
    print("\n" + "=" * 60)
    print(text)
    print("=" * 60)


def show_group_info(list_mgr: ListManager, users: list[User]) -> None:
    info = list_mgr.get_group_info(users)
    print("Department groups:")
    for group, count in sorted(info.items()):
        print(f"  - {group}: {count} people")


# ----- participant list -------------------------------------------------------
def resolve_problems(valid: list[User]) -> list[User] | None:
    """Ask how to handle format errors / duplicates. Returns None to terminate."""
    print("\nHow do you want to handle the problematic data?")
    print("1. Remove invalid/duplicate data and keep the remaining valid entries")
    print("2. Manually remove specific participants by name")
    print("3. Terminate this upload")
    choice = input("Select option: ").strip()
    if choice == "1":
        return valid
    if choice == "2":
        raw = input("Enter names to remove (comma-separated): ")
        names = {token.strip() for token in raw.split(",") if token.strip()}
        remaining = [user for user in valid if user.name not in names]
        print(f"Removed {len(valid) - len(remaining)} participant(s).")
        return remaining
    if choice == "3":
        return None
    print("Invalid choice. Upload terminated.")
    return None


def setup_user_list(list_mgr: ListManager) -> list[User] | None:
    while True:
        print("\n--- Participant List ---")
        print("1. Load built-in default team building list")
        print("2. Upload custom txt list")
        print("0. Exit")
        choice = input("Select option: ").strip()
        if choice == "0":
            return None
        if choice == "1":
            users, errors = list_mgr.load_default_list()
        elif choice == "2":
            path = input("Enter txt file path: ").strip().strip('"').strip("'")
            users, errors = list_mgr.load_custom_list(path)
        else:
            print("Invalid option.")
            continue

        if errors:
            print("Errors found in the list:")
            for error in errors:
                print(f"  {error}")

        valid, duplicates = list_mgr.validate_users(users)
        if duplicates:
            print("Duplicate names detected (later entries ignored):")
            for user in duplicates:
                print(f"  {user.name} ({user.department})")

        if errors or duplicates:
            resolved = resolve_problems(valid)
            if resolved is None:
                print("Upload terminated.")
                continue
            valid = resolved

        if not valid:
            print("No valid participants available.")
            continue

        print(f"Loaded {len(valid)} participants.")
        show_group_info(list_mgr, valid)
        return valid

def add_custom_prize(prize_mgr: PrizeManager, groups: list[str]) -> None:
    name = input("Prize name: ").strip()
    quantity_text = input("Quantity: ").strip()
    if not quantity_text.isdigit() or int(quantity_text) <= 0:
        print("Quantity must be a positive integer.")
        return
    drawer = input("Associated lottery person (blank for none): ").strip() or None

    avoid: list[str] = []
    if groups:
        print("Available groups:")
        for index, group in enumerate(groups, start=1):
            print(f"  {index}. {group}")
    raw = input("Avoidance groups (comma-separated numbers or names, blank for none): ")
    for token in [item.strip() for item in raw.split(",") if item.strip()]:
        if token.isdigit() and 1 <= int(token) <= len(groups):
            avoid.append(groups[int(token) - 1])
        elif token in groups:
            avoid.append(token)
        else:
            print(f"Unknown group ignored: {token}")

    try:
        prize_mgr.add_prize(name, int(quantity_text), avoid, drawer)
        print("Prize added.")
    except ValueError as exc:
        print(f"Cannot add prize: {exc}")

# ----- lottery process --------------------------------------------------------
def describe_rules(prize: Prize, lottery: LotteryManager) -> list[str]:
    lines = ["Rule: previous winners are excluded."]
    if lottery.is_cheng_ru_protection_active(prize):
        lines.append(
            "Fallback (Cheng Ru group protection): only Cheng Ru Group members are "
            "eligible; avoidance groups are overridden."
        )
    elif prize.associated_drawer == LI_BIN:
        lines.append("Special rule (Li Bin): relaxed filtering, avoidance groups ignored.")
    elif prize.avoid_groups:
        lines.append("Avoided groups: " + ", ".join(prize.avoid_groups))
    else:
        lines.append("Avoided groups: none")
    return lines


def await_command(prompt: str) -> str:
    """Wait for operator input. Returns 'next' to proceed or 'terminate' to end the round.
    'pause' suspends the lottery until 'continue' (resume) or 'terminate'."""
    while True:
        command = input(prompt).strip().lower()
        if command == "":
            return "next"
        if command == "terminate":
            return "terminate"
        if command == "pause":
            print("Lottery paused.")
            while True:
                resume = input("Type 'continue' to resume or 'terminate' to end this round: ")
                resume = resume.strip().lower()
                if resume == "continue":
                    print("Lottery resumed.")
                    return "next"
                if resume == "terminate":
                    return "terminate"
                print("Unknown command.")
        print("Unknown command. Use Enter, 'pause', or 'terminate'.")


def run_round(prize: Prize, lottery: LotteryManager, result_mgr: ResultManager) -> list[User]:
    winners: list[User] = []
    for draw_no in range(1, prize.quantity + 1):
        print(f"Currently drawing the {ordinal(draw_no)} winner...")
        time.sleep(DRAW_DELAY_SECONDS)
        user = lottery.draw_one(prize)
        if user is None:
            print("No eligible candidates remain for this prize. Round ends early.")
            break
        result_mgr.add_winner(user, prize)
        winners.append(user)
        print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")
        if draw_no == prize.quantity:
            break
        command = await_command(
            "Press Enter for the next draw, 'pause' to pause, or 'terminate' to end this round: "
        )
        if command == "terminate":
            print("Round terminated by operator.")
            break
    return winners


def print_round_results(prize: Prize, winners: list[User]) -> None:
    print(f"\nRound results for {prize.name} (winning order):")
    if not winners:
        print("  (no winners)")
    for index, user in enumerate(winners, start=1):
        print(f"  {index}. {user.name} ({user.department})")


def run_lottery(prizes: list[Prize], lottery: LotteryManager, result_mgr: ResultManager) -> None:
    total = len(prizes)
    for round_no, prize in enumerate(prizes, start=1):
        _print_header(f"Round {round_no}/{total}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Lottery person: {prize.associated_drawer or 'None'}")
        for line in describe_rules(prize, lottery):
            print(line)

        command = await_command(
            "Press Enter to start this round, or type 'terminate' to skip it: "
        )
        if command == "terminate":
            print("Round skipped.")
            continue

        round_winners = run_round(prize, lottery, result_mgr)
        lottery.finish_round(prize, round_winners)
        print_round_results(prize, round_winners)


def show_summary(prizes: list[Prize], result_mgr: ResultManager) -> None:
    _print_header("Final Winner List (by prize level, then winning order)")
    winners_map = result_mgr.get_winners_by_prize(prizes)
    for prize_name, winners in winners_map.items():
        print(f"\n{prize_name}")
        if not winners:
            print("  (no winners)")
        for winner in winners:
            print(
                f"  {winner.sequence}. {winner.user.name} ({winner.user.department}) "
                f"- {prize_name}"
            )


def export_prompt(prizes: list[Prize], result_mgr: ResultManager) -> None:

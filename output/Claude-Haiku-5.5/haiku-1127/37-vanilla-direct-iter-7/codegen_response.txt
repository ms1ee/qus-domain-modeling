import os

from lottery_system.config import DEFAULT_USER_LIST_TEXT
from lottery_system.models import User


class ListManager:
    """Loads, verifies and analyses participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_USER_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except UnicodeDecodeError:
            return [], ["Format Error: file is not valid UTF-8 text"]
        except OSError as exc:
            return [], [f"File read error: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    f"(expected format: Name,Department Group)"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). The first occurrence of a name is kept."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict:
        """Return a mapping of department group -> number of people."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.config import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the prize library and the lottery round order."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = name.strip().casefold()
        for prize in self._prizes:
            if prize.name.strip().casefold() == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Load the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                continue
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"Failed to load '{name}': {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        """Add a prize at the end of the round order."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if self._find(clean_name) is not None:
            raise ValueError(f"A prize named '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")

        groups: list[str] = []
        for group in avoid_groups or []:
            group_name = str(group).strip()
            if group_name and group_name not in groups:
                groups.append(group_name)

        drawer = associated_drawer.strip() if associated_drawer else None
        if drawer == "":
            drawer = None

        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        count = len(self._prizes)
        if sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in positions]
        return True

import os
import re
from dataclasses import dataclass
from datetime import datetime
import csv

from lottery_system.models import Prize, User

_LEVEL_WORDS = [
    "first", "second", "third", "fourth", "fifth",
    "sixth", "seventh", "eighth", "ninth", "tenth",
]
_LEVEL_PATTERN = re.compile(r"\b(" + "|".join(_LEVEL_WORDS) + r")\b", re.IGNORECASE)


def prize_level_rank(prize_name: str) -> int:
    """Lower rank means higher prize level (First=1, Second=2, Third=3...)."""
    match = _LEVEL_PATTERN.search(prize_name or "")
    if not match:
        return 99
    return _LEVEL_WORDS.index(match.group(1).lower()) + 1


def sort_prizes_by_level(prizes) -> list[Prize]:
    """Sort prizes from high to low level; ties keep addition order."""
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: (prize_level_rank(item[1].name), item[0]))
    return [prize for _, prize in indexed]


@dataclass
class Winner:
    user: User
    prize_name: str
    sequence: int
    won_at: str


class ResultManager:
    """Keeps the winner list in real time and produces summaries/exports."""

    def __init__(self):
        self._winners: list[Winner] = []
        self._won_keys: set[str] = set()

    @staticmethod
    def _key(user: User) -> str:
        return user.name.strip().casefold()

    def has_won(self, user: User) -> bool:
        return self._key(user) in self._won_keys

    def add_winner(self, user, prize) -> Winner:
        """Record a winner. `prize` may be a Prize or a prize name."""
        if self.has_won(user):
            raise ValueError(f"{user.name} has already won a prize.")
        prize_name = prize if isinstance(prize, str) else prize.name
        winner = Winner(
            user=user,
            prize_name=prize_name,
            sequence=len(self._winners) + 1,
            won_at=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        )
        self._winners.append(winner)
        self._won_keys.add(self._key(user))
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict:
        """Return {prize name: [Winner...]} ordered by prize level (high to low).

        Within the same prize, winners are ordered by winning time.
        """
        result: dict[str, list[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            matched = [w for w in self._winners if w.prize_name == prize.name]
            result[prize.name] = sorted(matched, key=lambda w: w.sequence)
        return result

    def _ordered_winners(self, prizes) -> list[Winner]:
        order_index = {}
        for index, prize in enumerate(prizes):
            order_index.setdefault(prize.name, index)
        fallback_index = len(prizes)

        def key(winner: Winner):
            return (
                prize_level_rank(winner.prize_name),
                order_index.get(winner.prize_name, fallback_index),
                winner.sequence,
            )

        return sorted(self._winners, key=key)

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export the complete winner list to current_dir/awards_result_YYYYMMDD.txt."""
        if not self._winners:
            return False, "No winners to export."

        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in self._ordered_winners(prizes):
                    writer.writerow([
                        winner.user.name,
                        winner.user.department,
                        winner.prize_name,
                        winner.won_at,
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

import random

from lottery_system.config import (
    CHENG_RU_GROUP,
    FALLBACK_DRAWERS,
    FALLBACK_TRIGGER_DRAWER,
    SPECIAL_NO_AVOID_DRAWER,
)
from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager


def _norm(text: str) -> str:
    return (text or "").strip().casefold()


class LotteryManager:
    """Executes lottery draws and applies the lottery rule engine."""

    def __init__(self, users, result_manager=None, rng=None):
        self.users: list[User] = list(users)
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        # Per-round results of Chen Qixian / Jin Duo: True if the round contained a Cheng Ru member.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.Random()

    # ----- helpers -------------------------------------------------------

    @staticmethod
    def _is_drawer(prize: Prize, drawer_name: str) -> bool:
        return prize.associated_drawer is not None and _norm(prize.associated_drawer) == _norm(drawer_name)

    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return _norm(user.department) == _norm(CHENG_RU_GROUP)

    def _fallback_active(self, prize: Prize) -> bool:
        """Fallback applies when Ma Jiaqi draws and the last two results of
        Chen Qixian / Jin Duo both contain no Cheng Ru group member."""
        if not self._is_drawer(prize, FALLBACK_TRIGGER_DRAWER):
            return False
        if len(self.last_two_draw_results) < 2:
            return False
        return not any(self.last_two_draw_results)

    # ----- rule engine ---------------------------------------------------

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        # Anti-duplication: previous winners are always excluded.
        pool = [user for user in self.users if not self.result_manager.has_won(user)]

        # Special rule: Li Bin's prizes only filter previous winners.
        if self._is_drawer(prize, SPECIAL_NO_AVOID_DRAWER):
            return pool

        # Fallback (Cheng Ru group protection): overrides avoidance groups.
        if self._fallback_active(prize):
            cheng_ru_pool = [user for user in pool if self._is_cheng_ru(user)]
            if cheng_ru_pool:
                return cheng_ru_pool

        # Basic rule: exclude avoidance groups of the prize.
        avoid = {_norm(group) for group in prize.avoid_groups}
        return [user for user in pool if _norm(user.department) not in avoid]

    def count_eligible(self, prize: Prize) -> int:
        return len(self._get_eligible_pool(prize))

    def describe_rules(self, prize: Prize) -> list[str]:
        rules: list[str] = []
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}")
        rules.append("Previous winners are excluded.")
        if self._is_drawer(prize, SPECIAL_NO_AVOID_DRAWER):
            rules.append(f"Special rule for {SPECIAL_NO_AVOID_DRAWER}: avoidance groups are not applied.")
        elif prize.avoid_groups:
            rules.append("Avoidance groups excluded: " + ", ".join(prize.avoid_groups))
        if self._is_drawer(prize, FALLBACK_TRIGGER_DRAWER):
            rules.append(
                f"Fallback rule: if the last two results of {FALLBACK_DRAWERS[0]} and "
                f"{FALLBACK_DRAWERS[1]} contain no {CHENG_RU_GROUP} member, only "
                f"{CHENG_RU_GROUP} members are eligible."
            )
            if self._fallback_active(prize):
                rules.append("Fallback is ACTIVE for this round: only Cheng Ru group members are eligible.")
        return rules

    # ----- draw control --------------------------------------------------

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.result_manager.add_winner(user, prize)
        return user

    def finish_round(self, prize: Prize, round_winners) -> None:
        """Update fallback state after a round has finished."""
        if self._is_drawer(prize, FALLBACK_TRIGGER_DRAWER):
            if self._fallback_active(prize):
                # The forced protection round has been consumed.
                self.last_two_draw_results.clear()
        elif any(self._is_drawer(prize, name) for name in FALLBACK_DRAWERS):
            self.last_two_draw_results.append(any(self._is_cheng_ru(user) for user in round_winners))
            del self.last_two_draw_results[:-2]

import re

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager

SEPARATOR = "=" * 60


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotterySystemCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ----- main loop -----------------------------------------------------

    def run(self):
        print(SEPARATOR)
        print("Company Team Building - Smart Lottery System")
        print(SEPARATOR)
        actions = {
            "1": self._load_list,
            "2": self._show_groups,
            "3": self._load_default_prizes,
            "4": self._add_custom_prize,
            "5": self._show_prizes,
            "6": self._reorder_prizes,
            "7": self._start_lottery,
            "8": self._show_summary,
            "9": self._export,
        }
        while True:
            self._print_menu()
            choice = input("Select an option: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid option, please try again.")
                continue
            action()

    def _print_menu(self):
        print()
        print("1. Load team building list (default / custom txt)")
        print("2. View department groups and headcounts")
        print("3. Load default prize library")
        print("4. Add custom prize")
        print("5. Show prize list")
        print("6. Adjust prize order")
        print("7. Start lottery")
        print("8. Show winner summary")
        print("9. Export winner list to file")
        print("0. Exit")

    # ----- list management ----------------------------------------------

    def _load_list(self):
        if self.result_manager.get_all_winners():
            print("Cannot reload the list after winners have been drawn.")
            return
        print("1. Built-in default list")
        print("2. Custom txt file (each line: Name,Department Group)")
        choice = input("Select list source: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = input("Enter txt file path: ").strip().strip('"')
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Invalid option.")
            return

        valid = self._review_list(users, errors)
        if valid is None:
            return
        self.users = valid
        self.lottery = LotteryManager(self.users, self.result_manager)
        print(f"Loaded {len(self.users)} valid participant(s).")

    def _review_list(self, users, errors):
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            if not valid:
                print("The list is empty.")
                return None
            return valid

        for error in errors:
            print(error)
        for dup in duplicates:
            print(f"Duplicate Name: {dup.name} ({dup.department}) appears more than once and will be ignored.")

        while True:
            choice = input(
                "Type 'remove' to drop invalid data and continue, "
                "or 'terminate' to cancel the upload: "
            ).strip().lower()
            if choice == "remove":
                if not valid:
                    print("No valid data remains.")
                    return None
                return valid
            if choice == "terminate":
                print("Upload terminated.")
                return None
            print("Please enter 'remove' or 'terminate'.")

    def _show_groups(self):
        if not self.users:
            print("No list loaded. Please use option 1 first.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for department, count in info.items():
            print(f"  {department}: {count} people")
        print(f"Total: {len(self.users)} people")

    # ----- prize management ---------------------------------------------

    def _load_default_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        if errors:
            for error in errors:
                print(error)
        else:
            print("Default prize library loaded.")

    def _add_custom_prize(self):
        name = input("Prize name: ").strip()
        quantity_raw = input("Quantity: ").strip()
        try:
            quantity = int(quantity_raw)
        except ValueError:
            print("Quantity must be an integer.")
            return
        if self.users:
            groups = ", ".join(self.list_manager.get_group_info(self.users).keys())
            print(f"Available groups: {groups}")
        avoid_raw = input("Avoidance groups (comma separated, blank for none): ")
        drawer = input("Associated lottery person (blank for none): ").strip()
        avoid = [group.strip() for group in avoid_raw.split(",") if group.strip()]
        try:
            self.prize_manager.add_prize(name, quantity, avoid, drawer or None)
        except ValueError as exc:
            print(f"Failed to add prize: {exc}")
            return
        print("Prize added.")

    def _show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            print(f"{index}. {prize.name} | quantity: {prize.quantity} | "
                  f"lottery person: {drawer} | avoid groups: {avoid}")

    def _reorder_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            print("At least two prizes are needed to reorder.")
            return
        self._show_prizes()
        raw = input("Enter new order as serial numbers (e.g. 2 1 3): ").strip()
        try:
            order = [int(item) for item in re.split(r"[\s,]+", raw) if item]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ----- lottery process ----------------------------------------------

    def _start_lottery(self):
        if self.lottery is None or not self.users:
            print("Please load the team building list first (option 1).")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Please load or add prizes first.")
            return
        print(f"Lottery starts with {len(prizes)} round(s).")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize)
        print("\nAll lottery rounds completed.")
        self._print_summary(prizes)

    def _run_round(self, index: int, total: int, prize):
        print()
        print(SEPARATOR)
        print(f"Round {index}/{total}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print("Rules:")
        for rule in self.lottery.describe_rules(prize):
            print(f"  - {rule}")
        print(SEPARATOR)

        drawn: list[User] = []
        for position in range(1, prize.quantity + 1):
            action = self._await_control(position)
            if action == "terminate":
                print("Round terminated. Skipping the remaining winners of this prize.")
                break
            print(f"Currently drawing the {ordinal(position)} winner...")
            user = self.lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize.")
                break
            drawn.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        self.lottery.finish_round(prize, drawn)
        print(f"Round {index} results ({prize.name}):")
        if drawn:
            for number, user in enumerate(drawn, start=1):
                print(f"  {number}. {user.name} ({user.department})")
        else:
            print("  No winners in this round.")

    def _await_control(self, position: int) -> str:
        """Wait for user control input before the next draw. Returns 'draw' or 'terminate'."""
        while True:
            command = input(
                f"[Enter] draw winner #{position} | 'pause' | 'terminate': "
            ).strip().lower()
            if command in ("", "continue"):
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = input("(paused) > ").strip().lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if resume == "terminate":
                        return "terminate"
                    print("Please type 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    # ----- results ------------------------------------------------------

    def _print_summary(self, prizes):
        grouped = self.result_manager.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            print("No winners yet.")

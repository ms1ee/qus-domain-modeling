"""Domain models shared by the lottery system services."""
from __future__ import annotations

import re
from dataclasses import dataclass, field

CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
MA_JIAQI = "Ma Jiaqi"
FALLBACK_DRAWERS = ("Chen Qixian", "Jin Duo")

_LEVEL_PATTERN = re.compile(r"^\s*(First|Second|Third)\s+Prize", re.IGNORECASE)
_LEVEL_MAP = {"first": 1, "second": 2, "third": 3}


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    @property
    def level(self) -> int | None:
        """Prize level parsed from the name (1 = First, 2 = Second, 3 = Third)."""
        match = _LEVEL_PATTERN.match(self.name)
        if match:
            return _LEVEL_MAP[match.group(1).lower()]
        return None

    @property
    def rule_description(self) -> str:
        """Human readable description of the rules applied to this prize."""
        parts: list[str] = []
        if self.associated_drawer:
            parts.append(f"Lottery person: {self.associated_drawer}.")
        if self.associated_drawer == LI_BIN:
            parts.append(
                "Special rule: only previous winners are excluded; "
                "avoidance groups are not applied."
            )
        else:
            parts.append("Previous winners are excluded.")
            if self.avoid_groups:
                parts.append(
                    "Members of avoidance groups ("
                    + ", ".join(self.avoid_groups)
                    + ") remain eligible but are drawn only after all other members."
                )
        if self.associated_drawer == MA_JIAQI:
            parts.append(
                "Fallback (Cheng Ru Group protection): if the last two results of "
                "Chen Qixian and Jin Duo contain no "
                f"{CHENG_RU_GROUP} member, only {CHENG_RU_GROUP} members are eligible."
            )
        return " ".join(parts)

    ("Ma Jiaqi", "Technology Department"),
    ("Li Bin", "Product Department"),

    ("Zhao Liu", "Cheng Ru Group"),
    ("Sun Qi", "Cheng Ru Group"),
    ("Zhou Ba", "Cheng Ru Group"),
    ("Wu Jiu", "Cheng Ru Group"),
    ("Zheng Shi", "Marketing Department"),
    ("Feng Yi", "Finance Department"),
    ("Chu Er", "Human Resources Department"),
    ("Wei San", "Technology Department"),
    ("Jiang Si", "Product Department"),
    ("Shen Wu", "Cheng Ru Group"),
    ("Han Liu", "Operations Department"),
    ("Yang Qi", "Operations Department"),
    ("Zhu Ba", "Technology Department"),
]

_SEPARATOR_PATTERN = re.compile(r"[,;\uff0c]")


class ListManager:
    """Loads, verifies and summarizes participant lists."""

    def __init__(self) -> None:
        self.source: str | None = None

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building participant list."""
        users = [User(name=name, department=dept) for name, dept in DEFAULT_PARTICIPANTS]
        self.source = "built-in default list"
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        users, errors = self._load_from_path(path)
        self.source = path
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file into users and format error messages."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.readlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in _SEPARATOR_PATTERN.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique users and duplicate-name entries."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of members in each department group."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.models import Prize

PRESET_PRIZES: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in PRESET_PRIZES:
            if any(p.name.lower() == name.lower() for p in self._prizes):
                continue
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"Cannot load preset prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the sequence. Raises ValueError on invalid input."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("quantity must be a positive integer")
        if any(p.name.lower() == clean_name.lower() for p in self._prizes):
            raise ValueError(f"prize '{clean_name}' already exists")

        groups: list[str] = []
        for group in avoid_groups or []:
            group = group.strip()
            if group and group not in groups:
                groups.append(group)

        drawer = associated_drawer.strip() if associated_drawer else None
        drawer = drawer or None

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        ``order`` contains 1-based current positions; the i-th element is the
        current position of the prize that should be placed at position i in
        the new sequence. Returns False if the order is not a valid permutation.
        """
        count = len(self._prizes)
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback."""
from __future__ import annotations

import random

from lottery_system.models import (
    CHENG_RU_GROUP,
    FALLBACK_DRAWERS,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)


def _is_cheng_ru(user: User) -> bool:
    return user.department.strip().lower() == CHENG_RU_GROUP.lower()


class LotteryManager:
    """Performs draws while enforcing the lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # Results (True = winner belongs to Cheng Ru Group) of the most recent
        # draws made by Chen Qixian and Jin Duo, at most two entries.
        self.last_two_draw_results: list[bool] = []

    def _available_users(self) -> list[User]:
        """Users who have not won any prize yet (anti-duplication)."""
        return [u for u in self._users if u.name not in self._winner_names]

    def _fallback_active(self) -> bool:
        """True when the last two results of Chen Qixian/Jin Duo contain no Cheng Ru member."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users that may be drawn for the given prize."""
        available = self._available_users()

        # Special rule: Li Bin only filters previous winners.
        if prize.associated_drawer == LI_BIN:
            return available

        # Fallback rule: Ma Jiaqi's prize is limited to Cheng Ru Group members.
        if prize.associated_drawer == MA_JIAQI and self._fallback_active():
            protected = [u for u in available if _is_cheng_ru(u)]
            if protected:
                return protected

        # Avoidance groups: members remain eligible but are drawn only after others.
        avoid = {g.strip().lower() for g in prize.avoid_groups}
        preferred = [u for u in available if u.department.strip().lower() not in avoid]
        if preferred:
            return preferred
        return [u for u in available if u.department.strip().lower() in avoid]

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None when no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winner_names.add(winner.name)
        if prize.associated_drawer in FALLBACK_DRAWERS:
            self.last_two_draw_results.append(_is_cheng_ru(winner))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return winner

"""Winner recording, ordering, display grouping and CSV/txt export."""
from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_time: datetime
    winning_order: int


def sort_prizes(prizes: list[Prize]) -> list[Prize]:
    """Sort prizes by level from high (First) to low; unknown levels keep addition order after leveled ones."""
    indexed = list(enumerate(prizes))
    indexed.sort(
        key=lambda item: (
            item[1].level if item[1].level is not None else 1000 + item[0],
            item[0],
        )
    )
    return [prize for _, prize in indexed]


class ResultManager:
    """Keeps the winner list and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._winner_names: set[str] = set()

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner. Raises ValueError if the person already won."""
        if user.name in self._winner_names:
            raise ValueError(f"{user.name} has already won a prize")
        winner = Winner(
            user=user,
            prize=prize,
            winning_time=datetime.now(),
            winning_order=len(self._winners) + 1,
        )
        self._winners.append(winner)
        self._winner_names.add(user.name)
        return winner

    def get_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Winners grouped by prize name, prizes ordered high level to low, winners in winning order."""
        ordered = sort_prizes(prizes)
        result: dict[str, list[Winner]] = {prize.name: [] for prize in ordered}
        for winner in self._winners:
            if winner.prize.name in result:
                result[winner.prize.name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to 'awards_result_YYYYMMDD.txt' in the current directory as CSV."""
        grouped = self.get_winners_by_prize(prizes)
        rows = [winner for winners in grouped.values() for winner in winners]
        if not rows:
            return False, "No winners to export."

        now = datetime.now()
        path = os.path.join(os.getcwd(), f"awards_result_{now:%Y%m%d}.txt")
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in rows:
                    writer.writerow([
                        winner.user.name,
                        winner.user.department,
                        winner.prize.name,
                        winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Command-line interface controlling the lottery process."""
from __future__ import annotations

from lottery_system.models import Prize, User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager, Winner

MENU = """
========== Team Building Smart Lottery ==========
 1. Load default participant list
 2. Upload custom participant list (txt)
 3. View department groups
 4. Load default prize library
 5. Add custom prize
 6. View prize sequence
 7. Reorder prize sequence
 8. Start lottery
 9. Export winner list
 0. Exit
================================================"""


def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer, e.g. 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive command-line controller for the whole lottery workflow."""

    def __init__(self) -> None:
        self._list_manager = ListManager()
        self._prize_manager = PrizeManager()
        self._users: list[User] = []
        self._results: ResultManager | None = None
        self._last_prizes: list[Prize] = []

    def run(self) -> None:
        """Main menu loop."""
        print("Welcome to the Team Building Smart Lottery System.")
        handlers = {
            "1": self._load_default_participants,
            "2": self._upload_custom_participants,
            "3": self._view_groups,
            "4": self._load_default_prizes,
            "5": self._add_custom_prize,
            "6": self._view_prizes,
            "7": self._reorder_prizes,
            "8": self._start_lottery,
            "9": self._export_results,
        }
        while True:
            print(MENU)
            choice = input("Select an option: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            handler = handlers.get(choice)
            if handler is None:
                print("Invalid option, please try again.")
                continue
            handler()

    # ----- participant list -------------------------------------------------

    def _load_default_participants(self) -> None:
        users, errors = self._list_manager.load_default_list()
        self._accept_participants(users, errors, "default list")

    def _upload_custom_participants(self) -> None:
        path = input("Enter the path of the txt file (Name,Department per line): ").strip().strip('"')
        if not path:
            print("Upload cancelled.")
            return
        users, errors = self._list_manager.load_custom_list(path)
        self._accept_participants(users, errors, path)

    def _accept_participants(self, users: list[User], errors: list[str], source: str) -> None:
        """Verify parsed users and let the operator decide how to handle problems."""
        valid_users, duplicates = self._list_manager.validate_users(users)
        if not errors and not duplicates:
            self._set_participants(valid_users)
            return

        print(f"Verification of '{source}' found problems:")
        for error in errors:
            print(f"  - {error}")
        for duplicate in duplicates:
            print(f"  - Duplicate name: {duplicate.name} ({duplicate.department})")

        while True:
            choice = input(
                "Enter 'r' to remove invalid/duplicate data and continue, "
                "or 't' to terminate the upload: "
            ).strip().lower()
            if choice == "r":
                if not valid_users:
                    print("No valid data remains; upload terminated.")
                    return
                self._set_participants(valid_users)
                return
            if choice == "t":
                print("Upload terminated; participant list unchanged.")
                return
            print("Please enter 'r' or 't'.")

    def _set_participants(self, users: list[User]) -> None:
        self._users = list(users)
        print(f"Loaded {len(self._users)} participants.")
        self._view_groups()

    def _view_groups(self) -> None:
        if not self._users:
            print("No participant list loaded.")
            return
        print("Department groups:")
        for department, count in self._list_manager.get_group_info(self._users).items():
            print(f"  {department}: {count} people")

    # ----- prizes -----------------------------------------------------------

    def _load_default_prizes(self) -> None:
        errors = self._prize_manager.load_default_prizes()
        if errors:
            for error in errors:
                print(f"  - {error}")
        else:
            print("Default prize library loaded.")

    def _add_custom_prize(self) -> None:
        name = input("Prize name: ").strip()
        quantity_text = input("Quantity: ").strip()
        if not quantity_text.isdigit() or int(quantity_text) < 1:
            print("Quantity must be a positive integer.")
            return
        if self._users:
            print("Available groups: " + ", ".join(self._list_manager.get_group_info(self._users)))
        groups_text = input("Avoidance groups (comma separated, optional): ").strip()
        avoid_groups = [g.strip() for g in groups_text.split(",") if g.strip()]
        drawer = input("Associated lottery person (optional): ").strip() or None
        try:
            prize = self._prize_manager.add_prize(name, int(quantity_text), avoid_groups, drawer)
        except ValueError as exc:
            print(f"Could not add prize: {exc}")
            return
        position = len(self._prize_manager.get_prizes())
        print(f"Prize '{prize.name}' added at sequence position {position}.")

    def _view_prizes(self) -> None:
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Prize sequence:")
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name} | Quantity: {prize.quantity}")
            print(f"     Rule: {prize.rule_description}")

    def _reorder_prizes(self) -> None:
        if not self._prize_manager.get_prizes():
            print("No prizes configured.")
            return
        self._view_prizes()
        text = input("Enter the new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(item) for item in text.split()]
        except ValueError:
            print("Invalid input: please enter integers only.")
            return
        if not self._prize_manager.reorder_prizes(order):
            print("Invalid order: each current serial number must appear exactly once.")
            return
        print("Prize sequence updated.")
        self._view_prizes()

    # ----- lottery process --------------------------------------------------

    def _start_lottery(self) -> None:
        if not self._users:
            print("Please load a participant list first.")
            return
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lottery = LotteryManager(self._users)
        results = ResultManager()
        self._results = results
        self._last_prizes = prizes

        print("\nLottery started. Commands during draws: [Enter] draw, 'pause', 'terminate'.")
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, len(prizes), lottery, results, prize)

        self._print_final_results(prizes, results)

    def _run_round(
        self,
        round_no: int,
        total_rounds: int,
        lottery: LotteryManager,
        results: ResultManager,
        prize: Prize,
    ) -> None:
        print("\n" + "-" * 60)
        print(f"Round {round_no}/{total_rounds}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {prize.rule_description}")
        print("-" * 60)

        round_winners: list[Winner] = []
        for position in range(1, prize.quantity + 1):
            if self._await_command(position) == "terminate":
                print("Round terminated by operator; remaining winners are skipped.")
                break
            print(f"Currently drawing the {ordinal(position)} winner...")
            user = lottery.draw(prize)
            if user is None:
                print("No eligible participants remain for this prize; the round ends.")
                break
            winner = results.add_winner(user, prize)
            round_winners.append(winner)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        print(f"\nResults of round {round_no}:")
        if not round_winners:
            print("  (no winners)")
        for winner in round_winners:
            print(
                f"  #{winner.winning_order}: {winner.user.name} "
                f"({winner.user.department}) - {prize.name}"
            )

    def _await_command(self, position: int) -> str:
        """Wait for the operator before the next draw. Returns 'draw' or 'terminate'."""
        while True:
            command = input(
                f"[Enter] draw the {ordinal(position)} winner | 'pause' | 'terminate': "
            ).strip().lower()
            if command == "":
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused.")
                while True:
                    resume = input("Type 'continue' to resume or 'terminate' to end this round: ").strip().lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        break
                    if resume == "terminate":
                        return "terminate"
                    print("Unknown command.")
                continue
            print("Unknown command.")

    def _print_final_results(self, prizes: list[Prize], results: ResultManager) -> None:
        print("\n========== Final Winner List ==========")
        for prize_name, winners in results.get_winners_by_prize(prizes).items():
            print(f"\n{prize_name}")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  #{winner.winning_order}: {winner.user.name} "
                    f"({winner.user.department}) at {winner.winning_time:%H:%M:%S}"
                )
        print("=======================================")

    # ----- export -----------------------------------------------------------

    def _export_results(self) -> None:
        if self._results is None or not self._results.get_winners():
            print("No lottery results to export. Run the lottery first.")
            return
        _, message = self._results.export_to_csv(self._last_prizes)
        print(message)


def main() -> int:
    """Program entry point."""
    try:
        LotteryCLI().run()
    except (KeyboardInterrupt, EOFError):
        print("\nExiting the lottery system.")
    return 0

"""Participant list loading, verification and group extraction."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_USER_LINES
from lottery_system.models import User


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (one 'Name,Department' per line)."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it into users and format errors."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines are skipped."""
        users: list[User] = []
        errors: list[str] = []
        for index, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            if "," in text:
                parts = [part.strip() for part in text.split(",")]
            elif "\t" in text:
                parts = [part.strip() for part in text.split("\t")]
            else:
                parts = [text]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {index}: Format Error - '{text}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence) and duplicate entries."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department -> number of people, sorted by department name."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

"""Prize library management."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZE_SPECS
from lottery_system.models import Prize


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        for prize in self._prizes:
            if prize.name == name:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid, drawer in DEFAULT_PRIZE_SPECS:
            if self._find(name) is not None:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            try:
                self.add_prize(name, quantity, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize '{clean_name}' already exists")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer")

        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = (group or "").strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in new order."""
        count = len(self._prizes)
        try:
            valid = sorted(order) == list(range(1, count + 1))
        except TypeError:
            return False
        if not valid:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: filtering, special rules, fallback and anti-duplication."""

from __future__ import annotations

import random

from lottery_system.constants import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User


def _norm(value: str | None) -> str:
    return (value or "").strip().casefold()


def _is_cheng_ru(user: User) -> bool:
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


class LotteryManager:
    """Executes lottery rounds while enforcing the lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.drawn_names: set[str] = set()
        # Cheng Ru group presence of the last two draws by Chen Qixian / Jin Duo.
        # True means the round produced at least one Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []
        self._round_winners: list[User] = []

    # ------------------------------------------------------------------
    # Rule engine
    # ------------------------------------------------------------------
    def is_cheng_ru_protection_active(self, prize: Prize) -> bool:
        """Fallback rule: Ma Jiaqi's pool is limited to Cheng Ru group members
        when the last two consecutive draws (Chen Qixian, Jin Duo) both
        contained no Cheng Ru group members."""
        return (
            _norm(prize.associated_drawer) == _norm(MA_JIAQI)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users eligible to win the given prize right now."""
        # Anti-duplication: previous winners (any prize) are always excluded.
        candidates = [user for user in self.users if user.name not in self.drawn_names]

        # Cheng Ru group protection (overrides avoidance groups).
        if self.is_cheng_ru_protection_active(prize):
            protected = [user for user in candidates if _is_cheng_ru(user)]
            if protected:
                return protected

        # Special rule: relaxed filtering for Li Bin.
        if _norm(prize.associated_drawer) == _norm(LI_BIN):
            return candidates

        # Basic filtering: exclude members of avoidance groups.
        avoid = {_norm(group) for group in prize.avoid_groups}
        return [user for user in candidates if _norm(user.department) not in avoid]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the current eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        parts = ["Previous winners are excluded"]
        if _norm(prize.associated_drawer) == _norm(LI_BIN):
            parts.append("Relaxed rule for Li Bin: group avoidance is not applied")
        elif prize.avoid_groups:
            parts.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        if self.is_cheng_ru_protection_active(prize):
            parts.append(
                "Cheng Ru Group protection ACTIVE: only Cheng Ru Group members are eligible"
            )
        return "; ".join(parts)

    # ------------------------------------------------------------------
    # Round execution
    # ------------------------------------------------------------------
    def start_round(self, prize: Prize) -> None:
        """Begin a new round for the given prize."""
        self._round_winners = []

    def draw_next(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no candidate exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self.drawn_names.add(winner.name)
        self._round_winners.append(winner)
        return winner

    def end_round(self, prize: Prize) -> list[User]:
        """Finish the round, update fallback history and return round winners."""
        winners = list(self._round_winners)
        drawer = _norm(prize.associated_drawer)
        if drawer in (_norm(CHEN_QIXIAN), _norm(JIN_DUO)):
            had_cheng_ru = any(_is_cheng_ru(user) for user in winners)
            self.last_two_draw_results = (self.last_two_draw_results + [had_cheng_ru])[-2:]
        self._round_winners = []
        return winners

"""Winner recording, grouping, ordering and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import User

_LEVEL_WORDS: list[tuple[str, int]] = [
    ("special", 0),
    ("grand", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
    ("sixth", 6),
]


def prize_level_rank(prize_name: str) -> int:
    """Lower rank means higher prize level. Unknown levels rank last."""
    lowered = (prize_name or "").casefold()
    for word, rank in _LEVEL_WORDS:
        if word in lowered:
            return rank
    return 999


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    prize_order: int  # winning order within the same prize (1-based)
    sequence: int     # global winning order (1-based)


class ResultManager:
    """Keeps the complete winner list of a lottery run."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize) -> Winner:
        """Record a winner. `prize` is a Prize (or any object with a name)."""
        prize_name = getattr(prize, "name", str(prize))
        prize_order = sum(1 for w in self._winners if w.prize_name == prize_name) + 1
        winner = Winner(
            user=user,
            prize_name=prize_name,
            prize_order=prize_order,
            sequence=len(self._winners) + 1,
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """All winners in winning time order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered from high to low prize level.
        Within each prize, winners keep their winning time order."""
        ordered_prizes = sorted(prizes, key=lambda p: prize_level_rank(p.name))
        grouped: dict[str, list[Winner]] = {}
        for prize in ordered_prizes:
            grouped[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return grouped

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        grouped = self.get_winners_by_prize(prizes)
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize Name", "Name", "Department", "Prize Order", "Winning Sequence"])
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        writer.writerow([
                            prize_name,
                            winner.user.name,
                            winner.user.department,
                            winner.prize_order,
                            winner.sequence,
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def ordinal(number: int) -> str:
    """Return an English ordinal such as 1st, 2nd, 11th."""
    if 11 <= number % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _ask(prompt: str) -> str:
    return input(prompt).strip()


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.results: ResultManager | None = None
        self.lottery: LotteryManager | None = None

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def run(self) -> None:
        print("=" * 60)
        print("  Company Team Building - Smart Lottery System")
        print("=" * 60)
        while True:
            self._print_main_menu()
            choice = _ask("Select an option: ")
            if choice == "1":
                self._load_list_flow()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self._prize_menu()
            elif choice == "4":
                self._start_lottery()
            elif choice == "5":
                self._show_summary()
            elif choice == "6":
                self._export_results()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    @staticmethod
    def _print_main_menu() -> None:
        print("\nMain Menu:")
        print("  1. Load participant list")
        print("  2. View department groups")
        print("  3. Prize configuration")
        print("  4. Start lottery")
        print("  5. View winner summary")
        print("  6. Export results")
        print("  0. Exit")

    # ------------------------------------------------------------------
    # List management
    # ------------------------------------------------------------------
    def _load_list_flow(self) -> None:
        print("\nList source:")
        print("  1. Built-in default team building list")
        print("  2. Custom txt file (one 'Name,Department' per line)")
        print("  0. Cancel")
        choice = _ask("Select source: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = _ask("Enter txt file path: ").strip('"')
            if not path:
                print("Cancelled.")
                return
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Cancelled.")
            return

        accepted = self._verify_users(users, errors)
        if accepted is None:
            print("Upload terminated. Current participant list unchanged.")
            return
        self.users = accepted
        print(f"Loaded {len(self.users)} participants.")
        self._show_groups()

    def _verify_users(self, users: list[User], errors: list[str]) -> list[User] | None:
        """Verify list content. Returns accepted users, or None if terminated."""
        if errors:
            print(f"\n{len(errors)} format error(s) found:")
            for error in errors:
                print(f"  {error}")
            answer = _ask("Remove invalid data and continue? (r = remove and continue / t = terminate): ").lower()
            if answer != "r":
                return None

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"\n{len(duplicates)} duplicate name(s) found:")
            for user in duplicates:
                print(f"  {user.name} ({user.department})")
            answer = _ask("Remove duplicates and continue? (r = remove and continue / t = terminate): ").lower()
            if answer != "r":
                return None

        if not valid:
            print("No valid participants found.")
            return None
        return valid

    def _show_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("\nDepartment groups:")
        for department, count in info.items():
            print(f"  {department}: {count} people")

    # ------------------------------------------------------------------
    # Prize configuration
    # ------------------------------------------------------------------
    def _prize_menu(self) -> None:

            print("  1. Load preset prize library")
            print("  2. Add custom prize")
            print("  3. View prizes")
            print("  4. Reorder lottery rounds")

                self._load_default_prizes()
            elif choice == "2":
                self._add_prize()
            elif choice == "3":
                self._view_prizes()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "0":

            else:

    def _load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            for error in errors:
                print(f"  {error}")
        else:
            print("Preset prize library loaded.")

    def _ask_avoid_groups(self) -> list[str]:
        group_names = list(self.list_manager.get_group_info(self.users).keys()) if self.users else []
        if group_names:
            print("Available groups:")
            for index, name in enumerate(group_names, start=1):
                print(f"  {index}. {name}")
        raw = _ask("Avoidance groups (comma-separated names or numbers, blank for none): ")
        groups: list[str] = []
        for token in [item.strip() for item in raw.split(",") if item.strip()]:
            if token.isdigit() and group_names and 1 <= int(token) <= len(group_names):
                groups.append(group_names[int(token) - 1])
            else:
                groups.append(token)

    def _add_prize(self) -> None:
        name = _ask("Prize name: ")
        if not name:
            print("Prize name cannot be empty.")
            return
        quantity_text = _ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            print("Quantity must be a positive integer.")
            return
        groups = self._ask_avoid_groups()
        drawer = _ask("Associated lottery person (leave blank for none): ")
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer or None)
        except ValueError as exc:
            print(f"Failed to add prize: {exc}")
            return
        print(f"Prize added: {prize.name} (quantity {prize.quantity}).")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nPrize sequence:")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) or "none"
            drawer = prize.associated_drawer or "none"
            print(f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                  f"avoid: {avoid} | drawer: {drawer}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        raw = _ask("Enter new order as serial numbers separated by spaces: ")
        try:
            order = [int(token) for token in raw.split()]
        except ValueError:
            print("Invalid input: please enter serial numbers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
            self._view_prizes()
        else:
            print("Invalid order: every prize position must appear exactly once.")

    # ------------------------------------------------------------------
    # Lottery process
    # ------------------------------------------------------------------
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load the participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        self.results = ResultManager()
        self.lottery = LotteryManager(self.users)
        print("\nLottery started!")
        total = len(prizes)
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)
        print("\nAll lottery rounds completed.")
        self._print_summary(prizes)

    def _run_round(self, index: int, total: int, prize: Prize) -> list[User]:
        assert self.lottery is not None and self.results is not None
        print("\n" + "-" * 60)
        print(f"Round {index}/{total}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {self.lottery.describe_rules(prize)}")
        pool_size = len(self.lottery.get_eligible_pool(prize))
        print(f"Eligible candidates: {pool_size}")
        print("-" * 60)

        if pool_size == 0:
            print("No eligible candidates, this round is skipped.")
            self.lottery.end_round(prize)
            return []

        self.lottery.start_round(prize)
        drawn = 0
        while drawn < prize.quantity:
            action = self._prompt_action()
            if action == "terminate":
                print("Round terminated by operator.")
                break
            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = self.lottery.draw_next(prize)
            if user is None:
                print("No more eligible candidates, round ends early.")
                break
            drawn += 1
            self.results.add_winner(user, prize)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        winners = self.lottery.end_round(prize)
        print(f"\nRound {index} results ({prize.name}):")
        if not winners:
            print("  (no winners)")
        for position, user in enumerate(winners, start=1):
            print(f"  {position}. {user.name} ({user.department})")
        return winners

    @staticmethod
    def _prompt_action() -> str:
        """Wait for operator input. Returns 'draw' or 'terminate'."""
        while True:
            command = _ask("[Enter] draw next | 'pause' | 'terminate': ").lower()
            if command in ("", "continue"):
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    paused_command = _ask("paused> ").lower()
                    if paused_command == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if paused_command == "terminate":
                        return "terminate"
                    print("Unknown command. Use 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    # ------------------------------------------------------------------
    # Results
    # ------------------------------------------------------------------
    def _print_summary(self, prizes: list[Prize]) -> None:
        if self.results is None:
            print("No lottery results yet.")
            return
        grouped = self.results.get_winners_by_prize(prizes)
        print("\n" + "=" * 60)
        print("  Winner Summary (by prize level, high to low)")
        print("=" * 60)
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                self._print_winner_line(winner)

    @staticmethod
    def _print_winner_line(winner: Winner) -> None:
        print(f"  {winner.prize_order}. {winner.user.name} ({winner.user.department})")

    def _show_summary(self) -> None:
        if self.results is None:
            print("No lottery has been run yet.")
            return
        self._print_summary(self.prize_manager.get_prizes())

    def _export_results(self) -> None:
        if self.results is None:
            print("No lottery results to export. Run the lottery first.")
            return
        success, message = self.results.export_to_csv(self.prize_manager.get_prizes())
        print(message)


def main() -> None:
    """Entry point for the interactive lottery system."""
    try:
        LotterySystemCLI().run()
    except (EOFError, KeyboardInterrupt):
        print("\nExited.")

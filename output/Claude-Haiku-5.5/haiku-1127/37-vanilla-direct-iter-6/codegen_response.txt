class InputReader:
    """Reads stdin in a background thread so lottery commands can be polled."""

    def __init__(self, stream=None) -> None:
        self._stream = stream if stream is not None else sys.stdin
        self._queue: queue.Queue = queue.Queue()
        self._thread = threading.Thread(target=self._pump, name="stdin-reader", daemon=True)
        self._thread.start()

    def _pump(self) -> None:
        try:
            for line in self._stream:
                self._queue.put(line.strip())
        except (OSError, ValueError):
            pass
        self._queue.put(None)

    def _unwrap(self, item):
        if item is None:
            self._queue.put(None)  # keep the EOF marker available for later reads
            raise EOFError("Input stream closed.")
        return item

    def get(self) -> str:
        """Blocking read of one input line."""
        return self._unwrap(self._queue.get())

    def poll(self):
        """Non-blocking read; returns None when no input is pending."""
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        return self._unwrap(item)


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotterySystemCLI:
    """Drives the list management, prize configuration, lottery and result workflows."""

    def __init__(self, reader: InputReader | None = None, draw_delay: float = 0.5) -> None:
        self.reader = reader if reader is not None else InputReader()
        self.draw_delay = draw_delay
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.lottery_completed = False

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str) -> str:
        print(prompt, end="", flush=True)
        return self.reader.get().strip()

    def _ask_int(self, prompt: str) -> int | None:
        raw = self._ask(prompt)
        if not raw:
            return None
        try:
            return int(raw)
        except ValueError:
            print("Please enter a valid integer.")
            return None

    def _set_users(self, users: list[User]) -> None:
        self.users = list(users)
        self.lottery = None
        self.lottery_completed = False
        print(f"List loaded successfully: {len(self.users)} valid participant(s).")

    def _accept_users(self, users: list[User], errors: list[str]) -> None:
        valid, duplicates = self.list_manager.validate_users(users)

        if errors:
            print("Format errors detected:")
            for err in errors:
                print(f"  {err}")
        if duplicates:
            print("Duplicate names detected (later entries will be removed):")
            for dup in duplicates:
                print(f"  {dup.name} ({dup.department})")

        if not errors and not duplicates:
            self._set_users(valid)
            return

        choice = self._ask(
            "Enter 'remove' to discard invalid/duplicate data and continue, "
            "or 'terminate' to cancel the upload: "
        ).lower()
        if choice == "remove":
            if valid:
                self._set_users(valid)
            else:
                print("No valid participants remain. Upload cancelled.")
        else:
            print("Upload terminated. Current list is unchanged.")

    def _print_prizes(self, prizes: list[Prize]) -> None:
        for index, prize in enumerate(prizes, start=1):
            drawer = prize.associated_drawer or "-"
            groups = ", ".join(prize.avoid_groups) or "-"
            print(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"lottery person: {drawer} | avoid groups: {groups}"
            )

    def run(self) -> None:
        print("Welcome to the Team Building Smart Lottery System.")
        try:
            while True:
                print(MAIN_MENU)
                choice = self._ask("Select an option: ")
                if choice == "0":
                    break
                if choice == "1":
                    users, errors = self.list_manager.load_default_list()
                    self._accept_users(users, errors)
                elif choice == "2":
                    path = self._ask("Path to custom txt list: ").strip('"').strip()
                    users, errors = self.list_manager.load_custom_list(path)
                    self._accept_users(users, errors)
                elif choice == "3":
                    self.show_groups()
                elif choice == "4":
                    self.prize_menu()
                elif choice == "5":
                    self.start_lottery()
                elif choice == "6":
                    self.show_summary()
                elif choice == "7":
                    self.export_results()
                else:
                    print("Invalid option, please try again.")
        except EOFError:
            print("\nInput closed.")
        print("Goodbye!")

    # ------------------------------------------------------------ list actions
    def show_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current valid department groups:")
        for department, count in info.items():

    def prize_menu(self) -> None:

            print(PRIZE_MENU)
            choice = self._ask("Select an option: ")
            if choice == "0":
                return
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    print("Some default prizes could not be added:")
                    for err in errors:
                        print(f"  {err}")
                else:
                    print("Default prize library loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._set_avoid_groups()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                prizes = self.prize_manager.get_prizes()
                if not prizes:
                    print("No prizes configured.")
                else:
                    self._print_prizes(prizes)
            else:
                print("Invalid option, please try again.")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity = self._ask_int("Quantity: ")
        if quantity is None or quantity < 1:
            print("Error: quantity must be a positive integer.")
            return
        groups_raw = self._ask("Avoidance groups (comma-separated, empty for none): ")
        drawer = self._ask("Associated lottery person (empty for none): ")
        groups = [g.strip() for g in groups_raw.split(",") if g.strip()]
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer or None)
        except ValueError as exc:
            print(f"Error: {exc}")
            return
        print("Prize added successfully.")

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes(prizes)
        index = self._ask_int("Prize serial number: ")
        if index is None or not 1 <= index <= len(prizes):
            print("Invalid prize serial number.")
            return
        if self.users:
            print("Available groups: " + ", ".join(self.list_manager.get_group_info(self.users).keys()))
        raw = self._ask("Avoidance groups (comma-separated, empty to clear): ")
        groups = [g.strip() for g in raw.split(",") if g.strip()]
        if self.users:
            known = {d.casefold() for d in self.list_manager.get_group_info(self.users)}
            unknown = [g for g in groups if g.casefold() not in known]
            if unknown:
                print(f"Warning: group(s) not in current list: {', '.join(unknown)}")
        self.prize_manager.set_avoid_groups(index, groups)
        print("Avoidance groups updated.")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            print("At least two prizes are required to reorder.")
            return
        self._print_prizes(prizes)
        raw = self._ask("Enter new order as serial numbers (e.g. 3 1 2): ")
        try:
            order = [int(token) for token in raw.replace(",", " ").split()]
        except ValueError:
            print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated:")
            self._print_prizes(self.prize_manager.get_prizes())
        else:
            print(f"Invalid order: must contain each serial number 1-{len(prizes)} exactly once.")

    # ---------------------------------------------------------- lottery actions
    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return
        if self.lottery_completed:
            print("This lottery session is already completed. Reload a list to start a new session.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)

        print("\nLottery starts!")
        print(LOTTERY_CONTROLS)
        total = len(prizes)
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)
        self.lottery_completed = True
        print("\nAll lottery rounds completed.")
        self.show_summary()

    def _handle_controls(self) -> bool:
        """Process pending control commands. Returns True if the current round must be terminated."""
        command = self.reader.poll()
        if not command:
            return False
        command = command.lower()
        if command == "pause":
            print("[PAUSED] Enter 'continue' to resume or 'terminate' to end this round.")
            while True:
                answer = self.reader.get().lower()
                if answer == "continue":
                    print("[RESUMED] Lottery continues.")
                    return False
                if answer == "terminate":
                    return True
                print("Lottery is paused. Only 'continue' or 'terminate' are accepted.")
        if command == "terminate":
            return True
        if command == "continue":
            print("Lottery is not paused.")
            return False
        print(f"Unknown command '{command}' ignored. {LOTTERY_CONTROLS}")
        return False

    def _run_round(self, index: int, total: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {index}/{total} ===")
        print(f"Prize: {prize.name} | Quantity: {prize.quantity}")
        print(f"Rule: {LotteryManager.describe_rule(prize)}")

        winners: list[User] = []
        terminated = False
        for position in range(1, prize.quantity + 1):
            if self._handle_controls():
                terminated = True
                break
            print(f"Currently drawing the {_ordinal(position)} winner...")
            time.sleep(self.draw_delay)
            user = self.lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize.")
                break
            winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        self.lottery.finish_round(prize, winners)
        if terminated:
            print("Round terminated manually; remaining prizes of this round are skipped.")
        if winners:
            print(f"Round results for {prize.name} (winning order):")
            for order, user in enumerate(winners, start=1):
                print(f"  {order}. {user.name} ({user.department})")
        else:
            print("No winners in this round.")

    def show_summary(self) -> None:
        if self.lottery is None:
            print("No lottery results yet.")
            return
        by_prize = self.lottery.results.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n==== Winner List (by prize level) ====")
        if not by_prize:
            print("No prizes configured.")
            return
        for prize_name, winners in by_prize.items():
            print(f"{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  {winner.prize_order}. {winner.user.name} ({winner.user.department}) "
                    f"- {winner.winning_time}"
                )

    def export_results(self) -> None:
        if self.lottery is None:
            print("No lottery results to export yet.")
            return
        _, message = self.lottery.results.export_to_csv(self.prize_manager.get_prizes())
        print(message)


def main() -> None:
    cli = LotterySystemCLI()
    try:
        cli.run()
    except KeyboardInterrupt:
        print("\nInterrupted. Exiting.")

def _name_key(name: str) -> str:
    return " ".join(name.split()).casefold()


class ListManager:
    """Loads participant lists and provides validation and group information."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for number, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {number}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Keep the first occurrence of each name; report later occurrences as duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = _name_key(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from lottery_system.models import Prize


DEFAULT_PRIZE_LIBRARY: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),

class PrizeManager:
    """Maintains the ordered list of prizes for the current lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, groups, drawer in DEFAULT_PRIZE_LIBRARY:
            try:
                self.add_prize(name, quantity, list(groups), drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        prize = Prize(clean_name, quantity, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def set_avoid_groups(self, index: int, groups: list[str]) -> bool:
        """Set avoidance groups of the prize at 1-based position index."""
        if not 1 <= index <= len(self._prizes):
            return False
        self._prizes[index - 1].avoid_groups = [g.strip() for g in groups if g and g.strip()]
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """order: 1-based current positions, listed in the new sequence."""
        n = len(self._prizes)
        if not all(isinstance(x, int) and not isinstance(x, bool) for x in order):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: candidate filtering, fallback protection and draws."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager


CHENG_RU_GROUP = "cheng ru group"
LI_BIN = "li bin"
MA_JIAQI = "ma jiaqi"
FALLBACK_DRAWERS = frozenset({"chen qixian", "jin duo"})


def _norm(text: str | None) -> str:
    return " ".join((text or "").split()).casefold()


def _is_cheng_ru(user: User) -> bool:
    return _norm(user.department) == CHENG_RU_GROUP


class LotteryManager:
    """Runs lottery rounds over a fixed participant list while enforcing all rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._won_names: set[str] = set()
        self.results = ResultManager()
        self.last_two_draw_results: list[bool] = []
        self._rng = random.SystemRandom()

    # ----------------------------------------------------------- rule helpers
    def _is_won(self, user: User) -> bool:
        return _norm(user.name) in self._won_names

    def _fallback_active(self) -> bool:
        """True when the last two Chen Qixian / Jin Duo rounds both had no Cheng Ru members."""
        return len(self.last_two_draw_results) >= 2 and self.last_two_draw_results[-2:] == [False, False]

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = [u for u in self._users if not self._is_won(u)]
        drawer = _norm(prize.associated_drawer)

        # Cheng Ru group protection: overrides avoidance groups for Ma Jiaqi's prize.
        if drawer == MA_JIAQI and self._fallback_active():
            protected = [u for u in available if _is_cheng_ru(u)]
            if protected:
                return protected

        # Special rule: Li Bin's prizes only exclude previous winners.
        if drawer == LI_BIN:
            return available

        avoid = {_norm(group) for group in prize.avoid_groups}
        return [u for u in available if _norm(u.department) not in avoid]

    @staticmethod
    def describe_rule(prize: Prize) -> str:
        drawer = _norm(prize.associated_drawer)
        parts: list[str] = []
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}")
        if drawer == LI_BIN:
            parts.append("Only previous winners are excluded (avoidance groups not applied)")
        else:
            parts.append("Previous winners are excluded")
            if prize.avoid_groups:
                parts.append("Avoidance group(s) excluded: " + ", ".join(prize.avoid_groups))
        if drawer == MA_JIAQI:
            parts.append(
                "Fallback: if the last two results of Chen Qixian and Jin Duo both lack "
                "Cheng Ru Group members, the pool is limited to Cheng Ru Group members"
            )
        return "; ".join(parts)

    # ----------------------------------------------------------------- draws
    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._won_names.add(_norm(user.name))
        self.results.add_winner(user, prize)
        return user

    def finish_round(self, prize: Prize, winners: list[User]) -> None:
        """Record the round outcome so the fallback rule can be evaluated later."""
        if _norm(prize.associated_drawer) in FALLBACK_DRAWERS:
            has_cheng_ru = any(_is_cheng_ru(u) for u in winners)
            self.last_two_draw_results.append(has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def draw_round(self, prize: Prize) -> list[User]:
        """Non-interactive convenience: draw all winners of a prize in one go."""
        winners: list[User] = []
        for _ in range(prize.quantity):
            user = self.draw_one(prize)
            if user is None:
                break
            winners.append(user)
        self.finish_round(prize, winners)
        return winners

from __future__ import annotations

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


TIME_FORMAT = "%Y-%m-%d %H:%M:%S"

_LEVEL_WORDS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}
_UNKNOWN_LEVEL = 999


def prize_level(name: str) -> int:
    """Numeric level of a prize (1 = highest). Unknown levels sort last."""
    lowered = name.casefold()
    for word, level in _LEVEL_WORDS.items():
        if re.search(rf"\b{word}\b", lowered):
            return level
    match = re.search(r"\b(\d+)(?:st|nd|rd|th)\b", lowered)
    if match:
        return int(match.group(1))
    return _UNKNOWN_LEVEL


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Sort prizes from high level to low; ties keep configured order."""
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: (prize_level(item[1].name), item[0]))
    return [prize for _, prize in indexed]


@dataclass(frozen=True)
class Winner:
    user: User
    prize_name: str
    prize_order: int      # winning order within the same prize (1-based)
    winning_time: str


class ResultManager:
    """Keeps the complete winner list for the lottery session."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        same_prize = sum(1 for w in self._winners if w.prize_name == prize.name)
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_order=same_prize + 1,
            winning_time=datetime.now().strftime(TIME_FORMAT),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        result: dict[str, list[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            if prize.name in result:
                continue
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        by_prize = self.get_winners_by_prize(prizes)
        filename = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(filename, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in by_prize.items():
                    for winner in winners:
                        writer.writerow(
                            [winner.user.name, winner.user.department, prize_name, winner.winning_time]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {filename}"

"""Data loading utilities for the Beijing Subway dataset.

The dataset lives in ``subway_data.json`` next to this module. The file has the
form::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str, "distance": int, "line": str}]}

Distances are expressed in meters.
"""

import json
import os

# Absolute path of src/subway_data.json, valid regardless of the working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read or has an invalid structure."""


def _validate_structure(data):
    """Check that the top-level structure of the dataset is usable.

    Detailed content checks (distances, line consistency, ...) are performed
    later by the data integrity module.
    """
    if not isinstance(data, dict):
        raise DataLoadError("Dataset root must be a JSON object")
    for key in ("stations", "connections"):
        if key not in data:
            raise DataLoadError(f"Dataset is missing the '{key}' section")
        if not isinstance(data[key], list):
            raise DataLoadError(f"Dataset section '{key}' must be a list")


def load_subway_data(path=DATA_FILE_PATH) -> dict:
    """Load and return the JSON content of the subway dataset.

    Args:
        path: Path of the JSON file. Defaults to ``src/subway_data.json``.

    Returns:
        The parsed dataset as a dictionary.

    Raises:
        DataLoadError: If the file is missing, unreadable, or malformed.
    """
    path = os.fspath(path)
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataLoadError(f"Unable to read data file: {exc}") from exc

    _validate_structure(data)
    return data


def save_subway_data(data, path=DATA_FILE_PATH) -> None:
    """Write the dataset dictionary back to a JSON file.

    Args:
        data: Dataset dictionary in the same format returned by ``load_subway_data``.
        path: Destination file path.

    Raises:
        DataLoadError: If the file cannot be written.
    """
    path = os.fspath(path)
    try:
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, ensure_ascii=False, indent=2)
            handle.write("\n")
    except OSError as exc:
        raise DataLoadError(f"Unable to write data file: {exc}") from exc

"""Core graph data structures for the subway network.

Stations are identified by integer numbers in the graph. The mapping from
station names to numbers is maintained by the caller (see ``network.py``).
The graph is a weighted undirected graph stored as adjacency lists.
"""


class Edge:
    """A weighted, undirected connection between two stations.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance in meters.
        line: Name of the subway line the connection belongs to.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by path algorithms (the distance in meters)."""
        return self.distance

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``.

        Raises:
            ValueError: If ``vertex`` is not an endpoint of this edge.
        """
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex} is not an endpoint of this edge")

    def __repr__(self):
        return (f"Edge({self.source!r}, {self.dest!r}, "
                f"distance={self.distance!r}, line={self.line!r})")


class Graph:
    """Weighted undirected graph implemented with adjacency lists.

    Vertices are station numbers (integers). Each edge is stored in the
    adjacency list of both of its endpoints.

    Attributes:
        station_lines: Mapping from station name to the list of line names
            the station belongs to.
    """

    def __init__(self):
        self._adj = {}
        self.station_lines = {}

    # ----- vertex management -------------------------------------------------

    def add_vertex(self, vertex):
        """Add a vertex if it is not already present."""
        self._adj.setdefault(vertex, [])

    def has_vertex(self, vertex):
        """Return True if the vertex exists in the graph."""
        return vertex in self._adj

    def vertices(self):
        """Return a list of all vertices."""
        return list(self._adj.keys())

    def vertex_count(self):
        """Return the number of vertices."""
        return len(self._adj)

    def set_station_lines(self, station_name, lines):
        """Record the line names a station belongs to."""
        self.station_lines[station_name] = list(lines)

    # ----- edge management ---------------------------------------------------

    def add_edge(self, edge):
        """Insert an undirected edge.

        If an edge between the same pair of stations on the same line already
        exists, it is replaced.

        Raises:
            ValueError: If the edge is a self-loop.
        """
        if edge.source == edge.dest:
            raise ValueError("Self-loop edges are not allowed")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        for old in self.find_edges(edge.source, edge.dest, edge.line):
            self._remove_edge_object(old)
        self._adj[edge.source].append(edge)
        self._adj[edge.dest].append(edge)

    def find_edges(self, u, v, line=None):
        """Return all edges between ``u`` and ``v`` (optionally on one line)."""
        if u not in self._adj:
            return []
        result = []
        for edge in self._adj[u]:
            if edge.other(u) == v and (line is None or edge.line == line):
                result.append(edge)
        return result

    def adjacent(self, vertex):
        """Return the list of edges incident to ``vertex``."""
        return self._adj.get(vertex, [])

    def edges(self):
        """Return every edge exactly once."""
        result = []
        for vertex, incident in self._adj.items():
            for edge in incident:
                if edge.source == vertex:
                    result.append(edge)
        return result

    def edge_count(self):
        """Return the number of undirected edges."""
        return len(self.edges())

    def _remove_edge_object(self, edge):
        """Remove an edge object from both endpoint adjacency lists."""
        self._adj[edge.source].remove(edge)
        self._adj[edge.dest].remove(edge)

"""Shortest path computation for the subway network.

Dijkstra's algorithm is run over a state space of ``(station, line)`` pairs so
that line changes (transfers) can be counted and constrained. A priority queue
implemented as a binary heap (``MinPQ``) provides the O((V+E) log V) bound.

Query modes:
    shortest_distance: minimise total distance in meters.
    least_transfer:    minimise number of transfers, then distance.
    comprehensive:     minimise a blended cost: distance plus penalties for
                       each transfer and each segment on a non-preferred line.
"""

TRANSFER_PENALTY_M = 1500.0
NON_PREFERRED_PENALTY_M = 300.0
VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")


class MinPQ:
    """Min-priority queue backed by a binary heap.

    Keys are compared with the ``<`` operator. Values are arbitrary objects.
    """

    def __init__(self):
        self._keys = []
        self._values = []

    def __len__(self):
        return len(self._keys)

    def is_empty(self):
        """Return True if the queue holds no elements."""
        return not self._keys

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._keys:
            raise IndexError("priority queue is empty")
        return self._keys[0]

    def insert(self, key, value):
        """Insert a (key, value) pair."""
        self._keys.append(key)
        self._values.append(value)
        self._sift_up(len(self._keys) - 1)

    def del_min(self):
        """Remove and return the (key, value) pair with the smallest key."""
        if not self._keys:
            raise IndexError("priority queue is empty")
        key, value = self._keys[0], self._values[0]
        last_key = self._keys.pop()
        last_value = self._values.pop()
        if self._keys:
            self._keys[0] = last_key
            self._values[0] = last_value
            self._sift_down(0)
        return key, value

    def _swap(self, i, j):
        self._keys[i], self._keys[j] = self._keys[j], self._keys[i]
        self._values[i], self._values[j] = self._values[j], self._values[i]

    def _sift_up(self, index):
        while index > 0:
            parent = (index - 1) // 2
            if self._keys[index] < self._keys[parent]:
                self._swap(index, parent)
                index = parent
            else:
                break

    def _sift_down(self, index):
        size = len(self._keys)
        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index
            if left < size and self._keys[left] < self._keys[smallest]:
                smallest = left
            if right < size and self._keys[right] < self._keys[smallest]:
                smallest = right
            if smallest == index:
                break
            self._swap(index, smallest)
            index = smallest


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of ``VALID_MODES``.
        max_transfers: Maximum allowed number of transfers, or None for no limit.
        preferred_lines: Line names the passenger prefers to ride.
    """

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])

    @property
    def mode(self):
        return self._mode

    @mode.setter
    def mode(self, value):
        if value not in VALID_MODES:
            raise ValueError(f"Unknown mode '{value}', expected one of {', '.join(VALID_MODES)}")
        self._mode = value

    @property
    def max_transfers(self):
        return self._max_transfers

    @max_transfers.setter
    def max_transfers(self, value):
        if value is not None:
            if isinstance(value, bool) or not isinstance(value, int) or value < 0:
                raise ValueError("max_transfers must be a non-negative integer or None")
        self._max_transfers = value


class ShortestPath:
    """Dijkstra-based shortest path search from a single source station.

    Attributes:
        dist_to: Mapping from station number to the total distance (meters)
            of the best path found from the source.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"Source station {source} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        # state (station, arrival_line) -> cost tuple
        self._cost = {}
        # state -> (distance, transfers, non_preferred_count, previous_state, edge)
        self._info = {}
        # station number -> best final state
        self._best_state = {}
        self._run()

    @staticmethod
    def _make_cost(mode, distance, transfers, non_preferred):
        """Build a comparable cost tuple for the given mode."""
        if mode == "shortest_distance":
            return (distance, non_preferred)
        if mode == "least_transfer":
            return (transfers, non_preferred, distance)
        # comprehensive
        return (distance + TRANSFER_PENALTY_M * transfers
                + NON_PREFERRED_PENALTY_M * non_preferred,)

    def _run(self):
        """Execute Dijkstra's algorithm over the (station, line) state space."""
        cfg = self.config
        mode = cfg.mode
        max_transfers = cfg.max_transfers
        preferred = set(cfg.preferred_lines)

        start = (self.source, None)
        start_cost = self._make_cost(mode, 0, 0, 0)
        self._cost[start] = start_cost
        self._info[start] = (0, 0, 0, None, None)

        pq = MinPQ()
        pq.insert(start_cost, start)

        while not pq.is_empty():
            cost, state = pq.del_min()
            if cost != self._cost[state]:
                continue  # stale queue entry
            vertex, line = state
            distance, transfers, non_preferred = self._info[state][0:3]

            for edge in self.graph.adjacent(vertex):
                nxt = edge.other(vertex)
                new_transfers = transfers + (1 if line is not None and edge.line != line else 0)
                if max_transfers is not None and new_transfers > max_transfers:
                    continue
                new_non_preferred = non_preferred + (
                    1 if preferred and edge.line not in preferred else 0)
                new_distance = distance + edge.weight
                new_cost = self._make_cost(mode, new_distance, new_transfers, new_non_preferred)
                next_state = (nxt, edge.line)
                old_cost = self._cost.get(next_state)
                if old_cost is None or new_cost < old_cost:
                    self._cost[next_state] = new_cost
                    self._info[next_state] = (new_distance, new_transfers,
                                              new_non_preferred, state, edge)
                    pq.insert(new_cost, next_state)

        # Select the best state for every reached station.
        for state, cost in self._cost.items():
            vertex = state[0]
            best = self._best_state.get(vertex)
            if best is None or cost < self._cost[best]:
                self._best_state[vertex] = state
        for vertex, state in self._best_state.items():
            self.dist_to[vertex] = self._info[state][0]

    def _require_state(self, target):
        state = self._best_state.get(target)
        if state is None:
            raise ValueError(f"No path from station {self.source} to station {target}")
        return state

    def has_path(self, target):
        """Return True if ``target`` is reachable under the current configuration."""
        return target in self._best_state

    def path_with_edges(self, target):
        """Backtrack the optimal path to ``target``.

        Returns:
            A tuple ``(stations, edges)`` where ``stations`` is the list of
            station numbers from source to target and ``edges`` the list of
            edges traversed between consecutive stations.

        Raises:
            ValueError: If the target is unreachable.
        """
        state = self._require_state(target)
        stations = []
        edges = []
        while state is not None:
            stations.append(state[0])
            info = self._info[state]
            if info[4] is not None:
                edges.append(info[4])
            state = info[3]
        stations.reverse()
        edges.reverse()
        return stations, edges

    def get_transfer_count(self, target):
        """Return the number of line transfers on the optimal path to ``target``."""
        return self._info[self._require_state(target)][1]

"""Fare and travel time calculation for the Beijing Subway.

Fare structure:
    0 - 6 km        : 3 yuan   (distance < 6000 m)
    6 - 12 km       : 4 yuan   (6000 m <= distance < 12000 m)
    12 - 22 km      : 5 yuan   (12000 m <= distance < 22000 m)
    22 - 32 km      : 6 yuan   (22000 m <= distance < 32000 m)
    over 32 km      : 7 yuan, plus 1 yuan for every further 20 km

Fractional meters are rounded up before tier lookup.
"""

import math

# (upper bound in meters exclusive, fare in yuan) for the basic/progressive tiers.
BASIC_FARE_TIERS = (
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
)
EXTENDED_BASE_DISTANCE_M = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP_M = 20000

AVERAGE_SPEED_KMH = 35.0
BOARDING_TIME_MIN = 2.0
TRANSFER_TIME_MIN = 5.0


def _validate_distance(distance):
    """Validate a non-negative, finite numeric distance in meters."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number of meters")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance) -> int:
    """Calculate the subway fare in yuan for a travel distance.

    Args:
        distance: Travel distance in meters (fractional values are rounded up).

    Returns:
        The fare in yuan as an integer.

    Raises:
        TypeError: If distance is not numeric.
        ValueError: If distance is negative, NaN or infinite.
    """
    _validate_distance(distance)
    meters = math.ceil(distance)
    for upper_bound, fare in BASIC_FARE_TIERS:
        if meters < upper_bound:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_BASE_DISTANCE_M) // EXTENDED_STEP_M


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimate the travel time in minutes.

    The estimate combines a fixed boarding allowance, travel time at an
    average operating speed, and a fixed walking allowance per transfer.

    Args:
        distance: Travel distance in meters.
        transfer_count: Number of line transfers.

    Returns:
        Estimated travel time in minutes, rounded to one decimal place.
    """
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise ValueError("transfer_count must be a non-negative integer")
    travel_minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    total = BOARDING_TIME_MIN + travel_minutes + transfer_count * TRANSFER_TIME_MIN
    return round(total, 1)

"""Subway network management.

``SubwayNetwork`` wraps the graph model and owns the mapping between official
station names and integer station numbers. It also supports data updates:
adding lines, modifying stations, and updating distances.
"""

import re

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
from models import Edge, Graph

_SPACE_RE = re.compile(r"\s+")
_SEPARATOR_CHARS_RE = re.compile(r"[\s\-'\u2019\u00b7]")


def _clean(text):
    """Collapse whitespace and strip a station or line name."""
    if not isinstance(text, str):
        raise TypeError("Names must be strings")
    return _SPACE_RE.sub(" ", text).strip()


def _normalize_key(text):
    """Build a lookup key that ignores case, spacing, punctuation and suffixes."""
    key = _SEPARATOR_CHARS_RE.sub("", text.casefold())
    if key.endswith("station"):
        key = key[:-len("station")]
    if key.endswith("\u7ad9"):  # Chinese suffix for "station"
        key = key[:-1]
    return key


class SubwayNetwork:
    """Subway network model built from the dataset.

    Attributes:
        graph: The weighted undirected ``Graph`` of the network.
        station_to_number: Mapping from official station name to station number.
        number_to_station: Mapping from station number to station name.
    """

    def __init__(self, data=None, path=DATA_FILE_PATH):
        if data is None:
            data = load_subway_data(path)
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        self._build(data)

    # ----- construction ------------------------------------------------------

    def _build(self, data):
        """Populate the network from a dataset dictionary."""
        for item in data.get("stations", []):
            name = self._register_station(item["name"])
            current = self.graph.station_lines[name]
            for line in item.get("lines", []):
                line = _clean(str(line))
                if line and line not in current:
                    current.append(line)

        for conn in data.get("connections", []):
            source = self._register_station(conn["source"])
            dest = self._register_station(conn["dest"])
            line = _clean(str(conn["line"]))
            distance = int(conn["distance"])
            self.graph.add_edge(Edge(self.station_to_number[source],
                                     self.station_to_number[dest],
                                     distance, line))

    def _register_station(self, name):
        """Register a station if unknown and return its cleaned name."""
        clean = _clean(name)
        if not clean:
            raise ValueError("Station name cannot be empty")
        if clean not in self.station_to_number:
            number = max(self.number_to_station, default=0) + 1
            self.station_to_number[clean] = number
            self.number_to_station[number] = clean
            self.graph.add_vertex(number)
            self.graph.set_station_lines(clean, [])
        return clean

    # ----- queries -----------------------------------------------------------

    def station_names(self):
        """Return all station names sorted alphabetically."""
        return sorted(self.station_to_number)

    def has_station(self, name):
        """Return True if the exact station name exists."""
        return _clean(name) in self.station_to_number

    def standardize_station_name(self, name):
        """Map user input to the official station name.

        Returns:
            The official name, or None if no station matches.
        """
        clean = _clean(name)
        if clean in self.station_to_number:
            return clean
        key = _normalize_key(clean)
        if not key:
            return None
        for candidate in self.station_to_number:
            if _normalize_key(candidate) == key:
                return candidate
        return None

    def number_of(self, name):
        """Return the station number of an official station name."""
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the official station name of a station number."""
        return self.number_to_station[number]

    def lines_of(self, name):
        """Return the line names a station belongs to."""
        return list(self.graph.station_lines.get(name, []))

    def all_lines(self):
        """Return every line name present in the network, sorted."""
        lines = set()
        for station_lines in self.graph.station_lines.values():
            lines.update(station_lines)
        for edge in self.graph.edges():
            lines.add(edge.line)
        return sorted(lines)

    def stations_on_line(self, line):
        """Return the names of stations served by a line, sorted by name."""
        numbers = set()
        for edge in self.graph.edges():
            if edge.line == line:
                numbers.add(edge.source)
                numbers.add(edge.dest)
        return sorted(self.number_to_station[n] for n in numbers)

    # ----- updates -----------------------------------------------------------

    def _require_station(self, name):
        canonical = self.standardize_station_name(name)
        if canonical is None:
            raise ValueError(f"Station '{name}' does not exist")
        return canonical

    @staticmethod
    def _validate_distance(distance):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise ValueError("Distance must be a number of meters")
        if distance <= 0:
            raise ValueError("Distance must be a positive number of meters")

    def add_connection(self, source, dest, distance, line):
        """Add a connection between two existing stations on a line."""
        src = self._require_station(source)
        dst = self._require_station(dest)
        if src == dst:
            raise ValueError("A station cannot connect to itself")
        line = _clean(line)
        if not line:
            raise ValueError("Line name cannot be empty")
        self._validate_distance(distance)
        self.graph.add_edge(Edge(self.number_of(src), self.number_of(dst),
                                 int(round(distance)), line))

    def add_line(self, line, station_names, distances):
        """Add a new line through an ordered list of stations.

        Args:
            line: New line name.
            station_names: Ordered station names along the line. Unknown
                stations are created.
            distances: Distances in meters between consecutive stations.

        Raises:
            ValueError: If the line already exists or the input is invalid.
        """
        line = _clean(line)
        if not line:
            raise ValueError("Line name cannot be empty")
        if line in self.all_lines():
            raise ValueError(f"Line '{line}' already exists")
        names = [_clean(s) for s in station_names]
        if len(names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(names) - 1:
            raise ValueError("Exactly one distance is required between consecutive stations")
        for distance in distances:
            self._validate_distance(distance)

        for name in names:
            registered = self._register_station(name)
            station_lines = self.graph.station_lines[registered]
            if line not in station_lines:
                station_lines.append(line)
        for a, b, distance in zip(names, names[1:], distances):
            self.add_connection(a, b, distance, line)

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance of existing connection(s) between two stations.

        Returns:
            The number of connections updated.
        """
        src = self._require_station(source)
        dst = self._require_station(dest)
        self._validate_distance(distance)
        edges = self.graph.find_edges(self.number_of(src), self.number_of(dst),
                                      _clean(line) if line else None)
        if not edges:
            raise ValueError(f"No connection between '{src}' and '{dst}'")
        for edge in edges:
            edge.distance = int(round(distance))
        return len(edges)

    def modify_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line affiliations.

        Returns:
            The station's name after the modification.
        """
        current = self._require_station(name)
        target = current
        if new_name is not None:
            target = _clean(new_name)
            if not target:
                raise ValueError("New station name cannot be empty")
            if target != current and self.standardize_station_name(target) is not None:
                raise ValueError(f"Station '{target}' already exists")
        if target != current:
            number = self.station_to_number.pop(current)
            self.station_to_number[target] = number
            self.number_to_station[number] = target
            self.graph.station_lines[target] = self.graph.station_lines.pop(current)
        if lines is not None:
            cleaned = []
            for line in lines:

"""Path visualization and export.

Builds a structured result describing a route (stations, line segments,
transfers, distance, fare, time) and renders it as console text or JSON.
"""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def _build_segments(names, edges):
    """Group consecutive edges that share a line into segments.

    Args:
        names: Station names along the path (length = len(edges) + 1).
        edges: Edges traversed along the path.
    """
    segments = []
    if not edges:
        return segments
    start = 0
    for i in range(1, len(edges) + 1):
        if i == len(edges) or edges[i].line != edges[start].line:
            seg_edges = edges[start:i]
            segments.append({
                "line": edges[start].line,
                "from": names[start],
                "to": names[i],
                "stations": names[start:i + 1],
                "stop_count": i - start,
                "distance_m": sum(e.distance for e in seg_edges),
            })
            start = i
    return segments


def build_route_result(network, start, end, mode, stations, edges, transfer_count,
                       max_transfers=None, preferred_lines=None):
    """Assemble the full route result and its summary.

    Args:
        network: The ``SubwayNetwork`` used for name lookups.
        start: Starting station name.
        end: Ending station name.
        mode: Query mode used.
        stations: Station numbers from source to target.
        edges: Edges along the path.
        transfer_count: Number of transfers on the path.
        max_transfers: Transfer limit used for the query (or None).
        preferred_lines: Preferred lines used for the query (or None).

    Returns:
        A dictionary with query info, path, segments, transfers, summary fields.
    """
    names = [network.name_of(n) for n in stations]
    total_m = sum(e.distance for e in edges)
    segments = _build_segments(names, edges)

    transfers = []
    for previous, following in zip(segments, segments[1:]):
        transfers.append({
            "station": previous["to"],
            "from_line": previous["line"],
            "to_line": following["line"],
        })

    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])

    fare = Sum_money(total_m)
    estimated_time = calculate_estimated_time(total_m, transfer_count)

    return {
        "query": {
            "start": start,
            "end": end,
            "mode": mode,
            "max_transfers": max_transfers,
            "line_preference": list(preferred_lines or []),
        },
        "path": names,
        "segments": segments,
        "transfers": transfers,
        "distance_m": total_m,
        "distance_km": round(total_m / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": estimated_time,
        "transfer_count": transfer_count,
        "lines": lines,
    }


def format_text(result):
    """Render a route result as human-readable console text."""
    query = result["query"]
    out = []
    out.append("=" * 60)
    out.append(f"Route: {query['start']} -> {query['end']}")
    out.append(f"Query mode: {query['mode']}")
    if query["max_transfers"] is not None:
        out.append(f"Max transfers: {query['max_transfers']}")
    if query["line_preference"]:
        out.append(f"Preferred lines: {', '.join(query['line_preference'])}")
    out.append("-" * 60)

    path = result["path"]
    out.append(f"Station sequence ({len(path)} stations):")
    for index, name in enumerate(path, 1):
        out.append(f"  {index:>3}. {name}")

    out.append("")
    out.append("Line segments:")
    if not result["segments"]:
        out.append("  (no travel required)")
    for index, seg in enumerate(result["segments"], 1):
        out.append(f"  [{index}] {seg['line']}: {seg['from']} -> {seg['to']} "
                   f"({seg['stop_count']} stops, {seg['distance_m']} m)")

    out.append("")
    out.append("Transfers:")
    if not result["transfers"]:
        out.append("  None")
    for transfer in result["transfers"]:
        out.append(f"  Transfer at {transfer['station']}: "
                   f"{transfer['from_line']} -> {transfer['to_line']}")

    out.append("")
    out.append("Summary:")
    out.append(f"  Total distance: {result['distance_m']} m "
               f"({result['distance_km']:.2f} km)")
    out.append(f"  Number of transfers: {result['transfer_count']}")
    out.append(f"  Lines involved: {', '.join(result['lines']) if result['lines'] else 'None'}")
    out.append(f"  Estimated time: {result['estimated_time_min']} min")
    out.append(f"  Fare: {result['fare']} yuan")
    out.append("=" * 60)
    return "\n".join(out)


def format_json(result):
    """Render a route result as an indented JSON document."""
    return json.dumps(result, ensure_ascii=False, indent=2)

"""Data integrity checks for the subway network.

Checks cover:
    * coverage of main lines and representative stations;
    * connection integrity (endpoints, line consistency, duplicates);
    * distance accuracy (positive, plausible values);
    * network connectivity.
"""

REQUIRED_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

REPRESENTATIVE_STATIONS = {
    "Line 1": ["Pingguoyuan", "Dongdan"],
    "Line 2": ["Xizhimen", "Dongzhimen"],
    "Line 4": ["Beijing Zoo"],
    "Line 5": ["Songjiazhuang"],
}

MAX_PLAUSIBLE_DISTANCE_M = 6000
MIN_PLAUSIBLE_DISTANCE_M = 50


class _UnionFind:
    """Minimal disjoint-set structure used for connectivity checks."""

    def __init__(self, items):
        self._parent = {item: item for item in items}

    def find(self, item):
        while self._parent[item] != item:
            self._parent[item] = self._parent[self._parent[item]]
            item = self._parent[item]
        return item

    def union(self, a, b):
        root_a, root_b = self.find(a), self.find(b)
        if root_a != root_b:
            self._parent[root_b] = root_a

    def component_count(self):
        return len({self.find(item) for item in self._parent})


def run_integrity_checks(network):
    """Run all integrity checks on a ``SubwayNetwork``.

    Returns:
        A dictionary with keys ``errors``, ``warnings`` (lists of strings),
        ``station_count`` and ``connection_count``.
    """
    data = network.to_dict()
    errors = []
    warnings = []

    stations = data["stations"]
    connections = data["connections"]
    station_names = {s["name"] for s in stations}
    station_lines = {s["name"]: set(s["lines"]) for s in stations}
    connection_lines = {c["line"] for c in connections}

    # 1. Coverage of main lines and representative stations.
    declared_lines = set()
    for lines in station_lines.values():
        declared_lines.update(lines)
    for line in REQUIRED_LINES:
        if line not in connection_lines and line not in declared_lines:
            warnings.append(f"Required line '{line}' is not present in the dataset")
    for line, representatives in REPRESENTATIVE_STATIONS.items():
        for station in representatives:
            if station not in station_names:
                warnings.append(f"Representative station '{station}' of {line} is missing")

    # 2. Station data.
    for station in stations:
        if not station["lines"]:
            errors.append(f"Station '{station['name']}' has no line affiliation")

    # 3. Connection integrity and distance accuracy.
    seen = set()
    union = _UnionFind(station_names)
    used_lines = set()
    for index, conn in enumerate(connections, 1):
        source, dest = conn["source"], conn["dest"]
        line, distance = conn["line"], conn["distance"]
        label = f"Connection #{index} ({source} - {dest}, {line})"

        if source not in station_names or dest not in station_names:
            errors.append(f"{label}: endpoint station is not in the station list")
            continue
        if source == dest:
            errors.append(f"{label}: station connects to itself")
            continue

        if isinstance(distance, bool) or not isinstance(distance, int) or distance <= 0:
            errors.append(f"{label}: distance must be a positive integer (meters)")
        elif distance > MAX_PLAUSIBLE_DISTANCE_M or distance < MIN_PLAUSIBLE_DISTANCE_M:
            warnings.append(f"{label}: distance {distance} m is outside the plausible range")

        for end in (source, dest):
            if line not in station_lines[end]:
                errors.append(f"{label}: line '{line}' is not listed for station '{end}'")

        key = (frozenset((source, dest)), line)
        if key in seen:
            warnings.append(f"{label}: duplicate connection")
        seen.add(key)

        union.union(source, dest)
        used_lines.add(line)

    # 4. Connectivity.
    isolated = [s for s in station_names if not any(
        c["source"] == s or c["dest"] == s for c in connections)]
    for name in sorted(isolated):
        warnings.append(f"Station '{name}' has no connections")
    components = union.component_count()
    if components > 1:
        warnings.append(f"Network has {components} disconnected components")

    for line in sorted(declared_lines - used_lines):
        warnings.append(f"Line '{line}' is declared on stations but has no connections")

    return {
        "errors": errors,
        "warnings": warnings,
        "station_count": len(station_names),
        "connection_count": len(connections),
    }


def format_integrity_report(result):
    """Render the output of ``run_integrity_checks`` as console text."""
    lines = [
        "Data integrity report",
        f"  Stations: {result['station_count']}",
        f"  Connections: {result['connection_count']}",
        f"  Errors: {len(result['errors'])}",
        f"  Warnings: {len(result['warnings'])}",
    ]
    for message in result["errors"]:
        lines.append(f"  [ERROR] {message}")
    for message in result["warnings"]:
        lines.append(f"  [WARN] {message}")
    if not result["errors"] and not result["warnings"]:
        lines.append("  Data passed all checks.")
    elif not result["errors"]:
        lines.append("  Data passed with warnings.")
    else:
        lines.append("  Data failed integrity checks.")
    return "\n".join(lines)

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import re

from data_check import format_integrity_report, run_integrity_checks
from data_loader import DATA_FILE_PATH, DataLoadError
from network import SubwayNetwork
from pathfinding import VALID_MODES, PathfindingConfig, ShortestPath
from route_report import build_route_result, format_json, format_text

INPUT_FORMAT_MESSAGE = (
    "Input format incorrect, please use 'Starting Station; Ending Station' format"
)
_SEPARATOR_RE = re.compile(r"\s*[;\uff1b,\uff0c]\s*")

HELP_TEXT = """Commands:
  <Start>; <End>      Query the route and fare, e.g. "Xizhimen; Dongzhimen"
  mode <name>         Change query mode: shortest_distance, least_transfer, comprehensive
  lines               List all subway lines
  help                Show this help
  quit                Exit the program"""


class QueryError(Exception):
    """A user-facing error in a query (input, station, or calculation)."""


def parse_args(argv=None):
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", default=None,
                        help="file that receives the most recent query result")
    parser.add_argument("--format", choices=["text", "json"], default="text",
                        help="output format (default: text)")
    parser.add_argument("--mode", choices=VALID_MODES, default="shortest_distance",
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="maximum number of allowed transfers")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line(s), comma separated, e.g. 'Line 1'")
    parser.add_argument("--check_data", action="store_true",
                        help="run data integrity checks and exit")
    return parser.parse_args(argv)


def parse_query(text):
    """Split a 'Starting Station; Ending Station' input into two names."""
    parts = [p.strip() for p in _SEPARATOR_RE.split(text.strip())]
    if len(parts) != 2 or not all(parts):
        raise QueryError(INPUT_FORMAT_MESSAGE)
    return parts[0], parts[1]


def resolve_station(network, name, role):
    """Standardize a station name and verify that it exists."""
    canonical = network.standardize_station_name(name)
    if canonical is None:
        raise QueryError(f"{role} station '{name}' does not exist, please check the station name")
    return canonical


def normalize_line_name(token, known_lines):
    """Map user input such as '1' or 'line 1' to a known line name."""
    text = " ".join(token.split())
    if re.fullmatch(r"\d+", text):
        text = f"Line {text}"
    for known in known_lines:
        if known.casefold() == text.casefold():
            return known
    raise QueryError(f"Unknown line '{token.strip()}' for --line_preference")


def parse_preferences(text, known_lines):
    """Parse a comma-separated line preference string."""
    if not text:
        return []
    result = []
    for token in text.split(","):
        if token.strip():
            name = normalize_line_name(token, known_lines)
            if name not in result:
                result.append(name)
    return result


def run_query(network, start, end, config):
    """Compute the route between two official station names.

    Returns:
        The route result dictionary built by ``build_route_result``.

    Raises:
        QueryError: If no path exists or path calculation fails.
    """
    try:
        source = network.number_of(start)
        target = network.number_of(end)
        searcher = ShortestPath(network.graph, source, config)
        if not searcher.has_path(target):
            raise QueryError(
                f"No route found from '{start}' to '{end}' with the current settings")
        stations, edges = searcher.path_with_edges(target)
        transfers = searcher.get_transfer_count(target)
    except (KeyError, ValueError) as exc:
        raise QueryError(f"Path calculation failed: {exc}") from exc

    return build_route_result(network, start, end, config.mode, stations, edges,
                              transfers, config.max_transfers, config.preferred_lines)


def render(result, output_format):
    """Render a result in the requested format."""
    if output_format == "json":
        return format_json(result)
    return format_text(result)


class QuerySession:
    """Interactive continuous-query session."""

    def __init__(self, network, config, output_format="text", output_file=None):
        self.network = network
        self.config = config
        self.output_format = output_format
        self.output_file = output_file

    def run(self):
        """Run the read-evaluate-print loop until the user quits."""
        print("Beijing Subway fare calculator. Type 'help' for commands.")
        while True:
            try:
                raw = input("Query> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            text = raw.strip()
            if not text:
                continue
            lowered = text.lower()
            if lowered in ("quit", "exit"):
                break
            if lowered == "help":
                print(HELP_TEXT)
            elif lowered == "lines":
                print(", ".join(self.network.all_lines()))
            elif lowered.startswith("mode"):
                self._change_mode(text)
            else:
                self._process_query(text)
        return 0

    def _change_mode(self, text):
        parts = text.split()
        if len(parts) != 2 or parts[1] not in VALID_MODES:
            print(f"Error: usage 'mode <{'|'.join(VALID_MODES)}>'")
            return
        self.config.mode = parts[1]
        print(f"Query mode set to {parts[1]}")

    def _process_query(self, text):
        try:
            start_raw, end_raw = parse_query(text)
            start = resolve_station(self.network, start_raw, "Starting")
            end = resolve_station(self.network, end_raw, "Ending")
            if start == end:
                raise QueryError("Starting station cannot be the same as the ending station")
            result = run_query(self.network, start, end, self.config)
        except QueryError as exc:
            print(f"Error: {exc}")
            return
        except Exception as exc:  # fare or other unexpected calculation failure
            print(f"Error: route or fare calculation failed: {exc}")
            return

        output = render(result, self.output_format)
        print(output)
        if self.output_file:
            try:
                with open(self.output_file, "w", encoding="utf-8") as handle:
                    handle.write(output + "\n")
                print(f"Result written to {self.output_file}")
            except OSError as exc:
                print(f"Error: unable to write output file: {exc}")


def main(argv=None):
    """Program entry point. Returns the process exit code."""
    args = parse_args(argv)

    try:
        network = SubwayNetwork(path=DATA_FILE_PATH)
    except (DataLoadError, KeyError, TypeError, ValueError) as exc:
        print(f"Error: unable to load subway data: {exc}")
        return 1

    if args.check_data:
        result = run_integrity_checks(network)
        print(format_integrity_report(result))
        return 1 if result["errors"] else 0

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer")
        return 2

    try:
        preferred = parse_preferences(args.line_preference, network.all_lines())
        config = PathfindingConfig(args.mode, args.max_transfers, preferred)
    except (QueryError, ValueError) as exc:
        print(f"Error: {exc}")
        return 2

    session = QuerySession(network, config, args.format, args.output_file)
    return session.run()

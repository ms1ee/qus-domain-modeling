class ListManager:
    """Loads, validates and summarizes participant lists."""

    def __init__(self):
        self.load_mode: str | None = None
        self.source_file_path: str | None = None
        self.loaded_at: datetime | None = None
        self.is_valid: bool = False

    def load_default_list(self) -> tuple[list[User], list[str]]:
        self.load_mode = "default"
        self.source_file_path = None
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        self._mark_loaded(users, errors)
        return users, errors

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        self.load_mode = "custom"
        self.source_file_path = path
        users, errors = self._load_from_path(path)
        self._mark_loaded(users, errors)
        return users, errors

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw in enumerate(lines, start=1):
            stripped = raw.strip()
            if not stripped:
                continue
            parts = stripped.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_number}: Format Error - '{raw}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def _mark_loaded(self, users, errors) -> None:
        self.loaded_at = datetime.now()
        self.is_valid = bool(users) and not errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from lottery_system.models import Prize


DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
    ("Technology Department Exclusive Coupon", 2, ["Product Department"], None),
]


class PrizeManager:
    """Manages the prize library and the lottery round order."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, avoid, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            self._prizes.append(Prize(name, quantity, list(avoid), drawer))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name must not be empty.")
        if not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._find(name) is not None:
            raise ValueError(f"Prize '{name}' already exists.")
        drawer = associated_drawer.strip() if associated_drawer else None
        self._prizes.append(
            Prize(
                name=name,
                quantity=quantity,
                avoid_groups=list(avoid_groups or []),
                associated_drawer=drawer or None,
            )
        )

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def get_prize_by_name(self, name) -> Prize | None:
        return self._find(name)

    def set_avoid_groups(self, prize_name, groups) -> bool:
        prize = self._find(prize_name)
        if prize is None:
            return False
        prize.avoid_groups = list(groups)
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

    def _find(self, name) -> Prize | None:
        for prize in self._prizes:
            if prize.name == name:
                return prize
        return None

import random

from lottery_system.models import Prize, User


CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"


def _norm(value) -> str:
    return (value or "").strip().casefold()


def _is_cheng_ru(user: User) -> bool:
    return _norm(user.department) == _norm(CHENG_RU_GROUP)


class LotteryManager:
    """Applies the lottery rule engine and performs draws."""

    def __init__(self, users):
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        self._winners: list[User] = []
        # Results (True = winner belongs to Cheng Ru group) of the last two
        # draws made by Chen Qixian / Jin Duo.
        self.last_two_draw_results: list[bool] = []

    @property
    def winners(self) -> list[User]:
        return list(self._winners)

    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def _base_pool(self) -> list[User]:
        return [u for u in self._users if u.name not in self._winner_names]

    def _is_cheng_ru_fallback_active(self) -> bool:
        return (
            len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        base = self._base_pool()
        drawer = _norm(prize.associated_drawer)

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group members,
        # overriding avoidance groups.
        if drawer == _norm(MA_JIAQI) and self._is_cheng_ru_fallback_active():
            return [u for u in base if _is_cheng_ru(u)]

        # Special rule: Li Bin only filters previous winners.
        if drawer == _norm(LI_BIN):
            return base

        avoid = {_norm(g) for g in prize.avoid_groups}
        if not avoid:
            return base
        return [u for u in base if _norm(u.department) not in avoid]

    def describe_rule(self, prize: Prize) -> str:
        parts = []
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}")
        parts.append("Previous winners are excluded")
        drawer = _norm(prize.associated_drawer)
        if drawer == _norm(LI_BIN):
            parts.append("Special rule: only previous winners are filtered (avoidance groups not applied)")
        elif drawer == _norm(MA_JIAQI) and self._is_cheng_ru_fallback_active():
            parts.append("Cheng Ru group protection: only Cheng Ru Group members are eligible")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        return "; ".join(parts)

    def draw_one(self, prize: Prize) -> User | None:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._record_winner(winner)

        drawer = _norm(prize.associated_drawer)
        if drawer in (_norm(CHEN_QIXIAN), _norm(JIN_DUO)):
            self.last_two_draw_results.append(_is_cheng_ru(winner))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return winner

    def _record_winner(self, user: User) -> None:
        if user.name in self._winner_names:
            raise ValueError(f"{user.name} has already won a prize.")
        self._winner_names.add(user.name)
        self._winners.append(user)

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""
    user: User
    prize_name: str = ""
    winning_order: int = 0
    won_at: datetime = field(default_factory=datetime.now)

    @property
    def name(self) -> str:
        return self.user.name

    @property
    def department(self) -> str:
        return self.user.department


class ResultManager:
    """Keeps winning records, groups them by prize and exports them."""

    def __init__(self):
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        if any(w.user.name == user.name for w in self._winners):
            raise ValueError(f"{user.name} has already won a prize.")
        winner = Winner(
            user=user,
            prize_name=prize.name,
            winning_order=len(self._winners) + 1,
            won_at=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            result.setdefault(prize.name, [])
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        by_prize = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for prize in prizes:
                    for winner in by_prize.get(prize.name, []):
                        writer.writerow([winner.name, winner.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from lottery_system.services.list_manager import ListManager

DRAWER_CHOICES = ["Chen Qixian", "Jin Duo", "Ma Jiaqi", "Li Bin"]


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def _ask(prompt: str) -> str:
    return input(prompt).strip()


class LotteryCLI:
    """Command-line interaction for the lottery process."""

    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.result_manager = ResultManager()

    # ------------------------------------------------------------------ #
    # Entry
    # ------------------------------------------------------------------ #
    def run(self) -> None:
        print("=" * 60)
        print("  Company Team Building - Smart Lottery System")
        print("=" * 60)
        users = self.setup_list()
        if users is None:
            print("Goodbye.")
            return
        self.users = users
        self.setup_prizes()
        print("Goodbye.")

    # ------------------------------------------------------------------ #
    # List management
    # ------------------------------------------------------------------ #
    def setup_list(self):
        while True:
            print("\n--- Participant List ---")
            print("1. Use built-in default team building list")
            print("2. Upload custom txt list (each line: Name,Department Group)")
            print("0. Exit")
            choice = _ask("Select: ")
            if choice == "0":
                return None
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
            elif choice == "2":
                path = _ask("Enter txt file path: ")
                users, errors = self.list_manager.load_custom_list(path)
            else:
                print("Invalid option.")
                continue

            if errors:
                print(f"\nFound {len(errors)} invalid line(s):")
                for err in errors:
                    print(f"  - {err}")
                action = self._ask_option(
                    "[r] remove invalid data and continue / [t] terminate upload: ",
                    ["r", "t"],
                )
                if action == "t":
                    print("Upload terminated.")
                    continue

            valid, duplicates = self.list_manager.validate_users(users)
            if duplicates:
                print(f"\nFound {len(duplicates)} duplicate name(s):")
                for user in duplicates:
                    print(f"  - {user.name} ({user.department})")
                action = self._ask_option(
                    "[r] remove duplicates (keep first) and continue / [t] terminate upload: ",
                    ["r", "t"],
                )
                if action == "t":
                    print("Upload terminated.")
                    continue

            if not valid:
                print("No valid participants found. Please try again.")
                continue

            self._show_groups(valid)
            return valid

    def _show_groups(self, users) -> None:
        info = self.list_manager.get_group_info(users)
        print(f"\nLoaded {len(users)} participant(s) in {len(info)} group(s):")
        for idx, (dept, count) in enumerate(info.items(), start=1):
            print(f"  {idx}. {dept}: {count} people")

    # ------------------------------------------------------------------ #
    # Prize configuration
    # ------------------------------------------------------------------ #
    def setup_prizes(self) -> None:
        while True:
            print("\n--- Prize Configuration ---")
            print("1. Load default prize library")
            print("2. Add custom prize")
            print("3. View prizes")
            print("4. Set avoid groups for a prize")
            print("5. Reorder prizes (lottery sequence)")
            print("6. View participant groups")
            print("7. Start lottery")
            print("0. Exit")
            choice = _ask("Select: ")
            if choice == "0":
                return
            if choice == "1":
                for err in self.prize_manager.load_default_prizes():
                    print(f"Note: {err}")
                print("Default prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._view_prizes()
            elif choice == "4":
                self._set_avoid_groups()
            elif choice == "5":
                self._reorder_prizes()
            elif choice == "6":
                self._show_groups(self.users)
            elif choice == "7":
                if self.start_lottery():
                    return
            else:
                print("Invalid option.")

    def _add_custom_prize(self) -> None:
        name = _ask("Prize name: ")
        try:
            quantity = int(_ask("Quantity: "))
        except ValueError:
            print("Quantity must be an integer.")
            return
        groups = self._select_groups()
        print("Associated lottery person (suggestions: " + ", ".join(DRAWER_CHOICES) + ")")
        drawer = _ask("Enter name (leave blank for none): ")
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer or None)
            print(f"Prize '{name.strip()}' added.")
        except ValueError as exc:
            print(f"Cannot add prize: {exc}")

    def _select_groups(self) -> list[str]:
        info = self.list_manager.get_group_info(self.users)
        groups = list(info.keys())
        if not groups:
            return []
        print("Available groups:")
        for idx, dept in enumerate(groups, start=1):
            print(f"  {idx}. {dept}")
        raw = _ask("Avoid groups (numbers separated by commas, blank for none): ")
        selected: list[str] = []
        for token in raw.replace(" ", "").split(","):
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                dept = groups[int(token) - 1]
                if dept not in selected:
                    selected.append(dept)
            else:
                print(f"Ignored invalid group selection: {token}")
        return selected

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nCurrent prize sequence:")
        for idx, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
            drawer = prize.associated_drawer or "None"
            print(f"  {idx}. {prize.name} | Quantity: {prize.quantity} | "
                  f"Lottery person: {drawer} | Avoid groups: {avoid}")

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        try:
            idx = int(_ask("Prize number: "))
            prize = prizes[idx - 1]
        except (ValueError, IndexError):
            print("Invalid prize number.")
            return
        groups = self._select_groups()
        self.prize_manager.set_avoid_groups(prize.name, groups)
        print(f"Avoid groups for '{prize.name}' set to: {', '.join(groups) or 'None'}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        raw = _ask("Enter new order as serial numbers (e.g. 2 1 3): ")
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(t) for t in tokens]
        except ValueError:
            print("Serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
            self._view_prizes()
        else:
            print("Invalid order: must contain every serial number exactly once.")

    # ------------------------------------------------------------------ #
    # Lottery process
    # ------------------------------------------------------------------ #
    def start_lottery(self) -> bool:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Please load or add prizes first.")
            return False
        if not self.users:
            print("No participants loaded.")
            return False

        self.result_manager = ResultManager()
        lottery = LotteryManager(self.users)

        print("\n" + "=" * 60)
        print("  LOTTERY START")
        print("=" * 60)
        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(round_no, prize, lottery)

        self._show_summary(prizes)
        self._ask_export(prizes)
        return True

    def _run_round(self, round_no, prize, lottery: LotteryManager) -> None:
        print(f"\n--- Round {round_no}: {prize.name} ---")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {lottery.describe_rule(prize)}")

        round_winners = []
        for draw_idx in range(1, prize.quantity + 1):
            command = self._before_draw()
            if command == "terminate":
                print("Current round terminated. Skipping remaining draws.")
                break

            print(f"Currently drawing the {ordinal(draw_idx)} winner...")
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            winner = self.result_manager.add_winner(user, prize)
            round_winners.append(winner)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        print(f"\nRound {round_no} results ({prize.name}):")
        if round_winners:
            for w in round_winners:
                print(f"  #{w.winning_order}: {w.name} ({w.department})")
        else:
            print("  No winners in this round.")

    def _before_draw(self) -> str:
        """Handles pause / continue / terminate commands before each draw.

        Returns 'next' to draw or 'terminate' to end the current round.
        """
        while True:
            cmd = _ask("Press Enter to draw next winner, or type 'pause' / 'terminate': ").lower()
            if cmd == "":
                return "next"
            if cmd == "terminate":
                return "terminate"
            if cmd == "pause":
                print("Lottery PAUSED. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    sub = _ask("> ").lower()
                    if sub == "continue":
                        print("Lottery resumed.")
                        return "next"
                    if sub == "terminate":
                        return "terminate"
                    print("Please type 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause', or 'terminate'.")

    def _show_summary(self, prizes) -> None:
        print("\n" + "=" * 60)
        print("  FINAL WINNER LIST")
        print("=" * 60)
        winners = self.result_manager.get_winners()
        if not winners:
            print("No winners.")
            return
        by_prize = self.result_manager.get_winners_by_prize(prizes)
        for prize in prizes:
            print(f"\n{prize.name}:")
            items = by_prize.get(prize.name, [])
            if not items:
                print("  (no winners)")
            for w in items:
                print(f"  {w.name} ({w.department})")

    def _ask_export(self, prizes) -> None:
        choice = _ask("\nExport winner list to CSV file? (y/n): ").lower()
        if choice == "y":
            _, message = self.result_manager.export_to_csv(prizes)
            print(message)

    # ------------------------------------------------------------------ #
    # Helpers
    # ------------------------------------------------------------------ #
    def _ask_option(self, prompt: str, options: list[str]) -> str:
        while True:
            value = _ask(prompt).lower()
            if value in options:
                return value
            print(f"Please enter one of: {', '.join(options)}")


def main() -> None:
    try:
        LotteryCLI().run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting.")

"""Global variable store shared across the dispatch simulator modules.

The store is a simple module-level dictionary. ``_init()`` resets it to the
default values defined in the PRD, and ``set_value`` / ``get_value`` provide
access by key. The key ``"riders"`` holds the list of ``Rider`` objects.
"""

_global_dict = {}


def _init():
    """Initialise (or reset) the global variable store with default values.

    Default values:
        funds: 1000 (system operating funds)
        riders: empty list of Rider objects
        rider_id_counter: 0 (last issued rider ID)
        is_running: True (main loop control flag)
        current_tick: 0 (simulated time)
    """
    global _global_dict
    _global_dict = {
        "funds": 1000,
        "riders": [],
        "rider_id_counter": 0,
        "is_running": True,
        "current_tick": 0,
        "config": {},
    }


def set_value(key, value):
    """Store a value under the given key.

    Args:
        key: Name of the global variable.
        value: Value to store.
    """
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under the given key.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` if the key is missing.
    """
    return _global_dict.get(key, default)

"""Grid-based path planning for the food delivery simulator.

The map is 800x800 units divided into an 8x8 grid of 100x100 cells. Streets
are the grid lines (coordinates that are multiples of 100). Restaurants and
customers sit on street intersections. Movement always follows streets:
straight lines plus right-angle turns at intersections.
"""

import logging
import math

GRID_SIZE = 8
CELL_SIZE = 100
MAP_SIZE = GRID_SIZE * CELL_SIZE

logger = logging.getLogger(__name__)


def clamp(value, low, high):
    """Clamp ``value`` into the inclusive range [low, high]."""
    return max(low, min(high, value))


def to_cell_corner(x, y):
    """Convert a raw map coordinate to the top-left corner of its grid cell.

    Coordinates are clamped to the map boundary first, so abnormal values
    (negative or above 800) are handled safely.

    Args:
        x: Raw x coordinate.
        y: Raw y coordinate.

    Returns:
        Tuple ``(x, y)`` of integers, both multiples of 100 within 0-800.
    """
    x = clamp(int(x), 0, MAP_SIZE)
    y = clamp(int(y), 0, MAP_SIZE)
    return ((x // CELL_SIZE) * CELL_SIZE, (y // CELL_SIZE) * CELL_SIZE)


def grid_cell_center(x, y):
    """Return the centre point of the grid cell containing ``(x, y)``.

    Uses ``center_x = (grid_x // 100) * 100 + 50`` as specified in the PRD.
    """
    x = clamp(int(x), 0, MAP_SIZE - 1)
    y = clamp(int(y), 0, MAP_SIZE - 1)
    return ((x // CELL_SIZE) * CELL_SIZE + 50, (y // CELL_SIZE) * CELL_SIZE + 50)


class PathPlanner:
    """Plans and advances movement along the street grid.

    Attributes:
        grid_size (int): Number of cells per side (8).
        cell_size (int): Size of one cell in map units (100).
        map_size (int): Total map size in map units (800).
    """

    def __init__(self, grid_size=GRID_SIZE, cell_size=CELL_SIZE):
        """Create a planner for a square grid.

        Args:
            grid_size: Number of cells along each axis.
            cell_size: Edge length of a single cell in map units.
        """
        self.grid_size = grid_size
        self.cell_size = cell_size
        self.map_size = grid_size * cell_size

    def is_on_street(self, x, y):
        """Return True when the point lies on a street (a grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def next_position(self, current, target, speed):
        """Advance from ``current`` toward ``target`` by at most ``speed`` units.

        Rules:
            * If the rider is on a vertical street (x on a grid line) and not
              yet level with the target, it moves vertically first.
            * Otherwise it moves horizontally along the current street.
            * Once the x coordinate matches the target, it moves vertically
              along the target's street. This produces right-angle turns only
              at intersections and never leaves the street network.

        Args:
            current: ``(x, y)`` current position.
            target: ``(x, y)`` destination.
            speed: Maximum distance to travel this tick.

        Returns:
            New ``(x, y)`` position after the move.
        """
        x, y = current
        tx, ty = target
        remaining = max(0, int(speed))
        cs = self.cell_size

        while remaining > 0 and (x, y) != (tx, ty):
            if x % cs == 0 and y != ty:
                # Special case: directly above/below or on a vertical street.
                step = min(remaining, abs(ty - y))
                y += step if ty > y else -step
            elif x != tx:
                step = min(remaining, abs(tx - x))
                x += step if tx > x else -step
            else:  # pragma: no cover - unreachable for street-aligned targets
                logger.warning("Off-street target %s from %s", target, current)
                break
            remaining -= step
        return (x, y)

    def plan_route(self, start, end):
        """Compute a street-following route from ``start`` to ``end``.

        The route contains at most one right-angle turn and is returned as a
        list of waypoints including both endpoints.

        Args:
            start: ``(x, y)`` starting point.
            end: ``(x, y)`` destination.

        Returns:
            List of ``(x, y)`` waypoints.
        """
        sx, sy = start
        ex, ey = end
        if sx % self.cell_size == 0:
            corner = (sx, ey)
        else:
            corner = (ex, sy)
        points = [(sx, sy)]
        for point in (corner, (ex, ey)):
            if point != points[-1]:
                points.append(point)
        return points

    @staticmethod
    def route_length(points):
        """Return the total Manhattan length of a list of waypoints."""
        total = 0
        for (x1, y1), (x2, y2) in zip(points, points[1:]):
            total += abs(x2 - x1) + abs(y2 - y1)
        return total

    @staticmethod
    def estimate_ticks(distance, speed):
        """Return the number of ticks needed to cover ``distance`` at ``speed``."""
        if speed <= 0:
            return 0
        return int(math.ceil(distance / speed))

"""Rider model: movement, state transitions and delivery event logging.

Rider states:
    0 (STATE_IDLE): available for new orders.
    1 (STATE_PICKING_UP): travelling to the restaurant of the current order.
    2 (STATE_DELIVERING): travelling to the customer with the picked-up order.

Log events recorded in ``Rider.log``:
    "spawn", "accepted_order", "arrived_at_merchant", "delivered_to_customer".
"""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

RIDER_SPEED = 20
START_X = 400
START_Y = 400

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

PICKUP_STATUSES = ("accepted", "picking_up")
DELIVERY_STATUSES = ("delivering",)

_planner = PathPlanner()


class Rider:
    """A delivery rider who moves on the street grid and serves orders.

    Attributes:
        rider_id (int): Unique identifier.
        x (int): Current x coordinate.
        y (int): Current y coordinate.
        speed (int): Units travelled per tick (fixed at 20).
        state (int): 0 idle, 1 picking up, 2 delivering.
        orders (list[Order]): Assigned orders, including completed history.
        special_case (bool): True while the rider is level with its target
            on one axis (directly above, below, left or right of it).
        log (list[dict]): Status change records with keys "event",
            "order_id", "tick", "x", "y" and "state".
    """

    def __init__(self, rider_id):
        """Create a rider at the default start position (400, 400).

        Args:
            rider_id: Unique integer identifier.
        """
        self.rider_id = int(rider_id)
        self.x = START_X
        self.y = START_Y
        self.speed = RIDER_SPEED
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn")

    @property
    def state_name(self):
        """Human-readable name of the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def position(self):
        """Current ``(x, y)`` position."""
        return (self.x, self.y)

    def active_orders(self):
        """Return orders that have not been completed yet."""
        return [order for order in self.orders if order.status != "completed"]

    def accept_pending_orders(self):
        """Accept newly assigned orders and update the rider state.

        Orders that were appended to ``orders`` but not yet accepted are
        marked as accepted and an "accepted_order" event is logged. An idle
        rider with work transitions to the picking-up state.
        """
        for order in self.orders:
            if order.status in ("pending", "assigned"):
                order.status = "accepted"
                order.rider_id = self.rider_id
                self._record("accepted_order", order.id)
        if self.state == STATE_IDLE and self._find_order(PICKUP_STATUSES):
            self.state = STATE_PICKING_UP

    def move(self):
        """Advance the rider by one tick.

        Handles order acceptance, movement toward the current target and the
        state transitions on arrival at the restaurant and at the customer.
        """
        self.accept_pending_orders()
        if self.state == STATE_IDLE:
            return
        if self.state == STATE_PICKING_UP:
            self._move_to_restaurant()
        elif self.state == STATE_DELIVERING:
            self._move_to_customer()

    def to_dict(self):
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [order.id for order in self.orders],
            "log": self.log,
        }

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _move_to_restaurant(self):
        """Move toward the restaurant of the next order awaiting pickup."""
        order = self._find_order(PICKUP_STATUSES)
        if order is None:
            self._resync_state()
            return
        order.status = "picking_up"
        self._step(order.res_pos)
        if self._at(order.res_pos):
            order.status = "delivering"
            self.state = STATE_DELIVERING
            self.special_case = False
            self._record("arrived_at_merchant", order.id)

    def _move_to_customer(self):
        """Move toward the customer of the order currently being delivered."""
        order = self._find_order(DELIVERY_STATUSES)
        if order is None:
            self._resync_state()
            return
        self._step(order.cus_pos)
        if self._at(order.cus_pos):
            tick = globalvar.get_value("current_tick", 0)
            order.complete(tick)
            self._resync_state()
            self._record("delivered_to_customer", order.id)

    def _resync_state(self):
        """Set the state from the statuses of the orders held."""
        if self._find_order(DELIVERY_STATUSES):
            self.state = STATE_DELIVERING
        elif self._find_order(PICKUP_STATUSES):
            self.state = STATE_PICKING_UP
        else:
            self.state = STATE_IDLE
            self.special_case = False

    def _find_order(self, statuses):
        """Return the first order (in assignment order) with a matching status."""
        for order in self.orders:
            if order.status in statuses:
                return order
        return None

    def _step(self, target):
        """Move one tick toward ``target`` along the street grid."""
        tx, ty = target
        self.x, self.y = _planner.next_position((self.x, self.y), (tx, ty), self.speed)
        # Special case: exactly one axis aligned with the target (same street).
        self.special_case = (self.x == tx) != (self.y == ty)

    def _at(self, target):
        """Return True when the rider stands exactly on ``target``."""
        return (self.x, self.y) == (target[0], target[1])

    def _record(self, event, order_id=None):
        """Append a status change record to the rider log."""
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": globalvar.get_value("current_tick", 0),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("rider=%s event=%s order=%s", self.rider_id, event, order_id)

"""Order data model.

An order is described by ``[restaurant_x, restaurant_y, customer_x, customer_y]``.
Locations are converted to the top-left corner of their grid cell on creation.

Order status values:
    "pending"      created, not yet assigned to a rider
    "assigned"     assigned to a rider, not yet accepted
    "accepted"     accepted by the rider, waiting for pickup
    "picking_up"   rider is travelling to the restaurant
    "delivering"   food picked up, rider is travelling to the customer
    "completed"    delivered to the customer
"""

import itertools

import globalvar
from path_planner import PathPlanner, to_cell_corner

_planner = PathPlanner()


class Order:
    """A food delivery order.

    Attributes:
        id (int): Unique order identifier.
        res_pos (tuple[int, int]): Restaurant location (grid cell corner).
        cus_pos (tuple[int, int]): Customer location (grid cell corner).
        status (str): Lifecycle status, see module docstring.
        rider_id (int | None): ID of the assigned rider.
        start_time (int): Tick at which the order was created.
        distance (float): Manhattan distance restaurant to customer.
        time_taken (int | None): Ticks from creation to completion.
        completion_time (int | None): Tick at which delivery completed.
    """

    _id_sequence = itertools.count(1)

    def __init__(self, res_pos, cus_pos):
        """Create an order and convert its locations to grid cell corners.

        Args:
            res_pos: ``(x, y)`` restaurant coordinates.
            cus_pos: ``(x, y)`` customer coordinates.
        """
        self.id = next(Order._id_sequence)
        self.res_pos = to_cell_corner(*res_pos)
        self.cus_pos = to_cell_corner(*cus_pos)
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("current_tick", 0)
        self.distance = float(
            _planner.route_length(_planner.plan_route(self.res_pos, self.cus_pos))
        )
        self.time_taken = None
        self.completion_time = None

    def assign_to(self, rider_id):
        """Record the assigned rider and mark the order as assigned.

        Args:
            rider_id: ID of the rider receiving the order.
        """
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def complete(self, tick):
        """Mark the order as delivered and record timing data.

        Args:
            tick: Simulation tick at which delivery finished.
        """
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time

    def efficiency(self):
        """Return distance covered per tick, or None if not yet completed."""
        if not self.time_taken:
            return None
        return self.distance / self.time_taken

    def to_list(self):
        """Return the order as ``[restaurant_x, restaurant_y, customer_x, customer_y]``."""
        return [self.res_pos[0], self.res_pos[1], self.cus_pos[0], self.cus_pos[1]]

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "completion_time": self.completion_time,
        }

"""Order management: validation, creation, assignment and rider selection."""

import logging
import math

import globalvar
from order_model import Order

COORD_MIN = 0
COORD_MAX = 1000

logger = logging.getLogger(__name__)


def validate_coordinates(values):
    """Validate four integer coordinates in the range 0-1000.

    Args:
        values: Sequence of four values
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

    Returns:
        A list of the four validated integers.

    Raises:
        ValueError: If the count is wrong, a value is not an integer, or a
            value is out of range.
    """
    values = list(values)
    if len(values) != 4:
        raise ValueError(
            "Expected 4 coordinates: restaurant_x restaurant_y customer_x customer_y"
        )
    result = []
    for value in values:
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"Coordinate {value!r} is not an integer")
        if not COORD_MIN <= value <= COORD_MAX:
            raise ValueError(
                f"Coordinate {value} is out of range {COORD_MIN}-{COORD_MAX}"
            )
        result.append(value)
    return result


def _euclidean(x1, y1, x2, y2):
    """Return the Euclidean distance between two points."""
    return math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)


def find_best_rider_for_order(order):
    """Select the rider that should receive ``order``.

    The rider with the fewest active orders is preferred. Ties are broken by
    the distance from the rider to the restaurant.

    Args:
        order: The ``Order`` to be assigned.

    Returns:
        The selected ``Rider``, or None when no rider exists.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        logger.warning("No riders available for order %s", order.id)
        return None
    rest_x, rest_y = order.res_pos

    def sort_key(rider):
        return (
            len(rider.active_orders()),
            _euclidean(rider.x, rider.y, rest_x, rest_y),
        )

    return min(riders, key=sort_key)


class OrderManager:
    """Keeps the order registry and performs order assignment.

    Attributes:
        orders (list[Order]): All orders created during this session.
    """

    def __init__(self):
        """Create an empty order registry."""
        self.orders = []

    def create_order(self, restaurant_xy, customer_xy):
        """Validate coordinates and create a new pending order.

        Args:
            restaurant_xy: ``(x, y)`` restaurant coordinates.
            customer_xy: ``(x, y)`` customer coordinates.

        Returns:
            The newly created ``Order``.

        Raises:
            ValueError: If coordinates are invalid.
        """
        values = validate_coordinates([*restaurant_xy, *customer_xy])
        order = Order((values[0], values[1]), (values[2], values[3]))
        self.orders.append(order)
        logger.info("Created order %s", order.id)
        return order

    def assign_order(self, order, rider):
        """Assign ``order`` to ``rider`` by appending it to the rider's orders.

        Args:
            order: The ``Order`` to assign.
            rider: The ``Rider`` receiving the order.
        """
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        rider.accept_pending_orders()
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)

    def pending_orders(self):
        """Return orders that have not been assigned to any rider yet."""
        return [order for order in self.orders if order.status == "pending"]

    def completed_orders(self):
        """Return orders that have been delivered."""
        return [order for order in self.orders if order.status == "completed"]

"""System initialisation and the core SystemManager.

SystemManager owns the system funds, the rider list, the simulation tick,
and the order flow: order submission, automatic rider recruitment, dispatch,
and the simulation loop.
"""

import json
import logging

import globalvar
from order import OrderManager, find_best_rider_for_order
from rider import Rider, STATE_IDLE

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruit_threshold": 300,
    "recruit_cost": 500,
}


def _load_config(path):
    """Load system parameters from a JSON file, falling back to defaults.

    Args:
        path: Path to a JSON file, or None to use defaults only.

    Returns:
        Dictionary of configuration parameters.
    """
    config = dict(DEFAULT_CONFIG)
    if path is None:
        return config
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError:
        logger.warning("Config file %s not found; using defaults", path)
        return config
    except (OSError, json.JSONDecodeError) as exc:
        logger.error("Could not read config %s: %s; using defaults", path, exc)
        return config
    for key in DEFAULT_CONFIG:
        if key in data:
            try:
                config[key] = int(data[key])
            except (TypeError, ValueError):
                logger.error("Invalid value for config key %s; using default", key)
    return config


def init_system(config_path=None):
    """Initialise global variables and return a new SystemManager.

    Sets funds to the configured initial value (default 1000), an empty rider
    list, rider ID counter 0, running state True and tick 0.

    Args:
        config_path: Optional path to a JSON configuration file.

    Returns:
        A ready-to-use ``SystemManager``.
    """
    config = _load_config(config_path)
    globalvar._init()
    globalvar.set_value("config", config)
    globalvar.set_value("funds", config["initial_funds"])
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("is_running", True)
    globalvar.set_value("current_tick", 0)
    logger.info("System initialised with funds=%s", config["initial_funds"])
    return SystemManager()


class SystemManager:
    """Manages funds, riders, orders and the simulation clock.

    Attributes:
        config (dict): Active configuration parameters.
        order_manager (OrderManager): Order registry and assignment helper.
    """

    def __init__(self):
        """Bind to the global state initialised by ``init_system``."""
        self.config = globalvar.get_value("config") or dict(DEFAULT_CONFIG)
        self.order_manager = OrderManager()

    # ------------------------------------------------------------------
    # State accessors
    # ------------------------------------------------------------------
    @property
    def funds(self):
        """Current system funds (may be negative)."""
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        """List of all riders."""
        return globalvar.get_value("riders")

    @property
    def current_tick(self):
        """Current simulated time."""
        return globalvar.get_value("current_tick", 0)

    @property
    def is_running(self):
        """True while the main loop should keep running."""
        return globalvar.get_value("is_running", False)

    def stop(self):
        """Request the main loop to stop."""
        globalvar.set_value("is_running", False)

    def idle_riders(self):
        """Return riders in the idle state."""
        return [rider for rider in self.riders if rider.state == STATE_IDLE]

    # ------------------------------------------------------------------
    # Recruitment and dispatch
    # ------------------------------------------------------------------
    def recruit_rider(self):
        """Create a new rider if funds allow, deducting the recruitment cost.

        Returns:
            The new ``Rider``, or None when funds are below the threshold.
        """
        threshold = self.config["recruit_threshold"]
        cost = self.config["recruit_cost"]
        if self.funds < threshold:
            return None
        globalvar.set_value("funds", self.funds - cost)
        new_id = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", new_id)
        rider = Rider(new_id)
        self.riders.append(rider)
        logger.info("Recruited rider %s (funds now %s)", new_id, self.funds)
        return rider

    def _should_recruit(self, pending_count):
        """Return True when funds allow and pending orders exceed idle riders."""
        return (
            self.funds >= self.config["recruit_threshold"]
            and pending_count > len(self.idle_riders())
        )

    def _try_dispatch(self, order):
        """Attempt to assign ``order`` to a rider, recruiting one if needed.

        Returns:
            Tuple ``(rider, new_rider)`` where ``rider`` is the assigned rider
            or None, and ``new_rider`` is True when a rider was recruited.
        """
        pending_count = len(self.order_manager.pending_orders())
        recruited = None
        if self._should_recruit(pending_count):
            recruited = self.recruit_rider()
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None, recruited is not None
        self.order_manager.assign_order(order, rider)
        return rider, recruited is not None

    def submit_order(self, restaurant_xy, customer_xy):
        """Create an order from user coordinates and dispatch it.

        Args:
            restaurant_xy: ``(x, y)`` restaurant coordinates.
            customer_xy: ``(x, y)`` customer coordinates.

        Returns:
            Dictionary with keys "order" (Order), "rider" (Rider or None)
            and "new_rider" (Rider recruited for this order, or None).
        """
        order = self.order_manager.create_order(restaurant_xy, customer_xy)
        rider, recruited = self._try_dispatch(order)
        new_rider = self.riders[-1] if recruited else None
        return {"order": order, "rider": rider, "new_rider": new_rider}

    def dispatch_pending(self):
        """Retry assignment for orders that are still pending."""
        for order in self.order_manager.pending_orders():
            self._try_dispatch(order)

    # ------------------------------------------------------------------
    # Simulation loop
    # ------------------------------------------------------------------
    def tick(self):
        """Advance the simulation by one tick and move every rider."""
        globalvar.set_value("current_tick", self.current_tick + 1)
        for rider in list(self.riders):
            rider.move()
        self.dispatch_pending()

    def run_ticks(self, count):
        """Advance the simulation by ``count`` ticks."""
        for _ in range(count):
            self.tick()

    # ------------------------------------------------------------------
    # Monitoring and persistence
    # ------------------------------------------------------------------
    def check_consistency(self):
        """Verify internal state and return a list of problems (empty if OK)."""
        issues = []
        ids = [rider.rider_id for rider in self.riders]
        if len(ids) != len(set(ids)):
            issues.append("Duplicate rider IDs detected")
        for rider in self.riders:
            has_work = bool(rider.active_orders())
            if rider.state == STATE_IDLE and has_work:
                issues.append(f"Rider {rider.rider_id} is idle but has active orders")
            if rider.state != STATE_IDLE and not has_work:
                issues.append(f"Rider {rider.rider_id} is busy without active orders")
        return issues

    def snapshot(self):
        """Return a JSON-serialisable snapshot of the whole system."""
        return {
            "config": self.config,
            "funds": self.funds,
            "current_tick": self.current_tick,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in self.order_manager.orders],
        }

    def save_state(self, path):
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(self.snapshot(), handle, indent=2)
        logger.info("State saved to %s", path)

"""Status display and ASCII map rendering.

The ASCII map shows the 8x8 street grid (800x800 units) using 17 characters
per side: every 100 units is two characters apart, with the street characters
'#' on the grid lines and '.' for the cell interiors. Symbols:
    R  rider          S  restaurant       C  customer
    #  street         .  empty cell space *  planned route
"""

from path_planner import MAP_SIZE
from rider import STATE_NAMES

CANVAS_SIZE = 17
STEPS = CANVAS_SIZE - 1


class DisplayManager:
    """Renders system status, order tables, logs and the ASCII map.

    Attributes:
        map_size (int): Map size in units (800).
    """

    def __init__(self, map_size=MAP_SIZE):
        """Create a display manager.

        Args:
            map_size: Map size in coordinate units.
        """
        self.map_size = map_size

    def to_char(self, x, y):
        """Convert a map coordinate to a ``(row, col)`` canvas index."""
        col = round(x * STEPS / self.map_size)
        row = round(y * STEPS / self.map_size)
        col = min(max(col, 0), STEPS)
        row = min(max(row, 0), STEPS)
        return row, col

    def _base_grid(self):
        """Return an empty canvas with streets and cell interiors."""
        return [
            ["#" if (r % 2 == 0 or c % 2 == 0) else "." for c in range(CANVAS_SIZE)]
            for r in range(CANVAS_SIZE)
        ]

    def _put(self, grid, point, symbol):
        """Place ``symbol`` at the canvas position of a map point."""
        row, col = self.to_char(point[0], point[1])
        grid[row][col] = symbol

    def _draw_segment(self, grid, start, end):
        """Draw a route segment with '*' over street and empty characters."""
        r1, c1 = self.to_char(start[0], start[1])
        r2, c2 = self.to_char(end[0], end[1])
        if r1 == r2:
            cells = [(r1, c) for c in range(min(c1, c2), max(c1, c2) + 1)]
        elif c1 == c2:
            cells = [(r, c1) for r in range(min(r1, r2), max(r1, r2) + 1)]
        else:  # pragma: no cover - routes are always axis-aligned
            return
        for row, col in cells:
            if grid[row][col] in "#.":
                grid[row][col] = "*"

    def render_map(self, riders=(), orders=(), route=None):
        """Render the ASCII map with riders, active orders and an optional route.

        Args:
            riders: Iterable of riders to mark with 'R'.
            orders: Iterable of orders; active ones show 'S' and 'C'.
            route: Optional list of waypoints drawn as connected segments.

        Returns:
            Multi-line string of the map.
        """
        grid = self._base_grid()
        if route:
            for start, end in zip(route, route[1:]):
                self._draw_segment(grid, start, end)
        for order in orders:
            if order.status == "completed":
                continue
            self._put(grid, order.res_pos, "S")
            self._put(grid, order.cus_pos, "C")
        for rider in riders:
            self._put(grid, (rider.x, rider.y), "R")

        lines = []
        for row_index, row in enumerate(grid):
            if row_index % 2 == 0:
                label = f"{row_index * self.map_size // STEPS:>4} "
            else:
                label = "     "
            lines.append(label + "".join(row))
        lines.append("     x: 0 .. 800 (left to right), y: 0 .. 800 (top to bottom)")
        lines.append("Legend: R rider, S restaurant, C customer, # street, "
                     ". cell interior, * route")
        return "\n".join(lines)

    def render_status(self, system):
        """Return a text snapshot of funds, riders and pending orders.

        Args:
            system: The ``SystemManager`` instance.
        """
        riders = system.riders
        lines = [
            f"System funds: {system.funds}",
            f"Simulation tick: {system.current_tick}",
            f"Riders: {len(riders)}",
            f"Pending orders: {len(system.order_manager.pending_orders())}",
        ]
        for rider in riders:
            order_ids = [order.id for order in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id}: position=({rider.x}, {rider.y}) "
                f"status={rider.state_name} active orders={len(rider.active_orders())} "
                f"order list={order_ids}"
            )
        issues = system.check_consistency()
        if issues:
            lines.append("Consistency issues:")
            lines.extend(f"  - {issue}" for issue in issues)
        else:
            lines.append("Consistency: OK")
        return "\n".join(lines)

    def render_orders(self, orders):
        """Return a table of order details.

        Args:
            orders: Iterable of ``Order`` objects.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        header = (
            f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
            f"{'Distance':>9} {'Taken':>6} {'Rider':>6} {'Status':>10}"
        )
        lines = [header, "-" * len(header)]
        for order in orders:
            restaurant = f"({order.res_pos[0]},{order.res_pos[1]})"
            customer = f"({order.cus_pos[0]},{order.cus_pos[1]})"
            taken = "-" if order.time_taken is None else str(order.time_taken)
            rider = "-" if order.rider_id is None else f"#{order.rider_id}"
            lines.append(
                f"{order.id:>4} {order.start_time:>6} {restaurant:>12} {customer:>12} "
                f"{order.distance:>9.0f} {taken:>6} {rider:>6} {order.status:>10}"
            )
            if order.efficiency() is not None:
                lines.append(f"      delivery efficiency: {order.efficiency():.2f} units/tick")
        return "\n".join(lines)

    def format_log_entry(self, rider, entry):
        """Return one formatted rider log line.

        Args:
            rider: The ``Rider`` that produced the entry.
            entry: A dictionary from ``rider.log``.
        """
        order_text = "-" if entry["order_id"] is None else f"#{entry['order_id']}"
        state = STATE_NAMES.get(entry["state"], "unknown")
        return (
            f"[tick {entry['tick']:>4}] Rider #{rider.rider_id} "
            f"{entry['event']:<22} order={order_text:<6} "
            f"at ({entry['x']}, {entry['y']}) state={state}"
        )

    def render_rider_logs(self, riders):
        """Return the full status-change history of every rider.

        Args:
            riders: Iterable of ``Rider`` objects.
        """
        riders = list(riders)
        if not riders:
            return "No riders yet."
        lines = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} (speed {rider.speed}, "
                         f"status {rider.state_name}):")
            for entry in rider.log:
                lines.append("  " + self.format_log_entry(rider, entry))
        return "\n".join(lines)

"""Command parsing, shortcuts, history and input validation."""

import logging

from order import validate_coordinates

logger = logging.getLogger(__name__)


class CommandError(ValueError):
    """Raised when a command line cannot be parsed or validated."""


ALIASES = {
    "s": "start",
    "st": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "w": "save",
    "h": "history",
    "?": "help",
    "q": "quit",
    "e": "exit",
}

COMMANDS = (
    "start", "status", "orders", "riders", "map", "tick",
    "save", "history", "help", "quit", "exit",
)


class InputManager:
    """Parses user command lines and keeps the command history.

    Attributes:
        command_history (list[str]): Raw command lines entered so far.
    """

    def __init__(self):
        """Create an input manager with an empty history."""
        self.command_history = []

    def parse_line(self, line):
        """Parse a raw input line into a command name and arguments.

        Shortcuts supported: single-letter aliases (see ``ALIASES``), ``!!``
        to repeat the last command and ``!n`` to repeat history entry n.

        Args:
            line: Raw text entered by the user.

        Returns:
            Tuple ``(command, args)``; ``command`` is None for empty input.

        Raises:
            CommandError: If the command is unknown or the reference invalid.
        """
        text = line.strip()
        if not text:
            return None, []
        if text == "!!":
            if not self.command_history:
                raise CommandError("No previous command in history.")
            text = self.command_history[-1]
        elif text.startswith("!") and text[1:].isdigit():
            index = int(text[1:])
            if not 1 <= index <= len(self.command_history):
                raise CommandError(f"History entry {index} does not exist.")
            text = self.command_history[index - 1]

        self.command_history.append(text)
        tokens = text.split()
        name = tokens[0].lower()
        name = ALIASES.get(name, name)
        if name not in COMMANDS:
            raise CommandError(
                f"Unknown command '{tokens[0]}'. Type 'help' for the command list."
            )
        return name, tokens[1:]

    def parse_coordinates(self, tokens):
        """Convert four command tokens into validated integer coordinates.

        Args:
            tokens: Sequence of four strings
                ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            List of four integers in the range 0-1000.

        Raises:
            CommandError: If the count is wrong or a token is not an integer.
            ValueError: If a value is out of range.
        """
        if len(tokens) != 4:
            raise CommandError(
                "start needs four integers: restaurant_x restaurant_y "
                "customer_x customer_y"
            )
        values = []
        for token in tokens:
            try:
                values.append(int(token))
            except ValueError:
                raise CommandError(f"'{token}' is not an integer.") from None
        return validate_coordinates(values)

"""Command-line entry point for the food delivery dispatch simulator.

Usage:
    python src/main.py [--config CONFIG_FILE] [--debug]

Each ``tick`` command advances simulated time by one unit (or N units when
given an argument). Type ``help`` inside the program for the command list.

import argparse
import logging
import sys

import globalvar
from display import DisplayManager
from input_manager import CommandError, InputManager
from path_planner import PathPlanner
from system import init_system

logger = logging.getLogger("dispatch")

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Create an order (restaurant then customer, 0-1000) and dispatch it
  status               Show funds, riders and their states
  orders               Show all orders with timing and assignment details
  riders               Show every rider status change log entry
  map                  Show the ASCII map with riders and active orders
  tick [n]             Advance simulated time by n ticks (default 1)
  save [file]          Save system state to a JSON file (default dispatch_state.json)
  history              Show command history
  help                 Show this help
  quit / exit          Leave the program
Shortcuts: s, st, o, r, m, t, w, h, ?, q, e, !! (repeat last), !n (repeat entry n)"""


class Application:
    """Interactive session tying together system, display and input handling.

    Attributes:
        system (SystemManager): Core simulation manager.
        display (DisplayManager): Rendering helper.
        inputs (InputManager): Command parser and history.
        planner (PathPlanner): Route planner used for route previews.
    """

    def __init__(self, config_path=None):
        """Initialise the system and register the command handlers.

        Args:
            config_path: Optional JSON configuration file.
        """
        self.system = init_system(config_path)
        self.display = DisplayManager()
        self.inputs = InputManager()
        self.planner = PathPlanner()
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self):
        """Run the main interaction loop until the system stops."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands.")
        while self.system.is_running:
            try:
                line = input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                self.system.stop()
                break
            self.handle_line(line)
        print("Goodbye.")

    def handle_line(self, line):
        """Parse and execute one command line, recovering from any error.

        Args:
            line: Raw input text.
        """
        try:
            command, args = self.inputs.parse_line(line)
            if command is None:
                return
            self.handlers[command](args)
        except CommandError as exc:
            print(f"Error: {exc}")
        except ValueError as exc:
            print(f"Invalid input: {exc}")
        except Exception as exc:  # Error recovery: keep the session alive.
            logger.exception("Unexpected error while handling command")
            print(f"Unexpected error: {exc}. System state has been preserved.")

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def cmd_start(self, args):
        """Create and dispatch an order, showing the route on the map."""
        if not args:
            args = input("Enter restaurant_x restaurant_y customer_x customer_y: ").split()
        values = self.inputs.parse_coordinates(args)
        result = self.system.submit_order(values[:2], values[2:])
        order = result["order"]
        rider = result["rider"]
        new_rider = result["new_rider"]

        print(f"Order #{order.id} created at tick {order.start_time}")
        print(f"Restaurant grid point: ({order.res_pos[0]}, {order.res_pos[1]})")
        print(f"Customer grid point:   ({order.cus_pos[0]}, {order.cus_pos[1]})")
        if new_rider is not None:
            print(f"New rider added: yes (Rider #{new_rider.rider_id}); "
                  f"funds now {self.system.funds}")
        else:
            print("New rider added: no")

        if rider is None:
            print("No rider available. The order stays pending and will be retried each tick.")
            return

        print(f"Assigned to Rider #{rider.rider_id} (status: {rider.state_name}, "
              f"active orders: {len(rider.active_orders())})")
        route = self.planner.plan_route(rider.position, order.res_pos)
        leg = self.planner.plan_route(order.res_pos, order.cus_pos)
        full_route = route + leg[1:]
        distance = self.planner.route_length(full_route)
        ticks = self.planner.estimate_ticks(distance, rider.speed)
        print(f"Speed: {rider.speed} units/tick; route length: {distance} units; "
              f"estimated time: {ticks} ticks")
        print(self.display.render_map(riders=[rider], orders=[order], route=full_route))

    def cmd_status(self, args):
        """Print the system status snapshot."""
        print(self.display.render_status(self.system))

    def cmd_orders(self, args):
        """Print the details of all orders."""
        print(self.display.render_orders(self.system.order_manager.orders))

    def cmd_riders(self, args):
        """Print the status-change history of every rider."""
        print(self.display.render_rider_logs(self.system.riders))

    def cmd_map(self, args):
        """Print the ASCII map with riders and active orders."""
        active = [o for o in self.system.order_manager.orders if o.status != "completed"]
        print(self.display.render_map(riders=self.system.riders, orders=active))

    def cmd_tick(self, args):
        """Advance the simulation and print the events that occurred."""
        steps = 1
        if args:
            try:
                steps = int(args[0])
            except ValueError:
                raise CommandError(f"'{args[0]}' is not an integer.") from None
            if not 1 <= steps <= 10000:
                raise CommandError("Tick count must be between 1 and 10000.")
        for _ in range(steps):
            before = {rider.rider_id: len(rider.log) for rider in self.system.riders}
            self.system.tick()
            for rider in self.system.riders:
                for entry in rider.log[before.get(rider.rider_id, 0):]:
                    print(self.display.format_log_entry(rider, entry))
        print(f"Now at tick {self.system.current_tick}; funds {self.system.funds}; "
              f"riders {len(self.system.riders)}")

    def cmd_save(self, args):
        """Save the system state to a JSON file."""
        path = args[0] if args else "dispatch_state.json"
        self.system.save_state(path)
        print(f"State saved to {path}")

    def cmd_history(self, args):
        """Print the command history."""
        if not self.inputs.command_history:
            print("No commands yet.")
            return
        for index, entry in enumerate(self.inputs.command_history, start=1):
            print(f"{index:>4}  {entry}")

    def cmd_help(self, args):
        """Print the help text."""
        print(HELP_TEXT)

    def cmd_quit(self, args):
        """Stop the main loop."""
        self.system.stop()


def main(argv=None):
    """Parse arguments, configure logging and run the application.

    Args:
        argv: Optional argument list (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code.

    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=None, help="Path to a JSON config file")
    parser.add_argument("--debug", action="store_true", help="Enable debug logging")
    options = parser.parse_args(argv)

    logging.basicConfig(
        level=logging.DEBUG if options.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    app = Application(config_path=options.config)
    app.run()
    return 0

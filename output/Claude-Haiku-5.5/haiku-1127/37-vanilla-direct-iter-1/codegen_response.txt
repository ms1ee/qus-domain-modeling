def normalize_text(value: str) -> str:
    """Normalize text for case-insensitive, whitespace-tolerant comparison."""
    return " ".join(str(value).split()).casefold()


def same_text(a: str, b: str) -> bool:
    """Return True when two strings are equal after normalization."""
    return normalize_text(a) == normalize_text(b)

class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building participant list."""
        users = [User(name, department) for name, department in DEFAULT_PARTICIPANTS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file into users and a list of error messages."""
        users: list[User] = []
        errors: list[str] = []

        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        for line_no, raw_line in enumerate(lines, start=1):
            line = raw_line.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))

        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique users and duplicate entries.

        The first occurrence of a name is kept as valid; later occurrences
        are returned as duplicates.
        """
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to number of people."""
        counts: dict[str, int] = {}
        display: dict[str, str] = {}
        for user in users:
            key = normalize_text(user.department)
            display.setdefault(key, user.department)
            counts[key] = counts.get(key, 0) + 1
        return {display[key]: count for key, count in counts.items()}

from lottery_system.models import Prize, normalize_text, same_text

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5),
    ("Second Prize - Shopping Cards", 3),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Manages the prize library and the lottery round sequence."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def find_prize(self, name: str) -> Prize | None:
        """Find a prize by name (case-insensitive)."""
        for prize in self._prizes:
            if same_text(prize.name, name):
                return prize
        return None

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize to the end of the round sequence."""
        clean_name = " ".join(str(name or "").split())
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        if self.find_prize(clean_name) is not None:
            raise ValueError(f"Prize already exists: {clean_name}")

        groups = self._clean_groups(avoid_groups)
        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def set_avoid_groups(self, name: str, groups: list[str]) -> bool:
        """Replace the avoidance groups of an existing prize."""
        prize = self.find_prize(name)
        if prize is None:
            return False
        prize.avoid_groups = self._clean_groups(groups)
        return True

    def get_prizes(self) -> list[Prize]:
        """Return prizes in lottery round order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        ``order`` contains 1-based current positions in the desired new order,
        e.g. [3, 1, 2] means the third prize becomes first, the first second,
        and the second third.
        """
        count = len(self._prizes)
        if not isinstance(order, list) or len(order) != count:
            return False
        if any(isinstance(item, bool) or not isinstance(item, int) for item in order):
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[index - 1] for index in order]
        return True

    @staticmethod
    def _clean_groups(groups: list[str] | None) -> list[str]:
        cleaned: list[str] = []
        for group in groups or []:
            text = " ".join(str(group).split())
            if text and not any(same_text(text, existing) for existing in cleaned):
                cleaned.append(text)
        return cleaned

"""Lottery rule engine: eligibility filtering and winner drawing."""

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    normalize_text,
    same_text,
)


class LotteryManager:
    """Runs draws while enforcing the lottery rules.

    Rules implemented:
    - Previous winners are always excluded (anti-duplication).
    - Prizes with a lottery person exclude that prize's avoidance groups,
      except for the special "Li Bin" rule which only excludes winners.
    - Cheng Ru Group protection: if the last two draws of Chen Qixian / Jin Duo
      contained no Cheng Ru Group member, Ma Jiaqi's prize pool is limited to
      Cheng Ru Group members (ignoring avoidance groups).
    """

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.winners: list[User] = []
        self._winner_keys: set[str] = set()
        # Records, for the most recent draws by Chen Qixian / Jin Duo, whether
        # the round contained a Cheng Ru Group member. Keeps at most two entries.
        self.last_two_draw_results: list[bool] = []
        self._random = random.Random()

    def is_winner(self, user: User) -> bool:
        """Return True if the user has already won a prize."""
        return normalize_text(user.name) in self._winner_keys

    def _is_cheng_ru(self, user: User) -> bool:
        return same_text(user.department, CHENG_RU_GROUP)

    def is_cheng_ru_fallback_active(self) -> bool:
        """Return True when the Cheng Ru Group protection is triggered."""
        return (
            len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users who may win the given prize right now."""
        available = [user for user in self.users if not self.is_winner(user)]
        drawer = prize.associated_drawer

        if drawer and same_text(drawer, MA_JIAQI) and self.is_cheng_ru_fallback_active():
            return [user for user in available if self._is_cheng_ru(user)]

        if drawer and same_text(drawer, LI_BIN):
            return available

        return [
            user
            for user in available
            if not any(same_text(user.department, group) for group in prize.avoid_groups)
        ]

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._random.choice(pool)
        self._winner_keys.add(normalize_text(winner.name))
        self.winners.append(winner)
        return winner

    def record_round_result(self, prize: Prize, round_winners: list[User]) -> None:
        """Update the Cheng Ru Group protection state after a prize round."""
        drawer = prize.associated_drawer
        if drawer and (same_text(drawer, CHEN_QIXIAN) or same_text(drawer, JIN_DUO)):
            contains_cheng_ru = any(self._is_cheng_ru(user) for user in round_winners)
            self.last_two_draw_results.append(contains_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

"""Winner recording, ordering, and CSV/txt export."""

import csv
import os
import re
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, normalize_text

_LEVEL_KEYWORDS = (
    ("grand", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
)


def prize_level_rank(prize_name: str) -> int | None:
    """Return the level rank of a prize (smaller is higher), or None if unknown."""
    lowered = prize_name.casefold()
    for keyword, rank in _LEVEL_KEYWORDS:
        if re.search(rf"\b{keyword}\b", lowered):
            return rank
    return None


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    win_order: int
    won_at: datetime


class ResultManager:
    """Maintains the real-time winner list and exports results."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._winner_keys: set[str] = set()

    def add_winner(self, user: User, prize: Prize | str) -> Winner:
        """Record a winner. Raises ValueError if the person already won."""
        key = normalize_text(user.name)
        if key in self._winner_keys:
            raise ValueError(f"{user.name} has already won a prize.")
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        winner = Winner(
            user=user,
            prize_name=prize_name,
            win_order=len(self._winners) + 1,
            won_at=datetime.now(),
        )
        self._winners.append(winner)
        self._winner_keys.add(key)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name.

        Prizes are ordered from high level to low level; within a prize,
        winners are in winning time order.
        """
        ordered = self._sort_prizes(prizes)
        result: dict[str, list[Winner]] = {prize.name: [] for prize in ordered}
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    @staticmethod
    def _sort_prizes(prizes: list[Prize]) -> list[Prize]:
        def sort_key(item: tuple[int, Prize]) -> tuple[int, int]:
            index, prize = item
            rank = prize_level_rank(prize.name)
            return (rank if rank is not None else 1000, index)

        return [prize for _, prize in sorted(enumerate(prizes), key=sort_key)]

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        now = datetime.now()
        path = os.path.join(os.getcwd(), f"awards_result_{now:%Y%m%d}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        writer.writerow(
                            [
                                winner.user.name,
                                winner.user.department,
                                prize_name,
                                winner.won_at.strftime("%Y-%m-%d %H:%M:%S"),
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _split_tokens(text: str) -> list[str]:
    return [token for token in re.split(r"[\s,]+", text.strip()) if token]


class LotteryCLI:
    """Interactive command-line controller for the whole lottery workflow."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ------------------------------------------------------------------ #
    # Helpers
    # ------------------------------------------------------------------ #
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    @staticmethod
    def _title(text: str) -> None:
        print()
        print("=" * 60)
        print(text)
        print("=" * 60)

    def _confirm(self, prompt: str) -> bool:
        return self._ask(prompt).lower() in ("y", "yes")

    @staticmethod
    def _parse_selection(text: str, options: list[str]) -> list[str] | None:
        """Parse comma/space separated option numbers. Returns None if invalid."""
        tokens = _split_tokens(text)
        chosen: list[str] = []
        for token in tokens:
            if not token.isdigit():
                return None
            index = int(token)
            if not 1 <= index <= len(options):
                return None
            if options[index - 1] not in chosen:
                chosen.append(options[index - 1])
        return chosen

    # ------------------------------------------------------------------ #
    # Entry
    # ------------------------------------------------------------------ #
    def run(self) -> int:
        try:
            return self._run_flow()
        except (EOFError, KeyboardInterrupt):
            print("\nLottery session ended.")
            return 1

    def _run_flow(self) -> int:
        self._title("Company Team Building Smart Lottery System")

        if not self._setup_user_list():
            print("No participant list loaded. Exiting.")
            return 0

        self._show_group_info()
        self._setup_prizes()

        if not self.prize_manager.get_prizes():
            print("No prizes configured. Exiting.")
            return 0

        if not self._confirm("\nStart the lottery now? (y/n): "):
            print("Lottery cancelled.")
            return 0

        self._run_lottery()
        self._show_summary()
        self._export_results()
        print("\nThank you for using the lottery system.")
        return 0

    # ------------------------------------------------------------------ #
    # List management
    # ------------------------------------------------------------------ #
    def _setup_user_list(self) -> bool:
        while True:
            self._title("List Loading")
            print("1. Load built-in default team building list")
            print("2. Upload custom txt list (format: Name,Department Group per line)")
            print("0. Exit")
            choice = self._ask("Select an option: ")

            if choice == "0":
                return False
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
            elif choice == "2":
                path = self._ask("Enter the txt file path: ")
                users, errors = self.list_manager.load_custom_list(path)
            else:
                print("Invalid option.")
                continue

            valid_users = self._verify_list(users, errors)
            if valid_users is None:
                continue

            self.users = valid_users
            print(f"Loaded {len(valid_users)} valid participants.")
            return True

    def _verify_list(self, users: list[User], errors: list[str]) -> list[User] | None:
        valid_users, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            return valid_users

        print("\nList verification found problems:")
        for error in errors:
            print(f"  - {error}")
        for duplicate in duplicates:
            print(
                f"  - Duplicate Name: '{duplicate.name}' ({duplicate.department}) "
                "appears more than once; only the first occurrence is kept"
            )
        print(f"Valid participants available: {len(valid_users)}")
        print("1. Remove invalid data and continue with valid participants")
        print("2. Terminate upload")
        choice = self._ask("Select an option: ")

        if choice == "1":
            if not valid_users:
                print("No valid participants remain.")
                return None
            return valid_users

        print("Upload terminated.")
        return None

        print(f"Total: {len(self.users)} people in {len(info)} groups")

    def _add_custom_prize(self) -> None:

        quantity_text = self._ask("Prize quantity: ")
        if not quantity_text.isdigit() or int(quantity_text) < 1:
            print("Error: quantity must be a positive integer.")
            return

        groups = self._choose_groups()
        try:

        except ValueError as exc:

    def _set_prize_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name}")

            print("Invalid prize number.")
            return
        prize = prizes[int(choice) - 1]
        groups = self._choose_groups()
        self.prize_manager.set_avoid_groups(prize.name, groups)
        print(f"Avoidance groups updated for {prize.name}.")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            print("At least two prizes are required to reorder.")
            return
        print("Current order:")
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name}")
        tokens = _split_tokens(self._ask("Enter new order as serial numbers (e.g. 2 1 3): "))
        if not tokens or not all(token.isdigit() for token in tokens):
            print("Invalid input: please enter serial numbers only.")
            return
        order = [int(token) for token in tokens]
        if self.prize_manager.reorder_prizes(order):
            print("Lottery round order updated.")
        else:
            print("Invalid order: each serial number must be used exactly once.")

    # ------------------------------------------------------------------ #
    # Lottery process
    # ------------------------------------------------------------------ #
    def _run_lottery(self) -> None:
        prizes = self.prize_manager.get_prizes()
        self.lottery = LotteryManager(self.users)
        self._title("Lottery Started")
        total = len(prizes)
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)
        print("\nAll lottery rounds completed.")

    def _describe_rules(self, prize) -> str:
        assert self.lottery is not None
        drawer = prize.associated_drawer
        rules = [f"Lottery person: {drawer or 'None'}"]
        if drawer and same_text(drawer, LI_BIN):
            rules.append("Only previous winners are excluded (avoidance groups not applied)")
        elif (
            drawer
            and same_text(drawer, MA_JIAQI)
            and self.lottery.is_cheng_ru_fallback_active()
        ):
            rules.append(
                f"Cheng Ru Group protection active: only {CHENG_RU_GROUP} members are eligible"
            )
        else:
            rules.append("Previous winners are excluded")
            if prize.avoid_groups:
                rules.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        return "; ".join(rules)

    def _await_command(self) -> str:
        """Wait for user input before each draw. Returns 'draw' or 'terminate'."""
        while True:
            command = self._ask("Press Enter to draw, or type 'pause' / 'terminate': ").lower()
            if command == "":
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused.")
                while True:
                    resume = self._ask("Type 'continue' to resume or 'terminate' to end this round: ").lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if resume == "terminate":
                        return "terminate"
                    print("Unknown command.")
            print("Unknown command.")

    def _run_round(self, index: int, total: int, prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {index}/{total}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {self._describe_rules(prize)}")

        round_winners: list[User] = []
        terminated = False

        for position in range(1, prize.quantity + 1):
            print(f"Currently drawing the {_ordinal(position)} winner...")
            command = self._await_command()
            if command == "terminate":
                terminated = True
                break

            winner = self.lottery.draw_one(prize)
            if winner is None:
                print("No eligible candidates remain for this prize; the round ends early.")
                break

            self.result_manager.add_winner(winner, prize)
            round_winners.append(winner)
            print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}!")

        self.lottery.record_round_result(prize, round_winners)

        print(f"\nRound results for {prize.name}:")
        if round_winners:
            for order, winner in enumerate(round_winners, start=1):
                print(f"  {order}. {winner.name} ({winner.department})")
        else:
            print("  (no winners)")

        if terminated:
            print(f"Round terminated early: {len(round_winners)}/{prize.quantity} winners drawn.")
        elif len(round_winners) < prize.quantity:
            print(f"Round ended with {len(round_winners)}/{prize.quantity} winners.")

    # ------------------------------------------------------------------ #
    # Results
    # ------------------------------------------------------------------ #
    def _show_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self._title("Final Winner List")
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}")
            if not winners:
                print("  (no winners)")
                continue
            for winner in winners:
                print(
                    f"  {winner.win_order}. {winner.user.name} "
                    f"({winner.user.department})"
                )

    def _export_results(self) -> None:
        if not self._confirm("\nExport the winner list to file? (y/n): "):
            return
        _, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(message)


def main() -> int:
    """Start the interactive lottery system."""
    return LotteryCLI().run()

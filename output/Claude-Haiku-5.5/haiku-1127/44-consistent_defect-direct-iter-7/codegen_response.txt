Below are the source files for the Beijing Subway fare calculation system. The data file `src/subway_data.json` is assumed to already exist and is not included.

Three PRD points are internally inconsistent, so I made these choices:

- **Fare tiers:** I followed the tier definitions (left-closed intervals) rather than the sample prices in the boundary bullets. So 6000 m costs 4 yuan, 12000 m costs 5, and 32000 m costs 7. The extended tier steps every 20 km, matching the 32-52 km = 7 yuan and 52-72 km = 8 yuan examples.
- **Path order:** `path_with_edges` returns stations from end to start, as the PRD's display rule specifies.
- **Separators:** the input parser accepts either `;` or `,` between the two station names.

        ValueError: if the JSON content does not have the expected structure.

        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")

"""Binary-heap based minimum priority queue used by Dijkstra's algorithm."""
from __future__ import annotations

import heapq
import itertools
from typing import Any, List, Tuple


class MinPQ:
    """A min-priority queue backed by a binary heap.

    Items with equal priority are dequeued in insertion order, which keeps the
    behaviour deterministic and avoids comparing the items themselves.
    Decrease-key is implemented lazily: callers insert a new entry with the
    lower priority and skip stale entries when they are dequeued.
    """

    def __init__(self) -> None:
        self._heap: List[Tuple[float, int, Any]] = []
        self._counter = itertools.count()

    def insert(self, priority: float, item: Any) -> None:
        """Add an item with the given priority (O(log n))."""
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def min(self) -> Tuple[float, Any]:
        """Return (priority, item) with the smallest priority without removing it."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        priority, _, item = self._heap[0]
        return priority, item

    def del_min(self) -> Tuple[float, Any]:
        """Remove and return (priority, item) with the smallest priority (O(log n))."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def is_empty(self) -> bool:
        """Return True when the queue contains no items."""
        return not self._heap

    def size(self) -> int:
        """Return the number of items currently stored."""
        return len(self._heap)

    def __len__(self) -> int:
        return len(self._heap)

"""Graph data structures used to model the subway network."""
from __future__ import annotations

from typing import Dict, Iterator, List, Optional, Sequence


class Edge:
    """A directed, weighted connection between two stations.

    Attributes:
        source: station number of the origin station.
        dest: station number of the destination station.
        weight: edge weight, equal to the actual distance in meters.
        line: name of the subway line the connection belongs to.
    """

    __slots__ = ("source", "dest", "weight", "line")

    def __init__(self, source: int, dest: int, distance: float, line: str) -> None:
        if distance < 0:
            raise ValueError("edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.weight = distance
        self.line = line

    @property
    def distance(self) -> float:
        """Distance in meters (alias of ``weight``)."""
        return self.weight

    @distance.setter
    def distance(self, value: float) -> None:
        if value < 0:
            raise ValueError("edge distance must be non-negative")
        self.weight = value

    def __repr__(self) -> str:
        return (
            f"Edge(source={self.source}, dest={self.dest}, "
            f"distance={self.weight}, line={self.line!r})"
        )


class Graph:
    """Weighted directed graph stored as adjacency lists.

    Station numbers are assigned by the caller. Each connection is usable only
    in the direction in which it was added.

    Attributes:
        station_lines: mapping of station name -> list of line names it belongs to.
    """

    def __init__(self) -> None:
        self._adj: Dict[int, List[Edge]] = {}
        self.station_lines: Dict[str, List[str]] = {}

    # ------------------------------------------------------------------ vertices
    def add_vertex(self, vertex: int) -> None:
        """Ensure that ``vertex`` exists in the graph."""
        self._adj.setdefault(vertex, [])

    def has_vertex(self, vertex: int) -> bool:
        """Return True if ``vertex`` is a node of the graph."""
        return vertex in self._adj

    def vertices(self) -> List[int]:
        """Return all station numbers in the graph."""
        return list(self._adj.keys())

    @property
    def num_vertices(self) -> int:
        return len(self._adj)

    # --------------------------------------------------------------------- edges
    def add_edge(self, edge: Edge) -> None:
        """Add a directed edge, creating its endpoints if necessary."""
        if not isinstance(edge, Edge):
            raise TypeError("add_edge expects an Edge instance")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adj[edge.source].append(edge)

    def adjacent(self, vertex: int) -> Sequence[Edge]:
        """Return the outgoing edges of ``vertex`` (do not mutate the result)."""
        return self._adj.get(vertex, [])

    def edges(self) -> Iterator[Edge]:
        """Iterate over every edge in the graph."""
        for edge_list in self._adj.values():
            for edge in edge_list:
                yield edge

    def edges_between(self, source: int, dest: int) -> List[Edge]:
        """Return all edges going from ``source`` to ``dest``."""
        return [edge for edge in self._adj.get(source, []) if edge.dest == dest]

    def remove_edge(self, source: int, dest: int, line: Optional[str] = None) -> int:
        """Remove edges from ``source`` to ``dest`` (optionally a single line).

        Returns the number of removed edges.
        """
        current = self._adj.get(source, [])
        kept = [
            edge
            for edge in current
            if not (edge.dest == dest and (line is None or edge.line == line))
        ]
        removed = len(current) - len(kept)
        self._adj[source] = kept
        return removed

    @property
    def num_edges(self) -> int:
        return sum(len(edge_list) for edge_list in self._adj.values())

    # ------------------------------------------------------------ station lines
    def set_station_lines(self, station_name: str, lines: Sequence[str]) -> None:
        """Set the list of lines that serve a station."""
        self.station_lines[station_name] = list(lines)

    def get_station_lines(self, station_name: str) -> List[str]:
        """Return the lines serving a station (empty list if unknown)."""
        return list(self.station_lines.get(station_name, []))

"""Fare and travel-time calculation for the Beijing Subway.

Fare structure (distances in meters, left-closed / right-open intervals):
    [0, 6000)          -> 3 yuan
    [6000, 12000)      -> 4 yuan
    [12000, 22000)     -> 5 yuan
    [22000, 32000)     -> 6 yuan
    [32000, 52000)     -> 7 yuan   (extended tier: +1 yuan per further 20 km)
    [52000, 72000)     -> 8 yuan
    ...
Fractional distances are rounded up (ceiling) to the next whole meter before
the tier is determined, so a boundary value always enters the next tier.
"""
from __future__ import annotations

import math
from typing import Tuple

MINUTES_PER_KM: float = 1.5
MINUTES_PER_TRANSFER: float = 5.0

# (lower bound in meters, fare in yuan) for the basic and progressive tiers.
BASE_FARE_TIERS: Tuple[Tuple[int, int], ...] = (
    (0, 3),
    (6000, 4),
    (12000, 5),
    (22000, 6),
)
EXTENDED_TIER_START: int = 32000
EXTENDED_TIER_FARE: int = 7
EXTENDED_STEP_METERS: int = 20000


def Sum_money(distance: float) -> int:
    """Return the fare in yuan for a trip of ``distance`` meters.

    Raises:
        ValueError: if the distance is negative or not a number.
    """
    if not isinstance(distance, (int, float)) or isinstance(distance, bool):
        raise ValueError("distance must be a number of meters")
    if distance != distance or distance < 0:  # NaN or negative
        raise ValueError("distance must be a non-negative number of meters")

    meters = math.ceil(distance)  # ceiling rounding of fractional distances

    if meters >= EXTENDED_TIER_START:
        return EXTENDED_TIER_FARE + (meters - EXTENDED_TIER_START) // EXTENDED_STEP_METERS

    fare = BASE_FARE_TIERS[0][1]
    for threshold, price in BASE_FARE_TIERS:
        if meters >= threshold:
            fare = price
    return fare


def calculate_estimated_time(distance: float, transfer_count: int) -> float:
    """Estimate travel time in minutes.

    Uses roughly 1.5 minutes per kilometer plus about 5 minutes per transfer.

    Args:
        distance: travel distance in meters.
        transfer_count: number of transfers on the route.
    """
    if distance < 0:
        raise ValueError("distance must be non-negative")
    if transfer_count < 0:
        raise ValueError("transfer_count must be non-negative")
    return (distance / 1000.0) * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

<file path="src/pathfinding.py">
"""Shortest-path computation over the subway graph.

Dijkstra's algorithm runs over search states ``(station, arriving_line,
transfers_so_far)``. A transfer happens whenever the arriving line differs
from the line of the next edge. Each mode uses its own edge cost:

* ``shortest_distance``: cost = distance in meters.
* ``least_transfer``:    cost = large penalty per transfer + distance (tie-break).
* ``comprehensive``:     cost = estimated minutes (1.5 min/km + 5 min/transfer).

A soft line preference adds a penalty to edges outside the preferred lines.
"""
from __future__ import annotations

import math
from typing import Dict, List, Optional, Tuple

from src.fare_calculator import MINUTES_PER_KM, MINUTES_PER_TRANSFER
from src.min_pq import MinPQ
from src.models import Edge, Graph

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES: Tuple[str, ...] = (
    MODE_SHORTEST_DISTANCE,
    MODE_LEAST_TRANSFER,
    MODE_COMPREHENSIVE,
)

# Transfer limit applied when the user does not specify --max_transfers.
# It bounds the search state space and guarantees termination.
DEFAULT_TRANSFER_LIMIT = 10
TRANSFER_COST_LEAST_TRANSFER = 1_000_000.0
PREFERENCE_PENALTY = {
    MODE_SHORTEST_DISTANCE: 1000.0,  # meters
    MODE_LEAST_TRANSFER: 1000.0,     # meters (tie-break scale)
    MODE_COMPREHENSIVE: 1.0,         # minutes
}

State = Tuple[int, Optional[str], int]


class PathfindingConfig:
    """User-facing options for a path query.

    Attributes:
        mode: one of VALID_MODES.
        max_transfers: maximum number of transfers allowed, or None for no limit.
        preferred_lines: lines favoured by the route search (soft preference).
    """

    def __init__(
        self,
        mode: str = MODE_SHORTEST_DISTANCE,
        max_transfers: Optional[int] = None,
        preferred_lines: Optional[List[str]] = None,
    ) -> None:
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines: List[str] = list(preferred_lines) if preferred_lines else []
        self.validate()

    def validate(self) -> None:
        """Raise ValueError if the configuration is invalid."""
        if self.mode not in VALID_MODES:
            raise ValueError(
                f"unknown mode '{self.mode}', expected one of {', '.join(VALID_MODES)}"
            )
        if self.max_transfers is not None:
            if not isinstance(self.max_transfers, int) or isinstance(self.max_transfers, bool):
                raise ValueError("max_transfers must be an integer")
            if self.max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")

    def __repr__(self) -> str:
        return (
            f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
            f"preferred_lines={self.preferred_lines!r})"
        )


class ShortestPath:
    """Single-source shortest path computation (Dijkstra with a binary heap).

    Attributes:
        dist_to: mapping of station number -> minimal cost (mode dependent;
            in ``shortest_distance`` mode this is the distance in meters).
    """

    def __init__(self, graph: Graph, source: int, config: Optional[PathfindingConfig] = None) -> None:
        if not graph.has_vertex(source):
            raise ValueError(f"unknown source station number {source}")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()

        self._limit: int = (
            self.config.max_transfers
            if self.config.max_transfers is not None
            else DEFAULT_TRANSFER_LIMIT
        )
        self._cost: Dict[State, float] = {}
        self._prev: Dict[State, Tuple[State, Edge]] = {}
        self._best: Dict[int, State] = {}
        self.dist_to: Dict[int, float] = {}
        self._run()

    # ------------------------------------------------------------------ search
    def _edge_cost(self, edge: Edge, is_transfer: bool) -> float:
        """Cost of traversing ``edge`` under the configured mode."""
        mode = self.config.mode
        if mode == MODE_COMPREHENSIVE:
            cost = edge.weight / 1000.0 * MINUTES_PER_KM
            if is_transfer:
                cost += MINUTES_PER_TRANSFER
        elif mode == MODE_LEAST_TRANSFER:
            cost = edge.weight + (TRANSFER_COST_LEAST_TRANSFER if is_transfer else 0.0)
        else:
            cost = edge.weight

        preferred = self.config.preferred_lines
        if preferred and edge.line not in preferred:
            cost += PREFERENCE_PENALTY[mode]
        return cost

    def _run(self) -> None:
        start: State = (self.source, None, 0)
        self._cost[start] = 0.0
        pq = MinPQ()
        pq.insert(0.0, start)

        while not pq.is_empty():
            cost, state = pq.del_min()
            if cost > self._cost.get(state, math.inf):
                continue  # stale heap entry
            node, line, transfers = state

            for edge in self.graph.adjacent(node):
                is_transfer = line is not None and edge.line != line
                new_transfers = transfers + 1 if is_transfer else transfers
                if new_transfers > self._limit:
                    continue
                new_state: State = (edge.dest, edge.line, new_transfers)
                new_cost = cost + self._edge_cost(edge, is_transfer)
                if new_cost < self._cost.get(new_state, math.inf):
                    self._cost[new_state] = new_cost
                    self._prev[new_state] = (state, edge)
                    pq.insert(new_cost, new_state)

        # Pick the best search state for every station (lowest cost, then fewest transfers).
        for state, cost in self._cost.items():
            node = state[0]
            best = self._best.get(node)
            if best is None or (cost, state[2]) < (self._cost[best], best[2]):
                self._best[node] = state
        for node, state in self._best.items():
            self.dist_to[node] = self._cost[state]

    # ----------------------------------------------------------------- queries
    def has_path_to(self, target: int) -> bool:
        """Return True if ``target`` is reachable from the source."""
        return target in self._best

    def path_with_edges(self, target: int) -> Tuple[List[int], List[Edge]]:
        """Return the route to ``target`` by backtracking predecessors.

        The station list is ordered from the target back to the source (end to
        start). ``edges[i]`` is the edge whose destination is ``stations[i]``
        and whose source is ``stations[i + 1]``. If the target is unreachable,
        ``([], [])`` is returned.
        """
        state = self._best.get(target)
        if state is None:
            return [], []

        stations: List[int] = []
        edges: List[Edge] = []
        current = state
        while current in self._prev:
            stations.append(current[0])
            prev_state, edge = self._prev[current]
            edges.append(edge)
            current = prev_state
        stations.append(current[0])  # the source station
        return stations, edges

    def get_transfer_count(self, target: int) -> int:
        """Return the number of transfers on the best route to ``target`` (0 if unreachable)."""
        state = self._best.get(target)
        if state is None:
            return 0
        return state[2]

    def get_total_distance(self, target: int) -> float:
        """Return the total distance in meters on the best route to ``target``."""
        _, edges = self.path_with_edges(target)
        return float(sum(edge.weight for edge in edges))

"""Subway network model: station registry, graph construction, updates and data checks."""
from __future__ import annotations

import difflib
from typing import Any, Dict, List, Optional, Sequence, Set, Tuple

from src.data_loader import DATA_FILE_PATH, load_subway_data
from src.models import Edge, Graph

# Main lines that must be represented in the dataset (PRD section 4.1).
REQUIRED_LINES: Tuple[str, ...] = (
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
)

MIN_PLAUSIBLE_DISTANCE_M = 100
MAX_PLAUSIBLE_DISTANCE_M = 5000
FUZZY_MATCH_CUTOFF = 0.5


class StationNotFoundError(ValueError):
    """Raised when a station name cannot be matched to any known station."""

    def __init__(self, name: str) -> None:
        super().__init__(f"station '{name}' not found")
        self.station_name = name


def normalize_station_name(name: str) -> str:
    """Normalize a station name for comparison (case, spacing and trailing '站')."""
    text = " ".join(str(name).split()).casefold()
    if text.endswith("站") and len(text) > 1:
        text = text[:-1]
    return text


class SubwayNetwork:
    """In-memory representation of the Beijing Subway network.

    Attributes:
        graph: weighted directed Graph keyed by station numbers.
    """

    def __init__(self, data: Optional[Dict[str, Any]] = None, data_path: str = DATA_FILE_PATH) -> None:
        if data is None:
            data = load_subway_data(data_path)
        self.graph = Graph()
        self.station_to_number: Dict[str, int] = {}  # station name -> station number
        self._names: List[str] = []                  # station number -> station name
        self._norm_index: Dict[str, str] = {}        # normalized name -> official name
        self._line_members: Dict[str, Dict[str, None]] = {}  # line -> ordered set of stations
        self._build(data)

    # ------------------------------------------------------------ construction
    def _build(self, data: Dict[str, Any]) -> None:
        for station in data.get("stations", []):
            name = str(station["name"])
            lines = [str(line) for line in station.get("lines", [])]
            self._ensure_station(name)
            self.graph.set_station_lines(name, lines)
            for line in lines:
                self._register_member(line, name)

        for conn in data.get("connections", []):
            self.add_connection(
                str(conn["source"]),
                str(conn["dest"]),
                int(conn["distance"]),
                str(conn["line"]),
            )

    def _ensure_station(self, name: str) -> int:
        """Return the number of ``name``, creating the station if needed."""
        number = self.station_to_number.get(name)
        if number is None:
            number = len(self._names)
            self.station_to_number[name] = number
            self._names.append(name)
            self._norm_index[normalize_station_name(name)] = name
            self.graph.add_vertex(number)
            if name not in self.graph.station_lines:
                self.graph.set_station_lines(name, [])
        return number

    def _register_member(self, line: str, station: str) -> None:
        self._line_members.setdefault(line, {})[station] = None

    def add_connection(self, source: str, dest: str, distance: int, line: str) -> None:
        """Add a single directed connection between two stations."""
        if distance <= 0:
            raise ValueError("connection distance must be positive")
        src_num = self._ensure_station(source)
        dst_num = self._ensure_station(dest)
        self.graph.add_edge(Edge(src_num, dst_num, distance, line))
        self._register_member(line, source)
        self._register_member(line, dest)

    # ---------------------------------------------------------------- lookups
    def station_names(self) -> List[str]:
        """Return all official station names, sorted."""
        return sorted(self.station_to_number)

    def lines(self) -> List[str]:
        """Return all line names, sorted."""
        return sorted(self._line_members)

    def stations_on_line(self, line: str) -> List[str]:
        """Return the stations served by ``line`` in dataset order."""
        return list(self._line_members.get(line, {}).keys())

    def name_of(self, number: int) -> str:
        """Return the station name for a station number."""
        return self._names[number]

    def number_of(self, name: str) -> int:
        """Return the station number for an official station name."""
        return self.station_to_number[name]

    def resolve_station(self, name: str) -> Tuple[str, bool]:
        """Standardize a user-supplied station name.

        Returns:
            (official_name, exact_match). When there is no exact match, the
            nearest official name is returned with exact_match=False.

        Raises:
            StationNotFoundError: if no station is reasonably similar.
        """
        text = normalize_station_name(name)
        if not text:
            raise StationNotFoundError(name)
        if text in self._norm_index:
            return self._norm_index[text], True
        matches = difflib.get_close_matches(
            text, list(self._norm_index.keys()), n=1, cutoff=FUZZY_MATCH_CUTOFF
        )
        if not matches:
            raise StationNotFoundError(name)
        return self._norm_index[matches[0]], False

    def transfer_stations(self) -> List[str]:
        """Return stations served by more than one line."""
        return [name for name, lines in self.graph.station_lines.items() if len(set(lines)) > 1]

    def _exact_number(self, name: str) -> int:
        if name not in self.station_to_number:
            raise StationNotFoundError(name)
        return self.station_to_number[name]

    # ---------------------------------------------------------------- updates
    def add_line(self, line_name: str, station_names: Sequence[str], distances: Sequence[int]) -> None:
        """Add a new line as a sequence of stations joined by bidirectional connections.

        Args:
            line_name: name of the new line (must not already exist).
            station_names: ordered station names along the line (at least 2).
            distances: distance in meters between consecutive stations.
        """
        line_name = line_name.strip()
        if not line_name:
            raise ValueError("line name must not be empty")
        if line_name in self._line_members:
            raise ValueError(f"line '{line_name}' already exists")
        names = [name.strip() for name in station_names]
        if len(names) < 2 or any(not name for name in names):
            raise ValueError("a line needs at least two named stations")
        if len(distances) != len(names) - 1:
            raise ValueError("expected one distance per pair of consecutive stations")

        for index, distance in enumerate(distances):
            a, b = names[index], names[index + 1]
            self.add_connection(a, b, int(distance), line_name)
            self.add_connection(b, a, int(distance), line_name)

        for name in names:
            lines = self.graph.get_station_lines(name)
            if line_name not in lines:
                lines.append(line_name)
                self.graph.set_station_lines(name, lines)

    def update_station_lines(self, name: str, lines: Sequence[str]) -> None:
        """Replace the list of lines serving an existing station."""
        official = self._names[self._exact_number(name)]
        new_lines = [str(line) for line in lines]
        self.graph.set_station_lines(official, new_lines)
        for line in new_lines:
            self._register_member(line, official)

    def rename_station(self, old_name: str, new_name: str) -> None:
        """Rename an existing station, keeping all connections."""
        new_name = new_name.strip()
        number = self._exact_number(old_name)
        if not new_name or new_name in self.station_to_number:
            raise ValueError(f"cannot rename to '{new_name}'")
        del self.station_to_number[old_name]
        self.station_to_number[new_name] = number
        self._names[number] = new_name
        self._norm_index.pop(normalize_station_name(old_name), None)
        self._norm_index[normalize_station_name(new_name)] = new_name
        self.graph.station_lines[new_name] = self.graph.station_lines.pop(old_name, [])
        for line, members in self._line_members.items():
            if old_name in members:
                self._line_members[line] = {
                    (new_name if member == old_name else member): None for member in members
                }

    def update_distance(self, source: str, dest: str, distance: int) -> int:
        """Update the distance between two adjacent stations in both directions.

        Returns:
            The number of edges updated.
        """
        if not isinstance(distance, int) or distance <= 0:
            raise ValueError("distance must be a positive integer number of meters")
        a = self._exact_number(source)
        b = self._exact_number(dest)
        count = 0
        for x, y in ((a, b), (b, a)):
            for edge in self.graph.edges_between(x, y):
                edge.distance = distance
                count += 1
        if count == 0:
            raise ValueError(f"no connection between '{source}' and '{dest}'")
        return count

    def to_data(self) -> Dict[str, Any]:
        """Export the current network in the subway_data.json structure."""
        stations = [
            {"name": name, "lines": self.graph.get_station_lines(name)}
            for name in self._names
        ]
        connections = [
            {
                "source": self._names[edge.source],
                "dest": self._names[edge.dest],
                "distance": int(round(edge.weight)),
                "line": edge.line,
            }
            for edge in self.graph.edges()
        ]
        return {"stations": stations, "connections": connections}

    # ------------------------------------------------------------- data checks
    def check_data(self) -> Tuple[bool, List[str]]:
        """Run data integrity checks.

        Checks station coverage of the main lines, connection integrity
        (duplicates, reverse links, distance plausibility) and consistency
        between declared station lines and actual connections.

        Returns:
            (passed, messages). ``passed`` is False when any error was found.
        """
        messages: List[str] = []
        errors = 0
        warnings = 0

        # 1. Station coverage of main lines.
        for line in REQUIRED_LINES:
            count = len(self.stations_on_line(line))
            if count < 2:
                messages.append(
                    f"[ERROR] coverage: line '{line}' has {count} station(s), expected at least 2"
                )
                errors += 1
            else:
                messages.append(f"[OK] coverage: line '{line}' has {count} stations")

        edges = list(self.graph.edges())
        keys = {(e.source, e.dest, e.line) for e in edges}
        incoming: Set[int] = {e.dest for e in edges}
        outgoing: Set[int] = {e.source for e in edges}
        for number, name in enumerate(self._names):
            if number not in incoming and number not in outgoing:
                messages.append(f"[WARN] connectivity: station '{name}' has no connections")
                warnings += 1

        # 2. Connection integrity.
        seen: Set[Tuple[int, int, str]] = set()
        for edge in edges:
            key = (edge.source, edge.dest, edge.line)
            src_name = self._names[edge.source]
            dst_name = self._names[edge.dest]
            if key in seen:
                messages.append(
                    f"[ERROR] connection: duplicate {src_name} -> {dst_name} on '{edge.line}'"
                )
                errors += 1
            seen.add(key)

            # 3. Distance accuracy.
            if not isinstance(edge.weight, (int, float)) or edge.weight <= 0:
                messages.append(
                    f"[ERROR] distance: {src_name} -> {dst_name} has invalid distance {edge.weight}"
                )
                errors += 1
            elif not (MIN_PLAUSIBLE_DISTANCE_M <= edge.weight <= MAX_PLAUSIBLE_DISTANCE_M):
                messages.append(
                    f"[WARN] distance: {src_name} -> {dst_name} distance {edge.weight} m is outside "
                    f"the plausible range {MIN_PLAUSIBLE_DISTANCE_M}-{MAX_PLAUSIBLE_DISTANCE_M} m"
                )
                warnings += 1

            if (edge.dest, edge.source, edge.line) not in keys:
                messages.append(
                    f"[WARN] connection: no reverse connection {dst_name} -> {src_name} on '{edge.line}'"
                )
                warnings += 1

        # 4. Declared station lines versus actual connection lines.
        touching: Dict[str, Set[str]] = {}
        for edge in edges:
            touching.setdefault(self._names[edge.source], set()).add(edge.line)
            touching.setdefault(self._names[edge.dest], set()).add(edge.line)
        for name, declared in self.graph.station_lines.items():
            actual = touching.get(name, set())
            if actual and set(declared) != actual:
                messages.append(
                    f"[WARN] consistency: station '{name}' declares lines {sorted(set(declared))} "
                    f"but connections use {sorted(actual)}"
                )
                warnings += 1

        messages.append(
            f"Summary: {errors} error(s), {warnings} warning(s), "
            f"{len(self._names)} stations, {len(edges)} connections"
        )
        return errors == 0, messages

def format_distance(meters: float) -> str:
    """Format a distance: meters for short trips, kilometers (with meters) for long ones."""
    if meters < 1000:
        return f"{meters:.0f} m"
    return f"{meters / 1000:.2f} km ({meters:.0f} m)"

def build_result(
    start_station: str,
    end_station: str,
    config: Any,
    notices: Sequence[str],
    stations_end_to_start: Sequence[str],
    edges_end_to_start: Sequence[Edge],
    transfer_count: int,
) -> PathResult:
    """Assemble a PathResult from the raw path returned by ShortestPath.

    Args:
        config: PathfindingConfig used for the query.
        stations_end_to_start: station names in the order returned by path_with_edges.
        edges_end_to_start: edges in the order returned by path_with_edges.
        transfer_count: transfers counted by the path search.
    """
    journey_names = list(reversed(stations_end_to_start))
    journey_edges = list(reversed(edges_end_to_start))

    segments: List[Dict[str, Any]] = []
    lines: List[str] = []
    distance = 0.0
    for index, edge in enumerate(journey_edges):
        distance += edge.weight
        if edge.line not in lines:
            lines.append(edge.line)
        if segments and segments[-1]["line"] == edge.line:
            segment = segments[-1]
            segment["end"] = journey_names[index + 1]
            segment["distance_m"] += edge.weight
            segment["stops"] += 1
        else:
            segments.append(
                {
                    "line": edge.line,
                    "start": journey_names[index],
                    "end": journey_names[index + 1],
                    "distance_m": edge.weight,
                    "stops": 1,
                }
            )

    transfers: List[Dict[str, str]] = []
    for index in range(1, len(journey_edges)):
        previous, current = journey_edges[index - 1], journey_edges[index]
        if previous.line != current.line:
            transfers.append(
                {
                    "station": journey_names[index],
                    "from_line": previous.line,
                    "to_line": current.line,
                }
            )

    fare = Sum_money(distance)
    estimated = calculate_estimated_time(distance, transfer_count)

    return PathResult(
        start_station=start_station,
        end_station=end_station,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        notices=list(notices),
        stations=list(stations_end_to_start),
        lines=lines,
        segments=segments,
        transfers=transfers,
        distance_m=distance,
        fare=fare,
        transfer_count=transfer_count,
        estimated_time=estimated,
    )


def render_text(result: PathResult) -> str:
    """Render a human-readable console report."""
    out: List[str] = []
    out.append("=" * 60)
    out.append("Beijing Subway Fare Query")
    out.append("=" * 60)
    out.append(f"Start: {result.start_station}    End: {result.end_station}")
    out.append(f"Mode: {result.mode}")
    if result.max_transfers is not None:
        out.append(f"Max transfers: {result.max_transfers}")
    if result.preferred_lines:
        out.append(f"Line preference: {', '.join(result.preferred_lines)}")
    for notice in result.notices:
        out.append(f"Note: {notice}")

    out.append("")
    out.append(f"Station sequence (end -> start, {len(result.stations)} stations):")
    out.append("  " + " -> ".join(result.stations))

    out.append("")
    out.append("Route segments:")
    for number, segment in enumerate(result.segments, start=1):
        out.append(
            f"  {number}. [{segment['line']}] {segment['start']} -> {segment['end']} "

    if result.transfers:
        for transfer in result.transfers:
            out.append(

"""Query orchestration: input parsing, station resolution and route/fare computation."""
from __future__ import annotations

import re
from typing import List, Optional, Tuple

from src.path_report import PathResult, build_result
from src.pathfinding import PathfindingConfig, ShortestPath
from src.subway_network import StationNotFoundError, SubwayNetwork

INPUT_FORMAT_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class QueryError(Exception):
    """Base class for user-facing query errors."""


class InputFormatError(QueryError):
    """Raised when the query text does not contain exactly two station names."""


class PathNotFoundError(QueryError):
    """Raised when no route exists between the two stations."""


def parse_query(text: str) -> Tuple[str, str]:
    """Split a 'Starting Station; Ending Station' query into two names.

    Both ';' and ',' (including full-width variants) are accepted as separators.
    """
    if not isinstance(text, str):
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    normalized = text.strip().replace("\uff1b", ";").replace("\uff0c", ",")
    parts = [part.strip() for part in re.split(r"[;,]", normalized)]
    if len(parts) != 2 or not all(parts):
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    return parts[0], parts[1]


def _resolve(network: SubwayNetwork, name: str, role: str, notices: List[str]) -> str:
    """Resolve a user station name, recording a notice when it was approximated."""
    try:
        official, exact = network.resolve_station(name)
    except StationNotFoundError as exc:
        raise QueryError(
            f"{role} station '{name}' does not exist, please check the station name"
        ) from exc
    if not exact:
        notices.append(
            f"{role} station '{name}' is not an exact match; matched to nearest station '{official}'"
        )
    return official


def run_query(
    network: SubwayNetwork,
    text: str,
    config: Optional[PathfindingConfig] = None,
) -> PathResult:
    """Parse a query, compute the optimal route and fare.

    Raises:
        QueryError: for any user-facing problem (format, unknown station,
            identical stations, no route, calculation failure).
    """
    config = config if config is not None else PathfindingConfig()
    start_input, end_input = parse_query(text)

    notices: List[str] = []
    start_name = _resolve(network, start_input, "Starting", notices)
    end_name = _resolve(network, end_input, "Ending", notices)
    if start_name == end_name:
        raise QueryError("Starting station cannot be the same as the ending station")

    source = network.number_of(start_name)
    target = network.number_of(end_name)
    try:
        search = ShortestPath(network.graph, source, config)
    except (ValueError, KeyError) as exc:
        raise QueryError(f"Path calculation failed: {exc}") from exc

    if not search.has_path_to(target):
        raise PathNotFoundError(
            f"No route found from '{start_name}' to '{end_name}' with the given options"
        )

    stations, edges = search.path_with_edges(target)
    station_names = [network.name_of(number) for number in stations]
    transfer_count = search.get_transfer_count(target)

    try:
        return build_result(
            start_name,
            end_name,
            config,
            notices,
            station_names,
            edges,
            transfer_count,
        )
    except (ValueError, TypeError) as exc:
        raise QueryError(f"Fare calculation failed: {exc}") from exc

EXIT_WORDS = {"quit", "exit", "q"}


def _non_negative_int(value: str) -> int:
    """argparse type: a non-negative integer."""
    try:
        number = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError(f"'{value}' is not an integer") from exc
    if number < 0:
        raise argparse.ArgumentTypeError("value must be non-negative")
    return number


def parse_line_preference(value: Optional[str]) -> List[str]:
    """Split a comma-separated --line_preference value into line names."""
    if not value:
        return []
    return [part.strip() for part in value.split(",") if part.strip()]


def build_arg_parser() -> argparse.ArgumentParser:
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (interactive CLI)."
    )
    parser.add_argument(
        "--output_file",
        default=None,
        help="export the query results of this session to the given file",
    )
    parser.add_argument(
        "--format",
        choices=["text", "json"],
        default="text",
        help="output format: text (default) or json",
    )
    parser.add_argument(
        "--mode",
        choices=list(VALID_MODES),
        default=MODE_SHORTEST_DISTANCE,
        help="path query mode (default: shortest_distance)",
    )
    parser.add_argument(
        "--max_transfers",
        type=_non_negative_int,

        help="maximum number of transfers allowed",
    )
    parser.add_argument(
        "--line_preference",
        default=None,
        help="preferred line(s), e.g. 'Line 1' or 'Line 1,Line 2'",

def _write_output_file(path: str, results: Sequence[PathResult], fmt: str) -> None:
    """Write all results of the session to ``path`` (overwriting previous content)."""
    if fmt == "json":
        content = json.dumps([result.to_dict() for result in results], ensure_ascii=False, indent=2)
    else:
        content = "\n\n".join(render_text(result) for result in results)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(content + "\n")


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Program entry point. Returns the process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    try:
        network = SubwayNetwork()
    except (OSError, ValueError, KeyError, TypeError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        passed, messages = network.check_data()
        for message in messages:
            print(message)
        print("Data check PASSED" if passed else "Data check FAILED")
        return 0 if passed else 1

    preferred = parse_line_preference(args.line_preference)
    known_lines = set(network.lines())
    for line in preferred:
        if line not in known_lines:
            print(f"Warning: line '{line}' is not present in the dataset")

    try:
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=preferred,
        )
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    print("Beijing Subway Fare Calculator")
    print(f"Mode: {config.mode}" + (f", max transfers: {config.max_transfers}" if config.max_transfers is not None else ""))
    print("Enter 'Starting Station; Ending Station' (or 'quit' to exit).")

    session: List[PathResult] = []
    while True:
        try:
            raw = input("\nQuery> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break

        text = raw.strip()
        if not text:
            continue
        if text.lower() in EXIT_WORDS:
            break

        try:
            result = run_query(network, text, config)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # defensive: never crash the interactive loop
            print(f"Unexpected error while processing query: {exc}")
            continue

        session.append(result)
        if args.format == "json":
            print(render_json(result))
        else:
            print(render_text(result))

        if args.output_file:
            try:
                _write_output_file(args.output_file, session, args.format)
            except OSError as exc:
                print(f"Error: could not write output file: {exc}")

    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Global variable store shared across the dispatch simulator.

The store is a simple module-level dictionary. It holds the system funds,
the rider list, the rider ID counter, the running flag, the order list and
the simulated clock. ``_init()`` resets every value to its default.
"""

from typing import Any, Dict

_store: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store to its default values.

    Default keys:
        funds: System funds (integer, may become negative).
        riders: Dynamic list of Rider objects.
        rider_id_counter: Auto-incrementing counter for rider IDs.
        running: Boolean flag controlling the main loop.
        orders: List of every Order created in this session.
        tick: Current simulated time (one unit per tick).
    """
    _store.clear()
    _store.update(
        {
            "funds": 1000,
            "riders": [],
            "rider_id_counter": 0,
            "running": True,
            "orders": [],
            "tick": 0,
        }
    )


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Value to store.
    """
    _store[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` when the key is missing.
    """
    return _store.get(key, default)

"""Grid-based path planning for riders on the street network.

The city is an 800x800 unit map divided into 8x8 cells of 100x100 units.
Streets are the grid lines where the x or y coordinate is a multiple of
100. Restaurants and customers sit on intersections (both coordinates are
multiples of 100). Riders always travel along streets.
"""

import logging
from typing import List, Sequence, Tuple

Point = Tuple[int, int]

CELL_SIZE = 100
MAP_SIZE = 800

logger = logging.getLogger(__name__)


class PathPlanner:
    """Plans axis-aligned routes that follow the street grid.

    Every planned route contains at most one right-angle turn. Each leg is
    either horizontal or vertical, so a rider never leaves the streets.

    Special cases:
        * Target directly above/below/left/right of the rider: a straight
          single-leg route is used.
        * Rider already at the target: an empty route is returned.

    Deadlock avoidance: riders never block one another. Each rider's
    movement depends only on its own route, so no circular wait can form.
    """

    @staticmethod
    def is_on_street(point: Sequence[int]) -> bool:
        """Return True if ``point`` lies on a street (grid line)."""
        x, y = int(point[0]), int(point[1])
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def is_intersection(point: Sequence[int]) -> bool:
        """Return True if ``point`` is a grid intersection."""
        x, y = int(point[0]), int(point[1])
        return x % CELL_SIZE == 0 and y % CELL_SIZE == 0

    @staticmethod
    def is_special_case(start: Sequence[int], goal: Sequence[int]) -> bool:
        """Return True when ``goal`` is directly above, below, left or right
        of ``start`` (a straight-line route)."""
        return start[0] == goal[0] or start[1] == goal[1]

    @classmethod
    def plan(cls, start: Sequence[int], goal: Sequence[int]) -> List[Point]:
        """Plan a street route from ``start`` to ``goal``.

        The returned list contains the waypoints after ``start`` (the corner
        point, when a turn is needed, followed by ``goal``). An empty list is
        returned when start and goal coincide.

        Args:
            start: Current (x, y) position.
            goal: Destination (x, y) position.

        Returns:
            List of (x, y) waypoints, ending with ``goal``.
        """
        sx, sy = int(start[0]), int(start[1])
        gx, gy = int(goal[0]), int(goal[1])
        if (sx, sy) == (gx, gy):
            return []
        if sx == gx or sy == gy:
            # Special case: directly above/below or left/right of the target.
            return [(gx, gy)]
        if sy % CELL_SIZE == 0:
            # Rider is on a horizontal street: travel horizontally first.
            corner = (gx, sy)
        elif sx % CELL_SIZE == 0:
            # Rider is on a vertical street: travel vertically first.
            corner = (sx, gy)
        else:
            # Should not happen, because riders always stay on streets.
            logger.warning("Off-street start %s; using fallback corner", start)
            corner = (gx, sy)
        return [corner, (gx, gy)]

    @classmethod
    def plan_route(cls, start: Sequence[int], stops: Sequence[Sequence[int]]) -> List[Point]:
        """Plan a continuous route through several destinations in order.

        Consecutive legs are chained so that the end of one leg is the start
        of the next. Duplicate waypoints are removed.

        Args:
            start: Starting (x, y) position.
            stops: Ordered list of (x, y) destinations.

        Returns:
            Combined list of waypoints.
        """
        route: List[Point] = []
        current = (int(start[0]), int(start[1]))
        for stop in stops:
            stop_point = (int(stop[0]), int(stop[1]))
            for waypoint in cls.plan(current, stop_point):
                if not route or route[-1] != waypoint:
                    route.append(waypoint)
            current = stop_point
        return route

    @staticmethod
    def route_length(start: Sequence[int], route: Sequence[Sequence[int]]) -> int:
        """Return the total travelled Manhattan length of ``route``.

        Args:
            start: Starting (x, y) position.
            route: Waypoints produced by ``plan``.

        Returns:
            Total distance in units.
        """
        total = 0
        cx, cy = int(start[0]), int(start[1])
        for wx, wy in route:
            total += abs(wx - cx) + abs(wy - cy)
            cx, cy = wx, wy
        return total

    @staticmethod
    def trace_points(start: Sequence[int], route: Sequence[Sequence[int]]) -> List[Point]:
        """Return every integer point along ``route``, excluding ``start``.

        Args:
            start: Starting (x, y) position.
            route: Waypoints produced by ``plan`` or ``plan_route``.

        Returns:
            List of integer (x, y) points along the route.
        """
        points: List[Point] = []
        cx, cy = int(start[0]), int(start[1])
        for wx, wy in route:
            if cy == wy and cx != wx:
                step = 1 if wx > cx else -1
                points.extend((x, cy) for x in range(cx + step, wx + step, step))
            elif cx == wx and cy != wy:
                step = 1 if wy > cy else -1
                points.extend((cx, y) for y in range(cy + step, wy + step, step))
            cx, cy = wx, wy
        return points

"""Order data model.

An order is placed with a restaurant position and a customer position in
the 0-800 map. Coordinates are snapped to the top-left corner of their grid
cell (a multiple of 100), which is the intersection used for pickup and
drop-off.
"""

from typing import Any, Dict, Optional, Sequence, Tuple

from path_planner import PathPlanner

CELL_SIZE = 100
GRID_CELLS = 8

Point = Tuple[int, int]

ORDER_STATUSES = ("pending", "assigned", "picking_up", "delivering", "completed")


def snap_to_grid(value: int) -> int:
    """Convert a raw coordinate to the top-left corner of its grid cell.

    Values beyond the last cell are clamped to the last cell.

    Args:
        value: Raw coordinate (0-1000).

    Returns:
        Grid-aligned coordinate (a multiple of 100 between 0 and 700).
    """
    cell = int(value) // CELL_SIZE
    cell = max(0, min(cell, GRID_CELLS - 1))
    return cell * CELL_SIZE


def snap_point(point: Sequence[int]) -> Point:
    """Snap an (x, y) point to its grid cell's top-left corner.

    Args:
        point: Raw (x, y) coordinates.

    Returns:
        Snapped (x, y) tuple.
    """
    return snap_to_grid(point[0]), snap_to_grid(point[1])


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        res_pos: Restaurant position (grid-snapped).
        cus_pos: Customer position (grid-snapped).
        status: One of pending, assigned, picking_up, delivering, completed.
        rider_id: ID of the assigned rider, or None.
        created_tick: Tick when the order was created.
        picked_up_tick: Tick when the rider reached the restaurant.
        completed_tick: Tick when the order was delivered.
        distance: Planned route length from restaurant to customer.
    """

    _id_counter = 0

    def __init__(
        self,
        res_pos: Sequence[int],
        cus_pos: Sequence[int],
        created_tick: int = 0,
        order_id: Optional[int] = None,
    ) -> None:
        """Create an order and snap both locations to the grid.

        Args:
            res_pos: Restaurant (x, y) location.
            cus_pos: Customer (x, y) location.
            created_tick: Simulated time of creation.
            order_id: Explicit ID (used when restoring saved state).
        """
        if order_id is None:
            Order._id_counter += 1
            order_id = Order._id_counter
        else:
            Order._id_counter = max(Order._id_counter, int(order_id))
        self.id = int(order_id)
        self.res_pos: Point = snap_point(res_pos)
        self.cus_pos: Point = snap_point(cus_pos)
        self.status = "pending"
        self.rider_id: Optional[int] = None
        self.created_tick = int(created_tick)
        self.picked_up_tick: Optional[int] = None
        self.completed_tick: Optional[int] = None
        self.distance = PathPlanner.route_length(
            self.res_pos, PathPlanner.plan(self.res_pos, self.cus_pos)
        )

    def assign_to(self, rider_id: int) -> None:
        """Record that the order has been assigned to a rider.

        Args:
            rider_id: ID of the rider receiving the order.
        """
        self.rider_id = int(rider_id)
        self.status = "assigned"

    def mark_picking_up(self) -> None:
        """Mark that the rider is heading to the restaurant."""
        self.status = "picking_up"

    def mark_picked_up(self, tick: int) -> None:
        """Mark that the rider has reached the restaurant.

        Args:
            tick: Current simulated time.
        """
        self.status = "delivering"
        self.picked_up_tick = int(tick)

    def mark_completed(self, tick: int) -> None:
        """Mark the order as delivered to the customer.

        Args:
            tick: Current simulated time.
        """
        self.status = "completed"
        self.completed_tick = int(tick)

    def time_taken(self) -> Optional[int]:
        """Return ticks from creation to delivery, or None if not completed."""
        if self.completed_tick is None:
            return None
        return self.completed_tick - self.created_tick

    def efficiency(self) -> Optional[float]:
        """Return delivery efficiency as distance units per tick.

        Returns:
            Efficiency value, or None if the order is not completed or the
            time taken is zero.
        """
        taken = self.time_taken()
        if not taken:
            return None
        return self.distance / taken

    def to_dict(self) -> Dict[str, Any]:
        """Serialise the order to a JSON-compatible dictionary."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_tick": self.created_tick,
            "picked_up_tick": self.picked_up_tick,
            "completed_tick": self.completed_tick,
            "distance": self.distance,
        }

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "Order":
        """Rebuild an order from ``to_dict`` output.

        Args:
            data: Dictionary produced by ``to_dict``.

        Returns:
            Restored Order instance.
        """
        order = cls(
            data["res_pos"],
            data["cus_pos"],
            created_tick=data.get("created_tick", 0),
            order_id=data["id"],
        )
        order.status = data.get("status", "pending")
        order.rider_id = data.get("rider_id")
        order.picked_up_tick = data.get("picked_up_tick")
        order.completed_tick = data.get("completed_tick")
        order.distance = data.get("distance", order.distance)
        return order

"""Rider model: state machine, movement along streets and event log.

Rider states:
    0 (IDLE): no active orders.
    1 (PICKING_UP): travelling to the restaurant of the current order.
    2 (DELIVERING): travelling to the customer of the current order.

Orders are served first-in first-out. An order is assigned by appending it
to ``rider.orders``; the state transitions and log entries happen in
``move()``.
"""

import logging
from typing import Any, Dict, List, Mapping, Optional, Tuple

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

SPAWN_X = 400
SPAWN_Y = 400
DEFAULT_SPEED = 20  # units per tick


def _current_tick() -> int:
    """Return the simulated time stored in the global variables."""
    return int(globalvar.get_value("tick", 0) or 0)


class Rider:
    """A delivery rider that moves along the street grid.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current position (integers, 0-800).
        speed: Movement speed in units per tick.
        state: Rider status enum (0 idle, 1 picking up, 2 delivering).
        orders: All orders assigned to this rider, including completed ones
            kept as delivery history.
        special_case: True while the current leg is a straight-line move
            (target directly above, below, left or right).
        target: Current destination (x, y) or None.
        route: Remaining waypoints for the current leg.
        log: Chronological list of event dictionaries with keys
            "event", "order_id", "tick", "x", "y" and "state".
    """

    def __init__(
        self,
        rider_id: int,
        x: int = SPAWN_X,
        y: int = SPAWN_Y,
        speed: int = DEFAULT_SPEED,
    ) -> None:
        """Create a rider and log its spawn event.

        Args:
            rider_id: Unique rider identifier.
            x: Starting x coordinate (default 400).
            y: Starting y coordinate (default 400).
            speed: Movement speed in units per tick.
        """
        self.rider_id = int(rider_id)
        self.x = int(x)
        self.y = int(y)
        self.speed = int(speed)
        self.state = IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.target: Optional[Tuple[int, int]] = None
        self.route: List[Tuple[int, int]] = []
        self.log: List[Dict[str, Any]] = []
        self.log_event("spawn", None)

    @property
    def position(self) -> Tuple[int, int]:
        """Return the current (x, y) position."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Return a human-readable name for the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def active_orders(self) -> List[Any]:
        """Return orders that are not yet completed, in FIFO order."""
        return [order for order in self.orders if order.status != "completed"]

    def current_order(self) -> Optional[Any]:
        """Return the order currently being served, or None."""
        active = self.active_orders()
        return active[0] if active else None

    def log_event(self, event: str, order_id: Optional[int]) -> Dict[str, Any]:
        """Append an event entry to the rider log.

        Args:
            event: One of spawn, accepted_order, arrived_at_merchant,
                delivered_to_customer.
            order_id: Related order ID, or None for spawn events.

        Returns:
            The log entry that was appended.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": _current_tick(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.info(
            "rider=%s tick=%s event=%s order=%s pos=(%s,%s)",
            self.rider_id,
            entry["tick"],
            event,
            order_id,
            self.x,
            self.y,
        )
        return entry

    def _set_target(self, target: Tuple[int, int]) -> None:
        """Set the destination for the current leg and plan its route.

        Args:
            target: Destination (x, y).
        """
        self.target = (int(target[0]), int(target[1]))
        self.route = PathPlanner.plan(self.position, self.target)
        self.special_case = PathPlanner.is_special_case(self.position, self.target)

    def _travel(self, budget: int) -> int:
        """Move along the planned route using up to ``budget`` units.

        Args:
            budget: Maximum distance to travel this tick.

        Returns:
            Unused budget (0 if the rider used it all).
        """
        remaining = budget
        while remaining > 0 and self.route:
            wx, wy = self.route[0]
            dx = wx - self.x
            dy = wy - self.y
            dist = abs(dx) + abs(dy)
            if dist == 0:
                self.route.pop(0)
                continue
            if dx != 0 and dy != 0:
                # Routes are axis-aligned; this branch is a safety net.
                logger.warning("rider=%s non-axis segment; snapping", self.rider_id)
                self.x, self.y = wx, wy
                self.route.pop(0)
                continue
            if dist <= remaining:
                self.x, self.y = wx, wy
                remaining -= dist
                self.route.pop(0)
            else:
                if dx != 0:
                    self.x += remaining if dx > 0 else -remaining
                else:
                    self.y += remaining if dy > 0 else -remaining
                remaining = 0
        return remaining

    def move(self) -> None:
        """Advance the rider by one tick.

        Handles the state transitions:
            IDLE -> PICKING_UP when an order is accepted.
            PICKING_UP -> DELIVERING on arrival at the restaurant.
            DELIVERING -> PICKING_UP (next order) or IDLE on delivery.
        """
        order = self.current_order()
        if order is None:
            if self.state != IDLE or self.route or self.target is not None:
                self.state = IDLE
                self.target = None
                self.route = []
                self.special_case = False
            return

        if self.state == IDLE:
            self.state = PICKING_UP
            order.mark_picking_up()
            self.log_event("accepted_order", order.id)
            self._set_target(order.res_pos)
        elif self.target is None:
            destination = order.res_pos if self.state == PICKING_UP else order.cus_pos
            self._set_target(destination)

        self._travel(self.speed)
        if self.route:
            return

        if self.state == PICKING_UP:
            self._arrive_at_merchant(order)
        elif self.state == DELIVERING:
            self._deliver(order)

    def _arrive_at_merchant(self, order: Any) -> None:
        """Handle arrival at the restaurant and head for the customer.

        Args:
            order: Order being picked up.
        """
        order.mark_picked_up(_current_tick())
        self.state = DELIVERING
        self.log_event("arrived_at_merchant", order.id)
        self._set_target(order.cus_pos)

    def _deliver(self, order: Any) -> None:
        """Handle delivery to the customer and move to the next order.

        Args:
            order: Order that has just been delivered.
        """
        order.mark_completed(_current_tick())
        self.log_event("delivered_to_customer", order.id)
        self.target = None
        self.route = []
        self.special_case = False
        next_order = self.current_order()
        if next_order is None:
            self.state = IDLE
            return
        self.state = PICKING_UP
        next_order.mark_picking_up()
        self.log_event("accepted_order", next_order.id)
        self._set_target(next_order.res_pos)

    def to_dict(self) -> Dict[str, Any]:
        """Serialise the rider to a JSON-compatible dictionary."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "target": list(self.target) if self.target else None,
            "route": [list(point) for point in self.route],
            "order_ids": [order.id for order in self.orders],
            "log": self.log,
        }

    @classmethod
    def from_dict(cls, data: Mapping[str, Any], orders_by_id: Mapping[int, Any]) -> "Rider":
        """Rebuild a rider from ``to_dict`` output.

        Args:
            data: Dictionary produced by ``to_dict``.
            orders_by_id: Lookup of Order objects keyed by order ID.

        Returns:
            Restored Rider instance.
        """
        rider = cls(
            data["rider_id"],
            data.get("x", SPAWN_X),
            data.get("y", SPAWN_Y),
            data.get("speed", DEFAULT_SPEED),
        )
        rider.state = int(data.get("state", IDLE))
        rider.special_case = bool(data.get("special_case", False))
        target = data.get("target")
        rider.target = (int(target[0]), int(target[1])) if target else None
        rider.route = [(int(p[0]), int(p[1])) for p in data.get("route", [])]
        rider.orders = [
            orders_by_id[oid] for oid in data.get("order_ids", []) if oid in orders_by_id
        ]
        rider.log = list(data.get("log", []))
        return rider

"""Order management: creation, assignment and rider selection."""

import logging
import math
from typing import List, Optional, Sequence

import globalvar
from order_model import Order
from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the rider that should receive ``order``.

    The rider with the fewest active orders is preferred. Ties are broken by
    the Euclidean distance from the rider to the restaurant:
    sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2).

    Args:
        order: Order that needs a rider.

    Returns:
        The selected Rider, or None when no rider exists.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    rest_x, rest_y = order.res_pos

    def sort_key(rider: Rider):
        """Sort riders by load first, then by distance to the restaurant."""
        distance = math.sqrt((rider.x - rest_x) ** 2 + (rider.y - rest_y) ** 2)
        return (len(rider.active_orders()), distance, rider.rider_id)

    return min(riders, key=sort_key)


class OrderManager:
    """Creates, assigns and reports on orders."""

    @staticmethod
    def _orders() -> List[Order]:
        """Return the global order list, creating it if necessary."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords: Sequence[int]) -> Order:
        """Create an order from ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Args:
            coords: Four integer coordinates (already validated).

        Returns:
            The newly created Order, stored in the global order list.

        Raises:
            ValueError: If ``coords`` does not contain exactly four values.
        """
        if len(coords) != 4:
            raise ValueError("an order needs exactly four coordinates")
        tick = int(globalvar.get_value("tick", 0) or 0)
        order = Order(coords[0:2], coords[2:4], created_tick=tick)
        self._orders().append(order)
        logger.info("order %s created res=%s cus=%s", order.id, order.res_pos, order.cus_pos)
        return order

    def pending_orders(self) -> List[Order]:
        """Return orders that have not been assigned to a rider yet."""
        return [order for order in self._orders() if order.status == "pending"]

    def completed_orders(self) -> List[Order]:
        """Return orders that have been delivered."""
        return [order for order in self._orders() if order.status == "completed"]

    def assign_order(self, order: Order) -> Optional[Rider]:
        """Assign ``order`` to the best available rider.

        Args:
            order: A pending order.

        Returns:
            The rider that received the order, or None if no rider exists.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

def load_config(path: Optional[str] = None) -> Dict[str, Any]:
    """Load system parameters from a JSON configuration file.

    Missing or invalid files fall back to ``DEFAULT_CONFIG``.

    Args:
        path: Path to the JSON configuration file, or None.

    Returns:
        Merged configuration dictionary.
    """
    config = dict(DEFAULT_CONFIG)
    config["spawn_position"] = list(DEFAULT_CONFIG["spawn_position"])
    if not path or not os.path.exists(path):
        return config
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.error("could not read config %s: %s", path, exc)
        return config
    if isinstance(data, dict):
        for key in DEFAULT_CONFIG:
            if key in data:
                config[key] = data[key]
    return config

class SystemManager:
    """Core manager for funds, riders, dispatch, ticks and state checks."""

    def __init__(self) -> None:
        """Create the manager. Global state must already be initialised."""
        self.order_manager = OrderManager()

    # ----- global state accessors -------------------------------------
    @property
    def config(self) -> Dict[str, Any]:
        """Return the active configuration dictionary."""
        return globalvar.get_value("config") or DEFAULT_CONFIG

    @property
    def funds(self) -> int:
        """Return the system fund balance (may be negative)."""
        return int(globalvar.get_value("funds", 0) or 0)

    @funds.setter
    def funds(self, value: int) -> None:
        """Set the system fund balance."""
        globalvar.set_value("funds", int(value))

    @property
    def riders(self) -> List[Rider]:
        """Return the live rider list."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def current_tick(self) -> int:
        """Return the current simulated time."""
        return int(globalvar.get_value("tick", 0) or 0)

    def is_running(self) -> bool:
        """Return True while the main loop should keep running."""
        return bool(globalvar.get_value("running", False))

    def stop(self) -> None:
        """Request the main loop to exit."""
        globalvar.set_value("running", False)

    def rider_by_id(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with ``rider_id``, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def idle_rider_count(self) -> int:
        """Return the number of riders without active orders."""
        return sum(1 for rider in self.riders if not rider.active_orders())

    # ----- recruitment -------------------------------------------------
    def should_recruit(self, pending_count: int) -> bool:
        """Return True when a new rider should be recruited.

        Args:
            pending_count: Number of pending orders.

        Returns:
            True if funds >= threshold and pending orders exceed idle riders.
        """
        threshold = int(self.config["recruit_threshold"])
        return self.funds >= threshold and pending_count > self.idle_rider_count()

    def recruit_rider(self) -> Optional[Rider]:
        """Create a new rider and deduct the recruitment cost.

        Returns:
            The new Rider, or None if funds are below the threshold.
        """
        cfg = self.config
        if self.funds < int(cfg["recruit_threshold"]):
            return None
        self.funds = self.funds - int(cfg["recruit_cost"])
        counter = int(globalvar.get_value("rider_id_counter", 0) or 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        spawn_x, spawn_y = cfg["spawn_position"]
        rider = Rider(counter, int(spawn_x), int(spawn_y), int(cfg["rider_speed"]))
        self.riders.append(rider)
        logger.info("recruited rider %s; funds now %s", counter, self.funds)
        return rider

    # ----- dispatch ----------------------------------------------------
    def dispatch_order(self, order: Order) -> Tuple[Optional[Rider], bool]:
        """Recruit if needed and assign ``order`` to a rider.

        Args:
            order: Pending order to dispatch.

        Returns:
            Tuple of (assigned rider or None, whether a new rider was added).
        """
        pending = len(self.order_manager.pending_orders())
        recruited = False
        if self.should_recruit(pending) and self.recruit_rider() is not None:
            recruited = True
        rider = self.order_manager.assign_order(order)
        return rider, recruited

    def dispatch_pending(self) -> int:
        """Try to dispatch every pending order.

        Returns:
            Number of orders assigned during this call.
        """
        assigned = 0
        for order in list(self.order_manager.pending_orders()):
            rider, _ = self.dispatch_order(order)
            if rider is not None:
                assigned += 1
        return assigned

    # ----- simulation clock -------------------------------------------
    def tick(self) -> None:
        """Advance the simulation by one unit of time."""
        globalvar.set_value("tick", self.current_tick + 1)
        try:
            self.dispatch_pending()
        except Exception:  # pragma: no cover - defensive recovery
            logger.exception("dispatch_pending failed during tick")
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                logger.exception("rider %s failed to move", rider.rider_id)

    def run_ticks(self, count: int) -> int:
        """Advance the simulation ``count`` ticks.

        Args:
            count: Number of ticks to run.

        Returns:
            Number of orders completed during these ticks.
        """
        before = len(self.order_manager.completed_orders())
        for _ in range(count):
            self.tick()
        return len(self.order_manager.completed_orders()) - before

    # ----- consistency -------------------------------------------------
    def check_consistency(self) -> List[str]:
        """Check and repair system state consistency.

        Repairs performed:
            * Riders that are busy without active orders are reset to idle.
            * Orders marked as assigned but held by no rider return to pending.

        Returns:
            Human-readable description of each issue found and repaired.
        """
        issues: List[str] = []
        riders_by_id = {rider.rider_id: rider for rider in self.riders}
        for rider in self.riders:
            if rider.state != IDLE and not rider.active_orders():
                issues.append(
                    f"rider #{rider.rider_id} was {rider.state_name} with no orders; reset to idle"
                )
                rider.state = IDLE
                rider.target = None
                rider.route = []
        for order in globalvar.get_value("orders") or []:
            if order.status in ("assigned", "picking_up", "delivering"):
                holder = riders_by_id.get(order.rider_id)
                if holder is None or order not in holder.orders:
                    issues.append(
                        f"order #{order.id} was {order.status} but no rider holds it; returned to pending"
                    )
                    order.status = "pending"
                    order.rider_id = None
        if self.funds < 0:
            issues.append(f"system funds are negative ({self.funds})")
        return issues

    # ----- persistence -------------------------------------------------
    def save_state(self, path: str) -> None:
        """Save riders, orders, funds and clock to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            ValueError: If the file cannot be written.
        """
        data = {
            "funds": self.funds,
            "tick": self.current_tick,
            "rider_id_counter": int(globalvar.get_value("rider_id_counter", 0) or 0),
            "config": self.config,
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in globalvar.get_value("orders") or []],
        }
        try:
            with open(path, "w", encoding="utf-8") as handle:
                json.dump(data, handle, indent=2)
        except OSError as exc:
            raise ValueError(f"could not save state to '{path}': {exc}") from exc
        logger.info("state saved to %s", path)

    def load_state(self, path: str) -> None:
        """Restore riders, orders, funds and clock from a JSON file.

        Args:
            path: Source file path.

        Raises:
            ValueError: If the file is missing or malformed.
        """
        try:
            with open(path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
            orders = [Order.from_dict(item) for item in data.get("orders", [])]
            by_id = {order.id: order for order in orders}
            riders = [Rider.from_dict(item, by_id) for item in data.get("riders", [])]
        except OSError as exc:
            raise ValueError(f"could not read state file '{path}': {exc}") from exc
        except (json.JSONDecodeError, KeyError, TypeError, ValueError) as exc:
            raise ValueError(f"state file '{path}' is malformed: {exc}") from exc
        globalvar.set_value("orders", orders)
        globalvar.set_value("riders", riders)
        globalvar.set_value("funds", int(data.get("funds", self.funds)))
        globalvar.set_value("tick", int(data.get("tick", 0)))
        globalvar.set_value("rider_id_counter", int(data.get("rider_id_counter", len(riders))))
        self.check_consistency()
        logger.info("state loaded from %s", path)

"""Command parsing, shortcut resolution, history and input validation."""

from typing import Callable, List, Tuple

COMMANDS = {
    "start": "start",
    "s": "start",
    "status": "status",
    "st": "status",
    "orders": "orders",
    "o": "orders",
    "riders": "riders",
    "r": "riders",
    "map": "map",
    "m": "map",
    "tick": "tick",
    "t": "tick",
    "save": "save",
    "load": "load",
    "history": "history",
    "hist": "history",
    "help": "help",
    "h": "help",
    "?": "help",
    "quit": "quit",
    "exit": "quit",
    "q": "quit",
}

COORD_LABELS = ["restaurant x", "restaurant y", "customer x", "customer y"]


class InputManager:
    """Parses user commands and validates coordinate input.

    Supports shortcut aliases, command history, '!!' (repeat last command)
    and '!n' (repeat command number n).
    """

    MIN_COORD = 0
    MAX_COORD = 1000

    HELP_TEXT = (
        "Commands:\n"
        "  start [rx ry cx cy]  (s)  Place an order; restaurant then customer coordinates (0-1000)\n"
        "  status               (st) Show funds, riders and their orders\n"
        "  orders               (o)  Show all orders\n"
        "  riders               (r)  Show every rider status change\n"
        "  map                  (m)  Show the ASCII map\n"
        "  tick [n]             (t)  Advance the simulation by n ticks (default 1)\n"
        "  save [file]          Save state to JSON (default dispatch_state.json)\n"
        "  load [file]          Load state from JSON\n"
        "  history              List previously entered commands\n"
        "  !! / !n              Repeat the last command / command number n\n"
        "  help                 (h, ?) Show this help\n"
        "  quit                 (q)  Exit (exit also works)"
    )

    def __init__(self) -> None:
        """Create an input manager with empty history."""
        self._history: List[str] = []

    @property
    def history(self) -> List[str]:
        """Return a copy of the command history."""
        return list(self._history)

    def record(self, line: str) -> None:
        """Add a command to the history.

        Args:
            line: Command line entered by the user.
        """
        self._history.append(line)

    def expand_history(self, line: str) -> str:
        """Resolve '!!' and '!n' history references.

        Args:
            line: Raw user input.

        Returns:
            The command to execute.

        Raises:
            ValueError: If the referenced history entry does not exist.
        """
        if line == "!!":
            if not self._history:
                raise ValueError("there is no previous command to repeat")
            return self._history[-1]
        if line.startswith("!") and line[1:].isdigit():
            index = int(line[1:])
            if not 1 <= index <= len(self._history):
                raise ValueError(f"no history entry number {index}")
            return self._history[index - 1]
        return line

    def parse_command(self, line: str) -> Tuple[str, List[str]]:
        """Split a command line into a canonical command name and arguments.

        Args:
            line: Command line text.

        Returns:
            Tuple of (canonical command name, argument tokens).

        Raises:
            ValueError: If the command is empty or unknown.
        """
        tokens = line.split()
        if not tokens:
            raise ValueError("empty command")
        name = tokens[0].lower()
        if name not in COMMANDS:
            raise ValueError(f"unknown command '{tokens[0]}'. Type 'help' for a list of commands.")
        return COMMANDS[name], tokens[1:]

    @staticmethod
    def parse_integer(text: str, label: str = "value") -> int:
        """Convert ``text`` to an integer with a clear error message.

        Args:
            text: Text to convert.
            label: Name used in the error message.

        Returns:
            Parsed integer.

        Raises:
            ValueError: If the text is not an integer.
        """
        try:
            return int(str(text).strip())
        except ValueError:
            raise ValueError(f"{label} must be an integer, got '{text}'") from None

    def validate_coordinate(self, value: int, label: str) -> int:
        """Check that a coordinate lies within 0-1000.

        Args:
            value: Integer coordinate.
            label: Name used in the error message.

        Returns:
            The validated value.

        Raises:
            ValueError: If the value is out of range.
        """
        if not self.MIN_COORD <= value <= self.MAX_COORD:
            raise ValueError(
                f"{label} must be between {self.MIN_COORD} and {self.MAX_COORD}, got {value}"
            )
        return value

    def parse_coordinates(self, tokens: List[str]) -> List[int]:
        """Parse four coordinate tokens into a validated list.

        Args:
            tokens: Exactly four tokens: restaurant x, y, customer x, y.

        Returns:
            List of four validated integers.

        Raises:
            ValueError: On wrong token count, non-integer or out-of-range values.
        """
        if len(tokens) != 4:
            raise ValueError(
                f"expected 4 coordinates (restaurant x y, customer x y), got {len(tokens)}"
            )
        coords = []
        for label, token in zip(COORD_LABELS, tokens):
            value = self.parse_integer(token, label)
            coords.append(self.validate_coordinate(value, label))
        return coords

    def read_coordinates(
        self, input_func: Callable[[str], str], output_func: Callable[[str], None] = print
    ) -> List[int]:
        """Prompt for the four coordinates, re-asking on invalid entries.

        Args:
            input_func: Function used to read a line (usually ``input``).
            output_func: Function used to show error messages.

        Returns:
            List of four validated integers.
        """
        coords: List[int] = []
        for label in COORD_LABELS:
            while True:
                text = input_func(f"{label} (0-1000): ")
                try:
                    value = self.parse_integer(text, label)
                    coords.append(self.validate_coordinate(value, label))
                    break
                except ValueError as exc:
                    output_func(f"Error: {exc}. Please try again.")
        return coords

                self.output("")
                self.system.stop()
                break
            line = raw.strip()
            if not line:
                continue
            try:
                line = self.inputs.expand_history(line)
                self.inputs.record(line)
                self.execute(line)
            except ValueError as exc:
                self.output(f"Error: {exc}")
            except Exception as exc:  # error recovery: keep the loop alive
                logger.exception("unexpected error while executing '%s'", line)
                self.output(f"Unexpected error: {exc}. Recovering system state.")
                self.recover()

    def execute(self, line: str) -> None:
        """Parse and run a single command line.

        Args:
            line: Command text (already history-expanded).

        Raises:
            ValueError: For unknown commands or invalid arguments.
        """
        command, args = self.inputs.parse_command(line)
        self._handlers[command](args)

    def recover(self) -> None:
        """Run the consistency check and report any repairs."""
        if self.system is None:
            return
        issues = self.system.check_consistency()
        for issue in issues:
            self.output(f"Recovered: {issue}")

    # ----- command handlers ------------------------------------------
    def _cmd_start(self, args: List[str]) -> None:
        """Place a new order and dispatch it."""
        if args:
            coords = self.inputs.parse_coordinates(args)
        else:
            coords = self.inputs.read_coordinates(self.input_func, self.output)
        order = self.system.order_manager.create_order(coords)
        rider, recruited = self.system.dispatch_order(order)
        self.output(self.display.format_new_order(order, rider, recruited))

    def _cmd_status(self, args: List[str]) -> None:
        """Show the system status snapshot."""
        issues = self.system.check_consistency()
        self.output(self.display.format_status(issues))

    def _cmd_orders(self, args: List[str]) -> None:
        """Show details of every order."""
        self.output(self.display.format_orders())

    def _cmd_riders(self, args: List[str]) -> None:
        """Show the rider status change log."""
        self.output(self.display.format_rider_log())

    def _cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map."""
        self.output(self.display.render_map())

    def _cmd_tick(self, args: List[str]) -> None:
        """Advance the simulation clock."""
        count = 1
        if args:
            count = InputManager.parse_integer(args[0], "tick count")
            if not 1 <= count <= MAX_TICKS_PER_COMMAND:
                raise ValueError(f"tick count must be between 1 and {MAX_TICKS_PER_COMMAND}")
        completed = self.system.run_ticks(count)
        self.output(
            f"Advanced {count} tick(s). Current tick: {self.system.current_tick}. "
            f"Orders completed in this step: {completed}. Funds: {self.system.funds}."
        )

    def _cmd_save(self, args: List[str]) -> None:
        """Save the state to a JSON file."""
        path = args[0] if args else DEFAULT_STATE_FILE
        self.system.save_state(path)
        self.output(f"State saved to {path}.")

    def _cmd_load(self, args: List[str]) -> None:
        """Load the state from a JSON file."""
        path = args[0] if args else DEFAULT_STATE_FILE
        self.system.load_state(path)
        self.output(f"State loaded from {path}.")

    def _cmd_history(self, args: List[str]) -> None:
        """List previously entered commands."""
        history = self.inputs.history
        if not history:
            self.output("No command history yet.")
            return
        self.output("\n".join(f"{i}: {cmd}" for i, cmd in enumerate(history, start=1)))

    def _cmd_help(self, args: List[str]) -> None:
        """Show command help."""
        self.output(InputManager.HELP_TEXT)

    def _cmd_quit(self, args: List[str]) -> None:
        """Exit the program."""
        self.system.stop()
        self.output("Goodbye.")

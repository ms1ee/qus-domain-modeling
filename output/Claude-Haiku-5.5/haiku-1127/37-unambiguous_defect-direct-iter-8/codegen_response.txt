"""Participant list loading, verification and group extraction."""

import os
import re

from lottery_system.defaults import DEFAULT_USER_LIST_TEXT
from lottery_system.models import User

# Accepted field separators: ASCII comma, full-width comma, or tab.
_SEPARATOR_PATTERN = re.compile(r"[,\uff0c\t]")


class ListManager:
    """Loads, verifies and summarizes participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_USER_LIST_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list from the given path."""
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it line by line."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines; return valid users and format errors."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text or text.startswith("#"):
                continue
            parts = [part.strip() for part in _SEPARATOR_PATTERN.split(text)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{text}' "
                    f"(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users and duplicates.

        The first occurrence of each name is kept as valid; every later
        occurrence of an already-seen name is reported in the duplicates list.
        """
        valid_users: list[User] = []
        duplicates: list[User] = []
        seen_names: set[str] = set()
        for user in users:
            key = user.name.strip().casefold()
            if key in seen_names:
                duplicates.append(user)
            else:
                seen_names.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of people."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management: presets, custom prizes, avoidance rules and ordering."""

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages (empty on success)."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a new prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer else None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer or None,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in lottery sequence (a copy of the list)."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes using 1-based current positions in the new order.

        Example: with prizes [A, B, C], order [3, 1, 2] yields [C, A, B].
        Returns False if the order is not a permutation of 1..n.
        """
        n = len(self._prizes)
        try:
            valid = sorted(order) == list(range(1, n + 1))
        except TypeError:
            valid = False
        if not valid:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Winner recording, grouping by prize level, and CSV export."""

import csv
import os
import re
from dataclasses import dataclass
from datetime import date
from typing import Any

from lottery_system.models import User

_LEVEL_PATTERN = re.compile(r"\b(first|second|third)\b", re.IGNORECASE)
_LEVEL_RANKS = {"first": 1, "second": 2, "third": 3}
_UNRANKED = 99


def prize_level_rank(prize_name: str) -> int:
    """Return the level rank of a prize (1 = highest). Unknown levels rank last."""
    match = _LEVEL_PATTERN.search(prize_name or "")
    if match is None:
        return _UNRANKED
    return _LEVEL_RANKS[match.group(1).lower()]


def sort_prizes_by_level(prizes: list[Any]) -> list[Any]:
    """Sort prizes from high to low level; ties keep their configured order."""
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda item: (prize_level_rank(item[1].name), item[0]))
    return [prize for _, prize in indexed]


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str = ""
    order: int = 0


class ResultManager:
    """Keeps the winner list in winning order and produces summaries/exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Any) -> Winner:
        """Record a winner for a prize (a Prize object or a prize name)."""
        prize_name = prize.name if hasattr(prize, "name") else str(prize)
        winner = Winner(user=user, prize_name=prize_name, order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Any]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low).

        Within each prize, winners are in winning time order.
        """
        grouped: dict[str, list[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            grouped[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return grouped

    def export_to_csv(self, prizes: list[Any]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt as CSV."""
        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Prize Name", "Name", "Department", "Win Order"])
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        writer.writerow(
                            [prize_name, winner.user.name, winner.user.department, winner.order]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

"""Lottery rule engine: eligibility filtering, special rules, and draws."""

import random
from typing import Any

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    User,
)
from lottery_system.services.result_manager import ResultManager

# Lottery persons whose draw results feed the Cheng Ru group protection check.
_TRACKED_DRAWERS = (CHEN_QIXIAN, JIN_DUO)
_PROTECTED_DRAWER = MA_JIAQI


def _same_text(a: str, b: str) -> bool:
    return (a or "").strip().casefold() == (b or "").strip().casefold()


def _drawer_is(prize: Any, drawer: str) -> bool:
    return prize.associated_drawer is not None and _same_text(prize.associated_drawer, drawer)


class LotteryManager:
    """Runs individual draws while enforcing all lottery rules."""

    def __init__(
        self,
        users: list[User],
        result_manager: ResultManager | None = None,
        rng: random.Random | None = None,
    ) -> None:
        self.users: list[User] = list(users)
        self.result_manager: ResultManager = (
            result_manager if result_manager is not None else ResultManager()
        )
        # Whether the last (up to two) draws by Chen Qixian / Jin Duo contained
        # a Cheng Ru group member. True = protected member won.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.Random()
        self._won_names: set[str] = set()

    # ------------------------------------------------------------------ rules
    @staticmethod
    def is_relaxed_prize(prize: Any) -> bool:
        """Special rule: prizes drawn by Li Bin use relaxed filtering."""
        return _drawer_is(prize, LI_BIN)

    def is_cheng_ru_protection_active(self, prize: Any) -> bool:
        """Fallback rule: Ma Jiaqi's pool is limited to Cheng Ru members when
        neither of the last two results of Chen Qixian / Jin Duo had a Cheng Ru member."""
        return (
            _drawer_is(prize, _PROTECTED_DRAWER)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def has_won(self, user: User) -> bool:
        """Anti-duplication check: whether this person already won any prize."""
        return user.name in self._won_names

    def _available_users(self) -> list[User]:
        return [user for user in self.users if not self.has_won(user)]

    def _get_eligible_pool(self, prize: Any) -> list[User]:
        """Compute the candidate pool for a prize."""
        available = self._available_users()

        if self.is_cheng_ru_protection_active(prize):
            protected = [u for u in available if _same_text(u.department, CHENG_RU_GROUP)]
            if protected:
                return protected
            # No protected member remains; fall back to the normal pool.

        if self.is_relaxed_prize(prize):
            return available

        avoid = {group.strip().casefold() for group in (prize.avoid_groups or [])}
        return [u for u in available if u.department.strip().casefold() not in avoid]

    def get_eligible_pool(self, prize: Any) -> list[User]:
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    # ------------------------------------------------------------------ draws
    def draw_once(self, prize: Any) -> User | None:
        """Draw one winner for the prize. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None

        user = self._rng.choice(pool)
        self._won_names.add(user.name)
        self.result_manager.add_winner(user, prize)

        drawer = prize.associated_drawer
        if drawer is not None and any(_same_text(drawer, d) for d in _TRACKED_DRAWERS):
            self._record_draw(_same_text(user.department, CHENG_RU_GROUP))
        return user

    def _record_draw(self, contains_cheng_ru: bool) -> None:
        self.last_two_draw_results = (self.last_two_draw_results + [contains_cheng_ru])[-2:]

    @property
    def winners(self) -> list:
        """All winners recorded so far, in winning order."""
        return self.result_manager.get_winners()

from lottery_system.models import LI_BIN, Prize, User

def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Interactive command-line controller."""

    def __init__(self, input_func=input, output_func=print) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.results: ResultManager | None = None

    # ---------------------------------------------------------------- helpers
    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    def _out(self, text: str = "") -> None:
        self._print(text)

    # ------------------------------------------------------------- main loop
    def run(self) -> None:
        self._out("=== Team Building Smart Lottery System ===")
        while True:
            self._print_main_menu()
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self.load_list_menu()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.prize_menu()
            elif choice == "4":
                self.start_lottery()
            elif choice == "5":
                self.results_menu()
            elif choice == "0":
                self._out("Goodbye.")
                break
            else:
                self._out("Invalid option, please try again.")

    def _print_main_menu(self) -> None:
        self._out("")
        self._out("---------- Main Menu ----------")
        self._out("1. Load participant list")
        self._out("2. View department groups")
        self._out("3. Prize configuration")
        self._out("4. Start lottery")
        self._out("5. Results (summary / export)")
        self._out("0. Exit")

    # ------------------------------------------------------------ list menus
    def load_list_menu(self) -> None:
        self._out("1. Built-in default list")
        self._out("2. Custom txt list (format: Name,Department per line)")
        self._out("0. Cancel")
        choice = self._ask("Select list source: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip()
            users, errors = self.list_manager.load_custom_list(path)
        else:
            self._out("Cancelled.")
            return
        self._verify_and_apply(users, errors)

    def _verify_and_apply(self, users: list[User], errors: list[str]) -> None:
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            self.users = valid
            self._out(f"List loaded successfully: {len(valid)} participants.")
            return

        self._out("List verification found problems:")
        for error in errors:
            self._out(f"  {error}")
        for dup in duplicates:
            self._out(f"  Duplicate name: {dup.name} ({dup.department})")
        self._out(
            f"Valid participants: {len(valid)}. "
            "Type 'r' to remove invalid data and continue, or 'q' to terminate the upload."
        )
        while True:
            command = self._ask("Your choice (r/q): ").strip().lower()
            if command == "r":
                self.users = valid
                self._out(f"Invalid data removed. Loaded {len(valid)} participants.")
                return
            if command == "q":
                self._out("Upload terminated. Previous list kept.")
                return
            self._out("Please enter 'r' or 'q'.")

    def show_groups(self) -> None:
        if not self.users:
            self._out("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._out("Current department groups:")
        for index, (department, count) in enumerate(info.items(), start=1):
            self._out(f"  {index}. {department}: {count} people")

    # ----------------------------------------------------------- prize menus
    def prize_menu(self) -> None:

            self._out("")
            self._out("---------- Prize Configuration ----------")
            self._out("1. View prizes")
            self._out("2. Load preset prizes")
            self._out("3. Add custom prize")
            self._out("4. Set avoidance groups / drawer for a prize")
            self._out("5. Reorder lottery sequence")
            self._out("0. Back")
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self.show_prizes()
            elif choice == "2":
                errors = self.prize_manager.load_default_prizes()
                for error in errors:
                    self._out(f"  {error}")
                self._out("Preset prizes loaded.")
            elif choice == "3":
                self.add_prize_interactive()
            elif choice == "4":
                self.set_prize_rules_interactive()
            elif choice == "5":
                self.reorder_interactive()
            elif choice == "0":
                return
            else:
                self._out("Invalid option, please try again.")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._out("No prizes configured.")
            return
        self._out("Lottery sequence:")
        for index, prize in enumerate(prizes, start=1):
            self._out(f"  {index}. {self._describe_prize_line(prize)}")

    def _describe_prize_line(self, prize: Prize) -> str:
        drawer = prize.associated_drawer or "None"
        avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
        return (
            f"{prize.name} | Quantity: {prize.quantity} | "

    def add_prize_interactive(self) -> None:
        name = self._ask("Prize name: ").strip()
        quantity = self._ask_positive_int("Quantity: ")
        if quantity is None:
            return
        avoid_text = self._ask(
            "Avoidance groups (comma separated, blank for none): "
        ).strip()
        avoid = [g.strip() for g in avoid_text.split(",") if g.strip()]
        drawer = self._ask("Associated lottery person (blank for none): ").strip()
        try:
            prize = self.prize_manager.add_prize(name, quantity, avoid, drawer or None)
            self._out(f"Prize added: {prize.name}")
        except ValueError as exc:
            self._out(f"Cannot add prize: {exc}")

    def set_prize_rules_interactive(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._out("No prizes configured.")
            return
        self.show_prizes()
        index = self._ask_index("Prize serial number: ", len(prizes))
        if index is None:
            return
        prize = prizes[index]
        avoid_text = self._ask(
            "Avoidance groups (comma separated, blank for none): "
        ).strip()
        prize.avoid_groups = [g.strip() for g in avoid_text.split(",") if g.strip()]
        drawer = self._ask("Associated lottery person (blank for none): ").strip()
        prize.associated_drawer = drawer or None
        self._out("Prize rules updated.")

    def reorder_interactive(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._out("No prizes configured.")
            return
        self.show_prizes()
        text = self._ask(
            "Enter the new order as serial numbers separated by spaces (blank to cancel): "
        ).strip()
        if not text:
            return
        try:
            order = [int(token) for token in text.split()]
        except ValueError:
            self._out("Invalid input: please enter integers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._out("Lottery sequence updated.")
            self.show_prizes()
        else:
            self._out("Invalid order: must contain every serial number exactly once.")

    def _ask_positive_int(self, prompt: str) -> int | None:
        for _ in range(3):
            text = self._ask(prompt).strip()
            if text.isdigit() and int(text) > 0:
                return int(text)
            self._out("Please enter a positive integer.")
        self._out("Too many invalid attempts; cancelled.")
        return None

    def _ask_index(self, prompt: str, size: int) -> int | None:
        text = self._ask(prompt).strip()
        if text.isdigit() and 1 <= int(text) <= size:
            return int(text) - 1
        self._out("Invalid serial number.")
        return None

    # ---------------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.users:
            self._out("Please load the participant list first (option 1).")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._out("Please configure prizes first (option 3).")
            return
        if self.results is not None and self.results.get_winners():
            self._out("Starting a new lottery: previous results are cleared.")

        self.results = ResultManager()
        lottery = LotteryManager(self.users, result_manager=self.results)
        self._out("Lottery started. During the draw you may type:")
        self._out("  Enter        -> draw the next winner")
        self._out("  pause        -> pause the lottery")
        self._out("  terminate    -> end the current round and skip to the next")

        for round_no, prize in enumerate(prizes, start=1):
            self._run_round(lottery, prize, round_no)

        self._out("")
        self._out("All rounds completed.")
        self._show_summary(prizes)

    def _rule_description(self, lottery: LotteryManager, prize: Prize) -> str:
        parts: list[str] = []
        if lottery.is_relaxed_prize(prize):
            parts.append(
                f"Relaxed rules ({LI_BIN}): only previous winners are excluded"
            )
        else:
            text = "Excludes previous winners"
            if prize.avoid_groups:
                text += " and members of groups: " + ", ".join(prize.avoid_groups)
            parts.append(text)
        if lottery.is_cheng_ru_protection_active(prize):
            parts.append("Cheng Ru group protection active: only Cheng Ru Group members eligible")
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}")
        return "; ".join(parts)

    def _run_round(self, lottery: LotteryManager, prize: Prize, round_no: int) -> None:
        self._out("")
        self._out(
            f"=== Round {round_no}: {prize.name} | Quantity: {prize.quantity} | "
            f"Rule: {self._rule_description(lottery, prize)} ==="
        )
        round_winners: list[User] = []
        index = 1
        while index <= prize.quantity:
            command = self._ask(
                f"Press Enter to draw the {_ordinal(index)} winner (pause / terminate): "
            ).strip().lower()
            if command == "pause":
                if not self._wait_for_resume():
                    self._out("Round terminated.")
                    break
            elif command == "terminate":
                self._out("Round terminated.")
                break
            elif command:
                self._out("Unknown command.")
                continue

            self._out(f"Currently drawing the {_ordinal(index)} winner...")
            user = lottery.draw_once(prize)
            if user is None:
                self._out("No eligible candidates remain for this prize; round ends.")
                break
            round_winners.append(user)
            self._out(
                f"Congratulations {user.name} ({user.department}) for winning {prize.name}"
            )
            index += 1

        self._out(f"Round {round_no} finished. Winners this round: {len(round_winners)}")

    def _wait_for_resume(self) -> bool:
        """Block until 'continue' (True) or 'terminate' (False)."""
        self._out("Lottery paused.")
        while True:
            command = self._ask("Type 'continue' to resume or 'terminate' to end this round: ")
            command = command.strip().lower()
            if command == "continue":
                self._out("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            self._out("Please type 'continue' or 'terminate'.")

    def _show_summary(self, prizes: list[Prize]) -> None:
        if self.results is None:
            self._out("No lottery results yet.")
            return
        grouped = self.results.get_winners_by_prize(prizes)
        self._out("")
        self._out("=========== Winner List ===========")
        if not any(grouped.values()):
            self._out("No winners were drawn.")
            return
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            self._out(f"[{prize_name}]")
            for winner in winners:
                self._out(
                    f"  #{winner.order} {winner.user.name} ({winner.user.department})"
                )

    # ---------------------------------------------------------------- results
    def results_menu(self) -> None:
        if self.results is None:
            self._out("No lottery results yet.")
            return
        self._out("1. Show winner summary")
        self._out("2. Export winner list to txt (CSV format)")
        self._out("0. Back")
        choice = self._ask("Select an option: ").strip()
        prizes = self.prize_manager.get_prizes()
        if choice == "1":
            self._show_summary(prizes)
        elif choice == "2":
            _, message = self.results.export_to_csv(prizes)
            self._out(message)
        else:
            return


def main() -> None:
    """Start the interactive lottery system."""
    cli = LotteryCLI()
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting lottery system.")

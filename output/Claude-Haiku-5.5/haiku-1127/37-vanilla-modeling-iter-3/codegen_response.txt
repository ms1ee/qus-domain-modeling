"""Core data models for the lottery system."""

from __future__ import annotations

import re
from dataclasses import dataclass, field

CUSTOM_PRIZE_LEVEL = 99

_LEVEL_WORDS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
}

_LEVEL_PATTERN = re.compile(
    r"\b(first|second|third|fourth|fifth)\s+prize\b", re.IGNORECASE
)


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize configured for a lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None


def prize_level(name: str) -> int:
    """Return the level of a prize derived from its name.

    Lower numbers mean higher prize levels (First Prize = 1).
    Prizes without a recognised level keyword are placed last.
    """
    match = _LEVEL_PATTERN.search(name)
    if not match:
        return CUSTOM_PRIZE_LEVEL
    return _LEVEL_WORDS[match.group(1).lower()]

from __future__ import annotations

from collections import Counter

from lottery_system.models import User

DEFAULT_LIST_LINES: list[str] = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Lin Xiao,Technology Department",
    "Tang Wei,Technology Department",
    "Wang Wu,Product Department",
    "Zhao Liu,Product Department",
    "Xu Lei,Product Department",
    "Sun Qi,Marketing Department",
    "Zhou Ba,Marketing Department",
    "Song Yu,Marketing Department",
    "Jin Duo,Operations Department",
    "Chen Qixian,Operations Department",
    "Ma Jiaqi,Human Resources Department",
    "Luo Hao,Human Resources Department",
    "Li Bin,Finance Department",
    "He Fang,Finance Department",
    "Wu Jiu,Cheng Ru Group",
    "Zheng Shi,Cheng Ru Group",
    "Liu Yang,Cheng Ru Group",
    "Gao Ming,Cheng Ru Group",
]


class ListManager:
    """Loads, validates and summarises participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse its lines into users and errors."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines, collecting format errors."""
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw in enumerate(lines, start=1):
            content = raw.strip()
            if not content:
                continue
            parts = [part.strip() for part in content.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_number}: '{content}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(
        self, users: list[User]
    ) -> tuple[list[User], list[User]]:
        """Split users into valid users (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department group names mapped to their member counts."""
        counts = Counter(user.department for user in users)
        return dict(sorted(counts.items()))

from __future__ import annotations

from lottery_system.models import Prize

DEFAULT_PRIZES: list[tuple[str, int, str | None]] = [
    ("Third Prize - Movie Tickets", 5, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Presets already present are skipped."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            if any(prize.name == name for prize in self._prizes):
                continue
            try:
                self.add_prize(name, quantity, [], drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name cannot be empty")
        if not isinstance(quantity, int) or quantity < 1:
            raise ValueError("quantity must be a positive integer")
        if any(prize.name == clean_name for prize in self._prizes):
            raise ValueError(f"prize '{clean_name}' already exists")
        drawer = (associated_drawer or "").strip() or None
        groups = list(dict.fromkeys(avoid_groups or []))
        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def update_prize(
        self,
        index: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> bool:
        """Update avoidance groups and lottery person of the prize at a 0-based index."""
        if not 0 <= index < len(self._prizes):
            return False
        prize = self._prizes[index]
        prize.avoid_groups = list(dict.fromkeys(avoid_groups or []))
        prize.associated_drawer = (associated_drawer or "").strip() or None
        return True

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if not all(isinstance(pos, int) for pos in order):
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in order]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
FALLBACK_DRAWER = "Ma Jiaqi"
TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_GROUP = "Cheng Ru Group"


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.last_two_draw_results: list[bool] = []
        self._winner_names: set[str] = set()
        self._rng = random.Random()

    def is_winner(self, user: User) -> bool:
        """Return True if the user has already won in any round."""
        return user.name in self._winner_names

    def _is_fallback_active(self, prize: Prize) -> bool:
        """Fallback applies when the lottery person is Ma Jiaqi and neither of the
        last two draws by Chen Qixian / Jin Duo contained Cheng Ru Group members."""
        return (
            prize.associated_drawer == FALLBACK_DRAWER
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the list of users who may win the given prize."""
        available = [user for user in self.users if not self.is_winner(user)]

        if self._is_fallback_active(prize):
            # Forced limitation, even if Cheng Ru Group is an avoidance group.
            return [user for user in available if user.department == PROTECTED_GROUP]

        if prize.associated_drawer == SPECIAL_DRAWER:
            # Special rule: only previous winners are filtered.
            return available

        avoided = set(prize.avoid_groups)
        return [user for user in available if user.department not in avoided]

    def draw_winner(self, prize: Prize) -> User | None:
        """Randomly draw one eligible winner for the prize, or None if none is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None

        winner = self._rng.choice(pool)
        self._winner_names.add(winner.name)

        if prize.associated_drawer in TRIGGER_DRAWERS:
            self.last_two_draw_results.append(winner.department == PROTECTED_GROUP)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

        return winner

    @staticmethod
    def describe_rules(prize: Prize) -> str:
        """Build a human-readable description of the prize's lottery rules."""
        parts: list[str] = []
        drawer = prize.associated_drawer or "none"
        parts.append(f"Lottery person: {drawer}")

        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("Special rule: only previous winners are excluded")
        else:
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            parts.append(f"Avoid groups: {groups}")

        if prize.associated_drawer == FALLBACK_DRAWER:
            parts.append(
                "Fallback: if the last two draws by Chen Qixian and Jin Duo contain "
                f"no {PROTECTED_GROUP} members, the pool is limited to {PROTECTED_GROUP} members"
            )

        parts.append("Previous winners are always excluded")
        return "; ".join(parts)

"""Winner recording, ordering, and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, prize_level


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    winning_order: int
    winning_time: datetime


class ResultManager:
    """Stores winners and produces ordered results and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for the given prize."""
        winner = Winner(
            user=user,
            prize_name=prize.name,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """Return all winners in the order they were drawn."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, each list in winning time order."""
        return {
            prize.name: [w for w in self._winners if w.prize_name == prize.name]
            for prize in prizes
        }

    def get_sorted_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Return winners sorted by prize level (high to low), then prize sequence,
        then winning time order."""
        prize_index = {prize.name: idx for idx, prize in enumerate(prizes)}
        fallback_index = len(prize_index)

        def sort_key(winner: Winner) -> tuple[int, int, int]:
            return (
                prize_level(winner.prize_name),
                prize_index.get(winner.prize_name, fallback_index),
                winner.winning_order,
            )

        return sorted(self._winners, key=sort_key)

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to awards_result_YYYYMMDD.txt in CSV format."""
        file_name = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in self.get_sorted_winners(prizes):
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from __future__ import annotations

from typing import Callable

from lottery_system.models import Prize, User

def ordinal(number: int) -> str:
    """Return the English ordinal form of an integer, e.g. 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive command-line application."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery_manager: LotteryManager | None = None

    # ----- helpers -------------------------------------------------------

    def _ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    def _set_users(self, users: list[User]) -> None:
        self.users = list(users)
        self.result_manager = ResultManager()
        self.lottery_manager = None

    # ----- main loop -----------------------------------------------------

    def run(self) -> None:
        self._print("=" * 50)
        self._print("  Team Building Smart Lottery System")
        self._print("=" * 50)
        while True:
            self._print("\n--- Main Menu ---")
            self._print("1. Load participant list")
            self._print("2. View department groups")
            self._print("3. Prize configuration")
            self._print("4. Start lottery")
            self._print("5. View winners")
            self._print("6. Export winner list")
            self._print("0. Exit")
            choice = self._ask("Select an option: ")

            if choice == "1":
                self._load_list_menu()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self._prize_menu()
            elif choice == "4":
                self._start_lottery()
            elif choice == "5":
                self._show_winners()
            elif choice == "6":
                self._export()
            elif choice == "0":
                self._print("Goodbye!")
                return
            else:
                self._print("Invalid option, please try again.")

    # ----- participant list ----------------------------------------------

    def _load_list_menu(self) -> None:
        self._print("\n--- Load Participant List ---")
        self._print("1. Built-in default list")
        self._print("2. Custom txt file (format: Name,Department Group per line)")
        self._print("0. Cancel")
        choice = self._ask("Select an option: ")

        if choice == "1":
            users, _errors = self.list_manager.load_default_list()
            valid, _dups = self.list_manager.validate_users(users)
            self._set_users(valid)
            self._print(f"Loaded {len(valid)} participants from the built-in list.")
        elif choice == "2":
            path = self._ask("Enter txt file path: ")
            valid = self._handle_custom_load(path)
            if valid is None:
                return
            if not valid:
                self._print("No valid participants loaded.")
                return
            self._set_users(valid)
            self._print(f"Loaded {len(valid)} participants from '{path}'.")
        else:
            self._print("Cancelled.")

    def _handle_custom_load(self, path: str) -> list[User] | None:
        users, errors = self.list_manager.load_custom_list(path)
        valid, duplicates = self.list_manager.validate_users(users)

        if not errors and not duplicates:
            return valid

        self._print("\nIssues detected in the uploaded list:")
        for error in errors:
            self._print(f"  - {error}")
        for dup in duplicates:
            self._print(
                f"  - Duplicate Name: '{dup.name}' ({dup.department}) appears more "
                "than once; the later entry is ignored"
            )

        choice = self._ask(
            "Enter 'r' to remove invalid/duplicate entries and continue, "
            "'m' to manually remove names, 't' to terminate upload: "
        ).lower()

        if choice == "t":
            self._print("Upload terminated.")
            return None
        if choice == "m":
            names = self._ask("Enter names to remove (comma separated): ")
            to_remove = {name.strip() for name in names.split(",") if name.strip()}
            valid = [user for user in valid if user.name not in to_remove]
            return valid
        if choice == "r":
            return valid

        self._print("Invalid choice. Upload terminated.")
        return None

    def _show_groups(self) -> None:
        if not self.users:
            self._print("No participant list loaded. Please load a list first.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._print("\n--- Department Groups ---")

    # ----- prize configuration -------------------------------------------

    def _prize_menu(self) -> None:
        while True:
            self._print("\n--- Prize Configuration ---")
            self._print("1. Load preset prizes")
            self._print("2. Add custom prize")
            self._print("3. Set avoidance groups / lottery person for a prize")
            self._print("4. Reorder prizes")
            self._print("5. View prizes")
            self._print("0. Back")
            choice = self._ask("Select an option: ")

            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                for error in errors:
                    self._print(f"Error: {error}")
                self._print("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._set_prize_rules()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                self._view_prizes()
            elif choice == "0":

    def _select_groups(self) -> list[str]:
        if not self.users:
            self._print("(No participant list loaded; no avoidance groups available.)")
            return []
        groups = list(self.list_manager.get_group_info(self.users).keys())
        self._print("Available department groups:")
        for idx, group in enumerate(groups, start=1):
            self._print(f"  {idx}. {group}")
        text = self._ask(
            "Select avoidance groups by number, comma separated (blank for none): "
        )
        selected: list[str] = []
        for token in text.split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                group = groups[int(token) - 1]
                if group not in selected:
                    selected.append(group)
            else:
                self._print(f"Ignoring invalid group selection: '{token}'")
        return selected

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        if not name:
            self._print("Prize name cannot be empty.")
            return
        quantity_text = self._ask("Quantity: ")
        if not quantity_text.isdigit() or int(quantity_text) < 1:
            self._print("Quantity must be a positive integer.")
            return
        groups = self._select_groups()
        drawer = self._ask("Associated lottery person (leave blank for none): ") or None
        try:
            self.prize_manager.add_prize(name, int(quantity_text), groups, drawer)
            self._print(f"Prize '{name}' added.")
        except ValueError as exc:
            self._print(f"Failed to add prize: {exc}")

    def _set_prize_rules(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._view_prizes()
        text = self._ask("Select prize number to configure: ")
        if not text.isdigit() or not 1 <= int(text) <= len(prizes):
            self._print("Invalid prize number.")
            return
        index = int(text) - 1
        groups = self._select_groups()
        drawer = self._ask("Associated lottery person (leave blank for none): ") or None
        if self.prize_manager.update_prize(index, groups, drawer):
            self._print(f"Rules updated for '{prizes[index].name}'.")
        else:
            self._print("Failed to update prize.")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._view_prizes()
        text = self._ask(
            "Enter new order as serial numbers separated by spaces (e.g. '2 1 3'): "
        )
        try:
            order = [int(token) for token in text.split()]
        except ValueError:
            self._print("Invalid input: please enter numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Prize order updated.")
            self._view_prizes()
        else:
            self._print(
                f"Invalid order: each number from 1 to {len(prizes)} must appear exactly once."
            )

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._print("\n--- Prize Sequence ---")
        for idx, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self._print(
                f"  {idx}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoid groups: {groups} | Lottery person: {drawer}"
            )

    # ----- lottery process -----------------------------------------------

    def _start_lottery(self) -> None:
        if not self.users:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured. Please configure prizes first.")
            return
        if self.lottery_manager is None:
            self.lottery_manager = LotteryManager(self.users)

        total = len(prizes)
        self._print("\n##### Lottery started #####")
        for idx, prize in enumerate(prizes, start=1):
            self._run_round(idx, total, prize)
        self._print("\n##### Lottery finished #####")
        self._show_winners()

    def _run_round(self, round_number: int, total: int, prize: Prize) -> None:
        assert self.lottery_manager is not None
        self._print(f"\n===== Round {round_number}/{total}: {prize.name} =====")
        self._print(f"Quantity: {prize.quantity}")
        self._print(f"Rules: {LotteryManager.describe_rules(prize)}")

        drawn: list[User] = []
        terminated = False

        while len(drawn) < prize.quantity and not terminated:
            command = self._ask(
                "Press Enter to draw the next winner, or type 'pause' / 'terminate': "
            ).lower()

            if command == "pause":
                self._print("Lottery paused.")
                while True:
                    resume = self._ask(
                        "Type 'continue' to resume or 'terminate' to end this round: "
                    ).lower()
                    if resume == "continue":
                        self._print("Lottery resumed.")
                        break
                    if resume == "terminate":
                        terminated = True
                        break
                    self._print("Unknown command.")
                continue

            if command == "terminate":
                terminated = True
                break

            if command != "":
                self._print("Unknown command.")
                continue

            self._print(
                f"Currently drawing the {ordinal(len(drawn) + 1)} winner "
                f"for {prize.name}..."
            )
            user = self.lottery_manager.draw_winner(prize)
            if user is None:
                self._print("No eligible participants remain for this prize. Round ends.")
                break

            self.result_manager.add_winner(user, prize)
            drawn.append(user)
            self._print(
                f"Congratulations {user.name} ({user.department}) for winning {prize.name}"
            )

        if terminated:
            skipped = prize.quantity - len(drawn)
            self._print(f"Round terminated. {skipped} prize(s) skipped.")

        self._print(f"Round results for {prize.name}:")
        if not drawn:
            self._print("  (no winners in this round)")
        for position, user in enumerate(drawn, start=1):
            self._print(f"  {position}. {user.name} ({user.department})")

    def _show_winners(self) -> None:
        prizes = self.prize_manager.get_prizes()
        winners = self.result_manager.get_sorted_winners(prizes)
        if not winners:
            self._print("No winners yet.")
            return
        self._print("\n===== Complete Winner List =====")
        current_prize: str | None = None
        for winner in winners:
            if winner.prize_name != current_prize:
                current_prize = winner.prize_name
                self._print(f"\n{current_prize}:")
            self._print(
                f"  [{winner.winning_order}] {winner.user.name} "
                f"({winner.user.department}) - {winner.prize_name}"
            )

    def _export(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not self.result_manager.get_all_winners():
            self._print("No winners to export yet.")
            return
        _success, message = self.result_manager.export_to_csv(prizes)
        self._print(message)


def main() -> int:
    """Entry point for the command-line application."""
    cli = LotterySystemCLI()
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting the lottery system.")
    return 0

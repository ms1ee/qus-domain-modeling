"""Domain models used across the lottery system."""

from __future__ import annotations

import re

# Level words recognised in prize names; lower number means higher prize level.
_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3, "fourth": 4, "fifth": 5}

# Level assigned to prizes without a recognised level word (sorted last).
DEFAULT_LEVEL = 999


def infer_prize_level(name: str) -> int:
    """Infer the prize level from a prize name such as 'First Prize - ...'."""
    lowered = name.lower()
    for word, level in _LEVEL_WORDS.items():
        if re.search(r"\b" + word + r"\b", lowered):
            return level
    return DEFAULT_LEVEL


class User:
    """A participant in the team building list."""

    def __init__(self, name: str, department: str) -> None:
        self.name = name
        self.department = department

    def __repr__(self) -> str:
        return f"User(name={self.name!r}, department={self.department!r})"


class Prize:
    """A prize with its lottery rules.

    avoid_groups: list of department names whose members may not win this prize.
    associated_drawer: name of the lottery person handling this prize, or None.
    """

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
        is_custom: bool = False,
        level: int | None = None,
        sequence_number: int = 0,
    ) -> None:
        self.name = name
        self.quantity = quantity
        self.avoid_groups: list[str] = list(avoid_groups) if avoid_groups else []
        self.associated_drawer = associated_drawer
        self.is_custom = is_custom
        self.level = level if level is not None else infer_prize_level(name)
        self.sequence_number = sequence_number

    def __repr__(self) -> str:
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
            f"avoid_groups={self.avoid_groups!r}, "
            f"associated_drawer={self.associated_drawer!r})"
        )

"""Participant list loading, validation and department group extraction."""

from __future__ import annotations

from lottery_system.config import DEFAULT_PARTICIPANTS_TEXT
from lottery_system.models import User

FORMAT_ERROR_PREFIX = "Format Error"
_ENCODINGS = ("utf-8-sig", "gbk")


def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
    """Parse 'Name,Department' lines into users and collect format errors."""
    users: list[User] = []
    errors: list[str] = []
    for line_no, raw in enumerate(lines, start=1):
        text = raw.strip().lstrip("\ufeff").strip()
        if not text:
            continue
        parts = [part.strip() for part in text.split(",")]
        if len(parts) != 2 or not parts[0] or not parts[1]:
            errors.append(
                f"{FORMAT_ERROR_PREFIX} at Line {line_no}: '{text}' "
                f"(expected format 'Name,Department')"
            )
            continue
        users.append(User(parts[0], parts[1]))
    return users, errors


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return _parse_lines(DEFAULT_PARTICIPANTS_TEXT.splitlines())

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list from the given path."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a line-based txt file ('Name,Department' per line)."""
        try:
            with open(path, "rb") as handle:
                raw = handle.read()
        except OSError as exc:
            reason = exc.strerror or str(exc)
            return [], [f"File Error: cannot read '{path}': {reason}"]

        text = None
        for encoding in _ENCODINGS:
            try:
                text = raw.decode(encoding)
                break
            except UnicodeDecodeError:
                continue
        if text is None:
            return [], ["File Error: unsupported file encoding (use UTF-8 or GBK)"]

        return _parse_lines(text.splitlines())

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique entries and duplicate-name entries.

        The first occurrence of each name is kept as valid; later occurrences
        are returned as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of members."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management: presets, custom prizes, rules and ordering."""

from __future__ import annotations

from lottery_system.config import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, [], drawer, is_custom=False)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
        is_custom: bool = True,
    ) -> Prize:
        """Append a prize to the end of the lottery sequence."""
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(prize.name == name for prize in self._prizes):
            raise ValueError(f"Prize '{name}' already exists.")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        groups = list(dict.fromkeys(groups))
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None

        prize = Prize(
            name,
            quantity,
            groups,
            drawer,
            is_custom=is_custom,
            sequence_number=len(self._prizes) + 1,
        )
        self._prizes.append(prize)
        return prize

    def update_rules(
        self,
        name: str,
        avoid_groups: list[str] | None,
        associated_drawer: str | None,
    ) -> bool:
        """Update avoidance groups and lottery person of an existing prize."""
        for prize in self._prizes:
            if prize.name == name:
                prize.avoid_groups = list(dict.fromkeys(avoid_groups or []))
                drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
                prize.associated_drawer = drawer
                return True
        return False

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        order holds 1-based current positions in the new order, e.g. [2, 1, 3]
        means the prize at position 2 becomes first. Must be a permutation of
        all current positions; otherwise returns False and changes nothing.
        """
        count = len(self._prizes)
        try:
            positions = [int(value) for value in order]
        except (TypeError, ValueError):
            return False
        if sorted(positions) != list(range(1, count + 1)):
            return False

        self._prizes = [self._prizes[position - 1] for position in positions]
        for index, prize in enumerate(self._prizes, start=1):
            prize.sequence_number = index
        return True

"""Lottery rule engine: filtering, special rules, fallback and anti-duplication."""

from __future__ import annotations

import random

from lottery_system.config import (
    CHENG_RU_GROUP,
    FALLBACK_DRAWER,
    FALLBACK_TRIGGER_DRAWERS,
    LI_BIN,
)
from lottery_system.models import Prize, User


class LotteryManager:
    """Performs individual draws while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # Whether each recent draw by Chen Qixian / Jin Duo contained a Cheng Ru
        # group member. Only the last two results are kept.
        self.last_two_draw_results: list[bool] = []

    @property
    def winner_names(self) -> frozenset[str]:
        """Names of everyone who has won so far (anti-duplication record)."""
        return frozenset(self._winner_names)

    def is_fallback_active(self) -> bool:
        """True when neither of the last two trigger draws hit the Cheng Ru group."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the candidates eligible for the given prize right now."""
        # Anti-duplication: previous winners are always excluded.
        available = [user for user in self._users if user.name not in self._winner_names]

        drawer = prize.associated_drawer

        # Special rule: Li Bin gets relaxed filtering (avoidance groups ignored).
        if drawer == LI_BIN:
            return available

        # Fallback: Cheng Ru group protection overrides avoidance groups.
        if drawer == FALLBACK_DRAWER and self.is_fallback_active():
            protected = [user for user in available if user.department == CHENG_RU_GROUP]
            if protected:
                return protected

        # Basic filtering: exclude members of the prize's avoidance groups.
        return [user for user in available if user.department not in prize.avoid_groups]

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no candidate is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None

        winner = random.choice(pool)
        self._winner_names.add(winner.name)

        if prize.associated_drawer in FALLBACK_TRIGGER_DRAWERS:
            self.last_two_draw_results.append(winner.department == CHENG_RU_GROUP)
            del self.last_two_draw_results[:-2]

        return winner

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        rules = ["Excludes previous winners"]
        drawer = prize.associated_drawer

        if drawer == LI_BIN:
            rules.append("Relaxed filtering for Li Bin (avoidance groups not applied)")
        elif prize.avoid_groups:
            rules.append("Avoids groups: " + ", ".join(prize.avoid_groups))

        if drawer == FALLBACK_DRAWER and self.is_fallback_active():
            rules.append(f"Fallback protection active: only {CHENG_RU_GROUP} members eligible")

        if drawer:
            rules.append(f"Lottery person: {drawer}")
        else:
            rules.append("No associated lottery person")
        return "; ".join(rules)

"""Winner recording, grouped summaries and CSV-style export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User

EXPORT_HEADER = ["Prize Level", "Prize Name", "Winning Order", "Name", "Department"]


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize: Prize
    winning_order: int
    won_at: datetime = field(default_factory=datetime.now)


class ResultManager:
    """Stores winners in winning order and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    @property
    def winners(self) -> list[Winner]:
        """All winners in the order they were drawn."""
        return list(self._winners)

    def clear(self) -> None:
        """Remove all recorded winners."""
        self._winners.clear()

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for a prize. Winning order is global and 1-based."""
        winner = Winner(user=user, prize=prize, winning_order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name.

        Keys are ordered by prize level from high to low (level 1 first), with
        ties broken by the configured prize order. Within each prize, winners
        are in winning time order.
        """
        ordered = sorted(enumerate(prizes), key=lambda item: (item[1].level, item[0]))
        grouped: dict[str, list[Winner]] = {}
        for _, prize in ordered:
            grouped[prize.name] = [w for w in self._winners if w.prize.name == prize.name]
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to awards_result_YYYYMMDD.txt in the current directory."""
        if not self._winners:
            return False, "No winners to export."

        file_name = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), file_name)
        grouped = self.get_winners_by_prize(prizes)

        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(EXPORT_HEADER)
                for winners in grouped.values():
                    for winner in winners:
                        writer.writerow([
                            winner.prize.level,
                            winner.prize.name,
                            winner.winning_order,
                            winner.user.name,
                            winner.user.department,
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"

        return True, f"Winner list successfully exported to {path}"

from __future__ import annotations

from typing import Callable

from lottery_system.models import Prize, User
from lottery_system.services.list_manager import ListManager

def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer, e.g. 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    """Interactive command-line controller."""

    def __init__(self, input_func: Callable[[str], str] = input) -> None:
        self._input = input_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    @staticmethod
    def _print(message: str = "") -> None:
        print(message)

    # --------------------------------------------------------------- main loop
    def run(self) -> None:
        self._print("Welcome to the Team Building Smart Lottery System")
        while True:
            self._print(MAIN_MENU)
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self._load_default_list()
            elif choice == "2":
                self._upload_custom_list()
            elif choice == "3":
                self._show_groups()
            elif choice == "4":
                self._prize_menu()
            elif choice == "5":
                self._start_lottery()
            elif choice == "6":
                self._print_summary()
            elif choice == "7":
                self._export()
            elif choice == "0":
                self._print("Goodbye!")
                return
            else:
                self._print("Invalid option, please try again.")

    # ------------------------------------------------------------ list module
    def _load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        if self._review_and_apply(users, errors):
            self._print(f"Built-in list loaded: {len(self.users)} participants.")

    def _upload_custom_list(self) -> None:
        path = self._ask("Enter the path of the txt file: ").strip().strip('"').strip("'")
        if not path:
            self._print("No path entered. Upload cancelled.")
            return
        users, errors = self.list_manager.load_custom_list(path)
        if self._review_and_apply(users, errors):
            self._print(f"Custom list loaded: {len(self.users)} participants.")

    def _review_and_apply(self, users: list[User], errors: list[str]) -> bool:
        """Validate loaded users; ask the operator how to handle any issues."""
        valid, duplicates = self.list_manager.validate_users(users)

        if not errors and not duplicates:
            self.users = valid
            return True

        self._print("Validation found the following issues:")
        for error in errors:
            self._print(f"  - {error}")
        for dup in duplicates:
            self._print(
                f"  - Duplicate Name: '{dup.name}' ({dup.department}) appears more than once; "
                f"first occurrence is kept"
            )
        self._print(f"Valid entries available: {len(valid)}")

        while True:
            choice = self._ask(
                "Enter R to remove invalid data and keep valid entries, "
                "or T to terminate the upload: "
            ).strip().lower()
            if choice == "r":
                self.users = valid
                self._print(f"Invalid data removed. Keeping {len(valid)} valid entries.")
                return True
            if choice == "t":
                self._print("Upload terminated. The previous participant list is unchanged.")
                return False
            self._print("Please enter R or T.")

    def _show_groups(self) -> None:
        if not self.users:
            self._print("No participant list loaded yet.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._print("Department groups:")
        for index, (group, count) in enumerate(info.items(), start=1):
            self._print(f"  {index}. {group}: {count} people")
        self._print(f"Total: {len(self.users)} participants in {len(info)} groups.")

    # ----------------------------------------------------------- prize module
    def _prize_menu(self) -> None:
        while True:
            self._print(PRIZE_MENU)
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self._show_prizes()
            elif choice == "2":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        self._print(f"  - {error}")
                else:
                    self._print("Preset prizes loaded.")
            elif choice == "3":
                self._add_custom_prize()
            elif choice == "4":
                self._edit_prize_rules()
            elif choice == "5":
                self._reorder_prizes()
            elif choice == "0":
                return
            else:
                self._print("Invalid option, please try again.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured yet.")
            return
        self._print("Current lottery sequence:")
        for index, prize in enumerate(prizes, start=1):
            self._print(f"  {index}. {self._format_prize(prize)}")

    @staticmethod
    def _format_prize(prize: Prize) -> str:
        groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "-"
        drawer = prize.associated_drawer or "-"
        return f"{prize.name} | quantity: {prize.quantity} | lottery person: {drawer} | avoid: {groups}"

    def _select_groups(self) -> list[str]:
        known: list[str] = []
        if self.users:
            known = list(self.list_manager.get_group_info(self.users).keys())
            self._print("Available groups:")
            for index, group in enumerate(known, start=1):
                self._print(f"  {index}. {group}")

        raw = self._ask(
            "Enter avoidance groups (numbers or names, comma-separated; blank for none): "
        ).strip()
        if not raw:
            return []

        selected: list[str] = []
        for token in (part.strip() for part in raw.split(",")):
            if not token:
                continue
            if token.isdigit() and known and 1 <= int(token) <= len(known):
                name = known[int(token) - 1]
            else:
                name = token
            if name not in selected:
                selected.append(name)
        return selected

    def _ask_drawer(self) -> str | None:
        raw = self._ask("Associated lottery person (blank for none): ").strip()
        return raw or None

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        if not name:
            self._print("Prize name cannot be empty.")
            return
        quantity_text = self._ask("Quantity: ").strip()
        try:
            quantity = int(quantity_text)
            if quantity < 1:
                raise ValueError
        except ValueError:
            self._print("Quantity must be a positive integer.")
            return

        groups = self._select_groups()
        drawer = self._ask_drawer()
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer, is_custom=True)
        except ValueError as exc:
            self._print(str(exc))
            return
        self._print(f"Prize added: {self._format_prize(prize)}")

    def _edit_prize_rules(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured yet.")
            return
        self._show_prizes()
        raw = self._ask("Enter the prize serial number to edit: ").strip()
        if not raw.isdigit() or not (1 <= int(raw) <= len(prizes)):
            self._print("Invalid serial number.")
            return
        prize = prizes[int(raw) - 1]
        groups = self._select_groups()
        drawer = self._ask_drawer()
        self.prize_manager.update_rules(prize.name, groups, drawer)
        self._print(f"Updated: {self._format_prize(prize)}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            self._print("At least two prizes are needed to reorder.")
            return
        self._show_prizes()
        raw = self._ask("Enter the new round order as serial numbers (e.g. 2 1 3): ")
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(token) for token in tokens]
        except ValueError:
            self._print("Invalid input: please enter integers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Lottery sequence updated.")
            self._show_prizes()
        else:
            self._print("Invalid order: every serial number must appear exactly once.")

    # -------------------------------------------------------- lottery control
    def _start_lottery(self) -> None:
        if not self.users:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure at least one prize first.")
            return

        if self.result_manager.winners:
            confirm = self._ask("Previous results will be cleared. Continue? (y/n): ").strip().lower()
            if confirm != "y":
                self._print("Lottery cancelled.")
                return
            self.result_manager.clear()

        lottery = LotteryManager(self.users)
        total_rounds = len(prizes)
        self._print(f"\nLottery starts: {total_rounds} round(s).")
        self._print("Commands during a round: Enter = draw next, 'pause', 'continue', 'terminate'.")

        for round_no, prize in enumerate(prizes, start=1):
            round_winners = self._run_round(lottery, prize, round_no, total_rounds)
            self._print(f"\n--- Round {round_no} results ({prize.name}) ---")
            if round_winners:
                for winner in round_winners:
                    self._print(
                        f"  #{winner.winning_order} {winner.user.name} ({winner.user.department})"
                    )
            else:
                self._print("  No winners in this round.")

        self._print("\nAll rounds completed.")
        self._print_summary()

    def _run_round(
        self,
        lottery: LotteryManager,
        prize: Prize,
        round_no: int,
        total_rounds: int,
    ) -> list[Winner]:
        self._print(
            f"\n=== Round {round_no}/{total_rounds}: {prize.name} | "
            f"Quantity: {prize.quantity} ==="
        )
        self._print(f"Rule: {lottery.describe_rules(prize)}")

        round_winners: list[Winner] = []
        drawn = 0
        paused = False

        while drawn < prize.quantity:
            if paused:
                prompt = "[PAUSED] Type 'continue' to resume or 'terminate' to end this round: "
            else:
                prompt = (
                    f"Press Enter to draw winner #{drawn + 1}, "
                    f"or type 'pause' / 'terminate': "
                )
            command = self._ask(prompt).strip().lower()

            if paused:
                if command == "continue":
                    paused = False
                    self._print("Lottery resumed.")
                elif command == "terminate":
                    self._print("Round terminated manually. Skipping remaining winners of this prize.")
                    break
                else:
                    self._print("Lottery is paused. Type 'continue' or 'terminate'.")
                continue

            if command == "pause":
                paused = True
                self._print("Lottery paused.")
                continue
            if command == "terminate":
                self._print("Round terminated manually. Skipping remaining winners of this prize.")
                break
            if command != "":
                self._print("Unknown command. Press Enter to draw, or type 'pause' / 'terminate'.")
                continue

            self._print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = lottery.draw(prize)
            if user is None:
                self._print("No eligible candidates remain for this prize. Round ends early.")
                break

            winner = self.result_manager.add_winner(user, prize)
            round_winners.append(winner)
            drawn += 1
            self._print(
                f"Congratulations {user.name} ({user.department}) for winning {prize.name}!"
            )

        return round_winners

    # --------------------------------------------------------- results module
    def _print_summary(self) -> None:
        if not self.result_manager.winners:
            self._print("No winners recorded yet.")
            return
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self._print("\n==================== Winner List ====================")
        for prize_name, winners in grouped.items():
            self._print(f"\n[{prize_name}]")
            if not winners:
                self._print("  (no winners drawn)")
                continue
            for winner in winners:
                self._print(
                    f"  #{winner.winning_order} {winner.user.name} "
                    f"({winner.user.department}) - {prize_name}"
                )

    def _export(self) -> None:
        if not self.result_manager.winners:
            self._print("No winners to export. Run the lottery first.")
            return
        _, message = self._export_result()
        self._print(message)

    def _export_result(self) -> tuple[bool, str]:
        return self.result_manager.export_to_csv(self.prize_manager.get_prizes())


def main() -> None:
    """Start the interactive lottery system."""
    cli = LotteryCLI()
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting the lottery system.")


if __name__ == "__main__":
    main()

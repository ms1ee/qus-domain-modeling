class ListManager:
    """Loads, validates and summarizes participant lists."""

    def __init__(self) -> None:
        self.load_mode: str | None = None
        self.file_path: str | None = None
        self.validation_status: str = "not_loaded"

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        self.load_mode = "default"
        self.file_path = None
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.validation_status = "invalid" if errors else "loaded"
        return users, errors

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        self.load_mode = "custom"
        self.file_path = path
        users, errors = self._load_from_path(path)
        self.validation_status = "invalid" if errors else "loaded"
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it into users and format error messages."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw in enumerate(lines, start=1):
            content = raw.strip()
            if not content:
                continue
            parsed = self._parse_line(content)
            if parsed is None:
                errors.append(
                    f"Line {line_number}: Format Error - '{content}' "
                    f"(expected 'Name,Department Group')"
                )
            else:
                users.append(User(name=parsed[0], department=parsed[1]))
        return users, errors

    @staticmethod
    def _parse_line(content: str) -> tuple[str, str] | None:
        parts = _SEPARATOR_PATTERN.split(content)
        if len(parts) != 2:
            return None
        name, department = parts[0].strip(), parts[1].strip()
        if not name or not department:
            return None
        return name, department

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique entries and duplicate entries.

        The first occurrence of a name is valid; later occurrences are duplicates.
        """
        valid: list[User] = []
        duplicates: list[User] = []
        seen: set[str] = set()
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

from __future__ import annotations

from lottery_system.models import Prize

PRESET_PRIZES: list[tuple[str, int, str]] = [
    ("Third Prize - Movie Tickets", 5, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
]

_LEVEL_WORDS: dict[str, int] = {"first": 1, "second": 2, "third": 3}
_CUSTOM_LEVEL_BASE = 1000


def _infer_level(name: str, index: int) -> int:
    """Preset prizes get their rank from the name; custom prizes rank after them."""
    words = name.split()
    first_word = words[0].casefold() if words else ""
    return _LEVEL_WORDS.get(first_word, _CUSTOM_LEVEL_BASE + index)


class PrizeManager:
    """Manages the prize library and the lottery sequence."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages (empty on success)."""
        errors: list[str] = []
        for name, quantity, drawer in PRESET_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize already exists: {name}")
                continue
            try:
                self.add_prize(name, quantity, [], drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize at the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name cannot be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("quantity must be a positive integer")
        if self._find(clean_name) is not None:
            raise ValueError(f"prize '{clean_name}' already exists")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        prize.level = _infer_level(clean_name, len(self._prizes))
        prize.sequence_number = len(self._prizes) + 1
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        count = len(self._prizes)
        if not isinstance(order, list) or len(order) != count:
            return False
        if any(isinstance(item, bool) or not isinstance(item, int) for item in order):
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        for index, prize in enumerate(self._prizes, start=1):
            prize.sequence_number = index
        return True

    def _find(self, name: str) -> Prize | None:
        target = name.strip().casefold()
        for prize in self._prizes:
            if prize.name.casefold() == target:
                return prize
        return None

"""Lottery rule engine: eligibility filtering, special rules and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import Prize, User

CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
MA_JIAQI = "Ma Jiaqi"
PROTECTED_DRAWERS = frozenset({"Chen Qixian", "Jin Duo"})


def _is_cheng_ru(user: User) -> bool:
    return user.department.casefold() == CHENG_RU_GROUP.casefold()


class LotteryManager:
    """Draws winners while enforcing the lottery rules.

    Rules:
    - Previous winners are never drawn again.
    - Members of a prize's avoidance groups are drawn only after all other
      eligible members (they remain eligible).
    - Lottery person 'Li Bin' only filters previous winners; avoidance groups
      are not applied.
    - Fallback: if neither of the last two draws made by Chen Qixian / Jin Duo
      produced a Cheng Ru Group member, Ma Jiaqi's pool is limited to Cheng Ru
      Group members (even if that group is an avoidance group).
    """

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        self.winners: list[User] = []
        self.last_two_draw_results: list[bool] = []

    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def is_cheng_ru_fallback_active(self) -> bool:
        """True when the last two protected draws both produced no Cheng Ru member."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users the next draw for this prize should be taken from."""
        candidates = [user for user in self._users if user.name not in self._winner_names]

        if prize.associated_drawer == MA_JIAQI and self.is_cheng_ru_fallback_active():
            protected = [user for user in candidates if _is_cheng_ru(user)]
            if protected:
                return protected

        if prize.associated_drawer == LI_BIN:
            return candidates

        avoid = {group.casefold() for group in prize.avoid_groups}
        non_avoid = [user for user in candidates if user.department.casefold() not in avoid]
        avoid_members = [user for user in candidates if user.department.casefold() in avoid]
        return non_avoid if non_avoid else avoid_members

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None when no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None

        winner = random.choice(pool)
        self._winner_names.add(winner.name)
        self.winners.append(winner)

        if prize.associated_drawer in PROTECTED_DRAWERS:
            self.last_two_draw_results.append(_is_cheng_ru(winner))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return winner

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        parts = ["previous winners are excluded"]
        if prize.associated_drawer == LI_BIN:
            parts.append("special rule for Li Bin: only previous winners are filtered, avoidance groups not applied")
        else:
            if prize.avoid_groups:
                parts.append(
                    "avoidance groups (" + ", ".join(prize.avoid_groups)
                    + ") are drawn only after other members"
                )
        if prize.associated_drawer == MA_JIAQI:
            parts.append(
                "fallback: if neither of the last two Chen Qixian/Jin Duo draws "
                "produced a Cheng Ru Group member, the pool is limited to Cheng Ru Group members"
            )
        return "; ".join(parts)

"""Winner recording, grouping, and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record: who won which prize, when, and in which order."""

    user: User
    prize: Prize
    winning_time: datetime
    winning_order: int


class ResultManager:
    """Stores winners and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner. winning_order is the position within the same prize."""
        order = sum(1 for winner in self._winners if winner.prize.name == prize.name) + 1
        winner = Winner(user=user, prize=prize, winning_time=datetime.now(), winning_order=order)
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name (keys follow the given prize order)."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize.name in result:
                result[winner.prize.name].append(winner)
        for winners in result.values():
            winners.sort(key=lambda item: item.winning_order)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export winners to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        if not self._winners:
            return False, "No winners to export."

        now = datetime.now()
        path = os.path.join(os.getcwd(), f"awards_result_{now:%Y%m%d}.txt")
        winners_by_prize = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize in sorted(prizes, key=lambda item: item.level):
                    for winner in winners_by_prize.get(prize.name, []):
                        writer.writerow([
                            winner.user.name,
                            winner.user.department,
                            winner.prize.name,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer (e.g. 1st, 12th)."""
    if 11 <= number % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryApp:
    """Interactive console application."""

    def __init__(self) -> None:
        self._list_manager = ListManager()
        self._prize_manager = PrizeManager()
        self._result_manager = ResultManager()
        self._users: list[User] = []

    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def run(self) -> None:
        print(SEPARATOR)
        print("  Company Team Building - Smart Lottery System")
        print(SEPARATOR)
        actions = {
            "1": self._load_participants,
            "2": self._show_groups,
            "3": self._configure_prizes,
            "4": self._start_lottery,
            "5": self._show_results,
        }
        while True:
            print("\nMain Menu")
            print("1. Load participant list")
            print("2. View department groups")
            print("3. Configure prizes")
            print("4. Start lottery")
            print("5. View / export results")
            print("0. Exit")
            choice = self._ask("Select an option: ")
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid option, please try again.")
            else:
                action()

    # ------------------------------------------------------------------ list

    def _load_participants(self) -> None:
        print("\nParticipant list source:")
        print("1. Built-in default team building list")
        print("2. Custom txt file (each line: Name,Department Group)")
        print("0. Back")
        choice = self._ask("Select an option: ")
        if choice == "1":
            users, errors = self._list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter path to txt file: ")
            users, errors = self._list_manager.load_custom_list(path)
        else:
            return

        accepted = self._verify_participants(users, errors)
        if accepted is None:
            print("Participant list was not loaded.")
            return
        self._users = accepted
        self._result_manager = ResultManager()
        print(f"Loaded {len(accepted)} participants. Previous results (if any) were cleared.")
        self._show_groups()

    def _verify_participants(self, users: list[User], errors: list[str]) -> list[User] | None:
        if errors:
            print("\nFormat errors found:")
            for error in errors:
                print(f"  - {error}")

        valid, duplicates = self._list_manager.validate_users(users)
        if duplicates:
            print("\nDuplicate names found (later occurrences are invalid):")
            for user in duplicates:
                print(f"  - {user.name} ({user.department})")

        if not errors and not duplicates:
            return valid

        print(f"\nValid records: {len(valid)}")
        while True:
            decision = self._ask(
                "Type 'remove' to discard invalid data and continue, "
                "or 'terminate' to abort the upload: "
            ).lower()
            if decision == "remove":
                if not valid:
                    print("No valid participants remain; upload aborted.")
                    return None
                print(f"Removed {len(errors) + len(duplicates)} invalid record(s).")
                return valid
            if decision == "terminate":
                print("Upload terminated.")
                return None
            print("Please type 'remove' or 'terminate'.")

    def _show_groups(self) -> None:
        if not self._users:
            print("No participant list loaded.")
            return
        info = self._list_manager.get_group_info(self._users)
        print("\nDepartment groups:")
        for department, count in info.items():
            print(f"  {department}: {count} people")
        print(f"Total: {len(self._users)} people")

    # ---------------------------------------------------------------- prizes

    def _configure_prizes(self) -> None:

    def _view_prizes(self) -> None:

        if not prizes:
            print("No prizes configured.")

        print("\nLottery sequence:")
        for index, prize in enumerate(prizes, start=1):

            print("No prizes configured.")
            return
        self._view_prizes()
        text = self._ask("Enter new order as serial numbers separated by spaces (e.g. 2 1 3): ")
        try:
            order = [int(item) for item in text.split()]
        except ValueError:
            print("Invalid input: please enter integers only.")
            return

            print("Lottery sequence updated.")
            self._view_prizes()
        else:

    # ---------------------------------------------------------------- lottery

    def _start_lottery(self) -> None:
        if not self._users:
            print("Please load a participant list first.")
            return
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return
        if self._result_manager.get_all_winners():
            confirm = self._ask("Existing results will be discarded. Start a new lottery? (y/n): ").lower()
            if confirm != "y":
                return
            self._result_manager = ResultManager()

        lottery = LotteryManager(self._users)
        total = len(prizes)
        print("\nLottery started!")
        for number, prize in enumerate(prizes, start=1):
            self._run_round(number, total, prize, lottery)

        print(f"\n{SEPARATOR}\nLottery finished!")
        self._print_summary(prizes)
        if self._ask("Export results to file? (y/n): ").lower() == "y":
            self._export(prizes)

    def _run_round(self, number: int, total: int, prize: Prize, lottery: LotteryManager) -> None:
        print(f"\n--- Round {number}/{total}: {prize.name} ---")
        print(f"Quantity: {prize.quantity}")
        print(f"Rules: {lottery.describe_rules(prize)}")

        drawn: list[Winner] = []
        while len(drawn) < prize.quantity:
            command = self._ask("Press Enter to draw (or 'pause' / 'terminate'): ").lower()
            if command == "pause":
                if not self._handle_pause():
                    break
                continue
            if command == "terminate":
                print("Round terminated, skipping remaining draws.")
                break
            if command not in ("", "draw"):
                print("Unknown command.")
                continue

            user = lottery.draw(prize)
            if user is None:
                print("No eligible participants remain for this prize; round ends.")
                break
            winner = self._result_manager.add_winner(user, prize)
            drawn.append(winner)
            print(f"Currently drawing the {ordinal(len(drawn))} winner...")
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        print(f"Round {number} finished with {len(drawn)} winner(s).")
        for winner in drawn:
            print(f"  {winner.winning_order}. {winner.user.name} ({winner.user.department})")

    def _handle_pause(self) -> bool:
        """Block until 'continue' (returns True) or 'terminate' (returns False)."""
        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = self._ask("(paused) > ").lower()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                print("Round terminated, skipping remaining draws.")
                return False
            print("Unknown command. Type 'continue' or 'terminate'.")

    # ---------------------------------------------------------------- results

    def _print_summary(self, prizes: list[Prize]) -> None:
        winners = self._result_manager.get_winners_by_prize(prizes)
        print("\nFinal winner list (prize level from high to low):")
        for prize in sorted(prizes, key=lambda item: item.level):
            print(f"\n{prize.name} (quantity {prize.quantity})")
            entries = winners.get(prize.name, [])
            if not entries:
                print("  No winners")
                continue
            for winner in entries:
                print(
                    f"  {winner.winning_order}. {winner.user.name} "
                    f"({winner.user.department}) - {prize.name}"
                )

    def _export(self, prizes: list[Prize]) -> None:
        _, message = self._result_manager.export_to_csv(prizes)[0], self._result_manager.export_to_csv(prizes)[1]
        print(message)

    def _show_results(self) -> None:
        prizes = self._prize_manager.get_prizes()
        if not self._result_manager.get_all_winners():
            print("No results yet.")
            return
        self._print_summary(prizes)
        if self._ask("Export results to file? (y/n): ").lower() == "y":
            self._export(prizes)


def main() -> None:
    try:
        LotteryApp().run()
    except (EOFError, KeyboardInterrupt):
        print("\nProgram terminated.")

"""Data loading utilities for the Beijing Subway network dataset.

The dataset lives in ``subway_data.json`` next to this module, so the
resolved path is valid regardless of the current working directory.
"""

import json
import os

DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read or is malformed."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway dataset from a JSON file.

    Args:
        path: Path of the JSON file. Defaults to ``DATA_FILE_PATH``.

    Returns:
        dict: The JSON content, with ``stations`` and ``connections`` lists.

    Raises:
        DataLoadError: If the file is missing, unreadable, not valid JSON,
            or does not contain the expected top-level structure.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Subway data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataLoadError(f"Cannot read subway data file: {exc}") from exc

    if not isinstance(data, dict):
        raise DataLoadError("Subway data must be a JSON object.")
    for key in ("stations", "connections"):
        if not isinstance(data.get(key), list):
            raise DataLoadError(f"Subway data has a missing or invalid '{key}' list.")
    return data

"""Binary min-heap priority queue used by the Dijkstra implementation."""


class MinPQ:
    """A minimum priority queue implemented as a binary heap.

    Each entry is a (key, item) pair. The entry with the smallest key is
    returned first. Insertion and removal both run in O(log n) time.
    """

    def __init__(self):
        # 1-based heap storage; index 0 is an unused sentinel.
        self._keys = [None]
        self._items = [None]
        self._size = 0

    def __len__(self):
        return self._size

    def is_empty(self):
        """Return True if the queue holds no entries."""
        return self._size == 0

    def min_key(self):
        """Return the smallest key without removing it."""
        if self.is_empty():
            raise IndexError("priority queue is empty")
        return self._keys[1]

    def insert(self, key, item):
        """Insert an item with the given priority key."""
        self._size += 1
        self._keys.append(key)
        self._items.append(item)
        self._swim(self._size)

    def del_min(self):
        """Remove and return the (key, item) pair with the smallest key."""
        if self.is_empty():
            raise IndexError("priority queue is empty")
        key, item = self._keys[1], self._items[1]
        self._exchange(1, self._size)
        self._keys.pop()
        self._items.pop()
        self._size -= 1
        self._sink(1)
        return key, item

    # ----- heap helpers -------------------------------------------------

    def _less(self, i, j):
        return self._keys[i] < self._keys[j]

    def _exchange(self, i, j):
        self._keys[i], self._keys[j] = self._keys[j], self._keys[i]
        self._items[i], self._items[j] = self._items[j], self._items[i]

    def _swim(self, k):
        """Move the element at index k up until heap order holds."""
        while k > 1 and self._less(k, k // 2):
            self._exchange(k, k // 2)
            k //= 2

    def _sink(self, k):
        """Move the element at index k down until heap order holds."""
        while 2 * k <= self._size:
            j = 2 * k
            if j < self._size and self._less(j + 1, j):
                j += 1
            if not self._less(j, k):
                break
            self._exchange(k, j)
            k = j

"""Graph data structures and subway network modeling.

The subway network is a weighted directed graph. Every station is a vertex
identified by an integer station number assigned by ``SubwayNetwork``; every
listed connection is an ``Edge`` that can only be traversed in its listed
direction.
"""


class Edge:
    """A directed, weighted connection between two station numbers.

    Attributes:
        source: Station number of the origin vertex.
        dest: Station number of the destination vertex.
        distance: Actual distance between the stations in meters.
        line: Name of the subway line operating this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Base weight of the edge (the distance in meters)."""
        return self.distance

    def __repr__(self):
        return (f"Edge({self.source!r} -> {self.dest!r}, "
                f"{self.distance} m, {self.line!r})")


class Graph:
    """Weighted directed graph with adjacency lists.

    Station numbers are assigned by the caller. The graph also keeps a
    ``station_lines`` mapping from station name to the lines serving it.
    """

    def __init__(self):
        self._adjacency = {}
        self._edge_count = 0
        self.station_lines = {}

    # ----- vertices -----------------------------------------------------

    def add_vertex(self, vertex):
        """Register a vertex (station number) if it is not present yet."""
        self._adjacency.setdefault(vertex, [])

    def has_vertex(self, vertex):
        """Return True if the vertex exists in the graph."""
        return vertex in self._adjacency

    def vertices(self):
        """Return the list of all vertex numbers."""
        return list(self._adjacency)

    def num_vertices(self):
        """Return the number of vertices."""
        return len(self._adjacency)

    def set_station_lines(self, station_name, lines):
        """Record the lines serving a station, removing duplicates in order.

        Args:
            station_name: Official station name.
            lines: Iterable of line names.
        """
        self.station_lines[station_name] = list(dict.fromkeys(lines))

    # ----- edges --------------------------------------------------------

    def add_edge(self, edge):
        """Add a directed edge, creating its endpoints if necessary.

        Raises:
            TypeError: If ``edge`` is not an ``Edge``.
            ValueError: If the distance is negative.
        """
        if not isinstance(edge, Edge):
            raise TypeError("add_edge expects an Edge instance")
        if edge.distance < 0:
            raise ValueError("edge distance must not be negative")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adjacency[edge.source].append(edge)
        self._edge_count += 1

    def adj(self, vertex):
        """Return the outgoing edges of a vertex (empty list if unknown)."""
        return self._adjacency.get(vertex, [])

    def edges(self):
        """Yield every edge in the graph."""
        for outgoing in self._adjacency.values():
            yield from outgoing

    def num_edges(self):
        """Return the number of directed edges."""
        return self._edge_count

    def update_edge_distance(self, source, dest, distance, line=None):
        """Update the distance of matching edges.

        Returns:
            int: Number of edges updated.
        """
        if distance <= 0:
            raise ValueError("distance must be positive")
        updated = 0
        for edge in self.adj(source):
            if edge.dest == dest and (line is None or edge.line == line):
                edge.distance = distance
                updated += 1
        return updated

    def remove_edges(self, source, dest, line=None):
        """Remove matching edges from the graph.

        Returns:
            int: Number of edges removed.
        """
        if source not in self._adjacency:
            return 0
        kept = []
        removed = 0
        for edge in self._adjacency[source]:
            if edge.dest == dest and (line is None or edge.line == line):
                removed += 1
            else:
                kept.append(edge)
        self._adjacency[source] = kept
        self._edge_count -= removed
        return removed


def _clean_name(value):
    """Normalize a station or line name by collapsing whitespace."""
    if not isinstance(value, str):
        raise ValueError(f"name must be a string, got {value!r}")
    cleaned = " ".join(value.split())
    if not cleaned:
        raise ValueError("name must not be empty")
    return cleaned


class SubwayNetwork:
    """Subway network built from the dataset dictionary.

    Attributes:
        graph: The ``Graph`` holding stations and directed connections.
        station_to_number: Mapping from station name to station number.
        number_to_station: List mapping station number to station name.
        line_stations: Mapping from line name to the stations it serves,
            in the order they were first encountered in the connections.
    """

    def __init__(self, data):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = []
        self.line_stations = {}
        self._build(data)

    # ----- construction -------------------------------------------------

    def _assign_number(self, name):
        """Return the number of a station, assigning a new one if needed."""
        if name not in self.station_to_number:
            number = len(self.number_to_station)
            self.station_to_number[name] = number
            self.number_to_station.append(name)
            self.graph.add_vertex(number)
        return self.station_to_number[name]

    def _register_station_lines(self, name, lines):
        """Merge lines into the station's recorded line list."""
        current = list(self.graph.station_lines.get(name, []))
        for line in lines:
            if line not in current:
                current.append(line)
        self.graph.set_station_lines(name, current)

    def _build(self, data):
        for station in data.get("stations", []):
            name = _clean_name(station.get("name"))
            self._assign_number(name)
            lines = [str(line).strip() for line in station.get("lines", [])]
            self._register_station_lines(name, lines)

        for conn in data.get("connections", []):
            source = _clean_name(conn.get("source"))
            dest = _clean_name(conn.get("dest"))
            distance = int(conn.get("distance"))
            line = _clean_name(conn.get("line"))
            source_no = self._assign_number(source)
            dest_no = self._assign_number(dest)
            self.graph.add_edge(Edge(source_no, dest_no, distance, line))
            self._register_station_lines(source, [line])
            self._register_station_lines(dest, [line])
            stations = self.line_stations.setdefault(line, [])
            for name in (source, dest):
                if name not in stations:
                    stations.append(name)

    # ----- queries ------------------------------------------------------

    def number(self, name):
        """Return the station number for a station name.

        Raises:
            KeyError: If the station is unknown.
        """
        return self.station_to_number[_clean_name(name)]

    def name_of(self, number):
        """Return the station name for a station number."""
        return self.number_to_station[number]

    def has_station(self, name):
        """Return True if the station exists in the network."""
        return _clean_name(name) in self.station_to_number

    def stations(self):
        """Return all station names in numbering order."""
        return list(self.number_to_station)

    def lines_of(self, name):
        """Return the lines serving a station."""
        return list(self.graph.station_lines.get(name, []))

    def transfer_stations(self):
        """Return a mapping of interchange stations to their lines."""
        return {name: list(lines)
                for name, lines in self.graph.station_lines.items()
                if len(lines) > 1}

    def all_lines(self):
        """Return the sorted list of all line names in the network."""
        return sorted(self.line_stations)


def build_network(data):
    """Build a ``SubwayNetwork`` from the raw dataset dictionary."""
    return SubwayNetwork(data)

"""Path calculation based on Dijkstra's shortest path algorithm.

The search runs over labels of the form ``(station_number, line, transfers)``
so that transfers can be counted and limited while the search proceeds.
Costs are measured in meters, with optional penalties for transfers and for
using lines outside the passenger's preference.
"""

from minpq import MinPQ

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Upper bound on transfers explored, which keeps the label space finite.
HARD_TRANSFER_LIMIT = 10
# Extra cost (meters) for each edge on a non-preferred line.
LINE_PREFERENCE_PENALTY_M = 500.0
# Dominant transfer penalty so that the fewest transfers are minimized first.
LEAST_TRANSFER_PENALTY_M = 1_000_000.0
# Composite score: roughly 5 minutes of travel time expressed in meters
# (5 min at 1.5 min/km is about 3.3 km).
COMPREHENSIVE_TRANSFER_PENALTY_M = 3300.0


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of ``VALID_MODES``.
        max_transfers: Maximum allowed transfers, or None for no explicit limit.
        preferred_lines: List of preferred line names (may be empty).
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"unknown mode {mode!r}; expected one of {VALID_MODES}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer")
            if max_transfers < 0:
                raise ValueError("max_transfers must not be negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = [str(line).strip()
                                for line in (preferred_lines or [])
                                if str(line).strip()]

    def transfer_penalty(self):
        """Return the cost penalty added for every transfer in this mode."""
        if self.mode == MODE_LEAST_TRANSFER:
            return LEAST_TRANSFER_PENALTY_M
        if self.mode == MODE_COMPREHENSIVE:
            return COMPREHENSIVE_TRANSFER_PENALTY_M
        return 0.0

    def effective_transfer_limit(self):
        """Return the transfer limit actually enforced during search."""
        if self.max_transfers is None:
            return HARD_TRANSFER_LIMIT
        return min(self.max_transfers, HARD_TRANSFER_LIMIT)


class PathResult:
    """Result of a path query.

    Attributes:
        station_numbers: Station numbers from start to end.
        edges: Edges traversed, in order from start to end.
        distance: Total physical distance in meters.
        transfer_count: Number of line changes along the path.
        cost: Optimization cost of the path (includes penalties).
    """

    def __init__(self, station_numbers, edges, distance, transfer_count, cost):
        self.station_numbers = station_numbers
        self.edges = edges
        self.distance = distance
        self.transfer_count = transfer_count
        self.cost = cost


class ShortestPath:
    """Dijkstra-based shortest path computation from a single source.

    Attributes:
        dist_to: Dictionary mapping reachable station numbers to the physical
            distance in meters of the best path found to them.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"unknown source station number: {source}")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._cost_to = {}
        self._best_label = {}
        self._label_cost = {}
        self._label_dist = {}
        self._edge_to = {}
        self._run()

    def _run(self):
        """Run Dijkstra's algorithm over (station, line, transfers) labels."""
        config = self.config
        transfer_limit = config.effective_transfer_limit()
        transfer_penalty = config.transfer_penalty()
        preferred = set(config.preferred_lines)

        start = (self.source, None, 0)
        self._label_cost[start] = 0.0
        self._label_dist[start] = 0
        queue = MinPQ()
        queue.insert(0.0, start)
        settled = set()

        while not queue.is_empty():
            cost, label = queue.del_min()
            if label in settled:
                continue
            settled.add(label)
            node, line, transfers = label

            # Labels are settled in order of cost, so the first label seen
            # for a station is its best label.
            if node not in self._best_label:
                self._best_label[node] = label
                self.dist_to[node] = self._label_dist[label]
                self._cost_to[node] = cost

            for edge in self.graph.adj(node):
                next_transfers = transfers
                step = float(edge.weight)
                if line is not None and edge.line != line:
                    next_transfers += 1
                    if next_transfers > transfer_limit:
                        continue
                    step += transfer_penalty
                if preferred and edge.line not in preferred:
                    step += LINE_PREFERENCE_PENALTY_M

                next_label = (edge.dest, edge.line, next_transfers)
                if next_label in settled:
                    continue
                next_cost = cost + step
                known = self._label_cost.get(next_label)
                if known is None or next_cost < known:
                    self._label_cost[next_label] = next_cost
                    self._label_dist[next_label] = self._label_dist[label] + edge.weight
                    self._edge_to[next_label] = (label, edge)
                    queue.insert(next_cost, next_label)

    def has_path_to(self, target):
        """Return True if the target station can be reached."""
        return target in self._best_label

    def cost_to(self, target):
        """Return the optimization cost to the target, or None if unreachable."""
        return self._cost_to.get(target)

    def get_transfer_count(self, target):
        """Return the number of transfers on the path to the target."""
        if target not in self._best_label:
            return 0
        return self._best_label[target][2]

    def path_with_edges(self, target):
        """Backtrack the path to the target.

        Returns:
            tuple: (list of station numbers from source to target,
                    list of Edge objects traversed in order).
            If the target is unreachable, returns ([], []).
        """
        if target not in self._best_label:
            return [], []
        label = self._best_label[target]
        nodes = []
        edges = []
        while True:
            nodes.append(label[0])
            link = self._edge_to.get(label)
            if link is None:
                break
            previous, edge = link
            edges.append(edge)
            label = previous
        nodes.reverse()
        edges.reverse()
        return nodes, edges


def find_path(graph, source, target, config=None):
    """Find the optimal path between two station numbers.

    Returns:
        PathResult, or None if no path satisfies the configuration.
    """
    search = ShortestPath(graph, source, config)
    if not search.has_path_to(target):
        return None
    nodes, edges = search.path_with_edges(target)
    distance = sum(edge.distance for edge in edges)
    return PathResult(nodes, edges, distance,
                      search.get_transfer_count(target), search.cost_to(target))


def find_shortest_distance_path(graph, source, target, max_transfers=None,
                                preferred_lines=None):
    """Shortest-distance query (minimizes physical distance)."""
    config = PathfindingConfig(MODE_SHORTEST_DISTANCE, max_transfers, preferred_lines)
    return find_path(graph, source, target, config)


def find_least_transfer_path(graph, source, target, max_transfers=None,
                             preferred_lines=None):
    """Minimum-transfer query (transfers first, distance as tie-breaker)."""
    config = PathfindingConfig(MODE_LEAST_TRANSFER, max_transfers, preferred_lines)
    return find_path(graph, source, target, config)


def find_comprehensive_path(graph, source, target, max_transfers=None,
                            preferred_lines=None):
    """Comprehensive query balancing distance and transfers."""
    config = PathfindingConfig(MODE_COMPREHENSIVE, max_transfers, preferred_lines)
    return find_path(graph, source, target, config)

"""Fare calculation based on the Beijing Subway fare structure.

Tiers (distance in meters, fares in yuan):
    [0, 6000)          3 yuan   (basic tier)
    [6000, 12000)      4 yuan   (progressive tier)
    [12000, 22000)     5 yuan   (progressive tier)
    [22000, 32000)     6 yuan   (progressive tier)
    (32000, ...)       7 yuan, plus 1 yuan for every further 20 km (extended tier)

Boundary values use the PRD-specified fares, which take precedence over the
left-closed tier rule: 6 km -> 3, 12 km -> 4, 22 km -> 6, 32 km -> 6 yuan.
Fractional distances are rounded up to the next whole meter before lookup.
"""

import math

BOUNDARY_FARE_OVERRIDES = {
    6000: 3,
    12000: 4,
    22000: 6,
    32000: 6,
}

EXTENDED_TIER_START_M = 32000
EXTENDED_STEP_M = 20000
EXTENDED_BASE_FARE = 7

# Average travel speed based estimate: minutes per kilometer and per transfer.
MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def Sum_money(distance):
    """Return the subway fare in yuan for a travel distance.

    Args:
        distance: Travel distance in meters (fractions are rounded up).

    Returns:
        int: Fare in yuan.

    Raises:
        TypeError: If distance is not a number.
        ValueError: If distance is negative or not finite.
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number of meters")
    if not math.isfinite(distance):
        raise ValueError("distance must be finite")
    if distance < 0:
        raise ValueError("distance must not be negative")

    meters = math.ceil(distance)
    if meters in BOUNDARY_FARE_OVERRIDES:
        return BOUNDARY_FARE_OVERRIDES[meters]
    if meters < 6000:
        return 3
    if meters < 12000:
        return 4
    if meters < 22000:
        return 5
    if meters < EXTENDED_TIER_START_M:
        return 6
    # Extended tier: 7 yuan for (32 km, 52 km), 8 yuan for [52 km, 72 km), ...
    return EXTENDED_BASE_FARE + (meters - EXTENDED_TIER_START_M) // EXTENDED_STEP_M


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes.

    Args:
        distance: Travel distance in meters.
        transfer_count: Number of transfers.

    Returns:
        float: Estimated time in minutes (about 1.5 min/km plus 5 min per transfer).
    """
    if distance < 0:
        raise ValueError("distance must not be negative")
    if transfer_count < 0:
        raise ValueError("transfer_count must not be negative")
    return (distance / 1000.0) * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""Path visualization, summary generation, and report export."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    """Format a distance: meters for short trips, kilometers for long trips."""
    meters = int(round(meters))
    if meters < 1000:
        return f"{meters} m"
    return f"{meters / 1000:.2f} km ({meters} m)"


def count_transfers(edges):
    """Count line changes between consecutive edges."""
    return sum(1 for index in range(1, len(edges))
               if edges[index].line != edges[index - 1].line)


def find_transfers(station_names, edges):
    """Return transfer details: station and the lines before and after it."""
    transfers = []
    for index in range(1, len(edges)):
        previous_line = edges[index - 1].line
        current_line = edges[index].line
        if previous_line != current_line:
            transfers.append({
                "station": station_names[index],
                "from_line": previous_line,
                "to_line": current_line,
            })
    return transfers


def involved_lines(edges):
    """Return the unique line names used on the path, in order of first use."""
    lines = []
    for edge in edges:
        if edge.line not in lines:
            lines.append(edge.line)
    return lines


def build_segments(station_names, edges):
    """Group consecutive edges on the same line into travel segments."""
    segments = []
    if not edges:
        return segments
    start = 0
    for index in range(1, len(edges) + 1):
        if index == len(edges) or edges[index].line != edges[start].line:
            segment_edges = edges[start:index]
            segments.append({
                "line": edges[start].line,
                "start": station_names[start],
                "end": station_names[index],
                "stations": index - start + 1,
                "distance_m": sum(edge.distance for edge in segment_edges),
            })
            start = index
    return segments


def build_summary(edges):
    """Extract key summary figures: distance, fare, transfers, and time."""
    distance = sum(edge.distance for edge in edges)
    transfers = count_transfers(edges)
    return {
        "distance_m": distance,
        "fare": Sum_money(distance),
        "transfer_count": transfers,
        "estimated_time_min": round(calculate_estimated_time(distance, transfers), 1),
    }


def build_report(query, station_names, edges):
    """Build a structured report dictionary for a computed path.

    Args:
        query: Dictionary describing the query (start, end, mode, ...).
        station_names: Station names from start to end.
        edges: Edges traversed from start to end.
    """
    summary = build_summary(edges)
    return {
        "query": dict(query),
        "path": list(station_names),
        "segments": build_segments(station_names, edges),
        "distance_m": summary["distance_m"],
        "distance_km": round(summary["distance_m"] / 1000.0, 3),
        "fare": summary["fare"],
        "estimated_time_min": summary["estimated_time_min"],
        "transfer_count": summary["transfer_count"],
        "transfers": find_transfers(station_names, edges),
        "lines": involved_lines(edges),
    }


def render_text(report):
    """Render a report as console-formatted text."""
    query = report["query"]
    lines = [
        "=" * 60,
        "Beijing Subway Route Report",
        "=" * 60,

        f"To:   {query['end']}",

"""Data management: integrity checks and update operations on the dataset.

All update functions operate on the raw dataset dictionary returned by
``data_loader.load_subway_data``. Use ``save_subway_data`` to persist changes.
"""

import json
import os
import re
import tempfile

from data_loader import DATA_FILE_PATH

REQUIRED_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

REPRESENTATIVE_STATIONS = [
    "Xizhimen", "Dongzhimen", "Dongdan", "Fuxingmen", "Zhichunlu",
]

MIN_PLAUSIBLE_DISTANCE_M = 30
MAX_PLAUSIBLE_DISTANCE_M = 10000


def _normalize(text):
    """Lowercase a name and drop whitespace and punctuation for comparison."""
    return re.sub(r"[^0-9a-z\u4e00-\u9fff]", "", str(text).lower())


def _is_positive_int(value):
    return isinstance(value, int) and not isinstance(value, bool) and value > 0


def check_data_integrity(data):
    """Run integrity checks on the dataset.

    Checks performed:
        * station data: names present, unique, with line information;
        * line coverage: main lines and representative stations present;
        * connections: valid fields, known endpoints, no self-loops or
          duplicates, lines consistent with station line lists;
        * distance accuracy: positive integers within a plausible range;
        * reverse connections present for each listed connection.

    Returns:
        tuple: (errors, warnings), each a list of message strings.
    """
    errors = []
    warnings = []

    stations = data.get("stations", [])
    connections = data.get("connections", [])

    # ----- station data ------------------------------------------------
    names = []
    station_lines = {}
    for index, station in enumerate(stations):
        name = station.get("name") if isinstance(station, dict) else None
        if not isinstance(name, str) or not name.strip():
            errors.append(f"Station entry #{index} has no valid name")
            continue
        name = " ".join(name.split())
        names.append(name)
        lines = station.get("lines")
        if not isinstance(lines, list) or not lines:
            warnings.append(f"Station '{name}' has no line information")
            lines = []
        station_lines.setdefault(name, set()).update(str(x) for x in lines)

    duplicates = {name for name in names if names.count(name) > 1}
    for name in sorted(duplicates):
        errors.append(f"Duplicate station name: '{name}'")

    name_set = set(names)
    normalized_names = {_normalize(name) for name in name_set}

    # ----- line and station coverage -----------------------------------
    all_lines = set()
    for lines in station_lines.values():
        all_lines.update(lines)
    for conn in connections:
        if isinstance(conn, dict) and isinstance(conn.get("line"), str):
            all_lines.add(conn["line"].strip())
    normalized_lines = {_normalize(line) for line in all_lines}

    for required in REQUIRED_LINES:
        if _normalize(required) not in normalized_lines:
            warnings.append(f"Main line '{required}' is not covered by the dataset")

    for representative in REPRESENTATIVE_STATIONS:
        if _normalize(representative) not in normalized_names:
            warnings.append(
                f"Representative station '{representative}' is not in the dataset")

    # ----- connection integrity and distance accuracy -------------------
    seen_keys = set()
    directed = set()
    for index, conn in enumerate(connections):
        if not isinstance(conn, dict):
            errors.append(f"Connection #{index} is not an object")
            continue
        source = conn.get("source")
        dest = conn.get("dest")
        line = conn.get("line")
        distance = conn.get("distance")
        if not isinstance(source, str) or not isinstance(dest, str) \
                or not isinstance(line, str):
            errors.append(f"Connection #{index} has missing source, dest, or line")
            continue
        source = " ".join(source.split())
        dest = " ".join(dest.split())
        line = line.strip()
        label = f"'{source}' -> '{dest}' ({line})"

        if source not in name_set:
            errors.append(f"Connection {label} references unknown station '{source}'")
        if dest not in name_set:
            errors.append(f"Connection {label} references unknown station '{dest}'")
        if source == dest:
            errors.append(f"Connection {label} is a self-loop")

        if not _is_positive_int(distance):
            errors.append(f"Connection {label} has invalid distance {distance!r}")
        elif not (MIN_PLAUSIBLE_DISTANCE_M <= distance <= MAX_PLAUSIBLE_DISTANCE_M):
            warnings.append(f"Connection {label} distance {distance} m is outside "
                            f"the plausible range")

        key = (source, dest, line)
        if key in seen_keys:
            errors.append(f"Duplicate connection {label}")
        seen_keys.add(key)
        directed.add((source, dest))

        for endpoint in (source, dest):
            if endpoint in station_lines and line not in station_lines[endpoint]:
                warnings.append(f"Station '{endpoint}' does not list line '{line}'")

    for source, dest in sorted(directed):
        if (dest, source) not in directed:
            warnings.append(f"Connection '{source}' -> '{dest}' has no reverse direction")

    return errors, warnings


def add_line(data, line_name, station_names, distances):
    """Add a new line with a sequence of stations and inter-station distances.

    Connections are created in both directions. Missing stations are added.

    Args:
        data: Raw dataset dictionary (modified in place).
        line_name: Name of the new line.
        station_names: Ordered list of station names on the line.
        distances: Distances in meters between consecutive stations.

    Returns:
        int: Number of connections added.

    Raises:
        ValueError: On invalid input or if the line already exists.
    """
    line_name = " ".join(str(line_name).split())
    if not line_name:
        raise ValueError("line name must not be empty")
    existing_lines = {conn.get("line") for conn in data["connections"]}
    for station in data["stations"]:
        existing_lines.update(station.get("lines", []))
    if line_name in existing_lines:
        raise ValueError(f"line '{line_name}' already exists")
    stations = [" ".join(str(name).split()) for name in station_names]
    if len(stations) < 2:
        raise ValueError("a line needs at least two stations")
    if len(set(stations)) != len(stations):
        raise ValueError("station names on a line must be distinct")
    if len(distances) != len(stations) - 1:
        raise ValueError("distances must have one entry per consecutive station pair")
    for distance in distances:
        if not _is_positive_int(distance):
            raise ValueError(f"invalid distance {distance!r}; must be a positive integer")

    by_name = {station["name"]: station for station in data["stations"]}
    for name in stations:
        if name not in by_name:
            entry = {"name": name, "lines": [line_name]}
            data["stations"].append(entry)
            by_name[name] = entry
        elif line_name not in by_name[name]["lines"]:
            by_name[name]["lines"].append(line_name)

    added = 0
    for index, distance in enumerate(distances):
        first, second = stations[index], stations[index + 1]
        data["connections"].append({"source": first, "dest": second,
                                    "distance": distance, "line": line_name})
        data["connections"].append({"source": second, "dest": first,
                                    "distance": distance, "line": line_name})
        added += 2
    return added


def modify_station(data, old_name, new_name=None, lines=None):
    """Rename a station and/or replace its line list.

    Connections referencing the old name are updated to the new name.

    Returns:
        dict: The updated station entry.

    Raises:
        KeyError: If the station does not exist.
        ValueError: If the new name already belongs to another station.
    """
    old_name = " ".join(old_name.split())
    station = next((s for s in data["stations"] if s["name"] == old_name), None)
    if station is None:
        raise KeyError(f"station '{old_name}' does not exist")
    if new_name is not None:
        new_name = " ".join(new_name.split())
        if not new_name:
            raise ValueError("new station name must not be empty")
        if new_name != old_name and any(s["name"] == new_name for s in data["stations"]):
            raise ValueError(f"station '{new_name}' already exists")
        station["name"] = new_name
        for conn in data["connections"]:
            if conn["source"] == old_name:
                conn["source"] = new_name
            if conn["dest"] == old_name:
                conn["dest"] = new_name
    if lines is not None:
        station["lines"] = list(dict.fromkeys(str(line).strip() for line in lines))
    return station


def update_distance(data, source, dest, distance, line=None):
    """Update the distance of the connection(s) from source to dest.

    Args:
        data: Raw dataset dictionary (modified in place).
        source: Origin station name.
        dest: Destination station name.
        distance: New distance in meters (positive integer).
        line: Optional line filter.

    Returns:
        int: Number of connections updated.

    Raises:
        ValueError: If the distance is invalid.
        LookupError: If no matching connection exists.
    """
    if not _is_positive_int(distance):
        raise ValueError("distance must be a positive integer (meters)")
    updated = 0
    for conn in data["connections"]:
        if conn["source"] == source and conn["dest"] == dest \
                and (line is None or conn["line"] == line):
            conn["distance"] = distance
            updated += 1
    if updated == 0:
        raise LookupError(f"no connection from '{source}' to '{dest}'")
    return updated


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write the dataset to a JSON file atomically."""
    directory = os.path.dirname(os.path.abspath(path))
    handle, temp_path = tempfile.mkstemp(dir=directory, suffix=".tmp")
    try:
        with os.fdopen(handle, "w", encoding="utf-8") as stream:
            json.dump(data, stream, ensure_ascii=False, indent=2)
        os.replace(temp_path, path)
    except OSError:
        if os.path.exists(temp_path):
            os.remove(temp_path)
        raise

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import difflib
import re
import sys

from data_loader import DATA_FILE_PATH, DataLoadError, load_subway_data
from data_manager import check_data_integrity
from models import SubwayNetwork
from path_report import build_report, render_json, render_text
from pathfinding import (
    MODE_SHORTEST_DISTANCE,
    VALID_MODES,
    PathfindingConfig,
    find_path,
)

EXIT_COMMANDS = ("quit", "exit", "q")
NEAREST_MATCH_CUTOFF = 0.6
_SEPARATOR_PATTERN = re.compile(r"[;,，；]")
_NOISE_PATTERN = re.compile(r"[\s\-_'’·•]+")
_STATION_SUFFIXES = ("station", "站")


class QueryError(Exception):
    """User-facing error raised while answering a query."""


def normalize_station_name(name):
    """Normalize a station name for lookup (case, spacing, and suffixes ignored)."""
    text = _NOISE_PATTERN.sub("", name).lower()
    for suffix in _STATION_SUFFIXES:
        if text.endswith(suffix) and len(text) > len(suffix):
            text = text[:-len(suffix)]
    return text


class StationResolver:
    """Maps user-entered station names onto official station names."""

    def __init__(self, station_names):
        self._names = list(station_names)
        self._exact = set(self._names)
        self._by_key = {}
        for name in self._names:
            self._by_key.setdefault(normalize_station_name(name), name)

    def resolve(self, raw, role):
        """Resolve a raw station name.

        Args:
            raw: Name typed by the user.
            role: "Starting" or "Ending", used in messages.

        Returns:
            tuple: (official station name, note string or None).

        Raises:
            QueryError: If the name is empty or no station is similar enough.
        """
        cleaned = " ".join(raw.split())
        if not cleaned:
            raise QueryError(f"{role} station name cannot be empty.")
        if cleaned in self._exact:
            return cleaned, None
        key = normalize_station_name(cleaned)
        if key in self._by_key:
            name = self._by_key[key]
            return name, f"Note: '{cleaned}' standardized to '{name}'."
        matches = difflib.get_close_matches(
            key, list(self._by_key), n=1, cutoff=NEAREST_MATCH_CUTOFF)
        if not matches:
            raise QueryError(
                f"{role} station '{cleaned}' does not exist, "
                f"please check the station name.")
        name = self._by_key[matches[0]]
        note = (f"Note: {role.lower()} station '{cleaned}' not found exactly; "
                f"matched to nearest station '{name}'.")
        return name, note


def parse_query(text):
    """Split 'Starting Station; Ending Station' into two raw station names.

    Raises:
        QueryError: If the input does not contain exactly two non-empty parts.
    """
    parts = [part.strip() for part in _SEPARATOR_PATTERN.split(text)]
    if len(parts) != 2 or not all(parts):
        raise QueryError("Input format incorrect, please use "
                         "'Starting Station; Ending Station' format.")
    return parts[0], parts[1]


def parse_line_preference(value):
    """Parse a comma-separated list of preferred line names."""
    if not value:
        return []
    return [item.strip() for item in value.split(",") if item.strip()]

def run_data_check(data):
    """Run the data integrity check, print results, and return an exit code."""
    errors, warnings = check_data_integrity(data)
    for warning in warnings:
        print(f"WARNING: {warning}")
    for error in errors:
        print(f"ERROR: {error}")
    if errors:
        print(f"Data integrity check FAILED: {len(errors)} error(s), "

        return 1
    print(f"Data integrity check passed with {len(warnings)} warning(s).")

def answer_query(network, resolver, text, config, output_format):
    """Answer one 'Starting; Ending' query.

    Returns:
        tuple: (list of notes, rendered report string).

    Raises:
        QueryError: For any user-facing failure.
    """
    raw_start, raw_end = parse_query(text)
    notes = []
    start, note = resolver.resolve(raw_start, "Starting")
    if note:
        notes.append(note)
    end, note = resolver.resolve(raw_end, "Ending")
    if note:
        notes.append(note)
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station.")

    try:
        result = find_path(network.graph, network.number(start),
                           network.number(end), config)
    except (KeyError, ValueError, IndexError) as exc:
        raise QueryError(f"Path calculation failed: {exc}") from exc
    if result is None:
        raise QueryError(f"No route found from '{start}' to '{end}' under the "
                         f"current settings (mode, max transfers, line preference).")

    names = [network.name_of(number) for number in result.station_numbers]
    query = {
        "start": start,
        "end": end,
        "mode": config.mode,
        "max_transfers": config.max_transfers,
        "line_preference": list(config.preferred_lines),
    }
    try:
        report = build_report(query, names, result.edges)
    except (TypeError, ValueError) as exc:
        raise QueryError(f"Fare calculation failed: {exc}") from exc

    if output_format == "json":
        rendered = render_json(report)
    else:
        rendered = render_text(report)
    return notes, rendered


def append_to_file(path, content):
    """Append rendered output to a file."""
    with open(path, "a", encoding="utf-8") as handle:
        handle.write(content + "\n\n")


def interactive_loop(network, config, output_format, output_file=None):
    """Run the continuous query loop until the user quits."""
    resolver = StationResolver(network.stations())
    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' to query (type 'quit' to exit).")
    while True:
        try:
            raw = input("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        try:
            notes, rendered = answer_query(network, resolver, text, config,
                                           output_format)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # keep the session alive on unexpected errors
            print(f"Unexpected error while answering query: {exc}")
            continue
        for note in notes:
            print(note)
        print(rendered)
        if output_file:
            try:
                append_to_file(output_file, rendered)
            except OSError as exc:
                print(f"Warning: could not write output file: {exc}")
    print("Goodbye.")
    return 0


def main(argv=None):
    """CLI entry point. Returns a process exit code."""
    args = parse_arguments(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except DataLoadError as exc:
        print(f"Error: {exc}")
        return 1

    if args.check_data:
        return run_data_check(data)

    try:
        config = PathfindingConfig(args.mode, args.max_transfers,
                                   parse_line_preference(args.line_preference))
    except ValueError as exc:
        print(f"Error: {exc}")
        return 2

    try:
        network = SubwayNetwork(data)
    except (KeyError, ValueError, TypeError) as exc:
        print(f"Error: could not build subway network: {exc}")
        return 1

    return interactive_loop(network, config, args.output_format, args.output_file)


if __name__ == "__main__":
    sys.exit(main())

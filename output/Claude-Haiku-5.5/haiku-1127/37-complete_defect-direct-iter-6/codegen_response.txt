def normalize_text(value: str) -> str:
    """Normalize a name or group for case-insensitive, whitespace-tolerant comparison."""
    return " ".join(str(value).split()).casefold()


def is_same_group(first: str, second: str) -> bool:
    """Return True when two department/group names refer to the same group."""
    return normalize_text(first) == normalize_text(second)

class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        if not isinstance(path, str) or not path.strip():
            return [], ["File Error: no file path provided."]
        cleaned = path.strip().strip('"').strip("'")
        if not os.path.isfile(cleaned):
            return [], [f"File Error: file not found: '{cleaned}'."]
        return self._load_from_path(cleaned)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse its lines into users and format errors."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except UnicodeDecodeError as exc:
            return [], [f"File Error: cannot decode '{path}' as UTF-8: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff").strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence of each name) and duplicates."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to number of people."""
        canonical: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            display = " ".join(user.department.split())
            key = normalize_text(display)
            name = canonical.setdefault(key, display)
            counts[name] = counts.get(name, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.models import Prize, normalize_text

DEFAULT_PRIZES: list[tuple[str, int, list[str], str | None]] = [
    ("Third Prize - Movie Tickets", 10, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 5, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 2, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(f"Cannot load default prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a prize to the end of the lottery sequence."""
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name must not be empty.")
        clean_name = " ".join(name.split())
        key = normalize_text(clean_name)
        if any(normalize_text(p.name) == key for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")

        groups: list[str] = []
        seen_groups: set[str] = set()
        for group in avoid_groups or []:
            if not isinstance(group, str):
                continue
            display = " ".join(group.split())
            if display and normalize_text(display) not in seen_groups:
                seen_groups.add(normalize_text(display))
                groups.append(display)

        drawer: str | None = None
        if associated_drawer is not None and str(associated_drawer).strip():
            drawer = " ".join(str(associated_drawer).split())

        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return the prizes in current lottery order (a shallow copy of the list)."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        count = len(self._prizes)
        try:
            positions = list(order)
        except TypeError:
            return False
        if not all(isinstance(p, int) and not isinstance(p, bool) for p in positions):
            return False
        if sorted(positions) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[p - 1] for p in positions]
        return True

"""Lottery rule engine: eligibility filtering, special rules, fallback protection and anti-duplication."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    is_same_group,
    normalize_text,
)

FALLBACK_SOURCE_DRAWERS: tuple[str, str] = (CHEN_QIXIAN, JIN_DUO)
FALLBACK_TARGET_DRAWER: str = MA_JIAQI


class LotteryManager:
    """Runs lottery draws according to the rule engine."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        self._rng = rng if rng is not None else random.Random()
        # Records whether the last two lottery persons' (Chen Qixian / Jin Duo) rounds
        # included a Cheng Ru Group member. True = protected group member won.
        self.last_two_draw_results: list[bool] = []

    # ------------------------------------------------------------------ helpers
    def has_won(self, user: User) -> bool:
        """Return True if the user has already won any prize."""
        return normalize_text(user.name) in self._winner_keys

    @staticmethod
    def _is_drawer(prize: Prize, drawer_name: str) -> bool:
        return prize.associated_drawer is not None and normalize_text(
            prize.associated_drawer
        ) == normalize_text(drawer_name)

    def _fallback_active(self, prize: Prize) -> bool:
        """Cheng Ru protection applies to Ma Jiaqi when neither of the last two
        lottery persons' results contained a Cheng Ru Group member."""
        return (
            self._is_drawer(prize, FALLBACK_TARGET_DRAWER)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    # ------------------------------------------------------------------ rules
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the users eligible to win the given prize right now."""
        # Anti-duplication: previous winners are always excluded.
        pool = [user for user in self._users if not self.has_won(user)]

        # Special rule: Li Bin's prizes only filter previous winners.
        if self._is_drawer(prize, LI_BIN):
            return pool

        # Fallback (Cheng Ru group protection) overrides avoidance groups.
        if self._fallback_active(prize):
            return [user for user in pool if is_same_group(user.department, CHENG_RU_GROUP)]

        avoided = {normalize_text(group) for group in prize.avoid_groups}
        return [user for user in pool if normalize_text(user.department) not in avoided]

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        rules: list[str] = []
        if self._is_drawer(prize, LI_BIN):
            rules.append(
                "Special rule (Li Bin): only previous winners are excluded; "
                "group avoidance is not applied."
            )
        else:
            rules.append("Previous winners are excluded.")
            if self._fallback_active(prize):
                rules.append(
                    "Cheng Ru Group protection: candidates are limited to Cheng Ru Group members."
                )
            elif prize.avoid_groups:
                rules.append("Avoided groups: " + ", ".join(prize.avoid_groups) + ".")
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}.")
        return " ".join(rules)

    # ------------------------------------------------------------------ draws
    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible candidate exists."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_keys.add(normalize_text(winner.name))
        return winner

    def finish_round(self, prize: Prize, round_winners: list[User]) -> None:
        """Update fallback state after a round has completed (or was terminated)."""
        if self._is_drawer(prize, FALLBACK_TARGET_DRAWER):
            # The protected round has been handled; restart the observation window.
            self.last_two_draw_results = []
            return

        if any(self._is_drawer(prize, source) for source in FALLBACK_SOURCE_DRAWERS):
            has_cheng_ru = any(
                is_same_group(user.department, CHENG_RU_GROUP) for user in round_winners
            )
            self.last_two_draw_results = (self.last_two_draw_results + [has_cheng_ru])[-2:]

"""Result management: recording winners, grouping them by prize and exporting to CSV."""

from __future__ import annotations

import csv
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path

from lottery_system.models import Prize, User, normalize_text


@dataclass(frozen=True)
class Winner:
    """A winning record."""

    user: User
    prize_name: str


class ResultManager:
    """Keeps the winner list in winning order and handles export."""

    CSV_HEADER: tuple[str, str, str] = ("Name", "Department", "Prize")

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> None:
        """Record a winner. Raises ValueError if the person already won."""
        key = normalize_text(user.name)
        if any(normalize_text(w.user.name) == key for w in self._winners):
            raise ValueError(f"{user.name} has already won a prize.")
        self._winners.append(Winner(user=user, prize_name=prize.name))

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, keeping winning order inside each group."""
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to awards_result_YYYYMMDD.txt in CSV format."""
        path = Path.cwd() / f"awards_result_{datetime.now():%Y%m%d}.txt"
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(self.CSV_HEADER)
                for prize in prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow(
                            (winner.user.name, winner.user.department, prize.name)
                        )
        except OSError as exc:
            return False, f"Export failed: cannot write '{path}': {exc}"
        return True, f"Winner list successfully exported to {path}"

def _ordinal(number: int) -> str:
    if 11 <= number % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotterySystemCLI:
    """Interactive command-line controller for the lottery system."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[[str], None] = print,
    ) -> None:
        self._input = input_func
        self._output = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.lottery_done = False

    # ------------------------------------------------------------------ io
    def say(self, message: str = "") -> None:
        self._output(message)

    def ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    # ------------------------------------------------------------------ main loop
    def run(self) -> int:
        self.say("=== Company Team Building Smart Lottery System ===")
        handlers: dict[str, Callable[[], None]] = {
            "1": self.load_default_list,
            "2": self.load_custom_list,
            "3": self.show_groups,
            "4": self.load_default_prizes,
            "5": self.add_custom_prize,
            "6": self.manage_prize_order,
            "7": self.start_lottery,
            "8": self.export_results,
        }
        try:
            while True:
                self.say()
                self.say("Main Menu:")
                for key, label in MENU_ITEMS:
                    self.say(f"  {key}. {label}")
                choice = self.ask("Select an option: ")
                if choice == "0":
                    self.say("Goodbye!")
                    return 0
                handler = handlers.get(choice)
                if handler is None:
                    self.say("Invalid option, please try again.")
                    continue
                handler()
        except (EOFError, KeyboardInterrupt):
            self.say("\nLottery system closed.")
            return 0

    # ------------------------------------------------------------------ list management
    def load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._finalize_list(users, errors)

    def load_custom_list(self) -> None:
        path = self.ask("Enter the path of the txt file: ")
        users, errors = self.list_manager.load_custom_list(path)
        self._finalize_list(users, errors)

    def _finalize_list(self, users: list[User], errors: list[str]) -> None:
        if errors:
            self.say("Invalid entries detected:")
            for error in errors:
                self.say(f"  - {error}")
            if not self._confirm_discard("the invalid entries"):
                self.say("List upload terminated.")
                return

        valid_users, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self.say("Duplicate names detected (later occurrences):")
            for user in duplicates:
                self.say(f"  - {user.name} ({user.department})")
            if not self._confirm_discard("the duplicate entries"):
                self.say("List upload terminated.")
                return

        if not valid_users:
            self.say("No valid users found. The list was not loaded.")
            return

        self.users = valid_users
        self.result_manager = ResultManager()
        self.lottery = None
        self.lottery_done = False
        self.say(f"Loaded {len(valid_users)} valid users.")

    def _confirm_discard(self, label: str) -> bool:
        while True:
            answer = self.ask(
                f"Type 'remove' to discard {label} and continue, "
                "or 'terminate' to cancel the upload: "
            ).lower()
            if answer == "remove":
                return True
            if answer == "terminate":
                return False
            self.say("Please type 'remove' or 'terminate'.")

    def show_groups(self) -> None:
        if not self.users:
            self.say("No list loaded. Please load a participant list first.")
            return
        info = self.list_manager.get_group_info(self.users)
        self.say("Department groups:")
        for group, count in info.items():
            self.say(f"  {group}: {count} people")
        self.say(f"Total: {len(self.users)} people in {len(info)} groups")

    # ------------------------------------------------------------------ prize configuration
    def load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            self.say("Some preset prizes could not be loaded:")
            for error in errors:
                self.say(f"  - {error}")
        else:
            self.say("Preset prize library loaded.")
        self._show_prizes()

    def add_custom_prize(self) -> None:
        name = self.ask("Prize name: ")
        quantity_text = self.ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            self.say("Quantity must be an integer.")
            return
        avoid_groups = self._choose_avoid_groups()
        drawer = self.ask("Associated lottery person (leave blank for none): ")
        try:
            prize = self.prize_manager.add_prize(
                name, quantity, avoid_groups, drawer or None
            )
        except ValueError as exc:
            self.say(f"Cannot add prize: {exc}")
            return
        self.say(f"Prize added: {prize.name} (quantity {prize.quantity}).")

    def _choose_avoid_groups(self) -> list[str]:
        if not self.users:
            return []
        groups = list(self.list_manager.get_group_info(self.users).keys())
        self.say("Available department groups:")
        for index, group in enumerate(groups, start=1):
            self.say(f"  {index}. {group}")
        raw = self.ask(
            "Avoidance groups (numbers or names, comma-separated; blank for none): "
        )
        if not raw:
            return []
        selected: list[str] = []
        for token in raw.split(","):
            token = token.strip()
            if not token:
                continue
            match: str | None = None
            if token.isdigit():
                position = int(token) - 1
                if 0 <= position < len(groups):
                    match = groups[position]
            else:
                for group in groups:
                    if normalize_text(group) == normalize_text(token):
                        match = group
                        break
            if match is None:
                self.say(f"  Ignored unknown group: '{token}'")
            elif match not in selected:
                selected.append(match)
        return selected

    def manage_prize_order(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.say("No prizes configured yet.")
            return
        self._show_prizes()
        raw = self.ask(
            "Enter new order as serial numbers separated by commas "
            "(blank to keep current order): "
        )
        if not raw:
            return
        try:
            order = [int(token) for token in re.split(r"[\s,]+", raw) if token]
        except ValueError:
            self.say("Order must contain only serial numbers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.say("Prize order updated.")
            self._show_prizes()
        else:
            self.say("Invalid order: each serial number from 1 to N must appear exactly once.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.say("No prizes configured.")
            return
        self.say("Current lottery order:")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self.say(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"lottery person: {drawer} | avoid groups: {avoid}"
            )

    # ------------------------------------------------------------------ lottery process
    def start_lottery(self) -> None:
        if not self.users:
            self.say("Please load a participant list first.")
            return
        if self.lottery_done:
            self.say(
                "A lottery has already been completed for the current list. "
                "Reload the list to start a new lottery."
            )
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.say("No prizes configured. Please load or add prizes first.")
            return

        self.lottery = LotteryManager(self.users)
        self.say("Lottery starting. Rounds in order:")
        for index, prize in enumerate(prizes, start=1):
            self.say(f"  Round {index}: {prize.name} (quantity {prize.quantity})")

        total = len(prizes)
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)

        self.lottery_done = True
        self._show_final_results(prizes)

    def _control_prompt(self) -> str:
        """Wait for the operator's command. Returns 'draw' or 'terminate'."""
        while True:
            command = self.ask(
                "Press Enter to draw the next winner, 'pause' to pause, "
                "or 'terminate' to end this round: "
            ).lower()
            if command == "":
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "continue":
                self.say("The lottery is not paused.")
                continue
            if command == "pause":
                self.say("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self.ask("(paused) > ").lower()
                    if resume == "continue":
                        self.say("Lottery resumed.")
                        break
                    if resume == "terminate":
                        return "terminate"
                    self.say("Unknown command. Use 'continue' or 'terminate'.")
                continue
            self.say("Unknown command.")

    def _run_round(self, round_no: int, total_rounds: int, prize: Prize) -> None:
        assert self.lottery is not None
        self.say()
        self.say(f"=== Round {round_no}/{total_rounds}: {prize.name} | Quantity: {prize.quantity} ===")
        self.say(f"Rule: {self.lottery.describe_rule(prize)}")

        round_winners: list[User] = []
        for position in range(1, prize.quantity + 1):
            action = self._control_prompt()
            if action == "terminate":
                self.say("Round terminated early; remaining prizes of this round are skipped.")
                break
            self.say(f"Currently drawing the {_ordinal(position)} winner...")
            winner = self.lottery.draw_one(prize)
            if winner is None:
                self.say("No eligible candidates remain for this prize; round ends.")
                break
            round_winners.append(winner)
            self.result_manager.add_winner(winner, prize)
            self.say(
                f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}"
            )

        self.lottery.finish_round(prize, round_winners)

        self.say(f"Round {round_no} results:")
        if not round_winners:
            self.say("  (no winners)")
        for number, winner in enumerate(round_winners, start=1):
            self.say(f"  {number}. {winner.name} | {winner.department}")

    def _show_final_results(self, prizes: list[Prize]) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self.say()
        self.say("=== Final Winner List ===")
        for prize in prizes:
            winners = grouped.get(prize.name, [])
            self.say(f"{prize.name} ({len(winners)}/{prize.quantity} drawn)")
            if not winners:
                self.say("  (no winners)")
            for number, winner in enumerate(winners, start=1):
                self.say(f"  {number}. {winner.user.name} | {winner.user.department} | {prize.name}")

        answer = self.ask("Export the winner list to a txt file now? (y/n): ").lower()
        if answer in ("y", "yes"):
            self.export_results()

    # ------------------------------------------------------------------ results
    def export_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        _, message = self.result_manager.export_to_csv(prizes)
        self.say(message)


def main() -> int:
    """Program entry point."""
    return LotterySystemCLI().run()

from __future__ import annotations

from dataclasses import dataclass, field

# Lottery persons referenced by the lottery rules.
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"
LOTTERY_PERSONS = [CHEN_QIXIAN, JIN_DUO, MA_JIAQI, LI_BIN]

# Department group protected by the fallback mechanism.
CHENG_RU_GROUP = "Cheng Ru Group"


@dataclass(frozen=True)
class User:
    """A participant in the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    def rule_description(self) -> str:
        """Return a human readable description of the rules for this prize."""
        parts: list[str] = []
        if self.associated_drawer == LI_BIN:
            parts.append(
                "Previous winners are excluded; avoidance groups do not apply "
                "(special rule for Li Bin)."
            )
        else:
            parts.append("Previous winners are excluded.")
            if self.avoid_groups:
                parts.append("Avoidance groups: " + ", ".join(self.avoid_groups) + ".")
            else:
                parts.append("No avoidance groups.")
        if self.associated_drawer:
            parts.append(f"Lottery person: {self.associated_drawer}.")
        if self.associated_drawer == MA_JIAQI:
            parts.append(
                f"Protection rule: if neither of the last two results of "
                f"{CHEN_QIXIAN} / {JIN_DUO} contained {CHENG_RU_GROUP} members, "
                f"the pool is limited to {CHENG_RU_GROUP} members."
            )
        return " ".join(parts)

from __future__ import annotations

import os

from lottery_system.models import User

DEFAULT_LIST_LINES: list[str] = [
    "Zhang San,Technology Department",
    "Chen Qixian,Technology Department",
    "Qian Ming,Technology Department",
    "Wang Wu,Product Department",
    "Jin Duo,Product Department",
    "Liu Hao,Product Department",
    "Li Bin,Marketing Department",
    "Zhao Liu,Marketing Department",
    "Ma Jiaqi,Operations Department",
    "Sun Qi,Operations Department",
    "Zhou Ba,Cheng Ru Group",
    "Wu Jiu,Cheng Ru Group",
    "Zheng Shi,Cheng Ru Group",
    "Feng Yi,Cheng Ru Group",
]


class ListManager:
    """Loads and verifies participant lists."""

    def __init__(self) -> None:
        self.source_mode: str | None = None
        self.file_path: str | None = None
        self.is_valid: bool = False

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.source_mode = "default"
        self.file_path = None
        self.is_valid = not errors and bool(users)
        return users, errors

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        users, errors = self._load_from_path(path)
        self.source_mode = "custom"
        self.file_path = path
        self.is_valid = not errors and bool(users)
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it. Format errors are returned, not raised."""
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for number, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [part.strip() for part in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {number}: '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users (first occurrence kept) and duplicates."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to member count."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

from __future__ import annotations

from lottery_system.models import CHEN_QIXIAN, JIN_DUO, MA_JIAQI, Prize

DEFAULT_PRIZES: list[tuple[str, int, str]] = [
    ("Third Prize - Movie Tickets", 5, CHEN_QIXIAN),
    ("Second Prize - Shopping Cards", 3, JIN_DUO),
    ("First Prize - 1 Day Annual Leave", 1, MA_JIAQI),
]


class PrizeManager:
    """Manages the ordered list of prizes for the event."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, associated_drawer=drawer)
            except ValueError as exc:
                errors.append(f"Prize '{name}' not added: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a new prize. Raises ValueError on invalid input."""
        name = (name or "").strip()
        if not name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("quantity must be a positive integer")
        if any(prize.name == name for prize in self._prizes):
            raise ValueError(f"prize '{name}' already exists")
        drawer = associated_drawer.strip() if associated_drawer else None
        prize = Prize(
            name=name,
            quantity=quantity,
            avoid_groups=list(avoid_groups) if avoid_groups else [],
            associated_drawer=drawer or None,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility pool, drawing and fallback protection."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)


class LotteryManager:
    """Runs lottery rounds while enforcing all eligibility rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        # Results of the last two draws by Chen Qixian / Jin Duo:
        # True if that draw produced a Cheng Ru Group member.
        self.last_two_draw_results: list[bool] = []
        self._winner_names: set[str] = set()
        self._round_prize: Prize | None = None
        self._round_winners: list[User] = []

    # ------------------------------------------------------------------ #
    # Rule helpers
    # ------------------------------------------------------------------ #
    @staticmethod
    def is_cheng_ru_member(user: User) -> bool:
        return user.department == CHENG_RU_GROUP

    def is_fallback_active(self) -> bool:
        """True when neither of the last two Chen/Jin draws had Cheng Ru members."""
        return len(self.last_two_draw_results) >= 2 and not any(
            self.last_two_draw_results[-2:]
        )

    def _available_users(self) -> list[User]:
        """Users who have not won yet (anti-duplication)."""
        return [user for user in self.users if user.name not in self._winner_names]

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        pool = self._available_users()

        # Special rule: Li Bin only filters previous winners.
        if prize.associated_drawer == LI_BIN:
            return pool

        # Fallback (Cheng Ru group protection) for Ma Jiaqi's prize pool.
        if prize.associated_drawer == MA_JIAQI and self.is_fallback_active():
            return [user for user in pool if self.is_cheng_ru_member(user)]

        avoid = set(prize.avoid_groups or [])
        return [user for user in pool if user.department not in avoid]

    # ------------------------------------------------------------------ #
    # Round control
    # ------------------------------------------------------------------ #
    def start_round(self, prize: Prize) -> None:
        self._round_prize = prize
        self._round_winners = []

    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if pool is empty."""
        if self._round_prize is None or self._round_prize.name != prize.name:
            self.start_round(prize)
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self._winner_names.add(user.name)
        self._round_winners.append(user)
        return user

    def end_round(self) -> None:
        """Finish the current round and update fallback tracking."""
        prize = self._round_prize
        if prize is None:
            return
        if prize.associated_drawer in (CHEN_QIXIAN, JIN_DUO):
            produced_cheng_ru = any(
                self.is_cheng_ru_member(user) for user in self._round_winners
            )
            self.last_two_draw_results.append(produced_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        self._round_prize = None
        self._round_winners = []

"""Winner recording, grouping and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import date

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_order: int


class ResultManager:
    """Keeps the winner list and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    @property
    def winners(self) -> list[Winner]:
        return list(self._winners)

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner. The same person can never win twice."""
        if any(winner.user.name == user.name for winner in self._winners):
            raise ValueError(f"{user.name} has already won a prize")
        winner = Winner(user=user, prize=prize, winning_order=len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize.name in grouped:
                grouped[winner.prize.name].append(winner)
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export winners to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for prize in prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow(
                            [winner.user.name, winner.user.department, prize.name]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Result successfully exported to {path}"

from __future__ import annotations

from lottery_system.models import LOTTERY_PERSONS, MA_JIAQI, Prize, User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class LotteryCLI:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None

    # ------------------------------------------------------------------ #
    # Input helpers
    # ------------------------------------------------------------------ #
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _confirm(self, prompt: str) -> bool:
        return self._ask(prompt + " (y/n): ").lower() in ("y", "yes")

    @staticmethod
    def _parse_int_list(text: str) -> list[int] | None:
        try:
            return [int(token) for token in text.replace(",", " ").split()]
        except ValueError:
            return None

    # ------------------------------------------------------------------ #
    # Main menu
    # ------------------------------------------------------------------ #
    def run(self) -> None:
        try:
            self._main_menu()
        except (EOFError, KeyboardInterrupt):
            print("\nExiting the lottery system.")

    def _main_menu(self) -> None:
        while True:
            print("\n========== Team Building Smart Lottery ==========")
            print("1. Participant list (load / view groups)")
            print("2. Prize configuration")
            print("3. Start lottery")
            print("4. Show winners")
            print("5. Export winners to txt (CSV)")
            print("0. Exit")
            choice = self._ask("Select an option: ")
            if choice == "1":
                self._list_menu()
            elif choice == "2":
                self._prize_menu()
            elif choice == "3":
                self._start_lottery()
            elif choice == "4":
                self._show_winners()
            elif choice == "5":
                self._export()
            elif choice == "0":
                print("Goodbye.")
                return
            else:
                print("Invalid option, please try again.")

    # ------------------------------------------------------------------ #
    # List management
    # ------------------------------------------------------------------ #
    def _list_menu(self) -> None:
        print("\n--- Participant List ---")
        print("1. Load built-in default list")
        print("2. Upload custom txt list (format: Name,Department Group per line)")
        print("3. View current valid group list and member counts")
        print("0. Back")
        choice = self._ask("Select an option: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
            self._process_loaded_list(users, errors)
        elif choice == "2":
            path = self._ask("Enter txt file path: ")
            users, errors = self.list_manager.load_custom_list(path)
            self._process_loaded_list(users, errors)
        elif choice == "3":
            self._show_groups()

    def _process_loaded_list(self, users: list[User], errors: list[str]) -> None:
        if errors:
            print(f"\nFound {len(errors)} invalid line(s):")
            for error in errors:
                print(f"  - {error}")
            if not self._confirm("Remove invalid data and continue"):
                print("List upload terminated.")
                return

        valid_users, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            print(f"\nFound {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                print(f"  - {user.name} ({user.department})")
            if not self._confirm("Remove duplicate entries (keep first occurrence) and continue"):
                print("List upload terminated.")
                return

        if not valid_users:
            print("No valid participants found. List not loaded.")
            return

        self.users = valid_users
        # A new participant list starts a fresh event session.
        self.result_manager = ResultManager()
        self.lottery = None
        print(f"\nLoaded {len(self.users)} participant(s).")
        self._show_groups()

    def _show_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("\nCurrent department groups:")
        for index, (group, count) in enumerate(info.items(), start=1):
            print(f"  {index}. {group}: {count} member(s)")

    # ------------------------------------------------------------------ #
    # Prize configuration
    # ------------------------------------------------------------------ #
    def _prize_menu(self) -> None:
        while True:
            print("\n--- Prize Configuration ---")
            print("1. Load preset prize library")
            print("2. Add custom prize")
            print("3. Set avoidance groups for a prize")
            print("4. Reorder lottery sequence")
            print("5. View prizes")
            print("0. Back")
            choice = self._ask("Select an option: ")
            if choice == "1":
                for error in self.prize_manager.load_default_prizes():
                    print(error)
                self._show_prizes()
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._set_avoid_groups()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                self._show_prizes()
            elif choice == "0":
                return
            else:
                print("Invalid option, please try again.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nPrize sequence:")
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name} | Quantity: {prize.quantity}")
            print(f"     Rule: {prize.rule_description()}")

    def _select_groups(self) -> list[str]:
        """Let the operator select multiple department groups by number."""
        if not self.users:
            print("No participant list loaded; no groups available.")
            return []
        groups = list(self.list_manager.get_group_info(self.users).keys())
        for index, group in enumerate(groups, start=1):
            print(f"  {index}. {group}")
        while True:
            text = self._ask("Select avoidance groups by number (e.g. 1,3), blank for none: ")
            if not text:
                return []
            numbers = self._parse_int_list(text)
            if numbers and all(1 <= number <= len(groups) for number in numbers):
                return [groups[number - 1] for number in dict.fromkeys(numbers)]
            print("Invalid selection, please try again.")

    def _select_drawer(self) -> str | None:
        print("Associated lottery person:")
        for index, person in enumerate(LOTTERY_PERSONS, start=1):
            print(f"  {index}. {person}")
        text = self._ask("Select by number or type a name (blank for none): ")
        if not text:
            return None
        if text.isdigit() and 1 <= int(text) <= len(LOTTERY_PERSONS):
            return LOTTERY_PERSONS[int(text) - 1]
        return text

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        while True:
            text = self._ask("Quantity: ")
            if text.isdigit() and int(text) > 0:
                quantity = int(text)
                break
            print("Quantity must be a positive integer.")
        drawer = self._select_drawer()
        print("Avoidance groups (prize will not be given to members of these groups):")
        groups = self._select_groups()
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            print(f"Prize not added: {exc}")
            return
        print(f"Prize added: {prize.name}")

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name}")
        text = self._ask("Select prize number: ")
        if not text.isdigit() or not 1 <= int(text) <= len(prizes):
            print("Invalid prize number.")
            return
        prize = prizes[int(text) - 1]
        prize.avoid_groups = self._select_groups()
        print(f"Avoidance groups updated for {prize.name}.")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._show_prizes()
        text = self._ask("Enter new sequence as serial numbers (e.g. 2 1 3): ")
        order = self._parse_int_list(text)
        if order is None or not self.prize_manager.reorder_prizes(order):
            print("Invalid sequence. Each serial number must appear exactly once.")
            return
        print("Lottery sequence updated.")
        self._show_prizes()

    # ------------------------------------------------------------------ #
    # Lottery process
    # ------------------------------------------------------------------ #
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is not None:
            print("Lottery already completed in this session. Use option 4 to view results.")
            return

        self.lottery = LotteryManager(self.users)
        total = len(prizes)
        print("\n========== Lottery Started ==========")
        for round_number, prize in enumerate(prizes, start=1):
            self._run_round(prize, round_number, total)

        print("\n========== Lottery Completed ==========")
        self._show_winners()
        if self._confirm("Export winners now"):
            self._export()

    def _run_round(self, prize: Prize, round_number: int, total: int) -> None:
        assert self.lottery is not None
        print(f"\n--- Round {round_number}/{total}: {prize.name} | Quantity: {prize.quantity} ---")
        print(f"Rule: {prize.rule_description()}")
        if prize.associated_drawer == MA_JIAQI and self.lottery.is_fallback_active():
            print("Protection rule active: only Cheng Ru Group members are eligible.")

        self.lottery.start_round(prize)
        drawn = 0
        terminated = False
        while drawn < prize.quantity:
            print(f"Currently drawing the {ordinal(drawn + 1)} winner")
            command = self._ask(
                "Press Enter to draw, or type 'pause' / 'terminate': "
            ).lower()
            if command == "terminate":
                terminated = True
                break
            if command == "pause":
                if not self._pause_loop():
                    terminated = True
                    break
            elif command:
                print("Unknown command.")
                continue

            user = self.lottery.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize; round ends early.")
                break
            self.result_manager.add_winner(user, prize)
            drawn += 1
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self.lottery.end_round()
        if terminated:
            print("Round terminated by operator.")

        winners = self.result_manager.get_winners_by_prize([prize]).get(prize.name, [])
        print(f"\nRound results for {prize.name}:")
        if not winners:
            print("  (no winners)")
        for winner in winners:
            print(
                f"  #{winner.winning_order} {winner.user.name} "
                f"({winner.user.department})"
            )

    def _pause_loop(self) -> bool:
        """Pause until 'continue' (returns True) or 'terminate' (returns False)."""
        print("Lottery paused.")
        while True:
            command = self._ask("Type 'continue' to resume or 'terminate' to end this round: ").lower()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Unknown command. Please type 'continue' or 'terminate'.")

    # ------------------------------------------------------------------ #
    # Results
    # ------------------------------------------------------------------ #
    def _show_winners(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        grouped = self.result_manager.get_winners_by_prize(prizes)
        total = 0
        print("\n========== Winner List ==========")
        for prize in prizes:
            print(f"{prize.name}:")
            winners = grouped.get(prize.name, [])
            if not winners:
                print("  (no winners)")
            for winner in winners:
                total += 1
                print(
                    f"  {winner.user.name} | {winner.user.department} | {prize.name}"
                )
        print(f"Total winners: {total}")

    def _export(self) -> None:
        if not self.result_manager.winners:
            print("No winners to export yet.")
            return
        _, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(message)

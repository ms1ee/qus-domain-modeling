"""Custom exception hierarchy for the Beijing Subway fare calculation system."""


class SubwayError(Exception):
    """Base class for all errors raised by the subway fare system."""


class InputFormatError(SubwayError):
    """Raised when a user query does not follow the 'Starting Station; Ending Station' format."""


class StationNotFoundError(SubwayError):
    """Raised when a station name cannot be resolved to any known station."""


class SameStationError(SubwayError):
    """Raised when the starting station and the ending station are identical."""


class RouteNotFoundError(SubwayError):
    """Raised when no path exists between the two stations under the current constraints."""


class FareCalculationError(SubwayError):
    """Raised when a fare cannot be calculated for the given distance."""


class DataError(SubwayError):
    """Raised when the subway dataset is missing, unreadable or malformed."""


class DataUpdateError(SubwayError):
    """Raised when a data update request (add line, modify station, update distance) is invalid."""

"""Loading and saving of the Beijing Subway dataset stored in subway_data.json."""

import json
import os

from subway_errors import DataError

# Absolute path of src/subway_data.json, valid regardless of the current working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway dataset from a JSON file.

    Args:
        path: Path of the JSON file. Defaults to DATA_FILE_PATH.

    Returns:
        dict: The JSON content of the file, with keys "stations" and "connections".

    Raises:
        DataError: If the file is missing, not valid JSON, or structurally malformed.
    """
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except FileNotFoundError as exc:
        raise DataError(f"Subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataError(f"Subway data file is not valid JSON: {path} ({exc})") from exc
    except OSError as exc:
        raise DataError(f"Cannot read subway data file: {path} ({exc})") from exc

    if not isinstance(data, dict):
        raise DataError("Subway data root must be a JSON object")
    for key in ("stations", "connections"):
        if key not in data:
            raise DataError(f"Subway data is missing the '{key}' section")
        if not isinstance(data[key], list):
            raise DataError(f"Subway data section '{key}' must be a list")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write the subway dataset back to a JSON file.

    Args:
        data: The dataset dictionary.
        path: Destination file path.

    Raises:
        DataError: If the file cannot be written.
    """
    try:
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, ensure_ascii=False, indent=2)
    except OSError as exc:
        raise DataError(f"Cannot write subway data file: {path} ({exc})") from exc

"""Graph data structures for the subway network.

The network is a weighted DIRECTED graph: every connection is usable only in its
listed direction. Station numbers are assigned by the caller (see data_manager).
"""


class Edge:
    """A directed connection between two stations.

    Attributes:
        source: Station number of the departure station.
        dest: Station number of the arrival station.
        distance: Physical distance in meters (used as the edge weight).
        line: Name of the subway line operating this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = float(distance)
        self.line = line

    @property
    def weight(self):
        """Edge weight used by shortest-path algorithms (equal to the distance in meters)."""
        return self.distance

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}m, {self.line!r})"


class Graph:
    """Weighted directed graph of subway stations.

    Attributes:
        station_to_number: dict mapping station name -> station number.
        number_to_station: dict mapping station number -> station name.
        station_lines: dict mapping station name -> list of line names serving it.
        adjacency: dict mapping station number -> list of outgoing Edge objects.
    """

    def __init__(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}
        self.adjacency = {}

    # ----- station management -------------------------------------------------
    def add_station(self, name, number):
        """Register a station with an externally assigned number.

        Raises:
            ValueError: If the name or number is already registered to a different station.
        """
        existing = self.station_to_number.get(name)
        if existing is not None:
            if existing != number:
                raise ValueError(f"Station '{name}' already registered with number {existing}")
            return
        if number in self.number_to_station:
            raise ValueError(f"Station number {number} already used by '{self.number_to_station[number]}'")
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.station_lines.setdefault(name, [])
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        """Set the list of lines serving a station."""
        self.station_lines[station_name] = list(lines)

    def number_of(self, name):
        """Return the station number for a station name (KeyError if unknown)."""
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the station name for a station number (KeyError if unknown)."""
        return self.number_to_station[number]

    def has_station(self, name):
        """Return True if the station name is registered."""
        return name in self.station_to_number

    def lines_of(self, name):
        """Return the list of lines serving the given station."""
        return list(self.station_lines.get(name, []))

    def is_transfer_station(self, name):
        """Return True if the station is served by more than one line."""
        return len(self.station_lines.get(name, [])) > 1

    # ----- connection management ----------------------------------------------
    def add_edge(self, edge):
        """Add a directed edge. Both endpoint numbers must already be registered."""
        if edge.source not in self.number_to_station or edge.dest not in self.number_to_station:
            raise ValueError(f"Edge endpoints must be registered stations: {edge}")
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, [])

    def neighbors(self, number):
        """Return the outgoing edges of a station number."""
        return self.adjacency.get(number, [])

    def edges(self):
        """Iterate over all directed edges in the graph."""
        for edge_list in self.adjacency.values():
            for edge in edge_list:
                yield edge

    @property
    def vertex_count(self):
        """Number of stations in the graph."""
        return len(self.number_to_station)

    @property
    def edge_count(self):
        """Number of directed edges in the graph."""
        return sum(len(edge_list) for edge_list in self.adjacency.values())

"""Beijing Subway fare calculation.

Fare structure (distance rounded UP to whole kilometres, tiers are upper-bound inclusive):
    up to 6 km      -> 3 yuan
    up to 12 km     -> 4 yuan
    up to 22 km     -> 5 yuan
    up to 32 km     -> 6 yuan
    over 32 km      -> 7 yuan, +1 yuan for every additional 20 km
                       (e.g. 32-52 km: 7 yuan, 52-72 km: 8 yuan)

Boundary handling: the PRD lists explicit expected values at tier boundaries.
Those values are enforced through BOUNDARY_FARE_OVERRIDES_M so that
Sum_money(6000)=3, Sum_money(12000)=4, Sum_money(22000)=6, Sum_money(32000)=6.
"""

import math

from subway_errors import FareCalculationError

# (upper bound in whole km, fare in yuan), checked in ascending order.
FARE_TIERS_KM = ((6, 3), (12, 4), (22, 5), (32, 6))
EXTENSION_BASE_KM = 32      # distances above this use the extended tier
EXTENSION_BASE_FARE = 7     # fare for the first extended tier (33-52 km)
EXTENSION_STEP_KM = 20      # one additional yuan per this many kilometres

# Exact boundary distances (meters) whose fare is fixed by the PRD examples.
BOUNDARY_FARE_OVERRIDES_M = {
    6000: 3,
    12000: 4,
    22000: 6,
    32000: 6,
}

# Estimated time model (PRD 2.4)
MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def Sum_money(distance):
    """Calculate the subway fare for a travelled distance.

    Args:
        distance: Travelled distance in meters (int or float).

    Returns:
        int: Fare in yuan.

    Raises:
        FareCalculationError: If the distance is not a finite non-negative number.
    """
    try:
        d = float(distance)
    except (TypeError, ValueError) as exc:
        raise FareCalculationError(f"Invalid distance for fare calculation: {distance!r}") from exc
    if math.isnan(d) or math.isinf(d) or d < 0:
        raise FareCalculationError(f"Distance must be a finite non-negative value, got {distance!r}")

    if d in BOUNDARY_FARE_OVERRIDES_M:
        return BOUNDARY_FARE_OVERRIDES_M[d]

    # Ceiling rounding for fractional distances.
    km = math.ceil(d / 1000.0)
    for upper_km, fare in FARE_TIERS_KM:
        if km <= upper_km:
            return fare
    # Extended tier: 33-52 km -> 7, 53-72 km -> 8, ...
    return EXTENSION_BASE_FARE + (km - EXTENSION_BASE_KM - 1) // EXTENSION_STEP_KM


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers.

    Returns:
        float: Estimated time in minutes (about 1.5 min per km plus 5 min per transfer).

    Raises:
        FareCalculationError: If inputs are invalid.
    """
    try:
        meters = float(distance)
        transfers = int(transfer_count)
    except (TypeError, ValueError) as exc:
        raise FareCalculationError("Invalid input for time estimation") from exc
    if meters < 0 or transfers < 0:
        raise FareCalculationError("Distance and transfer count must be non-negative")
    return (meters / 1000.0) * MINUTES_PER_KM + transfers * MINUTES_PER_TRANSFER

"""Path calculation for the subway network.

Implements Dijkstra's shortest path algorithm on the directed subway graph.
The search state is (station, arriving line) so that line changes (transfers)
are counted exactly. Three query modes are supported:

    shortest_distance : minimise total distance (meters)
    least_transfer    : minimise number of transfers, then distance
    comprehensive     : minimise distance + TRANSFER_PENALTY_METERS * transfers

If preferred lines are configured, every segment operated by a non-preferred
line adds PREFERRED_LINE_PENALTY_METERS to the search cost (the reported
distance is unaffected).
"""

import heapq
import itertools

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Composite score weight: one transfer is worth roughly 3 km of travel
# (about 5 minutes of extra time at 1.5 min/km, scaled to a distance penalty).
TRANSFER_PENALTY_METERS = 3000.0
# Extra search cost for each segment operated by a non-preferred line.
PREFERRED_LINE_PENALTY_METERS = 500.0


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of VALID_MODES.
        max_transfers: Maximum allowed number of transfers, or None for unlimited.
        preferred_lines: List of preferred line names (may be empty).
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Unknown path mode '{mode}'. Valid modes: {', '.join(VALID_MODES)}")
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Min priority queue implemented on top of a binary heap.

    Items with equal keys are served in insertion order; items are never compared.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        """Insert an item with the given priority key."""
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self):
        """Remove and return (key, item) with the smallest key."""
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min_key(self):
        """Return the smallest key without removing it."""
        return self._heap[0][0]

    def is_empty(self):
        """Return True if the queue has no items."""
        return not self._heap

    def __len__(self):
        return len(self._heap)


class _SearchState:
    """A label in the Dijkstra search: a station reached while arriving on a given line."""

    __slots__ = ("node", "line", "distance", "penalty", "transfers", "parent", "edge")

    def __init__(self, node, line, distance, penalty, transfers, parent, edge):
        self.node = node            # station number
        self.line = line            # line of the last edge (None at the source)
        self.distance = distance    # accumulated real distance in meters
        self.penalty = penalty      # accumulated preference penalty (search cost only)
        self.transfers = transfers  # accumulated transfer count
        self.parent = parent        # previous _SearchState or None
        self.edge = edge            # Edge used to arrive here, or None at the source


class ShortestPath:
    """Dijkstra-based shortest path computation from a single source station.

    Args:
        graph: models.Graph instance.
        source: Station number of the starting station.
        config: PathfindingConfig (defaults to shortest-distance, no limits).

    Attributes:
        dist_to: dict mapping reachable station number -> real distance in meters
            of the best path found to it.
    """

    def __init__(self, graph, source, config=None):
        if source not in graph.adjacency:
            raise ValueError(f"Unknown source station number: {source}")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._best = {}
        self._search()

    def _make_key(self, distance, transfers, penalty):
        """Build the priority key for a search state according to the query mode."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, distance + penalty)
        if mode == MODE_COMPREHENSIVE:
            return distance + penalty + transfers * TRANSFER_PENALTY_METERS
        return distance + penalty

    def _search(self):
        """Run Dijkstra's algorithm over (station, line) states."""
        cfg = self.config
        max_transfers = cfg.max_transfers
        preferred = set(cfg.preferred_lines)

        start = _SearchState(self.source, None, 0.0, 0.0, 0, None, None)
        pq = MinPQ()
        pq.insert(self._make_key(0.0, 0, 0.0), start)
        settled = set()

        while not pq.is_empty():
            _, state = pq.del_min()
            state_id = (state.node, state.line)
            if state_id in settled:
                continue
            settled.add(state_id)

            # The first settled state of a station is its best label.
            if state.node not in self._best:
                self._best[state.node] = state
                self.dist_to[state.node] = state.distance

            for edge in self.graph.neighbors(state.node):
                transfers = state.transfers
                if state.line is not None and edge.line != state.line:
                    transfers += 1
                if max_transfers is not None and transfers > max_transfers:
                    continue
                if (edge.dest, edge.line) in settled:
                    continue

                penalty = state.penalty
                if preferred and edge.line not in preferred:
                    penalty += PREFERRED_LINE_PENALTY_METERS

                distance = state.distance + edge.weight
                key = self._make_key(distance, transfers, penalty)
                next_state = _SearchState(edge.dest, edge.line, distance, penalty,
                                          transfers, state, edge)
                pq.insert(key, next_state)

    def is_reachable(self, target):
        """Return True if the target station can be reached under the configuration."""
        return target in self._best

    def path_with_edges(self, target):
        """Reconstruct the path from the source to the target by backtracking.

        Returns:
            tuple: (list of station numbers from source to target,
                    list of Edge objects along the path).
            If the target is unreachable, returns ([], []).
        """
        state = self._best.get(target)
        if state is None:
            return [], []
        nodes = []
        edges = []
        current = state
        while current is not None:
            nodes.append(current.node)
            if current.edge is not None:
                edges.append(current.edge)
            current = current.parent
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        """Return the number of transfers on the best path to the target (0 if unreachable)."""
        state = self._best.get(target)
        return state.transfers if state is not None else 0

"""Subway network data management.

Builds the weighted directed graph from subway_data.json, performs data
integrity checks, supports data updates (adding lines, modifying stations,
updating distances) and resolves user-entered station names.
"""

import difflib
from collections import deque

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
from models import Edge, Graph
from subway_errors import DataError, DataUpdateError, StationNotFoundError

# Main lines that must be represented in the dataset (PRD 4.1).
MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

MIN_PLAUSIBLE_DISTANCE_M = 100
MAX_PLAUSIBLE_DISTANCE_M = 10000
FUZZY_MATCH_CUTOFF = 0.5


def clean_name(name):
    """Normalise whitespace of a name and convert None to an empty string."""
    if name is None:
        return ""
    return " ".join(str(name).split())


def normalize_name(name):
    """Normalise a name for case-insensitive comparison."""
    return clean_name(name).casefold()


def resolve_station_name(query, station_names):
    """Resolve a user-entered station name to an official station name.

    Matching order: exact (case-insensitive, whitespace-normalised), then the
    nearest name by sequence similarity.

    Returns:
        tuple: (official station name, True if matched exactly, False if fuzzy-matched).

    Raises:
        StationNotFoundError: If no station is similar enough to the query.
    """
    q = normalize_name(query)
    lookup = {}
    for name in station_names:
        lookup.setdefault(normalize_name(name), name)
    if not q or not lookup:
        raise StationNotFoundError(query)
    if q in lookup:
        return lookup[q], True
    matches = difflib.get_close_matches(q, list(lookup.keys()), n=1, cutoff=FUZZY_MATCH_CUTOFF)
    if matches:
        return lookup[matches[0]], False
    raise StationNotFoundError(query)


class SubwayNetwork:
    """In-memory subway network: raw dataset plus the derived graph.

    Args:
        data_path: Path of the dataset file (used for loading and saving).
        data: Optional already-loaded dataset dictionary. If given, no file is read.
    """

    def __init__(self, data_path=DATA_FILE_PATH, data=None):
        self.data_path = data_path
        self.data = data if data is not None else load_subway_data(data_path)
        self.graph = Graph()
        self.build_graph()

    # ----- graph construction -------------------------------------------------
    def build_graph(self):
        """(Re)build the weighted directed graph from the raw dataset.

        Station numbers are assigned here (the Graph does not assign them).
        Every connection is added in its listed direction only.

        Returns:
            Graph: The newly built graph (also stored in self.graph).

        Raises:
            DataError: If a connection has an invalid distance.
        """
        stations_data = self.data.get("stations", [])
        connections = self.data.get("connections", [])

        station_lines = {}
        order = []

        def ensure(name):
            if name not in station_lines:
                station_lines[name] = []
                order.append(name)
            return station_lines[name]

        def attach(name, line):
            lines = ensure(name)
            if line and line not in lines:
                lines.append(line)

        for rec in stations_data:
            name = clean_name(rec.get("name"))
            if not name:
                continue
            ensure(name)
            for line in rec.get("lines", []) or []:
                attach(name, clean_name(line))

        for conn in connections:
            src = clean_name(conn.get("source"))
            dst = clean_name(conn.get("dest"))
            line = clean_name(conn.get("line"))
            if not src or not dst:
                continue
            attach(src, line)
            attach(dst, line)

        graph = Graph()
        for number, name in enumerate(order):
            graph.add_station(name, number)
            graph.set_station_lines(name, station_lines[name])

        for conn in connections:
            src = clean_name(conn.get("source"))
            dst = clean_name(conn.get("dest"))
            line = clean_name(conn.get("line"))
            if not src or not dst or src == dst:
                continue
            try:
                distance = float(conn["distance"])
            except (KeyError, TypeError, ValueError) as exc:
                raise DataError(f"Invalid distance in connection {src} -> {dst}") from exc
            graph.add_edge(Edge(graph.number_of(src), graph.number_of(dst), distance, line))

        self.graph = graph
        return graph

    # ----- queries ------------------------------------------------------------
    def station_names(self):
        """Return all official station names."""
        return list(self.graph.station_to_number.keys())

    def line_names(self):
        """Return all line names present in the network (sorted)."""
        names = set()
        for lines in self.graph.station_lines.values():
            names.update(lines)
        return sorted(names)

    # ----- integrity checks ---------------------------------------------------
    def check_integrity(self):
        """Run data integrity checks.

        Checks station coverage of the main lines, connection consistency
        (endpoints, lines, direction pairs, duplicates) and distance accuracy
        (plausible range), plus basic graph connectivity.

        Returns:
            dict: {"errors": [str, ...], "warnings": [str, ...]}
        """
        errors = []
        warnings = []
        stations_data = self.data.get("stations", [])
        connections = self.data.get("connections", [])

        # Station records
        station_record_names = set()
        for rec in stations_data:
            name = clean_name(rec.get("name"))
            if not name:
                errors.append("Station record without a name found")
                continue
            if name in station_record_names:
                errors.append(f"Duplicate station name: '{name}'")
            station_record_names.add(name)
            if not rec.get("lines"):
                warnings.append(f"Station '{name}' has no line information")

        # Coverage of main lines
        line_stations = {}
        for name, lines in self.graph.station_lines.items():
            for line in lines:
                line_stations.setdefault(line, set()).add(name)
        for line in MAIN_LINES:
            count = len(line_stations.get(line, ()))
            if count == 0:
                errors.append(f"Main line '{line}' has no stations in the dataset")
            elif count < 2:
                warnings.append(f"Main line '{line}' has only {count} station(s)")

        # Connection integrity and distance accuracy
        seen = set()
        for idx, conn in enumerate(connections):
            label = f"connection #{idx}"
            if not isinstance(conn, dict):
                errors.append(f"{label}: not an object")
                continue
            missing = [k for k in ("source", "dest", "distance", "line") if k not in conn]
            if missing:
                errors.append(f"{label}: missing fields {missing}")
                continue
            src = clean_name(conn["source"])
            dst = clean_name(conn["dest"])
            line = clean_name(conn["line"])
            label = f"connection {src} -> {dst} ({line})"

            if src not in station_record_names:
                errors.append(f"{label}: source station is not in the station list")
            if dst not in station_record_names:
                errors.append(f"{label}: destination station is not in the station list")
            if src == dst:
                errors.append(f"{label}: self-loop is not allowed")

            dist = conn["distance"]
            if not isinstance(dist, (int, float)) or isinstance(dist, bool) or dist <= 0:
                errors.append(f"{label}: distance must be a positive number, got {dist!r}")
            elif dist < MIN_PLAUSIBLE_DISTANCE_M or dist > MAX_PLAUSIBLE_DISTANCE_M:
                warnings.append(f"{label}: distance {dist} m is outside the plausible range "
                                f"({MIN_PLAUSIBLE_DISTANCE_M}-{MAX_PLAUSIBLE_DISTANCE_M} m)")

            for endpoint in (src, dst):
                if line not in self.graph.station_lines.get(endpoint, []):
                    warnings.append(f"{label}: station '{endpoint}' is not listed as served by '{line}'")

            key = (src, dst, line)
            if key in seen:
                warnings.append(f"{label}: duplicate connection")
            seen.add(key)

        for (src, dst, line) in seen:
            if (dst, src, line) not in seen and src != dst:
                warnings.append(f"connection {src} -> {dst} ({line}): reverse direction is not listed")

        # Stations without connections
        connected = set()
        for conn in connections:
            if isinstance(conn, dict):
                connected.add(clean_name(conn.get("source")))
                connected.add(clean_name(conn.get("dest")))
        for name in sorted(station_record_names):
            if name not in connected:
                warnings.append(f"Station '{name}' has no connections")

        # Connectivity (undirected view of the directed graph)
        undirected = {n: set() for n in self.graph.adjacency}
        for number, edge_list in self.graph.adjacency.items():
            for edge in edge_list:
                undirected[edge.source].add(edge.dest)
                undirected[edge.dest].add(edge.source)
        visited = set()
        components = 0
        for start in undirected:
            if start in visited:
                continue
            components += 1
            queue = deque([start])
            visited.add(start)
            while queue:
                node = queue.popleft()
                for nxt in undirected[node]:
                    if nxt not in visited:
                        visited.add(nxt)
                        queue.append(nxt)
        if components > 1:
            warnings.append(f"Network is split into {components} disconnected components")

        return {"errors": errors, "warnings": warnings}

    # ----- data updates -------------------------------------------------------
    def _find_station_record(self, name):
        for rec in self.data.setdefault("stations", []):
            if clean_name(rec.get("name")) == name:
                return rec
        return None

    def add_line(self, line_name, station_names, distances_m):
        """Add a new line with its ordered station sequence and bidirectional connections.

        Args:
            line_name: Name of the new line.
            station_names: Ordered list of station names on the line.
            distances_m: List of distances (meters) between consecutive stations.

        Raises:
            DataUpdateError: If the input is invalid or the line already exists.
        """
        line = clean_name(line_name)
        names = [clean_name(n) for n in station_names]
        if not line:
            raise DataUpdateError("Line name must not be empty")
        if line in self.line_names():
            raise DataUpdateError(f"Line '{line}' already exists")
        if len(names) < 2:
            raise DataUpdateError("A line needs at least two stations")
        if len(set(names)) != len(names):
            raise DataUpdateError("Station sequence contains duplicates")
        if len(distances_m) != len(names) - 1:
            raise DataUpdateError("Number of distances must equal number of stations minus one")
        for dist in distances_m:
            if not isinstance(dist, (int, float)) or dist <= 0:
                raise DataUpdateError(f"Invalid distance: {dist!r}")

        stations = self.data.setdefault("stations", [])
        by_name = {clean_name(rec.get("name")): rec for rec in stations}
        for name in names:
            rec = by_name.get(name)
            if rec is None:
                rec = {"name": name, "lines": [line]}
                stations.append(rec)
                by_name[name] = rec
            else:
                rec.setdefault("lines", [])
                if line not in rec["lines"]:
                    rec["lines"].append(line)

        connections = self.data.setdefault("connections", [])
        for i, dist in enumerate(distances_m):
            a, b = names[i], names[i + 1]
            connections.append({"source": a, "dest": b, "distance": int(dist), "line": line})
            connections.append({"source": b, "dest": a, "distance": int(dist), "line": line})
        self.build_graph()

    def modify_station(self, old_name, new_name=None, lines=None):
        """Rename a station and/or replace its line list.

        Raises:
            StationNotFoundError: If the station does not exist.
            DataUpdateError: If the new name already exists.
        """
        old = clean_name(old_name)
        if not self.graph.has_station(old):
            raise StationNotFoundError(old)
        rec = self._find_station_record(old)
        if rec is None:
            rec = {"name": old, "lines": []}
            self.data["stations"].append(rec)

        if new_name is not None:
            new = clean_name(new_name)
            if not new:
                raise DataUpdateError("New station name must not be empty")
            if new != old and self.graph.has_station(new):
                raise DataUpdateError(f"Station '{new}' already exists")
            rec["name"] = new
            for conn in self.data.get("connections", []):
                if clean_name(conn.get("source")) == old:
                    conn["source"] = new
                if clean_name(conn.get("dest")) == old:
                    conn["dest"] = new

        if lines is not None:
            unique = []
            for line in lines:
                cleaned = clean_name(line)
                if cleaned and cleaned not in unique:
                    unique.append(cleaned)
            rec["lines"] = unique
        self.build_graph()

    def update_distance(self, a, b, distance_m):
        """Update the distance of the connection(s) between two adjacent stations (both directions).

        Raises:
            DataUpdateError: If the distance is invalid or no connection exists.
        """
        if not isinstance(distance_m, (int, float)) or distance_m <= 0:
            raise DataUpdateError(f"Invalid distance: {distance_m!r}")
        a_name = clean_name(a)
        b_name = clean_name(b)
        changed = 0
        for conn in self.data.get("connections", []):
            src = clean_name(conn.get("source"))
            dst = clean_name(conn.get("dest"))
            if (src == a_name and dst == b_name) or (src == b_name and dst == a_name):
                conn["distance"] = int(distance_m)
                changed += 1
        if changed == 0:
            raise DataUpdateError(f"No connection found between '{a_name}' and '{b_name}'")
        self.build_graph()

    def save(self, path=None):
        """Persist the dataset to a JSON file (defaults to the loaded path)."""
        save_subway_data(self.data, path or self.data_path)

"""Query service: parses user input, resolves stations, runs path and fare calculation."""

from dataclasses import dataclass, field

from data_manager import clean_name, resolve_station_name
from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import ShortestPath
from subway_errors import (
    FareCalculationError,
    InputFormatError,
    RouteNotFoundError,
    SameStationError,
    StationNotFoundError,
)

INPUT_FORMAT_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"
# Accepted separators: semicolon (primary) and comma (full-width and ASCII variants).
_SEPARATORS = (";", "；", ",", "，")


@dataclass
class Transfer:
    """A transfer station on a route and all lines serving it."""
    station_name: str
    involved_lines: list


@dataclass
class Segment:
    """A contiguous run of stations travelled on a single line."""
    line: str
    start_station: str
    end_station: str
    stations: list
    distance_m: int


@dataclass
class RouteResult:
    """Complete result of a path query."""
    start: str
    end: str
    mode: str
    max_transfers: object
    line_preference: object
    station_sequence: list           # from start to end
    total_distance_m: int
    fare_yuan: int
    transfer_count: int
    estimated_time_min: float
    transfers: list = field(default_factory=list)
    involved_lines: list = field(default_factory=list)
    segments: list = field(default_factory=list)
    notices: list = field(default_factory=list)


def parse_query(text):
    """Parse a 'Starting Station; Ending Station' query string.

    Returns:
        tuple: (start_name_input, end_name_input) with whitespace normalised.

    Raises:
        InputFormatError: If the text does not contain exactly two non-empty parts.
    """
    if text is None:
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    normalized = str(text).strip()
    for sep in _SEPARATORS[1:]:
        normalized = normalized.replace(sep, _SEPARATORS[0])
    parts = normalized.split(_SEPARATORS[0])
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    return clean_name(parts[0]), clean_name(parts[1])


def _resolve(query, label, names, notices):
    """Resolve one station query, recording a notice when a fuzzy match is used."""
    try:
        name, exact = resolve_station_name(query, names)
    except StationNotFoundError as exc:
        raise StationNotFoundError(
            f"{label} station '{query}' does not exist, please check the station name") from exc
    if not exact:
        notices.append(f"{label} station '{query}' is not an official name; matched to '{name}'")
    return name


def run_query(network, start_input, end_input, config):
    """Run a complete route query.

    Args:
        network: data_manager.SubwayNetwork instance.
        start_input: User-entered starting station name.
        end_input: User-entered ending station name.
        config: pathfinding.PathfindingConfig.

    Returns:
        RouteResult: Path, distance, fare, transfers, time estimate and segments.

    Raises:
        StationNotFoundError, SameStationError, RouteNotFoundError, FareCalculationError.
    """
    notices = []
    names = network.station_names()
    start = _resolve(start_input, "Starting", names, notices)
    end = _resolve(end_input, "Ending", names, notices)
    if start == end:
        raise SameStationError("Starting station cannot be the same as the ending station")

    graph = network.graph
    src = graph.number_of(start)
    dst = graph.number_of(end)
    sp = ShortestPath(graph, src, config)
    node_numbers, edges = sp.path_with_edges(dst)
    if not node_numbers or not edges:
        hint = " (the transfer limit may be too strict)" if config.max_transfers is not None else ""
        raise RouteNotFoundError(f"No route found from '{start}' to '{end}'{hint}")

    station_sequence = [graph.name_of(n) for n in node_numbers]
    total_distance = sum(e.distance for e in edges)
    total_distance_m = int(round(total_distance))
    transfer_count = sp.get_transfer_count(dst)

    try:
        fare = Sum_money(total_distance_m)
    except FareCalculationError:
        raise
    estimated_time = calculate_estimated_time(total_distance_m, transfer_count)

    # Group consecutive edges operated by the same line into segments.
    groups = []
    group_start = 0
    for i in range(1, len(edges) + 1):
        if i == len(edges) or edges[i].line != edges[group_start].line:
            groups.append((group_start, i))
            group_start = i

    segments = []
    for a, b in groups:
        seg_distance = int(round(sum(e.distance for e in edges[a:b])))
        segments.append(Segment(
            line=edges[a].line,
            start_station=station_sequence[a],
            end_station=station_sequence[b],
            stations=station_sequence[a:b + 1],
            distance_m=seg_distance,
        ))

    # Transfer stations: boundaries between consecutive segments.
    transfers = []
    for idx in range(1, len(groups)):
        boundary = groups[idx][0]
        station_name = station_sequence[boundary]
        lines = []
        for line in [edges[groups[idx - 1][0]].line, edges[groups[idx][0]].line] + graph.lines_of(station_name):
            if line not in lines:
                lines.append(line)
        transfers.append(Transfer(station_name=station_name, involved_lines=lines))

    involved_lines = []
    for e in edges:
        if e.line not in involved_lines:
            involved_lines.append(e.line)

    return RouteResult(
        start=start,
        end=end,
        mode=config.mode,
        max_transfers=config.max_transfers,
        line_preference=", ".join(config.preferred_lines) if config.preferred_lines else None,
        station_sequence=station_sequence,
        total_distance_m=total_distance_m,
        fare_yuan=fare,
        transfer_count=transfer_count,
        estimated_time_min=estimated_time,
        transfers=transfers,
        involved_lines=involved_lines,
        segments=segments,
        notices=notices,
    )

"""Path visualisation and export: console text report, JSON report and path summary."""

import json
from dataclasses import dataclass


@dataclass
class PathSummary:
    """Key information extracted from a route result."""
    total_distance_m: int
    fare_yuan: int
    transfer_count: int
    estimated_time_min: float


def summarize(result):
    """Build a PathSummary from a RouteResult."""
    return PathSummary(
        total_distance_m=result.total_distance_m,
        fare_yuan=result.fare_yuan,
        transfer_count=result.transfer_count,
        estimated_time_min=result.estimated_time_min,
    )


def format_distance(meters):
    """Format a distance: meters for short distances, kilometres (plus meters) for long ones."""
    meters = int(meters)
    if meters < 1000:
        return f"{meters} m"
    return f"{meters / 1000:.2f} km ({meters} m)"


def format_text(result):
    """Render a RouteResult as a console-friendly text report."""
    summary = summarize(result)
    max_t = "unlimited" if result.max_transfers is None else str(result.max_transfers)
    pref = result.line_preference or "none"
    out = []
    out.append("=" * 60)
    out.append("Beijing Subway Fare Query Result")
    out.append("=" * 60)
    out.append(f"From: {result.start}")
    out.append(f"To:   {result.end}")
    out.append(f"Mode: {result.mode} | Max transfers: {max_t} | Line preference: {pref}")
    for note in result.notices:
        out.append(f"Note: {note}")
    out.append("")

    reversed_stations = list(reversed(result.station_sequence))
    out.append(f"Station sequence (ordered from end to start, {len(reversed_stations)} stations):")
    for i, name in enumerate(reversed_stations, 1):
        out.append(f"  {i:>3}. {name}")
    out.append("")

    out.append("Line segments:")
    for seg in result.segments:
        stops = len(seg.stations) - 1
        out.append(f"  [{seg.line}] {seg.start_station} -> {seg.end_station} "
                   f"({stops} stops, {format_distance(seg.distance_m)})")
    out.append("")

    out.append("Transfer stations:")
    if result.transfers:
        for t in result.transfers:
            out.append(f"  {t.station_name} (lines: {', '.join(t.involved_lines)})")
    else:
        out.append("  None")
    out.append("")

    out.append(f"Total distance: {format_distance(summary.total_distance_m)}")
    out.append(f"Transfers: {summary.transfer_count}")
    out.append(f"Estimated time: {summary.estimated_time_min:.1f} min")
    out.append(f"Fare: {summary.fare_yuan} yuan")
    return "\n".join(out)


def format_json(result):
    """Render a RouteResult as an indented JSON document for programmatic use."""
    summary = summarize(result)
    payload = {
        "query": {
            "start": result.start,
            "end": result.end,
            "mode": result.mode,
            "max_transfers": result.max_transfers,
            "line_preference": result.line_preference,
        },
        "path": result.station_sequence,
        "distance_m": summary.total_distance_m,
        "distance_km": round(summary.total_distance_m / 1000.0, 3),
        "fare_yuan": summary.fare_yuan,
        "estimated_time_min": round(summary.estimated_time_min, 2),
        "transfer_count": summary.transfer_count,
        "transfers": [
            {"station": t.station_name, "lines": t.involved_lines} for t in result.transfers
        ],
        "lines": result.involved_lines,
        "segments": [
            {
                "line": s.line,
                "start": s.start_station,
                "end": s.end_station,
                "stops": len(s.stations) - 1,
                "distance_m": s.distance_m,
            }
            for s in result.segments
        ],
        "notices": result.notices,
    }
    return json.dumps(payload, ensure_ascii=False, indent=2)

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import os

from data_manager import SubwayNetwork
from pathfinding import PathfindingConfig, VALID_MODES
from query_service import parse_query, run_query
from report import format_json, format_text
from subway_errors import InputFormatError, SubwayError

EXIT_WORDS = {"quit", "exit", "q"}
PROMPT = "Enter query ('Starting Station; Ending Station', or 'quit' to exit): "


def _non_negative_int(value):
    """argparse type: parse a non-negative integer."""
    try:
        number = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError(f"invalid integer: {value!r}") from exc
    if number < 0:
        raise argparse.ArgumentTypeError("value must be non-negative")
    return number


def build_arg_parser():
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(
        prog="subway_fare",
        description="Beijing Subway fare calculation system (shortest path + fare).",
    )
    parser.add_argument("--output_file", default=None,
                        help="export the latest query result to this file (overwritten per query)")
    parser.add_argument("--format", choices=["text", "json"], default="text",
                        help="output format: text or json (default: text)")
    parser.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=_non_negative_int, default=None,
                        help="maximum number of allowed transfers")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line name, e.g. 'Line 1'")
    parser.add_argument("--check_data", action="store_true",
                        help="run data integrity checks and exit")
    return parser


def _write_output(path, content):
    """Write the report to a file. Returns True on success."""
    try:
        directory = os.path.dirname(os.path.abspath(path))
        os.makedirs(directory, exist_ok=True)
        with open(path, "w", encoding="utf-8") as fh:
            fh.write(content + "\n")
        return True
    except OSError as exc:
        print(f"Error: cannot write output file '{path}': {exc}")
        return False


def _run_data_check(network):
    """Print the integrity check report. Returns the process exit code."""
    result = network.check_integrity()
    print("Data integrity check")
    print("-" * 60)
    if not result["errors"] and not result["warnings"]:
        print("No problems found.")
    for err in result["errors"]:
        print(f"[ERROR]   {err}")
    for warn in result["warnings"]:
        print(f"[WARNING] {warn}")
    print("-" * 60)
    print(f"Errors: {len(result['errors'])}, Warnings: {len(result['warnings'])}")
    return 1 if result["errors"] else 0


def main(argv=None):
    """Program entry point. Returns the process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    try:
        network = SubwayNetwork()
    except SubwayError as exc:
        print(f"Error: failed to load subway data: {exc}")
        return 1

    if args.check_data:
        return _run_data_check(network)

    preferred = [args.line_preference] if args.line_preference else []
    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=preferred)
    except ValueError as exc:
        print(f"Error: {exc}")
        return 2

    if preferred and preferred[0] not in network.line_names():
        print(f"Warning: line preference '{preferred[0]}' is not a known line; it will have no effect.")

    print("Beijing Subway Fare Calculator")
    print("Type a query like 'Xizhimen; Dongzhimen'. Type 'quit' to exit.")

    while True:
        try:
            raw = input(PROMPT)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in EXIT_WORDS:
            break

        try:
            start_input, end_input = parse_query(text)
            result = run_query(network, start_input, end_input, config)
            output = format_json(result) if args.format == "json" else format_text(result)
        except InputFormatError as exc:
            print(f"Error: {exc}")
            continue
        except SubwayError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # defensive: never crash the interactive loop
            print(f"Error: unexpected failure while processing the query: {exc}")
            continue

        print(output)
        if args.output_file:
            _write_output(args.output_file, output)
        print()

    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""Grid-based path planning for the 800x800 delivery map.

The map is divided into an 8x8 grid of 100x100 unit cells. Lines whose x or y
coordinate is a multiple of 100 are streets. Movement prefers streets: a rider
that is on a street moves along it, a rider that is off the streets first snaps
onto the nearest street, and a rider directly above or below its target moves
straight towards it.
"""

import logging
from typing import List, Tuple

logger = logging.getLogger(__name__)

MAP_SIZE = 800
CELL_SIZE = 100

Point = Tuple[int, int]


class PathPlanner:
    """Plans axis-aligned, street-first movement on the map.

    All methods are stateless; the planner only computes positions. Riders do
    not block each other, so path conflicts cannot create deadlocks: every
    step strictly reduces the remaining distance to the target or moves a
    rider onto a street, which guarantees progress every tick.
    """

    @staticmethod
    def clamp(value: int) -> int:
        """Clamp a coordinate into the valid map range ``[0, MAP_SIZE]``.

        Args:
            value: Coordinate to clamp.

        Returns:
            The clamped coordinate.
        """
        return max(0, min(MAP_SIZE, value))

    @staticmethod
    def _sign(delta: int) -> int:
        """Return -1, 0 or 1 according to the sign of ``delta``."""
        return (delta > 0) - (delta < 0)

    @classmethod
    def nearest_street(cls, value: int) -> int:
        """Return the closest street line (multiple of 100) to a coordinate.

        Args:
            value: A single x or y coordinate.

        Returns:
            The nearest grid line coordinate inside the map.
        """
        return cls.clamp((value + CELL_SIZE // 2) // CELL_SIZE * CELL_SIZE)

    @staticmethod
    def is_on_street(x: int, y: int) -> bool:
        """Check whether a point lies on a street.

        Args:
            x: X coordinate.
            y: Y coordinate.

        Returns:
            True if x or y is aligned with a grid line.
        """
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @classmethod
    def is_special_case(cls, x: int, y: int, tx: int, ty: int) -> bool:
        """Determine whether the current position needs special handling.

        A special case is a rider directly above or below its target, or a
        rider that is currently off the street network.

        Args:
            x: Current X coordinate.
            y: Current Y coordinate.
            tx: Target X coordinate.
            ty: Target Y coordinate.

        Returns:
            True if the special-case movement rules apply.
        """
        return x == tx or not cls.is_on_street(x, y)

    @classmethod
    def next_step(cls, x: int, y: int, tx: int, ty: int) -> Point:
        """Compute the position after one unit of movement towards a target.

        Priority rules:
            1. Already at the target: stay.
            2. Directly above/below the target: move vertically.
            3. On a horizontal street: move horizontally.
            4. On a vertical street: move vertically to the nearest street.
            5. Off the streets: snap to the nearest street (cheapest snap).

        Args:
            x: Current X coordinate.
            y: Current Y coordinate.
            tx: Target X coordinate.
            ty: Target Y coordinate.

        Returns:
            The new ``(x, y)`` position after a single unit step.
        """
        if (x, y) == (tx, ty):
            return x, y

        # Special case: directly above or below the target.
        if x == tx:
            return x, cls.clamp(y + cls._sign(ty - y))

        # Street-first: travel horizontally along a horizontal street.
        if y % CELL_SIZE == 0:
            return cls.clamp(x + cls._sign(tx - x)), y

        # On a vertical street but off horizontal streets: reach a horizontal street.
        if x % CELL_SIZE == 0:
            ys = cls.nearest_street(y)
            return x, y + cls._sign(ys - y)

        # Off the street network: snap to the closest street line.
        ys = cls.nearest_street(y)
        xs = cls.nearest_street(x)
        if abs(ys - y) <= abs(xs - x):
            return x, y + cls._sign(ys - y)
        return x + cls._sign(xs - x), y

    @classmethod
    def plan_route(cls, start: Point, goal: Point, max_steps: int = 10000) -> List[Point]:
        """Compute the full unit-step route from a start point to a goal.

        Args:
            start: Starting ``(x, y)`` position.
            goal: Destination ``(x, y)`` position.
            max_steps: Safety limit on the number of steps.

        Returns:
            List of positions starting with ``start`` and ending at ``goal``
            (or at the last reached position if the limit is hit).
        """
        x, y = start
        path: List[Point] = [(x, y)]
        for _ in range(max_steps):
            if (x, y) == tuple(goal):
                break
            x, y = cls.next_step(x, y, goal[0], goal[1])
            path.append((x, y))
        if path[-1] != tuple(goal) and len(path) > 1:
            logger.warning("Route planning stopped before reaching goal %s", goal)
        return path

"""Rider model: state machine, movement and order handling."""

import logging
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_START_X = 435
DEFAULT_START_Y = 385
DEFAULT_SPEED = 10


class Rider:
    """A delivery rider that moves on the grid and carries orders.

    Attributes:
        rider_id: Unique integer identifier.
        x: Current X coordinate (0-800).
        y: Current Y coordinate (0-800).
        speed: Movement speed in units per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Assigned orders; ``orders[0]`` is the order being served (FIFO).
        special_case: True while special movement rules apply.
        distance_travelled: Total number of unit steps taken.
        log: Chronological list of status-change entries. Each entry is a dict
            with at least the keys ``"event"`` and ``"order_id"``.
    """

    def __init__(
        self,
        rider_id: int,
        x: int = DEFAULT_START_X,
        y: int = DEFAULT_START_Y,
        speed: int = DEFAULT_SPEED,
    ) -> None:
        """Create a new idle rider and log its spawn event.

        Args:
            rider_id: Unique identifier of the rider.
            x: Initial X coordinate.
            y: Initial Y coordinate.
            speed: Movement speed in units per tick.

        Raises:
            TypeError: If ``rider_id`` is not an integer.
        """
        if isinstance(rider_id, bool) or not isinstance(rider_id, int):
            raise TypeError("rider_id must be an integer")
        self.rider_id: int = rider_id
        self.x: int = int(x)
        self.y: int = int(y)
        self.speed: int = int(speed)
        self.state: int = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.distance_travelled: int = 0
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)
        logger.info("Rider #%d spawned at (%d, %d)", self.rider_id, self.x, self.y)

    # ------------------------------------------------------------------ helpers
    @property
    def position(self) -> Tuple[int, int]:
        """Current ``(x, y)`` position."""
        return self.x, self.y

    @property
    def state_name(self) -> str:
        """Human-readable name of the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def is_idle(self) -> bool:
        """Return True if the rider is available for new orders."""
        return self.state == STATE_IDLE

    @staticmethod
    def _current_tick() -> int:
        """Return the global simulated time."""
        return int(globalvar.get_value("tick", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append a status-change entry to the rider log.

        Args:
            event: Event name, e.g. ``"accepted_order"``.
            order_id: Related order id, or None.
        """
        self.log.append(
            {
                "event": event,
                "order_id": order_id,
                "tick": self._current_tick(),
                "rider_id": self.rider_id,
                "x": self.x,
                "y": self.y,
                "state": self.state,
            }
        )

    # ------------------------------------------------------------ order intake
    def accept_order(self, order: Any) -> None:
        """Accept an order by appending it to ``orders`` and updating state.

        An idle rider moves to the picking-up state. The order's status is
        updated accordingly and an ``accepted_order`` log entry is written.

        Args:
            order: The ``Order`` instance to accept.
        """
        self.orders.append(order)
        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP
            order.status = "picking_up"
        self._record("accepted_order", order.id)
        logger.info("Rider #%d accepted order #%d", self.rider_id, order.id)

    # ---------------------------------------------------------------- movement
    def move(self) -> None:
        """Advance the rider by one tick.

        The rider travels up to ``speed`` unit steps towards the current
        target. Reaching the restaurant switches the rider to the delivering
        state; reaching the customer completes the order.
        """
        if not self.orders:
            if self.state != STATE_IDLE:
                self.state = STATE_IDLE
                self.special_case = False
            return

        if self.state == STATE_IDLE:
            self.state = STATE_PICKING_UP
            self.orders[0].status = "picking_up"

        order = self.orders[0]
        target = order.res_pos if self.state == STATE_PICKING_UP else order.cus_pos
        self._travel_towards(target)

        if self.position == tuple(target):
            if self.state == STATE_PICKING_UP:
                self.state = STATE_DELIVERING
                order.status = "delivering"
                self._record("arrived_at_merchant", order.id)
                logger.info("Rider #%d arrived at merchant for order #%d",
                            self.rider_id, order.id)
            else:
                self._complete(order)

    def _travel_towards(self, target: Tuple[int, int]) -> None:
        """Take up to ``speed`` unit steps along the planned street route.

        Args:
            target: Destination ``(x, y)`` position.
        """
        budget = self.speed
        tx, ty = target
        while budget > 0 and self.position != (tx, ty):
            self.special_case = PathPlanner.is_special_case(self.x, self.y, tx, ty)
            self.x, self.y = PathPlanner.next_step(self.x, self.y, tx, ty)
            self.distance_travelled += 1
            budget -= 1

    def _complete(self, order: Any) -> None:
        """Finish delivery of an order and release it from the rider.

        Args:
            order: The order that was delivered (must be ``orders[0]``).
        """
        order.mark_completed(self._current_tick())
        self.orders.remove(order)
        self._record("delivered_to_customer", order.id)
        self.special_case = False
        if self.orders:
            self.state = STATE_PICKING_UP
            self.orders[0].status = "picking_up"
        else:
            self.state = STATE_IDLE
        logger.info("Rider #%d delivered order #%d", self.rider_id, order.id)

    # ------------------------------------------------------------ serialization
    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serializable snapshot of the rider.

        Returns:
            Dictionary with the rider attributes and order ids.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [o.id for o in self.orders],
            "log": list(self.log),
        }

"""System-level state management: funds, riders, time and persistence."""

import json
import logging
from typing import Any, Dict, List, Optional

import globalvar
from rider import (
    DEFAULT_SPEED,
    DEFAULT_START_X,
    DEFAULT_START_Y,
    STATE_IDLE,
    Rider,
)

logger = logging.getLogger(__name__)

# Configurable system parameters (can be overridden by a JSON config file).
PARAMS: Dict[str, int] = {
    "initial_funds": 1000,
    "recruit_threshold": 300,
    "recruit_cost": 300,
    "rider_speed": DEFAULT_SPEED,
    "start_x": DEFAULT_START_X,
    "start_y": DEFAULT_START_Y,
}


def init_system() -> None:
    """Initialize all core global variables.

    Sets system funds, an empty rider list, the rider id counter, the running
    flag, the simulated clock, the pending order queue and the order registry.
    """
    globalvar._init()
    globalvar.set_value("funds", int(PARAMS["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("pending_orders", [])
    globalvar.set_value("all_orders", [])
    globalvar.set_value("order_id_counter", 0)
    logger.info("System initialized with funds=%s", PARAMS["initial_funds"])


# ----------------------------------------------------------------- accessors
def _riders_list() -> List[Rider]:
    """Return the live rider list, creating it if missing."""
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    return riders


def get_riders() -> List[Rider]:
    """Return the list of all riders."""
    return _riders_list()


def get_funds() -> int:
    """Return the current system fund balance."""
    return int(globalvar.get_value("funds", 0) or 0)


def adjust_funds(delta: int) -> int:
    """Add ``delta`` (may be negative) to the system funds.

    Args:
        delta: Amount to add. Negative values deduct funds.

    Returns:
        The new fund balance.
    """
    new_value = get_funds() + int(delta)
    globalvar.set_value("funds", new_value)
    return new_value


def get_tick() -> int:
    """Return the current simulated time in ticks."""
    return int(globalvar.get_value("tick", 0) or 0)


def get_idle_riders() -> List[Rider]:
    """Return all riders currently in the idle state."""
    return [r for r in _riders_list() if r.state == STATE_IDLE]


def find_rider(rider_id: int) -> Optional[Rider]:
    """Look up a rider by id.

    Args:
        rider_id: Identifier to search for.

    Returns:
        The matching rider or None.
    """
    for rider in _riders_list():
        if rider.rider_id == rider_id:
            return rider
    return None


# ------------------------------------------------------------- rider CRUD
def create_rider() -> Rider:
    """Create a new rider with an auto-incremented id at the start position.

    Returns:
        The newly created rider, already appended to the rider list.
    """
    counter = int(globalvar.get_value("rider_id_counter", 0) or 0) + 1
    globalvar.set_value("rider_id_counter", counter)
    rider = Rider(
        counter,
        x=PARAMS["start_x"],
        y=PARAMS["start_y"],
        speed=PARAMS["rider_speed"],
    )
    _riders_list().append(rider)
    logger.info("Created rider #%d", counter)
    return rider


def recruit_if_needed(pending_count: int) -> Optional[Rider]:
    """Recruit a rider when funds allow and pending orders exceed idle riders.

    Args:
        pending_count: Number of orders waiting for assignment.

    Returns:
        The new rider if one was recruited, otherwise None.
    """
    idle = len(get_idle_riders())
    if get_funds() >= PARAMS["recruit_threshold"] and pending_count > idle:
        adjust_funds(-PARAMS["recruit_cost"])
        logger.info("Recruiting rider: pending=%d idle=%d", pending_count, idle)
        return create_rider()
    return None


def remove_rider(rider_id: int) -> bool:
    """Remove an idle rider from the system.

    Args:
        rider_id: Identifier of the rider to remove.

    Returns:
        True if the rider was removed, False if it does not exist.

    Raises:
        ValueError: If the rider still has assigned orders.
    """
    rider = find_rider(rider_id)
    if rider is None:
        return False
    if not rider.is_idle() or rider.orders:
        raise ValueError(f"Rider #{rider_id} is busy and cannot be removed")
    _riders_list().remove(rider)
    logger.info("Removed rider #%d", rider_id)
    return True


# ------------------------------------------------------------- time control
def advance_tick() -> int:
    """Advance the simulation by one tick and move every rider.

    Returns:
        The new tick value.
    """
    tick = get_tick() + 1
    globalvar.set_value("tick", tick)
    for rider in list(_riders_list()):
        rider.move()
    return tick


# ----------------------------------------------------------- consistency
def check_consistency() -> List[str]:
    """Verify and repair the state of riders and their orders.

    Repairs performed:
        * Riders without orders are set to idle.
        * Riders with orders are never idle.
        * Completed orders are removed from rider lists.

    Returns:
        A list of human-readable descriptions of issues that were fixed.
    """
    issues: List[str] = []
    for rider in _riders_list():
        for order in list(rider.orders):
            if getattr(order, "status", "") == "completed":
                rider.orders.remove(order)
                issues.append(f"Removed completed order #{order.id} from rider #{rider.rider_id}")
        if not rider.orders and rider.state != STATE_IDLE:
            rider.state = STATE_IDLE
            issues.append(f"Reset rider #{rider.rider_id} to idle")
        elif rider.orders and rider.state == STATE_IDLE:
            rider.state = 1
            rider.orders[0].status = "picking_up"
            issues.append(f"Rider #{rider.rider_id} had orders while idle; set to picking up")
    for issue in issues:
        logger.warning("Consistency repair: %s", issue)
    return issues


# ------------------------------------------------------------ persistence
def load_config(path: str) -> Dict[str, Any]:
    """Load system parameters from a JSON configuration file.

    The file may contain a ``"params"`` object whose known keys override the
    defaults in :data:`PARAMS`.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        The parsed JSON document.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the JSON is invalid or a parameter is not an integer.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    params = data.get("params", {}) if isinstance(data, dict) else {}
    for key in PARAMS:
        if key in params:
            value = params[key]
            if isinstance(value, bool) or not isinstance(value, int):
                raise ValueError(f"Parameter '{key}' must be an integer")
            PARAMS[key] = value
    logger.info("Loaded configuration from %s", path)
    return data


def save_config(path: str) -> None:
    """Persist parameters, funds, riders and order history to a JSON file.

    Args:
        path: Destination file path.

    Raises:
        OSError: If the file cannot be written.
    """
    orders = globalvar.get_value("all_orders", []) or []
    pending = globalvar.get_value("pending_orders", []) or []
    data = {
        "params": dict(PARAMS),
        "state": {
            "funds": get_funds(),
            "tick": get_tick(),
            "rider_id_counter": int(globalvar.get_value("rider_id_counter", 0) or 0),
            "order_id_counter": int(globalvar.get_value("order_id_counter", 0) or 0),
        },
        "riders": [r.to_dict() for r in _riders_list()],
        "pending_order_ids": [o.id for o in pending],
        "orders": [o.to_dict() for o in orders],
    }
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2)
    logger.info("Saved state to %s", path)

"""Order data model and coordinate conversion."""

import logging
import math
from typing import Any, Dict, List, Sequence, Tuple

import globalvar

logger = logging.getLogger(__name__)

MAP_SIZE = 800
CELL_SIZE = 100
GRID_CELLS = 8

ORDER_PENDING = "pending"
ORDER_ASSIGNED = "assigned"
ORDER_PICKING_UP = "picking_up"
ORDER_DELIVERING = "delivering"
ORDER_COMPLETED = "completed"


def validate_coordinate(value: Any) -> int:
    """Validate a single map coordinate.

    Args:
        value: Value to validate.

    Returns:
        The coordinate as an integer.

    Raises:
        TypeError: If the value is not an integer.
        ValueError: If the value is outside the range 0-800.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise TypeError(f"Coordinate {value!r} is not an integer")
    if not 0 <= value <= MAP_SIZE:
        raise ValueError(f"Coordinate {value} is outside the range 0-{MAP_SIZE}")
    return value


def to_grid_center(x: int, y: int) -> Tuple[int, int]:
    """Convert a coordinate to the center point of its grid cell.

    Uses ``center = (coord // 100) * 100 + 50`` with the cell index clamped to
    the valid 8x8 range, so that the coordinate 800 maps to the last cell.

    Args:
        x: X coordinate (0-800).
        y: Y coordinate (0-800).

    Returns:
        The ``(center_x, center_y)`` tuple.
    """
    gx = min(max(int(x) // CELL_SIZE, 0), GRID_CELLS - 1)
    gy = min(max(int(y) // CELL_SIZE, 0), GRID_CELLS - 1)
    return gx * CELL_SIZE + CELL_SIZE // 2, gy * CELL_SIZE + CELL_SIZE // 2


def _validate_point(point: Sequence[Any]) -> Tuple[int, int]:
    """Validate an ``[x, y]`` point.

    Args:
        point: Sequence of exactly two integer coordinates.

    Returns:
        The validated ``(x, y)`` tuple.

    Raises:
        TypeError: If the point is not a two-element sequence of integers.
        ValueError: If a coordinate is out of range.
    """
    if not isinstance(point, (list, tuple)) or len(point) != 2:
        raise TypeError("A point must be a sequence of two integers")
    return validate_coordinate(point[0]), validate_coordinate(point[1])


def _registry() -> List["Order"]:
    """Return the global list of all orders ever created."""
    registry = globalvar.get_value("all_orders")
    if registry is None:
        registry = []
        globalvar.set_value("all_orders", registry)
    return registry


def _next_order_id() -> int:
    """Return the next unique order id."""
    value = int(globalvar.get_value("order_id_counter", 0) or 0) + 1
    globalvar.set_value("order_id_counter", value)
    return value


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        raw_restaurant: Restaurant coordinates as entered by the user.
        raw_customer: Customer coordinates as entered by the user.
        res_pos: Restaurant grid-cell center ``(x, y)``.
        cus_pos: Customer grid-cell center ``(x, y)``.
        status: One of pending, assigned, picking_up, delivering, completed.
        rider_id: Id of the assigned rider, or None.
        start_tick: Simulated time at which the order was created.
        assigned_tick: Simulated time at which the order was assigned.
        complete_tick: Simulated time at which the order was delivered.
        distance: Straight-line distance between the grid centers.
        time_taken: Ticks from creation to delivery, or None.
        efficiency: Distance covered per tick, or None.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: ``[restaurant_x, restaurant_y]`` in the range 0-800.
            cus_pos: ``[customer_x, customer_y]`` in the range 0-800.

        Raises:
            TypeError: If the coordinates are not integers.
            ValueError: If a coordinate is outside the range 0-800.
        """
        rx, ry = _validate_point(res_pos)
        cx, cy = _validate_point(cus_pos)
        self.id: int = _next_order_id()
        self.raw_restaurant: Tuple[int, int] = (rx, ry)
        self.raw_customer: Tuple[int, int] = (cx, cy)
        self.res_pos: Tuple[int, int] = to_grid_center(rx, ry)
        self.cus_pos: Tuple[int, int] = to_grid_center(cx, cy)
        self.status: str = ORDER_PENDING
        self.rider_id: Any = None
        self.start_tick: int = int(globalvar.get_value("tick", 0) or 0)
        self.assigned_tick: Any = None
        self.complete_tick: Any = None
        self.distance: float = round(
            math.hypot(self.res_pos[0] - self.cus_pos[0],
                       self.res_pos[1] - self.cus_pos[1]), 2)
        self.time_taken: Any = None
        self.efficiency: Any = None
        _registry().append(self)
        logger.info("Order #%d created: restaurant=%s customer=%s",
                    self.id, self.res_pos, self.cus_pos)

    def assign_to(self, rider_id: int) -> None:
        """Assign this order to a rider.

        The order is appended to the rider's ``orders`` list through
        ``Rider.accept_order``, which also performs the idle-to-picking-up
        state transition and writes the log entry.

        Args:
            rider_id: Identifier of the target rider.

        Raises:
            ValueError: If no rider with the given id exists.
        """
        riders = globalvar.get_value("riders") or []
        rider = next((r for r in riders if r.rider_id == rider_id), None)
        if rider is None:
            raise ValueError(f"No rider with id {rider_id}")
        self.rider_id = rider_id
        self.assigned_tick = int(globalvar.get_value("tick", 0) or 0)
        self.status = ORDER_ASSIGNED
        rider.accept_order(self)
        logger.info("Order #%d assigned to rider #%d", self.id, rider_id)

    def mark_completed(self, tick: int) -> None:
        """Record delivery completion, time taken and efficiency.

        Args:
            tick: Simulated time of delivery.
        """
        self.status = ORDER_COMPLETED
        self.complete_tick = tick
        self.time_taken = max(tick - self.start_tick, 0)
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 3)
        else:
            self.efficiency = self.distance
        logger.info("Order #%d completed at tick %d", self.id, tick)

    def to_dict(self) -> Dict[str, Any]:

        Returns:
            Dictionary with the order attributes.

"""Order assignment: rider selection, recruitment and dispatching."""

import logging
import math
from typing import List, Optional, Tuple

import globalvar
import system
from order_model import Order
from rider import Rider

logger = logging.getLogger(__name__)


def euclidean_distance(x1: float, y1: float, x2: float, y2: float) -> float:
    """Return the Euclidean distance between two points.

    Args:
        x1: X coordinate of the first point.
        y1: Y coordinate of the first point.
        x2: X coordinate of the second point.
        y2: Y coordinate of the second point.

    Returns:
        ``sqrt((x1 - x2)^2 + (y1 - y2)^2)``.
    """
    return math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the best rider to serve an order.

    Selection is closest-first by Euclidean distance from the rider to the
    restaurant. For load balancing, riders with fewer assigned orders are
    preferred first; idle riders (zero orders) therefore always win over busy
    ones, and ties are broken by rider id.

    Args:
        order: The order to be assigned.

    Returns:
        The selected rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    rx, ry = order.res_pos
    return min(
        riders,
        key=lambda r: (
            len(r.orders),
            euclidean_distance(r.x, r.y, rx, ry),
            r.rider_id,
        ),
    )


def dispatch_pending_orders() -> Tuple[List[Tuple[Order, Rider]], List[Rider]]:
    """Assign all pending orders in FIFO order, recruiting riders as needed.

    Before each assignment the recruitment rule is applied repeatedly: while
    funds allow and pending orders exceed idle riders, a new rider is created.
    Orders that cannot be assigned (no riders and insufficient funds) remain
    in the pending queue.

    Returns:
        A tuple ``(assignments, new_riders)`` where ``assignments`` is a list of
        ``(order, rider)`` pairs and ``new_riders`` lists riders recruited in
        this call.
    """
    pending = globalvar.get_value("pending_orders")
    if pending is None:
        pending = []
        globalvar.set_value("pending_orders", pending)

    assignments: List[Tuple[Order, Rider]] = []
    new_riders: List[Rider] = []

    while pending:
        order = pending[0]
        while True:
            recruited = system.recruit_if_needed(len(pending))
            if recruited is None:
                break
            new_riders.append(recruited)

        best = find_best_rider_for_order(order)
        if best is None:
            break

        order.assign_to(best.rider_id)
        pending.pop(0)
        assignments.append((order, best))

    return assignments, new_riders

"""Status display and ASCII map visualization."""

import logging
from typing import Any, Iterable, List, Sequence, Tuple

from rider import STATE_NAMES

logger = logging.getLogger(__name__)


class DisplayManager:
    """Renders simulator status, order tables, rider logs and the ASCII map.

    The ASCII map is a 17x17 character canvas. Each character represents 50
    map units, so street lines (multiples of 100) fall on even indices and the
    8x8 grid cells fall on odd indices. Symbols:
        '#' street, '.' empty cell, 'R' rider, 'S' restaurant,
        'C' customer, '*' route line segment.
    """

    CANVAS = 17
    STEP = 50

    @classmethod
    def point_to_cell(cls, x: int, y: int) -> Tuple[int, int]:
        """Convert a map coordinate to a ``(row, col)`` canvas index.

        Args:
            x: X coordinate (0-800).
            y: Y coordinate (0-800).

        Returns:
            Tuple ``(row, col)`` inside the canvas.
        """
        col = min(max(int(x) // cls.STEP, 0), cls.CANVAS - 1)
        row = min(max(int(y) // cls.STEP, 0), cls.CANVAS - 1)
        return row, col

    def render_map(
        self,
        riders: Iterable[Any] = (),
        orders: Iterable[Any] = (),
        route: Sequence[Tuple[int, int]] = (),
    ) -> str:
        """Render the ASCII map with riders, orders and an optional route.

        Args:
            riders: Riders to mark with 'R'.
            orders: Orders to mark with 'S' and 'C' (completed orders skipped).
            route: Sequence of positions drawn as connected '*' segments.

        Returns:
            The multi-line map string.
        """
        n = self.CANVAS
        canvas = [
            ["#" if (r % 2 == 0 or c % 2 == 0) else "." for c in range(n)]
            for r in range(n)
        ]

        cells = [self.point_to_cell(x, y) for x, y in route]
        for (r1, c1), (r2, c2) in zip(cells, cells[1:]):
            for r in range(min(r1, r2), max(r1, r2) + 1):
                for c in range(min(c1, c2), max(c1, c2) + 1):
                    canvas[r][c] = "*"

        for order in orders:
            if getattr(order, "status", "") == "completed":
                continue
            r, c = self.point_to_cell(*order.res_pos)
            canvas[r][c] = "S"
            r, c = self.point_to_cell(*order.cus_pos)
            canvas[r][c] = "C"

        for rider in riders:
            r, c = self.point_to_cell(rider.x, rider.y)
            canvas[r][c] = "R"

        lines = ["     x: 0 .. 800 (1 char = 50 units, streets every 100)"]
        for r in range(n):
            label = f"{r * self.STEP:>4} " if r % 2 == 0 else "     "
            lines.append(label + " ".join(canvas[r]))
        lines.append("Legend: R rider, S restaurant, C customer, "
                     "# street, . empty, * route")
        return "\n".join(lines)

    def render_status(self, funds: int, riders: Sequence[Any], pending_count: int) -> str:
        """Render the system status snapshot.

        Args:
            funds: Current system funds.
            riders: All riders.
            pending_count: Number of orders waiting for assignment.

        Returns:
            The multi-line status string.
        """
        lines = [
            "=== System Status ===",
            f"Funds: {funds}",
            f"Riders: {len(riders)}",
            f"Pending orders: {pending_count}",
        ]
        for rider in riders:
            order_ids = [o.id for o in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id} | position ({rider.x}, {rider.y}) | "
                f"{STATE_NAMES.get(rider.state, 'unknown')} | "
                f"orders: {len(rider.orders)} {order_ids}"
            )
        return "\n".join(lines)

    def render_orders(self, orders: Sequence[Any]) -> str:
        """Render a table of all orders.

        Args:
            orders: All orders in creation order.

        Returns:
            The multi-line table string.
        """
        if not orders:
            return "No orders yet."
        lines = ["=== Orders ==="]
        for o in orders:
            time_taken = "-" if o.time_taken is None else f"{o.time_taken} ticks"
            rider = "-" if o.rider_id is None else f"#{o.rider_id}"
            lines.append(
                f"Order #{o.id} | start tick {o.start_tick} | "
                f"restaurant {o.res_pos} | customer {o.cus_pos} | "
                f"distance {o.distance} | time {time_taken} | "
                f"rider {rider} | status {o.status}"
            )
        return "\n".join(lines)

    def render_rider_logs(self, riders: Sequence[Any]) -> str:
        """Render the status-change history of every rider.

        Args:
            riders: All riders.

        Returns:
            The multi-line log string.
        """
        if not riders:
            return "No riders."
        lines = ["=== Rider Records ==="]
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id}:")
            if not rider.log:
                lines.append("  (no records)")
            for entry in rider.log:
                order = "-" if entry.get("order_id") is None else f"#{entry['order_id']}"
                state = STATE_NAMES.get(entry.get("state", 0), "unknown")
                lines.append(
                    f"  tick {entry.get('tick', 0):>5} | {entry['event']:<22} | "
                    f"order {order:<6} | position ({entry.get('x')}, {entry.get('y')}) | "
                    f"state {state}"
                )
        return "\n".join(lines)

import logging
from typing import Callable, List, Optional, Sequence, Tuple

import globalvar

    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "save": "save",
    "history": "history", "hist": "history",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Place an order (restaurant x y, customer x y; 0-800).
                       Shortcut: s. Prompts for coordinates when omitted.
  status               Show funds, riders and their order lists (shortcut: st).
  orders               Show all orders with time, distance and rider (shortcut: o).
  riders               Show every rider status-change record (shortcut: r).
  map                  Show the ASCII map with riders and active orders (shortcut: m).
  save [path]          Save state to a JSON file (default dispatch_state.json).
  history              Show command history.
  !!                   Repeat the previous command.   !N  Repeat history entry N.
  help                 Show this help (shortcut: h or ?).
  quit / exit          Exit the program (shortcut: q).

def parse_coordinates(tokens: Sequence[str]) -> List[int]:
    """Parse and validate four integer coordinates.

    Commas are accepted as separators in addition to whitespace.

    Args:
        tokens: Raw text tokens.

    Returns:
        A list of four integers ``[rx, ry, cx, cy]``.

    Raises:
        ValueError: If the count, type or range of any value is invalid.
    """
    text = " ".join(tokens).replace(",", " ")
    parts = text.split()
    if len(parts) != 4:
        raise ValueError(
            "Expected 4 integers: restaurant_x restaurant_y customer_x customer_y"
        )
    values: List[int] = []
    for part in parts:
        try:
            value = int(part)
        except ValueError:
            raise ValueError(f"'{part}' is not an integer") from None
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"Coordinate {value} is out of range 0-{MAP_SIZE}")
        values.append(value)
    return values


class InputManager:
    """Runs the interactive command loop and dispatches commands.

    Attributes:
        display: The DisplayManager used for all rendering.
        history: Previously entered command lines.
    """

    def __init__(
        self,
        input_func: Optional[Callable[[str], str]] = None,
        output_func: Optional[Callable[..., None]] = None,
    ) -> None:
        """Create the command manager.

        Args:
            input_func: Function used to read a line (defaults to ``input``).
            output_func: Function used to print output (defaults to ``print``).
        """
        self._input = input_func or input
        self._print = output_func or print
        self.display = DisplayManager()
        self.history: List[str] = []
        self._setup_readline()

    @staticmethod
    def _setup_readline() -> None:
        """Enable line editing and arrow-key history when readline exists."""
        try:
            import readline  # noqa: F401  pylint: disable=unused-import
        except ImportError:
            logger.debug("readline is not available on this platform")

    # ---------------------------------------------------------- main loop
    def run(self) -> None:
        """Run the command loop until the running flag is cleared."""
        self._print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while globalvar.get_value("running", True):
            try:
                line = self._input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self._print("")
                break
            line = line.strip()
            if not line:
                continue
            line = self._expand_history(line)
            if line is None:
                continue
            self.history.append(line)
            try:
                self.execute(line)
            except Exception as exc:  # error recovery: keep the loop alive
                logger.exception("Command failed: %s", line)
                self._print(f"Error: {exc}")
            for issue in system.check_consistency():
                self._print(f"[recovered] {issue}")

    def _expand_history(self, line: str) -> Optional[str]:
        """Resolve ``!!`` and ``!N`` shortcuts to previous commands.

        Args:
            line: Raw input line.

        Returns:
            The expanded command, or None if the shortcut is invalid.
        """
        if line == "!!":
            if not self.history:
                self._print("No previous command.")
                return None
            return self.history[-1]
        if line.startswith("!") and line[1:].isdigit():
            index = int(line[1:])
            if not 1 <= index <= len(self.history):
                self._print(f"No history entry {index}.")
                return None
            return self.history[index - 1]
        return line

    def execute(self, line: str) -> None:
        """Parse and run a single command line.

        Args:
            line: The command text.
        """
        parts = line.split()
        name = COMMAND_ALIASES.get(parts[0].lower())
        args = parts[1:]
        if name is None:
            self._print(f"Unknown command '{parts[0]}'. Type 'help' for a list.")
            return
        handler = getattr(self, f"cmd_{name}")
        handler(args)

    # ------------------------------------------------------------ commands
    def cmd_help(self, _args: List[str]) -> None:
        """Print the help text."""
        self._print(HELP_TEXT)

    def cmd_history(self, _args: List[str]) -> None:
        """Print the numbered command history."""
        if not self.history:
            self._print("History is empty.")
            return
        for index, entry in enumerate(self.history, start=1):
            self._print(f"{index:>4}  {entry}")

    def cmd_quit(self, _args: List[str]) -> None:
        """Stop the main loop."""
        globalvar.set_value("running", False)
        self._print("Goodbye.")

    def cmd_status(self, _args: List[str]) -> None:
        """Print the system status snapshot."""
        pending = globalvar.get_value("pending_orders") or []
        self._print(self.display.render_status(
            system.get_funds(), system.get_riders(), len(pending)))

    def cmd_orders(self, _args: List[str]) -> None:
        """Print every order with time, distance and rider information."""
        self._print(self.display.render_orders(globalvar.get_value("all_orders") or []))

    def cmd_riders(self, _args: List[str]) -> None:
        """Print every rider's recorded status changes."""
        self._print(self.display.render_rider_logs(system.get_riders()))

    def cmd_map(self, _args: List[str]) -> None:
        """Print the ASCII map with riders and active orders."""
        orders = [o for o in (globalvar.get_value("all_orders") or [])
                  if o.status != "completed"]
        self._print(self.display.render_map(system.get_riders(), orders))

    def cmd_save(self, args: List[str]) -> None:
        """Save state to a JSON file.

        Args:
            args: Optional output path.
        """
        path = args[0] if args else DEFAULT_SAVE_PATH
        system.save_config(path)
        self._print(f"State saved to {path}")

    def cmd_start(self, args: List[str]) -> None:
        """Place an order, dispatch it and simulate until it is delivered.

        Args:
            args: Optional four coordinate tokens. When omitted the coordinates
                are requested interactively.
        """
        tokens = list(args)
        if not tokens:
            tokens = [self._input("Enter restaurant x y and customer x y (0-800): ")]
        try:
            coords = parse_coordinates(tokens)
        except ValueError as exc:
            self._print(f"Invalid input: {exc}")
            return
        self.place_order(coords)

    # ------------------------------------------------------- order workflow
    def place_order(self, coords: Sequence[int]) -> None:
        """Create an order, dispatch it and report the delivery route.

        Args:
            coords: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.
        """
        order = Order(coords[0:2], coords[2:4])
        pending = globalvar.get_value("pending_orders")
        pending.append(order)

        assignments, new_riders = dispatch_pending_orders()
        if order.status == "pending" or order.rider_id is None:
            self._print(
                f"Order #{order.id} queued: no rider available and funds are "
                f"insufficient to recruit one (funds {system.get_funds()})."
            )
            return

        rider = system.find_rider(order.rider_id)
        route: List[Tuple[int, int]] = [(rider.x, rider.y)]
        ticks = 0
        while order.status != "completed" and ticks < MAX_SIMULATION_TICKS:
            system.advance_tick()
            _, more_new = dispatch_pending_orders()
            new_riders.extend(more_new)
            ticks += 1
            route.append((rider.x, rider.y))

        if order.status != "completed":
            self._print(f"Order #{order.id} did not finish within {MAX_SIMULATION_TICKS} ticks.")
            return

        self._print_order_report(order, rider, route, new_riders, coords)

    def _print_order_report(
        self,
        order: Order,
        rider,
        route: List[Tuple[int, int]],
        new_riders: list,
        coords: Sequence[int],
    ) -> None:
        """Print the summary and route map for a delivered order.

        Args:
            order: The completed order.
            rider: The rider that served the order.
            route: Positions visited by the rider, one per tick.
            new_riders: Riders recruited while processing this order.
            coords: The coordinates entered by the user.
        """
        rgrid = to_grid_center(coords[0], coords[1])
        cgrid = to_grid_center(coords[2], coords[3])
        recruited = "yes" if new_riders else "no"
        lines = [
            f"Order #{order.id} delivered.",
            f"  Restaurant: input ({coords[0]}, {coords[1]}) -> grid {rgrid}",
            f"  Customer:   input ({coords[2]}, {coords[3]}) -> grid {cgrid}",
            f"  New rider added: {recruited}"
            + (f" (rider #{', #'.join(str(r.rider_id) for r in new_riders)})"
               if new_riders else ""),
            f"  Assigned rider: #{rider.rider_id} | speed {rider.speed} units/tick",
            f"  Time taken: {order.time_taken} ticks | distance {order.distance} | "
            f"efficiency {order.efficiency} units/tick",
            "  Route map:",
            self.display.render_map(riders=[rider], route=route),

        self._print("\n".join(lines))

Usage:
    python src/main.py [--config path/to/config.json]

import argparse
import logging
import os
import sys
from typing import List, Optional

if __package__ in (None, ""):
    # Allow running as a script: `python src/main.py`
    sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import system  # noqa: E402  pylint: disable=wrong-import-position
from input_manager import InputManager  # noqa: E402  pylint: disable=wrong-import-position


def configure_logging() -> None:
    """Configure logging from the DISPATCH_LOG_LEVEL environment variable."""
    level_name = os.environ.get("DISPATCH_LOG_LEVEL", "WARNING").upper()
    level = getattr(logging, level_name, logging.WARNING)
    logging.basicConfig(
        level=level,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        stream=sys.stderr,
    )


def main(argv: Optional[List[str]] = None) -> int:
    """Run the simulator.

    Args:
        argv: Command-line arguments (defaults to ``sys.argv[1:]``).

    Returns:
        Process exit code.
    """
    configure_logging()
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=None,
                        help="optional JSON file with system parameters")
    args = parser.parse_args(argv)

    if args.config:
        try:
            system.load_config(args.config)
        except (OSError, ValueError) as exc:
            print(f"Warning: could not load config '{args.config}': {exc}. "
                  "Using defaults.")

    system.init_system()
    try:
        InputManager().run()
    except Exception:  # last-resort recovery
        logging.getLogger(__name__).exception("Unexpected fatal error")
        print("A fatal error occurred. See the log for details.")
        return 1
    return 0


if __name__ == "__main__":

from __future__ import annotations

from dataclasses import dataclass, field

_LEVEL_WORDS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
}
DEFAULT_LEVEL = 99


def parse_level_from_name(name: str) -> int:
    """Derive a prize level from a leading ordinal word such as 'First Prize'.

    Lower numbers represent higher prize levels. Unknown names get DEFAULT_LEVEL.
    """
    words = name.strip().split()
    if not words:
        return DEFAULT_LEVEL
    return _LEVEL_WORDS.get(words[0].casefold(), DEFAULT_LEVEL)


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round.

    avoid_groups: department groups whose members must not win this prize.
    associated_drawer: the lottery person bound to this prize (or None).
    level: prize level, 1 is the highest; derived from the name when omitted.
    """

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None
    level: int | None = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if self.level is None:
            self.level = parse_level_from_name(self.name)

"""Participant list loading, verification and group extraction."""
from __future__ import annotations

from datetime import datetime

from lottery_system.default_data import DEFAULT_LIST_TEXT
from lottery_system.models import User

LIST_FORMAT_HINT = "expected 'Name,Department'"


def parse_list_lines(lines: list[str]) -> tuple[list[User], list[str]]:
    """Parse line-based participant data.

    Blank lines and lines starting with '#' are ignored.
    Returns (parsed users, format error messages).
    """
    users: list[User] = []
    errors: list[str] = []
    for line_no, raw in enumerate(lines, start=1):
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        parts = [part.strip() for part in line.split(",")]
        if len(parts) != 2 or not parts[0] or not parts[1]:
            errors.append(
                f"Format Error at Line {line_no}: '{line}' ({LIST_FORMAT_HINT})"
            )
            continue
        users.append(User(name=parts[0], department=parts[1]))
    return users, errors


class ListManager:
    """Loads, verifies and summarizes participant lists."""

    def __init__(self) -> None:
        self.source_mode: str | None = None
        self.file_path: str | None = None
        self.loaded_at: datetime | None = None

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        users, errors = parse_list_lines(DEFAULT_LIST_TEXT.splitlines())
        self.source_mode = "default"
        self.file_path = None
        self.loaded_at = datetime.now()
        return users, errors

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt participant list from the given path."""
        self.source_mode = "custom"
        self.file_path = path
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a txt file. Returns (users, errors)."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: '{path}'"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        self.loaded_at = datetime.now()
        return parse_list_lines(lines)

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid first occurrences, duplicate name entries)."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department group name to member count."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

from __future__ import annotations

from lottery_system.models import Prize

DEFAULT_PRIZES: list[dict] = [
    {
        "name": "Third Prize - Movie Tickets",
        "quantity": 5,
        "avoid_groups": [],
        "associated_drawer": "Chen Qixian",
    },
    {
        "name": "Second Prize - Shopping Cards",
        "quantity": 3,
        "avoid_groups": [],
        "associated_drawer": "Jin Duo",
    },
    {
        "name": "First Prize - 1 Day Annual Leave",
        "quantity": 1,
        "avoid_groups": [],
        "associated_drawer": "Ma Jiaqi",
    },
]


class PrizeManager:
    """Manages the ordered prize library. Starts with no prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for spec in DEFAULT_PRIZES:
            try:
                self.add_prize(
                    name=spec["name"],
                    quantity=spec["quantity"],
                    avoid_groups=list(spec["avoid_groups"]),
                    associated_drawer=spec["associated_drawer"],
                )
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the lottery order and return it."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer else None
        if not drawer:
            drawer = None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current lottery order (a copy of the list)."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` holds 1-based current positions in the new order.

        Returns True on success, False if `order` is not a permutation of positions.
        """
        count = len(self._prizes)
        try:
            if sorted(order) != list(range(1, count + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback."""
from __future__ import annotations

import random

from lottery_system.models import Prize, User

PROTECTED_GROUP = "Cheng Ru group"
LI_BIN = "Li Bin"
MA_JIAQI = "Ma Jiaqi"
FALLBACK_TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")


def _norm(text: str | None) -> str:
    return (text or "").strip().casefold()


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.winner_names: set[str] = set()
        # Results (True = contained a Cheng Ru group member) of the most recent
        # draws made by Chen Qixian / Jin Duo, at most two entries.
        self.last_two_draw_results: list[bool] = []

    # ----- rule helpers -------------------------------------------------
    @staticmethod
    def is_protected(user: User) -> bool:
        return _norm(user.department) == _norm(PROTECTED_GROUP)

    def is_fallback_active(self) -> bool:
        """True when the last two Chen/Jin draws both had no Cheng Ru member."""
        return len(self.last_two_draw_results) == 2 and not any(
            self.last_two_draw_results
        )

    def _available_users(self) -> list[User]:
        """Users who have not won yet (anti-duplication)."""
        return [u for u in self.users if u.name not in self.winner_names]

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        available = self._available_users()
        drawer = _norm(prize.associated_drawer)

        # Fallback (Cheng Ru group protection): forced pool for Ma Jiaqi.
        if drawer == _norm(MA_JIAQI) and self.is_fallback_active():
            forced = [u for u in available if self.is_protected(u)]
            if forced:
                return forced

        # Special rule: Li Bin's prizes use relaxed filtering (no group avoidance).
        if drawer == _norm(LI_BIN):
            return available

        # Basic filtering: exclude the prize's avoidance groups.
        avoided = {_norm(group) for group in prize.avoid_groups}
        return [u for u in available if _norm(u.department) not in avoided]

    def describe_rule(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        rules = ["previous winners excluded"]
        drawer = _norm(prize.associated_drawer)
        if drawer == _norm(LI_BIN):
            rules.append("relaxed filtering for Li Bin (group avoidance not applied)")
        elif prize.avoid_groups:
            rules.append("avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            rules.append("no group avoidance")
        if drawer == _norm(MA_JIAQI):
            if self.is_fallback_active():
                rules.append(
                    f"{PROTECTED_GROUP} protection ACTIVE: pool limited to its members"
                )
            else:
                rules.append(
                    f"{PROTECTED_GROUP} protection armed (triggers if the last two "
                    f"Chen Qixian/Jin Duo draws had no {PROTECTED_GROUP} winner)"
                )
        return "; ".join(rules)

    # ----- drawing ------------------------------------------------------
    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        winner: User | None = None
        if pool:
            winner = random.choice(pool)
            self.winner_names.add(winner.name)

        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(name) for name in FALLBACK_TRIGGER_DRAWERS}:
            contains_protected = winner is not None and self.is_protected(winner)
            self.last_two_draw_results.append(contains_protected)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return winner

"""Winner recording, grouping and CSV-style export."""
from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_order: int
    winning_time: datetime

    @property
    def name(self) -> str:
        return self.user.name

    @property
    def department(self) -> str:
        return self.user.department


class ResultManager:
    """Keeps the complete ordered winner list for the event."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    @property
    def winners(self) -> list[Winner]:
        return list(self._winners)

    def add_winner(self, user: User, prize: Prize) -> Winner:
        winner = Winner(
            user=user,
            prize=prize,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name.

        Keys are ordered by prize level (high to low); ties keep the
        configured lottery order. Each list is in winning time order.
        """
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda item: (item[1].level, item[0]))
        result: dict[str, list[Winner]] = {}
        for _, prize in indexed:
            result[prize.name] = [
                w for w in self._winners if w.prize.name == prize.name
            ]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to awards_result_YYYYMMDD.txt in the current directory."""
        grouped = self.get_winners_by_prize(prizes)
        level_map = {p.name: p.level for p in prizes}
        file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(
                    [
                        "Prize Level",
                        "Prize Name",
                        "Winning Order",
                        "Name",
                        "Department",
                        "Winning Time",
                    ]
                )
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        writer.writerow(
                            [
                                level_map.get(prize_name, ""),
                                prize_name,
                                winner.winning_order,
                                winner.name,
                                winner.department,
                                winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

from __future__ import annotations

from lottery_system.models import Prize, User
from lottery_system.services.list_manager import ListManager

SEPARATOR = "-" * 64


def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer, e.g. 1st, 2nd, 11th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def ask(prompt: str) -> str:
    """Read a stripped line from the user."""
    return input(prompt).strip()


class LotterySession:
    """Holds the state of one lottery event and drives the interactive menus."""

    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.completed_prizes: set[str] = set()

    # ----- main loop ----------------------------------------------------
    def run(self) -> None:
        print("=" * 64)
        print("  Company Team Building Smart Lottery System")
        print("=" * 64)
        while True:
            print()
            print("1. Load participant list")
            print("2. View department groups")
            print("3. Configure prizes")
            print("4. Start lottery")
            print("5. View winner summary")
            print("6. Export results to file")
            print("0. Exit")
            choice = ask("Select an option: ")
            if choice == "1":
                self._load_participants()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self._prize_menu()
            elif choice == "4":
                self._run_lottery()
            elif choice == "5":
                self._show_results()
            elif choice == "6":
                self._export()
            elif choice == "0":
                print("Goodbye.")
                return
            else:
                print("Invalid option, please try again.")

    # ----- list management ----------------------------------------------
    def _load_participants(self) -> None:
        print("Participant list source:")
        print("  1. Built-in default team building list")
        print("  2. Custom txt file (one 'Name,Department' per line)")
        choice = ask("Select source [1/2]: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
            for error in errors:
                print(error)
        elif choice == "2":
            path = ask("Enter txt file path: ").strip('"').strip("'")
            users, errors = self.list_manager.load_custom_list(path)
            users = self._verify_custom_list(users, errors)
            if users is None:
                print("Upload terminated. Current list is unchanged.")
                return
        else:
            print("Invalid source selection.")
            return

        if not users:
            print("No valid participants loaded. Current list is unchanged.")
            return

        self.users = users
        self.lottery = LotteryManager(users)
        self.result_manager = ResultManager()
        self.completed_prizes = set()
        print(f"Loaded {len(users)} participants. Previous results (if any) were reset.")
        self._show_groups()

    def _verify_custom_list(
        self, users: list[User], errors: list[str]
    ) -> list[User] | None:
        """Detect duplicates/format errors and let the operator decide.

        Returns the list to use, or None if the upload is terminated.
        """
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            print(f"Verification passed: {len(valid)} participants.")
            return valid

        print("Verification found issues:")
        for error in errors:
            print(f"  {error}")
        for dup in duplicates:
            print(f"  Duplicate Name: '{dup.name}' ({dup.department})")

        print("Options:")
        print("  1. Remove invalid data (format errors and duplicates) and continue")
        print("  2. Remove specific entries by name and continue")
        print("  3. Terminate upload")
        choice = ask("Select option [1/2/3]: ")
        if choice == "1":
            return valid
        if choice == "2":
            names = ask("Enter names to remove (comma-separated): ")
            to_remove = {n.strip().casefold() for n in names.split(",") if n.strip()}
            remaining = [u for u in valid if u.name.strip().casefold() not in to_remove]
            removed = len(valid) - len(remaining)
            print(f"Removed {removed} entries.")
            return remaining
        return None

    def _show_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for group, count in info.items():
            print(f"  {group}: {count} people")
        print(f"  Total: {sum(info.values())} people in {len(info)} groups")

    # ----- prize configuration ------------------------------------------
    def _prize_menu(self) -> None:
        while True:
            print()
            print("Prize configuration:")
            print("  1. Load preset prizes")
            print("  2. Add custom prize")
            print("  3. View prizes (lottery order)")
            print("  4. Reorder lottery sequence")
            print("  0. Back")
            choice = ask("Select an option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                for error in errors:
                    print(f"Error: {error}")
                if not errors:
                    print("Preset prizes loaded.")
            elif choice == "2":
                self._add_prize()
            elif choice == "3":
                self._view_prizes()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "0":
                return
            else:
                print("Invalid option.")

    def _add_prize(self) -> None:
        name = ask("Prize name: ")
        try:
            quantity = int(ask("Quantity: "))
        except ValueError:
            print("Error: quantity must be an integer.")
            return
        if self.users:
            info = self.list_manager.get_group_info(self.users)
            print("Available department groups: " + ", ".join(info.keys()))
        groups_text = ask("Avoidance groups (comma-separated, blank for none): ")
        groups = [g.strip() for g in groups_text.split(",") if g.strip()]
        drawer = ask("Associated lottery person (blank for none): ")
        try:
            prize = self.prize_manager.add_prize(
                name, quantity, groups, drawer or None
            )
        except ValueError as exc:
            print(f"Error: {exc}")
            return
        print(f"Prize added at position {len(self.prize_manager.get_prizes())}: {prize.name}")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for index, prize in enumerate(prizes, start=1):
            print(f"  {index}. {prize.name} | quantity: {prize.quantity}")
            print(f"     drawer: {prize.associated_drawer or 'None'}")
            print(f"     avoid groups: {', '.join(prize.avoid_groups) or 'None'}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        text = ask("Enter new order as serial numbers separated by spaces: ")
        try:
            order = [int(token) for token in text.split()]
        except ValueError:
            print("Error: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
        else:
            print(f"Error: order must contain each number 1-{len(prizes)} exactly once.")

    # ----- lottery process ----------------------------------------------
    def _run_lottery(self) -> None:
        if not self.users or self.lottery is None:
            print("Load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Configure at least one prize first.")
            return
        pending = [p for p in prizes if p.name not in self.completed_prizes]
        if not pending:
            print("All prizes have already been drawn.")
            return
        print(f"Starting lottery with {len(pending)} round(s).")
        for round_no, prize in enumerate(prizes, start=1):
            if prize.name in self.completed_prizes:
                continue
            self._run_round(round_no, prize)
        print()
        print("Lottery finished. Winner summary:")
        self._show_results()

    def _run_round(self, round_no: int, prize: Prize) -> None:
        assert self.lottery is not None
        print()
        print(SEPARATOR)
        print(f"Round {round_no}: {prize.name}")
        print(f"Quantity: {prize.quantity} | Drawer: {prize.associated_drawer or 'None'}")
        print(f"Rule: {self.lottery.describe_rule(prize)}")
        print(SEPARATOR)

        round_winners: list[Winner] = []
        drawn = 0
        while drawn < prize.quantity:
            command = ask(
                f"[Enter] draw the {ordinal(drawn + 1)} winner | "
                "'pause' | 'terminate': "
            ).casefold()
            if command == "terminate":
                print("Round terminated by operator. Remaining winners skipped.")
                break
            if command == "pause":
                if not self._pause_round():
                    print("Round terminated by operator. Remaining winners skipped.")
                    break
            elif command:
                print(f"Unknown command '{command}'.")
                continue

            print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            user = self.lottery.draw(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            winner = self.result_manager.add_winner(user, prize)
            drawn += 1
            round_winners.append(winner)
            print(
                f"Congratulations {user.name} ({user.department}) "
                f"for winning {prize.name}!"
            )

        self.completed_prizes.add(prize.name)
        print()
        print(f"Round {round_no} results ({len(round_winners)} winner(s)):")
        if not round_winners:
            print("  (none)")
        for winner in round_winners:
            print(f"  {winner.winning_order}. {winner.name} ({winner.department})")

    def _pause_round(self) -> bool:
        """Pause the lottery. Returns True to resume, False to terminate the round."""
        print("Lottery paused.")
        while True:
            command = ask("Type 'continue' to resume or 'terminate' to end this round: ")
            command = command.casefold()
            if command == "continue":
                print("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            print("Unknown command.")

    # ----- results ------------------------------------------------------
    def _show_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            print("No winners yet.")
            return
        print(SEPARATOR)
        print("Winner list (by prize level, then winning order):")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            print(f"{prize_name}:")
            for winner in winners:
                print(
                    f"  {winner.winning_order:>3}. {winner.name} "
                    f"({winner.department})"
                )
        print(SEPARATOR)

    def _export(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not any(self.result_manager.get_winners_by_prize(prizes).values()):
            print("No winners to export.")
            return
        _, message = self.result_manager.export_to_csv(prizes)
        print(message)


def main() -> int:
    """Entry point for the interactive program."""
    session = LotterySession()
    try:
        session.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting lottery system.")
    return 0

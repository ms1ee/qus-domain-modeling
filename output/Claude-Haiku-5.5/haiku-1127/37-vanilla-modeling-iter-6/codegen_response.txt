"""Domain models used across the lottery system."""

from dataclasses import dataclass, field
from typing import Optional

from lottery_system.config import DEFAULT_LEVEL, LEVEL_KEYWORDS


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round.

    avoid_groups: department groups whose members are excluded from this prize.
    associated_drawer: the lottery person bound to this prize (or None).
    """

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    @property
    def level(self) -> int:
        """Prize level derived from the name; smaller value means higher prize."""
        lowered = self.name.strip().lower()
        for keyword, value in LEVEL_KEYWORDS.items():
            if lowered.startswith(keyword):
                return value
        return DEFAULT_LEVEL

"""Participant list loading, verification and group extraction."""

from lottery_system.config import DEFAULT_PARTICIPANT_LINES
from lottery_system.models import User


class ListManager:
    """Loads participant lists from the built-in data or a custom txt file."""

    def __init__(self) -> None:
        self.load_mode: str = ""
        self.source_file_path: str = ""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        self.load_mode = "default"
        self.source_file_path = ""
        return self._parse_lines(DEFAULT_PARTICIPANT_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        self.load_mode = "custom"
        self.source_file_path = path
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{line}' "
                    "(expected format 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    @staticmethod
    def validate_users(users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). Only the first occurrence of a name is valid."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = " ".join(user.name.split()).casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    @staticmethod
    def get_group_info(users: list[User]) -> dict[str, int]:
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items(), key=lambda item: item[0]))

"""Prize library management."""

from lottery_system.config import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the ordered list of prizes. Starts with no prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, [], drawer)
            except ValueError as exc:
                errors.append(f"Cannot load preset prize '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if any(p.name.casefold() == clean_name.casefold() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """order holds 1-based current positions in the desired new sequence."""
        n = len(self._prizes)
        try:
            is_permutation = sorted(order) == list(range(1, n + 1))
        except TypeError:
            return False
        if not is_permutation:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, drawing and fallback protection."""

import random

from lottery_system.config import (
    CHENG_RU_GROUP,
    CHEN_QIXIAN,
    FALLBACK_TARGET_DRAWER,
    FALLBACK_TRIGGER_DRAWERS,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User


def _norm(text: str | None) -> str:
    return " ".join((text or "").split()).casefold()


class LotteryManager:
    """Performs lottery rounds while enforcing all eligibility rules."""

    def __init__(self, users: list[User]) -> None:
        self._users: list[User] = list(users)
        self.winners: list[User] = []
        self._winner_keys: set[str] = set()
        # Records whether each recent draw by Chen Qixian / Jin Duo contained
        # Cheng Ru group members (True) or not (False). Keeps last two entries.
        self.last_two_draw_results: list[bool] = []
        self._round_hits: list[User] = []

    # ----- helpers -------------------------------------------------------
    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return _norm(user.department) == _norm(CHENG_RU_GROUP)

    def _is_previous_winner(self, user: User) -> bool:
        return _norm(user.name) in self._winner_keys

    @staticmethod
    def _is_li_bin(prize: Prize) -> bool:
        return _norm(prize.associated_drawer) == _norm(LI_BIN)

    @staticmethod
    def _is_ma_jiaqi(prize: Prize) -> bool:
        return _norm(prize.associated_drawer) == _norm(FALLBACK_TARGET_DRAWER)

    def _is_fallback_active(self, prize: Prize) -> bool:
        if not self._is_ma_jiaqi(prize):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and recent == [False, False]

    # ----- eligibility ---------------------------------------------------
    def _eligible_pool(self, prize: Prize) -> list[User]:
        pool = [u for u in self._users if not self._is_previous_winner(u)]
        if self._is_li_bin(prize):
            # Special rule: only previous winners are filtered.
            return pool
        if self._is_fallback_active(prize):
            # Fallback: only Cheng Ru group members, overriding avoidance groups.
            return [u for u in pool if self._is_cheng_ru(u)]
        avoid = {_norm(group) for group in prize.avoid_groups}
        return [u for u in pool if _norm(u.department) not in avoid]

    def pool_size(self, prize: Prize) -> int:
        return len(self._eligible_pool(prize))

    def describe_rule(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        if self._is_li_bin(prize):
            parts.append("Special rule for Li Bin: avoidance groups are not applied")
        elif prize.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}")
        if self._is_ma_jiaqi(prize):
            if self._is_fallback_active(prize):
                parts.append(
                    f"Fallback active: only {CHENG_RU_GROUP} members are eligible"
                )
            else:
                parts.append(
                    f"Fallback applies if {CHEN_QIXIAN} and {JIN_DUO} both missed "
                    f"{CHENG_RU_GROUP} members"
                )
        return "; ".join(parts)

    # ----- drawing -------------------------------------------------------
    def start_round(self, prize: Prize) -> None:
        self._round_hits = []

    def draw_one(self, prize: Prize) -> User | None:
        pool = self._eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        self._winner_keys.add(_norm(user.name))
        self._round_hits.append(user)
        return user

    def end_round(self, prize: Prize) -> list[User]:
        hits = list(self._round_hits)
        drawer = _norm(prize.associated_drawer)
        if drawer in (_norm(CHEN_QIXIAN), _norm(JIN_DUO)):
            self.last_two_draw_results.append(any(self._is_cheng_ru(u) for u in hits))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(MA_JIAQI):
            # The fallback check for this consecutive sequence is consumed.
            self.last_two_draw_results = []
        self._round_hits = []
        return hits

"""Winner recording, grouped summaries and CSV/txt export."""

import csv
import datetime
import os
from dataclasses import dataclass

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize: Prize
    winning_order: int
    winning_time: datetime.datetime

    @property
    def prize_name(self) -> str:
        return self.prize.name


class ResultManager:
    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._order_counter: int = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        self._order_counter += 1
        winner = Winner(
            user=user,
            prize=prize,
            winning_order=self._order_counter,
            winning_time=datetime.datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Prize names ordered by level (high to low), then addition order.
        Winners within each prize are in winning time order."""
        ordered = sorted(prizes, key=lambda prize: prize.level)  # stable sort
        result: dict[str, list[Winner]] = {}
        for prize in ordered:
            result.setdefault(prize.name, [])
        for winner in self._winners:
            if winner.prize_name in result:
                result[winner.prize_name].append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        date_str = datetime.date.today().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_str}.txt")
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winners in grouped.values():
                    for winner in winners:
                        writer.writerow([
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from lottery_system.models import User

def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotterySystemCLI:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.has_results = False

    # ----- utilities -----------------------------------------------------
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    def _print_menu(self) -> None:
        print("\n========== Team Building Smart Lottery ==========")
        print("1. Load built-in default team building list")
        print("2. Load custom txt list (Name,Department Group per line)")
        print("3. View department groups")
        print("4. Load preset prizes")
        print("5. Add custom prize")
        print("6. View and reorder prizes")
        print("7. Start lottery")
        print("8. View winner list")
        print("9. Export winner list to file")
        print("0. Exit")

    def run(self) -> None:
        print("Welcome to the Company Team Building Smart Lottery System.")
        while True:
            self._print_menu()
            try:
                choice = self._ask("Select an option: ")
                if choice == "1":
                    self._load_default_list()
                elif choice == "2":
                    self._load_custom_list()
                elif choice == "3":
                    self._view_groups()
                elif choice == "4":
                    self._load_presets()
                elif choice == "5":
                    self._add_prize()
                elif choice == "6":
                    self._view_and_reorder_prizes()
                elif choice == "7":
                    self._start_lottery()
                elif choice == "8":
                    self._view_winners()
                elif choice == "9":
                    self._export_results()
                elif choice == "0":
                    print("Goodbye!")
                    return
                else:
                    print("Invalid option, please try again.")
            except (EOFError, KeyboardInterrupt):
                print("\nInput closed. Exiting.")
                return

    # ----- list management ----------------------------------------------
    def _load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._apply_loaded_list(users, errors, "built-in default list")

    def _load_custom_list(self) -> None:
        path = self._ask("Enter the path of the txt file: ")
        if not path:
            print("No path entered. Upload cancelled.")
            return
        users, errors = self.list_manager.load_custom_list(path)
        self._apply_loaded_list(users, errors, f"file '{path}'")

    def _apply_loaded_list(self, users: list[User], errors: list[str], source: str) -> None:
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            self.users = valid
            print(f"Loaded {len(valid)} participants from {source}.")
            return

        print(f"Verification of {source} found issues:")
        for error in errors:
            print(f"  {error}")
        for dup in duplicates:
            print(f"  Duplicate Name: '{dup.name}' ({dup.department}) already exists, ignored.")
        print(f"Valid participants: {len(valid)}")
        print("1. Remove invalid data and keep valid participants")
        print("2. Terminate upload")
        choice = self._ask("Select an option: ")
        if choice == "1":
            if not valid:
                print("No valid participants remain. Participant list unchanged.")
                return
            self.users = valid
            print(f"Loaded {len(valid)} participants from {source}.")
        else:
            print("Upload terminated. Participant list unchanged.")

    def _view_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current department groups:")
        for index, (group, count) in enumerate(info.items(), start=1):
            print(f"  {index}. {group}: {count} people")
        print(f"Total participants: {len(self.users)}")

    # ----- prize management ---------------------------------------------
    def _load_presets(self) -> None:

        if errors:
            for error in errors:
                print(error)
        else:
            print("Preset prizes loaded.")

    def _add_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity_text = self._ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            print("Quantity must be a positive integer.")
            return

        groups = sorted({u.department for u in self.users})
        avoid: list[str] = []
        if groups:
            print("Available department groups:")
            for index, group in enumerate(groups, start=1):
                print(f"  {index}. {group}")
        raw_avoid = self._ask(
            "Avoidance groups (serial numbers or names, comma-separated, blank for none): "
        )
        for token in [t.strip() for t in raw_avoid.split(",") if t.strip()]:
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                avoid.append(groups[int(token) - 1])
            else:
                avoid.append(token)

        drawer = self._ask("Associated lottery person (blank for none): ")
        try:
            prize = self.prize_manager.add_prize(name, quantity, avoid, drawer or None)
            print(f"Prize added: {prize.name} (quantity {prize.quantity}).")
        except ValueError as exc:
            print(f"Cannot add prize: {exc}")

    def _view_and_reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._print_prizes(prizes)
        raw = self._ask(
            "Enter new order as serial numbers separated by spaces (blank to keep): "
        )
        if not raw:
            return
        try:
            order = [int(token) for token in raw.split()]
        except ValueError:
            print("Invalid order: please enter integers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self._print_prizes(self.prize_manager.get_prizes())
        else:
            print("Invalid order: every serial number must be used exactly once.")

    @staticmethod
    def _print_prizes(prizes) -> None:
        print("Prize sequence:")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            print(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoid: {avoid} | Lottery person: {drawer}"
            )

    # ----- lottery process ----------------------------------------------
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        self.result_manager = ResultManager()
        self.has_results = False
        lottery = LotteryManager(self.users)
        print("\nLottery begins!")
        print("Commands: [Enter] draw next winner | pause | terminate (end current round)")

        for round_no, prize in enumerate(prizes, start=1):
            print(f"\n--- Round {round_no}: {prize.name} ---")
            print(f"Quantity: {prize.quantity}")
            print(f"Rule: {lottery.describe_rule(prize)}")
            print(f"Eligible participants at start: {lottery.pool_size(prize)}")
            round_winners = self._run_round(lottery, prize)
            print(f"Round {round_no} results:")
            if not round_winners:
                print("  (no winners in this round)")
            for winner in round_winners:
                print(
                    f"  #{winner.winning_order} {winner.user.name} "
                    f"({winner.user.department}) - {prize.name}"
                )

        self.has_results = True
        print("\nAll rounds completed.")
        self._print_summary(prizes)

    def _run_round(self, lottery: LotteryManager, prize) -> list[Winner]:
        lottery.start_round(prize)
        round_winners: list[Winner] = []
        drawn = 0
        while drawn < prize.quantity:
            print(f"Currently drawing the {_ordinal(drawn + 1)} winner...")
            if self._await_action() == "terminate":
                print("Round terminated manually; remaining winners are skipped.")
                break
            user = lottery.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize; round ends early.")
                break
            drawn += 1
            winner = self.result_manager.add_winner(user, prize)
            round_winners.append(winner)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")
        lottery.end_round(prize)
        return round_winners

    def _await_action(self) -> str:
        """Returns 'draw' or 'terminate'. Handles pause/continue commands."""
        while True:
            command = self._ask("Press Enter to draw (or type pause / terminate): ").lower()
            if command == "":
                return "draw"
            if command == "terminate":
                return "terminate"
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("(paused) > ").lower()
                    if resume == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if resume == "terminate":
                        return "terminate"
                    print("Unknown command. Use 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    def _print_summary(self, prizes) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n========== Final Winner List ==========")
        for prize_name, winners in grouped.items():
            print(f"{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  #{winner.winning_order} {winner.user.name} "
                    f"({winner.user.department}) at "
                    f"{winner.winning_time.strftime('%Y-%m-%d %H:%M:%S')}"
                )

    # ----- results ------------------------------------------------------
    def _view_winners(self) -> None:
        if not self.has_results:
            print("No lottery results yet.")
            return
        self._print_summary(self.prize_manager.get_prizes())

    def _export_results(self) -> None:
        if not self.has_results:
            print("No lottery results to export.")
            return

"""Grid-based path planning for the 800x800 street map.

The map is divided into 100x100 unit cells. Streets run along the grid
lines (x or y is a multiple of 100). Restaurants, customers and street
intersections are the points where both coordinates are multiples of 100.
Paths are computed over the intersection graph with Dijkstra's algorithm,
penalising right-angle turns so that straight runs are preferred.
"""

import heapq
import itertools
import logging
import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_NODES = MAP_SIZE // CELL_SIZE + 1  # 9 intersections per axis
STEP_COST = CELL_SIZE                    # cost of moving one block
TURN_PENALTY = 10                        # extra cost for a right-angle turn
DIRECTIONS = ((CELL_SIZE, 0), (-CELL_SIZE, 0), (0, CELL_SIZE), (0, -CELL_SIZE))

logger = logging.getLogger("dispatch.path")


def snap_to_cell(x, y):
    """Snap a point to the top-left corner of the grid cell containing it.

    Args:
        x: X coordinate (0-800).
        y: Y coordinate (0-800).

    Returns:
        Tuple ``(cell_x, cell_y)`` with both values multiples of 100.
    """
    return (int(x) // CELL_SIZE * CELL_SIZE, int(y) // CELL_SIZE * CELL_SIZE)


def is_intersection(x, y):
    """Return True if the point lies on a street intersection."""
    return x % CELL_SIZE == 0 and y % CELL_SIZE == 0


def is_on_street(x, y):
    """Return True if the point lies on any street (grid line)."""
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


def in_bounds(x, y):
    """Return True if the point lies inside the 800x800 map."""
    return 0 <= x <= MAP_SIZE and 0 <= y <= MAP_SIZE


def manhattan(a, b):
    """Return the Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def euclidean(a, b):
    """Return the Euclidean distance between two points."""
    return math.hypot(a[0] - b[0], a[1] - b[1])


def route_length(points):
    """Return the total travelled distance along a list of points."""
    return sum(manhattan(points[i], points[i + 1]) for i in range(len(points) - 1))


class PathPlanner:
    """Plans axis-aligned routes along streets between two points."""

    def entry_nodes(self, x, y):
        """Return the intersections a point can leave from, with costs.

        A point on an intersection returns itself with cost 0. A point in
        the middle of a block returns the two surrounding intersections,
        with the distance needed to reach each one.

        Args:
            x: X coordinate of the start point.
            y: Y coordinate of the start point.

        Returns:
            List of ``((node_x, node_y), cost)`` tuples.

        Raises:
            ValueError: If the point is not on a street.
        """
        if is_intersection(x, y):
            return [((x, y), 0)]
        if not is_on_street(x, y):
            raise ValueError(f"Point ({x}, {y}) is not on a street")
        base_x = x - x % CELL_SIZE
        base_y = y - y % CELL_SIZE
        if y % CELL_SIZE == 0:
            candidates = [
                ((base_x, y), x - base_x),
                ((base_x + CELL_SIZE, y), base_x + CELL_SIZE - x),
            ]
        else:
            candidates = [
                ((x, base_y), y - base_y),
                ((x, base_y + CELL_SIZE), base_y + CELL_SIZE - y),
            ]
        return [(node, cost) for node, cost in candidates if in_bounds(*node)]

    def plan(self, start, goal):
        """Compute the waypoints from ``start`` to ``goal`` along streets.

        Args:
            start: ``(x, y)`` current position of the rider.
            goal: ``(x, y)`` destination intersection.

        Returns:
            List of waypoints (intersections) ending at ``goal``. The start
            point itself is not included. An empty list means no movement.

        Raises:
            ValueError: If the goal is not an intersection or no route exists.
        """
        start = (int(start[0]), int(start[1]))
        goal = (int(goal[0]), int(goal[1]))
        if start == goal:
            return []
        if not is_intersection(*goal):
            raise ValueError(f"Goal {goal} is not a street intersection")

        entries = self.entry_nodes(*start)
        if not entries:
            raise ValueError(f"No street access from {start}")
        for node, _ in entries:
            if node == goal:
                return [goal]

        best = {}
        parent = {}
        heap = []
        counter = itertools.count()
        for node, cost in entries:
            state = (node, None)
            best[state] = cost
            parent[state] = None
            heapq.heappush(heap, (cost, next(counter), state))

        found = None
        while heap:
            cost, _, state = heapq.heappop(heap)
            if cost > best.get(state, math.inf):
                continue
            node, direction = state
            if node == goal:
                found = state
                break
            for step in DIRECTIONS:
                nxt = (node[0] + step[0], node[1] + step[1])
                if not in_bounds(*nxt):
                    continue
                new_cost = cost + STEP_COST
                if direction is not None and step != direction:
                    new_cost += TURN_PENALTY
                next_state = (nxt, step)
                if new_cost < best.get(next_state, math.inf):
                    best[next_state] = new_cost
                    parent[next_state] = state
                    heapq.heappush(heap, (new_cost, next(counter), next_state))

        if found is None:
            raise ValueError(f"No route from {start} to {goal}")

        nodes = []
        state = found
        while state is not None:
            nodes.append(state[0])
            state = parent[state]
        nodes.reverse()
        if nodes and nodes[0] == start:
            nodes = nodes[1:]
        logger.debug("Planned route %s -> %s with %d waypoints", start, goal, len(nodes))
        return nodes

"""Rider model: state machine, street movement, and event logging.

State machine:
    IDLE (0)        -> PICKING_UP (1) when an order is accepted
    PICKING_UP (1)  -> DELIVERING (2) on arrival at the restaurant
    DELIVERING (2)  -> PICKING_UP (1) when a delivery completes and more
                                       orders remain, otherwise IDLE (0)

Riders travel only along streets. Riders do not block each other, so
streets are shared and no lock-based deadlock can occur.
"""

import logging
import math

import globalvar
from path_planner import PathPlanner, is_intersection, route_length

logger = logging.getLogger("dispatch.rider")

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

DEFAULT_SPEED = 20
SPAWN_X = 400
SPAWN_Y = 400


def _now():
    """Return the current simulation tick."""
    return globalvar.get_value("time", 0) or 0


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    Attributes:
        rider_id: Unique integer identifier.
        x, y: Current integer coordinates.
        speed: Movement budget in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: All orders ever assigned (completed ones kept as history).
        special_case: True when the rider is mid-block or directly above or
            below its current target.
        log: List of event dicts with keys "event", "order_id", "time",
            "x", "y", "state" and "rider_id".
        path: Remaining waypoints for the current leg.
        target: Current destination intersection.
        distance_travelled: Total distance moved so far.
    """

    def __init__(self, rider_id, x=SPAWN_X, y=SPAWN_Y, speed=DEFAULT_SPEED):
        """Create a rider.

        Args:
            rider_id: Unique rider identifier.
            x: Starting X coordinate (default 400).
            y: Starting Y coordinate (default 400).
            speed: Units moved per tick (default 20).
        """
        self.rider_id = rider_id
        self.x = x
        self.y = y
        self.speed = speed
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.path = []
        self.target = None
        self.distance_travelled = 0
        self._planner = PathPlanner()
        self._record("spawn", None)
        logger.info("Rider #%s spawned at (%s, %s)", rider_id, x, y)

    @property
    def position(self):
        """Return the current position as an ``(x, y)`` tuple."""
        return (self.x, self.y)

    def current_order(self):
        """Return the order currently being served, or None."""
        for order in self.orders:
            if order.status != "completed":
                return order
        return None

    def active_order_count(self):
        """Return the number of assigned orders that are not completed."""
        return sum(1 for order in self.orders if order.status != "completed")

    def accept_order(self, order):
        """Handle a newly assigned order (called by ``Order.assign_to``).

        Logs the acceptance. If the rider is idle the pickup leg starts
        immediately; otherwise the order is queued behind current work.

        Args:
            order: The order that has just been appended to ``self.orders``.
        """
        self._record("accepted_order", order.id)
        if self.state == IDLE and self.current_order() is order:
            self._begin_pickup(order)

    def move(self):
        """Advance the rider by one tick.

        Spends up to ``speed`` units of movement along the planned route and
        processes any arrival events (restaurant pickup, customer delivery)
        that result from this movement.
        """
        order = self.current_order()
        if order is None:
            if self.state != IDLE:
                self._go_idle()
            return
        if self.state == IDLE:
            self._begin_pickup(order)

        self.distance_travelled += self._advance()

        for _ in range(3):
            if self.path:
                break
            order = self.current_order()
            if order is None:
                self._go_idle()
                break
            if self.state == PICKING_UP and self.position == order.restaurant:
                self._arrive_at_merchant(order)
            elif self.state == DELIVERING and self.position == order.customer:
                self._deliver(order)
            else:
                break

    def route_to_order(self, order):
        """Return the planned route from the rider's position through ``order``.

        The route includes pickup and delivery legs of all unfinished orders
        queued up to and including ``order``. Used for route previews.

        Args:
            order: The order whose route should be previewed.

        Returns:
            List of ``(x, y)`` points starting at the rider's position.
        """
        points = [self.position]
        cur = self.position
        stops = []
        current = self.current_order()
        for queued in self.orders:
            if queued.status == "completed":
                continue
            if queued is current and self.state == DELIVERING:
                stops.append(queued.customer)
            else:
                stops.append(queued.restaurant)
                stops.append(queued.customer)
            if queued is order:
                break
        for stop in stops:
            if stop == cur:
                continue
            points.extend(self._planner.plan(cur, stop))
            cur = stop
        return points

    def remaining_distance(self):
        """Return the length of the remaining route for the current leg."""
        return route_length([self.position] + self.path)

    def estimated_ticks(self):
        """Return the estimated number of ticks to finish the current leg."""
        if self.speed <= 0:
            return math.inf
        return math.ceil(self.remaining_distance() / self.speed)

    def _record(self, event, order_id):
        """Append an event entry to the rider log."""
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": _now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "rider_id": self.rider_id,
        })

    def _set_route(self, goal):
        """Plan a route to ``goal`` and update the special-case flag."""
        self.target = goal
        self.path = self._planner.plan(self.position, goal)
        self.special_case = (self.x == goal[0]) or not is_intersection(self.x, self.y)

    def _begin_pickup(self, order):
        """Switch to the picking-up state and route to the restaurant."""
        self.state = PICKING_UP
        order.status = "picking_up"
        self._set_route(order.restaurant)
        logger.info("Rider #%s heading to restaurant %s for order #%s",
                    self.rider_id, order.restaurant, order.id)

    def _arrive_at_merchant(self, order):
        """Handle arrival at the restaurant and start the delivery leg."""
        self.state = DELIVERING
        order.status = "delivering"
        order.pickup_time = _now()
        self._record("arrived_at_merchant", order.id)
        self._set_route(order.customer)

    def _deliver(self, order):
        """Handle delivery to the customer and move on to the next order."""
        order.complete(_now())
        self._record("delivered_to_customer", order.id)
        logger.info("Rider #%s delivered order #%s", self.rider_id, order.id)
        nxt = self.current_order()
        if nxt is None:
            self._go_idle()
        else:
            self.state = PICKING_UP
            nxt.status = "picking_up"
            self._set_route(nxt.restaurant)

    def _go_idle(self):
        """Release the rider back to the idle pool."""
        self.state = IDLE
        self.path = []
        self.target = None
        self.special_case = False

    def _advance(self):
        """Move along the planned waypoints using up to ``speed`` units.

        Returns:
            The number of units actually moved.
        """
        remaining = self.speed
        moved = 0
        while remaining > 0 and self.path:
            wx, wy = self.path[0]
            dx = wx - self.x
            dy = wy - self.y
            dist = abs(dx) + abs(dy)
            if dist <= remaining:
                self.x, self.y = wx, wy
                remaining -= dist
                moved += dist
                self.path.pop(0)
            else:
                if dx != 0:
                    self.x += remaining if dx > 0 else -remaining
                else:
                    self.y += remaining if dy > 0 else -remaining
                moved += remaining
                remaining = 0
        return moved

"""Order data model: coordinates, lifecycle, and delivery statistics."""

import logging

import globalvar
from path_planner import MAP_SIZE, manhattan, snap_to_cell

logger = logging.getLogger("dispatch.order")

COMPLETED = "completed"


def _now():
    """Return the current simulation tick."""
    return globalvar.get_value("time", 0) or 0


def _validate_point(point):
    """Validate an ``(x, y)`` input pair.

    Raises:
        ValueError: If the point is malformed, non-integer or out of range.
    """
    try:
        x, y = point
    except (TypeError, ValueError):
        raise ValueError(f"Location must be an (x, y) pair, got {point!r}")
    for value in (x, y):
        if not isinstance(value, int) or isinstance(value, bool):
            raise ValueError(f"Coordinates must be integers, got {value!r}")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"Coordinate {value} out of range 0-{MAP_SIZE}")
    return int(x), int(y)


def _find_rider(rider_id):
    """Return the rider with ``rider_id`` from the global roster, or None."""
    for rider in globalvar.get_value("riders") or []:
        if rider.rider_id == rider_id:
            return rider
    return None


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order number.
        restaurant: Snapped ``(x, y)`` restaurant location.
        customer: Snapped ``(x, y)`` customer location.
        status: "pending", "assigned", "picking_up", "delivering" or "completed".
        rider_id: Id of the assigned rider, or None.
        start_time: Tick at which the order was created.
        pickup_time: Tick at which the rider reached the restaurant.
        complete_time: Tick at which the order was delivered.
        distance: Manhattan distance between restaurant and customer.
        time_taken: Ticks between creation and completion.
        efficiency: Distance covered per tick of total time taken.
    """

    _id_counter = 0

    def __init__(self, res_pos, cus_pos):
        """Create an order from raw restaurant and customer coordinates.

        Args:
            res_pos: ``(x, y)`` restaurant coordinates in the range 0-800.
            cus_pos: ``(x, y)`` customer coordinates in the range 0-800.

        Raises:
            ValueError: If coordinates are malformed or out of range.
        """
        restaurant = _validate_point(res_pos)
        customer = _validate_point(cus_pos)
        Order._id_counter += 1
        self.id = Order._id_counter
        self.restaurant = snap_to_cell(*restaurant)
        self.customer = snap_to_cell(*customer)
        self.status = "pending"
        self.rider_id = None
        self.start_time = _now()
        self.pickup_time = None
        self.complete_time = None
        self.distance = manhattan(self.restaurant, self.customer)
        self.time_taken = None
        self.efficiency = None

    def assign_to(self, rider_id):
        """Assign this order to the rider with ``rider_id``.

        The order is appended to ``rider.orders`` and the rider is notified.

        Args:
            rider_id: Identifier of the target rider.

        Raises:
            ValueError: If the rider does not exist or the order is completed.
        """
        if self.status == COMPLETED:
            raise ValueError(f"Order #{self.id} is already completed")
        rider = _find_rider(rider_id)
        if rider is None:
            raise ValueError(f"Rider #{rider_id} does not exist")
        self.rider_id = rider_id
        self.status = "assigned"
        rider.orders.append(self)
        rider.accept_order(self)
        logger.info("Order #%s assigned to rider #%s", self.id, rider_id)

    def complete(self, time_now):
        """Mark the order as delivered and record timing statistics.

        Args:
            time_now: Tick at which delivery happened.
        """
        self.status = COMPLETED
        self.complete_time = time_now
        self.time_taken = max(0, time_now - self.start_time)
        self.efficiency = round(self.distance / max(self.time_taken, 1), 2)

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.restaurant),
            "customer": list(self.customer),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

"""Order creation, assignment, and dispatch logic."""

import logging

import globalvar
from order_model import Order
from path_planner import euclidean

logger = logging.getLogger("dispatch.dispatch")


def _find_rider(rider_id):
    """Return the rider with ``rider_id`` from the global roster, or None."""
    for rider in globalvar.get_value("riders") or []:
        if rider.rider_id == rider_id:
            return rider
    return None


def find_best_rider_for_order(order):
    """Choose the rider that should receive ``order``.

    The rider with the fewest unfinished orders is preferred. Ties are broken
    by the Euclidean distance from the rider to the restaurant, so the
    closest rider wins among equally loaded riders.

    Args:
        order: The order that needs a rider.

    Returns:
        The selected Rider, or None if the roster is empty.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None

    def sort_key(rider):
        return (
            rider.active_order_count(),
            euclidean((rider.x, rider.y), order.restaurant),
            rider.rider_id,
        )

    return min(riders, key=sort_key)


class OrderManager:
    """Creates orders, keeps the pending queue, and dispatches riders."""

    def __init__(self, system):
        """Create a manager bound to a SystemManager.

        Args:
            system: SystemManager used for rider recruitment.
        """
        self.system = system

    def create_order(self, res_pos, cus_pos):
        """Create an order and add it to the history and pending queue.

        Args:
            res_pos: ``(x, y)`` restaurant coordinates.
            cus_pos: ``(x, y)`` customer coordinates.

        Returns:
            The newly created Order.
        """
        order = Order(res_pos, cus_pos)
        globalvar.get_value("orders").append(order)
        globalvar.get_value("pending_orders").append(order)
        logger.info("Order #%s created: %s -> %s", order.id, order.restaurant, order.customer)
        return order

    def dispatch_pending(self):
        """Assign pending orders to riders, recruiting when needed.

        For each pending order a rider is recruited if pending orders exceed
        idle riders and funds allow it. The order is then assigned to the best
        rider. Processing stops when no rider is available.

        Returns:
            Tuple ``(assigned_orders, new_rider_ids)``.
        """
        pending = globalvar.get_value("pending_orders")
        assigned = []
        new_rider_ids = []
        for order in list(pending):
            recruited = self.system.maybe_recruit(len(pending))
            if recruited is not None:
                new_rider_ids.append(recruited.rider_id)
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            pending.remove(order)
            order.assign_to(rider.rider_id)
            assigned.append(order)
        return assigned, new_rider_ids

    def place_order(self, res_pos, cus_pos):
        """Create an order and try to dispatch it immediately.

        Args:
            res_pos: ``(x, y)`` restaurant coordinates.
            cus_pos: ``(x, y)`` customer coordinates.

        Returns:
            Dict with keys "order" (Order), "rider" (Rider or None when the
            order is still pending) and "new_riders" (list of new rider ids).
        """
        order = self.create_order(res_pos, cus_pos)
        _, new_rider_ids = self.dispatch_pending()
        rider = _find_rider(order.rider_id) if order.rider_id is not None else None
        return {"order": order, "rider": rider, "new_riders": new_rider_ids}

"""System-level management: funds, rider roster, recruitment, and ticks."""

import logging

import globalvar
from rider import DEFAULT_SPEED, IDLE, Rider

logger = logging.getLogger("dispatch.system")


def init_system():
    """Initialise global variables to their startup defaults."""
    globalvar._init()
    logger.info("System initialised with funds=%s", globalvar.get_value("funds"))


class SystemManager:
    """Manages system funds, the rider roster, recruitment and simulation ticks."""

    def get_funds(self):
        """Return the current system fund balance."""
        return globalvar.get_value("funds")

    def adjust_funds(self, delta):
        """Add ``delta`` (may be negative) to the fund balance.

        Returns:
            The new fund balance.
        """
        new_value = int(globalvar.get_value("funds")) + int(delta)
        globalvar.set_value("funds", new_value)
        return new_value

    def idle_rider_count(self):
        """Return the number of riders in the idle state."""
        return sum(1 for rider in globalvar.get_value("riders") if rider.state == IDLE)

    def add_rider(self):
        """Create a new rider at (400, 400) and add it to the roster.

        Returns:
            The newly created Rider.
        """
        counter = globalvar.get_value("rider_id_counter") + 1
        globalvar.set_value("rider_id_counter", counter)
        speed = globalvar.get_value("rider_speed", DEFAULT_SPEED)
        rider = Rider(counter, speed=speed)
        globalvar.get_value("riders").append(rider)
        return rider

    def maybe_recruit(self, pending_count):
        """Recruit a rider when funds and demand allow it.

        A rider is created when funds are at least the recruit threshold and
        the number of pending orders exceeds the number of idle riders. The
        recruit cost is deducted from the funds.

        Args:
            pending_count: Number of orders waiting for a rider.

        Returns:
            The new Rider, or None if no rider was recruited.
        """
        min_funds = globalvar.get_value("recruit_min_funds", 300)
        cost = globalvar.get_value("recruit_cost", 500)
        if self.get_funds() >= min_funds and pending_count > self.idle_rider_count():
            self.adjust_funds(-cost)
            rider = self.add_rider()
            logger.info("Recruited rider #%s (cost %s)", rider.rider_id, cost)
            return rider
        return None

    def remove_rider(self, rider_id):
        """Remove an idle rider that has no unfinished orders.

        Returns:
            True if the rider was removed, otherwise False.
        """
        riders = globalvar.get_value("riders")
        for rider in riders:
            if rider.rider_id == rider_id:
                if rider.state != IDLE or rider.active_order_count() > 0:
                    return False
                riders.remove(rider)
                logger.info("Removed rider #%s", rider_id)
                return True
        return False

    def tick(self):
        """Advance simulated time by one tick and move every rider.

        Returns:
            The new simulation time.
        """
        new_time = globalvar.get_value("time", 0) + 1
        globalvar.set_value("time", new_time)
        for rider in list(globalvar.get_value("riders")):
            rider.move()
        return new_time

    def is_running(self):
        """Return the main loop running flag."""
        return bool(globalvar.get_value("running"))

    def stop(self):
        """Request that the main loop terminate."""
        globalvar.set_value("running", False)

    def check_consistency(self):
        """Check internal state for inconsistencies.

        Returns:
            List of human-readable issue descriptions (empty when healthy).
        """
        issues = []
        funds = globalvar.get_value("funds")
        if not isinstance(funds, int):
            issues.append("system funds is not an integer")
        riders = globalvar.get_value("riders")
        ids = [rider.rider_id for rider in riders]
        if len(ids) != len(set(ids)):
            issues.append("duplicate rider ids detected")
        if ids and max(ids) > globalvar.get_value("rider_id_counter"):
            issues.append("rider id counter is behind existing rider ids")
        for rider in riders:
            active = rider.active_order_count()
            if (rider.state == IDLE) != (active == 0):
                issues.append(
                    f"rider #{rider.rider_id} state {rider.state} inconsistent "
                    f"with {active} active order(s)"
                )
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    issues.append(
                        f"order #{order.id} listed under rider #{rider.rider_id} "
                        f"but assigned to #{order.rider_id}"
                    )
        for order in globalvar.get_value("pending_orders"):
            if order.status != "pending":
                issues.append(f"pending order #{order.id} has status {order.status}")
        return issues

from order_model import COMPLETED
from path_planner import MAP_SIZE, CELL_SIZE
from rider import STATE_NAMES

SCALE = 50                          # map units per character column/row
MAP_CHARS = MAP_SIZE // SCALE + 1   # 17 characters per axis (9 intersections)

SYMBOL_RIDER = "R"
SYMBOL_RESTAURANT = "S"
SYMBOL_CUSTOMER = "C"
SYMBOL_EMPTY = "."
SYMBOL_STREET = "#"
SYMBOL_ROUTE = "*"

MAP_LEGEND = ("Legend: R rider  S restaurant  C customer  "
              ". intersection  # street  * planned route")


def _to_col(x):
    """Convert a map X coordinate to a character column."""
    return int(round(x / SCALE))


def _to_row(y):
    """Convert a map Y coordinate to a character row (north at the top)."""
    return int(round((MAP_SIZE - y) / SCALE))


class DisplayManager:
    """Builds text output for status, orders, rider logs and the map."""

    def render_map(self, riders=(), active_orders=(), route=None):
        """Render the street map as an ASCII string.

        Args:
            riders: Riders to draw as 'R'.
            active_orders: Orders whose restaurants ('S') and customers ('C')
                are drawn.
            route: Optional list of ``(x, y)`` points drawn as '*'.

        Returns:
            Multi-line string containing the map and a legend.
        """
        grid = [[" "] * MAP_CHARS for _ in range(MAP_CHARS)]
        for row in range(MAP_CHARS):
            for col in range(MAP_CHARS):
                if row % 2 == 0 and col % 2 == 0:
                    grid[row][col] = SYMBOL_EMPTY
                elif row % 2 == 0 or col % 2 == 0:
                    grid[row][col] = SYMBOL_STREET

        if route:
            for start, end in zip(route, route[1:]):
                self._draw_segment(grid, start, end, SYMBOL_ROUTE)
            if route:
                self._plot(grid, route[0], SYMBOL_ROUTE)

        for order in active_orders:
            if order.status == COMPLETED:
                continue
            self._plot(grid, order.restaurant, SYMBOL_RESTAURANT)
            self._plot(grid, order.customer, SYMBOL_CUSTOMER)

        for rider in riders:
            self._plot(grid, rider.position, SYMBOL_RIDER)

        lines = ["".join(row) for row in grid]
        lines.append(MAP_LEGEND)
        return "\n".join(lines)

    def render_status(self, funds, riders, pending_count):
        """Render the system status snapshot.

        Args:
            funds: Current fund balance.
            riders: List of Rider objects.
            pending_count: Number of orders waiting for a rider.

        Returns:
            Multi-line status string.
        """
        idle = sum(1 for rider in riders if rider.state == 0)
        lines = [
            f"System funds: {funds}",
            f"Riders: {len(riders)} (idle: {idle})   Pending orders: {pending_count}",
        ]
        if not riders:
            lines.append("  No riders on duty.")
        for rider in riders:
            active = [o for o in rider.orders if o.status != COMPLETED]
            order_text = ", ".join(f"#{o.id}({o.status})" for o in active) or "none"
            lines.append(
                f"  Rider #{rider.rider_id}: location=({rider.x}, {rider.y}) "
                f"status={STATE_NAMES[rider.state]} speed={rider.speed} "
                f"active orders={len(active)} [{order_text}]"
            )
        return "\n".join(lines)

    def render_orders(self, orders):
        """Render the order table with timing and distance details.

        Args:
            orders: Iterable of Order objects.

        Returns:
            Multi-line table string.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        lines = [
            f"{'ID':<5}{'Start':<7}{'Restaurant':<13}{'Customer':<13}"
            f"{'Dist':<7}{'Taken':<7}{'Eff':<7}{'Rider':<7}Status"
        ]
        for order in orders:
            taken = "-" if order.time_taken is None else str(order.time_taken)
            eff = "-" if order.efficiency is None else str(order.efficiency)
            rider = "-" if order.rider_id is None else f"#{order.rider_id}"
            lines.append(
                f"{order.id:<5}{order.start_time:<7}"
                f"{str(order.restaurant):<13}{str(order.customer):<13}"
                f"{order.distance:<7}{taken:<7}{eff:<7}{rider:<7}{order.status}"
            )
        return "\n".join(lines)

    def render_rider_logs(self, riders):
        """Render every recorded status change for all riders.

        Args:
            riders: List of Rider objects.

        Returns:
            Multi-line log string.
        """
        if not riders:
            return "No riders on duty."
        lines = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id}:")
            for entry in rider.log:
                order_text = "-" if entry["order_id"] is None else f"#{entry['order_id']}"
                lines.append(
                    f"  [t={entry['time']:>4}] {entry['event']:<22} "
                    f"order={order_text:<5} at ({entry['x']}, {entry['y']}) "
                    f"state={STATE_NAMES.get(entry['state'], entry['state'])}"
                )
        return "\n".join(lines)

    def _plot(self, grid, point, symbol):
        """Place ``symbol`` at the character cell corresponding to ``point``."""
        row = _to_row(point[1])
        col = _to_col(point[0])
        if 0 <= row < MAP_CHARS and 0 <= col < MAP_CHARS:
            grid[row][col] = symbol

    def _draw_segment(self, grid, start, end, symbol):
        """Draw an axis-aligned route segment with ``symbol``."""
        dist = max(abs(end[0] - start[0]), abs(end[1] - start[1]))
        steps = max(1, dist // 10)
        for i in range(steps + 1):
            t = i / steps
            x = start[0] + (end[0] - start[0]) * t
            y = start[1] + (end[1] - start[1]) * t
            self._plot(grid, (x, y), symbol)

"""Command parsing, coordinate validation, and input history."""

from path_planner import MAP_SIZE

COMMAND_ALIASES = {
    "start": "start", "s": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "save": "save", "w": "save",
    "history": "history", "hist": "history",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}

try:  # Enables arrow-key history on platforms that provide readline.
    import readline  # noqa: F401
except ImportError:
    readline = None


class InputManager:
    """Reads, parses and validates user commands."""

    def __init__(self):
        """Create an input manager with an empty command history."""
        self.history = []

    def read_line(self, prompt):
        """Read one line from the user and record it in the history.

        Args:
            prompt: Prompt text to display.

        Returns:
            The raw input line.

        Raises:
            EOFError: When input is closed.
        """
        line = input(prompt)
        if line.strip():
            self.history.append(line.strip())
        return line

    def parse(self, line):
        """Split a command line into a canonical command and arguments.

        Args:
            line: Raw input line.

        Returns:
            Tuple ``(command, args)``. ``command`` is None for blank input.

        Raises:
            ValueError: If the command is unknown.
        """
        tokens = line.strip().split()
        if not tokens:
            return None, []
        name = tokens[0].lower()
        command = COMMAND_ALIASES.get(name)
        if command is None:
            raise ValueError(f"Unknown command '{name}'. Type 'help' for the command list.")
        return command, tokens[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Validate four integer coordinates in the range 0-800.

        Args:
            tokens: List of four string tokens.

        Returns:
            List of four integers ``[rx, ry, cx, cy]``.

        Raises:
            ValueError: If the count, type or range is wrong.
        """
        if len(tokens) != 4:
            raise ValueError("expected 4 integers: restaurant_x restaurant_y customer_x customer_y")
        values = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise ValueError(f"'{token}' is not an integer")
            if not 0 <= value <= MAP_SIZE:
                raise ValueError(f"coordinate {value} is out of range 0-{MAP_SIZE}")
            values.append(value)
        return values

    def prompt_order(self):
        """Interactively ask for order coordinates until valid or cancelled.

        Returns:
            List of four integers, or None if the user cancelled with a blank line.
        """
        print("Enter restaurant_x restaurant_y customer_x customer_y "
              f"(0-{MAP_SIZE}). Leave blank to cancel.")
        while True:
            line = self.read_line("order> ")
            if not line.strip():
                return None
            try:
                return self.parse_coordinates(line.split())
            except ValueError as exc:
                print(f"Invalid input: {exc}. Please try again.")

"""Configuration loading and JSON persistence of system state."""

import json
import logging
import os

import globalvar

CONFIG_FILE = "config.json"

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "rider_speed": 20,
    "recruit_min_funds": 300,
    "recruit_cost": 500,
    "state_file": "dispatch_state.json",
}

logger = logging.getLogger("dispatch.config")


def load_config(path=CONFIG_FILE):
    """Load system parameters from a JSON file, falling back to defaults.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        Dict of configuration values.
    """
    settings = dict(DEFAULT_CONFIG)
    if not os.path.exists(path):
        return settings
    try:
        with open(path, encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read config %s: %s; using defaults", path, exc)
        return settings
    if not isinstance(data, dict):
        logger.warning("Config %s is not a JSON object; using defaults", path)
        return settings
    for key in DEFAULT_CONFIG:
        if key in data:
            settings[key] = data[key]
    return settings


def snapshot_state():
    """Build a JSON-serialisable snapshot of the current system state.

    Returns:
        Dict containing funds, time, riders (with logs) and order history.
    """
    riders = globalvar.get_value("riders") or []
    orders = globalvar.get_value("orders") or []
    pending = globalvar.get_value("pending_orders") or []
    return {
        "funds": globalvar.get_value("funds"),
        "time": globalvar.get_value("time"),
        "rider_id_counter": globalvar.get_value("rider_id_counter"),
        "riders": [
            {
                "rider_id": rider.rider_id,
                "x": rider.x,
                "y": rider.y,
                "speed": rider.speed,
                "state": rider.state,
                "special_case": rider.special_case,
                "order_ids": [order.id for order in rider.orders],
                "log": rider.log,
            }
            for rider in riders
        ],
        "orders": [order.to_dict() for order in orders],
        "pending_order_ids": [order.id for order in pending],
    }


def save_state(path, data=None):
    """Write the system state to a JSON file atomically.

    Args:
        path: Destination file path.
        data: Optional pre-built snapshot; built from globals when omitted.

    Raises:
        OSError: If the file cannot be written.
    """
    if data is None:
        data = snapshot_state()
    temp_path = path + ".tmp"
    with open(temp_path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2)
    os.replace(temp_path, path)
    logger.info("State saved to %s", path)

"""Command-line entry point for the food delivery dispatch simulator.

Run with:  python src/main.py
"""

import logging
import math
import sys

import globalvar
from config import load_config, save_state, snapshot_state
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from order_model import COMPLETED
from path_planner import MAP_SIZE, route_length

from system import SystemManager, init_system

LOG_FILE = "dispatch.log"
MAX_TICKS_PER_COMMAND = 10000

logger = logging.getLogger("dispatch")

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy]  [s]  Create an order (restaurant then customer, 0-800).
                            Without arguments you will be prompted.
  status               [st] Show system funds, riders and their orders.
  orders               [o]  Show all orders with timing and distance.
  riders               [r]  Show every rider status change log.
  map                  [m]  Show the ASCII street map.
  tick [n]             [t]  Advance the simulation by n ticks (default 1).
  save                 [w]  Save state to the JSON state file.
  history              Show previously entered commands.
  help                 [h]  Show this help.
  quit                 [q]  Exit (also: exit)."""


class DispatchApp:
    """Interactive application that wires managers together and runs the loop."""

    def __init__(self, config):
        """Initialise the system and all managers.

        Args:
            config: Configuration dict produced by ``load_config``.
        """
        init_system()
        globalvar.set_value("funds", int(config["initial_funds"]))
        globalvar.set_value("rider_speed", int(config["rider_speed"]))
        globalvar.set_value("recruit_min_funds", int(config["recruit_min_funds"]))
        globalvar.set_value("recruit_cost", int(config["recruit_cost"]))
        self.state_file = str(config["state_file"])
        self.system = SystemManager()
        self.orders = OrderManager(self.system)
        self.display = DisplayManager()
        self.inputs = InputManager()
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
        }

    def run(self):
        """Run the interactive main loop until quit or end of input."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands.")
        while self.system.is_running():
            try:
                line = self.inputs.read_line("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            try:
                self.execute(line)
            except EOFError:
                print()
                break
            except ValueError as exc:
                print(f"Error: {exc}")
            except Exception:
                logger.exception("Unexpected error while executing %r", line)
                print("Internal error: the command was aborted, state was kept. "
                      f"See {LOG_FILE} for details.")
                self._recover()
            self._report_inconsistencies()
        print("Goodbye.")

    def execute(self, line):
        """Parse and run one command line.

        Args:
            line: Raw user input.
        """
        command, args = self.inputs.parse(line)
        if command is None:
            return
        self._handlers[command](args)

    def cmd_start(self, args):
        """Create an order, dispatch it, and show the assigned rider's route."""
        if args:
            coords = InputManager.parse_coordinates(args)
        else:
            coords = self.inputs.prompt_order()
            if coords is None:
                print("Order entry cancelled.")
                return
        rx, ry, cx, cy = coords
        result = self.orders.place_order((rx, ry), (cx, cy))
        order = result["order"]
        rider = result["rider"]
        new_riders = result["new_riders"]
        now = globalvar.get_value("time")

        print(f"Order #{order.id} created at tick {now}.")
        print(f"  Restaurant grid location: ({order.restaurant[0]}, {order.restaurant[1]})")
        print(f"  Customer grid location:   ({order.customer[0]}, {order.customer[1]})")
        if new_riders:
            ids = ", ".join(f"#{rid}" for rid in new_riders)
            print(f"  New rider added: yes (Rider {ids})")
        else:
            print("  New rider added: no")

        if rider is None:
            print("  No rider available (insufficient funds). The order is pending "
                  "and will be retried every tick.")
            return

        print(f"  Assigned to Rider #{rider.rider_id} "
              f"(status: {STATE_NAMES[rider.state]}, "
              f"active orders: {rider.active_order_count()})")
        route = rider.route_to_order(order)
        distance = route_length(route)
        ticks = math.ceil(distance / rider.speed) if rider.speed > 0 else 0
        print(f"  Speed: {rider.speed} units/tick   Route length: {distance} units   "
              f"Estimated time: {ticks} tick(s)")
        print(self.display.render_map(riders=[rider], active_orders=[order], route=route))

    def cmd_status(self, args):
        """Show the system status snapshot."""
        riders = globalvar.get_value("riders")
        pending = globalvar.get_value("pending_orders")
        print(f"Time: tick {globalvar.get_value('time')}")
        print(self.display.render_status(self.system.get_funds(), riders, len(pending)))

    def cmd_orders(self, args):
        """Show the full order table."""
        print(self.display.render_orders(globalvar.get_value("orders")))

    def cmd_riders(self, args):
        """Show the rider status change logs."""
        print(self.display.render_rider_logs(globalvar.get_value("riders")))

    def cmd_map(self, args):
        """Show the ASCII map with riders and active orders."""
        orders = globalvar.get_value("orders")
        active = [o for o in orders if o.status != COMPLETED]
        print(self.display.render_map(riders=globalvar.get_value("riders"),
                                      active_orders=active))

    def cmd_tick(self, args):
        """Advance the simulation by one or more ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise ValueError(f"'{args[0]}' is not an integer tick count")
            if not 1 <= count <= MAX_TICKS_PER_COMMAND:
                raise ValueError(f"tick count must be between 1 and {MAX_TICKS_PER_COMMAND}")
        for _ in range(count):
            self._step()
        print(f"Time is now tick {globalvar.get_value('time')}. "
              f"Riders: {len(globalvar.get_value('riders'))}, "
              f"pending orders: {len(globalvar.get_value('pending_orders'))}.")

    def cmd_save(self, args):
        """Save the system state to the JSON state file."""
        try:
            save_state(self.state_file, snapshot_state())
        except OSError as exc:
            raise ValueError(f"could not save state: {exc}")
        print(f"State saved to {self.state_file}.")

    def cmd_history(self, args):
        """Print the command history of this session."""
        if not self.inputs.history:
            print("No commands entered yet.")
            return
        for index, entry in enumerate(self.inputs.history, start=1):
            print(f"  {index:>3}  {entry}")

    def cmd_help(self, args):
        """Print the command reference."""
        print(HELP_TEXT)

    def cmd_quit(self, args):
        """Stop the main loop."""
        self.system.stop()

    def _step(self):
        """Run one simulation tick: dispatch pending orders, then move riders."""
        assigned, new_riders = self.orders.dispatch_pending()
        for rider_id in new_riders:
            print(f"  [t={globalvar.get_value('time')}] Rider #{rider_id} recruited.")
        for order in assigned:
            print(f"  [t={globalvar.get_value('time')}] Order #{order.id} "
                  f"assigned to Rider #{order.rider_id}.")
        self.system.tick()

    def _recover(self):
        """Attempt to restore consistency after an unexpected error."""
        try:
            self.orders.dispatch_pending()
        except Exception:
            logger.exception("Recovery dispatch failed")

    def _report_inconsistencies(self):
        """Log and display any consistency problems detected in the state."""
        for issue in self.system.check_consistency():
            logger.warning("Consistency issue: %s", issue)
            print(f"Warning: {issue}")


def _configure_logging():
    """Configure file logging, falling back to stderr if the file is unusable."""
    log_format = "%(asctime)s %(levelname)s %(name)s: %(message)s"
    try:
        logging.basicConfig(filename=LOG_FILE, level=logging.INFO, format=log_format)
    except OSError:
        logging.basicConfig(stream=sys.stderr, level=logging.WARNING, format=log_format)


def main():
    """Program entry point.

    Returns:
        Process exit code.
    """
    _configure_logging()
    config = load_config()
    app = DispatchApp(config)
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

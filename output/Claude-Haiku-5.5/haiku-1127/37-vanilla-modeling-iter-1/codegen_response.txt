"""Domain models and shared constants for the lottery system."""
from __future__ import annotations

import re
from dataclasses import dataclass, field

# Department whose members are protected by the fallback rule.
CHENG_RU_GROUP = "Cheng Ru Group"

# Special lottery person: only previous winners are filtered.
SPECIAL_DRAWER_LI_BIN = "Li Bin"

# Fallback rule: if the last two draws by these persons contained no
# Cheng Ru Group member, the next draw by FALLBACK_TARGET_DRAWER is restricted.
FALLBACK_TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"

UNRANKED_LEVEL = 9999

_ORDINAL_VALUES = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}
_ORDINAL_RE = re.compile(
    r"\b(first|second|third|fourth|fifth|sixth|seventh|eighth|ninth|tenth)\b",
    re.IGNORECASE,
)


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None


def prize_level(name: str) -> int:
    """Return the rank of a prize from its name (First=1, Second=2, ...).

    Prizes without an ordinal word are ranked last.
    """
    match = _ORDINAL_RE.search(name or "")
    if match is None:
        return UNRANKED_LEVEL
    return _ORDINAL_VALUES[match.group(1).lower()]


def sort_prizes_by_level(prizes: list[Prize]) -> list[Prize]:
    """Return prizes ordered from highest level to lowest.

    The sort is stable, so prizes of equal level keep their input order.
    """
    return sorted(prizes, key=lambda prize: prize_level(prize.name))

class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (each line: Name,Department Group)."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it into users and error messages."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines are ignored."""
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw in enumerate(lines, start=1):
            content = raw.strip()
            if not content:
                continue
            parts = content.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_number}: '{content}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users (first occurrence of each name) and
        duplicates (later occurrences of an already seen name)."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of members per department group."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management and round ordering."""
from __future__ import annotations

from lottery_system.models import Prize

DEFAULT_PRIZES: list[tuple[str, int]] = [
    ("Third Prize - Movie Tickets", 10),
    ("Second Prize - Shopping Cards", 5),
    ("First Prize - 1 Day Annual Leave", 2),
]


class PrizeManager:
    """Stores prizes and the lottery round sequence."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = name.strip().casefold()
        for prize in self._prizes:
            if prize.name.casefold() == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize '{name}' already exists, skipped.")
                continue
            self._prizes.append(Prize(name, quantity))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Add a custom prize at the end of the round sequence.

        Raises ValueError for an empty name, a non-positive quantity or a
        duplicate name.
        """
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize '{clean_name}' already exists.")
        groups = [group.strip() for group in (avoid_groups or []) if group and group.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(Prize(clean_name, quantity, groups, drawer))

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current round order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the
        new order and must be a permutation of 1..n."""
        count = len(self._prizes)
        try:
            is_permutation = sorted(order) == list(range(1, count + 1))
        except TypeError:
            return False
        if not is_permutation:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering and fair random drawing."""
from __future__ import annotations

import random

from lottery_system.models import (
    CHENG_RU_GROUP,
    FALLBACK_TARGET_DRAWER,
    FALLBACK_TRIGGER_DRAWERS,
    SPECIAL_DRAWER_LI_BIN,
    Prize,
    User,
)


def _same(a: str, b: str) -> bool:
    return a.strip().casefold() == b.strip().casefold()


def _is_cheng_ru(user: User) -> bool:
    return _same(user.department, CHENG_RU_GROUP)


def _user_key(user: User) -> str:
    return user.name.strip().casefold()


class LotteryManager:
    """Applies lottery rules and performs the draws.

    Rules:
    - Previous winners are never drawn again (anti-duplication).
    - Members of the prize's avoidance groups are excluded.
    - Special person "Li Bin": only previous winners are filtered.
    - Fallback: if the last two draws by Chen Qixian / Jin Duo contained no
      Cheng Ru Group member, Ma Jiaqi's pool is restricted to Cheng Ru Group
      members (overriding avoidance groups).
    """

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._won_keys: set[str] = set()
        self._winners: list[User] = []
        self._rng = rng if rng is not None else random.Random()
        # Results (True = contained a Cheng Ru Group member) of the most recent
        # draws by the fallback trigger persons, at most the last two kept.
        self.last_two_draw_results: list[bool] = []

    @property
    def winners(self) -> list[User]:
        """Users who have won so far, in winning order."""
        return list(self._winners)

    def is_winner(self, user: User) -> bool:
        return _user_key(user) in self._won_keys

    def is_fallback_active(self) -> bool:
        """True when the last two trigger-person draws had no Cheng Ru member."""
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the participants that may be drawn for the given prize."""
        available = [user for user in self._users if not self.is_winner(user)]
        drawer = prize.associated_drawer

        if drawer and _same(drawer, SPECIAL_DRAWER_LI_BIN):
            return available

        if drawer and _same(drawer, FALLBACK_TARGET_DRAWER) and self.is_fallback_active():
            return [user for user in available if _is_cheng_ru(user)]

        avoided = {group.strip().casefold() for group in prize.avoid_groups}
        return [
            user
            for user in available
            if user.department.strip().casefold() not in avoided
        ]

    def eligible_count(self, prize: Prize) -> int:
        """Number of participants currently eligible for the prize."""
        return len(self._get_eligible_pool(prize))

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._won_keys.add(_user_key(user))
        self._winners.append(user)

        drawer = prize.associated_drawer
        if drawer and any(_same(drawer, trigger) for trigger in FALLBACK_TRIGGER_DRAWERS):
            self.last_two_draw_results.append(_is_cheng_ru(user))
            del self.last_two_draw_results[:-2]
        return user

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize."""
        rules = ["previous winners are excluded"]
        drawer = prize.associated_drawer
        if drawer and _same(drawer, SPECIAL_DRAWER_LI_BIN):
            rules.append("special rule for Li Bin: avoidance groups are NOT applied")
        elif drawer and _same(drawer, FALLBACK_TARGET_DRAWER) and self.is_fallback_active():
            rules.append(
                f"fallback active: only {CHENG_RU_GROUP} members are eligible "
                "(avoidance groups overridden)"
            )
        elif prize.avoid_groups:
            rules.append("avoided groups: " + ", ".join(prize.avoid_groups))
        return "; ".join(rules) + "."

"""Winner recording, summary grouping and CSV export."""
from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, sort_prizes_by_level


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    winning_order: int
    winning_time: str


class ResultManager:
    """Keeps the real-time winner list and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    @property
    def winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def add_winner(self, user: User, prize: Prize) -> None:
        """Record a winner for a prize."""
        record = Winner(
            user=user,
            prize_name=prize.name,
            winning_order=len(self._winners) + 1,
            winning_time=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        )
        self._winners.append(record)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, following the order of `prizes`.

        Each list is ordered by winning order within the prize.
        """
        grouped: dict[str, list[Winner]] = {}
        for prize in prizes:
            if prize.name in grouped:
                continue
            grouped[prize.name] = sorted(
                (winner for winner in self._winners if winner.prize_name == prize.name),
                key=lambda winner: winner.winning_order,
            )
        return grouped

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to ./awards_result_YYYYMMDD.txt in CSV format.

        Rows are ordered by prize level (high to low), then winning order.
        """
        ordered_prizes = sort_prizes_by_level(prizes)
        grouped = self.get_winners_by_prize(ordered_prizes)
        file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize in ordered_prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow(
                            [
                                winner.user.name,
                                winner.user.department,
                                winner.prize_name,
                                winner.winning_time,
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: cannot write '{path}': {exc}"
        return True, f"Winner list successfully exported to {path}"

def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


class InputReader:
    """Reads stdin in a background thread so commands can be polled
    while a lottery round is running."""

    def __init__(self, stream: TextIO | None = None) -> None:
        self._stream = stream if stream is not None else sys.stdin
        self._queue: queue.Queue[str] = queue.Queue()
        self._eof = False
        thread = threading.Thread(target=self._pump, daemon=True)
        thread.start()

    def _pump(self) -> None:
        while True:
            line = self._stream.readline()
            if line == "":
                self._queue.put(EOF_TOKEN)
                return
            self._queue.put(line.strip())

    def read(self, prompt: str = "") -> str:
        """Blocking read of one input line."""
        if prompt:
            print(prompt, end="", flush=True)
        if self._eof:
            return EOF_TOKEN
        item = self._queue.get()
        if item == EOF_TOKEN:
            self._eof = True
        return item

    def poll(self) -> str | None:
        """Non-blocking read. Returns None when no input is pending."""
        if self._eof:
            return EOF_TOKEN
        try:
            item = self._queue.get_nowait()
        except queue.Empty:
            return None
        if item == EOF_TOKEN:
            self._eof = True
        return item


class LotterySystemCLI:
    """Interactive command-line controller for the whole system."""

    def __init__(self, input_reader: InputReader | None = None, draw_delay: float = 0.5) -> None:
        self._input = input_reader if input_reader is not None else InputReader()
        self._delay = draw_delay
        self._list_manager = ListManager()
        self._prize_manager = PrizeManager()
        self._users: list[User] = []
        self._results = ResultManager()

    def _ask(self, prompt: str) -> str:
        """Prompt for text. End of input is returned as an empty string."""
        value = self._input.read(prompt)
        if value == EOF_TOKEN:
            return ""
        return value.strip()

    # --------------------------------------------------------------- main loop
    def run(self) -> int:
        print(BANNER)
        while True:
            print(MAIN_MENU)
            choice = self._input.read("Select an option: ")
            if choice == EOF_TOKEN:
                print()
                break
            if choice == "1":
                self._load_participants()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self._prize_menu()
            elif choice == "4":
                self._start_lottery()
            elif choice == "5":
                self._show_winner_list()
            elif choice == "6":
                self._export()
            elif choice == "0":
                print("Goodbye!")
                break
            else:
                print("Invalid option, please try again.")
        return 0

    # ------------------------------------------------------- list management
    def _load_participants(self) -> None:
        print("1. Built-in team building list")
        print("2. Upload custom txt list (each line: Name,Department Group)")
        choice = self._ask("Choose list source (1/2): ")
        if choice == "1":
            users, errors = self._list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip('"')
            users, errors = self._list_manager.load_custom_list(path)
        else:
            print("Invalid choice. Participant list unchanged.")
            return

        valid, duplicates = self._list_manager.validate_users(users)
        problems = list(errors)
        problems.extend(
            f"Duplicate Name: '{user.name}' ({user.department}) repeats an earlier entry"
            for user in duplicates
        )
        if problems:
            print(f"Found {len(problems)} problem(s) in the list:")
            for problem in problems:
                print(f"  - {problem}")
            print("Enter 'r' to remove invalid data and keep the valid entries,")
            print("or 't' to terminate the upload.")
            decision = self._ask("Your choice (r/t): ").lower()
            if decision != "r":
                print("Upload terminated. Participant list unchanged.")
                return

        if not users:
            print("No valid participants found. Participant list unchanged.")

        self._lottery = None

    def _start_lottery(self) -> None:

            print("Please load a participant list first.")
            return
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Please set up prizes first.")
            return

        self._results = ResultManager()
        self._lottery = LotteryManager(self._users)
        print("\n##### Lottery started #####")
        print("Control commands: 'pause', 'continue', 'terminate'")
        for index, prize in enumerate(prizes, start=1):
            if not self._run_round(prize, index, len(prizes)):
                print("Lottery process exited.")
                break
        else:
            print("\n##### All lottery rounds completed #####")
        self._show_winner_list()

    def _run_round(self, prize: Prize, index: int, total: int) -> bool:
        """Run one round. Returns False if the whole lottery must stop."""
        assert self._lottery is not None
        self._paused = False
        print(f"\n--- Round {index}/{total}: {prize.name} ---")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {self._lottery.describe_rules(prize)}")

        round_winners: list[User] = []
        for position in range(1, prize.quantity + 1):
            status = self._apply_control()
            if status == "exit":
                self._print_round_summary(index, round_winners)
                return False
            if status == "terminate":
                print("Current round terminated by operator.")
                break
            if self._lottery.eligible_count(prize) == 0:
                print("No eligible participants remain for this prize. Round ends.")
                break

            print(f"Currently drawing the {_ordinal(position)} winner...", flush=True)
            time.sleep(self._delay)
            user = self._lottery.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            self._results.add_winner(user, prize)
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        self._print_round_summary(index, round_winners)
        return True

    def _print_round_summary(self, index: int, winners: list[User]) -> None:
        print(f"Round {index} results ({len(winners)} winner(s)):")
        for position, user in enumerate(winners, start=1):
            print(f"  {position}. {user.name} ({user.department})")

    def _apply_control(self) -> str:
        """Process pending control commands.

        Returns "run" to keep drawing, "terminate" to skip the current round,
        or "exit" when the input stream has ended.
        """
        while True:
            if self._paused:
                command = self._input.read("[PAUSED] Enter 'continue' or 'terminate': ")
            else:
                pending = self._input.poll()
                if pending is None:
                    return "run"
                command = pending
            command = command.strip().lower()
            if command == EOF_TOKEN:
                return "exit"
            if command == "pause":
                if not self._paused:
                    self._paused = True
                    print("Lottery paused. Enter 'continue' to resume or 'terminate' to skip this round.")
            elif command == "continue":
                if self._paused:
                    self._paused = False
                    print("Lottery resumed.")
            elif command == "terminate":
                self._paused = False
                return "terminate"
            elif command:
                print(f"Unknown command '{command}'. Use 'pause', 'continue' or 'terminate'.")

    # ---------------------------------------------------------------- results
    def _show_winner_list(self) -> None:
        prizes = sort_prizes_by_level(self._prize_manager.get_prizes())
        grouped = self._results.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            print("No winners yet.")
            return
        print("\n========== Winner List ==========")
        for prize in prizes:
            winners = grouped.get(prize.name, [])
            if not winners:
                continue
            print(f"\n{prize.name}:")
            for position, winner in enumerate(winners, start=1):
                print(
                    f"  {position}. {winner.user.name} | {winner.user.department} | "
                    f"{prize.name} | {winner.winning_time}"
                )

    def _export(self) -> None:
        if not self._results.winners:
            print("No winners to export yet. Run the lottery first.")
            return
        success, message = self._results.export_to_csv(self._prize_manager.get_prizes())
        print(message)
        if not success:
            print("Please check the file permissions and try again.")


def main() -> int:
    """Application entry point."""
    cli = LotterySystemCLI()
    try:
        return cli.run()
    except KeyboardInterrupt:
        print("\nInterrupted. Exiting.")
        return 130


if __name__ == "__main__":
    sys.exit(main())

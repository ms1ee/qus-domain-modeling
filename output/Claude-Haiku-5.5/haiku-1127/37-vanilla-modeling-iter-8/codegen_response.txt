from __future__ import annotations

from dataclasses import dataclass, field


# Special lottery persons and group names referenced by the rule engine.
LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
CHENG_RU_GROUP = "Cheng Ru group"


def normalize(text: str) -> str:
    """Normalize a name or group label for case-insensitive comparison."""
    return " ".join(str(text).split()).casefold()


def same_name(first: str, second: str) -> bool:
    """Return True if two names/labels are equal after normalization."""
    return normalize(first) == normalize(second)


def prize_level(name: str) -> int:
    """Derive the prize level (1 = highest) from the prize name.

    Prizes named with "First", "Second" or "Third" are ranked 1, 2 and 3.
    Custom prizes without such keywords are ranked after all preset prizes,
    preserving their addition order.
    """
    lowered = name.casefold()
    for keyword, level in (("first", 1), ("second", 2), ("third", 3)):
        if keyword in lowered:
            return level
    return 99


@dataclass(frozen=True)
class User:
    """A participant with a name and a department group."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that can be drawn in a lottery round."""

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None


def describe_prize_rules(prize: Prize) -> str:
    """Build a human-readable description of the rules applied to a prize."""
    parts: list[str] = []
    drawer = prize.associated_drawer

    if drawer:
        parts.append(f"Drawn by {drawer}")
    else:
        parts.append("No associated lottery person")

    if drawer and same_name(drawer, LI_BIN):
        parts.append(
            "Special rule: only previous winners are excluded; "
            "avoidance groups do not apply"
        )
    else:
        parts.append("Excludes previous winners")
        if prize.avoid_groups:
            parts.append("Avoids group(s): " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")

    if drawer and same_name(drawer, MA_JIAQI):
        parts.append(
            "Fallback: if the last two results of Chen Qixian and Jin Duo "
            "contain no Cheng Ru group winners, the pool is limited to "
            "Cheng Ru group members"
        )

    return "; ".join(parts)

def _read_text(path: str) -> str:
    """Read a text file trying UTF-8 (with optional BOM) first, then GBK."""
    with open(path, "rb") as handle:
        raw = handle.read()
    for encoding in ("utf-8-sig", "gbk"):
        try:
            return raw.decode(encoding)
        except UnicodeDecodeError:
            continue
    raise OSError("unsupported text encoding (expected UTF-8 or GBK)")


def _parse_lines(lines: Iterable[str]) -> tuple[list[User], list[str]]:
    """Parse "Name,Department" lines into users and format error messages."""
    users: list[User] = []
    errors: list[str] = []
    for line_number, raw in enumerate(lines, start=1):
        stripped = raw.strip()
        if not stripped:
            continue
        parts = stripped.split(",")
        if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
            errors.append(
                f"Format Error at Line {line_number}: '{stripped}' "
                "(expected 'Name,Department Group')"
            )
            continue
        users.append(User(name=parts[0].strip(), department=parts[1].strip()))
    return users, errors


class ListManager:
    """Loads, validates and summarizes participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building participant list."""
        return _parse_lines(DEFAULT_PARTICIPANT_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt participant list from the given path."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: '{path}'"]
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a participant file, reporting read errors."""
        try:
            content = _read_text(path)
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return _parse_lines(content.splitlines())

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique entries and duplicate-name entries.

        The first occurrence of a name is kept as valid; later occurrences
        are returned as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of members per department group, sorted by name."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items(), key=lambda item: item[0].casefold()))

from __future__ import annotations

from lottery_system.models import Prize, normalize


DEFAULT_PRIZES: list[tuple[str, int, str]] = [
    ("Third Prize - Movie Tickets", 3, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
]


class PrizeManager:
    """Maintains the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        key = normalize(name)
        for prize in self._prizes:
            if normalize(prize.name) == key:
                return prize
        return None

    def load_default_prizes(self) -> list[str]:
        """Add the preset prize library. Returns error messages, if any."""
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize already exists and was skipped: {name}")
                continue
            self._prizes.append(
                Prize(name=name, quantity=quantity, avoid_groups=[], associated_drawer=drawer)
            )
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a custom prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize already exists: {clean_name}")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")

        groups: list[str] = []
        seen: set[str] = set()
        for group in avoid_groups or []:
            cleaned = str(group).strip()
            if cleaned and normalize(cleaned) not in seen:
                seen.add(normalize(cleaned))
                groups.append(cleaned)

        drawer = (associated_drawer or "").strip() or None
        prize = Prize(
            name=clean_name,
            quantity=quantity,
            avoid_groups=groups,
            associated_drawer=drawer,
        )
        self._prizes.append(prize)
        return prize

    def set_avoid_groups(self, name: str, avoid_groups: list[str]) -> bool:
        """Replace the avoidance groups of an existing prize."""
        prize = self._find(name)
        if prize is None:
            return False
        groups: list[str] = []
        seen: set[str] = set()
        for group in avoid_groups:
            cleaned = str(group).strip()
            if cleaned and normalize(cleaned) not in seen:
                seen.add(normalize(cleaned))
                groups.append(cleaned)
        prize.avoid_groups = groups
        return True

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        `order` contains 1-based current positions in the desired new order.
        Returns False if it is not a permutation of all current positions.
        """
        count = len(self._prizes)
        if not isinstance(order, list) or len(order) != count:
            return False
        if not all(isinstance(item, int) and not isinstance(item, bool) for item in order):
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
    describe_prize_rules,
    normalize,
    same_name,
)


def is_cheng_ru(user: User) -> bool:
    """Return True if the user belongs to the Cheng Ru group."""
    return normalize(user.department) == normalize(CHENG_RU_GROUP)


class LotteryManager:
    """Applies lottery rules and draws winners one at a time.

    Usage per round: begin_round(prize), draw_next(prize) repeatedly,
    then end_round(prize).
    """

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        # Results of the last two Chen Qixian / Jin Duo rounds:
        # True if the round produced at least one Cheng Ru group winner.
        self.last_two_draw_results: list[bool] = []
        self._winner_names: set[str] = set()
        self._winners: list[User] = []
        self._round_has_cheng_ru: bool = False
        self._fallback_active: bool = False

    @property
    def winners(self) -> list[User]:
        """All winners drawn so far, in winning order."""
        return list(self._winners)

    def is_fallback_triggered(self, prize: Prize) -> bool:
        """Return True if the prize is drawn by Ma Jiaqi and the Cheng Ru
        protection fallback is in effect (last two results both without
        Cheng Ru group members)."""
        if not prize.associated_drawer or not same_name(prize.associated_drawer, MA_JIAQI):
            return False
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute the users who may be drawn for the given prize."""
        available = [
            user for user in self.users if normalize(user.name) not in self._winner_names
        ]

        if self.is_fallback_triggered(prize):
            forced = [user for user in available if is_cheng_ru(user)]
            if forced:
                return forced

        drawer = prize.associated_drawer
        if drawer and same_name(drawer, LI_BIN):
            return available

        avoided = {normalize(group) for group in prize.avoid_groups}
        return [user for user in available if normalize(user.department) not in avoided]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the current eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def describe_rule(self, prize: Prize) -> str:
        """Return the rule description shown to the operator for a prize."""
        text = describe_prize_rules(prize)
        if self.is_fallback_triggered(prize):
            text += " [Fallback ACTIVE for this round]"
        return text

    def begin_round(self, prize: Prize) -> None:
        """Prepare internal state before drawing for a prize."""
        self._round_has_cheng_ru = False
        self._fallback_active = self.is_fallback_triggered(prize)

    def draw_next(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self._winners.append(user)
        self._winner_names.add(normalize(user.name))
        if is_cheng_ru(user):
            self._round_has_cheng_ru = True
        return user

    def end_round(self, prize: Prize) -> None:
        """Update fallback tracking after a round finishes or is terminated."""
        drawer = prize.associated_drawer
        if drawer and (same_name(drawer, CHEN_QIXIAN) or same_name(drawer, JIN_DUO)):
            self.last_two_draw_results.append(self._round_has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer and same_name(drawer, MA_JIAQI) and self._fallback_active:
            # The protection has been applied; start counting afresh.
            self.last_two_draw_results = []
        self._round_has_cheng_ru = False
        self._fallback_active = False

"""Winner recording, grouped summaries and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, prize_level


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    winning_order: int
    winning_time: datetime


class ResultManager:
    """Stores winners and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._counter: int = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for a prize and return the created record."""
        self._counter += 1
        winner = Winner(
            user=user,
            prize_name=prize.name,
            winning_order=self._counter,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def all_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name.

        Keys are ordered by prize level (high to low); prizes with the same
        level keep their configured sequence. Winners within each prize are
        ordered by winning time.
        """
        ordered = sorted(
            enumerate(prizes),
            key=lambda item: (prize_level(item[1].name), item[0]),
        )
        result: dict[str, list[Winner]] = {}
        for _, prize in ordered:
            if prize.name in result:
                continue
            matched = [w for w in self._winners if w.prize_name == prize.name]
            result[prize.name] = sorted(matched, key=lambda w: w.winning_order)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to awards_result_YYYYMMDD.txt in
        the current working directory, formatted as CSV."""
        grouped = self.get_winners_by_prize(prizes)
        rows = [winner for winners in grouped.values() for winner in winners]
        if not rows:
            return False, "No winners to export."

        file_name = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in rows:
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"

        return True, f"Winner list successfully exported to {path}"

def _ask(prompt: str) -> str:
    """Read a stripped line from stdin; exit cleanly on end of input."""
    try:
        return input(prompt).strip()
    except EOFError:
        print()
        raise SystemExit(0)


def _ask_int(prompt: str, minimum: int = 1) -> int:
    """Prompt until the user enters an integer >= minimum."""
    while True:
        text = _ask(prompt)
        try:
            value = int(text)
        except ValueError:
            print("Please enter a valid integer.")
            continue
        if value < minimum:
            print(f"Please enter a number >= {minimum}.")
            continue
        return value


def _ordinal(number: int) -> str:
    """Format an integer as an English ordinal (1st, 2nd, 3rd, 4th...)."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _split_items(text: str) -> list[str]:
    """Split a comma-separated input into non-empty stripped items."""
    return [item.strip() for item in text.split(",") if item.strip()]


class LotterySystemCLI:
    """Drives the interactive lottery workflow."""

        self.prize_manager = PrizeManager()
        self.results = ResultManager()

    def run(self) -> None:

        print("Welcome to the Company Team Building Smart Lottery System.")
        while True:
            print(MENU_TEXT)
            choice = _ask("Select an option: ")
            if choice == "1":
                self._load_default_list()
            elif choice == "2":
                self._upload_custom_list()
            elif choice == "3":
                self._view_groups()
            elif choice == "4":
                self._load_preset_prizes()
            elif choice == "5":
                self._add_custom_prize()
            elif choice == "6":
                self._set_avoid_groups()
            elif choice == "7":
                self._reorder_prizes()
            elif choice == "8":
                self._view_prizes()
            elif choice == "9":
                self._start_lottery()
            elif choice == "10":
                self._show_winners()
            elif choice == "11":
                self._export_results()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    # ----------------------------------------------------- list management
    def _load_default_list(self) -> None:

        path = _ask("Enter the path of the txt file (one 'Name,Department' per line): ")
        path = path.strip('"').strip("'")

        self._apply_loaded(users, errors, f"file '{path}'")

    def _apply_loaded(self, users: list[User], errors: list[str], source: str) -> None:
        """Validate loaded users and let the operator decide on issues."""
        valid, duplicates = self.list_manager.validate_users(users)

        if not errors and not duplicates:
            if not valid:
                print(f"No valid participants found in {source}.")
                return
            self.users = valid
            print(f"Loaded {len(valid)} participants from {source}.")
            return

        print(
            f"Validation of {source} found {len(errors)} format error(s) "
            f"and {len(duplicates)} duplicate name(s):"
        )
        for error in errors:
            print("  " + error)
        for duplicate in duplicates:
            print(
                f"  Duplicate Name: {duplicate.name} ({duplicate.department}) "
                "- later entry ignored"
            )

        if not valid:
            print("No valid participants remain. Upload terminated.")
            return

        choice = _ask(
            "Remove invalid data and continue? [R]emove and continue / [T]erminate upload: "
        ).lower()
        if choice in ("r", "remove"):
            self.users = valid
            print(f"Loaded {len(valid)} valid participants from {source}.")
        else:
            print("Upload terminated. Participant list unchanged.")

    def _set_avoid_groups(self) -> None:

        if not prizes:
            print("No prizes configured.")
            return
        self._print_prize_list(prizes)
        index = _ask_int("Select prize number: ", minimum=1)

            print("No prizes configured.")
            return
        self._print_prize_list(prizes)
        text = _ask(
            "Enter new order as serial numbers (e.g. '2 1 3', blank to cancel): "

        if not text:
            print("Reorder cancelled.")
            return
        try:
            order = [int(item) for item in text.replace(",", " ").split()]
        except ValueError:
            print("Invalid input: serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize sequence updated.")
            self._print_prize_list(self.prize_manager.get_prizes())
        else:
            print(
                f"Invalid order: provide each serial number from 1 to {len(prizes)} exactly once."
            )

    def _print_prize_list(self, prizes: list[Prize]) -> None:
        for position, prize in enumerate(prizes, start=1):
            print(f"  {position}. {prize.name} (quantity {prize.quantity})")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nPrize configuration (lottery sequence):")
        for position, prize in enumerate(prizes, start=1):
            print(f"  {position}. {prize.name} - quantity {prize.quantity}")
            print(f"     Rule: {describe_prize_rules(prize)}")

    # ------------------------------------------------------------- lottery
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure at least one prize first.")
            return
        if self.results.all_winners():
            confirm = _ask("Previous results will be discarded. Start a new lottery? (y/n): ")
            if confirm.lower() not in ("y", "yes"):
                print("Lottery cancelled.")
                return

        self.results = ResultManager()
        self.lottery = LotteryManager(self.users)
        print("\nLottery started. Commands: [Enter] = draw next, 'pause', 'terminate'.")

        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize)

        print("\nAll lottery rounds completed.")
        self._print_summary(prizes)

    def _run_round(self, index: int, total: int, prize: Prize) -> None:
        assert self.lottery is not None
        print(f"\n=== Round {index}/{total}: {prize.name} ===")
        print(f"Quantity: {prize.quantity}")

        self.lottery.begin_round(prize)
        print(f"Rule: {self.lottery.describe_rule(prize)}")

        round_winners: list[Winner] = []
        terminated = False

        while len(round_winners) < prize.quantity and not terminated:
            command = _ask("> ").lower()

            if command in ("terminate", "t"):
                terminated = True
                print("Round terminated by operator.")
                break

            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    paused_command = _ask("(paused) > ").lower()
                    if paused_command == "continue":
                        print("Lottery resumed.")
                        break
                    if paused_command in ("terminate", "t"):
                        terminated = True
                        print("Round terminated by operator.")
                        break
                    print("While paused, only 'continue' or 'terminate' is accepted.")
                continue

            if command == "continue":
                print("Lottery is not paused.")
                continue

            if command != "":
                print("Unknown command. Use [Enter], 'pause' or 'terminate'.")
                continue

            position = len(round_winners) + 1
            print(f"Currently drawing the {_ordinal(position)} winner...")
            user = self.lottery.draw_next(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends early.")
                break

            winner = self.results.add_winner(user, prize)
            round_winners.append(winner)
            print(
                f"Congratulations {user.name} ({user.department}) "
                f"for winning {prize.name}"
            )

        self.lottery.end_round(prize)

        print(f"\nRound {index} results ({len(round_winners)}/{prize.quantity} drawn):")
        if not round_winners:
            print("  No winners in this round.")
        for winner in round_winners:
            print(
                f"  {winner.winning_order}. {winner.user.name} "
                f"({winner.user.department}) - {winner.prize_name}"
            )

    def _print_summary(self, prizes: list[Prize]) -> None:
        grouped = self.results.get_winners_by_prize(prizes)
        print("\n==================== Complete Winner List ====================")
        if not any(grouped.values()):
            print("No winners were drawn.")
            return
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}:")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(
                    f"  - {winner.user.name} ({winner.user.department}) "
                    f"at {winner.winning_time:%Y-%m-%d %H:%M:%S}"
                )

    def _show_winners(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not self.results.all_winners():
            print("No winners recorded yet.")
            return
        self._print_summary(prizes)

    def _export_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        success, message = self.results.export_to_csv(prizes)
        print(message)


def main() -> None:
    """Application entry point."""
    try:
        LotterySystemCLI().run()
    except KeyboardInterrupt:
        print("\nInterrupted. Exiting.")

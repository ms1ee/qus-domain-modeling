"""
Data loading utilities for the Beijing Subway fare calculation system.

The network data is read from ``subway_data.json`` which lives next to this
module, so the path is valid regardless of the current working directory.
"""

import json
import os

# Absolute path of the subway dataset (src/subway_data.json).
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")

# Top-level keys that must be present in the dataset.
REQUIRED_KEYS = ("stations", "connections")


def load_subway_data(path=DATA_FILE_PATH):
    """
    Load the subway dataset from a JSON file.

    Args:
        path: Path to the JSON file. Defaults to ``DATA_FILE_PATH``.

    Returns:
        dict: The JSON content of the file, with keys ``stations`` and
        ``connections``.

    Raises:
        OSError: If the file cannot be opened.
        ValueError: If the JSON is malformed or required keys are missing.
    """
    with open(path, "r", encoding="utf-8") as handle:
        try:
            data = json.load(handle)
        except json.JSONDecodeError as exc:
            raise ValueError(f"malformed JSON in '{path}': {exc}") from exc

    if not isinstance(data, dict):
        raise ValueError(f"subway data in '{path}' must be a JSON object")

    for key in REQUIRED_KEYS:
        if key not in data:
            raise ValueError(f"subway data in '{path}' is missing the '{key}' key")

    return data

"""
Core graph data model for the subway network.

Stations are identified by integer numbers assigned by the caller
(see ``network_builder.build_graph``); the ``Graph`` keeps the mapping
between station names and numbers, the lines serving each station, and
an undirected adjacency list of ``Edge`` objects.
"""


class Edge:
    """
    An undirected, weighted connection between two adjacent stations.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance between the stations in meters.
        line: Name of the subway line operating this connection.
    """

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by path algorithms (equal to the distance in meters)."""
        return self.distance

    def other_end(self, node):
        """
        Return the endpoint opposite to ``node``.

        Raises:
            ValueError: If ``node`` is not an endpoint of this edge.
        """
        if node == self.source:
            return self.dest
        if node == self.dest:
            return self.source
        raise ValueError(f"node {node} is not an endpoint of edge {self!r}")

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"


class Graph:
    """
    Weighted undirected graph of subway stations.

    Attributes:
        station_to_number: dict mapping station name -> station number.
        number_to_station: dict mapping station number -> station name.
        station_lines: dict mapping station name -> list of line names.
        adjacency: dict mapping station number -> list of incident ``Edge``.
    """

    def __init__(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}
        self.adjacency = {}
        self._edges = []

    # ----------------------------------------------------------- stations
    def add_station(self, station_name, number):
        """
        Register a station under a caller-assigned number.

        Re-registering the same name with the same number is a no-op.

        Raises:
            TypeError: If ``number`` is not an integer.
            ValueError: If the name or number is already bound to a different station.
        """
        if not isinstance(number, int) or isinstance(number, bool):
            raise TypeError("station number must be an integer")

        if station_name in self.station_to_number:
            if self.station_to_number[station_name] != number:
                raise ValueError(f"station '{station_name}' is already registered with another number")
            return

        if number in self.number_to_station:
            raise ValueError(f"station number {number} is already used by '{self.number_to_station[number]}'")

        self.station_to_number[station_name] = number
        self.number_to_station[number] = station_name
        self.adjacency.setdefault(number, [])
        self.station_lines.setdefault(station_name, [])

    def set_station_lines(self, station_name, lines):
        """Set the (de-duplicated, order-preserving) list of lines serving a station."""
        self.station_lines[station_name] = list(dict.fromkeys(str(line) for line in lines))

    def has_station(self, station_name):
        """Return True if a station with this name is registered."""
        return station_name in self.station_to_number

    def get_number(self, station_name):
        """Return the number of a station by name."""
        return self.station_to_number[station_name]

    def get_name(self, number):
        """Return the name of a station by number."""
        return self.number_to_station[number]

    def lines_of(self, station_name):
        """Return a copy of the list of lines serving the station."""
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        """A station served by two or more lines is a transfer (interchange) station."""
        return len(self.lines_of(station_name)) >= 2

    def station_names(self):
        """Return all station names sorted alphabetically."""
        return sorted(self.station_to_number)

    def all_lines(self):
        """Return all distinct line names in the network, sorted."""
        lines = set()
        for line_list in self.station_lines.values():
            lines.update(line_list)
        return sorted(lines)

    def station_count(self):
        """Return the number of registered stations."""
        return len(self.station_to_number)

    # -------------------------------------------------------------- edges
    def add_edge(self, edge):
        """
        Add an undirected edge. Both endpoint numbers must be registered.

        Raises:
            KeyError: If an endpoint station number is unknown.
        """
        for number in (edge.source, edge.dest):
            if number not in self.adjacency:
                raise KeyError(f"station number {number} is not registered in the graph")

        self.adjacency[edge.source].append(edge)
        if edge.dest != edge.source:
            self.adjacency[edge.dest].append(edge)
        self._edges.append(edge)

    def neighbors(self, number):
        """Return the list of edges incident to the station with this number."""
        return self.adjacency.get(number, [])

    def edges(self):
        """Return a copy of the list of all unique edges."""
        return list(self._edges)

    def edge_count(self):
        """Return the number of unique edges."""
        return len(self._edges)

"""
Builds a ``Graph`` from the raw subway dataset.

Station numbers are assigned here (the caller), as required by the
``Graph`` model. Stations referenced only by connections are registered
automatically, and the lines of every connection are added to the lines
served by both of its endpoints.
"""

from models import Edge, Graph


def build_graph(data):
    """
    Construct the weighted undirected subway graph from the dataset.

    Args:
        data: dict with ``stations`` (list of {name, lines}) and
              ``connections`` (list of {source, dest, distance, line}).

    Returns:
        Graph: The populated graph.

    Raises:
        KeyError / TypeError / ValueError: If the dataset is malformed.
    """
    station_lines = {}
    order = []

    # 1. Collect stations and their lines from the station list.
    for entry in data.get("stations", []):
        name = str(entry["name"]).strip()
        if not name:
            raise ValueError("station name cannot be empty")
        if name not in station_lines:
            station_lines[name] = []
            order.append(name)
        for line in entry.get("lines", []):
            if line not in station_lines[name]:
                station_lines[name].append(line)

    connections = data.get("connections", [])

    # 2. Make sure every connection endpoint is known and serves the line.
    for conn in connections:
        for endpoint in (str(conn["source"]).strip(), str(conn["dest"]).strip()):
            if endpoint not in station_lines:
                station_lines[endpoint] = []
                order.append(endpoint)
            if conn["line"] not in station_lines[endpoint]:
                station_lines[endpoint].append(conn["line"])

    # 3. Build the graph with caller-assigned numbers (0..n-1).
    graph = Graph()
    numbers = {}
    for index, name in enumerate(order):
        graph.add_station(name, index)
        numbers[name] = index
        graph.set_station_lines(name, station_lines[name])

    # 4. Add the connections as undirected weighted edges.
    for conn in connections:
        source = numbers[str(conn["source"]).strip()]
        dest = numbers[str(conn["dest"]).strip()]
        distance = conn["distance"]
        if not isinstance(distance, (int, float)) or isinstance(distance, bool):
            raise TypeError(f"distance of connection {conn!r} must be numeric")
        graph.add_edge(Edge(source, dest, distance, conn["line"]))

    return graph

"""
Shortest-path computation for the subway network.

Implements:
  * ``MinPQ``        - binary min-heap priority queue (O(log n) insert/delete-min).
  * ``PathfindingConfig`` - query configuration (mode, transfer limit, line preference).
  * ``ShortestPath`` - Dijkstra's algorithm driven by ``MinPQ``.

Search states are triples ``(station_number, arrival_line, transfers_so_far)``
so that transfer limits and least-transfer optimisation are exact. The
transfer count increases whenever consecutive edges belong to different lines.

Complexity: O((V+E) log V) for each fixed transfer bound, where the number
of states is bounded by V * L * (max_transfers + 1) with L the number of lines.
"""

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Transfer limit used when the caller does not specify one.
DEFAULT_MAX_TRANSFERS = 10

# Equivalent distance penalty (meters) charged per transfer in comprehensive mode.
TRANSFER_PENALTY_METERS = 500.0

# Extra cost (meters) charged for travelling on a line that is not preferred.
LINE_PREFERENCE_PENALTY_METERS = 1000.0


class MinPQ:
    """
    Binary min-heap priority queue.

    Items are stored with their priority; ties are broken by insertion order
    so that comparison never falls back to comparing the items themselves.
    """

    def __init__(self):
        self._heap = []
        self._counter = 0

    def insert(self, priority, item):
        """Insert an item with the given (comparable) priority."""
        self._heap.append((priority, self._counter, item))
        self._counter += 1
        self._swim(len(self._heap) - 1)

    def del_min(self):
        """
        Remove and return the ``(priority, item)`` pair with the smallest priority.

        Raises:
            IndexError: If the queue is empty.
        """
        if self.is_empty():
            raise IndexError("priority queue underflow")
        top = self._heap[0]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._sink(0)
        return top[0], top[2]

    def min_priority(self):
        """Return the smallest priority without removing it."""
        if self.is_empty():
            raise IndexError("priority queue is empty")
        return self._heap[0][0]

    def is_empty(self):
        """Return True if the queue holds no items."""
        return not self._heap

    def __len__(self):
        return len(self._heap)

    def _less(self, i, j):
        a = self._heap[i]
        b = self._heap[j]
        return (a[0], a[1]) < (b[0], b[1])

    def _swap(self, i, j):
        self._heap[i], self._heap[j] = self._heap[j], self._heap[i]

    def _swim(self, k):
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._swap(k, parent)
                k = parent
            else:
                break

    def _sink(self, k):
        n = len(self._heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and self._less(right, left):
                child = right
            if self._less(child, k):
                self._swap(child, k)
                k = child
            else:
                break


class PathfindingConfig:
    """
    Configuration of a path query.

    Attributes:
        mode: One of ``shortest_distance``, ``least_transfer``, ``comprehensive``.
        max_transfers: Maximum number of transfers allowed (None = default limit).
        preferred_lines: List of line names that are preferred (others are penalised).
    """

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class ShortestPath:
    """
    Dijkstra-based path finder from a single source station.

    Args:
        graph: The ``Graph`` to search.
        source: Station number of the starting station.
        config: Optional ``PathfindingConfig``.

    Attributes:
        dist_to: dict mapping reachable station number -> travelled distance (meters)
                 of the selected path.
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()

        if self.config.mode not in VALID_MODES:
            raise ValueError(f"unknown path mode '{self.config.mode}', expected one of {VALID_MODES}")

        max_transfers = self.config.max_transfers
        if max_transfers is None:
            max_transfers = DEFAULT_MAX_TRANSFERS
        if not isinstance(max_transfers, int) or isinstance(max_transfers, bool) or max_transfers < 0:
            raise ValueError("max_transfers must be a non-negative integer")
        self._max_transfers = max_transfers

        self._preferred = set(self.config.preferred_lines)

        if source not in graph.adjacency:
            raise ValueError(f"source station number {source} is not in the graph")

        self.dist_to = {}
        # state -> (cost tuple, actual distance, distance-cost, transfers)
        self._labels = {}
        # state -> (previous state, edge used)
        self._pred = {}
        # station number -> best final state
        self._best_state = {}

        self._run()

    # ------------------------------------------------------------ helpers
    def _cost_of(self, dist_cost, transfers):
        """Build the comparable cost tuple for the configured mode."""
        mode = self.config.mode
        if mode == "shortest_distance":
            return (dist_cost,)
        if mode == "least_transfer":
            return (transfers, dist_cost)
        # comprehensive: distance plus a penalty per transfer
        return (dist_cost + transfers * TRANSFER_PENALTY_METERS,)

    # ----------------------------------------------------------- Dijkstra
    def _run(self):
        """Run Dijkstra's algorithm over (station, line, transfers) states."""
        start = (self.source, None, 0)
        zero_cost = self._cost_of(0.0, 0)
        self._labels[start] = (zero_cost, 0.0, 0.0, 0)

        pq = MinPQ()
        pq.insert(zero_cost, start)
        done = set()

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in done:
                continue
            done.add(state)

            node, line, transfers = state
            _, distance, dist_cost, _ = self._labels[state]

            for edge in self.graph.neighbors(node):
                nxt = edge.other_end(node)
                if nxt == node:
                    continue

                is_transfer = line is not None and edge.line != line
                new_transfers = transfers + (1 if is_transfer else 0)
                if new_transfers > self._max_transfers:
                    continue

                step = float(edge.weight)
                if self._preferred and edge.line not in self._preferred:
                    step += LINE_PREFERENCE_PENALTY_METERS

                new_dist_cost = dist_cost + step
                new_distance = distance + float(edge.weight)
                new_cost = self._cost_of(new_dist_cost, new_transfers)
                new_state = (nxt, edge.line, new_transfers)

                if new_state in done:
                    continue

                known = self._labels.get(new_state)
                if known is None or new_cost < known[0]:
                    self._labels[new_state] = (new_cost, new_distance, new_dist_cost, new_transfers)
                    self._pred[new_state] = (state, edge)
                    pq.insert(new_cost, new_state)

        # Select the best finalised state for every reachable station.
        best = {}
        for state in self._labels:
            if state not in done:
                continue
            node = state[0]
            cost = self._labels[state][0]
            current = best.get(node)
            if current is None or cost < self._labels[current][0]:
                best[node] = state

        self._best_state = best
        for node, state in best.items():
            self.dist_to[node] = self._labels[state][1]

    # ------------------------------------------------------------- queries
    def has_path_to(self, target):
        """Return True if ``target`` (station number) is reachable under the constraints."""
        return target in self._best_state

    def path_with_edges(self, target):
        """
        Reconstruct the path from the source to ``target``.

        Returns:
            (list[station number], list[Edge]): Station numbers in travel order
            (source first, target last) and the edges traversed between them.
            Both lists are empty if the target is unreachable.
        """
        if target not in self._best_state:
            return [], []

        state = self._best_state[target]
        nodes = []
        edges = []
        while True:
            nodes.append(state[0])
            if state not in self._pred:
                break
            previous, edge = self._pred[state]
            edges.append(edge)
            state = previous

        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        """
        Return the number of line changes on the path to ``target``.

        Returns 0 if the target is unreachable or equals the source.
        """
        _, edges = self.path_with_edges(target)
        transfers = 0
        for previous, current in zip(edges, edges[1:]):
            if previous.line != current.line:
                transfers += 1
        return transfers

"""
Fare and travel-time calculation for the Beijing Subway.

Fare structure (distance rounded up to whole kilometers):
  * 0-6 km    : 3 yuan
  * 6-12 km   : 4 yuan
  * 12-22 km  : 5 yuan
  * 22-32 km  : 6 yuan
  * > 32 km   : 6 yuan + 1 yuan for every further (or part of) 20 km
"""

import math
from dataclasses import dataclass


@dataclass(frozen=True)
class FareTier:
    """A fare band: distances up to ``max_distance_km`` (inclusive) cost ``fare_yuan``."""

    min_distance_km: float
    max_distance_km: float
    fare_yuan: int


# Base fare tiers (ceiling-rounded kilometers are matched against these bounds).
FARE_TIERS = (
    FareTier(0.0, 6.0, 3),
    FareTier(6.0, 12.0, 4),
    FareTier(12.0, 22.0, 5),
    FareTier(22.0, 32.0, 6),
)

# Extended fare: beyond the last tier's upper bound.
EXTENSION_START_KM = 32.0
EXTENSION_STEP_KM = 20.0
EXTENSION_STEP_FARE = 1

# Average operating speed used for time estimation (km/h).
AVERAGE_SPEED_KMH = 35.0
# Time cost of each transfer (minutes), covering walking and waiting.
TRANSFER_TIME_MINUTES = 5.0


def _validate_distance(distance):
    """Ensure the distance is a non-negative real number."""
    if not isinstance(distance, (int, float)) or isinstance(distance, bool):
        raise TypeError("distance must be a number of meters")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("distance must be a finite number")
    if distance < 0:
        raise ValueError("distance cannot be negative")


def ceil_kilometers(distance):
    """
    Round a distance in meters up to whole kilometers.

    Any fraction of a kilometer counts as a full kilometer, which implements
    the ceiling rounding required by the fare structure.
    """
    _validate_distance(distance)
    return int(math.ceil(distance / 1000.0))


def Sum_money(distance):
    """
    Compute the subway fare for a travelled distance.

    Args:
        distance: Travelled distance in meters.

    Returns:
        int: Fare in yuan.

    Raises:
        TypeError / ValueError: If the distance is not a valid non-negative number.
    """
    kilometers = ceil_kilometers(distance)

    for tier in FARE_TIERS:
        if kilometers <= tier.max_distance_km:
            return tier.fare_yuan

    extra_km = kilometers - EXTENSION_START_KM
    steps = int(math.ceil(extra_km / EXTENSION_STEP_KM))
    return FARE_TIERS[-1].fare_yuan + steps * EXTENSION_STEP_FARE


def calculate_estimated_time(distance, transfer_count):
    """
    Estimate the travel time of a journey in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers on the journey.

    Returns:
        float: Estimated travel time in minutes.
    """
    _validate_distance(distance)
    if not isinstance(transfer_count, int) or isinstance(transfer_count, bool) or transfer_count < 0:
        raise ValueError("transfer_count must be a non-negative integer")

    riding_minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    transfer_minutes = transfer_count * TRANSFER_TIME_MINUTES
    return float(riding_minutes + transfer_minutes)

"""
Query service: validates user input, runs the path search, and assembles
a ``RouteResult`` containing the path, distance, fare, transfers and time.
"""

from dataclasses import dataclass, field

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


class RouteError(Exception):
    """Raised when a query cannot be answered (bad station, no path, calculation failure)."""


@dataclass
class RouteResult:
    """Outcome of a successful route query."""

    start_station: str
    end_station: str
    mode: str
    max_transfers: object
    preferred_lines: list
    station_names: list
    edges: list
    total_distance_m: float
    fare_yuan: int
    transfer_count: int
    transfers: list = field(default_factory=list)
    lines: list = field(default_factory=list)
    estimated_time_minutes: float = 0.0

    @property
    def total_distance_km(self):
        """Total distance converted from meters to kilometers."""
        return self.total_distance_m / 1000.0


def parse_query(text):
    """
    Parse a query of the form ``Starting Station; Ending Station``.

    Full-width semicolons are accepted as separators too.

    Returns:
        (str, str): The stripped starting and ending station names.

    Raises:
        ValueError: If the format is wrong or a name is empty.
    """
    normalized = text.replace("\uff1b", ";")
    parts = normalized.split(";")
    if len(parts) != 2:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    start = " ".join(parts[0].split())
    end = " ".join(parts[1].split())
    if not start or not end:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    return start, end


def resolve_station(graph, name, role):
    """
    Standardise a station name against the official names in the graph.

    Matching is whitespace-normalised and case-insensitive.

    Args:
        graph: The subway ``Graph``.
        name: The user-supplied name.
        role: "Starting" or "Ending" (used in error messages).

    Returns:
        str: The official station name.

    Raises:
        RouteError: If no unique matching station exists.
    """
    cleaned = " ".join(name.split())
    if not cleaned:
        raise RouteError(f"{role} station name cannot be empty")
    if graph.has_station(cleaned):
        return cleaned

    lowered = cleaned.casefold()
    matches = [station for station in graph.station_names() if station.casefold() == lowered]
    if len(matches) == 1:
        return matches[0]

    raise RouteError(f"{role} station '{cleaned}' does not exist, please check the station name")


def find_transfers(station_names, edges):
    """
    Identify transfer stations along a path.

    Returns:
        list[dict]: Each item has ``station``, ``from_line`` and ``to_line``.
    """
    transfers = []
    for index in range(len(edges) - 1):
        if edges[index].line != edges[index + 1].line:
            transfers.append({
                "station": station_names[index + 1],
                "from_line": edges[index].line,
                "to_line": edges[index + 1].line,
            })
    return transfers


def unique_lines(edges):
    """Return the lines used by the path, in order of first use."""
    lines = []
    for edge in edges:
        if edge.line not in lines:
            lines.append(edge.line)
    return lines


def query_route(graph, start_name, end_name, config=None):
    """
    Compute the best route and its fare between two stations.

    Args:
        graph: The subway ``Graph``.
        start_name: Official starting station name.
        end_name: Official ending station name.
        config: Optional ``PathfindingConfig``.

    Returns:
        RouteResult

    Raises:
        RouteError: On invalid stations, identical endpoints, no route, or
                    calculation failures.
    """
    config = config if config is not None else PathfindingConfig()

    if not graph.has_station(start_name):
        raise RouteError(f"Starting station '{start_name}' does not exist, please check the station name")
    if not graph.has_station(end_name):
        raise RouteError(f"Ending station '{end_name}' does not exist, please check the station name")
    if start_name == end_name:
        raise RouteError("Starting station cannot be the same as the ending station")

    try:
        search = ShortestPath(graph, graph.get_number(start_name), config)
    except ValueError as exc:
        raise RouteError(f"Path calculation failed: {exc}") from exc

    target = graph.get_number(end_name)
    if not search.has_path_to(target):
        raise RouteError(
            f"No route found from '{start_name}' to '{end_name}' "
            f"under the current settings (mode={config.mode}, "
            f"max_transfers={config.max_transfers}, line_preference={config.preferred_lines or 'none'})"
        )

    try:

"""
Report generation for route query results.

Supported formats:
  * ``text`` - human-readable console report with line segments and transfers.
  * ``json`` - structured data for programmatic integration.
"""

import json

from route_service import RouteResult

SUPPORTED_FORMATS = ("text", "json")


def format_distance(meters):
    """
    Format a distance showing both meters and kilometers.

    Short distances lead with meters, long distances lead with kilometers.
    """
    if meters < 1000:
        return f"{meters:.0f} m ({meters / 1000.0:.2f} km)"
    return f"{meters / 1000.0:.2f} km ({meters:.0f} m)"


def build_segments(station_names, edges):
    """
    Group consecutive edges operated by the same line into segments.

    Returns:
        list[dict]: Each item has ``line``, ``start``, ``end``, ``hops``, ``distance_m``.
    """
    segments = []
    for index, edge in enumerate(edges):
        if segments and segments[-1]["line"] == edge.line:
            segment = segments[-1]
            segment["end"] = station_names[index + 1]
            segment["hops"] += 1
            segment["distance_m"] += edge.weight
        else:
            segments.append({
                "line": edge.line,
                "start": station_names[index],
                "end": station_names[index + 1],
                "hops": 1,
                "distance_m": float(edge.weight),
            })
    return segments


def _max_transfers_label(max_transfers):
    return "unlimited (default limit applies)" if max_transfers is None else str(max_transfers)


def render_text(result):
    """Render a ``RouteResult`` as a console-formatted text report."""
    if not isinstance(result, RouteResult):
        raise TypeError("render_text expects a RouteResult")

    divider = "=" * 60
    preference = ", ".join(result.preferred_lines) if result.preferred_lines else "none"
    segments = build_segments(result.station_names, result.edges)

    out = []
    out.append(divider)
    out.append("Beijing Subway Fare Query")
    out.append(divider)
    out.append(f"Query:           {result.start_station} -> {result.end_station}")
    out.append(f"Mode:            {result.mode}")
    out.append(f"Max transfers:   {_max_transfers_label(result.max_transfers)}")
    out.append(f"Line preference: {preference}")
    out.append("")

    out.append(f"Station sequence ({len(result.station_names)} stations):")
    for index, name in enumerate(result.station_names, start=1):
        marker = ""
        if 1 < index < len(result.station_names) and result.station_names[index - 1] is not None:
            if any(t["station"] == name for t in result.transfers):
                marker = "  [transfer]"
        out.append(f"  {index}. {name}{marker}")
    out.append("")

    out.append("Line segments:")
    for segment in segments:
        out.append(
            f"  [{segment['line']}] {segment['start']} -> {segment['end']} "
            f"({segment['hops']} interval(s), {format_distance(segment['distance_m'])})"
        )
    out.append("")

    if result.transfers:
        out.append(f"Transfer stations ({len(result.transfers)}):")
        for transfer in result.transfers:
            out.append(f"  - {transfer['station']}: {transfer['from_line']} -> {transfer['to_line']}")
    else:
        out.append("Transfer stations: none (no transfer required)")
    out.append("")

    out.append("Lines involved: " + (", ".join(result.lines) if result.lines else "none"))
    out.append("")

    out.append("Summary:")
    out.append(f"  Total distance:    {format_distance(result.total_distance_m)}")
    out.append(f"  Transfers:         {result.transfer_count}")
    out.append(f"  Estimated time:    {result.estimated_time_minutes:.1f} min")
    out.append(f"Fare: {result.fare_yuan} yuan")
    out.append(divider)
    return "\n".join(out)


def build_json_data(result):
    """Build the JSON-serialisable dictionary for a ``RouteResult``."""
    return {
        "query": {
            "start_station": result.start_station,
            "end_station": result.end_station,
            "mode": result.mode,
            "max_transfers": result.max_transfers,
            "line_preference": result.preferred_lines,
        },
        "path": list(result.station_names),
        "segments": build_segments(result.station_names, result.edges),
        "distance_meters": round(result.total_distance_m, 2),
        "distance_kilometers": round(result.total_distance_km, 3),
        "fare_yuan": result.fare_yuan,
        "estimated_time_minutes": round(result.estimated_time_minutes, 1),
        "transfer_count": result.transfer_count,
        "transfers": list(result.transfers),
        "lines": list(result.lines),
    }


def render_json(result):
    """Render a ``RouteResult`` as an indented JSON string."""
    if not isinstance(result, RouteResult):
        raise TypeError("render_json expects a RouteResult")
    return json.dumps(build_json_data(result), ensure_ascii=False, indent=2)


def render_report(result, fmt="text"):
    """
    Render a route result in the requested format.

    Args:
        result: A ``RouteResult``.
        fmt: "text" or "json".

    Raises:
        ValueError: If the format is unsupported.
    """
    if fmt == "text":
        return render_text(result)
    if fmt == "json":
        return render_json(result)
    raise ValueError(f"unsupported output format '{fmt}', expected one of {SUPPORTED_FORMATS}")

"""
Data integrity checks for the subway dataset.

Checks performed:
  * Station coverage: each main line has representative stations.
  * Connection integrity: endpoints exist, lines are served by both endpoints,
    no duplicate connections, graph connectivity, no isolated stations.
  * Distance accuracy: distances are positive numbers, with suspicious values flagged.
"""

from dataclasses import dataclass, field

MAIN_LINES = (
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
)

# Minimum number of stations expected on each main line.
MIN_REPRESENTATIVE_STATIONS = 2
# Distances above this (meters) between adjacent stations are reported as suspicious.
MAX_PLAUSIBLE_DISTANCE_M = 5000


@dataclass
class IntegrityCheck:
    """Result of a data integrity check."""

    passed: bool
    messages: list = field(default_factory=list)


def _is_number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def check_integrity(data):
    """
    Run all integrity checks on the raw subway dataset.

    Args:
        data: dict loaded from ``subway_data.json``.

    Returns:
        IntegrityCheck: ``passed`` is False if any error was found; ``messages``
        lists every error, warning and a closing summary.
    """
    errors = []
    warnings = []

    stations = data.get("stations") if isinstance(data, dict) else None
    connections = data.get("connections") if isinstance(data, dict) else None
    if not isinstance(stations, list) or not isinstance(connections, list):
        return IntegrityCheck(False, ["[ERROR] dataset must contain 'stations' and 'connections' lists"])

    # ---- Station data ----------------------------------------------------
    station_lines = {}
    for index, entry in enumerate(stations):
        if not isinstance(entry, dict) or not isinstance(entry.get("name"), str) or not entry["name"].strip():
            errors.append(f"[ERROR] station entry #{index} has no valid name")
            continue
        name = entry["name"]
        if name in station_lines:
            errors.append(f"[ERROR] duplicate station name '{name}'")
            continue
        lines = entry.get("lines", [])
        if not isinstance(lines, list):
            errors.append(f"[ERROR] station '{name}' has an invalid line list")
            lines = []
        station_lines[name] = set(lines)

    # Station coverage of main lines.
    line_counts = {}
    for lines in station_lines.values():
        for line in lines:
            line_counts[line] = line_counts.get(line, 0) + 1
    for line in MAIN_LINES:
        count = line_counts.get(line, 0)
        if count == 0:
            errors.append(f"[ERROR] main line '{line}' has no stations in the dataset")
        elif count < MIN_REPRESENTATIVE_STATIONS:
            warnings.append(f"[WARN] main line '{line}' has only {count} representative station(s)")

    # ---- Connection data -------------------------------------------------
    seen = set()
    adjacency = {name: set() for name in station_lines}
    for index, conn in enumerate(connections):
        if not isinstance(conn, dict):
            errors.append(f"[ERROR] connection #{index} is not an object")
            continue
        source = conn.get("source")
        dest = conn.get("dest")
        line = conn.get("line")
        distance = conn.get("distance")
        label = f"{source} - {dest} ({line})"

        if source not in station_lines:
            errors.append(f"[ERROR] connection {label}: unknown source station '{source}'")
        if dest not in station_lines:
            errors.append(f"[ERROR] connection {label}: unknown destination station '{dest}'")

        if source in station_lines and dest in station_lines:
            if line not in station_lines[source] or line not in station_lines[dest]:
                errors.append(f"[ERROR] connection {label}: line not served by both endpoints")
            adjacency[source].add(dest)
            adjacency[dest].add(source)

        if not _is_number(distance) or distance <= 0:
            errors.append(f"[ERROR] connection {label}: distance must be a positive number, got {distance!r}")
        else:
            if isinstance(distance, float) and not distance.is_integer():
                warnings.append(f"[WARN] connection {label}: distance {distance} is not whole meters")
            if distance > MAX_PLAUSIBLE_DISTANCE_M:
                warnings.append(f"[WARN] connection {label}: distance {distance} m looks implausibly large")

        key = (frozenset((source, dest)), line)
        if key in seen:
            errors.append(f"[ERROR] duplicate connection {label}")
        seen.add(key)

    # Isolated stations and connectivity.
    isolated = [name for name, neighbours in adjacency.items() if not neighbours]
    for name in isolated:
        warnings.append(f"[WARN] station '{name}' has no connections")

    if station_lines:
        start = next(iter(station_lines))
        visited = {start}
        stack = [start]
        while stack:
            current = stack.pop()
            for nxt in adjacency.get(current, ()):
                if nxt not in visited:
                    visited.add(nxt)
                    stack.append(nxt)
        unreachable = len(station_lines) - len(visited)
        if unreachable > 0:
            warnings.append(f"[WARN] {unreachable} station(s) are not connected to the main network")

    messages = errors + warnings
    passed = not errors
    summary = (
        f"Summary: {len(station_lines)} stations, {len(connections)} connections, "
        f"{len(errors)} error(s), {len(warnings)} warning(s)"
    )
    messages.append(summary)
    if passed and not warnings:
        messages.append("[OK] data integrity check passed")
    return IntegrityCheck(passed, messages)

"""
Data update utilities for the subway dataset.

All functions operate on the in-memory dataset dictionary (as returned by
``data_loader.load_subway_data``); ``save_subway_data`` persists changes.
"""

import json

from data_loader import DATA_FILE_PATH


def _find_station(data, name):
    for entry in data["stations"]:
        if entry.get("name") == name:
            return entry
    return None


def _line_exists(data, line_name):
    if any(line_name in entry.get("lines", []) for entry in data["stations"]):
        return True
    return any(conn.get("line") == line_name for conn in data["connections"])


def _validate_distance(distance):
    if not isinstance(distance, int) or isinstance(distance, bool) or distance <= 0:
        raise ValueError("distance must be a positive integer number of meters")


def add_line(data, line_name, station_names, connections):
    """
    Add a new line with its stations and connections.

    Args:
        data: The dataset dictionary (modified in place).
        line_name: Name of the new line.
        station_names: Ordered list of station names on the line. Existing
                       stations gain the new line; unknown ones are created.
        connections: List of dicts with ``source``, ``dest`` and ``distance``
                     (meters). Endpoints must be in ``station_names``.

    Returns:
        dict: The updated dataset.

    Raises:
        ValueError: On invalid or duplicate input.
    """
    line_name = str(line_name).strip()
    if not line_name:
        raise ValueError("line name cannot be empty")
    if _line_exists(data, line_name):
        raise ValueError(f"line '{line_name}' already exists")

    stations = []
    for name in station_names:
        cleaned = str(name).strip()
        if cleaned and cleaned not in stations:
            stations.append(cleaned)
    if len(stations) < 2:
        raise ValueError("a line needs at least two stations")

    for name in stations:
        entry = _find_station(data, name)
        if entry is None:
            data["stations"].append({"name": name, "lines": [line_name]})
        elif line_name not in entry.setdefault("lines", []):
            entry["lines"].append(line_name)

    for conn in connections:
        source = str(conn["source"]).strip()
        dest = str(conn["dest"]).strip()
        if source not in stations or dest not in stations:
            raise ValueError(f"connection {source} - {dest} references stations outside the line")
        _validate_distance(conn["distance"])
        data["connections"].append({
            "source": source,
            "dest": dest,
            "distance": conn["distance"],
            "line": line_name,
        })
    return data


def modify_station(data, name, new_name=None, lines=None):
    """
    Modify a station's name and/or lines.

    Renaming also updates every connection that references the station.

    Raises:
        KeyError: If the station does not exist.
        ValueError: If the new name is already taken.
    """
    entry = _find_station(data, name)
    if entry is None:
        raise KeyError(f"station '{name}' does not exist")

    if lines is not None:
        entry["lines"] = list(dict.fromkeys(str(line) for line in lines))

    if new_name is not None and new_name != name:
        new_name = str(new_name).strip()
        if not new_name:
            raise ValueError("new station name cannot be empty")
        if _find_station(data, new_name) is not None:
            raise ValueError(f"station '{new_name}' already exists")
        entry["name"] = new_name
        for conn in data["connections"]:
            if conn.get("source") == name:
                conn["source"] = new_name
            if conn.get("dest") == name:
                conn["dest"] = new_name
    return entry


def update_distance(data, source, dest, distance, line=None):
    """
    Update the distance of the connection(s) between two stations.

    Args:
        line: If given, only the connection on this line is updated.

    Returns:
        int: Number of connections updated.

    Raises:
        KeyError: If no matching connection exists.
        ValueError: If the distance is not a positive integer.
    """
    _validate_distance(distance)
    matches = [
        conn for conn in data["connections"]
        if {conn.get("source"), conn.get("dest")} == {source, dest}
        and (line is None or conn.get("line") == line)
    ]
    if not matches:
        raise KeyError(f"no connection between '{source}' and '{dest}'")
    for conn in matches:
        conn["distance"] = distance
    return len(matches)


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write the dataset back to a JSON file (UTF-8, indented)."""
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")
    return path

Command-line entry point for the Beijing Subway fare calculation system.

Usage examples:
  python src/main.py
  python src/main.py --mode least_transfer --format json
  python src/main.py --max_transfers 2 --line_preference "Line 1" --output_file result.txt

Interactive input format: ``Starting Station; Ending Station``.

import argparse
import os
import sys

# Make sibling modules importable regardless of how the script is launched.
_HERE = os.path.dirname(os.path.abspath(__file__))
if _HERE not in sys.path:
    sys.path.insert(0, _HERE)

from data_loader import DATA_FILE_PATH, load_subway_data  # noqa: E402
from integrity import check_integrity  # noqa: E402
from network_builder import build_graph  # noqa: E402

from route_service import RouteError, parse_query, query_route, resolve_station  # noqa: E402

EXIT_WORDS = {"quit", "exit", "q"}


def build_arg_parser():
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (shortest-path based)."
    )
    parser.add_argument("--output_file", default=None,
                        help="append query results to this file")
    parser.add_argument("--format", dest="format", choices=SUPPORTED_FORMATS, default="text",
                        help="output format: text or json (default: text)")
    parser.add_argument("--mode", choices=VALID_MODES, default="shortest_distance",
                        help="path query mode (default: shortest_distance)")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="maximum number of transfers allowed")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line name, e.g. 'Line 1'")
    parser.add_argument("--check_data", action="store_true",
                        help="run data integrity check and exit")
    return parser


def process_query(text, graph, config, fmt):
    """
    Parse, validate and answer one query.

    Returns:
        str: The rendered report.

    Raises:
        RouteError / ValueError: On invalid input or when no route is available.
    """
    start_raw, end_raw = parse_query(text)
    start = resolve_station(graph, start_raw, "Starting")
    end = resolve_station(graph, end_raw, "Ending")
    if start == end:
        raise RouteError("Starting station cannot be the same as the ending station")
    result = query_route(graph, start, end, config)
    return render_report(result, fmt)


def append_to_file(path, text):
    """Append a report to the output file, separated from earlier reports."""
    with open(path, "a", encoding="utf-8") as handle:
        handle.write(text)
        handle.write("\n\n")


def main(argv=None):
    """Run the CLI. Returns the process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except (OSError, ValueError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        result = check_integrity(data)
        for message in result.messages:
            print(message)
        return 0 if result.passed else 1

    try:
        graph = build_graph(data)
    except (KeyError, TypeError, ValueError) as exc:
        print(f"Error: failed to build the subway network: {exc}", file=sys.stderr)
        return 1

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer", file=sys.stderr)
        return 2

    preferred_lines = []
    if args.line_preference:
        line = " ".join(args.line_preference.split())
        if line not in graph.all_lines():
            print(f"Error: unknown line preference '{line}'. Known lines: {', '.join(graph.all_lines())}",
                  file=sys.stderr)
            return 2
        preferred_lines = [line]

    config = PathfindingConfig(
        mode=args.mode,
        max_transfers=args.max_transfers,
        preferred_lines=preferred_lines,
    )

    print("Beijing Subway Fare Calculation System")
    print(f"Mode: {args.mode} | Format: {args.format}"
          + (f" | Output file: {args.output_file}" if args.output_file else ""))
    print("Enter a query as 'Starting Station; Ending Station' (or 'quit' to exit).")

    while True:
        try:
            raw = input("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()

        if not text:
            continue
        if text.lower() in EXIT_WORDS:
            break

        try:
            report = process_query(text, graph, config, args.format)

        except Exception as exc:  # defensive: keep the session alive
            print(f"Error: unexpected failure while answering the query: {exc}")
            continue

        print(report)

        if args.output_file:
            try:
                append_to_file(args.output_file, report)
            except OSError as exc:
                print(f"Error: could not write to '{args.output_file}': {exc}")

    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

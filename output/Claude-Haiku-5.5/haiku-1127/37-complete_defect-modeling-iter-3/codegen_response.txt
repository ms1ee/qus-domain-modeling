CHENG_RU_GROUP = "Cheng Ru Group"

CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"


def normalize_text(text) -> str:
    """Normalize text for case-insensitive, whitespace-insensitive comparison."""
    return " ".join(str(text).split()).casefold()


def is_same_text(first, second) -> bool:
    """Return True if two strings are equal after normalization."""
    return normalize_text(first) == normalize_text(second)


def is_cheng_ru_group(department) -> bool:
    """Return True if the department is the Cheng Ru group."""
    return is_same_text(department, CHENG_RU_GROUP)

import os

DEFAULT_USER_RECORDS = [
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Cheng Ru Group"),
    ("Li Bin", "Marketing Department"),
    ("Zhang San", "Technology Department"),
    ("Li Si", "Product Department"),
    ("Wang Wu", "Cheng Ru Group"),
    ("Zhao Liu", "Finance Department"),
    ("Sun Qi", "Operations Department"),
    ("Zhou Ba", "Cheng Ru Group"),
    ("Wu Jiu", "Technology Department"),
    ("Zheng Shi", "Product Department"),
    ("Feng Yi", "Marketing Department"),
    ("Chu Er", "Cheng Ru Group"),
    ("Wei San", "Finance Department"),
    ("Jiang Si", "Operations Department"),
    ("Shen Wu", "Technology Department"),
    ("Han Liu", "Product Department"),
]


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        """Load the built-in team building participant list."""
        users = [User(name, department) for name, department in DEFAULT_USER_RECORDS]
        return users, []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        """Parse a txt file into users; returns (valid users, error messages)."""
        users: List[User] = []
        errors: List[str] = []

        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.readlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]

        for line_number, raw in enumerate(lines, start=1):
            content = raw.strip()
            if not content:
                continue
            parts = content.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_number}: '{content}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))

        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        """Split users into (valid users keeping first occurrence, duplicate entries)."""
        seen = set()
        valid: List[User] = []
        duplicates: List[User] = []
        for user in users:
            key = normalize_text(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        """Return a mapping of department group name to number of members."""
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management."""

from typing import List, Optional

from lottery_system.constants import normalize_text
from lottery_system.models import Prize


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    PRESET_PRIZES = [
        ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
        ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
        ("First Prize - 1 Day Annual Leave", 1, [], "Li Bin"),
    ]

    def __init__(self):
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        """Load the preset prize library. Returns a list of error messages."""
        errors: List[str] = []
        for name, quantity, avoid_groups, drawer in self.PRESET_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        """Append a prize to the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(normalize_text(p.name) == normalize_text(clean_name) for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups: List[str] = []
        for group in avoid_groups or []:
            cleaned = str(group).strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)

        drawer: Optional[str] = None
        if associated_drawer and str(associated_drawer).strip():
            drawer = str(associated_drawer).strip()

        self._prizes.append(Prize(clean_name, quantity, groups, drawer))

    def get_prizes(self) -> List[Prize]:
        """Return the prizes in the current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        """Reorder prizes. order contains 1-based current positions in the new order."""
        total = len(self._prizes)
        try:
            positions = [int(item) for item in order]
        except (TypeError, ValueError):
            return False
        if sorted(positions) != list(range(1, total + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in positions]
        return True

"""Lottery rule engine: eligibility filtering, drawing and fallback protection."""

import random
from typing import List, Optional

from lottery_system.constants import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    is_cheng_ru_group,
    is_same_text,
    normalize_text,
)
from lottery_system.models import Prize, User


class LotteryManager:
    """Runs lottery draws for a set of participants while enforcing rules."""

    TRACKED_DRAWERS = (CHEN_QIXIAN, JIN_DUO)
    PROTECTED_DRAWER = MA_JIAQI

    def __init__(self, users):
        self._users: List[User] = list(users)
        self._winner_keys = set()
        self._winners: List[User] = []
        self._random = random.Random()
        # Results (True = contained Cheng Ru group members) of the last two
        # rounds drawn by Chen Qixian / Jin Duo.
        self.last_two_draw_results: List[bool] = []

    @property
    def winners(self) -> List[User]:
        """Winners in the order they were drawn."""
        return list(self._winners)

    def has_won(self, user: User) -> bool:
        """Return True if the person has already won any prize."""
        return normalize_text(user.name) in self._winner_keys

    def _is_cheng_ru_protection_active(self) -> bool:
        """Protection triggers when the last two tracked results both lack Cheng Ru members."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        """Compute the set of participants that may win the given prize."""
        available = [user for user in self._users if not self.has_won(user)]
        drawer = prize.associated_drawer

        # Special rule: Li Bin only triggers the previous-winner filter.
        if drawer and is_same_text(drawer, LI_BIN):
            return available

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru group members,
        # even if that group is an avoidance group of this prize.
        if drawer and is_same_text(drawer, self.PROTECTED_DRAWER) and self._is_cheng_ru_protection_active():
            protected = [user for user in available if is_cheng_ru_group(user.department)]
            if protected:
                return protected

        avoided = {normalize_text(group) for group in prize.avoid_groups}
        return [user for user in available if normalize_text(user.department) not in avoided]

    def eligible_pool(self, prize: Prize) -> List[User]:
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def draw_one(self, prize: Prize) -> Optional[User]:
        """Draw one winner for the prize. Returns None if no eligible participant remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._random.choice(pool)
        self._winners.append(winner)
        self._winner_keys.add(normalize_text(winner.name))
        return winner

    def end_round(self, prize: Prize, round_winners) -> None:
        """Record the outcome of a finished round for the fallback mechanism."""
        drawer = prize.associated_drawer
        if drawer and any(is_same_text(drawer, name) for name in self.TRACKED_DRAWERS):
            hit = any(is_cheng_ru_group(user.department) for user in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def describe_rules(self, prize: Prize) -> str:
        """Return a human-readable description of the rules applied to a prize round."""
        parts = ["Previous winners are excluded"]
        drawer = prize.associated_drawer

        if drawer:
            parts.append(f"Lottery person: {drawer}")
        else:
            parts.append("No lottery person specified")

        if drawer and is_same_text(drawer, LI_BIN):
            parts.append("Special rule: avoidance groups are not applied")
        elif prize.avoid_groups:
            parts.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")

        if drawer and is_same_text(drawer, self.PROTECTED_DRAWER):
            if self._is_cheng_ru_protection_active():
                parts.append(f"Cheng Ru group protection active: pool limited to {CHENG_RU_GROUP} members")
            else:
                parts.append("Cheng Ru group protection not triggered")

        return "; ".join(parts)

"""Winner recording, grouping and CSV export."""

import csv
import datetime
import os
from dataclasses import dataclass
from typing import Dict, List, Tuple

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize_name: str
    winning_order: int
    win_time: datetime.datetime


class ResultManager:
    """Keeps all winning records of the current lottery."""

    def __init__(self):
        self._winners: List[Winner] = []

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner for a prize."""
        record = Winner(
            user=user,
            prize_name=prize.name,
            winning_order=len(self._winners) + 1,
            win_time=datetime.datetime.now(),
        )
        self._winners.append(record)
        return record

    def get_winners(self) -> List[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        """Group winners by prize name, keeping the order of the given prizes."""
        grouped: Dict[str, List[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            if winner.prize_name in grouped:
                grouped[winner.prize_name].append(winner)
        return grouped

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        """Export the complete winner list as CSV into the current directory."""
        date_text = datetime.datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_text}.txt")

        grouped = self.get_winners_by_prize(prizes)
        ordered: List[Winner] = []
        for prize in prizes:
            ordered.extend(grouped.get(prize.name, []))
        known_names = {prize.name for prize in prizes}
        ordered.extend(w for w in self._winners if w.prize_name not in known_names)

        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Winning Order", "Name", "Department", "Prize"])
                for winner in ordered:
                    writer.writerow([
                        winner.winning_order,
                        winner.user.name,
                        winner.user.department,
                        winner.prize_name,
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"

        return True, f"Result successfully exported to {path}"

"""Command-line interface for the lottery system."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


CMD_PAUSE = "pause"
CMD_CONTINUE = "continue"
CMD_TERMINATE = "terminate"
ACTION_DRAW = "draw"


def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer, e.g. 1st, 12th."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _split_tokens(raw: str):
    return [token.strip() for token in raw.split(",") if token.strip()]


class LotterySystemCLI:
    """Interactive command-line controller."""

    def __init__(self, input_func=input, output_func=print):
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.list_loaded = False
        self.lottery_manager = None
        self.result_manager = ResultManager()
        self._handlers = {
            "1": self._load_default_list,
            "2": self._load_custom_list,
            "3": self._view_groups,
            "4": self._load_preset_prizes,
            "5": self._add_prize,
            "6": self._view_prizes,
            "7": self._reorder_prizes,
            "8": self._start_lottery,
            "9": self._show_results,
            "10": self._export_results,
        }

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str = "") -> str:
        return self._input(prompt).strip()

    def _ask_positive_int(self, prompt: str) -> int:
        while True:
            raw = self._ask(prompt)
            if raw.isdigit() and int(raw) > 0:
                return int(raw)
            self._print("Please enter a positive whole number.")

    def _group_names(self):
        return list(self.list_manager.get_group_info(self.users).keys())

    # --------------------------------------------------------------- main loop
    def run(self) -> None:
        self._print("=== Company Team Building Smart Lottery System ===")
        try:
            while True:
                self._print_menu()
                choice = self._ask("Select an option: ")
                if choice == "0":
                    break
                handler = self._handlers.get(choice)
                if handler is None:
                    self._print("Invalid option, please try again.")
                    continue
                handler()
        except (EOFError, KeyboardInterrupt):
            self._print("\nInput closed, exiting system.")
        self._print("Goodbye!")

    def _print_menu(self) -> None:
        self._print("")
        self._print("---------------- Main Menu ----------------")
        self._print("1.  Load built-in default participant list")
        self._print("2.  Upload custom txt participant list")
        self._print("3.  View department groups")
        self._print("4.  Load preset prizes")
        self._print("5.  Add custom prize")
        self._print("6.  View prizes")
        self._print("7.  Reorder prizes")
        self._print("8.  Start lottery")
        self._print("9.  View winner list")
        self._print("10. Export winner list (CSV)")
        self._print("0.  Exit")
        self._print("-------------------------------------------")

    # ------------------------------------------------------- list management
    def _load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._finalize_list(users, errors)

    def _load_custom_list(self) -> None:
        path = self._ask("Enter the path of the custom txt list: ").strip('"')
        if not path:
            self._print("No path entered.")
            return
        users, errors = self.list_manager.load_custom_list(path)
        self._finalize_list(users, errors)

    def _finalize_list(self, users, errors) -> bool:
        if errors:
            self._print(f"Found {len(errors)} invalid record(s):")
            for error in errors:
                self._print("  - " + error)
            choice = self._ask(
                "Enter 'c' to skip invalid records and continue, or 't' to terminate upload: "
            ).lower()
            if choice != "c":
                self._print("Upload terminated. Participant list unchanged.")
                return False

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self._print(f"Found {len(duplicates)} duplicate name(s):")
            for dup in duplicates:
                self._print(f"  - {dup.name} ({dup.department})")
            choice = self._ask(
                "Enter 'r' to remove duplicates (keep first occurrence), or 't' to terminate upload: "
            ).lower()
            if choice != "r":
                self._print("Upload terminated. Participant list unchanged.")
                return False

        if not valid:
            self._print("No valid participants found. Participant list unchanged.")
            return False

        self.users = valid
        self.list_loaded = True
        self.lottery_manager = None
        self._print(f"Participant list loaded: {len(valid)} participant(s).")
        return True

    def _view_groups(self) -> None:
        if not self.list_loaded:
            self._print("Please load a participant list first.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._print("Department groups:")
        for index, (department, count) in enumerate(info.items(), start=1):
            self._print(f"  {index}. {department}: {count} person(s)")
        self._print(f"Total: {len(self.users)} participant(s)")

    # ----------------------------------------------------------- prize config
    def _load_preset_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            for error in errors:
                self._print("Error: " + error)
        else:
            self._print("Preset prizes loaded.")

    def _select_avoid_groups(self):
        groups = self._group_names()
        if not groups:
            self._print("No participant list loaded; no avoidance groups available.")
            return []
        self._print("Available department groups:")
        for index, department in enumerate(groups, start=1):
            self._print(f"  {index}. {department}")
        raw = self._ask("Select avoidance group numbers (comma-separated, blank for none): ")
        selected = []
        for token in _split_tokens(raw):
            if token.isdigit() and 1 <= int(token) <= len(groups):
                group = groups[int(token) - 1]
                if group not in selected:
                    selected.append(group)
            else:
                self._print(f"Ignoring invalid group selection: {token}")
        return selected

    def _add_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity = self._ask_positive_int("Quantity: ")
        avoid_groups = self._select_avoid_groups()
        drawer = self._ask("Associated lottery person (blank for none, e.g. Chen Qixian): ")
        try:
            self.prize_manager.add_prize(name, quantity, avoid_groups, drawer or None)
        except ValueError as exc:
            self._print(f"Error: {exc}")
            return
        self._print("Prize added.")

    def _view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured.")
            return
        self._print("Prize sequence:")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
            drawer = prize.associated_drawer or "None"
            self._print(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Lottery person: {drawer} | Avoid groups: {avoid}"
            )

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes to reorder.")
            return
        self._view_prizes()
        raw = self._ask(
            "Enter new order as serial numbers separated by commas (e.g. 2,1,3); blank to keep: "
        )
        if not raw:
            return
        try:
            order = [int(token) for token in _split_tokens(raw)]
        except ValueError:
            self._print("Invalid input: please enter serial numbers only.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Prize order updated.")
            self._view_prizes()
        else:
            self._print("Invalid order: each serial number must appear exactly once.")

    # ------------------------------------------------------- lottery process
    def _start_lottery(self) -> None:
        if not self.list_loaded:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured. Load presets or add prizes first.")
            return
        if self.result_manager.get_winners():
            confirm = self._ask("Previous lottery results will be cleared. Continue? (y/n): ")
            if confirm.lower() != "y":
                return

        self.result_manager = ResultManager()
        self.lottery_manager = LotteryManager(self.users)
        self._print("")
        self._print("=== Lottery started ===")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize)
        self._print("")
        self._print("=== All rounds completed ===")
        self._show_final_results(prizes)

    def _await_command(self) -> str:
        """Wait for the operator's command before the next draw.

        Returns ACTION_DRAW to proceed, or CMD_TERMINATE to end the round.
        """
        while True:
            command = self._ask("[Enter] draw next | 'pause' | 'terminate': ").lower()
            if command in ("", CMD_CONTINUE):
                return ACTION_DRAW
            if command == CMD_TERMINATE:
                return CMD_TERMINATE
            if command == CMD_PAUSE:
                self._print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    resume = self._ask("(paused) ").lower()
                    if resume == CMD_CONTINUE:
                        self._print("Lottery resumed.")
                        return ACTION_DRAW
                    if resume == CMD_TERMINATE:
                        return CMD_TERMINATE
                    self._print("Unknown command. Use 'continue' or 'terminate'.")
            self._print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    def _run_round(self, index: int, total: int, prize) -> None:
        self._print("")
        self._print(f"--- Round {index}/{total}: {prize.name} ---")
        self._print(f"Quantity: {prize.quantity}")
        self._print(f"Rules: {self.lottery_manager.describe_rules(prize)}")

        round_winners = []
        terminated = False
        while len(round_winners) < prize.quantity:
            if self._await_command() == CMD_TERMINATE:
                terminated = True
                break
            position = len(round_winners) + 1
            self._print(f"Currently drawing the {ordinal(position)} winner...")
            winner = self.lottery_manager.draw_one(prize)
            if winner is None:
                self._print("No eligible participants remain for this prize. Round ends early.")
                break
            round_winners.append(winner)
            self.result_manager.add_winner(winner, prize)
            self._print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}!")

        self.lottery_manager.end_round(prize, round_winners)

        if terminated:
            self._print(f"Round terminated. {len(round_winners)} of {prize.quantity} winner(s) drawn.")
        self._print(f"Results of round {index}:")
        if round_winners:
            for order, winner in enumerate(round_winners, start=1):
                self._print(f"  {order}. {winner.name} ({winner.department})")
        else:
            self._print("  (no winners)")

    # ---------------------------------------------------------------- results
    def _show_final_results(self, prizes) -> None:
        if not self.result_manager.get_winners():
            self._print("No winners recorded yet.")
            return
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self._print("")
        self._print("=== Winner List ===")
        for prize in prizes:
            winners = grouped.get(prize.name, [])
            self._print(f"{prize.name}:")
            if not winners:
                self._print("  (no winners)")
            for winner in winners:
                self._print(
                    f"  #{winner.winning_order} {winner.user.name} ({winner.user.department})"
                )
        self._print(f"Total winners: {len(self.result_manager.get_winners())}")

    def _show_results(self) -> None:
        self._show_final_results(self.prize_manager.get_prizes())

    def _export_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        success, message = self.result_manager.export_to_csv(prizes)
        self._print(message)


def main() -> None:
    """Start the interactive lottery system."""
    LotterySystemCLI().run()

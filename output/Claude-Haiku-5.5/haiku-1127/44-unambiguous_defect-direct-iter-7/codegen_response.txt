    """Load and validate the subway dataset.

    Args:
        path: Path of the JSON file to read.

    Returns:
        The parsed JSON content as a dictionary with the keys
        ``stations`` and ``connections``.

    Raises:
        FileNotFoundError: If the file does not exist.
        ValueError: If the JSON is malformed or does not match the expected
            structure.
    """
    path = os.fspath(path)
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError:
        raise FileNotFoundError(f"subway data file not found: {path}") from None
    except json.JSONDecodeError as exc:
        raise ValueError(f"invalid JSON in {path}: {exc}") from exc
    return validate_subway_data(data)


def _is_number(value) -> bool:
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def validate_subway_data(data) -> dict:
    """Check that ``data`` follows the subway data structure.

    Raises:
        ValueError: On the first structural problem found.
    """
    if not isinstance(data, dict):
        raise ValueError("subway data must be a JSON object")
    stations = data.get("stations")
    connections = data.get("connections")
    if not isinstance(stations, list):
        raise ValueError("'stations' must be a list")
    if not isinstance(connections, list):
        raise ValueError("'connections' must be a list")

    for index, station in enumerate(stations):
        if not isinstance(station, dict):
            raise ValueError(f"stations[{index}] must be an object")
        name = station.get("name")
        if not isinstance(name, str) or not name.strip():
            raise ValueError(f"stations[{index}].name must be a non-empty string")
        lines = station.get("lines", [])
        if not isinstance(lines, list) or not all(isinstance(x, str) for x in lines):
            raise ValueError(f"stations[{index}].lines must be a list of strings")

    for index, conn in enumerate(connections):
        if not isinstance(conn, dict):
            raise ValueError(f"connections[{index}] must be an object")
        for key in ("source", "dest", "line"):
            if not isinstance(conn.get(key), str) or not conn.get(key).strip():
                raise ValueError(f"connections[{index}].{key} must be a non-empty string")
        if not _is_number(conn.get("distance")):
            raise ValueError(f"connections[{index}].distance must be a number (meters)")
    return data

"""Binary min-heap priority queue used by Dijkstra's algorithm."""


class MinPQ:
    """Min priority queue implemented as an array-based binary heap.

    Insertion and removal of the minimum both run in O(log n), which gives
    Dijkstra's algorithm its O((V + E) log V) running time.
    Keys are compared with the ``<`` operator, so tuples work as keys.
    """

    def __init__(self) -> None:
        self._keys: list = []
        self._items: list = []

    def insert(self, item, key) -> None:
        """Add ``item`` with priority ``key``."""
        self._keys.append(key)
        self._items.append(item)
        self._swim(len(self._keys) - 1)

    def del_min(self):
        """Remove and return the item with the smallest key.

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._keys:
            raise IndexError("priority queue is empty")
        self._exch(0, len(self._keys) - 1)
        self._keys.pop()
        item = self._items.pop()
        if self._keys:
            self._sink(0)
        return item

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._keys:
            raise IndexError("priority queue is empty")
        return self._keys[0]

    def is_empty(self) -> bool:
        return not self._keys

    def __len__(self) -> int:
        return len(self._keys)

    # ----- heap helpers -------------------------------------------------
    def _less(self, i: int, j: int) -> bool:
        return self._keys[i] < self._keys[j]

    def _exch(self, i: int, j: int) -> None:
        self._keys[i], self._keys[j] = self._keys[j], self._keys[i]
        self._items[i], self._items[j] = self._items[j], self._items[i]

    def _swim(self, k: int) -> None:
        """Move the element at index ``k`` up until heap order holds."""
        while k > 0:
            parent = (k - 1) // 2
            if not self._less(k, parent):
                break
            self._exch(k, parent)
            k = parent

    def _sink(self, k: int) -> None:
        """Move the element at index ``k`` down until heap order holds."""
        n = len(self._keys)
        while True:
            child = 2 * k + 1
            if child >= n:
                break
            if child + 1 < n and self._less(child + 1, child):
                child += 1
            if not self._less(child, k):
                break
            self._exch(k, child)
            k = child

<file path="src/models.py">
"""Graph data structures and the subway network model.

``Graph`` is a generic weighted undirected graph whose vertices are integer
station numbers. The numbers are assigned by the caller (``SubwayNetwork``),
not by ``Graph`` itself. ``SubwayNetwork`` maps official station names to
numbers, keeps line information, and offers data update operations.
"""

from __future__ import annotations


class Edge:
    """An undirected connection between two stations on one subway line.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance in meters (also used as the edge weight).
        line: Name of the subway line the connection belongs to.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source: int, dest: int, distance: float, line: str) -> None:
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self) -> float:
        """Edge weight used by path algorithms (equal to the distance)."""
        return self.distance

    @weight.setter
    def weight(self, value: float) -> None:
        self.distance = value

    def other(self, vertex: int) -> int:
        """Return the endpoint of this edge that is not ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"vertex {vertex} is not an endpoint of {self!r}")

    def __repr__(self) -> str:
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Attributes:
        station_lines: Mapping from station name to the list of line names
            the station belongs to (interchange stations have several).
    """

    def __init__(self) -> None:
        self._adj: dict[int, list[Edge]] = {}
        self._edges: list[Edge] = []
        self.station_lines: dict[str, list[str]] = {}

    # ----- vertices -----------------------------------------------------
    def add_vertex(self, vertex: int) -> None:
        """Add a vertex if it does not exist yet."""
        self._adj.setdefault(vertex, [])

    def has_vertex(self, vertex: int) -> bool:
        return vertex in self._adj

    def vertices(self) -> list[int]:
        return list(self._adj.keys())

    def set_station_lines(self, station_name: str, lines) -> None:
        """Record the lines a station belongs to."""
        self.station_lines[station_name] = list(lines)

    # ----- edges --------------------------------------------------------
    def add_edge(self, edge: Edge) -> None:
        """Add an undirected edge (stored in both endpoints' adjacency lists)."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self._adj[edge.dest].append(edge)
        self._edges.append(edge)

    def neighbors(self, vertex: int) -> list[Edge]:
        """Return the edges incident to ``vertex`` (do not modify the list)."""
        return self._adj.get(vertex, [])

    def edges_between(self, u: int, v: int) -> list[Edge]:
        """Return all edges joining ``u`` and ``v``."""
        return [edge for edge in self.neighbors(u) if edge.other(u) == v]

    def edges(self) -> list[Edge]:
        return list(self._edges)

    @property
    def vertex_count(self) -> int:
        return len(self._adj)

    @property
    def edge_count(self) -> int:
        return len(self._edges)


def _clean_name(value, what: str) -> str:
    if not isinstance(value, str) or not value.strip():
        raise ValueError(f"{what} must be a non-empty string")
    return value.strip()


def _validate_distance(distance) -> int:
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("distance must be a number of meters")
    value = int(round(distance))
    if value <= 0:
        raise ValueError("distance must be a positive number of meters")
    return value


class SubwayNetwork:
    """The Beijing Subway network: stations, lines and connections.

    Attributes:
        graph: The underlying weighted undirected ``Graph``.
        station_to_number: Official station name -> station number.
        number_to_station: Station number -> official station name.
    """

    def __init__(self) -> None:
        self.graph = Graph()
        self.station_to_number: dict[str, int] = {}
        self.number_to_station: dict[int, str] = {}
        self._line_order: list[str] = []
        self._next_number = 0

    # ----- queries ------------------------------------------------------
    def has_station(self, name: str) -> bool:
        return name in self.station_to_number

    def station_names(self) -> list[str]:
        return list(self.station_to_number.keys())

    def station_count(self) -> int:
        return len(self.station_to_number)

    def number_of(self, name: str) -> int:
        """Return the station number of ``name`` (KeyError if unknown)."""
        return self.station_to_number[name]

    def name_of(self, number: int) -> str:
        """Return the station name of ``number`` (KeyError if unknown)."""
        return self.number_to_station[number]

    def station_lines(self, name: str) -> list[str]:
        """Return a copy of the line names a station belongs to."""
        return list(self.graph.station_lines.get(name, []))

    def lines(self) -> list[str]:
        """Return all line names in the order they were first seen."""
        return list(self._line_order)

    def stations_on_line(self, line: str) -> list[str]:
        """Return the names of all stations serving ``line``."""
        return [name for name in self.station_names() if line in self.station_lines(name)]

    def transfer_stations(self) -> list[str]:
        """Return stations served by two or more lines."""
        return [name for name in self.station_names() if len(self.station_lines(name)) >= 2]

    # ----- construction and updates ------------------------------------
    def _register_line(self, line: str) -> None:
        if line not in self._line_order:
            self._line_order.append(line)

    def add_station(self, name: str, lines=None) -> int:
        """Add a station (or merge lines into an existing one).

        Returns:
            The station number assigned to ``name``.
        """
        name = _clean_name(name, "station name")
        new_lines = [_clean_name(line, "line name") for line in (lines or [])]
        if name in self.station_to_number:
            number = self.station_to_number[name]
            base = self.station_lines(name)
        else:
            number = self._next_number
            self._next_number += 1
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
            base = []
        merged = list(base)
        for line in new_lines:
            if line not in merged:
                merged.append(line)
        self.graph.set_station_lines(name, merged)
        for line in new_lines:
            self._register_line(line)
        return number

    def update_station_lines(self, name: str, lines) -> None:
        """Replace the line list of an existing station."""
        if name not in self.station_to_number:
            raise ValueError(f"station '{name}' does not exist")
        cleaned: list[str] = []
        for line in lines:
            line = _clean_name(line, "line name")
            if line not in cleaned:
                cleaned.append(line)
            self._register_line(line)
        self.graph.set_station_lines(name, cleaned)

    def rename_station(self, old_name: str, new_name: str) -> None:
        """Rename an existing station, keeping its number and connections."""
        new_name = _clean_name(new_name, "station name")
        if old_name not in self.station_to_number:
            raise ValueError(f"station '{old_name}' does not exist")
        if new_name != old_name and new_name in self.station_to_number:
            raise ValueError(f"station '{new_name}' already exists")
        number = self.station_to_number.pop(old_name)
        lines = self.graph.station_lines.pop(old_name, [])
        self.station_to_number[new_name] = number
        self.number_to_station[number] = new_name
        self.graph.station_lines[new_name] = lines

    def add_connection(self, source: str, dest: str, distance, line: str) -> Edge:
        """Add a connection between two adjacent stations on ``line``.

        Stations that do not exist yet are created and assigned ``line``.
        """
        source = _clean_name(source, "station name")
        dest = _clean_name(dest, "station name")
        line = _clean_name(line, "line name")
        if source == dest:
            raise ValueError("a connection must join two different stations")
        dist = _validate_distance(distance)
        if source not in self.station_to_number:
            self.add_station(source, [line])
        if dest not in self.station_to_number:
            self.add_station(dest, [line])
        edge = Edge(self.number_of(source), self.number_of(dest), dist, line)
        self.graph.add_edge(edge)
        self._register_line(line)
        return edge

    def update_distance(self, source: str, dest: str, distance, line: str = None) -> int:
        """Update the distance of the connection(s) between two stations.

        Returns:
            The number of edges updated.
        Raises:
            ValueError: If no matching connection exists.
        """
        if source not in self.station_to_number or dest not in self.station_to_number:
            raise ValueError("both stations must exist")
        dist = _validate_distance(distance)
        matched = [
            edge
            for edge in self.graph.edges_between(self.number_of(source), self.number_of(dest))
            if line is None or edge.line == line
        ]
        if not matched:
            raise ValueError(f"no connection between '{source}' and '{dest}'")
        for edge in matched:
            edge.distance = dist
        return len(matched)

    def add_line(self, line: str, station_names, distances) -> None:
        """Add a new line from an ordered station sequence.

        Args:
            line: Name of the new line (must not exist yet).
            station_names: Ordered station names along the line.
            distances: Distances in meters between consecutive stations
                (``len(station_names) - 1`` values).
        """
        line = _clean_name(line, "line name")
        if line in self._line_order:
            raise ValueError(f"line '{line}' already exists")
        names = [_clean_name(n, "station name") for n in station_names]
        if len(names) < 2:
            raise ValueError("a line needs at least two stations")
        if len(distances) != len(names) - 1:
            raise ValueError("expected one distance per pair of consecutive stations")
        for name in names:
            self.add_station(name, [line])
        for index in range(len(names) - 1):
            self.add_connection(names[index], names[index + 1], distances[index], line)


def build_network(data: dict) -> SubwayNetwork:
    """Build a ``SubwayNetwork`` from the parsed subway data dictionary."""
    network = SubwayNetwork()
    for station in data.get("stations", []):
        network.add_station(station["name"], station.get("lines", []))
    for conn in data.get("connections", []):
        network.add_connection(conn["source"], conn["dest"], conn["distance"], conn["line"])
    return network

"""Shortest-path computation with Dijkstra's algorithm.

Dijkstra is run over states ``(station_number, line_of_last_edge)``. This lets
the search account for transfers (a change of line at a station) exactly,
which is required by the minimum-transfer and comprehensive modes.

Cost models:
    shortest_distance : minimise total distance, transfers break ties
    least_transfer    : minimise transfers, distance breaks ties
    comprehensive     : distance plus a penalty per transfer
Line preference adds a soft penalty to edges that are not on a preferred line.
"""

from __future__ import annotations

from min_pq import MinPQ
from models import Edge, Graph

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance (meters) charged for each transfer in comprehensive mode.
TRANSFER_PENALTY_METERS = 1000.0
# Equivalent distance (meters) charged for each edge outside preferred lines.
PREFERENCE_PENALTY_METERS = 500.0


class PathNotFoundError(Exception):
    """Raised when no path satisfies the query constraints."""


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of ``VALID_MODES``.
        max_transfers: Maximum number of transfers allowed, or ``None`` for
            no limit.
        preferred_lines: Lines that are preferred (empty list = no preference).
    """

    def __init__(self, mode: str = MODE_COMPREHENSIVE, max_transfers=None,
                 preferred_lines=None) -> None:
        if mode not in VALID_MODES:
            raise ValueError(f"unknown mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer or None")
            if max_transfers < 0:
                raise ValueError("max_transfers must not be negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines: list[str] = list(preferred_lines or [])


class _Label:
    """Best known information about a search state."""

    __slots__ = ("cost", "distance", "transfers", "penalty", "prev", "edge")

    def __init__(self, cost, distance, transfers, penalty, prev, edge) -> None:
        self.cost = cost
        self.distance = distance
        self.transfers = transfers
        self.penalty = penalty
        self.prev = prev
        self.edge = edge


class ShortestPath:
    """Single-source optimal paths over a subway ``Graph``.

    Attributes:
        dist_to: Station number -> real travelled distance in meters of the
            best path found (reachable stations only).
        transfers_to: Station number -> number of transfers on that path.
    """

    def __init__(self, graph: Graph, source: int, config: PathfindingConfig = None) -> None:
        if not graph.has_vertex(source):
            raise ValueError(f"source station {source} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to: dict[int, float] = {}
        self.transfers_to: dict[int, int] = {}
        self._labels: dict[tuple, _Label] = {}
        self._best_state: dict[int, tuple] = {}
        self._run()

    # ----- cost model ---------------------------------------------------
    def _cost(self, distance: float, transfers: int, penalty: float) -> tuple:
        total = distance + penalty
        mode = self.config.mode
        if mode == MODE_SHORTEST_DISTANCE:
            return (total, transfers)
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, total)
        return (total + TRANSFER_PENALTY_METERS * transfers,)

    # ----- Dijkstra -----------------------------------------------------
    def _run(self) -> None:
        preferred = set(self.config.preferred_lines)
        max_transfers = self.config.max_transfers

        start = (self.source, None)
        start_cost = self._cost(0.0, 0, 0.0)
        self._labels[start] = _Label(start_cost, 0.0, 0, 0.0, None, None)

        pq = MinPQ()
        pq.insert(start, start_cost)
        settled: set[tuple] = set()

        while not pq.is_empty():
            state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)
            node, line = state
            label = self._labels[state]

            for edge in self.graph.neighbors(node):
                next_node = edge.other(node)
                next_state = (next_node, edge.line)
                if next_state in settled:
                    continue
                transfers = label.transfers
                if line is not None and edge.line != line:
                    transfers += 1
                if max_transfers is not None and transfers > max_transfers:
                    continue
                distance = label.distance + edge.weight
                penalty = label.penalty
                if preferred and edge.line not in preferred:
                    penalty += PREFERENCE_PENALTY_METERS
                cost = self._cost(distance, transfers, penalty)
                existing = self._labels.get(next_state)
                if existing is None or cost < existing.cost:
                    self._labels[next_state] = _Label(
                        cost, distance, transfers, penalty, state, edge
                    )
                    pq.insert(next_state, cost)

        # Select the best state for every reachable station.
        for state, label in self._labels.items():
            if state not in settled:
                continue
            node = state[0]
            current = self._best_state.get(node)
            if current is None or label.cost < self._labels[current].cost:
                self._best_state[node] = state

        for node, state in self._best_state.items():
            label = self._labels[state]
            self.dist_to[node] = label.distance
            self.transfers_to[node] = label.transfers

    # ----- results ------------------------------------------------------
    def has_path_to(self, target: int) -> bool:
        return target in self._best_state

    def path_with_edges(self, target: int) -> tuple[list[int], list[Edge]]:
        """Return the station numbers and edges of the path source -> target.

        Raises:
            PathNotFoundError: If ``target`` cannot be reached under the
                configured constraints.
        """
        if not self.has_path_to(target):
            raise PathNotFoundError(f"station {target} is unreachable from the source")
        state = self._best_state[target]
        nodes: list[int] = []
        edges: list[Edge] = []
        while True:
            label = self._labels[state]
            nodes.append(state[0])
            if label.prev is None:
                break
            edges.append(label.edge)
            state = label.prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target: int) -> int:
        """Return the number of line transfers on the path to ``target``."""
        if not self.has_path_to(target):
            raise PathNotFoundError(f"station {target} is unreachable from the source")
        return self.transfers_to[target]

"""Beijing Subway fare and travel time estimation.

Fare structure (distance in meters, fractional meters are rounded up):
    0 to < 6 km       : 3 yuan
    6 km to < 12 km   : 4 yuan
    12 km to < 22 km  : 5 yuan
    22 km to < 32 km  : 6 yuan
    >= 32 km          : 7 yuan, plus 1 yuan for every further 20 km
"""

import math

BASE_FARE = 3
FARE_BREAKPOINTS_METERS = (
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
)
EXTENDED_START_METERS = 32000
EXTENDED_START_FARE = 7
EXTENDED_STEP_METERS = 20000

AVERAGE_SPEED_KMH = 35.0
BOARDING_MINUTES = 2.0
TRANSFER_MINUTES = 5.0


class FareCalculationError(ValueError):
    """Raised when a fare or time cannot be calculated from the given input."""


def _normalize_distance(distance) -> int:
    """Validate a distance in meters and round fractional values up."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise FareCalculationError("distance must be a number of meters")
    if math.isnan(distance) or math.isinf(distance):
        raise FareCalculationError("distance must be a finite number")
    if distance < 0:
        raise FareCalculationError("distance must not be negative")
    return math.ceil(distance)


def Sum_money(distance) -> int:
    """Return the subway fare in yuan for a trip of ``distance`` meters.

    Boundary values follow the official tiers: 6000 m costs 4 yuan,
    12000 m costs 5 yuan, 22000 m costs 6 yuan and 32000 m costs 7 yuan.
    """
    meters = _normalize_distance(distance)
    for limit, fare in FARE_BREAKPOINTS_METERS:
        if meters < limit:
            return fare
    return EXTENDED_START_FARE + (meters - EXTENDED_START_METERS) // EXTENDED_STEP_METERS


def fare_tier_name(distance) -> str:
    """Return a human-readable name of the fare tier for ``distance``."""
    meters = _normalize_distance(distance)
    if meters < 6000:
        return "Basic tier (0-6 km)"
    if meters < 12000:
        return "Progressive tier (6-12 km)"
    if meters < 22000:
        return "Progressive tier (12-22 km)"
    if meters < 32000:
        return "Progressive tier (22-32 km)"
    return "Extended tier (over 32 km)"


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimate the travel time in minutes.

    The estimate combines average running speed, a fixed boarding allowance
    and a penalty for each line transfer.

    Args:
        distance: Travel distance in meters.
        transfer_count: Number of line transfers.

    Returns:
        Estimated time in minutes, rounded to one decimal place.
    """
    meters = _normalize_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int):
        raise FareCalculationError("transfer_count must be an integer")
    if transfer_count < 0:
        raise FareCalculationError("transfer_count must not be negative")
    running = (meters / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    minutes = running + BOARDING_MINUTES + transfer_count * TRANSFER_MINUTES
    return round(minutes, 1)

"""Validation and standardisation of user input.

Covers the "Starting Station; Ending Station" query format, station name
standardisation, station existence checks and line preference parsing.
"""

from __future__ import annotations

import difflib
import re
import unicodedata

from models import SubwayNetwork

QUERY_SEPARATOR_PATTERN = r"[;；,，]"
LIST_SEPARATOR_PATTERN = r"[,，;；]"


class ValidationError(ValueError):
    """Base class for invalid user input."""


class InputFormatError(ValidationError):
    """The query does not follow the 'Starting Station; Ending Station' format."""


class StationNotFoundError(ValidationError):
    """A station name does not match any station in the network."""

    def __init__(self, message: str, suggestions=None) -> None:
        super().__init__(message)
        self.suggestions = list(suggestions or [])


def normalize_text(text: str) -> str:
    """Apply Unicode normalisation and collapse whitespace."""
    text = unicodedata.normalize("NFKC", str(text)).strip()
    return re.sub(r"\s+", " ", text)


def parse_query(text: str) -> tuple[str, str]:
    """Split a query such as ``"Xizhimen; Guomen"`` into two station names.

    Raises:
        InputFormatError: If the text does not contain exactly two non-empty
            parts separated by ';' (full-width or ASCII) or ','.
    """
    message = "Input format incorrect, please use 'Starting Station; Ending Station' format"
    if text is None or not str(text).strip():
        raise InputFormatError(message)
    parts = re.split(QUERY_SEPARATOR_PATTERN, str(text))
    if len(parts) != 2:
        raise InputFormatError(message)
    start, end = normalize_text(parts[0]), normalize_text(parts[1])
    if not start or not end:
        raise InputFormatError(message)
    return start, end


def resolve_station(name: str, network: SubwayNetwork, role: str = "Starting") -> str:
    """Map a user-entered name to the official station name.

    Matching order: exact name, case-insensitive name, name with/without the
    trailing character '站' and name ignoring spaces.

    Args:
        name: Name entered by the user.
        network: The subway network.
        role: 'Starting' or 'Ending', used in error messages.

    Raises:
        StationNotFoundError: If no station matches.
    """
    cleaned = normalize_text(name)
    names = network.station_names()
    if cleaned in network.station_to_number:
        return cleaned

    casefold_map = {n.casefold(): n for n in names}
    if cleaned.casefold() in casefold_map:
        return casefold_map[cleaned.casefold()]

    candidates = [cleaned]
    if cleaned.endswith("站"):
        candidates.append(cleaned[:-1].rstrip())
    else:
        candidates.append(cleaned + "站")
    compact = cleaned.replace(" ", "")
    candidates.append(compact)
    for candidate in candidates:
        if candidate and candidate in network.station_to_number:
            return candidate
        if candidate and candidate.casefold() in casefold_map:
            return casefold_map[candidate.casefold()]

    suggestions = difflib.get_close_matches(cleaned, names, n=3, cutoff=0.6)
    message = f"{role} station '{cleaned}' does not exist, please check the station name"
    if suggestions:
        message += f". Did you mean: {', '.join(suggestions)}?"
    raise StationNotFoundError(message, suggestions)


def parse_line_preferences(text, network: SubwayNetwork) -> list[str]:
    """Parse a comma-separated list of line names such as ``"Line 1, Line 2"``.

    Returns:
        The official line names, in the order given, without duplicates.

    Raises:
        ValidationError: If a line name does not exist in the network.
    """
    if text is None:
        return []
    lookup = {line.casefold(): line for line in network.lines()}
    result: list[str] = []
    for part in re.split(LIST_SEPARATOR_PATTERN, str(text)):
        cleaned = normalize_text(part)
        if not cleaned:
            continue
        official = lookup.get(cleaned.casefold())
        if official is None:
            available = ", ".join(network.lines())
            raise ValidationError(
                f"Line '{cleaned}' does not exist, available lines: {available}"
            )
        if official not in result:
            result.append(official)
    return result

"""High-level route query: combines validation, path search and fare calculation."""

from __future__ import annotations

from dataclasses import dataclass, field

from fare_calculator import Sum_money, calculate_estimated_time
from input_validator import ValidationError, resolve_station
from models import Edge, SubwayNetwork
from pathfinding import PathfindingConfig, PathNotFoundError, ShortestPath


@dataclass
class Segment:
    """A continuous ride on one line, between two interchange points."""

    line: str
    stations: list[str] = field(default_factory=list)
    distance_m: int = 0

    @property
    def start(self) -> str:
        return self.stations[0]

    @property
    def end(self) -> str:
        return self.stations[-1]

    @property
    def stop_count(self) -> int:
        return len(self.stations) - 1


@dataclass
class RouteResult:
    """Complete result of a route query."""

    start: str
    end: str
    mode: str
    max_transfers: int | None
    preferred_lines: list[str]
    stations: list[str]
    edges: list[Edge]
    segments: list[Segment]
    distance_m: int
    fare: int
    transfer_count: int
    transfer_stations: list[str]
    lines: list[str]
    estimated_time_min: float

    @property
    def distance_km(self) -> float:
        return self.distance_m / 1000.0


def build_segments(station_names: list[str], edges: list[Edge]) -> list[Segment]:
    """Group a path into per-line segments; each line change is a transfer."""
    segments: list[Segment] = []
    for index, edge in enumerate(edges):
        if not segments or segments[-1].line != edge.line:
            segments.append(Segment(line=edge.line, stations=[station_names[index]]))
        segments[-1].stations.append(station_names[index + 1])
        segments[-1].distance_m += int(edge.distance)
    return segments


def query_route(network: SubwayNetwork, start_name: str, end_name: str,
                config: PathfindingConfig = None) -> RouteResult:
    """Compute the route, fare and summary between two stations.

    Raises:
        StationNotFoundError: If a station name is unknown (subclass of
            ValidationError).
        ValidationError: If start and end are the same station.
        PathNotFoundError: If no route satisfies the constraints.
        FareCalculationError: If the fare cannot be calculated.
    """
    config = config if config is not None else PathfindingConfig()
    start = resolve_station(start_name, network, "Starting")
    end = resolve_station(end_name, network, "Ending")
    if start == end:
        raise ValidationError("Starting station cannot be the same as the ending station")

    source = network.number_of(start)
    target = network.number_of(end)
    search = ShortestPath(network.graph, source, config)
    if not search.has_path_to(target):
        if config.max_transfers is not None:
            raise PathNotFoundError(
                f"No route found from '{start}' to '{end}' within "
                f"{config.max_transfers} transfer(s)"
            )
        raise PathNotFoundError(f"No route found from '{start}' to '{end}'")

    node_numbers, edges = search.path_with_edges(target)
    names = [network.name_of(number) for number in node_numbers]
    segments = build_segments(names, edges)
    distance = sum(int(edge.distance) for edge in edges)
    transfer_count = len(segments) - 1
    transfer_stations = [segment.end for segment in segments[:-1]]
    lines: list[str] = []
    for segment in segments:
        if segment.line not in lines:
            lines.append(segment.line)

    fare = Sum_money(distance)
    estimated = calculate_estimated_time(distance, transfer_count)

    return RouteResult(
        start=start,
        end=end,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        stations=names,
        edges=edges,
        segments=segments,
        distance_m=distance,
        fare=fare,
        transfer_count=transfer_count,
        transfer_stations=transfer_stations,
        lines=lines,
        estimated_time_min=estimated,
    )

"""Rendering of route results as console text or structured JSON."""

from __future__ import annotations

import json

from route_service import RouteResult

SEPARATOR = "=" * 60

def format_text_report(result: RouteResult) -> str:
    """Render a route result as a human-readable console report."""
    max_text = "unlimited" if result.max_transfers is None else str(result.max_transfers)
    preference = ", ".join(result.preferred_lines) if result.preferred_lines else "none"
    lines = [
        SEPARATOR,
        f"Route: {result.start} -> {result.end}",
        f"Mode: {result.mode} | Max transfers: {max_text} | Line preference: {preference}",
        RULE,
        f"Station sequence ({len(result.stations)} stations):",
    ]
    for index, name in enumerate(result.stations, start=1):
        lines.append(f"  {index:>3}. {name}")

    lines.append("")
    lines.append("Line segments:")
    for segment in result.segments:
        lines.append(
            f"  [{segment.line}] {segment.start} -> {segment.end} "
            f"({segment.stop_count} stops, {segment.distance_m} m)"
        )

    lines.append("")
    lines.append("Transfer details:")
    if result.transfer_count == 0:
        lines.append("  No transfer required")
    else:
        for index, station in enumerate(result.transfer_stations):
            before = result.segments[index].line
            after = result.segments[index + 1].line
            lines.append(f"  Transfer at {station}: {before} -> {after}")
    lines.append(f"Lines involved: {', '.join(result.lines)}")

    lines.append(RULE)
    lines.append(
        f"Total distance: {result.distance_m} m ({result.distance_km:.2f} km)"
    )
    lines.append(f"Transfers: {result.transfer_count}")
    lines.append(f"Estimated time: {result.estimated_time_min:.1f} min")
    lines.append(f"Fare: {result.fare} yuan")
    lines.append(SEPARATOR)
    return "\n".join(lines)


def route_to_dict(result: RouteResult) -> dict:
    """Convert a route result into a JSON-serialisable dictionary."""
    return {
        "query": {
            "start": result.start,
            "end": result.end,
            "mode": result.mode,
            "max_transfers": result.max_transfers,
            "line_preference": result.preferred_lines,
        },
        "path": result.stations,
        "distance_m": result.distance_m,
        "distance_km": round(result.distance_km, 2),
        "fare": result.fare,
        "estimated_time_min": result.estimated_time_min,
        "transfer_count": result.transfer_count,
        "transfer_stations": result.transfer_stations,
        "lines": result.lines,
        "segments": [
            {
                "line": segment.line,
                "start": segment.start,
                "end": segment.end,
                "stations": segment.stations,
                "distance_m": segment.distance_m,
            }
            for segment in result.segments
        ],
    }

    return json.dumps([route_to_dict(result) for result in results],
                      ensure_ascii=False, indent=2)

    if output_format == "json":
        return render_json(results)
    return "\n\n".join(format_text_report(result) for result in results) + "\n"

"""Data integrity checks and data maintenance helpers."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

from data_loader import save_subway_data
from models import SubwayNetwork

LEVEL_ERROR = "ERROR"
LEVEL_WARNING = "WARNING"

CATEGORY_COVERAGE = "coverage"
CATEGORY_CONNECTION = "connection"
CATEGORY_DISTANCE = "distance"

MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

# Plausible range of inter-station distances in meters.
MIN_REASONABLE_DISTANCE = 200
MAX_REASONABLE_DISTANCE = 5000


@dataclass(frozen=True)
class CheckIssue:
    """A single finding of the integrity check."""

    level: str
    category: str
    message: str

    def __str__(self) -> str:
        return f"[{self.level}] ({self.category}) {self.message}"


def check_station_coverage(network: SubwayNetwork) -> list[CheckIssue]:
    """Verify that main lines and stations are represented in the data."""
    issues: list[CheckIssue] = []
    if network.station_count() == 0:
        issues.append(CheckIssue(LEVEL_ERROR, CATEGORY_COVERAGE, "No stations are loaded"))
        return issues

    available = {line.casefold(): line for line in network.lines()}
    for line in MAIN_LINES:
        official = available.get(line.casefold())
        if official is None:
            issues.append(CheckIssue(
                LEVEL_WARNING, CATEGORY_COVERAGE,
                f"Main line '{line}' is not present in the network data"))
            continue
        count = len(network.stations_on_line(official))
        if count < 2:
            issues.append(CheckIssue(
                LEVEL_WARNING, CATEGORY_COVERAGE,
                f"Main line '{line}' has only {count} station(s)"))

    for name in network.station_names():
        if not network.station_lines(name):
            issues.append(CheckIssue(
                LEVEL_WARNING, CATEGORY_COVERAGE,
                f"Station '{name}' has no line assignment"))
    return issues


def _connected_components(network: SubwayNetwork) -> list[list[int]]:
    graph = network.graph
    seen: set[int] = set()
    components: list[list[int]] = []
    for start in graph.vertices():
        if start in seen:
            continue
        component = []
        queue = deque([start])
        seen.add(start)
        while queue:
            node = queue.popleft()
            component.append(node)
            for edge in graph.neighbors(node):
                other = edge.other(node)
                if other not in seen:
                    seen.add(other)
                    queue.append(other)
        components.append(component)
    return components


def check_connections(network: SubwayNetwork) -> list[CheckIssue]:
    """Verify connection relationships between stations."""
    issues: list[CheckIssue] = []
    graph = network.graph

    for name in network.station_names():
        if not graph.neighbors(network.number_of(name)):
            issues.append(CheckIssue(
                LEVEL_WARNING, CATEGORY_CONNECTION,
                f"Station '{name}' has no connections"))

    seen: set[tuple] = set()
    for edge in graph.edges():
        src = network.name_of(edge.source)
        dst = network.name_of(edge.dest)
        for name in (src, dst):
            if edge.line not in network.station_lines(name):
                issues.append(CheckIssue(
                    LEVEL_WARNING, CATEGORY_CONNECTION,
                    f"Connection {src}-{dst} is on line '{edge.line}', but station "
                    f"'{name}' is not registered on that line"))
        key = (min(edge.source, edge.dest), max(edge.source, edge.dest), edge.line)
        if key in seen:
            issues.append(CheckIssue(
                LEVEL_WARNING, CATEGORY_CONNECTION,
                f"Duplicate connection {src}-{dst} on line '{edge.line}'"))
        seen.add(key)

    components = _connected_components(network)
    if len(components) > 1:
        sizes = sorted((len(c) for c in components), reverse=True)
        smallest = min(components, key=len)
        sample = [network.name_of(n) for n in smallest[:5]]
        issues.append(CheckIssue(
            LEVEL_WARNING, CATEGORY_CONNECTION,
            f"Network has {len(components)} disconnected groups (sizes {sizes[:5]}); "
            f"smallest group includes {sample}"))
    return issues


def check_distances(network: SubwayNetwork) -> list[CheckIssue]:
    """Verify that inter-station distances are positive and plausible."""
    issues: list[CheckIssue] = []
    for edge in network.graph.edges():
        src = network.name_of(edge.source)
        dst = network.name_of(edge.dest)
        if edge.distance <= 0:
            issues.append(CheckIssue(
                LEVEL_ERROR, CATEGORY_DISTANCE,
                f"Connection {src}-{dst} has non-positive distance {edge.distance}"))
        elif not MIN_REASONABLE_DISTANCE <= edge.distance <= MAX_REASONABLE_DISTANCE:
            issues.append(CheckIssue(
                LEVEL_WARNING, CATEGORY_DISTANCE,
                f"Connection {src}-{dst} on '{edge.line}' has suspicious distance "
                f"{edge.distance} m (expected {MIN_REASONABLE_DISTANCE}-"
                f"{MAX_REASONABLE_DISTANCE} m)"))
    return issues


def run_integrity_check(network: SubwayNetwork) -> list[CheckIssue]:
    """Run all integrity checks and return the combined findings."""
    return (check_station_coverage(network)
            + check_connections(network)
            + check_distances(network))


def has_errors(issues: list[CheckIssue]) -> bool:
    return any(issue.level == LEVEL_ERROR for issue in issues)


def format_check_report(issues: list[CheckIssue]) -> str:
    """Render integrity check findings as text."""
    if not issues:
        return "Data integrity check passed: no issues found."
    errors = sum(1 for issue in issues if issue.level == LEVEL_ERROR)
    warnings = len(issues) - errors
    lines = ["Data integrity check results:"]
    lines.extend(f"  {issue}" for issue in issues)
    lines.append(f"Summary: {errors} error(s), {warnings} warning(s)")
    return "\n".join(lines)


def export_network_dict(network: SubwayNetwork) -> dict:
    """Convert the network back into the subway data file structure."""
    stations = [
        {"name": name, "lines": network.station_lines(name)}
        for name in network.station_names()
    ]
    connections = []
    for edge in network.graph.edges():
        connections.append({
            "source": network.name_of(edge.source),
            "dest": network.name_of(edge.dest),
            "distance": int(edge.distance),
            "line": edge.line,
        })
    return {"stations": stations, "connections": connections}


def save_network(network: SubwayNetwork, path) -> None:
    """Persist the (possibly updated) network to a JSON data file."""
    save_subway_data(export_network_dict(network), path)

def _write_output_file(results, path: str, output_format: str) -> None:
    content = render_results(results, output_format)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(content)


def run_interactive(network: SubwayNetwork, config: PathfindingConfig,
                    output_format: str = "text", output_file: str = None,
                    input_func=input, out=print) -> int:
    """Run the continuous query loop until the user quits.

    Each successful query is printed and, if ``output_file`` is given, the
    file is rewritten with all results of the session so far.
    """
    results = []
    out("Beijing Subway fare calculator. Type 'help' for instructions.")
    while True:
        try:
            raw = input_func("Query> ")
        except (EOFError, KeyboardInterrupt):
            out("")
            break
        text = raw.strip()
        if not text:
            continue
        lowered = text.lower()
        if lowered in EXIT_WORD_LIST:
            break
        if lowered in HELP_WORD_LIST:
            out(HELP_TEXT)
            continue

        try:
            start, end = parse_query(text)
            result = query_route(network, start, end, config)
        except ValidationError as exc:
            out(f"Error: {exc}")
            continue
        except PathNotFoundError as exc:
            out(f"Error: {exc}")
            continue
        except FareCalculationError as exc:
            out(f"Fare calculation error: {exc}")
            continue
        except Exception as exc:  # last-resort guard so the session keeps running
            out(f"Unexpected error while calculating the route: {exc}")
            continue

        results.append(result)
        if output_format == "json":
            out(render_json([result]))
        else:
            out(format_text_report(result))

        if output_file:
            try:
                _write_output_file(results, output_file, output_format)
                out(f"Results saved to {output_file}")
            except OSError as exc:
                out(f"Error: could not write output file '{output_file}': {exc}")
    return 0


def main(argv=None) -> int:
    """Program entry point. Returns the process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    try:
        raw_data = load_subway_data(DATA_FILE_PATH)
        network = build_network(raw_data)
    except (OSError, ValueError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        issues = run_integrity_check(network)
        print(format_check_report(issues))
        return 1 if has_errors(issues) else 0

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must not be negative", file=sys.stderr)
        return 2

    try:
        preferred = parse_line_preferences(args.line_preference, network)
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=preferred)
    except (ValidationError, ValueError) as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    return run_interactive(network, config, args.output_format, args.output_file)

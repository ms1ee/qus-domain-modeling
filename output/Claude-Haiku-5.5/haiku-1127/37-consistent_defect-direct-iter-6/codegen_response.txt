"""Participant list loading, validation and group extraction."""

from __future__ import annotations

import re
from collections.abc import Iterable

from lottery_system.default_data import DEFAULT_USER_LINES
from lottery_system.models import User

_SEPARATOR_PATTERN = re.compile(r"[,;，；]")


class ListManager:
    """Loads and validates the participant list."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_USER_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list in 'Name,Department' format."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read a txt file and parse it into users and error messages."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: {path}"]
        except UnicodeDecodeError:
            return [], [f"File Error: '{path}' is not a valid UTF-8 text file"]
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: Iterable[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Blank lines are skipped."""
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = _SEPARATOR_PATTERN.split(text)
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error at Line {line_no}: '{text}' "
                    "(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid unique users and duplicate occurrences.

        The first occurrence of a name is kept as valid; later occurrences
        are reported as duplicates.
        """
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return the number of people in each department group."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from collections.abc import Iterable

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the prize list and the lottery round order."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: Iterable[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Add a prize to the end of the round order.

        Raises ValueError on invalid input.
        """
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if any(p.name == clean_name for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer else ""
        self._prizes.append(
            Prize(
                name=clean_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer or None,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return copies of the prizes in current round order."""
        return [
            Prize(
                name=p.name,
                quantity=p.quantity,
                avoid_groups=list(p.avoid_groups),
                associated_drawer=p.associated_drawer,
            )
            for p in self._prizes
        ]

    def set_avoid_groups(self, name: str, groups: Iterable[str]) -> bool:
        """Replace the avoidance groups of the named prize."""
        prize = self._find(name)
        if prize is None:
            return False
        prize.avoid_groups = [g.strip() for g in groups if g and g.strip()]
        return True

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order holds 1-based current positions in the new order."""
        count = len(self._prizes)
        if len(order) != count:
            return False
        if sorted(order) != list(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

    def _find(self, name: str) -> Prize | None:
        for prize in self._prizes:
            if prize.name == name:
                return prize
        return None

"""Lottery rule engine: eligibility filtering and drawing."""

from __future__ import annotations

import random
from collections.abc import Iterable

from lottery_system.constants import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User


class LotteryManager:
    """Runs individual draws while enforcing the lottery rules."""

    def __init__(self, users: Iterable[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._rng = rng if rng is not None else random.Random()
        self._winner_names: set[str] = set()
        # Cheng Ru group flags of the most recent draws by Chen Qixian / Jin Duo.
        # True means the winner belonged to the Cheng Ru group.
        self.last_two_draw_results: list[bool] = []

    @property
    def users(self) -> list[User]:
        return list(self._users)

    def mark_as_winner(self, user: User) -> None:
        """Record a person as already having won (e.g. from earlier rounds)."""
        self._winner_names.add(user.name)

    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def _candidates(self) -> list[User]:
        """All participants who have not won yet."""
        return [u for u in self._users if u.name not in self._winner_names]

    def _cheng_ru_protection_active(self) -> bool:
        """True when the last two Chen/Jin draws both had no Cheng Ru member."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the pool the next winner of this prize is drawn from.

        - Previous winners are always excluded.
        - Cheng Ru protection (Ma Jiaqi): if protection is active, the pool is
          limited to Cheng Ru group members, even if that group is an avoidance group.
        - Li Bin: only previous winners are filtered; avoidance groups are not applied.
        - Otherwise, members of avoidance groups stay eligible but are only drawn
          after all other members are exhausted.
        """
        candidates = self._candidates()
        if not candidates:
            return []

        drawer = prize.associated_drawer
        if drawer == MA_JIAQI and self._cheng_ru_protection_active():
            forced = [u for u in candidates if u.department == CHENG_RU_GROUP]
            if forced:
                return forced

        if drawer == LI_BIN:
            return candidates

        avoid = set(prize.avoid_groups)
        primary = [u for u in candidates if u.department not in avoid]
        if primary:
            return primary
        return candidates

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if nobody is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_names.add(winner.name)
        self._update_protection_state(prize, winner)
        return winner

    def _update_protection_state(self, prize: Prize, winner: User) -> None:
        drawer = prize.associated_drawer
        if drawer in (CHEN_QIXIAN, JIN_DUO):
            self.last_two_draw_results.append(winner.department == CHENG_RU_GROUP)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == MA_JIAQI:
            # The Ma Jiaqi round consumes the protection window.
            self.last_two_draw_results = []

"""Winner recording, grouping by prize level, and CSV export."""

from __future__ import annotations

import csv
import io
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User

_UNRANKED_LEVEL = 99
_CSV_HEADER = ["Name", "Department", "Prize Name", "Winning Time"]


@dataclass
class Winner:
    """A recorded winning result."""

    user: User
    prize_name: str
    sequence: int
    won_at: datetime


def prize_level_rank(name: str) -> int:
    """Return the level rank of a prize (First=1, Second=2, Third=3)."""
    lowered = name.lower()
    for rank, keyword in enumerate(("first", "second", "third"), start=1):
        if keyword in lowered:
            return rank
    return _UNRANKED_LEVEL


class ResultManager:
    """Keeps the winner list and produces summaries and exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []

    def add_winner(self, user: User, prize: Prize | str) -> Winner:
        """Record a winner in winning order and return the record."""
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        winner = Winner(
            user=user,
            prize_name=prize_name,
            sequence=len(self._winners) + 1,
            won_at=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered from high to low prize level.

        Within each prize, winners are in winning time order.
        """
        ordered = sorted(prizes, key=lambda p: prize_level_rank(p.name))
        result: dict[str, list[Winner]] = {}
        for prize in ordered:
            result[prize.name] = [w for w in self._winners if w.prize_name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        buffer = io.StringIO()
        writer = csv.writer(buffer, lineterminator="\n")
        writer.writerow(_CSV_HEADER)
        grouped = self.get_winners_by_prize(prizes)
        for prize_name, winners in grouped.items():
            for winner in winners:
                writer.writerow(
                    [
                        winner.user.name,
                        winner.user.department,
                        prize_name,
                        f"{winner.won_at:%Y-%m-%d %H:%M:%S}",
                    ]
                )
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                handle.write(buffer.getvalue())
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

from __future__ import annotations

from collections.abc import Callable

SEPARATOR = "-" * 60


def ordinal(number: int) -> str:
    """Return the English ordinal form of a positive integer."""
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def describe_rules(prize: Prize) -> str:
    """Build a human-readable rule description for a prize round."""
    rules = ["previous winners are excluded"]
    drawer = prize.associated_drawer
    if drawer == LI_BIN:
        rules.append("special rule for Li Bin: avoidance groups are not applied")
    elif prize.avoid_groups:
        rules.append(
            "avoidance groups "
            + ", ".join(prize.avoid_groups)
            + " are drawn after other members"
        )
    if drawer == MA_JIAQI:
        rules.append(
            "Cheng Ru Group protection may limit the pool to Cheng Ru Group members"
        )
    if drawer:
        rules.append(f"associated lottery person: {drawer}")
    return "; ".join(rules)


class LotterySystemCLI:
    """Interactive console application."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.completed_prizes: set[str] = set()

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    def _ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    def _choose(self, prompt: str, options: set[str]) -> str:
        while True:
            value = self._ask(prompt).lower()
            if value in options:
                return value
            self._print("Invalid input, please try again.")

    def _ensure_lottery(self) -> LotteryManager:
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)
            for winner in self.result_manager.get_all_winners():
                self.lottery.mark_as_winner(winner.user)
        return self.lottery

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    def _print_menu(self) -> None:
        self._print("\n========== Team Building Smart Lottery ==========")
        self._print("1. Load participant list")
        self._print("2. View department groups")
        self._print("3. Configure prizes")
        self._print("4. Start lottery")
        self._print("5. View winners")
        self._print("6. Export winners (CSV txt)")
        self._print("0. Exit")

    def run(self) -> None:
        while True:
            self._print_menu()
            choice = self._ask("Select an option: ")
            if choice == "1":
                self.load_list_menu()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.prize_menu()
            elif choice == "4":
                self.run_lottery()
            elif choice == "5":
                self.show_results()
            elif choice == "6":
                self.export_results()
            elif choice == "0":
                self._print("Goodbye!")
                break
            else:
                self._print("Invalid option, please try again.")

    # ------------------------------------------------------------------
    # List management
    # ------------------------------------------------------------------
    def load_list_menu(self) -> None:
        self._print("\n[Load Participant List]")
        self._print("1. Built-in default list")
        self._print("2. Custom txt file (format: Name,Department per line)")
        self._print("0. Cancel")
        choice = self._ask("Select: ")
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ")
            if not path:
                self._print("Path is empty. Upload cancelled.")
                return
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

        if errors:
            self._print(f"Found {len(errors)} problematic line(s):")
            for error in errors:
                self._print(f"  {error}")
            action = self._choose(
                "Enter 'r' to remove invalid data and continue, "
                "or 't' to terminate upload: ",
                {"r", "t"},
            )
            if action == "t":
                self._print("Upload terminated. Current list unchanged.")
                return

        valid_users, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self._print(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                self._print(f"  Duplicate: {user.name} ({user.department})")
            action = self._choose(
                "Enter 'r' to remove duplicates and continue, "
                "or 't' to terminate upload: ",

                self._print("Upload terminated. Current list unchanged.")
                return

        if not valid_users:
            self._print("No valid participants found. List not loaded.")
            return

        self.users = valid_users
        self.lottery = None
        self._print(f"Loaded {len(valid_users)} participant(s).")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured yet.")
            return
        self._print("\n[Prize Order]")
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) or "None"
            drawer = prize.associated_drawer or "None"
            self._print(
                f"{index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoidance groups: {groups} | Associated person: {drawer}"

    def _load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            for error in errors:
                self._print(f"  Skipped: {error}")
        else:
            self._print("Preset prizes loaded.")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        quantity_text = self._ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._print("Quantity must be an integer. Prize not added.")
            return
        groups_text = self._ask(
            "Avoidance groups (comma-separated, empty for none): "
        )
        groups = [g.strip() for g in groups_text.split(",") if g.strip()]
        drawer = self._ask("Associated lottery person (empty for none): ") or None
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            self._print(f"Failed to add prize: {exc}")
            return
        self._print(f"Prize '{name.strip()}' added.")

    def _set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("No prizes configured yet.")
            return
        self.show_prizes()
        text = self._ask("Enter prize serial number: ")
        try:
            index = int(text)
        except ValueError:
            self._print("Invalid serial number.")
            return
        if not 1 <= index <= len(prizes):
            self._print("Serial number out of range.")
            return
        groups_text = self._ask(
            "Avoidance groups (comma-separated, empty to clear): "
        )
        groups = [g.strip() for g in groups_text.split(",") if g.strip()]
        if self.prize_manager.set_avoid_groups(prizes[index - 1].name, groups):
            self._print("Avoidance groups updated.")
        else:
            self._print("Failed to update avoidance groups.")

    def _reorder_prizes(self) -> None:
        if not self.prize_manager.get_prizes():
            self._print("No prizes configured yet.")
            return
        self.show_prizes()
        text = self._ask(
            "Enter new order as serial numbers separated by spaces (e.g. 2 1 3): "
        )
        try:
            order = [int(part) for part in text.split()]
        except ValueError:
            self._print("Order must contain only serial numbers.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._print("Prize order updated.")
        else:
            self._print("Invalid order. Each serial number must appear exactly once.")

    # ------------------------------------------------------------------
    # Lottery process
    # ------------------------------------------------------------------
    def run_lottery(self) -> None:
        if not self.users:
            self._print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._print("Please configure prizes first.")
            return
        pending = [p for p in prizes if p.name not in self.completed_prizes]
        if not pending:
            self._print("All lottery rounds have already been completed.")
            return
        if self._ask("Start the lottery now? (y/n): ").lower() not in ("y", "yes"):
            return

        lottery = self._ensure_lottery()
        self._print(f"Lottery started with {len(prizes)} round(s).")
        for round_no, prize in enumerate(prizes, start=1):
            if prize.name in self.completed_prizes:
                continue
            self._run_round(round_no, prize, lottery)
            self.completed_prizes.add(prize.name)
        self._print("\nAll lottery rounds finished. Use option 5 to view winners.")

    def _run_round(self, round_no: int, prize: Prize, lottery: LotteryManager) -> None:
        self._print(SEPARATOR)
        self._print(f"Round {round_no}: {prize.name}")
        self._print(f"Quantity: {prize.quantity}")
        self._print(f"Rules: {describe_rules(prize)}")
        self._print(SEPARATOR)
        self._print(
            "Commands: Enter = draw next, 'pause' = pause, 'terminate' = skip round"
        )

        round_winners: list[Winner] = []
        drawn = 0
        paused = False
        while drawn < prize.quantity:
            if paused:
                command = self._ask(
                    "[PAUSED] Type 'continue' to resume or 'terminate' to skip: "
                ).lower()
                if command == "continue":
                    paused = False
                    self._print("Lottery resumed.")
                elif command == "terminate":
                    self._print("Current round terminated.")
                    break
                else:
                    self._print("Unknown command.")
                continue

            command = self._ask("Draw next winner (Enter / pause / terminate): ").lower()
            if command == "pause":
                paused = True
                self._print("Lottery paused.")
                continue
            if command == "terminate":
                self._print("Current round terminated.")
                break
            if command not in ("", "continue"):
                self._print("Unknown command.")
                continue

            self._print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
            winner_user = lottery.draw(prize)
            if winner_user is None:
                self._print("No eligible candidates remain for this round.")
                break
            drawn += 1
            winner = self.result_manager.add_winner(winner_user, prize)
            round_winners.append(winner)
            self._print(
                f"Congratulations {winner_user.name} ({winner_user.department}) "
                f"for winning {prize.name}"
            )

        self._print(f"\nRound {round_no} finished: {drawn}/{prize.quantity} awarded.")
        if round_winners:
            self._print("Round results (winning order):")
            for winner in round_winners:
                self._print(
                    f"  {winner.sequence}. {winner.user.name} "
                    f"({winner.user.department})"
                )

    # ------------------------------------------------------------------
    # Results
    # ------------------------------------------------------------------
    def show_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            self._print("No winners yet.")
            return
        self._print("\n[Winner List]")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            self._print(f"\n{prize_name}")
            for winner in winners:
                self._print(
                    f"  {winner.sequence}. {winner.user.name} "
                    f"({winner.user.department})"
                )

    def export_results(self) -> None:
        prizes = self.prize_manager.get_prizes()
        _, message = (None, None)
        success, message = self.result_manager.export_to_csv(prizes)
        self._print(message if success else f"Error: {message}")


def main() -> None:
    """Entry point for the console application."""
    cli = LotterySystemCLI()
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting lottery system.")

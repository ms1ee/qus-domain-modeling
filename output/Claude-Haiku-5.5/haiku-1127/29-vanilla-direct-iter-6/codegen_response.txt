"""Global variable store for the food delivery dispatch simulator.

This module keeps a single process-wide dictionary of system-level values
such as system funds, the rider list, the rider ID counter, the running
flag and the simulated tick counter. Other modules access these values via
``get_value`` and ``set_value``.
"""

from typing import Any

_GLOBALS: dict = {}


def _init() -> None:
    """Reset all global variables to their default values.

    Defaults:
        funds: 1000 (used for rider recruitment and operating costs).
        riders: empty list of Rider objects.
        rider_id_counter: 0 (riders receive IDs 1, 2, 3, ...).
        running: True (controls the main execution loop).
        tick: 0 (simulated time, one unit per tick).
        orders: empty list of Order objects (full order history).
        grid_size: 800 (map is 800 x 800 units).
        cell_size: 100 (each grid cell is 100 x 100 units).
    """
    _GLOBALS.clear()
    _GLOBALS.update(
        {
            "funds": 1000,
            "riders": [],
            "rider_id_counter": 0,
            "running": True,
            "tick": 0,
            "orders": [],
            "grid_size": 800,
            "cell_size": 100,
        }
    )


def set_value(key: str, value: Any) -> None:
    """Store a value under the given global key.

    Args:
        key: Name of the global variable (e.g. ``"funds"``, ``"riders"``).
        value: Value to store.
    """
    _GLOBALS[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` if the key is missing.
    """
    return _GLOBALS.get(key, default)

"""Grid-based path planning for riders.

The map is an 800x800 unit plane divided into an 8x8 grid of 100x100 cells.
Lines whose x or y coordinate is a multiple of 100 are streets. Restaurant
and customer locations are cell centers (e.g. 150, 250, ...). Movement is
axis-aligned (horizontal or vertical) and prefers following streets.
"""

GRID_SIZE = 800
CELL_SIZE = 100
GRID_DIM = GRID_SIZE // CELL_SIZE


def is_on_street(value: int) -> bool:
    """Return True if a coordinate lies on a grid street line.

    Args:
        value: One coordinate (x or y) in units.

    Returns:
        True when the value is a multiple of the cell size.
    """
    return value % CELL_SIZE == 0


def cell_index(value: int) -> int:
    """Return the grid index (0-7) containing a coordinate.

    Values at or beyond the map edge are clamped to the last cell.

    Args:
        value: One coordinate in units.

    Returns:
        Integer cell index between 0 and GRID_DIM - 1.
    """
    return min(max(int(value), 0) // CELL_SIZE, GRID_DIM - 1)


def cell_center(value: int) -> int:
    """Convert a coordinate to the center of its grid cell.

    Implements ``center = (grid // 100) * 100 + 50`` with clamping so that
    the map edge value 800 maps to the center of the last cell (750).

    Args:
        value: One coordinate in units (0-800).

    Returns:
        The center coordinate of the containing cell.
    """
    return cell_index(value) * CELL_SIZE + CELL_SIZE // 2


def _sign(value: int) -> int:
    """Return -1, 0 or 1 according to the sign of ``value``."""
    if value > 0:
        return 1
    if value < 0:
        return -1
    return 0


class PathPlanner:
    """Computes routes and single-unit movement steps on the street grid.

    Rules:
        * Movement is horizontal or vertical only (right-angle turns).
        * A rider that stands on a street line (x multiple of 100) but
          not on a horizontal street moves along that street first.
        * Otherwise horizontal movement is preferred first.
        * When the rider is directly above or below the target, it moves
          straight vertically (special case).
    """

    def __init__(self, grid_size: int = GRID_SIZE, cell_size: int = CELL_SIZE) -> None:
        """Create a planner for a grid of the given size.

        Args:
            grid_size: Map width/height in units.
            cell_size: Size of one grid cell in units.
        """
        self.grid_size = grid_size
        self.cell_size = cell_size

    def in_bounds(self, x: int, y: int) -> bool:
        """Return True if the point lies inside the map (0..grid_size)."""
        return 0 <= x <= self.grid_size and 0 <= y <= self.grid_size

    def route(self, start: tuple, end: tuple) -> list:
        """Return a right-angle waypoint list from ``start`` to ``end``.

        The corner is chosen so that the first segment follows a street when
        possible.

        Args:
            start: (x, y) starting point.
            end: (x, y) destination point.

        Returns:
            List of (x, y) waypoints beginning with ``start`` and ending with
            ``end`` (consecutive duplicates removed).
        """
        sx, sy = start
        ex, ey = end
        if sx != ex and sy != ey and is_on_street(sx) and not is_on_street(sy):
            corner = (sx, ey)
        else:
            corner = (ex, sy)
        points = [(sx, sy)]
        for point in (corner, (ex, ey)):
            if point != points[-1]:
                points.append(point)
        return points

    @staticmethod
    def route_length(points: list) -> int:
        """Return the total Manhattan length of an axis-aligned route.

        Args:
            points: List of (x, y) waypoints.

        Returns:
            Total travelled distance in units.
        """
        total = 0
        for (ax, ay), (bx, by) in zip(points, points[1:]):
            total += abs(bx - ax) + abs(by - ay)
        return total

    @staticmethod
    def next_step(x: int, y: int, tx: int, ty: int) -> tuple:
        """Return the position after moving exactly one unit toward a target.

        Args:
            x: Current x coordinate.
            y: Current y coordinate.
            tx: Target x coordinate.
            ty: Target y coordinate.

        Returns:
            (x, y) after one unit of axis-aligned movement. Returns the input
            position unchanged when already at the target.
        """
        if x == tx and y == ty:
            return x, y
        if x == tx:
            # Directly above or below the target: move straight vertically.
            return x, y + _sign(ty - y)
        if y == ty:
            return x + _sign(tx - x), y
        if is_on_street(x) and not is_on_street(y):
            return x, y + _sign(ty - y)
        return x + _sign(tx - x), y

"""Order data model.

An order is created from user input in the form
``[restaurant_x, restaurant_y, customer_x, customer_y]`` with coordinates in
the range 0-800. Coordinates are converted to grid cell centers.
"""

import itertools
import logging

from globalvar import get_value
from path_planner import PathPlanner, cell_center

logger = logging.getLogger(__name__)

COORD_MIN = 0
COORD_MAX = 800

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def _current_tick() -> int:
    """Return the current simulated tick (0 if unavailable)."""
    value = get_value("tick")
    return value if isinstance(value, int) else 0


def _validate_point(point, label: str) -> tuple:
    """Validate a raw (x, y) input pair.

    Args:
        point: Sequence of two integers.
        label: Name used in error messages (e.g. "restaurant").

    Returns:
        A tuple (x, y) of validated integers.

    Raises:
        ValueError: If the point is malformed, not integer or out of range.
    """
    try:
        x, y = point
    except (TypeError, ValueError):
        raise ValueError(f"{label} location must contain exactly 2 values")
    for value in (x, y):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} coordinates must be integers")
        if not COORD_MIN <= value <= COORD_MAX:
            raise ValueError(
                f"{label} coordinates must be in range {COORD_MIN}-{COORD_MAX}"
            )
    return int(x), int(y)


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier (auto-incremented).
        res_input: Raw restaurant coordinates as entered by the user.
        cus_input: Raw customer coordinates as entered by the user.
        res_pos: Restaurant grid cell center (x, y).
        cus_pos: Customer grid cell center (x, y).
        status: One of "pending", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: ID of the assigned rider, or None.
        start_tick: Tick at which the order was created.
        pickup_tick: Tick at which the rider reached the restaurant.
        completion_tick: Tick at which the order was delivered.
        distance: Route length from restaurant to customer in units.
    """

    _counter = itertools.count(1)

    def __init__(self, res_pos, cus_pos) -> None:
        """Create an order and convert coordinates to grid cell centers.

        Args:
            res_pos: Restaurant (x, y) in range 0-800.
            cus_pos: Customer (x, y) in range 0-800.

        Raises:
            ValueError: If coordinates are malformed, non-integer or out of
                range.
        """
        self.res_input = _validate_point(res_pos, "restaurant")
        self.cus_input = _validate_point(cus_pos, "customer")
        self.id = next(Order._counter)
        self.res_pos = (cell_center(self.res_input[0]), cell_center(self.res_input[1]))
        self.cus_pos = (cell_center(self.cus_input[0]), cell_center(self.cus_input[1]))
        self.status = STATUS_PENDING
        self.rider_id = None
        self.start_tick = _current_tick()
        self.pickup_tick = None
        self.completion_tick = None
        planner = PathPlanner()
        self.distance = planner.route_length(planner.route(self.res_pos, self.cus_pos))
        logger.info(
            "Order %d created: restaurant=%s customer=%s",
            self.id,
            self.res_pos,
            self.cus_pos,
        )

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_picked_up(self, tick: int) -> None:
        """Record arrival at the restaurant and switch to delivering.

        Args:
            tick: Current simulated tick.
        """
        self.pickup_tick = tick
        self.status = STATUS_DELIVERING

    def complete(self, tick: int) -> None:
        """Mark the order as delivered.

        Args:
            tick: Current simulated tick.
        """
        self.completion_tick = tick
        self.status = STATUS_COMPLETED
        logger.info("Order %d completed at tick %d", self.id, tick)

    @property
    def time_taken(self):
        """Ticks from creation to completion, or None if not completed."""
        if self.completion_tick is None:
            return None
        return self.completion_tick - self.start_tick

    @property
    def efficiency(self):
        """Delivery efficiency in units per tick, or None if not completed.

        Returns None when the order is incomplete or completed in zero ticks.
        """
        taken = self.time_taken
        if not taken:
            return None
        return self.distance / taken

    def __repr__(self) -> str:
        return f"Order(id={self.id}, status={self.status}, rider={self.rider_id})"

"""Rider model: state machine, movement and order handling.

Rider states:
    0 (IDLE): available for new orders.
    1 (PICKING_UP): en route to the restaurant.
    2 (DELIVERING): en route to the customer.

Orders are assigned by appending them to ``rider.orders``. The state
transitions and their log entries happen inside ``move()``.
"""

import logging

from globalvar import get_value
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

DEFAULT_RIDER_X = 435
DEFAULT_RIDER_Y = 385
DEFAULT_SPEED = 10


def _current_tick() -> int:
    """Return the current simulated tick (0 if unavailable)."""
    value = get_value("tick")
    return value if isinstance(value, int) else 0


class Rider:
    """A delivery rider that moves on the street grid and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates in units.
        speed: Distance covered per tick (units/tick).
        state: Current status (0 idle, 1 picking up, 2 delivering).
        orders: List of assigned Order objects (first one is active).
        special_case: Flag set when the rider is directly above/below its
            target or is waiting for a higher-priority rider to pass.
        log: List of event dictionaries with keys "event", "order_id",
            "tick", "x", "y", "state".
    """

    def __init__(self, rider_id: int) -> None:
        """Create a rider at the default starting position.

        Args:
            rider_id: Unique identifier of the rider.
        """
        self.rider_id = rider_id
        self.x = DEFAULT_RIDER_X
        self.y = DEFAULT_RIDER_Y
        self.speed = DEFAULT_SPEED
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.planner = PathPlanner()
        self.log_event("spawn", None)

    @property
    def position(self) -> tuple:
        """Current (x, y) position."""
        return (self.x, self.y)

    def current_order(self):
        """Return the order currently being served, or None."""
        return self.orders[0] if self.orders else None

    def log_event(self, event: str, order_id):
        """Append a status-change record to the rider log.

        Args:
            event: Event name ("spawn", "accepted_order",
                "arrived_at_merchant" or "delivered_to_customer").
            order_id: Related order identifier, or None.

        Returns:
            The created log entry dictionary.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "tick": _current_tick(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %d: %s order=%s", self.rider_id, event, order_id)
        return entry

    def move(self) -> None:
        """Advance the rider by one tick.

        Handles idle → picking up → delivering → idle transitions, moves the
        rider up to ``speed`` units along the grid and logs key events.
        """
        order = self.current_order()
        if order is None:
            # Consistency check: no orders means the rider must be idle.
            self.state = IDLE
            self.special_case = False
            return

        if self.state == IDLE:
            self._start_pickup(order)

        if self.state == PICKING_UP:
            self._advance(order.res_pos)
            if self.position == order.res_pos:
                self.state = DELIVERING
                order.mark_picked_up(_current_tick())
                self.log_event("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            self._advance(order.cus_pos)
            if self.position == order.cus_pos:
                order.complete(_current_tick())
                self.orders.remove(order)
                self.log_event("delivered_to_customer", order.id)
                next_order = self.current_order()
                if next_order is None:
                    self.state = IDLE
                    self.special_case = False
                else:
                    self._start_pickup(next_order)
        else:
            # Unknown state: recover to a consistent state.
            logger.warning("Rider %d had invalid state %s", self.rider_id, self.state)
            self.state = PICKING_UP

    def _start_pickup(self, order) -> None:
        """Begin serving an order: switch to picking up and log acceptance."""
        self.state = PICKING_UP
        order.status = "picking_up"
        self.log_event("accepted_order", order.id)

    def _advance(self, target: tuple) -> None:
        """Move up to ``speed`` unit steps toward ``target``.

        Deadlock/conflict avoidance: a rider yields when the next cell is
        occupied by a rider with a lower ID. Lower IDs never yield, so the
        scheme cannot deadlock.

        Args:
            target: (x, y) destination point.
        """
        tx, ty = target
        self.special_case = self.x == tx and self.y != ty
        budget = self.speed
        while budget > 0 and self.position != (tx, ty):
            nx, ny = self.planner.next_step(self.x, self.y, tx, ty)
            if self._blocked(nx, ny):
                self.special_case = True
                break
            self.x, self.y = nx, ny
            budget -= 1

    def _blocked(self, x: int, y: int) -> bool:
        """Return True if a higher-priority rider occupies (x, y)."""
        for other in get_value("riders") or []:
            if other is self:
                continue
            if other.rider_id < self.rider_id and other.position == (x, y):
                return True
        return False

    def __repr__(self) -> str:
        return (
            f"Rider(id={self.rider_id}, pos=({self.x}, {self.y}), "
            f"state={STATE_NAMES.get(self.state, self.state)}, orders={len(self.orders)})"
        )

"""Order assignment logic.

Implements the closest-rider-first assignment strategy based on Euclidean
distance, with a per-rider capacity limit that balances load among riders.
"""

import logging
import math

from globalvar import get_value
from order_model import Order, STATUS_PENDING
from rider import Rider

logger = logging.getLogger(__name__)

MAX_ORDERS_PER_RIDER = 3


def _euclidean(rider: Rider, point: tuple) -> float:
    """Return the Euclidean distance between a rider and a point."""
    return math.sqrt((rider.x - point[0]) ** 2 + (rider.y - point[1]) ** 2)


def find_best_rider_for_order(order: Order) -> Rider:
    """Choose the best rider for an order.

    The closest rider (Euclidean distance to the restaurant) is preferred.
    Riders already holding ``MAX_ORDERS_PER_RIDER`` orders are excluded. Ties
    are broken by fewer assigned orders and then by lower rider ID.

    Args:
        order: The order to assign.

    Returns:
        The selected Rider, or None if no rider has spare capacity.
    """
    riders = get_value("riders") or []
    candidates = [r for r in riders if len(r.orders) < MAX_ORDERS_PER_RIDER]
    if not candidates:
        return None
    return min(
        candidates,
        key=lambda r: (
            _euclidean(r, order.res_pos),
            len(r.orders),
            r.rider_id,
        ),
    )


class OrderManager:
    """Creates, assigns and tracks orders."""

    @staticmethod
    def create_order(res_pos, cus_pos) -> Order:
        """Create an order and register it in the global order history.

        Args:
            res_pos: Restaurant (x, y) in range 0-800.
            cus_pos: Customer (x, y) in range 0-800.

        Returns:
            The created Order.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        order = Order(res_pos, cus_pos)
        get_value("orders").append(order)
        return order

    @staticmethod
    def pending_orders() -> list:
        """Return all orders that have not yet been assigned to a rider."""
        return [o for o in get_value("orders") or [] if o.status == STATUS_PENDING]

    @staticmethod
    def assign(order: Order):
        """Assign an order to the best available rider.

        Args:
            order: The order to assign.

        Returns:
            The rider that received the order, or None if none is available.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            logger.info("No rider available for order %d", order.id)
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        logger.info("Order %d assigned to rider %d", order.id, rider.rider_id)
        return rider

"""System management: funds, rider recruitment, order dispatch and ticks."""

import logging
import math

from globalvar import _init, get_value, set_value
from order import OrderManager
from path_planner import PathPlanner
from order_model import STATUS_COMPLETED
from rider import IDLE, Rider

logger = logging.getLogger(__name__)

INITIAL_FUNDS = 1000
RECRUIT_THRESHOLD = 300
RECRUIT_COST = 300


def init_system() -> None:
    """Initialize all core parameters of the system.

    Sets funds to 1000, an empty rider list, rider ID counter to 0, the
    running flag to True, the tick counter to 0 and an empty order history.
    """
    _init()
    set_value("funds", INITIAL_FUNDS)
    set_value("riders", [])
    set_value("rider_id_counter", 0)
    set_value("running", True)
    set_value("tick", 0)
    set_value("orders", [])
    logger.info("System initialised with funds=%d", INITIAL_FUNDS)


class SystemManager:
    """Core coordinator for funds, riders, orders and simulated time."""

    def __init__(self) -> None:
        """Create the manager; expects ``init_system`` to have been called."""
        self.order_manager = OrderManager()
        self.planner = PathPlanner()

    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return get_value("funds")

    @property
    def riders(self) -> list:
        """List of all riders."""
        return get_value("riders")

    @property
    def tick_count(self) -> int:
        """Current simulated tick."""
        return get_value("tick")

    @property
    def running(self) -> bool:
        """Whether the main loop should keep running."""
        return bool(get_value("running"))

    def shutdown(self) -> None:
        """Request the main loop to stop."""
        set_value("running", False)
        logger.info("System shutdown requested")

    def rider_by_id(self, rider_id: int):
        """Return the rider with the given ID, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def idle_riders(self) -> list:
        """Return riders that are idle and hold no orders."""
        return [r for r in self.riders if r.state == IDLE and not r.orders]

    def create_rider(self) -> Rider:
        """Create a new rider, register it and deduct the recruitment cost.

        Returns:
            The newly created Rider.
        """
        rider_id = get_value("rider_id_counter") + 1
        set_value("rider_id_counter", rider_id)
        rider = Rider(rider_id)
        self.riders.append(rider)
        set_value("funds", self.funds - RECRUIT_COST)
        logger.info("Recruited rider %d (funds now %d)", rider_id, self.funds)
        return rider

    def try_recruit(self) -> list:
        """Recruit riders while funds allow and pending orders exceed idle riders.

        Returns:
            List of riders created by this call (possibly empty).
        """
        created = []
        while (
            self.funds >= RECRUIT_THRESHOLD
            and len(self.order_manager.pending_orders()) > len(self.idle_riders())
        ):
            created.append(self.create_rider())
        return created

    def _dispatch_pending(self) -> None:
        """Try to recruit and assign all pending orders in arrival order."""
        pending = self.order_manager.pending_orders()
        if not pending:
            return
        self.try_recruit()
        for order in pending:
            if self.order_manager.assign(order) is None:
                break

    def submit_order(self, restaurant_pos, customer_pos) -> dict:
        """Create an order, recruit if needed and assign it.

        Args:
            restaurant_pos: Restaurant (x, y) in range 0-800.
            customer_pos: Customer (x, y) in range 0-800.

        Returns:
            Dictionary with keys:
                "order": the created Order,
                "new_riders": list of riders recruited for this order,
                "rider": assigned Rider or None,
                "route": waypoint list from rider to restaurant to customer,
                "distance": total route length in units,
                "eta": estimated ticks to delivery (direct route estimate).

        Raises:
            ValueError: If the coordinates are invalid.
        """
        before = len(self.riders)
        order = self.order_manager.create_order(restaurant_pos, customer_pos)
        self._dispatch_pending()
        new_riders = list(self.riders[before:])
        rider = self.rider_by_id(order.rider_id) if order.rider_id is not None else None

        route = []
        distance = 0
        eta = None
        if rider is not None:
            first = self.planner.route(rider.position, order.res_pos)
            second = self.planner.route(order.res_pos, order.cus_pos)
            route = first + second[1:]
            distance = self.planner.route_length(route)
            eta = math.ceil(distance / rider.speed) if rider.speed > 0 else None
        return {
            "order": order,
            "new_riders": new_riders,
            "rider": rider,
            "route": route,
            "distance": distance,
            "eta": eta,
        }

    def _completed_count(self) -> int:
        """Return the number of completed orders in history."""
        return sum(1 for o in get_value("orders") or [] if o.status == STATUS_COMPLETED)

    def advance(self, ticks: int = 1) -> int:
        """Advance the simulation by a number of ticks.

        Each tick: increment the clock, dispatch pending orders and move every
        rider. Exceptions from a single rider are logged and do not stop the
        simulation (error recovery).

        Args:
            ticks: Number of ticks to simulate.

        Returns:
            Number of orders completed during these ticks.
        """
        completed_before = self._completed_count()
        for _ in range(max(0, int(ticks))):
            set_value("tick", self.tick_count + 1)

"""JSON persistence for system state and configuration files."""

import json
import logging
import os

logger = logging.getLogger(__name__)


def save_state(path: str, system) -> None:
    """Write the current system state to a JSON file.

    Args:
        path: Destination file path.
        system: SystemManager instance whose ``to_dict`` is serialized.

    Raises:
        OSError: If the file cannot be written.
    """
    data = system.to_dict()
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2)
    logger.info("System state saved to %s", path)


def load_config(path: str) -> dict:
    """Load system parameters from a JSON configuration file.

    Missing or invalid files produce an empty configuration (with a warning)
    so that the program can still start with defaults.

    Args:
        path: Path to the configuration file.

    Returns:
        Dictionary of configuration values (empty if unavailable).
    """
    if not os.path.exists(path):
        return {}
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read configuration %s: %s", path, exc)
        return {}
    if not isinstance(data, dict):
        logger.warning("Configuration %s must contain a JSON object", path)
        return {}
    return data

"""Status display and ASCII map visualization."""

from globalvar import get_value
from order_model import STATUS_COMPLETED
from path_planner import GRID_DIM, cell_index
from rider import STATE_NAMES

SYMBOL_RIDER = "R"
SYMBOL_RESTAURANT = "S"
SYMBOL_CUSTOMER = "C"
SYMBOL_EMPTY = "."
SYMBOL_STREET = "#"


class DisplayManager:
    """Produces text output for status, orders, rider logs and the map."""

    def render_map(self, riders, orders, route=None) -> str:
        """Render the 8x8 ASCII map.

        Symbols: 'R' rider, 'S' restaurant, 'C' customer, '#' route/street,
        '.' empty. Priority when cells overlap: R > S > C > # > '.'.

        Args:
            riders: Iterable of Rider objects.
            orders: Iterable of Order objects (completed orders are hidden).
            route: Optional list of (x, y) waypoints drawn as a line.

        Returns:
            Multi-line string with the map and a legend.
        """
        grid = [[SYMBOL_EMPTY] * GRID_DIM for _ in range(GRID_DIM)]

        if route:
            for (ax, ay), (bx, by) in zip(route, route[1:]):
                steps = max(abs(bx - ax), abs(by - ay))
                if steps == 0:
                    self._set(grid, ax, ay, SYMBOL_STREET, only_empty=True)
                    continue
                for i in range(steps + 1):
                    px = ax + (bx - ax) * i // steps
                    py = ay + (by - ay) * i // steps
                    self._set(grid, px, py, SYMBOL_STREET, only_empty=True)

        for order in orders:
            if order.status == STATUS_COMPLETED:
                continue
            self._set(grid, order.cus_pos[0], order.cus_pos[1], SYMBOL_CUSTOMER)
            self._set(grid, order.res_pos[0], order.res_pos[1], SYMBOL_RESTAURANT)

        for rider in riders:
            self._set(grid, rider.x, rider.y, SYMBOL_RIDER)

        lines = [" " * 11 + " ".join(str(c) for c in range(GRID_DIM))]
        for row in reversed(range(GRID_DIM)):
            start = row * 100
            label = f"y{start:03d}-{start + 100:03d} |"
            lines.append(label + " " + " ".join(grid[row]))
        lines.append("x cells: 0..7 (each cell 100 units)")
        lines.append(
            "Legend: R=Rider S=Restaurant C=Customer #=Street/Route .=Empty"
        )
        return "\n".join(lines)

    @staticmethod
    def _set(grid, x, y, symbol, only_empty=False) -> None:
        """Place a symbol in the grid cell containing (x, y)."""
        col = cell_index(x)
        row = cell_index(y)
        if only_empty and grid[row][col] != SYMBOL_EMPTY:
            return
        grid[row][col] = symbol

    @staticmethod
    def format_status(system) -> str:
        """Return a snapshot of funds, riders and their orders.

        Args:
            system: SystemManager instance.

        Returns:
            Multi-line status text.
        """
        lines = [
            f"Tick: {system.tick_count}",
            f"System funds: {system.funds}",
            f"Riders: {len(system.riders)}",
        ]
        for rider in system.riders:
            order_text = ", ".join(
                f"#{o.id}({o.status})" for o in rider.orders
            ) or "none"
            lines.append(
                f"  Rider #{rider.rider_id}: position=({rider.x}, {rider.y}) "
                f"state={STATE_NAMES.get(rider.state, rider.state)} "
                f"orders={len(rider.orders)} [{order_text}]"
            )
        return "\n".join(lines)

    @staticmethod
    def format_orders(orders) -> str:
        """Return a table of all orders.

        Args:
            orders: Iterable of Order objects.

        Returns:
            Multi-line table text.
        """
        orders = list(orders)
        if not orders:
            return "No orders yet."
        lines = [
            "ID   Start  Restaurant   Customer     Distance  Taken  Rider  Status"
        ]
        for o in orders:
            taken = o.time_taken if o.time_taken is not None else "-"
            rider = f"#{o.rider_id}" if o.rider_id is not None else "-"
            lines.append(
                f"{o.id:<4} {o.start_tick:<6} {str(o.res_pos):<12} "
                f"{str(o.cus_pos):<12} {o.distance:<9} {str(taken):<6} "
                f"{rider:<6} {o.status}"
            )
        return "\n".join(lines)

    @staticmethod
    def format_rider_log(riders) -> str:
        """Return every recorded status change for all riders.

        Args:
            riders: Iterable of Rider objects.

        Returns:
            Multi-line text, grouped by rider.
        """
        riders = list(riders)
        if not riders:
            return "No riders yet."
        lines = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id}:")
            for entry in rider.log:
                order_text = (
                    f"order #{entry['order_id']}" if entry["order_id"] is not None else "-"
                )
                lines.append(
                    f"  t={entry['tick']:<4} {entry['event']:<22} {order_text:<12} "
                    f"pos=({entry['x']}, {entry['y']}) "
                    f"state={STATE_NAMES.get(entry['state'], entry['state'])}"
                )
        return "\n".join(lines)

    @staticmethod
    def format_route(route) -> str:
        """Return a readable representation of a waypoint list."""
        if not route:
            return "-"
        return " -> ".join(f"({x}, {y})" for x, y in route)

"""Command-line parsing, validation and command dispatch."""

import logging

from globalvar import get_value
from order_model import COORD_MAX, COORD_MIN
from persistence import save_state
from rider import STATE_NAMES

try:  # Command history and line editing where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    readline = None

logger = logging.getLogger(__name__)

DEFAULT_SAVE_PATH = "dispatch_state.json"

WELCOME = (
    "Food Delivery Dispatch Simulator\n"
    "Type 'help' for commands. Shortcuts: s=start st=status o=orders "
    "r=riders m=map t=tick h=help q=quit"
)

HELP_TEXT = """Commands:
  start [rx ry cx cy]  (s)   Submit an order: restaurant then customer (0-800)
  status               (st)  Show funds, riders, states and order lists
  orders               (o)   Show all orders with distance and time taken
  riders               (r)   Show every rider status change log
  map                  (m)   Show the ASCII map with riders and orders
  tick [n]             (t)   Advance the simulation by n ticks (default 1)
  save [path]                Save system state to JSON
  history                    Show command history
  help                 (h)   Show this help
  quit / exit          (q)   Leave the program"""

COMMANDS = {
    "start": "start", "s": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "save": "save",
    "history": "history",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}


class InputManager:
    """Parses user commands and dispatches them to the system and display."""

    def __init__(self, system, display, input_func=input, output_func=print) -> None:
        """Create the input manager.

        Args:
            system: SystemManager instance.
            display: DisplayManager instance.
            input_func: Callable used to read a line (injectable for tests).
            output_func: Callable used to print output.
        """
        self.system = system
        self.display = display
        self._input = input_func
        self._print = output_func
        self.history = []
        self._handlers = {
            "start": self._cmd_start,
            "status": self._cmd_status,
            "orders": self._cmd_orders,
            "riders": self._cmd_riders,
            "map": self._cmd_map,
            "tick": self._cmd_tick,
            "save": self._cmd_save,
            "history": self._cmd_history,
            "help": self._cmd_help,
            "quit": self._cmd_quit,
        }

    @staticmethod
    def parse_command(line: str):
        """Split a raw input line into a canonical command and arguments.

        Args:
            line: Raw user input.

        Returns:
            Tuple (command, args). Command is an empty string for blank input
            and the lowercased token for unknown commands.
        """
        tokens = line.strip().split()
        if not tokens:
            return "", []
        word = tokens[0].lower()
        return COMMANDS.get(word, word), tokens[1:]

    @staticmethod
    def parse_coordinates(tokens) -> list:
        """Validate four integer coordinates in the range 0-800.

        Args:
            tokens: Sequence of string tokens.

        Returns:
            List of four integers [rx, ry, cx, cy].

        Raises:
            ValueError: If the count, type or range is invalid.
        """
        if len(tokens) != 4:
            raise ValueError(
                "expected 4 integers: restaurant_x restaurant_y customer_x customer_y"
            )
        values = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise ValueError(f"'{token}' is not an integer")
            if not COORD_MIN <= value <= COORD_MAX:
                raise ValueError(
                    f"coordinate {value} is out of range {COORD_MIN}-{COORD_MAX}"
                )
            values.append(value)
        return values

    def run(self) -> None:
        """Run the interactive main loop until the system stops."""
        self._print(WELCOME)
        while self.system.running:
            try:
                line = self._input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self._print("\nExiting.")
                break
            if not line or not line.strip():
                continue
            self.history.append(line.strip())
            try:
                self.execute(line)
            except Exception as exc:  # error recovery: keep the session alive
                logger.exception("Command failed: %s", line)
                self._print(f"Error: {exc}. Please try again.")
        self.system.shutdown()

    def execute(self, line: str) -> None:
        """Execute a single command line.

        Args:
            line: Raw user input.
        """
        command, args = self.parse_command(line)
        if not command:
            return
        handler = self._handlers.get(command)
        if handler is None:
            self._print(f"Unknown command '{command}'. Type 'help' for options.")
            return
        handler(args)

    def _cmd_start(self, args) -> None:
        """Handle the start command (order input)."""
        tokens = list(args)
        if not tokens:
            self._print("Enter: restaurant_x restaurant_y customer_x customer_y")
            tokens = self._input("order> ").split()
        try:
            values = self.parse_coordinates(tokens)
        except ValueError as exc:
            self._print(f"Invalid input: {exc}")
            return

        res = (values[0], values[1])
        cus = (values[2], values[3])
        try:
            result = self.system.submit_order(res, cus)
        except ValueError as exc:
            self._print(f"Invalid order: {exc}")
            return

        order = result["order"]
        self._print(f"Order #{order.id} created at tick {order.start_tick}.")
        self._print(f"  Restaurant input {res} -> grid center {order.res_pos}")
        self._print(f"  Customer input {cus} -> grid center {order.cus_pos}")
        if result["new_riders"]:
            ids = ", ".join(f"#{r.rider_id}" for r in result["new_riders"])
            self._print(
                f"  New rider added: {ids} (funds now {self.system.funds})"
            )
        else:
            self._print("  No new rider added.")

        rider = result["rider"]
        if rider is None:
            self._print("  No rider available; the order remains pending.")
            return

        self._print(
            f"  Assigned to rider #{rider.rider_id} "
            f"(speed {rider.speed} units/tick)"
        )
        self._print(
            f"  Route distance {result['distance']} units, "
            f"estimated time {result['eta']} ticks"
        )
        self._print("  Route: " + self.display.format_route(result["route"]))
        self._print(
            self.display.render_map(
                get_value("riders") or [], [order], route=result["route"]
            )
        )

    def _cmd_status(self, args) -> None:
        """Handle the status command."""
        self._print(self.display.format_status(self.system))

    def _cmd_orders(self, args) -> None:
        """Handle the orders command."""
        self._print(self.display.format_orders(get_value("orders") or []))

    def _cmd_riders(self, args) -> None:
        """Handle the riders command."""
        self._print(self.display.format_rider_log(get_value("riders") or []))

    def _cmd_map(self, args) -> None:
        """Handle the map command."""
        self._print(
            self.display.render_map(
                get_value("riders") or [], get_value("orders") or []
            )
        )

    def _cmd_tick(self, args) -> None:
        """Handle the tick command: advance the simulation."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                self._print(f"'{args[0]}' is not an integer")
                return
            if count < 1:
                self._print("Tick count must be at least 1")
                return
        completed = self.system.advance(count)
        self._print(
            f"Advanced {count} tick(s). Current tick: {self.system.tick_count}. "
            f"Orders completed: {completed}."
        )

    def _cmd_save(self, args) -> None:
        """Handle the save command."""
        path = args[0] if args else DEFAULT_SAVE_PATH
        try:
            save_state(path, self.system)
        except OSError as exc:
            self._print(f"Could not save to {path}: {exc}")
            return
        self._print(f"State saved to {path}")

    def _cmd_history(self, args) -> None:
        """Handle the history command."""
        if not self.history:
            self._print("No commands yet.")
            return
        for index, entry in enumerate(self.history, start=1):
            self._print(f"{index:>4}  {entry}")

    def _cmd_help(self, args) -> None:
        """Handle the help command."""
        self._print(HELP_TEXT)

    def _cmd_quit(self, args) -> None:
        """Handle the quit/exit command."""
        self.system.shutdown()
        self._print("Goodbye.")

"""Entry point for the food delivery dispatch simulator.

Usage:
    python src/main.py [--config config.json] [--log-level WARNING]
"""

import argparse
import logging
import sys

from display_manager import DisplayManager
from input_manager import InputManager
from persistence import load_config
from globalvar import set_value
from system import SystemManager, init_system


def parse_args(argv=None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: Optional list of arguments (defaults to sys.argv).

    Returns:
        Parsed argument namespace.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument(
        "--config", default="config.json", help="JSON configuration file path"
    )
    parser.add_argument(
        "--log-level",
        default="WARNING",
        choices=["DEBUG", "INFO", "WARNING", "ERROR"],
        help="Logging verbosity",
    )
    return parser.parse_args(argv)


def main(argv=None) -> int:
    """Start the simulator.

    Args:
        argv: Optional list of command-line arguments.

    Returns:
        Process exit code (0 on normal exit).
    """
    args = parse_args(argv)
    logging.basicConfig(
        level=getattr(logging, args.log_level),
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    init_system()
    config = load_config(args.config)
    if "funds" in config:
        try:
            set_value("funds", int(config["funds"]))
        except (TypeError, ValueError):
            logging.getLogger(__name__).warning("Invalid 'funds' in configuration")

    system = SystemManager()
    display = DisplayManager()
    InputManager(system, display).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

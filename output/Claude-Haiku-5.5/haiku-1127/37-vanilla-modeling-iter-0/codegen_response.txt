"""Domain models and shared constants for the lottery system."""
from __future__ import annotations

import re
from dataclasses import dataclass, field

CHENG_RU_GROUP = "Cheng Ru group"

LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"

LOTTERY_PERSONS = [LI_BIN, CHEN_QIXIAN, JIN_DUO, MA_JIAQI]

UNRANKED_LEVEL = 99

_LEVEL_WORDS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}

_LEVEL_PATTERN = re.compile(
    r"^\s*(first|second|third|fourth|fifth|sixth|seventh|eighth|ninth|tenth)\s+prize\b",
    re.IGNORECASE,
)


@dataclass(frozen=True)
class User:
    """A participant of the team building event."""

    name: str
    department: str


@dataclass
class Prize:
    """A prize that is drawn in one lottery round.

    avoid_groups: department groups that must not win this prize.
    associated_drawer: the lottery person bound to this prize (or None).
    """

    name: str
    quantity: int
    avoid_groups: list[str] = field(default_factory=list)
    associated_drawer: str | None = None

    @property
    def level(self) -> int:
        """Prize level derived from the name prefix (1 = highest).

        Names such as "First Prize - ..." yield 1, "Second Prize - ..." yield 2, etc.
        Prizes without a recognisable level get the lowest rank.
        """
        match = _LEVEL_PATTERN.match(self.name)
        if match:
            return _LEVEL_WORDS[match.group(1).lower()]
        return UNRANKED_LEVEL

DEFAULT_LIST_LINES = [

    "Wang Wu,Product Department",

    "Liu San,Product Department",

    "Zheng Shi,Finance Department",
    "Feng Yi,Human Resources Department",

class ListManager:
    """Loads and validates participant lists."""

    def __init__(self):
        self.load_mode: str | None = None
        self.source_file_path: str | None = None
        self.is_valid: bool = False

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in default team building list."""
        self.load_mode = "default"
        self.source_file_path = None
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.is_valid = bool(users) and not errors
        return users, errors

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list (each line: "Name,Department Group")."""
        self.load_mode = "custom"
        self.source_file_path = path
        users, errors = self._load_from_path(path)
        self.is_valid = bool(users) and not errors
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a txt file. Format errors are reported per line."""
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                content = handle.read()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(content.splitlines())

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_number, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_number}: Format Error - '{line}' "
                    "(expected 'Name,Department Group')"
                )
                continue
            users.append(User(name=parts[0].strip(), department=parts[1].strip()))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        """Split users into (valid_users, duplicates).

        The first occurrence of a name is kept as valid; later occurrences
        of the same name are returned as duplicates.
        """
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        """Return a mapping of department group name to member count."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.models import Prize

PRESET_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Li Bin"),
    ("Second Prize - Shopping Cards", 3, [], "Chen Qixian"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
    ("Technology Department Exclusive Coupon", 2, ["Product Department"], "Jin Duo"),
]


class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Load the preset prize library. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, groups, drawer in PRESET_PRIZES:
            try:
                self.add_prize(name, quantity, groups, drawer)
            except ValueError as exc:
                errors.append(f"Prize '{name}': {exc}")
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        """Add a prize at the end of the current sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("prize name must not be empty")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("quantity must be a positive integer")
        if any(p.name == clean_name for p in self._prizes):
            raise ValueError(f"prize '{clean_name}' already exists")
        groups = list(dict.fromkeys(avoid_groups or []))
        drawer = associated_drawer.strip() if associated_drawer else None
        drawer = drawer or None
        prize = Prize(name=clean_name, quantity=quantity, avoid_groups=groups,
                      associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery sequence."""
        return list(self._prizes)

    def set_avoid_groups(self, prize_name: str, groups: list[str]) -> bool:
        """Replace the avoidance groups of the named prize."""
        for prize in self._prizes:
            if prize.name == prize_name:
                prize.avoid_groups = list(dict.fromkeys(groups or []))
                return True
        return False

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order holds 1-based current positions in the new order."""
        try:
            positions = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        total = len(self._prizes)
        if sorted(positions) != list(range(1, total + 1)):
            return False
        self._prizes = [self._prizes[p - 1] for p in positions]
        return True

"""Lottery rule engine: eligibility filtering, special rules and fallback."""
from __future__ import annotations

import random

from lottery_system.models import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
    Prize,
    User,
)


class LotteryManager:
    """Draws winners while enforcing the lottery rules."""

    def __init__(self, users):
        self.users: list[User] = list(users)
        self.winner_names: set[str] = set()
        # Outcomes (True = winner belongs to Cheng Ru group) of the most recent
        # draws by Chen Qixian / Jin Duo, keeping at most two entries.
        self.last_two_draw_results: list[bool] = []

    def is_winner(self, user: User) -> bool:
        return user.name in self.winner_names

    def is_fallback_active(self) -> bool:
        """Fallback triggers when the last two results both contain no Cheng Ru member."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        pool = [u for u in self.users if u.name not in self.winner_names]
        drawer = prize.associated_drawer
        if drawer == LI_BIN:
            # Special rule: only previous winners are filtered.
            return pool
        if drawer == MA_JIAQI and self.is_fallback_active():
            # Fallback: force Cheng Ru group members, overriding avoidance groups.
            return [u for u in pool if u.department == CHENG_RU_GROUP]
        avoid = set(prize.avoid_groups)
        return [u for u in pool if u.department not in avoid]

    def get_eligible_count(self, prize: Prize) -> int:
        return len(self._get_eligible_pool(prize))

    def get_rule_description(self, prize: Prize) -> str:
        drawer = prize.associated_drawer
        if drawer == LI_BIN:
            return ("Previous winners are excluded; avoidance groups are NOT applied "
                    "(special rule for Li Bin).")
        if drawer == MA_JIAQI and self.is_fallback_active():
            return ("Fallback ACTIVE: only Cheng Ru group members are eligible "
                    "(previous winners excluded).")
        parts = ["Previous winners are excluded"]
        if prize.avoid_groups:
            parts.append("avoidance groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("no avoidance groups")
        if drawer == MA_JIAQI:
            parts.append("fallback: if the last two Chen Qixian/Jin Duo draws both had no "
                         "Cheng Ru group member, only Cheng Ru group members are eligible")
        return "; ".join(parts) + "."

    def draw_winner(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self.winner_names.add(winner.name)
        self._record_draw(prize, winner)
        return winner

    def _record_draw(self, prize: Prize, winner: User) -> None:
        if prize.associated_drawer in (CHEN_QIXIAN, JIN_DUO):
            self.last_two_draw_results.append(winner.department == CHENG_RU_GROUP)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def end_round(self, prize: Prize) -> None:
        """Called when a round finishes; resets the fallback state after Ma Jiaqi's round."""
        if prize.associated_drawer == MA_JIAQI:
            self.last_two_draw_results.clear()

"""Winner recording, grouping and CSV-style export."""
from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize_name: str
    prize_level: int
    winning_order: int
    winning_time: datetime


class ResultManager:
    """Stores all winning records of the event."""

    def __init__(self):
        self._winners: list[Winner] = []
        self._order_counter: int = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        self._order_counter += 1
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=prize.level,
            winning_order=self._order_counter,
            winning_time=datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low).

        Within each prize, winners are ordered by winning time/order.
        """
        result: dict[str, list[Winner]] = {}
        for prize in sorted(prizes, key=lambda p: p.level):
            winners = [w for w in self._winners if w.prize_name == prize.name]
            winners.sort(key=lambda w: w.winning_order)
            result[prize.name] = winners
        return result

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt in CSV format."""
        grouped = self.get_winners_by_prize(prizes)
        rows = [w for winners in grouped.values() for w in winners]
        if not rows:
            return False, "No winners to export."

        now = datetime.now()
        file_path = os.path.join(os.getcwd(), f"awards_result_{now:%Y%m%d}.txt")
        try:
            with open(file_path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for w in rows:
                    writer.writerow([
                        w.user.name,
                        w.user.department,
                        w.prize_name,
                        w.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {file_path}"

from __future__ import annotations

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager

from lottery_system.services.result_manager import ResultManager, Winner

SEPARATOR = "=" * 64


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Interactive console application controlling the whole lottery process."""

    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.groups: dict[str, int] = {}

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt).strip()

    # ---------------------------------------------------------------- main flow
    def run(self) -> None:
        print(SEPARATOR)
        print("Company Team Building Event - Smart Lottery System")
        print(SEPARATOR)
        users = self._setup_participants()
        if users is None:
            print("Goodbye!")
            return
        self._setup_prizes()
        print("Goodbye!")

    # ------------------------------------------------------ participant list
    def _setup_participants(self):
        while True:
            print("\n--- Participant List ---")
            print("1. Load built-in default list")
            print("2. Upload custom txt list (format: Name,Department Group per line)")
            print("0. Exit")
            choice = self._ask("Select an option: ")
            if choice == "0":
                return None
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
                source = "built-in default list"
            elif choice == "2":
                path = self._ask("Enter the txt file path: ")
                users, errors = self.list_manager.load_custom_list(path)
                source = path
            else:
                print("Invalid option.")
                continue

            valid, duplicates = self.list_manager.validate_users(users)
            if errors or duplicates:
                print("\nIssues found in the list:")
                for error in errors:
                    print(f"  - {error}")
                for dup in duplicates:
                    print(f"  - Duplicate Name: '{dup.name}' ({dup.department}) appears more "
                          "than once; the first occurrence is kept.")
                print("Enter 'r' to remove invalid data and continue, "
                      "or 't' to terminate this upload.")
                action = self._ask("Your choice (r/t): ").lower()
                if action == "t":
                    print("Upload terminated.")
                    continue
                if action != "r":
                    print("Invalid choice. Upload terminated.")
                    continue

            if not valid:
                print("No valid participants found. Please try again.")
                continue

            self.users = valid
            self.groups = self.list_manager.get_group_info(valid)
            print(f"\nLoaded {len(valid)} participants from {source}.")
            print("Department groups:")
            for group, count in self.groups.items():
                print(f"  - {group}: {count} people")
            return valid

    # ----------------------------------------------------------- prize config
    def _setup_prizes(self) -> None:
        while True:
            print("\n--- Prize Configuration ---")
            print("1. Load preset prize library")
            print("2. Add custom prize")
            print("3. Set avoidance groups for a prize")
            print("4. Reorder lottery sequence")
            print("5. View prizes")
            print("6. Start lottery")
            print("0. Exit")
            choice = self._ask("Select an option: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        print(f"  - {error}")
                else:
                    print("Preset prizes loaded.")
            elif choice == "2":
                self._add_custom_prize()
            elif choice == "3":
                self._configure_avoid_groups()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                self._show_prizes()
            elif choice == "6":
                if not self.prize_manager.get_prizes():
                    print("Please add at least one prize first.")
                    continue
                self._run_lottery()
                return
            elif choice == "0":
                return
            else:
                print("Invalid option.")

    def _parse_group_selection(self, text: str) -> list[str] | None:
        """Accept group numbers (from the group list) or exact group names, comma separated."""
        group_names = list(self.groups.keys())
        selected: list[str] = []
        if not text.strip():
            return []
        for token in [t.strip() for t in text.split(",") if t.strip()]:
            if token.isdigit():
                index = int(token)
                if not 1 <= index <= len(group_names):
                    return None
                name = group_names[index - 1]
            elif token in group_names:
                name = token
            else:
                return None
            if name not in selected:
                selected.append(name)
        return selected

    def _show_group_choices(self) -> None:
        print("Available department groups:")
        for idx, (group, count) in enumerate(self.groups.items(), start=1):
            print(f"  {idx}. {group} ({count} people)")

    def _add_custom_prize(self) -> None:
        name = self._ask("Prize name: ")
        try:
            quantity = int(self._ask("Quantity: "))
        except ValueError:
            print("Quantity must be an integer.")
            return
        self._show_group_choices()
        groups_text = self._ask("Avoidance groups (numbers or names, comma separated; empty for none): ")
        groups = self._parse_group_selection(groups_text)
        if groups is None:
            print("Invalid group selection. Prize not added.")
            return
        print("Associated lottery person (optional):")
        print("  0. None")
        for idx, person in enumerate(LOTTERY_PERSONS, start=1):
            print(f"  {idx}. {person}")
        drawer_text = self._ask("Select (0-4): ")
        drawer = None
        if drawer_text.isdigit() and 1 <= int(drawer_text) <= len(LOTTERY_PERSONS):
            drawer = LOTTERY_PERSONS[int(drawer_text) - 1]
        elif drawer_text not in ("", "0"):
            print("Invalid selection. Prize not added.")
            return
        try:
            prize = self.prize_manager.add_prize(name, quantity, groups, drawer)
            print(f"Prize added: {prize.name} (quantity {prize.quantity}).")
        except ValueError as exc:
            print(f"Prize not added: {exc}")

    def _configure_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._show_prizes()
        try:
            index = int(self._ask("Select prize number: "))
            prize = prizes[index - 1]
        except (ValueError, IndexError):
            print("Invalid prize number.")
            return
        self._show_group_choices()
        groups_text = self._ask("Avoidance groups (numbers or names, comma separated; empty to clear): ")
        groups = self._parse_group_selection(groups_text)
        if groups is None:
            print("Invalid group selection.")
            return
        self.prize_manager.set_avoid_groups(prize.name, groups)
        print(f"Avoidance groups for '{prize.name}': {', '.join(groups) if groups else 'none'}")

    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._show_prizes()
        text = self._ask("Enter new order as serial numbers (e.g. 3 1 2): ")
        try:
            order = [int(x) for x in text.replace(",", " ").split()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Lottery sequence updated.")
            self._show_prizes()
        else:
            print("Invalid order: every prize number must appear exactly once.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Lottery sequence:")
        for idx, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            print(f"  {idx}. {prize.name} | quantity: {prize.quantity} | "
                  f"avoid groups: {groups} | drawer: {drawer}")

    # -------------------------------------------------------- lottery process
    def _await_command(self) -> str:
        """Wait for user input before a draw. Returns 'draw' or 'skip'."""
        while True:
            cmd = self._ask("Press Enter to draw (commands: pause / terminate): ").lower()
            if cmd in ("", "continue"):
                return "draw"
            if cmd == "terminate":
                return "skip"
            if cmd == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    sub = self._ask("> ").lower()
                    if sub == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if sub == "terminate":
                        return "skip"
                    print("Unknown command. Use 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause', or 'terminate'.")

    def _run_lottery(self) -> None:
        lottery = LotteryManager(self.users)
        prizes = self.prize_manager.get_prizes()

        for round_no, prize in enumerate(prizes, start=1):
            print("\n" + SEPARATOR)
            print(f"Round {round_no}: {prize.name} | Quantity: {prize.quantity}")
            print(f"Rule: {lottery.get_rule_description(prize)}")
            print(SEPARATOR)

            round_winners: list[Winner] = []
            drawn = 0
            while drawn < prize.quantity:
                if lottery.get_eligible_count(prize) == 0:
                    print("No eligible participants remain for this prize. Round ends.")
                    break
                action = self._await_command()
                if action == "skip":
                    remaining = prize.quantity - drawn
                    print(f"Round terminated. Skipping {remaining} remaining winner(s).")
                    break
                print(f"Currently drawing the {ordinal(drawn + 1)} winner...")
                user = lottery.draw_winner(prize)
                if user is None:
                    print("No eligible participants remain for this prize. Round ends.")
                    break
                winner = self.result_manager.add_winner(user, prize)
                round_winners.append(winner)
                drawn += 1
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

            lottery.end_round(prize)

            print(f"\nRound {round_no} results:")
            if not round_winners:
                print("  No winners in this round.")
            for w in round_winners:
                print(f"  #{w.winning_order} {w.user.name} ({w.user.department}) - {w.prize_name}")

        self._show_summary(prizes)
        self._export_prompt(prizes)

    def _show_summary(self, prizes: list[Prize]) -> None:
        print("\n" + SEPARATOR)
        print("Final Winner List (by prize level, high to low)")
        print(SEPARATOR)
        grouped = self.result_manager.get_winners_by_prize(prizes)
        for prize_name, winners in grouped.items():
            print(f"\n{prize_name}:")
            if not winners:
                print("  (no winners)")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {w.winning_time:%Y-%m-%d %H:%M:%S}")

    def _export_prompt(self, prizes: list[Prize]) -> None:
        answer = self._ask("\nExport the winner list to a txt (CSV format) file? (y/n): ").lower()
        if answer in ("y", "yes"):
            _, message = self.result_manager.export_to_csv(prizes)
            print(message)
        else:
            print("Export skipped.")


def main() -> int:
    try:
        LotteryCLI().run()

        print("\nExiting.")

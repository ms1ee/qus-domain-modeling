"""Process-wide global variable store.

Holds the shared system state (funds, rider list, rider ID counter, running
flag, simulated tick, order records). Access it through ``set_value`` and
``get_value`` so that all modules see the same values.
"""

import config

_global_dict = {}


def _init():
    """Reset the global store to its default values.

    Keys created:
        riders: list of Rider objects (dynamic rider pool).
        funds: integer system funds (may become negative).
        rider_id_counter: last issued rider ID (0 at startup).
        running: boolean main-loop control flag.
        tick: current simulated time in ticks.
        orders: list of every Order accepted into the system.
        order_history: list of completed Order objects.
    """
    global _global_dict
    _global_dict = {
        "riders": [],
        "funds": config.INITIAL_FUNDS,
        "rider_id_counter": 0,
        "running": True,
        "tick": 0,
        "orders": [],
        "order_history": [],
    }


def set_value(key, value):
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Any Python object to store.
    """
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key is absent.

    Returns:
        The stored value, or ``default`` when the key does not exist.
    """
    return _global_dict.get(key, default)

<file path="src/grid.py">
"""Coordinate helpers for the 8x8 grid (800x800 units, 100-unit cells).

Streets run along grid lines (x or y is a multiple of 100). Restaurant and
customer locations are cell centers (``cell * 100 + 50``). Each location is
served from the street intersection at the top-left corner of its cell
(the "access node"), which is always a multiple of 100 on both axes.
"""

import config


def is_valid_coordinate(value):
    """Return True when ``value`` is an ``int`` within the map bounds.

    Args:
        value: Candidate coordinate.

    Returns:
        bool: True if the value is an integer in ``[COORD_MIN, COORD_MAX]``.
    """
    return (
        isinstance(value, int)
        and not isinstance(value, bool)
        and config.COORD_MIN <= value <= config.COORD_MAX
    )


def clamp_coordinate(value):
    """Clamp a numeric value into the map bounds and return an int.

    Args:
        value: Numeric coordinate (int or float).

    Returns:
        int: Value limited to ``[COORD_MIN, COORD_MAX]``.
    """
    return max(config.COORD_MIN, min(config.COORD_MAX, int(value)))


def cell_index(value):
    """Return the grid cell index (0-7) containing a coordinate.

    Values equal to the map edge (800) belong to the last cell.

    Args:
        value: Coordinate value.

    Returns:
        int: Cell index in ``[0, GRID_CELLS - 1]``.
    """
    clamped = min(clamp_coordinate(value), config.MAP_SIZE - 1)
    return clamped // config.CELL_SIZE


def snap_to_center(x, y):
    """Transform raw input coordinates to the center of their grid cell.

    Implements ``center = (grid // 100) * 100 + 50`` for both axes.

    Args:
        x: Raw x coordinate.
        y: Raw y coordinate.

    Returns:
        tuple[int, int]: Cell center coordinates.
    """
    center_x = cell_index(x) * config.CELL_SIZE + config.CELL_CENTER_OFFSET
    center_y = cell_index(y) * config.CELL_SIZE + config.CELL_CENTER_OFFSET
    return center_x, center_y


def access_node(point):
    """Return the street intersection serving the cell that contains ``point``.

    Args:
        point: ``(x, y)`` location (usually a cell center).

    Returns:
        tuple[int, int]: Intersection coordinates (both multiples of 100).
    """
    x, y = point
    return (
        cell_index(x) * config.CELL_SIZE,
        cell_index(y) * config.CELL_SIZE,
    )


def is_on_road(x, y):
    """Return True when the point lies on a street (grid line).

    Args:
        x: x coordinate.
        y: y coordinate.

    Returns:
        bool: True if ``x`` or ``y`` is a multiple of the cell size.
    """
    return x % config.CELL_SIZE == 0 or y % config.CELL_SIZE == 0


def snap_to_road(x, y):
    """Move an off-road point horizontally onto the nearest vertical street.

    Args:
        x: x coordinate.
        y: y coordinate.

    Returns:
        tuple[int, int]: A point that lies on a street.
    """
    x = clamp_coordinate(x)
    y = clamp_coordinate(y)
    if is_on_road(x, y):
        return x, y
    half = config.CELL_SIZE // 2
    street_x = min(config.MAP_SIZE, ((x + half) // config.CELL_SIZE) * config.CELL_SIZE)
    return street_x, y


def manhattan(a, b):
    """Return the Manhattan (street-following) distance between two points.

    Args:
        a: ``(x, y)`` point.
        b: ``(x, y)`` point.

    Returns:
        int: ``|ax - bx| + |ay - by|``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

<file path="src/path_planner.py">
"""Grid-based path planning for rider movement.

Riders travel only along streets. The planner returns one movement step per
tick; each step never leaves the street network and always makes progress
toward the target, so riders cannot livelock. Riders do not block each other,
which means path conflicts never stall a rider.
"""

import config
import grid


def _toward(value, target, step):
    """Move ``value`` toward ``target`` by at most ``step`` units.

    Args:
        value: Current coordinate.
        target: Desired coordinate.
        step: Maximum movement in this call (positive int).

    Returns:
        int: New coordinate, never overshooting ``target``.
    """
    if value == target:
        return value
    delta = target - value
    move = min(step, abs(delta))
    return value + move if delta > 0 else value - move


class PathPlanner:
    """Stateless planner that computes street-constrained movement steps."""

    @staticmethod
    def next_step(x, y, tx, ty, speed):
        """Compute the next position of a rider heading to ``(tx, ty)``.

        Movement rules:
            * At an intersection, travel horizontally until aligned with the
              target column, then vertically (right-angle turns).
            * On a horizontal street (mid-block), move horizontally toward the
              target column.
            * On a vertical street (mid-block), move vertically toward the
              target row.
            * Directly above/below the target (same column) is a special
              case: the rider moves vertically only and ``special_case`` is
              reported as True.

        Args:
            x: Current x coordinate (must be on a street).
            y: Current y coordinate (must be on a street).
            tx: Target x coordinate (normally a street intersection).
            ty: Target y coordinate (normally a street intersection).
            speed: Maximum units to travel this tick.

        Returns:
            tuple[int, int, bool]: ``(new_x, new_y, special_case)``.
        """
        if (x, y) == (tx, ty):
            return x, y, False

        step = max(1, int(speed))
        on_horizontal = y % config.CELL_SIZE == 0
        on_vertical = x % config.CELL_SIZE == 0

        if on_horizontal and on_vertical:
            if x != tx:
                new_x, new_y = _toward(x, tx, step), y
            else:
                new_x, new_y = x, _toward(y, ty, step)
        elif on_horizontal:
            new_x, new_y = _toward(x, tx, step), y
        elif on_vertical:
            new_x, new_y = x, _toward(y, ty, step)
        else:
            # Off-road point (should not happen): join the nearest vertical street.
            half = config.CELL_SIZE // 2
            street_x = min(
                config.MAP_SIZE,
                ((x + half) // config.CELL_SIZE) * config.CELL_SIZE,
            )
            new_x, new_y = _toward(x, street_x, step), y

        special = new_x == tx and new_y != ty
        return new_x, new_y, special

    @staticmethod
    def route_length(a, b):
        """Return the street-network length estimate between two points.

        Args:
            a: ``(x, y)`` start point.
            b: ``(x, y)`` end point.

        Returns:
            int: Manhattan distance, which equals the L-shaped route length.
        """
        return grid.manhattan(a, b)

"""Order data model.

An order is defined by a restaurant location and a customer location. Both
inputs are transformed to grid cell centers, and each is served from the
street intersection of its cell (the access node).
"""

import itertools

import globalvar
import grid

STATUS_CREATED = "created"
STATUS_ASSIGNED = "assigned"
STATUS_PICKED_UP = "picked_up"
STATUS_COMPLETED = "completed"


def _validate_point(point, label):
    """Validate an ``(x, y)`` pair of integer coordinates.

    Args:
        point: Sequence of two integers.
        label: Human-readable name used in error messages.

    Returns:
        tuple[int, int]: The validated point.

    Raises:
        ValueError: If the point is malformed or out of range.
    """
    if len(point) != 2:
        raise ValueError(f"{label} location must have exactly two coordinates")
    x, y = point
    for value in (x, y):
        if not grid.is_valid_coordinate(value):
            raise ValueError(
                f"{label} coordinate {value!r} must be an integer in "
                f"[{grid.config.COORD_MIN}, {grid.config.COORD_MAX}]"
            )
    return x, y


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier (auto-incremented).
        res_input: Raw restaurant coordinates as entered.
        cus_input: Raw customer coordinates as entered.
        res_pos: Restaurant grid cell center ``(x, y)``.
        cus_pos: Customer grid cell center ``(x, y)``.
        restaurant_node: Street intersection used for pickup.
        customer_node: Street intersection used for delivery.
        distance: Manhattan distance from restaurant to customer node.
        status: One of created, assigned, picked_up, completed.
        rider_id: ID of the assigned rider, or None.
        created_tick, start_tick, pickup_tick, complete_tick: Timing data.
    """

    _id_sequence = itertools.count(1)

    def __init__(self, res_pos, cus_pos):
        """Create an order from raw restaurant and customer coordinates.

        Args:
            res_pos: ``(x, y)`` restaurant location, integers in 0-800.
            cus_pos: ``(x, y)`` customer location, integers in 0-800.

        Raises:
            ValueError: If either location is malformed or out of range.
        """
        res_pos = _validate_point(res_pos, "Restaurant")
        cus_pos = _validate_point(cus_pos, "Customer")

        self.id = next(Order._id_sequence)
        self.res_input = res_pos
        self.cus_input = cus_pos
        self.res_pos = grid.snap_to_center(*res_pos)
        self.cus_pos = grid.snap_to_center(*cus_pos)
        self.restaurant_node = grid.access_node(self.res_pos)
        self.customer_node = grid.access_node(self.cus_pos)
        self.distance = grid.manhattan(self.restaurant_node, self.customer_node)

        self.status = STATUS_CREATED
        self.rider_id = None
        self.created_tick = globalvar.get_value("tick", 0)
        self.start_tick = None
        self.pickup_tick = None
        self.complete_tick = None

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider.

        The caller is responsible for appending the order to the rider's
        ``orders`` list.

        Args:
            rider_id: ID of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED
        self.start_tick = globalvar.get_value("tick", 0)

    def mark_picked_up(self, tick):
        """Record that the rider collected the food at the restaurant.

        Args:
            tick: Simulation tick of the pickup.
        """
        self.status = STATUS_PICKED_UP
        self.pickup_tick = tick

    def mark_completed(self, tick):
        """Record delivery completion.

        Args:
            tick: Simulation tick of the delivery.
        """
        self.status = STATUS_COMPLETED
        self.complete_tick = tick

    @property
    def time_taken(self):
        """Ticks from assignment to completion, or None if not completed."""
        if self.start_tick is None or self.complete_tick is None:
            return None
        return self.complete_tick - self.start_tick

    @property
    def efficiency(self):
        """Delivery efficiency in distance units per tick, or None."""
        taken = self.time_taken
        if not taken:
            return None
        return self.distance / taken

    def to_dict(self):
        """Return a JSON-serializable representation of the order.

        Returns:
            dict: Order attributes.
        """
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "distance": self.distance,
            "status": self.status,
            "rider_id": self.rider_id,
            "created_tick": self.created_tick,
            "start_tick": self.start_tick,
            "pickup_tick": self.pickup_tick,
            "complete_tick": self.complete_tick,
            "time_taken": self.time_taken,
        }

    def __repr__(self):
        return (
            f"Order(id={self.id}, status={self.status!r}, "
            f"rider_id={self.rider_id})"
        )

"""Rider model: state machine, street movement and event logging.

State machine (derived from the rider's order list):
    idle (0)        -> no active orders
    picking up (1)  -> at least one order assigned but not yet collected
    delivering (2)  -> all active orders collected, heading to customers

Multiple orders are served with a greedy nearest-stop policy: the rider
always heads for the nearest pending stop (restaurant of an assigned order or
customer of a carried order). Deliveries win ties so food is not held longer
than necessary.
"""

import logging
import random

import config
import globalvar
import grid
from order_model import STATUS_ASSIGNED, STATUS_COMPLETED, STATUS_PICKED_UP
from path_planner import PathPlanner

logger = logging.getLogger("dispatch.rider")

STATE_NAMES = {
    config.STATE_IDLE: "idle",
    config.STATE_PICKING_UP: "picking up",
    config.STATE_DELIVERING: "delivering",
}

EVENT_SPAWN = "spawn"
EVENT_ACCEPTED = "accepted_order"
EVENT_ARRIVED_MERCHANT = "arrived_at_merchant"
EVENT_DELIVERED = "delivered_to_customer"
EVENT_STATE_CHANGE = "state_change"


class Rider:
    """A delivery rider that moves along streets and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current position (integers, always on a street).
        speed: Movement in units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Active orders (assigned or carried).
        special_case: True while directly above/below the current target.
        log: List of event dictionaries (keys include "event" and "order_id").
        distance_traveled: Total units travelled since spawn.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        """Create a rider on a street.

        Args:
            rider_id: Unique integer identifier.
            x: Optional starting x; random road point when omitted.
            y: Optional starting y; random road point when omitted.
            speed: Optional units per tick; random within configured range.

        Raises:
            ValueError: If the supplied coordinates are out of range.
        """
        self.rider_id = int(rider_id)
        if x is None or y is None:
            x, y = self.random_road_point()
        if not (grid.is_valid_coordinate(x) and grid.is_valid_coordinate(y)):
            raise ValueError(f"Rider start position ({x},{y}) is out of range")
        if not grid.is_on_road(x, y):
            x, y = grid.snap_to_road(x, y)
        self.x = x
        self.y = y
        if speed is None:
            speed = random.randint(config.RIDER_SPEED_MIN, config.RIDER_SPEED_MAX)
        self.speed = max(1, int(speed))
        self.state = config.STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.distance_traveled = 0
        self._acknowledged = set()
        self._log_event(EVENT_SPAWN, None)

    @staticmethod
    def random_road_point():
        """Pick a random integer point lying on a street.

        Returns:
            tuple[int, int]: A road point within the map.
        """
        span = config.MAP_SIZE
        step = config.CELL_SIZE
        if random.random() < 0.5:
            y = random.randrange(0, span + 1, step)
            x = random.randint(0, span)
        else:
            x = random.randrange(0, span + 1, step)
            y = random.randint(0, span)
        return x, y

    @property
    def position(self):
        """Current ``(x, y)`` position."""
        return (self.x, self.y)

    @property
    def is_idle(self):
        """True when the rider has no active orders."""
        return not self.orders

    @property
    def state_name(self):
        """Human-readable state label."""
        return STATE_NAMES.get(self.state, "unknown")

    def refresh_state(self):
        """Register newly assigned orders and recompute the rider state.

        Safe to call at any time; it is idempotent.
        """
        self._acknowledge_new_orders()
        self._update_state()

    def move(self):
        """Advance the rider by one tick.

        Processes pickups and deliveries at the current position, moves one
        movement step toward the nearest pending stop, processes arrivals
        again, and refreshes the state machine.
        """
        self.refresh_state()
        self._process_arrivals()
        target = self._choose_target()
        if target is not None:
            tx, ty = target
            nx, ny, special = PathPlanner.next_step(self.x, self.y, tx, ty, self.speed)
            self.distance_traveled += abs(nx - self.x) + abs(ny - self.y)
            self.x, self.y = nx, ny
            self.special_case = special
            self._process_arrivals()
        else:
            self.special_case = False
        self.refresh_state()

    def to_dict(self):
        """Return a JSON-serializable snapshot of the rider.

        Returns:
            dict: Rider attributes including active order IDs.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "order_ids": [order.id for order in self.orders],
            "special_case": self.special_case,
            "distance_traveled": self.distance_traveled,
        }

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _log_event(self, event, order_id, **extra):
        """Append an event entry to the rider log, trimming old entries."""
        entry = {
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("tick", 0),
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        entry.update(extra)
        self.log.append(entry)
        overflow = len(self.log) - config.MAX_RIDER_LOG
        if overflow > 0:
            del self.log[:overflow]

    def _acknowledge_new_orders(self):
        """Log an accepted_order event for every newly assigned order."""
        for order in self.orders:
            if order.status == STATUS_ASSIGNED and order.id not in self._acknowledged:
                self._acknowledged.add(order.id)
                self._log_event(EVENT_ACCEPTED, order.id)

    def _update_state(self):
        """Derive the state from the order list and log any transition."""
        if any(order.status == STATUS_ASSIGNED for order in self.orders):
            new_state = config.STATE_PICKING_UP
        elif self.orders:
            new_state = config.STATE_DELIVERING
        else:
            new_state = config.STATE_IDLE
        if new_state != self.state:
            old_state = self.state
            self.state = new_state
            self._log_event(EVENT_STATE_CHANGE, None, from_state=old_state, to_state=new_state)

    def _choose_target(self):
        """Select the nearest pending stop, preferring deliveries on ties.

        Returns:
            tuple[int, int] | None: Target intersection, or None if no stops.
        """
        here = self.position
        best = None
        for order in self.orders:
            if order.status == STATUS_ASSIGNED:
                candidate = (grid.manhattan(here, order.restaurant_node), 1, order.restaurant_node)
            elif order.status == STATUS_PICKED_UP:
                candidate = (grid.manhattan(here, order.customer_node), 0, order.customer_node)
            else:
                continue
            if best is None or candidate[:2] < best[:2]:
                best = candidate
        return None if best is None else best[2]

    def _process_arrivals(self):
        """Handle pickups and deliveries that occur at the current position."""
        here = self.position
        tick = globalvar.get_value("tick", 0)

        for order in list(self.orders):
            if order.status == STATUS_ASSIGNED and order.restaurant_node == here:
                order.mark_picked_up(tick)
                self._log_event(EVENT_ARRIVED_MERCHANT, order.id)

        for order in list(self.orders):
            if order.status == STATUS_PICKED_UP and order.customer_node == here:
                order.mark_completed(tick)
                self.orders.remove(order)
                self._acknowledged.discard(order.id)
                self._log_event(EVENT_DELIVERED, order.id)
                self._record_completion(order)

    def _record_completion(self, order):
        """Store a completed order in the history and credit the delivery fee."""
        history = globalvar.get_value("order_history")
        if history is None:
            history = []
            globalvar.set_value("order_history", history)
        history.append(order)
        funds = globalvar.get_value("funds", 0)
        globalvar.set_value("funds", funds + config.DELIVERY_FEE)
        logger.info(
            "Order %s delivered by rider %s at tick %s",
            order.id,
            self.rider_id,
            order.complete_tick,
        )

"""Order management: creation, rider matching and assignment.

``find_best_rider_for_order`` implements the distance-first strategy with a
load-balancing penalty: the score of a rider is its Manhattan distance to the
restaurant plus a fixed penalty per already-assigned order.
"""

import logging

import config
import globalvar
import grid
from order_model import STATUS_CREATED, Order

logger = logging.getLogger("dispatch.order")


class NoRiderAvailableError(RuntimeError):
    """Raised when no rider can accept a new order."""


def find_best_rider_for_order(order):
    """Return the most suitable rider for ``order``.

    Riders already at ``MAX_ORDERS_PER_RIDER`` are skipped. The score is
    ``manhattan(rider, restaurant_node) + LOAD_PENALTY * active_orders``;
    ties are broken by the lower rider ID.

    Args:
        order: Order to be assigned.

    Returns:
        Rider | None: The selected rider, or None if no rider has capacity.
    """
    riders = globalvar.get_value("riders", [])
    best_rider = None
    best_key = None
    for rider in riders:
        load = len(rider.orders)
        if load >= config.MAX_ORDERS_PER_RIDER:
            continue
        distance = grid.manhattan(rider.position, order.restaurant_node)
        score = distance + config.LOAD_PENALTY * load
        key = (score, rider.rider_id)
        if best_key is None or key < best_key:
            best_key = key
            best_rider = rider
    return best_rider

        Args:
            values: Sequence of four integers.

        Returns:
            tuple[tuple[int, int], tuple[int, int]]: Restaurant and customer points.

        Raises:

            ValueError: If the coordinates are invalid.

        rider.refresh_state()

def apply_config_file(path):
    """Load tunable parameters from a JSON configuration file.

    The file may contain the parameters at top level or inside a
    ``"parameters"`` object. Unknown keys are ignored.

    Args:
        path: Path to the JSON file.

    Returns:
        dict: The parameters that were applied.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the JSON is malformed or a value is invalid.
    """
    with open(path, encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError("configuration file must contain a JSON object")
    params = data.get("parameters", data)
    if not isinstance(params, dict):
        raise ValueError("'parameters' must be a JSON object")

    applied = {}
    for name in config.TUNABLE_PARAMS:
        if name not in params:
            continue
        value = params[name]
        if isinstance(value, bool) or not isinstance(value, int) or value < 0:
            raise ValueError(f"parameter {name} must be a non-negative integer")
        applied[name] = value

    speed_min = applied.get("RIDER_SPEED_MIN", config.RIDER_SPEED_MIN)
    speed_max = applied.get("RIDER_SPEED_MAX", config.RIDER_SPEED_MAX)
    if speed_min < 1 or speed_min > speed_max:
        raise ValueError("rider speed range must satisfy 1 <= MIN <= MAX")

    for name, value in applied.items():
        setattr(config, name, value)
    return applied


class SystemManager:
    """Manages funds, the rider pool, the simulation clock and consistency.

    The state itself lives in ``globalvar``; this class provides the
    operations on that state.
    """

    @property
    def riders(self):
        """The live list of Rider objects."""
        return globalvar.get_value("riders", [])

    @property
    def funds(self):
        """Current system funds (integer, may be negative)."""
        return globalvar.get_value("funds", 0)

    @property
    def tick(self):
        """Current simulated time in ticks."""
        return globalvar.get_value("tick", 0)

    @property
    def running(self):
        """Main-loop control flag."""
        return globalvar.get_value("running", False)

    def stop(self):
        """Request the main loop to terminate."""
        globalvar.set_value("running", False)

    def adjust_funds(self, delta):
        """Add ``delta`` (may be negative) to the system funds.

        Args:
            delta: Integer amount to add.
        """
        globalvar.set_value("funds", self.funds + int(delta))

    def has_idle_rider(self):
        """Return True if at least one rider has no active orders."""
        return any(rider.is_idle for rider in self.riders)

    def can_recruit(self):
        """Return True when the pool has room and funds cover the recruit cost."""
        return (
            len(self.riders) < config.MAX_RIDERS
            and self.funds >= config.RIDER_RECRUIT_COST
        )

    def recruit_rider(self):
        """Create a new rider, charging the recruitment cost.

        Returns:
            Rider | None: The new rider, or None if recruitment is not possible.
        """
        if not self.can_recruit():
            return None
        rider_id = self._next_rider_id()
        self.adjust_funds(-config.RIDER_RECRUIT_COST)
        rider = Rider(rider_id)
        self.riders.append(rider)
        logger.info("Recruited rider %s at (%s,%s)", rider.rider_id, rider.x, rider.y)
        return rider

    def advance(self, ticks=1):
        """Advance the simulation clock and move every rider.

        A failure in one rider is logged and does not stop the simulation.

        Args:
            ticks: Number of ticks to simulate.
        """
        for _ in range(int(ticks)):
            globalvar.set_value("tick", self.tick + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception:  # noqa: BLE001 - keep the system running
                    logger.exception("Rider %s failed during tick %s", rider.rider_id, self.tick)

    def check_consistency(self):
        """Detect and repair inconsistent state.

        Repairs performed: removing completed orders from rider lists,
        snapping off-road riders onto streets, and recomputing rider states.

        Returns:
            list[str]: Descriptions of each issue found (empty when healthy).
        """
        issues = []
        seen_ids = set()
        for rider in self.riders:
            if rider.rider_id in seen_ids:
                issues.append(f"duplicate rider id {rider.rider_id}")
            seen_ids.add(rider.rider_id)

            stale = [o for o in rider.orders if o.status == STATUS_COMPLETED]
            for order in stale:
                rider.orders.remove(order)
                issues.append(f"removed completed order {order.id} from rider {rider.rider_id}")

            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    issues.append(
                        f"order {order.id} owner mismatch on rider {rider.rider_id}"
                    )

            if not grid.is_on_road(rider.x, rider.y):
                old = (rider.x, rider.y)
                rider.x, rider.y = grid.snap_to_road(rider.x, rider.y)
                issues.append(f"rider {rider.rider_id} moved off-road from {old}")

            before = rider.state
            rider.refresh_state()
            if rider.state != before:
                issues.append(f"rider {rider.rider_id} state corrected to {rider.state}")

        for issue in issues:
            logger.warning("Consistency: %s", issue)
        return issues

    def snapshot(self):
        """Return a JSON-serializable snapshot of the whole system.

        Returns:
            dict: Tick, funds, rider IDs, riders, orders and parameters.
        """
        return {
            "tick": self.tick,
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "parameters": {name: getattr(config, name) for name in config.TUNABLE_PARAMS},
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in globalvar.get_value("orders", [])],
            "order_history": [order.to_dict() for order in globalvar.get_value("order_history", [])],
        }

    def save_state(self, path):
        """Write the system snapshot to a JSON file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(self.snapshot(), handle, indent=2)
        logger.info("State saved to %s", path)

    def load_config(self, path):
        """Apply parameter overrides from a JSON configuration file.

        Args:
            path: Path to the configuration file.

        Returns:
            dict: Applied parameters, or an empty dict if loading failed.
        """
        try:
            return apply_config_file(path)
        except (OSError, ValueError) as exc:
            logger.warning("Could not load configuration %s: %s", path, exc)
            return {}

    def _next_rider_id(self):
        """Increment and return the rider ID counter."""
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        return counter


def init_system(config_path=None):
    """Initialize global state and return the system manager.

    Args:
        config_path: Optional JSON configuration file applied before the
            default state is created.

    Returns:
        SystemManager: The manager bound to the initialized globals.
    """
    if config_path and os.path.exists(config_path):
        try:
            applied = apply_config_file(config_path)
            logger.info("Applied configuration overrides: %s", applied)
        except (OSError, ValueError) as exc:
            logger.warning("Ignoring configuration %s: %s", config_path, exc)

    globalvar._init()
    manager = SystemManager()
    globalvar.set_value("system_manager", manager)
    logger.info("System initialized with funds %s", manager.funds)
    return manager

def _cell_label(center):
    """Return a short ``[col, row]`` label for a cell center."""
    return f"[{grid.cell_index(center[0])}, {grid.cell_index(center[1])}]"


def _move_toward(value, target, step):
    """Move ``value`` toward ``target`` by at most ``step``."""
    if value == target:
        return value
    delta = target - value
    move = min(step, abs(delta))
    return value + move if delta > 0 else value - move

        while (x, y) != (tx, ty):

            y = _move_toward(y, ty, 10)
            points.append((x, y))
        return points

                self._put(canvas, point[0], point[1], "*")

                for x, y in self._segment(previous, point):
                    self._put(canvas, x, y, "*")

            lines.append(label + "".join(row))
        lines.append("     x: 0 .. 800 (street intersections every 100 units)")

"""Command parsing, input validation and command history."""

import logging

import config

try:  # Enables line editing and up/down history where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - not available on some platforms
    readline = None

logger = logging.getLogger("dispatch.input")

COMMAND_ALIASES = {
    "s": "start",
    "st": "start",
    "stat": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "h": "help",
    "?": "help",
    "hist": "history",
    "q": "quit",
    "x": "exit",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Place an order (restaurant then customer, 0-800).
                       Without arguments you will be prompted.
  status  (s)          Show system funds, riders, states and order lists.
  orders  (o)          Show details for all orders.
  riders  (r)          Show the status-change log of every rider.
  map     (m)          Show the ASCII map with riders and active orders.
  tick [n] (t)         Advance the simulation by n ticks (default 1).
  history (hist)       Show previously entered commands.
  help    (h, ?)       Show this help.
  quit    (q), exit    Save state and exit.
Shortcut: !! repeats the previous command.
"""


class InputManager:
    """Reads and validates user commands.

    Args:
        reader: Callable used to read a line (defaults to ``input``).
        writer: Callable used to print messages (defaults to ``print``).
    """

    def __init__(self, reader=input, writer=print):
        self._reader = reader
        self._writer = writer
        self._history = []

    def read_line(self, prompt="dispatch> "):
        """Read one raw line from the user.

        Args:
            prompt: Prompt text.

        Returns:
            str: The stripped input line.
        """
        return self._reader(prompt).strip()

    def read_command(self):
        """Read and parse the next command, honouring ``!!`` and history.

        Returns:
            tuple[str, list[str]]: Canonical command name and its arguments.
        """
        line = self.read_line()
        if line == "!!":
            if not self._history:
                self._writer("No previous command to repeat.")
                return "", []
            line = self._history[-1]
            self._writer(f"Repeating: {line}")
        if line:
            self._history.append(line)
        return self.parse_command(line)

    @staticmethod
    def parse_command(line):
        """Split a command line into a canonical name and arguments.

        Args:
            line: Raw command text.

        Returns:
            tuple[str, list[str]]: Command name (lower case) and arguments.
        """
        tokens = line.split()
        if not tokens:
            return "", []
        name = tokens[0].lower()
        return COMMAND_ALIASES.get(name, name), tokens[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Validate four integer coordinates.

        Args:
            tokens: Sequence of four values (strings or ints).

        Returns:
            list[int]: The validated integers.

        Raises:
            ValueError: If the count, type or range is wrong.
        """
        tokens = list(tokens)
        if len(tokens) != 4:
            raise ValueError(
                "expected four integers: restaurant_x restaurant_y customer_x customer_y "
                f"(got {len(tokens)})"
            )
        values = []
        for token in tokens:
            text = str(token).strip()
            try:
                value = int(text)
            except ValueError:
                raise ValueError(f"'{text}' is not an integer") from None
            if not config.COORD_MIN <= value <= config.COORD_MAX:
                raise ValueError(
                    f"coordinate {value} is outside the allowed range "
                    f"{config.COORD_MIN}-{config.COORD_MAX}"
                )
            values.append(value)
        return values

    def prompt_coordinates(self):

"""Command-line entry point for the food delivery dispatch simulator.

Run with::

    python src/main.py [--config dispatch_config.json] [--log-level INFO]
"""

import argparse
import logging
import sys

import config
import system
from display import DisplayManager
from input_manager import HELP_TEXT, InputManager
from order import NoRiderAvailableError, OrderManager
from order_model import STATUS_COMPLETED

logger = logging.getLogger("dispatch.main")


class DispatchApp:
    """Interactive command loop wiring all managers together.

    Args:
        config_path: Optional JSON configuration file.
        reader: Input callable (defaults to ``input``).
        writer: Output callable (defaults to ``print``).
    """

    def __init__(self, config_path=None, reader=input, writer=print):
        self.manager = system.init_system(config_path)
        self.orders = OrderManager(self.manager)
        self.display = DisplayManager()
        self.inputs = InputManager(reader, writer)
        self.out = writer
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self):
        """Run the main loop until the user quits.

        Returns:
            int: Process exit code (0 on normal exit).
        """
        self.out("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.manager.running:
            try:
                command, args = self.inputs.read_command()
            except (EOFError, KeyboardInterrupt):
                self.out("")
                self.out("Exiting.")
                self.manager.stop()
                break
            if not command:
                continue
            self.dispatch(command, args)
            self._self_check()
        self._save_on_exit()
        return 0

    def dispatch(self, command, args):
        """Execute one command, reporting errors without terminating.

        Args:
            command: Canonical command name.
            args: Command arguments.
        """
        handler = self._handlers.get(command)
        if handler is None:
            self.out(f"Unknown command '{command}'. Type 'help' for the list of commands.")
            return
        try:
            handler(args)
        except (ValueError, NoRiderAvailableError) as exc:
            self.out(f"Error: {exc}")
        except Exception:  # noqa: BLE001 - recover and keep running
            logger.exception("Unexpected failure while running '%s'", command)
            self.out("Unexpected internal error; the system state has been kept consistent.")

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def cmd_start(self, args):
        """Place an order, assign a rider and simulate until delivery."""
        if args:
            values = self.inputs.parse_coordinates(args)
        else:
            values = self.inputs.prompt_coordinates()
            if values is None:
                self.out("Order input cancelled.")
                return

        order = self.orders.create_order(values)
        rider, new_rider = self.orders.assign_order(order)
        self.out(self.display.render_order_placed(order, rider, new_rider))

        trail = [rider.position]
        ticks = 0
        while order.status != STATUS_COMPLETED:
            if ticks >= config.MAX_SIMULATION_TICKS:
                self.out(
                    f"Warning: order #{order.id} not completed after "
                    f"{config.MAX_SIMULATION_TICKS} ticks; stopping the preview."
                )
                break
            self.manager.advance(1)
            ticks += 1
            trail.append(rider.position)

        self.out(self.display.render_route(order, rider, trail))

    def cmd_status(self, args):
        """Show funds, riders and their order lists."""
        self.out(self.display.render_status(self.manager.funds, self.manager.riders, self.manager.tick))
        active = self.orders.active_orders()
        self.out(f"Active orders: {len(active)}")

    def cmd_orders(self, args):
        """Show the details of every order."""
        self.out(self.display.render_orders(self.orders.all_orders()))

    def cmd_riders(self, args):
        """Show the status-change log of every rider."""
        self.out(self.display.render_rider_logs(self.manager.riders))

    def cmd_map(self, args):
        """Show the ASCII map of riders and active orders."""
        self.out(self.display.render_map(riders=self.manager.riders, orders=self.orders.active_orders()))

    def cmd_tick(self, args):
        """Advance the simulation clock by a number of ticks."""
        count = 1
        if args:
            try:
                count = int(args[0])
            except ValueError:
                raise ValueError(f"'{args[0]}' is not an integer tick count") from None
        if not 1 <= count <= config.MAX_MANUAL_TICKS:
            raise ValueError(f"tick count must be between 1 and {config.MAX_MANUAL_TICKS}")
        self.manager.advance(count)
        self.out(f"Advanced {count} tick(s). Current tick: {self.manager.tick}")

    def cmd_history(self, args):
        """Show previously entered commands."""
        history = self.inputs.history()
        if not history:
            self.out("No command history yet.")
            return
        for index, line in enumerate(history, start=1):
            self.out(f"{index:>4}  {line}")

    def cmd_help(self, args):
        """Show the command reference."""
        self.out(HELP_TEXT)

    def cmd_quit(self, args):
        """Stop the main loop."""
        self.manager.stop()
        self.out("Goodbye.")

    # ------------------------------------------------------------------
    # Support
    # ------------------------------------------------------------------
    def _self_check(self):
        """Run consistency checks after each command and report repairs."""
        issues = self.manager.check_consistency()
        if issues:
            self.out(f"Note: repaired {len(issues)} consistency issue(s); see the log for details.")

    def _save_on_exit(self):
        """Persist the system state; failures are logged, not raised."""
        try:
            self.manager.save_state(config.STATE_FILE)
        except OSError as exc:
            logger.error("Could not save state to %s: %s", config.STATE_FILE, exc)


def setup_logging(level_name):
    """Configure logging to the log file, falling back to warnings on stderr.

    Args:
        level_name: Logging level name such as ``"INFO"``.
    """
    level = getattr(logging, str(level_name).upper(), logging.INFO)
    fmt = "%(asctime)s %(levelname)s %(name)s: %(message)s"
    try:
        logging.basicConfig(filename=config.LOG_FILE, level=level, format=fmt, encoding="utf-8")
    except OSError:
        logging.basicConfig(level=logging.WARNING, format=fmt)


def main(argv=None):
    """Parse arguments and run the dispatch simulator.

    Args:
        argv: Optional argument list (defaults to ``sys.argv[1:]``).

    Returns:
        int: Process exit code.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument(
        "--config",
        default=config.CONFIG_FILE,
        help="JSON configuration file with system parameters",
    )
    parser.add_argument("--log-level", default="INFO", help="logging level (default: INFO)")
    args = parser.parse_args(argv)

    setup_logging(args.log_level)
    app = DispatchApp(config_path=args.config)
    return app.run()


if __name__ == "__main__":
    sys.exit(main())

from __future__ import annotations

import os
import re

from lottery_system.constants import DEFAULT_USER_RECORDS
from lottery_system.models import User, normalize_key

# Accept ASCII/full-width commas and semicolons as separators.
_SEPARATOR_PATTERN = re.compile(r"[,;，；]")


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in default team building list."""
        users = [User(name, department) for name, department in DEFAULT_USER_RECORDS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each line: "Name,Department"."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []

        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        try:
            lines = self._read_lines(path)
        except OSError as exc:
            return [], [f"File Error: cannot read file ({exc})"]

        for line_no, raw_line in enumerate(lines, start=1):
            text = raw_line.strip()
            if not text:
                continue
            parts = [part.strip() for part in _SEPARATOR_PATTERN.split(text)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{text}' "
                    f"(expected 'Name,Department')"
                )
                continue
            users.append(User(name=parts[0], department=parts[1]))

        return users, errors

    @staticmethod
    def _read_lines(path: str) -> list[str]:
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=encoding) as handle:
                    return handle.read().splitlines()
            except UnicodeDecodeError:
                continue
        raise OSError(f"unable to decode file {path}")

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid (first occurrence) and duplicate entries."""
        seen: set[str] = set()
        valid_users: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = normalize_key(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid_users.append(user)
        return valid_users, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return a mapping of department name to number of people."""
        canonical: dict[str, str] = {}
        counts: dict[str, int] = {}
        for user in users:
            department = " ".join(user.department.split())
            key = normalize_key(department)
            if key not in canonical:
                canonical[key] = department
            display = canonical[key]
            counts[display] = counts.get(display, 0) + 1
        return counts

from __future__ import annotations

from lottery_system.constants import DEFAULT_PRIZE_RECORDS
from lottery_system.models import Prize, normalize_key


class PrizeManager:
    """Manages the ordered list of prizes."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity in DEFAULT_PRIZE_RECORDS:
            try:
                self.add_prize(name, quantity)
            except ValueError as exc:
                errors.append(f"Prize Error: '{name}': {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize to the end of the lottery sequence."""
        prize_name = (name or "").strip()
        if not prize_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if any(normalize_key(p.name) == normalize_key(prize_name) for p in self._prizes):
            raise ValueError(f"Prize '{prize_name}' already exists.")

        groups: list[str] = []
        seen_groups: set[str] = set()
        for group in avoid_groups or []:
            cleaned = " ".join((group or "").split())
            key = normalize_key(cleaned)
            if cleaned and key not in seen_groups:
                seen_groups.add(key)
                groups.append(cleaned)

        drawer = None
        if associated_drawer and associated_drawer.strip():
            drawer = associated_drawer.strip()

        self._prizes.append(
            Prize(
                name=prize_name,
                quantity=quantity,
                avoid_groups=groups,
                associated_drawer=drawer,
            )
        )

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. ``order`` holds 1-based current positions."""
        count = len(self._prizes)
        try:
            if len(order) != count or set(order) != set(range(1, count + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

from __future__ import annotations

import random

from lottery_system.constants import (
    CHEN_QIXIAN,
    CHENG_RU_GROUP,
    JIN_DUO,
    LI_BIN,
    MA_JIAQI,
)
from lottery_system.models import Prize, User, normalize_key


def _in_group(user: User, group: str) -> bool:
    return normalize_key(user.department) == normalize_key(group)


def _same_person(a: str | None, b: str) -> bool:
    return normalize_key(a or "") == normalize_key(b)


class LotteryManager:
    """Applies lottery rules and tracks winners across all rounds."""

    def __init__(self, users: list[User], rng: random.Random | None = None) -> None:
        self._users: list[User] = list(users)
        self._winners: list[User] = []
        self._winner_keys: set[str] = set()
        self._rng = rng or random.Random()
        self._current_prize: Prize | None = None
        self._round_winners: list[User] = []
        # Results of the two most recent rounds run by Chen Qixian / Jin Duo.
        # True means that round contained a Cheng Ru group member.
        self.last_two_draw_results: list[bool] = []

    @property
    def winners(self) -> list[User]:
        """All winners drawn so far, in winning order."""
        return list(self._winners)

    def _is_winner(self, user: User) -> bool:
        return normalize_key(user.name) in self._winner_keys

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the candidates eligible for the next pick of ``prize``."""
        base = [user for user in self._users if not self._is_winner(user)]
        drawer = prize.associated_drawer

        # Special rule: Li Bin only filters previous winners.
        if _same_person(drawer, LI_BIN):
            return base

        # Cheng Ru group protection for Ma Jiaqi's prize pool.
        if _same_person(drawer, MA_JIAQI) and self.last_two_draw_results == [False, False]:
            forced = [user for user in base if _in_group(user, CHENG_RU_GROUP)]
            if forced:
                return forced

        # Avoidance groups: their members remain eligible but are drawn after
        # all other members.
        preferred = [
            user
            for user in base
            if not any(_in_group(user, group) for group in prize.avoid_groups)
        ]
        if preferred:
            return preferred
        return base

    def start_round(self, prize: Prize) -> None:
        """Begin a lottery round for ``prize``."""
        self._current_prize = prize
        self._round_winners = []

    def draw_next(self) -> User | None:
        """Draw one winner for the active round. Returns None if no candidate."""
        if self._current_prize is None:
            raise RuntimeError("No active round. Call start_round() first.")
        pool = self._get_eligible_pool(self._current_prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self._winners.append(user)
        self._winner_keys.add(normalize_key(user.name))
        self._round_winners.append(user)
        return user

    def end_round(self) -> list[User]:
        """Finish the active round and return its winners."""
        if self._current_prize is None:
            return []
        prize = self._current_prize
        if _same_person(prize.associated_drawer, CHEN_QIXIAN) or _same_person(
            prize.associated_drawer, JIN_DUO
        ):
            hit = any(_in_group(user, CHENG_RU_GROUP) for user in self._round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

        result = list(self._round_winners)
        self._current_prize = None
        self._round_winners = []
        return result

"""Winner recording, ordering and CSV export."""

from __future__ import annotations

import csv
import datetime
import os
import re
from dataclasses import dataclass

from lottery_system.models import Prize, User

_LEVEL_WORDS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}
_LEVEL_PATTERN = re.compile(r"^\s*([A-Za-z]+)\b")
_UNRANKED = 99


def prize_level(prize_name: str) -> int:
    """Return the prize rank derived from its name (1 = highest).

    Names without a recognised ordinal word (e.g. "Custom Gift") rank last.
    """
    match = _LEVEL_PATTERN.match(prize_name or "")
    if not match:
        return _UNRANKED
    return _LEVEL_WORDS.get(match.group(1).lower(), _UNRANKED)


@dataclass
class Winner:
    """A single winning record."""

    user: User
    prize_name: str
    sequence: int
    won_at: datetime.datetime


class ResultManager:
    """Stores winners and produces ordered results."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._sequence = 0

    def add_winner(self, user: User, prize: Prize) -> None:
        """Record that ``user`` won ``prize``."""
        self._sequence += 1
        prize_name = prize.name if isinstance(prize, Prize) else str(prize)
        self._winners.append(
            Winner(
                user=user,
                prize_name=prize_name,
                sequence=self._sequence,
                won_at=datetime.datetime.now(),
            )
        )

    def get_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return sorted(self._winners, key=lambda w: w.sequence)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map each prize name to its winners in winning-time order."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            winners = [w for w in self._winners if w.prize_name == prize.name]
            result[prize.name] = sorted(winners, key=lambda w: w.sequence)
        return result

    def get_ordered_winners(self, prizes: list[Prize]) -> list[Winner]:
        """Winners ordered by prize level (high to low), then winning time."""
        indexed = sorted(
            enumerate(prizes),
            key=lambda item: (prize_level(item[1].name), item[0]),
        )
        by_prize = self.get_winners_by_prize(prizes)
        ordered: list[Winner] = []
        for _, prize in indexed:
            ordered.extend(by_prize.get(prize.name, []))
        return ordered

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to current_dir/awards_result_YYYYMMDD.txt."""
        date_text = datetime.datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_text}.txt")
        rows = self.get_ordered_winners(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8-sig") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for winner in rows:
                    writer.writerow(
                        [
                            winner.user.name,
                            winner.user.department,
                            winner.prize_name,
                            winner.won_at.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

MENU = """

def _ask(prompt: str) -> str:
    return input(prompt).strip()


def _ordinal(number: int) -> str:
    if 10 <= number % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(number % 10, "th")
    return f"{number}{suffix}"


def _same(a: str | None, b: str) -> bool:
    return normalize_key(a or "") == normalize_key(b)


class LotteryCLI:
    """Interactive command-line controller."""

    def __init__(self) -> None:
        self._list_manager = ListManager()
        self._prize_manager = PrizeManager()
        self._users: list[User] = []
        self._results: ResultManager | None = None
        self._last_prizes: list[Prize] = []

    # ------------------------------------------------------------------ main loop
    def run(self) -> None:
        print(BANNER)
        actions = {
            "1": self._load_participants,
            "2": self._show_participants,
            "3": self._load_default_prizes,
            "4": self._add_custom_prize,
            "5": self._view_prizes,
            "6": self._reorder_prizes,
            "7": self._start_lottery,
            "8": self._export_results,
        }
        while True:
            print(MENU)
            choice = _ask("Select an option: ")
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid option, please try again.")
                continue
            action()

    # ------------------------------------------------------- list management
    def _load_participants(self) -> None:
        print(
            "\nLoad participant list:\n"
            "  1. Built-in default list\n"
            "  2. Custom txt file (each line: Name,Department)\n"
            "  0. Cancel"
        )
        choice = _ask("Select: ")
        if choice == "1":
            users, errors = self._list_manager.load_default_list()
        elif choice == "2":
            path = _ask("Enter txt file path: ")
            users, errors = self._list_manager.load_custom_list(path)
        else:
            return

        valid, duplicates = self._list_manager.validate_users(users)
        if errors or duplicates:
            print("\nValidation issues found:")
            for error in errors:
                print(f"  - {error}")
            for duplicate in duplicates:
                print(f"  - Duplicate name: {duplicate.name} ({duplicate.department})")
            print(
                f"Total: {len(errors)} format error(s), "
                f"{len(duplicates)} duplicate(s)."
            )
            print(
                "  1. Remove invalid/duplicate entries and continue\n"
                "  2. Terminate upload (keep current list)"
            )
            if _ask("Select: ") != "1":
                print("Upload terminated. Participant list unchanged.")
                return

        if not valid:
            print("No valid participants found. Participant list unchanged.")
            return
        self._users = valid
        print(f"Loaded {len(valid)} participant(s).")

    def _show_participants(self) -> None:
        if not self._users:
            print("No participant list loaded.")
            return
        print(f"\nTotal participants: {len(self._users)}")
        print("Department groups:")
        for department, count in self._list_manager.get_group_info(self._users).items():
            print(f"  - {department}: {count} person(s)")

    # ------------------------------------------------------- prize management
    def _load_default_prizes(self) -> None:
        errors = self._prize_manager.load_default_prizes()
        for error in errors:
            print(error)
        if not errors:
            print("Default prizes loaded.")

    def _add_custom_prize(self) -> None:
        name = _ask("Prize name: ")
        quantity = self._ask_positive_int("Quantity: ")
        groups = self._choose_avoid_groups()
        drawer = self._choose_drawer()
        try:
            self._prize_manager.add_prize(name, quantity, groups, drawer)
        except ValueError as exc:
            print(f"Error: {exc}")
            return
        print(f"Prize '{name}' added to position {len(self._prize_manager.get_prizes())}.")

    @staticmethod
    def _ask_positive_int(prompt: str) -> int:
        while True:
            raw = _ask(prompt)
            if raw.isdigit() and int(raw) > 0:
                return int(raw)
            print("Please enter a positive integer.")

    def _choose_avoid_groups(self) -> list[str]:
        groups = list(self._list_manager.get_group_info(self._users).keys()) if self._users else []
        if groups:
            print("Available department groups:")
            for index, group in enumerate(groups, start=1):
                print(f"  {index}. {group}")
        raw = _ask("Avoidance groups (numbers or names, comma-separated; blank for none): ")
        if not raw:
            return []
        selected: list[str] = []
        for token in (part.strip() for part in raw.split(",")):
            if not token:
                continue
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                selected.append(groups[int(token) - 1])
            else:
                selected.append(token)
        return selected

    def _choose_drawer(self) -> str | None:
        print("Associated lottery person:")
        for index, person in enumerate(LOTTERY_PERSONS, start=1):
            print(f"  {index}. {person}")
        print("  0. None")
        raw = _ask("Select: ")
        if raw in ("", "0"):
            return None
        if raw.isdigit() and 1 <= int(raw) <= len(LOTTERY_PERSONS):
            return LOTTERY_PERSONS[int(raw) - 1]
        for person in LOTTERY_PERSONS:
            if _same(raw, person):
                return person
        print("Invalid selection; no lottery person assigned.")
        return None

    def _view_prizes(self) -> None:
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nPrize sequence:")
        for index, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
            drawer = prize.associated_drawer or "None"
            print(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoidance groups: {groups} | Lottery person: {drawer}"
            )

    def _reorder_prizes(self) -> None:
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._view_prizes()
        raw = _ask("Enter new order as serial numbers, e.g. '3 1 2': ")
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(token) for token in tokens]
        except ValueError:
            print("Invalid input: please enter numbers only.")
            return
        if self._prize_manager.reorder_prizes(order):
            print("Prize order updated.")
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ---------------------------------------------------------- lottery flow
    def _start_lottery(self) -> None:
        if not self._users:
            print("Please load a participant list first.")
            return
        prizes = self._prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured. Load default prizes or add custom prizes first.")
            return

        lottery = LotteryManager(self._users)
        results = ResultManager()
        self._results = results
        self._last_prizes = prizes

        print("\nLottery started!")
        print("Commands: Enter = draw next winner, 'pause' = pause, "
              "'terminate' = end current round.")
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, len(prizes), prize, lottery, results)

        print("\nAll rounds completed.")
        self._show_summary(prizes, results)
        if _ask("Export results now? (y/n): ").lower() == "y":
            self._export_results()

    def _run_round(
        self,
        index: int,
        total: int,
        prize: Prize,
        lottery: LotteryManager,
        results: ResultManager,
    ) -> None:
        print("\n" + "=" * 60)
        print(f"Round {index}/{total}: {prize.name}")
        print(f"Quantity: {prize.quantity}")
        print(f"Rule: {self._describe_rule(prize)}")
        print("=" * 60)

        lottery.start_round(prize)
        terminated = False
        for position in range(1, prize.quantity + 1):
            print(f"Currently drawing the {_ordinal(position)} winner...")
            if not self._await_command():
                terminated = True
                break
            user = lottery.draw_next()
            if user is None:
                print("No eligible candidates remain for this prize.")
                break
            results.add_winner(user, prize)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        round_winners = lottery.end_round()
        if terminated:
            print("Round terminated by operator.")
        print(f"Round finished: {len(round_winners)}/{prize.quantity} winner(s) drawn.")

    @staticmethod
    def _await_command() -> bool:
        """Handle operator commands before a draw. Returns False to terminate."""
        while True:
            command = _ask("Press Enter to draw (or type pause / terminate): ").lower()
            if command in ("", "continue"):
                return True
            if command == "terminate":
                return False
            if command == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    command = _ask("(paused) > ").lower()
                    if command == "continue":
                        print("Lottery resumed.")
                        return True
                    if command == "terminate":
                        return False
                    print("Unknown command. Use 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    @staticmethod
    def _describe_rule(prize: Prize) -> str:
        drawer = prize.associated_drawer
        if _same(drawer, LI_BIN):
            parts = [
                "Only previous winners are excluded "
                "(Li Bin rule: avoidance groups do not apply)."
            ]
        else:
            parts = ["Previous winners are excluded."]
            if prize.avoid_groups:
                parts.append(
                    "Members of avoidance group(s) "
                    f"{', '.join(prize.avoid_groups)} are drawn after other members."
                )
        if drawer:
            parts.append(f"Lottery person: {drawer}.")
        if _same(drawer, MA_JIAQI):
            parts.append(
                "Cheng Ru protection: if neither of the last two "
                f"{CHEN_QIXIAN}/{JIN_DUO} rounds produced a {CHENG_RU_GROUP} winner, "
                f"only {CHENG_RU_GROUP} members are eligible."
            )
        return " ".join(parts)

    def _show_summary(self, prizes: list[Prize], results: ResultManager) -> None:
        print("\n==================== Final Winner List ====================")
        by_prize = results.get_winners_by_prize(prizes)
        for prize in sorted(prizes, key=lambda p: prize_level(p.name)):
            winners = by_prize.get(prize.name, [])
            print(f"\n{prize.name} ({len(winners)}/{prize.quantity})")
            if not winners:
                print("  (no winners)")
            for winner in winners:
                print(f"  {winner.user.name} - {winner.user.department}")

    # ------------------------------------------------------------- export
    def _export_results(self) -> None:
        if self._results is None:
            print("No lottery results yet.")
            return
        _, message = self._results.export_to_csv(self._last_prizes)
        print(message)


def main() -> None:
    """Program entry point."""

        LotteryCLI().run()

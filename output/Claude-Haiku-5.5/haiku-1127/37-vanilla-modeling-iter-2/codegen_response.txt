from lottery_system.data import DEFAULT_LIST_TEXT
from lottery_system.models import User


class ListManager:
    """Loads and verifies participant lists."""

    def __init__(self):
        self.source_type: str | None = None
        self.file_path: str | None = None
        self.is_valid: bool = False

    def load_default_list(self) -> tuple[list[User], list[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        self.source_type = "default"
        self.file_path = None
        self.is_valid = bool(users)
        return users, errors

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        users, errors = self._load_from_path(path)
        self.source_type = "custom"
        self.file_path = path
        self.is_valid = bool(users)
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            if not raw.strip():
                continue
            parts = raw.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_no}: Format Error - '{raw.strip()}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Return (valid_users, duplicates). The first occurrence of a name is valid;
        later occurrences with the same name are reported as duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip().casefold()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the prize library and lottery round order. Starts empty."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        existing = {p.name.casefold() for p in self._prizes}
        for name, quantity, drawer in DEFAULT_PRIZES:
            if name.casefold() in existing:
                continue
            try:
                self.add_prize(name, quantity, [], drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        name = (name or "").strip()
        if not name:
            raise ValueError("prize name must not be empty")
        if not isinstance(quantity, int) or isinstance(quantity, bool) or quantity < 1:
            raise ValueError("quantity must be a positive integer")
        groups: list[str] = []
        for group in avoid_groups or []:
            cleaned = group.strip()
            if cleaned and cleaned not in groups:
                groups.append(cleaned)
        drawer = associated_drawer.strip() if associated_drawer else None
        if drawer == "":
            drawer = None
        prize = Prize(name, quantity, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """order[k] is the 1-based current position of the prize that should
        occupy new position k+1."""
        n = len(self._prizes)
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[pos - 1] for pos in order]
        return True

"""Winner records, ordering, display grouping and CSV/TXT export."""

import csv
import os
import re
from dataclasses import dataclass
from datetime import date, datetime

from lottery_system.models import Prize, User

_LEVEL_WORDS = {"first": 1, "second": 2, "third": 3}


def prize_level(name: str) -> int:
    """Map 'First'/'Second'/'Third' in a prize name to level 1/2/3.
    Prizes without a level word are ranked after leveled prizes."""
    match = re.search(r"\b(first|second|third)\b", name, re.IGNORECASE)
    if match:
        return _LEVEL_WORDS[match.group(1).lower()]
    return 99


@dataclass
class Winner:
    user: User
    prize: Prize
    winning_time: datetime
    sequence: int


def _ordered_prizes(prizes: list[Prize]) -> list[Prize]:
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda pair: (prize_level(pair[1].name), pair[0]))
    return [prize for _, prize in indexed]


class ResultManager:
    """Keeps all winners in real time and produces ordered views/exports."""

    HEADER = ["Name", "Department", "Prize Name", "Winning Time"]

    def __init__(self):
        self._winners: list[Winner] = []
        self._counter = 0

    @property
    def winners(self) -> list[Winner]:
        return list(self._winners)

    def add_winner(self, user: User, prize: Prize) -> Winner:
        self._counter += 1
        winner = Winner(user, prize, datetime.now(), self._counter)
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Keys are prize names ordered by level (high to low, i.e. First first);
        values are winners of that prize in winning-time order."""
        result: dict[str, list[Winner]] = {}
        for prize in _ordered_prizes(prizes):
            bucket = result.setdefault(prize.name, [])
            bucket.extend(w for w in self._winners if w.prize is prize)
        return result

    def _ordered_winners(self, prizes: list[Prize]) -> list[Winner]:
        ordered: list[Winner] = []
        for bucket in self.get_winners_by_prize(prizes).values():
            ordered.extend(bucket)
        return ordered

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        winners = self._ordered_winners(prizes)
        if not winners:
            return False, "No winners to export."
        filename = f"awards_result_{date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(self.HEADER)
                for winner in winners:
                    writer.writerow([
                        winner.user.name,
                        winner.user.department,
                        winner.prize.name,
                        f"{winner.winning_time:%Y-%m-%d %H:%M:%S}",
                    ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and draw execution."""

import random

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager, Winner

CHENG_RU_GROUP = "Cheng Ru Group"
LI_BIN = "Li Bin"
FALLBACK_DRAWER = "Ma Jiaqi"
TRACKED_DRAWERS = ("Chen Qixian", "Jin Duo")
CONSECUTIVE_THRESHOLD = 2


def _same(a: str, b: str) -> bool:
    return a.strip().casefold() == b.strip().casefold()


class LotteryManager:
    """Runs lottery draws while enforcing the lottery rules."""

    def __init__(self, users: list[User]):
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Results (True = contains Cheng Ru group member) of the most recent
        # draws by Chen Qixian / Jin Duo, at most CONSECUTIVE_THRESHOLD entries.
        self.last_two_draw_results: list[bool] = []
        self._winner_names: set[str] = set()

    @property
    def winner_names(self) -> set[str]:
        return set(self._winner_names)

    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    @staticmethod
    def is_cheng_ru(user: User) -> bool:
        return _same(user.department, CHENG_RU_GROUP)

    def _fallback_active(self, prize: Prize) -> bool:
        drawer = prize.associated_drawer
        if not drawer or not _same(drawer, FALLBACK_DRAWER):
            return False
        if len(self.last_two_draw_results) < CONSECUTIVE_THRESHOLD:
            return False
        return not any(self.last_two_draw_results[-CONSECUTIVE_THRESHOLD:])

    def _compute_pool(self, prize: Prize) -> tuple[list[User], bool]:
        """Return (eligible users, whether the Cheng Ru protection was applied)."""
        available = [u for u in self.users if u.name not in self._winner_names]
        drawer = prize.associated_drawer

        # Special rule: Li Bin only filters previous winners.
        if drawer and _same(drawer, LI_BIN):
            return available, False

        # Fallback: force Cheng Ru group members for Ma Jiaqi.
        if self._fallback_active(prize):
            forced = [u for u in available if self.is_cheng_ru(u)]
            if forced:
                return forced, True

        avoid = {g.strip().casefold() for g in prize.avoid_groups}
        pool = [u for u in available if u.department.strip().casefold() not in avoid]
        return pool, False

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        return self._compute_pool(prize)[0]

    def describe_rule(self, prize: Prize) -> str:
        parts: list[str] = []
        drawer = prize.associated_drawer
        if drawer and _same(drawer, LI_BIN):
            parts.append("previous winners excluded only (avoidance groups not applied)")
        else:
            parts.append("previous winners excluded")
            if prize.avoid_groups:
                parts.append("avoidance groups: " + ", ".join(prize.avoid_groups))
            if drawer and _same(drawer, FALLBACK_DRAWER):
                parts.append("Cheng Ru Group protection applies if Chen Qixian/Jin Duo "
                             "produced no Cheng Ru Group winner in their last two draws")
        if drawer:
            parts.append(f"lottery person: {drawer}")
        return "; ".join(parts)

    def draw(self, prize: Prize) -> User | None:
        pool, forced = self._compute_pool(prize)
        if not pool:
            return None
        winner_user = random.choice(pool)
        self._winner_names.add(winner_user.name)
        self.result_manager.add_winner(winner_user, prize)

        drawer = prize.associated_drawer
        if drawer and any(_same(drawer, d) for d in TRACKED_DRAWERS):
            self.last_two_draw_results.append(self.is_cheng_ru(winner_user))
            self.last_two_draw_results = self.last_two_draw_results[-CONSECUTIVE_THRESHOLD:]
        elif forced:
            # Protection has been used; restart the consecutive-failure window.
            self.last_two_draw_results = []
        return winner_user

    def get_winner_records(self) -> list[Winner]:
        return self.result_manager.winners

"""Command-line interface for the lottery process."""

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import Winner

MENU = """
==========================================
 Team Building Smart Lottery System
==========================================
 1. Load participant list
 2. View department groups
 3. Prize configuration
 4. Adjust prize order
 5. Start lottery
 6. View winners / export results
 0. Exit
"""


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery: LotteryManager | None = None
        self.last_prizes: list = []

    # ---------- helpers ----------
    @staticmethod
    def _ask(prompt: str) -> str:
        return input(prompt)

    def _group_map(self) -> dict[str, str]:
        groups = self.list_manager.get_group_info(self.users)
        return {name.casefold(): name for name in groups}

    # ---------- main loop ----------
    def run(self) -> None:
        print("Default prizes are not loaded automatically. Use menu 3 to load them.")
        while True:
            print(MENU)
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self._load_list()
            elif choice == "2":
                self._show_groups()
            elif choice == "3":
                self._prize_menu()
            elif choice == "4":
                self._reorder_prizes()
            elif choice == "5":
                self._start_lottery()
            elif choice == "6":
                self._winners_menu()
            elif choice == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid option, please try again.")

    # ---------- list management ----------
    def _load_list(self) -> None:
        print("1. Built-in default team building list")
        print("2. Custom txt file (each line: Name,Department Group)")
        choice = self._ask("Select source (1/2): ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip().strip('"')
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Invalid choice. List not changed.")
            return

        valid, duplicates = self.list_manager.validate_users(users)
        issues = list(errors) + [
            f"Duplicate name: '{u.name}' ({u.department})" for u in duplicates
        ]
        if issues:
            print("Verification found the following problems:")
            for issue in issues:
                print(f"  - {issue}")
            print(f"Valid entries: {len(valid)}")
            answer = self._ask(
                "Type 'remove' to discard invalid/duplicate data and continue, "
                "or 'terminate' to cancel the upload: "
            ).strip().lower()
            if answer != "remove":
                print("Upload terminated. Previous list kept.")
                return

        if not valid:
            print("No valid participants found. List not changed.")
            return
        self.users = valid
        print(f"Loaded {len(valid)} participants.")

    def _show_groups(self) -> None:
        if not self.users:
            print("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for group, count in info.items():
            print(f"  - {group}: {count} people")
        print(f"Total: {len(self.users)} people")

    # ---------- prize configuration ----------
    def _prize_menu(self) -> None:
        while True:
            print("\n--- Prize configuration ---")
            print("1. View prizes")
            print("2. Load preset prizes")
            print("3. Add custom prize")
            print("0. Back")
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self._show_prizes()
            elif choice == "2":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for err in errors:
                        print(f"  Failed: {err}")
                else:
                    print("Preset prizes loaded.")
            elif choice == "3":
                self._add_prize()
            elif choice == "0":
                return
            else:
                print("Invalid option.")

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for idx, prize in enumerate(prizes, start=1):
            groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            print(f"  {idx}. {prize.name} | quantity: {prize.quantity} | "
                  f"avoid groups: {groups} | lottery person: {drawer}")

    def _add_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        qty_text = self._ask("Quantity: ").strip()
        try:
            quantity = int(qty_text)
        except ValueError:
            print("Quantity must be an integer.")
            return

        avoid: list[str] = []
        groups_text = self._ask(
            "Avoidance groups (comma-separated group names, empty for none): "
        ).strip()
        if groups_text:
            known = self._group_map()
            if not known:
                print("No participant list loaded; avoidance groups ignored.")
            for raw in groups_text.split(","):
                key = raw.strip().casefold()
                if not key:
                    continue
                if key in known:
                    avoid.append(known[key])
                else:
                    print(f"Warning: unknown group '{raw.strip()}' ignored.")

        drawer = self._ask("Associated lottery person (empty for none): ").strip() or None
        try:
            prize = self.prize_manager.add_prize(name, quantity, avoid, drawer)
            print(f"Prize added: {prize.name} (quantity {prize.quantity})")
        except ValueError as exc:
            print(f"Failed to add prize: {exc}")

    # ---------- ordering ----------
    def _reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self._show_prizes()
        text = self._ask(
            "Enter new order as serial numbers separated by spaces (empty to keep): "
        ).strip()
        if not text:
            print("Order unchanged.")
            return
        try:
            order = [int(x) for x in text.replace(",", " ").split()]
        except ValueError:
            print("Serial numbers must be integers.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated:")
            self._show_prizes()
        else:
            print("Invalid order: each serial number must appear exactly once.")

    # ---------- lottery process ----------
    def _start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lottery = LotteryManager(self.users)
        self.last_prizes = prizes
        print("\nLottery started!")
        for round_no, prize in enumerate(prizes, start=1):
            print(f"\n--- Round {round_no}/{len(prizes)} ---")
            self._run_round(lottery, round_no, prize)
        self.lottery = lottery

        print("\nAll rounds completed.")
        self._print_summary(lottery, prizes)
        if self._ask("Export results to txt now? (y/n): ").strip().lower() == "y":
            self._export(lottery, prizes)

    def _run_round(self, lottery: LotteryManager, round_no: int, prize) -> None:
        print(f"Prize: {prize.name} | Quantity: {prize.quantity}")
        print(f"Rule: {lottery.describe_rule(prize)}")
        drawn: list = []
        while len(drawn) < prize.quantity:
            target = ordinal(len(drawn) + 1)
            cmd = self._ask(
                f"Press Enter to draw the {target} winner "
                f"(commands: pause / terminate): "
            ).strip().lower()
            if cmd == "pause":
                cmd = self._wait_while_paused()
            if cmd == "terminate":
                print("Round terminated. Skipping remaining winners of this prize.")
                break
            if cmd not in ("", "continue"):
                print(f"Unknown command '{cmd}'.")
                continue

            print(f"Currently drawing the {target} winner...")
            user = lottery.draw(prize)
            if user is None:
                print("No eligible participants remain for this prize. Round ends.")
                break
            drawn.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        print(f"Round {round_no} results ({prize.name}):")
        if not drawn:
            print("  (no winners)")
        for idx, user in enumerate(drawn, start=1):
            print(f"  {idx}. {user.name} ({user.department})")

    def _wait_while_paused(self) -> str:
        print("Lottery paused.")
        while True:
            cmd = self._ask(
                "Type 'continue' to resume or 'terminate' to skip this round: "
            ).strip().lower()
            if cmd == "continue":
                print("Lottery resumed.")
                return "continue"
            if cmd == "terminate":
                return "terminate"
            print("Please type 'continue' or 'terminate'.")

    # ---------- results ----------
    def _print_summary(self, lottery: LotteryManager, prizes: list) -> None:
        grouped = lottery.result_manager.get_winners_by_prize(prizes)
        print("\n========== Winner List ==========")
        for prize_name, winners in grouped.items():
            print(f"{prize_name}:")
            if not winners:
                print("  (no winners)")
            for idx, w in enumerate(winners, start=1):
                self._print_winner_line(idx, w)
        print("=================================")

    @staticmethod
    def _print_winner_line(idx: int, winner: Winner) -> None:
        print(f"  {idx}. {winner.user.name} ({winner.user.department}) "
              f"- {winner.winning_time:%H:%M:%S}")

    def _export(self, lottery: LotteryManager, prizes: list) -> None:
        ok, message = lottery.result_manager.export_to_csv(prizes)
        print(message)

    def _winners_menu(self) -> None:
        if self.lottery is None:
            print("No lottery results yet.")
            return
        self._print_summary(self.lottery, self.last_prizes)
        if self._ask("Export results to txt? (y/n): ").strip().lower() == "y":
            self._export(self.lottery, self.last_prizes)


def main() -> None:
    try:
        LotteryCLI().run()
    except (KeyboardInterrupt, EOFError):
        print("\nExiting.")

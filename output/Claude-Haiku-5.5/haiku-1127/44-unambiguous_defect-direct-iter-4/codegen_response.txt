"""Loading and saving of the Beijing Subway dataset (src/subway_data.json).

The dataset JSON has the form:
    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str, "distance": int, "line": str}]}
"""

import json
import os

_THIS_DIR = os.path.dirname(os.path.abspath(__file__))

# Absolute path of src/subway_data.json, valid regardless of the working directory.
DATA_FILE_PATH = os.path.join(_THIS_DIR, "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read, parsed or saved."""


def load_subway_data(path=DATA_FILE_PATH) -> dict:
    """Read the subway dataset file and return its JSON content as a dict.

    Args:
        path: Path to the JSON dataset file.

    Returns:
        The parsed dataset dictionary with 'stations' and 'connections' lists.

    Raises:
        DataLoadError: If the file is missing, unreadable, not valid JSON,
            or does not have the expected structure.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Subway data file is not valid JSON: {path} ({exc})") from exc
    except OSError as exc:
        raise DataLoadError(f"Cannot read subway data file: {path} ({exc})") from exc

    if not isinstance(data, dict):
        raise DataLoadError("Subway data must be a JSON object")
    stations = data.get("stations", [])
    connections = data.get("connections", [])
    if not isinstance(stations, list) or not isinstance(connections, list):
        raise DataLoadError("'stations' and 'connections' must both be lists")
    data["stations"] = stations
    data["connections"] = connections
    return data


def save_subway_data(data: dict, path=DATA_FILE_PATH) -> None:
    """Write a dataset dictionary back to a JSON file.

    Args:
        data: Dataset dictionary in the same format returned by load_subway_data.
        path: Destination file path.

    Raises:
        DataLoadError: If the file cannot be written.
    """
    try:
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, ensure_ascii=False, indent=2)
            handle.write("\n")
    except OSError as exc:
        raise DataLoadError(f"Cannot write subway data file: {path} ({exc})") from exc

"""Domain models: a weighted undirected graph and the Beijing Subway network built on it.

Graph nodes are integer station numbers assigned by the SubwayNetwork builder.
Station names are kept in mapping dictionaries and in Graph.station_lines.
"""

import math


class StationNotFoundError(LookupError):
    """Raised when a station name or number does not exist in the network."""


class Edge:
    """A weighted, undirected connection between two stations.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        weight: Distance in meters (the edge weight used by Dijkstra's algorithm).
        line: Name of the subway line the connection belongs to.
    """

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.weight = distance
        self.line = line

    @property
    def distance(self):
        """Distance in meters; identical to ``weight``."""
        return self.weight

    @distance.setter
    def distance(self, value):
        self.weight = value

    def other(self, vertex):
        """Return the endpoint of this edge opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex!r} is not an endpoint of this edge")

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.weight!r}, {self.line!r})"


class Graph:
    """Weighted undirected graph with adjacency lists.

    Station numbers are supplied by the caller. ``station_lines`` maps station
    names to the list of lines serving that station.
    """

    def __init__(self):
        self._adjacency = {}
        self._edges = []
        self.station_lines = {}

    def add_vertex(self, vertex):
        """Add a vertex (station number) if it does not already exist."""
        if vertex not in self._adjacency:
            self._adjacency[vertex] = []

    def has_vertex(self, vertex) -> bool:
        """Return True if the vertex exists in the graph."""
        return vertex in self._adjacency

    def vertices(self) -> list:
        """Return a list of all vertices."""
        return list(self._adjacency)

    def set_station_lines(self, station_name, lines):
        """Record the lines serving a station, identified by its name."""
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge: Edge):
        """Add an undirected edge; it is visible from both endpoints.

        Raises:
            ValueError: For self-loops or negative weights.
        """
        if edge.source == edge.dest:
            raise ValueError("Self-loop edges are not allowed")
        if edge.weight < 0:
            raise ValueError("Edge weight must be non-negative")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adjacency[edge.source].append(edge)
        self._adjacency[edge.dest].append(edge)
        self._edges.append(edge)

    def adjacent(self, vertex) -> list:
        """Return the list of edges incident to ``vertex`` (do not modify it)."""
        if vertex not in self._adjacency:
            raise KeyError(f"Unknown vertex {vertex!r}")
        return self._adjacency[vertex]

    def find_edges(self, a, b, line=None) -> list:
        """Return all edges between vertices ``a`` and ``b``, optionally on one line."""
        return [
            edge
            for edge in self._adjacency.get(a, [])
            if edge.other(a) == b and (line is None or edge.line == line)
        ]

    def remove_edge(self, edge: Edge):
        """Remove an edge from the graph."""
        self._adjacency[edge.source].remove(edge)
        self._adjacency[edge.dest].remove(edge)
        self._edges.remove(edge)

    def edges(self) -> list:
        """Return a list of all distinct edges."""
        return list(self._edges)

    def num_vertices(self) -> int:
        """Return the number of vertices."""
        return len(self._adjacency)

    def num_edges(self) -> int:
        """Return the number of distinct edges."""
        return len(self._edges)


def _clean_name(name) -> str:
    """Normalize a station or line name by converting to str and stripping spaces."""
    if name is None:
        return ""
    return str(name).strip()


def _clean_lines(lines) -> list:
    """Return a de-duplicated list of cleaned, non-empty line names in order."""
    if isinstance(lines, str):
        lines = [lines]
    result = []
    for line in lines or []:
        cleaned = _clean_name(line)
        if cleaned and cleaned not in result:
            result.append(cleaned)
    return result


def _validate_distance(distance) -> int:
    """Validate a distance in meters and return it as a positive integer."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("Distance must be a number of meters")
    if not math.isfinite(distance):
        raise ValueError("Distance must be finite")
    value = int(round(distance))
    if value <= 0:
        raise ValueError("Distance must be positive")
    return value


class SubwayNetwork:
    """The Beijing Subway network: a Graph plus name/number mappings and line indexes.

    Attributes:
        graph: The weighted undirected Graph keyed by station numbers.
        station_to_number: Mapping station name -> station number.
        number_to_station: Mapping station number -> station name.
        line_stations: Mapping line name -> list of station names on that line.
        build_warnings: Messages describing records skipped while loading.
    """

    def __init__(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        self.line_stations = {}
        self.build_warnings = []
        self._next_number = 0

    @classmethod
    def from_data(cls, data: dict):
        """Build a network from a dataset dictionary (see data_loader)."""
        network = cls()
        network._load(data)
        return network

    def _load(self, data: dict):
        """Populate the network from stations and connections, recording problems."""
        for item in data.get("stations", []):
            if not isinstance(item, dict) or not isinstance(item.get("name"), str) \
                    or not item["name"].strip():
                self.build_warnings.append(f"Skipped station record without a valid name: {item!r}")
                continue
            try:
                self.add_station(item["name"], item.get("lines", []))
            except ValueError as exc:
                self.build_warnings.append(f"Skipped station {item['name']!r}: {exc}")

        for conn in data.get("connections", []):
            try:
                source = conn["source"]
                dest = conn["dest"]
                distance = conn["distance"]
                line = conn["line"]
            except (KeyError, TypeError):
                self.build_warnings.append(f"Skipped malformed connection record: {conn!r}")
                continue
            for name in (source, dest):
                if not self.has_station(name) and _clean_name(name):
                    self.add_station(name, [line])
                    self.build_warnings.append(
                        f"Station '{_clean_name(name)}' is referenced by a connection "
                        f"but not listed; it was added automatically"
                    )
            try:
                self.add_connection(source, dest, distance, line)
            except ValueError as exc:
                self.build_warnings.append(
                    f"Skipped connection {_clean_name(source)} - {_clean_name(dest)} "
                    f"({_clean_name(line)}): {exc}"
                )

    def _index_line(self, line: str, station_name: str):
        """Record that ``station_name`` is served by ``line``."""
        stations = self.line_stations.setdefault(line, [])
        if station_name not in stations:
            stations.append(station_name)

    def _ensure_station_line(self, station_name: str, line: str):
        """Make sure the station's line list contains ``line``."""
        current = self.graph.station_lines[station_name]
        if line not in current:
            self.graph.set_station_lines(station_name, current + [line])
        self._index_line(line, station_name)

    def add_station(self, name, lines=()) -> int:
        """Add a station (or merge lines into an existing one) and return its number."""
        name = _clean_name(name)
        if not name:
            raise ValueError("Station name must not be empty")
        lines = _clean_lines(lines)
        if name in self.station_to_number:
            number = self.station_to_number[name]
            current = self.graph.station_lines[name]
            self.graph.set_station_lines(name, current + [l for l in lines if l not in current])
        else:
            number = self._next_number
            self._next_number += 1
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
            self.graph.set_station_lines(name, lines)
        for line in lines:
            self._index_line(line, name)
        return number

    def add_connection(self, source, dest, distance, line):
        """Add a connection between two existing stations on a line.

        Raises:
            StationNotFoundError: If a station does not exist.
            ValueError: For invalid distances, self-connections or duplicates.
        """
        source_name = _clean_name(source)
        dest_name = _clean_name(dest)
        line = _clean_name(line)
        a = self.get_station_number(source_name)
        b = self.get_station_number(dest_name)
        if not line:
            raise ValueError("Line name must not be empty")
        if a == b:
            raise ValueError("Source and destination are the same station")
        value = _validate_distance(distance)
        if self.graph.find_edges(a, b, line):
            raise ValueError(f"Connection already exists on {line}")
        self.graph.add_edge(Edge(a, b, value, line))
        self._ensure_station_line(source_name, line)
        self._ensure_station_line(dest_name, line)

    def has_station(self, name) -> bool:
        """Return True if a station with this name exists."""
        return _clean_name(name) in self.station_to_number

    def get_station_number(self, name) -> int:
        """Return the station number for a station name."""
        key = _clean_name(name)
        if key not in self.station_to_number:
            raise StationNotFoundError(f"Station '{key}' does not exist")
        return self.station_to_number[key]

    def station_name(self, number) -> str:
        """Return the station name for a station number."""
        if number not in self.number_to_station:
            raise StationNotFoundError(f"No station with number {number!r}")
        return self.number_to_station[number]

    def station_names(self) -> list:
        """Return all station names sorted alphabetically."""
        return sorted(self.station_to_number)

    def lines(self) -> list:
        """Return all line names sorted alphabetically."""
        return sorted(self.line_stations)

    def stations_on_line(self, line) -> list:
        """Return the station names served by a line."""
        return list(self.line_stations.get(line, []))

    def station_lines(self, name) -> list:
        """Return the lines serving a station."""
        key = _clean_name(name)
        if key not in self.graph.station_lines:
            raise StationNotFoundError(f"Station '{key}' does not exist")
        return list(self.graph.station_lines[key])

    def is_transfer_station(self, name) -> bool:
        """Return True if the station is served by more than one line."""
        return len(self.station_lines(name)) > 1

    def transfer_stations(self) -> list:
        """Return the sorted names of all multi-line interchange stations."""
        return sorted(
            name for name, lines in self.graph.station_lines.items() if len(lines) > 1
        )

    def add_line(self, line_name, station_names, distances):
        """Add a new line with an ordered station sequence and consecutive distances.

        Existing stations are reused (the new line is added to them); unknown
        stations are created.

        Args:
            line_name: Name of the new line (must not already exist).
            station_names: Ordered list of station names along the line.
            distances: Distances in meters between consecutive stations.

        Raises:
            ValueError: If the input is inconsistent. Nothing is changed on error.
        """
        line = _clean_name(line_name)
        if not line:
            raise ValueError("Line name must not be empty")
        if line in self.line_stations:
            raise ValueError(f"Line '{line}' already exists")
        names = [_clean_name(n) for n in station_names]
        if len(names) < 2:
            raise ValueError("A line needs at least two stations")
        if any(not n for n in names):
            raise ValueError("Station names must not be empty")
        if len(set(names)) != len(names):
            raise ValueError("Station sequence must not contain duplicates")
        if len(distances) != len(names) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        values = [_validate_distance(d) for d in distances]

        for name in names:
            if self.has_station(name):
                self._ensure_station_line(name, line)
            else:
                self.add_station(name, [line])
        for i, value in enumerate(values):
            self.add_connection(names[i], names[i + 1], value, line)

    def modify_station(self, old_name, new_name=None, lines=None) -> str:
        """Rename a station and/or replace the lines serving it.

        Args:
            old_name: Current station name.
            new_name: New name (optional).
            lines: New list of lines (optional; replaces the existing list).

        Returns:
            The resulting station name.

        Raises:
            StationNotFoundError: If the station does not exist.
            ValueError: If the new name is taken or the new line list is empty.
        """
        old = _clean_name(old_name)
        number = self.get_station_number(old)
        target = _clean_name(new_name) if new_name else old
        if target != old and target in self.station_to_number:
            raise ValueError(f"Station '{target}' already exists")
        if lines is not None:
            new_lines = _clean_lines(lines)
            if not new_lines:
                raise ValueError("A station must be served by at least one line")
        else:
            new_lines = list(self.graph.station_lines[old])
        old_lines = list(self.graph.station_lines[old])

        if target != old:
            del self.station_to_number[old]
            self.station_to_number[target] = number
            self.number_to_station[number] = target
            self.graph.station_lines[target] = self.graph.station_lines.pop(old)
            for line, names in self.line_stations.items():
                self.line_stations[line] = [target if n == old else n for n in names]

        for line in old_lines:
            if line not in new_lines and line in self.line_stations:
                self.line_stations[line] = [n for n in self.line_stations[line] if n != target]
        self.graph.set_station_lines(target, new_lines)
        for line in new_lines:
            self._index_line(line, target)
        return target

    def update_distance(self, source, dest, distance, line=None) -> int:
        """Update the distance of the connection(s) between two stations.

        Args:
            source: Name of one station.
            dest: Name of the other station.
            distance: New distance in meters.
            line: Optional line to restrict the update to.

        Returns:
            The number of connections updated.

        Raises:
            StationNotFoundError: If a station does not exist.
            ValueError: If the distance is invalid or no connection is found.
        """
        a = self.get_station_number(source)
        b = self.get_station_number(dest)
        value = _validate_distance(distance)
        line_key = _clean_name(line) if line else None
        edges = self.graph.find_edges(a, b, line_key)
        if not edges:
            raise ValueError(
                f"No connection between '{_clean_name(source)}' and '{_clean_name(dest)}'"
            )
        for edge in edges:
            edge.weight = value
        return len(edges)

    def export_data(self) -> dict:
        """Export the network to a dataset dictionary (same format as subway_data.json)."""
        stations = [
            {"name": name, "lines": list(self.graph.station_lines[name])}
            for name in self.station_names()
        ]
        connections = [
            {
                "source": self.number_to_station[edge.source],
                "dest": self.number_to_station[edge.dest],
                "distance": edge.weight,
                "line": edge.line,
            }
            for edge in self.graph.edges()
        ]
        return {"stations": stations, "connections": connections}

<file path="src/pathfinding.py">
"""Shortest path computation for the Beijing Subway network.

Dijkstra's algorithm is run over a state space of (station, arriving line) pairs,
which allows transfers to be counted exactly. A binary-heap priority queue (MinPQ)
drives the search, giving O((V+E) log V) behaviour.

Cost model per mode:
    shortest_distance : total distance in meters.
    least_transfer    : (number of transfers, total distance) compared lexicographically.
    comprehensive     : distance plus TRANSFER_PENALTY_METERS per transfer.
Lines outside the preferred list (if any are configured) add PREFERENCE_PENALTY_METERS
to the search cost; reported distances are always the real distances.
"""

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)
DEFAULT_MODE = MODE_COMPREHENSIVE

# Search-only penalties (in meters-equivalent); they do not change reported distances.
TRANSFER_PENALTY_METERS = 1000.0
PREFERENCE_PENALTY_METERS = 500.0


class PathNotFoundError(Exception):
    """Raised when no path exists between two stations under the current settings."""


class PathfindingConfig:
    """Configuration for path queries.

    Attributes:
        mode: One of VALID_MODES.
        max_transfers: Maximum number of transfers allowed, or None for no limit.
        preferred_lines: Lines favoured by the search (list of line names).
    """

    def __init__(self, mode=DEFAULT_MODE, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])
        self.validate()

    def validate(self):
        """Raise ValueError if the configuration is invalid."""
        if self.mode not in VALID_MODES:
            raise ValueError(f"Unknown mode '{self.mode}'; expected one of {', '.join(VALID_MODES)}")
        if self.max_transfers is not None:
            if isinstance(self.max_transfers, bool) or not isinstance(self.max_transfers, int):
                raise ValueError("max_transfers must be an integer")
            if self.max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class MinPQ:
    """Min priority queue implemented as an array-based binary heap.

    Items are (key, value) pairs; ordering uses only the key.
    """

    def __init__(self):
        self._keys = []
        self._values = []

    def __len__(self):
        return len(self._keys)

    def size(self) -> int:
        """Return the number of items in the queue."""
        return len(self._keys)

    def is_empty(self) -> bool:
        """Return True if the queue holds no items."""
        return not self._keys

    def insert(self, key, value):
        """Insert a value with the given priority key."""
        self._keys.append(key)
        self._values.append(value)
        self._swim(len(self._keys) - 1)

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._keys:
            raise IndexError("Priority queue is empty")
        return self._keys[0]

    def del_min(self):
        """Remove and return the (key, value) pair with the smallest key."""
        if not self._keys:
            raise IndexError("Priority queue is empty")
        top_key, top_value = self._keys[0], self._values[0]
        last_key = self._keys.pop()
        last_value = self._values.pop()
        if self._keys:
            self._keys[0] = last_key
            self._values[0] = last_value
            self._sink(0)
        return top_key, top_value

    def _swim(self, index):
        """Move the item at ``index`` up until the heap property holds."""
        keys = self._keys
        while index > 0:
            parent = (index - 1) // 2
            if keys[index] < keys[parent]:
                self._swap(index, parent)
                index = parent
            else:
                break

    def _sink(self, index):
        """Move the item at ``index`` down until the heap property holds."""
        keys = self._keys
        size = len(keys)
        while True:
            left = 2 * index + 1
            if left >= size:
                break
            child = left
            right = left + 1
            if right < size and keys[right] < keys[left]:
                child = right
            if keys[child] < keys[index]:
                self._swap(index, child)
                index = child
            else:
                break

    def _swap(self, i, j):
        self._keys[i], self._keys[j] = self._keys[j], self._keys[i]
        self._values[i], self._values[j] = self._values[j], self._values[i]


class _SearchState:
    """A reached (station, arriving line) state in the Dijkstra search."""

    __slots__ = ("node", "line", "cost", "distance", "transfers", "prev", "edge")

    def __init__(self, node, line, cost, distance, transfers, prev, edge):
        self.node = node
        self.line = line
        self.cost = cost
        self.distance = distance
        self.transfers = transfers
        self.prev = prev
        self.edge = edge


class ShortestPath:
    """Single-source shortest paths from a source station using Dijkstra's algorithm.

    Attributes:
        dist_to: Mapping station number -> actual distance (meters) from the source
            along the best path found to that station.
    """

    def __init__(self, graph, source, config=None):
        """Run the search from ``source`` (a station number) over ``graph``."""
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()
        if not graph.has_vertex(source):
            raise ValueError(f"Unknown source station number {source!r}")
        self.dist_to = {}
        self._best = {}
        self._search()

    def _step_cost(self, base, transfer, effective_distance):
        """Return the cost tuple after traversing one edge from a state with cost ``base``."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            increment = (1.0 if transfer else 0.0, effective_distance)
        elif mode == MODE_COMPREHENSIVE:
            increment = (effective_distance + (TRANSFER_PENALTY_METERS if transfer else 0.0), 0.0)
        else:
            increment = (effective_distance, 0.0)
        return (base[0] + increment[0], base[1] + increment[1])

    def _search(self):
        """Execute Dijkstra's algorithm over (station, line) states."""
        preferred = set(self.config.preferred_lines)
        max_transfers = self.config.max_transfers
        start = _SearchState(self.source, None, (0.0, 0.0), 0, 0, None, None)

        queue = MinPQ()
        queue.insert(start.cost, start)
        best_cost = {(self.source, None): start.cost}
        settled = set()

        while not queue.is_empty():
            cost, state = queue.del_min()
            key = (state.node, state.line)
            if key in settled:
                continue
            settled.add(key)
            # States are settled in non-decreasing cost order, so the first one
            # settled for a station is its best state.
            if state.node not in self._best:
                self._best[state.node] = state

            for edge in self.graph.adjacent(state.node):
                nxt = edge.other(state.node)
                if nxt == self.source:
                    continue
                transfer = state.line is not None and edge.line != state.line
                transfers = state.transfers + (1 if transfer else 0)
                if max_transfers is not None and transfers > max_transfers:
                    continue
                next_key = (nxt, edge.line)
                if next_key in settled:
                    continue
                penalty = PREFERENCE_PENALTY_METERS if preferred and edge.line not in preferred else 0.0
                new_cost = self._step_cost(cost, transfer, edge.weight + penalty)
                old_cost = best_cost.get(next_key)
                if old_cost is None or new_cost < old_cost:
                    best_cost[next_key] = new_cost
                    queue.insert(
                        new_cost,
                        _SearchState(nxt, edge.line, new_cost, state.distance + edge.weight,
                                     transfers, state, edge),
                    )

        for node, state in self._best.items():
            self.dist_to[node] = state.distance

    def has_path(self, target) -> bool:
        """Return True if ``target`` (station number) is reachable under the config."""
        return target in self._best

    def _state_for(self, target):
        state = self._best.get(target)
        if state is None:
            raise PathNotFoundError(f"No path found to station number {target!r} "
                                    f"under the current settings")
        return state

    def path_with_edges(self, target):
        """Return the station numbers and edges of the best path from source to target.

        Returns:
            (nodes, edges): ``nodes`` is the list of station numbers from source to
            target; ``edges`` is the list of Edge objects between consecutive nodes.

        Raises:
            PathNotFoundError: If the target cannot be reached.
        """
        state = self._state_for(target)
        nodes = []
        edges = []
        current = state
        while current.prev is not None:
            nodes.append(current.node)
            edges.append(current.edge)
            current = current.prev
        nodes.append(current.node)
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target) -> int:
        """Return the number of line transfers on the best path to ``target``."""
        return self._state_for(target).transfers

"""Fare and travel-time calculation based on the Beijing Subway fare structure.

Fare tiers (distance in meters):
    [0, 6000)         -> 3 yuan
    [6000, 12000)     -> 4 yuan
    [12000, 22000)    -> 5 yuan
    [22000, 32000)    -> 6 yuan
    >= 32000          -> 7 yuan, plus 1 yuan for every further 20 km (floor-based)
"""

import math

# (upper bound exclusive in meters, fare in yuan) for the basic and progressive tiers.
FARE_TIERS = ((6000, 3), (12000, 4), (22000, 5), (32000, 6))
EXTENDED_THRESHOLD_METERS = 32000
EXTENDED_STEP_METERS = 20000
EXTENDED_BASE_FARE = 7

AVERAGE_SPEED_KMH = 35.0        # average in-service speed used for time estimates
TRANSFER_TIME_MINUTES = 5.0     # walking and waiting time per transfer
BOARDING_TIME_MINUTES = 2.0     # fixed boarding overhead per trip


def _validate_distance(distance) -> float:
    """Ensure ``distance`` is a finite, non-negative real number in meters."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number of meters")
    if not math.isfinite(distance):
        raise ValueError("Distance must be finite")
    if distance < 0:
        raise ValueError("Distance must be non-negative")
    return distance


def Sum_money(distance) -> int:
    """Return the subway fare in yuan for a journey of ``distance`` meters.

    Boundary values follow the tier table: 6000 m -> 4 yuan, 12000 m -> 5 yuan,
    22000 m -> 6 yuan, 32000 m -> 7 yuan.
    """
    value = _validate_distance(distance)
    for upper_bound, fare in FARE_TIERS:
        if value < upper_bound:
            return fare
    extra_steps = (value - EXTENDED_THRESHOLD_METERS) // EXTENDED_STEP_METERS
    return EXTENDED_BASE_FARE + int(extra_steps)


def describe_fare_tier(distance) -> str:
    """Return a human-readable description of the fare tier for ``distance`` meters."""
    value = _validate_distance(distance)
    if value < 6000:
        return "Basic tier (0-6 km)"
    if value < 12000:
        return "Progressive tier (6-12 km)"
    if value < 22000:
        return "Progressive tier (12-22 km)"
    if value < 32000:
        return "Progressive tier (22-32 km)"
    return "Extended tier (over 32 km)"


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimate travel time in minutes from distance (meters) and transfer count.

    The estimate is in-vehicle time at an average speed, plus a fixed boarding
    overhead and a fixed time penalty per transfer.
    """
    value = _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise ValueError("Transfer count must be a non-negative integer")
    travel_minutes = (value / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    total = travel_minutes + BOARDING_TIME_MINUTES + transfer_count * TRANSFER_TIME_MINUTES
    return float(round(total, 1))

"""Input parsing and validation for the command-line interface.

Handles station-name standardization, "Starting Station; Ending Station" format
parsing, existence checks and logical checks.
"""

import re
import unicodedata

FORMAT_HINT = "Input format incorrect, please use 'Starting Station; Ending Station' format"

_SEPARATOR_PATTERN = re.compile(r"[;；,，]")


class ValidationError(ValueError):
    """Raised when user input fails format, existence or logic validation."""


def normalize_key(name) -> str:
    """Return a comparison key for a name: NFKC-normalized, lower-case, no spaces,
    and without a trailing 站 (station) suffix."""
    text = unicodedata.normalize("NFKC", str(name)).strip()
    text = re.sub(r"\s+", "", text).lower()
    if len(text) > 1 and text.endswith("站"):
        text = text[:-1]
    return text


def parse_station_pair(text) -> tuple:
    """Split 'Starting Station; Ending Station' input into two raw names.

    Accepts ';', '；', ',' or '，' as separators.

    Raises:
        ValidationError: If the input does not contain exactly two non-empty parts.
    """
    if text is None:
        raise ValidationError(FORMAT_HINT)
    parts = [part.strip() for part in _SEPARATOR_PATTERN.split(str(text))]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValidationError(FORMAT_HINT)
    return parts[0], parts[1]


def resolve_station(raw_name, official_names, role="Starting") -> str:
    """Map a user-entered station name to its official name.

    Args:
        raw_name: The name typed by the user.
        official_names: Iterable of official station names.
        role: "Starting" or "Ending", used in error messages.

    Raises:
        ValidationError: If no official station matches.
    """
    key = normalize_key(raw_name)
    for name in official_names:
        if normalize_key(name) == key:
            return name
    raise ValidationError(
        f"{role} station '{raw_name}' does not exist, please check the station name"
    )


def validate_query(text, official_names) -> tuple:
    """Parse and validate a full query. Returns (start_name, end_name) official names.

    Raises:
        ValidationError: On format, existence or logical errors.
    """
    raw_start, raw_end = parse_station_pair(text)
    names = list(official_names)
    start = resolve_station(raw_start, names, "Starting")
    end = resolve_station(raw_end, names, "Ending")
    if start == end:
        raise ValidationError("Starting station cannot be the same as the ending station")
    return start, end


def resolve_line(raw_name, known_lines):
    """Return the official line name matching ``raw_name``, or None if unknown."""
    key = normalize_key(raw_name)
    for line in known_lines:
        if normalize_key(line) == key:
            return line
    return None

"""Route query service: runs path search, builds route details, fare and time estimates."""

from dataclasses import dataclass, field
from typing import Optional

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


class FareCalculationError(Exception):
    """Raised when the fare or time for a computed route cannot be calculated."""


@dataclass
class RouteSegment:
    """A maximal run of consecutive stops on the same line."""

    line: str
    start: str
    end: str
    edge_count: int
    distance_m: float

    @property
    def stop_count(self) -> int:
        """Number of stations covered by the segment, including both ends."""
        return self.edge_count + 1


@dataclass
class TransferInfo:
    """A transfer between two line segments at a station."""

    station: str
    from_line: str
    to_line: str


@dataclass
class RouteResult:
    """Complete result of a route query."""

    start: str
    destination: str
    mode: str
    max_transfers: Optional[int]
    preferred_lines: list = field(default_factory=list)
    stations: list = field(default_factory=list)
    segments: list = field(default_factory=list)
    transfers: list = field(default_factory=list)
    lines: list = field(default_factory=list)
    distance_m: float = 0
    fare: int = 0
    transfer_count: int = 0
    estimated_minutes: float = 0.0

    @property
    def distance_km(self) -> float:
        """Total distance in kilometers."""
        return self.distance_m / 1000.0

    @property
    def stop_count(self) -> int:
        """Number of stations on the route."""
        return len(self.stations)

    def to_summary(self) -> dict:
        """Return the key figures of the route as a dictionary."""
        return {
            "total_distance_km": round(self.distance_km, 3),
            "total_distance_m": self.distance_m,
            "fare_yuan": self.fare,
            "transfer_count": self.transfer_count,
            "estimated_time_minutes": self.estimated_minutes,
            "stop_count": self.stop_count,
        }


def _build_segments(names, edges) -> list:
    """Group consecutive edges that share a line into RouteSegment objects."""
    segments = []
    for index, edge in enumerate(edges):
        end_name = names[index + 1]
        if segments and segments[-1].line == edge.line:
            current = segments[-1]
            current.end = end_name
            current.edge_count += 1
            current.distance_m += edge.weight
        else:
            segments.append(RouteSegment(
                line=edge.line,
                start=names[index],
                end=end_name,
                edge_count=1,
                distance_m=edge.weight,
            ))
    return segments


def _build_transfers(segments) -> list:
    """Derive transfer points from consecutive segments."""
    transfers = []
    for previous, following in zip(segments, segments[1:]):
        transfers.append(TransferInfo(
            station=previous.end,
            from_line=previous.line,
            to_line=following.line,
        ))
    return transfers


def query_route(network, start_name, destination_name, config=None) -> RouteResult:
    """Compute the route between two stations and assemble the full result.

    Args:
        network: A models.SubwayNetwork instance.
        start_name: Official name of the starting station.
        destination_name: Official name of the ending station.
        config: PathfindingConfig (defaults are used if None).

    Raises:
        ValueError: If the start and end stations are identical.
        StationNotFoundError: If a station does not exist.
        PathNotFoundError: If no route satisfies the configuration.
        FareCalculationError: If the fare or time cannot be computed.
    """
    config = config if config is not None else PathfindingConfig()
    if start_name == destination_name:
        raise ValueError("Starting station cannot be the same as the ending station")

    source = network.get_station_number(start_name)
    target = network.get_station_number(destination_name)
    search = ShortestPath(network.graph, source, config)
    nodes, edges = search.path_with_edges(target)

    names = [network.station_name(number) for number in nodes]
    segments = _build_segments(names, edges)
    transfers = _build_transfers(segments)
    lines = []
    for segment in segments:
        if segment.line not in lines:
            lines.append(segment.line)
    distance = sum(edge.weight for edge in edges)
    transfer_count = search.get_transfer_count(target)

    try:
        fare = Sum_money(distance)
        minutes = calculate_estimated_time(distance, transfer_count)
    except (ValueError, TypeError) as exc:
        raise FareCalculationError(f"Fare calculation failed: {exc}") from exc

    return RouteResult(
        start=start_name,
        destination=destination_name,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        stations=names,
        segments=segments,
        transfers=transfers,
        lines=lines,
        distance_m=distance,
        fare=fare,
        transfer_count=transfer_count,
        estimated_minutes=minutes,
    )

class IntegrityIssue:

class IntegrityReport:

def _is_valid_distance(value) -> bool:

def check_data_integrity(data, network=None) -> IntegrityReport:
    """Run data coverage, connection and distance checks on a dataset.

    Args:
        data: Dataset dictionary (stations and connections).
        network: Optional prebuilt SubwayNetwork for data; built from ``data`` if None.

    Returns:
        An IntegrityReport describing all findings.
    """
    report = IntegrityReport()
    if network is None:
        network = SubwayNetwork.from_data(data)
    stations = data.get("stations", []) if isinstance(data, dict) else []
    connections = data.get("connections", []) if isinstance(data, dict) else []

    report.stats = {
        "stations": len(network.station_names()),
        "lines": len(network.lines()),
        "connections": network.graph.num_edges(),
    }
    for warning in network.build_warnings:
        report.add(LEVEL_WARNING, "loading", warning)

    # --- Station records ---
    raw_lines = {}
    seen_names = set()
    for item in stations:
        if not isinstance(item, dict) or not isinstance(item.get("name"), str) \
                or not item["name"].strip():
            report.add(LEVEL_ERROR, "stations", f"Malformed station record: {item!r}")
            continue
        name = item["name"].strip()
        if name in seen_names:
            report.add(LEVEL_WARNING, "stations", f"Duplicate station record: '{name}'")
        seen_names.add(name)
        lines = item.get("lines", [])
        if not isinstance(lines, list):
            report.add(LEVEL_ERROR, "stations", f"Station '{name}' has a non-list 'lines' field")
            continue
        if not lines:
            report.add(LEVEL_WARNING, "stations", f"Station '{name}' is not assigned to any line")
        raw_lines.setdefault(name, set()).update(
            str(line).strip() for line in lines if isinstance(line, str))

    # --- Station data coverage ---
    line_keys = {normalize_key(line) for line in network.lines()}
    for expected, aliases in EXPECTED_MAIN_LINES.items():
        if not any(normalize_key(alias) in line_keys for alias in aliases):
            report.add(LEVEL_ERROR, "coverage", f"Main line '{expected}' is missing from the dataset")
    station_keys = {normalize_key(name) for name in network.station_names()}
    for line, station in REPRESENTATIVE_STATIONS.items():
        if normalize_key(station) not in station_keys:
            report.add(LEVEL_WARNING, "coverage",
                       f"Representative station '{station}' of {line} is not in the dataset")

    # --- Connection integrity and distance accuracy ---
    distances_by_pair = {}
    for index, conn in enumerate(connections, 1):
        if not isinstance(conn, dict) or any(k not in conn for k in ("source", "dest", "distance", "line")):
            report.add(LEVEL_ERROR, "connections", f"Malformed connection record #{index}: {conn!r}")
            continue
        src = str(conn["source"]).strip()
        dst = str(conn["dest"]).strip()
        line = str(conn["line"]).strip()
        label = f"{src} - {dst} ({line})"
        if src == dst:
            report.add(LEVEL_ERROR, "connections", f"Self-connection at {label}")
            continue
        ok = True
        for name in (src, dst):
            if name not in raw_lines:
                report.add(LEVEL_ERROR, "connections", f"Connection {label} references unknown station '{name}'")
                ok = False
            elif line not in raw_lines[name]:
                report.add(LEVEL_ERROR, "connections",
                           f"Station '{name}' in connection {label} is not listed as serving {line}")
                ok = False
        distance = conn["distance"]
        if not _is_valid_distance(distance):
            report.add(LEVEL_ERROR, "distances", f"Invalid distance {distance!r} for {label}")
            continue
        if distance < PLAUSIBLE_MIN_DISTANCE or distance > PLAUSIBLE_MAX_DISTANCE:
            report.add(LEVEL_WARNING, "distances",
                       f"Distance {distance} m for {label} is outside the plausible range "
                       f"({PLAUSIBLE_MIN_DISTANCE}-{PLAUSIBLE_MAX_DISTANCE} m)")
        if ok:
            key = (frozenset((src, dst)), line)
            distances_by_pair.setdefault(key, []).append(distance)

    for (pair, line), values in distances_by_pair.items():
        names = " - ".join(sorted(pair))
        if len(set(values)) > 1:
            report.add(LEVEL_ERROR, "distances",
                       f"Conflicting distances {sorted(set(values))} recorded for {names} ({line})")
        elif len(values) > 1:
            report.add(LEVEL_WARNING, "connections",
                       f"Duplicate connection recorded for {names} ({line})")

    # --- Network connectivity ---
    graph = network.graph
    for name in network.station_names():
        number = network.station_to_number[name]
        if not graph.adjacent(number):
            report.add(LEVEL_WARNING, "connectivity", f"Station '{name}' has no connections")
    components = []
    visited = set()
    for number in graph.vertices():
        if number in visited:
            continue
        size = 0
        queue = deque([number])
        visited.add(number)
        while queue:
            current = queue.popleft()
            size += 1
            for edge in graph.adjacent(current):
                nxt = edge.other(current)
                if nxt not in visited:
                    visited.add(nxt)
                    queue.append(nxt)
        components.append(size)
    if len(components) > 1:
        report.add(LEVEL_WARNING, "connectivity",

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import os
import sys

from data_loader import DATA_FILE_PATH, DataLoadError, load_subway_data
from data_manager import check_data_integrity
from input_validator import ValidationError, resolve_line, validate_query
from models import SubwayNetwork
from pathfinding import (
    DEFAULT_MODE,
    MODE_COMPREHENSIVE,
    MODE_LEAST_TRANSFER,
    MODE_SHORTEST_DISTANCE,
    PathfindingConfig,
    PathNotFoundError,
)
from report import format_json, format_json_session, format_text, format_text_session
from route_service import FareCalculationError, query_route

FORMAT_TEXT = "text"
FORMAT_JSON = "json"

HELP_TEXT = """Commands:
  Enter a query in the form 'Starting Station; Ending Station'
  (',' also works as separator), e.g.  西直门; 国贸
  help   - show this help
  quit   - exit the program"""


def _non_negative_int(value):
    """argparse type: a non-negative integer."""
    try:
        number = int(value)
    except (TypeError, ValueError) as exc:
        raise argparse.ArgumentTypeError(f"'{value}' is not an integer") from exc
    if number < 0:
        raise argparse.ArgumentTypeError("must be non-negative")
    return number


def build_parser() -> argparse.ArgumentParser:
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (Dijkstra-based route planner).")
    parser.add_argument("--output_file", default=None,
                        help="Export query results to this file (rewritten with all results of the session).")
    parser.add_argument("--format", dest="format", choices=[FORMAT_TEXT, FORMAT_JSON],
                        default=FORMAT_TEXT, help="Output format: text or json (default: text).")
    parser.add_argument("--mode", choices=[MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE],
                        default=DEFAULT_MODE,
                        help="Path query mode (default: comprehensive).")
    parser.add_argument("--max_transfers", type=_non_negative_int, default=None,
                        help="Maximum number of transfers allowed.")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line(s), e.g. 'Line 1' or 'Line 1,Line 2'.")
    parser.add_argument("--check_data", action="store_true",
                        help="Run data integrity check and exit.")
    return parser


def build_config(args, network) -> PathfindingConfig:
    """Build a PathfindingConfig from parsed arguments, resolving line preferences.

    Raises:
        ValueError: If a preferred line is unknown or other settings are invalid.
    """
    preferred = []
    if args.line_preference:
        for raw in args.line_preference.split(","):
            if not raw.strip():
                continue
            official = resolve_line(raw, network.lines())
            if official is None:
                raise ValueError(
                    f"Unknown line preference '{raw.strip()}'. Known lines: {', '.join(network.lines())}")
            if official not in preferred:
                preferred.append(official)
    return PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                             preferred_lines=preferred)


def write_output_file(path, results, fmt):
    """Write all session results to ``path`` in the requested format."""
    directory = os.path.dirname(os.path.abspath(path))
    os.makedirs(directory, exist_ok=True)
    if fmt == FORMAT_JSON:
        content = format_json_session(results)
    else:
        content = format_text_session(results)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(content + "\n")


def process_query(network, text, config):
    """Validate a raw query string and compute the route result.

    Raises:
        ValidationError, PathNotFoundError, FareCalculationError, ValueError.
    """
    start, end = validate_query(text, network.station_names())
    return query_route(network, start, end, config)


def _print_error(message):
    print(f"Error: {message}")


def main(argv=None) -> int:
    """Program entry point. Returns the process exit code."""
    parser = build_parser()
    args = parser.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except DataLoadError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 1
    network = SubwayNetwork.from_data(data)

    if args.check_data:
        report = check_data_integrity(data, network)
        print(report.format())
        return 0 if report.passed else 1

    try:
        config = build_config(args, network)
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    print("Beijing Subway fare calculator")
    print(HELP_TEXT)
    print(f"Mode: {config.mode}"
          + (f", max transfers: {config.max_transfers}" if config.max_transfers is not None else "")
          + (f", line preference: {', '.join(config.preferred_lines)}" if config.preferred_lines else ""))

    results = []
    while True:
        try:
            raw = input("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        lowered = text.lower()
        if lowered in ("quit", "exit", "q"):
            break
        if lowered == "help":
            print(HELP_TEXT)
            continue

        try:
            result = process_query(network, text, config)
        except ValidationError as exc:
            _print_error(str(exc))
            continue
        except PathNotFoundError as exc:
            _print_error(f"{exc}. Try raising --max_transfers or changing --mode.")
            continue
        except FareCalculationError as exc:
            _print_error(str(exc))
            continue
        except ValueError as exc:
            _print_error(str(exc))
            continue
        except Exception as exc:  # keep the session alive for unexpected failures
            _print_error(f"Unexpected error while calculating the route: {exc}")
            continue

        if args.format == FORMAT_JSON:
            print(format_json(result))
        else:
            print(format_text(result))
        results.append(result)

        if args.output_file:
            try:
                write_output_file(args.output_file, results, args.format)
                print(f"Results saved to {args.output_file}")
            except OSError as exc:
                _print_error(f"Cannot write output file '{args.output_file}': {exc}")

    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

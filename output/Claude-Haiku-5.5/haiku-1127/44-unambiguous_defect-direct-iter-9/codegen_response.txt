"""Loading of the Beijing Subway dataset stored in ``subway_data.json``.

The data file lives next to this module, so ``DATA_FILE_PATH`` is valid no
matter which working directory the program is launched from.
"""

import json
import os

_THIS_DIR = os.path.dirname(os.path.abspath(__file__))

#: Absolute path of ``src/subway_data.json``.
DATA_FILE_PATH = os.path.join(_THIS_DIR, "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway dataset cannot be read or is malformed."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway dataset from a JSON file.

    Args:
        path: Path of the JSON file. Defaults to ``DATA_FILE_PATH``.

    Returns:
        dict: The JSON content, with ``"stations"`` and ``"connections"`` keys.

    Raises:
        DataLoadError: If the file is missing, is not valid JSON, or does not
            contain the expected top-level structure.
    """
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"subway data file is not valid JSON: {exc}") from exc
    except OSError as exc:
        raise DataLoadError(f"cannot read subway data file: {exc}") from exc

    if not isinstance(data, dict):
        raise DataLoadError("subway data must be a JSON object")
    for key in ("stations", "connections"):
        if key not in data or not isinstance(data[key], list):
            raise DataLoadError(f"subway data is missing the '{key}' list")
    return data

"""Core graph data structures used to model the subway network.

Station numbers (vertices) are plain hashable values assigned by the caller
(see ``subway_network.SubwayNetwork``). ``Graph`` only manages vertices,
undirected weighted edges and the per-station line affiliation table.
"""


class Edge:
    """An undirected weighted edge between two station numbers.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        distance: Physical distance between the stations, in meters.
        line: Name of the subway line that operates this connection.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("edge distance must be a number of meters")
        if distance < 0:
            raise ValueError("edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Edge weight used by shortest-path algorithms (equals the distance)."""
        return self.distance

    def other_endpoint(self, vertex):
        """Return the endpoint opposite to ``vertex``.

        Raises:
            ValueError: If ``vertex`` is not an endpoint of this edge.
        """
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"vertex {vertex!r} is not an endpoint of {self!r}")

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"


class Graph:
    """Weighted undirected graph with adjacency lists.

    Attributes:
        station_lines: dict mapping station name -> list of line names.
    """

    def __init__(self):
        self._adjacency = {}
        self._edges = []
        self.station_lines = {}

    # ----- vertex management -------------------------------------------------
    def add_vertex(self, vertex):
        """Add a vertex if it is not already present."""
        self._adjacency.setdefault(vertex, [])

    def has_vertex(self, vertex):
        """Return True if ``vertex`` is part of the graph."""
        return vertex in self._adjacency

    def vertices(self):
        """Return a list of all vertices."""
        return list(self._adjacency)

    def set_station_lines(self, station_name, lines):
        """Record the line names a station belongs to."""
        self.station_lines[station_name] = list(lines)

    # ----- edge management ---------------------------------------------------
    def add_edge(self, edge):
        """Insert an undirected edge, creating its endpoints if needed.

        Raises:
            ValueError: If the edge is a self-loop.
        """
        if edge.source == edge.dest:
            raise ValueError("self-loops are not supported")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adjacency[edge.source].append(edge)
        self._adjacency[edge.dest].append(edge)
        self._edges.append(edge)

    def remove_edge(self, edge):
        """Remove an edge from the graph.

        Raises:
            ValueError: If the edge is not part of the graph.
        """
        if not any(existing is edge for existing in self._edges):
            raise ValueError("edge is not part of the graph")
        self._edges = [existing for existing in self._edges if existing is not edge]
        self._adjacency[edge.source] = [
            existing for existing in self._adjacency[edge.source] if existing is not edge
        ]
        self._adjacency[edge.dest] = [
            existing for existing in self._adjacency[edge.dest] if existing is not edge
        ]

    def adjacent_edges(self, vertex):
        """Return the list of edges incident to ``vertex`` (do not mutate)."""
        if vertex not in self._adjacency:
            raise KeyError(f"vertex {vertex!r} is not in the graph")
        return self._adjacency[vertex]

    def neighbors(self, vertex):
        """Yield ``(edge, neighbor_vertex)`` pairs for ``vertex``."""
        for edge in self.adjacent_edges(vertex):
            yield edge, edge.other_endpoint(vertex)

    def find_edges(self, u, v):
        """Return all edges directly connecting ``u`` and ``v``."""
        if u not in self._adjacency:
            return []
        return [edge for edge in self._adjacency[u] if edge.other_endpoint(u) == v]

    def edges(self):
        """Return a list of all edges."""
        return list(self._edges)

    @property
    def num_vertices(self):
        return len(self._adjacency)

    @property
    def num_edges(self):
        return len(self._edges)

"""Binary-heap based minimum priority queue used by Dijkstra's algorithm."""

import heapq
import itertools


class MinPQ:
    """Minimum priority queue implemented on top of a binary heap.

    Items are inserted with a numeric key; ``del_min`` removes the item with
    the smallest key in O(log n). A monotonically increasing counter breaks
    ties so that arbitrary (unorderable) items can be stored safely.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        """Insert ``item`` with priority ``key``."""
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self):
        """Remove and return the ``(key, item)`` pair with the smallest key.

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("priority queue is empty")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min_key(self):
        """Return the smallest key without removing it."""
        if not self._heap:
            raise IndexError("priority queue is empty")
        return self._heap[0][0]

    def is_empty(self):
        """Return True if the queue holds no items."""
        return not self._heap

    def __len__(self):
        return len(self._heap)

"""Shortest path computation with Dijkstra's algorithm.

The search runs over states ``(station_number, arrival_line)`` so that line
changes (transfers) can be counted and constrained exactly. Three query modes
are supported:

* ``shortest_distance``  - minimise the physical distance.
* ``least_transfer``     - minimise the number of transfers, then distance.
* ``comprehensive``      - distance plus a penalty per transfer.

``preferred_lines`` adds a soft penalty to edges that do not belong to the
preferred lines, and ``max_transfers`` prunes any route that exceeds the limit.
"""

from priority_queue import MinPQ

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

MODE_LABELS = {
    MODE_SHORTEST_DISTANCE: "Shortest distance",
    MODE_LEAST_TRANSFER: "Minimum transfers",
    MODE_COMPREHENSIVE: "Comprehensive optimal",
}

#: Equivalent distance (meters) added for every transfer in comprehensive mode.
TRANSFER_PENALTY_M = 1000.0
#: Cost added per transfer in least-transfer mode (dominates distance).
LEAST_TRANSFER_WEIGHT = 1.0e9
#: Penalty (meters) added for every edge outside the preferred lines.
LINE_PREFERENCE_PENALTY_M = 500.0


class PathfindingError(Exception):
    """Raised for invalid pathfinding requests."""


class PathNotFoundError(PathfindingError):
    """Raised when no path satisfies the given constraints."""


class PathfindingConfig:
    """Configuration for a path query.

    Attributes:
        mode: One of ``VALID_MODES``.
        max_transfers: Maximum number of allowed transfers, or None (no limit).
        preferred_lines: List of preferred line names (may be empty).
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(
                f"unsupported mode {mode!r}; expected one of {', '.join(VALID_MODES)}"
            )
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer or None")
        if preferred_lines is None:
            preferred = []
        elif isinstance(preferred_lines, str):
            preferred = [preferred_lines]
        else:
            preferred = list(preferred_lines)
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = preferred

    def __repr__(self):
        return (
            f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
            f"preferred_lines={self.preferred_lines!r})"
        )


class _StateInfo:
    """Best known information for one ``(station, line)`` search state."""

    __slots__ = ("cost", "distance", "transfers", "prev", "edge")

    def __init__(self, cost, distance, transfers, prev, edge):
        self.cost = cost            # objective value used by the priority queue
        self.distance = distance    # real travelled distance in meters
        self.transfers = transfers  # number of transfers so far
        self.prev = prev            # predecessor state (or None for the source)
        self.edge = edge            # edge used to reach this state (or None)


class ShortestPath:
    """Single-source shortest path search over a :class:`models.Graph`.

    Attributes:
        dist_to: dict mapping reachable station number -> real distance (m)
            of the best route found to it.
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise PathfindingError(f"station number {source!r} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}   # station number -> its best finalized state
        self._states = {}       # state -> _StateInfo
        self._run()

    # ----- cost model --------------------------------------------------------
    def _step_cost(self, distance, is_transfer, line):
        """Objective increment for traversing one edge."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            cost = distance + (LEAST_TRANSFER_WEIGHT if is_transfer else 0.0)
        elif mode == MODE_COMPREHENSIVE:
            cost = distance + (TRANSFER_PENALTY_M if is_transfer else 0.0)
        else:
            cost = float(distance)
        preferred = self.config.preferred_lines
        if preferred and line not in preferred:
            cost += LINE_PREFERENCE_PENALTY_M
        return cost

    # ----- Dijkstra ----------------------------------------------------------
    def _run(self):
        """Execute Dijkstra's algorithm with a priority queue.

        Complexity: O((V + E) log V) for a fixed number of lines per station.
        """
        start = (self.source, None)
        self._states[start] = _StateInfo(0.0, 0, 0, None, None)
        pq = MinPQ()
        pq.insert(0.0, start)
        max_transfers = self.config.max_transfers

        while not pq.is_empty():
            cost, state = pq.del_min()
            info = self._states[state]
            if cost > info.cost:
                continue  # stale queue entry
            node, line = state
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = info.distance

            for edge, neighbor in self.graph.neighbors(node):
                is_transfer = line is not None and edge.line != line
                transfers = info.transfers + (1 if is_transfer else 0)
                if max_transfers is not None and transfers > max_transfers:
                    continue
                distance = info.distance + edge.distance
                new_cost = info.cost + self._step_cost(edge.distance, is_transfer, edge.line)
                next_state = (neighbor, edge.line)
                existing = self._states.get(next_state)
                if existing is None or new_cost < existing.cost:
                    self._states[next_state] = _StateInfo(
                        new_cost, distance, transfers, state, edge
                    )
                    pq.insert(new_cost, next_state)

    # ----- queries -----------------------------------------------------------
    def has_path_to(self, target):
        """Return True if ``target`` is reachable under the current constraints."""
        return target in self._best_state

    def _require_path(self, target):
        if not self.graph.has_vertex(target):
            raise PathfindingError(f"station number {target!r} is not in the graph")
        if target not in self._best_state:
            raise PathNotFoundError(f"no path from {self.source!r} to {target!r}")
        return self._best_state[target]

    def path_with_edges(self, target):
        """Backtrack the route from the source to ``target``.

        Returns:
            tuple: ``(stations, edges)`` where ``stations`` is the ordered list of
            station numbers (source first, target last) and ``edges`` the ordered
            list of :class:`models.Edge` objects traversed (``len(stations) - 1``
            items).

        Raises:
            PathNotFoundError: If the target is unreachable.
        """
        state = self._require_path(target)
        nodes = []
        edges = []
        while state is not None:
            info = self._states[state]
            nodes.append(state[0])
            if info.edge is not None:
                edges.append(info.edge)
            state = info.prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        """Return the number of line changes on the best route to ``target``."""
        state = self._require_path(target)
        return self._states[state].transfers

    def get_distance(self, target):
        """Return the real distance (meters) of the best route to ``target``."""
        self._require_path(target)
        return self.dist_to[target]

"""Beijing Subway fare calculation and journey time estimation.

Fare structure:
* 0-6 km: 3 yuan
* 6-12 km: 4 yuan (boundary 6000 m belongs to this tier)
* 12-22 km: 5 yuan (boundary 12000 m belongs to this tier)
* 22-32 km: 6 yuan (boundary 22000 m belongs to this tier)
* over 32 km: 7 yuan at 32 km, then +1 yuan for every further 20 km
  (any started 20 km block is charged, i.e. the distance is rounded up).
"""

import math
from bisect import bisect_right

#: Lower bounds (meters) of the basic tiers 6 km, 12 km, 22 km and 32 km.
BASIC_FARE_THRESHOLDS_M = (6000, 12000, 22000, 32000)
#: Fares (yuan) for the tiers 0-6, 6-12, 12-22, 22-32 and >32 km.
BASIC_FARES = (3, 4, 5, 6, 7)
EXTENSION_START_M = 32000
EXTENSION_STEP_M = 20000
EXTENSION_STEP_FARE = 1

#: Average train speed used for time estimation (km/h).
TRAIN_SPEED_KMH = 35.0
#: Fixed time for boarding and platform handling (minutes).
BOARDING_MINUTES = 3.0
#: Walking and waiting time added per transfer (minutes).
TRANSFER_MINUTES = 5.0


class FareError(ValueError):
    """Raised when a fare or time cannot be calculated from the given input."""


def _validate_distance(distance):
    """Validate a distance in meters and return it as a float-compatible number."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise FareError("distance must be a number of meters")
    if math.isnan(distance) or math.isinf(distance):
        raise FareError("distance must be a finite number")
    if distance < 0:
        raise FareError("distance must not be negative")
    return distance


def Sum_money(distance):
    """Compute the subway fare (yuan) for a travelled distance.

    Args:
        distance: Travelled distance in meters.

    Returns:
        int: The fare in yuan.

    Raises:
        FareError: If the distance is not a valid non-negative number.
    """
    d = _validate_distance(distance)
    if d <= EXTENSION_START_M:
        # bisect_right places boundary values in the higher tier, e.g. 6000 -> 4 yuan.
        return BASIC_FARES[bisect_right(BASIC_FARE_THRESHOLDS_M, d)]
    blocks = math.ceil((d - EXTENSION_START_M) / EXTENSION_STEP_M)
    return BASIC_FARES[-1] + blocks * EXTENSION_STEP_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimate the journey time in minutes.

    Args:
        distance: Travelled distance in meters.
        transfer_count: Number of transfers on the route.

    Returns:
        float: Estimated travel time in minutes, rounded to one decimal.

    Raises:
        FareError: If the inputs are invalid.
    """
    d = _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise FareError("transfer_count must be a non-negative integer")
    riding_minutes = (d / 1000.0) / TRAIN_SPEED_KMH * 60.0
    total = BOARDING_MINUTES + riding_minutes + transfer_count * TRANSFER_MINUTES
    return round(total, 1)

"""Subway network model: station registry, graph construction and data management.

This module owns the station-name <-> station-number mapping and builds the
:class:`models.Graph` from the dataset. It also provides the data update
operations (adding lines, modifying stations and updating distances) and the
station name standardisation used by the query parser.
"""

from models import Edge, Graph

_STATION_SUFFIXES = ("station", "站")


class StationNotFoundError(Exception):
    """Raised when a station name is not part of the network."""


def _normalize_station_key(name):
    """Build a comparison key: no whitespace, lower case, no 'station' suffix."""
    key = "".join(name.split()).lower()
    for suffix in _STATION_SUFFIXES:
        if key.endswith(suffix) and len(key) > len(suffix):
            key = key[: -len(suffix)]
            break
    return key


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("distance must be a number of meters")
    if distance < 0:
        raise ValueError("distance must be non-negative")
    return distance


class SubwayNetwork:
    """Container for the subway graph and its station/line bookkeeping.

    Attributes:
        graph: The weighted undirected :class:`models.Graph`.
        station_to_number: dict mapping station name -> station number.
        number_to_station: dict mapping station number -> station name.
        line_stations: dict mapping line name -> set of station names.
    """

    def __init__(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        self.line_stations = {}

    # ----- construction ------------------------------------------------------
    @classmethod
    def from_data(cls, data):
        """Build a network from the dict returned by ``load_subway_data``."""
        network = cls()
        for station in data.get("stations", []):
            network.add_station(station["name"], station.get("lines", []))
        for connection in data.get("connections", []):
            network.add_connection(
                connection["source"],
                connection["dest"],
                connection["distance"],
                connection["line"],
            )
        return network

    # ----- station management ------------------------------------------------
    def add_station(self, name, lines=()):
        """Add a station (or merge lines into an existing one). Returns its number."""
        if not isinstance(name, str) or not name.strip():
            raise ValueError("station name must be a non-empty string")
        name = " ".join(name.split())
        if name in self.station_to_number:
            number = self.station_to_number[name]
            merged = list(self.graph.station_lines.get(name, []))
        else:
            number = len(self.number_to_station)
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
            merged = []
        for line in lines:
            if line not in merged:
                merged.append(line)
        self.graph.set_station_lines(name, merged)
        for line in merged:
            self.line_stations.setdefault(line, set()).add(name)
        return number

    def has_station(self, name):
        return name in self.station_to_number

    def get_number(self, name):
        """Return the station number for an exact station name.

        Raises:
            StationNotFoundError: If the station is unknown.
        """
        if name not in self.station_to_number:
            raise StationNotFoundError(f"station {name!r} does not exist")
        return self.station_to_number[name]

    def resolve_station_name(self, raw):
        """Standardise user input to an official station name.

        Matching ignores case, whitespace and a trailing 'Station'/'站' word.

        Returns:
            str or None: The official name, or None if nothing matches.
        """
        if not isinstance(raw, str):
            return None
        collapsed = " ".join(raw.split())
        if not collapsed:
            return None
        if collapsed in self.station_to_number:
            return collapsed
        key = _normalize_station_key(collapsed)
        for name in self.station_to_number:
            if _normalize_station_key(name) == key:
                return name
        return None

    def station_names(self):
        """Return all station names, sorted alphabetically."""
        return sorted(self.station_to_number)

    def lines(self):
        """Return all line names, sorted alphabetically."""
        return sorted(self.line_stations)

    def stations_on_line(self, line):
        """Return the sorted station names served by ``line``."""
        return sorted(self.line_stations.get(line, set()))

    def transfer_stations(self):
        """Return ``{station name: sorted lines}`` for multi-line interchanges."""
        result = {}
        for name in self.station_names():
            lines = self.graph.station_lines.get(name, [])
            if len(lines) > 1:
                result[name] = sorted(lines)
        return result

    # ----- connections -------------------------------------------------------
    def add_connection(self, source, dest, distance, line):
        """Add a direct connection between two stations served by ``line``.

        Unknown stations are created automatically and both endpoints are
        registered as serving ``line``.
        """
        if not isinstance(line, str) or not line.strip():
            raise ValueError("line name must be a non-empty string")
        _validate_distance(distance)
        src = self.add_station(source, [line])
        dst = self.add_station(dest, [line])
        edge = Edge(src, dst, distance, line)
        self.graph.add_edge(edge)
        return edge

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance of the connection(s) between two stations.

        Returns:
            int: Number of edges updated.

        Raises:
            StationNotFoundError: If a station is unknown.
            ValueError: If no matching connection exists or distance is invalid.
        """
        _validate_distance(distance)
        a = self.get_number(station_a)
        b = self.get_number(station_b)
        edges = [
            edge for edge in self.graph.find_edges(a, b)
            if line is None or edge.line == line
        ]
        if not edges:
            raise ValueError(f"no connection between {station_a!r} and {station_b!r}")
        for edge in edges:
            edge.distance = distance
        return len(edges)

    # ----- line management ---------------------------------------------------
    def add_new_line(self, line, stations, distances):
        """Add a new line through an ordered list of stations.

        Args:
            line: Name of the new line (must not already exist).
            stations: Ordered list of station names (at least two).
            distances: Either one distance (meters) used for every segment, or a
                list with ``len(stations) - 1`` distances.

        Raises:
            ValueError: If the line exists or the inputs are invalid.
        """
        if not isinstance(line, str) or not line.strip():
            raise ValueError("line name must be a non-empty string")
        if line in self.line_stations:
            raise ValueError(f"line {line!r} already exists")
        stations = list(stations)
        if len(stations) < 2:
            raise ValueError("a new line needs at least two stations")
        segments = len(stations) - 1
        if isinstance(distances, (int, float)) and not isinstance(distances, bool):
            distances = [distances] * segments
        distances = list(distances)
        if len(distances) != segments:
            raise ValueError("number of distances must equal number of segments")
        for value in distances:
            _validate_distance(value)
        for index in range(segments):
            self.add_connection(stations[index], stations[index + 1], distances[index], line)

    def modify_station(self, old_name, new_name=None, lines=None):
        """Rename a station and/or replace its line affiliations.

        Returns:
            str: The station's (possibly new) name.
        """
        if old_name not in self.station_to_number:
            raise StationNotFoundError(f"station {old_name!r} does not exist")
        current = old_name
        if new_name and new_name != old_name:
            if new_name in self.station_to_number:
                raise ValueError(f"station {new_name!r} already exists")
            number = self.station_to_number.pop(old_name)
            self.station_to_number[new_name] = number
            self.number_to_station[number] = new_name
            old_lines = self.graph.station_lines.pop(old_name, [])
            for line in old_lines:
                members = self.line_stations.get(line, set())
                members.discard(old_name)
                members.add(new_name)
            self.graph.set_station_lines(new_name, old_lines)
            current = new_name
        if lines is not None:
            new_lines = list(dict.fromkeys(lines))
            for line in self.graph.station_lines.get(current, []):
                if line not in new_lines:
                    self.line_stations.get(line, set()).discard(current)
            for line in new_lines:
                self.line_stations.setdefault(line, set()).add(current)
            self.graph.set_station_lines(current, new_lines)
        return current

"""Data integrity checks for the subway dataset (triggered by ``--check_data``).

Checks performed:
* Coverage: main lines and representative hub stations are present.
* Connections: endpoints are declared stations, no self-loops, no duplicates,
  lines are consistent with station affiliations.
* Distances: positive integers within a plausible inter-station range.
* Connectivity: every station is reachable from every other station.
"""

from collections import defaultdict, deque
from dataclasses import dataclass, field

REQUIRED_LINES = (
    "Line 1",
    "Line 2",
    "Line 4",
    "Line 5",
    "Line 6",
    "Line 8",
    "Line 10",
    "Line 13",
    "Line 14",
    "Line 15",
    "Changping Line",
    "Yizhuang Line",
    "Fangshan Line",
    "Airport Line",
)
REPRESENTATIVE_STATIONS = ("Xizhimen", "Dongzhimen", "Tiananmen East")
MIN_PLAUSIBLE_DISTANCE_M = 150
MAX_PLAUSIBLE_DISTANCE_M = 8000


@dataclass
class IntegrityReport:
    """Outcome of an integrity check."""

    errors: list = field(default_factory=list)
    warnings: list = field(default_factory=list)

    @property
    def ok(self):
        return not self.errors

    def format(self):
        """Render the report as human-readable text."""
        out = ["Data integrity check"]
        if not self.errors and not self.warnings:
            out.append("  All checks passed.")
        for message in self.errors:
            out.append(f"  [ERROR] {message}")
        for message in self.warnings:
            out.append(f"  [WARN]  {message}")
        if self.ok:
            out.append(f"Result: PASSED ({len(self.warnings)} warning(s))")
        else:
            out.append(f"Result: FAILED ({len(self.errors)} error(s))")
        return "\n".join(out)


def check_integrity(data):
    """Run all integrity checks on the raw dataset dict.

    Returns:
        IntegrityReport: The collected errors and warnings.
    """
    report = IntegrityReport()
    stations = data.get("stations")
    connections = data.get("connections")
    if not isinstance(stations, list) or not isinstance(connections, list):
        report.errors.append("dataset must contain 'stations' and 'connections' lists")
        return report

    # ----- stations ----------------------------------------------------------
    declared = {}
    for station in stations:
        if not isinstance(station, dict):
            report.errors.append(f"station entry is not an object: {station!r}")
            continue
        name = station.get("name")
        if not isinstance(name, str) or not name.strip():
            report.errors.append(f"station entry has invalid name: {name!r}")
            continue
        if name in declared:
            report.errors.append(f"station {name!r} is declared more than once")
            continue
        lines = station.get("lines", [])
        if not isinstance(lines, list) or not lines:
            report.warnings.append(f"station {name!r} has no line information")
            lines = []
        declared[name] = {line for line in lines if isinstance(line, str)}

    # ----- coverage ----------------------------------------------------------
    all_lines = set()
    for lines in declared.values():
        all_lines.update(lines)
    for connection in connections:
        if isinstance(connection, dict) and isinstance(connection.get("line"), str):
            all_lines.add(connection["line"])
    for line in REQUIRED_LINES:
        if line not in all_lines:
            report.errors.append(f"main line {line!r} is missing from the network data")
    for hub in REPRESENTATIVE_STATIONS:
        if hub not in declared:
            report.errors.append(f"representative station {hub!r} is missing")

    # ----- connections -------------------------------------------------------
    seen = set()
    adjacency = defaultdict(set)
    for index, connection in enumerate(connections, start=1):
        if not isinstance(connection, dict):
            report.errors.append(f"connection #{index} is not an object")
            continue
        src = connection.get("source")
        dst = connection.get("dest")
        distance = connection.get("distance")
        line = connection.get("line")
        label = f"connection #{index} ({src!r} - {dst!r})"
        if not all(isinstance(v, str) for v in (src, dst, line)):
            report.errors.append(f"{label} has missing or invalid source/dest/line")
            continue
        undeclared = [name for name in (src, dst) if name not in declared]
        if undeclared:
            report.errors.append(
                f"{label} references undeclared station(s): {', '.join(undeclared)}"
            )
            continue
        if src == dst:
            report.errors.append(f"{label} is a self-loop")
            continue
        if isinstance(distance, bool) or not isinstance(distance, int) or distance <= 0:
            report.errors.append(f"{label} has invalid distance {distance!r} (positive integer meters required)")
        elif not MIN_PLAUSIBLE_DISTANCE_M <= distance <= MAX_PLAUSIBLE_DISTANCE_M:
            report.warnings.append(
                f"{label} distance {distance} m is outside the plausible range "
                f"{MIN_PLAUSIBLE_DISTANCE_M}-{MAX_PLAUSIBLE_DISTANCE_M} m"
            )
        for name in (src, dst):
            if line not in declared[name]:
                report.warnings.append(
                    f"{label}: line {line!r} is not listed for station {name!r}"
                )
        key = (frozenset((src, dst)), line)
        if key in seen:
            report.errors.append(f"{label} duplicates an existing connection on {line!r}")
        seen.add(key)
        adjacency[src].add(dst)
        adjacency[dst].add(src)

    # ----- connectivity ------------------------------------------------------
    if declared:
        start = next(iter(declared))
        visited = {start}
        queue = deque([start])
        while queue:
            current = queue.popleft()
            for neighbor in adjacency[current]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
        unreachable = sorted(set(declared) - visited)
        if unreachable:
            sample = ", ".join(unreachable[:10])
            report.errors.append(
                f"{len(unreachable)} station(s) are not connected to the network: {sample}"
            )
    return report

"""Parsing and validation of user queries in the 'Starting Station; Ending Station' format."""

import re

_QUERY_SPLIT = re.compile(r"[;；,，]")


class QueryError(Exception):
    """Base class for user-input problems."""


class InputFormatError(QueryError):
    """The query does not contain exactly two station names."""

    def __init__(self):
        super().__init__(
            "Input format incorrect, please use 'Starting Station; Ending Station' format"
        )


class UnknownStationError(QueryError):
    """A station name does not exist in the network."""

    def __init__(self, role, name):
        super().__init__(
            f"{role} station '{name}' does not exist, please check the station name"
        )
        self.role = role
        self.name = name


class SameStationError(QueryError):
    """Starting and ending stations are identical."""

    def __init__(self):
        super().__init__("Starting station cannot be the same as the ending station")


def parse_query(text):
    """Split a query into raw starting and ending station names.

    Accepted separators: ``;``, ``,`` and their full-width forms.

    Returns:
        tuple: ``(start_raw, end_raw)``.

    Raises:
        InputFormatError: If the text does not contain exactly two non-empty parts.
    """
    if not isinstance(text, str):
        raise InputFormatError()
    parts = [part.strip() for part in _QUERY_SPLIT.split(text.strip())]
    if len(parts) != 2 or not all(parts):
        raise InputFormatError()
    return parts[0], parts[1]


def resolve_query(text, network):
    """Parse a query and standardise both station names against the network.

    Returns:
        tuple: Official ``(start, end)`` station names.

    Raises:
        InputFormatError, UnknownStationError, SameStationError
    """
    raw_start, raw_end = parse_query(text)
    start = network.resolve_station_name(raw_start)
    if start is None:
        raise UnknownStationError("Starting", raw_start)
    end = network.resolve_station_name(raw_end)
    if end is None:
        raise UnknownStationError("Ending", raw_end)
    if start == end:
        raise SameStationError()
    return start, end


def parse_line_preference(text):
    """Parse a comma-separated list of preferred line names, e.g. ``"Line 1, Line 2"``."""
    if not text:
        return []
    return [part.strip() for part in text.split(",") if part.strip()]

"""Route planning: combines Dijkstra path search with fare and time estimation."""

from dataclasses import dataclass, field

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


class RouteError(Exception):
    """Raised when a route cannot be planned for the given stations."""


@dataclass
class RouteSegment:
    """A continuous ride on one line.

    Attributes:
        line: Line name.
        stations: Station names travelled on this line, in order.
        distance_m: Distance travelled on this line, in meters.
    """

    line: str
    stations: list = field(default_factory=list)
    distance_m: float = 0


@dataclass
class TransferInfo:
    """A change of line at an interchange station."""

    station: str
    from_line: str
    to_line: str


@dataclass
class RouteResult:
    """Complete answer to a path query."""

    start: str
    end: str
    mode: str
    max_transfers: object
    preferred_lines: list
    stations: list
    segments: list
    transfers: list
    total_distance_m: float
    fare: int
    transfer_count: int
    estimated_time_min: float

    @property
    def lines(self):
        """Unique line names involved in the journey, in travel order."""
        result = []
        for segment in self.segments:
            if segment.line not in result:
                result.append(segment.line)
        return result


def _build_segments(names, edges):
    segments = []
    for index, edge in enumerate(edges):
        if not segments or segments[-1].line != edge.line:
            segments.append(RouteSegment(edge.line, [names[index]], 0))
        segments[-1].stations.append(names[index + 1])
        segments[-1].distance_m += edge.distance
    return segments


def _build_transfers(segments):
    transfers = []
    for index in range(1, len(segments)):
        transfers.append(
            TransferInfo(
                station=segments[index].stations[0],
                from_line=segments[index - 1].line,
                to_line=segments[index].line,
            )
        )
    return transfers


def plan_route(network, start_name, end_name, config=None):
    """Plan the optimal route between two stations.

    Args:
        network: A :class:`subway_network.SubwayNetwork`.
        start_name: Official starting station name.
        end_name: Official ending station name.
        config: Optional :class:`pathfinding.PathfindingConfig`.

    Returns:
        RouteResult: Path, segments, transfers, distance, fare and time.

    Raises:
        RouteError: If the stations are identical.
        StationNotFoundError: If a station is unknown.
        PathfindingError / PathNotFoundError: If no route satisfies the constraints.
    """
    config = config if config is not None else PathfindingConfig()
    if start_name == end_name:
        raise RouteError("Starting station cannot be the same as the ending station")

    source = network.get_number(start_name)
    target = network.get_number(end_name)
    search = ShortestPath(network.graph, source, config)
    node_ids, edges = search.path_with_edges(target)

    names = [network.number_to_station[node] for node in node_ids]
    segments = _build_segments(names, edges)
    transfers = _build_transfers(segments)
    total_distance = sum(edge.distance for edge in edges)
    transfer_count = search.get_transfer_count(target)
    fare = Sum_money(total_distance)
    estimated = calculate_estimated_time(total_distance, transfer_count)

    return RouteResult(
        start=start_name,
        end=end_name,
        mode=config.mode,
        max_transfers=config.max_transfers,
        preferred_lines=list(config.preferred_lines),
        stations=names,
        segments=segments,
        transfers=transfers,
        total_distance_m=total_distance,
        fare=fare,
        transfer_count=transfer_count,
        estimated_time_min=estimated,
    )

import json

def build_summary(result):
    """Build a JSON-serialisable summary dictionary for a route result."""

        "query": {
            "start": result.start,
            "end": result.end,
            "mode": result.mode,
            "max_transfers": result.max_transfers,
            "line_preference": result.preferred_lines,
        },
        "path": list(result.stations),
        "distance": {
            "meters": total,
            "kilometers": round(total / 1000.0, 3),
        },
        "fare": result.fare,
        "estimated_time_minutes": result.estimated_time_min,
        "transfer_count": result.transfer_count,
        "lines": result.lines,
        "transfers": [
            {"station": t.station, "from_line": t.from_line, "to_line": t.to_line}
            for t in result.transfers
        ],
        "segments": [
            {
                "line": s.line,
                "from": s.stations[0],
                "to": s.stations[-1],
                "stops": len(s.stations) - 1,
                "distance_m": s.distance_m,
            }
            for s in result.segments
        ],
    }


def format_json(result):
    """Return the route result as an indented JSON string."""
    return json.dumps(build_summary(result), ensure_ascii=False, indent=2)


def format_text(result):
    """Return a human-readable console report of the route."""
    out = []
    out.append(f"Route: {result.start} -> {result.end}")
    out.append(f"Mode: {MODE_LABELS.get(result.mode, result.mode)}")
    if result.max_transfers is not None:
        out.append(f"Max transfers: {result.max_transfers}")
    if result.preferred_lines:
        out.append(f"Preferred lines: {', '.join(result.preferred_lines)}")
    out.append("")

    out.append(f"Station sequence ({len(result.stations)} stations):")
    for index, name in enumerate(result.stations, start=1):
        out.append(f"  {index:>3}. {name}")
    out.append("")

    out.append("Line segments:")
    for segment in result.segments:
        stops = len(segment.stations) - 1
        out.append(
            f"  [{segment.line}] {segment.stations[0]} -> {segment.stations[-1]} "
            f"({stops} stops, {segment.distance_m:.0f} m)"
        )
    out.append("")

    out.append("Transfers:")
    if not result.transfers:
        out.append("  None (direct journey)")
    else:
        for transfer in result.transfers:
            out.append(
                f"  - {transfer.station}: {transfer.from_line} -> {transfer.to_line}"
            )
    out.append(f"Lines involved: {', '.join(result.lines)}")
    out.append("")

    kilometers = result.total_distance_m / 1000.0
    out.append(f"Total distance: {result.total_distance_m:.0f} m ({kilometers:.2f} km)")
    out.append(f"Transfer count: {result.transfer_count}")
    out.append(f"Estimated time: {result.estimated_time_min:.1f} min")
    out.append(f"Fare: {result.fare} yuan")
    return "\n".join(out)

def _non_negative_int(value):
    try:
        parsed = int(value)
    except ValueError as exc:
        raise argparse.ArgumentTypeError("must be an integer") from exc
    if parsed < 0:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    return parsed


def build_arg_parser():
    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(

    parser.add_argument(

        choices=("text", "json"),
        default="text",
        help="output format: text (default) or json",
    )
    parser.add_argument(
        "--mode",
        choices=VALID_MODES,
        default=MODE_SHORTEST_DISTANCE,

    parser.add_argument(

def _export(path, content):
    """Write query output to a file. Returns True on success."""
    try:
        with open(path, "w", encoding="utf-8") as handle:
            handle.write(content)
            handle.write("\n")
    except OSError as exc:
        print(f"Error: could not write output file '{path}': {exc}")
        return False
    print(f"Result exported to {path}")
    return True


def answer_query(network, text, config, fmt):
    """Process one query string and return the formatted report.

    Raises:
        QueryError, StationNotFoundError, RouteError, PathfindingError, FareError
    """
    start, end = resolve_query(text, network)
    result = plan_route(network, start, end, config)
    return format_json(result) if fmt == "json" else format_text(result)


def _run_loop(network, config, fmt, output_file):
    """Interactive continuous query loop. Returns the process exit code."""
    print("Beijing Subway Fare Calculator")
    print(HELP_TEXT)
    while True:
        try:
            raw = input("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = raw.strip()
        if not text:
            continue
        lowered = text.lower()
        if lowered in ("quit", "exit", "q"):
            break
        if lowered in ("help", "?"):
            print(HELP_TEXT)
            continue
        try:
            output = answer_query(network, text, config, fmt)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except StationNotFoundError as exc:
            print(f"Error: {exc}")
            continue
        except RouteError as exc:
            print(f"Error: {exc}")
            continue
        except PathfindingError:
            print(
                "Error: no route found between these stations under the current settings "
                "(check --max_transfers and --line_preference)."
            )
            continue
        except FareError as exc:
            print(f"Fare calculation error: {exc}")
            continue
        except Exception as exc:  # last-resort guard to keep the session alive
            print(f"Unexpected error while processing the query: {exc}")
            continue
        print(output)
        if output_file:
            _export(output_file, output)
    print("Goodbye.")
    return 0


def main(argv=None):
    """Program entry point. Returns the process exit code."""
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    try:
        data = load_subway_data()
    except DataLoadError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        report = check_integrity(data)
        print(report.format())
        return 0 if report.ok else 1

    network = SubwayNetwork.from_data(data)
    preferred = parse_line_preference(args.line_preference)
    known_lines = set(network.lines())
    for line in preferred:
        if line not in known_lines:
            print(f"Warning: line '{line}' is not in the network data and will be ignored.")

    try:
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=preferred,
        )
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    return _run_loop(network, config, args.format, args.output_file)

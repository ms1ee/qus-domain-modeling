"""Global variable store shared by all simulator modules.

The store is a plain dictionary keyed by string names. ``_init()`` resets the
store to its default values. It is also executed when this module is first
imported, so ``get_value`` is always usable even before ``system.init_system``
is called.
"""

import logging
from copy import deepcopy
from typing import Any

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruit_threshold": 300,
    "recruit_cost": 300,
    "rider_speed": 10,
    "rider_start": [435, 385],
}

_global_dict = {}


def _init() -> None:
    """Reset the global store to its default values.

    Keys created:
        funds: System funds (int, may become negative through operating costs).
        riders: List of Rider objects.
        rider_id_counter: Last issued rider ID.
        is_running: Main loop control flag.
        current_tick: Simulated time in ticks.
        orders: Full order history (list of Order objects).
        pending_orders: Orders waiting for a rider.
        order_id_counter: Last issued order ID.
        config: System parameters (copy of DEFAULT_CONFIG).
    """
    global _global_dict
    _global_dict = {
        "funds": DEFAULT_CONFIG["initial_funds"],
        "riders": [],
        "rider_id_counter": 0,
        "is_running": True,
        "current_tick": 0,
        "orders": [],
        "pending_orders": [],
        "order_id_counter": 0,
        "config": deepcopy(DEFAULT_CONFIG),
    }
    logger.debug("Global variables initialised")


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Value to store.
    """
    _global_dict[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value or ``default``.
    """
    return _global_dict.get(key, default)


def next_value(key: str) -> int:
    """Increment an integer counter and return the new value.

    Args:
        key: Name of the counter variable.

    Returns:
        The incremented counter value (starting at 1).
    """
    value = int(_global_dict.get(key, 0)) + 1
    _global_dict[key] = value
    return value

"""Grid-based path planning for the 800x800 street map.

The map is divided into an 8x8 grid of 100x100 cells. Points whose x or y
coordinate is a multiple of 100 lie on streets. Movement is performed one
unit at a time along horizontal or vertical axes, which yields right-angle
turns and straight-line movement. Streets are preferred whenever the planner
has a choice between two axes.
"""

import logging
from typing import Iterable, List, Optional, Sequence, Set, Tuple

logger = logging.getLogger(__name__)

MAP_SIZE = 800
CELL_SIZE = 100
GRID_SIZE = MAP_SIZE // CELL_SIZE

Point = Tuple[int, int]


def to_cell_index(value: int) -> int:
    """Return the grid cell index (0-7) containing a coordinate.

    Coordinates equal to the map edge (800) are assigned to the last cell.

    Args:
        value: Coordinate value in units (0-800).

    Returns:
        Cell index in the range 0..GRID_SIZE-1.
    """
    return min(max(int(value) // CELL_SIZE, 0), GRID_SIZE - 1)


def to_cell_center(x: int, y: int) -> Point:
    """Convert a coordinate to the centre point of its grid cell.

    Implements ``center_x = (grid_x // 100) * 100 + 50``.

    Args:
        x: Input x coordinate (0-800).
        y: Input y coordinate (0-800).

    Returns:
        Tuple (center_x, center_y).
    """
    center_x = to_cell_index(x) * CELL_SIZE + CELL_SIZE // 2
    center_y = to_cell_index(y) * CELL_SIZE + CELL_SIZE // 2
    return center_x, center_y


class PathPlanner:
    """Computes unit-step routes on the street grid.

    Attributes:
        grid_size: Number of cells per side.
        cell_size: Size of one cell in units.
        map_size: Total map size in units.
    """

    def __init__(self, grid_size: int = GRID_SIZE, cell_size: int = CELL_SIZE) -> None:
        """Create a planner for a square grid.

        Args:
            grid_size: Number of cells per side.
            cell_size: Size of a cell in units.
        """
        self.grid_size = grid_size
        self.cell_size = cell_size
        self.map_size = grid_size * cell_size

    def is_in_bounds(self, point: Point) -> bool:
        """Return True if the point lies inside the map."""
        x, y = point
        return 0 <= x <= self.map_size and 0 <= y <= self.map_size

    def is_on_street(self, point: Point) -> bool:
        """Return True if the point lies on a street (grid line)."""
        x, y = point
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def prefers_vertical(self, point: Point) -> bool:
        """Decide the preferred axis when both axes are possible.

        A rider standing on a vertical street (x on a grid line, y not on one)
        prefers to keep moving vertically; otherwise horizontal movement is
        preferred.

        Args:
            point: Current position.

        Returns:
            True if vertical movement should be tried first.
        """
        x, y = point
        return x % self.cell_size == 0 and y % self.cell_size != 0

    def candidate_moves(self, position: Point, target: Point,
                        prefer_vertical: bool = False) -> List[Point]:
        """List the unit moves that reduce the distance to the target.

        Special cases:
            * Same x as target (directly above or below): vertical only.
            * Same y as target (directly left or right): horizontal only.

        Args:
            position: Current position.
            target: Destination position.
            prefer_vertical: Whether vertical moves are tried first.

        Returns:
            Ordered list of candidate next points (most preferred first).
        """
        x, y = position
        tx, ty = target
        if (x, y) == (tx, ty):
            return []
        sx = (tx > x) - (tx < x)
        sy = (ty > y) - (ty < y)
        if x == tx:
            return [(x, y + sy)]
        if y == ty:
            return [(x + sx, y)]
        horizontal = (x + sx, y)
        vertical = (x, y + sy)
        if prefer_vertical:
            return [vertical, horizontal]
        return [horizontal, vertical]

    def next_step(self, position: Point, target: Point,
                  occupied: Optional[Set[Point]] = None,
                  prefer_vertical: Optional[bool] = None) -> Optional[Point]:
        """Return the next unit position toward the target.

        Args:
            position: Current position.
            target: Destination position.
            occupied: Points that other riders occupy (avoided to prevent conflicts).
            prefer_vertical: Axis preference; computed from the street layout if None.

        Returns:
            The next point, or None if every useful move is blocked.
        """
        if prefer_vertical is None:
            prefer_vertical = self.prefers_vertical(position)
        for candidate in self.candidate_moves(position, target, prefer_vertical):
            if not self.is_in_bounds(candidate):
                continue
            if occupied and candidate in occupied:
                continue
            return candidate
        return None

    def plan_path(self, start: Sequence[int], target: Sequence[int],
                  occupied: Optional[Iterable[Point]] = None) -> List[Point]:
        """Compute the list of unit points from start to target.

        The start point is excluded from the result. Without occupancy
        information the path always reaches the target.

        Args:
            start: Starting coordinates.
            target: Destination coordinates.
            occupied: Optional set of blocked points.

        Returns:
            List of points visited in order (ending at target if reachable).
        """
        blocked = set(occupied) if occupied is not None else None
        pos = (int(start[0]), int(start[1]))
        goal = (int(target[0]), int(target[1]))
        path: List[Point] = []
        while pos != goal:
            nxt = self.next_step(pos, goal, blocked)
            if nxt is None:
                logger.debug("Path from %s to %s blocked at %s", start, target, pos)
                break
            path.append(nxt)
            pos = nxt
        return path

    def plan_route(self, start: Sequence[int], restaurant: Sequence[int],
                   customer: Sequence[int]) -> List[Point]:
        """Plan the continuous route start -> restaurant -> customer.

        Args:
            start: Rider position.
            restaurant: Restaurant grid centre.
            customer: Customer grid centre.

        Returns:
            Concatenated list of route points.
        """
        to_restaurant = self.plan_path(start, restaurant)
        to_customer = self.plan_path(restaurant, customer)
        return to_restaurant + to_customer

"""Order data model.

An order stores the raw input coordinates, the grid-cell centre points used
for navigation, its lifecycle status and delivery statistics.
"""

import logging
import math
from typing import Any, Dict, Optional, Sequence, Tuple

import globalvar
from path_planner import MAP_SIZE, to_cell_center

logger = logging.getLogger(__name__)

Point = Tuple[int, int]


def _validate_point(point: Sequence[Any], label: str) -> Point:
    """Validate an ``[x, y]`` pair of integers in the range 0-800.

    Args:
        point: Sequence of two integers.
        label: Human readable name used in error messages.

    Returns:
        A tuple of two ints.

    Raises:
        ValueError: If the format, type or range is invalid.
    """
    if not isinstance(point, (list, tuple)) or len(point) != 2:
        raise ValueError(f"{label} location must be given as [x, y]")
    values = []
    for value in point:
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"{label} coordinates must be integers, got {value!r}")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"{label} coordinates must be in range 0-{MAP_SIZE}, got {value}")
        values.append(value)
    return values[0], values[1]


class Order:
    """A food delivery order.

    Attributes:
        id: Unique order identifier.
        raw_restaurant: Restaurant coordinates as entered by the user.
        raw_customer: Customer coordinates as entered by the user.
        restaurant_x, restaurant_y: Grid-cell centre of the restaurant.
        customer_x, customer_y: Grid-cell centre of the customer.
        status: One of "created", "assigned", "picked_up", "completed".
        rider_id: ID of the assigned rider, or None.
        start_time: Tick at which the order was created.
        pickup_time: Tick at which the rider reached the restaurant.
        completion_time: Tick at which the order was delivered.
        distance: Euclidean distance between restaurant and customer centres.
        time_taken: Ticks between creation and completion.
    """

    STATUS_CREATED = "created"
    STATUS_ASSIGNED = "assigned"
    STATUS_PICKED_UP = "picked_up"
    STATUS_COMPLETED = "completed"

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order from restaurant and customer coordinates.

        Args:
            res_pos: [restaurant_x, restaurant_y] (0-800).
            cus_pos: [customer_x, customer_y] (0-800).

        Raises:
            ValueError: If the coordinates are malformed or out of range.
        """
        raw_res = _validate_point(res_pos, "restaurant")
        raw_cus = _validate_point(cus_pos, "customer")
        self.id = globalvar.next_value("order_id_counter")
        self.raw_restaurant = raw_res
        self.raw_customer = raw_cus
        self.restaurant_x, self.restaurant_y = to_cell_center(*raw_res)
        self.customer_x, self.customer_y = to_cell_center(*raw_cus)
        self.status = self.STATUS_CREATED
        self.rider_id: Optional[int] = None
        self.start_time = int(globalvar.get_value("current_tick", 0))
        self.pickup_time: Optional[int] = None
        self.completion_time: Optional[int] = None
        self.distance = float(math.hypot(self.restaurant_x - self.customer_x,
                                         self.restaurant_y - self.customer_y))
        self.time_taken: Optional[int] = None
        logger.debug("Order %d created: %s -> %s", self.id, raw_res, raw_cus)

    @property
    def res_center(self) -> Point:
        """Restaurant grid-cell centre as a tuple."""
        return self.restaurant_x, self.restaurant_y

    @property
    def cus_center(self) -> Point:
        """Customer grid-cell centre as a tuple."""
        return self.customer_x, self.customer_y

    @property
    def efficiency(self) -> Optional[float]:
        """Delivery efficiency in distance units per tick, or None if unfinished."""
        if self.time_taken is None:
            return None
        return round(self.distance / max(self.time_taken, 1), 2)

    def assign_to(self, rider_id: int) -> None:
        """Record the rider assigned to this order.

        The caller is responsible for appending the order to ``rider.orders``.

        Args:
            rider_id: Identifier of the assigned rider.
        """
        self.rider_id = rider_id
        if self.status == self.STATUS_CREATED:
            self.status = self.STATUS_ASSIGNED

    def mark_picked_up(self, tick: int) -> None:
        """Mark that the rider has collected the food at the restaurant.

        Args:
            tick: Current simulation tick.
        """
        self.status = self.STATUS_PICKED_UP
        self.pickup_time = tick

    def complete(self, tick: int) -> None:
        """Mark the order as delivered and compute its statistics.

        Args:
            tick: Current simulation tick.
        """
        self.status = self.STATUS_COMPLETED
        self.completion_time = tick
        self.time_taken = tick - self.start_time
        logger.info("Order %d completed in %d ticks", self.id, self.time_taken)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.raw_restaurant),
            "customer": list(self.raw_customer),
            "restaurant_center": [self.restaurant_x, self.restaurant_y],
            "customer_center": [self.customer_x, self.customer_y],
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "distance": round(self.distance, 2),
            "time_taken": self.time_taken,
        }

"""Rider model: state machine, movement and order execution.

State machine:
    IDLE (0) -> PICKING_UP (1): a new order is present when move() runs.
    PICKING_UP (1) -> DELIVERING (2): the rider reaches the restaurant centre.
    DELIVERING (2) -> IDLE (0) (or PICKING_UP for the next queued order):
        the rider reaches the customer centre and the order completes.

Every transition appends an entry to ``Rider.log``.
"""

import logging
from typing import Any, Dict, Iterable, List, Optional, Tuple

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

Point = Tuple[int, int]

_PLANNER = PathPlanner()


class Rider:
    """A delivery rider that moves on the street grid and executes orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current position in units.
        speed: Units moved per tick.
        state: 0 = idle, 1 = picking up, 2 = delivering.
        orders: Queue of assigned orders; orders[0] is the active order.
        special_case: True when the rider is aligned with its target or blocked.
        log: List of event dictionaries with keys "event" and "order_id".
        blocked_ticks: Consecutive ticks in which the rider could not move.
        distance_travelled: Total units moved.
    """

    STATE_IDLE = 0
    STATE_PICKING_UP = 1
    STATE_DELIVERING = 2
    STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}
    BLOCKED_TICKS_BEFORE_DETOUR = 2

    def __init__(self, rider_id: int) -> None:
        """Create a new idle rider at the configured start position.

        Args:
            rider_id: Unique identifier assigned by the system.

        Raises:
            ValueError: If the configured speed is not positive.
        """
        config = globalvar.get_value("config", globalvar.DEFAULT_CONFIG)
        start = config.get("rider_start", globalvar.DEFAULT_CONFIG["rider_start"])
        self.rider_id = int(rider_id)
        self.x = int(start[0])
        self.y = int(start[1])
        self.speed = int(config.get("rider_speed", globalvar.DEFAULT_CONFIG["rider_speed"]))
        if self.speed <= 0:
            raise ValueError("rider speed must be a positive integer")
        self.state = self.STATE_IDLE
        self.orders: List[Any] = []
        self.special_case = False
        self.log: List[Dict[str, Any]] = []
        self.blocked_ticks = 0
        self.distance_travelled = 0
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # Properties and helpers
    # ------------------------------------------------------------------
    @property
    def position(self) -> Point:
        """Current position as a tuple."""
        return self.x, self.y

    @property
    def state_name(self) -> str:
        """Human readable name of the current state."""
        return self.STATE_NAMES.get(self.state, "unknown")

    @property
    def is_idle(self) -> bool:
        """True when the rider has no state work and no orders."""
        return self.state == self.STATE_IDLE and not self.orders

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event entry to the rider log.

        Args:
            event: Event name (spawn, accepted_order, arrived_at_merchant,
                delivered_to_customer).
            order_id: Related order ID or None.
        """
        entry = {
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "tick": int(globalvar.get_value("current_tick", 0)),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %d: %s (order %s)", self.rider_id, event, order_id)

    def _target(self) -> Point:
        """Return the grid centre the rider is currently heading to."""
        order = self.orders[0]
        if self.state == self.STATE_PICKING_UP:
            return order.res_center
        return order.cus_center

    def _complete_current_order(self, tick: int) -> None:
        """Finish the active order and remove it from the rider's queue."""
        order = self.orders.pop(0)
        order.complete(tick)
        self._record("delivered_to_customer", order.id)

    # ------------------------------------------------------------------
    # Simulation
    # ------------------------------------------------------------------
    def move(self, occupied: Optional[Iterable[Point]] = None) -> None:
        """Advance the rider by one tick.

        Performs up to ``speed`` unit steps toward the current target, applying
        state transitions when targets are reached. Positions in ``occupied``
        are avoided; a rider blocked for several ticks changes its axis
        preference to break deadlocks.

        Args:
            occupied: Positions currently held by other riders.
        """
        blocked_cells = set(occupied) if occupied is not None else set()
        tick = int(globalvar.get_value("current_tick", 0))

        if not self.orders:
            self.state = self.STATE_IDLE
            self.special_case = False
            self.blocked_ticks = 0
            return

        if self.state == self.STATE_IDLE:
            self.state = self.STATE_PICKING_UP
            self._record("accepted_order", self.orders[0].id)

        budget = self.speed
        moved = False
        while self.orders:
            target = self._target()
            if self.position == target:
                if self.state == self.STATE_PICKING_UP:
                    order = self.orders[0]
                    self.state = self.STATE_DELIVERING
                    order.mark_picked_up(tick)
                    self._record("arrived_at_merchant", order.id)
                    continue
                self._complete_current_order(tick)
                if self.orders:
                    self.state = self.STATE_PICKING_UP
                    self._record("accepted_order", self.orders[0].id)
                else:
                    self.state = self.STATE_IDLE
                continue

            if budget <= 0:
                break
            prefer_vertical = _PLANNER.prefers_vertical(self.position)
            if self.blocked_ticks >= self.BLOCKED_TICKS_BEFORE_DETOUR:
                prefer_vertical = not prefer_vertical
            nxt = _PLANNER.next_step(self.position, target, blocked_cells,
                                     prefer_vertical=prefer_vertical)
            if nxt is None:
                break
            self.x, self.y = nxt
            self.distance_travelled += 1
            budget -= 1
            moved = True

        if self.orders:
            target = self._target()
            at_target = self.position == target
            aligned = self.x == target[0] or self.y == target[1]
            if moved or at_target:
                self.blocked_ticks = 0
            else:
                self.blocked_ticks += 1
            self.special_case = aligned or self.blocked_ticks > 0
        else:
            self.blocked_ticks = 0
            self.special_case = False

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
        }

"""Order assignment: nearest-rider selection and pending order management."""

import logging
import math
from typing import List, Optional

import globalvar
from order_model import Order
from rider import Rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order: Order) -> Optional[Rider]:
    """Select the closest rider to the order's restaurant.

    Distance is Euclidean: sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2).
    Ties are broken by the current workload (fewer orders wins), then by ID,
    which balances load among equally placed riders.

    Args:
        order: The order to assign.

    Returns:
        The best Rider, or None if no riders exist.
    """
    riders = globalvar.get_value("riders", [])
    rx, ry = order.res_center
    best: Optional[Rider] = None
    best_key = None
    for rider in riders:
        distance = math.hypot(rider.x - rx, rider.y - ry)
        key = (round(distance, 6), len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    return best


class OrderManager:
    """Creates orders, keeps the order history and assigns pending orders."""

    @property
    def orders(self) -> List[Order]:
        """Full order history."""
        return globalvar.get_value("orders", [])

    @property
    def pending(self) -> List[Order]:
        """Orders that have not yet been assigned to a rider."""
        return globalvar.get_value("pending_orders", [])

    def create_order(self, res_pos, cus_pos) -> Order:
        """Create an order and queue it for assignment.

        Args:
            res_pos: [restaurant_x, restaurant_y].
            cus_pos: [customer_x, customer_y].

        Returns:
            The created Order.

        Raises:
            ValueError: If coordinates are invalid.
        """
        order = Order(res_pos, cus_pos)
        self.orders.append(order)
        self.pending.append(order)
        logger.info("Order %d queued", order.id)
        return order

    def assign_pending(self) -> List[Order]:
        """Assign every pending order that can be matched to a rider.

        Returns:
            List of orders that were assigned in this call.
        """
        assigned: List[Order] = []
        remaining: List[Order] = []
        for order in self.pending:
            rider = find_best_rider_for_order(order)
            if rider is None:
                remaining.append(order)
                continue
            rider.orders.append(order)
            order.assign_to(rider.rider_id)
            assigned.append(order)
            logger.info("Order %d assigned to rider %d", order.id, rider.rider_id)
        self.pending[:] = remaining
        return assigned

    def find_order(self, order_id: int) -> Optional[Order]:
        """Return the order with the given ID, or None."""
        for order in self.orders:
            if order.id == order_id:
                return order
        return None

import json

def init_system() -> None:
    """Initialise all global variables to their default values."""
    globalvar._init()
    logger.info("System initialised with funds %s", globalvar.get_value("funds"))


def _is_int(value: Any) -> bool:
    """Return True for ints that are not bools."""
    return isinstance(value, int) and not isinstance(value, bool)


def _validate_config_value(key: str, value: Any) -> Any:
    """Validate one configuration entry.

    Args:
        key: Configuration key.
        value: Raw value from the configuration file.

    Returns:
        The normalised value.

    Raises:
        ValueError: If the value is invalid for the key.
    """
    if key == "rider_start":
        if not (isinstance(value, (list, tuple)) and len(value) == 2
                and all(_is_int(v) and 0 <= v <= MAP_SIZE for v in value)):
            raise ValueError("rider_start must be [x, y] with values 0-800")
        return [int(value[0]), int(value[1])]
    if not _is_int(value) or value < 0:
        raise ValueError(f"{key} must be a non-negative integer")
    if key == "rider_speed" and value == 0:
        raise ValueError("rider_speed must be positive")
    return value


def load_config(path: str) -> Dict[str, Any]:
    """Load system parameters from a JSON file.

    Invalid or missing entries fall back to the defaults and a warning is logged.

    Args:
        path: Path to the JSON configuration file.

    Returns:
        The effective configuration dictionary.
    """
    config = dict(globalvar.DEFAULT_CONFIG)
    config["rider_start"] = list(globalvar.DEFAULT_CONFIG["rider_start"])
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError:
        logger.warning("Configuration file %s not found; using defaults", path)
        data = {}
    except (OSError, json.JSONDecodeError) as exc:
        logger.error("Cannot read configuration %s: %s", path, exc)
        data = {}
    if not isinstance(data, dict):
        logger.error("Configuration root must be an object; using defaults")
        data = {}
    for key in globalvar.DEFAULT_CONFIG:
        if key not in data:
            continue
        try:
            config[key] = _validate_config_value(key, data[key])
        except ValueError as exc:
            logger.warning("Ignoring config entry: %s", exc)
    globalvar.set_value("config", config)
    globalvar.set_value("funds", int(config["initial_funds"]))
    return config


def export_state(path: str) -> None:
    """Write system parameters, riders and order history to a JSON file.

    Args:
        path: Destination file path.

    Raises:
        OSError: If the file cannot be written.
    """
    data = {
        "config": globalvar.get_value("config", globalvar.DEFAULT_CONFIG),
        "funds": globalvar.get_value("funds"),
        "current_tick": globalvar.get_value("current_tick"),
        "riders": [rider.to_dict() for rider in globalvar.get_value("riders", [])],
        "orders": [order.to_dict() for order in globalvar.get_value("orders", [])],

    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2)
    logger.info("State exported to %s", path)


class SystemManager:
    """Coordinates funds, riders, orders and the simulation clock.

    The state itself lives in ``globalvar`` so every module shares it.
    """

    def __init__(self) -> None:
        """Create the managers used by the system."""
        self.order_manager = OrderManager()
        self.path_planner = PathPlanner()
        self.display = DisplayManager()

    # ------------------------------------------------------------------
    # State accessors
    # ------------------------------------------------------------------
    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return int(globalvar.get_value("funds", 0))

    @funds.setter
    def funds(self, value: int) -> None:
        globalvar.set_value("funds", int(value))

    @property
    def riders(self) -> List[Rider]:
        """All riders managed by the system."""
        return globalvar.get_value("riders", [])

    @property
    def current_tick(self) -> int:
        """Current simulated time."""
        return int(globalvar.get_value("current_tick", 0))

    @property
    def rider_id_counter(self) -> int:
        """Last issued rider ID."""
        return int(globalvar.get_value("rider_id_counter", 0))

    @property
    def is_running(self) -> bool:
        """Main loop control flag."""
        return bool(globalvar.get_value("is_running", True))

    @is_running.setter
    def is_running(self, value: bool) -> None:
        globalvar.set_value("is_running", bool(value))

    @property
    def pending_count(self) -> int:
        """Number of orders waiting for a rider."""
        return len(self.order_manager.pending)

    def config(self, key: str) -> Any:
        """Return a configuration value by key."""
        cfg = globalvar.get_value("config", globalvar.DEFAULT_CONFIG)
        return cfg.get(key, globalvar.DEFAULT_CONFIG[key])

    def get_rider(self, rider_id: int) -> Optional[Rider]:
        """Return the rider with the given ID, or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    @property
    def idle_riders(self) -> List[Rider]:
        """Riders that are idle and have no assigned orders."""
        return [rider for rider in self.riders if rider.is_idle]

    # ------------------------------------------------------------------
    # Recruitment
    # ------------------------------------------------------------------
    def recruit_rider(self) -> Optional[Rider]:
        """Create one new rider if funds allow.

        Returns:
            The new Rider, or None if funds are below the recruitment threshold.
        """
        if self.funds < self.config("recruit_threshold"):
            return None
        self.funds = self.funds - self.config("recruit_cost")
        rider_id = globalvar.next_value("rider_id_counter")
        rider = Rider(rider_id)
        self.riders.append(rider)
        logger.info("Recruited rider %d; funds now %d", rider_id, self.funds)
        return rider

    def maybe_recruit(self) -> bool:
        """Recruit riders while pending orders exceed idle riders and funds allow.

        Returns:
            True if at least one rider was added.
        """
        added = False
        while (self.pending_count > len(self.idle_riders)
               and self.funds >= self.config("recruit_threshold")):
            if self.recruit_rider() is None:
                break
            added = True
        return added

    # ------------------------------------------------------------------
    # Operations
    # ------------------------------------------------------------------
    def submit_order(self, res_pos, cus_pos) -> Tuple[Any, bool]:
        """Create an order, recruit if necessary and assign it.

        Args:
            res_pos: [restaurant_x, restaurant_y].
            cus_pos: [customer_x, customer_y].

        Returns:
            Tuple (order, new_rider_added).

        Raises:
            ValueError: If coordinates are invalid.
        """
        order = self.order_manager.create_order(res_pos, cus_pos)
        added = self.maybe_recruit()
        self.order_manager.assign_pending()
        return order, added

    def tick(self) -> List[Tuple[int, Dict[str, Any]]]:
        """Advance the simulation by one tick.

        Every rider moves once (avoiding other riders), pending orders are
        re-evaluated and the system state is checked.

        Returns:
            List of (rider_id, log_entry) tuples generated during this tick.
        """
        globalvar.set_value("current_tick", self.current_tick + 1)
        before = {rider.rider_id: len(rider.log) for rider in self.riders}
        for rider in list(self.riders):
            occupied = {other.position for other in self.riders if other is not rider}
            rider.move(occupied)
        self.maybe_recruit()
        self.order_manager.assign_pending()
        events: List[Tuple[int, Dict[str, Any]]] = []
        for rider in self.riders:
            for entry in rider.log[before.get(rider.rider_id, 0):]:
                events.append((rider.rider_id, entry))
        for issue in self.check_consistency():
            logger.warning(issue)
        return events

    def check_consistency(self) -> List[str]:
        """Verify and automatically repair the system state.

        Repairs performed: invalid rider states are reset to idle, orders
        duplicated across riders are removed, and mismatched rider IDs on
        orders are corrected.

        Returns:
            List of human readable descriptions of issues found.
        """
        issues: List[str] = []
        seen = {}
        for rider in self.riders:
            if rider.state not in (0, 1, 2):
                issues.append(f"Rider {rider.rider_id} had invalid state {rider.state}; reset to idle")
                rider.state = Rider.STATE_IDLE
            kept = []
            for order in rider.orders:
                if order.status == "completed" or order.id in seen:
                    issues.append(f"Removed order {order.id} from rider {rider.rider_id}")
                    continue
                if order.rider_id != rider.rider_id:
                    issues.append(f"Order {order.id} rider mismatch repaired")
                    order.rider_id = rider.rider_id
                seen[order.id] = rider.rider_id
                kept.append(order)
            rider.orders[:] = kept
            if not rider.orders and rider.state != Rider.STATE_IDLE:
                issues.append(f"Rider {rider.rider_id} had no orders; reset to idle")
                rider.state = Rider.STATE_IDLE
        self.order_manager.pending[:] = [
            order for order in self.order_manager.pending if order.status != "completed"
        ]
        if not isinstance(globalvar.get_value("funds"), int):
            issues.append("Funds were not an integer; reset to initial value")
            self.funds = int(self.config("initial_funds"))
        return issues

"""Text display: ASCII map, status, order and rider reports.

Map legend:
    '#' street (grid line), '.' empty space, 'S' restaurant, 'C' customer,
    'R' rider, '*' route. Each canvas character spans 50 units, so grid cell
    centres (x = 150, 250, ...) fall on odd indices and streets on even ones.
"""

from typing import Any, Iterable, List, Sequence, Tuple

from path_planner import CELL_SIZE, MAP_SIZE

Point = Tuple[int, int]
CANVAS_UNIT = CELL_SIZE // 2


class DisplayManager:
    """Builds human-readable text for the command line interface.

    Attributes:
        map_symbols: Mapping from element name to its ASCII symbol.
    """

    def __init__(self) -> None:
        """Create the display manager with the standard symbol set."""
        self.map_symbols = {
            "rider": "R",
            "restaurant": "S",
            "customer": "C",
            "empty": ".",
            "street": "#",
            "route": "*",
        }

    def _index(self, value: int) -> int:
        """Convert a coordinate to a canvas index."""
        return min(max(int(value) // CANVAS_UNIT, 0), MAP_SIZE // CANVAS_UNIT)

    def render_map(self, riders: Iterable[Any] = (), orders: Iterable[Any] = (),
                   route: Sequence[Point] = ()) -> str:
        """Render the 8x8 grid as ASCII art.

        Args:
            riders: Riders to draw.
            orders: Orders whose restaurant and customer points are drawn
                (completed orders are hidden).
            route: Route points drawn as connected '*' marks.

        Returns:
            The multi-line map string.
        """
        size = MAP_SIZE // CANVAS_UNIT + 1
        canvas: List[List[str]] = []
        for row in range(size):
            y = row * CANVAS_UNIT
            line = []
            for col in range(size):
                x = col * CANVAS_UNIT
                on_street = x % CELL_SIZE == 0 or y % CELL_SIZE == 0
                line.append(self.map_symbols["street" if on_street else "empty"])
            canvas.append(line)

        def put(x: int, y: int, key: str) -> None:
            canvas[self._index(y)][self._index(x)] = self.map_symbols[key]

        for x, y in route:
            put(x, y, "route")
        for order in orders:
            if order.status == "completed":
                continue
            put(order.restaurant_x, order.restaurant_y, "restaurant")
            put(order.customer_x, order.customer_y, "customer")
        for rider in riders:
            put(rider.x, rider.y, "rider")

        lines = [
            "Map (1 char = 50 units): # street  . empty  S restaurant  "
            "C customer  R rider  * route",
        ]
        for row, cells in enumerate(canvas):
            lines.append(f"{row * CANVAS_UNIT:>4} " + "".join(cells))
        lines.append(f"     x: 0 -> {MAP_SIZE} (left to right), "
                     f"y: 0 -> {MAP_SIZE} (top to bottom)")
        return "\n".join(lines)

    def format_status(self, funds: int, riders: Sequence[Any], pending: int = 0) -> str:
        """Format the system snapshot shown by the 'status' command.

        Args:
            funds: Current system funds.
            riders: All riders.
            pending: Number of orders waiting for a rider.

        Returns:
            Multi-line status text.
        """
        lines = [
            f"System funds: {funds}",
            f"Riders: {len(riders)}",
            f"Pending orders: {pending}",
        ]
        if not riders:
            lines.append("  (no riders yet)")
        for rider in riders:
            order_ids = [order.id for order in rider.orders]
            lines.append(
                f"  Rider {rider.rider_id}: position ({rider.x}, {rider.y}), "
                f"status {rider.state_name}, speed {rider.speed}, "
                f"orders {len(rider.orders)} {order_ids}"
            )
        return "\n".join(lines)

    def format_orders(self, orders: Sequence[Any]) -> str:
        """Format a table of all orders.

        Args:
            orders: Order history.

        Returns:
            Multi-line table text.
        """
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':<14} {'Customer':<14} "
                  f"{'Distance':>9} {'Time':>6} {'Rider':>6} {'Status':<10}")
        lines = [header, "-" * len(header)]
        for order in orders:
            restaurant = f"({order.restaurant_x},{order.restaurant_y})"
            customer = f"({order.customer_x},{order.customer_y})"
            time_text = "-" if order.time_taken is None else str(order.time_taken)
            rider_text = "-" if order.rider_id is None else str(order.rider_id)
            lines.append(
                f"{order.id:>4} {order.start_time:>6} {restaurant:<14} {customer:<14} "
                f"{order.distance:>9.1f} {time_text:>6} {rider_text:>6} {order.status:<10}"
            )
        return "\n".join(lines)

    def format_rider_logs(self, riders: Sequence[Any]) -> str:
        """Format the status change history of every rider.

        Args:
            riders: All riders.

        Returns:
            Multi-line log text.
        """
        if not riders:
            return "No riders yet."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider {rider.rider_id} (current: {rider.state_name}, "
                         f"travelled {rider.distance_travelled} units):")
            for entry in rider.log:
                order_text = "-" if entry["order_id"] is None else str(entry["order_id"])
                lines.append(
                    f"  t={entry['tick']:>4}  {entry['event']:<22} order={order_text:<5} "
                    f"pos=({entry['x']},{entry['y']})"
                )
        return "\n".join(lines)

"""Command parsing, shortcut resolution, command history and input validation."""

from typing import Callable, List, Optional, Sequence, Tuple

MAP_MIN = 0
MAP_MAX = 800
COORDINATE_LABELS = ("restaurant x", "restaurant y", "customer x", "customer y")


class InputManager:
    """Reads and validates user commands.

    Attributes:
        commandHistory: Every accepted command in entry order.
    """

    ALIASES = {
        "s": "start",
        "st": "status",
        "o": "orders",
        "r": "riders",
        "t": "tick",
        "m": "map",
        "h": "help",
        "?": "help",
        "q": "quit",
        "x": "exit",
        "hist": "history",
    }

    def __init__(self) -> None:
        """Create an input manager with an empty history."""
        self.commandHistory: List[str] = []

    def read_command(self, input_func: Callable[[str], str] = input) -> Optional[str]:
        """Read one raw command line.

        Supports the '!!' shortcut which repeats the last command.

        Args:
            input_func: Function used to read input (injectable for testing).

        Returns:
            The command text, or None for an empty line.

        Raises:
            ValueError: If '!!' is used without any previous command.
        """
        raw = input_func("dispatch> ").strip()
        if not raw:
            return None
        if raw == "!!":
            if not self.commandHistory:
                raise ValueError("no previous command to repeat")
            raw = self.commandHistory[-1]
            print(f"Repeating: {raw}")
        self.commandHistory.append(raw)
        return raw

    def parse_command(self, line: str) -> Tuple[str, List[str]]:
        """Split a command line into its name and arguments.

        Args:
            line: Raw command text.

        Returns:
            Tuple (command_name, argument_list) with shortcuts resolved.

        Raises:
            ValueError: If the line is empty.
        """
        tokens = line.split()
        if not tokens:
            raise ValueError("empty command")
        name = tokens[0].lower()
        return self.ALIASES.get(name, name), tokens[1:]

    def parse_coordinates(self, tokens: Sequence[str]) -> List[int]:
        """Validate four integer coordinates in the range 0-800.

        Args:
            tokens: Exactly four string tokens.

        Returns:
            List of four ints [restaurant_x, restaurant_y, customer_x, customer_y].

        Raises:
            ValueError: If the count, type or range is invalid.
        """
        if len(tokens) != 4:
            raise ValueError(f"expected 4 integers, got {len(tokens)}")
        values: List[int] = []
        for label, token in zip(COORDINATE_LABELS, tokens):
            try:
                value = int(token)
            except ValueError:
                raise ValueError(f"{label} must be an integer, got '{token}'") from None
            if not MAP_MIN <= value <= MAP_MAX:
                raise ValueError(f"{label} must be in range {MAP_MIN}-{MAP_MAX}, got {value}")
            values.append(value)
        return values

    def prompt_coordinates(self, input_func: Callable[[str], str] = input) -> Optional[List[int]]:
        """Interactively ask for the four order coordinates.

        Args:
            input_func: Function used to read input.

        Returns:
            Validated coordinates, or None if the user cancelled.
        """
        print("Enter: restaurant_x restaurant_y customer_x customer_y (0-800), "
              "or 'cancel' to abort.")
        while True:
            raw = input_func("coordinates> ").strip()
            if raw.lower() in ("cancel", "c"):
                return None
            tokens = raw.replace(",", " ").split()
            try:
                return self.parse_coordinates(tokens)
            except ValueError as exc:
                print(f"Invalid input: {exc}. Try again or type 'cancel'.")

  start [rx ry cx cy]  Create an order (restaurant and customer coordinates 0-800).
                       Without arguments you will be prompted for them.
  status (s)           Show system funds and every rider's status.
  orders (o)           Show all orders with time, distance and rider.

def cmd_start(manager: SystemManager, inp: InputManager, args: List[str]) -> None:
    """Create an order, show the assigned rider's route and the map."""
    if args and len(args) != 4:
        raise ValueError("start expects 4 integers: restaurant_x restaurant_y customer_x customer_y")
    if args:
        coords = inp.parse_coordinates(args)
    else:
        coords = inp.prompt_coordinates(input)
        if coords is None:
            print("Order entry cancelled.")
            return

    order, added = manager.submit_order((coords[0], coords[1]), (coords[2], coords[3]))
    rx, ry = order.raw_restaurant
    cx, cy = order.raw_customer
    print(f"Order #{order.id} created at tick {order.start_time}.")
    print(f"Restaurant: input ({rx},{ry}) -> grid cell "
          f"({to_cell_index(rx)},{to_cell_index(ry)}), centre {order.res_center}")
    print(f"Customer:   input ({cx},{cy}) -> grid cell "

    route = manager.path_planner.plan_route((rider.x, rider.y), order.res_center, order.cus_center)

    print(f"Assigned rider: #{rider.rider_id} at ({rider.x},{rider.y}), speed {rider.speed} units/tick")
    print(f"Route length: {len(route)} units, estimated time: {eta} ticks "
          f"(use 'tick' to advance the simulation)")
    print(manager.display.render_map(riders=[rider], orders=[order], route=route))

def _parse_config_argument(argv: List[str]) -> Optional[str]:
    """Return the path given with --config, or None."""
    if "--config" not in argv:
        return None
    index = argv.index("--config")
    if index + 1 >= len(argv):
        raise ValueError("--config requires a file path")
    return argv[index + 1]


def main(argv: Optional[List[str]] = None) -> int:
    """Run the interactive simulator.

    Args:
        argv: Command line arguments (defaults to sys.argv[1:]).

    Returns:
        Process exit code.
    """
    args = list(sys.argv[1:] if argv is None else argv)
    level = logging.DEBUG if "--debug" in args else logging.WARNING
    logging.basicConfig(level=level,
                        format="%(asctime)s %(levelname)s %(name)s: %(message)s")
    try:
        config_path = _parse_config_argument(args)
    except ValueError as exc:
        print(f"Error: {exc}")
        return 2

    init_system()
    if config_path:
        load_config(config_path)

    manager = SystemManager()
    inp = InputManager()
    handlers = build_handlers()

    print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
    print("Type 'help' for the list of commands.")

    while manager.is_running:
        try:
            line = inp.read_command()
        except ValueError as exc:
            print(f"Error: {exc}")
            continue
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if line is None:
            continue

        try:
            name, command_args = inp.parse_command(line)
        except ValueError as exc:
            print(f"Error: {exc}")
            continue

        if name in ("quit", "exit"):
            manager.is_running = False
            break

        handler = handlers.get(name)
        if handler is None:
            print(f"Unknown command '{name}'. Type 'help' for the list of commands.")
            continue
        try:
            handler(manager, inp, command_args)
        except ValueError as exc:
            print(f"Error: {exc}")
        except Exception:
            logger.exception("Command '%s' failed", name)
            print("An unexpected error occurred; the system state has been preserved.")
            manager.check_consistency()

    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

def ordinal(n: int) -> str:
    """Return the English ordinal representation of n (1 -> 1st)."""
    if 11 <= n % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Interactive command-line controller for the whole lottery process."""

    def __init__(self, input_func=input, output_func=print):
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.lottery: LotteryManager | None = None
        self.users = []
        self.lottery_done = False

    # ------------------------------------------------------------------ helpers
    def ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    def say(self, message: str = "") -> None:
        self._print(message)

    # --------------------------------------------------------------- entry point
    def run(self) -> None:
        self.say("=== Company Team Building Smart Lottery System ===")
        if not self.setup_participants():
            self.say("Exiting the system.")
            return
        self.main_menu()
        self.say("Goodbye!")

    # ------------------------------------------------------- list management
    def setup_participants(self) -> bool:
        while True:
            self.say("\n-- Participant List --")
            self.say("1. Load built-in default team building list")
            self.say("2. Upload custom txt list (format: Name,Department Group per line)")
            self.say("0. Exit")
            choice = self.ask("Select an option: ")

            if choice == "0":
                return False
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
            elif choice == "2":
                path = self.ask("Enter the txt file path: ")
                users, errors = self.list_manager.load_custom_list(path)
            else:
                self.say("Invalid option, please try again.")
                continue

            if errors:
                self.say(f"Found {len(errors)} invalid entr{'y' if len(errors) == 1 else 'ies'}:")
                for error in errors:
                    self.say(f"  {error}")
                action = self.ask(
                    "Type 'remove' to discard invalid data and continue, "
                    "or 'terminate' to abandon this upload: "
                ).lower()
                if action != "remove":
                    self.say("Upload terminated.")
                    continue

            valid_users, duplicates = self.list_manager.validate_users(users)
            if duplicates:
                self.say(f"Found {len(duplicates)} duplicate name(s):")
                for user in duplicates:
                    self.say(f"  {user.name} ({user.department})")
                action = self.ask(
                    "Type 'remove' to drop duplicates and continue, "
                    "or 'terminate' to abandon this upload: "
                ).lower()
                if action != "remove":
                    self.say("Upload terminated.")
                    continue

            if not valid_users:
                self.say("No valid participants remain. Please load another list.")
                continue

            self.users = valid_users
            self.say("Participant list loaded successfully.")
            self.show_groups()
            return True

    def show_groups(self) -> None:
        info = self.list_manager.get_group_info(self.users)
        self.say(f"\nValid participants: {len(self.users)}")
        self.say("Department groups:")
        for department, count in info.items():
            self.say(f"  {department}: {count} people")

    def group_names(self) -> list[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    # --------------------------------------------------------- main menu
    def main_menu(self) -> None:
        while True:
            self.say("\n-- Main Menu --")
            self.say("1. Load default prize library")
            self.say("2. Add custom prize")
            self.say("3. View prizes")
            self.say("4. Reorder prizes")
            self.say("5. View department groups")
            self.say("6. Start lottery")
            self.say("0. Exit")
            choice = self.ask("Select an option: ")

            if choice == "0":
                return
            if choice == "1":
                self.load_default_prizes()
            elif choice == "2":
                self.add_custom_prize()
            elif choice == "3":
                self.view_prizes()
            elif choice == "4":
                self.reorder_prizes()
            elif choice == "5":
                self.show_groups()
            elif choice == "6":
                self.start_lottery()
            else:
                self.say("Invalid option, please try again.")

    # ------------------------------------------------------ prize configuration
    def load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        if errors:
            self.say("Some default prizes could not be loaded:")
            for error in errors:
                self.say(f"  {error}")
        else:
            self.say("Default prize library loaded.")

    def _ask_positive_int(self, prompt: str) -> int:
        while True:
            raw = self.ask(prompt)
            try:
                value = int(raw)
            except ValueError:
                value = 0
            if value > 0:
                return value
            self.say("Please enter a positive integer.")

    def _ask_avoid_groups(self, groups: list[str]) -> list[str]:
        if not groups:
            return []
        while True:
            self.say("Department groups available to avoid:")
            for index, group in enumerate(groups, start=1):
                self.say(f"  {index}. {group}")
            raw = self.ask("Enter numbers or names separated by commas (blank for none): ")
            if not raw:
                return []
            selected: list[str] = []
            invalid: list[str] = []
            tokens = [t.strip() for t in raw.split(",") if t.strip()]
            for token in tokens:
                match = None
                if token.isdigit() and 1 <= int(token) <= len(groups):
                    match = groups[int(token) - 1]
                else:
                    for group in groups:
                        if group.lower() == token.lower():
                            match = group
                            break
                if match is None:
                    invalid.append(token)
                elif match not in selected:
                    selected.append(match)
            if invalid:
                self.say(f"Unknown group(s): {', '.join(invalid)}. Please try again.")
                continue
            return selected

    def add_custom_prize(self) -> None:
        name = self.ask("Prize name: ")
        if not name:
            self.say("Prize name must not be empty.")
            return
        quantity = self._ask_positive_int("Quantity: ")
        avoid_groups = self._ask_avoid_groups(self.group_names())
        drawer = self.ask("Associated lottery person (leave blank for none): ") or None
        try:
            self.prize_manager.add_prize(name, quantity, avoid_groups, drawer)
        except ValueError as error:
            self.say(f"Failed to add prize: {error}")
        else:
            self.say("Prize added successfully.")

    def view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.say("No prizes configured.")
            return
        self.say("Prize list (in lottery order):")
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) or "None"
            drawer = prize.associated_drawer or "None"
            self.say(
                f"  {index}. {prize.name} | Quantity: {prize.quantity} | "
                f"Avoid groups: {avoid} | Associated person: {drawer}"
            )

    def reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.say("No prizes configured.")
            return
        self.view_prizes()
        raw = self.ask("Enter new order as serial numbers separated by spaces or commas (e.g. 2 1 3): ")
        try:
            order = [int(token) for token in raw.replace(",", " ").split()]
        except ValueError:
            order = None
        if order is None or not self.prize_manager.reorder_prizes(order):
            self.say("Invalid order. Each serial number must appear exactly once.")
        else:
            self.say("Prize order updated.")

    # ------------------------------------------------------- lottery process
    def describe_rule(self, prize: Prize) -> str:
        if prize.associated_drawer == SPECIAL_DRAWER:
            return (
                "Previous winners are excluded. Special rule for "
                f"{SPECIAL_DRAWER}: group avoidance is not applied."
            )
        parts = ["Previous winners are excluded"]
        if prize.avoid_groups:
            parts.append(f"members of avoided groups ({', '.join(prize.avoid_groups)}) are excluded")
        if prize.associated_drawer == FALLBACK_DRAWER:
            parts.append(
                f"fallback protection: if neither of the last two {TRACKED_DRAWERS[0]}/"
                f"{TRACKED_DRAWERS[1]} draws included {CHENG_RU_GROUP} members, "
                f"only {CHENG_RU_GROUP} members are eligible"
            )
        if prize.associated_drawer:
            parts.append(f"associated lottery person: {prize.associated_drawer}")
        return "; ".join(parts)

    def wait_for_draw_command(self) -> bool:
        """Return True to draw the next winner, False to terminate the round."""
        while True:
            command = self.ask("[Enter] draw next | 'pause' | 'terminate': ").lower()
            if command == "":
                return True
            if command == "terminate":
                return False
            if command == "pause":
                self.say("Lottery paused. Type 'continue' to resume or 'terminate' to skip this round.")
                while True:
                    resume = self.ask("(paused) ").lower()
                    if resume == "continue":
                        self.say("Lottery resumed.")
                        return True
                    if resume == "terminate":
                        return False
                    self.say("Type 'continue' or 'terminate'.")
            if command == "continue":
                self.say("Lottery is not paused.")
                continue
            self.say("Unknown command.")

    def run_round(self, round_no: int, prize: Prize) -> None:
        self.say(f"\n--- Round {round_no}: {prize.name} ---")
        self.say(f"Quantity: {prize.quantity}")
        self.say(f"Rule: {self.describe_rule(prize)}")

        round_winners = []
        for index in range(1, prize.quantity + 1):
            if not self.wait_for_draw_command():
                self.say("Round terminated and skipped.")
                break
            self.say(f"Currently drawing the {ordinal(index)} winner...")
            user = self.lottery.draw(prize)
            if user is None:
                self.say("No eligible participants remain for this prize. Round ends.")
                break
            winner = self.result_manager.add_winner(user, prize)
            round_winners.append(winner)
            self.say(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

        self.say(f"Round {round_no} results:")
        if not round_winners:
            self.say("  (no winners)")
        for position, winner in enumerate(round_winners, start=1):
            self.say(f"  {position}. {winner.user.name} ({winner.user.department})")

    def start_lottery(self) -> None:
        if self.lottery_done:
            self.say("The lottery has already been completed in this session.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.say("No prizes configured. Please load or add prizes first.")
            return

        self.lottery = LotteryManager(self.users)
        self.say("\n=== Lottery started ===")
        for round_no, prize in enumerate(prizes, start=1):
            self.run_round(round_no, prize)
        self.lottery_done = True
        self.say("\n=== Lottery finished ===")
        self.show_final_results(prizes)
        self.offer_export(prizes)

    def show_final_results(self, prizes: list[Prize]) -> None:
        grouped = self.result_manager.get_winners_by_prize(prizes)
        self.say("\n=== Complete Winner List ===")
        self.say("Name | Department | Prize")
        total = 0
        for winners in grouped.values():
            for winner in winners:
                total += 1
                self.say(f"{winner.user.name} | {winner.user.department} | {winner.prize.name}")
        if total == 0:
            self.say("(no winners)")

    def offer_export(self, prizes: list[Prize]) -> None:
        answer = self.ask("Export the winner list to a file? (y/n): ").lower()
        if answer in ("y", "yes"):
            _, message = self.result_manager.export_to_csv(prizes)
            self.say(message)


def main(input_func=input, output_func=print) -> None:
    cli = LotteryCLI(input_func, output_func)
    try:
        cli.run()
    except (EOFError, KeyboardInterrupt):
        output_func("\nInput closed. Exiting.")

Xu Ba,Operations Department

class ListManager:
    """Loads, validates and summarizes participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> tuple[list[User], list[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> tuple[list[User], list[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found: '{path}'"]
        except (OSError, UnicodeDecodeError) as error:
            return [], [f"File Error: cannot read '{path}' ({error})"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> tuple[list[User], list[str]]:
        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = text.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department Group', got '{text}'"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users) -> tuple[list[User], list[User]]:
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> dict[str, int]:
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize library management: presets, custom prizes, ordering."""

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 2, "Ma Jiaqi"),
]


class PrizeManager:
    """Keeps the ordered list of prizes for the lottery."""

    def __init__(self):
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        errors: list[str] = []
        for name, quantity, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, [], drawer)
            except ValueError as error:
                errors.append(str(error))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(p.name.lower() == clean_name.lower() for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        groups = list(dict.fromkeys(groups))
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(Prize(clean_name, quantity, groups, drawer))

    def get_prizes(self) -> list[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes; order contains 1-based current positions in the new order."""
        try:
            if not all(isinstance(i, int) and not isinstance(i, bool) for i in order):
                return False
        except TypeError:
            return False
        if sorted(order) != list(range(1, len(self._prizes) + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

"""Lottery rule engine: eligibility pools, fallback protection and drawing."""

import random

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
FALLBACK_DRAWER = "Ma Jiaqi"
TRACKED_DRAWERS = ("Chen Qixian", "Jin Duo")
CHENG_RU_GROUP = "Cheng Ru Group"


class LotteryManager:
    """Draws winners while enforcing all lottery rules."""

    def __init__(self, users):
        self._users: list[User] = list(users)
        self._winner_names: set[str] = set()
        # Results (True = contained Cheng Ru group member) of the most recent
        # draws by the tracked lottery persons, at most two entries.
        self.last_two_draw_results: list[bool] = []

    @staticmethod
    def _norm(text: str) -> str:
        return (text or "").strip().lower()

    def _is_cheng_ru(self, user: User) -> bool:
        return self._norm(user.department) == self._norm(CHENG_RU_GROUP)

    def _is_fallback_active(self, prize: Prize) -> bool:
        return (
            prize.associated_drawer == FALLBACK_DRAWER
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        # Basic filtering: previous winners are always excluded.
        pool = [u for u in self._users if u.name not in self._winner_names]

        # Special rule for Li Bin: only previous winners are filtered.
        if prize.associated_drawer == SPECIAL_DRAWER:
            return pool

        # Fallback (Cheng Ru group protection): overrides avoidance groups.
        if self._is_fallback_active(prize):
            return [u for u in pool if self._is_cheng_ru(u)]

        avoided = {self._norm(g) for g in prize.avoid_groups}
        return [u for u in pool if self._norm(u.department) not in avoided]

    def draw(self, prize: Prize) -> User | None:
        """Draw one winner for the prize; returns None when no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._winner_names.add(winner.name)
        if prize.associated_drawer in TRACKED_DRAWERS:
            self.last_two_draw_results.append(self._is_cheng_ru(winner))
            del self.last_two_draw_results[:-2]
        return winner

"""Winner recording, grouping and CSV-style export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record."""

    user: User
    prize: Prize
    winning_order: int = 0


class ResultManager:
    """Maintains the winner list and exports it."""

    def __init__(self):
        self._winners: list[Winner] = []

    def add_winner(self, user, prize) -> Winner:
        winner = Winner(user, prize, len(self._winners) + 1)
        self._winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes) -> dict:
        grouped: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            grouped.setdefault(winner.prize.name, []).append(winner)
        return grouped

    def export_to_csv(self, prizes) -> tuple[bool, str]:
        date_str = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date_str}.txt")
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for winners in grouped.values():
                    for winner in winners:
                        writer.writerow(
                            [winner.user.name, winner.user.department, winner.prize.name]
                        )
        except OSError as error:
            return False, f"Export failed: {error}"
        return True, f"Winner list successfully exported to {path}"

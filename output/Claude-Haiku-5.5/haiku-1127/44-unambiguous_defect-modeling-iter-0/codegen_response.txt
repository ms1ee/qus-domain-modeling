"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network dataset is stored in ``subway_data.json`` located in the
same directory as this module, so it can be found regardless of the current
working directory.
"""

import json
import os

# Absolute path of the subway dataset, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path):
    """
    Load the subway network dataset from a JSON file.

    Args:
        path: Path to the JSON file.

    Returns:
        dict: The JSON content, expected to contain "stations" and "connections".

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the file is not valid JSON or does not contain a JSON object.
    """
    with open(path, "r", encoding="utf-8") as fh:
        data = json.load(fh)
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object at the top level")
    return data


def save_subway_data(data, path):
    """
    Persist the subway network dataset to a JSON file.

    Args:
        data: Dictionary with "stations" and "connections" keys.
        path: Destination file path.

    Raises:
        OSError: If the file cannot be written.
    """
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, ensure_ascii=False, indent=2)
        fh.write("\n")

"""
Core graph data structures for the subway network.

Stations are represented as integer nodes. The caller assigns the station
numbers (see ``network.SubwayNetwork``). Edges are undirected and carry a
weight equal to the physical distance in meters and the name of the line.
"""

import sys


class Edge:
    """
    A weighted, undirected connection between two stations on a subway line.

    Attributes:
        source (int): Station number of one endpoint.
        dest (int): Station number of the other endpoint.
        distance (float): Physical distance in meters.
        line (str): Name of the subway line this connection belongs to.
    """

    __slots__ = ("source", "dest", "distance", "line")

    def __init__(self, source, dest, distance, line):
        self.source = int(source)
        self.dest = int(dest)
        self.distance = float(distance)
        self.line = str(line)
        if self.distance < 0:
            raise ValueError("Edge distance must be non-negative")

    @property
    def weight(self):
        """The edge weight used by shortest path algorithms (distance in meters)."""
        return self.distance

    def other(self, node):
        """
        Return the endpoint of this edge opposite to ``node``.

        Raises:
            ValueError: If ``node`` is not an endpoint of this edge.
        """
        if node == self.source:
            return self.dest
        if node == self.dest:
            return self.source
        raise ValueError(f"Node {node} is not an endpoint of {self!r}")

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class Graph:
    """
    Weighted undirected graph with adjacency lists.

    Nodes are station numbers. ``station_lines`` maps station names to the
    list of line names serving that station; a station served by more than one
    line is a transfer (interchange) station.
    """

    def __init__(self):
        self._adjacency = {}      # station number -> list[Edge]
        self._edges = []          # all edges, each stored once
        self.station_lines = {}   # station name -> list[str]

    # ------------------------------------------------------------------
    # Node and station management
    # ------------------------------------------------------------------
    def add_node(self, node):
        """Ensure that a node exists in the graph."""
        self._adjacency.setdefault(int(node), [])

    def has_node(self, node):
        """Return True if the node exists in the graph."""
        return node in self._adjacency

    def set_station_lines(self, station_name, lines):
        """Set the line affiliations of a station (duplicates are removed, order kept)."""
        unique = list(dict.fromkeys(str(line) for line in lines))
        self.station_lines[station_name] = unique

    # ------------------------------------------------------------------
    # Edge management
    # ------------------------------------------------------------------
    def add_edge(self, edge):
        """
        Add an undirected edge to the graph.

        Raises:
            ValueError: If the edge is a self-loop.
        """
        if edge.source == edge.dest:
            raise ValueError("Self-loop edges are not allowed")
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self._adjacency[edge.source].append(edge)
        self._adjacency[edge.dest].append(edge)
        self._edges.append(edge)

    def adjacent_edges(self, node):
        """Return the list of edges incident to ``node`` (empty if unknown)."""
        return self._adjacency.get(node, [])

    def find_edges(self, a, b):
        """Return all edges directly connecting nodes ``a`` and ``b``."""
        return [e for e in self._adjacency.get(a, []) if e.other(a) == b]

    def update_edge_distance(self, a, b, distance):
        """
        Update the distance of all edges between nodes ``a`` and ``b``.

        Returns:
            int: Number of edges updated.
        """
        edges = self.find_edges(a, b)
        for edge in edges:
            edge.distance = float(distance)
        return len(edges)

    def nodes(self):
        """Return a list of all node numbers."""
        return list(self._adjacency.keys())

    def edges(self):
        """Return a list of all edges."""
        return list(self._edges)

    # ------------------------------------------------------------------
    # Statistics
    # ------------------------------------------------------------------
    @property
    def node_count(self):
        """Number of nodes (stations)."""
        return len(self._adjacency)

    @property
    def edge_count(self):
        """Number of undirected edges (connections)."""
        return len(self._edges)

    def memory_usage_mb(self):
        """Approximate memory footprint of the graph structure in megabytes."""
        total = sys.getsizeof(self._adjacency) + sys.getsizeof(self._edges)
        total += sys.getsizeof(self.station_lines)
        for lst in self._adjacency.values():
            total += sys.getsizeof(lst)
        for edge in self._edges:
            total += sys.getsizeof(edge) + sys.getsizeof(edge.line)
        return total / (1024 * 1024)

"""
Subway network construction and data management.

This module assigns integer station numbers, builds the ``Graph`` from the
raw dataset, handles transfer-station information, and offers update
operations (adding lines, modifying stations, updating distances).
"""

from dataclasses import dataclass, field

from models import Edge, Graph


@dataclass
class Station:
    """Read-only view of a station."""

    station_id: str
    name: str
    lines: list = field(default_factory=list)

    @property
    def is_transfer_station(self):
        """A station served by more than one line is an interchange station."""
        return len(self.lines) > 1


@dataclass
class Line:
    """A subway line with its ordered station names."""

    name: str
    stations: list = field(default_factory=list)


class SubwayNetwork:
    """
    The Beijing Subway network: station registry plus weighted graph.

    Attributes:
        graph (Graph): The weighted undirected graph of stations.
        station_to_number (dict): Mapping station name -> station number.
        number_to_station (list): Station names indexed by station number.
    """

    def __init__(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = []

    # ------------------------------------------------------------------
    # Station registry
    # ------------------------------------------------------------------
    def _ensure_station(self, name):
        """Return the number of a station, registering it if necessary."""
        if name in self.station_to_number:
            return self.station_to_number[name]
        number = len(self.number_to_station)
        self.number_to_station.append(name)
        self.station_to_number[name] = number
        self.graph.add_node(number)
        self.graph.station_lines.setdefault(name, [])
        return number

    def add_station(self, name, lines=()):
        """
        Add a station (or merge lines into an existing station).

        Returns:
            int: The station number.
        """
        name = str(name).strip()
        if not name:
            raise ValueError("Station name must not be empty")
        number = self._ensure_station(name)
        merged = list(self.graph.station_lines.get(name, []))
        for line in lines:
            if line not in merged:
                merged.append(line)
        self.graph.set_station_lines(name, merged)
        return number

    def has_station(self, name):
        """Return True if the station exists in the network."""
        return name in self.station_to_number

    def number_of(self, name):
        """Return the station number for a station name."""
        if name not in self.station_to_number:
            raise KeyError(f"Unknown station: {name}")
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the station name for a station number."""
        return self.number_to_station[number]

    def station_names(self):
        """Return all station names in registration order."""
        return list(self.number_to_station)

    def get_station(self, name):
        """Return a Station view for the given name."""
        if not self.has_station(name):
            raise KeyError(f"Unknown station: {name}")
        number = self.station_to_number[name]
        return Station(station_id=str(number), name=name,
                       lines=list(self.graph.station_lines.get(name, [])))

    def stations(self):
        """Return Station views for all stations."""
        return [self.get_station(name) for name in self.number_to_station]

    def is_transfer_station(self, name):
        """Return True if the station is served by more than one line."""
        return len(self.graph.station_lines.get(name, [])) > 1

    # ------------------------------------------------------------------
    # Connections and lines
    # ------------------------------------------------------------------
    def add_connection(self, source, dest, distance, line):
        """
        Add a direct connection between two stations on a line.

        Returns:
            Edge: The created edge.
        """
        if source == dest:
            raise ValueError("A station cannot be connected to itself")
        distance = float(distance)
        if distance < 0:
            raise ValueError("Connection distance must be non-negative")
        a = self._ensure_station(source)
        b = self._ensure_station(dest)
        for name in (source, dest):
            lines = list(self.graph.station_lines.get(name, []))
            if line not in lines:
                lines.append(line)
                self.graph.set_station_lines(name, lines)
        edge = Edge(a, b, distance, line)
        self.graph.add_edge(edge)
        return edge

    def line_names(self):
        """Return a sorted list of every line name present in the network."""
        names = set()
        for lines in self.graph.station_lines.values():
            names.update(lines)
        for edge in self.graph.edges():
            names.add(edge.line)
        return sorted(names)

    def stations_on_line(self, line):
        """Return the names of stations served by a line, in registration order."""
        return [name for name in self.number_to_station
                if line in self.graph.station_lines.get(name, [])]

    def get_line(self, line):
        """Return a Line view with the stations served by the line."""
        return Line(name=line, stations=self.stations_on_line(line))

    def add_line(self, line_name, station_names, distances):
        """
        Add a new line by specifying its ordered stations and segment distances.

        Args:
            line_name: Name of the new line.
            station_names: Ordered list of station names on the line.
            distances: Distances in meters between consecutive stations.

        Raises:
            ValueError: If the input sequence is inconsistent.
        """
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must be one less than number of stations")
        for name in station_names:
            self.add_station(name, [line_name])
        for i, distance in enumerate(distances):
            self.add_connection(station_names[i], station_names[i + 1], distance, line_name)

    # ------------------------------------------------------------------
    # Updates
    # ------------------------------------------------------------------
    def update_station(self, old_name, new_name=None, lines=None):
        """
        Rename a station and/or replace its line affiliations.

        Returns:
            str: The (possibly new) station name.
        """
        if old_name not in self.station_to_number:
            raise KeyError(f"Unknown station: {old_name}")
        number = self.station_to_number[old_name]
        target = old_name
        if new_name and new_name != old_name:
            if new_name in self.station_to_number:
                raise ValueError(f"Station already exists: {new_name}")
            del self.station_to_number[old_name]
            self.station_to_number[new_name] = number
            self.number_to_station[number] = new_name
            self.graph.station_lines[new_name] = self.graph.station_lines.pop(old_name, [])
            target = new_name
        if lines is not None:
            self.graph.set_station_lines(target, lines)
        return target

    def update_distance(self, a, b, distance):
        """
        Update the distance between two directly connected stations.

        Returns:
            int: Number of edges updated.

        Raises:
            KeyError: If either station is unknown or no connection exists.
        """
        if distance < 0:
            raise ValueError("Distance must be non-negative")
        na = self.number_of(a)
        nb = self.number_of(b)
        count = self.graph.update_edge_distance(na, nb, distance)
        if count == 0:
            raise KeyError(f"No direct connection between {a} and {b}")
        return count

    def to_dict(self):
        """Export the network in the same structure as subway_data.json."""
        stations = [{"name": name, "lines": list(self.graph.station_lines.get(name, []))}
                    for name in self.number_to_station]
        connections = [{
            "source": self.name_of(edge.source),
            "dest": self.name_of(edge.dest),
            "distance": int(round(edge.distance)),
            "line": edge.line,
        } for edge in self.graph.edges()]
        return {"stations": stations, "connections": connections}


def build_network(data):
    """
    Build a SubwayNetwork from the parsed subway dataset.

    Args:
        data: dict with "stations" ([{name, lines}]) and
              "connections" ([{source, dest, distance, line}]).

    Returns:
        SubwayNetwork: The constructed network.

    Raises:
        ValueError: If the dataset is malformed.
    """
    if not isinstance(data, dict):
        raise ValueError("Subway dataset must be a dictionary")
    network = SubwayNetwork()
    try:
        for item in data.get("stations", []):
            network.add_station(item["name"], item.get("lines", []))
        for item in data.get("connections", []):
            network.add_connection(item["source"], item["dest"], item["distance"], item["line"])
    except (KeyError, TypeError) as exc:
        raise ValueError(f"Malformed subway dataset entry: {exc}") from exc
    return network

"""
Path calculation for the subway network.

Contains:
  * ``MinPQ``: an indexed binary min-heap supporting decrease-key, which gives
    Dijkstra's algorithm its O((V+E) log V) running time.
  * ``PathfindingConfig``: query mode, transfer limit and line preferences.
  * ``ShortestPath``: Dijkstra's algorithm over a state space of
    (station, arriving line, transfers so far), so that transfers at
    multi-line interchange stations are handled correctly.
"""

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance (meters) penalty per transfer used by the comprehensive mode.
TRANSFER_PENALTY_METERS = 1000.0
# Equivalent distance (meters) penalty for riding a line outside the preferences.
PREFERENCE_PENALTY_METERS = 300.0


class MinPQ:
    """
    Indexed binary min-heap priority queue.

    Items are non-negative integers. Each item may be present at most once;
    its key can be lowered with ``decrease_key``.

    Attributes:
        size (int): Number of items currently in the queue.
        capacity (int): Number of item slots currently allocated.
    """

    def __init__(self, capacity=16):
        size = max(1, int(capacity))
        self._keys = [None] * size      # item -> key
        self._qp = [-1] * size          # item -> position in heap, -1 if absent
        self._heap = []                 # heap-ordered list of items

    @property
    def size(self):
        return len(self._heap)

    @property
    def capacity(self):
        return len(self._qp)

    def is_empty(self):
        """Return True if the queue has no items."""
        return not self._heap

    def contains(self, item):
        """Return True if the item is currently in the queue."""
        return 0 <= item < len(self._qp) and self._qp[item] != -1

    def _ensure_slot(self, item):
        if item >= len(self._qp):
            grow = item + 1 - len(self._qp)
            self._qp.extend([-1] * grow)
            self._keys.extend([None] * grow)

    def insert(self, item, key):
        """Insert an item with the given key."""
        if item < 0:
            raise ValueError("Items must be non-negative integers")
        self._ensure_slot(item)
        if self.contains(item):
            raise ValueError(f"Item {item} is already in the queue")
        self._keys[item] = key
        self._heap.append(item)
        self._qp[item] = len(self._heap) - 1
        self._swim(len(self._heap) - 1)

    def min_item(self):
        """Return the item with the smallest key without removing it."""
        if not self._heap:
            raise IndexError("Priority queue is empty")
        return self._heap[0]

    def min_key(self):
        """Return the smallest key in the queue."""
        return self._keys[self.min_item()]

    def decrease_key(self, item, key):
        """Lower the key of an item already in the queue."""
        if not self.contains(item):
            raise KeyError(f"Item {item} is not in the queue")
        if key > self._keys[item]:
            raise ValueError("New key must not be greater than the current key")
        self._keys[item] = key
        self._swim(self._qp[item])

    def del_min(self):
        """Remove and return the item with the smallest key."""
        if not self._heap:
            raise IndexError("Priority queue is empty")
        top = self._heap[0]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._qp[last] = 0
            self._sink(0)
        self._qp[top] = -1
        self._keys[top] = None
        return top

    # Heap helpers -------------------------------------------------------
    def _greater(self, i, j):
        return self._keys[self._heap[i]] > self._keys[self._heap[j]]

    def _swap(self, i, j):
        a = self._heap[i]
        b = self._heap[j]
        self._heap[i] = b
        self._heap[j] = a
        self._qp[b] = i
        self._qp[a] = j

    def _swim(self, k):
        while k > 0:
            parent = (k - 1) // 2
            if self._greater(parent, k):
                self._swap(parent, k)
                k = parent
            else:
                break

    def _sink(self, k):
        n = len(self._heap)
        while True:
            child = 2 * k + 1
            if child >= n:
                break
            if child + 1 < n and self._greater(child, child + 1):
                child += 1
            if not self._greater(k, child):
                break
            self._swap(k, child)
            k = child


class PathfindingConfig:
    """
    Configuration of a path query.

    Attributes:
        mode (str): One of VALID_MODES.
        max_transfers (int | None): Maximum allowed transfers, None means unlimited.
        preferred_lines (list[str]): Lines the passenger prefers to ride.
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Unsupported mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer")
            if max_transfers < 0:
                raise ValueError("max_transfers must not be negative")
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = preferred_lines.split(",")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = [str(line).strip() for line in preferred_lines if str(line).strip()]


def count_transfers(edges):
    """Count line changes along a sequence of consecutive edges."""
    return sum(1 for prev, nxt in zip(edges, edges[1:]) if prev.line != nxt.line)


class ShortestPath:
    """
    Dijkstra-based optimal path computation from one source station.

    The search runs over states (station, arriving line, transfers so far).
    The cost ordering depends on the mode:
      * shortest_distance: total distance (plus preference penalties)
      * least_transfer:    (transfers, distance) lexicographically
      * comprehensive:     distance + TRANSFER_PENALTY_METERS * transfers

    Attributes:
        algorithm (str): Name of the algorithm ("Dijkstra").
        dist_to (dict): Station number -> travelled distance in meters on the
            best path found for that station.
    """

    algorithm = "Dijkstra"

    def __init__(self, graph, source, config=None):
        if not graph.has_node(source):
            raise ValueError(f"Source station {source} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._state_node = []
        self._state_line = []
        self._state_trans = []
        self._state_dist = []
        self._state_pen = []
        self._state_key = []
        self._state_prev = []
        self._state_edge = []
        self._state_index = {}
        self._best_state = {}
        self._run()

    def _key(self, distance, transfers, penalty):
        """Compute the priority key of a state according to the query mode."""
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, distance + penalty)
        if mode == MODE_COMPREHENSIVE:
            return distance + penalty + TRANSFER_PENALTY_METERS * transfers
        return distance + penalty

    def _add_state(self, node, line, transfers, distance, penalty, key, prev, edge):
        sid = len(self._state_node)
        self._state_node.append(node)
        self._state_line.append(line)
        self._state_trans.append(transfers)
        self._state_dist.append(distance)
        self._state_pen.append(penalty)
        self._state_key.append(key)
        self._state_prev.append(prev)
        self._state_edge.append(edge)
        self._state_index[(node, line, transfers)] = sid
        return sid

    def _run(self):
        """Execute Dijkstra's algorithm over the state space."""
        cfg = self.config
        preferred = set(cfg.preferred_lines)
        max_transfers = cfg.max_transfers

        pq = MinPQ(capacity=max(16, 2 * self.graph.edge_count + 1))
        start_key = self._key(0.0, 0, 0.0)
        start = self._add_state(self.source, None, 0, 0.0, 0.0, start_key, -1, None)
        pq.insert(start, start_key)

        settled = set()
        while not pq.is_empty():
            s = pq.del_min()
            settled.add(s)
            node = self._state_node[s]
            line_in = self._state_line[s]
            trans_s = self._state_trans[s]
            dist_s = self._state_dist[s]
            pen_s = self._state_pen[s]

            for edge in self.graph.adjacent_edges(node):
                nxt = edge.other(node)
                line = edge.line
                new_trans = trans_s + (1 if line_in is not None and line != line_in else 0)
                if max_transfers is not None and new_trans > max_transfers:
                    continue
                new_dist = dist_s + edge.distance
                new_pen = pen_s + (PREFERENCE_PENALTY_METERS
                                   if preferred and line not in preferred else 0.0)
                new_key = self._key(new_dist, new_trans, new_pen)

                sid = self._state_index.get((nxt, line, new_trans))
                if sid is None:
                    sid = self._add_state(nxt, line, new_trans, new_dist, new_pen,
                                          new_key, s, edge)
                    pq.insert(sid, new_key)
                elif sid not in settled and new_key < self._state_key[sid]:
                    self._state_dist[sid] = new_dist
                    self._state_pen[sid] = new_pen
                    self._state_key[sid] = new_key
                    self._state_prev[sid] = s
                    self._state_edge[sid] = edge
                    pq.decrease_key(sid, new_key)

        # Choose the best state for every station.
        for sid in range(len(self._state_node)):
            node = self._state_node[sid]
            if node not in self._best_state or \
                    self._state_key[sid] < self._state_key[self._best_state[node]]:
                self._best_state[node] = sid
        for node, sid in self._best_state.items():
            self.dist_to[node] = self._state_dist[sid]

    def has_path_to(self, target):
        """Return True if the target station is reachable under the configuration."""
        return target in self._best_state

    def path_with_edges(self, target):
        """
        Reconstruct the path from the source to ``target``.

        Returns:
            tuple: (list of station numbers from source to target,
                    list of Edge objects along the path).
            Both lists are empty if the target is unreachable.
        """
        if target not in self._best_state:
            return [], []
        sid = self._best_state[target]
        nodes = []
        edges = []
        while sid != -1:
            nodes.append(self._state_node[sid])
            if self._state_prev[sid] != -1:
                edges.append(self._state_edge[sid])
            sid = self._state_prev[sid]
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):

"""
Fare and travel time calculation for Beijing Subway trips.

Fare structure:
  * 0 - under 6 km:        3 yuan
  * 6 km - under 12 km:    4 yuan
  * 12 km - under 22 km:   5 yuan
  * 22 km - under 32 km:   6 yuan
  * 32 km and above:       7 yuan, +1 yuan for every further 20 km

Each lower boundary belongs to the higher tier (e.g. exactly 6000 m costs 4 yuan).
Fractional meters are rounded to the nearest whole meter before lookup.
"""

# (exclusive upper bound in meters, fare in yuan) for the basic and progressive tiers.
FARE_TIERS = (
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
)
EXTENDED_START_METERS = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP_METERS = 20000

# Average travel speed of the subway in meters per minute (35 km/h).
AVERAGE_SPEED_M_PER_MIN = 35000.0 / 60.0
# Fixed boarding/alighting allowance per trip, in minutes.
BOARDING_MINUTES = 2.0
# Extra time per transfer (walking and waiting), in minutes.
TRANSFER_MINUTES = 4.0


def Sum_money(distance):
    """
    Calculate the subway fare for a travel distance.

    Args:
        distance: Travel distance in meters (int or float).

    Returns:
        int: Fare in yuan.

    Raises:
        ValueError: If the distance is negative or not a finite number.
    """
    try:
        meters = float(distance)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"Invalid distance: {distance!r}") from exc
    if meters != meters or meters in (float("inf"), float("-inf")):
        raise ValueError("Distance must be a finite number")
    if meters < 0:
        raise ValueError("Distance must not be negative")

    meters = int(round(meters))
    for upper, fare in FARE_TIERS:
        if meters < upper:
            return fare
    # Extended tier: fare rises by 1 yuan for each full 20 km beyond 32 km.
    return EXTENDED_BASE_FARE + int((meters - EXTENDED_START_METERS) // EXTENDED_STEP_METERS)


def calculate_estimated_time(distance, transfer_count):
    """
    Estimate the travel time of a trip.

    Args:
        distance: Travel distance in meters.
        transfer_count: Number of transfers on the trip.

    Returns:
        float: Estimated travel time in minutes.

    Raises:
        ValueError: If the distance or transfer count is invalid.
    """
    meters = float(distance)
    if meters < 0:
        raise ValueError("Distance must not be negative")
    if transfer_count < 0:
        raise ValueError("Transfer count must not be negative")
    riding = meters / AVERAGE_SPEED_M_PER_MIN
    return riding + BOARDING_MINUTES + TRANSFER_MINUTES * transfer_count

class Query:

class Route:
    """
    A computed trip with its derived summary information.

    Attributes:
        mode (str): Query mode used.
        station_names (list[str]): Ordered station names from start to end.
        edges (list[Edge]): Connections traversed.
        total_distance_meters (float): Total distance in meters.
        transfers (list[dict]): Transfer details (station, from_line, to_line).
        lines (list[str]): Lines involved, in order of first use.
        segments (list[dict]): Continuous rides on one line.
        fare (int): Fare in yuan.
        estimated_time_minutes (float): Estimated travel time.
    """

    def __init__(self, mode, station_names, edges, total_distance_meters, transfers,

        self.mode = mode
        self.station_names = station_names
        self.edges = edges
        self.total_distance_meters = total_distance_meters
        self.transfers = transfers
        self.lines = lines
        self.segments = segments
        self.fare = fare
        self.estimated_time_minutes = estimated_time_minutes

    @property
    def transfer_count(self):
        """Number of transfers required."""
        return len(self.transfers)

    @property
    def total_distance_km(self):
        """Total distance in kilometers."""
        return self.total_distance_meters / 1000.0

    def summary(self):
        """Return the key figures of the route."""
        return {
            "distance_meters": self.total_distance_meters,
            "distance_km": self.total_distance_km,
            "fare": self.fare,
            "transfer_count": self.transfer_count,
            "estimated_time_minutes": self.estimated_time_minutes,
        }


def build_route(station_names, edges, mode):
    """
    Build a Route from an ordered station list and the edges between them.

    Args:
        station_names: Station names from start to end (len = len(edges) + 1).
        edges: Edge objects along the path.
        mode: Query mode string.

    Returns:
        Route: The assembled route.
    """
    total = sum(edge.distance for edge in edges)

    transfers = []
    for i in range(1, len(edges)):
        if edges[i - 1].line != edges[i].line:
            transfers.append({
                "station": station_names[i],
                "from_line": edges[i - 1].line,
                "to_line": edges[i].line,
            })

    lines = []
    for edge in edges:
        if edge.line not in lines:
            lines.append(edge.line)

    segments = []
    i = 0
    while i < len(edges):
        j = i
        while j + 1 < len(edges) and edges[j + 1].line == edges[i].line:
            j += 1
        segment_distance = sum(edges[k].distance for k in range(i, j + 1))
        segments.append({
            "line": edges[i].line,
            "from_station": station_names[i],
            "to_station": station_names[j + 1],
            "stops": j - i + 1,
            "distance_meters": segment_distance,
        })
        i = j + 1

    fare = Sum_money(total)
    estimated_time = calculate_estimated_time(total, len(transfers))

    return Route(
        mode=mode,
        station_names=list(station_names),
        edges=list(edges),
        total_distance_meters=round(total, 1),
        transfers=transfers,
        lines=lines,
        segments=segments,
        fare=fare,
        estimated_time_minutes=round(estimated_time, 1),
    )


def plan_route(network, query):
    """
    Compute the route for a query.

    Args:
        network: SubwayNetwork instance.
        query: Query describing start, end and constraints.

    Returns:
        Route: The computed route.

    Raises:
        KeyError: If a station is unknown.
        NoPathError: If no path satisfies the constraints.
        ValueError: If the configuration is invalid.
    """
    config = PathfindingConfig(query.mode, query.max_transfers, query.preferred_lines)
    source = network.number_of(query.start)
    target = network.number_of(query.end)
    shortest = ShortestPath(network.graph, source, config)
    if not shortest.has_path_to(target):
        if query.max_transfers is not None:
            raise NoPathError(
                f"No route from '{query.start}' to '{query.end}' within "
                f"{query.max_transfers} transfer(s)")
        raise NoPathError(f"No route found from '{query.start}' to '{query.end}'")
    nodes, edges = shortest.path_with_edges(target)
    names = [network.name_of(node) for node in nodes]
    return build_route(names, edges, query.mode)

    lines.append(f"Lines involved: {', '.join(route.lines)}")

    lines.append(f"Total distance: {route.total_distance_meters:.0f} m "

    lines.append(f"Estimated time: {route.estimated_time_minutes:.1f} minutes")
    lines.append(f"Fare: {route.fare} yuan")

"""
Data integrity checks for the subway network dataset.

Checks performed:
  * station coverage of the main lines
  * connection integrity (endpoints served by the line, duplicates, connectivity)
  * distance accuracy (positive and plausible distances)
"""

from collections import deque

REQUIRED_MAIN_LINES = (
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
)
# Distances above this value (meters) between adjacent stations are suspicious.
MAX_PLAUSIBLE_DISTANCE_METERS = 10000


class DataIntegrityChecker:
    """
    Runs integrity checks on a SubwayNetwork.

    Attributes:
        last_check_result (str): Summary of the most recent check run.
        errors (list[str]): Errors found by the most recent run.
        warnings (list[str]): Warnings found by the most recent run.
    """

    def __init__(self):
        self.last_check_result = "NOT_RUN"
        self.errors = []
        self.warnings = []

    def check_station_coverage(self, network):
        """Verify that representative stations of the main lines are present."""
        present = set(network.line_names())
        for line in REQUIRED_MAIN_LINES:
            if line not in present:
                self.warnings.append(f"Main line '{line}' is missing from the dataset")
            elif not network.stations_on_line(line):
                self.warnings.append(f"Main line '{line}' has no stations")

    def check_connections(self, network):
        """Verify connection relationships between stations and lines."""
        graph = network.graph
        seen = set()
        for edge in graph.edges():
            a = network.name_of(edge.source)
            b = network.name_of(edge.dest)
            for name in (a, b):
                if edge.line not in graph.station_lines.get(name, []):
                    self.errors.append(
                        f"Station '{name}' is connected on '{edge.line}' but is not served by it")
            key = (min(edge.source, edge.dest), max(edge.source, edge.dest), edge.line)
            if key in seen:
                self.warnings.append(
                    f"Duplicate connection between '{a}' and '{b}' on '{edge.line}'")
            seen.add(key)

        nodes = graph.nodes()
        if nodes:
            visited = {nodes[0]}
            queue = deque([nodes[0]])
            while queue:
                node = queue.popleft()
                for edge in graph.adjacent_edges(node):
                    nxt = edge.other(node)
                    if nxt not in visited:
                        visited.add(nxt)
                        queue.append(nxt)
            unreachable = [network.name_of(n) for n in nodes if n not in visited]
            if unreachable:
                self.errors.append(
                    f"{len(unreachable)} station(s) are not connected to the network: "
                    + ", ".join(unreachable[:10]))

    def check_distances(self, network):
        """Verify that connection distances are positive and plausible."""
        for edge in network.graph.edges():
            a = network.name_of(edge.source)
            b = network.name_of(edge.dest)
            if edge.distance <= 0:
                self.errors.append(f"Non-positive distance between '{a}' and '{b}': {edge.distance}")
            elif edge.distance > MAX_PLAUSIBLE_DISTANCE_METERS:
                self.warnings.append(
                    f"Unusually long distance between '{a}' and '{b}': {edge.distance} m")

    def run(self, network):
        """
        Run all checks.

        Returns:
            bool: True if no errors were found.
        """
        self.errors = []
        self.warnings = []
        self.check_station_coverage(network)
        self.check_connections(network)
        self.check_distances(network)
        passed = not self.errors
        status = "PASSED" if passed else "FAILED"
        self.last_check_result = (f"{status}: {len(self.errors)} error(s), "
                                  f"{len(self.warnings)} warning(s)")
        return passed

    def report(self):
        """Return a human-readable report of the last check run."""
        lines = [f"Data integrity check: {self.last_check_result}"]
        for error in self.errors:
            lines.append(f"  ERROR: {error}")
        for warning in self.warnings:
            lines.append(f"  WARNING: {warning}")
        return "\n".join(lines)

# Accepted separators between starting and ending station.
SEPARATORS = (";", "；", ",", "，")
EXIT_COMMANDS = ("quit", "exit")


class InputFormatError(Exception):
    """Raised when the query text does not follow the expected format."""


class StationNotFoundError(Exception):
    """Raised when a station name does not match any official station."""

    def __init__(self, role, name, suggestions=None):
        self.role = role
        self.name = name
        self.suggestions = list(suggestions or [])
        message = f"{role} station '{name}' does not exist, please check the station name"
        if self.suggestions:
            message += ". Did you mean: " + ", ".join(self.suggestions) + "?"
        super().__init__(message)


class LogicError(Exception):
    """Raised when a query is logically invalid."""


def normalize_station_name(name):
    """Normalize a station name for comparison (full-width to half-width, spacing, case)."""
    text = unicodedata.normalize("NFKC", str(name))
    text = re.sub(r"\s+", " ", text).strip()
    return text.casefold()


class StationResolver:
    """Maps user-entered station names to official station names."""

    def __init__(self, station_names):
        self._lookup = {}
        names = list(station_names)
        # Official names take priority over aliases.
        for name in names:
            self._lookup.setdefault(normalize_station_name(name), name)
        for name in names:
            key = normalize_station_name(name)
            alias = key[:-1] if key.endswith("站") else key + "站"
            if alias:
                self._lookup.setdefault(alias, name)
        self._keys = list(self._lookup.keys())

    def resolve(self, raw, role="Starting"):
        """
        Return the official station name for user input.

        Raises:
            InputFormatError: If the input is empty.
            StationNotFoundError: If no matching station exists.
        """
        key = normalize_station_name(raw)
        if not key:
            raise InputFormatError(f"{role} station name is empty")
        if key in self._lookup:
            return self._lookup[key]
        suggestions = []
        for close in difflib.get_close_matches(key, self._keys, n=3, cutoff=0.6):
            name = self._lookup[close]
            if name not in suggestions:
                suggestions.append(name)
        raise StationNotFoundError(role, str(raw).strip(), suggestions)

    """Create the command-line argument parser."""
    parser = argparse.ArgumentParser(

    parser.add_argument("--max_transfers", type=int, default=None,

    parser.add_argument("--line_preference", default=None,

def parse_line_preference(value):
    """Split the --line_preference argument into a list of line names."""
    if not value:
        return []
    return [part.strip() for part in value.split(",") if part.strip()]


def parse_query_text(text):
    """
    Split user input into (starting station, ending station).

    Raises:
        InputFormatError: If the separator is missing or a station part is empty.
    """
    stripped = text.strip()
    for separator in SEPARATORS:
        if separator in stripped:
            left, right = stripped.split(separator, 1)
            if left.strip() and right.strip():
                return left.strip(), right.strip()
            break
    raise InputFormatError(
        "Input format incorrect, please use 'Starting Station; Ending Station' format")


def answer_query(network, resolver, start_raw, end_raw, mode, max_transfers, preferred_lines):
    """
    Validate a query and compute its route.

    Returns:
        tuple: (Query, Route)

    Raises:
        StationNotFoundError, LogicError, NoPathError, ValueError, KeyError
    """
    start = resolver.resolve(start_raw, "Starting")
    end = resolver.resolve(end_raw, "Ending")
    if start == end:
        raise LogicError("Starting station cannot be the same as the ending station")
    query = Query(start, end, mode, max_transfers, preferred_lines)
    route = plan_route(network, query)
    return query, route


def run_interactive(network, resolver, exporter, mode, max_transfers, preferred_lines):
    """Run the continuous query loop until the user quits or input ends."""
    prompt = "Enter query as 'Starting Station; Ending Station' ('quit' to exit): "
    while True:
        try:
            text = input(prompt)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = text.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        try:
            start_raw, end_raw = parse_query_text(text)
            query, route = answer_query(network, resolver, start_raw, end_raw,
                                        mode, max_transfers, preferred_lines)
        except InputFormatError as exc:
            print(f"Error: {exc}")
        except StationNotFoundError as exc:
            print(f"Error: {exc}")
        except LogicError as exc:
            print(f"Error: {exc}")
        except NoPathError as exc:
            print(f"Path error: {exc}")
        except (ValueError, KeyError, ArithmeticError) as exc:
            print(f"Calculation error: {exc}")
        else:
            try:
                output = exporter.export(query, route)
            except OSError as exc:
                print(f"Error: failed to write output file: {exc}")
                output = exporter.render(query, route)
            print(output)

    print("Goodbye.")

"""
Data loading utilities for the Beijing Subway network.

The network dataset is stored in ``subway_data.json`` located in the same
directory as this module, so it can be found regardless of the working
directory the program is started from.
"""

import json
import os

# Absolute path of the bundled subway dataset (src/subway_data.json).
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and minimally validate the subway dataset.

    Args:
        path: Path to the JSON file. Defaults to DATA_FILE_PATH.

    Returns:
        dict: The JSON content with at least the keys "stations" and "connections".

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the JSON is malformed or lacks the required structure.
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)

    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")
    if not isinstance(data.get("stations"), list):
        raise ValueError("Subway data must contain a 'stations' list")
    if not isinstance(data.get("connections"), list):
        raise ValueError("Subway data must contain a 'connections' list")
    return data

"""
Core data models for the subway fare system.

Contains the weighted undirected graph (Graph), its edges (Edge), and the
result/value objects used by reporting and the CLI (Query, Transfer, Segment,
Route, PathSummary, Report).
"""

from dataclasses import dataclass, field
from typing import Dict, List, Optional


class Edge:
    """A weighted, line-tagged connection between two adjacent stations.

    Attributes:
        source: Station number of one endpoint.
        dest: Station number of the other endpoint.
        weight: Edge weight, equal to the distance in meters.
        line: Name of the subway line this connection belongs to.
    """

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.line = line
        self.weight = 0.0
        self.distance = distance  # goes through the validating setter

    @property
    def distance(self):
        """Distance in meters (same value as ``weight``)."""
        return self.weight

    @distance.setter
    def distance(self, value):
        value = float(value)
        if value != value or value <= 0:
            raise ValueError(f"Edge distance must be a positive number, got {value}")
        self.weight = value

    def other(self, node):
        """Return the endpoint opposite to ``node``."""
        if node == self.source:
            return self.dest
        if node == self.dest:
            return self.source
        raise ValueError(f"Node {node} is not an endpoint of this edge")

    def __repr__(self):
        return f"Edge({self.source}-{self.dest}, {self.weight:.0f}m, line={self.line!r})"


class Graph:
    """Weighted undirected graph of subway stations.

    Station numbers are assigned by the caller (see SubwayNetwork); the graph
    only stores the mapping and the adjacency structure.

    Attributes:
        station_to_number: dict mapping station name -> station number.
        number_to_station: dict mapping station number -> station name.
        station_lines: dict mapping station name -> list of line names.
        adjacency_list: dict mapping station number -> list of incident Edge objects.
        edges: list of all Edge objects (each undirected edge stored once).
    """

    def __init__(self):
        self.station_to_number: Dict[str, int] = {}
        self.number_to_station: Dict[int, str] = {}
        self.station_lines: Dict[str, List[str]] = {}
        self.adjacency_list: Dict[int, List[Edge]] = {}
        self.edges: List[Edge] = []

    # ----- station management -------------------------------------------------
    def register_station(self, name, number):
        """Register a station with an externally assigned number."""
        if name in self.station_to_number:
            raise ValueError(f"Station '{name}' is already registered")
        if number in self.number_to_station:
            raise ValueError(f"Station number {number} is already in use")
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.station_lines.setdefault(name, [])
        self.adjacency_list.setdefault(number, [])

    def has_station(self, name):
        """Return True if a station with this exact name is registered."""
        return name in self.station_to_number

    def set_station_lines(self, name, lines):
        """Set the (de-duplicated, order-preserving) line list of a station."""
        if name not in self.station_to_number:
            raise KeyError(f"Unknown station '{name}'")
        self.station_lines[name] = list(dict.fromkeys(lines))

    def number_of(self, name):
        """Return the station number for a station name."""
        return self.station_to_number[name]

    def name_of(self, number):
        """Return the station name for a station number."""
        return self.number_to_station[number]

    def station_count(self):
        """Return the number of registered stations."""
        return len(self.station_to_number)

    def station_names(self):
        """Return the list of all station names in number order."""
        return [self.number_to_station[n] for n in sorted(self.number_to_station)]

    def rename_station(self, old_name, new_name):
        """Rename a station while keeping its number and edges."""
        if old_name not in self.station_to_number:
            raise KeyError(f"Unknown station '{old_name}'")
        if new_name in self.station_to_number:
            raise ValueError(f"Station '{new_name}' already exists")
        number = self.station_to_number.pop(old_name)
        self.station_to_number[new_name] = number
        self.number_to_station[number] = new_name
        self.station_lines[new_name] = self.station_lines.pop(old_name, [])

    # ----- edge management ----------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge to the adjacency lists of both endpoints."""
        if edge.source not in self.number_to_station or edge.dest not in self.number_to_station:
            raise KeyError("Both endpoints of an edge must be registered stations")
        self.adjacency_list.setdefault(edge.source, []).append(edge)
        if edge.dest != edge.source:
            self.adjacency_list.setdefault(edge.dest, []).append(edge)
        self.edges.append(edge)

    def adjacent(self, number):
        """Return the list of edges incident to a station number."""
        return self.adjacency_list.get(number, [])

"""
Path calculation for the subway network.

Provides:
  * MinPQ: a binary min-heap priority queue used by Dijkstra's algorithm.
  * PathfindingConfig: query mode, transfer limit and line preference.
  * ShortestPath: Dijkstra's algorithm over (station, line) states, which
    allows counting transfers and supports the three query modes:
      - shortest_distance: minimise total distance (meters).
      - least_transfer:    minimise transfers, distance breaks ties.
      - comprehensive:     distance plus a penalty per transfer.

Time complexity is O((V + E) log V) (times the constant number of line
states per station) thanks to the heap-based priority queue.
"""

import math
from typing import Dict, List, Optional, Tuple

from models import Edge, Graph

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance of one transfer in comprehensive mode:
# about 5 minutes at 1.5 minutes per kilometre (5 / 1.5 km ~ 3333 m).
TRANSFER_PENALTY_METERS = 3333.0
# Extra cost for riding an edge that is not on a preferred line.
PREFERENCE_PENALTY_METERS = 500.0
# Large weight that makes one transfer dominate any distance in least_transfer mode.
LEAST_TRANSFER_WEIGHT = 1.0e9

# A search state is (station number, line of the edge used to arrive).
State = Tuple[int, Optional[str]]


def _normalize_line(name):
    """Normalize a line name for case/space-insensitive comparison."""
    return str(name).strip().lower().replace(" ", "")


class Node:
    """A heap entry holding a priority and an item."""

    __slots__ = ("priority", "item")

    def __init__(self, priority, item):
        self.priority = priority
        self.item = item


class MinPQ:
    """Binary min-heap priority queue.

    insert and del_min run in O(log n); the minimum is at the root.
    """

    def __init__(self):
        self._heap: List[Node] = []

    @property
    def size(self):
        """Number of elements currently in the queue."""
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def is_empty(self):
        """Return True if the queue holds no elements."""
        return not self._heap

    def insert(self, priority, item):
        """Insert an item with the given priority."""
        self._heap.append(Node(priority, item))
        self._swim(len(self._heap) - 1)

    def min_node(self):
        """Return (without removing) the node with the smallest priority."""
        if not self._heap:
            raise IndexError("MinPQ is empty")
        return self._heap[0]

    def del_min(self):
        """Remove and return the node with the smallest priority."""
        if not self._heap:
            raise IndexError("MinPQ is empty")
        heap = self._heap
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sink(0)
        return top

    def _swim(self, k):
        heap = self._heap
        while k > 0:
            parent = (k - 1) // 2
            if heap[k].priority < heap[parent].priority:
                heap[k], heap[parent] = heap[parent], heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        heap = self._heap
        n = len(heap)
        while True:
            left = 2 * k + 1
            right = left + 1
            smallest = k
            if left < n and heap[left].priority < heap[smallest].priority:
                smallest = left
            if right < n and heap[right].priority < heap[smallest].priority:
                smallest = right
            if smallest == k:
                break
            heap[k], heap[smallest] = heap[smallest], heap[k]
            k = smallest


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: One of VALID_MODES.
        max_transfers: Maximum allowed transfers, or None for unlimited.
        preferred_lines: List of preferred line names (may be empty).
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Unknown mode '{mode}', expected one of {VALID_MODES}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer or None")
            if max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines)


class ShortestPath:
    """Dijkstra-based path finder from a single source station.

    Attributes:
        dist_to: dict station number -> travelled distance (meters) of the best path.
        edge_to: dict station number -> last Edge of the best path.
        transfers_to: dict station number -> number of transfers on the best path.
    """

    def __init__(self, graph, source, config=None):
        if source not in graph.number_to_station:
            raise ValueError(f"Unknown source station number {source}")
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self._preferred = {_normalize_line(p) for p in self.config.preferred_lines}

        self.dist_to: Dict[int, float] = {}
        self.edge_to: Dict[int, Optional[Edge]] = {}
        self.transfers_to: Dict[int, int] = {}
        self.cost_to: Dict[int, float] = {}

        # state -> (cost, distance, transfers, parent_state, edge_used)
        self._states: Dict[State, tuple] = {}
        self._best_state: Dict[int, State] = {}
        self._run()

    def _increment(self, edge, switched):
        """Cost added by riding ``edge``, depending on mode and line changes."""
        inc = edge.weight
        if switched:
            if self.config.mode == MODE_LEAST_TRANSFER:
                inc += LEAST_TRANSFER_WEIGHT
            elif self.config.mode == MODE_COMPREHENSIVE:
                inc += TRANSFER_PENALTY_METERS
        if self._preferred and _normalize_line(edge.line) not in self._preferred:
            inc += PREFERENCE_PENALTY_METERS
        return inc

    def _run(self):
        """Run Dijkstra over (station, line) states."""
        max_transfers = self.config.max_transfers
        states = self._states
        start: State = (self.source, None)
        states[start] = (0.0, 0.0, 0, None, None)

        pq = MinPQ()
        pq.insert(0.0, start)
        settled = set()

        while not pq.is_empty():
            state = pq.del_min().item
            if state in settled:
                continue
            settled.add(state)
            cost, dist, transfers, _, _ = states[state]
            node, line = state

            for edge in self.graph.adjacent(node):
                nxt = edge.other(node)
                switched = line is not None and edge.line != line
                new_transfers = transfers + (1 if switched else 0)
                if max_transfers is not None and new_transfers > max_transfers:
                    continue
                new_state: State = (nxt, edge.line)
                if new_state in settled:
                    continue
                new_cost = cost + self._increment(edge, switched)
                new_dist = dist + edge.weight
                old = states.get(new_state)
                if old is None or new_cost < old[0]:
                    states[new_state] = (new_cost, new_dist, new_transfers, state, edge)
                    pq.insert(new_cost, new_state)

        # Pick the best state for every station.
        best: Dict[int, State] = {}
        for st, info in states.items():
            number = st[0]
            if number not in best or info[0] < states[best[number]][0]:
                best[number] = st
        self._best_state = best
        for number, st in best.items():
            cost, dist, transfers, _, edge = states[st]
            self.cost_to[number] = cost
            self.dist_to[number] = dist
            self.transfers_to[number] = transfers
            self.edge_to[number] = edge

    def has_path(self, target):
        """Return True if ``target`` is reachable under the current configuration."""
        return target in self._best_state

    def path_with_edges(self, target):
        """Return (station numbers, edges) of the best path from source to target.

        Returns ([], []) if the target is unreachable.
        """
        if target not in self._best_state:
            return [], []
        states = self._states
        state = self._best_state[target]
        nodes = []
        edges = []
        while state is not None:
            _, _, _, parent, edge = states[state]
            nodes.append(state[0])
            if edge is not None:
                edges.append(edge)
            state = parent
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        """Return the number of transfers on the best path to target (0 if unreachable)."""
        return self.transfers_to.get(target, 0)

"""
Fare and travel-time calculation for Beijing Subway.

Fare structure (distance in meters, intervals are left-closed, right-open):
  [0, 6 km)        -> 3 yuan
  [6 km, 12 km)    -> 4 yuan
  [12 km, 22 km)   -> 5 yuan
  [22 km, 32 km)   -> 6 yuan
  [32 km, ...)     -> 7 yuan, plus 1 yuan for every additional 20 km
                      (32-52 km: 7 yuan, 52-72 km: 8 yuan, ...)

Fractional distances are rounded up to the next whole meter before the
tier lookup, so the boundary values themselves enter the next tier.
"""

import math

# (lower bound inclusive, upper bound exclusive, fare in yuan) in meters.
FARE_TIERS = (
    (0, 6000, 3),
    (6000, 12000, 4),
    (12000, 22000, 5),
    (22000, 32000, 6),
)
EXTENSION_START_METERS = 32000
EXTENSION_BASE_FARE = 7
EXTENSION_STEP_METERS = 20000

# Travel-time model.
MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def Sum_money(distance):
    """Return the subway fare in yuan for a travelled distance in meters.

    Args:
        distance: Distance in meters (int or float, fractional values are rounded up).

    Returns:
        int: Fare in yuan.

    Raises:
        ValueError: If the distance is negative or not a finite number.
    """
    try:
        value = float(distance)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"Invalid distance: {distance!r}") from exc
    if math.isnan(value) or math.isinf(value):
        raise ValueError(f"Distance must be finite, got {distance!r}")
    if value < 0:
        raise ValueError(f"Distance must be non-negative, got {distance!r}")

    meters = math.ceil(value)
    for lower, upper, fare in FARE_TIERS:
        if lower <= meters < upper:
            return fare
    # Extended tier: 7 yuan at 32 km, +1 yuan every 20 km.
    return EXTENSION_BASE_FARE + (meters - EXTENSION_START_METERS) // EXTENSION_STEP_METERS


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes.

    Args:
        distance: Distance in meters.
        transfer_count: Number of transfers.

    Returns:
        float: About 1.5 minutes per kilometre plus about 5 minutes per transfer.
    """
    if distance < 0:
        raise ValueError("Distance must be non-negative")
    if transfer_count < 0:
        raise ValueError("Transfer count must be non-negative")
    return (distance / 1000.0) * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""
Subway network management.

SubwayNetwork builds the weighted undirected Graph from the dataset, assigns
station numbers, resolves user-supplied station names to official names, and
supports data updates (adding lines, modifying stations, updating distances).
"""

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph


class SubwayNetwork:
    """Subway network model backed by a Graph.

    Attributes:
        graph: The Graph instance holding stations, lines and edges.
        source_path: Path of the dataset file the network was loaded from.
    """

    def __init__(self, data=None, path=None):
        self.source_path = path or DATA_FILE_PATH
        if data is None:
            data = load_subway_data(self.source_path)
        self.graph = Graph()
        self._normalized_index = {}
        self._build(data)

    # ----- helpers ------------------------------------------------------------
    @staticmethod
    def normalize_name(name):
        """Normalize a station name for lookup (strip, lowercase, drop spaces and trailing '站')."""
        text = str(name).strip().replace(" ", "").lower()
        if text.endswith("站") and len(text) > 1:
            text = text[:-1]
        return text

    def _register_station(self, name, lines):
        """Create a new station with the next free number."""
        if self.graph.has_station(name):
            raise ValueError(f"Station '{name}' already exists")
        number = self.graph.station_count() + 1
        self.graph.register_station(name, number)
        self.graph.set_station_lines(name, lines)
        self._normalized_index[self.normalize_name(name)] = name
        return number

    def _ensure_station(self, name, lines):
        """Return the number of a station, creating it if needed."""
        if self.graph.has_station(name):
            return self.graph.number_of(name)
        return self._register_station(name, lines)

    def _build(self, data):
        """Populate the graph from the dataset dictionary."""
        stations = data.get("stations")
        connections = data.get("connections")
        if not isinstance(stations, list) or not isinstance(connections, list):
            raise ValueError("Dataset must contain 'stations' and 'connections' lists")

        for station in stations:
            name = str(station["name"]).strip()
            lines = [str(line).strip() for line in station.get("lines", [])]
            if self.graph.has_station(name):
                merged = self.graph.station_lines[name] + lines
                self.graph.set_station_lines(name, merged)
            else:
                self._register_station(name, lines)

        for conn in connections:
            src = str(conn["source"]).strip()
            dst = str(conn["dest"]).strip()
            line = str(conn["line"]).strip()
            distance = float(conn["distance"])
            if distance <= 0:
                raise ValueError(f"Non-positive distance between '{src}' and '{dst}'")
            src_no = self._ensure_station(src, [line])
            dst_no = self._ensure_station(dst, [line])
            self.graph.add_edge(Edge(src_no, dst_no, distance, line))

    # ----- queries ------------------------------------------------------------
    def has_station(self, name):
        """Return True if the exact official station name exists."""
        return self.graph.has_station(name)

    def resolve_station(self, name):
        """Map a user-supplied name to the official station name, or None if unknown."""
        if name is None:
            return None
        stripped = str(name).strip()
        if self.graph.has_station(stripped):
            return stripped
        return self._normalized_index.get(self.normalize_name(stripped))

    def number_of(self, name):
        """Return the number of an official station name."""
        return self.graph.number_of(name)

    def name_of(self, number):
        """Return the official name of a station number."""
        return self.graph.name_of(number)

    def lines_of(self, name):
        """Return the lines serving a station."""
        return list(self.graph.station_lines.get(name, []))

    def station_names(self):
        """Return all official station names, sorted alphabetically."""
        return sorted(self.graph.station_to_number)

    def line_names(self):
        """Return all line names found in stations or connections, sorted."""
        names = set()
        for lines in self.graph.station_lines.values():
            names.update(lines)
        for edge in self.graph.edges:
            names.add(edge.line)
        return sorted(names)

    def line_stations(self, line):
        """Return the stations served by a line, in station-number order."""
        result = []
        for number in sorted(self.graph.number_to_station):
            name = self.graph.name_of(number)
            if line in self.graph.station_lines.get(name, []):
                result.append(name)
        return result

    def transfer_stations(self):
        """Return names of stations served by more than one line."""
        return [name for name in self.station_names() if len(self.graph.station_lines[name]) > 1]

    # ----- updates ------------------------------------------------------------
    def add_line(self, line_name, station_names, distances):
        """Add a new line as a chain of stations with given inter-station distances.

        Args:
            line_name: Name of the new line (must not exist yet).
            station_names: Ordered list of station names (existing or new).
            distances: Distances in meters between consecutive stations.
        """
        line = str(line_name).strip()
        if not line:
            raise ValueError("Line name must not be empty")
        if line in self.line_names():
            raise ValueError(f"Line '{line}' already exists")
        names = [str(s).strip() for s in station_names]
        if len(names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(set(names)) != len(names):
            raise ValueError("Station names on a line must be unique")
        if len(distances) != len(names) - 1:
            raise ValueError("Number of distances must be one less than number of stations")
        for distance in distances:
            if float(distance) <= 0:
                raise ValueError("Distances must be positive")

        numbers = []
        for name in names:
            if self.graph.has_station(name):
                number = self.graph.number_of(name)
                current = self.graph.station_lines[name]
                if line not in current:
                    self.graph.set_station_lines(name, current + [line])
            else:
                number = self._register_station(name, [line])
            numbers.append(number)

        for i, distance in enumerate(distances):
            self.graph.add_edge(Edge(numbers[i], numbers[i + 1], float(distance), line))
        return numbers

    def modify_station(self, old_name, new_name=None, lines=None):
        """Rename a station and/or replace its line list. Returns the official name."""
        official = self.resolve_station(old_name)
        if official is None:
            raise KeyError(f"Unknown station '{old_name}'")
        if new_name is not None and str(new_name).strip() and str(new_name).strip() != official:
            target = str(new_name).strip()
            if self.graph.has_station(target):
                raise ValueError(f"Station '{target}' already exists")
            self.graph.rename_station(official, target)
            self._normalized_index.pop(self.normalize_name(official), None)
            self._normalized_index[self.normalize_name(target)] = target
            official = target
        if lines is not None:
            self.graph.set_station_lines(official, [str(x).strip() for x in lines])
        return official

    def update_distance(self, name_a, name_b, distance, line=None):
        """Update the distance between two adjacent stations. Returns the number of edges updated."""
        a = self.resolve_station(name_a)
        b = self.resolve_station(name_b)
        if a is None or b is None:
            raise KeyError("Both stations must exist")
        if float(distance) <= 0:
            raise ValueError("Distance must be positive")
        na = self.graph.number_of(a)
        nb = self.graph.number_of(b)
        updated = 0
        for edge in self.graph.adjacent(na):
            if edge.other(na) == nb and (line is None or edge.line == line):
                edge.distance = float(distance)
                updated += 1
        if updated == 0:
            raise KeyError(f"No connection between '{a}' and '{b}'")
        return updated

"""
Path visualization and export.

Builds Route and PathSummary objects from a path, and renders them as console
text or JSON reports. Reports can be written to a file.
"""

import json
import os

from fare_calculator import Sum_money, calculate_estimated_time
from models import PathSummary, Query, Report, Route, Segment, Transfer


def build_route(graph, station_numbers, edges):
    """Create a Route (names, distance, transfers, lines, time) from a path.

    Args:
        graph: The Graph used for name lookup.
        station_numbers: Ordered station numbers from source to target.
        edges: Edges between consecutive stations (len = len(station_numbers) - 1).
    """
    if not station_numbers:
        raise ValueError("Cannot build a route from an empty path")
    if len(station_numbers) != len(edges) + 1:
        raise ValueError("Path and edge lists are inconsistent")

    names = [graph.name_of(n) for n in station_numbers]
    total = sum(edge.weight for edge in edges)

    involved = []
    transfers = []
    for i, edge in enumerate(edges):
        if edge.line not in involved:
            involved.append(edge.line)
        if i > 0 and edges[i - 1].line != edge.line:
            transfers.append(Transfer(names[i], edges[i - 1].line, edge.line))

    segments = []
    seg_start = 0
    seg_dist = 0.0
    for i, edge in enumerate(edges):
        seg_dist += edge.weight
        is_last = i == len(edges) - 1 or edges[i + 1].line != edge.line
        if is_last:
            segments.append(Segment(edge.line, names[seg_start], names[i + 1], seg_dist))
            seg_start = i + 1
            seg_dist = 0.0

    transfer_count = len(transfers)
    estimated = calculate_estimated_time(total, transfer_count)
    return Route(
        station_sequence=names,
        total_distance_m=total,
        transfer_count=transfer_count,
        involved_lines=involved,
        estimated_time_minutes=estimated,
        transfers=transfers,
        segments=segments,
    )


def build_summary(route):
    """Extract key figures (km, fare, transfers, time) from a route."""
    fare = Sum_money(route.total_distance_m)
    return PathSummary(
        total_distance_km=route.total_distance_m / 1000.0,
        fare_yuan=fare,
        transfer_count=route.transfer_count,
        estimated_time_minutes=route.estimated_time_minutes,
    )


def format_distance(meters):
    """Format a distance: meters for short trips, kilometres for long ones (both shown)."""
    km = meters / 1000.0
    if meters < 1000:
        return f"{meters:.0f} m ({km:.2f} km)"
    return f"{km:.2f} km ({meters:.0f} m)"


def render_text(route, query, summary):
    """Render a console-formatted report."""
    rule = "=" * 60
    max_t = "unlimited" if query.max_transfers is None else str(query.max_transfers)
    out = [
        rule,
        "Beijing Subway Fare Query Result",
        rule,
        f"Start station   : {query.start_station}",
        f"End station     : {query.end_station}",
        f"Query mode      : {query.mode}",
        f"Max transfers   : {max_t}",
        f"Line preference : {query.line_preference or 'none'}",
        "",
        f"Station sequence ({len(route.station_sequence)} stations):",
    ]
    for index, name in enumerate(route.station_sequence, start=1):
        out.append(f"  {index:>3}. {name}")

    out.append("")
    out.append("Line segments:")
    if route.segments:
        for seg in route.segments:
            out.append(
                f"  [{seg.line}] {seg.from_station} -> {seg.to_station} "
                f"({format_distance(seg.distance_m)})"
            )
    else:
        out.append("  none")

    out.append("")
    out.append("Transfers:")
    if route.transfers:
        for transfer in route.transfers:
            out.append(f"  - {transfer.station_name}: {transfer.from_line} -> {transfer.to_line}")
    else:
        out.append("  none (direct route)")

    out.extend([
        "",
        "Summary:",
        f"  Total distance : {format_distance(route.total_distance_m)}",
        f"  Transfers      : {summary.transfer_count}",
        f"  Estimated time : {summary.estimated_time_minutes:.1f} minutes",
        f"  Lines involved : {', '.join(route.involved_lines) if route.involved_lines else 'none'}",
        "",
        f"Fare: {summary.fare_yuan} yuan",
        rule,
    ])
    return "\n".join(out)


def render_json(route, query, summary):
    """Render a structured JSON report for programmatic use."""
    payload = {
        "query": {
            "start_station": query.start_station,
            "end_station": query.end_station,
            "mode": query.mode,
            "max_transfers": query.max_transfers,
            "line_preference": query.line_preference,
        },
        "path": route.station_sequence,
        "distance": {
            "meters": round(route.total_distance_m, 2),
            "kilometers": round(summary.total_distance_km, 3),
        },
        "fare_yuan": summary.fare_yuan,
        "estimated_time_minutes": round(summary.estimated_time_minutes, 2),
        "transfer_count": summary.transfer_count,
        "involved_lines": route.involved_lines,
        "transfers": [
            {"station": t.station_name, "from_line": t.from_line, "to_line": t.to_line}
            for t in route.transfers
        ],
        "segments": [
            {
                "line": s.line,
                "from_station": s.from_station,
                "to_station": s.to_station,
                "distance_meters": round(s.distance_m, 2),
            }
            for s in route.segments
        ],
    }
    return json.dumps(payload, ensure_ascii=False, indent=2)


def create_report(route, query, summary, fmt="text"):
    """Create a Report in the requested format ('text' or 'json')."""
    if fmt == "text":
        content = render_text(route, query, summary)
    elif fmt == "json":
        content = render_json(route, query, summary)
    else:
        raise ValueError(f"Unsupported output format '{fmt}'")
    return Report(format=fmt, content=content)


def write_report(report, path):
    """Write a report's content to a file (UTF-8), creating directories as needed."""
    directory = os.path.dirname(os.path.abspath(path))
    os.makedirs(directory, exist_ok=True)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(report.content)
        handle.write("\n")
    report.output_file_path = path
    return report

"""
Data integrity checks for the subway network.

Checks performed:
  * Station coverage: representative stations of main lines exist.
  * Connection integrity: endpoints exist, lines are consistent, edges are
    stored in both adjacency lists, no duplicate edges, network connectivity.
  * Distance accuracy: distances are finite, positive and plausible.
"""

import math

# Representative stations for main lines used by the coverage check.
MAIN_LINE_REPRESENTATIVES = {
    "Line 1": ["苹果园", "西单", "四惠东"],
    "Line 2": ["西直门", "建国门", "东直门"],
    "Line 4": ["安河桥北", "北京南站"],
    "Line 5": ["天通苑北", "宋家庄"],
    "Line 6": ["海淀五路居", "草房"],
    "Line 8": ["朱辛庄", "鼓楼大街"],
    "Line 10": ["巴沟", "宋家庄"],
    "Line 13": ["西直门", "东直门"],
    "Line 14": ["张郭庄", "望京南"],
    "Line 15": ["望京东"],
}

# Plausible inter-station distance range in meters.
MIN_DISTANCE_M = 50.0
MAX_DISTANCE_M = 15000.0


class DataIntegrityChecker:
    """Runs integrity checks against a SubwayNetwork.

    Attributes:
        main_line_representatives: dict line name -> representative station names.
        distance_tolerance: (min, max) plausible distance range in meters.
        errors: list of error messages from the last run.
        warnings: list of warning messages from the last run.
    """

    def __init__(self, network, representatives=None, min_distance_m=MIN_DISTANCE_M,
                 max_distance_m=MAX_DISTANCE_M):
        self.network = network
        self.main_line_representatives = (
            dict(representatives) if representatives is not None else dict(MAIN_LINE_REPRESENTATIVES)
        )
        self.distance_tolerance = (float(min_distance_m), float(max_distance_m))
        self.errors = []
        self.warnings = []

    def run(self):
        """Execute all checks. Returns True when no errors were found."""
        self.errors = []
        self.warnings = []
        self.check_station_coverage()
        self.check_connection_integrity()
        self.check_distance_accuracy()
        return not self.errors

    def check_station_coverage(self):
        """Verify that representative stations of main lines are present."""
        known_lines = set(self.network.line_names())
        for line, stations in self.main_line_representatives.items():
            if line not in known_lines:
                self.errors.append(f"Main line '{line}' is missing from the network")
                continue
            for station in stations:
                if self.network.resolve_station(station) is None:
                    self.errors.append(f"Representative station '{station}' of {line} is missing")

    def check_connection_integrity(self):
        """Verify endpoints, line consistency, symmetry, duplicates and connectivity."""
        graph = self.network.graph
        if graph.station_count() == 0:
            self.errors.append("Network contains no stations")
            return

        for number in graph.number_to_station:
            name = graph.name_of(number)
            if not graph.station_lines.get(name):
                self.warnings.append(f"Station '{name}' has no line assignment")
            if not graph.adjacent(number):
                self.errors.append(f"Station '{name}' has no connections")

        seen = set()
        for edge in graph.edges:
            src = graph.name_of(edge.source)
            dst = graph.name_of(edge.dest)
            for name in (src, dst):
                if edge.line not in graph.station_lines.get(name, []):
                    self.errors.append(
                        f"Connection {src}-{dst} uses line '{edge.line}' not served by '{name}'"
                    )
            if not any(e is edge for e in graph.adjacent(edge.dest)):
                self.errors.append(f"Connection {src}-{dst} is not bidirectional")
            key = (min(src, dst), max(src, dst), edge.line)
            if key in seen:
                self.warnings.append(f"Duplicate connection {src}-{dst} on {edge.line}")
            seen.add(key)

        numbers = list(graph.number_to_station)
        reached = {numbers[0]}
        stack = [numbers[0]]
        while stack:
            current = stack.pop()
            for edge in graph.adjacent(current):
                other = edge.other(current)
                if other not in reached:
                    reached.add(other)
                    stack.append(other)
        if len(reached) != len(numbers):
            self.errors.append(
                f"{len(numbers) - len(reached)} station(s) are not reachable from "
                f"'{graph.name_of(numbers[0])}'"
            )

    def check_distance_accuracy(self):
        """Verify that every distance is finite, positive and within the plausible range."""
        graph = self.network.graph
        low, high = self.distance_tolerance
        for edge in graph.edges:
            src = graph.name_of(edge.source)
            dst = graph.name_of(edge.dest)
            value = edge.weight
            if math.isnan(value) or math.isinf(value) or value <= 0:
                self.errors.append(f"Invalid distance {value} between {src} and {dst}")
            elif value < low or value > high:
                self.warnings.append(
                    f"Suspicious distance {value:.0f} m between {src} and {dst} "
                    f"(expected {low:.0f}-{high:.0f} m)"
                )

    def report_lines(self):
        """Return human-readable lines summarizing the last run."""
        lines = []
        if not self.errors and not self.warnings:
            lines.append("Data integrity check passed: no issues found.")
        for message in self.errors:
            lines.append(f"[ERROR] {message}")
        for message in self.warnings:
            lines.append(f"[WARNING] {message}")
        if self.errors:
            lines.append(f"Data integrity check FAILED with {len(self.errors)} error(s).")
        elif self.warnings:
            lines.append(f"Data integrity check passed with {len(self.warnings)} warning(s).")
        return lines

"""
Command-line interface for the Beijing Subway fare calculation system.

Supports interactive continuous queries in the format
"Starting Station; Ending Station", plus command-line options for output
format, query mode, transfer limit, line preference, output file and data
integrity checking.
"""

import argparse
import json
import sys

from data_loader import DATA_FILE_PATH
from integrity import DataIntegrityChecker
from models import Query
from network import SubwayNetwork
from pathfinding import (
    MODE_SHORTEST_DISTANCE,
    VALID_MODES,
    PathfindingConfig,
    ShortestPath,
)
from report import build_route, build_summary, create_report, write_report

FORMAT_ERROR_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"
QUIT_COMMANDS = ("q", "quit", "exit")


class InputFormatError(ValueError):
    """Raised when the query line does not follow the 'Start; End' format."""


class QueryError(Exception):
    """Raised for user-facing query failures (unknown station, no route, etc.)."""


def parse_query_text(text):
    """Split a 'Starting Station; Ending Station' line into two stripped names."""
    if text.count(";") != 1:
        raise InputFormatError(FORMAT_ERROR_MESSAGE)
    start, end = (part.strip() for part in text.split(";"))
    if not start or not end:
        raise InputFormatError(FORMAT_ERROR_MESSAGE)
    return start, end


def execute_query(network, start_name, end_name, config):
    """Validate stations, compute the route and summary.

    Returns:
        tuple: (Query, Route, PathSummary)

    Raises:
        QueryError: On validation, path, or fare failures.
    """
    start = network.resolve_station(start_name)
    if start is None:
        raise QueryError(
            f"Starting station '{start_name}' does not exist, please check the station name"
        )
    end = network.resolve_station(end_name)
    if end is None:
        raise QueryError(
            f"Ending station '{end_name}' does not exist, please check the station name"
        )
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")

    try:
        start_no = network.number_of(start)
        end_no = network.number_of(end)
        finder = ShortestPath(network.graph, start_no, config)
        if not finder.has_path(end_no):
            if config.max_transfers is not None:
                raise QueryError(
                    f"No route from '{start}' to '{end}' within {config.max_transfers} transfer(s)"
                )
            raise QueryError(f"No route found from '{start}' to '{end}'")
        nodes, edges = finder.path_with_edges(end_no)
        route = build_route(network.graph, nodes, edges)
    except QueryError:
        raise
    except Exception as exc:
        raise QueryError(f"Path calculation failed: {exc}") from exc

    try:
        summary = build_summary(route)
    except Exception as exc:
        raise QueryError(f"Fare calculation failed: {exc}") from exc

    query = Query(
        start_station=start,
        end_station=end,
        mode=config.mode,
        max_transfers=config.max_transfers,
        line_preference=", ".join(config.preferred_lines) if config.preferred_lines else None,
    )
    return query, route, summary


def run_interactive(network, config, fmt, output_file=None, input_func=input):
    """Run the continuous query loop until the user quits or input ends.

    Returns:
        int: Process exit code (0 on normal exit).
    """
    print("Beijing Subway Fare Calculator")
    print("Enter 'Starting Station; Ending Station' to query, or 'q' to quit.")
    while True:
        try:
            line = input_func("\nQuery> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = line.strip()
        if not text:
            continue
        if text.lower() in QUIT_COMMANDS:
            break

        try:
            start_name, end_name = parse_query_text(text)
        except InputFormatError as exc:
            print(f"Error: {exc}")
            continue

        try:
            query, route, summary = execute_query(network, start_name, end_name, config)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # defensive: never crash the loop
            print(f"Error: unexpected failure during query: {exc}")
            continue

        try:
            report = create_report(route, query, summary, fmt)
        except ValueError as exc:
            print(f"Error: {exc}")
            continue

        print(report.content)
        if output_file:
            try:
                write_report(report, output_file)
                print(f"Report written to {output_file}")
            except OSError as exc:
                print(f"Error: could not write output file '{output_file}': {exc}")
    return 0


def build_parser():
    """Create the argument parser."""
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system (CLI)."
    )
    parser.add_argument("--output_file", default=None,
                        help="Write query results to this file.")
    parser.add_argument("--format", choices=["text", "json"], default="text",
                        help="Output format (default: text).")
    parser.add_argument("--mode", choices=list(VALID_MODES), default=MODE_SHORTEST_DISTANCE,
                        help="Path query mode (default: shortest_distance).")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="Maximum number of transfers allowed.")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line name, e.g. 'Line 1'.")
    parser.add_argument("--check_data", action="store_true",
                        help="Run data integrity checks and exit.")
    return parser


def main(argv=None):
    """CLI entry point. Returns the process exit code."""
    parser = build_parser()
    args = parser.parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be non-negative")

    try:
        network = SubwayNetwork(path=DATA_FILE_PATH)
    except (OSError, ValueError, KeyError, TypeError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        checker = DataIntegrityChecker(network)
        passed = checker.run()
        for line in checker.report_lines():
            print(line)
        return 0 if passed else 1

    preference = args.line_preference.strip() if args.line_preference else None
    if preference:
        known = {name.lower().replace(" ", "") for name in network.line_names()}
        if preference.lower().replace(" ", "") not in known:
            print(f"Warning: line '{preference}' is not in the network data.", file=sys.stderr)

    try:
        config = PathfindingConfig(
            mode=args.mode,
            max_transfers=args.max_transfers,
            preferred_lines=[preference] if preference else [],
        )
    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    return run_interactive(network, config, args.format, args.output_file)

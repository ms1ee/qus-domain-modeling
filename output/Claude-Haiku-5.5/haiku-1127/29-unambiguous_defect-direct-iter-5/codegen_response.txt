"""Process-wide key/value store shared by the dispatch simulator modules.

The store holds the rider list (key ``"riders"``), the order list
(``"orders"``), the system funds, ID counters, the running flag, the
simulation tick, and the loaded configuration.
"""

from typing import Any, Dict

_store: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store to its default values.

    After this call the rider list is empty, the rider and order ID counters
    are zero, the running flag is True, and the tick is 0. Funds are set to 0
    here and overwritten by ``system.init_system``.
    """
    _store.clear()
    _store.update({
        "riders": [],
        "orders": [],
        "completed_orders": [],
        "funds": 0,
        "rider_id_counter": 0,
        "order_id_counter": 0,
        "running": True,
        "tick": 0,
        "config": {},
        "rng": None,
    })


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Value to store.
    """
    _store[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key is missing.

    Returns:
        The stored value, or ``default`` when the key does not exist.
    """
    return _store.get(key, default)


def next_id(counter_key: str) -> int:
    """Increment the counter stored under ``counter_key`` and return it.

    Args:
        counter_key: Name of an integer counter in the store.

    Returns:
        The new counter value (first call returns 1).
    """
    value = int(_store.get(counter_key, 0)) + 1
    _store[counter_key] = value
    return value

"""Grid-based path planning on the 800x800 street map.

The map is an 8x8 grid of 100x100 blocks. Streets are the lines where ``x``
or ``y`` is a multiple of 100. Locations (restaurants, customers, and
intersections) sit at integer hundreds. Riders only move along streets, one
unit per step. Breadth-first search gives shortest right-angle routes, and
neighbours closer to the goal are explored first so that routes tend to run
straight whenever possible.
"""

import random
from collections import deque
from typing import Deque, Dict, List, Optional, Sequence, Tuple

MAP_SIZE = 800
BLOCK_SIZE = 100
GRID_DIMENSION = MAP_SIZE // BLOCK_SIZE  # 8 cells per side

Point = Tuple[int, int]


def is_on_map(x: int, y: int) -> bool:
    """Return True when the point lies inside the 0-800 map bounds.

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.
    """
    return 0 <= x <= MAP_SIZE and 0 <= y <= MAP_SIZE


def is_on_road(x: int, y: int) -> bool:
    """Return True when the point lies on a street.

    A point is on a street when its x or y coordinate is a multiple of 100.

    Args:
        x: Horizontal coordinate.
        y: Vertical coordinate.
    """
    if not is_on_map(x, y):
        return False
    return x % BLOCK_SIZE == 0 or y % BLOCK_SIZE == 0


def manhattan(a: Sequence[int], b: Sequence[int]) -> int:
    """Return the Manhattan distance between two points.

    Args:
        a: First point as ``(x, y)``.
        b: Second point as ``(x, y)``.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def snap_coordinate(value: float) -> int:
    """Clamp a coordinate to 0-800 and snap it to the nearest multiple of 100.

    Args:
        value: Raw coordinate value.

    Returns:
        The nearest integer hundred within the map bounds.
    """
    clamped = max(0.0, min(float(MAP_SIZE), float(value)))
    snapped = int(clamped / BLOCK_SIZE + 0.5) * BLOCK_SIZE
    return max(0, min(MAP_SIZE, snapped))


def snap_point(x: float, y: float) -> Point:
    """Snap both coordinates of a point to the nearest location (integer hundred).

    Args:
        x: Raw horizontal coordinate.
        y: Raw vertical coordinate.

    Returns:
        A ``(x, y)`` tuple where both coordinates are multiples of 100.
    """
    return (snap_coordinate(x), snap_coordinate(y))


def grid_cell(point: Sequence[int]) -> Tuple[int, int]:
    """Return the (column, row) index of the 8x8 grid cell containing a point.

    Args:
        point: Point as ``(x, y)``.

    Returns:
        Tuple ``(col, row)`` with both values in the range 0-7.
    """
    col = min(int(point[0]) // BLOCK_SIZE, GRID_DIMENSION - 1)
    row = min(int(point[1]) // BLOCK_SIZE, GRID_DIMENSION - 1)
    return (col, row)


class PathPlanner:
    """Plans street-constrained routes between map points.

    Attributes:
        _rng: Random generator used to pick random road positions.
    """

    def __init__(self, rng: Optional[random.Random] = None) -> None:
        """Create a planner.

        Args:
            rng: Optional random generator. A new one is created when omitted.
        """
        self._rng = rng if rng is not None else random.Random()

    @staticmethod
    def neighbors(point: Point, goal: Point) -> List[Point]:
        """Return the street neighbours of a point, closest-to-goal first.

        Args:
            point: Current road point.
            goal: Destination used to order the neighbours.

        Returns:
            List of neighbouring points that lie on a street.
        """
        x, y = point
        candidates = [(x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)]
        candidates = [p for p in candidates if is_on_road(p[0], p[1])]
        candidates.sort(key=lambda p: manhattan(p, goal))
        return candidates

    def find_path(self, start: Sequence[int], goal: Sequence[int]) -> Optional[List[Point]]:
        """Compute the shortest street path from ``start`` to ``goal``.

        The start point is excluded from the result and the goal point is
        included as the final element.

        Args:
            start: Starting point ``(x, y)`` on a street.
            goal: Destination point ``(x, y)`` on a street.

        Returns:
            List of points to visit in order (empty when already at the goal),
            or None when either point is off-street or the goal is unreachable.
        """
        start_pt = (int(start[0]), int(start[1]))
        goal_pt = (int(goal[0]), int(goal[1]))
        if not is_on_road(*start_pt) or not is_on_road(*goal_pt):
            return None
        if start_pt == goal_pt:
            return []

        parent: Dict[Point, Optional[Point]] = {start_pt: None}
        queue: Deque[Point] = deque([start_pt])
        while queue:
            current = queue.popleft()
            if current == goal_pt:
                break
            for nxt in self.neighbors(current, goal_pt):
                if nxt not in parent:
                    parent[nxt] = current
                    queue.append(nxt)

        if goal_pt not in parent:
            return None

        path: List[Point] = []
        node: Optional[Point] = goal_pt
        while node is not None and node != start_pt:
            path.append(node)
            node = parent[node]
        path.reverse()
        return path

    def random_road_point(self) -> Point:
        """Return a random point on a street, used to place new riders.

        Returns:
            A random ``(x, y)`` point where x or y is a multiple of 100.
        """
        if self._rng.random() < 0.5:
            y = self._rng.randint(0, GRID_DIMENSION) * BLOCK_SIZE
            x = self._rng.randint(0, MAP_SIZE)
        else:
            x = self._rng.randint(0, GRID_DIMENSION) * BLOCK_SIZE
            y = self._rng.randint(0, MAP_SIZE)
        return (x, y)

    @staticmethod
    def is_straight(a: Sequence[int], b: Sequence[int]) -> bool:
        """Return True when two points share an x or y coordinate.

        Args:
            a: First point.
            b: Second point.
        """
        return a[0] == b[0] or a[1] == b[1]

"""Order data model for the food delivery dispatch system.

An order is described by ``[restaurant_x, restaurant_y, customer_x, customer_y]``.
Coordinates are snapped to the nearest location on the street grid.
"""

from typing import Any, Dict, List, Optional, Sequence

from globalvar import get_value, next_id
from path_planner import PathPlanner, Point, snap_point

STATUS_CREATED = "created"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


class Order:
    """A single food delivery order.

    Attributes:
        id: Unique order identifier.
        res_pos: Restaurant location ``(x, y)``.
        cus_pos: Customer location ``(x, y)``.
        status: One of the ``STATUS_*`` constants. ``"completed"`` after delivery.
        rider_id: ID of the assigned rider, or None.
        start_time: Tick when the order was created.
        assigned_time: Tick when the order was assigned to a rider.
        completed_time: Tick when the order was delivered.
        distance: Street distance from restaurant to customer, in units.
    """

    def __init__(self, res_pos: Sequence[int], cus_pos: Sequence[int]) -> None:
        """Create an order and compute its delivery distance.

        Args:
            res_pos: Restaurant coordinates ``(x, y)``; snapped to the grid.
            cus_pos: Customer coordinates ``(x, y)``; snapped to the grid.
        """
        self.id: int = next_id("order_id_counter")
        self.res_pos: Point = snap_point(res_pos[0], res_pos[1])
        self.cus_pos: Point = snap_point(cus_pos[0], cus_pos[1])
        self.status: str = STATUS_CREATED
        self.rider_id: Optional[int] = None
        self.start_time: int = int(get_value("tick", 0))
        self.assigned_time: Optional[int] = None
        self.completed_time: Optional[int] = None
        route = PathPlanner().find_path(self.res_pos, self.cus_pos)
        self.distance: int = len(route) if route is not None else 0

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to a rider.

        Args:
            rider_id: ID of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED
        self.assigned_time = int(get_value("tick", 0))

    def complete(self, tick: int) -> None:
        """Mark the order as delivered.

        Args:
            tick: Simulation tick at which delivery happened.
        """
        self.status = STATUS_COMPLETED
        self.completed_time = int(tick)

    @property
    def time_taken(self) -> Optional[int]:
        """Ticks from order creation to delivery, or None if not completed."""
        if self.completed_time is None:
            return None
        return self.completed_time - self.start_time

    def to_list(self) -> List[int]:
        """Return the order in the ``[rx, ry, cx, cy]`` format."""
        return [self.res_pos[0], self.res_pos[1], self.cus_pos[0], self.cus_pos[1]]

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "coordinates": self.to_list(),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "assigned_time": self.assigned_time,
            "completed_time": self.completed_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

"""Order management: input validation, rider assignment, and completion accounting."""

import logging
from typing import TYPE_CHECKING, List, Sequence

from globalvar import get_value, set_value
from order_model import STATUS_COMPLETED, Order
from path_planner import MAP_SIZE, manhattan

if TYPE_CHECKING:
    from rider import Rider

logger = logging.getLogger(__name__)

# Extra distance-equivalent cost added per active order a rider already holds.
# This spreads load across riders instead of overloading the closest one.
LOAD_PENALTY = 300


class NoRiderAvailableError(RuntimeError):
    """Raised when no rider exists and none can be recruited."""


def validate_coordinates(values: Sequence[object]) -> List[int]:
    """Validate the four order coordinates.

    Args:
        values: Sequence ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

    Returns:
        A list of four integers.

    Raises:
        ValueError: If the count is wrong, a value is not an integer, or a
            value lies outside 0-800.
    """
    if len(values) != 4:
        raise ValueError(f"expected 4 coordinates, got {len(values)}")
    result: List[int] = []
    for value in values:
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"coordinate {value!r} is not an integer")
        if not 0 <= value <= MAP_SIZE:
            raise ValueError(f"coordinate {value} is outside the range 0-{MAP_SIZE}")
        result.append(value)
    return result


def find_best_rider_for_order(order: Order) -> "Rider":
    """Select the most suitable rider for an order.

    The cost of a rider is the Manhattan distance from its current position to
    the restaurant plus ``LOAD_PENALTY`` for each order it already holds. The
    rider with the lowest cost wins, and ties go to the lower rider ID. If no
    rider exists, one is recruited when funds allow.

    Args:
        order: Order that needs a rider.

    Returns:
        The selected Rider.

    Raises:
        NoRiderAvailableError: If no rider exists and recruitment fails.
    """
    from system import recruit_rider  # local import avoids a circular dependency

    riders = get_value("riders", [])
    if not riders:
        if recruit_rider() is None:
            raise NoRiderAvailableError("no riders available and recruitment failed")
        riders = get_value("riders", [])

    best = None
    best_key = None
    for rider in riders:
        distance = manhattan(rider.position(), order.res_pos)
        key = (distance + LOAD_PENALTY * len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best = rider
            best_key = key
    if best is None:
        raise NoRiderAvailableError("no rider could be selected")
    return best


def record_completion(order: Order) -> None:
    """Record a delivered order and credit the delivery fee to system funds.

    Args:
        order: The order that has just been delivered.
    """
    config = get_value("config", {})
    fee = int(config.get("delivery_fee", 0))
    set_value("funds", int(get_value("funds", 0)) + fee)
    get_value("completed_orders", []).append(order)
    logger.info(
        "Order %s completed by rider %s in %s ticks",
        order.id,
        order.rider_id,
        order.time_taken,
    )


class OrderManager:
    """Creates, assigns, and tracks orders."""

    def create_order(self, coordinates: Sequence[object]) -> Order:
        """Validate coordinates and create a new order.

        Args:
            coordinates: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            The newly created Order, which is also stored in the global order list.

        Raises:
            ValueError: If the coordinates are invalid.
        """
        values = validate_coordinates(coordinates)
        order = Order(values[:2], values[2:])
        get_value("orders", []).append(order)
        logger.info("Created order %s: %s", order.id, order.to_list())
        return order

    def assign_order(self, order: Order) -> "Rider":
        """Assign an order to the best rider and append it to that rider's list.

        Args:
            order: Order to assign.

        Returns:
            The rider that received the order.
        """
        rider = find_best_rider_for_order(order)
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

    def discard_order(self, order: Order) -> None:
        """Remove an order from the global list, used when assignment fails.

        Args:
            order: Order to remove.
        """
        orders = get_value("orders", [])
        if order in orders:
            orders.remove(order)

    def active_orders(self) -> List[Order]:
        """Return all orders that are not yet completed."""
        return [o for o in get_value("orders", []) if o.status != STATUS_COMPLETED]

    def all_orders(self) -> List[Order]:
        """Return every order created in this session."""
        return list(get_value("orders", []))

"""Rider model: state machine, street movement, and event logging.

State machine::

    IDLE (0)  --accepted order-->  PICKING_UP (1)
    PICKING_UP (1)  --reached restaurant-->  DELIVERING (2)
    DELIVERING (2)  --delivered to customer-->  IDLE (0)
"""

import logging
import random
from typing import Any, Dict, List, Optional

from globalvar import get_value
from order import record_completion
from order_model import STATUS_DELIVERING, STATUS_PICKING_UP, Order
from path_planner import PathPlanner, Point

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {
    IDLE: "idle",
    PICKING_UP: "picking up",
    DELIVERING: "delivering",
}

SPEED_CHOICES = (60, 80, 100)
MAX_WAIT_TICKS = 3


class Rider:
    """A delivery rider that moves along streets and serves orders in sequence.

    Attributes:
        rider_id: Unique identifier.
        x: Current horizontal coordinate.
        y: Current vertical coordinate.
        speed: Movement speed in units per tick.
        state: Current state (IDLE, PICKING_UP, or DELIVERING).
        orders: Assigned orders; the first element is the active order.
        special_case: True when the current target is directly above or below
            the rider (same x or y), i.e. a straight-line movement.
        path: Remaining street points to visit, in order.
        target: Current destination point, or None when idle.
        wait_ticks: Consecutive ticks spent waiting for a blocked point.
        distance_travelled: Total units moved.
        log: Event history entries with keys ``tick``, ``event``, ``order_id``,
            ``x``, ``y``, ``state``, and ``rider_id``.
    """

    def __init__(self, rider_id: int, rng: Optional[random.Random] = None) -> None:
        """Spawn a rider on a random street with an idle state.

        Args:
            rider_id: Unique identifier assigned by the system.
            rng: Optional random generator for placement and speed.
        """
        self.rider_id: int = rider_id
        self._rng: random.Random = rng if rng is not None else random.Random()
        self.planner = PathPlanner(self._rng)
        self.x, self.y = self.planner.random_road_point()
        self.speed: int = self._rng.choice(SPEED_CHOICES)
        self.state: int = IDLE
        self.orders: List[Order] = []
        self.special_case: bool = False
        self.path: List[Point] = []
        self.target: Optional[Point] = None
        self.wait_ticks: int = 0
        self.distance_travelled: int = 0
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    def position(self) -> Point:
        """Return the current position as ``(x, y)``."""
        return (self.x, self.y)

    @property
    def state_name(self) -> str:
        """Human-readable name of the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def move(self) -> None:
        """Advance the rider by one tick.

        An idle rider with orders accepts the first one and starts moving to
        the restaurant. The rider moves up to ``speed`` street units along its
        path, and state transitions happen when it reaches a target.
        """
        if self.state == IDLE:
            if not self.orders:
                return
            self._start_pickup()

        self._advance()

        for _ in range(2):
            if self.state == IDLE or self.path or self.target is None:
                break
            self._reach_target()

    def _start_pickup(self) -> None:
        """Accept the next order and head toward its restaurant."""
        order = self.orders[0]
        order.status = STATUS_PICKING_UP
        self.state = PICKING_UP
        self._set_target(order.res_pos)
        self._record("accepted_order", order.id)

    def _set_target(self, target: Point) -> None:
        """Set a new destination and compute the street path to reach it.

        Args:
            target: Destination point on a street.
        """
        self.target = target
        path = self.planner.find_path(self.position(), target)
        if path is None:
            logger.warning("Rider %s found no path to %s", self.rider_id, target)
            path = []
        self.path = path
        self.special_case = (
            (self.x == target[0] or self.y == target[1])
            and self.position() != target
        )

    def _advance(self) -> None:
        """Move along the path by up to ``speed`` units, respecting blocking."""
        steps = self.speed
        while steps > 0 and self.path:
            if self._is_blocked():
                self.wait_ticks += 1
                break
            self.wait_ticks = 0
            self.x, self.y = self.path.pop(0)
            self.distance_travelled += 1
            steps -= 1

    def _is_blocked(self) -> bool:
        """Return True when the next point is held by another rider.

        A rider waits at most ``MAX_WAIT_TICKS`` ticks, then proceeds. This
        prevents deadlocks when two riders block each other.
        """
        if not self.path:
            return False
        if self.wait_ticks >= MAX_WAIT_TICKS:
            return False
        nxt = self.path[0]
        for other in get_value("riders", []):
            if other is not self and other.position() == nxt:
                return True
        return False

    def _reach_target(self) -> None:
        """Handle arrival at the current target and perform the state transition."""
        if not self.orders:
            self._become_idle()
            return
        order = self.orders[0]
        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.status = STATUS_DELIVERING
            self._record("arrived_at_merchant", order.id)
            self._set_target(order.cus_pos)
        elif self.state == DELIVERING:
            tick = int(get_value("tick", 0))
            order.complete(tick)
            self._record("delivered_to_customer", order.id)
            self.orders.pop(0)
            record_completion(order)
            self._become_idle()

    def _become_idle(self) -> None:
        """Reset the rider to the idle state with no active destination."""
        self.state = IDLE
        self.target = None
        self.path = []
        self.special_case = False
        self.wait_ticks = 0

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an entry to the rider's event log.

        Args:
            event: Event name, e.g. ``"accepted_order"``.
            order_id: Related order ID, or None.
        """
        self.log.append({
            "tick": int(get_value("tick", 0)),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

"""System-level management: initialisation, funds, recruitment, clock, and persistence."""

import json
import logging
import math
import os
import random
from typing import Any, Dict, List, Optional

import globalvar
from globalvar import get_value, next_id, set_value
from order import NoRiderAvailableError, OrderManager  # noqa: F401 (re-exported)
from path_planner import PathPlanner, is_on_road, snap_coordinate
from rider import IDLE, Rider

logger = logging.getLogger(__name__)

DEFAULT_CONFIG: Dict[str, Any] = {
    "initial_funds": 10000,
    "recruit_cost": 500,
    "max_riders": 10,
    "delivery_fee": 80,
    "operating_cost_per_tick": 1,
    "random_seed": None,
}

_INT_KEYS = ("initial_funds", "recruit_cost", "max_riders", "delivery_fee", "operating_cost_per_tick")


def load_config(path: Optional[str] = None) -> Dict[str, Any]:
    """Load system parameters from a JSON file, falling back to defaults.

    The file may contain a top-level ``"parameters"`` object, or the parameters
    may be given at the top level. Unknown keys are ignored.

    Args:
        path: Path to the JSON configuration file, or None to use defaults.

    Returns:
        A dictionary of system parameters.
    """
    config = dict(DEFAULT_CONFIG)
    if not path:
        return config
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except FileNotFoundError:
        logger.info("No configuration file at %s; using defaults", path)
        return config
    except (OSError, json.JSONDecodeError) as exc:
        logger.warning("Could not read configuration %s: %s", path, exc)
        return config

    if not isinstance(data, dict):
        logger.warning("Configuration %s is not a JSON object; using defaults", path)
        return config
    params = data.get("parameters", data)
    if not isinstance(params, dict):
        return config

    for key in _INT_KEYS:
        if key in params:
            try:
                config[key] = int(params[key])
            except (TypeError, ValueError):
                logger.warning("Invalid value for %s in %s; using default", key, path)
    if "random_seed" in params:
        config["random_seed"] = params["random_seed"]
    return config


def init_system(config_path: Optional[str] = None) -> "SystemManager":
    """Initialise global state and return the system manager.

def recruit_rider() -> Optional[Rider]:
    """Recruit a new rider if the limit and funds allow.

    The recruitment cost is deducted from system funds. The new rider gets the
    next auto-incremented ID and is appended to the global rider list.

    Returns:
        The new Rider, or None when recruitment is not possible.
    """
    config = get_value("config", {})
    riders = get_value("riders", [])
    if len(riders) >= int(config.get("max_riders", 10)):
        logger.info("Recruitment skipped: rider limit reached")
        return None
    cost = int(config.get("recruit_cost", 0))
    funds = int(get_value("funds", 0))
    if funds < cost:
        logger.warning("Recruitment skipped: funds %s below cost %s", funds, cost)
        return None
    set_value("funds", funds - cost)
    rider_id = next_id("rider_id_counter")
    rider = Rider(rider_id, rng=get_value("rng"))
    riders.append(rider)
    logger.info("Recruited rider %s at (%s, %s)", rider.rider_id, rider.x, rider.y)
    return rider


class SystemManager:
    """Coordinates funds, riders, orders, and the simulation clock.

    Attributes:
        order_manager: Order creation and assignment helper.
        planner: Path planner used for route previews.
    """

    def __init__(self) -> None:
        """Create the manager using the current global state."""
        self.order_manager = OrderManager()
        self.planner = PathPlanner(get_value("rng"))

    @property
    def funds(self) -> int:
        """Current system fund balance (may be negative)."""
        return int(get_value("funds", 0))

    @property
    def riders(self) -> List[Rider]:
        """The live rider list."""
        return get_value("riders", [])

    @property
    def tick_count(self) -> int:
        """Current simulation tick."""
        return int(get_value("tick", 0))

    @property
    def config(self) -> Dict[str, Any]:
        """Loaded system parameters."""
        return get_value("config", {})

    def stop(self) -> None:
        """Clear the running flag so the main loop exits."""
        set_value("running", False)

    def should_recruit(self) -> bool:
        """Return True when riders are insufficient and funds allow recruitment.

        Recruitment is considered when the rider limit is not reached, funds
        cover the cost, and either no riders exist or every rider is busy.
        """
        cfg = self.config
        riders = self.riders
        if len(riders) >= int(cfg.get("max_riders", 10)):
            return False
        if self.funds < int(cfg.get("recruit_cost", 0)):
            return False
        if not riders:
            return True
        return all(rider.orders for rider in riders)

    def submit_order(self, coordinates: List[object]) -> Dict[str, Any]:
        """Create an order, recruit if needed, assign it, and preview its route.

        Args:
            coordinates: ``[restaurant_x, restaurant_y, customer_x, customer_y]``.

        Returns:
            Dictionary with keys ``order``, ``rider``, ``new_rider``,
            ``queued_ahead``, and ``route`` (the route dictionary).

        Raises:
            ValueError: If the coordinates are invalid.
            NoRiderAvailableError: If no rider can serve the order.
        """
        order = self.order_manager.create_order(coordinates)
        try:
            new_rider = False
            if self.should_recruit():
                new_rider = recruit_rider() is not None
            rider = self.order_manager.assign_order(order)
        except Exception:
            self.order_manager.discard_order(order)
            raise
        queued_ahead = len(rider.orders) - 1
        route = self.plan_route(rider, order)
        return {
            "order": order,
            "rider": rider,
            "new_rider": new_rider,
            "queued_ahead": queued_ahead,
            "route": route,
        }

    def plan_route(self, rider: Rider, order: Any) -> Dict[str, Any]:
        """Build the route from the rider's position through restaurant to customer.

        Args:
            rider: Rider assigned to the order.
            order: The order being routed.

        Returns:
            Dictionary with ``points``, ``pickup_distance``, ``delivery_distance``,
            and ``eta_ticks``. The ETA ignores orders queued ahead of this one.
        """
        to_restaurant = self.planner.find_path(rider.position(), order.res_pos) or []
        to_customer = self.planner.find_path(order.res_pos, order.cus_pos) or []
        speed = max(1, int(rider.speed))
        eta = math.ceil(len(to_restaurant) / speed) + math.ceil(len(to_customer) / speed)
        return {
            "points": [rider.position()] + to_restaurant + to_customer,
            "pickup_distance": len(to_restaurant),
            "delivery_distance": len(to_customer),
            "eta_ticks": eta,
        }

    def tick(self) -> int:
        """Advance the simulation by one tick.

        Increments the clock, charges operating costs for each rider, and moves
        every rider. Exceptions from a single rider are logged and do not stop
        the other riders.

        Returns:
            The new tick value.
        """
        set_value("tick", self.tick_count + 1)
        riders = list(self.riders)
        cost = int(self.config.get("operating_cost_per_tick", 0)) * len(riders)
        set_value("funds", self.funds - cost)
        for rider in riders:
            try:
                rider.move()
            except Exception:  # keep the simulation running after an unexpected error
                logger.exception("Rider %s failed during tick %s", rider.rider_id, self.tick_count)
        return self.tick_count

    def check_consistency(self) -> List[str]:
        """Detect and repair inconsistent system state.

        Repairs performed: removes completed orders from rider lists, resets
        riders that have a target but no orders, snaps off-street riders onto
        the nearest street, and restores integer funds.

        Returns:
            Human-readable descriptions of the issues found and repaired.
        """
        issues: List[str] = []
        riders = self.riders

        if not isinstance(get_value("funds"), int):
            issues.append("funds were not an integer; reset to integer value")
            set_value("funds", int(get_value("funds", 0) or 0))

        seen_ids = set()
        for rider in riders:
            if rider.rider_id in seen_ids:
                issues.append(f"duplicate rider id {rider.rider_id} detected")
            seen_ids.add(rider.rider_id)

            active = [o for o in rider.orders if o.status != "completed"]
            if len(active) != len(rider.orders):
                issues.append(f"rider {rider.rider_id} held completed orders; removed")
                rider.orders[:] = active

            if rider.state != IDLE and not rider.orders:
                issues.append(f"rider {rider.rider_id} was busy without orders; reset to idle")
                rider._become_idle()

            if not is_on_road(rider.x, rider.y):
                new_x = rider.x if rider.x % 100 == 0 else rider.x
                new_y = rider.y if rider.y % 100 == 0 else snap_coordinate(rider.y)
                if not is_on_road(new_x, new_y):
                    new_x = snap_coordinate(rider.x)
                issues.append(f"rider {rider.rider_id} was off-street; moved to ({new_x}, {new_y})")
                rider.x, rider.y = new_x, new_y
                rider.path = []
                rider.target = None
                rider.state = IDLE if not rider.orders else rider.state
        for issue in issues:
            logger.warning("Consistency: %s", issue)
        return issues

    def save_state(self, path: str) -> None:
        """Write parameters, funds, riders, and order history to a JSON file.

        The file is written to a temporary name and then moved into place, so a
        crash during the write does not corrupt the previous file.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        data = {
            "parameters": self.config,
            "funds": self.funds,
            "tick": self.tick_count,
            "rider_id_counter": get_value("rider_id_counter", 0),
            "order_id_counter": get_value("order_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in get_value("orders", [])],
        }
        tmp_path = path + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        os.replace(tmp_path, path)
        logger.info("State saved to %s", path)

class DisplayManager:

    LEGEND = (

        """Return a snapshot of funds and every rider's state.

        Args:

            f"=== System status (tick {tick}) ===",
            f"System funds: {funds}",
            f"Riders: {len(riders)} / {max_riders}",

            lines.append("No riders on duty.")
            return "\n".join(lines)
        lines.append(f"{'ID':>4} {'Position':>12} {'State':<12} {'Orders':>6}  Order IDs")

            position = f"({rider.x}, {rider.y})"

        for o in orders:
            time_text = str(o.time_taken) if o.time_taken is not None else "-"
            rider_text = str(o.rider_id) if o.rider_id is not None else "-"
            restaurant = f"({o.res_pos[0]}, {o.res_pos[1]})"

    def render_map(
        self,
        riders: Iterable[Rider] = (),
        orders: Iterable = (),
        route: Optional[Iterable[Tuple[int, int]]] = None,
    ) -> str:
        """Render the 800x800 street map as ASCII art at 50-unit resolution.

        Args:
            riders: Riders to mark on the map.
            orders: Orders whose active restaurants and customers are marked.
            route: Optional sequence of points drawn with ``*``.

        Returns:
            Multi-line string containing the map and a legend.
        """
        rider_marks = {}
        for rider in riders:
            rider_marks[rider.position()] = str(rider.rider_id % 10)

        active = [o for o in orders if o.status != STATUS_COMPLETED]
        restaurants: Set[Tuple[int, int]] = {o.res_pos for o in active}
        customers: Set[Tuple[int, int]] = {o.cus_pos for o in active}
        route_points: Set[Tuple[int, int]] = set(route or [])

        steps = range(0, MAP_SIZE + 1, MAP_STEP)
        header = "      " + "".join(
            str(x // BLOCK_SIZE) if x % BLOCK_SIZE == 0 else " " for x in steps
        )
        lines = [self.LEGEND, header + "   <- x in hundreds"]
        for y in reversed(list(steps)):
            label = f"{y:>5} " if y % BLOCK_SIZE == 0 else "      "
            row = "".join(
                self._symbol((x, y), rider_marks, restaurants, customers, route_points)
                for x in steps
            )
            lines.append(label + row)
        return "\n".join(lines)

    @staticmethod
    def _symbol(
        point: Tuple[int, int],
        rider_marks: dict,
        restaurants: Set[Tuple[int, int]],
        customers: Set[Tuple[int, int]],
        route_points: Set[Tuple[int, int]],
    ) -> str:
        """Return the map character for a single point, by display priority.

        Riders are drawn first, then restaurants, customers, the route, and
        finally the streets and empty space.
        """
        if point in rider_marks:
            return rider_marks[point]
        if point in restaurants:
            return "R"
        if point in customers:
            return "C"
        if point in route_points:
            return "*"
        x, y = point
        on_x = x % BLOCK_SIZE == 0
        on_y = y % BLOCK_SIZE == 0
        if on_x and on_y:
            return "+"
        if on_y:
            return "-"
        if on_x:
            return "|"
        return " "

class ValidationError(ValueError):
    """Raised when user input is malformed or out of range."""


class InputManager:
    """Reads user input, parses commands, and validates coordinates.

    Attributes:
        history: Non-empty command lines entered during this session.
    """

    def __init__(self, input_func: Callable[[str], str] = input) -> None:
        """Create the input manager.

        Args:
            input_func: Function used to read a line, injectable for testing.
        """
        self._input = input_func
        self.history: List[str] = []

    def read_line(self, prompt: str) -> str:
        """Read one line of input and record it in the history.

        Args:
            prompt: Prompt text shown to the user.

        Raises:
            EOFError: If input is closed.
        """
        line = self._input(prompt)
        if line.strip():
            self.history.append(line.strip())
        return line

    @staticmethod
    def parse_command(line: str) -> Optional[Tuple[str, List[str]]]:
        """Parse a command line into a canonical command name and arguments.

        Commas are treated as spaces, so ``start 100,200,300,400`` works.

        Args:
            line: Raw input line.

        Returns:
            ``(command, args)``, or None for an empty line.

        Raises:
            ValidationError: If the command is unknown.
        """
        tokens = line.replace(",", " ").split()
        if not tokens:
            return None
        word = tokens[0].lower()
        command = COMMAND_ALIASES.get(word)
        if command is None:
            raise ValidationError(f"unknown command '{tokens[0]}'. Type 'help' for the list.")
        return command, tokens[1:]

    @staticmethod
    def parse_coordinates(tokens: List[str]) -> List[int]:
        """Convert four tokens into validated integer coordinates.

        Args:
            tokens: Four string tokens.

        Returns:
            List of four integers in the range 0-800.

        Raises:
            ValidationError: If the count is wrong, a token is not an integer,
                or a value is out of range.
        """
        if len(tokens) != 4:
            raise ValidationError(f"expected 4 integers, got {len(tokens)}")
        values: List[int] = []
        for token in tokens:
            try:
                value = int(token)
            except ValueError:
                raise ValidationError(f"'{token}' is not an integer") from None
            if not 0 <= value <= MAP_SIZE:
                raise ValidationError(f"coordinate {value} is out of range 0-{MAP_SIZE}")
            values.append(value)
        return values

    def read_order_coordinates(self, args: List[str]) -> Optional[List[int]]:
        """Obtain the four order coordinates from arguments or by prompting.

        When arguments are given they are validated directly. Otherwise the user
        is prompted until the input is valid. An empty line cancels the order.

        Args:
            args: Arguments that followed the ``start`` command.

        Returns:
            ``[restaurant_x, restaurant_y, customer_x, customer_y]``, or None if cancelled.

        Raises:
            ValidationError: If arguments were supplied and are invalid.
        """
        if args:
            return self.parse_coordinates(args)
        prompt = "Enter restaurant_x restaurant_y customer_x customer_y (0-800), blank to cancel: "
        while True:
            text = self.read_line(prompt)
            if not text.strip():
                return None
            try:
                return self.parse_coordinates(text.replace(",", " ").split())
            except ValidationError as exc:
                print(f"Invalid input: {exc}. Try again or press Enter to cancel.")

    @staticmethod
    def parse_wait_ticks(args: List[str]) -> int:
        """Parse the optional tick count of the ``wait`` command.

        Args:
            args: Command arguments; empty means one tick.

        Returns:
            Number of ticks, between 1 and 1000.

        Raises:
            ValidationError: If the value is not a valid positive integer.
        """
        if not args:
            return 1
        try:
            value = int(args[0])
        except ValueError:
            raise ValidationError(f"'{args[0]}' is not an integer") from None
        if not 1 <= value <= MAX_WAIT_TICKS:
            raise ValidationError(f"tick count must be between 1 and {MAX_WAIT_TICKS}")
        return value

class DispatchApp:
    """Runs the interactive command loop.

    Attributes:
        sysmgr: The system manager.
        display: Output formatter.
        inputs: Input reader and parser.
    """

    def __init__(self, sysmgr: SystemManager, display: DisplayManager, inputs: InputManager) -> None:
        """Create the application.

        Args:
            sysmgr: Initialised system manager.
            display: Display manager for output formatting.
            inputs: Input manager for reading commands.
        """
        self.sysmgr = sysmgr
        self.display = display
        self.inputs = inputs
        self._handlers: Dict[str, Callable[[List[str]], None]] = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "wait": self.cmd_wait,
            "history": self.cmd_history,
            "help": self.cmd_help,
        }

    def run(self) -> None:
        """Run the main loop until the user quits or input closes."""
        print("Food delivery dispatch simulator. Type 'help' for commands.")
        while get_value("running", True):
            try:
                line = self.inputs.read_line(f"[t={self.sysmgr.tick_count}] dispatch> ")
            except (EOFError, KeyboardInterrupt):
                print()
                self.sysmgr.stop()
                break

            try:
                parsed = InputManager.parse_command(line)
            except ValidationError as exc:
                print(f"Error: {exc}")
                parsed = None

            if parsed is not None:
                command, args = parsed
                if command == "quit":
                    self.sysmgr.stop()
                    break
                self._execute(command, args)

            if get_value("running", True):
                self.sysmgr.tick()

    def _execute(self, command: str, args: List[str]) -> None:
        """Run a command handler and convert failures into user messages.

        Args:
            command: Canonical command name.
            args: Arguments following the command.
        """
        handler = self._handlers.get(command)
        if handler is None:
            print(f"Error: command '{command}' is not implemented.")
            return
        try:
            handler(args)
        except (ValidationError, ValueError) as exc:
            print(f"Error: {exc}")
        except NoRiderAvailableError as exc:
            print(f"Order could not be assigned: {exc}")
        except Exception:  # error recovery: log, repair state, keep running
            logging.getLogger(__name__).exception("Command '%s' failed", command)
            print("An internal error occurred. System state was checked and repaired where possible.")
            for issue in self.sysmgr.check_consistency():
                print(f"  repaired: {issue}")

    def cmd_start(self, args: List[str]) -> None:
        """Place an order, assign a rider, and show the route map.

        Args:
            args: Optional four coordinates. If omitted the user is prompted.
        """
        coords = self.inputs.read_order_coordinates(args)
        if coords is None:
            print("Order cancelled.")
            return

        result = self.sysmgr.submit_order(coords)
        order = result["order"]
        rider = result["rider"]
        route = result["route"]
        config = self.sysmgr.config

        from path_planner import grid_cell  # local import keeps top-level imports short

        res_cell = grid_cell(order.res_pos)
        cus_cell = grid_cell(order.cus_pos)
        print(f"Order #{order.id} created at tick {order.start_time}.")
        print(
            f"  Restaurant: ({order.res_pos[0]}, {order.res_pos[1]}) "
            f"grid cell {res_cell}"

            f"  Customer:   ({order.cus_pos[0]}, {order.cus_pos[1]}) "
            f"grid cell {cus_cell}"
        )
        if result["new_rider"]:
            print(
                f"New rider #{rider.rider_id} recruited "
                f"(cost {config.get('recruit_cost', 0)}); funds now {self.sysmgr.funds}."
            )
        print(
            f"Assigned to rider #{rider.rider_id} at ({rider.x}, {rider.y}), "
            f"speed {rider.speed} units/tick, orders ahead: {result['queued_ahead']}."
        )
        print(
            f"Route: pickup {route['pickup_distance']} units, delivery "
            f"{route['delivery_distance']} units, estimated {route['eta_ticks']} ticks "
            f"(excludes queued orders)."
        )
        print(self.display.render_map(self.sysmgr.riders, get_value("orders", []), route["points"]))

    def cmd_status(self, args: List[str]) -> None:
        """Show funds, rider states, and order lists.

        Args:
            args: Unused.
        """
        config = self.sysmgr.config
        print(
            self.display.format_status(
                self.sysmgr.funds,
                self.sysmgr.riders,
                self.sysmgr.tick_count,
                int(config.get("max_riders", 10)),
            )
        )
        for issue in self.sysmgr.check_consistency():
            print(f"  repaired: {issue}")

    def cmd_orders(self, args: List[str]) -> None:
        """Show details of every order.

        Args:
            args: Unused.
        """
        print(self.display.format_orders(get_value("orders", [])))

    def cmd_riders(self, args: List[str]) -> None:
        """Show the full status change history of every rider.

        Args:
            args: Unused.
        """
        print(self.display.format_rider_history(self.sysmgr.riders))

    def cmd_map(self, args: List[str]) -> None:
        """Show the ASCII map with riders and active orders.

        Args:
            args: Unused.
        """
        print(self.display.render_map(self.sysmgr.riders, get_value("orders", [])))

    def cmd_wait(self, args: List[str]) -> None:
        """Advance the simulation by several ticks.

        The main loop already ticks once after each command, so this command
        performs ``n - 1`` additional ticks.

        Args:
            args: Optional tick count.
        """
        ticks = InputManager.parse_wait_ticks(args)
        for _ in range(ticks - 1):
            self.sysmgr.tick()
        print(f"Advanced {ticks} tick(s).")

    def cmd_history(self, args: List[str]) -> None:
        """Print the commands entered in this session.

        Args:
            args: Unused.
        """
        if not self.inputs.history:
            print("No commands entered yet.")
            return
        for index, entry in enumerate(self.inputs.history, start=1):
            print(f"{index:>4}  {entry}")

    def cmd_help(self, args: List[str]) -> None:
        """Print the help text.

        Args:
            args: Unused.
        """
        print(HELP_TEXT)


def main(argv: Optional[List[str]] = None) -> int:
    """Parse arguments, initialise the system, run the loop, and save state.

    Args:
        argv: Optional argument list; defaults to ``sys.argv[1:]``.

    Returns:
        Process exit code.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=CONFIG_PATH, help="JSON configuration file")
    parser.add_argument("--state", default=STATE_PATH, help="file to save state to on exit")
    options = parser.parse_args(argv)

    configure_logging()
    sysmgr = init_system(options.config)
    app = DispatchApp(sysmgr, DisplayManager(), InputManager())
    try:
        app.run()
    finally:
        try:
            sysmgr.save_state(options.state)
            print(f"State saved to {options.state}. Goodbye.")
        except OSError as exc:
            logging.getLogger(__name__).error("Could not save state: %s", exc)
            print(f"Warning: could not save state: {exc}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

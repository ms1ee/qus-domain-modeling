"""Process-wide global variable store.

The store is a simple dictionary shared by the simulator modules. It holds
shared references such as the rider list (key ``"riders"``) and the simulated
clock (key ``"current_tick"``).
"""

_global_dict = {}


def _init():
    """Reset the global variable dictionary to an empty state."""
    global _global_dict
    _global_dict = {}


def set_value(key, value):
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable.
        value: Any Python object to store.
    """
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when ``key`` is not present.

    Returns:
        The stored value, or ``default`` if the key does not exist.
    """
    return _global_dict.get(key, default)


def current_tick():
    """Return the current simulated tick as an integer (0 if unset)."""
    return get_value("current_tick") or 0

"""Grid map and street-based path planning.

The city is an 800x800 coordinate plane divided into an 8x8 grid of 100x100
cells. Every grid line is a street; points whose x or y is a multiple of 100
lie on a street, and points where both are multiples of 100 are intersections
(restaurant and customer locations).
"""

import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = MAP_SIZE // CELL_SIZE


class GridMap:
    """Describes the coordinate system and street layout of the city."""

    def __init__(self, map_size=MAP_SIZE, cell_size=CELL_SIZE):
        """Create a grid map.

        Args:
            map_size: Side length of the square map in units.
            cell_size: Side length of one grid cell in units.

        Raises:
            ValueError: If the sizes are not positive or do not divide evenly.
        """
        if map_size <= 0 or cell_size <= 0 or map_size % cell_size != 0:
            raise ValueError("map_size must be a positive multiple of cell_size")
        self.map_size = map_size
        self.cell_size = cell_size
        self.grid_dimension = map_size // cell_size

    def contains(self, x, y):
        """Return True if the point lies inside the map bounds."""
        return 0 <= x <= self.map_size and 0 <= y <= self.map_size

    def is_street(self, x, y):
        """Return True if the point lies on a street (a grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def is_intersection(self, x, y):
        """Return True if the point is a grid intersection."""
        return x % self.cell_size == 0 and y % self.cell_size == 0

    def cell_of(self, x, y):
        """Return the (column, row) index of the cell containing the point."""
        col = min(x // self.cell_size, self.grid_dimension - 1)
        row = min(y // self.cell_size, self.grid_dimension - 1)
        return col, row

    def cell_center(self, value):
        """Return the center coordinate of the cell containing ``value``.

        Implements ``center = (value // 100) * 100 + 50``.
        """
        return (value // self.cell_size) * self.cell_size + self.cell_size // 2


class PathPlanner:
    """Computes street-following routes and per-tick movement.

    Routes always travel along streets using right-angle turns. Riders may
    share streets freely and never wait for one another, so no path conflict
    or deadlock can arise between riders.
    """

    def __init__(self, grid_map=None):
        """Create a planner bound to a grid map.

        Args:
            grid_map: Optional GridMap; a default 8x8 map is used otherwise.
        """
        self.grid_map = grid_map or GridMap()
        self.grid_dimension = self.grid_map.grid_dimension

    @staticmethod
    def next_position(x, y, tx, ty, speed):
        """Advance a position toward a target by at most ``speed`` units.

        Movement is horizontal first, then vertical (right-angle turn). When
        the rider is directly above or below the target, only vertical
        movement is used (straight-line special case).

        Args:
            x, y: Current coordinates.
            tx, ty: Target coordinates.
            speed: Maximum distance to move this tick.

        Returns:
            Tuple (new_x, new_y).
        """
        remaining = max(0, int(speed))
        while remaining > 0:
            if x != tx:
                step = min(remaining, abs(tx - x))
                x += step if tx > x else -step
                remaining -= step
            elif y != ty:
                step = min(remaining, abs(ty - y))
                y += step if ty > y else -step
                remaining -= step
            else:
                break
        return x, y

    @staticmethod
    def plan_route(start, end):
        """Return the corner points of a street route from ``start`` to ``end``.

        Args:
            start: Tuple (x, y).
            end: Tuple (x, y).

        Returns:
            List of (x, y) waypoints beginning with ``start`` and ending with
            ``end``. A straight route has two points, an L-shaped route three.
        """
        sx, sy = start
        ex, ey = end
        if sx == ex or sy == ey:
            return [(sx, sy), (ex, ey)]
        return [(sx, sy), (ex, sy), (ex, ey)]

    @staticmethod
    def route_length(route):
        """Return the total Manhattan length of a list of waypoints."""
        total = 0
        for (ax, ay), (bx, by) in zip(route, route[1:]):
            total += abs(bx - ax) + abs(by - ay)
        return total

    def plan_multi_stop(self, start, stops):
        """Build one continuous route that visits ``stops`` in order.

        Used for the restaurant-then-customer path of an order.

        Args:
            start: Tuple (x, y) where the rider currently is.
            stops: Ordered list of (x, y) destinations.

        Returns:
            List of waypoints starting at ``start``.
        """
        route = [tuple(start)]
        current = tuple(start)
        for stop in stops:
            leg = self.plan_route(current, stop)
            route.extend(leg[1:])
            current = tuple(stop)
        return route

    def optimize_stops(self, start, stops):
        """Greedily reorder stops by nearest Manhattan distance.

        Args:
            start: Tuple (x, y) origin.
            stops: List of (x, y) destinations.

        Returns:
            New list with the stops in a nearest-first visiting order.
        """
        remaining = list(stops)
        ordered = []
        current = tuple(start)
        while remaining:
            nxt = min(remaining,
                      key=lambda p: abs(p[0] - current[0]) + abs(p[1] - current[1]))
            ordered.append(nxt)
            remaining.remove(nxt)
            current = nxt
        return ordered

    @staticmethod
    def estimate_ticks(distance, speed):
        """Return the number of ticks needed to cover ``distance``."""
        if speed <= 0:
            raise ValueError("speed must be positive")
        return math.ceil(distance / speed)

"""Rider model: state machine, movement and event logging.

Rider states:
    0 (IDLE): available, no active order.
    1 (PICKING_UP): en route to the restaurant.
    2 (DELIVERING): en route to the customer.

Events recorded in ``Rider.log``:
    "spawn", "accepted_order", "arrived_at_merchant", "delivered_to_customer".
"""

import logging

import globalvar
from path_planner import MAP_SIZE, PathPlanner

LOGGER = logging.getLogger("dispatch.rider")

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

DEFAULT_RIDER_SPEED = 20
SPAWN_X = 400
SPAWN_Y = 400


class Rider:
    """A delivery rider that moves along streets and serves orders."""

    def __init__(self, rider_id, speed=DEFAULT_RIDER_SPEED):
        """Create a rider spawned at (400, 400) in the idle state.

        Args:
            rider_id: Unique integer identifier.
            speed: Movement speed in units per tick.
        """
        self.rider_id = int(rider_id)
        self.x = SPAWN_X
        self.y = SPAWN_Y
        self.speed = int(speed)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._log_event("spawn", None)

    # ------------------------------------------------------------------
    # Logging and queries
    # ------------------------------------------------------------------
    def _log_event(self, event, order_id):
        """Append a status-change record to the rider log."""
        self.log.append({
            "time": globalvar.current_tick(),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
        })

    def state_name(self):
        """Return the human-readable name of the current state."""
        return STATE_NAMES.get(self.state, "unknown")

    def current_order(self):
        """Return the first order that is not completed, or None."""
        for order in self.orders:
            if order.status != "completed":
                return order
        return None

    def active_order_count(self):
        """Return the number of orders not yet completed."""
        return sum(1 for order in self.orders if order.status != "completed")

    def is_idle(self):
        """Return True if the rider has no active order."""
        return self.active_order_count() == 0

    # ------------------------------------------------------------------
    # State transitions and movement
    # ------------------------------------------------------------------
    def _accept(self, order):
        """Transition from idle (or after a delivery) to picking up."""
        self.state = PICKING_UP
        order.status = "picking_up"
        self._log_event("accepted_order", order.id)

    def _target(self, order):
        """Return the destination for the current state."""
        if self.state == DELIVERING:
            return order.customer_x, order.customer_y
        return order.restaurant_x, order.restaurant_y

    def _handle_arrival(self, order):
        """Apply state transitions when the rider reaches a destination."""
        pos = (self.x, self.y)
        if self.state == PICKING_UP and pos == (order.restaurant_x, order.restaurant_y):
            self.state = DELIVERING
            order.status = "delivering"
            self._log_event("arrived_at_merchant", order.id)
        elif self.state == DELIVERING and pos == (order.customer_x, order.customer_y):
            order.complete(globalvar.current_tick())
            self._log_event("delivered_to_customer", order.id)
            next_order = self.current_order()
            if next_order is None:
                self.state = IDLE
                self.special_case = False
            else:
                self._accept(next_order)

    def move(self):
        """Advance the rider by one tick.

        Performs the idle-to-picking-up transition when a new order is
        assigned, moves toward the current destination along streets, and
        applies arrival transitions.
        """
        order = self.current_order()
        if order is None:
            self.state = IDLE
            self.special_case = False
            return
        if self.state == IDLE:
            self._accept(order)
        target_x, target_y = self._target(order)
        # Special case: rider is directly above/below or beside the target.
        self.special_case = self.x == target_x or self.y == target_y
        self.x, self.y = PathPlanner.next_position(
            self.x, self.y, target_x, target_y, self.speed)
        self._handle_arrival(order)

    def recover(self):
        """Repair an inconsistent rider after an unexpected error."""
        if self.state not in STATE_NAMES:
            LOGGER.warning("Rider %s had invalid state %s; resetting",
                           self.rider_id, self.state)
            self.state = IDLE
        if not (0 <= self.x <= MAP_SIZE and 0 <= self.y <= MAP_SIZE):
            LOGGER.warning("Rider %s was off-map; returning to spawn", self.rider_id)
            self.x, self.y = SPAWN_X, SPAWN_Y
        order = self.current_order()
        if order is None:
            self.state = IDLE
            self.special_case = False
        elif self.state == IDLE:
            self._accept(order)

    def to_dict(self):
        """Return a JSON-serialisable summary of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [order.id for order in self.orders],
        }

"""Order data model.

An order is described by ``[restaurant_x, restaurant_y, customer_x,
customer_y]`` with coordinates in the range 0-800. Coordinates are snapped to
the top-left corner of their grid cell, which is also a street intersection.
"""

import globalvar
from path_planner import CELL_SIZE, MAP_SIZE

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def snap_to_grid(value):
    """Convert a coordinate to the top-left corner of its grid cell.

    Args:
        value: Integer coordinate (values are clamped to 0-800).

    Returns:
        Integer multiple of 100 in the range 0-800.
    """
    clamped = max(0, min(int(value), MAP_SIZE))
    return (clamped // CELL_SIZE) * CELL_SIZE


class Order:
    """A food delivery order from a restaurant to a customer."""

    _id_counter = 0

    def __init__(self, res_pos, cus_pos):
        """Create an order and snap its coordinates to the grid.

        Args:
            res_pos: Restaurant position as (x, y) in 0-800.
            cus_pos: Customer position as (x, y) in 0-800.
        """
        Order._id_counter += 1
        self.id = Order._id_counter
        self.restaurant_x = snap_to_grid(res_pos[0])
        self.restaurant_y = snap_to_grid(res_pos[1])
        self.customer_x = snap_to_grid(cus_pos[0])
        self.customer_y = snap_to_grid(cus_pos[1])
        self.status = STATUS_PENDING
        self.rider_id = None
        self.start_time = globalvar.current_tick()
        self.completion_time = None
        self.distance = (abs(self.customer_x - self.restaurant_x)
                         + abs(self.customer_y - self.restaurant_y))
        self.time_taken = None

    def assign_to(self, rider_id):
        """Record the rider responsible for this order."""
        self.rider_id = rider_id
        if self.status == STATUS_PENDING:
            self.status = STATUS_ASSIGNED

    def complete(self, tick):
        """Mark the order delivered and record completion statistics.

        Args:
            tick: Simulated tick at which delivery finished.
        """
        self.status = STATUS_COMPLETED
        self.completion_time = tick
        self.time_taken = tick - self.start_time

    def efficiency(self):
        """Return delivery distance per tick, or None if not completed."""
        if not self.time_taken:
            return None
        return self.distance / self.time_taken

    def to_dict(self):
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

"""Order management: creation, assignment, recruitment and completion."""

import logging

import globalvar
from order_model import Order
from rider import Rider  # noqa: F401  (used for type reference)

LOGGER = logging.getLogger("dispatch.order")


def find_best_rider_for_order(order):
    """Choose the rider that should serve ``order``.

    Strategy: only riders with the fewest active orders are considered (the
    order always goes to the least-loaded rider), and among those the rider
    with the smallest Manhattan distance to the restaurant wins.

    Args:
        order: The Order to be assigned.

    Returns:
        The selected Rider, or None if no rider exists.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    min_load = min(rider.active_order_count() for rider in riders)
    candidates = [r for r in riders if r.active_order_count() == min_load]
    return min(
        candidates,
        key=lambda r: (abs(r.x - order.restaurant_x) + abs(r.y - order.restaurant_y),
                       r.rider_id),
    )


class OrderManager:
    """Creates, assigns and tracks orders for the dispatch system."""

    def __init__(self, system):
        """Create an order manager bound to a SystemManager.

        Args:
            system: The SystemManager that owns funds and riders.
        """
        self.system = system
        self.orders = []

    def create_order(self, restaurant_pos, customer_pos):
        """Create and register a new pending order.

        Args:
            restaurant_pos: (x, y) restaurant location.
            customer_pos: (x, y) customer location.

        Returns:
            The created Order.
        """
        order = Order(restaurant_pos, customer_pos)
        self.orders.append(order)
        return order

    def pending_orders(self):
        """Return orders that have not yet been assigned to a rider."""
        return [o for o in self.orders if o.status == "pending"]

    def completed_orders(self):
        """Return the list of delivered orders."""
        return [o for o in self.orders if o.status == "completed"]

    def _recruit_if_needed(self):
        """Recruit riders while pending orders exceed idle riders.

        Returns:
            True if at least one rider was recruited.
        """
        added = False
        while len(self.pending_orders()) > len(self.system.idle_riders()):
            if self.system.recruit_rider() is None:
                break
            added = True
        return added

    def assign(self, order):
        """Assign ``order`` to the best rider and append it to that rider.

        Args:
            order: The pending Order.

        Returns:
            The Rider that received the order, or None if none is available.
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        LOGGER.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

    def submit_order(self, restaurant_pos, customer_pos):
        """Create an order, recruit if needed, and assign it.

        Args:
            restaurant_pos: (x, y) restaurant location.
            customer_pos: (x, y) customer location.

        Returns:
            Tuple (order, rider_or_none, new_rider_added).
        """
        order = self.create_order(restaurant_pos, customer_pos)
        added = self._recruit_if_needed()
        rider = self.assign(order)
        return order, rider, added

    def assign_pending(self):
        """Try to assign every pending order; return how many were assigned."""
        assigned = 0
        for order in self.pending_orders():
            self._recruit_if_needed()
            if self.assign(order) is not None:
                assigned += 1
        return assigned

"""System-wide state management, configuration and persistence."""

import json
import logging
import os

import globalvar
from rider import DEFAULT_RIDER_SPEED, IDLE, STATE_NAMES, Rider

LOGGER = logging.getLogger("dispatch.system")

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruit_cost": 500,
    "recruit_min_funds": 300,
    "rider_speed": DEFAULT_RIDER_SPEED,
    "map_size": 800,
    "cell_size": 100,
}


def load_config(path=None):
    """Load system parameters from a JSON file, merged over defaults.

    Args:
        path: Optional path to a JSON configuration file.

    Returns:
        Dictionary of configuration values.
    """
    config = dict(DEFAULT_CONFIG)
    if not path or not os.path.exists(path):
        return config
    try:
        with open(path, encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        LOGGER.warning("Could not read config %s: %s; using defaults", path, exc)
        return config
    if not isinstance(data, dict):
        LOGGER.warning("Config %s is not a JSON object; using defaults", path)
        return config
    for key in DEFAULT_CONFIG:
        if key in data:
            try:
                config[key] = int(data[key])
            except (TypeError, ValueError):
                LOGGER.warning("Invalid value for %s in config; ignored", key)
    return config


class SystemManager:
    """Owns system funds, the rider list and the running flag."""

    def __init__(self, config=None):
        """Create the system manager with the given configuration.

        Args:
            config: Optional configuration dictionary (see DEFAULT_CONFIG).
        """
        cfg = dict(DEFAULT_CONFIG)
        if config:
            cfg.update(config)
        self.funds = int(cfg["initial_funds"])
        self.recruit_cost = int(cfg["recruit_cost"])
        self.recruit_min_funds = int(cfg["recruit_min_funds"])
        self.rider_speed = int(cfg["rider_speed"])
        self.riders = []
        self.rider_id_counter = 0
        self.running = True
        globalvar.set_value("riders", self.riders)

    # ------------------------------------------------------------------
    # Rider CRUD and recruitment
    # ------------------------------------------------------------------
    def next_rider_id(self):
        """Increment and return the rider ID counter."""
        self.rider_id_counter += 1
        return self.rider_id_counter

    def add_rider(self):
        """Create a rider at (400, 400) in the idle state and register it."""
        rider = Rider(self.next_rider_id(), speed=self.rider_speed)
        self.riders.append(rider)
        LOGGER.info("Rider %s added", rider.rider_id)
        return rider

    def recruit_rider(self):
        """Recruit a new rider if funds allow.

        Returns:
            The new Rider, or None if funds are below the recruit threshold.
        """
        if self.funds < self.recruit_min_funds:
            return None
        self.funds -= self.recruit_cost
        return self.add_rider()

    def get_rider(self, rider_id):
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def idle_riders(self):
        """Return riders that have no active order."""
        return [r for r in self.riders if r.is_idle()]

    def remove_rider(self, rider_id):
        """Remove an idle rider from the system.

        Returns:
            True if the rider was removed, False otherwise.
        """
        rider = self.get_rider(rider_id)
        if rider is None or not rider.is_idle():
            return False
        self.riders.remove(rider)
        return True

    # ------------------------------------------------------------------
    # Simulation clock and consistency
    # ------------------------------------------------------------------
    def tick(self):
        """Advance simulated time by one tick and move every rider.

        Exceptions raised by an individual rider are logged and the rider is
        repaired, so one faulty rider cannot stop the simulation.

        Returns:
            The new tick value.
        """
        new_tick = globalvar.current_tick() + 1
        globalvar.set_value("current_tick", new_tick)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                LOGGER.exception("Error moving rider %s; recovering", rider.rider_id)
                rider.recover()
        return new_tick

    def check_consistency(self):
        """Verify internal invariants and return a list of problem descriptions."""
        issues = []
        seen = set()
        for rider in self.riders:
            if rider.rider_id in seen:
                issues.append(f"Duplicate rider id {rider.rider_id}")
            seen.add(rider.rider_id)
            if rider.state not in STATE_NAMES:
                issues.append(f"Rider {rider.rider_id} has invalid state {rider.state}")
            order = rider.current_order()
            if rider.state != IDLE and order is None:
                issues.append(f"Rider {rider.rider_id} is busy without an active order")
            if order is not None and order.rider_id != rider.rider_id:
                issues.append(f"Order {order.id} is not linked to rider {rider.rider_id}")
        return issues

    def shutdown(self):
        """Stop the main loop."""
        self.running = False


def init_system(config_path=None):
    """Initialise global variables and create the SystemManager.

    Args:
        config_path: Optional JSON configuration file path.

    Returns:
        The initialised SystemManager.
    """
    globalvar._init()
    config = load_config(config_path)
    globalvar.set_value("config", config)
    globalvar.set_value("current_tick", 0)
    return SystemManager(config)


def save_state(path, system, orders):
    """Persist system parameters, riders and orders to a JSON file.

    Args:
        path: Destination file path.
        system: The SystemManager to save.
        orders: Iterable of Order objects.
    """
    data = {
        "config": globalvar.get_value("config"),
        "funds": system.funds,
        "rider_id_counter": system.rider_id_counter,
        "current_tick": globalvar.current_tick(),
        "riders": [r.to_dict() for r in system.riders],
        "orders": [o.to_dict() for o in orders],
    }
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2)

"""Status reports, order/rider tables and ASCII map rendering."""

from path_planner import CELL_SIZE, GRID_DIMENSION
from rider import STATE_NAMES
import globalvar

CANVAS_STEP = CELL_SIZE // 2  # each canvas character covers 50 units


class DisplayManager:
    """Produces text output for the command-line interface."""

    def __init__(self, system, map_dimension=GRID_DIMENSION):
        """Create a display bound to the system manager.

        Args:
            system: The SystemManager whose state is displayed.
            map_dimension: Number of grid cells per side (8).
        """
        self.system = system
        self.map_dimension = map_dimension

    def status_report(self):
        """Return the system status snapshot as text."""
        riders = self.system.riders
        lines = [
            "=== System Status ===",
            f"System funds: {self.system.funds}",
            f"Current tick: {globalvar.current_tick()}",
            f"Riders: {len(riders)} (idle: {len(self.system.idle_riders())})",
        ]
        if not riders:
            lines.append("No riders on duty.")
        for rider in riders:
            order_ids = [o.id for o in rider.orders]
            lines.append(
                f"  Rider #{rider.rider_id}: location ({rider.x}, {rider.y}), "
                f"status {rider.state_name()}, active orders "
                f"{rider.active_order_count()}, order list {order_ids or '-'}"
            )
        return "\n".join(lines)

    def orders_report(self, orders):
        """Return a table of order details.

        Args:
            orders: Iterable of Order objects.
        """
        orders = list(orders)
        if not orders:
            return "No orders."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Dist':>6} {'Taken':>6} {'Rider':>6} {'Status':>12}")
        lines = [header, "-" * len(header)]
        for order in orders:
            rider = "-" if order.rider_id is None else f"#{order.rider_id}"
            taken = "-" if order.time_taken is None else str(order.time_taken)
            lines.append(
                f"{order.id:>4} {order.start_time:>6} "
                f"{f'({order.restaurant_x},{order.restaurant_y})':>12} "
                f"{f'({order.customer_x},{order.customer_y})':>12} "
                f"{order.distance:>6} {taken:>6} {rider:>6} {order.status:>12}"
            )
        return "\n".join(lines)

    def riders_report(self):
        """Return every recorded status change for all riders."""
        riders = self.system.riders
        if not riders:
            return "No riders on duty."
        lines = []
        for rider in riders:
            lines.append(f"Rider #{rider.rider_id} (current state: {rider.state_name()})")
            for entry in rider.log:
                order_id = "-" if entry["order_id"] is None else f"#{entry['order_id']}"
                lines.append(
                    f"  t={entry['time']:>5}  {entry['event']:<22} order {order_id:<6} "
                    f"at ({entry['x']}, {entry['y']})"
                )
        return "\n".join(lines)

    def _mark(self, canvas, x, y, symbol):
        """Place ``symbol`` on the canvas at map coordinates (x, y)."""
        size = len(canvas)
        row = min(max(y // CANVAS_STEP, 0), size - 1)
        col = min(max(x // CANVAS_STEP, 0), size - 1)
        canvas[row][col] = symbol

    def _draw_segment(self, canvas, start, end, symbol):
        """Draw an axis-aligned route segment on the canvas."""
        size = len(canvas)
        ar = min(max(start[1] // CANVAS_STEP, 0), size - 1)
        ac = min(max(start[0] // CANVAS_STEP, 0), size - 1)
        br = min(max(end[1] // CANVAS_STEP, 0), size - 1)
        bc = min(max(end[0] // CANVAS_STEP, 0), size - 1)
        for row in range(min(ar, br), max(ar, br) + 1):
            for col in range(min(ac, bc), max(ac, bc) + 1):
                canvas[row][col] = symbol

    def map_text(self, orders=None, route=None):
        """Render the ASCII map.

        Symbols: 'R' rider, 'S' restaurant, 'C' customer, '#' street,
        '.' empty block, '*' planned route.

        Args:
            orders: Optional iterable of orders whose S/C points and status
                are displayed.
            route: Optional list of waypoints drawn as connected segments.

        Returns:
            Multi-line string of the map and a short order status list.
        """
        size = self.map_dimension * 2 + 1
        canvas = [
            ['#' if (r % 2 == 0 or c % 2 == 0) else '.' for c in range(size)]
            for r in range(size)
        ]
        if route:
            if len(route) == 1:
                self._mark(canvas, route[0][0], route[0][1], '*')
            for start, end in zip(route, route[1:]):
                self._draw_segment(canvas, start, end, '*')
        orders = list(orders or [])
        for order in orders:
            self._mark(canvas, order.restaurant_x, order.restaurant_y, 'S')
            self._mark(canvas, order.customer_x, order.customer_y, 'C')
        for rider in self.system.riders:
            self._mark(canvas, rider.x, rider.y, 'R')

        lines = ["".join(row) for row in canvas]
        lines.append("Legend: R rider, S restaurant, C customer, # street, "
                     ". block, * route (1 char = 50 units, origin top-left)")
        for order in orders:
            lines.append(
                f"Order #{order.id}: S({order.restaurant_x},{order.restaurant_y}) -> "
                f"C({order.customer_x},{order.customer_y}) [{order.status}]"
            )
        return "\n".join(lines)

"""Command parsing, shortcut resolution, history and input validation."""

COORD_MIN = 0
COORD_MAX = 1000


class InputError(ValueError):
    """Raised for malformed or out-of-range user input."""


class InputManager:
    """Reads and validates user commands."""

    SHORTCUTS = {
        "s": "start",
        "st": "status",
        "o": "orders",
        "r": "riders",
        "m": "map",
        "t": "tick",
        "u": "run",
        "h": "help",
        "?": "help",
        "hist": "history",
        "q": "quit",
        "e": "exit",
    }

    def __init__(self):
        """Create the input manager with an empty command history."""
        self.command_history = []

    def read_line(self, prompt="> "):
        """Read one line from standard input.

        Returns:
            The raw line, or "quit" when input is closed.
        """
        try:
            return input(prompt)
        except EOFError:
            return "quit"

    def parse_command(self, line):
        """Split a command line into a normalised name and argument list.

        Supports shortcut keys, "!!" (repeat last command) and "!n" (repeat
        the n-th command from history).

        Args:
            line: Raw input text.

        Returns:
            Tuple (command_name, args). Empty input yields ("", []).

        Raises:
            InputError: If a history reference is invalid.
        """
        stripped = line.strip()
        if not stripped:
            return "", []
        if stripped.startswith("!"):
            ref = stripped[1:]
            if not self.command_history:
                raise InputError("No command history available yet.")
            if ref == "!":
                stripped = self.command_history[-1]
            elif ref.isdigit() and 1 <= int(ref) <= len(self.command_history):
                stripped = self.command_history[int(ref) - 1]
            else:
                raise InputError("Unknown history reference. Use !! or !<number>.")
        self.command_history.append(stripped)
        parts = stripped.split()
        name = parts[0].lower()
        name = self.SHORTCUTS.get(name, name)
        return name, parts[1:]

    def parse_coordinates(self, tokens):
        """Validate four integer coordinates.

        Accepts forms such as ``100 200 300 400`` or ``[100, 200, 300, 400]``.

        Args:
            tokens: Iterable of string tokens.

        Returns:
            Tuple (restaurant_x, restaurant_y, customer_x, customer_y).

        Raises:
            InputError: On wrong count, non-integer values or out-of-range values.
        """
        text = " ".join(tokens)
        for ch in "[](),":
            text = text.replace(ch, " ")
        parts = text.split()
        if len(parts) != 4:
            raise InputError(
                "Expected 4 coordinates: restaurant_x restaurant_y "
                "customer_x customer_y.")
        values = []
        for part in parts:
            try:
                value = int(part)
            except ValueError:
                raise InputError(f"'{part}' is not an integer coordinate.")
            if not COORD_MIN <= value <= COORD_MAX:
                raise InputError(
                    f"Coordinate {value} is out of range "
                    f"{COORD_MIN}-{COORD_MAX}.")
            values.append(value)
        return tuple(values)

    def history(self):
        """Return a copy of the command history list."""
        return list(self.command_history)

"""Command-line entry point for the Food Delivery Order Dispatch simulator.

Run from the project root with:  python src/main.py [config.json]
"""

import logging
import sys

from display import DisplayManager
from input_manager import InputError, InputManager
from order import OrderManager
from path_planner import PathPlanner
from system import init_system, save_state

LOGGER = logging.getLogger("dispatch.main")

MAX_RUN_TICKS = 10000

HELP_TEXT = """Commands:
  start [rx ry cx cy]  Place an order (restaurant then customer, 0-1000)  [s]
  status               Show funds, riders and their orders               [st]
  orders               Show all order details                            [o]
  riders               Show every rider status change log                [r]
  map                  Show the ASCII map with riders and orders         [m]
  tick [n]             Advance simulated time by n ticks (default 1)     [t]
  run                  Advance time until all orders are delivered       [u]
  save [path]          Save state to a JSON file
  history              Show command history                              [hist]
  !! / !n              Repeat the last / n-th command
  help                 Show this help                                    [h]
  quit | exit          Leave the program                                 [q/e]"""


class DispatchApp:
    """Wires together the managers and runs the interactive loop."""

    def __init__(self, config_path=None):
        """Initialise all subsystems.

        Args:
            config_path: Optional JSON configuration file.
        """
        self.system = init_system(config_path)
        self.orders = OrderManager(self.system)
        self.display = DisplayManager(self.system)
        self.inputs = InputManager()
        self.planner = PathPlanner()
        self.handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "run": self.cmd_run,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self):
        """Run the interactive main loop until the system stops."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands.")
        while self.system.running:
            try:
                line = self.inputs.read_line("dispatch> ")
            except KeyboardInterrupt:
                print()
                break
            try:
                name, args = self.inputs.parse_command(line)
            except InputError as exc:
                print(f"Input error: {exc}")
                continue
            if not name:
                continue
            handler = self.handlers.get(name)
            if handler is None:
                print(f"Unknown command '{name}'. Type 'help' for commands.")
                continue
            try:
                handler(args)
            except InputError as exc:
                print(f"Input error: {exc}")
            except Exception:  # keep the simulator alive on unexpected errors
                LOGGER.exception("Unexpected error while running '%s'", name)
                print("An unexpected error occurred; checking system state.")
                for issue in self.system.check_consistency():
                    print(f"Warning: {issue}")
        print("Goodbye.")

    def advance(self, steps):
        """Advance the simulation and report newly delivered orders."""
        for _ in range(steps):
            delivered_before = {o.id for o in self.orders.completed_orders()}
            tick = self.system.tick()
            self.orders.assign_pending()
            for order in self.orders.completed_orders():
                if order.id not in delivered_before:
                    print(f"[t={tick}] Order #{order.id} delivered by rider "
                          f"#{order.rider_id} (took {order.time_taken} ticks)")
        for issue in self.system.check_consistency():
            print(f"Warning: {issue}")

    def _active_orders(self):
        """Return orders that are not yet completed."""
        return [o for o in self.orders.orders if o.status != "completed"]

    def cmd_start(self, args):
        """Place a new order and show the assigned rider's route."""
        if not args:
            args = self.inputs.read_line(
                "Enter restaurant_x restaurant_y customer_x customer_y: ").split()
        rx, ry, cx, cy = self.inputs.parse_coordinates(args)
        order, rider, added = self.orders.submit_order((rx, ry), (cx, cy))
        print(f"Restaurant grid point: ({order.restaurant_x}, {order.restaurant_y})")
        print(f"Customer grid point: ({order.customer_x}, {order.customer_y})")
        if added:
            print(f"New rider added: yes (rider #{self.system.riders[-1].rider_id}, "
                  f"funds now {self.system.funds})")
        else:
            print("New rider added: no")
        if rider is None:
            print(f"No rider available (funds {self.system.funds}); "
                  f"order #{order.id} is pending.")
            return
        print(f"Order #{order.id} assigned to rider #{rider.rider_id}.")
        legs = [(order.restaurant_x, order.restaurant_y),
                (order.customer_x, order.customer_y)]
        route = self.planner.plan_multi_stop((rider.x, rider.y), legs)
        total = self.planner.route_length(route)
        ticks = self.planner.estimate_ticks(total, rider.speed)
        print(f"Route distance: {total} units | Speed: {rider.speed} units/tick | "
              f"Estimated time: {ticks} ticks (excludes queued orders)")
        print(self.display.map_text(orders=[order], route=route))

    def cmd_status(self, args):
        """Print the system status snapshot."""
        print(self.display.status_report())

    def cmd_orders(self, args):
        """Print the table of all orders."""
        print(self.display.orders_report(self.orders.orders))

    def cmd_riders(self, args):
        """Print the rider status change history."""
        print(self.display.riders_report())

    def cmd_map(self, args):
        """Print the ASCII map with riders and active orders."""
        print(self.display.map_text(orders=self._active_orders()))

    def cmd_tick(self, args):
        """Advance the simulation by one or more ticks."""
        steps = 1
        if args:
            try:
                steps = int(args[0])
            except ValueError:
                raise InputError("Tick count must be an integer.")
        if not 1 <= steps <= MAX_RUN_TICKS:
            raise InputError(f"Tick count must be between 1 and {MAX_RUN_TICKS}.")
        self.advance(steps)

    def cmd_run(self, args):
        """Advance until every order is delivered (bounded by MAX_RUN_TICKS)."""
        if not self._active_orders():
            print("No active orders to run.")
            return
        steps = 0
        while self._active_orders() and steps < MAX_RUN_TICKS:
            self.advance(1)
            steps += 1
        if self._active_orders():
            print(f"Stopped after {steps} ticks; some orders are still pending "
                  f"(funds {self.system.funds}).")
        else:
            print(f"All orders delivered after {steps} ticks.")

    def cmd_save(self, args):
        """Save the state to a JSON file."""
        path = args[0] if args else "dispatch_state.json"
        try:
            save_state(path, self.system, self.orders.orders)
        except OSError as exc:
            print(f"Could not save state: {exc}")
            return
        print(f"State saved to {path}.")

    def cmd_history(self, args):
        """Print the numbered command history."""
        history = self.inputs.history()
        if not history:
            print("No command history.")
            return
        for index, entry in enumerate(history, start=1):
            print(f"{index:>4}  {entry}")

    def cmd_help(self, args):
        """Print the command help text."""
        print(HELP_TEXT)

    def cmd_quit(self, args):
        """Stop the main loop."""
        self.system.shutdown()


def main(argv=None):
    """Program entry point.

    Args:
        argv: Optional argument list; the first item may be a config path.

    Returns:
        Process exit code.
    """
    args = sys.argv[1:] if argv is None else list(argv)
    logging.basicConfig(level=logging.WARNING,
                        format="%(asctime)s %(levelname)s %(name)s: %(message)s")
    config_path = args[0] if args else "config.json"
    app = DispatchApp(config_path)
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

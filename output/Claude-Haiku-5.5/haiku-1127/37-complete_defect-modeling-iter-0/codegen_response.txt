from __future__ import annotations

import os

from lottery_system.default_data import DEFAULT_PARTICIPANTS
from lottery_system.models import User


class ListManager:
    """Loads and validates participant lists."""

    ENCODINGS = ("utf-8-sig", "gbk")

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Return the built-in default team building list."""
        users = [User(name, department) for name, department in DEFAULT_PARTICIPANTS]
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        errors: list[str] = []
        users: list[User] = []

        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]

        try:
            lines = self._read_lines(path)
        except OSError as exc:
            return [], [f"File Error: unable to read file: {exc}"]
        except UnicodeDecodeError:
            return [], ["File Error: unable to decode file (use UTF-8 or GBK)"]

        for line_no, raw in enumerate(lines, start=1):
            text = raw.strip()
            if not text:
                continue
            parts = text.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Line {line_no}: Format Error - expected 'Name,Department Group', got '{text}'"
                )
                continue
            name, department = parts[0].strip(), parts[1].strip()
            if not name or not department:
                errors.append(
                    f"Line {line_no}: Format Error - name and department group must not be empty"
                )
                continue
            users.append(User(name, department))

        return users, errors

    def _read_lines(self, path: str) -> list[str]:
        with open(path, "rb") as handle:
            raw = handle.read()
        last_error: UnicodeDecodeError | None = None
        for encoding in self.ENCODINGS:
            try:
                return raw.decode(encoding).splitlines()
            except UnicodeDecodeError as exc:
                last_error = exc
        assert last_error is not None
        raise last_error

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into valid users (first occurrence kept) and duplicates."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department group names mapped to their member counts."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return counts

"""Prize library management."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def _find(self, name: str) -> Prize | None:
        return next((p for p in self._prizes if p.name == name), None)

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize '{name}' already exists; skipped.")
                continue
            self._prizes.append(Prize(name, quantity, list(avoid_groups), drawer))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Add a custom prize at the end of the lottery sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Quantity must be a positive integer.")
        if self._find(clean_name) is not None:
            raise ValueError(f"Prize '{clean_name}' already exists.")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        groups = list(dict.fromkeys(groups))
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        prize = Prize(clean_name, quantity, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> list[Prize]:
        """Return prizes in lottery sequence order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. `order` contains 1-based current positions in the new order."""
        count = len(self._prizes)
        if len(order) != count:
            return False
        if set(order) != set(range(1, count + 1)):
            return False
        self._prizes = [self._prizes[position - 1] for position in order]
        return True

"""Winner tracking, grouping and CSV export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime

from lottery_system.models import Prize, User


@dataclass
class Winner:
    """A winning record linking a participant to a prize."""

    user: User
    prize: Prize
    order: int
    win_time: datetime = field(default_factory=datetime.now)

    @property
    def prize_name(self) -> str:
        return self.prize.name


class ResultManager:
    """Maintains the winner list and produces result views and exports."""

    EXPORT_PREFIX = "awards_result_"

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._won_names: set[str] = set()

    def has_won(self, user: User) -> bool:
        return user.name in self._won_names

    def add_winner(self, user: User, prize: Prize) -> Winner:
        """Record a winner. A person can never win twice."""
        if self.has_won(user):
            raise ValueError(f"{user.name} has already won a prize.")
        winner = Winner(user=user, prize=prize, order=len(self._winners) + 1)
        self._winners.append(winner)
        self._won_names.add(user.name)
        return winner

    def get_winners(self) -> list[Winner]:
        """Return all winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, following the given prize order."""
        result: dict[str, list[Winner]] = {prize.name: [] for prize in prizes}
        for winner in self._winners:
            result.setdefault(winner.prize_name, []).append(winner)
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the complete winner list to awards_result_YYYYMMDD.txt in CSV format."""
        if not self._winners:
            return False, "No winners to export; nothing was exported."

        by_prize = self.get_winners_by_prize(prizes)
        file_name = f"{self.EXPORT_PREFIX}{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", encoding="utf-8", newline="") as handle:
                writer = csv.writer(handle)
                writer.writerow(["Name", "Department", "Prize"])
                for winners in by_prize.values():
                    for winner in winners:
                        writer.writerow([winner.user.name, winner.user.department, winner.prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and draw execution."""

from __future__ import annotations

import random
from typing import Iterable

from lottery_system.constants import CHAIN_DRAWERS, CHENG_RU_GROUP, LI_BIN, MA_JIAQI
from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager, Winner


class LotteryManager:
    """Applies lottery rules and performs draws."""

    def __init__(self, users: Iterable[User], rng: random.Random | None = None) -> None:
        self.users: list[User] = list(users)
        self.result_manager = ResultManager()
        # Results (True = winner from Cheng Ru group) of the most recent draws
        # made by Chen Qixian / Jin Duo.
        self.last_two_draw_results: list[bool] = []
        self._rng = rng if rng is not None else random.Random()

    def get_available_users(self) -> list[User]:
        """Participants who have not won any prize yet (anti-duplication)."""
        return [user for user in self.users if not self.result_manager.has_won(user)]

    def is_fallback_active(self) -> bool:
        """True when the last two chain draws produced no Cheng Ru group winner."""
        return len(self.last_two_draw_results) == 2 and not any(self.last_two_draw_results)

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        pool = self.get_available_users()
        drawer = prize.associated_drawer

        # Fallback (Cheng Ru group protection) overrides everything else for Ma Jiaqi.
        if drawer == MA_JIAQI and self.is_fallback_active():
            return [user for user in pool if user.department == CHENG_RU_GROUP]

        # Special rule for Li Bin: only previous winners are excluded.
        if drawer == LI_BIN:
            return pool

        avoid = set(prize.avoid_groups)
        return [user for user in pool if user.department not in avoid]

    def get_eligible_pool(self, prize: Prize) -> list[User]:
        """Public accessor for the eligible pool of a prize."""
        return self._get_eligible_pool(prize)

    def describe_rules(self, prize: Prize) -> str:
        """Human readable description of the rules applied to a prize round."""
        parts: list[str] = []
        drawer = prize.associated_drawer
        if drawer:
            parts.append(f"Associated lottery person: {drawer}")
        else:
            parts.append("No associated lottery person")

        if drawer == LI_BIN:
            parts.append("Special rule: only previous winners are excluded (avoid groups not applied)")
        elif prize.avoid_groups:
            parts.append("Avoided groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoided groups")

        if drawer == MA_JIAQI:
            if self.is_fallback_active():
                parts.append(
                    f"Fallback active: only {CHENG_RU_GROUP} members are eligible"
                )
            else:
                parts.append("Fallback inactive: Cheng Ru group protection not triggered")

        parts.append("Previous winners are always excluded")
        return "; ".join(parts)

    def _record_draw_result(self, hit_cheng_ru: bool) -> None:
        self.last_two_draw_results.append(hit_cheng_ru)
        self.last_two_draw_results = self.last_two_draw_results[-2:]

    def draw_one(self, prize: Prize) -> Winner | None:
        """Draw a single winner for the prize. Returns None if no eligible person remains."""
        pool = self._get_eligible_pool(prize)
        winner: Winner | None = None
        if pool:
            user = self._rng.choice(pool)
            winner = self.result_manager.add_winner(user, prize)

        if prize.associated_drawer in CHAIN_DRAWERS:
            hit = winner is not None and winner.user.department == CHENG_RU_GROUP
            self._record_draw_result(hit)
        return winner

from __future__ import annotations

from typing import Callable

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager


def ordinal(n: int) -> str:
    """Return the English ordinal form of n, e.g. 1st, 2nd, 11th."""
    if 11 <= n % 100 <= 13:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def parse_int_list(text: str) -> list[int] | None:
    """Parse integers separated by spaces or commas. Returns None on invalid input."""
    tokens = text.replace(",", " ").split()
    try:
        return [int(token) for token in tokens]
    except ValueError:
        return None


class LotteryApp:
    """Interactive command-line application controller."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._output = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.lottery_manager: LotteryManager | None = None

    # ------------------------------------------------------------------ helpers
    def _say(self, message: str = "") -> None:
        self._output(message)

    def _ask(self, prompt: str) -> str:
        return self._input(prompt)

    def _confirm(self, prompt: str) -> bool:
        return self._ask(f"{prompt} (y/n): ").strip().lower() in ("y", "yes")

    def _group_names(self) -> list[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    # --------------------------------------------------------------- main loop
    def run(self) -> int:
        self._say("=== Company Team Building Smart Lottery System ===")
        while True:
            self._print_main_menu()
            choice = self._ask("Select an option: ").strip()
            if choice == "1":
                self.load_participants()
            elif choice == "2":
                self.show_groups()
            elif choice == "3":
                self.prize_menu()
            elif choice == "4":
                self.start_lottery()
            elif choice == "5":
                self.show_results()
            elif choice == "0":
                self._say("Goodbye!")
                return 0
            else:
                self._say("Invalid option, please try again.")

    def _print_main_menu(self) -> None:
        self._say("\n--- Main Menu ---")
        self._say("  1. Load participant list")
        self._say("  2. View department groups")
        self._say("  3. Prize configuration")
        self._say("  4. Start lottery")
        self._say("  5. View / export results")
        self._say("  0. Exit")

    # ------------------------------------------------------- participant list
    def load_participants(self) -> None:
        self._say("\nSelect list source:")
        self._say("  1. Built-in default team building list")
        self._say("  2. Custom txt file (each line: Name,Department Group)")
        self._say("  0. Cancel")
        choice = self._ask("Choice: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = self._ask("Enter txt file path: ").strip().strip('"')
            users, errors = self.list_manager.load_custom_list(path)
        else:
            return

        valid = self._review_import(users, errors)
        if valid is None:
            self._say("Upload terminated. Participant list unchanged.")
            return
        self.users = valid
        self._say(f"Participant list loaded: {len(valid)} participants.")

    def _review_import(self, users: list[User], errors: list[str]) -> list[User] | None:
        valid, duplicates = self.list_manager.validate_users(users)
        if not errors and not duplicates:
            if not valid:
                self._say("No participants found in the list.")
                return None
            return valid

        self._say("\nIssues found in the list:")
        for error in errors:
            self._say(f"  - {error}")
        for dup in duplicates:
            self._say(f"  - Duplicate name: {dup.name} ({dup.department}) - later entry ignored")
        self._say(f"Valid participants: {len(valid)}")
        self._say("Options:")
        self._say("  1. Remove all invalid and duplicate entries and continue")
        self._say("  2. Manually remove specific participants by name, then continue")
        self._say("  3. Terminate upload")
        choice = self._ask("Choice: ").strip()

        if choice == "1":
            result = valid
        elif choice == "2":
            text = self._ask("Enter names to remove (comma separated): ")
            to_remove = {name.strip() for name in text.split(",") if name.strip()}
            result = [user for user in valid if user.name not in to_remove]
        else:
            return None

        if not result:
            self._say("No valid participants remain.")
            return None
        return result

    def show_groups(self) -> None:
        if not self.users:
            self._say("No participant list loaded. Please load a list first.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._say("Current valid department groups:")
        for index, (department, count) in enumerate(info.items(), start=1):
            self._say(f"  {index}. {department}: {count} people")
        self._say(f"Total: {len(self.users)} people")

    # ---------------------------------------------------------- prize config
    def prize_menu(self) -> None:
        while True:
            self._say("\n--- Prize Configuration ---")
            self._print_prizes()
            self._say("  1. Load preset prizes")
            self._say("  2. Add custom prize")
            self._say("  3. Change lottery order")
            self._say("  0. Back")
            choice = self._ask("Choice: ").strip()
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                if errors:
                    for error in errors:
                        self._say(f"  - {error}")
                else:
                    self._say("Preset prizes loaded.")
            elif choice == "2":
                self.add_prize()
            elif choice == "3":
                self.reorder_prizes()
            elif choice == "0":
                return
            else:
                self._say("Invalid option, please try again.")

    def _print_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("  (no prizes configured)")
            return
        for index, prize in enumerate(prizes, start=1):
            avoid = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
            drawer = prize.associated_drawer or "none"
            self._say(
                f"  {index}. {prize.name} | quantity: {prize.quantity} | "
                f"avoid groups: {avoid} | lottery person: {drawer}"
            )

    def _select_avoid_groups(self) -> list[str] | None:
        groups = self._group_names()
        if not groups:
            self._say("Load the participant list first to select avoid groups; no avoid groups set.")
            return []
        self._say("Department groups:")
        for index, group in enumerate(groups, start=1):
            self._say(f"  {index}. {group}")
        text = self._ask("Select groups to avoid by number (comma separated, blank for none): ").strip()
        if not text:
            return []
        numbers = parse_int_list(text)
        if numbers is None or any(n < 1 or n > len(groups) for n in numbers):
            self._say("Invalid group selection.")
            return None
        return list(dict.fromkeys(groups[n - 1] for n in numbers))

    def add_prize(self) -> None:
        name = self._ask("Prize name: ").strip()
        if not name:
            self._say("Prize name cannot be empty.")
            return
        quantity_text = self._ask("Quantity: ").strip()
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._say("Quantity must be a positive integer.")
            return
        avoid_groups = self._select_avoid_groups()
        if avoid_groups is None:
            return
        drawer = self._ask("Associated lottery person (blank for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, quantity, avoid_groups, drawer)
        except ValueError as exc:
            self._say(f"Failed to add prize: {exc}")
            return
        self._say("Prize added.")

    def reorder_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            self._say("At least two prizes are needed to change the order.")
            return
        text = self._ask(
            f"Enter new order as serial numbers 1-{len(prizes)} (e.g. 2 1 3): "
        )
        order = parse_int_list(text)
        if order is None or not self.prize_manager.reorder_prizes(order):
            self._say("Invalid order. Each serial number from 1 to N must appear exactly once.")
            return
        self._say("Lottery order updated.")

    # ----------------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.users:
            self._say("Please load the participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("Please configure prizes first.")
            return
        if self.lottery_manager and self.lottery_manager.result_manager.get_winners():
            if not self._confirm("A previous lottery has results. Starting a new lottery clears them. Continue"):
                return

        self.lottery_manager = LotteryManager(self.users)
        self._say("\n=== Lottery started ===")
        self._say("Commands: [Enter] draw next winner | pause | terminate (end current round)")
        total = len(prizes)
        for index, prize in enumerate(prizes, start=1):
            self._run_round(index, total, prize)
        self._say("\n=== Lottery finished ===")
        self.show_results()

    def _run_round(self, index: int, total: int, prize) -> None:
        manager = self.lottery_manager
        assert manager is not None
        self._say(f"\n--- Round {index}/{total}: {prize.name} ---")
        self._say(f"Quantity: {prize.quantity}")
        self._say(f"Rules: {manager.describe_rules(prize)}")

        round_winners = []
        while len(round_winners) < prize.quantity:
            position = len(round_winners) + 1
            command = self._ask(
                f"[{ordinal(position)} winner] Press Enter to draw, or type pause/terminate: "
            ).strip().lower()

            if command == "pause":
                if not self._pause_round():
                    self._say("Round terminated by operator.")
                    break
                continue
            if command == "terminate":
                self._say("Round terminated by operator.")
                break
            if command:
                self._say(f"Unknown command '{command}'.")
                continue

            self._say(f"Currently drawing the {ordinal(position)} winner...")
            winner = manager.draw_one(prize)
            if winner is None:
                self._say("No eligible participants remain for this prize. Round ends.")
                break
            round_winners.append(winner)
            self._say(
                f"Congratulations {winner.user.name} ({winner.user.department}) "
                f"for winning {prize.name}!"
            )

        self._say(f"Round {index} results (in winning order):")
        if not round_winners:
            self._say("  (no winners)")
        for position, winner in enumerate(round_winners, start=1):
            self._say(f"  {ordinal(position)}: {winner.user.name} ({winner.user.department})")

    def _pause_round(self) -> bool:
        """Pause until 'continue' (returns True) or 'terminate' (returns False)."""
        self._say("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
        while True:
            command = self._ask("(paused) > ").strip().lower()
            if command == "continue":
                self._say("Lottery resumed.")
                return True
            if command == "terminate":
                return False
            self._say("Only 'continue' or 'terminate' is accepted while paused.")

    # ----------------------------------------------------------------- results
    def show_results(self) -> None:
        if not self.lottery_manager or not self.lottery_manager.result_manager.get_winners():
            self._say("No results yet.")
            return
        result_manager = self.lottery_manager.result_manager
        prizes = self.prize_manager.get_prizes()
        by_prize = result_manager.get_winners_by_prize(prizes)

        self._say("\n=== Winner List ===")
        for prize_name, winners in by_prize.items():
            self._say(f"{prize_name}:")
            if not winners:
                self._say("  (no winners)")
            for winner in winners:
                self._say(f"  - {winner.user.name} ({winner.user.department})")

        if self._confirm("Export results to a CSV-formatted txt file"):
            _, message = result_manager.export_to_csv(prizes)
            self._say(message)


def main(argv: list[str] | None = None) -> int:
    """Program entry point."""
    app = LotteryApp()
    try:
        return app.run()
    except (EOFError, KeyboardInterrupt):
        print("\nExiting.")
        return 0

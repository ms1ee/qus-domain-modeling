"""Global variable storage shared by all simulator modules.

The store is a plain dictionary keyed by variable name. ``_init`` resets
every variable to its default value and is invoked on import so that
``get_value`` is always usable.
"""

from typing import Any, Dict

DEFAULT_CONFIG: Dict[str, int] = {
    "initial_funds": 1000,
    "recruit_cost": 300,
    "recruit_fund_threshold": 300,
    "max_run_ticks": 2000,
}

_store: Dict[str, Any] = {}


def _init() -> None:
    """Reset the global store to its default values.

    Defaults:
        funds: 1000
        riders: empty list
        rider_id_counter: 0
        is_running: True
        orders / pending_orders: empty lists
        tick: 0
        config: copy of DEFAULT_CONFIG
    """
    _store.clear()
    _store.update(
        {
            "funds": DEFAULT_CONFIG["initial_funds"],
            "riders": [],
            "rider_id_counter": 0,
            "is_running": True,
            "orders": [],
            "pending_orders": [],
            "tick": 0,
            "config": dict(DEFAULT_CONFIG),
        }
    )


def set_value(key: str, value: Any) -> None:
    """Store ``value`` under ``key``.

    Args:
        key: Name of the global variable (e.g. ``"riders"``).
        value: Value to store.
    """
    _store[key] = value


def get_value(key: str, default: Any = None) -> Any:
    """Return the global variable stored under ``key``.

    Args:
        key: Name of the global variable.
        default: Value returned when the key does not exist.

    Returns:
        The stored value, or ``default`` if the key is missing.
    """
    return _store.get(key, default)

"""Grid-based path planning for the 800x800 map.

The map is divided into an 8x8 grid of 100x100 cells. Grid lines (where x
or y is a multiple of 100) are streets. Routes are built from axis-aligned
segments, preferring corners that keep each segment on a street.
"""

from dataclasses import dataclass, field
from typing import List, Sequence, Tuple

Point = Tuple[int, int]

MAP_SIZE = 800
CELL_SIZE = 100


@dataclass
class Route:
    """A connected sequence of waypoints.

    Attributes:
        waypoints: Ordered list of points, starting at the origin.
        total_distance: Total travelled distance (Manhattan length of the
            axis-aligned segments).
    """

    waypoints: List[Point] = field(default_factory=list)
    total_distance: float = 0.0


class PathPlanner:
    """Computes axis-aligned, street-prioritising paths on the grid."""

    cell_size: int = CELL_SIZE
    map_size: int = MAP_SIZE

    @staticmethod
    def is_on_street(coord_x: int, coord_y: int) -> bool:
        """Return True if the point lies on a street (grid line)."""
        return coord_x % CELL_SIZE == 0 or coord_y % CELL_SIZE == 0

    @classmethod
    def plan_path(cls, start: Point, goal: Point) -> List[Point]:
        """Plan an axis-aligned path from ``start`` to ``goal``.

        The returned list excludes ``start`` and ends with ``goal``.
        Two L-shaped candidates are considered; the one whose segments
        lie on more streets is chosen. Points on the same row or column
        are connected by a single straight segment (special case).

        Args:
            start: Current position (x, y).
            goal: Destination position (x, y).

        Returns:
            Waypoints to visit in order. Empty if already at the goal.
        """
        if start == goal:
            return []
        sx, sy = start
        gx, gy = goal
        if sx == gx or sy == gy:
            return [goal]

        corner_h = (gx, sy)  # horizontal segment first
        corner_v = (sx, gy)  # vertical segment first
        score_h = int(sy % CELL_SIZE == 0) + int(gx % CELL_SIZE == 0)
        score_v = int(sx % CELL_SIZE == 0) + int(gy % CELL_SIZE == 0)
        corner = corner_h if score_h >= score_v else corner_v
        return [corner, goal]

    @classmethod
    def build_route(cls, origin: Point, stops: Sequence[Point]) -> Route:
        """Build a continuous route visiting ``stops`` in order.

        Args:
            origin: Starting position of the rider.
            stops: Successive destinations (e.g. restaurant, customer).

        Returns:
            A Route whose waypoints begin with ``origin``.
        """
        waypoints: List[Point] = [tuple(origin)]
        total = 0.0
        current = tuple(origin)
        for stop in stops:
            for point in cls.plan_path(current, tuple(stop)):
                total += abs(point[0] - current[0]) + abs(point[1] - current[1])
                waypoints.append(point)
                current = point
        return Route(waypoints=waypoints, total_distance=float(total))

"""Order data model: coordinate conversion, lifecycle and bookkeeping."""

import itertools
import math
from typing import Any, Dict, Optional, Sequence

import globalvar

MIN_COORD = 0
MAX_COORD = 800
CELL_SIZE = 100
GRID_DIMENSION = 8

_order_ids = itertools.count(1)


def to_cell_center(value: int) -> int:
    """Convert a raw coordinate to the centre of its grid cell.

    Implements ``center = (value // 100) * 100 + 50``. The value 800 lies
    on the outer boundary and is mapped into the last cell.

    Args:
        value: Integer coordinate in the range 0-800.

    Returns:
        Centre coordinate of the containing cell.
    """
    cell = min(int(value) // CELL_SIZE, GRID_DIMENSION - 1)
    return cell * CELL_SIZE + CELL_SIZE // 2


def _validate_point(point: Sequence[int]) -> tuple:
    """Validate an (x, y) pair and return it as a tuple of ints.

    Raises:
        ValueError: If the point is malformed or out of range.
    """
    if len(point) != 2:
        raise ValueError(f"point must have two coordinates, got {point!r}")
    x, y = point
    for value in (x, y):
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError(f"coordinate {value!r} is not an integer")
        if not MIN_COORD <= value <= MAX_COORD:
            raise ValueError(
                f"coordinate {value} out of range {MIN_COORD}-{MAX_COORD}"
            )
    return int(x), int(y)


class Order:
    """A food delivery order from a restaurant to a customer.

    Attributes:
        id: Unique order identifier.
        restaurant_x, restaurant_y: Grid-centre coordinates of the restaurant.
        customer_x, customer_y: Grid-centre coordinates of the customer.
        restaurant_pos, customer_pos: Tuples of the centre coordinates.
        status: "pending", "assigned" or "completed".
        rider_id: Identifier of the assigned rider, or None.
        start_time: Tick at which the order was created.
        completion_time: Tick at which the order was delivered, or None.
        distance: Euclidean distance between restaurant and customer.
        time_taken: Ticks from creation to completion, or None.
        route: Planned Route for the assigned rider, or None.
    """

    def __init__(
        self,
        res_pos: Sequence[int],
        cus_pos: Sequence[int],
        start_time: Optional[int] = None,
    ) -> None:
        """Create an order from raw user coordinates.

        Args:
            res_pos: Restaurant position (x, y), range 0-800.
            cus_pos: Customer position (x, y), range 0-800.
            start_time: Creation tick; defaults to the current tick.

        Raises:
            ValueError: If coordinates are invalid.
        """
        res_x, res_y = _validate_point(res_pos)
        cus_x, cus_y = _validate_point(cus_pos)

        self.id: int = next(_order_ids)
        self.restaurant_x = to_cell_center(res_x)
        self.restaurant_y = to_cell_center(res_y)
        self.customer_x = to_cell_center(cus_x)
        self.customer_y = to_cell_center(cus_y)
        self.restaurant_pos = (self.restaurant_x, self.restaurant_y)
        self.customer_pos = (self.customer_x, self.customer_y)

        self.status: str = "pending"
        self.rider_id: Optional[int] = None
        if start_time is None:
            start_time = int(globalvar.get_value("tick", 0) or 0)
        self.start_time: int = int(start_time)
        self.completion_time: Optional[int] = None
        self.distance: float = math.hypot(
            self.customer_x - self.restaurant_x,
            self.customer_y - self.restaurant_y,
        )
        self.time_taken: Optional[int] = None
        self.route = None

    def assign_to(self, rider_id: int) -> None:
        """Mark the order as assigned to the rider with ``rider_id``."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def complete(self, completion_time: int) -> None:
        """Mark the order as delivered and record efficiency data.

        Args:
            completion_time: Tick at which delivery finished.
        """
        self.completion_time = int(completion_time)
        self.time_taken = self.completion_time - self.start_time
        self.status = "completed"

    @property
    def efficiency(self) -> float:
        """Distance covered per tick of total order time (0 if unknown)."""
        if not self.time_taken:
            return 0.0
        return self.distance / self.time_taken

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable representation of the order."""
        return {
            "id": self.id,
            "restaurant": [self.restaurant_x, self.restaurant_y],
            "customer": [self.customer_x, self.customer_y],
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": round(self.distance, 2),
            "time_taken": self.time_taken,
        }

    def __repr__(self) -> str:
        return (
            f"Order(id={self.id}, restaurant={self.restaurant_pos}, "
            f"customer={self.customer_pos}, status={self.status})"
        )

"""Order assignment: selects the best rider for an order."""

import logging
import math
from typing import Optional

import globalvar
from rider import Rider

logger = logging.getLogger(__name__)


def euclidean_distance(x1: float, y1: float, x2: float, y2: float) -> float:
    """Return the Euclidean distance between two points."""
    return math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)


def find_best_rider_for_order(order) -> Optional[Rider]:
    """Select the rider that should receive ``order``.

    Strategy: closest rider first, by Euclidean distance from the rider's
    current position to the restaurant. Idle riders (state 0 with no
    orders) are preferred; if none exist, all riders are considered so
    that busy riders share the load. Ties are broken by fewer assigned
    orders and then by rider id.

    Args:
        order: The Order to be assigned.

    Returns:
        The chosen Rider, or None if no rider exists.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        logger.info("No riders available for order %s", order.id)
        return None

    rest_x, rest_y = order.restaurant_pos
    idle = [r for r in riders if r.state == 0 and not r.orders]
    candidates = idle if idle else list(riders)

    best = min(
        candidates,
        key=lambda r: (
            euclidean_distance(r.x, r.y, rest_x, rest_y),
            len(r.orders),
            r.rider_id,
        ),
    )
    logger.debug("Order %s assigned candidate rider %s", order.id, best.rider_id)
    return best

<file path="src/rider.py">
"""Rider model: movement, state machine and order lifecycle.

State machine:
    0 (idle)        -> 1 (picking up)  when an order is present in ``orders``
    1 (picking up)  -> 2 (delivering)  on arrival at the restaurant
    2 (delivering)  -> 0 (idle)        on delivery; the order is removed
"""

import logging
from typing import Any, Dict, List, Optional, Tuple

from globalvar import get_value
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

INITIAL_X = 435
INITIAL_Y = 385
INITIAL_SPEED = 10

Point = Tuple[int, int]


class Rider:
    """A delivery rider that moves on the grid and serves orders.

    Attributes:
        rider_id: Unique identifier.
        x, y: Current coordinates.
        speed: Distance covered per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Assigned orders; ``orders[0]`` is the one being served.
        special_case: True while the rider is aligned with its target on
            one axis (directly above/below or beside it).
        path: Remaining waypoints of the current leg.
        log: Chronological list of event dictionaries. Each entry has the
            keys "event" and "order_id" plus "time", "x", "y", "state".
    """

    def __init__(self, rider_id: int) -> None:
        """Create an idle rider at the default depot position.

        Args:
            rider_id: Unique identifier assigned by the system.
        """
        self.rider_id: int = rider_id
        self.x: int = INITIAL_X
        self.y: int = INITIAL_Y
        self.speed: int = INITIAL_SPEED
        self.state: int = STATE_IDLE
        self.orders: List[Any] = []
        self.special_case: bool = False
        self.path: List[Point] = []
        self.log: List[Dict[str, Any]] = []
        self._record("spawn", None)

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    @property
    def position(self) -> Point:
        """Current (x, y) position."""
        return (self.x, self.y)

    def current_order(self) -> Optional[Any]:
        """Return the order currently being served, or None."""
        return self.orders[0] if self.orders else None

    def current_target(self) -> Optional[Point]:
        """Return the active destination for the current state."""
        order = self.current_order()
        if order is None:
            return None
        if self.state == STATE_DELIVERING:
            return order.customer_pos
        return order.restaurant_pos

    def _now(self) -> int:
        return int(get_value("tick", 0) or 0)

    def _record(self, event: str, order_id: Optional[int]) -> None:
        """Append an event to the rider's log."""
        self.log.append(
            {
                "event": event,
                "order_id": order_id,
                "time": self._now(),
                "x": self.x,
                "y": self.y,
                "state": self.state,
                "rider_id": self.rider_id,
            }
        )

    def _is_aligned(self, target: Point) -> bool:
        """True if the rider shares a row or column with the target."""
        return self.position != target and (
            self.x == target[0] or self.y == target[1]
        )

    def _advance(self, budget: int) -> None:
        """Consume up to ``budget`` distance along the planned path.

        Segments are axis-aligned, so each tick moves the rider along a
        single street direction (horizontal or vertical).
        """
        while budget > 0 and self.path:
            wx, wy = self.path[0]
            dx = wx - self.x
            dy = wy - self.y
            dist = abs(dx) + abs(dy)
            if dist <= budget:
                self.x, self.y = wx, wy
                self.path.pop(0)
                budget -= dist
            else:
                if dx != 0:
                    self.x += budget if dx > 0 else -budget
                else:
                    self.y += budget if dy > 0 else -budget
                budget = 0

    def _arrive(self, order: Any) -> None:
        """Handle arrival at the current destination and transition state."""
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            self.path = []
            self._record("arrived_at_merchant", order.id)
            logger.info("Rider %s arrived at restaurant for order %s",
                        self.rider_id, order.id)
        elif self.state == STATE_DELIVERING:
            order.complete(self._now())
            self.orders.remove(order)
            self.state = STATE_IDLE
            self.path = []
            self.special_case = False
            self._record("delivered_to_customer", order.id)
            logger.info("Rider %s delivered order %s", self.rider_id, order.id)

    # ------------------------------------------------------------------
    # Public behaviour
    # ------------------------------------------------------------------
    def move(self) -> None:
        """Advance the rider by one tick.

        Performs any state transition triggered by the current conditions
        (accepting a new order, arriving at a destination) and moves the
        rider up to ``speed`` units along the street grid.
        """
        order = self.current_order()

        if self.state == STATE_IDLE:
            if order is None:
                return
            self.state = STATE_PICKING_UP
            self.path = []
            self._record("accepted_order", order.id)

        if order is None:
            logger.warning("Rider %s had an active state without orders; "
                           "resetting to idle", self.rider_id)
            self.state = STATE_IDLE
            self.path = []
            self.special_case = False
            return

        target = self.current_target()
        if target is None:
            return

        if not self.path and self.position != target:
            self.path = PathPlanner.plan_path(self.position, target)

        self.special_case = self._is_aligned(target)
        self._advance(self.speed)

        if self.position == target and not self.path:
            self._arrive(order)

    def to_dict(self) -> Dict[str, Any]:
        """Return a JSON-serialisable snapshot of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [order.id for order in self.orders],
            "log": list(self.log),
        }

    def __repr__(self) -> str:
        return (f"Rider(id={self.rider_id}, pos=({self.x}, {self.y}), "
                f"state={self.state}, orders={len(self.orders)})")

"""System management: funds, rider recruitment, simulated time and dispatch."""

import json
import logging
import os
from typing import Any, Dict, List, Optional, Tuple

import globalvar
from globalvar import DEFAULT_CONFIG, get_value, set_value
from order import find_best_rider_for_order
from order_model import Order
from path_planner import PathPlanner
from rider import STATE_IDLE, Rider

logger = logging.getLogger(__name__)


def load_config(path: str) -> Dict[str, Any]:
    """Load system parameters from a JSON configuration file.

    Recognised keys: initial_funds, recruit_cost, recruit_fund_threshold,
    max_run_ticks. Each must be a non-negative integer.

    Args:
        path: Path to the JSON file containing a single object.

    Returns:
        A copy of the active configuration after applying the file.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the content is malformed.
    """
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError("configuration file must contain a JSON object")

    config = get_value("config")
    for key in DEFAULT_CONFIG:
        if key not in data:
            continue
        value = data[key]
        if isinstance(value, bool) or not isinstance(value, int) or value < 0:
            raise ValueError(f"invalid value for '{key}': {value!r}")
        config[key] = value
    if "initial_funds" in data:
        set_value("funds", config["initial_funds"])
    logger.info("Loaded configuration from %s", path)
    return dict(config)


def init_system(config_path: Optional[str] = None) -> "SystemManager":
    """Initialise global state and return the system manager.

    Args:
        config_path: Optional JSON configuration file. Missing or invalid
            files fall back to the built-in defaults.

    Returns:
        A ready-to-use SystemManager.
    """
    globalvar._init()
    if config_path and os.path.exists(config_path):
        try:
            load_config(config_path)
        except (OSError, ValueError) as exc:
            logger.error("Failed to load config %s (%s); using defaults",
                         config_path, exc)
            globalvar._init()
    return SystemManager()


class SystemManager:
    """Facade over the global simulation state.

    Responsible for funds, the rider list, rider recruitment, the tick
    loop and dispatching pending orders.
    """

    # ------------------------------------------------------------------
    # State accessors
    # ------------------------------------------------------------------
    def _cfg(self, key: str) -> int:
        return int(get_value("config")[key])

    @property
    def funds(self) -> int:
        """System funds (may be negative)."""
        return int(get_value("funds", 0))

    @funds.setter
    def funds(self, value: int) -> None:
        set_value("funds", int(value))

    @property
    def riders(self) -> List[Rider]:
        """The dynamic list of riders."""
        return get_value("riders")

    @property
    def rider_id_counter(self) -> int:
        """Last rider id issued."""
        return int(get_value("rider_id_counter", 0))

    @property
    def is_running(self) -> bool:
        """Main loop control flag."""
        return bool(get_value("is_running", True))

    @is_running.setter
    def is_running(self, value: bool) -> None:
        set_value("is_running", bool(value))

    @property
    def tick_count(self) -> int:
        """Current simulated time in ticks."""
        return int(get_value("tick", 0))

    @property
    def orders(self) -> List[Order]:
        """All orders ever created in this session."""
        return get_value("orders")

    @property
    def pending_orders(self) -> List[Order]:
        """Orders waiting for a rider."""
        return get_value("pending_orders")

    # ------------------------------------------------------------------
    # Riders
    # ------------------------------------------------------------------
    def idle_rider_count(self) -> int:
        """Number of riders that are idle and have no assigned orders."""
        return sum(1 for r in self.riders
                   if r.state == STATE_IDLE and not r.orders)

    def recruit_rider(self) -> Rider:
        """Create a new rider and deduct the recruitment cost.

        Returns:
            The newly created Rider.
        """
        counter = self.rider_id_counter + 1
        set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        self.funds = self.funds - self._cfg("recruit_cost")
        logger.info("Recruited rider %s (funds now %s)", counter, self.funds)
        return rider

    def try_recruit(self, pending_count: int) -> bool:
        """Recruit riders while funds allow and pending orders exceed idle riders.

        Args:
            pending_count: Number of orders waiting for assignment.

        Returns:
            True if at least one rider was added.
        """
        added = False
        idle = self.idle_rider_count()
        while (self.funds >= self._cfg("recruit_fund_threshold")
               and pending_count > idle):
            self.recruit_rider()
            idle += 1
            added = True
        return added

    def rider_for_order(self, order: Order) -> Optional[Rider]:
        """Return the rider currently responsible for ``order``, if any."""
        for rider in self.riders:
            if rider.rider_id == order.rider_id:
                return rider
        return None

    # ------------------------------------------------------------------
    # Orders
    # ------------------------------------------------------------------
    def submit_order(
        self, res_pos: Tuple[int, int], cus_pos: Tuple[int, int]
    ) -> Tuple[Order, Optional[Rider], bool]:
        """Create an order, recruit if needed, and dispatch it.

        Args:
            res_pos: Restaurant coordinates (x, y), 0-800.
            cus_pos: Customer coordinates (x, y), 0-800.

        Returns:
            Tuple (order, assigned_rider_or_None, new_rider_added).
        """
        order = Order(res_pos, cus_pos, start_time=self.tick_count)
        self.orders.append(order)
        self.pending_orders.append(order)
        added = self.try_recruit(len(self.pending_orders))
        self.dispatch_pending()
        return order, self.rider_for_order(order), added

    def dispatch_pending(self) -> int:
        """Assign every pending order to its best rider.

        Returns:
            Number of orders assigned during this call.
        """
        assigned = 0
        for order in list(self.pending_orders):
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            rider.orders.append(order)
            order.assign_to(rider.rider_id)
            order.route = PathPlanner.build_route(
                rider.position, [order.restaurant_pos, order.customer_pos]
            )
            self.pending_orders.remove(order)
            assigned += 1
            logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return assigned

    # ------------------------------------------------------------------
    # Time
    # ------------------------------------------------------------------
    def tick(self) -> None:
        """Advance simulated time by one unit and move every rider.

        Exceptions raised by an individual rider are logged and the rider
        is reset to a consistent state, so the simulation keeps running.
        """
        set_value("tick", self.tick_count + 1)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - defensive recovery
                logger.exception("Rider %s failed during move; recovering",
                                 rider.rider_id)
                rider.path = []
                if not rider.orders:
                    rider.state = STATE_IDLE
        self.dispatch_pending()

    def run_until_order_complete(self, order: Order,
                                 max_ticks: Optional[int] = None) -> int:
        """Run ticks until ``order`` is delivered.

        Args:
            order: The order to follow.
            max_ticks: Safety bound on the number of ticks.

        Returns:
            Number of ticks elapsed.

        Raises:
            RuntimeError: If the order is not completed within ``max_ticks``.
        """
        limit = max_ticks if max_ticks is not None else self._cfg("max_run_ticks")
        start = self.tick_count
        while order.status != "completed":
            if self.tick_count - start >= limit:
                raise RuntimeError(
                    f"order {order.id} not completed within {limit} ticks")
            self.tick()
        return self.tick_count - start

    # ------------------------------------------------------------------
    # Consistency and persistence
    # ------------------------------------------------------------------
    def check_consistency(self) -> List[str]:
        """Return a list of detected state inconsistencies (empty if none)."""
        issues: List[str] = []
        ids = [r.rider_id for r in self.riders]
        if len(set(ids)) != len(ids):
            issues.append("duplicate rider ids detected")
        for rider in self.riders:
            if rider.state != STATE_IDLE and not rider.orders:
                issues.append(f"rider {rider.rider_id} is busy without orders")
            for order in rider.orders:
                if order.rider_id != rider.rider_id:
                    issues.append(
                        f"order {order.id} mismatched with rider {rider.rider_id}")
                if order.status == "completed":
                    issues.append(
                        f"completed order {order.id} still held by rider "
                        f"{rider.rider_id}")
            if not (0 <= rider.x <= 800 and 0 <= rider.y <= 800):
                issues.append(f"rider {rider.rider_id} outside map")
        return issues

    def save_state(self, path: str) -> None:
        """Persist parameters, riders and order history as JSON.

        Args:
            path: Destination file path.

        Raises:
            OSError: If the file cannot be written.
        """
        data = {
            "funds": self.funds,
            "tick": self.tick_count,
            "rider_id_counter": self.rider_id_counter,
            "config": dict(get_value("config")),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.orders],
        }
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        logger.info("Saved simulation state to %s", path)

"""Status display and ASCII map visualisation.

The map is rendered on a 17x17 character canvas: one character per 50
units, so that street grid lines (multiples of 100) and cell centres
(50, 150, ...) are both represented. Symbols: R rider, S restaurant,
C customer, '#' street, '.' empty cell interior, '*' planned route.
"""

import math
from typing import Dict, Iterable, List, Optional, Sequence, Tuple

from path_planner import Route

Point = Tuple[int, int]

STATE_NAMES: Dict[int, str] = {0: "idle", 1: "picking up", 2: "delivering"}
RESOLUTION = 50
MAP_SIZE = 800
CANVAS_SIZE = MAP_SIZE // RESOLUTION + 1
SAMPLE_STEP = 10


def _sign(value: int) -> int:
    return (value > 0) - (value < 0)


class DisplayManager:
    """Produces text output for the command-line interface."""

    def __init__(self) -> None:
        """Create the display manager with its symbol mapping."""
        self.map_symbols: Dict[str, str] = {
            "rider": "R",
            "restaurant": "S",
            "customer": "C",
            "empty": ".",
            "street": "#",
            "route": "*",
        }

    # ------------------------------------------------------------------
    # Map
    # ------------------------------------------------------------------
    @staticmethod
    def _index(coord: int) -> int:
        return min(max(int(coord) // RESOLUTION, 0), CANVAS_SIZE - 1)

    def _blank_canvas(self) -> List[List[str]]:
        canvas: List[List[str]] = []
        for row in range(CANVAS_SIZE):
            line = []
            for col in range(CANVAS_SIZE):
                on_street = row % 2 == 0 or col % 2 == 0
                line.append(self.map_symbols["street"] if on_street
                            else self.map_symbols["empty"])
            canvas.append(line)
        return canvas

    def _place(self, canvas: List[List[str]], x: int, y: int, symbol: str) -> None:
        canvas[self._index(y)][self._index(x)] = symbol

    def _draw_route(self, canvas: List[List[str]], waypoints: Sequence[Point]) -> None:
        """Draw connected axis-aligned segments between waypoints."""
        symbol = self.map_symbols["route"]
        for start, end in zip(waypoints, waypoints[1:]):
            dx = end[0] - start[0]
            dy = end[1] - start[1]
            length = abs(dx) + abs(dy)
            for step in range(0, length + 1, SAMPLE_STEP):
                t = min(step, length)
                x = start[0] + _sign(dx) * min(t, abs(dx))
                y = start[1] + _sign(dy) * min(t, abs(dy))
                self._place(canvas, x, y, symbol)
            self._place(canvas, end[0], end[1], symbol)
        if len(waypoints) == 1:
            self._place(canvas, waypoints[0][0], waypoints[0][1], symbol)

    def render_map(self, riders: Iterable, orders: Iterable = (),
                   route: Optional[Route] = None) -> str:
        """Render the ASCII map with riders, orders and an optional route.

        Args:
            riders: Riders to mark with 'R'.
            orders: Orders whose restaurant ('S') and customer ('C') are
                shown; completed orders are skipped.
            route: Optional Route drawn with '*'.

        Returns:
            Multi-line string containing the map and a legend.
        """
        canvas = self._blank_canvas()
        if route is not None and route.waypoints:
            self._draw_route(canvas, route.waypoints)
        for order in orders:
            if getattr(order, "status", "") == "completed":
                continue
            self._place(canvas, order.restaurant_x, order.restaurant_y,
                        self.map_symbols["restaurant"])
            self._place(canvas, order.customer_x, order.customer_y,
                        self.map_symbols["customer"])
        for rider in riders:
            self._place(canvas, rider.x, rider.y, self.map_symbols["rider"])

        lines = ["Map (y increases upward, x increases to the right):"]
        for row in reversed(range(CANVAS_SIZE)):
            label = f"{row * RESOLUTION:>4} |"
            lines.append(label + " " + " ".join(canvas[row]))
        lines.append("     " + " ".join("^" if c % 2 == 0 else " "
                                        for c in range(CANVAS_SIZE)))
        lines.append("Legend: R=rider S=restaurant C=customer "
                     "#=street .=empty *=route")
        return "\n".join(lines)

    # ------------------------------------------------------------------
    # Text reports
    # ------------------------------------------------------------------
    def render_status(self, funds: int, riders: Sequence,
                      issues: Sequence[str] = ()) -> str:
        """Render funds, rider count and per-rider details."""
        lines = [f"System funds: {funds}", f"Riders: {len(riders)}"]
        for rider in riders:
            lines.append(
                f"  Rider {rider.rider_id}: position=({rider.x}, {rider.y}) "
                f"status={STATE_NAMES.get(rider.state, rider.state)} "
                f"orders={len(rider.orders)} "
                f"order_ids={[o.id for o in rider.orders]}"
            )
        if issues:
            lines.append("Consistency warnings:")
            lines.extend(f"  - {issue}" for issue in issues)
        return "\n".join(lines)

    def render_orders(self, orders: Sequence) -> str:
        """Render a table of all orders."""
        if not orders:
            return "No orders yet."
        header = (f"{'ID':>4} {'Start':>6} {'Restaurant':>12} {'Customer':>12} "
                  f"{'Dist':>8} {'Taken':>6} {'Rider':>6} {'Status':>10}")
        lines = [header, "-" * len(header)]
        for o in orders:
            taken = "-" if o.time_taken is None else str(o.time_taken)
            rider = "-" if o.rider_id is None else str(o.rider_id)
            lines.append(
                f"{o.id:>4} {o.start_time:>6} "
                f"{str(o.restaurant_pos):>12} {str(o.customer_pos):>12} "
                f"{o.distance:>8.1f} {taken:>6} {rider:>6} {o.status:>10}"
            )
        return "\n".join(lines)

    def render_rider_logs(self, riders: Sequence) -> str:
        """Render every recorded status change for every rider."""
        if not riders:
            return "No riders."
        lines: List[str] = []
        for rider in riders:
            lines.append(f"Rider {rider.rider_id} log:")
            if not rider.log:
                lines.append("  (no events)")
            for entry in rider.log:
                order = "-" if entry["order_id"] is None else entry["order_id"]
                lines.append(
                    f"  t={entry['time']:>5} event={entry['event']:<22} "
                    f"order={order} pos=({entry['x']}, {entry['y']}) "
                    f"state={entry['state']}"
                )
        return "\n".join(lines)

    def render_route(self, route: Route, speed: int) -> str:
        """Render the waypoint list and estimated travel time."""
        lines = [f"Planned route (rider speed {speed} units/tick):"]
        for index, (x, y) in enumerate(route.waypoints):
            lines.append(f"  {index}: ({x}, {y})")
        ticks = math.ceil(route.total_distance / speed) if speed > 0 else 0
        lines.append(f"Total distance: {route.total_distance:.0f} units, "
                     f"estimated time: {ticks} ticks")
        return "\n".join(lines)

    def render_help(self) -> str:
        """Return the command reference."""
        return "\n".join([
            "Commands:",
            "  start [rx ry cx cy]  Place an order (restaurant then customer, 0-800)",
            "  status (s)           Funds, riders and their states",
            "  orders (o)           All orders with timing and rider",
            "  riders (r)           Event log of every rider",
            "  map (m)              ASCII map of riders and orders",
            "  tick [n] (t)         Advance simulated time by n ticks",
            "  save [path] (w)      Save state to JSON",
            "  history (hist)       Show command history",
            "  !! / !n              Repeat last / n-th command",
            "  help (h, ?)          Show this help",
            "  quit (q) / exit (x)  Leave the program",
        ])

"""Command parsing, validation and history for the command-line interface."""

import logging
import re
from typing import Callable, List, Sequence, Tuple

logger = logging.getLogger(__name__)

COORD_MIN = 0
COORD_MAX = 800

COMMAND_ALIASES = {
    "s": "status",
    "o": "orders",
    "r": "riders",
    "m": "map",
    "t": "tick",
    "h": "help",
    "?": "help",
    "q": "quit",
    "x": "exit",
    "w": "save",
    "hist": "history",
}

KNOWN_COMMANDS = (
    "start", "status", "orders", "riders", "map", "tick",
    "save", "history", "help", "quit", "exit",
)


class InputError(ValueError):
    """Raised when user input cannot be parsed or validated."""


class InputManager:
    """Reads, expands, parses and validates user commands.

    Attributes:
        command_history: Every accepted command line, in order.
    """

    def __init__(self, reader: Callable[[str], str] = input) -> None:
        """Create an input manager.

        Args:
            reader: Callable used to read one line (defaults to ``input``).
        """
        self._reader = reader
        self.command_history: List[str] = []

    def read_line(self, prompt: str = "delivery> ") -> str:
        """Read a stripped line of user input."""
        return self._reader(prompt).strip()

    def expand_shortcut(self, line: str) -> str:
        """Expand history shortcuts '!!' and '!n'.

        Raises:
            InputError: If the referenced history entry does not exist.
        """
        if line == "!!":
            if not self.command_history:
                raise InputError("no previous command in history")
            return self.command_history[-1]
        match = re.fullmatch(r"!(\d+)", line)
        if match:
            index = int(match.group(1))
            if not 1 <= index <= len(self.command_history):
                raise InputError(f"history entry {index} does not exist")
            return self.command_history[index - 1]
        return line

    def record(self, line: str) -> None:
        """Append a command line to the history."""
        self.command_history.append(line)

    def parse_command(self, line: str) -> Tuple[str, List[str]]:
        """Split a line into a canonical command name and its arguments.

        Raises:
            InputError: If the line is empty or the command is unknown.
        """
        tokens = line.split()
        if not tokens:
            raise InputError("empty command")
        name = tokens[0].lower()
        name = COMMAND_ALIASES.get(name, name)
        if name not in KNOWN_COMMANDS:
            raise InputError(
                f"unknown command '{tokens[0]}'. Type 'help' for a list.")
        return name, tokens[1:]

    @staticmethod
    def validate_coordinates(values: Sequence[int]) -> List[int]:
        """Check that every value lies within the map range 0-800.

        Raises:
            InputError: If any value is out of range.
        """
        for value in values:
            if not COORD_MIN <= value <= COORD_MAX:
                raise InputError(
                    f"coordinate {value} out of range {COORD_MIN}-{COORD_MAX}")
        return list(values)

    @staticmethod
    def parse_coordinates(tokens: Sequence[str]) -> List[int]:
        """Parse four integer coordinates separated by spaces or commas.

        Returns:
            [restaurant_x, restaurant_y, customer_x, customer_y]

        Raises:
            InputError: On wrong count, non-integer or out-of-range values.
        """
        normalized = " ".join(tokens).replace(",", " ").split()
        if len(normalized) != 4:
            raise InputError(
                "expected four integers: restaurant_x restaurant_y "
                "customer_x customer_y")
        values: List[int] = []
        for token in normalized:
            try:
                values.append(int(token))
            except ValueError:
                raise InputError(f"'{token}' is not an integer") from None
        return InputManager.validate_coordinates(values)

    @staticmethod
    def parse_positive_int(tokens: Sequence[str], default: int,
                           maximum: int) -> int:
        """Parse an optional positive integer argument within a bound.

        Raises:
            InputError: If the argument is not an integer in 1..maximum.
        """
        if not tokens:
            return default
        try:
            value = int(tokens[0])
        except ValueError:
            raise InputError(f"'{tokens[0]}' is not an integer") from None
        if not 1 <= value <= maximum:
            raise InputError(f"value must be between 1 and {maximum}")
        return value

"""Command-line entry point for the food delivery dispatch simulator.

Run with:  python src/main.py [config.json]
"""

import logging
import os
import sys

SRC_DIR = os.path.dirname(os.path.abspath(__file__))
if SRC_DIR not in sys.path:
    sys.path.insert(0, SRC_DIR)

from display import DisplayManager  # noqa: E402
from input_manager import InputError, InputManager  # noqa: E402
from system import SystemManager, init_system  # noqa: E402

logger = logging.getLogger("dispatch")

BANNER = (
    "Food Delivery Order Dispatch System\n"
    "Type 'help' for commands, 'quit' to exit."
)
MAX_MANUAL_TICKS = 10000
DEFAULT_STATE_FILE = "simulation_state.json"


class Application:
    """Wires the system, display and input managers into a CLI loop."""

    def __init__(self, system: SystemManager, display: DisplayManager,
                 inputs: InputManager) -> None:
        """Create the application.

        Args:
            system: Core simulation facade.
            display: Text and map renderer.
            inputs: Command reader, parser and history.
        """
        self.system = system
        self.display = display
        self.inputs = inputs
        self._handlers = {
            "start": self.cmd_start,
            "status": self.cmd_status,
            "orders": self.cmd_orders,
            "riders": self.cmd_riders,
            "map": self.cmd_map,
            "tick": self.cmd_tick,
            "save": self.cmd_save,
            "history": self.cmd_history,
            "help": self.cmd_help,
            "quit": self.cmd_quit,
            "exit": self.cmd_quit,
        }

    def run(self) -> None:
        """Run the interactive loop until the system stops."""
        print(BANNER)
        while self.system.is_running:
            try:
                line = self.inputs.read_line()
            except (EOFError, KeyboardInterrupt):
                print()
                self.system.is_running = False
                break
            if not line:
                continue
            try:
                line = self.inputs.expand_shortcut(line)
                name, args = self.inputs.parse_command(line)
                self.inputs.record(line)
                self._handlers[name](args)
            except InputError as exc:
                print(f"Input error: {exc}")
            except RuntimeError as exc:
                print(f"Simulation error: {exc}")
            except Exception:  # keep the session alive on unexpected faults
                logger.exception("Unexpected failure while handling command")
                print("An unexpected error occurred; system state preserved.")
        print("Goodbye.")

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------
    def cmd_start(self, args) -> None:
        """Place an order and simulate its delivery."""
        if args:
            coords = InputManager.parse_coordinates(args)
        else:
            line = self.inputs.read_line(
                "Enter restaurant_x restaurant_y customer_x customer_y: ")
            coords = InputManager.parse_coordinates(line.split())

        rx, ry, cx, cy = coords
        order, rider, added = self.system.submit_order((rx, ry), (cx, cy))

        print(f"Restaurant grid position: {order.restaurant_pos}")
        print(f"Customer grid position:   {order.customer_pos}")
        print(f"New rider added: {'yes' if added else 'no'}")
        if rider is None:
            print("No rider is available and funds are insufficient; "
                  "the order is pending.")
            return

        print(f"Assigned rider: {rider.rider_id} (speed {rider.speed} units/tick)")
        print(self.display.render_route(order.route, rider.speed))
        print(self.display.render_map(self.system.riders, [order], order.route))

        ticks = self.system.run_until_order_complete(order)
        print(f"Order {order.id} delivered in {ticks} ticks "
              f"(distance {order.distance:.1f}).")

    def cmd_status(self, _args) -> None:
        """Show funds, riders and consistency warnings."""
        print(self.display.render_status(
            self.system.funds, self.system.riders,
            self.system.check_consistency()))

    def cmd_orders(self, _args) -> None:
        """Show the order table."""
        print(self.display.render_orders(self.system.orders))

    def cmd_riders(self, _args) -> None:
        """Show the event log of every rider."""
        print(self.display.render_rider_logs(self.system.riders))

    def cmd_map(self, _args) -> None:
        """Show the ASCII map."""
        print(self.display.render_map(self.system.riders, self.system.orders))

    def cmd_tick(self, args) -> None:
        """Advance the simulation by one or more ticks."""
        count = InputManager.parse_positive_int(args, 1, MAX_MANUAL_TICKS)
        for _ in range(count):
            self.system.tick()
        print(f"Advanced {count} tick(s); current time {self.system.tick_count}.")

    def cmd_save(self, args) -> None:
        """Persist the simulation state to a JSON file."""
        path = args[0] if args else DEFAULT_STATE_FILE
        self.system.save_state(path)
        print(f"State saved to {path}.")

    def cmd_history(self, _args) -> None:
        """Print the command history."""
        if not self.inputs.command_history:
            print("History is empty.")
            return
        for index, entry in enumerate(self.inputs.command_history, start=1):
            print(f"  {index:>3}  {entry}")

    def cmd_help(self, _args) -> None:
        """Print the command reference."""
        print(self.display.render_help())

    def cmd_quit(self, _args) -> None:
        """Stop the main loop."""
        self.system.is_running = False


def main(argv=None) -> int:
    """Program entry point.

    Args:
        argv: Optional argument list; the first element may be a config path.

    Returns:
        Process exit code.
    """
    argv = sys.argv[1:] if argv is None else list(argv)
    level_name = os.environ.get("DISPATCH_LOG_LEVEL", "WARNING").upper()
    level = getattr(logging, level_name, logging.WARNING)
    if not isinstance(level, int):
        level = logging.WARNING
    logging.basicConfig(
        level=level,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )

    config_path = argv[0] if argv else os.path.join(os.getcwd(), "config.json")
    system = init_system(config_path)
    app = Application(system, DisplayManager(), InputManager())
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

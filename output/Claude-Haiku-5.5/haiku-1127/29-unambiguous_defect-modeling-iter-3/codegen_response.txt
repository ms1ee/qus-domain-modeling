"""Grid geometry for the 800x800 food delivery dispatch map.

The map is an 8x8 grid of 100x100 unit cells. Streets run along the grid
lines, meaning a point is on a road when its x or y coordinate is a multiple
of 100. Points where both coordinates are multiples of 100 are intersections,
and they are the only valid restaurant and customer locations.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE  # 8 cells per side

INTERSECTIONS = tuple(
    (x, y)
    for x in range(0, MAP_SIZE + 1, CELL_SIZE)
    for y in range(0, MAP_SIZE + 1, CELL_SIZE)
)


def clamp(value, low=0, high=MAP_SIZE):
    """Clamp ``value`` into the inclusive range [low, high].

    Args:
        value: The number to clamp.
        low: Lower bound (inclusive).
        high: Upper bound (inclusive).

    Returns:
        int: The clamped value.
    """
    return max(low, min(high, int(value)))


def snap_to_grid(value):
    """Snap a coordinate to the nearest multiple of ``CELL_SIZE``.

    The result is clamped to the map bounds so it always lies on a valid
    grid line between 0 and ``MAP_SIZE``.

    Args:
        value: Raw coordinate value.

    Returns:
        int: Snapped coordinate.
    """
    value = int(value)
    snapped = ((value + CELL_SIZE // 2) // CELL_SIZE) * CELL_SIZE
    return clamp(snapped)


def is_in_bounds(x, y):
    """Return True if the point lies inside the map bounds.

    Args:
        x: X coordinate.
        y: Y coordinate.

    Returns:
        bool: Whether the point is within 0..MAP_SIZE on both axes.
    """
    return 0 <= x <= MAP_SIZE and 0 <= y <= MAP_SIZE


def is_road(x, y):
    """Return True if the point lies on a street (a grid line).

    Args:
        x: X coordinate.
        y: Y coordinate.

    Returns:
        bool: True when the point is in bounds and on a grid line.
    """
    return is_in_bounds(x, y) and (x % CELL_SIZE == 0 or y % CELL_SIZE == 0)


def is_intersection(x, y):
    """Return True if the point is a grid intersection (restaurant/customer spot).

    Args:
        x: X coordinate.
        y: Y coordinate.

    Returns:
        bool: True when both coordinates are multiples of ``CELL_SIZE``.
    """
    return is_in_bounds(x, y) and x % CELL_SIZE == 0 and y % CELL_SIZE == 0


def manhattan(a, b):
    """Return the Manhattan distance between two points.

    Args:
        a: Tuple (x, y).
        b: Tuple (x, y).

    Returns:
        int: |ax - bx| + |ay - by|.
    """
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

import copy
import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 5000,
    "recruitment_cost": 800,
    "max_riders": 10,

def load_json(path):
    """Read and parse a JSON file.

    Args:
        path: File path to read.

    Returns:
        The parsed JSON object.

    Raises:
        OSError: If the file cannot be read.
        ValueError: If the content is not valid JSON.
    """
    with open(path, "r", encoding="utf-8") as handle:
        return json.load(handle)


def save_json(path, data):
    """Write ``data`` to ``path`` as formatted JSON.

    Args:
        path: Destination file path.
        data: JSON-serialisable object.

    Raises:
        OSError: If the file cannot be written.
    """
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, indent=2, ensure_ascii=False)


def load_config(path=None):
    """Load system parameters, falling back to defaults for anything invalid.

    Args:
        path: Optional path to a JSON configuration file.

    Returns:
        dict: Complete configuration dictionary.
    """
    config = copy.deepcopy(DEFAULT_CONFIG)
    if not path:
        return config
    if not os.path.exists(path):
        logger.warning("Config file %s not found; using defaults.", path)
        return config
    try:
        data = load_json(path)
    except (OSError, ValueError) as exc:
        logger.error("Could not read config %s: %s; using defaults.", path, exc)
        return config
    if not isinstance(data, dict):
        logger.error("Config %s must contain a JSON object; using defaults.", path)
        return config

    for key in DEFAULT_CONFIG:
        if key not in data:
            continue
        value = data[key]
        if key == "speed_options":
            if (isinstance(value, list) and value
                    and all(isinstance(v, int) and not isinstance(v, bool) and v > 0
                            for v in value)):
                config[key] = value
            else:
                logger.warning("Invalid speed_options in config; keeping default.")
        elif key == "state_file":
            if isinstance(value, str) and value:
                config[key] = value
            else:
                logger.warning("Invalid state_file in config; keeping default.")
        else:
            if isinstance(value, int) and not isinstance(value, bool) and value >= 0:
                config[key] = value
            else:
                logger.warning("Invalid value for %s in config; keeping default.", key)
    return config

"""Grid-based path planning restricted to the street network."""

from collections import deque

from grid import is_road

_STEPS = ((1, 0), (-1, 0), (0, 1), (0, -1))


class PathPlanner:
    """Computes unit-step routes along streets between two points.

    Routes only pass through road points (x or y on a 100-unit grid line),
    so movement always follows streets and turns at right angles. When the
    start and goal are directly above or below each other (or left or right),
    a straight-line fast path is tried first.
    """

    def __init__(self, prefer_streets=True):
        """Create a planner.

        Args:
            prefer_streets: When True, the straight fast path is tried before
                the general search. Routes always stay on streets either way.
        """
        self.prefer_streets = prefer_streets

    def plan(self, start, goal):
        """Compute a street route from ``start`` to ``goal``.

        Args:
            start: Tuple (x, y) of the origin, which must be on a road.
            goal: Tuple (x, y) of the destination, which must be on a road.

        Returns:
            list[tuple[int, int]]: Unit-step points from ``start`` (exclusive)
            to ``goal`` (inclusive). Empty if the two points are equal.

        Raises:
            ValueError: If either point is not on a road.
            RuntimeError: If no route exists.
        """
        start = (int(start[0]), int(start[1]))
        goal = (int(goal[0]), int(goal[1]))
        if start == goal:
            return []
        if not is_road(*start) or not is_road(*goal):
            raise ValueError("Route endpoints must lie on a street: %r -> %r" % (start, goal))

        aligned = start[0] == goal[0] or start[1] == goal[1]
        if self.prefer_streets and aligned:
            straight = self._straight(start, goal)
            if straight is not None:
                return straight
        return self._search(start, goal)

    @staticmethod
    def _straight(start, goal):
        """Return a straight unit-step path if every point on it is a road.

        Args:
            start: Origin point.
            goal: Destination point, aligned with ``start`` on one axis.

        Returns:
            list of points, or None if the straight line leaves the streets.
        """
        sx, sy = start
        gx, gy = goal
        if sx == gx:
            step = 1 if gy > sy else -1
            points = [(sx, y) for y in range(sy + step, gy + step, step)]
        else:
            step = 1 if gx > sx else -1
            points = [(x, sy) for x in range(sx + step, gx + step, step)]
        if all(is_road(x, y) for x, y in points):
            return points
        return None

    @staticmethod
    def _search(start, goal):
        """Breadth-first search over street points.

        Args:
            start: Origin point.
            goal: Destination point.

        Returns:
            list of points from start (exclusive) to goal (inclusive).

        Raises:
            RuntimeError: If the goal cannot be reached.
        """
        parent = {start: None}
        queue = deque([start])
        found = False
        while queue:
            current = queue.popleft()
            if current == goal:
                found = True
                break
            for dx, dy in _STEPS:
                nxt = (current[0] + dx, current[1] + dy)
                if nxt in parent or not is_road(*nxt):
                    continue
                parent[nxt] = current
                queue.append(nxt)
        if not found:
            raise RuntimeError("No street route from %r to %r" % (start, goal))

        path = []
        node = goal
        while node != start:
            path.append(node)
            node = parent[node]
        path.reverse()
        return path

    @staticmethod
    def path_length(path):
        """Return the number of unit steps in a planned path.

        Args:
            path: List of points as returned by ``plan``.

        Returns:
            int: Number of steps.
        """
        return len(path)

"""Rider model: state machine, street movement and event logging."""

import logging

import globalvar
from grid import INTERSECTIONS
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {
    IDLE: "idle",
    PICKING_UP: "picking up",
    DELIVERING: "delivering",
}

DEFAULT_SPEEDS = (10, 15, 20)
MAX_LOG_ENTRIES = 5000
_SPAWN_STRIDE = 13  # coprime with the 81 intersections, spreads riders out

_PLANNER = PathPlanner()


def _current_tick():
    """Return the current simulated tick from the global store."""
    tick = globalvar.get_value("current_tick", 0)
    return int(tick) if tick is not None else 0


def _spawn_point(rider_id):
    """Return a deterministic starting intersection (always on a road)."""
    return INTERSECTIONS[(rider_id * _SPAWN_STRIDE) % len(INTERSECTIONS)]


class Rider:
    """A delivery rider that moves along streets and serves assigned orders.

    Orders are assigned by appending them to ``orders``. The first order in
    the list is the active one. ``move()`` advances the rider by one tick and
    performs all state transitions and their log entries.

    Attributes:
        rider_id (int): Unique identifier.
        x (int), y (int): Current position on the street grid.
        speed (int): Units travelled per tick.
        state (int): 0 idle, 1 picking up, 2 delivering.
        orders (list[Order]): Assigned orders, active order first.
        special_case (bool): True while directly above or below the target.
        path (list[tuple]): Remaining unit-step points of the current leg.
        log (list[dict]): Status change entries with keys "event" and
            "order_id" (plus time, rider_id, position, state, description).
    """

    def __init__(self, rider_id):
        """Create a rider at a deterministic street intersection.

        Args:
            rider_id: Unique positive integer identifier.
        """
        self.rider_id = int(rider_id)
        self.x, self.y = _spawn_point(self.rider_id)
        speeds = tuple(globalvar.get_value("speed_options") or DEFAULT_SPEEDS)
        self.speed = int(speeds[(self.rider_id - 1) % len(speeds)])
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.path = []
        self.log = []
        self._record("spawn", None, "rider recruited")

    @property
    def position(self):
        """Return the current position as a tuple (x, y)."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Return the human-readable name of the current state."""
        return STATE_NAMES[self.state]

    def _record(self, event, order_id, description=""):
        """Append a status change entry to the rider log.

        Args:
            event: Event name, e.g. "accepted_order".
            order_id: Related order id, or None.
            description: Optional free text.

        Returns:
            dict: The created log entry.
        """
        entry = {
            "time": _current_tick(),
            "event": event,
            "order_id": order_id,
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "description": description,
        }
        self.log.append(entry)
        if len(self.log) > MAX_LOG_ENTRIES:
            del self.log[:len(self.log) - MAX_LOG_ENTRIES]
        logger.debug("rider %s event=%s order=%s", self.rider_id, event, order_id)
        return entry

    def move(self):
        """Advance the rider by one tick.

        Handles idle-to-picking-up when an order is present, moves along the
        street route toward the restaurant or customer, and performs the
        arrival transitions (picking up to delivering, delivering to idle).
        """
        tick = _current_tick()
        if not self.orders:
            if self.state != IDLE:
                logger.warning("rider %s had no orders while state=%s; resetting",
                               self.rider_id, self.state)
                self.state = IDLE
                self.path = []
            return

        if self.state == IDLE:
            self.state = PICKING_UP
            self.path = []
            order = self.orders[0]
            self._record("accepted_order", order.id,
                         "accepted order, heading to restaurant %r" % (order.restaurant_pos,))

        order = self.orders[0]
        target = order.restaurant_pos if self.state == PICKING_UP else order.customer_pos
        if self.position != target:
            self._step_towards(order, target)
        if self.position == target:
            self._arrive(order, tick)

    def _step_towards(self, order, target):
        """Move up to ``speed`` unit steps along the street route to ``target``.

        Args:
            order: The active order whose distance counter is updated.
            target: Destination tuple (x, y).
        """
        if not self.path:
            self.path = _PLANNER.plan(self.position, target)
            # Directly above or below the target: straight-line special case.
            self.special_case = self.x == target[0] and self.y != target[1]
        if not self.path:
            return
        steps = min(self.speed, len(self.path))
        travelled = self.path[:steps]
        del self.path[:steps]
        self.x, self.y = travelled[-1]
        order.distance += steps

    def _arrive(self, order, tick):
        """Apply the state transition for reaching the current target.

        Args:
            order: The active order.
            tick: Current simulated tick.
        """
        self.path = []
        self.special_case = False
        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.mark_picked_up(tick)
            self._record("arrived_at_merchant", order.id,
                         "picked up at restaurant %r" % (order.restaurant_pos,))
        elif self.state == DELIVERING:
            self.orders.remove(order)
            order.complete(tick)
            self.state = IDLE
            self._record("delivered_to_customer", order.id,
                         "delivered to customer %r" % (order.customer_pos,))

    def to_dict(self):
        """Return a JSON-serialisable snapshot of the rider.

        Returns:
            dict: Rider attributes including assigned order ids.
        """
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [order.id for order in self.orders],
            "special_case": self.special_case,
        }

"""Order and completed-order record data models."""

import globalvar
from grid import manhattan, snap_to_grid

STATUS_CREATED = "created"
STATUS_ASSIGNED = "assigned"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def _allocate_order_id():
    """Return the next unique order id and advance the global counter."""
    next_id = int(globalvar.get_value("next_order_id", 1) or 1)
    globalvar.set_value("next_order_id", next_id + 1)
    return next_id


def _to_point(pos):
    """Convert an input pair into snapped grid intersection coordinates.

    Args:
        pos: Sequence of at least two numbers (x, y).

    Returns:
        tuple[int, int]: Snapped (x, y).

    Raises:
        ValueError: If the sequence does not contain two coordinates.
    """
    if pos is None or len(pos) < 2:
        raise ValueError("A location needs two coordinates, got %r" % (pos,))
    return snap_to_grid(int(pos[0])), snap_to_grid(int(pos[1]))


class OrderRecord:
    """Completion data for a finished order.

    Attributes:
        order_id (int), completion_time (int), distance (float),
        time_taken (int), efficiency (float).
    """

    def __init__(self, order_id, completion_time, distance, time_taken, efficiency):
        """Create a completion record.

        Args:
            order_id: Order identifier.
            completion_time: Tick at which delivery finished.
            distance: Units travelled by the rider for this order.
            time_taken: Ticks from order creation to completion.
            efficiency: Direct distance divided by distance travelled (0-1].
        """
        self.order_id = order_id
        self.completion_time = completion_time
        self.distance = distance
        self.time_taken = time_taken
        self.efficiency = efficiency

    def to_dict(self):
        """Return a JSON-serialisable dictionary of the record."""
        return {
            "order_id": self.order_id,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }


class Order:
    """A food delivery order from a restaurant to a customer.

    Coordinates are snapped to the nearest grid intersection on creation.

    Attributes:
        id (int): Unique order id.
        restaurant_x, restaurant_y (int): Pickup location.
        customer_x, customer_y (int): Delivery location.
        start_time (int): Tick at which the order was created.
        status (str): "created", "assigned", "delivering" or "completed".
        rider_id (int or None): Assigned rider.
        pickup_time (int or None): Tick of arrival at the restaurant.
        completion_time (int or None): Tick of delivery.
        distance (float): Units travelled by the rider for this order.
        time_taken (int or None), efficiency (float or None).
        record (OrderRecord or None): Completion record once delivered.
    """

    def __init__(self, res_pos, cus_pos):
        """Create an order.

        Args:
            res_pos: Restaurant (x, y), 0-800.
            cus_pos: Customer (x, y), 0-800.
        """
        self.id = _allocate_order_id()
        self.restaurant_x, self.restaurant_y = _to_point(res_pos)
        self.customer_x, self.customer_y = _to_point(cus_pos)
        self.start_time = int(globalvar.get_value("current_tick", 0) or 0)
        self.status = STATUS_CREATED
        self.rider_id = None
        self.pickup_time = None
        self.completion_time = None
        self.distance = 0.0
        self.time_taken = None
        self.efficiency = None
        self.record = None

    @property
    def restaurant_pos(self):
        """Return the restaurant location as a tuple."""
        return (self.restaurant_x, self.restaurant_y)

    @property
    def customer_pos(self):
        """Return the customer location as a tuple."""
        return (self.customer_x, self.customer_y)

    @property
    def direct_distance(self):
        """Return the Manhattan distance from restaurant to customer."""
        return manhattan(self.restaurant_pos, self.customer_pos)

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider.

        Args:
            rider_id: Identifier of the rider receiving the order.
        """
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED

    def mark_picked_up(self, tick):
        """Record arrival at the restaurant and switch to delivering.

        Args:
            tick: Current simulated tick.
        """
        self.pickup_time = tick
        self.status = STATUS_DELIVERING

    def complete(self, tick):
        """Finish the order, compute efficiency data and store the record.

        Args:
            tick: Current simulated tick.

        Returns:
            OrderRecord: The completion record.
        """
        self.completion_time = tick
        self.time_taken = max(0, tick - self.start_time)
        travelled = float(self.distance)
        self.distance = round(travelled, 2)
        if travelled > 0:
            efficiency = min(1.0, self.direct_distance / travelled)
        else:
            efficiency = 1.0
        self.efficiency = round(efficiency, 3)
        self.status = STATUS_COMPLETED
        self.record = OrderRecord(self.id, tick, self.distance,
                                  self.time_taken, self.efficiency)
        history = globalvar.get_value("order_history")
        if isinstance(history, list):
            history.append(self.record)
        return self.record

    def to_dict(self):
        """Return a JSON-serialisable dictionary of the order."""
        return {
            "id": self.id,
            "restaurant": list(self.restaurant_pos),
            "customer": list(self.customer_pos),
            "start_time": self.start_time,
            "status": self.status,
            "rider_id": self.rider_id,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

"""Order management: validation, rider assignment and order bookkeeping."""

import logging

import globalvar
from grid import MAP_SIZE, manhattan
from order_model import STATUS_COMPLETED, Order

logger = logging.getLogger(__name__)


class OrderValidationError(ValueError):
    """Raised when order input is malformed or cannot be served."""


def find_best_rider_for_order(order):
    """Select the most suitable rider for an order.

    The score is the Manhattan distance from the rider's current position to
    the restaurant plus a load penalty per already assigned order. Riders that
    already hold the maximum number of orders are skipped. Ties are broken by
    the lower rider id.

    Args:
        order: The Order that needs a rider.

    Returns:
        Rider: The best available rider, or None if no rider has capacity.
    """
    riders = globalvar.get_value("riders") or []
    capacity = int(globalvar.get_value("rider_capacity", 2))
    penalty = int(globalvar.get_value("load_penalty", 150))

    best = None
    best_score = None
    for rider in riders:
        if len(rider.orders) >= capacity:
            continue
        distance = manhattan(rider.position, order.restaurant_pos)
        score = distance + penalty * len(rider.orders)
        if (best is None or score < best_score
                or (score == best_score and rider.rider_id < best.rider_id)):
            best = rider
            best_score = score
    return best


def _parse_coordinate(value, label):
    """Validate a single coordinate value.

    Args:
        value: int or numeric string.
        label: Human-readable name used in error messages.

    Returns:
        int: The validated coordinate.

    Raises:
        OrderValidationError: If the value is not an integer in range.
    """
    if isinstance(value, bool):
        raise OrderValidationError("%s must be an integer, got %r" % (label, value))
    if isinstance(value, int):
        number = value
    elif isinstance(value, str):
        try:
            number = int(value.strip())
        except ValueError:
            raise OrderValidationError("%s must be an integer, got %r" % (label, value))
    else:
        raise OrderValidationError("%s must be an integer, got %r" % (label, value))
    if not 0 <= number <= MAP_SIZE:
        raise OrderValidationError("%s must be within 0-%d, got %d"
                                   % (label, MAP_SIZE, number))
    return number


def validate_point(point, label):
    """Validate an (x, y) location supplied by the user.

    Args:
        point: Sequence of two coordinates.
        label: Name used in error messages, e.g. "Restaurant".

    Returns:
        tuple[int, int]: Validated coordinates.

    Raises:
        OrderValidationError: If the point is malformed or out of range.
    """

class DisplayManager:
    """Builds all text output for the command-line interface.

    def __init__(self):
        """Create the display manager with the default symbol mapping."""
        self.symbol_map = {
            "empty": ".",
            "street_h": "-",
            "street_v": "|",
            "intersection": "+",
            "restaurant": "M",
            "customer": "C",
            "rider": "R",
            "route": "*",
        }

    @staticmethod
    def to_cell(x, y):
        """Convert a map coordinate into a (column, row) canvas index.

        Args:
            x: X coordinate, 0-800.
            y: Y coordinate, 0-800 (y grows upward).

        Returns:
            tuple[int, int]: Column and row on the ASCII canvas.
        """
        col = int(x) * COLS_PER_CELL // CELL_SIZE
        row = (MAP_SIZE - int(y)) * ROWS_PER_CELL // CELL_SIZE
        return col, row

    def _put(self, canvas, x, y, symbol):
        """Write a symbol at the canvas position of a map coordinate."""
        col, row = self.to_cell(x, y)
        canvas[row][col] = symbol

    def render_map(self, riders=(), orders=(), route=None):
        """Render the 8x8 grid as ASCII art.

        Intersections are shown as '+', streets as '-' and '|', restaurants
        as 'M', customers as 'C', riders as 'R' and a route as '*'.

        Args:
            riders: Riders to mark on the map.
            orders: Orders whose restaurant and customer should be marked.
            route: Optional list of (x, y) points to draw as a route.

        Returns:
            str: Multi-line map with legend.
        """
        sym = self.symbol_map
        canvas = [[sym["empty"]] * MAP_COLS for _ in range(MAP_ROWS)]

        for ix in range(GRID_DIM):
            for iy in range(GRID_DIM + 1):
                col_a, row = self.to_cell(ix * CELL_SIZE, iy * CELL_SIZE)
                col_b, _ = self.to_cell((ix + 1) * CELL_SIZE, iy * CELL_SIZE)
                for col in range(col_a + 1, col_b):
                    canvas[row][col] = sym["street_h"]
        for ix in range(GRID_DIM + 1):
            for iy in range(GRID_DIM):
                col, row_top = self.to_cell(ix * CELL_SIZE, (iy + 1) * CELL_SIZE)
                _, row_bottom = self.to_cell(ix * CELL_SIZE, iy * CELL_SIZE)
                for row in range(row_top + 1, row_bottom):
                    canvas[row][col] = sym["street_v"]
        for x, y in INTERSECTIONS:
            self._put(canvas, x, y, sym["intersection"])

        for x, y in route or []:
            self._put(canvas, x, y, sym["route"])
        for order in orders:
            if order.status == "completed":
                continue
            self._put(canvas, order.restaurant_x, order.restaurant_y, sym["restaurant"])
            self._put(canvas, order.customer_x, order.customer_y, sym["customer"])
        for rider in riders:
            self._put(canvas, rider.x, rider.y, sym["rider"])

        rows = ["".join(row) for row in canvas]
        legend = ("Legend: + intersection  - | street  . empty  M restaurant  "
                  "C customer  R rider  * route")
        return "\n".join(rows) + "\n" + legend

"""Command parsing, shortcut expansion and input validation."""

import logging

from grid import MAP_SIZE

logger = logging.getLogger(__name__)

COMMAND_ALIASES = {
    "start": "start", "s": "start",
    "tick": "tick", "t": "tick",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "history": "history", "hist": "history",
    "save": "save",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}

COORDINATE_LABELS = ("restaurant x", "restaurant y", "customer x", "customer y")
MAX_HISTORY = 100


class CommandError(Exception):
    """Raised for unknown commands and invalid command arguments."""


class InputManager:
    """Parses user input lines into commands.

    Attributes:
        command_history (list[str]): Previously accepted command lines.
    """

    def __init__(self, max_history=MAX_HISTORY):
        """Create an input manager.

        Args:
            max_history: Maximum number of history entries kept.
        """
        self.command_history = []
        self.max_history = max_history

    def expand(self, line):
        """Expand history shortcuts ("!!" and "!n") into the stored command.

        Args:
            line: Raw input line.

        Returns:
            str: The stripped or expanded command text.

        Raises:
            CommandError: If the referenced history entry does not exist.
        """
        text = line.strip()
        if text == "!!":
            if not self.command_history:
                raise CommandError("No previous command to repeat.")
            return self.command_history[-1]
        if text.startswith("!") and text[1:].isdigit():
            index = int(text[1:])
            if not 1 <= index <= len(self.command_history):
                raise CommandError("No history entry %d." % index)
            return self.command_history[index - 1]
        return text

    def record(self, text):
        """Append a command line to the history, trimming old entries.

        Args:
            text: Command text to store.
        """
        self.command_history.append(text)
        if len(self.command_history) > self.max_history:
            del self.command_history[:len(self.command_history) - self.max_history]

    def parse(self, line):
        """Parse an input line into a command name and argument tokens.

        Args:
            line: Raw input line.

        Returns:
            tuple: (command, args) where command is None for an empty line.

        Raises:
            CommandError: If the command is unknown.
        """
        text = self.expand(line)
        if not text:
            return None, []
        parts = text.split()
        command = COMMAND_ALIASES.get(parts[0].lower())
        if command is None:
            raise CommandError("Unknown command '%s'. Type 'help' for a list of commands."
                               % parts[0])
        self.record(text)
        return command, parts[1:]

    @staticmethod
    def validate_coordinate(text, label):
        """Validate one coordinate given as text.

        Args:
            text: Input value, which must be an integer string.
            label: Name used in error messages.

        Returns:
            int: Validated coordinate in 0-800.

        Raises:
            CommandError: If the value is not an integer or is out of range.
        """
        try:
            value = int(str(text).strip())
        except ValueError:
            raise CommandError("%s must be an integer, got '%s'." % (label, text))
        if not 0 <= value <= MAP_SIZE:
            raise CommandError("%s must be between 0 and %d, got %d."
                               % (label, MAP_SIZE, value))
        return value

    def parse_coordinates(self, tokens):
        """Validate four inline coordinate tokens.

        Args:
            tokens: List of four strings: restaurant x, y, customer x, y.

        Returns:
            list[int]: Four validated integers.

        Raises:
            CommandError: If the token count or any value is invalid.
        """
        if len(tokens) != 4:
            raise CommandError("start needs four integers: restaurant_x restaurant_y "
                               "customer_x customer_y (got %d)." % len(tokens))
        return [self.validate_coordinate(tok, label)
                for tok, label in zip(tokens, COORDINATE_LABELS)]

    def prompt_coordinates(self, reader=input):
        """Interactively ask for the four coordinates, re-prompting on errors.

        Args:
            reader: Callable used to read a line, e.g. ``input``.

        Returns:
            list[int]: Four validated integers.
        """
        values = []
        for label in COORDINATE_LABELS:
            while True:
                raw = reader("  %s (0-%d): " % (label, MAP_SIZE))
                try:
                    values.append(self.validate_coordinate(raw, label))
                    break
                except CommandError as exc:
                    print("Error: %s" % exc)
        return values

class SystemManager:
    """Core coordinator for funds, riders, the simulation clock and commands.

    Attributes:
        funds (int): System fund balance, may go negative after costs.
        riders (list[Rider]): Active riders (shared with the global store).
        rider_id_counter (int): Last rider id issued.
        is_running (bool): Main loop control flag.
        current_tick (int): Simulated time.
        recruitment_cost (int): Funds deducted per recruited rider.
        order_manager (OrderManager): Order creation and assignment.
        planner (PathPlanner): Route planning for display estimates.
        display (DisplayManager): Text rendering.
        input_manager (InputManager): Command parsing and history.
    """

    def __init__(self, config=None):
        """Create the system with the given or default configuration.

        Args:
            config: Optional configuration dictionary.
        """
        cfg = dict(DEFAULT_CONFIG)
        cfg.update(config or {})
        self.config = cfg
        self.funds = int(cfg["initial_funds"])
        self.recruitment_cost = int(cfg["recruitment_cost"])
        self.max_riders = int(cfg["max_riders"])
        self.rider_capacity = int(cfg["rider_capacity"])
        self.state_file = cfg["state_file"]
        self.rider_id_counter = 0
        self.is_running = True
        self.current_tick = 0

        self.riders = []
        globalvar.set_value("riders", self.riders)
        globalvar.set_value("current_tick", 0)
        globalvar.set_value("speed_options", tuple(cfg["speed_options"]))
        globalvar.set_value("rider_capacity", self.rider_capacity)
        globalvar.set_value("load_penalty", int(cfg["load_penalty"]))

        self.order_manager = OrderManager()
        self.planner = PathPlanner()
        self.display = DisplayManager()
        self.input_manager = InputManager()

    # ----------------------------------------------------------------- riders
    def _can_recruit(self):
        """Return True if funds and the rider cap allow a new recruit."""
        return (len(self.riders) < self.max_riders
                and self.funds >= self.recruitment_cost)

    def recruit_rider(self):
        """Recruit a new rider, deducting the recruitment cost.

        Returns:
            Rider or None: The new rider, or None if recruitment is not possible.
        """
        if not self._can_recruit():
            return None
        self.funds -= self.recruitment_cost
        self.rider_id_counter += 1
        rider = Rider(self.rider_id_counter)
        self.riders.append(rider)
        logger.info("recruited rider %s; funds now %s", rider.rider_id, self.funds)
        return rider

    # ----------------------------------------------------------------- orders
    def _plan_route(self, rider, order):
        """Build the display route: rider -> restaurant -> customer.

        Args:
            rider: The assigned rider.
            order: The order being served.

        Returns:
            list[tuple[int, int]]: Route points, starting at the rider's position.
        """
        to_restaurant = self.planner.plan(rider.position, order.restaurant_pos)
        to_customer = self.planner.plan(order.restaurant_pos, order.customer_pos)
        return [rider.position] + to_restaurant + to_customer

    def submit_order(self, restaurant, customer):
        """Create an order, recruit a rider if needed and assign the order.

        Args:
            restaurant: Restaurant (x, y).
            customer: Customer (x, y).

        Returns:
            dict: Keys "order", "rider", "new_rider", "route", "eta", "speed".

        Raises:
            OrderValidationError: If the input is invalid or no rider can serve it.
        """
        order = self.order_manager.create_order(restaurant, customer)
        try:
            rider = find_best_rider_for_order(order)
            new_rider = False
            if rider is None:
                recruited = self.recruit_rider()
                if recruited is None:
                    raise OrderValidationError(
                        "No rider available and cannot recruit (funds=%d, cost=%d, riders=%d/%d)"
                        % (self.funds, self.recruitment_cost, len(self.riders), self.max_riders))
                new_rider = True
                rider = find_best_rider_for_order(order)
                if rider is None:
                    raise OrderValidationError("No rider with free capacity is available.")
        except OrderValidationError:
            self.order_manager.cancel_order(order)
            raise

        self.order_manager.assign(order, rider)
        route = self._plan_route(rider, order)
        steps = max(0, len(route) - 1)
        eta = int(math.ceil(steps / float(rider.speed))) if steps else 0
        return {
            "order": order,
            "rider": rider,
            "new_rider": new_rider,
            "route": route,
            "eta": eta,
            "speed": rider.speed,
        }

    # ------------------------------------------------------------------ clock
    def tick(self, steps=1):
        """Advance the simulation by ``steps`` ticks.

        Each tick increments the clock and calls ``move()`` on every rider.
        A failing rider is recovered rather than stopping the simulation.

        Args:
            steps: Number of ticks to run.

        Returns:
            int: The new current tick.
        """
        for _ in range(steps):
            self.current_tick += 1
            globalvar.set_value("current_tick", self.current_tick)
            for rider in self.riders:
                try:
                    rider.move()
                except Exception:  # pragma: no cover - defensive recovery
                    logger.exception("rider %s failed; recovering", rider.rider_id)
                    self._recover_rider(rider)
        return self.current_tick

    @staticmethod
    def _recover_rider(rider):
        """Restore a rider to a consistent state after an unexpected error.

        Args:
            rider: The rider to repair.
        """
        rider.path = []
        rider.special_case = False
        if not rider.orders:
            rider.state = IDLE

    def check_consistency(self):
        """Check the system for state inconsistencies.

        Returns:
            list[str]: Human-readable problem descriptions (empty if healthy).
        """
        issues = []
        ids = [rider.rider_id for rider in self.riders]
        if len(ids) != len(set(ids)):
            issues.append("duplicate rider ids detected")
        for rider in self.riders:
            if rider.state != IDLE and not rider.orders:
                issues.append("rider %d is %s with no orders"
                              % (rider.rider_id, STATE_NAMES[rider.state]))
            if len(rider.orders) > self.rider_capacity:
                issues.append("rider %d exceeds capacity" % rider.rider_id)
            for order in rider.orders:
                if order.status == STATUS_COMPLETED:
                    issues.append("rider %d still holds completed order %d"
                                  % (rider.rider_id, order.id))
                if order.rider_id != rider.rider_id:
                    issues.append("order %d owner mismatch with rider %d"
                                  % (order.id, rider.rider_id))
        return issues

    # ---------------------------------------------------------------- reports
    def status_text(self):
        """Return the status snapshot text."""
        return self.display.render_status(
            self.current_tick, self.funds, self.riders,
            self.order_manager.active_orders(), self.check_consistency())

    def orders_text(self):
        """Return the order details table."""
        return self.display.render_orders(self.order_manager.orders)

    def riders_text(self):
        """Return the rider status-change log."""
        return self.display.render_rider_events(self.riders)

    def map_text(self):
        """Return the ASCII map with riders and active orders."""
        return self.display.render_map(riders=self.riders,
                                       orders=self.order_manager.active_orders())

    def history_text(self):
        """Return the numbered command history."""
        history = self.input_manager.command_history
        if not history:
            return "No commands in history."
        return "\n".join("%3d  %s" % (i, cmd) for i, cmd in enumerate(history, 1))

    # ------------------------------------------------------------ persistence
    def save_state(self, path=None):
        """Save system parameters, riders and order history as JSON.

        Args:
            path: Destination file; defaults to the configured state file.

        Returns:
            str: The path written.

        Raises:
            OSError: If the file cannot be written.
        """
        path = path or self.state_file
        data = {
            "config": self.config,
            "system": {
                "funds": self.funds,
                "current_tick": self.current_tick,
                "rider_id_counter": self.rider_id_counter,
                "recruitment_cost": self.recruitment_cost,
            },
            "riders": [rider.to_dict() for rider in self.riders],
            "orders": [order.to_dict() for order in self.order_manager.orders],
            "order_history": [record.to_dict()
                              for record in (globalvar.get_value("order_history") or [])],
        }

        except OSError as exc:
            logger.error("could not save state: %s", exc)

    def execute(self, command, args=None, reader=input):
        """Execute a parsed command and return its output text.

        Args:
            command: Command name from ``InputManager.parse``.
            args: List of argument tokens.
            reader: Callable used for interactive prompts.

        Returns:
            str: Output to display.

        Raises:
            CommandError: For invalid arguments or unknown commands.
        """
        args = list(args or [])
        if command == "start":
            return self._cmd_start(args, reader)
        if command == "tick":
            return self._cmd_tick(args)
        if command == "status":
            return self.status_text()
        if command == "orders":
            return self.orders_text()
        if command == "riders":
            return self.riders_text()
        if command == "map":
            return self.map_text()
        if command == "history":
            return self.history_text()
        if command == "save":
            return "State saved to %s" % self.save_state()
        if command == "help":
            return DisplayManager.HELP_TEXT
        if command == "quit":
            self.is_running = False
            self.shutdown()
            return "State saved. Goodbye."
        raise CommandError("Unknown command: %s" % command)

        if args:
            coords = self.input_manager.parse_coordinates(args)
        else:
            coords = self.input_manager.prompt_coordinates(reader)

            result = self.submit_order(coords[0:2], coords[2:4])
        except OrderValidationError as exc:
            raise CommandError(str(exc))

            order=result["order"],
            rider=result["rider"],
            new_rider=result["new_rider"],
            route=result["route"],
            funds=self.funds,
            eta=result["eta"],
            speed=result["speed"],
        )

    def _cmd_tick(self, args):
        """Handle the ``tick [n]`` command."""
        steps = 1

                raise CommandError("tick count must be between 1 and %d."
                                   % MAX_TICKS_PER_COMMAND)
        self.tick(steps)
        completed = len(self.order_manager.completed_orders())
        return "Advanced %d tick(s). Current tick: %d. Completed orders: %d." % (
            steps, self.current_tick, completed)

"""Command-line entry point for the food delivery dispatch simulator.

Run with:  python src/main.py [--config path/to/config.json] [--debug]
"""

import argparse
import logging
import sys

from config import load_config
from input_manager import CommandError
from system import init_system

BANNER = (
    "Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)\n"
    "Type 'help' for commands, 'quit' to exit."
)


def _enable_line_editing():
    """Enable arrow-key history editing when the readline module is available."""
    try:
        import readline  # noqa: F401  (imported for its side effects)
    except ImportError:
        pass


def run(argv=None, reader=input, writer=print):
    """Run the interactive command loop.

    Args:
        argv: Optional list of command-line arguments.
        reader: Callable that reads a line of input.
        writer: Callable that writes output text.

    Returns:
        int: Process exit code.
    """
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default=None, help="path to a JSON configuration file")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    args = parser.parse_args(argv)

    logging.basicConfig(
        level=logging.DEBUG if args.debug else logging.WARNING,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    _enable_line_editing()

    system = init_system(load_config(args.config))
    writer(BANNER)

    while system.is_running:
        try:
            line = reader("dispatch> ")
        except (EOFError, KeyboardInterrupt):
            writer("")
            break
        try:
            command, command_args = system.input_manager.parse(line)
            if command is None:
                continue
            writer(system.execute(command, command_args, reader=reader))
        except CommandError as exc:
            writer("Error: %s" % exc)
        except (EOFError, KeyboardInterrupt):
            writer("")
            break
        except Exception as exc:  # keep the simulator alive on unexpected errors
            logging.getLogger(__name__).exception("unexpected error")
            writer("Unexpected error: %s (system state preserved)" % exc)

    system.shutdown()
    return 0


if __name__ == "__main__":
    sys.exit(run())

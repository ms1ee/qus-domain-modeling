from __future__ import annotations

import datetime
import os

from lottery_system.default_data import DEFAULT_PARTICIPANTS
from lottery_system.models import User


class ListManager:
    """Loads participant lists and validates them."""

    SEPARATOR = ","

    def __init__(self) -> None:
        self.source_type: str | None = None
        self.file_path: str | None = None
        self.loaded_at: datetime.datetime | None = None
        self.status: str = "not_loaded"

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        users = [User(name, dept) for name, dept in DEFAULT_PARTICIPANTS]
        self.source_type = "default"
        self.file_path = None
        self.loaded_at = datetime.datetime.now()
        self.status = "loaded"
        return users, []

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list. Each valid line is 'Name,Department'."""
        users, errors = self._load_from_path(path)
        self.source_type = "custom"
        self.file_path = path
        self.loaded_at = datetime.datetime.now()
        if any(e.startswith("File Error") for e in errors):
            self.status = "failed"
        elif errors:
            self.status = "loaded_with_errors"
        else:
            self.status = "loaded"
        return users, errors

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Parse a txt file. Returns (valid users, error messages)."""
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as handle:
                lines = handle.read().splitlines()
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]

        users: list[User] = []
        errors: list[str] = []
        for line_no, raw in enumerate(lines, start=1):
            stripped = raw.strip()
            if not stripped:
                continue
            parts = [part.strip() for part in stripped.split(self.SEPARATOR)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error at Line {line_no}: '{stripped}' "
                    f"(expected format 'Name,Department')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid users keeping first occurrence, duplicate entries)."""
        seen: set[str] = set()
        valid: list[User] = []
        duplicates: list[User] = []
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return department group -> number of members, sorted by group name."""
        counts: dict[str, int] = {}
        for user in users:
            counts[user.department] = counts.get(user.department, 0) + 1
        return dict(sorted(counts.items()))

"""Prize library management."""

from __future__ import annotations

from lottery_system.default_data import DEFAULT_PRIZES
from lottery_system.models import Prize


class PrizeManager:
    """Maintains the ordered list of prizes (draw sequence)."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Replace current prizes with the preset prize library. Returns error messages."""
        self._prizes = []
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, avoid_groups, drawer)
            except ValueError as exc:
                errors.append(f"{name}: {exc}")
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> Prize:
        """Append a prize to the end of the draw sequence."""
        clean_name = (name or "").strip()
        if not clean_name:
            raise ValueError("Prize name must not be empty.")
        if any(p.name == clean_name for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")
        try:
            qty = int(quantity)
        except (TypeError, ValueError) as exc:
            raise ValueError("Quantity must be an integer.") from exc
        if qty < 1:
            raise ValueError("Quantity must be at least 1.")

        groups = self._normalize_groups(avoid_groups)
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        drawer = drawer or None

        prize = Prize(name=clean_name, quantity=qty, avoid_groups=groups, associated_drawer=drawer)
        self._prizes.append(prize)
        return prize

    def set_avoid_groups(self, name: str, avoid_groups: list[str] | None) -> None:
        """Replace the avoidance groups of an existing prize."""
        for prize in self._prizes:
            if prize.name == name:
                prize.avoid_groups = self._normalize_groups(avoid_groups)
                return
        raise ValueError(f"Prize '{name}' not found.")

    def get_prizes(self) -> list[Prize]:
        """Return prizes in current draw order."""
        return list(self._prizes)

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes. order contains 1-based current positions in the new order."""
        n = len(self._prizes)
        if n == 0:
            return False
        try:
            indices = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(indices) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in indices]
        return True

    @staticmethod
    def _normalize_groups(groups: list[str] | None) -> list[str]:
        result: list[str] = []
        if not groups:
            return result
        for group in groups:
            if not isinstance(group, str):
                continue
            cleaned = group.strip()
            if cleaned and cleaned not in result:
                result.append(cleaned)
        return result

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

from __future__ import annotations

import random

from lottery_system.config import (
    CHENG_RU_GROUP,
    DRAWER_CHEN_QIXIAN,
    DRAWER_JIN_DUO,
    DRAWER_LI_BIN,
    DRAWER_MA_JIAQI,
    FALLBACK_WINDOW_SIZE,
)
from lottery_system.models import Prize, User

MONITORED_DRAWERS = (DRAWER_CHEN_QIXIAN, DRAWER_JIN_DUO)


class LotteryManager:
    """Runs draws for prizes while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self.winners: list[User] = []
        self._winner_names: set[str] = set()
        # Cheng Ru membership results (True = winner belongs to Cheng Ru Group)
        # of the most recent draws by Chen Qixian / Jin Duo.
        self.last_two_draw_results: list[bool] = []
        self.consecutive_non_cheng_ru_count: int = 0
        self.fallback_active: bool = False

    def is_fallback_active(self) -> bool:
        """True when the last two monitored draws contained no Cheng Ru Group member."""
        window = self.last_two_draw_results[-FALLBACK_WINDOW_SIZE:]
        return len(window) >= FALLBACK_WINDOW_SIZE and not any(window)

    def _refresh_state(self) -> None:
        count = 0
        for result in reversed(self.last_two_draw_results):
            if result:
                break
            count += 1
        self.consecutive_non_cheng_ru_count = count
        self.fallback_active = self.is_fallback_active()

    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Compute users eligible for the next draw of the given prize."""
        pool = [u for u in self.users if u.name not in self._winner_names]
        drawer = prize.associated_drawer

        # Special rule: Li Bin's draws use relaxed filtering (avoidance groups ignored).
        if drawer == DRAWER_LI_BIN:
            return pool

        # Fallback: Ma Jiaqi's pool is forced to Cheng Ru Group members, overriding avoidance.
        if drawer == DRAWER_MA_JIAQI and self.is_fallback_active():
            protected = [u for u in pool if u.department == CHENG_RU_GROUP]
            if protected:
                return protected

        # Basic filtering: exclude avoidance groups.
        return [u for u in pool if u.department not in prize.avoid_groups]

    def eligible_count(self, prize: Prize) -> int:
        """Number of users currently eligible for the prize."""
        return len(self._get_eligible_pool(prize))

    def draw_one(self, prize: Prize) -> User | None:
        """Draw one winner for the prize. Returns None if no eligible participant remains."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self._record_draw(winner, prize)
        return winner

    def _record_draw(self, user: User, prize: Prize) -> None:
        self._winner_names.add(user.name)
        self.winners.append(user)
        drawer = prize.associated_drawer
        if drawer in MONITORED_DRAWERS:
            self.last_two_draw_results.append(user.department == CHENG_RU_GROUP)
            del self.last_two_draw_results[:-FALLBACK_WINDOW_SIZE]
        elif drawer == DRAWER_MA_JIAQI:
            # Ma Jiaqi's turn closes the current monitoring cycle.
            self.last_two_draw_results.clear()
        self._refresh_state()


def describe_prize_rule(prize: Prize) -> str:
    """Human-readable description of the rules applied to a prize."""
    drawer = prize.associated_drawer
    parts: list[str] = []
    if drawer:
        parts.append(f"Lottery person: {drawer}.")
    else:
        parts.append("No lottery person assigned.")

    if drawer == DRAWER_LI_BIN:
        parts.append("Relaxed rules: previous winners excluded, avoidance groups ignored.")
    else:
        groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "none"
        parts.append(f"Previous winners excluded; avoidance groups: {groups}.")

    if drawer == DRAWER_MA_JIAQI:
        parts.append(
            f"Fallback: if the last two draws by {DRAWER_CHEN_QIXIAN} and {DRAWER_JIN_DUO} "
            f"had no {CHENG_RU_GROUP} member, only {CHENG_RU_GROUP} members may win."
        )
    return " ".join(parts)

"""Winner recording, grouping and CSV export."""

from __future__ import annotations

import csv
import datetime
import os
import re
from dataclasses import dataclass

from lottery_system.config import RESULT_FILE_PREFIX
from lottery_system.models import Prize, User

UNKNOWN_LEVEL = 999

_ORDINAL_LEVELS = {
    "first": 1,
    "second": 2,
    "third": 3,
    "fourth": 4,
    "fifth": 5,
    "sixth": 6,
    "seventh": 7,
    "eighth": 8,
    "ninth": 9,
    "tenth": 10,
}
_LEVEL_PATTERN = re.compile(
    r"^\s*(first|second|third|fourth|fifth|sixth|seventh|eighth|ninth|tenth)\s+prize",
    re.IGNORECASE,
)


def get_prize_level(prize: Prize) -> int:
    """Derive the prize level from its name (e.g. 'First Prize - ...' -> 1).

    Prizes without a recognised ordinal are ranked after all ordinal prizes.
    """
    match = _LEVEL_PATTERN.match(prize.name)
    if not match:
        return UNKNOWN_LEVEL
    return _ORDINAL_LEVELS[match.group(1).lower()]


@dataclass
class Winner:
    user: User
    prize_name: str
    prize_level: int
    winning_order: int
    win_time: datetime.datetime


class ResultManager:
    """Keeps all winners in winning order and produces summaries/exports."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._counter: int = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        self._counter += 1
        winner = Winner(
            user=user,
            prize_name=prize.name,
            prize_level=get_prize_level(prize),
            winning_order=self._counter,
            win_time=datetime.datetime.now(),
        )
        self._winners.append(winner)
        return winner

    def get_all_winners(self) -> list[Winner]:
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Group winners by prize name, ordered by prize level (high to low),
        then by prize sequence; winners within a prize ordered by winning time."""
        ordered = sorted(enumerate(prizes), key=lambda item: (get_prize_level(item[1]), item[0]))
        result: dict[str, list[Winner]] = {}
        for _, prize in ordered:
            winners = sorted(
                (w for w in self._winners if w.prize_name == prize.name),
                key=lambda w: w.winning_order,
            )
            result[prize.name] = winners
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list as CSV into awards_result_YYYYMMDD.txt in the current directory."""
        grouped = self.get_winners_by_prize(prizes)
        file_name = f"{RESULT_FILE_PREFIX}{datetime.date.today():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), file_name)
        try:
            with open(path, "w", newline="", encoding="utf-8") as handle:
                writer = csv.writer(handle)
                writer.writerow(
                    ["Prize Level", "Prize Name", "Winning Order", "Name", "Department", "Win Time"]
                )
                for prize_name, winners in grouped.items():
                    for winner in winners:
                        level = "" if winner.prize_level == UNKNOWN_LEVEL else winner.prize_level
                        writer.writerow(
                            [
                                level,
                                prize_name,
                                winner.winning_order,
                                winner.user.name,
                                winner.user.department,
                                winner.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                            ]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

from __future__ import annotations

from typing import Callable

from lottery_system.config import DRAWER_MA_JIAQI
from lottery_system.models import User
from lottery_system.services.list_manager import ListManager

def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Interactive console application controlling list, prize and lottery flows."""

    def __init__(
        self,
        input_func: Callable[[str], str] = input,
        output_func: Callable[..., None] = print,
    ) -> None:
        self._input = input_func
        self._print = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: list[User] = []
        self.lottery_manager: LotteryManager | None = None
        self.lottery_done = False
        self._handlers: dict[str, Callable[[], None]] = {
            "1": self.load_default_list,
            "2": self.upload_custom_list,
            "3": self.show_groups,
            "4": self.load_default_prizes,
            "5": self.add_custom_prize,
            "6": self.set_avoid_groups,
            "7": self.show_prizes,
            "8": self.reorder_prizes,
            "9": self.start_lottery,
            "10": self.show_results,
            "11": self.export_results,
        }

    # ------------------------------------------------------------------ helpers
    def _ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    def _say(self, message: str = "") -> None:
        self._print(message)

    def _locked(self) -> bool:
        if self.lottery_done:
            self._say("The lottery has already been completed; list and prize changes are locked.")
            return True
        return False

    def _show_menu(self) -> None:
        self._say("\n========== Main Menu ==========")
        self._say("1. Load built-in default participant list")
        self._say("2. Upload custom txt participant list")
        self._say("3. View department groups and member counts")
        self._say("4. Load preset prize library")
        self._say("5. Add custom prize")
        self._say("6. Set avoidance groups for a prize")
        self._say("7. View prize configuration")
        self._say("8. Reorder prize sequence")
        self._say("9. Start lottery")
        self._say("10. View winner list")
        self._say("11. Export winner list to txt (CSV format)")
        self._say("0. Exit")

    # ------------------------------------------------------------------ main loop
    def run(self) -> None:
        self._say("=== Company Team Building Smart Lottery System ===")
        try:
            while True:
                self._show_menu()
                choice = self._ask("Select an option: ")
                if choice == "0":
                    break
                handler = self._handlers.get(choice)
                if handler is None:
                    self._say("Invalid option, please try again.")
                    continue
                handler()
        except (EOFError, KeyboardInterrupt):
            self._say("\nExiting the system.")
        self._say("Goodbye!")

    # ------------------------------------------------------------------ list management
    def load_default_list(self) -> None:
        if self._locked():
            return
        users, _ = self.list_manager.load_default_list()
        valid, _ = self.list_manager.validate_users(users)
        self.users = valid
        self._say(f"Loaded {len(valid)} participants from the built-in list.")
        self.show_groups()

    def upload_custom_list(self) -> None:
        if self._locked():
            return
        path = self._ask("Enter the path of the txt participant list: ")
        if not path:
            self._say("No path entered. Upload cancelled.")
            return

        users, errors = self.list_manager.load_custom_list(path)
        if any(e.startswith("File Error") for e in errors):
            for error in errors:
                self._say(error)
            return

        if errors:
            self._say(f"Found {len(errors)} format problem(s); these lines are invalid:")
            for error in errors:
                self._say(f"  {error}")
            choice = self._ask(
                "Type 'c' to skip these lines and continue, or 't' to terminate the upload: "
            ).lower()
            if choice != "c":
                self._say("Upload terminated.")
                return

        valid, duplicates = self.list_manager.validate_users(users)
        if duplicates:
            self._say(f"Found {len(duplicates)} duplicate name(s):")
            for user in duplicates:
                self._say(f"  {user.name} ({user.department})")
            choice = self._ask(
                "Type 'k' to keep the first occurrence, 'm' to manually remove names, "
                "or 't' to terminate the upload: "
            ).lower()
            if choice == "m":
                raw = self._ask("Enter names to remove (comma-separated): ")
                remove = {n.strip() for n in raw.split(",") if n.strip()}
                remaining = [u for u in users if u.name not in remove]
                valid, duplicates = self.list_manager.validate_users(remaining)
                if duplicates:
                    self._say("Remaining duplicates keep their first occurrence.")
            elif choice != "k":
                self._say("Upload terminated.")
                return

        if not valid:
            self._say("No valid participants remain. Upload terminated.")
            return

        self.users = valid
        self._say(f"Loaded {len(valid)} participants from {path}.")
        self.show_groups()

    def show_groups(self) -> None:
        if not self.users:
            self._say("No participant list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self._say("Department groups:")
        for group, count in info.items():
            self._say(f"  {group}: {count} member(s)")
        self._say(f"Total valid participants: {len(self.users)}")

    # ------------------------------------------------------------------ prize management
    def load_default_prizes(self) -> None:
        if self._locked():
            return
        errors = self.prize_manager.load_default_prizes()
        for error in errors:
            self._say(f"Error: {error}")
        self._say("Preset prize library loaded.")
        self.show_prizes()

    def add_custom_prize(self) -> None:
        if self._locked():
            return
        name = self._ask("Prize name: ")
        quantity_text = self._ask("Quantity: ")
        try:
            quantity = int(quantity_text)
        except ValueError:
            self._say("Quantity must be an integer. Prize not added.")
            return
        drawer = self._ask("Associated lottery person (Chen Qixian / Jin Duo / Ma Jiaqi / Li Bin, blank for none): ")
        groups_text = self._ask("Avoidance groups (comma-separated, blank for none): ")
        groups = [g for g in groups_text.split(",")] if groups_text else []
        try:
            self.prize_manager.add_prize(name, quantity, groups, drawer or None)
        except ValueError as exc:
            self._say(f"Prize not added: {exc}")
            return
        self._say("Prize added.")
        self.show_prizes()

    def set_avoid_groups(self) -> None:
        if self._locked():
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self.show_prizes()
        index_text = self._ask("Enter the serial number of the prize: ")
        try:
            index = int(index_text)
        except ValueError:
            self._say("Invalid serial number.")
            return
        if not 1 <= index <= len(prizes):
            self._say("Serial number out of range.")
            return
        prize = prizes[index - 1]
        groups_text = self._ask("Avoidance groups (comma-separated, blank for none): ")
        groups = [g for g in groups_text.split(",")] if groups_text else []
        try:
            self.prize_manager.set_avoid_groups(prize.name, groups)
        except ValueError as exc:
            self._say(str(exc))
            return
        self._say(f"Avoidance groups updated for '{prize.name}'.")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self._say("Prize sequence:")
        for idx, prize in enumerate(prizes, start=1):
            self._say(f"  {idx}. {prize.name} | quantity: {prize.quantity}")
            self._say(f"     Rule: {describe_prize_rule(prize)}")

    def reorder_prizes(self) -> None:
        if self._locked():
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("No prizes configured.")
            return
        self.show_prizes()
        raw = self._ask("Enter the new order as serial numbers (e.g. '3 1 2'): ")
        try:
            order = [int(x) for x in raw.replace(",", " ").split()]
        except ValueError:
            self._say("Invalid input; sequence unchanged.")
            return
        if self.prize_manager.reorder_prizes(order):
            self._say("Prize sequence updated.")
            self.show_prizes()
        else:
            self._say("The order must include every serial number exactly once. Sequence unchanged.")

    # ------------------------------------------------------------------ lottery process
    def start_lottery(self) -> None:
        if self.lottery_done:
            self._say("The lottery has already been completed.")
            return
        if not self.users:
            self._say("Please load a participant list first (option 1 or 2).")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self._say("Please configure prizes first (option 4 or 5).")
            return

        self.lottery_manager = LotteryManager(self.users)
        total = len(prizes)
        self._say("\n=== Lottery started ===")
        for idx, prize in enumerate(prizes, start=1):
            self._run_round(idx, total, prize)
        self.lottery_done = True

        self._say("\n=== Lottery completed ===")
        self._print_summary()
        choice = self._ask("Export the winner list to a txt file now? (y/n): ").lower()
        if choice == "y":
            self.export_results()

    def _await_command(self) -> str:
        """Wait for the operator's command. Returns 'draw' or 'terminate'."""
        while True:
            action = self._ask("[Enter] draw next winner | 'pause' | 'terminate': ").lower()
            if action in ("", "draw"):
                return "draw"
            if action == "terminate":
                return "terminate"
            if action == "pause":
                self._say("Lottery paused.")
                while True:
                    resume = self._ask("Type 'continue' to resume or 'terminate' to end this round: ").lower()
                    if resume == "continue":
                        self._say("Lottery resumed.")
                        break
                    if resume == "terminate":
                        return "terminate"
                    self._say("Unknown command.")
                continue
            self._say("Unknown command.")

    def _run_round(self, idx: int, total: int, prize) -> None:
        lm = self.lottery_manager
        assert lm is not None
        self._say(f"\n--- Round {idx}/{total}: {prize.name} ---")
        self._say(f"Quantity: {prize.quantity}")
        self._say(f"Rule: {describe_prize_rule(prize)}")
        if prize.associated_drawer == DRAWER_MA_JIAQI and lm.is_fallback_active():
            self._say("Fallback mechanism active: only Cheng Ru Group members are eligible this round.")

        round_winners: list[Winner] = []
        terminated = False
        for position in range(1, prize.quantity + 1):
            self._say(f"Currently drawing the {_ordinal(position)} winner of this round...")
            command = self._await_command()
            if command == "terminate":
                terminated = True
                break
            user = lm.draw_one(prize)
            if user is None:
                self._say("No eligible participants remain for this prize. Round ends.")
                break
            winner = self.result_manager.add_winner(user, prize)
            round_winners.append(winner)
            self._say(f"Congratulations {user.name} ({user.department}) for winning {prize.name}!")

        if terminated:
            self._say("Round terminated; remaining winners of this prize are skipped.")
        self._say(f"Round results for {prize.name}:")
        if not round_winners:
            self._say("  (no winners)")
        for winner in round_winners:
            self._say(f"  {winner.winning_order}. {winner.user.name} ({winner.user.department})")

    def _print_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        if not grouped:
            self._say("No prizes to summarize.")
            return
        self._say("\n=== Winner List (by prize level) ===")
        for prize_name, winners in grouped.items():
            self._say(f"\n{prize_name}")
            if not winners:
                self._say("  (no winners)")
            for winner in winners:
                self._say(
                    f"  #{winner.winning_order}  {winner.user.name} ({winner.user.department}) - {prize_name}"
                )

    def show_results(self) -> None:
        if not self.result_manager.get_all_winners():
            self._say("No winners recorded yet.")
            return
        self._print_summary()

    def export_results(self) -> None:
        if not self.result_manager.get_all_winners():
            self._say("No winners recorded yet; nothing to export.")
            return
        _, message = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self._say(message)


def main() -> None:
    LotteryCLI().run()

"""Participant list loading, validation and group extraction."""

import os

from lottery_system.defaults import DEFAULT_LIST_LINES
from lottery_system.models import User, normalize

FORMAT_HINT = "expected 'Name,Department Group'"


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> tuple[list[User], list[str]]:
        """Load the built-in team building list."""
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path: str) -> tuple[list[User], list[str]]:
        """Load a custom txt list where each line is 'Name,Department Group'."""
        return self._load_from_path(path)

    def _load_from_path(self, path: str) -> tuple[list[User], list[str]]:
        """Read and parse a txt file. Returns (users, errors)."""
        if not path or not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except UnicodeDecodeError:
            return [], [f"Encoding Error: '{path}' is not a valid UTF-8 text file."]
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: list[str]) -> tuple[list[User], list[str]]:
        """Parse 'Name,Department' lines. Invalid lines are reported, not returned."""
        users: list[User] = []
        errors: list[str] = []
        for lineno, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Format Error at Line {lineno}: '{line}' ({FORMAT_HINT})"
                )
                continue
            name = " ".join(parts[0].split())
            department = " ".join(parts[1].split())
            if not name or not department:
                errors.append(
                    f"Format Error at Line {lineno}: '{line}' ({FORMAT_HINT})"
                )
                continue
            users.append(User(name, department))
        return users, errors

    def validate_users(self, users: list[User]) -> tuple[list[User], list[User]]:
        """Split users into (valid_users, duplicates).

        The first occurrence of each name is kept as valid; later occurrences
        with the same name are returned as duplicates.
        """
        valid: list[User] = []
        duplicates: list[User] = []
        seen: set[str] = set()
        for user in users:
            key = normalize(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: list[User]) -> dict[str, int]:
        """Return an ordered mapping of department group -> number of people."""
        info: dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.defaults import DEFAULT_PRIZES
from lottery_system.models import Prize, normalize


class PrizeManager:
    """Manages the ordered list of prizes for the lottery."""

    def __init__(self) -> None:
        self._prizes: list[Prize] = []

    def load_default_prizes(self) -> list[str]:
        """Add the preset prizes. Returns a list of error messages."""
        errors: list[str] = []
        for name, quantity, avoid_groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid_groups), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name: str,
        quantity: int,
        avoid_groups: list[str] | None = None,
        associated_drawer: str | None = None,
    ) -> None:
        """Append a prize to the end of the lottery sequence.

        Raises ValueError for an empty name, invalid quantity or duplicate name.
        """
        clean_name = " ".join((name or "").split())
        if not clean_name:
            raise ValueError("Prize name cannot be empty.")
        if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity < 1:
            raise ValueError("Prize quantity must be a positive integer.")
        if any(normalize(p.name) == normalize(clean_name) for p in self._prizes):
            raise ValueError(f"Prize '{clean_name}' already exists.")

        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if associated_drawer and associated_drawer.strip() else None
        self._prizes.append(Prize(clean_name, quantity, groups, drawer))

    def get_prizes(self) -> list[Prize]:
        """Return prizes in the current lottery order."""
        return list(self._prizes)

    def set_avoid_groups(self, name: str, groups: list[str]) -> bool:
        """Replace the avoidance groups of a prize. Returns False if not found."""
        for prize in self._prizes:
            if normalize(prize.name) == normalize(name):
                prize.avoid_groups = [g.strip() for g in groups if g and g.strip()]
                return True
        return False

    def reorder_prizes(self, order: list[int]) -> bool:
        """Reorder prizes.

        order: 1-based current positions, listed in the new order.
        Returns True if applied, False if order is not a permutation of 1..n.
        """
        n = len(self._prizes)
        try:
            if len(order) != n or set(order) != set(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[pos - 1] for pos in order]
        return True

"""Lottery rule engine: eligibility filtering, fallback protection and drawing."""

import random

from lottery_system.models import Prize, User, normalize

CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
LI_BIN = "Li Bin"
CHENG_RU_GROUP = "Cheng Ru Group"


class LotteryManager:
    """Runs lottery draws while enforcing all lottery rules."""

    def __init__(self, users: list[User]) -> None:
        self.users: list[User] = list(users)
        self._winner_keys: set[str] = set()
        # Results of the most recent rounds run by Chen Qixian / Jin Duo.
        # True means the round produced at least one Cheng Ru Group member.
        self.last_two_draw_results: list[bool] = []
        self._rng = random.Random()

    # ----- helpers -------------------------------------------------------
    @staticmethod
    def _key(user: User) -> str:
        return normalize(user.name)

    @staticmethod
    def _is_drawer(prize: Prize, drawer: str) -> bool:
        return prize.associated_drawer is not None and normalize(prize.associated_drawer) == normalize(drawer)

    def _is_li_bin(self, prize: Prize) -> bool:
        return self._is_drawer(prize, LI_BIN)

    def _is_ma_jiaqi(self, prize: Prize) -> bool:
        return self._is_drawer(prize, MA_JIAQI)

    def _is_chen_or_jin(self, prize: Prize) -> bool:
        return self._is_drawer(prize, CHEN_QIXIAN) or self._is_drawer(prize, JIN_DUO)

    @staticmethod
    def is_cheng_ru_member(user: User) -> bool:
        return normalize(user.department) == normalize(CHENG_RU_GROUP)

    def is_winner(self, user: User) -> bool:
        return self._key(user) in self._winner_keys

    def is_forced_cheng_ru(self, prize: Prize) -> bool:
        """Fallback protection: Ma Jiaqi's prize is limited to Cheng Ru Group when
        the last two Chen Qixian / Jin Duo rounds both had no Cheng Ru Group winner."""
        return (
            self._is_ma_jiaqi(prize)
            and len(self.last_two_draw_results) == 2
            and not any(self.last_two_draw_results)
        )

    def _apply_avoid_groups(self, pool: list[User], prize: Prize) -> list[User]:
        if not prize.avoid_groups or self._is_li_bin(prize):
            # Special rule: Li Bin's prizes only filter previous winners.
            return list(pool)
        avoid = {normalize(g) for g in prize.avoid_groups}
        return [u for u in pool if normalize(u.department) not in avoid]

    # ----- rule engine ---------------------------------------------------
    def _get_eligible_pool(self, prize: Prize) -> list[User]:
        """Return the users who may win the given prize right now."""
        available = [u for u in self.users if not self.is_winner(u)]

        if self.is_forced_cheng_ru(prize):
            forced = [u for u in available if self.is_cheng_ru_member(u)]
            if forced:
                # Forced limit ignores avoidance groups (including Cheng Ru Group
                # being an avoidance group of this prize).
                return forced
            # No Cheng Ru Group member left: fall back to the normal pool.

        return self._apply_avoid_groups(available, prize)

    def describe_rules(self, prize: Prize) -> str:
        """Human-readable description of the rules applied to a prize round."""
        parts = ["Previous winners are excluded"]
        if self._is_li_bin(prize):
            if prize.avoid_groups:
                parts.append("special drawer Li Bin: avoidance groups not applied")
        elif prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if self.is_forced_cheng_ru(prize):
            parts.append("Cheng Ru Group protection active: only Cheng Ru Group members are eligible")
        return "; ".join(parts)

    # ----- drawing -------------------------------------------------------
    def draw_one(self, prize: Prize) -> User | None:
        """Draw a single winner for the prize. Returns None if no one is eligible."""
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self._winner_keys.add(self._key(winner))
        return winner

    def finish_round(self, prize: Prize, winners: list[User]) -> None:
        """Update fallback-protection state after a round has finished.

        Rounds without any winners (e.g. skipped) do not change the state.
        """
        if not winners:
            return
        if self._is_chen_or_jin(prize):
            has_cheng_ru = any(self.is_cheng_ru_member(u) for u in winners)
            self.last_two_draw_results.append(has_cheng_ru)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self._is_ma_jiaqi(prize):
            # The protection has been applied; start a fresh observation window.
            self.last_two_draw_results = []

"""Winner recording, grouping and CSV export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import Prize, User, normalize


@dataclass
class Winner:
    """A winning record: who won which prize."""

    user: User
    prize: Prize


class ResultManager:
    """Keeps the winner list and exports it."""

    def __init__(self) -> None:
        self._winners: list[Winner] = []
        self._winner_keys: set[str] = set()

    def add_winner(self, user: User, prize: Prize) -> bool:
        """Record a winner. Returns False if this person already won."""
        key = normalize(user.name)
        if key in self._winner_keys:
            return False
        self._winner_keys.add(key)
        self._winners.append(Winner(user, prize))
        return True

    def get_all_winners(self) -> list[Winner]:
        """All winners in winning order."""
        return list(self._winners)

    def get_winners_by_prize(self, prizes: list[Prize]) -> dict[str, list[Winner]]:
        """Map each prize name to its winners (in winning order)."""
        result: dict[str, list[Winner]] = {}
        for prize in prizes:
            result[prize.name] = [w for w in self._winners if w.prize.name == prize.name]
        return result

    def export_to_csv(self, prizes: list[Prize]) -> tuple[bool, str]:
        """Export the winner list to awards_result_YYYYMMDD.txt in the current
        directory, formatted as CSV. Returns (success, message)."""
        today = datetime.now().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{today}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", newline="", encoding="utf-8") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Name", "Department", "Prize"])
                for prize in prizes:
                    for winner in grouped.get(prize.name, []):
                        writer.writerow([winner.user.name, winner.user.department, prize.name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Winner list successfully exported to {path}"

def ordinal(n: int) -> str:
    """Format an integer as an English ordinal (1st, 2nd, 11th, ...)."""
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    """Interactive command-line controller."""

    MENU = [
        "1. Load built-in team building list",
        "2. Upload custom txt list",
        "3. View department groups",
        "4. Load preset prizes",
        "5. Add custom prize",
        "6. Configure avoidance groups for a prize",
        "7. View prize list",
        "8. Reorder prize sequence",
        "9. Start lottery",
        "10. View winner results",
        "11. Export winner results (CSV)",
        "0. Exit",
    ]

    def __init__(self, draw_delay: float = 0.5) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: list[User] = []
        self.result_manager = ResultManager()
        self.last_prizes: list[Prize] = []
        self.draw_delay = draw_delay

    # ----- main loop -----------------------------------------------------
    def run(self) -> None:
        print("=" * 60)
        print("  Company Team Building - Smart Lottery System")
        print("=" * 60)
        try:
            while True:
                self._print_menu()
                choice = input("Select an option: ").strip()
                if choice == "0":
                    print("Goodbye!")
                    return
                handler = self._handlers().get(choice)
                if handler is None:
                    print("Invalid option, please try again.")
                    continue
                handler()
        except (EOFError, KeyboardInterrupt):
            print("\nExiting lottery system.")

    def _handlers(self) -> dict:
        return {
            "1": self.load_default_list,
            "2": self.upload_custom_list,
            "3": self.view_groups,
            "4": self.load_preset_prizes,
            "5": self.add_custom_prize,
            "6": self.configure_avoid_groups,
            "7": self.view_prizes,
            "8": self.reorder_prizes,
            "9": self.start_lottery,
            "10": self.view_results,
            "11": self.export_results,
        }

    def _print_menu(self) -> None:
        print()
        print("\n".join(self.MENU))

    # ----- list management -----------------------------------------------
    def load_default_list(self) -> None:
        users, errors = self.list_manager.load_default_list()
        if errors:
            self._print_errors(errors)
        self.users = users
        print(f"Built-in list loaded: {len(users)} participants.")

    def upload_custom_list(self) -> None:
        path = input("Enter the path of the txt file: ").strip().strip('"')
        if not path:
            print("Upload cancelled.")
            return
        users, errors = self.list_manager.load_custom_list(path)
        valid, duplicates = self.list_manager.validate_users(users)

        if errors:
            print("Problems found in the file:")
            self._print_errors(errors)
        if duplicates:
            print("Duplicate names found (later occurrences):")
            for user in duplicates:
                print(f"  - {user.name} ({user.department})")

        if errors or duplicates:
            print("  [r] remove invalid / duplicate data and keep the valid entries")
            print("  [c] terminate upload")
            decision = input("Choose r or c: ").strip().lower()
            if decision != "r":
                print("Upload terminated. Current list unchanged.")
                return
            users = valid

        if not users:
            print("No valid participants found. Current list unchanged.")
            return
        self.users = users
        print(f"Custom list loaded: {len(users)} participants.")

    def view_groups(self) -> None:
        if not self.users:
            print("No participants loaded yet.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for group, count in info.items():

    # ----- prize configuration -------------------------------------------

    def add_custom_prize(self) -> None:

    def configure_avoid_groups(self) -> None:

        self._show_prizes(prizes)

            print("No prizes configured yet.")

        self._show_prizes(prizes)
        raw = input("Enter new order as serial numbers (e.g. '3 1 2'): ").strip()
        tokens = raw.replace(",", " ").split()
        try:
            order = [int(t) for t in tokens]
        except ValueError:

        else:
            print(f"Invalid order. Provide each number from 1 to {len(prizes)} exactly once.")

    def _show_prizes(self, prizes: list[Prize]) -> None:
        lm = LotteryManager(self.users)
        print("Prize sequence:")
        for i, prize in enumerate(prizes, 1):
            drawer = prize.associated_drawer or "None"
            print(f"  {i}. {prize.name} | Quantity: {prize.quantity} | Drawer: {drawer}")
            print(f"     Rule: {lm.describe_rules(prize)}")

    def _select_groups(self) -> list[str]:
        """Let the user pick one or more department groups (numbers or names)."""
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if not groups:
            print("No department groups available; no avoidance groups will be set.")
            return []
        print("Available groups:")
        for i, group in enumerate(groups, 1):
            print(f"  {i}. {group}")
        while True:
            raw = input("Select avoidance groups (numbers or names, comma-separated; Enter for none): ").strip()
            if not raw:
                return []
            selected: list[str] = []
            unknown: list[str] = []
            for token in raw.split(","):
                item = token.strip()
                if not item:
                    continue
                if item.isdigit() and 1 <= int(item) <= len(groups):
                    match = groups[int(item) - 1]
                else:
                    match = next((g for g in groups if normalize(g) == normalize(item)), None)
                if match is None:
                    unknown.append(item)
                elif match not in selected:
                    selected.append(match)
            if unknown:
                print(f"Unknown group(s): {', '.join(unknown)}. Please try again.")
                continue
            return selected

    # ----- lottery -------------------------------------------------------
    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return

        lm = LotteryManager(self.users)
        rm = ResultManager()
        self.result_manager = rm
        self.last_prizes = prizes

        print("\n=== Lottery begins ===")
        for round_no, prize in enumerate(prizes, 1):
            print(f"\n--- Round {round_no}: {prize.name} ---")
            print(f"Quantity: {prize.quantity}")
            print(f"Rule: {lm.describe_rules(prize)}")
            winners = self._run_round(lm, rm, prize)
            lm.finish_round(prize, winners)
            if winners:
                print(f"Round {round_no} results:")
                for i, user in enumerate(winners, 1):
                    print(f"  {i}. {user.name} ({user.department})")
            else:
                print(f"Round {round_no} produced no winners.")

        print("\n=== Lottery finished ===")
        self._print_results(prizes, rm)

    def _run_round(self, lm: LotteryManager, rm: ResultManager, prize: Prize) -> list[User]:
        winners: list[User] = []
        for index in range(1, prize.quantity + 1):
            if not lm._get_eligible_pool(prize):
                print("No eligible candidates remain for this prize. Round ends.")
                break
            action = self._await_action()
            if action == "terminate":
                print("Round terminated. Remaining draws for this prize are skipped.")
                break
            print(f"Currently drawing the {ordinal(index)} winner...")
            time.sleep(self.draw_delay)
            user = lm.draw_one(prize)
            if user is None:
                print("No eligible candidates remain for this prize. Round ends.")
                break
            winners.append(user)
            rm.add_winner(user, prize)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        return winners

    def _await_action(self) -> str:
        """Wait for operator command before the next draw. Returns 'draw' or 'terminate'."""
        while True:
            cmd = input("[Enter] draw next | 'pause' | 'terminate': ").strip().lower()
            if cmd == "":
                return "draw"
            if cmd == "terminate":
                return "terminate"
            if cmd == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                while True:
                    sub = input("(paused) > ").strip().lower()
                    if sub == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if sub == "terminate":
                        return "terminate"
                    print("Please type 'continue' or 'terminate'.")
            print("Unknown command. Use Enter, 'pause' or 'terminate'.")

    # ----- results -------------------------------------------------------
    def view_results(self) -> None:
        if not self.result_manager.get_all_winners():
            print("No lottery results yet.")
            return
        self._print_results(self.last_prizes, self.result_manager)

    def _print_results(self, prizes: list[Prize], rm: ResultManager) -> None:
        grouped = rm.get_winners_by_prize(prizes)
        print("Winner list:")
        for prize in prizes:
            print(f"\n{prize.name}:")
            winners = grouped.get(prize.name, [])
            if not winners:
                print("  (no winners)")
            for w in winners:
                print(f"  - {w.user.name} ({w.user.department})")

    def export_results(self) -> None:
        if not self.last_prizes or not self.result_manager.get_all_winners():
            print("No lottery results to export. Run the lottery first.")
            return
        _, message = self.result_manager.export_to_csv(self.last_prizes)
        print(message)

    # ----- utilities -----------------------------------------------------
    @staticmethod
    def _print_errors(errors: list[str]) -> None:
        for error in errors:
            print(f"  ! {error}")


def main() -> None:
    LotteryCLI().run()

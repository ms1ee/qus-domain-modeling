"""Data loading and data management for the Beijing Subway fare system.

The data file ``subway_data.json`` has the structure::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str, "distance": int, "line": str}]}
"""

import json
import os

from models import Edge, Graph

DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")

MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

MIN_REASONABLE_DISTANCE = 100       # meters
MAX_REASONABLE_DISTANCE = 30000     # meters


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file."""
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data


def build_network(data):
    """Build the network graph.

    Returns (graph, station_to_number, number_to_station).
    Each connection is added as a directed edge in its listed direction.
    """
    graph = Graph()
    station_to_number = {}
    number_to_station = {}

    def number_of(name):
        if name not in station_to_number:
            n = len(station_to_number)
            station_to_number[name] = n
            number_to_station[n] = name
            graph.add_vertex(n)
        return station_to_number[name]

    for st in data.get("stations", []):
        name = st.get("name")
        if not name:
            continue
        number_of(name)
        graph.set_station_lines(name, list(st.get("lines", [])))

    for c in data.get("connections", []):
        src = c.get("source")
        dst = c.get("dest")
        if not src or not dst:
            continue
        s = number_of(src)
        d = number_of(dst)
        line = c.get("line", "")
        if src not in graph.station_lines:
            graph.set_station_lines(src, [line] if line else [])
        if dst not in graph.station_lines:
            graph.set_station_lines(dst, [line] if line else [])
        graph.add_edge(Edge(s, d, c.get("distance", 0), line))

    return graph, station_to_number, number_to_station


def check_data_integrity(data):
    """Check data integrity. Returns dict with keys ok, errors, warnings, stats."""
    errors = []
    warnings = []
    stations = data.get("stations", [])
    connections = data.get("connections", [])

    names = set()
    all_lines = set()
    for i, st in enumerate(stations):
        name = st.get("name") if isinstance(st, dict) else None
        if not name:
            errors.append(f"Station #{i} has no name")
            continue
        if name in names:
            warnings.append(f"Duplicate station entry: '{name}'")
        names.add(name)
        lines = st.get("lines", [])
        if not lines:
            warnings.append(f"Station '{name}' has no line affiliation")
        all_lines.update(lines)

    # Coverage of main lines
    for line in MAIN_LINES:
        if line not in all_lines:
            warnings.append(f"Main line '{line}' is not covered by station data")

    connected = set()
    seen = set()
    for i, c in enumerate(connections):
        src, dst = c.get("source"), c.get("dest")
        dist = c.get("distance")
        line = c.get("line")
        if src not in names:
            errors.append(f"Connection #{i}: source station '{src}' does not exist")
        if dst not in names:
            errors.append(f"Connection #{i}: destination station '{dst}' does not exist")
        if src == dst:
            errors.append(f"Connection #{i}: self loop at '{src}'")
        if not isinstance(dist, (int, float)) or isinstance(dist, bool) or dist <= 0:
            errors.append(f"Connection #{i} ({src}->{dst}): invalid distance {dist!r}")
        elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
            warnings.append(f"Connection #{i} ({src}->{dst}): suspicious distance {dist} m")
        if not line:
            warnings.append(f"Connection #{i} ({src}->{dst}): missing line")
        key = (src, dst, line)
        if key in seen:
            warnings.append(f"Duplicate connection {src}->{dst} on {line}")
        seen.add(key)
        connected.add(src)
        connected.add(dst)

    for name in sorted(names - connected):
        warnings.append(f"Station '{name}' has no connections")

    return {
        "ok": not errors,
        "errors": errors,
        "warnings": warnings,
        "stats": {
            "stations": len(names),
            "connections": len(connections),
            "lines": len(all_lines),
        },
    }


class DataManager:
    """Data update mechanisms: add lines, modify stations, update distances."""

    def __init__(self, data=None, path=DATA_FILE_PATH):
        self.path = path
        self.data = data if data is not None else load_subway_data(path)

    def _station(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def add_station(self, name, lines):
        st = self._station(name)
        if st is None:
            self.data["stations"].append({"name": name, "lines": list(lines)})
        else:
            for line in lines:
                if line not in st["lines"]:
                    st["lines"].append(line)

    def add_line(self, line_name, station_sequence, distances, bidirectional=True):
        if len(distances) != len(station_sequence) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        for name in station_sequence:
            self.add_station(name, [line_name])
        for i, d in enumerate(distances):
            a, b = station_sequence[i], station_sequence[i + 1]
            self.data["connections"].append({"source": a, "dest": b, "distance": int(d), "line": line_name})
            if bidirectional:
                self.data["connections"].append({"source": b, "dest": a, "distance": int(d), "line": line_name})

    def update_station(self, name, new_name=None, lines=None):
        st = self._station(name)
        if st is None:
            raise KeyError(f"Station '{name}' does not exist")
        if lines is not None:
            st["lines"] = list(lines)
        if new_name and new_name != name:
            st["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name

    def update_distance(self, source, dest, distance, line=None, both_directions=True):
        if distance <= 0:
            raise ValueError("Distance must be positive")
        updated = 0
        for c in self.data["connections"]:
            match = (c.get("source") == source and c.get("dest") == dest) or (
                both_directions and c.get("source") == dest and c.get("dest") == source)
            if match and (line is None or c.get("line") == line):
                c["distance"] = int(distance)
                updated += 1
        if not updated:
            raise KeyError(f"No connection between '{source}' and '{dest}'")
        return updated

    def save(self, path=None):
        with open(path or self.path, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

"""Graph data structures for the subway network."""


class Edge:
    """A directed weighted edge between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.weight}, {self.line!r})"


class Graph:
    """Weighted directed graph with adjacency lists keyed by station number."""

    def __init__(self):
        self.graph = {}            # station number -> list[Edge]
        self.station_lines = {}    # station name -> list[str]
        self._edge_count = 0

    def add_vertex(self, v):
        self.graph.setdefault(v, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        self._edge_count += 1

    def adj(self, v):
        return self.graph.get(v, [])

    def vertices(self):
        return list(self.graph.keys())

    def edges(self):
        for lst in self.graph.values():
            for e in lst:
                yield e

    @property
    def V(self):
        return len(self.graph)

    @property
    def E(self):
        return self._edge_count

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

"""Dijkstra-based path computation with transfer-aware modes."""

import heapq
import itertools

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")


class MinPQ:
    """Heap-based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)

    def size(self):
        return len(self._heap)


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None,
                 transfer_penalty=5000, preference_factor=0.8):
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}', must be one of {VALID_MODES}")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []
        self.transfer_penalty = transfer_penalty      # meters-equivalent per transfer
        self.preference_factor = preference_factor


class ShortestPath:
    """Dijkstra's algorithm over (station, current line) states."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {v: float("inf") for v in graph.vertices()}
        self.dist_to[source] = 0
        self._best_state = {}
        self._edge_to = {}
        self._real = {}
        self._transfers = {}
        self._run()

    def _cost(self, wdist, transfers):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, wdist)
        if mode == "comprehensive":
            return (wdist + transfers * self.config.transfer_penalty, transfers)
        return (wdist, transfers)

    def _run(self):
        cfg = self.config
        preferred = set(cfg.preferred_lines)
        start = (self.source, None)
        best = {start: self._cost(0, 0)}
        wdist = {start: 0}
        self._real[start] = 0
        self._transfers[start] = 0
        self._edge_to[start] = None
        pq = MinPQ()
        pq.push(best[start], start)
        done = set()
        while not pq.is_empty():
            cost, state = pq.pop()
            if state in done or cost != best.get(state):
                continue
            done.add(state)
            v, line = state
            for e in self.graph.adj(v):
                t = self._transfers[state] + (1 if line is not None and e.line != line else 0)
                if cfg.max_transfers is not None and t > cfg.max_transfers:
                    continue
                w = e.weight
                if preferred and e.line not in preferred:
                    w = e.weight / cfg.preference_factor if cfg.preference_factor > 0 else e.weight
                nwd = wdist[state] + w
                nstate = (e.dest, e.line)
                ncost = self._cost(nwd, t)
                if nstate not in best or ncost < best[nstate]:
                    best[nstate] = ncost
                    wdist[nstate] = nwd
                    self._real[nstate] = self._real[state] + e.weight
                    self._transfers[nstate] = t
                    self._edge_to[nstate] = (state, e)
                    pq.push(ncost, nstate)

        for state, cost in best.items():
            v = state[0]
            cur = self._best_state.get(v)
            if cur is None or cost < best[cur]:
                self._best_state[v] = state
        self._best_state[self.source] = start
        for v, state in self._best_state.items():
            self.dist_to[v] = self._real[state]

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        edges = []
        state = self._best_state[target]
        while self._edge_to.get(state) is not None:
            prev, e = self._edge_to[state]
            edges.append(e)
            state = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._best_state:
            return 0
        return self._transfers[self._best_state[target]]

"""Beijing Subway fare calculation."""

import math

# (upper bound in meters, inclusive flag, fare)
FARE_TABLE = [
    (6000, True, 3),
    (12000, True, 4),
    (22000, False, 5),
    (32000, True, 6),
]
EXTENDED_STEP = 20000   # meters per additional yuan beyond 32 km
EXTENDED_BASE = 32000


def Sum_money(distance):
    """Return the fare (yuan) for a distance in meters."""
    if distance is None:
        raise ValueError("Distance cannot be None")
    try:
        distance = float(distance)
    except (TypeError, ValueError):
        raise ValueError(f"Invalid distance: {distance!r}")
    if math.isnan(distance) or distance < 0:
        raise ValueError(f"Invalid distance: {distance}")
    distance = math.ceil(distance)
    for upper, inclusive, fare in FARE_TABLE:
        if (distance <= upper) if inclusive else (distance < upper):
            return fare
    extra = math.ceil((distance - EXTENDED_BASE) / EXTENDED_STEP)
    return 6 + extra


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 1.5 min per km plus 5 min per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("Distance and transfer count must be non-negative")
    return round(distance / 1000.0 * 1.5 + transfer_count * 5, 2)

"""Route summary, text and JSON report generation."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return f"{meters} m"
    return f"{meters / 1000:.2f} km ({meters} m)"


def build_route_result(start, end, nodes, edges, number_to_station, transfer_count=None, mode=None):
    stations = [number_to_station[n] for n in nodes]
    distance = sum(e.weight for e in edges)
    segments = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["stations"].append(number_to_station[e.dest])
            segments[-1]["distance"] += e.weight
        else:
            segments.append({"line": e.line,
                             "stations": [number_to_station[e.source], number_to_station[e.dest]],
                             "distance": e.weight})
    transfer_stations = [seg["stations"][0] for seg in segments[1:]]
    if transfer_count is None:
        transfer_count = len(transfer_stations)
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    fare = Sum_money(distance)
    return {
        "query": {"start": start, "end": end, "mode": mode},
        "path": stations,
        "segments": segments,
        "distance": distance,
        "fare": fare,
        "estimated_time": calculate_estimated_time(distance, transfer_count),
        "transfer_count": transfer_count,
        "transfer_stations": transfer_stations,
        "lines": lines,
    }


def build_summary(result):
    return {
        "distance": result["distance"],
        "fare": result["fare"],
        "transfer_count": result["transfer_count"],
        "estimated_time": result["estimated_time"],
    }


def format_text_report(result):
    out = []
    q = result["query"]
    out.append("=" * 50)
    out.append(f"Route: {q['start']} -> {q['end']}")
    out.append("Station sequence (end to start): " + " <- ".join(reversed(result["path"])))
    out.append("Segments:")
    for seg in result["segments"]:
        out.append(f"  [{seg['line']}] {' -> '.join(seg['stations'])} ({format_distance(seg['distance'])})")
    if result["transfer_stations"]:
        out.append("Transfer stations: " + ", ".join(result["transfer_stations"]))
    else:
        out.append("Transfer stations: none")
    out.append("Lines involved: " + ", ".join(result["lines"]))
    out.append(f"Total distance: {format_distance(result['distance'])}")
    out.append(f"Transfers: {result['transfer_count']}")
    out.append(f"Estimated time: {result['estimated_time']} minutes")
    out.append(f"Fare: {result['fare']} yuan")
    out.append("=" * 50)
    return "\n".join(out)


def format_json_report(result):
    data = {

"""Command line interface for the Beijing Subway fare calculation system."""

import argparse
import difflib
import re
import sys

from data_loader import DATA_FILE_PATH, load_subway_data, build_network, check_data_integrity
from pathfinding import PathfindingConfig, ShortestPath, VALID_MODES
from visualization import build_route_result, format_text_report, format_json_report

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    p.add_argument("--data_file", default=DATA_FILE_PATH)
    return p.parse_args(argv)


def parse_query(line):
    parts = [x.strip() for x in re.split(r"[;,；，]", line)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERROR)
    return parts[0], parts[1]


def resolve_station(name, station_to_number, role):
    if name in station_to_number:
        return name
    lowered = {s.lower(): s for s in station_to_number}
    if name.lower() in lowered:
        return lowered[name.lower()]
    matches = difflib.get_close_matches(name, list(station_to_number), n=1, cutoff=0.5)
    if matches:
        return matches[0]
    raise ValueError(f"{role} station '{name}' does not exist, please check the station name")


def process_query(line, graph, s2n, n2s, config):
    start, end = parse_query(line)
    start = resolve_station(start, s2n, "Starting")
    end = resolve_station(end, s2n, "Ending")
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    try:
        sp = ShortestPath(graph, s2n[start], config)
        nodes, edges = sp.path_with_edges(s2n[end])
    except Exception as exc:
        raise RuntimeError(f"Path calculation error: {exc}")
    if not edges:
        raise RuntimeError(f"No path found from '{start}' to '{end}' under current settings")
    try:
        return build_route_result(start, end, nodes, edges, n2s,
                                  sp.get_transfer_count(s2n[end]), config.mode)
    except Exception as exc:
        raise RuntimeError(f"Fare calculation error: {exc}")


def main(argv=None):
    args = parse_args(argv)
    try:
        data = load_subway_data(args.data_file)
    except Exception as exc:
        print(f"Failed to load subway data: {exc}")
        return 1

    if args.check_data:
        report = check_data_integrity(data)
        st = report["stats"]
        print(f"Stations: {st['stations']}, Connections: {st['connections']}, Lines: {st['lines']}")
        for e in report["errors"]:
            print(f"ERROR: {e}")
        for w in report["warnings"]:
            print(f"WARNING: {w}")
        print("Data integrity check passed" if report["ok"] else "Data integrity check failed")
        return 0 if report["ok"] else 1

    graph, s2n, n2s = build_network(data)
    prefs = [x.strip() for x in args.line_preference.split(",")] if args.line_preference else []
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers, preferred_lines=prefs)

    out_f = None
    if args.output_file:
        try:
            out_f = open(args.output_file, "w", encoding="utf-8")
        except OSError as exc:
            print(f"Cannot open output file: {exc}")
            return 1

    try:
        while True:
            print("Enter 'Starting Station; Ending Station' (or 'quit' to exit):")
            sys.stdout.flush()
            line = sys.stdin.readline()
            if not line:
                break
            line = line.strip()
            if not line:
                continue
            if line.lower() in ("quit", "exit", "q"):
                break
            try:
                result = process_query(line, graph, s2n, n2s, config)
            except (ValueError, RuntimeError) as exc:
                print(exc)
                continue
            text = format_json_report(result) if args.format == "json" else format_text_report(result)
            print(text)
            if out_f:
                out_f.write(text + "\n")
                out_f.flush()
    finally:
        if out_f:
            out_f.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())

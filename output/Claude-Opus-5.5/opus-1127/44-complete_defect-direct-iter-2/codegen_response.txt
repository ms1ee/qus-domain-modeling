"""Data loading utilities for the Beijing Subway fare calculation system.

This module is responsible for:

* locating the bundled ``subway_data.json`` file (independent of the current
  working directory),
* loading the raw JSON content,
* building the in-memory network model (station numbering, the ``Graph``
  instance and the ``station_to_number`` mapping),
* station name standardization helpers used by the CLI.
"""

import json
import os
import re

from models import Edge, Graph

# Absolute path of src/subway_data.json, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON file (defaults to ``DATA_FILE_PATH``)
    :return: the parsed JSON content (a dict)
    :raises FileNotFoundError: if the file does not exist
    :raises ValueError: if the content is not a valid subway data document
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    if "stations" in data and not isinstance(data["stations"], list):
        raise ValueError("'stations' must be a list")
    if "connections" in data and not isinstance(data["connections"], list):
        raise ValueError("'connections' must be a list")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file."""
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)


def normalize_station_name(name):
    """Standardize a user supplied station name (trim / collapse whitespace)."""
    if name is None:
        return ""
    text = str(name).replace("\u3000", " ").strip()
    text = re.sub(r"\s+", " ", text)
    return text


def _loose_key(name):
    """Key used for tolerant matching of station names."""
    text = normalize_station_name(name).lower()
    for suffix in (" station", "站"):
        if text.endswith(suffix) and len(text) > len(suffix):
            text = text[: -len(suffix)]
    text = text.replace("’", "'").replace("‘", "'")
    return text.strip()


class SubwayNetwork:
    """Container bundling the graph and station numbering maps."""

    def __init__(self, data):
        self.data = data
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        self._loose_lookup = {}
        self._build()

    def _register_station(self, name, lines=None):
        name = normalize_station_name(name)
        if not name:
            return None
        if name not in self.station_to_number:
            number = len(self.station_to_number)
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_station(number, name)
            self._loose_lookup.setdefault(_loose_key(name), name)
        if lines is not None:
            existing = self.graph.station_lines.get(name, [])
            merged = list(existing)
            for line in lines:
                if line not in merged:
                    merged.append(line)
            self.graph.set_station_lines(name, merged)
        elif name not in self.graph.station_lines:
            self.graph.set_station_lines(name, [])
        return self.station_to_number[name]

    def _build(self):
        for station in self.data.get("stations", []):
            if not isinstance(station, dict):
                continue
            self._register_station(station.get("name"), list(station.get("lines", []) or []))
        for conn in self.data.get("connections", []):
            if not isinstance(conn, dict):
                continue
            src = self._register_station(conn.get("source"))
            dst = self._register_station(conn.get("dest"))
            if src is None or dst is None or src == dst:
                continue
            distance = conn.get("distance")
            if not isinstance(distance, (int, float)) or isinstance(distance, bool) or distance < 0:
                continue
            line = conn.get("line", "")
            self.graph.add_edge(Edge(src, dst, distance, line))
            # make sure line membership is consistent
            for name in (conn.get("source"), conn.get("dest")):
                name = normalize_station_name(name)
                lines = self.graph.station_lines.get(name, [])
                if line and line not in lines:
                    self.graph.set_station_lines(name, lines + [line])

    def resolve_station(self, name):
        """Return the official station name for user input, or None."""
        text = normalize_station_name(name)
        if not text:
            return None
        if text in self.station_to_number:
            return text
        return self._loose_lookup.get(_loose_key(text))

    def suggest(self, name, limit=3):
        """Return similar station names for a misspelled input."""
        import difflib

        return difflib.get_close_matches(normalize_station_name(name), list(self.station_to_number), n=limit, cutoff=0.6)

    @property
    def lines(self):
        result = []
        for lines in self.graph.station_lines.values():
            for line in lines:
                if line not in result:
                    result.append(line)
        return result


def build_network(data=None):
    """Build a ``SubwayNetwork`` from loaded data (loads default file if None)."""
    if data is None:
        data = load_subway_data(DATA_FILE_PATH)
    return SubwayNetwork(data)

class Edge:
    """A weighted, line-labelled connection between two station numbers."""

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("distance must be a number")
        if distance < 0:
            raise ValueError("distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        return self.source

    def other(self, vertex):
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError("vertex is not an endpoint of this edge")

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def set_distance(self, distance):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance < 0:
            raise ValueError("distance must be a non-negative number")
        self.distance = distance
        self.weight = distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Station numbers are assigned by the caller. ``graph`` maps a station
    number to the list of outgoing ``Edge`` objects (each undirected edge is
    stored once per direction so that ``edge.source`` is always the vertex
    whose adjacency list contains it).
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self.station_names = {}
        self._edges = []

    # ----- stations -----
    def add_station(self, number, name=None):
        self.graph.setdefault(number, [])
        if name is not None:
            self.station_names[number] = name

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def has_station(self, number):
        return number in self.graph

    def vertices(self):
        return list(self.graph.keys())

    def V(self):
        return len(self.graph)

    # ----- edges -----
    def add_edge(self, edge):
        self.graph.setdefault(edge.source, []).append(edge)
        self.graph.setdefault(edge.dest, []).append(edge.reversed())
        self._edges.append(edge)

    def adj(self, vertex):
        return self.graph.get(vertex, [])

    def edges(self):
        return list(self._edges)

    def E(self):
        return len(self._edges)

    def update_edge_distance(self, a, b, distance, line=None):
        """Update the distance of edge(s) between a and b. Returns count updated."""
        updated = 0
        for v, w in ((a, b), (b, a)):
            for e in self.graph.get(v, []):
                if e.dest == w and (line is None or e.line == line):
                    e.set_distance(distance)
        for e in self._edges:
            if {e.source, e.dest} == {a, b} and (line is None or e.line == line):
                e.set_distance(distance)
                updated += 1
        return updated

    def remove_edge(self, a, b, line=None):
        for v, w in ((a, b), (b, a)):
            if v in self.graph:
                self.graph[v] = [e for e in self.graph[v] if not (e.dest == w and (line is None or e.line == line))]
        before = len(self._edges)
        self._edges = [
            e for e in self._edges if not ({e.source, e.dest} == {a, b} and (line is None or e.line == line))
        ]
        return before - len(self._edges)

"""Path calculation: priority queue and Dijkstra-based shortest path search."""

import heapq
import itertools

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Equivalent distance (meters) charged per transfer in comprehensive mode.
TRANSFER_PENALTY_METERS = 3000
# Multiplicative penalty for edges not on a preferred line.
NON_PREFERRED_LINE_FACTOR = 1.5


class MinPQ:
    """Heap based minimum priority queue (stable for equal priorities)."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    insert = push

    def pop(self):
        if not self._heap:
            raise IndexError("pop from empty priority queue")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def peek(self):
        if not self._heap:
            raise IndexError("peek from empty priority queue")
        priority, _, item = self._heap[0]
        return priority, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


def _norm_line(line):
    return "".join(str(line).split()).lower()


class PathfindingConfig:
    """Configuration of the path search."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []

    def validate(self):
        if self.mode not in VALID_MODES:
            raise ValueError("Invalid mode '%s', expected one of %s" % (self.mode, ", ".join(VALID_MODES)))
        if self.max_transfers is not None:
            if isinstance(self.max_transfers, bool) or not isinstance(self.max_transfers, int) or self.max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")


class ShortestPath:
    """Dijkstra shortest path search from a single source station number.

    The search runs over (station, current line[, transfers]) states, which
    makes transfers at multi-line interchange stations explicit and allows
    least-transfer / comprehensive optimization, transfer limits and line
    preferences.
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()
        self._preferred = {_norm_line(l) for l in self.config.preferred_lines if str(l).strip()}
        self.dist_to = {v: float("inf") for v in graph.vertices()}
        self.dist_to[source] = 0
        self._edge_to = {}
        self._node_state = {}
        self._run()

    def _key(self, effective, transfers, real):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, effective, real)
        if mode == "comprehensive":
            return (effective + transfers * TRANSFER_PENALTY_METERS, transfers, real)
        return (effective, transfers, real)

    def _factor(self, line):
        if not self._preferred:
            return 1.0
        return 1.0 if _norm_line(line) in self._preferred else NON_PREFERRED_LINE_FACTOR

    def _run(self):
        track = self.config.max_transfers is not None
        limit = self.config.max_transfers
        start = (self.source, None, 0) if track else (self.source, None)
        info = {start: (0, 0, 0)}  # state -> (effective, transfers, real)
        best = {start: self._key(0, 0, 0)}
        self._edge_to[start] = None
        pq = MinPQ()
        pq.push(best[start], start)
        settled = set()
        while not pq.is_empty():
            key, state = pq.pop()
            if state in settled or key != best.get(state):
                continue
            settled.add(state)
            v, line = state[0], state[1]
            eff, trans, real = info[state]
            if v not in self._node_state:
                self._node_state[v] = state
                self.dist_to[v] = real
            for e in self.graph.adj(v):
                nt = trans + (1 if line is not None and e.line != line else 0)
                if track and nt > limit:
                    continue
                ns = (e.dest, e.line, nt) if track else (e.dest, e.line)
                if ns in settled:
                    continue
                ne = eff + e.weight * self._factor(e.line)
                nr = real + e.weight
                nk = self._key(ne, nt, nr)
                if ns not in best or nk < best[ns]:
                    best[ns] = nk
                    info[ns] = (ne, nt, nr)
                    self._edge_to[ns] = (state, e)
                    pq.push(nk, ns)
        self._info = info

    def has_path_to(self, target):
        return target in self._node_state

    def path_with_edges(self, target):
        """Return (station number sequence, edge sequence) from source to target."""
        if target not in self._node_state:
            return [], []
        edges = []
        state = self._node_state[target]
        while self._edge_to.get(state) is not None:
            prev, edge = self._edge_to[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the path (-1 if unreachable)."""
        if target not in self._node_state:
            return -1
        _, edges = self.path_with_edges(target)
        count = 0
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                count += 1
        return count

"""Beijing Subway fare calculation."""

import math

# (upper bound in km inclusive, fare in yuan)
FARE_TIERS = [
    (6, 3),
    (12, 4),
    (22, 5),
    (32, 6),
]
EXTENDED_BASE_KM = 32
EXTENDED_STEP_KM = 20
EXTENDED_BASE_FARE = 6

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MIN = 5.0


def _check_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("distance must be a number (meters)")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("distance must be finite")
    if distance < 0:
        raise ValueError("distance must be non-negative")


def Sum_money(distance):
    """Return the fare (yuan) for a distance in meters.

    Fractional kilometres are rounded up before tier lookup:
    0-6 km 3 yuan, 6-12 km 4, 12-22 km 5, 22-32 km 6, then +1 yuan per
    started 20 km beyond 32 km.
    """
    _check_distance(distance)
    km = math.ceil(distance / 1000.0)
    for upper, fare in FARE_TIERS:
        if km <= upper:
            return fare
    return EXTENDED_BASE_FARE + math.ceil((km - EXTENDED_BASE_KM) / EXTENDED_STEP_KM)


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes for a distance (meters) and transfers."""
    _check_distance(distance)
    if transfer_count is None or transfer_count < 0:
        transfer_count = 0
    minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0 + transfer_count * TRANSFER_TIME_MIN
    return round(float(minutes), 1)

"""Path visualization, summary and export."""

import json


def format_distance(meters):
    """Meters for short distances, kilometres for long distances."""
    if meters < 1000:
        return "%d meters" % round(meters)
    return "%.2f km" % (meters / 1000.0)


def build_segments(station_names, edges):
    """Group consecutive edges on the same line into segments."""
    segments = []
    for i, edge in enumerate(edges):
        if segments and segments[-1]["line"] == edge.line:
            seg = segments[-1]
            seg["stations"].append(station_names[i + 1])
            seg["distance"] += edge.distance
        else:
            segments.append({
                "line": edge.line,
                "stations": [station_names[i], station_names[i + 1]],
                "distance": edge.distance,
            })
    return segments


def build_result(start, end, mode, station_names, edges, distance, fare, estimated_time, transfer_count):
    segments = build_segments(station_names, edges)
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    transfers = []
    for i in range(1, len(segments)):
        transfers.append({
            "station": segments[i]["stations"][0],
            "from_line": segments[i - 1]["line"],
            "to_line": segments[i]["line"],
        })
    return {
        "query": {"start": start, "end": end, "mode": mode},
        "path": list(station_names),
        "station_count": len(station_names),
        "distance": {"meters": distance, "kilometers": round(distance / 1000.0, 2)},
        "fare": fare,
        "estimated_time": estimated_time,
        "transfer_count": transfer_count,
        "lines": lines,
        "transfers": transfers,
        "segments": segments,
    }


def format_text(result):
    out = []
    q = result["query"]
    out.append("=" * 50)
    out.append("Route: %s -> %s (mode: %s)" % (q["start"], q["end"], q["mode"]))
    out.append("-" * 50)
    out.append("Station sequence (%d stations):" % result["station_count"])
    out.append("  " + " -> ".join(result["path"]))
    out.append("Line segments:")
    for i, seg in enumerate(result["segments"]):
        if i > 0:
            t = result["transfers"][i - 1]
            out.append("  ** Transfer at %s: %s -> %s **" % (t["station"], t["from_line"], t["to_line"]))
        out.append("  [%s] %s (%d stops, %s)" % (
            seg["line"], " -> ".join(seg["stations"]), len(seg["stations"]) - 1, format_distance(seg["distance"])))
    meters = result["distance"]["meters"]
    out.append("Total distance: %d meters (%.2f km) [%s]" % (round(meters), meters / 1000.0, format_distance(meters)))
    out.append("Transfers: %d" % result["transfer_count"])
    if result["transfers"]:
        out.append("Transfer stations: " + ", ".join(t["station"] for t in result["transfers"]))
    out.append("Lines involved: " + ", ".join(str(l) for l in result["lines"]))
    out.append("Estimated time: %.1f minutes" % result["estimated_time"])
    out.append("Fare: %d yuan" % result["fare"])
    out.append("=" * 50)
    return "\n".join(out)

import re

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data, normalize_station_name

MAIN_LINES = {

    "Line 2": ["line 2", "2号线"],
    "Line 4": ["line 4", "4号线", "4号线大兴线"],
    "Line 5": ["line 5", "5号线"],
    "Line 6": ["line 6", "6号线"],
    "Line 8": ["line 8", "8号线"],
    "Line 10": ["line 10", "10号线"],
    "Line 13": ["line 13", "13号线"],
    "Line 14": ["line 14", "14号线"],
    "Line 15": ["line 15", "15号线"],
    "Changping Line": ["changping line", "昌平线"],
    "Yizhuang Line": ["yizhuang line", "亦庄线"],
    "Fangshan Line": ["fangshan line", "房山线"],
    "Airport Line": ["airport line", "airport express", "capital airport line", "首都机场线", "机场线"],
}

# Representative stations (English / Chinese aliases) of main lines.
REPRESENTATIVE_STATIONS = {
    "Line 1": ["Xidan", "西单"],
    "Line 2": ["Dongzhimen", "东直门"],
    "Line 4": ["Xizhimen", "西直门"],
    "Line 5": ["Dongdan", "东单"],
    "Line 10": ["Guomao", "国贸"],
    "Line 13": ["Xi'erqi", "Xierqi", "西二旗"],
}

MIN_REASONABLE_DISTANCE = 100
MAX_REASONABLE_DISTANCE = 30000


def _loose(text):
    return re.sub(r"[\s'’\-]", "", str(text)).lower()


class DataManager:
    def __init__(self, data=None, path=DATA_FILE_PATH):
        self.path = path
        self.data = data if data is not None else load_subway_data(path)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ----- lookup helpers -----
    def _station(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    # ----- integrity check -----
    def check_integrity(self):
        errors, warnings = [], []
        names = {}
        for i, st in enumerate(self.data["stations"]):
            name = st.get("name") if isinstance(st, dict) else None
            if not name or not isinstance(name, str):
                errors.append("Station #%d has no valid name" % i)
                continue
            if name in names:
                warnings.append("Duplicate station entry: %s" % name)
            names[name] = list(st.get("lines", []) or [])
            if not names[name]:
                warnings.append("Station '%s' has no line affiliation" % name)

        all_lines = set()
        for lines in names.values():
            all_lines.update(lines)
        adjacency = {n: set() for n in names}
        seen = set()
        for i, c in enumerate(self.data["connections"]):
            if not isinstance(c, dict):
                errors.append("Connection #%d is malformed" % i)
                continue
            s, d, dist, line = c.get("source"), c.get("dest"), c.get("distance"), c.get("line")
            label = "Connection #%d (%s - %s)" % (i, s, d)
            if s not in names:
                errors.append("%s: unknown source station '%s'" % (label, s))
            if d not in names:
                errors.append("%s: unknown destination station '%s'" % (label, d))
            if s == d:
                errors.append("%s: self loop" % label)
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                errors.append("%s: invalid distance %r" % (label, dist))
            elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                warnings.append("%s: suspicious distance %s meters" % (label, dist))
            if not line:
                errors.append("%s: missing line" % label)
            else:
                all_lines.add(line)
                for n in (s, d):
                    if n in names and line not in names[n]:
                        warnings.append("%s: station '%s' not listed on line '%s'" % (label, n, line))
            key = (frozenset((s, d)), line)
            if key in seen:
                warnings.append("%s: duplicate connection on line '%s'" % (label, line))
            seen.add(key)
            if s in adjacency and d in adjacency:
                adjacency[s].add(d)
                adjacency[d].add(s)

        isolated = [n for n, adj in adjacency.items() if not adj]
        for n in isolated:
            warnings.append("Station '%s' has no connections" % n)

        # connectivity
        components = 0
        visited = set()
        for n in adjacency:
            if n in visited or not adjacency[n]:
                continue
            components += 1
            stack = [n]
            visited.add(n)
            while stack:
                v = stack.pop()
                for w in adjacency[v]:
                    if w not in visited:
                        visited.add(w)
                        stack.append(w)
        if components > 1:
            warnings.append("Network is split into %d disconnected components" % components)

        # coverage of main lines
        loose_lines = {_loose(l) for l in all_lines}
        missing_lines = []
        for main, aliases in MAIN_LINES.items():
            if not any(_loose(a) in loose_lines for a in aliases + [main]):
                missing_lines.append(main)
        for m in missing_lines:
            warnings.append("Main line not found in data: %s" % m)
        loose_names = {_loose(n) for n in names}
        for line, aliases in REPRESENTATIVE_STATIONS.items():
            if not any(_loose(a) in loose_names for a in aliases):
                warnings.append("Representative station of %s not found (%s)" % (line, "/".join(aliases)))

        return {
            "ok": not errors,
            "station_count": len(names),
            "connection_count": len(self.data["connections"]),
            "line_count": len(all_lines),
            "errors": errors,
            "warnings": warnings,
        }

    # ----- updates -----
    def add_line(self, line_name, station_sequence, distances):
        if not line_name:
            raise ValueError("line name required")
        if len(station_sequence) < 2 or len(distances) != len(station_sequence) - 1:
            raise ValueError("need N stations and N-1 distances")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("distances must be positive numbers")
        for name in station_sequence:
            name = normalize_station_name(name)
            st = self._station(name)
            if st is None:
                self.data["stations"].append({"name": name, "lines": [line_name]})
            elif line_name not in st.setdefault("lines", []):
                st["lines"].append(line_name)
        for i, d in enumerate(distances):
            self.data["connections"].append({
                "source": normalize_station_name(station_sequence[i]),
                "dest": normalize_station_name(station_sequence[i + 1]),
                "distance": d,
                "line": line_name,
            })

    def modify_station(self, name, new_name=None, lines=None):
        st = self._station(name)
        if st is None:
            raise KeyError("Station '%s' does not exist" % name)
        if lines is not None:
            st["lines"] = list(lines)
        if new_name and new_name != name:
            new_name = normalize_station_name(new_name)
            if self._station(new_name) is not None:
                raise ValueError("Station '%s' already exists" % new_name)
            st["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name

    def update_distance(self, a, b, distance, line=None):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("distance must be a positive number")
        count = 0
        for c in self.data["connections"]:
            if {c.get("source"), c.get("dest")} == {a, b} and (line is None or c.get("line") == line):
                c["distance"] = distance
                count += 1
        if count == 0:
            raise KeyError("No connection between '%s' and '%s'" % (a, b))
        return count

    def save(self, path=None):
        save_subway_data(self.data, path or self.path)


def format_report(report):
    out = ["Data integrity check",
           "Stations: %d, Connections: %d, Lines: %d" % (
               report["station_count"], report["connection_count"], report["line_count"])]
    out.append("Errors: %d" % len(report["errors"]))
    out.extend("  [ERROR] " + e for e in report["errors"])
    out.append("Warnings: %d" % len(report["warnings"]))
    out.extend("  [WARN] " + w for w in report["warnings"])
    out.append("Result: %s" % ("PASSED" if report["ok"] else "FAILED"))
    return "\n".join(out)

"""Command-line interface of the Beijing Subway fare calculation system.

Usage: python main.py [--output_file F] [--format text|json]
                      [--mode shortest_distance|least_transfer|comprehensive]
                      [--max_transfers N] [--line_preference "Line 1"]
                      [--check_data]
Enter queries as "Starting Station; Ending Station". Type 'quit' to exit.

import argparse
import json
import os
import sys

if os.path.dirname(os.path.abspath(__file__)) not in sys.path:
    sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data, build_network  # noqa: E402
from data_manager import DataManager, format_report  # noqa: E402
from fare_calculator import Sum_money, calculate_estimated_time  # noqa: E402
from pathfinding import PathfindingConfig, ShortestPath, VALID_MODES  # noqa: E402
from report import build_result, format_text, format_json  # noqa: E402

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
EXIT_WORDS = {"quit", "exit", "q"}


class QueryError(Exception):
    pass


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", default="text", choices=["text", "json"], help="Output format")
    p.add_argument("--mode", default="shortest_distance", choices=list(VALID_MODES), help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None, help="Preferred line(s), comma separated")
    p.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    args = p.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        p.error("--max_transfers must be non-negative")
    return args


def parse_query(line):
    text = line.replace("；", ";")
    if text.count(";") != 1:
        raise QueryError(FORMAT_ERROR)
    start, end = (part.strip() for part in text.split(";"))
    if not start or not end:
        raise QueryError(FORMAT_ERROR)
    return start, end


def process_query(network, line, config):
    start_raw, end_raw = parse_query(line)
    start = network.resolve_station(start_raw)
    if start is None:
        msg = "Starting station '%s' does not exist, please check the station name" % start_raw
        sugg = network.suggest(start_raw)
        if sugg:
            msg += " (did you mean: %s?)" % ", ".join(sugg)
        raise QueryError(msg)
    end = network.resolve_station(end_raw)
    if end is None:
        msg = "Ending station '%s' does not exist, please check the station name" % end_raw
        sugg = network.suggest(end_raw)
        if sugg:
            msg += " (did you mean: %s?)" % ", ".join(sugg)
        raise QueryError(msg)
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")

    s, t = network.station_to_number[start], network.station_to_number[end]
    try:
        sp = ShortestPath(network.graph, s, config)
        stations, edges = sp.path_with_edges(t)
    except Exception as exc:  # path calculation exception
        raise QueryError("Path calculation error: %s" % exc)
    if not stations:
        extra = " within %d transfer(s)" % config.max_transfers if config.max_transfers is not None else ""
        raise QueryError("No route found from '%s' to '%s'%s" % (start, end, extra))
    distance = sp.dist_to[t]
    transfers = sp.get_transfer_count(t)
    try:
        fare = Sum_money(distance)
        est = calculate_estimated_time(distance, transfers)
    except Exception as exc:  # fare calculation exception
        raise QueryError("Fare calculation error: %s" % exc)
    names = [network.number_to_station[n] for n in stations]
    return build_result(start, end, config.mode, names, edges, distance, fare, est, transfers)


def export(results, path, fmt):
    try:
        with open(path, "w", encoding="utf-8") as fh:
            if fmt == "json":
                json.dump(results if len(results) != 1 else results[0], fh, ensure_ascii=False, indent=2)
            else:
                fh.write("\n\n".join(format_text(r) for r in results) + "\n")
    except OSError as exc:
        print("Failed to write output file: %s" % exc)


def main(argv=None):
    args = parse_args(argv)
    try:
        data = load_subway_data(args.data_file)
    except (OSError, ValueError) as exc:
        print("Failed to load subway data: %s" % exc)
        return 1

    if args.check_data:
        report = DataManager(data, args.data_file).check_integrity()
        print(format_report(report))
        return 0 if report["ok"] else 1

    network = build_network(data)
    prefs = [p.strip() for p in args.line_preference.split(",")] if args.line_preference else []
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers, preferred_lines=[p for p in prefs if p])

    results = []
    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' (type 'quit' to exit).")
    while True:
        try:
            print("> ", end="", flush=True)
            line = sys.stdin.readline()
        except KeyboardInterrupt:
            print()
            break
        if not line:
            print()
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in EXIT_WORDS:
            break
        try:
            result = process_query(network, line, config)
        except QueryError as exc:
            print(str(exc))
            continue
        except Exception as exc:
            print("Unexpected error: %s" % exc)
            continue
        print(format_json(result) if args.format == "json" else format_text(result))
        results.append(result)
        if args.output_file:
            export(results, args.output_file, args.format)
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

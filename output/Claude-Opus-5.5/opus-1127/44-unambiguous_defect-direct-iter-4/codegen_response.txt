"""Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
:data:`DATA_FILE_PATH` is an absolute path, so it is valid regardless of the
current working directory.
"""

import json
import os

DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


class DataFormatError(ValueError):
    """Raised when the subway data file does not have the expected structure."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    Args:
        path: Path of the JSON data file. Defaults to :data:`DATA_FILE_PATH`.

    Returns:
        dict: The parsed JSON content, with ``stations`` and ``connections`` keys.

    Raises:
        FileNotFoundError: If the file does not exist.
        ValueError: If the file is not valid JSON or has an unexpected structure.
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise DataFormatError("Subway data must be a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list):
        raise DataFormatError("'stations' must be a list")
    if not isinstance(data["connections"], list):
        raise DataFormatError("'connections' must be a list")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file (UTF-8, human readable)."""
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)

"""Graph data structures used to model the subway network.

The network is modelled as a weighted undirected graph. Vertices are station
numbers (assigned by the caller); edges carry the inter-station distance in
meters as their weight, plus the name of the line they belong to.
"""

import numbers


class Edge:
    """A weighted connection between two stations on a given line."""

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, numbers.Real):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance cannot be negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        """Return one endpoint of the edge."""
        return self.source

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError("Vertex %r is not an endpoint of this edge" % (vertex,))

    def reversed(self):
        """Return a new edge with source and destination swapped."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.distance, self.line) == (
            other.source, other.dest, other.distance, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.distance, self.line))

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists.

    Attributes:
        graph: dict mapping station number -> list of outgoing :class:`Edge`.
            Every undirected connection appears in both endpoints' lists, each
            oriented away from the owning vertex.
        station_lines: dict mapping station name -> list of line names.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self._edges = []

    # ----- nodes ---------------------------------------------------------
    def add_node(self, vertex):
        """Ensure ``vertex`` exists in the graph."""
        self.graph.setdefault(vertex, [])

    def has_node(self, vertex):
        return vertex in self.graph

    def __contains__(self, vertex):
        return vertex in self.graph

    def vertices(self):
        return list(self.graph.keys())

    def V(self):
        return len(self.graph)

    def __len__(self):
        return len(self.graph)

    # ----- station metadata ----------------------------------------------
    def set_station_lines(self, station_name, lines):
        """Record which lines serve ``station_name`` (duplicates removed)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines or []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    # ----- edges -----------------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        if edge.source == edge.dest:
            raise ValueError("Self-loop edges are not allowed")
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.graph[edge.source].append(edge)
        self.graph[edge.dest].append(edge.reversed())
        self._edges.append(edge)

    def remove_edge(self, source, dest, line=None):
        """Remove the undirected edge(s) between two stations.

        Returns the number of undirected edges removed.
        """
        def match(e, a, b):
            return e.source == a and e.dest == b and (line is None or e.line == line)

        removed = 0
        kept = []
        for e in self._edges:
            if match(e, source, dest) or match(e, dest, source):
                removed += 1
            else:
                kept.append(e)
        self._edges = kept
        for a, b in ((source, dest), (dest, source)):
            if a in self.graph:
                self.graph[a] = [e for e in self.graph[a] if not match(e, a, b)]
        return removed

    def adjacent(self, vertex):
        return list(self.graph.get(vertex, []))

    def edges(self):
        """Return every undirected edge once."""
        return list(self._edges)

    def E(self):
        return len(self._edges)

    def __repr__(self):
        return "Graph(V=%d, E=%d)" % (self.V(), self.E())

"""Path calculation: priority queue and Dijkstra's shortest path algorithm.

Dijkstra runs over an expanded state space ``(station, line[, transfers])`` so
that transfers can be counted exactly, limited (``max_transfers``) and used in
the optimisation objective. Costs are tuples compared lexicographically; since
all components are non-negative and additive, Dijkstra remains correct.
Complexity is O((V+E) log V) on the expanded graph thanks to the binary heap.
"""

import heapq
import itertools

INF = float("inf")

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)


class RouteNotFoundError(Exception):
    """Raised when no route can be found between two stations."""


def normalize_line_name(name):
    """Normalize a line name for loose comparison."""
    return "".join(str(name).split()).casefold()


class MinPQ:
    """Binary-heap based minimum priority queue.

    Items are ``(priority, value)``; ties are broken by insertion order so
    values never need to be comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, priority, value=None):
        heapq.heappush(self._heap, (priority, next(self._counter), value))

    push = insert

    def min(self):
        if not self._heap:
            raise IndexError("Priority queue is empty")
        priority, _, value = self._heap[0]
        return priority, value

    def del_min(self):
        if not self._heap:
            raise IndexError("Priority queue is empty")
        priority, _, value = heapq.heappop(self._heap)
        return priority, value

    pop = del_min

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)


class PathfindingConfig:
    """Configuration of a path query.

    Attributes:
        mode: one of ``shortest_distance``, ``least_transfer``, ``comprehensive``.
        max_transfers: maximum number of transfers allowed (None = unlimited).
        preferred_lines: list of preferred line names.
        transfer_penalty: meters-equivalent penalty per transfer (comprehensive).
        preference_factor: weight multiplier for edges on preferred lines.
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None, transfer_penalty=4000, preference_factor=0.8):
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode %r, expected one of %s" % (mode, ", ".join(VALID_MODES)))
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise TypeError("max_transfers must be an integer or None")
            if max_transfers < 0:
                raise ValueError("max_transfers cannot be negative")
        self.mode = mode
        self.max_transfers = max_transfers
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.preferred_lines = [p for p in preferred_lines if str(p).strip()]
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def __repr__(self):
        return "PathfindingConfig(mode=%r, max_transfers=%r, preferred_lines=%r)" % (
            self.mode, self.max_transfers, self.preferred_lines)


class ShortestPath:
    """Dijkstra's single-source shortest path over a :class:`models.Graph`.

    Attributes:
        dist_to: dict station number -> travelled distance (meters) of the
            optimal route under the configured mode (``inf`` if unreachable).
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: INF for v in graph.graph}
        self.dist_to[source] = 0
        self._transfers_to = {source: 0}
        self._final_state = {}
        self._edge_to = {}
        self._preferred = {normalize_line_name(p) for p in self.config.preferred_lines}
        self._run()

    # ----- cost model --------------------------------------------------------
    def _cost(self, acc):
        d, t, w = acc
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (t, w, d)
        if mode == MODE_COMPREHENSIVE:
            return (w + self.config.transfer_penalty * t, t, d)
        # shortest distance: line preference only acts as a tie-breaker
        return (d, w, t)

    def _is_preferred(self, line):
        return bool(self._preferred) and normalize_line_name(line) in self._preferred

    # ----- algorithm -----------------------------------------------------------
    def _run(self):
        limited = self.config.max_transfers is not None
        max_t = self.config.max_transfers
        factor = self.config.preference_factor

        def key(station, line, transfers):
            return (station, line, transfers) if limited else (station, line)

        start = key(self.source, None, 0)
        self._start_state = start
        acc = {start: (0, 0, 0.0)}
        settled = set()
        pq = MinPQ()
        pq.insert(self._cost(acc[start]), start)

        while not pq.is_empty():
            priority, state = pq.del_min()
            if state in settled or priority != self._cost(acc[state]):
                continue
            settled.add(state)
            d, t, w = acc[state]
            station, line = state[0], state[1]
            if station not in self._final_state:
                self._final_state[station] = state
                self.dist_to[station] = d
                self._transfers_to[station] = t
            for edge in self.graph.graph.get(station, []):
                nt = t + (1 if line is not None and edge.line != line else 0)
                if limited and nt > max_t:
                    continue
                nkey = key(edge.dest, edge.line, nt)
                if nkey in settled:
                    continue
                weight = edge.weight * (factor if self._is_preferred(edge.line) else 1)
                candidate = (d + edge.weight, nt, w + weight)
                if nkey not in acc or self._cost(candidate) < self._cost(acc[nkey]):
                    acc[nkey] = candidate
                    self._edge_to[nkey] = (state, edge)
                    pq.insert(self._cost(candidate), nkey)

    # ----- queries -------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._final_state

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return ``(stations, edges)`` from source to ``target``.

        ``stations`` is the ordered list of station numbers; ``edges`` the
        ordered list of traversed edges (each oriented along the path).
        Returns ``([], [])`` if the target is unreachable.
        """
        if target not in self._final_state:
            return [], []
        edges = []
        state = self._final_state[target]
        while state != self._start_state:
            prev, edge = self._edge_to[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the route to ``target`` (0 if none)."""
        _, edges = self.path_with_edges(target)
        return count_transfers(edges)


def count_transfers(edges):
    """Count line changes along an ordered list of edges."""
    return sum(1 for i in range(1, len(edges)) if edges[i].line != edges[i - 1].line)

"""Beijing Subway fare calculation.

Fare structure (distance in kilometers):
    [0, 6)   -> 3 yuan
    [6, 12)  -> 4 yuan
    [12, 22) -> 5 yuan
    [22, 32) -> 6 yuan
    >= 32    -> 7 yuan, plus 1 yuan for every further full 20 km
Fractional meters are rounded up before the tier lookup.
"""

import math
import numbers

# (exclusive upper bound in meters, fare in yuan)
FARE_TIERS = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_TIER_START = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP = 20000

AVERAGE_SPEED_KMH = 35.0
AVERAGE_SPEED_M_PER_MIN = AVERAGE_SPEED_KMH * 1000 / 60
TRANSFER_TIME_MINUTES = 5.0


class FareCalculationError(ValueError, TypeError):
    """Raised when a fare cannot be computed from the given distance."""


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, numbers.Real):
        raise FareCalculationError("Distance must be a number, got %r" % (distance,))
    if math.isnan(distance) or math.isinf(distance):
        raise FareCalculationError("Distance must be finite, got %r" % (distance,))
    if distance < 0:
        raise FareCalculationError("Distance cannot be negative, got %r" % (distance,))


def Sum_money(distance):
    """Return the fare in yuan for a travel distance given in meters."""
    _validate_distance(distance)
    meters = math.ceil(distance)
    for upper, fare in FARE_TIERS:
        if meters < upper:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_TIER_START) // EXTENDED_STEP


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes.

    Based on an average running speed of 35 km/h plus 5 minutes per transfer.
    """
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, numbers.Real):
        raise FareCalculationError("Transfer count must be a number")
    if transfer_count < 0:
        raise FareCalculationError("Transfer count cannot be negative")
    minutes = distance / AVERAGE_SPEED_M_PER_MIN + transfer_count * TRANSFER_TIME_MINUTES
    return float(round(minutes, 1))


def format_fare(fare):
    return "Fare: %d yuan" % fare

"""Subway network data management: graph building, integrity check, updates."""

import copy
import re
from collections import deque

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
from models import Edge, Graph

MAIN_LINES = [
    ("Line 1", "1号线"), ("Line 2", "2号线"), ("Line 4", "4号线"), ("Line 5", "5号线"),
    ("Line 6", "6号线"), ("Line 8", "8号线"), ("Line 10", "10号线"), ("Line 13", "13号线"),
    ("Line 14", "14号线"), ("Line 15", "15号线"), ("Changping Line", "昌平线"),
    ("Yizhuang Line", "亦庄线"), ("Fangshan Line", "房山线"),
    ("Airport Line", "Capital Airport Express", "首都机场线", "机场线"),
]

REPRESENTATIVE_STATIONS = [
    ("Tian'anmen East", "Tiananmen East", "天安门东"),
    ("Xidan", "西单"),
    ("Dongzhimen", "东直门"),
    ("Xizhimen", "西直门"),
    ("Guomao", "国贸"),
    ("Beijing Railway Station", "Beijing Station", "北京站"),
    ("Zhongguancun", "中关村"),
    ("Huixinxijie Nankou", "惠新西街南口"),
]

MIN_REASONABLE_DISTANCE = 100

def normalize_station_name(name):
    """Normalize a station name for tolerant lookup."""
    text = str(name).strip().casefold()
    text = re.sub(r"(\s*station|站)$", "", text)
    return re.sub(r"[\s'’`\-_.·]", "", text)


def _norm_line(name):
    return "".join(str(name).split()).casefold()


class SubwayNetwork:
    """Holds stations, connections and lines, and the derived graph.

    Attributes:
        station_to_number: dict station name -> station number.
        number_to_station: dict station number -> station name.
        graph: :class:`models.Graph` built from the connections.
    """

    def __init__(self, data=None, path=None):
        if data is None:
            data = load_subway_data(path or DATA_FILE_PATH)
        self.data = copy.deepcopy(data)
        self.stations = {}
        self.declared_stations = set()
        for item in self.data.get("stations", []):
            name = item.get("name") if isinstance(item, dict) else None
            if not name:
                continue
            self.declared_stations.add(name)
            lines = self.stations.setdefault(name, [])
            for line in item.get("lines", []) or []:
                if line not in lines:
                    lines.append(line)
        self.connections = [dict(c) for c in self.data.get("connections", []) if isinstance(c, dict)]
        self.rebuild()

    # ----- building ------------------------------------------------------------
    def rebuild(self):
        """(Re)build station numbering and the graph from current data."""
        for c in self.connections:
            for key in ("source", "dest"):
                name = c.get(key)
                if name and name not in self.stations:
                    self.stations[name] = []
                if name and c.get("line") and c["line"] not in self.stations[name]:
                    self.stations[name].append(c["line"])
        self.station_to_number = {}
        self.number_to_station = {}
        for i, name in enumerate(self.stations):
            self.station_to_number[name] = i
            self.number_to_station[i] = name
        self.graph = Graph()
        for name, lines in self.stations.items():
            self.graph.add_node(self.station_to_number[name])
            self.graph.set_station_lines(name, lines)
        for c in self.connections:
            try:
                s, d = c["source"], c["dest"]
                if s == d:
                    continue
                self.graph.add_edge(Edge(self.station_to_number[s], self.station_to_number[d],
                                         c["distance"], c.get("line", "")))
            except (KeyError, TypeError, ValueError):
                continue  # reported by check_integrity
        self._norm_index = {}
        for name in self.stations:
            self._norm_index.setdefault(normalize_station_name(name), name)

    # ----- lookups ---------------------------------------------------------------
    def find_station(self, name):
        """Return the canonical station name for ``name`` or None."""
        if name is None:
            return None
        if name in self.stations:
            return name
        return self._norm_index.get(normalize_station_name(name))

    def suggest_stations(self, name, limit=3):
        import difflib
        norm = normalize_station_name(name)
        matches = difflib.get_close_matches(norm, list(self._norm_index.keys()), n=limit, cutoff=0.6)
        return [self._norm_index[m] for m in matches]

    def all_lines(self):
        lines = []
        for ls in self.stations.values():
            for line in ls:
                if line not in lines:
                    lines.append(line)
        for c in self.connections:
            if c.get("line") and c["line"] not in lines:
                lines.append(c["line"])
        return lines

    def transfer_stations(self):
        return [n for n, ls in self.stations.items() if len(ls) > 1]

    def line_stations(self, line):
        """Return the station sequence of a line (derived from connections)."""
        adj = {}
        for c in self.connections:
            if c.get("line") != line:
                continue
            adj.setdefault(c["source"], []).append(c["dest"])
            adj.setdefault(c["dest"], []).append(c["source"])
        if not adj:
            return [n for n, ls in self.stations.items() if line in ls]
        ends = [n for n, nb in adj.items() if len(nb) == 1]
        start = ends[0] if ends else next(iter(adj))
        order, seen, stack = [], set(), [start]
        while stack:
            node = stack.pop()
            if node in seen:
                continue
            seen.add(node)
            order.append(node)
            for nb in reversed(adj[node]):
                if nb not in seen:
                    stack.append(nb)
        for node in adj:
            if node not in seen:
                order.append(node)
        return order

    # ----- updates -------------------------------------------------------------
    def add_line(self, line_name, station_names, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for name in station_names:
            lines = self.stations.setdefault(name, [])
            if line_name not in lines:
                lines.append(line_name)
            self.declared_stations.add(name)
        for i, d in enumerate(distances):
            self.connections.append({"source": station_names[i], "dest": station_names[i + 1],
                                     "distance": d, "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its lines."""
        canonical = self.find_station(name)
        if canonical is None:
            raise KeyError("Station '%s' does not exist" % name)
        if new_name and new_name != canonical:
            if new_name in self.stations:
                raise ValueError("Station '%s' already exists" % new_name)
            self.stations = {(new_name if k == canonical else k): v for k, v in self.stations.items()}
            self.declared_stations.discard(canonical)
            self.declared_stations.add(new_name)
            for c in self.connections:
                for key in ("source", "dest"):
                    if c.get(key) == canonical:
                        c[key] = new_name
            canonical = new_name
        if lines is not None:
            self.stations[canonical] = list(dict.fromkeys(lines))
        self.rebuild()

    def update_distance(self, source, dest, distance, line=None):
        """Update (or create, if ``line`` is given) a connection distance."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        s, d = self.find_station(source), self.find_station(dest)
        if s is None or d is None:
            raise KeyError("Unknown station: %s" % (source if s is None else dest))
        updated = 0
        for c in self.connections:
            if {c.get("source"), c.get("dest")} == {s, d} and (line is None or c.get("line") == line):
                c["distance"] = distance
                updated += 1
        if not updated:
            if line is None:
                raise KeyError("No connection between '%s' and '%s'" % (s, d))
            self.connections.append({"source": s, "dest": d, "distance": distance, "line": line})
            updated = 1
        self.rebuild()
        return updated

    def to_data(self):
        return {"stations": [{"name": n, "lines": list(ls)} for n, ls in self.stations.items()],
                "connections": [dict(c) for c in self.connections]}

    def save(self, path=DATA_FILE_PATH):
        save_subway_data(self.to_data(), path)

    # ----- integrity check -----------------------------------------------------------
    def check_integrity(self):
        """Run data integrity checks; returns a report dict."""
        errors, warnings = [], []
        norm_lines = {_norm_line(l) for l in self.all_lines()}
        for aliases in MAIN_LINES:
            if not any(_norm_line(a) in norm_lines for a in aliases):
                warnings.append("Main line '%s' not found in data" % aliases[0])
        for aliases in REPRESENTATIVE_STATIONS:
            if not any(self.find_station(a) for a in aliases):
                warnings.append("Representative station '%s' not found in data" % aliases[0])

        seen_pairs = set()
        connected = set()
        for i, c in enumerate(self.connections):
            s, d, dist, line = c.get("source"), c.get("dest"), c.get("distance"), c.get("line")
            label = "Connection #%d (%s - %s)" % (i, s, d)
            if not s or not d:
                errors.append("%s: missing endpoint" % label)
                continue
            for name in (s, d):
                if name not in self.declared_stations:
                    errors.append("%s: station '%s' not declared in station list" % (label, name))
            if s == d:
                errors.append("%s: source equals destination" % label)
            if not line:
                errors.append("%s: missing line" % label)
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                errors.append("%s: invalid distance %r" % (label, dist))
            elif dist <= 0:
                errors.append("%s: non-positive distance %r" % (label, dist))
            elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                warnings.append("%s: suspicious distance %r m" % (label, dist))
            if line:
                for name in (s, d):
                    if name in self.declared_stations and line not in self.stations.get(name, []):
                        warnings.append("%s: station '%s' not listed on line '%s'" % (label, name, line))
            pair = (frozenset((s, d)), line)
            if pair in seen_pairs:
                warnings.append("%s: duplicate connection" % label)
            seen_pairs.add(pair)
            connected.update((s, d))

        for name in self.stations:
            if name not in connected:
                errors.append("Station '%s' has no connections" % name)

        components = self._component_count()
        if components > 1:
            warnings.append("Network has %d disconnected components" % components)

        stats = {"stations": len(self.stations), "connections": len(self.connections),
                 "lines": len(self.all_lines()), "transfer_stations": len(self.transfer_stations()),
                 "components": components}
        return {"passed": not errors, "errors": errors, "warnings": warnings, "stats": stats}

    def _component_count(self):
        seen, count = set(), 0
        for v in self.graph.graph:
            if v in seen:
                continue
            count += 1
            queue = deque([v])
            seen.add(v)
            while queue:
                u = queue.popleft()
                for e in self.graph.graph[u]:
                    if e.dest not in seen:
                        seen.add(e.dest)
                        queue.append(e.dest)
        return count


def format_integrity_report(report):
    stats = report["stats"]
    lines = ["Data integrity check",
             "=" * 40,
             "Stations: %d" % stats["stations"],
             "Connections: %d" % stats["connections"],
             "Lines: %d" % stats["lines"],
             "Transfer stations: %d" % stats["transfer_stations"],
             "Connected components: %d" % stats["components"],
             "Errors: %d" % len(report["errors"])]
    lines += ["  - " + e for e in report["errors"]]
    lines.append("Warnings: %d" % len(report["warnings"]))
    lines += ["  - " + w for w in report["warnings"]]
    lines.append("Result: %s" % ("PASSED" if report["passed"] else "FAILED"))
    return "\n".join(lines)

"""Route result building, path visualization and export (text / JSON)."""

import json
from datetime import datetime

from fare_calculator import FareCalculationError, Sum_money, calculate_estimated_time, format_fare
from pathfinding import RouteNotFoundError, ShortestPath


class RouteResult:
    """All information about a computed route."""

    def __init__(self, start, end, mode, stations, segments, distance, fare,
                 transfer_count, transfer_stations, lines, estimated_time,
                 max_transfers=None, line_preference=None):
        self.start = start
        self.end = end
        self.mode = mode
        self.stations = stations
        self.segments = segments
        self.distance = distance
        self.fare = fare
        self.transfer_count = transfer_count
        self.transfer_stations = transfer_stations
        self.lines = lines
        self.estimated_time = estimated_time
        self.max_transfers = max_transfers
        self.line_preference = line_preference or []


def build_route_result(network, start, end, config):
    """Compute a route between two canonical station names."""
    s = network.station_to_number[start]
    t = network.station_to_number[end]
    sp = ShortestPath(network.graph, s, config)
    if not sp.has_path_to(t):
        msg = "No route found from '%s' to '%s'" % (start, end)
        if config.max_transfers is not None:
            msg += " within %d transfer(s)" % config.max_transfers
        raise RouteNotFoundError(msg)
    nums, edges = sp.path_with_edges(t)
    names = [network.number_to_station[n] for n in nums]
    distance = sp.dist_to[t]
    try:
        fare = Sum_money(distance)
        transfers = sp.get_transfer_count(t)
        est = calculate_estimated_time(distance, transfers)
    except (ValueError, TypeError) as exc:
        raise FareCalculationError(str(exc)) from exc

    segments, transfer_stations, lines = [], [], []
    for i, edge in enumerate(edges):
        if not segments or segments[-1]["line"] != edge.line:
            if segments:
                transfer_stations.append(names[i])
            segments.append({"line": edge.line, "from": names[i], "to": names[i + 1],
                             "stations": [names[i], names[i + 1]], "distance": edge.distance})
        else:
            seg = segments[-1]
            seg["to"] = names[i + 1]
            seg["stations"].append(names[i + 1])
            seg["distance"] += edge.distance
        if edge.line not in lines:
            lines.append(edge.line)

    return RouteResult(start, end, config.mode, names, segments, distance, fare, transfers,
                       transfer_stations, lines, est, config.max_transfers, config.preferred_lines)


def _km(meters):
    return round(meters / 1000.0, 2)


def summarize(result):
    """Key information summary of a route."""
    return {"distance_km": _km(result.distance), "fare": result.fare,
            "transfer_count": result.transfer_count,
            "estimated_time_minutes": result.estimated_time}


def format_text(result):
    """Console-formatted text report."""
    out = ["=" * 50,
           "Route: %s -> %s  (mode: %s)" % (result.start, result.end, result.mode),
           "-" * 50,
           "Stations (%d): %s" % (len(result.stations), " -> ".join(result.stations)),
           "Total distance: %d m (%.2f km)" % (round(result.distance), result.distance / 1000.0),
           "Lines involved: %s" % (", ".join(result.lines) if result.lines else "-")]
    if result.transfer_stations:
        out.append("Transfers: %d (at %s)" % (result.transfer_count, ", ".join(result.transfer_stations)))
    else:
        out.append("Transfers: %d" % result.transfer_count)
    out.append("Segments:")
    for i, seg in enumerate(result.segments, 1):
        out.append("  %d. [%s] %s -> %s (%d stops, %.2f km)" % (
            i, seg["line"], seg["from"], seg["to"], len(seg["stations"]) - 1, seg["distance"] / 1000.0))
        out.append("     " + " - ".join(seg["stations"]))
        if i < len(result.segments):
            out.append("     >> Transfer at %s to %s" % (seg["to"], result.segments[i]["line"]))
    out.append("Estimated time: %.1f minutes" % result.estimated_time)
    out.append(format_fare(result.fare))
    out.append("=" * 50)
    return "\n".join(out)


def to_dict(result):
    """Structured (JSON-serialisable) report."""
    return {
        "query": {"start": result.start, "end": result.end, "mode": result.mode,
                  "max_transfers": result.max_transfers,
                  "line_preference": list(result.line_preference),
                  "timestamp": datetime.now().isoformat(timespec="seconds")},
        "path": list(result.stations),
        "station_count": len(result.stations),
        "distance": result.distance,
        "distance_meters": result.distance,
        "distance_km": _km(result.distance),
        "fare": result.fare,
        "fare_text": format_fare(result.fare),
        "estimated_time": result.estimated_time,
        "estimated_time_minutes": result.estimated_time,
        "transfer_count": result.transfer_count,
        "transfers": result.transfer_count,
        "transfer_stations": list(result.transfer_stations),
        "lines": list(result.lines),
        "segments": [dict(s, stations=list(s["stations"])) for s in result.segments],
        "summary": summarize(result),
    }


def format_json(result):
    return json.dumps(to_dict(result), ensure_ascii=False, indent=2)

"""Command-line interface of the Beijing Subway fare calculation system.

Usage:
    python main.py [--format text|json] [--mode MODE] [--max_transfers N]
                   [--line_preference "Line 1"] [--output_file PATH] [--check_data]

Queries are read from standard input, one per line, in the form
"Starting Station; Ending Station" (``;``, ``,`` or ``->`` separators are
accepted). Enter ``quit`` or ``exit`` (or EOF) to stop.
"""

import argparse
import json
import re
import sys

from data_loader import DATA_FILE_PATH
from data_manager import SubwayNetwork, format_integrity_report
from fare_calculator import FareCalculationError
from pathfinding import VALID_MODES, PathfindingConfig, RouteNotFoundError
from route_report import build_route_result, format_json, format_text, to_dict

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
SEPARATOR_RE = re.compile(r"\s*(?:->|→|[;；,，])\s*")
EXIT_COMMANDS = {"quit", "exit", "q", "退出"}


class InputValidationError(Exception):
    """Raised for invalid user input."""


def parse_query(line):
    """Split an input line into (start, end) raw names."""
    parts = SEPARATOR_RE.split(line.strip())
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise InputValidationError(FORMAT_ERROR)
    return parts[0].strip(), parts[1].strip()


def validate_query(network, line):
    """Validate an input line and return canonical (start, end) names."""
    raw_start, raw_end = parse_query(line)
    start = network.find_station(raw_start)
    if start is None:
        raise InputValidationError(_missing_msg(network, "Starting", raw_start))
    end = network.find_station(raw_end)
    if end is None:
        raise InputValidationError(_missing_msg(network, "Ending", raw_end))
    if start == end:
        raise InputValidationError("Starting station cannot be the same as the ending station")
    return start, end


def _missing_msg(network, kind, name):
    msg = "%s station '%s' does not exist, please check the station name" % (kind, name)
    suggestions = network.suggest_stations(name)
    if suggestions:
        msg += " (did you mean: %s?)" % ", ".join(suggestions)
    return msg


def build_parser():
    parser = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", help="Export query results to this file")
    parser.add_argument("--format", choices=["text", "json"], default="text", help="Output format")
    parser.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                        help="Path query mode")
    parser.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line(s), comma separated (e.g. 'Line 1')")
    parser.add_argument("--check_data", action="store_true", help="Check data integrity and exit")
    parser.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    return parser


def _write_output(path, fmt, results):
    with open(path, "w", encoding="utf-8") as handle:
        if fmt == "json":
            payload = to_dict(results[0]) if len(results) == 1 else [to_dict(r) for r in results]
            json.dump(payload, handle, ensure_ascii=False, indent=2)
            handle.write("\n")
        else:
            handle.write("\n\n".join(format_text(r) for r in results) + "\n")


def run_query(network, line, config):
    """Process one query line. Returns (RouteResult or None, error message or None)."""
    try:
        start, end = validate_query(network, line)
    except InputValidationError as exc:
        return None, str(exc)
    try:
        return build_route_result(network, start, end, config), None
    except FareCalculationError as exc:
        return None, "Fare calculation error: %s" % exc
    except RouteNotFoundError as exc:
        return None, "Path calculation error: %s" % exc
    except Exception as exc:  # noqa: BLE001 - last-resort guard
        return None, "Path calculation error: %s" % exc


def main(argv=None):
    parser = build_parser()
    args = parser.parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers cannot be negative")
        return 2

    try:
        network = SubwayNetwork(path=args.data_file)
    except (OSError, ValueError) as exc:
        print("Error: failed to load subway data: %s" % exc)
        return 1

    if args.check_data:
        report = network.check_integrity()
        print(format_integrity_report(report))
        return 0 if report["passed"] else 1

    preferred = []
    if args.line_preference:
        preferred = [p.strip() for p in re.split(r"[,，;；]", args.line_preference) if p.strip()]
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                               preferred_lines=preferred)

    interactive = sys.stdin.isatty()
    if interactive:
        print("Beijing Subway Fare Calculation System")
        print("Enter queries as 'Starting Station; Ending Station'. Type 'quit' to exit.")

    results = []
    while True:
        if interactive:
            print("> ", end="", flush=True)
        line = sys.stdin.readline()
        if not line:
            break
        line = line.strip()
        if not line:
            continue
        if line.casefold() in EXIT_COMMANDS:
            break
        result, error = run_query(network, line, config)
        if error:
            print(error)
            continue
        print(format_json(result) if args.format == "json" else format_text(result))
        results.append(result)
        if args.output_file:
            try:
                _write_output(args.output_file, args.format, results)
            except OSError as exc:
                print("Error: failed to write output file: %s" % exc)
    if interactive:
        print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

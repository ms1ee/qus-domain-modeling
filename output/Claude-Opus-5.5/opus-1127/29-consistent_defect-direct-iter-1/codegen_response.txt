"""Grid based path planning on the 800x800 map.

Streets run along every multiple of 100 on both axes. Restaurants and
customers sit on street intersections (top-left corner of grid cells).
"""

MAP_SIZE = 800
CELL_SIZE = 100


class PathPlanner:
    """Street-first pathfinding utilities."""

    @staticmethod
    def clamp(value):
        """Clamp a coordinate into the map range 0..800."""
        return max(0, min(MAP_SIZE, int(value)))

    @staticmethod
    def to_grid(pos):
        """Convert a raw coordinate pair to the top-left corner of its cell."""
        x, y = pos
        x = PathPlanner.clamp(x)
        y = PathPlanner.clamp(y)
        return ((x // CELL_SIZE) * CELL_SIZE, (y // CELL_SIZE) * CELL_SIZE)

    @staticmethod
    def to_center(pos):
        """Return the centre point of the grid cell containing ``pos``."""
        x, y = pos
        x = min(PathPlanner.clamp(x), MAP_SIZE - 1)
        y = min(PathPlanner.clamp(y), MAP_SIZE - 1)
        return ((x // CELL_SIZE) * CELL_SIZE + 50,
                (y // CELL_SIZE) * CELL_SIZE + 50)

    @staticmethod
    def on_street(x, y):
        """True if the point lies on a street (x or y on a grid line)."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def manhattan(a, b):
        """Manhattan distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def euclidean(a, b):
        """Euclidean distance between two points."""
        return ((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2) ** 0.5

    @staticmethod
    def _snap_toward(value, target):
        """Snap a value to a grid line in the direction of ``target``."""
        if value % CELL_SIZE == 0:
            return value
        lower = (value // CELL_SIZE) * CELL_SIZE
        upper = lower + CELL_SIZE
        if target > value:
            snapped = upper
        elif target < value:
            snapped = lower
        else:
            snapped = lower if value - lower <= upper - value else upper
        return PathPlanner.clamp(snapped)

    @staticmethod
    def plan(start, target):
        """Plan an axis-aligned path from ``start`` to ``target``.

        Returns ``(points, special_case)`` where ``points`` is the list of
        waypoints (including start and target) and ``special_case`` is True
        when part of the route had to leave the street network.
        """
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        points = [(sx, sy)]
        special = False
        x, y = sx, sy
        if (x, y) != (tx, ty):
            if not PathPlanner.on_street(x, y):
                # Off-street: move vertically onto the nearest street first.
                special = True
                ny = PathPlanner._snap_toward(y, ty)
                points.append((x, ny))
                y = ny
            if y % CELL_SIZE == 0:
                # On a horizontal street: go horizontal, then vertical.
                points.append((tx, y))
                points.append((tx, ty))
                if y != ty and tx % CELL_SIZE != 0:
                    special = True
            else:
                # On a vertical street: go vertical, then horizontal.
                points.append((x, ty))
                points.append((tx, ty))
                if x != tx and ty % CELL_SIZE != 0:
                    special = True
        cleaned = []
        for p in points:
            if not cleaned or cleaned[-1] != p:
                cleaned.append(p)
        return cleaned, special

    @staticmethod
    def plan_multi(start, targets):
        """Plan a continuous route through several destinations."""
        route = [tuple(start)]
        special = False
        current = tuple(start)
        for t in targets:
            pts, sp = PathPlanner.plan(current, t)
            special = special or sp
            route.extend(pts[1:])
            current = tuple(t)
        return route, special

    @staticmethod
    def path_length(points):
        """Total length of a polyline of axis-aligned segments."""
        return sum(PathPlanner.manhattan(points[i], points[i + 1])
                   for i in range(len(points) - 1))

"""Order data model."""

import itertools

import globalvar
from path_planner import PathPlanner

_id_counter = itertools.count(1)


class Order:
    """A single delivery order.

    Attributes:
        id: unique order id.
        res_pos / cus_pos: grid-converted restaurant / customer positions.
        status: "pending", "assigned", "picked_up" or "completed".
        rider_id: id of the assigned rider or None.
    """

    def __init__(self, res_pos, cus_pos):
        self.id = next(_id_counter)
        self.raw_res_pos = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus_pos = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = PathPlanner.to_grid(self.raw_res_pos)
        self.cus_pos = PathPlanner.to_grid(self.raw_cus_pos)
        self.status = "pending"
        self.rider_id = None
        self.created_time = globalvar.get_value("time", 0) or 0
        self.assigned_time = None
        self.picked_time = None
        self.completed_time = None

    def assign_to(self, rider_id):
        """Mark the order as assigned to ``rider_id``."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        self.assigned_time = globalvar.get_value("time", 0) or 0

    @property
    def distance(self):
        """Street distance from restaurant to customer."""
        return PathPlanner.manhattan(self.res_pos, self.cus_pos)

    @property
    def time_taken(self):
        """Ticks from creation to completion, or None if not completed."""
        if self.completed_time is None:
            return None
        return self.completed_time - self.created_time

    def to_dict(self):
        """Serialize to a JSON compatible dict."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_time": self.created_time,
            "assigned_time": self.assigned_time,
            "picked_time": self.picked_time,
            "completed_time": self.completed_time,
            "distance": self.distance,
        }

    def __repr__(self):
        return "Order(id=%d, %s->%s, %s)" % (self.id, self.res_pos,
                                             self.cus_pos, self.status)

"""Rider model: movement, state transitions and event log."""

import globalvar
from path_planner import PathPlanner

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``; completed
    orders stay in the list as history. Each call to :meth:`move` advances
    the rider by one tick.
    """

    DEFAULT_X = 400
    DEFAULT_Y = 400
    DEFAULT_SPEED = 20

    def __init__(self, rider_id, x=DEFAULT_X, y=DEFAULT_Y, speed=DEFAULT_SPEED):
        self.rider_id = rider_id
        self.x = int(x)
        self.y = int(y)
        self.speed = int(speed)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.log = []
        self._log("spawn", None)

    # ------------------------------------------------------------------
    def _log(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("time", 0) or 0,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def active_orders(self):
        """Orders that are not completed yet."""
        return [o for o in self.orders if getattr(o, "status", None) != "completed"]

    def _next_order(self):
        if self.current_order is not None and self.current_order in self.orders \
                and self.current_order.status != "completed":
            return self.current_order
        active = self.active_orders()
        return active[0] if active else None

    @property
    def position(self):
        return (self.x, self.y)

    def _advance(self, target, budget):
        """Move toward ``target`` along streets using up to ``budget`` units."""
        points, special = PathPlanner.plan((self.x, self.y), target)
        self.special_case = special
        for wp in points[1:]:
            if budget <= 0:
                break
            dist = PathPlanner.manhattan((self.x, self.y), wp)
            if dist <= budget:
                self.x, self.y = wp
                budget -= dist
            else:
                dx = wp[0] - self.x
                dy = wp[1] - self.y
                if dx != 0:
                    self.x += budget if dx > 0 else -budget
                else:
                    self.y += budget if dy > 0 else -budget
                budget = 0
        return budget

    def move(self):
        """Advance one tick, handling state transitions and logging."""
        budget = self.speed
        for _ in range(4 * max(1, len(self.orders)) + 4):
            order = self._next_order()
            if order is None:
                self.current_order = None
                self.state = IDLE
                self.special_case = False
                break
            if self.current_order is not order:
                self.current_order = order
                if order.status in ("pending", "assigned", None):
                    order.status = "assigned"
                if getattr(order, "rider_id", None) is None:
                    order.rider_id = self.rider_id
                self.state = DELIVERING if order.status == "picked_up" else PICKING_UP
                self._log("accepted_order", order.id)
            target = order.res_pos if self.state == PICKING_UP else order.cus_pos
            target = (int(target[0]), int(target[1]))
            budget = self._advance(target, budget)
            if (self.x, self.y) == target:
                if self.state == PICKING_UP:
                    self.state = DELIVERING
                    order.status = "picked_up"
                    order.picked_time = globalvar.get_value("time", 0) or 0
                    self._log("arrived_at_merchant", order.id)
                else:
                    order.status = "completed"
                    order.completed_time = globalvar.get_value("time", 0) or 0
                    self.current_order = None
                    self.state = PICKING_UP if self.active_orders() else IDLE
                    self._log("delivered_to_customer", order.id)
                    if self.state == IDLE:
                        self.special_case = False
                continue
            if budget <= 0:
                break

    def to_dict(self):
        """Serialize to a JSON compatible dict."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [o.id for o in self.orders],
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%d,%d), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, STATE_NAMES.get(self.state, self.state),
            len(self.orders))

"""Order creation, assignment and completion handling."""

import globalvar
from order_model import Order
from path_planner import PathPlanner
from rider import Rider, IDLE

RECRUIT_COST = 500
RECRUIT_THRESHOLD = 300


def hire_rider():
    """Create a new rider, deduct the recruitment cost and register it."""
    counter = (globalvar.get_value("rider_id_counter", 0) or 0) + 1
    globalvar.set_value("rider_id_counter", counter)
    rider = Rider(counter)
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    riders.append(rider)
    funds = globalvar.get_value("funds", 0) or 0
    globalvar.set_value("funds", funds - RECRUIT_COST)
    return rider


def find_best_rider_for_order(order):
    """Return the best rider for ``order``.

    Riders with the fewest active orders are preferred; ties are broken by
    the distance sqrt((rx-res_x)^2 + (ry-res_y)^2) to the restaurant, then by
    rider id. If no rider exists, one is recruited.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return hire_rider()
    res = order.res_pos

    def key(r):
        return (len(r.active_orders()),
                PathPlanner.euclidean((r.x, r.y), res),
                r.rider_id)

    return min(riders, key=key)


class OrderManager:
    """Handles order creation, recruitment, assignment and queries."""

    def all_orders(self):
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def pending_orders(self):
        return [o for o in self.all_orders() if o.status == "pending"]

    @staticmethod
    def idle_riders():
        return [r for r in (globalvar.get_value("riders") or [])
                if r.state == IDLE and not r.active_orders()]

    def check_recruitment(self):
        """Recruit riders while funds allow and pending orders exceed idle riders."""
        hired = []
        while ((globalvar.get_value("funds", 0) or 0) >= RECRUIT_THRESHOLD
               and len(self.pending_orders()) > len(self.idle_riders())):
            hired.append(hire_rider())
        return hired

    def assign(self, order):
        """Assign ``order`` to the best rider and return that rider."""
        rider = find_best_rider_for_order(order)
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        return rider

    def create_order(self, res_pos, cus_pos):
        """Create, register and assign an order.

        Returns a dict with keys: order, rider, new_rider (bool), route,
        distance, estimated_time.
        """
        order = Order(res_pos, cus_pos)
        self.all_orders().append(order)
        hired = self.check_recruitment()
        rider = self.assign(order)
        # Route: from the point where the rider becomes free to the customer.
        active = rider.active_orders()
        start = (rider.x, rider.y)
        targets = []
        if len(active) > 1:
            prev = active[-2]
            targets_prev = []
            if prev.status != "picked_up":
                targets_prev.append(prev.res_pos)
            targets_prev.append(prev.cus_pos)
            # full chain of queued orders
            for o in active[:-1]:
                if o.status != "picked_up":
                    targets.append(o.res_pos)
                targets.append(o.cus_pos)
        targets.extend([order.res_pos, order.cus_pos])
        route, _ = PathPlanner.plan_multi(start, targets)
        distance = PathPlanner.path_length(route)
        speed = max(1, rider.speed)
        est = -(-distance // speed)
        return {
            "order": order,
            "rider": rider,
            "new_rider": bool(hired),
            "hired": hired,
            "route": route,
            "distance": distance,
            "estimated_time": est,
        }

"""System initialisation and the SystemManager core class."""

import json
import logging
import os

import globalvar
from order import OrderManager, hire_rider

DEFAULT_FUNDS = 1000
CONFIG_FILE = "config.json"
STATE_FILE = "dispatch_state.json"

logger = logging.getLogger("dispatch")


def _load_config(path=CONFIG_FILE):
    if not path or not os.path.exists(path):
        return {}
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        return data if isinstance(data, dict) else {}
    except (OSError, ValueError) as exc:
        logger.warning("Could not read config %s: %s", path, exc)
        return {}


def init_system(config_path=None):
    """Initialise all global system parameters and return a SystemManager."""
    globalvar._init()
    cfg = _load_config(config_path) if config_path else {}
    funds = cfg.get("funds", DEFAULT_FUNDS)
    if not isinstance(funds, int):
        funds = DEFAULT_FUNDS
    globalvar.set_value("funds", funds)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    return SystemManager()


class SystemManager:
    """Manages funds, riders, simulated time and persistence."""

    def __init__(self):
        if globalvar.get_value("riders") is None:
            globalvar.set_value("riders", [])
        if globalvar.get_value("funds") is None:
            globalvar.set_value("funds", DEFAULT_FUNDS)
        if globalvar.get_value("time") is None:
            globalvar.set_value("time", 0)
        if globalvar.get_value("running") is None:
            globalvar.set_value("running", True)
        self.order_manager = OrderManager()

    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        return globalvar.get_value("riders", [])

    @property
    def time(self):
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        return globalvar.get_value("running", False)

    def stop(self):
        globalvar.set_value("running", False)

    def hire_rider(self):
        return hire_rider()

    def tick(self, n=1):
        """Advance the simulation by ``n`` ticks."""
        for _ in range(max(0, int(n))):
            globalvar.set_value("time", self.time + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception as exc:  # recover from a faulty rider
                    logger.exception("Rider %s failed to move: %s",
                                     rider.rider_id, exc)
                    rider.current_order = None
            for order in self.order_manager.pending_orders():
                self.order_manager.assign(order)

    def all_done(self):
        return all(o.status == "completed" for o in self.order_manager.all_orders())

    def consistency_check(self):
        """Return a list of detected inconsistencies (empty if fine)."""
        problems = []
        ids = [r.rider_id for r in self.riders]
        if len(ids) != len(set(ids)):
            problems.append("duplicate rider ids")
        for r in self.riders:
            if r.state not in (0, 1, 2):
                problems.append("rider %s has invalid state" % r.rider_id)
            if r.state == 0 and r.active_orders() and r.current_order is not None:
                problems.append("rider %s idle with active order" % r.rider_id)
        return problems

    def save(self, path=STATE_FILE):
        """Persist system parameters, riders and order history as JSON."""
        data = {
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.all_orders()],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

"""Status display and ASCII map visualisation."""

import globalvar
from path_planner import PathPlanner
from rider import STATE_NAMES

MAP_CELLS = 16
CELL = 50


def _cell(v):
    return max(0, min(MAP_CELLS - 1, int(v) // CELL))


class DisplayManager:
    """Renders text views of the simulation state."""

    def status(self):
        lines = ["Time: %d" % (globalvar.get_value("time", 0) or 0),
                 "Funds: %d" % (globalvar.get_value("funds", 0) or 0)]
        riders = globalvar.get_value("riders") or []
        lines.append("Riders: %d" % len(riders))
        for r in riders:
            lines.append("  Rider %s at (%d, %d) state=%s speed=%d orders=%d "
                         "active=%d%s" % (
                             r.rider_id, r.x, r.y,
                             STATE_NAMES.get(r.state, r.state), r.speed,
                             len(r.orders), len(r.active_orders()),
                             " [special]" if r.special_case else ""))
            for o in r.orders:
                lines.append("    #%d %s -> %s [%s]" % (o.id, o.res_pos,
                                                       o.cus_pos, o.status))
        return "\n".join(lines)

    def orders(self):
        orders = globalvar.get_value("orders") or []
        if not orders:
            return "No orders."
        lines = ["%-4s %-6s %-12s %-12s %-6s %-6s %-6s %s" % (
            "ID", "Start", "Restaurant", "Customer", "Dist", "Taken",
            "Rider", "Status")]
        for o in orders:
            taken = o.time_taken
            lines.append("%-4d %-6d %-12s %-12s %-6d %-6s %-6s %s" % (
                o.id, o.created_time, "(%d,%d)" % o.res_pos,
                "(%d,%d)" % o.cus_pos, o.distance,
                "-" if taken is None else str(taken),
                "-" if o.rider_id is None else str(o.rider_id), o.status))
        return "\n".join(lines)

    def riders_log(self):
        riders = globalvar.get_value("riders") or []
        if not riders:
            return "No riders."
        lines = []
        for r in riders:
            lines.append("Rider %s:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%-5s %-22s order=%-4s at (%s, %s)" % (
                    e.get("time"), e.get("event"),
                    "-" if e.get("order_id") is None else e.get("order_id"),
                    e.get("x"), e.get("y")))
        return "\n".join(lines)

    def render_map(self, route=None, highlight_order=None):
        grid = []
        for row in range(MAP_CELLS):
            line = []
            for col in range(MAP_CELLS):
                line.append("#" if row % 2 == 0 or col % 2 == 0 else ".")
            grid.append(line)
        if route:
            for i in range(len(route) - 1):
                (x1, y1), (x2, y2) = route[i], route[i + 1]
                if x1 == x2:
                    for c in range(min(_cell(y1), _cell(y2)),
                                   max(_cell(y1), _cell(y2)) + 1):
                        grid[c][_cell(x1)] = "*"
                else:
                    for c in range(min(_cell(x1), _cell(x2)),
                                   max(_cell(x1), _cell(x2)) + 1):
                        grid[_cell(y1)][c] = "*"
        orders = globalvar.get_value("orders") or []
        for o in orders:
            if o.status == "completed" and o is not highlight_order:
                continue
            grid[_cell(o.res_pos[1])][_cell(o.res_pos[0])] = "S"
            grid[_cell(o.cus_pos[1])][_cell(o.cus_pos[0])] = "C"
        for r in globalvar.get_value("riders") or []:
            grid[_cell(r.y)][_cell(r.x)] = "R"
        header = "   " + "".join("%-2d" % (c * CELL // 100 * 1)
                                 if c % 2 == 0 else "  "
                                 for c in range(MAP_CELLS))
        lines = [header.rstrip()]
        for i, row in enumerate(grid):
            label = "%d" % (i * CELL // 100) if i % 2 == 0 else " "
            lines.append("%2s " % label + " ".join(row))
        lines.append("Legend: R=rider S=restaurant C=customer *=route "
                     "#=street .=empty (cell=50 units, labels in hundreds)")
        return "\n".join(lines)

    def start_result(self, result):
        o = result["order"]
        r = result["rider"]
        lines = [
            "Order #%d created." % o.id,
            "Restaurant grid position: (%d, %d)" % o.res_pos,
            "Customer grid position:   (%d, %d)" % o.cus_pos,

"""Command parsing and input validation."""

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "q": "quit", "exit": "quit", "m": "map", "t": "tick", "h": "help",
    "?": "help", "hist": "history", "!!": "repeat",
}

COMMANDS = ("start", "status", "orders", "riders", "quit", "map", "tick",
            "run", "help", "history", "save", "repeat")

MIN_COORD = 0
MAX_COORD = 1000


class InputManager:
    """Parses commands, resolves shortcuts and keeps a command history."""

    def __init__(self):
        self.history = []

    def parse(self, line):
        """Return (command, args). Unknown commands are returned as-is."""
        tokens = (line or "").strip().split()
        if not tokens:
            return "", []
        cmd = tokens[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        return cmd, tokens[1:]

    def record(self, line):
        line = (line or "").strip()
        if line and not line.startswith("!!"):
            self.history.append(line)
            if len(self.history) > 500:
                del self.history[:-500]

    def last(self):
        return self.history[-1] if self.history else None

    @staticmethod
    def validate_coords(tokens):
        """Validate four integer coordinates.

        Returns (coords, error). ``coords`` is a list of 4 ints clamped to
        the 0-800 map, or None if invalid.
        """
        text = " ".join(tokens).replace(",", " ").replace("[", " ") \
            .replace("]", " ")
        parts = text.split()
        if len(parts) != 4:
            return None, "Please enter exactly 4 integers: rx ry cx cy"
        values = []
        for p in parts:
            try:
                v = int(p)
            except ValueError:
                return None, "Invalid integer: %r" % p
            if v < MIN_COORD or v > MAX_COORD:
                return None, "Coordinate %d out of range (%d-%d)" % (
                    v, MIN_COORD, MAX_COORD)
            values.append(min(v, 800))
        return values, None

    @staticmethod
    def help_text():
        return ("Commands:\n"
                "  start (s) [rx ry cx cy]  create an order\n"
                "  status (st)              system snapshot\n"
                "  orders (o)               all orders\n"
                "  riders (r)               rider event logs\n"
                "  map (m)                  ASCII map\n"
                "  tick (t) [n]             advance n ticks\n"
                "  run                      run until all orders complete\n"
                "  save                     save state to JSON\n"
                "  history (hist), !!       command history / repeat last\n"
                "  quit / exit (q)          exit")

"""Command line entry point of the food delivery dispatch simulator."""

import logging
import sys

try:  # command history / arrow keys where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

import globalvar
from display import DisplayManager
from input_manager import InputManager
from system import init_system


def main(argv=None):
    """Run the interactive main loop. Each loop iteration is one tick."""
    logging.basicConfig(level=logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")
    system = init_system("config.json")
    display = DisplayManager()
    inp = InputManager()
    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
    while globalvar.get_value("running", False):
        try:
            line = input("[t=%d] > " % system.time)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        cmd, args = inp.parse(line)
        if cmd == "repeat":
            last = inp.last()
            if not last:
                print("No previous command.")
                continue
            print(last)
            line = last
            cmd, args = inp.parse(line)
        inp.record(line)
        advance = 1
        try:
            if cmd == "":
                pass
            elif cmd == "quit":
                system.stop()
                advance = 0
            elif cmd == "help":
                print(inp.help_text())
                advance = 0
            elif cmd == "start":
                tokens = args
                if not tokens:
                    tokens = input("Enter restaurant_x restaurant_y "
                                   "customer_x customer_y: ").split()
                coords, err = inp.validate_coords(tokens)
                if err:
                    print("Error:", err)
                    advance = 0
                else:
                    result = system.order_manager.create_order(
                        (coords[0], coords[1]), (coords[2], coords[3]))
                    print(display.start_result(result))
            elif cmd == "status":
                print(display.status())
            elif cmd == "orders":
                print(display.orders())
            elif cmd == "riders":
                print(display.riders_log())
            elif cmd == "map":
                print(display.render_map())
            elif cmd == "tick":
                n = 1
                if args:
                    try:
                        n = max(0, int(args[0]))
                    except ValueError:
                        print("Error: tick count must be an integer")
                        n = 0
                system.tick(n)
                advance = 0
                print("Time is now %d" % system.time)
            elif cmd == "run":
                steps = 0
                while not system.all_done() and steps < 100000:
                    system.tick()
                    steps += 1
                advance = 0
                print("Ran %d ticks. Time is now %d" % (steps, system.time))
            elif cmd == "history":
                for i, h in enumerate(inp.history, 1):
                    print("%3d  %s" % (i, h))
                advance = 0
            elif cmd == "save":
                print("Saved to", system.save())
                advance = 0
            else:
                print("Unknown command %r. Type 'help'." % cmd)
                advance = 0
            if advance:
                system.tick(advance)
            problems = system.consistency_check()
            if problems:
                logging.warning("Consistency issues: %s", "; ".join(problems))
        except (EOFError, KeyboardInterrupt):
            break
        except Exception as exc:  # graceful recovery
            logging.exception("Unexpected error: %s", exc)
            print("An error occurred; the system state has been preserved.")
    try:
        system.save()
    except OSError as exc:
        logging.warning("Could not save state: %s", exc)
    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Global variable registry shared by all modules of the dispatch system.

The registry is a plain dictionary hidden behind ``set_value``/``get_value``.
Well-known keys:

* ``"riders"``            -- list[Rider]
* ``"orders"``            -- list[Order] (order history)
* ``"funds"``             -- int, system funds (may be negative)
* ``"rider_id_counter"``  -- int, last rider id handed out
* ``"order_id_counter"``  -- int, last order id handed out
* ``"running"``           -- bool, main loop flag
* ``"time"``              -- int, current simulated tick
"""

_global_dict = None


def _init():
    """(Re)initialise the global registry to an empty dictionary."""
    global _global_dict
    _global_dict = {}


def _ensure():
    """Lazily initialise the registry if ``_init`` was never called."""
    if _global_dict is None:
        _init()


def set_value(key, value):
    """Store ``value`` under ``key``."""
    _ensure()
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` (``default`` if missing)."""
    _ensure()
    return _global_dict.get(key, default)


def has_key(key):
    """Return True if ``key`` is present in the registry."""
    _ensure()
    return key in _global_dict

"""Grid map and coordinate conversion utilities.

The map is 800x800 units divided into an 8x8 grid of 100x100 cells.
Grid lines (x or y a multiple of 100) are streets.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE


def clamp_coord(value):
    """Clamp a coordinate into the valid range 0..MAP_SIZE."""
    return max(0, min(MAP_SIZE, int(value)))


def to_grid_index(value):
    """Convert a raw coordinate into a grid index 0..GRID_DIM-1.

    The boundary value 800 belongs to the last cell (index 7).
    """
    value = clamp_coord(value)
    return min(value // CELL_SIZE, GRID_DIM - 1)


def to_cell_center(value):
    """Convert a raw coordinate into the coordinate of its cell center.

    ``center = (v // 100) * 100 + 50`` with boundary handling for 800.
    """
    return to_grid_index(value) * CELL_SIZE + CELL_SIZE // 2


def to_cell_center_point(pos):
    """Convert an ``(x, y)`` point to its cell center point."""
    return (to_cell_center(pos[0]), to_cell_center(pos[1]))


def to_grid_point(pos):
    """Convert an ``(x, y)`` point to its ``(grid_x, grid_y)`` indices."""
    return (to_grid_index(pos[0]), to_grid_index(pos[1]))


def is_street(x, y):
    """Return True if the point lies on a street (a grid line)."""
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


class GridCell:
    """A single 100x100 cell of the map."""

    def __init__(self, grid_x, grid_y, symbol="."):
        self.grid_x = grid_x
        self.grid_y = grid_y
        self.center_x = grid_x * CELL_SIZE + CELL_SIZE // 2
        self.center_y = grid_y * CELL_SIZE + CELL_SIZE // 2
        self.symbol = symbol

    def __repr__(self):
        return "GridCell(%d, %d)" % (self.grid_x, self.grid_y)


class GridMap:
    """The 8x8 grid map containing 64 cells."""

    def __init__(self, width=MAP_SIZE, height=MAP_SIZE, cell_size=CELL_SIZE):
        self.width = width
        self.height = height
        self.cell_size = cell_size
        self.grid_dimension = width // cell_size
        self.cells = [[GridCell(gx, gy) for gx in range(self.grid_dimension)]
                      for gy in range(self.grid_dimension)]

    def cell_at(self, x, y):
        """Return the cell containing raw coordinate ``(x, y)``."""
        return self.cells[to_grid_index(y)][to_grid_index(x)]

    def all_cells(self):
        """Return a flat list of all cells."""
        return [c for row in self.cells for c in row]

"""Grid-based path planning.

Routes prefer streets (grid lines).  A route from a start point to a goal
cell center is built as:

1. a short vertical hop onto the horizontal street nearest the start
   (in the direction of the goal),
2. a horizontal run along that street to the vertical street bordering the
   goal cell (on the side facing the start),
3. a vertical run along that street to the goal's y coordinate,
4. a short horizontal hop into the goal cell center.

When the rider is directly above or below the target (same x), the route
leaves the block sideways onto the neighbouring street, travels vertically
and re-enters -- this is flagged as a *special case*.

Riders never block each other (streets have unlimited capacity), so path
conflicts and deadlocks cannot occur.
"""

from grid_map import CELL_SIZE, MAP_SIZE


class Waypoint:
    """A point along a route."""

    def __init__(self, x, y, sequence):
        self.x = x
        self.y = y
        self.sequence = sequence

    def __repr__(self):
        return "Waypoint(%d, %d, #%d)" % (self.x, self.y, self.sequence)


def manhattan(a, b):
    """Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class Route:
    """A polyline route made of axis-aligned segments."""

    def __init__(self, points, special_case=False):
        self.points = [tuple(p) for p in points]
        self.special_case = special_case

    @property
    def total_distance(self):
        """Total length of the route in map units."""
        return sum(manhattan(self.points[i], self.points[i + 1])
                   for i in range(len(self.points) - 1))

    @property
    def waypoints(self):
        """Route points as :class:`Waypoint` objects."""
        return [Waypoint(p[0], p[1], i) for i, p in enumerate(self.points)]

    def estimated_time(self, speed):
        """Estimated ticks needed at ``speed`` units/tick (ceiling)."""
        if speed <= 0:
            return 0
        return -(-self.total_distance // speed)

    def extend(self, other):
        """Append another route that starts where this one ends."""
        pts = other.points
        if self.points and pts and self.points[-1] == pts[0]:
            pts = pts[1:]
        self.points = _simplify(self.points + list(pts))
        self.special_case = self.special_case or other.special_case
        return self

    def __repr__(self):
        return "Route(%s, dist=%d)" % (self.points, self.total_distance)


def _simplify(points):
    """Remove duplicate consecutive points and collinear middle points."""
    out = []
    for p in points:
        if out and out[-1] == p:
            continue
        out.append(p)
    i = 1
    while i < len(out) - 1:
        a, b, c = out[i - 1], out[i], out[i + 1]
        if (a[0] == b[0] == c[0]) or (a[1] == b[1] == c[1]):
            # Only collapse if b lies between a and c (no U-turn).
            if min(a[0], c[0]) <= b[0] <= max(a[0], c[0]) and \
                    min(a[1], c[1]) <= b[1] <= max(a[1], c[1]):
                out.pop(i)
                continue
        i += 1
    return out


class PathPlanner:
    """Street-preferring path planner on the 8x8 grid."""

    def __init__(self, cell_size=CELL_SIZE, map_size=MAP_SIZE):
        self.cell_size = cell_size
        self.map_size = map_size

    def _line_toward(self, value, target):
        """Nearest grid line to ``value`` lying toward ``target``."""
        cs = self.cell_size
        if value % cs == 0:
            return value
        lo = (value // cs) * cs
        hi = min(lo + cs, self.map_size)
        return hi if target > value else lo

    def plan(self, start, goal):
        """Plan a route from ``start`` to ``goal``; returns a :class:`Route`."""
        sx, sy = int(start[0]), int(start[1])
        gx, gy = int(goal[0]), int(goal[1])
        if (sx, sy) == (gx, gy):
            return Route([(sx, sy)], False)
        special = (sx == gx)
        hy = self._line_toward(sy, gy)
        vx = self._line_toward(gx, sx)
        points = [(sx, sy), (sx, hy), (vx, hy), (vx, gy), (gx, gy)]
        return Route(_simplify(points), special)

    def plan_multi(self, points):
        """Plan a continuous route through several destinations."""
        points = [tuple(p) for p in points]
        if not points:
            return Route([], False)
        route = Route([points[0]], False)
        for i in range(len(points) - 1):
            route.extend(self.plan(points[i], points[i + 1]))
        return route

    def distance(self, start, goal):
        """Street distance between two points."""
        return self.plan(start, goal).total_distance

"""Rider model: movement, state machine and status log."""

import logging
from enum import IntEnum

import globalvar
from grid_map import clamp_coord
from path_planner import PathPlanner, manhattan

logger = logging.getLogger("dispatch.rider")


class RiderState(IntEnum):
    """Rider state enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2


STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``.  Each call to
    :meth:`move` advances the rider by one tick; state transitions and log
    entries happen there.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        start = globalvar.get_value("rider_start", (400, 400))
        self.rider_id = rider_id
        self.x = clamp_coord(start[0] if x is None else x)
        self.y = clamp_coord(start[1] if y is None else y)
        self.speed = int(speed if speed is not None
                         else globalvar.get_value("rider_speed", 50))
        if self.speed <= 0:
            self.speed = 50
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.current_order = None
        self.path = []
        self.current_route = None
        self.completed_orders = []
        self.distance_travelled = 0
        self._planner = PathPlanner()
        self._log("spawn", None)

    # ------------------------------------------------------------------ helpers
    def _now(self):
        return globalvar.get_value("time", 0)

    def _log(self, event, order):
        """Append a status record to the rider log."""
        self.log.append({
            "time": self._now(),
            "event": event,
            "order_id": getattr(order, "id", None) if order is not None else None,
            "x": self.x,
            "y": self.y,
            "state": int(self.state),
        })
        limit = globalvar.get_value("max_log_entries", 5000)
        if limit and len(self.log) > limit:
            del self.log[:len(self.log) - limit]
        logger.debug("Rider %s %s order=%s at (%d,%d)", self.rider_id, event,
                     self.log[-1]["order_id"], self.x, self.y)

    def pending_orders(self):
        """Orders assigned to this rider that are not completed."""
        return [o for o in self.orders if getattr(o, "status", "") != "completed"]

    def _set_route(self, goal):
        route = self._planner.plan((self.x, self.y), goal)
        self.current_route = route
        self.path = list(route.points[1:])
        self.special_case = route.special_case

    def _advance(self, budget):
        """Move along ``self.path`` using up to ``budget`` units."""
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            dx, dy = tx - self.x, ty - self.y
            dist = abs(dx) + abs(dy)
            if dist == 0:
                self.path.pop(0)
                continue
            if dist <= budget:
                self.x, self.y = tx, ty
                budget -= dist
                self.distance_travelled += dist
                self.path.pop(0)
            else:
                step = budget
                if dx:
                    s = min(step, abs(dx))
                    self.x += s if dx > 0 else -s
                    step -= s
                if step and dy:
                    s = min(step, abs(dy))
                    self.y += s if dy > 0 else -s
                self.distance_travelled += budget
                budget = 0

    def _at(self, pos):
        return not self.path and (self.x, self.y) == (int(pos[0]), int(pos[1]))

    # --------------------------------------------------------------- simulation
    def move(self):
        """Advance the rider by one tick."""
        if self.state != RiderState.IDLE and (
                self.current_order is None
                or self.current_order.status == "completed"):
            self.reset()
        if self.state == RiderState.IDLE:
            pending = self.pending_orders()
            if not pending:
                return
            order = pending[0]
            self.current_order = order
            self.state = RiderState.PICKING_UP
            order.status = "picking_up"
            if getattr(order, "rider_id", None) is None:
                order.rider_id = self.rider_id
            order.accept_time = self._now()
            self._log("accepted_order", order)
            self._set_route(order.res_pos)

        self._advance(self.speed)
        order = self.current_order
        if self.state == RiderState.PICKING_UP and self._at(order.res_pos):
            self.state = RiderState.DELIVERING
            order.status = "delivering"
            order.pickup_time = self._now()
            self._log("arrived_at_merchant", order)
            self._set_route(order.cus_pos)
        elif self.state == RiderState.DELIVERING and self._at(order.cus_pos):
            self._complete(order)

    def _complete(self, order):
        now = self._now()
        order.status = "completed"
        order.completion_time = now
        start = getattr(order, "start_time", None)
        order.time_taken = now - start if start is not None else None
        self.state = RiderState.IDLE
        self._log("delivered_to_customer", order)
        if order in self.orders:
            self.orders.remove(order)
        self.completed_orders.append(order.id)
        if len(self.completed_orders) > 10000:
            del self.completed_orders[:-10000]
        self.current_order = None
        self.path = []
        self.current_route = None
        self.special_case = False
        funds = globalvar.get_value("funds")
        if funds is not None:
            globalvar.set_value("funds", funds + int(getattr(order, "fee", 0)))

    def reset(self):
        """Recover to a consistent idle state (used for error recovery)."""
        self.state = RiderState.IDLE
        self.current_order = None
        self.path = []
        self.current_route = None
        self.special_case = False
        self.x = clamp_coord(self.x)
        self.y = clamp_coord(self.y)

    # ---------------------------------------------------------------- estimates
    def remaining_route_distance(self):
        """Distance left on the current path."""
        pos = (self.x, self.y)
        dist = 0
        for p in self.path:
            dist += manhattan(pos, p)
            pos = p
        return dist

    def estimate_workload(self):
        """Return ``(remaining_distance, final_position)`` for queued work."""
        pos = (self.x, self.y)
        dist = 0
        cur = self.current_order
        if cur is not None and cur.status != "completed":
            if self.state == RiderState.PICKING_UP:
                dist += self.remaining_route_distance()
                pos = tuple(cur.res_pos)
                dist += manhattan(pos, cur.cus_pos)
                pos = tuple(cur.cus_pos)
            elif self.state == RiderState.DELIVERING:
                dist += self.remaining_route_distance()
                pos = tuple(cur.cus_pos)
        for o in self.pending_orders():
            if o is cur:
                continue
            dist += manhattan(pos, o.res_pos) + manhattan(o.res_pos, o.cus_pos)
            pos = tuple(o.cus_pos)
        return dist, pos

    @property
    def state_name(self):
        return STATE_NAMES.get(int(self.state), "unknown")

    def to_dict(self):
        """Serializable representation."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": int(self.state),
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "completed_orders": list(self.completed_orders),
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%d,%d), state=%s)" % (
            self.rider_id, self.x, self.y, self.state_name)

import globalvar
from grid_map import to_cell_center_point, to_grid_point
from path_planner import PathPlanner


class Order:
    """A delivery order from a restaurant to a customer.

    Input coordinates are converted to grid cell centers.
    Status lifecycle: pending -> assigned -> picking_up -> delivering -> completed
    """

    def __init__(self, res_pos, cus_pos, start_time=None):
        if len(res_pos) != 2 or len(cus_pos) != 2:
            raise ValueError("positions must be (x, y) pairs")
        counter = globalvar.get_value("order_id_counter", 0) + 1
        globalvar.set_value("order_id_counter", counter)
        self.id = counter
        self.raw_res = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = to_cell_center_point(self.raw_res)
        self.cus_pos = to_cell_center_point(self.raw_cus)
        self.res_grid = to_grid_point(self.raw_res)
        self.cus_grid = to_grid_point(self.raw_cus)
        self.start_time = globalvar.get_value("time", 0) if start_time is None else start_time
        self.status = "pending"
        self.rider_id = None
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.distance = PathPlanner().distance(self.res_pos, self.cus_pos)
        base = globalvar.get_value("base_fee", 30)
        per = globalvar.get_value("fee_per_100_units", 10)
        self.fee = int(base + (self.distance // 100) * per)

    @property
    def restaurant_x(self):
        return self.res_pos[0]

    @property
    def restaurant_y(self):
        return self.res_pos[1]

    @property
    def customer_x(self):
        return self.cus_pos[0]

    @property
    def customer_y(self):
        return self.cus_pos[1]

    def assign_to(self, rider_id):
        """Mark the order as assigned to ``rider_id``."""
        self.rider_id = rider_id
        self.status = "assigned"
        self.assign_time = globalvar.get_value("time", 0)

    def to_dict(self):
        """Serializable representation."""
        return {
            "id": self.id,
            "raw_restaurant": list(self.raw_res),
            "raw_customer": list(self.raw_cus),
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "start_time": self.start_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "rider_id": self.rider_id,
            "status": self.status,
            "fee": self.fee,
        }

    def __repr__(self):
        return "Order(#%d %s->%s %s)" % (self.id, self.res_pos, self.cus_pos, self.status)

"""Order management: creation, assignment and history."""

import logging

import globalvar
from order_model import Order
from path_planner import manhattan

logger = logging.getLogger("dispatch.order")


def find_best_rider_for_order(order, riders=None):
    """Return the best rider for ``order`` (closest rider first).

    For an idle rider the cost is its Manhattan distance to the restaurant.
    For a busy rider the cost is its remaining queued work plus the distance
    from its final drop-off to the restaurant, which balances load among
    competing riders.  Ties are broken by fewer pending orders, then lower id.
    Returns ``None`` if there are no riders.
    """
    if riders is None:
        riders = globalvar.get_value("riders") or []
    best, best_key = None, None
    for rider in riders:
        work, end = rider.estimate_workload()
        cost = work + manhattan(end, order.res_pos)
        key = (cost, len(rider.pending_orders()), rider.rider_id)
        if best_key is None or key < best_key:
            best, best_key = rider, key
    return best


class OrderManager:
    """Handles order creation, assignment and completion records."""

    def __init__(self, assignment_strategy="nearest"):
        self.assignment_strategy = assignment_strategy
        if globalvar.get_value("orders") is None:
            globalvar.set_value("orders", [])

    def _orders(self):
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create and register an order from ``[rx, ry, cx, cy]``."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        orders = self._orders()
        orders.append(order)
        limit = globalvar.get_value("max_order_history", 10000)
        if limit and len(orders) > limit:
            for o in list(orders):
                if len(orders) <= limit:
                    break
                if o.status == "completed":
                    orders.remove(o)
        logger.info("Created order %d", order.id)
        return order

    def assign_to_rider(self, order, rider):
        """Assign ``order`` to ``rider``."""
        order.assign_to(rider.rider_id)
        rider.orders.append(order)

    def assign_order(self, order):
        """Assign ``order`` to the best rider; returns the rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is not None:
            self.assign_to_rider(order, rider)
        return rider

    def get_orders(self):
        return list(self._orders())

    def get_order(self, order_id):
        for o in self._orders():
            if o.id == order_id:
                return o
        return None

    def completed_orders(self):
        return [o for o in self._orders() if o.status == "completed"]

"""JSON configuration and data persistence."""

import json
import logging
import os

logger = logging.getLogger("dispatch.config")

DEFAULTS = {
    "initial_funds": 1000,
    "recruitment_threshold": 300,
    "rider_cost": 300,
    "rider_speed": 50,
    "rider_start": [400, 400],
    "max_riders": 20,
    "base_fee": 30,
    "fee_per_100_units": 10,
    "max_order_history": 10000,
    "max_log_entries": 5000,
    "data_file": "dispatch_data.json",
}


class Configuration:
    """System parameters, optionally loaded from a JSON file."""

    def __init__(self, file_path=None):
        self.file_path = file_path
        self.parameters = dict(DEFAULTS)
        if file_path and os.path.exists(file_path):
            self.load()

    def load(self):
        """Load parameters from ``file_path``; invalid files are ignored."""
        try:
            with open(self.file_path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            params = data.get("parameters", data) if isinstance(data, dict) else {}
            for key, value in params.items():
                if key in DEFAULTS:
                    self.parameters[key] = value
        except (OSError, ValueError) as exc:
            logger.warning("Could not load config %s: %s", self.file_path, exc)

    def save(self, path=None):
        """Save parameters to JSON."""
        path = path or self.file_path
        if not path:
            return False
        with open(path, "w", encoding="utf-8") as fh:
            json.dump({"parameters": self.parameters}, fh, indent=2)
        return True

    def get(self, key, default=None):
        return self.parameters.get(key, DEFAULTS.get(key, default))

    @staticmethod
    def save_data(path, data):
        """Persist a data snapshot (system, riders, orders) as JSON."""
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return True

class SystemManager:

    def __init__(self, configuration=None):

        self.path_planner = PathPlanner()

    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def riders(self):
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def orders(self):
        return self.order_manager.get_orders()

    @property
    def running(self):
        return globalvar.get_value("running", False)

    @property
    def current_time(self):
        return globalvar.get_value("time", 0)

    @property
    def rider_id_counter(self):
        return globalvar.get_value("rider_id_counter", 0)

    @property
    def recruitment_threshold(self):
        return globalvar.get_value("recruitment_threshold", 300)

    # ------------------------------------------------------------------ riders
    def recruit_rider(self):
        """Create a new rider, paying the rider cost."""
        new_id = self.rider_id_counter + 1
        globalvar.set_value("rider_id_counter", new_id)
        rider = Rider(new_id)
        self.riders.append(rider)
        self.funds = self.funds - int(self.config.get("rider_cost"))
        logger.info("Recruited rider %d", new_id)
        return rider

    def auto_recruit(self):
        """Recruit a rider if funds reach the threshold; returns it or None."""
        if self.funds >= self.recruitment_threshold and \
                len(self.riders) < int(self.config.get("max_riders")):
            return self.recruit_rider()
        return None

    def get_rider(self, rider_id):
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    # ------------------------------------------------------------------ orders
    def submit_order(self, coords):
        """Create and assign an order from four coordinates.

        Returns a dict with keys ``order``, ``rider``, ``recruited``,
        ``route``, ``eta`` and ``busy``.
        """
        coords = InputManager.validate_coordinates(coords)
        recruited = self.auto_recruit()
        if not self.riders:
            recruited = self.recruit_rider()
        order = self.order_manager.create_order(coords)
        rider = find_best_rider_for_order(order, self.riders)
        _, start = rider.estimate_workload()
        busy = bool(rider.pending_orders()) or rider.state != RiderState.IDLE
        route = self.path_planner.plan_multi([start, order.res_pos, order.cus_pos])
        self.order_manager.assign_to_rider(order, rider)
        return {
            "order": order,
            "rider": rider,
            "recruited": recruited,
            "route": route,
            "eta": route.estimated_time(rider.speed),
            "busy": busy,
        }

    # -------------------------------------------------------------- simulation
    def tick(self, n=1):
        """Advance simulated time by ``n`` ticks."""
        for _ in range(max(0, int(n))):
            globalvar.set_value("time", self.current_time + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception:  # pragma: no cover - defensive recovery
                    logger.exception("Rider %s failed to move; recovering", rider.rider_id)
                    rider.reset()

    def check_consistency(self):
        """Repair inconsistent state; returns a list of fix descriptions."""
        fixes = []
        for rider in self.riders:
            if int(rider.state) not in (0, 1, 2):
                rider.reset()
                fixes.append("Rider %s had invalid state; reset" % rider.rider_id)
            elif rider.state != RiderState.IDLE and (
                    rider.current_order is None
                    or rider.current_order.status == "completed"):
                rider.reset()
                fixes.append("Rider %s busy without order; reset" % rider.rider_id)
            if rider.x != clamp_coord(rider.x) or rider.y != clamp_coord(rider.y):
                rider.x, rider.y = clamp_coord(rider.x), clamp_coord(rider.y)
                fixes.append("Rider %s position clamped" % rider.rider_id)
        ids = {r.rider_id for r in self.riders}
        for order in self.order_manager.get_orders():
            if order.status == "completed":
                continue
            if order.rider_id not in ids:
                if not self.riders:
                    self.recruit_rider()
                rider = find_best_rider_for_order(order, self.riders)
                self.order_manager.assign_to_rider(order, rider)
                fixes.append("Order %d reassigned to rider %s" % (order.id, rider.rider_id))
        return fixes

    def stop(self):
        globalvar.set_value("running", False)

    def snapshot(self):
        """Return a JSON-serializable snapshot of the system."""
        return {
            "parameters": self.config.parameters,
            "system": {
                "funds": self.funds,
                "time": self.current_time,
                "rider_id_counter": self.rider_id_counter,
            },
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.get_orders()],
        }

    def save_data(self, path=None):
        """Persist the snapshot to a JSON file."""
        path = path or self.config.get("data_file")
        return Configuration.save_data(path, self.snapshot())

def _idx(v):
    return max(0, min(SIZE - 1, int(round(v / UNIT))))


class DisplayManager:
    """Formats system information for the command line."""

    def render_map(self, riders=(), orders=(), route=None):
        """Return an ASCII map.

        Even rows/columns are streets ('#'), odd/odd positions are blocks
        ('.'). Routes are drawn with '-', '|' and '+'; 'S' restaurant,
        'C' customer, 'R' rider.
        """
        grid = [["#" if (r % 2 == 0 or c % 2 == 0) else "." for c in range(SIZE)]
                for r in range(SIZE)]
        if route is not None and len(route.points) > 1:
            pts = route.points
            for a, b in zip(pts, pts[1:]):
                ra, ca, rb, cb = _idx(a[1]), _idx(a[0]), _idx(b[1]), _idx(b[0])
                if ra == rb:
                    for c in range(min(ca, cb), max(ca, cb) + 1):
                        grid[ra][c] = "-"
                else:
                    for r in range(min(ra, rb), max(ra, rb) + 1):
                        grid[r][ca] = "|"
            for p in pts[1:-1]:
                grid[_idx(p[1])][_idx(p[0])] = "+"
        for o in orders:
            grid[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = "C"
        for o in orders:
            grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = "S"
        for rd in riders:
            grid[_idx(rd.y)][_idx(rd.x)] = "R"
        header = "    " + " ".join(str(i // 2) if i % 2 else " " for i in range(SIZE))
        lines = [header]
        for r in range(SIZE):
            label = str(r // 2) if r % 2 else " "
            lines.append(" %s  %s" % (label, " ".join(grid[r])))
        lines.append("Legend: R=Rider S=Restaurant C=Customer #=Street .=Block "
                     "-|+=Route (grid lines every 100 units)")
        return "\n".join(lines)

class InputError(ValueError):
    """Raised for invalid user input."""


COMMAND_ALIASES = {
    "start": "start", "n": "start", "new": "start", "1": "start",
    "status": "status", "s": "status", "2": "status",
    "orders": "orders", "o": "orders", "3": "orders",
    "riders": "riders", "r": "riders", "4": "riders",
    "map": "map", "m": "map", "5": "map",
    "tick": "tick", "t": "tick", "6": "tick",
    "history": "history", "h": "history",
    "save": "save",
    "check": "check",
    "help": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}


class InputManager:
    """Parses commands, keeps command history, validates input."""

    def __init__(self, max_history=500):
        self.command_history = []
        self.max_history = max_history

    def read_line(self, prompt):
        """Read a line from stdin; EOF/interrupt becomes ``quit``."""
        try:
            return input(prompt)
        except (EOFError, KeyboardInterrupt):
            print()
            return "quit"

    def record(self, raw):
        """Add a command to history."""
        self.command_history.append(raw)
        if len(self.command_history) > self.max_history:
            del self.command_history[:-self.max_history]

    def resolve_history(self, raw):
        """Expand ``!!`` and ``!n`` history references."""
        raw = raw.strip()
        if raw == "!!":
            if not self.command_history:
                raise InputError("No command history.")
            return self.command_history[-1]
        if raw.startswith("!"):
            try:
                n = int(raw[1:])
                return self.command_history[n - 1]
            except (ValueError, IndexError):
                raise InputError("Invalid history reference: %s" % raw)
        return raw

    def parse(self, raw):
        """Parse a raw line into ``(command, args)``.

        Unknown commands yield ``(None, args)``; empty lines ``("", [])``.
        """
        raw = self.resolve_history(raw)
        tokens = tokenize(raw)
        if not tokens:
            return "", []
        self.record(raw)
        return COMMAND_ALIASES.get(tokens[0].lower()), tokens[1:]

    @staticmethod
    def validate_coordinates(tokens):
        """Validate four integer coordinates in range 0..800."""
        if isinstance(tokens, str):
            tokens = tokenize(tokens)
        tokens = list(tokens)
        if len(tokens) != 4:
            raise InputError("Expected 4 coordinates: restaurant_x restaurant_y "
                             "customer_x customer_y (got %d)." % len(tokens))
        values = []
        for t in tokens:
            if isinstance(t, bool):
                raise InputError("Coordinate %r is not an integer." % (t,))
            if isinstance(t, int):
                v = t
            else:
                try:
                    v = int(str(t).strip())
                except ValueError:
                    raise InputError("Coordinate %r is not an integer." % (t,))
            if not 0 <= v <= MAP_SIZE:
                raise InputError("Coordinate %d out of range 0-%d." % (v, MAP_SIZE))
            values.append(v)
        return values


def tokenize(raw):
    """Split a line on whitespace/commas, stripping brackets."""
    for ch in "[](),":
        raw = raw.replace(ch, " ")
    return raw.split()

"""Command-line entry point of the Food Delivery Order Dispatch System."""

import argparse
import logging
import sys

import globalvar
from input_manager import InputError
from system import init_system

try:  # enables arrow-key command history where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

HELP = """Commands (shortcut in brackets):
  start [n|1] rx ry cx cy  Create an order (coords 0-800); prompts if omitted
  status [s|2]             Funds and rider states
  orders [o|3]             All orders
  riders [r|4]             Rider status-change history
  map [m|5]                ASCII map of riders and active orders
  tick [t|6] [N]           Advance simulation N ticks (default 1)
  history [h]              Command history (!! repeat last, !n repeat n-th)
  save [file]              Save system data to JSON
  check                    Run a state consistency check
  help [?]                 This help
  quit / exit [q]          Exit
Every command iteration advances the simulation by one tick."""


def handle(sm, cmd, args):
    """Execute a command; returns True if an automatic tick should follow."""
    dm, im = sm.display_manager, sm.input_manager
    if cmd == "start":
        if not args:
            args = im.read_line("Enter restaurant_x restaurant_y customer_x customer_y: ")
        result = sm.submit_order(args)
        print(dm.format_order_result(result))
    elif cmd == "status":
        print(dm.format_status(sm.funds, sm.current_time, sm.riders))
    elif cmd == "orders":
        print(dm.format_orders(sm.orders))
    elif cmd == "riders":
        print(dm.format_rider_logs(sm.riders))
    elif cmd == "map":
        print(dm.format_map(sm.riders, sm.orders))
    elif cmd == "tick":
        n = 1
        if args:
            try:
                n = int(args[0])
            except ValueError:
                raise InputError("Tick count must be an integer.")
            if not 1 <= n <= 10000:
                raise InputError("Tick count must be between 1 and 10000.")
        sm.tick(n)
        print("Advanced %d tick(s); time is now %d." % (n, sm.current_time))
        return False
    elif cmd == "history":
        for i, c in enumerate(im.command_history, 1):
            print("%4d  %s" % (i, c))
    elif cmd == "save":
        path = args[0] if args else None
        sm.save_data(path)
        print("Data saved to %s." % (path or sm.config.get("data_file")))
    elif cmd == "check":
        fixes = sm.check_consistency()
        print("\n".join(fixes) if fixes else "System state is consistent.")
    elif cmd == "help":
        print(HELP)
    elif cmd == "quit":
        sm.stop()
        return False
    return True


def main(argv=None):
    """Run the interactive dispatch simulator."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", help="JSON configuration file")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    parser.add_argument("--no-save", action="store_true", help="do not save data on exit")
    opts = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if opts.debug else logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")

    sm = init_system(opts.config)
    im = sm.input_manager
    print("Food Delivery Order Dispatch System - 8x8 grid (800x800 units)")
    print("Type 'help' for commands.")
    while globalvar.get_value("running"):
        raw = im.read_line("[t=%d funds=%d] dispatch> " % (sm.current_time, sm.funds))
        try:
            cmd, args = im.parse(raw)
            if cmd == "":
                auto_tick = True
            elif cmd is None:
                print("Unknown command. Type 'help' for the list of commands.")
                auto_tick = False
            else:
                auto_tick = handle(sm, cmd, args)
            if auto_tick:
                sm.tick(1)
        except InputError as exc:
            print("Input error: %s" % exc)
        except Exception as exc:  # graceful recovery
            logging.getLogger("dispatch").exception("Unexpected error")
            print("Error: %s. Recovering system state..." % exc)
            for fix in sm.check_consistency():
                print("  " + fix)

    if not opts.no_save:
        try:
            sm.save_data()
            print("Data saved to %s." % sm.config.get("data_file"))
        except OSError as exc:
            print("Could not save data: %s" % exc)
    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

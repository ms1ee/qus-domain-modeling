"""Grid based path planning.

The map is an 800x800 coordinate system split into 8x8 cells of 100x100.
Lines with ``x % 100 == 0`` or ``y % 100 == 0`` are streets. Restaurants
and customers are located at cell centres (``... 50``). Riders travel
along streets and only leave a street for the final stretch into a
cell centre (and the first stretch out of one).
"""

MAP_SIZE = 800
CELL_SIZE = 100


def _clamp(value, low=0, high=MAP_SIZE):
    return max(low, min(high, value))


def manhattan(a, b):
    """Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class PathPlanner:
    """Street-first path planner with right-angle turns."""

    def __init__(self):
        self.last_special = False

    # ------------------------------------------------------------------ #
    # Coordinate helpers
    # ------------------------------------------------------------------ #
    @staticmethod
    def to_center(value):
        """Convert a raw coordinate to the centre of its grid cell.

        ``center = (v // 100) * 100 + 50``; values are clamped to 0..800
        and the boundary value 800 maps to the last cell (750).
        """
        v = _clamp(int(value))
        center = (v // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2
        if center > MAP_SIZE:
            center = MAP_SIZE - CELL_SIZE // 2
        return center

    @classmethod
    def to_center_point(cls, pos):
        """Convert an (x, y) pair to the centre point of its grid cell."""
        return (cls.to_center(pos[0]), cls.to_center(pos[1]))

    @staticmethod
    def to_cell(pos):
        """Return the (column, row) grid cell index (0..7) for a point."""
        cx = min(int(pos[0]) // CELL_SIZE, MAP_SIZE // CELL_SIZE - 1)
        cy = min(int(pos[1]) // CELL_SIZE, MAP_SIZE // CELL_SIZE - 1)
        return (max(cx, 0), max(cy, 0))

    @staticmethod
    def is_on_street(x, y):
        """Return True if the point lies on a street line."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    # ------------------------------------------------------------------ #
    # Planning
    # ------------------------------------------------------------------ #
    def plan(self, start, target):
        """Plan a path from ``start`` to ``target``.

        Returns a list of waypoints (excluding the start point). Each
        consecutive pair of points differs in only one axis, so the path
        consists of straight segments and right-angle turns.
        ``self.last_special`` is set when the "directly above/below"
        special case required a detour.
        """
        self.last_special = False
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        if (sx, sy) == (tx, ty):
            return []
        pts = []

        # 1. Leave a cell interior by stepping onto a horizontal street.
        if not self.is_on_street(sx, sy):
            base = (sy // CELL_SIZE) * CELL_SIZE
            if ty > sy:
                ny = base + CELL_SIZE
            else:
                ny = base
            ny = _clamp(ny)
            sy = ny
            pts.append((sx, sy))

        # 2. Determine the horizontal street used to enter the target.
        if ty % CELL_SIZE == 0:
            ey = ty
        else:
            base = (ty // CELL_SIZE) * CELL_SIZE
            ey = base if sy <= ty else base + CELL_SIZE
            ey = _clamp(ey)

        # 3. Travel along streets to (tx, ey).
        if sy == ey and sy % CELL_SIZE == 0:
            pts.append((tx, ey))
        else:
            if sx % CELL_SIZE == 0:
                vx = sx
            else:
                # On a horizontal street but not on a vertical one.
                low = _clamp((sx // CELL_SIZE) * CELL_SIZE)
                high = _clamp((sx // CELL_SIZE + 1) * CELL_SIZE)
                vx = min((low, high),
                         key=lambda c: (abs(sx - c) + abs(c - tx), c))
                if sx == tx:
                    # Directly above/below the target: detour required.
                    self.last_special = True
                pts.append((vx, sy))
            pts.append((vx, ey))
            pts.append((tx, ey))

        # 4. Final stretch into the target.
        pts.append((tx, ty))

        # Remove duplicates / zero-length segments.
        cleaned = []
        prev = (int(start[0]), int(start[1]))
        for p in pts:
            if p != prev:
                cleaned.append(p)
                prev = p
        return cleaned

    def plan_route(self, start, targets):
        """Plan a continuous multi-destination route.

        Returns the full list of points including the start point.
        """
        route = [(int(start[0]), int(start[1]))]
        pos = route[0]
        for t in targets:
            pts = self.plan(pos, t)
            route.extend(pts)
            pos = (int(t[0]), int(t[1]))
        return route

    @staticmethod
    def path_length(start, points):
        """Length of the polyline ``start -> points...``."""
        total = 0
        prev = start
        for p in points:
            total += manhattan(prev, p)
            prev = p
        return total

    @staticmethod
    def route_length(route):
        """Length of a full route (list of points including the start)."""
        if not route:
            return 0
        return PathPlanner.path_length(route[0], route[1:])

"""Order data model."""

import globalvar
from path_planner import PathPlanner, manhattan


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique order number.
        raw: the raw user input ``[rx, ry, cx, cy]``.
        res_pos / cus_pos: converted grid-centre positions.
        status: "pending", "assigned", "picked_up" or "completed".
        rider_id: id of the assigned rider (or None).
        start_time / assign_time / accept_time / pickup_time /
        complete_time: simulation ticks of the lifecycle events.
        distance: Manhattan distance restaurant -> customer.
        route_distance: street route length restaurant -> customer.
        travel_distance: distance actually travelled by the rider.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.raw = [int(res_pos[0]), int(res_pos[1]),
                    int(cus_pos[0]), int(cus_pos[1])]
        self.res_pos = PathPlanner.to_center_point(res_pos)
        self.cus_pos = PathPlanner.to_center_point(cus_pos)
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0)
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.complete_time = None
        self.distance = manhattan(self.res_pos, self.cus_pos)
        planner = PathPlanner()
        self.route_distance = planner.path_length(
            self.res_pos, planner.plan(self.res_pos, self.cus_pos))
        self.travel_distance = 0

    def assign_to(self, rider_id):
        """Assign the order to the rider with ``rider_id``."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        self.assign_time = globalvar.get_value("time", 0)

    def mark_picked_up(self, time):
        """Record the pickup at the restaurant."""
        self.status = "picked_up"
        self.pickup_time = time

    def mark_completed(self, time):
        """Record the delivery to the customer."""
        self.status = "completed"
        self.complete_time = time

    @property
    def time_taken(self):
        """Ticks from creation to completion (None if not completed)."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.start_time

    @property
    def efficiency(self):
        """Route distance / travelled distance ratio (1.0 is ideal)."""
        if self.status != "completed" or not self.travel_distance:
            return None
        total = self.route_distance + 0.0
        return round(total / self.travel_distance, 3) if total else None

    def to_dict(self):
        """Serialisable representation."""
        return {
            "id": self.id,
            "raw": self.raw,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "assign_time": self.assign_time,
            "accept_time": self.accept_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
            "route_distance": self.route_distance,
            "travel_distance": self.travel_distance,
            "time_taken": self.time_taken,
        }

    def __repr__(self):
        return (f"Order(id={self.id}, res={self.res_pos}, "
                f"cus={self.cus_pos}, status={self.status})")

"""Rider model: movement, state machine and order handling."""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up",
               DELIVERING: "delivering"}

# Starting positions (all on road intersections).
SPAWN_POINTS = [(400, 400), (0, 0), (800, 0), (0, 800), (800, 800),
                (400, 0), (0, 400), (800, 400), (400, 800), (200, 200),
                (600, 600), (200, 600), (600, 200)]

DEFAULT_SPEED = 50


class Rider:
    """A delivery rider.

    Attributes:
        rider_id: unique id.
        x, y: current position.
        speed: units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: assigned orders (first element is the active one).
        special_case: True while executing a special-case detour.
        log: list of event dicts (keys include "event" and "order_id").
    """

    def __init__(self, rider_id):
        self.rider_id = rider_id
        sx, sy = SPAWN_POINTS[(rider_id - 1) % len(SPAWN_POINTS)]
        self.x = sx
        self.y = sy
        config = globalvar.get_value("config") or {}
        try:
            self.speed = int(config.get("rider_speed", DEFAULT_SPEED))
        except (TypeError, ValueError):
            self.speed = DEFAULT_SPEED
        if self.speed <= 0:
            self.speed = DEFAULT_SPEED
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.path = []
        self.total_distance = 0
        self.completed_orders = 0
        self._plan_key = None
        self._planner = PathPlanner()
        self.log = []
        self._add_log("spawn", None)

    # ------------------------------------------------------------------ #
    def _now(self):
        return globalvar.get_value("time", 0)

    def _add_log(self, event, order_id):
        entry = {
            "time": self._now(),
            "event": event,
            "order_id": order_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)

    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def position(self):
        """Current (x, y) position."""
        return (self.x, self.y)

    # ------------------------------------------------------------------ #
    def _ensure_plan(self, order):
        target = order.res_pos if self.state == PICKING_UP else order.cus_pos
        key = (order.id, self.state)
        if key != self._plan_key:
            self.path = self._planner.plan((self.x, self.y), target)
            self.special_case = self._planner.last_special
            self._plan_key = key

    def _advance(self, order):
        budget = self.speed
        moved = 0
        while budget > 0 and self.path:
            wx, wy = self.path[0]
            dx = wx - self.x
            dy = wy - self.y
            if dx != 0:
                step = min(budget, abs(dx))
                self.x += step if dx > 0 else -step
                budget -= step
                moved += step
            elif dy != 0:
                step = min(budget, abs(dy))
                self.y += step if dy > 0 else -step
                budget -= step
                moved += step
            if (self.x, self.y) == (wx, wy):
                self.path.pop(0)
        self.total_distance += moved
        if hasattr(order, "travel_distance"):
            order.travel_distance += moved
        if not self.path:
            self.special_case = False

    def move(self):
        """Advance the rider by one tick.

        Handles state transitions (idle -> picking up -> delivering ->
        idle) and records the corresponding log entries.
        """
        if self.state not in (IDLE, PICKING_UP, DELIVERING):
            self.state = IDLE
        if not self.orders:
            if self.state != IDLE:
                self.state = IDLE
            self.path = []
            self._plan_key = None
            self.special_case = False
            return
        order = self.orders[0]
        now = self._now()

        if self.state == IDLE:
            self.state = PICKING_UP
            if getattr(order, "rider_id", None) is None:
                order.assign_to(self.rider_id)
            order.accept_time = now
            self._add_log("accepted_order", order.id)

        self._ensure_plan(order)
        self._advance(order)

        if self.path:
            return

        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.mark_picked_up(now)
            self._add_log("arrived_at_merchant", order.id)
            self._ensure_plan(order)
        elif self.state == DELIVERING:
            order.mark_completed(now)
            self.orders.pop(0)
            self.completed_orders += 1
            self.state = IDLE
            self.path = []
            self._plan_key = None
            self._add_log("delivered_to_customer", order.id)
            config = globalvar.get_value("config") or {}
            fee = config.get("delivery_fee", 0)
            funds = globalvar.get_value("funds")
            if funds is not None and fee:
                globalvar.set_value("funds", funds + fee)

    # ------------------------------------------------------------------ #
    def workload(self):
        """Estimate the final position and remaining distance of all
        currently assigned orders.

        Returns:
            (final_position, remaining_distance)
        """
        pos = (self.x, self.y)
        dist = 0
        for idx, o in enumerate(self.orders):
            if idx == 0 and self.state == DELIVERING:
                legs = [o.cus_pos]
            else:
                legs = [o.res_pos, o.cus_pos]
            for t in legs:
                pts = self._planner.plan(pos, t)
                dist += self._planner.path_length(pos, pts)
                pos = tuple(t)
        return pos, dist

    def to_dict(self):
        """Serialisable representation."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "total_distance": self.total_distance,
            "completed_orders": self.completed_orders,
            "log": list(self.log),
        }

    def __repr__(self):
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state}, orders={len(self.orders)})")

"""System initialisation and resource management."""

import json
import logging
import os

import globalvar

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 10000,
    "rider_cost": 1000,
    "rider_speed": 50,
    "max_riders": 10,
    "delivery_fee": 30,
    "map_size": 800,
    "cell_size": 100,
    "state_file": "dispatch_state.json",
}

CONFIG_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                           "config.json")


def load_config(path=None):
    """Load configuration from a JSON file merged over the defaults."""
    config = dict(DEFAULT_CONFIG)
    path = path or CONFIG_FILE
    try:
        if path and os.path.exists(path):
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                for key, value in data.items():
                    if key in DEFAULT_CONFIG:
                        config[key] = value
    except (OSError, ValueError) as exc:
        logger.warning("Could not load config %s: %s", path, exc)
    return config


def get_config(key):
    """Return a configuration value."""
    cfg = globalvar.get_value("config") or DEFAULT_CONFIG
    return cfg.get(key, DEFAULT_CONFIG.get(key))


def init_system(config_path=None):
    """Initialise all global system parameters."""
    globalvar._init()
    config = load_config(config_path)
    globalvar.set_value("config", config)
    globalvar.set_value("funds", int(config["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("last_rider_added", False)
    return config


def can_recruit():
    """Return True if funds and rider limits allow recruiting a rider."""
    funds = globalvar.get_value("funds", 0) or 0
    riders = globalvar.get_value("riders") or []
    return (funds >= get_config("rider_cost")
            and len(riders) < get_config("max_riders"))


def recruit_rider():
    """Create a new rider, deduct its cost and add it to the rider list."""
    from rider import Rider
    counter = (globalvar.get_value("rider_id_counter", 0) or 0) + 1
    globalvar.set_value("rider_id_counter", counter)
    rider = Rider(counter)
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    riders.append(rider)
    funds = globalvar.get_value("funds", 0) or 0
    globalvar.set_value("funds", funds - get_config("rider_cost"))
    logger.info("Recruited rider %s", counter)
    return rider


class SystemManager:
    """Facade over the global system state."""

    def __init__(self, config_path=None, initialise=True):
        if initialise:
            init_system(config_path)

    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        return globalvar.get_value("riders") or []

    @property
    def orders(self):
        return globalvar.get_value("orders") or []

    @property
    def time(self):
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        return bool(globalvar.get_value("running", False))

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    def recruit(self):
        """Recruit a rider (see :func:`recruit_rider`)."""
        return recruit_rider()

    def tick(self, count=1):
        """Advance the simulation ``count`` ticks."""
        for _ in range(max(0, int(count))):
            globalvar.set_value("time", self.time + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception as exc:  # pragma: no cover - recovery
                    logger.exception("Rider %s failed: %s",
                                     rider.rider_id, exc)
                    self.check_consistency()

    def check_consistency(self):
        """Repair inconsistent states. Returns a list of fixes applied."""
        fixes = []
        for rider in self.riders:
            if rider.state not in (0, 1, 2):
                rider.state = 0
                fixes.append(f"rider {rider.rider_id}: invalid state reset")
            if rider.state != 0 and not rider.orders:
                rider.state = 0
                rider.path = []
                fixes.append(f"rider {rider.rider_id}: busy without orders")
            rider.x = max(0, min(800, int(rider.x)))
            rider.y = max(0, min(800, int(rider.y)))
            completed = [o for o in rider.orders if o.status == "completed"]
            for o in completed:
                rider.orders.remove(o)
                fixes.append(f"rider {rider.rider_id}: removed done order")
        ids = [r.rider_id for r in self.riders]
        if len(ids) != len(set(ids)):
            fixes.append("duplicate rider ids detected")
        return fixes

    def save_state(self, path=None):
        """Persist system parameters, riders and order history as JSON."""
        path = path or get_config("state_file")
        data = {
            "config": globalvar.get_value("config"),
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.orders],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

"""Order management: creation, assignment and completion tracking."""

import logging
import math

import globalvar
import system
from order_model import Order
from path_planner import PathPlanner, manhattan

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Select the most suitable rider for ``order``.

    Strategy:
      1. The idle rider closest (Manhattan distance) to the restaurant.
      2. Otherwise recruit a new rider if funds/limits allow (or if no
         rider exists at all).
      3. Otherwise the busy rider with the smallest estimated finishing
         distance (current workload + distance to the restaurant), which
         balances load between riders.

    Sets the global flag ``last_rider_added``. Does not append the order.
    """
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    globalvar.set_value("last_rider_added", False)
    target = order.res_pos

    idle = [r for r in riders if r.state == 0 and not r.orders]
    if idle:
        return min(idle, key=lambda r: (manhattan((r.x, r.y), target),
                                        r.rider_id))

    if not riders or system.can_recruit():
        rider = system.recruit_rider()
        globalvar.set_value("last_rider_added", True)
        return rider

    def score(r):
        pos, dist = r.workload()
        return (dist + manhattan(pos, target), len(r.orders), r.rider_id)

    return min(riders, key=score)


class OrderManager:
    """Creates orders, assigns them and reports results."""

    def __init__(self):
        self.planner = PathPlanner()

    @property
    def orders(self):
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create and register an order from ``[rx, ry, cx, cy]``."""
        if len(coords) != 4:
            raise ValueError("an order needs exactly 4 coordinates")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        self.orders.append(order)
        return order

    def assign(self, order):
        """Assign ``order`` to the best rider.

        Returns a dict with keys: rider, added, route, distance, eta.
        """
        rider = find_best_rider_for_order(order)
        added = bool(globalvar.get_value("last_rider_added", False))
        start_pos, pre_dist = rider.workload()
        route = self.planner.plan_route(start_pos,
                                        [order.res_pos, order.cus_pos])
        route_len = self.planner.route_length(route)
        total = pre_dist + route_len
        eta = math.ceil(total / rider.speed) if rider.speed else None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        logger.info("Order %s assigned to rider %s", order.id,
                    rider.rider_id)
        return {"rider": rider, "added": added, "route": route,
                "distance": route_len, "total_distance": total,
                "eta": eta}

    def place_order(self, coords):
        """Create and assign an order in one step."""
        order = self.create_order(coords)
        result = self.assign(order)
        result["order"] = order

        return [o for o in self.orders if o.status == "completed"]

    def statistics(self):
        """Simple efficiency statistics for completed orders."""
        done = self.completed_orders()
        if not done:
            return {"completed": 0, "avg_time": None, "avg_efficiency": None}
        times = [o.time_taken for o in done]
        effs = [o.efficiency for o in done if o.efficiency]
        return {
            "completed": len(done),
            "avg_time": round(sum(times) / len(times), 2),
            "avg_efficiency": round(sum(effs) / len(effs), 3) if effs
            else None,
        }

"""Status display and ASCII map visualisation."""

import globalvar
from path_planner import PathPlanner
from rider import STATE_NAMES

STEP = 50
SIZE = 800 // STEP + 1  # 17 points per axis


def _idx(v):
    return max(0, min(SIZE - 1, int(round(v / STEP))))


class DisplayManager:
    """Builds textual views of the simulation state.

    Map legend:
        +  street intersection   -  horizontal street   |  vertical street
        .  block (empty cell)    *  route               M  restaurant
        C  customer              0-9 rider (last digit of the id)
    """

    LEGEND = ("Legend: +/-/| streets  . block  * route  M restaurant  "
              "C customer  digit = rider id")

    def __init__(self, out=print):
        self.out = out

    # ------------------------------------------------------------------ #
    @staticmethod
    def _base_grid():
        grid = []
        for j in range(SIZE):
            row = []
            for i in range(SIZE):
                if i % 2 == 0 and j % 2 == 0:
                    row.append("+")
                elif i % 2 == 0:
                    row.append("|")
                elif j % 2 == 0:
                    row.append("-")
                else:
                    row.append(".")
            grid.append(row)
        return grid

    @staticmethod
    def _draw_route(grid, route):
        for a, b in zip(route, route[1:]):
            ai, aj = _idx(a[0]), _idx(a[1])
            bi, bj = _idx(b[0]), _idx(b[1])
            if aj == bj:
                step = 1 if bi >= ai else -1
                for i in range(ai, bi + step, step):
                    grid[aj][i] = "*"
            elif ai == bi:
                step = 1 if bj >= aj else -1
                for j in range(aj, bj + step, step):
                    grid[j][ai] = "*"
            else:
                grid[aj][ai] = "*"
                grid[bj][bi] = "*"

    def render_map(self, riders=None, orders=None, route=None):
        """Return the ASCII map as a string."""
        riders = riders if riders is not None else (
            globalvar.get_value("riders") or [])
        if orders is None:
            orders = [o for o in (globalvar.get_value("orders") or [])
                      if o.status != "completed"]
        grid = self._base_grid()
        if route:
            self._draw_route(grid, route)
        for o in orders:
            grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = "M"
            grid[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = "C"
        for r in riders:
            grid[_idx(r.y)][_idx(r.x)] = str(r.rider_id % 10)
        lines = ["     " + " ".join(f"{i * STEP // 100}" if i % 2 == 0
                                    else " " for i in range(SIZE)),
                 ]
        for j, row in enumerate(grid):
            label = f"{j * STEP:>3}" if j % 2 == 0 else "   "
            lines.append(f"{label}  " + " ".join(row))
        lines.append("(x axis in hundreds, y axis in units; y grows down)")
        lines.append(self.LEGEND)
        return "\n".join(lines)

    # ------------------------------------------------------------------ #
    def show_map(self, route=None):
        """Print the map."""
        self.out(self.render_map(route=route))

    def show_status(self):
        """Print funds and the state of every rider."""
        riders = globalvar.get_value("riders") or []
        self.out(f"Time: {globalvar.get_value('time', 0)}  "
                 f"Funds: {globalvar.get_value('funds', 0)}  "
                 f"Riders: {len(riders)}")
        if not riders:
            self.out("  (no riders yet)")
        for r in riders:
            ids = [o.id for o in r.orders]
            flag = " [special]" if r.special_case else ""
            self.out(f"  Rider {r.rider_id}: pos=({r.x},{r.y}) "
                     f"speed={r.speed} state={r.state}"
                     f"({STATE_NAMES.get(r.state, '?')}) "
                     f"orders={len(r.orders)} {ids}{flag}")

    def show_orders(self):
        """Print every order with its details."""
        orders = globalvar.get_value("orders") or []
        if not orders:
            self.out("No orders yet.")
            return
        self.out(f"{'ID':>3} {'Start':>5} {'Restaurant':>11} "

        for o in orders:

            self.out(f"{o.id:>3} {o.start_time:>5} "

                     f"{o.route_distance:>5} {taken:>5} {rid:>5}  "

    def show_rider_logs(self):

        for r in riders:

        self.out(f"  Route distance: {result['distance']}  "

"""Command parsing and input validation."""

import re

try:  # command history / line editing when available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - platform dependent
    readline = None

COMMANDS = ("start", "status", "orders", "riders", "map", "tick",
            "save", "history", "help", "quit", "exit")

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "m": "map", "t": "tick", "h": "help", "?": "help",
    "hist": "history", "q": "quit", "x": "exit",
}


class InputManager:
    """Parses user commands and validates coordinates."""

    MIN_COORD = 0
    MAX_COORD = 800

    def __init__(self, reader=input):
        self.reader = reader
        self.history = []

    def parse(self, line):
        """Parse a command line into ``(command, args)``.

        Supports shortcuts, ``!!`` (repeat last) and ``!n`` (repeat the
        n-th history entry). Returns ``(None, [])`` for empty input.
        Raises ValueError for unknown commands.
        """
        line = (line or "").strip()
        if not line:
            return None, []
        if line == "!!":
            if not self.history:
                raise ValueError("history is empty")
            line = self.history[-1]
        elif re.fullmatch(r"!\d+", line):
            n = int(line[1:])
            if n < 1 or n > len(self.history):
                raise ValueError(f"no history entry {n}")
            line = self.history[n - 1]
        tokens = line.split(None, 1)
        cmd = tokens[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        if cmd not in COMMANDS:
            raise ValueError(f"unknown command '{tokens[0]}' "
                             "(type 'help')")
        self.history.append(line)
        args = tokens[1].strip() if len(tokens) > 1 else ""
        return cmd, args

    @classmethod
    def parse_coordinates(cls, text):
        """Parse ``[rx, ry, cx, cy]`` from text.

        Accepts separators of spaces/commas and optional brackets.
        Raises ValueError with a helpful message on invalid input.
        """
        if isinstance(text, (list, tuple)):
            tokens = [str(t) for t in text]
        else:
            cleaned = str(text).strip().strip("[]()")
            tokens = [t for t in re.split(r"[\s,]+", cleaned) if t]
        if len(tokens) != 4:
            raise ValueError("please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x "
                             "customer_y")
        values = []
        for t in tokens:
            try:
                values.append(int(t))
            except ValueError:
                raise ValueError(f"'{t}' is not an integer") from None
        for v in values:
            if v < cls.MIN_COORD or v > cls.MAX_COORD:
                raise ValueError(f"coordinate {v} out of range "
                                 f"{cls.MIN_COORD}-{cls.MAX_COORD}")
        return values

    @staticmethod
    def parse_tick_count(text):
        """Parse an optional positive tick count (default 1)."""
        text = (text or "").strip()
        if not text:
            return 1
        try:
            n = int(text)
        except ValueError:
            raise ValueError("tick count must be an integer") from None
        if n < 1 or n > 100000:
            raise ValueError("tick count must be between 1 and 100000")
        return n

    def read(self, prompt):
        """Read a line from the user."""
        return self.reader(prompt)

"""Command-line entry point of the food delivery dispatch simulator."""

import logging
import sys

import globalvar
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from system import SystemManager

HELP = """Commands (shortcut in brackets):
  start [rx ry cx cy] (s)  create an order (coordinates 0-800)
  status (st)              funds and rider states
  orders (o)               all orders
  riders (r)               rider status change log
  map (m)                  show the ASCII map
  tick [n] (t)             advance the simulation n ticks
  save                     save state to JSON
  history (hist)           command history; '!!' repeats, '!n' runs n-th
  help (h, ?)              this help
  quit / exit (q, x)       leave the program
Every command line also advances the simulation by one tick."""


def handle(cmd, args, sysm, om, dm, im):
    """Execute one command. Returns True if a tick should follow."""
    if cmd == "start":
        text = args or im.read("Enter restaurant_x restaurant_y "
                               "customer_x customer_y: ")
        coords = im.parse_coordinates(text)
        result = om.place_order(coords)
        dm.show_assignment(result)
    elif cmd == "status":
        dm.show_status()
    elif cmd == "orders":
        dm.show_orders()
        stats = om.statistics()
        print(f"Completed: {stats['completed']}  "
              f"Avg time: {stats['avg_time']}  "
              f"Avg efficiency: {stats['avg_efficiency']}")
    elif cmd == "riders":
        dm.show_rider_logs()
    elif cmd == "map":
        dm.show_map()
    elif cmd == "tick":
        n = im.parse_tick_count(args)
        sysm.tick(n)
        print(f"Advanced {n} tick(s); time is now {sysm.time}.")
        return False
    elif cmd == "save":
        print(f"State saved to {sysm.save_state()}")
    elif cmd == "history":
        for i, h in enumerate(im.history, 1):
            print(f"{i:>3}  {h}")
    elif cmd == "help":
        print(HELP)
    elif cmd in ("quit", "exit"):
        sysm.stop()
        return False
    return True


def main(argv=None):
    """Run the interactive main loop."""
    argv = sys.argv[1:] if argv is None else argv
    level = logging.DEBUG if "--debug" in argv else logging.WARNING
    logging.basicConfig(level=level,
                        format="%(levelname)s %(name)s: %(message)s")
    sysm = SystemManager()
    om = OrderManager()
    dm = DisplayManager()
    im = InputManager()
    print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
    print("Type 'help' for commands.")
    while globalvar.get_value("running", False):
        try:
            line = im.read(f"[t={sysm.time}] dispatch> ")
        except (EOFError, KeyboardInterrupt):
            print()
            sysm.stop()
            break
        try:
            cmd, args = im.parse(line)
        except ValueError as exc:
            print(f"Error: {exc}")
            continue
        if cmd is None:
            sysm.tick()
            continue
        try:
            if handle(cmd, args, sysm, om, dm, im):
                sysm.tick()
        except (EOFError, KeyboardInterrupt):
            print()
            sysm.stop()
        except ValueError as exc:
            print(f"Error: {exc}")
        except Exception as exc:  # recovery
            logging.exception("Unexpected error")
            print(f"Unexpected error: {exc}. Recovering...")
            for fix in sysm.check_consistency():
                print(f"  fixed: {fix}")
    try:
        sysm.save_state()
    except OSError as exc:
        print(f"Could not save state: {exc}")
    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

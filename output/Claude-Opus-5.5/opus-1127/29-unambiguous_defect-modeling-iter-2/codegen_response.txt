# Food Delivery Order Dispatch System (CLI simulator)

Run: `python main.py [--config config.json] [--debug] [--no-save]`

Commands (shortcut in brackets):

* `start` (`s`) – enter `restaurant_x restaurant_y customer_x customer_y` (0-800).
  Can be given inline: `start 120 340 560 780`.
* `status` (`st`) – funds, riders, states, order lists and map.
* `orders` (`o`) – all orders with timing, distance and rider.
* `riders` (`r`) – status-change log of every rider.
* `map` (`m`) – ASCII map.
* `step [n]` (`n`) – advance simulation `n` ticks.
* `history` (`hi`), `!N` – show / re-run command N.
* `save` – write state to the JSON state file.
* `quit` / `exit` (`q`).

Every iteration of the main loop advances the simulation by one tick.

Map coordinates: 800x800, cells of 100x100. Streets run along multiples of
100; restaurants/customers sit at cell centres (`(v // 100) * 100 + 50`).

"""Configuration defaults and loading for the dispatch simulator."""

import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 10000,
    "recruitment_cost": 1000,
    "rider_speed": 100,
    "max_riders": 10,
    "map_size": 800,
    "cell_size": 100,
    "state_file": "dispatch_state.json",
}


def load_config(path=None):
    """Load configuration from a JSON file merged over the defaults.

    Unknown keys are ignored; invalid files fall back to defaults.

    :param path: path of a JSON configuration file, or None.
    :return: configuration dictionary.
    """
    cfg = dict(DEFAULT_CONFIG)
    if not path or not os.path.exists(path):
        return cfg
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        if isinstance(data, dict):
            for key, default in DEFAULT_CONFIG.items():
                if key in data:
                    value = data[key]
                    if isinstance(default, int) and not isinstance(value, bool):
                        try:
                            value = int(value)
                        except (TypeError, ValueError):
                            logger.warning("Invalid config value for %s", key)
                            continue
                    cfg[key] = value
    except (OSError, ValueError) as exc:
        logger.warning("Could not load config %s: %s", path, exc)
    return cfg

"""Grid map definition and grid-based path planning.

Streets run along every line where x or y is a multiple of the cell size.
Restaurants and customers are located at cell centres, and riders reach
them through the midpoint of one of the four streets surrounding the cell.
"""

import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = MAP_SIZE // CELL_SIZE


def manhattan(a, b):
    """Return the Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class GridMap:
    """8x8 grid over an 800x800 coordinate system."""

    def __init__(self, map_size=MAP_SIZE, cell_size=CELL_SIZE):
        self.map_size = int(map_size)
        self.cell_size = int(cell_size)
        self.grid_dimension = self.map_size // self.cell_size

    def clamp(self, value):
        """Clamp a coordinate to [0, map_size]."""
        return max(0, min(self.map_size, int(round(value))))

    def to_grid_center(self, value):
        """Convert an input coordinate to the centre of its grid cell.

        ``center = (v // 100) * 100 + 50``; the value ``map_size`` (and
        anything beyond) is mapped into the last cell.
        """
        v = max(0, min(self.map_size - 1, int(value)))
        return (v // self.cell_size) * self.cell_size + self.cell_size // 2

    def to_grid_cell(self, value):
        """Return the grid cell index (0..dimension-1) of a coordinate."""
        v = max(0, min(self.map_size - 1, int(value)))
        return v // self.cell_size

    def is_on_street(self, x, y):
        """True when the point lies on a street (grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    def in_bounds(self, x, y):
        """True when the point lies inside the map."""
        return 0 <= x <= self.map_size and 0 <= y <= self.map_size


DEFAULT_GRID = GridMap()


def to_grid_center(value):
    """Module-level helper using the default grid."""
    return DEFAULT_GRID.to_grid_center(value)


class Route:
    """A planned route: ordered waypoints with length and estimated time."""

    def __init__(self, waypoints, speed=1):
        self.waypoints = [tuple(p) for p in waypoints]
        self.speed = max(1, int(speed))

    @property
    def length(self):
        """Total route length in map units."""
        return sum(manhattan(self.waypoints[i], self.waypoints[i + 1])
                   for i in range(len(self.waypoints) - 1))

    @property
    def estimated_time(self):
        """Estimated number of ticks needed to travel the route."""
        return int(math.ceil(self.length / float(self.speed)))

    def __repr__(self):
        return "Route(%s, length=%d)" % (self.waypoints, self.length)


class PathPlanner:
    """Plans street-following paths on the grid.

    Riders never block each other (streets have unlimited capacity), so
    planned paths are conflict- and deadlock-free by construction.
    """

    def __init__(self, grid=None):
        self.grid = grid or DEFAULT_GRID

    # ------------------------------------------------------------------ utils
    def _norm(self, p):
        return (self.grid.clamp(p[0]), self.grid.clamp(p[1]))

    @staticmethod
    def simplify(points):
        """Remove duplicate and collinear intermediate waypoints."""
        out = []
        for p in points:
            p = tuple(p)
            if out and out[-1] == p:
                continue
            if len(out) >= 2:
                a, b = out[-2], out[-1]
                if (a[0] == b[0] == p[0]) or (a[1] == b[1] == p[1]):
                    out[-1] = p
                    continue
            out.append(p)
        return out

    @staticmethod
    def length_of(points):
        """Length of a polyline of axis-aligned segments."""
        return sum(manhattan(points[i], points[i + 1])
                   for i in range(len(points) - 1))

    def _anchors(self, p):
        """Intersections reachable straight along the street of ``p``."""
        cs = self.grid.cell_size
        x, y = p
        if x % cs == 0 and y % cs == 0:
            cands = [(x, y)]
        elif x % cs == 0:
            lo = (y // cs) * cs
            cands = [(x, lo), (x, lo + cs)]
        else:
            lo = (x // cs) * cs
            cands = [(lo, y), (lo + cs, y)]
        return [c for c in cands if self.grid.in_bounds(*c)]

    def _snap_points(self, p):
        """Nearest street points for an off-street location."""
        cs = self.grid.cell_size
        x, y = p
        lx = (x // cs) * cs
        ly = (y // cs) * cs
        cands = [(lx, y), (lx + cs, y), (x, ly), (x, ly + cs)]
        return [c for c in cands if self.grid.in_bounds(*c)]

    # --------------------------------------------------------------- planning
    def plan_path(self, start, end):
        """Return the shortest street path from ``start`` to ``end``.

        The returned list contains ``start`` and ``end`` and only right-angle
        turns. Off-street endpoints are first snapped to the nearest street.
        """
        start = self._norm(start)
        end = self._norm(end)
        if start == end:
            return [start]
        g = self.grid
        cs = g.cell_size
        if not g.is_on_street(*start):
            best = None
            for s in self._snap_points(start):
                cand = self.simplify([start] + self.plan_path(s, end))
                if best is None or self.length_of(cand) < self.length_of(best):
                    best = cand
            return best
        if not g.is_on_street(*end):
            best = None
            for e in self._snap_points(end):
                cand = self.simplify(self.plan_path(start, e) + [end])
                if best is None or self.length_of(cand) < self.length_of(best):
                    best = cand
            return best
        sx, sy = start
        ex, ey = end
        if sx % cs == 0 and sx == ex:
            return [start, end]
        if sy % cs == 0 and sy == ey:
            return [start, end]
        best = None
        best_len = None
        for a in self._anchors(start):
            for b in self._anchors(end):
                for corner in ((b[0], a[1]), (a[0], b[1])):
                    cand = self.simplify([start, a, corner, b, end])
                    length = self.length_of(cand)
                    if best is None or length < best_len:
                        best, best_len = cand, length
        return best

    def plan_route(self, start, end, speed=1):
        """Plan a path and wrap it into a :class:`Route`."""
        return Route(self.plan_path(start, end), speed)

    def path_length(self, start, end):
        """Length of the planned path between two points."""
        return self.length_of(self.plan_path(start, end))

    def access_points(self, center):
        """Street midpoints surrounding a cell centre."""
        cx, cy = int(center[0]), int(center[1])
        h = self.grid.cell_size // 2
        cands = [(cx - h, cy), (cx + h, cy), (cx, cy - h), (cx, cy + h)]
        return [c for c in cands if self.grid.in_bounds(*c)]

    def best_access_point(self, from_pos, center):
        """Return ``(point, path_length)`` of the closest access point."""
        best = None
        for p in self.access_points(center):
            length = self.path_length(from_pos, p)
            if best is None or length < best[1]:
                best = (p, length)
        if best is None:
            p = self._norm(center)
            best = (p, self.path_length(from_pos, p))
        return best

    def is_special_case(self, start, end):
        """True if a detour is needed (e.g. target directly above/below)."""
        cs = self.grid.cell_size
        sx, sy = self._norm(start)
        ex, ey = self._norm(end)
        if not self.grid.is_on_street(sx, sy):
            return True
        if sx == ex and sx % cs != 0 and sy != ey:
            return True
        if sy == ey and sy % cs != 0 and sx != ex:
            return True
        return False

"""Order data model."""

import globalvar
from path_planner import DEFAULT_GRID, manhattan


def _next_order_id():
    counter = (globalvar.get_value("order_id_counter", 0) or 0) + 1
    globalvar.set_value("order_id_counter", counter)
    return counter


def _pair(pos, name):
    try:
        x, y = pos
        return int(x), int(y)
    except (TypeError, ValueError):
        raise ValueError("%s must be a pair of integers" % name)


class Order:
    """A delivery order from a restaurant to a customer.

    Input coordinates are automatically converted to grid cell centres.
    Status lifecycle: pending -> assigned -> picking_up -> delivering ->
    completed.
    """

    def __init__(self, res_pos, cus_pos, start_time=None):
        rx, ry = _pair(res_pos, "res_pos")
        cx, cy = _pair(cus_pos, "cus_pos")
        self.id = _next_order_id()
        self.raw_res_pos = (rx, ry)
        self.raw_cus_pos = (cx, cy)
        self.res_pos = (DEFAULT_GRID.to_grid_center(rx),
                        DEFAULT_GRID.to_grid_center(ry))
        self.cus_pos = (DEFAULT_GRID.to_grid_center(cx),
                        DEFAULT_GRID.to_grid_center(cy))
        self.status = "pending"
        self.rider_id = None
        if start_time is None:
            start_time = globalvar.get_value("tick", 0) or 0
        self.start_time = int(start_time)
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.pickup_point = None
        self.dropoff_point = None
        self.distance = manhattan(self.res_pos, self.cus_pos)
        self.efficiency = None

    # Convenience accessors matching the [rx, ry, cx, cy] format.
    @property
    def restaurant_x(self):
        return self.res_pos[0]

    @property
    def restaurant_y(self):
        return self.res_pos[1]

    @property
    def customer_x(self):
        return self.cus_pos[0]

    @property
    def customer_y(self):
        return self.cus_pos[1]

    @property
    def completed(self):
        """True once the order has been delivered."""
        return self.status == "completed"

    def as_list(self):
        """Return ``[restaurant_x, restaurant_y, customer_x, customer_y]``."""
        return [self.res_pos[0], self.res_pos[1],
                self.cus_pos[0], self.cus_pos[1]]

    def assign_to(self, rider_id):
        """Assign this order to the rider with ``rider_id``."""
        self.rider_id = rider_id
        self.assign_time = globalvar.get_value("tick", 0) or 0
        if self.status == "pending":
            self.status = "assigned"

    def to_dict(self):
        """JSON-serialisable representation."""
        return {
            "id": self.id,
            "raw_restaurant": list(self.raw_res_pos),
            "raw_customer": list(self.raw_cus_pos),
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "efficiency": self.efficiency,
        }

    def __repr__(self):
        return "Order(#%d %s->%s %s)" % (self.id, self.res_pos,
                                         self.cus_pos, self.status)

"""Rider model: movement, state transitions and status log."""

import logging
from enum import IntEnum

import globalvar
from path_planner import PathPlanner, DEFAULT_GRID

logger = logging.getLogger(__name__)

DEFAULT_SPEED = 100

SPAWN_POINTS = [
    (400, 400), (0, 0), (800, 800), (0, 800), (800, 0),
    (400, 0), (0, 400), (800, 400), (400, 800), (200, 600),
]

_planner = PathPlanner()


class RiderState(IntEnum):
    """Rider status enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2


STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class Rider:
    """A delivery rider moving along the street grid."""

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = int(rider_id)
        if x is None or y is None:
            sx, sy = SPAWN_POINTS[(self.rider_id - 1) % len(SPAWN_POINTS)]
            x = sx if x is None else x
            y = sy if y is None else y
        self.x = DEFAULT_GRID.clamp(x)
        self.y = DEFAULT_GRID.clamp(y)
        if speed is None:
            speed = globalvar.get_value("rider_speed") or DEFAULT_SPEED
        self.speed = max(1, int(speed))
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.route = []
        self.current_order = None
        self.completed_orders = []
        self.distance_travelled = 0
        self.log = []
        self._record("spawn", None)

    # ---------------------------------------------------------------- helpers
    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(int(self.state), "unknown")

    def _record(self, event, order_id):
        entry = {
            "time": globalvar.get_value("tick", 0) or 0,
            "event": event,
            "order_id": order_id,
            "x": self.x,
            "y": self.y,
            "state": int(self.state),
        }
        self.log.append(entry)
        logger.debug("Rider %d: %s", self.rider_id, entry)

    def pending_orders(self):
        """Orders assigned to this rider and not yet delivered."""
        return [o for o in self.orders
                if getattr(o, "status", None) != "completed"]

    def is_available(self):
        """True if the rider is idle with no queued orders."""
        return self.state == RiderState.IDLE and not self.pending_orders()

    def final_position(self):
        """Estimated position after finishing every queued order."""
        pending = self.pending_orders()
        if not pending:
            return (self.x, self.y)
        last = pending[-1]
        if getattr(last, "dropoff_point", None):
            return tuple(last.dropoff_point)
        pickup = getattr(last, "pickup_point", None)
        if not pickup:
            pickup = _planner.best_access_point((self.x, self.y),
                                                last.res_pos)[0]
        return _planner.best_access_point(pickup, last.cus_pos)[0]

    def reset(self):
        """Recover to a consistent idle state (error recovery)."""
        if self.current_order is not None and \
                self.current_order.status != "completed":
            self.current_order.status = "assigned"
        self.state = RiderState.IDLE
        self.route = []
        self.current_order = None
        self.special_case = False

    def _advance(self):
        """Move up to ``speed`` units along the remaining route."""
        budget = self.speed
        while budget > 0 and self.route:
            tx, ty = self.route[0]
            d = abs(tx - self.x) + abs(ty - self.y)
            if d <= budget:
                self.x, self.y = tx, ty
                budget -= d
                self.distance_travelled += d
                self.route.pop(0)
            else:
                if tx != self.x:
                    self.x += budget if tx > self.x else -budget
                else:
                    self.y += budget if ty > self.y else -budget
                self.distance_travelled += budget
                budget = 0

    # -------------------------------------------------------------- lifecycle
    def _accept(self, order):
        if getattr(order, "rider_id", None) != self.rider_id:
            order.assign_to(self.rider_id)
        order.status = "picking_up"
        order.accept_time = globalvar.get_value("tick", 0) or 0
        pickup, _ = _planner.best_access_point((self.x, self.y), order.res_pos)
        order.pickup_point = pickup
        self.special_case = _planner.is_special_case((self.x, self.y), pickup)
        self.route = _planner.plan_path((self.x, self.y), pickup)[1:]
        self.current_order = order
        self.state = RiderState.PICKING_UP
        self._record("accepted_order", order.id)

    def _arrive_at_merchant(self, order):
        here = (self.x, self.y)
        dropoff, length = _planner.best_access_point(here, order.cus_pos)
        order.dropoff_point = dropoff
        order.distance = length
        order.pickup_time = globalvar.get_value("tick", 0) or 0
        order.status = "delivering"
        self.special_case = _planner.is_special_case(here, dropoff)
        self.route = _planner.plan_path(here, dropoff)[1:]
        self.state = RiderState.DELIVERING
        self._record("arrived_at_merchant", order.id)

    def _deliver(self, order):
        now = globalvar.get_value("tick", 0) or 0
        order.status = "completed"
        order.completion_time = now
        order.time_taken = now - order.start_time
        if order.time_taken > 0:
            order.efficiency = round(order.distance / float(order.time_taken), 2)
        if order in self.orders:
            self.orders.remove(order)
        self.completed_orders.append(order)
        self.current_order = None
        self.route = []
        self.special_case = False
        self.state = RiderState.IDLE
        self._record("delivered_to_customer", order.id)

    def move(self):
        """Advance the rider by one tick.

        Idle riders accept their next queued order; picking-up riders move
        towards the restaurant and switch to delivering once they reach it;
        delivering riders move to the customer and become idle on delivery.
        """
        if self.state == RiderState.IDLE:
            pending = self.pending_orders()
            if pending:
                self._accept(pending[0])
            return
        order = self.current_order
        if order is None:
            self.reset()
            return
        if self.state == RiderState.PICKING_UP:
            self._advance()
            if not self.route:
                self._arrive_at_merchant(order)
        elif self.state == RiderState.DELIVERING:
            self._advance()
            if not self.route:
                self._deliver(order)
        else:
            self.reset()

    def to_dict(self):
        """JSON-serialisable representation."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": int(self.state),
            "orders": [o.id for o in self.orders],
            "completed_orders": [o.id for o in self.completed_orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(#%d at (%d,%d) %s, %d orders)" % (
            self.rider_id, self.x, self.y, self.state_name, len(self.orders))

import logging

import globalvar
from config import DEFAULT_CONFIG
from rider import Rider, RiderState
from path_planner import DEFAULT_GRID

logger = logging.getLogger(__name__)


def init_system(config=None):
    """Initialise all global system parameters.

    :param config: optional dict overriding :data:`DEFAULT_CONFIG`.
    """
    cfg = dict(DEFAULT_CONFIG)
    if config:
        cfg.update(config)
    globalvar._init()
    globalvar.set_value("config", cfg)
    globalvar.set_value("funds", int(cfg["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)
    globalvar.set_value("recruitment_cost", int(cfg["recruitment_cost"]))
    globalvar.set_value("max_riders", int(cfg["max_riders"]))
    globalvar.set_value("rider_speed", int(cfg["rider_speed"]))
    logger.info("System initialised with funds %s", cfg["initial_funds"])


def _riders():
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    return riders


def can_recruit():
    """True when funds allow and the rider limit is not reached."""
    funds = globalvar.get_value("funds", DEFAULT_CONFIG["initial_funds"])
    cost = globalvar.get_value("recruitment_cost",
                               DEFAULT_CONFIG["recruitment_cost"])
    max_riders = globalvar.get_value("max_riders", DEFAULT_CONFIG["max_riders"])
    return funds >= cost and len(_riders()) < max_riders


def recruit_rider(force=False):
    """Create a new rider, deducting the recruitment cost.

    :param force: recruit even if funds are insufficient (funds may go
        negative).
    :return: the new Rider, or None if recruitment is not allowed.
    """
    if not force and not can_recruit():
        return None
    funds = globalvar.get_value("funds", DEFAULT_CONFIG["initial_funds"])
    cost = globalvar.get_value("recruitment_cost",
                               DEFAULT_CONFIG["recruitment_cost"])

    globalvar.set_value("rider_id_counter", counter)
    rider = Rider(counter)
    _riders().append(rider)
    globalvar.set_value("funds", int(funds) - int(cost))
    logger.info("Recruited rider %d", counter)
    return rider


def try_recruit_rider():
    """Recruit a rider if allowed, otherwise return None."""
    return recruit_rider(force=False)


def advance_tick():
    """Advance the simulation by one tick and move every rider.

    :return: the new tick number.
    """
    tick = (globalvar.get_value("tick", 0) or 0) + 1
    globalvar.set_value("tick", tick)
    for rider in list(_riders()):
        try:
            rider.move()
        except Exception:  # pragma: no cover - defensive recovery
            logger.exception("Rider %s failed to move; resetting",
                             getattr(rider, "rider_id", "?"))
            try:
                rider.reset()
            except Exception:
                pass
    return tick


def check_consistency():
    """Check and repair global state. Returns a list of applied fixes."""
    fixes = []
    riders = globalvar.get_value("riders")
    if not isinstance(riders, list):
        globalvar.set_value("riders", [])
        fixes.append("rider list reset")
        riders = []

    if not isinstance(funds, int) or isinstance(funds, bool):
        try:
            globalvar.set_value("funds", int(funds))
        except (TypeError, ValueError):
            globalvar.set_value("funds", 0)
        fixes.append("funds normalised")

        globalvar.set_value("orders", [])
        fixes.append("order list reset")
    for r in riders:
        if int(r.state) not in (0, 1, 2):
            r.reset()
            fixes.append("rider %d invalid state" % r.rider_id)
        if r.state != RiderState.IDLE and r.current_order is None:
            r.reset()
            fixes.append("rider %d without current order" % r.rider_id)

        if (cx, cy) != (r.x, r.y):
            r.x, r.y = cx, cy
            fixes.append("rider %d position clamped" % r.rider_id)
    for f in fixes:
        logger.warning("Consistency fix: %s", f)
    return fixes

"""Order creation, assignment (distance-first) and bookkeeping."""

import logging

import globalvar
import system
from order_model import Order
from path_planner import PathPlanner, Route

logger = logging.getLogger(__name__)

_planner = PathPlanner()


def rider_distance_to_order(rider, order, from_queue_end=False):
    """Street distance from the rider to the order's restaurant."""
    start = rider.final_position() if from_queue_end else (rider.x, rider.y)
    return _planner.best_access_point(start, order.res_pos)[1]


def find_best_rider_for_order(order):
    """Select the most suitable rider for ``order``.

    1. The nearest idle rider (street distance), ties broken by id.
    2. Otherwise a newly recruited rider if funds allow.
    3. Otherwise the least loaded rider, then the nearest queue end.
    4. With no riders at all, a rider is force-recruited.

    The order is not appended; the caller appends it to ``rider.orders``.
    """
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    available = [r for r in riders if r.is_available()]
    if available:
        return min(available,
                   key=lambda r: (rider_distance_to_order(r, order),
                                  r.rider_id))
    new_rider = system.try_recruit_rider()
    if new_rider is not None:
        return new_rider
    if riders:
        return min(riders,
                   key=lambda r: (len(r.pending_orders()),
                                  rider_distance_to_order(r, order, True),
                                  r.rider_id))
    logger.warning("No riders and insufficient funds; forcing recruitment")
    return system.recruit_rider(force=True)


class OrderManager:
    """Handles order creation, assignment and completion statistics."""

    def __init__(self, planner=None):
        self.planner = planner or _planner

    @property
    def orders(self):
        """All orders known to the system."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create and register an order from ``[rx, ry, cx, cy]``."""
        if len(coords) != 4:
            raise ValueError("exactly four coordinates are required")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        self.orders.append(order)
        return order

    def preview_route(self, rider, order):
        """Planned route rider -> restaurant -> customer."""
        if rider.pending_orders():
            start = rider.final_position()
        else:
            start = (rider.x, rider.y)
        pickup, _ = self.planner.best_access_point(start, order.res_pos)
        dropoff, _ = self.planner.best_access_point(pickup, order.cus_pos)
        first = self.planner.plan_path(start, pickup)
        second = self.planner.plan_path(pickup, dropoff)
        waypoints = PathPlanner.simplify(first + second[1:])
        return Route(waypoints, rider.speed), pickup, dropoff

    def assign_order(self, order):
        """Assign ``order`` to the best rider.

        :return: (rider, recruited: bool)
        """
        before = globalvar.get_value("rider_id_counter", 0) or 0
        rider = find_best_rider_for_order(order)
        recruited = (globalvar.get_value("rider_id_counter", 0) or 0) > before
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider, recruited

    def submit(self, coords):
        """Create, assign and preview an order.

        :return: dict with keys order, rider, recruited, route, pickup,
            dropoff, queue_position.
        """
        order = self.create_order(coords)
        before = globalvar.get_value("rider_id_counter", 0) or 0
        rider = find_best_rider_for_order(order)
        recruited = (globalvar.get_value("rider_id_counter", 0) or 0) > before
        route, pickup, dropoff = self.preview_route(rider, order)
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return {
            "order": order,
            "rider": rider,
            "recruited": recruited,
            "route": route,
            "pickup": pickup,
            "dropoff": dropoff,
            "queue_position": len(rider.pending_orders()),
        }

    def completed_orders(self):
        """Orders already delivered."""
        return [o for o in self.orders if o.completed]

    def active_orders(self):
        """Orders not yet delivered."""
        return [o for o in self.orders if not o.completed]

    def statistics(self):
        """Delivery efficiency summary."""
        done = self.completed_orders()
        times = [o.time_taken for o in done if o.time_taken is not None]
        return {
            "total": len(self.orders),
            "completed": len(done),
            "active": len(self.orders) - len(done),
            "average_time": round(sum(times) / len(times), 2) if times else None,
        }

class DisplayManager:
    """Formats status information and renders the ASCII map."""

    STEP = 50

    def __init__(self, grid=None):
        self.grid = grid or DEFAULT_GRID

    def _idx(self, v):
        n = self.grid.map_size // self.STEP
        return max(0, min(n, int(round(v / float(self.STEP)))))

    def render_map(self, riders, orders=None, route=None):
        """Render an ASCII map (y grows downwards).

        :param riders: riders to draw.
        :param orders: orders whose restaurant/customer are drawn.
        :param route: optional list of waypoints drawn as '*'.
        """
        cs = self.grid.cell_size
        n = self.grid.map_size // self.STEP + 1
        cells = []
        for r in range(n):
            row = []
            for c in range(n):
                x, y = c * self.STEP, r * self.STEP
                if x % cs == 0 and y % cs == 0:
                    row.append("+")
                elif x % cs == 0:
                    row.append("|")
                elif y % cs == 0:
                    row.append("-")
                else:
                    row.append(".")
            cells.append(row)
        if route:
            for i in range(len(route) - 1):
                (x1, y1), (x2, y2) = route[i], route[i + 1]
                c1, r1, c2, r2 = (self._idx(x1), self._idx(y1),
                                  self._idx(x2), self._idx(y2))
                if r1 == r2:
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        cells[r1][c] = "*"
                else:
                    for r in range(min(r1, r2), max(r1, r2) + 1):
                        cells[r][c1] = "*"
        for o in orders or []:
            cells[self._idx(o.res_pos[1])][self._idx(o.res_pos[0])] = "R"
            cells[self._idx(o.cus_pos[1])][self._idx(o.cus_pos[0])] = "C"
        seen = {}
        for rd in riders or []:
            key = (self._idx(rd.y), self._idx(rd.x))
            seen[key] = seen.get(key, 0) + 1
            cells[key[0]][key[1]] = ("@" if seen[key] > 1
                                     else str(rd.rider_id % 10))
        header = "x/100 " + " ".join(
            str(c // 2) if c % 2 == 0 else " " for c in range(n))
        lines = [header]
        for r in range(n):
            lines.append("%5d " % (r * self.STEP) + " ".join(cells[r]))
        lines.append(LEGEND)
        return "\n".join(lines)

"""Command parsing, input validation and command history."""

from collections import deque

try:  # optional line editing / arrow-key history
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "m": "map", "n": "step", "t": "step", "hi": "history",
    "h": "help", "?": "help", "q": "quit", "exit": "quit",
}

COMMANDS = {"start", "status", "orders", "riders", "map", "step",
            "history", "help", "save", "quit"}


class InputManager:
    """Parses commands and validates coordinate input."""

    def __init__(self, max_history=200, min_coord=0, max_coord=800):
        self.command_history = deque(maxlen=max_history)
        self.min_coord = min_coord
        self.max_coord = max_coord

    def read_line(self, prompt="> "):
        """Read a line from stdin (raises EOFError/KeyboardInterrupt)."""
        return input(prompt)

    def resolve_history(self, line):
        """Expand ``!N`` / ``!!`` references to earlier commands."""
        line = line.strip()
        if line == "!!":
            if not self.command_history:
                raise ValueError("no previous command")
            return self.command_history[-1]
        if line.startswith("!"):
            try:
                idx = int(line[1:])
            except ValueError:
                raise ValueError("usage: !N where N is a history number")
            if not 1 <= idx <= len(self.command_history):
                raise ValueError("no such history entry: %d" % idx)
            return self.command_history[idx - 1]
        return line

    def parse_command(self, line):
        """Return ``(command, args)``; unknown commands are returned as-is."""
        line = self.resolve_history(line)
        if not line:
            return "", []
        self.command_history.append(line)
        parts = line.split(None, 1)
        cmd = parts[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        args = parts[1] if len(parts) > 1 else ""
        return cmd, args

    def parse_coordinates(self, text):
        """Parse ``[rx, ry, cx, cy]`` into four validated integers.

        :raises ValueError: with a user friendly message.
        """
        cleaned = str(text)
        for ch in "[](),;":
            cleaned = cleaned.replace(ch, " ")
        parts = cleaned.split()
        if len(parts) != 4:
            raise ValueError("exactly 4 integers are required "
                             "(restaurant_x restaurant_y customer_x customer_y)")
        values = []
        for p in parts:
            try:
                v = int(p)
            except ValueError:
                raise ValueError("'%s' is not an integer" % p)
            if not self.min_coord <= v <= self.max_coord:
                raise ValueError("coordinate %d out of range %d-%d" % (
                    v, self.min_coord, self.max_coord))
            values.append(v)
        return values

    def parse_step_count(self, text, maximum=1000):
        """Parse the argument of ``step`` (default 1)."""
        text = str(text).strip()
        if not text:
            return 1
        try:
            n = int(text)
        except ValueError:
            raise ValueError("step count must be an integer")
        if not 1 <= n <= maximum:
            raise ValueError("step count must be between 1 and %d" % maximum)
        return n

    def history_text(self):
        """Numbered command history."""
        if not self.command_history:
            return "No commands yet."
        return "\n".join("%3d  %s" % (i + 1, c)
                         for i, c in enumerate(self.command_history))

class SystemManager:

        cfg = globalvar.get_value("config")
        self.grid = GridMap(cfg["map_size"], cfg["cell_size"])
        self.planner = PathPlanner(self.grid)
        self.order_manager = OrderManager(self.planner)

        self.input_manager = InputManager(max_coord=self.grid.map_size)
        self.state_file = cfg.get("state_file", "dispatch_state.json")
        self.save_on_exit = save_on_exit

    @property
    def funds(self):

    @property

        return globalvar.get_value("recruitment_cost", 0)

    def tick(self, n=1):
        """Advance the simulation ``n`` ticks."""
        for _ in range(n):
            system.advance_tick()
            system.check_consistency()
        return self.current_tick

    # -------------------------------------------------------------- commands
    def handle_command(self, line):
        """Execute one command line and return its output text."""
        try:
            cmd, args = self.input_manager.parse_command(line)
        except ValueError as exc:
            return "Error: %s" % exc
        if not cmd:
            return ""
        handler = getattr(self, "cmd_" + cmd, None)
        if handler is None:
            return "Unknown command '%s'. Type 'help' for commands." % cmd
        try:
            return handler(args)
        except ValueError as exc:
            return "Error: %s" % exc
        except Exception as exc:  # graceful recovery
            logger.exception("Command %s failed", cmd)
            system.check_consistency()
            return "Internal error (%s); system state recovered." % exc

    def cmd_start(self, args):
        if not args.strip():
            try:
                args = self.input_manager.read_line(
                    "Enter restaurant_x restaurant_y customer_x customer_y "

            except (EOFError, KeyboardInterrupt):

    def cmd_map(self, args):

    def cmd_step(self, args):
        n = self.input_manager.parse_step_count(args)
        # The main loop advances one more tick after the command.
        if n > 1:
            self.tick(n - 1)
        return "Advancing %d tick(s)." % n

    def cmd_history(self, args):
        return self.input_manager.history_text()

    def cmd_help(self, args):
        return self.display.help_text()

    def cmd_save(self, args):
        path = args.strip() or self.state_file
        self.save_state(path)
        return "State saved to %s" % path

    def cmd_quit(self, args):
        globalvar.set_value("running", False)
        return "Goodbye."

    # ----------------------------------------------------------- persistence
    def to_dict(self):
        """JSON-serialisable snapshot of the whole system."""
        return {
            "config": globalvar.get_value("config"),
            "funds": self.funds,
            "tick": self.current_tick,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.orders],

        path = path or self.state_file
        with open(path, "w", encoding="utf-8") as fh:

        """Interactive main loop; each iteration is one simulated tick."""

        print("Funds: %d. Type 'help' for commands." % self.funds)

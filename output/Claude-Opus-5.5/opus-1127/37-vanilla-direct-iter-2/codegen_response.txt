"""Core data models for the lottery system."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A participant in the lottery."""

    name: str
    department: str

    def __post_init__(self):
        self.name = str(self.name).strip()
        self.department = str(self.department).strip()

    def __str__(self):
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize configuration (one lottery round)."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        self.name = str(self.name).strip()
        if self.avoid_groups is None:
            self.avoid_groups = []
        elif isinstance(self.avoid_groups, str):
            self.avoid_groups = [g.strip() for g in self.avoid_groups.split(",") if g.strip()]
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer if drawer else None

    def __str__(self):
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "None"
        return (f"{self.name} | Quantity: {self.quantity} | "
                f"Avoid Groups: {avoid} | Drawer: {drawer}")

class ListManager:
    """Loads, validates and analyses participant lists."""

    def __init__(self):
        self.users: List[User] = []

    # ------------------------------------------------------------------ load
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if path is None or not str(path).strip():
            return [], ["File Error: no file path provided"]
        return self._load_from_path(str(path).strip())

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"File Error: cannot read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for n, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {n}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # -------------------------------------------------------------- validate
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    # ----------------------------------------------------------------- group
    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and draw order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            if any(p.name == name for p in self.prizes):
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        prize = Prize(str(name).strip(), qty, avoid_groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def get_prize(self, index: int) -> Optional[Prize]:
        if 0 <= index < len(self.prizes):
            return self.prizes[index]
        return None

    def set_avoid_groups(self, index: int, groups) -> bool:
        prize = self.get_prize(index)
        if prize is None:
            return False
        prize.avoid_groups = [str(g).strip() for g in (groups or []) if str(g).strip()]
        return True

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

import csv
import os
import re
from dataclasses import dataclass, field
from datetime import datetime
from typing import Dict, List, Optional, Tuple

from lottery_system.models import Prize, User

_LEVEL_PATTERNS = [
    (r"\b(special|grand)\b", 0),
    (r"\b(first|1st)\b", 1),
    (r"\b(second|2nd)\b", 2),
    (r"\b(third|3rd)\b", 3),
    (r"\b(fourth|4th)\b", 4),
    (r"\b(fifth|5th)\b", 5),
    (r"\b(sixth|6th)\b", 6),
]


def prize_level(name: str) -> int:
    """Return a numeric level for a prize name (lower is higher level)."""
    lower = (name or "").lower()
    for pattern, level in _LEVEL_PATTERNS:
        if re.search(pattern, lower):
            return level
    return 100


def sort_prizes_by_level(prizes) -> List[Prize]:
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda t: (prize_level(t[1].name), t[0]))
    return [p for _, p in indexed]


@dataclass
class Winner:
    user: User
    prize: Prize
    win_time: datetime = field(default_factory=datetime.now)
    seq: int = 0

    @property
    def prize_name(self) -> str:
        return self.prize.name if isinstance(self.prize, Prize) else str(self.prize)


class ResultManager:
    """Keeps the winner list and provides summaries/exports."""

    def __init__(self):
        self.winners: List[Winner] = []
        self._seq = 0

    def has_won(self, user) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def winner_names(self):
        return {w.user.name for w in self.winners}

    def add_winner(self, user, prize) -> Optional[Winner]:
        if self.has_won(user):
            return None
        self._seq += 1
        winner = Winner(user=user, prize=prize, win_time=datetime.now(), seq=self._seq)
        self.winners.append(winner)
        return winner

    def clear(self):
        self.winners = []
        self._seq = 0

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            if prize.name in result:
                continue
            ws = [w for w in self.winners if w.prize_name == prize.name]
            ws.sort(key=lambda w: (w.win_time, w.seq))
            result[prize.name] = ws
        # winners for prizes not in the given list
        extra = [w for w in self.winners if w.prize_name not in result]
        for w in sorted(extra, key=lambda w: (prize_level(w.prize_name), w.win_time, w.seq)):
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes, path=None) -> Tuple[bool, str]:
        if path is None:
            filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
            path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, ws in grouped.items():
                    for w in ws:
                        writer.writerow([
                            w.user.name,
                            w.user.department,
                            prize_name,
                            w.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

SPECIAL_DRAWER = "Li Bin"
WATCH_DRAWERS = ("Chen Qixian", "Jin Duo")
TARGET_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def _norm(text) -> str:
    return "".join(str(text or "").split()).casefold()


def is_protected_group(department) -> bool:
    d = _norm(department)
    return d in ("chengrugroup", "chengru")


class LotteryManager:
    """Applies filtering rules and performs draws."""

    def __init__(self, users, result_manager: Optional[ResultManager] = None, rng=None):
        self.users: List[User] = list(users)
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winner_names = set(self.result_manager.winner_names())
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()

    # ------------------------------------------------------------- rules
    def is_fallback_active(self, prize: Prize) -> bool:
        if _norm(prize.associated_drawer) != _norm(TARGET_DRAWER):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        available = [u for u in self.users if u.name not in self.winner_names]
        if self.is_fallback_active(prize):
            protected = [u for u in available if is_protected_group(u.department)]
            if protected:
                return protected
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return available
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in available if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        drawer = prize.associated_drawer
        if self.is_fallback_active(prize):
            parts.append(
                f"FALLBACK ACTIVE: pool limited to '{PROTECTED_GROUP}' members "
                f"(previous two drawers' results contained no {PROTECTED_GROUP} member)"
            )
        elif _norm(drawer) == _norm(SPECIAL_DRAWER):
            parts.append(f"Special drawer {SPECIAL_DRAWER}: avoidance groups not applied")
        elif prize.avoid_groups:
            parts.append("Avoidance groups excluded: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")
        if drawer:
            parts.append(f"Drawer: {drawer}")
        return "; ".join(parts)

    # -------------------------------------------------------------- draw
    def record_winner(self, user: User, prize: Prize) -> bool:
        if user.name in self.winner_names:
            return False
        winner = self.result_manager.add_winner(user, prize)
        self.winner_names.add(user.name)
        return winner is not None

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.record_winner(user, prize)
        return user

    def finish_round(self, prize: Prize, round_winners: List[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in WATCH_DRAWERS}:
            contains = any(is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(TARGET_DRAWER):
            self.last_two_draw_results = []

    def draw_round(self, prize: Prize) -> List[User]:
        """Non-interactive full round draw."""
        winners: List[User] = []
        for _ in range(prize.quantity):
            user = self.draw_one(prize)
            if user is None:
                break
            winners.append(user)
        self.finish_round(prize, winners)
        return winners

class ExitApp(Exception):
    pass


def ask(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        raise ExitApp()


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class App:
    def __init__(self):

        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery_manager = None

    def menu_load_list(self):
        print("\n1. Load built-in default list\n2. Upload custom txt list")
        choice = ask("Choose: ").strip()
        if choice == "1":
            users, errors = self.list_manager.load_default_list()
        elif choice == "2":
            path = ask("Enter txt file path: ").strip()
            users, errors = self.list_manager.load_custom_list(path)
        else:
            print("Invalid choice.")
            return
        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            print("\nProblems detected in the list:")
            for e in errors:
                print("  " + e)
            for d in duplicates:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            if not valid:
                print("No valid data. Upload terminated.")
                return
            ans = ask("Type 'r' to remove invalid data and continue, 't' to terminate upload: ")
            if ans.strip().lower() != "r":
                print("Upload terminated.")
                return
        if not valid:
            print("No valid data loaded.")
            return
        self.users = valid
        self.lottery_manager = None
        print(f"Loaded {len(valid)} valid participants.")

    def menu_view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("\nCurrent valid groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    # --------------------------------------------------------------- prize
    def _groups(self):
        return list(self.list_manager.get_group_info(self.users).keys())

        groups = self._groups()
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = ask("Avoidance groups (numbers or names, comma-separated, empty for none): ").strip()
        if not raw:
            return []
        selected = []
        for token in raw.replace("，", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                g = groups[int(token) - 1]
            else:
                g = token
            if g not in selected:
                selected.append(g)
        return selected

    def _show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return False
        print("\nPrize order:")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")

    def run_round(self, prize):

                    print("Lottery resumed.")
                continue
            if paused:
                print("Lottery is paused.")

            if user is None:
                print("No eligible participants left for this prize.")
                break

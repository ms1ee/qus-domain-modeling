from __future__ import annotations

from typing import List, Optional


class User:
    """A participant of the lottery."""

    def __init__(self, name: str, department: str):
        self.name = str(name).strip()
        self.department = str(department).strip()

    def __repr__(self) -> str:
        return f"User(name={self.name!r}, department={self.department!r})"

    def __eq__(self, other) -> bool:
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self) -> int:
        return hash((self.name, self.department))


class Prize:
    """A prize configured for a lottery round."""

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups: List[str] = list(avoid_groups) if avoid_groups else []
        if associated_drawer is not None:
            associated_drawer = str(associated_drawer).strip() or None
        self.associated_drawer: Optional[str] = associated_drawer

    def __repr__(self) -> str:
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
            f"avoid_groups={self.avoid_groups!r}, "
            f"associated_drawer={self.associated_drawer!r})"
        )

from __future__ import annotations

import os
import re
from typing import Dict, List, Tuple

from lottery_system.models import User

DEFAULT_LIST_TEXT = """\
Zhang San,Technology Department
Li Si,Technology Department
Wang Wu,Technology Department
Zhao Liu,Technology Department
Sun Qi,Product Department
Zhou Ba,Product Department
Wu Jiu,Product Department
Zheng Shi,Marketing Department
Feng Yi,Marketing Department
Chen Er,Marketing Department
Chu Wei,Human Resources Department
Wei Jiang,Human Resources Department
Shen Han,Cheng Ru Group
Yang Zhu,Cheng Ru Group
Qin You,Cheng Ru Group
Xu He,Cheng Ru Group
"""

_SPLIT_RE = re.compile(r"[,;，；]")


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self):
        self.users: List[User] = []

    # ------------------------------------------------------------------ parsing
    def _parse_lines(self, lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff")
            if not line:
                continue
            parts = [p.strip() for p in _SPLIT_RE.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File not found: {path}"]
        content = None
        for enc in ("utf-8-sig", "gbk", "latin-1"):
            try:
                with open(path, "r", encoding=enc) as f:
                    content = f.read()
                break
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"Unable to read file {path}: {exc}"]
        if content is None:
            return [], [f"Unable to decode file: {path}"]
        return self._parse_lines(content.splitlines())

    # --------------------------------------------------------------- validation
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize configuration management."""

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Holds the prize list in lottery order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self.prizes}
        for name, qty in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize already exists: {name}")
                continue
            try:
                self.add_prize(name, qty)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name,
        quantity,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ) -> Prize:
        name = str(name).strip() if name is not None else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError(f"Prize quantity must be positive: {quantity}")
        groups = [str(g).strip() for g in (avoid_groups or []) if str(g).strip()]
        # remove duplicate groups while preserving order
        groups = list(dict.fromkeys(groups))
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def set_avoid_groups(self, index: int, groups: List[str]) -> bool:
        """Set avoidance groups for the prize at 0-based index."""
        if not 0 <= index < len(self.prizes):
            return False
        cleaned = [str(g).strip() for g in groups if str(g).strip()]
        self.prizes[index].avoid_groups = list(dict.fromkeys(cleaned))
        return True

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Winner bookkeeping, display and export."""

from __future__ import annotations

import csv
import os
import re
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User

_LEVEL_PATTERNS = [
    (re.compile(r"\b(grand|special)\b", re.I), 0),
    (re.compile(r"\bfirst\b", re.I), 1),
    (re.compile(r"\bsecond\b", re.I), 2),
    (re.compile(r"\bthird\b", re.I), 3),
    (re.compile(r"\bfourth\b", re.I), 4),
    (re.compile(r"\bfifth\b", re.I), 5),
    (re.compile(r"\b(participation|consolation)\b", re.I), 50),
]


def prize_level(prize_name: str) -> int:
    for pattern, level in _LEVEL_PATTERNS:
        if pattern.search(prize_name):
            return level
    return 10


def sort_prizes_by_level(prizes) -> list:
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda t: (prize_level(t[1].name), t[0]))
    return [p for _, p in indexed]


class Winner:
    def __init__(self, user: User, prize: Prize, time: datetime | None = None):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()

    def __repr__(self) -> str:
        return f"Winner(user={self.user!r}, prize={self.prize.name!r})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self._names = set()

    def add_winner(self, user: User, prize: Prize) -> bool:
        if user.name in self._names:
            return False
        self._names.add(user.name)
        self.winners.append(Winner(user, prize))
        return True

    def is_winner(self, user: User) -> bool:
        return user.name in self._names

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for prize in sort_prizes_by_level(prizes):
            result.setdefault(prize.name, [])
        for w in self.winners:
            if w.prize.name in result:
                result[w.prize.name].append(w)
        for name in result:
            result[name].sort(key=lambda w: w.time)
        return result

    def format_summary(self, prizes) -> str:
        lines = ["===== Complete Winner List ====="]
        grouped = self.get_winners_by_prize(prizes)
        any_winner = False
        for name, winners in grouped.items():
            lines.append(f"[{name}]")
            if not winners:
                lines.append("  (no winners)")
            for i, w in enumerate(winners, 1):
                any_winner = True
                lines.append(
                    f"  {i}. {w.user.name} ({w.user.department}) - {name} "
                    f"@ {w.time.strftime('%Y-%m-%d %H:%M:%S')}"
                )
        if not any_winner:
            lines.append("No winners yet.")
        return "\n".join(lines)

    def export_to_csv(self, prizes):
        grouped = self.get_winners_by_prize(prizes)
        rows = [w for ws in grouped.values() for w in ws]
        if not rows:
            return False, "No winners to export."
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for w in rows:
                    writer.writerow(
                        [
                            w.user.name,
                            w.user.department,
                            w.prize.name,
                            w.time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and round execution."""

from __future__ import annotations

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"


def _norm(text: str) -> str:
    return " ".join(str(text).lower().split())


def is_cheng_ru(department: str) -> bool:
    key = _norm(department).replace(" ", "").replace("_", "").replace("-", "")
    return "chengru" in key


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryManager:
    def __init__(self, users, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()

    # ------------------------------------------------------------- rules
    def is_fallback_active(self, prize: Prize) -> bool:
        drawer = _norm(prize.associated_drawer or "")
        if drawer != _norm(FALLBACK_DRAWER):
            return False
        if len(self.last_two_draw_results) < 2:
            return False
        return not any(self.last_two_draw_results[-2:])

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        available = [u for u in self.users if u.name not in self.winner_names]
        if self.is_fallback_active(prize):
            return [u for u in available if is_cheng_ru(u.department)]
        drawer = _norm(prize.associated_drawer or "")
        if drawer == _norm(SPECIAL_DRAWER):
            return available
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        if not avoid:
            return available
        preferred = [u for u in available if _norm(u.department) not in avoid]
        if preferred:
            return preferred
        # Avoidance group members remain eligible, but only after others.
        return [u for u in available if _norm(u.department) in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = []
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}")
        if self.is_fallback_active(prize):
            parts.append(
                "Fallback active: prize pool limited to Cheng Ru group members"
            )
        elif _norm(prize.associated_drawer or "") == _norm(SPECIAL_DRAWER):
            parts.append("Special rule: only previous winners are excluded")
        else:
            if prize.avoid_groups:
                parts.append(
                    "Avoidance groups (drawn only after other members): "
                    + ", ".join(prize.avoid_groups)
                )
            parts.append("Previous winners are excluded")
        return "; ".join(parts)

    # ------------------------------------------------------------- drawing
    def mark_winner(self, user: User) -> bool:
        if user.name in self.winner_names:
            return False
        self.winner_names.add(user.name)
        self.winners.append(user)
        return True

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.mark_winner(user)
        return user

    def record_round_result(self, prize: Prize, round_winners: List[User]) -> None:
        if not prize.associated_drawer:
            return
        contains = any(is_cheng_ru(u.department) for u in round_winners)
        self.last_two_draw_results.append(contains)
        self.last_two_draw_results = self.last_two_draw_results[-2:]

    def run_round(
        self,
        prize: Prize,
        result_manager=None,
        input_func: Callable[[str], str] = input,
        output: Callable[[str], None] = print,
    ) -> List[User]:
        output("")
        output(f"===== Prize: {prize.name} | Quantity: {prize.quantity} =====")
        output(f"Rules: {self.describe_rules(prize)}")
        output(
            "Commands: [Enter] draw next, 'pause' to pause, "
            "'continue' to resume, 'terminate' to end this round"
        )
        round_winners: List[User] = []
        i = 1
        terminated = False
        while i <= prize.quantity and not terminated:
            try:
                cmd = input_func(f"Ready to draw the {_ordinal(i)} winner > ")
            except EOFError:
                cmd = ""
            cmd = (cmd or "").strip().lower()
            if cmd == "terminate":
                output("Round terminated.")
                break
            if cmd == "pause":
                output("Lottery paused. Type 'continue' to resume or 'terminate' to end.")
                while True:
                    try:
                        c2 = input_func("(paused) > ")
                    except EOFError:
                        c2 = "continue"
                    c2 = (c2 or "").strip().lower()
                    if c2 == "continue":
                        output("Lottery resumed.")
                        break
                    if c2 == "terminate":
                        output("Round terminated.")
                        terminated = True
                        break
                    output("Paused. Please type 'continue' or 'terminate'.")
                continue
            if cmd not in ("", "continue", "draw"):
                output("Unknown command.")
                continue
            output(f"Currently drawing the {_ordinal(i)} winner...")
            user = self.draw_one(prize)
            if user is None:
                output("No eligible participants left for this prize.")
                break
            if result_manager is not None:
                result_manager.add_winner(user, prize)
            round_winners.append(user)
            output(
                f"Congratulations {user.name} ({user.department}) "
                f"for winning {prize.name}"
            )
            i += 1
        self.record_round_result(prize, round_winners)
        output(f"--- Results of round '{prize.name}' ---")
        if not round_winners:
            output("  (no winners)")
        for n, u in enumerate(round_winners, 1):
            output(f"  {n}. {u.name} ({u.department})")
        return round_winners

"""Command-line interface for the lottery system."""

from __future__ import annotations

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager

MENU = """
========== Team Building Smart Lottery ==========
 1. Load default list
 2. Upload custom list (txt)
 3. View group information
 4. Load preset prizes
 5. Add custom prize
 6. Configure avoidance groups for a prize
 7. Reorder prizes
 8. View prizes
 9. Start lottery
10. View complete winner list
11. Export results
 0. Exit
"""


class App:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.result_manager = ResultManager()

    def ask(self, prompt):
        try:
            return input(prompt)
        except EOFError:
            return "0"

    def groups(self):
        return list(self.list_manager.get_group_info(self.users).keys())

    def _accept(self, users, errors):
        valid, dups = self.list_manager.validate_users(users)
        for e in errors:
            print(e)
        for d in dups:
            print(f"Duplicate name: {d.name} ({d.department})")
        if errors or dups:
            choice = self.ask(
                "Invalid data found. 'r' = remove invalid data and continue, "
                "'t' = terminate upload: "
            ).strip().lower()
            if choice != "r":
                print("Upload terminated.")
                return
        if not valid:
            print("No valid users loaded.")
            return
        self.users = valid
        self.list_manager.users = valid
        print(f"Loaded {len(valid)} valid participants.")

    def select_groups(self):
        groups = self.groups()
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = self.ask("Avoidance groups (numbers or names, comma separated, empty for none): ")
        result = []
        for token in raw.replace(";", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                result.append(groups[int(token) - 1])
            else:
                result.append(token)
        return result

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(
                f"  {i}. {p.name} x{p.quantity} | avoid: "
                f"{', '.join(p.avoid_groups) or 'none'} | drawer: "
                f"{p.associated_drawer or 'none'}"
            )

    def run(self):
        while True:
            print(MENU)
            c = self.ask("Select: ").strip()
            if c == "0":
                print("Bye.")
                return
            elif c == "1":
                self._accept(*self.list_manager.load_default_list())
            elif c == "2":
                path = self.ask("Path of txt list: ").strip().strip('"')
                self._accept(*self.list_manager.load_custom_list(path))
            elif c == "3":
                info = self.list_manager.get_group_info(self.users)
                if not info:
                    print("No list loaded.")
                for g, n in info.items():
                    print(f"  {g}: {n} people")
            elif c == "4":
                for e in self.prize_manager.load_default_prizes():
                    print(e)
                self.show_prizes()
            elif c == "5":
                name = self.ask("Prize name: ")
                qty = self.ask("Quantity: ")
                drawer = self.ask("Associated lottery person (empty for none): ").strip() or None
                groups = self.select_groups()
                try:
                    self.prize_manager.add_prize(name, qty, groups, drawer)
                    print("Prize added.")
                except ValueError as exc:
                    print(f"Error: {exc}")
            elif c == "6":
                self.show_prizes()
                idx = self.ask("Prize number: ").strip()
                if not idx.isdigit():
                    print("Invalid number.")
                    continue
                groups = self.select_groups()
                if self.prize_manager.set_avoid_groups(int(idx) - 1, groups):
                    print("Avoidance groups updated.")
                else:
                    print("Invalid prize number.")
            elif c == "7":
                self.show_prizes()
                raw = self.ask("New order (e.g. 3,2,1): ")
                try:
                    order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
                except ValueError:
                    order = []
                print("Reordered." if self.prize_manager.reorder_prizes(order) else "Invalid order.")
            elif c == "8":
                self.show_prizes()
            elif c == "9":
                prizes = self.prize_manager.get_prizes()
                if not self.users or not prizes:
                    print("Please load a list and configure prizes first.")
                    continue
                self.result_manager = ResultManager()
                lm = LotteryManager(self.users)
                for p in prizes:
                    lm.run_round(p, self.result_manager, input_func=self.ask)
                print(self.result_manager.format_summary(prizes))
            elif c == "10":
                print(self.result_manager.format_summary(self.prize_manager.get_prizes()))
            elif c == "11":
                ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
                print(msg)
            else:
                print("Invalid option.")


def main():
    App().run()


if __name__ == "__main__":
    main()

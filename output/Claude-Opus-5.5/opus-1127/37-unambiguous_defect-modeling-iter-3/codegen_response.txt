from __future__ import annotations

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass(frozen=True)
class User:
    """A lottery participant."""
    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None
    level: Optional[int] = None  # 1 = highest level; None = unspecified

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None

    def describe(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "Not specified"
        return (f"{self.name} | Quantity: {self.quantity} | "
                f"Avoid groups: {avoid} | Drawer: {drawer}")

class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.load_mode: str = ""
        self.source_path: str = ""

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        self.load_mode = "default"
        self.source_path = "<built-in>"
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        self.load_mode = "custom"
        self.source_path = str(path)
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if path is None or str(path).strip() == "":
            return [], ["File Error: no file path provided"]
        path = str(path).strip()
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"File Error: cannot read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: List[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            normalized = line.replace("，", ",").replace("\t", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if idx == 1 and len(parts) == 2 and parts[0].lower() == "name" \
                    and parts[1].lower() in ("department", "group", "dept"):
                continue  # header line
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' - expected format 'Name,Department'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # --------------------------------------------------------------- validation
    @staticmethod
    def validate_users(users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    @staticmethod
    def get_group_info(users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    # name, quantity, avoid groups, drawer, level
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian", 3),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo", 2),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi", 1),

class PrizeManager:
    """Holds the ordered list of prizes (lottery round sequence)."""

    def __init__(self) -> None:
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer, level in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer, level=level)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None,
                  level: Optional[int] = None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else name
        if not name or not isinstance(name, str):
            raise ValueError("Prize name cannot be empty")
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists")
        try:
            if isinstance(quantity, bool):
                raise ValueError
            qty = int(quantity)
            if isinstance(quantity, float) and quantity != qty:
                raise ValueError
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError(f"Prize quantity must be positive: {quantity}")
        prize = Prize(name, qty, list(avoid_groups or []), associated_drawer, level)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self._prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine and draw execution."""
from __future__ import annotations

import random
from typing import List, Optional

from lottery_system.models import Prize, User


def _norm(text) -> str:
    return " ".join(str(text or "").strip().lower().split())


def _norm_group(text) -> str:
    t = _norm(text)
    if t.endswith(" group"):
        t = t[: -len(" group")].strip()
    return t


class LotteryManager:
    RELAXED_DRAWER = "Li Bin"
    TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
    TARGET_DRAWER = "Ma Jiaqi"
    PROTECTED_GROUP = "Cheng Ru"

    def __init__(self, users, rng: Optional[random.Random] = None) -> None:
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()

    # --------------------------------------------------------------- helpers
    def is_protected_member(self, user: User) -> bool:
        return _norm_group(user.department) == _norm_group(self.PROTECTED_GROUP)

    def is_relaxed(self, prize: Prize) -> bool:
        return _norm(prize.associated_drawer) == _norm(self.RELAXED_DRAWER)

    def is_fallback_triggered(self, prize: Prize) -> bool:
        if _norm(prize.associated_drawer) != _norm(self.TARGET_DRAWER):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def has_won(self, user: User) -> bool:
        return user.name in self.winner_names

    # --------------------------------------------------------------- engine
    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        not_won = [u for u in self.users if not self.has_won(u)]
        if self.is_fallback_triggered(prize):
            protected = [u for u in not_won if self.is_protected_member(u)]
            if protected:
                return protected
        if self.is_relaxed(prize):
            return not_won
        avoid = {_norm_group(g) for g in (prize.avoid_groups or [])}
        return [u for u in not_won if _norm_group(u.department) not in avoid]

    def get_rule_description(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        if self.is_fallback_triggered(prize):
            parts.append(f"Fallback triggered: pool limited to '{self.PROTECTED_GROUP}' group "
                         "members (avoidance ignored)")
        elif self.is_relaxed(prize):
            parts.append(f"Relaxed rule for drawer {self.RELAXED_DRAWER}: avoidance groups ignored")
        elif prize.avoid_groups:
            parts.append("Avoiding groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")
        if prize.associated_drawer:
            parts.append(f"Drawer: {prize.associated_drawer}")
        return "; ".join(parts)

    def mark_winner(self, user: User) -> bool:
        if self.has_won(user):
            return False
        self.winners.append(user)
        self.winner_names.add(user.name)
        return True

    def draw_winner(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.mark_winner(user)
        return user

    def draw_round(self, prize: Prize) -> List[User]:
        result = []
        for _ in range(prize.quantity):
            u = self.draw_winner(prize)
            if u is None:
                break
            result.append(u)
        self.finish_round(prize, result)
        return result

    def finish_round(self, prize: Prize, round_winners) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in self.TRIGGER_DRAWERS}:
            self.last_two_draw_results.append(
                any(self.is_protected_member(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(self.TARGET_DRAWER):
            self.last_two_draw_results = []

"""Winner records, summaries and export."""
from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize: Prize
    win_time: datetime = field(default_factory=datetime.now)
    order: int = 0

    @property
    def prize_name(self) -> str:
        return self.prize.name if isinstance(self.prize, Prize) else str(self.prize)


def _sorted_prizes(prizes) -> List[Prize]:
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda t: (t[1].level is None,
                                t[1].level if t[1].level is not None else 0, t[0]))
    return [p for _, p in indexed]


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []
        self._counter = 0

    @property
    def total_winners(self) -> int:
        return len(self.winners)

    def add_winner(self, user: User, prize: Prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self._counter += 1
        self.winners.append(Winner(user, prize, datetime.now(), self._counter))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for prize in _sorted_prizes(prizes):
            ws = [w for w in self.winners if w.prize_name == prize.name]
            ws.sort(key=lambda w: (w.win_time, w.order))
            result[prize.name] = ws
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Prize", "Name", "Department", "Winning Order", "Win Time"])
                for prize_name, ws in grouped.items():
                    for w in ws:
                        writer.writerow([prize_name, w.user.name, w.user.department, w.order,
                                         w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""
from __future__ import annotations

from typing import List, Optional

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryCLI:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_manager: Optional[LotteryManager] = None
        self.drawn_prizes = set()

    @staticmethod
    def ask(prompt: str) -> str:
        try:
            return input(prompt).strip()
        except EOFError:
            return "0"

    # ------------------------------------------------------------ list module
    def _set_users(self, users: List[User]) -> None:
        self.users = users
        self.lottery_manager = None
        self.result_manager = ResultManager()
        self.drawn_prizes = set()
        print(f"{len(users)} valid participants loaded.")

    def _handle_loaded(self, users, errors) -> None:
        if errors:
            print("The following problems were detected:")
            for e in errors:
                print("  " + e)
            if not users:
                print("No valid data. Upload terminated.")
                return
            c = self.ask("Remove invalid data and continue (r) or terminate upload (t)? ").lower()
            if c != "r":
                print("Upload terminated.")
                return
        valid, dups = self.list_manager.validate_users(users)
        if dups:
            print("Duplicate names detected:")
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            c = self.ask("Remove duplicates and continue (r) or terminate upload (t)? ").lower()
            if c != "r":
                print("Upload terminated.")
                return
        self._set_users(valid)

    def list_menu(self) -> None:
        while True:
            print("\n--- List Management ---")
            print("1. Load built-in default list")
            print("2. Upload custom txt list (format per line: Name,Department)")
            print("3. View groups")
            print("4. View participants")
            print("0. Back")
            c = self.ask("Choice: ")
            if c == "1":
                self._handle_loaded(*self.list_manager.load_default_list())
            elif c == "2":
                path = self.ask("txt file path: ")
                self._handle_loaded(*self.list_manager.load_custom_list(path))
            elif c == "3":
                info = self.list_manager.get_group_info(self.users)
                if not info:
                    print("No list loaded.")
                for i, (g, n) in enumerate(info.items(), 1):
                    print(f"  {i}. {g}: {n} people")
            elif c == "4":
                if not self.users:
                    print("No list loaded.")
                for u in self.users:
                    print(f"  {u.name} ({u.department})")
            elif c == "0":
                return
            else:
                print("Invalid choice.")

    # ----------------------------------------------------------- prize module
    def _select_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            raw = self.ask("Avoidance groups (numbers or names, comma separated, empty for none): ")
        else:
            raw = self.ask("Avoidance group names (comma separated, empty for none): ")
        result = []
        for token in raw.replace("，", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                name = groups[int(token) - 1]
            else:
                name = token
            if name not in result:
                result.append(name)
        return result

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            mark = " [drawn]" if p.name in self.drawn_prizes else ""
            print(f"  {i}. {p.describe()}{mark}")

    def prize_menu(self) -> None:
        while True:
            print("\n--- Prize Configuration ---")
            print("1. Load preset prize library")
            print("2. Add custom prize")
            print("3. View prizes (round order)")
            print("4. Reorder prizes")
            print("0. Back")
            c = self.ask("Choice: ")
            if c == "1":
                errors = self.prize_manager.load_default_prizes()
                for e in errors:
                    print("  " + e)
                self.show_prizes()
            elif c == "2":
                name = self.ask("Prize name: ")
                qty = self.ask("Quantity: ")
                avoid = self._select_groups()
                drawer = self.ask("Associated lottery person (empty for none): ") or None
                try:
                    self.prize_manager.add_prize(name, qty, avoid, drawer)
                    print("Prize added.")
                except ValueError as exc:
                    print(f"Error: {exc}")
            elif c == "3":
                self.show_prizes()
            elif c == "4":
                self.show_prizes()
                raw = self.ask("New order as serial numbers (e.g. 3,1,2): ")
                try:
                    order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
                except ValueError:
                    order = []
                if self.prize_manager.reorder_prizes(order):
                    print("Prizes reordered.")
                    self.show_prizes()
                else:
                    print("Invalid order.")
            elif c == "0":
                return
            else:
                print("Invalid choice.")

    # --------------------------------------------------------------- lottery
    def run_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = [p for p in self.prize_manager.get_prizes() if p.name not in self.drawn_prizes]
        if not prizes:
            print("No prizes left to draw. Please configure prizes.")
            return
        if self.lottery_manager is None:
            self.lottery_manager = LotteryManager(self.users)
        lm = self.lottery_manager
        auto = False
        for idx, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {idx}: {prize.name}")
            print(f"Quantity: {prize.quantity}")
            print(f"Rules: {lm.get_rule_description(prize)}")
            print("=" * 60)
            round_winners: List[User] = []
            while len(round_winners) < prize.quantity:
                n = len(round_winners) + 1
                if not auto:
                    cmd = self.ask(f"Enter to draw the {ordinal(n)} winner "
                                   "(commands: pause / terminate / auto): ").lower()
                    if cmd == "pause":
                        print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                        stop = False
                        while True:
                            c = self.ask("> ").lower()
                            if c == "continue":
                                print("Lottery resumed.")
                                break
                            if c in ("terminate", "0"):
                                stop = True
                                break
                            print("Paused. Type 'continue' or 'terminate'.")
                        if stop:
                            print("Current round terminated.")
                            break
                        continue
                    if cmd in ("terminate", "0"):
                        print("Current round terminated.")
                        break
                    if cmd == "auto":
                        auto = True
                    elif cmd not in ("", "draw", "continue"):
                        print("Unknown command.")
                        continue
                print(f"Currently drawing the {ordinal(n)} winner...")
                user = lm.draw_winner(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            lm.finish_round(prize, round_winners)
            self.drawn_prizes.add(prize.name)
            print(f"\nRound {idx} results - {prize.name}:")
            if not round_winners:
                print("  No winners.")
            for i, u in enumerate(round_winners, 1):
                print(f"  {i}. {u.name} ({u.department})")
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for prize_name, ws in grouped.items():
            if not ws:
                continue
            print(f"[{prize_name}]")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {prize_name} "
                      f"@ {w.win_time.strftime('%H:%M:%S')}")

    # ------------------------------------------------------------------ main
    def run(self) -> None:
        print("Welcome to the Team Building Smart Lottery System")
        while True:
            print("\n===== Main Menu =====")
            print("1. List management")
            print("2. Prize configuration")
            print("3. Start lottery")
            print("4. View winner summary")
            print("5. Export results")
            print("0. Exit")
            c = self.ask("Choice: ")
            if c == "1":
                self.list_menu()
            elif c == "2":
                self.prize_menu()
            elif c == "3":
                self.run_lottery()
            elif c == "4":
                self.show_summary()
            elif c == "5":
                ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
                print(msg)
            elif c == "0":
                print("Goodbye!")
                return
            else:
                print("Invalid choice.")


def main() -> None:
    LotteryCLI().run()


if __name__ == "__main__":
    main()

"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
``DATA_FILE_PATH`` is computed from this file's location, so it is valid
regardless of the current working directory.
"""

import json
import os

#: Absolute path of the bundled subway network data file.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON data file (defaults to ``DATA_FILE_PATH``)
    :return: the parsed JSON content (a dict with ``stations`` and ``connections``)
    :raises FileNotFoundError: if the file does not exist
    :raises ValueError: if the file is not valid JSON
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid JSON in subway data file '{path}': {exc}") from exc
    return data

"""
Core data structures: weighted undirected graph (``Graph``), its edges
(``Edge``) and a heap based priority queue (``MinPQ``).
"""

import heapq
import itertools


class Edge:
    """A weighted, undirected connection between two stations on one line.

    ``source`` and ``dest`` are station numbers. ``weight`` always equals
    ``distance`` (meters).
    """

    def __init__(self, source, dest, distance, line):
        if distance is None or distance < 0:
            raise ValueError(f"Edge distance must be non-negative, got {distance!r}")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Weight of the edge (equal to its distance in meters)."""
        return self.distance

    def either(self):
        """Return one endpoint of the edge."""
        return self.source

    def other(self, v):
        """Return the endpoint of the edge that is not ``v``."""
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError(f"Vertex {v!r} is not an endpoint of {self!r}")

    def connects(self, a, b):
        """Whether this edge joins ``a`` and ``b`` (in any direction)."""
        return (self.source == a and self.dest == b) or (self.source == b and self.dest == a)

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller; the graph simply stores
    whatever vertex identifiers appear in the edges it receives.
    """

    def __init__(self):
        self.adj = {}             # station number -> list[Edge]
        self.edges = []           # every edge once
        self.station_lines = {}   # station name -> list of line names

    # ----- vertices -------------------------------------------------
    def add_vertex(self, v):
        """Register a vertex (no-op if it already exists)."""
        self.adj.setdefault(v, [])

    def vertices(self):
        return list(self.adj.keys())

    @property
    def vertex_count(self):
        return len(self.adj)

    @property
    def edge_count(self):
        return len(self.edges)

    # ----- station metadata -----------------------------------------
    def set_station_lines(self, station_name, lines):
        """Set the lines a station belongs to (duplicates removed, order kept)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines or []))

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        """A station served by more than one line is a transfer station."""
        return len(self.station_lines.get(station_name, [])) > 1

    # ----- edges ------------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge (stored in the adjacency of both endpoints)."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self.adj[edge.dest].append(edge)
        self.edges.append(edge)

    def adjacent(self, v):
        """Edges incident to ``v``."""
        return list(self.adj.get(v, []))

    def degree(self, v):
        return len(self.adj.get(v, []))

    def find_edges(self, a, b, line=None):
        """All edges between ``a`` and ``b`` (optionally restricted to a line)."""
        return [e for e in self.adj.get(a, [])
                if e.connects(a, b) and (line is None or e.line == line)]

    def remove_edges_between(self, a, b, line=None):
        """Remove edges between ``a`` and ``b``; return number removed."""
        doomed = self.find_edges(a, b, line)
        for e in doomed:
            for v in (e.source, e.dest):
                lst = self.adj.get(v, [])
                if e in lst:
                    lst.remove(e)
            if e in self.edges:
                self.edges.remove(e)
        return len(doomed)

    def __repr__(self):
        return f"Graph(V={self.vertex_count}, E={self.edge_count})"


class MinPQ:
    """Binary-heap based minimum priority queue.

    Items with equal priority are returned in insertion order, and items
    themselves never need to be comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    insert = push

    def pop(self):
        """Remove and return ``(priority, item)`` with the smallest priority."""
        if not self._heap:
            raise IndexError("pop from an empty priority queue")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    del_min = pop

    def peek(self):
        if not self._heap:
            raise IndexError("peek from an empty priority queue")
        priority, _, item = self._heap[0]
        return priority, item

    def is_empty(self):
        return not self._heap

    @property
    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

"""
Path calculation based on Dijkstra's algorithm.

The search runs on an expanded state space ``(station, current line,
transfers)`` so that transfers can be counted exactly, limited by
``max_transfers`` and minimised in the ``least_transfer`` mode.

Modes:
  * ``shortest_distance`` - minimise total distance (ties: fewer transfers)
  * ``least_transfer``    - minimise transfers (ties: shorter distance)
  * ``comprehensive``     - minimise distance + transfers * transfer penalty
"""

import math

from models import MinPQ

MODES = ("shortest_distance", "least_transfer", "comprehensive")

#: Distance-equivalent penalty (meters) of one transfer in comprehensive mode.
TRANSFER_PENALTY = 3000
#: Cost multiplier applied to edges on preferred lines.
PREFERRED_LINE_FACTOR = 0.8


class PathfindingConfig:
    """Configuration of a path query."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None,
                 transfer_penalty=TRANSFER_PENALTY, preference_factor=PREFERRED_LINE_FACTOR):
        if mode not in MODES:
            raise ValueError(f"Unknown mode '{mode}', expected one of {', '.join(MODES)}")
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        if isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = [l for l in (preferred_lines or []) if l]
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class _State:
    __slots__ = ("cost", "dist", "eff", "transfers", "parent", "edge")

    def __init__(self, cost, dist, eff, transfers, parent, edge):
        self.cost = cost
        self.dist = dist
        self.eff = eff
        self.transfers = transfers
        self.parent = parent
        self.edge = edge


class ShortestPath:
    """Single-source optimal paths on a ``Graph`` according to a config."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.mode = self.config.mode
        self.max_transfers = self.config.max_transfers
        self.preferred_lines = set(self.config.preferred_lines)
        self.dist_to = {}
        self._states = {}
        self._best = {}
        self._run()

    # ----- internals ----------------------------------------------------
    def _cost(self, eff, transfers):
        if self.mode == "least_transfer":
            return (transfers, eff)
        if self.mode == "comprehensive":
            return (eff + transfers * self.config.transfer_penalty, transfers)
        return (eff, transfers)

    def _run(self):
        limit = self.max_transfers
        start = (self.source, None, 0)
        self._states[start] = _State(self._cost(0, 0), 0, 0, 0, None, None)
        pq = MinPQ()
        pq.push(self._states[start].cost, start)
        settled = set()
        adj = self.graph.adj

        while pq:
            _, key = pq.pop()
            if key in settled:
                continue
            settled.add(key)
            rec = self._states[key]
            v, line, _t = key
            for e in adj.get(v, []):
                w = e.other(v)
                nt = rec.transfers + (1 if (line is not None and e.line != line) else 0)
                if limit is not None and nt > limit:
                    continue
                nkey = (w, e.line, nt if limit is not None else 0)
                if nkey in settled:
                    continue
                factor = self.config.preference_factor if e.line in self.preferred_lines else 1
                neff = rec.eff + e.weight * factor
                ncost = self._cost(neff, nt)
                old = self._states.get(nkey)
                if old is None or ncost < old.cost:
                    self._states[nkey] = _State(ncost, rec.dist + e.weight, neff, nt, key, e)
                    pq.push(ncost, nkey)

        for key, rec in self._states.items():
            v = key[0]
            cur = self._best.get(v)
            if cur is None or rec.cost < self._states[cur].cost:
                self._best[v] = key
        self._best[self.source] = start
        for v, key in self._best.items():
            self.dist_to[v] = self._states[key].dist

    # ----- queries --------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return ``(stations, edges)`` from source to target, or ``([], [])``."""
        if target not in self._best:
            return [], []
        edges = []
        key = self._best[target]
        while key is not None:
            rec = self._states[key]
            if rec.edge is not None:
                edges.append(rec.edge)
            key = rec.parent
        edges.reverse()
        stations = [self.source]
        cur = self.source
        for e in edges:
            cur = e.other(cur)
            stations.append(cur)
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Transfers on the chosen path to ``target`` (-1 if unreachable)."""
        if target not in self._best:
            return -1
        return self._states[self._best[target]].transfers

    def get_lines(self, target):
        """Ordered list of distinct lines used on the path to ``target``."""
        _, edges = self.path_with_edges(target)
        return list(dict.fromkeys(e.line for e in edges))

"""
Beijing Subway fare calculation.

Tiers (left-closed, right-open, in meters):
  [0, 6000)      -> 3 yuan
  [6000, 12000)  -> 4 yuan
  [12000, 22000) -> 5 yuan
  [22000, 32000) -> 6 yuan
  >= 32000       -> 7 yuan + 1 yuan per additional full 20 km
Fractional distances are rounded up to the next whole meter.
"""

import math

#: (min meters inclusive, max meters exclusive, fare yuan)
FARE_TIERS = [
    (0, 6000, 3),
    (6000, 12000, 4),
    (12000, 22000, 5),
    (22000, 32000, 6),
]
EXTENDED_START = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_INCREMENT_DISTANCE = 20000
EXTENDED_INCREMENT_FARE = 1

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError(f"Distance must be a number, got {type(distance).__name__}")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters."""
    _validate_distance(distance)
    meters = math.ceil(distance)
    for low, high, fare in FARE_TIERS:
        if low <= meters < high:
            return fare
    extra = (meters - EXTENDED_START) // EXTENDED_INCREMENT_DISTANCE
    return EXTENDED_BASE_FARE + extra * EXTENDED_INCREMENT_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes (1.5 min/km, 5 min per transfer)."""
    _validate_distance(distance)
    if transfer_count is None or transfer_count < 0:
        transfer_count = 0
    return float(distance / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER)

def _norm(text):
    return re.sub(r"[\s\-_'’.,()（）]", "", str(text)).casefold()


def _line_matches(data_line, alias):
    d, a = _norm(data_line), _norm(alias)
    if d == a:
        return True
    return d.startswith(a) and not d[len(a)].isdigit()


@dataclass
class IntegrityReport:
    """Result of a data integrity check."""
    station_coverage_valid: bool = True
    connection_integrity_valid: bool = True
    distance_accuracy_valid: bool = True
    errors: list = field(default_factory=list)
    warnings: list = field(default_factory=list)
    stats: dict = field(default_factory=dict)

    @property
    def is_valid(self):
        return (self.station_coverage_valid and self.connection_integrity_valid
                and self.distance_accuracy_valid)

    def to_text(self):
        def flag(ok):
            return "PASS" if ok else "FAIL"
        lines = ["Data integrity check report", "=" * 40]
        for k, v in self.stats.items():
            lines.append(f"{k}: {v}")
        lines.append("-" * 40)
        lines.append(f"Station data coverage: {flag(self.station_coverage_valid)}")
        lines.append(f"Connection integrity: {flag(self.connection_integrity_valid)}")
        lines.append(f"Distance data accuracy: {flag(self.distance_accuracy_valid)}")
        if self.errors:
            lines.append("Errors:")
            lines.extend(f"  - {e}" for e in self.errors)
        if self.warnings:
            lines.append("Warnings:")
            lines.extend(f"  - {w}" for w in self.warnings)
        lines.append("Overall result: " + ("PASS" if self.is_valid else "FAIL"))
        return "\n".join(lines)


class SubwayNetwork:
    """Beijing Subway network: stations, lines, connections and graph model."""

    def __init__(self, data=None, data_file=DATA_FILE_PATH):
        self.data_source_file = data_file
        if data is None:
            data = load_subway_data(data_file)
        if not isinstance(data, dict):
            raise ValueError("Subway data must be a JSON object")
        self.station_order = []
        self.stations = {}           # name -> list of lines
        for st in data.get("stations", []) or []:
            name = str(st.get("name", "")).strip()
            if not name:
                continue
            if name not in self.stations:
                self.station_order.append(name)
                self.stations[name] = []
            for line in st.get("lines", []) or []:
                if line not in self.stations[name]:
                    self.stations[name].append(line)
        self.connections = []
        for c in data.get("connections", []) or []:
            self.connections.append({
                "source": str(c.get("source", "")).strip(),
                "dest": str(c.get("dest", "")).strip(),
                "distance": c.get("distance"),
                "line": c.get("line"),
            })
        self.rebuild()

    # ----- model building ---------------------------------------------------
    def rebuild(self):
        """(Re)build station numbering, the Graph and the adjacency dictionary."""
        self.station_to_number = {}
        self.number_to_station = {}
        for name in self.station_order:
            self._assign_number(name)
        for c in self.connections:
            for name in (c["source"], c["dest"]):
                if name and name not in self.station_to_number:
                    self._assign_number(name)

        self.graph = Graph()
        for name, num in self.station_to_number.items():
            self.graph.add_vertex(num)
            self.graph.set_station_lines(name, self.stations.get(name, []))

        self.adjacency = {name: [] for name in self.station_to_number}
        self.lines = {}
        for c in self.connections:
            s, d, dist, line = c["source"], c["dest"], c["distance"], c["line"]
            if not s or not d or not isinstance(dist, (int, float)) or dist < 0:
                continue
            self.graph.add_edge(Edge(self.station_to_number[s], self.station_to_number[d], dist, line))
            self.adjacency[s].append({"station": d, "distance": dist, "line": line})
            self.adjacency[d].append({"station": s, "distance": dist, "line": line})
            seq = self.lines.setdefault(line, [])
            for n in (s, d):
                if n not in seq:
                    seq.append(n)
        # also register lines that only appear on stations
        for name, lines in self.stations.items():
            for line in lines:
                self.lines.setdefault(line, [])
                if name not in self.lines[line]:
                    self.lines[line].append(name)
        self._lookup = {}
        for name in self.station_to_number:
            for key in self._lookup_keys(name):
                self._lookup.setdefault(key, name)

    def _assign_number(self, name):
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name

    @staticmethod
    def _lookup_keys(name):
        n = _norm(name)
        keys = [n]
        for suffix in ("station", "站"):
            if n.endswith(suffix) and len(n) > len(suffix):
                keys.append(n[: -len(suffix)])
        return keys

    @property
    def graph_dict(self):
        return self.adjacency

    # ----- lookup -------------------------------------------------------------
    def find_station(self, name):
        """Return the official station name for user input, or None."""
        if name is None:
            return None
        name = str(name).strip()
        if name in self.station_to_number:
            return name
        for key in self._lookup_keys(name):
            if key in self._lookup:
                return self._lookup[key]
        return None

    def station_exists(self, name):
        return self.find_station(name) is not None

    def get_station_lines(self, name):
        return list(self.stations.get(name, []))

    def is_transfer_station(self, name):
        return len(self.stations.get(name, [])) > 1

    def transfer_stations(self):
        return [n for n in self.station_order if self.is_transfer_station(n)]

    def all_line_names(self):
        return sorted(l for l in self.lines if l)

    def find_line(self, name):
        for line in self.lines:
            if line and _line_matches(line, name):
                return line
        return None

    # ----- data updates ------------------------------------------------------
    def add_station(self, name, lines=None):
        name = name.strip()
        if not name:
            raise ValueError("Station name cannot be empty")
        if name not in self.stations:
            self.station_order.append(name)
            self.stations[name] = []
        for line in lines or []:
            if line not in self.stations[name]:
                self.stations[name].append(line)
        self.rebuild()

    def add_line(self, line_name, station_names, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        for d in distances:
            if not isinstance(d, (int, float)) or d <= 0:
                raise ValueError(f"Invalid distance {d!r}")
        for name in station_names:
            if name not in self.stations:
                self.station_order.append(name)
                self.stations[name] = []
            if line_name not in self.stations[name]:
                self.stations[name].append(line_name)
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.connections.append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        if name not in self.stations:
            raise KeyError(f"Station '{name}' does not exist")
        if new_name and new_name != name:
            if new_name in self.stations:
                raise ValueError(f"Station '{new_name}' already exists")
            self.stations[new_name] = self.stations.pop(name)
            self.station_order[self.station_order.index(name)] = new_name
            for c in self.connections:
                if c["source"] == name:
                    c["source"] = new_name
                if c["dest"] == name:
                    c["dest"] = new_name
            name = new_name
        if lines is not None:
            self.stations[name] = list(dict.fromkeys(lines))
        self.rebuild()

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance between two adjacent stations; returns count updated."""
        if not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        count = 0
        for c in self.connections:
            if {c["source"], c["dest"]} == {station_a, station_b} and (line is None or c["line"] == line):
                c["distance"] = distance
                count += 1
        if count == 0:
            raise KeyError(f"No connection between '{station_a}' and '{station_b}'")
        self.rebuild()
        return count

    def to_dict(self):
        return {
            "stations": [{"name": n, "lines": list(self.stations[n])} for n in self.station_order],
            "connections": [dict(c) for c in self.connections],
        }

    def save(self, path=None):
        path = path or self.data_source_file
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, ensure_ascii=False, indent=2)

    # ----- integrity check --------------------------------------------------
    def check_integrity(self):
        rep = IntegrityReport()
        rep.stats = {
            "Stations": len(self.station_to_number),
            "Connections": len(self.connections),
            "Lines": len([l for l in self.lines if l]),
            "Transfer stations": len(self.transfer_stations()),
        }
        if not self.station_order:
            rep.station_coverage_valid = False
            rep.errors.append("No station data found")

        # coverage of main lines
        data_lines = [l for l in self.lines if l]
        for main, aliases in MAIN_LINES.items():
            if not any(_line_matches(dl, a) for dl in data_lines for a in aliases):
                rep.warnings.append(f"Main line '{main}' not found in data")
        for main, reps in REPRESENTATIVE_STATIONS.items():
            for aliases in reps:
                if not any(self.find_station(a) for a in aliases):
                    rep.warnings.append(
                        f"Representative station '{aliases[0]}' of {main} not found")

        # connection integrity
        seen = {}
        for i, c in enumerate(self.connections):
            s, d, line = c["source"], c["dest"], c["line"]
            label = f"connection #{i + 1} ({s} - {d})"
            if not s or not d:
                rep.connection_integrity_valid = False
                rep.errors.append(f"{label}: missing endpoint")
                continue
            if s == d:
                rep.connection_integrity_valid = False
                rep.errors.append(f"{label}: connects a station to itself")
            for n in (s, d):
                if n not in self.stations:
                    rep.connection_integrity_valid = False
                    rep.errors.append(f"{label}: station '{n}' not in station list")
                elif line and self.stations[n] and line not in self.stations[n]:
                    rep.warnings.append(f"{label}: station '{n}' is not listed on line '{line}'")
            if not line:
                rep.warnings.append(f"{label}: missing line name")
            dist = c["distance"]
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                rep.distance_accuracy_valid = False
                rep.errors.append(f"{label}: invalid distance {dist!r}")
            else:
                if not (MIN_REASONABLE_DISTANCE <= dist <= MAX_REASONABLE_DISTANCE):
                    rep.warnings.append(f"{label}: unusual distance {dist} m")
                key = (frozenset((s, d)), line)
                if key in seen and seen[key] != dist:
                    rep.distance_accuracy_valid = False
                    rep.errors.append(f"{label}: inconsistent duplicate distance "
                                      f"({seen[key]} vs {dist})")
                seen.setdefault(key, dist)

        isolated = [n for n in self.station_order if not self.adjacency.get(n)]
        for n in isolated:
            rep.connection_integrity_valid = False
            rep.errors.append(f"Station '{n}' has no connections")

        # connectivity
        if self.station_to_number:
            start = next(iter(self.station_to_number))
            visited = {start}
            stack = [start]
            while stack:
                v = stack.pop()
                for nb in self.adjacency.get(v, []):
                    if nb["station"] not in visited:
                        visited.add(nb["station"])
                        stack.append(nb["station"])
            unreachable = len(self.station_to_number) - len(visited)
            if unreachable:
                rep.warnings.append(f"{unreachable} station(s) are not reachable from '{start}'")
        return rep

"""
Route building, path visualization and export (text / JSON).

import json
from dataclasses import dataclass, field

from fare_calculator import Sum_money, calculate_estimated_time


@dataclass
class RouteSegment:
    order: int
    line: str
    stations: list
    distance: int

    @property
    def start(self):
        return self.stations[0]

    @property
    def end(self):
        return self.stations[-1]


@dataclass
class Route:
    stations: list
    segments: list = field(default_factory=list)
    total_distance: int = 0
    transfer_count: int = 0
    transfer_stations: list = field(default_factory=list)
    lines: list = field(default_factory=list)
    fare: int = 0
    estimated_time: float = 0.0


def build_route(path, edges, number_to_station):
    """Build a Route from a station-number path and its edges."""
    names = [number_to_station.get(n, str(n)) for n in path]
    segments = []
    for i, e in enumerate(edges):
        to_name = names[i + 1]
        if segments and segments[-1].line == e.line:
            segments[-1].stations.append(to_name)
            segments[-1].distance += e.distance
        else:
            segments.append(RouteSegment(len(segments) + 1, e.line, [names[i], to_name], e.distance))
    total = sum(e.distance for e in edges)
    transfers = max(len(segments) - 1, 0)
    route = Route(
        stations=names,
        segments=segments,
        total_distance=total,
        transfer_count=transfers,
        transfer_stations=[s.start for s in segments[1:]],
        lines=list(dict.fromkeys(s.line for s in segments)),
    )
    route.fare = Sum_money(total)
    route.estimated_time = calculate_estimated_time(total, transfers)
    return route


def format_distance(meters):
    """Meters for short distances, kilometers for long ones (both shown)."""
    km = meters / 1000.0
    if meters < 1000:
        return f"{meters:g} meters ({km:.2f} km)"
    return f"{km:.2f} km ({meters:g} meters)"


def format_text_report(route, start, end, mode=None):
    out = ["=" * 50, f"Route: {start} -> {end}"]
    if mode:
        out.append(f"Mode: {mode}")
    out.append("-" * 50)
    out.append(f"Station sequence ({len(route.stations)} stations):")
    out.append("  " + " -> ".join(route.stations))
    out.append("Line segments:")
    for i, seg in enumerate(route.segments):
        stops = len(seg.stations) - 1
        out.append(f"  {seg.order}. [{seg.line}] {seg.start} -> {seg.end} "
                   f"({stops} stop{'s' if stops != 1 else ''}, {format_distance(seg.distance)})")
        out.append("       " + " -> ".join(seg.stations))
        if i + 1 < len(route.segments):
            out.append(f"     Transfer at {seg.end}: {seg.line} -> {route.segments[i + 1].line}")
    out.append(f"Transfer count: {route.transfer_count}")
    out.append("Transfer stations: " + (", ".join(route.transfer_stations) or "None"))
    out.append("Lines involved: " + ", ".join(route.lines))
    out.append(f"Total distance: {format_distance(route.total_distance)}")
    out.append(f"Estimated time: {route.estimated_time:.1f} minutes")
    out.append(f"Fare: {route.fare} yuan")
    out.append("=" * 50)
    return "\n".join(out)


def build_json_report(route, start, end, mode=None, max_transfers=None, line_preference=None):
    return {
        "query": {
            "start": start,
            "end": end,
            "mode": mode,
            "max_transfers": max_transfers,
            "line_preference": line_preference,
        },
        "path": list(route.stations),
        "segments": [
            {"order": s.order, "line": s.line, "stations": list(s.stations),
             "distance": s.distance} for s in route.segments
        ],
        "distance": {
            "meters": route.total_distance,
            "kilometers": round(route.total_distance / 1000.0, 3),
        },
        "fare": route.fare,
        "estimated_time": round(route.estimated_time, 1),
        "transfer_count": route.transfer_count,
        "transfer_stations": list(route.transfer_stations),
        "lines": list(route.lines),
    }


def format_json_report(*args, **kwargs):
    return json.dumps(build_json_report(*args, **kwargs), ensure_ascii=False, indent=2)

"""
Command-line interface of the Beijing Subway fare calculation system.

Usage:
    python main.py [--format text|json] [--output_file FILE]
                   [--mode shortest_distance|least_transfer|comprehensive]
                   [--max_transfers N] [--line_preference "Line 1"] [--check_data]

Enter queries as "Starting Station; Ending Station". Enter 'q' to quit.
"""

import argparse
import json
import sys

from data_loader import DATA_FILE_PATH
from data_manager import SubwayNetwork
from pathfinding import MODES, PathfindingConfig, ShortestPath
from report import build_json_report, build_route, format_text_report

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
QUIT_WORDS = {"q", "quit", "exit"}


class QueryError(Exception):
    """User-facing query error."""


def parse_query_input(text):
    """Split 'Start; End' into a tuple; raise QueryError on bad format."""
    if text is None:
        raise QueryError(FORMAT_ERROR)
    text = text.replace("；", ";").strip()
    parts = text.split(";")
    if len(parts) != 2:
        raise QueryError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryError(FORMAT_ERROR)
    return start, end


def validate_stations(network, start, end):
    s = network.find_station(start)
    if s is None:
        raise QueryError(f"Starting station '{start}' does not exist, please check the station name")
    e = network.find_station(end)
    if e is None:
        raise QueryError(f"Ending station '{end}' does not exist, please check the station name")
    if s == e:
        raise QueryError("Starting station cannot be the same as the ending station")
    return s, e


def run_query(network, start, end, config):
    """Compute a route between two (validated) station names."""
    try:
        sp = ShortestPath(network.graph, network.station_to_number[start], config)
        target = network.station_to_number[end]
        path, edges = sp.path_with_edges(target)
    except Exception as exc:  # path calculation exception
        raise QueryError(f"Path calculation error: {exc}") from exc
    if not path:
        msg = f"No path found from '{start}' to '{end}'"
        if config.max_transfers is not None:
            msg += f" within {config.max_transfers} transfer(s)"
        raise QueryError(msg)
    try:
        return build_route(path, edges, network.number_to_station)
    except Exception as exc:  # fare calculation exception
        raise QueryError(f"Fare calculation error: {exc}") from exc


def build_arg_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", default="text", choices=["text", "json"], help="Output format")
    p.add_argument("--mode", default="shortest_distance", choices=list(MODES), help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="Preferred line(s), comma separated (e.g. 'Line 1')")
    p.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    return p


def main(argv=None):
    args = build_arg_parser().parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer")
        return 2

    try:
        network = SubwayNetwork(data_file=args.data_file)
    except Exception as exc:
        print(f"Error: failed to load subway data: {exc}")
        return 1

    if args.check_data:
        print(network.check_integrity().to_text())
        return 0

    preferred = []
    if args.line_preference:
        for raw in args.line_preference.replace("，", ",").split(","):
            raw = raw.strip()
            if not raw:
                continue
            real = network.find_line(raw)
            if real is None:
                print(f"Warning: line '{raw}' not found in data, preference ignored")
            else:
                preferred.append(real)
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                               preferred_lines=preferred)

    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' to query, or 'q' to quit.")
    json_results = []
    text_written = False

    while True:
        try:
            print("Please enter query: ", end="", flush=True)
            line = sys.stdin.readline()
        except KeyboardInterrupt:
            print()
            break
        if line == "":
            print()
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in QUIT_WORDS:
            break
        try:
            start_in, end_in = parse_query_input(line)
            start, end = validate_stations(network, start_in, end_in)
            route = run_query(network, start, end, config)
        except QueryError as exc:
            print(str(exc))
            continue
        except Exception as exc:
            print(f"Unexpected error: {exc}")
            continue

        if args.format == "json":
            data = build_json_report(route, start, end, args.mode, args.max_transfers,
                                     args.line_preference)
            output = json.dumps(data, ensure_ascii=False, indent=2)
        else:
            data = None
            output = format_text_report(route, start, end, args.mode)
        print(output)

        if args.output_file:
            try:
                if args.format == "json":
                    json_results.append(data)
                    payload = json_results[0] if len(json_results) == 1 else json_results
                    with open(args.output_file, "w", encoding="utf-8") as fh:
                        json.dump(payload, fh, ensure_ascii=False, indent=2)
                else:
                    with open(args.output_file, "a" if text_written else "w", encoding="utf-8") as fh:
                        fh.write(output + "\n")
                    text_written = True
                print(f"Result exported to {args.output_file}")
            except OSError as exc:
                print(f"Error: failed to write output file: {exc}")

    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

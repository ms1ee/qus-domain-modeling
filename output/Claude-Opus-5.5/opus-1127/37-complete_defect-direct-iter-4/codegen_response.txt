"""Core data models for the lottery system."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A participant of the lottery."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)

    def __str__(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer if self.associated_drawer else "None"
        return (f"{self.name} x{self.quantity} "
                f"(Avoid Groups: {avoid}; Drawer: {drawer})")

class ListManager:
    """Manages participant lists."""

    def __init__(self) -> None:
        self.users: List[User] = []

    # ------------------------------------------------------------------ #
    def _parse_lines(self, lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff")
            if not line:
                continue
            normalized = line.replace("\uff0c", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        """Load built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_DATA.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        """Load a custom txt list."""
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        path = str(path)
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"File Error: cannot read file: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file: {exc}"]
        return self._parse_lines(lines)

    # ------------------------------------------------------------------ #
    def validate_users(self, users: List[User]) -> Tuple[List[User], List[User]]:
        """Split users into valid users and duplicate-name entries."""
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users: List[User]) -> Dict[str, int]:
        """Return department -> number of people."""
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and draw order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None,
                  associated_drawer: Optional[str] = None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else name
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be positive")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        prize = Prize(name, qty, groups, drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def _norm(text: Optional[str]) -> str:
    return " ".join((text or "").lower().split())


def is_protected_group(department: str) -> bool:
    d = _norm(department)
    return d in ("cheng ru group", "cheng ru")


class LotteryManager:
    """Applies filtering rules and performs draws."""

    def __init__(self, users: List[User]) -> None:
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self._winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = random.SystemRandom()

    # ------------------------------------------------------------------ #
    def _is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def _fallback_active(self, prize: Prize) -> bool:
        return (_norm(prize.associated_drawer) == _norm(FALLBACK_DRAWER)
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if not self._is_winner(u)]
        if self._fallback_active(prize):
            protected = [u for u in base if is_protected_group(u.department)]
            if protected:
                return protected
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return base
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in base if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        rules = ["Previous winners are excluded"]
        if self._fallback_active(prize):
            rules.append(f"Fallback mechanism active: pool limited to "
                         f"'{PROTECTED_GROUP}' members (avoidance ignored)")
        elif _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            rules.append(f"Special drawer {SPECIAL_DRAWER}: avoidance groups not applied")
        elif prize.avoid_groups:
            rules.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        else:
            rules.append("No avoidance groups")
        if prize.associated_drawer:
            rules.insert(0, f"Drawer: {prize.associated_drawer}")
        return "; ".join(rules)

    def draw_one(self, prize: Prize, pool: Optional[List[User]] = None) -> Optional[User]:
        if pool is None:
            pool = self._get_eligible_pool(prize)
        pool = [u for u in pool if not self._is_winner(u)]
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winners.append(user)
        self._winner_names.add(user.name)
        return user

    def record_round_result(self, prize: Prize, round_winners: List[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in WATCHED_DRAWERS}:
            self.last_two_draw_results.append(
                any(is_protected_group(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_DRAWER):
            self.last_two_draw_results = []

"""Result management and export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime
from typing import Dict, List, Tuple

from lottery_system.models import User


@dataclass
class Winner:
    user: User
    prize_name: str

    @property
    def name(self) -> str:
        return self.user.name

    @property
    def department(self) -> str:
        return self.user.department


def _prize_name(prize) -> str:
    return prize.name if hasattr(prize, "name") else str(prize)


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []

    def add_winner(self, user: User, prize) -> Winner:
        w = Winner(user, _prize_name(prize))
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes or []:
            result[_prize_name(p)] = []
        for w in self.winners:
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        if not self.winners:
            return False, "No winners to export"
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for pname, winners in self.get_winners_by_prize(prizes).items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, pname])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _input(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        return "terminate"


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryApp:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_done = False

    # ---------------- list ---------------- #
    def _accept_list(self, users, errors) -> None:
        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            print("\nList verification found problems:")
            for e in errors:
                print("  " + e)
            for d in duplicates:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            choice = _input("Enter 'r' to remove invalid data and continue, "
                            "'t' to terminate upload: ").strip().lower()
            if choice != "r":
                print("Upload terminated. List unchanged.")
                return
        if not valid:
            print("No valid users loaded.")
            return
        self.users = valid
        print(f"Loaded {len(valid)} valid users.")

    def load_default(self) -> None:
        self._accept_list(*self.list_manager.load_default_list())

    def load_custom(self) -> None:
        path = _input("Enter txt file path: ").strip().strip('"')
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            print("Upload failed.")
            return
        self._accept_list(users, errors)

    def view_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("\nValid groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    # ---------------- prizes ---------------- #
    def load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Default prizes loaded.")

    def add_custom_prize(self) -> None:
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid: List[str] = []
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = _input("Avoid groups (numbers or names separated by commas, blank for none): ").strip()
        for item in raw.replace("\uff0c", ",").split(","):
            item = item.strip()
            if not item:
                continue
            if item.isdigit() and 1 <= int(item) <= len(groups):
                g = groups[int(item) - 1]
            else:
                g = item
            if g not in avoid:
                avoid.append(g)
        drawer = _input("Associated lottery person (blank for none): ").strip() or None
        try:
            p = self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Added prize: {p}")
        except ValueError as exc:
            print(f"Error: {exc}")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nCurrent prize order:")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")

    def reorder(self) -> None:
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = _input("Enter new order as serial numbers (e.g. 3,1,2), blank to keep: ").strip()
        if not raw:
            return
        try:
            order = [int(x) for x in raw.replace("\uff0c", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.show_prizes()
        else:
            print("Invalid order. Must be a permutation of current serial numbers.")

    # ---------------- lottery ---------------- #
    def run_lottery(self) -> None:
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        self.result_manager = ResultManager()
        engine = LotteryManager(self.users)
        print("\nCommands during draw: Enter = draw next, 'pause', 'continue', 'terminate'")
        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            print(f"Rules: {engine.describe_rules(prize)}")
            print("=" * 60)
            pool = engine._get_eligible_pool(prize)
            round_winners: List[User] = []
            terminated = False
            for i in range(1, prize.quantity + 1):
                cmd = _input(f"[{_ordinal(i)} winner] Press Enter to draw: ").strip().lower()
                while cmd == "pause":
                    print("Lottery paused.")
                    cmd = _input("Type 'continue' to resume or 'terminate' to skip round: ").strip().lower()
                    if cmd == "continue":
                        print("Lottery resumed.")
                        break
                    if cmd != "terminate":
                        cmd = "pause"
                if cmd == "terminate":
                    print("Current round terminated and skipped.")
                    terminated = True
                    break
                print(f"Currently drawing the {_ordinal(i)} winner...")
                user = engine.draw_one(prize, pool)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                round_winners.append(user)
                self.result_manager.add_winner(user, prize)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            engine.record_round_result(prize, round_winners)
            print(f"\nRound {rnd} results ({prize.name}):")
            if round_winners:
                for k, u in enumerate(round_winners, 1):
                    print(f"  {k}. {u.name} ({u.department})")
            else:
                print("  No winners.")
            if terminated:
                continue
        self.lottery_done = True
        self.show_summary()

    def show_summary(self) -> None:
        prizes = self.prize_manager.get_prizes()
        data = self.result_manager.get_winners_by_prize(prizes)
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        print("\n========== Complete Winner List ==========")
        for pname, winners in data.items():
            print(f"{pname}:")
            if not winners:
                print("  (none)")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {pname}")

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ---------------- main loop ---------------- #
    def run(self) -> None:
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_default_prizes, "5": self.add_custom_prize,
            "6": self.show_prizes, "7": self.reorder, "8": self.run_lottery,
            "9": self.show_summary, "10": self.export,
        }
        while True:
            print("\n===== Team Building Smart Lottery System =====")
            print(" 1. Load default list        2. Upload custom txt list")
            print(" 3. View groups              4. Load preset prizes")
            print(" 5. Add custom prize         6. View prizes")
            print(" 7. Reorder prizes           8. Start lottery")
            print(" 9. View all winners        10. Export results")
            print(" 0. Exit")
            choice = _input("Select: ").strip()
            if choice in ("0", "terminate", "exit", "quit"):
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid option.")


def main() -> None:
    LotteryApp().run()


if __name__ == "__main__":
    main()

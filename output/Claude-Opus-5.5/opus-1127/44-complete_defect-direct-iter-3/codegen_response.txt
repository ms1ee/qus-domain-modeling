"""Data loading utilities for the Beijing Subway fare calculation system.

The network data lives in ``subway_data.json`` next to this module.  The
path is computed from this file's location so it is valid regardless of
the current working directory.
"""

import json
import os

DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    Raises:
        FileNotFoundError: if the file does not exist.
        ValueError: if the file content is not valid JSON or is malformed.
    """
    if path is None:
        path = DATA_FILE_PATH
    if not os.path.exists(path):
        raise FileNotFoundError(f"Subway data file not found: {path}")
    with open(path, "r", encoding="utf-8") as f:
        try:
            data = json.load(f)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid JSON in subway data file: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write the subway data back to a JSON file."""
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

"""Core data structures: Edge, Graph and MinPQ (heap based priority queue)."""

import heapq
import itertools


class Edge:
    """A weighted, line-labelled edge between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        return self.source

    def other(self, v):
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError(f"Vertex {v} is not an endpoint of this edge")

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}m, {self.line})"


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Station numbers are assigned by the caller.  ``graph`` maps each station
    number to a list of outgoing :class:`Edge` objects; each undirected
    connection is stored as two oriented edges.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self.E = 0

    @property
    def V(self):
        return len(self.graph)

    def add_vertex(self, v):
        if v not in self.graph:
            self.graph[v] = []

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines) if lines else []

    def add_edge(self, edge):
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        if edge.source != edge.dest:
            self.graph[edge.dest].append(edge.reversed())
        self.E += 1

    def remove_edge(self, a, b, line=None):
        """Remove the connection(s) between a and b (optionally only for a line)."""
        removed = 0
        for u, w in ((a, b), (b, a)):
            if u in self.graph:
                before = len(self.graph[u])
                self.graph[u] = [e for e in self.graph[u]
                                 if not (e.dest == w and (line is None or e.line == line))]
                removed += before - len(self.graph[u])
        self.E -= removed // 2
        return removed > 0

    def adj(self, v):
        return self.graph.get(v, [])

    def vertices(self):
        return list(self.graph.keys())

    def edges(self):
        result = []
        for v, lst in self.graph.items():
            for e in lst:
                if e.source <= e.dest if isinstance(e.source, int) else True:
                    result.append(e)
        return result


class MinPQ:
    """Minimum priority queue based on a binary heap (heapq)."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def del_min(self):
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def min(self):
        if not self._heap:
            raise IndexError("min from empty priority queue")
        priority, _, item = self._heap[0]
        return priority, item

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

"""Dijkstra-based path finding with transfer awareness.

The search runs over states ``(station, arriving line[, transfers])`` so that
transfers at interchange stations are counted correctly.  Costs are tuples
that are added component-wise, which keeps Dijkstra correct while allowing
different optimisation modes.
"""

from models import MinPQ

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY_METERS = 3000      # comprehensive mode penalty per transfer
PREFERENCE_FACTOR = 0.8             # cost factor for preferred lines


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class ShortestPath:
    """Single-source shortest path computation using Dijkstra's algorithm."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: float("inf") for v in graph.vertices()}
        self._transfers_to = {}
        self._parent = {}         # state -> (prev_state, edge)
        self._node_state = {}     # node -> best final state
        if source not in graph.graph:
            raise ValueError(f"Source station {source} is not in the graph")
        self._run()

    # ------------------------------------------------------------------ #
    def _cost(self, wdist, transfers, dist):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, wdist, dist)
        if mode == "comprehensive":
            return (wdist + TRANSFER_PENALTY_METERS * transfers, transfers, dist)
        return (wdist, transfers, dist)

    def _run(self):
        cfg = self.config
        limit = cfg.max_transfers
        preferred = set(cfg.preferred_lines)
        start = (self.source, None, 0 if limit is not None else None)
        # state -> (wdist, transfers, dist)
        values = {start: (0.0, 0, 0)}
        best_cost = {start: self._cost(0.0, 0, 0)}
        done = set()
        pq = MinPQ()
        pq.insert(best_cost[start], start)

        while not pq.is_empty():
            cost, state = pq.del_min()
            if state in done or cost != best_cost.get(state):
                continue
            done.add(state)
            node, line, _ = state
            wdist, transfers, dist = values[state]
            if node not in self._node_state:
                self._node_state[node] = state
                self.dist_to[node] = dist
                self._transfers_to[node] = transfers
            for edge in self.graph.adj(node):
                new_t = transfers + (1 if line is not None and edge.line != line else 0)
                if limit is not None and new_t > limit:
                    continue
                factor = PREFERENCE_FACTOR if edge.line in preferred else 1.0
                new_w = wdist + edge.weight * factor
                new_d = dist + edge.weight
                new_state = (edge.dest, edge.line, new_t if limit is not None else None)
                if new_state in done:
                    continue
                new_cost = self._cost(new_w, new_t, new_d)
                old = best_cost.get(new_state)
                if old is None or new_cost < old:
                    best_cost[new_state] = new_cost
                    values[new_state] = (new_w, new_t, new_d)
                    self._parent[new_state] = (state, edge)
                    pq.insert(new_cost, new_state)

    # ------------------------------------------------------------------ #
    def has_path_to(self, target):
        return target in self._node_state

    def distance_to(self, target):
        return self.dist_to.get(target, float("inf"))

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target."""
        if target not in self._node_state:
            return [], []
        edges = []
        state = self._node_state[target]
        while state in self._parent:
            prev, edge = self._parent[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        path = [self.source] + [e.dest for e in edges]
        return path, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                count += 1
        return count

"""Beijing Subway fare calculation.

Fare structure (distance rounded up to whole kilometres):
    0-6 km   : 3 yuan
    6-12 km  : 4 yuan
    12-22 km : 5 yuan
    22-32 km : 6 yuan
    > 32 km  : +1 yuan for every additional 20 km (or part thereof)
"""

import math

# (upper bound in km inclusive, fare in yuan)
FARE_TIERS = [
    (6, 3),
    (12, 4),
    (22, 5),
    (32, 6),
]
EXTENDED_STEP_KM = 20
AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters."""
    if distance is None:
        raise ValueError("Distance cannot be None")
    try:
        distance = float(distance)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"Invalid distance: {distance!r}") from exc
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError(f"Invalid distance: {distance!r}")
    if distance < 0:
        raise ValueError("Distance cannot be negative")
    km = math.ceil(distance / 1000.0)
    for upper, fare in FARE_TIERS:
        if km <= upper:
            return fare
    last_upper, last_fare = FARE_TIERS[-1]
    return last_fare + math.ceil((km - last_upper) / EXTENDED_STEP_KM)


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes from distance (meters) and transfers."""
    if distance is None or distance < 0:
        raise ValueError("Distance must be non-negative")
    if transfer_count is None or transfer_count < 0:
        raise ValueError("Transfer count must be non-negative")
    ride = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    return round(ride + transfer_count * TRANSFER_TIME_MINUTES, 1)

"""Subway network data management: graph building, integrity checks, updates."""

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
from models import Edge, Graph

MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

# Representative stations (English name, Chinese name)
REPRESENTATIVE_STATIONS = [
    ("Xidan", "西单"), ("Tian'anmen East", "天安门东"), ("Guomao", "国贸"),
    ("Dongzhimen", "东直门"), ("Xizhimen", "西直门"), ("Beijing Railway Station", "北京站"),
    ("Zhongguancun", "中关村"), ("Yonghegong", "雍和宫"), ("Sanyuanqiao", "三元桥"),
]

MIN_DISTANCE = 50
MAX_DISTANCE = 40000


def _normalize_line(name):
    s = str(name).lower().replace(" ", "").replace("号线", "").replace("line", "")
    return s


_CN_LINE_ALIASES = {
    "changpingline": "昌平", "yizhuangline": "亦庄", "fangshanline": "房山", "airportline": "机场",
}


class SubwayDataManager:
    """Holds subway data, builds the graph and supports updates/checks."""

    def __init__(self, data=None, path=DATA_FILE_PATH):
        self.path = path
        self.data = data if data is not None else load_subway_data(path)
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = None
        self.build()

    # ------------------------------------------------------------------ #
    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        g = Graph()
        for st in self.data.get("stations", []):
            name = st.get("name")
            if not name or name in self.station_to_number:
                continue
            self._assign(name)
            g.add_vertex(self.station_to_number[name])
            g.set_station_lines(name, st.get("lines", []))
        for c in self.data.get("connections", []):
            src, dst = c.get("source"), c.get("dest")
            if not src or not dst:
                continue
            for n in (src, dst):
                if n not in self.station_to_number:
                    self._assign(n)
                    g.add_vertex(self.station_to_number[n])
                    g.set_station_lines(n, [c.get("line")] if c.get("line") else [])
            try:
                dist = int(c.get("distance", 0))
            except (TypeError, ValueError):
                continue
            if dist < 0:
                continue
            g.add_edge(Edge(self.station_to_number[src], self.station_to_number[dst],
                            dist, c.get("line", "")))
        self.graph = g
        return g

    def _assign(self, name):
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name

    # ------------------------------------------------------------------ #
    def lines(self):
        result = []
        for st in self.data.get("stations", []):
            for line in st.get("lines", []):
                if line not in result:
                    result.append(line)
        for c in self.data.get("connections", []):
            if c.get("line") and c["line"] not in result:
                result.append(c["line"])
        return result

    def line_stations(self, line):
        return [st["name"] for st in self.data.get("stations", []) if line in st.get("lines", [])]

    def transfer_stations(self):
        return [st["name"] for st in self.data.get("stations", []) if len(st.get("lines", [])) > 1]

    def find_station(self, name):
        """Return standardized station name or None."""
        if name is None:
            return None
        name = name.strip()
        if name in self.station_to_number:
            return name
        low = name.lower()
        for s in self.station_to_number:
            if s.lower() == low:
                return s
        compact = low.replace(" ", "")
        for s in self.station_to_number:
            if s.lower().replace(" ", "") == compact:
                return s
        return None

    def suggest(self, name, n=3):
        import difflib
        return difflib.get_close_matches(name, list(self.station_to_number), n=n, cutoff=0.6)

    # ------------------------------------------------------------------ #
    def check_integrity(self):
        """Return a report dict with keys: ok, errors, warnings, stats."""
        errors, warnings = [], []
        stations = self.data.get("stations", [])
        conns = self.data.get("connections", [])
        names = set()
        for st in stations:
            name = st.get("name")
            if not name:
                errors.append("Station entry without name")
                continue
            if name in names:
                warnings.append(f"Duplicate station entry: {name}")
            names.add(name)
            if not st.get("lines"):
                warnings.append(f"Station '{name}' has no line information")

        # Coverage of main lines
        all_lines = self.lines()
        norm = [_normalize_line(l) for l in all_lines]
        missing_lines = []
        for ml in MAIN_LINES:
            key = _normalize_line(ml)
            cn = _CN_LINE_ALIASES.get(key)
            found = any(n == key or (cn and cn in n) for n in norm)
            if not found:
                missing_lines.append(ml)
        if missing_lines:
            warnings.append("Main lines not found in data: " + ", ".join(missing_lines))
        missing_rep = [en for en, cn in REPRESENTATIVE_STATIONS if en not in names and cn not in names]
        if missing_rep and len(missing_rep) == len(REPRESENTATIVE_STATIONS):
            warnings.append("No representative stations found in data")
        elif missing_rep:
            warnings.append("Representative stations missing: " + ", ".join(missing_rep))

        # Connection integrity and distance accuracy
        seen = set()
        connected = set()
        for i, c in enumerate(conns):
            src, dst, line = c.get("source"), c.get("dest"), c.get("line")
            if src not in names:
                errors.append(f"Connection #{i}: unknown source station '{src}'")
            if dst not in names:
                errors.append(f"Connection #{i}: unknown dest station '{dst}'")
            if src == dst:
                errors.append(f"Connection #{i}: self loop at '{src}'")
            connected.update([src, dst])
            key = (frozenset((src, dst)), line)
            if key in seen:
                warnings.append(f"Duplicate connection {src} - {dst} ({line})")
            seen.add(key)
            d = c.get("distance")
            if not isinstance(d, (int, float)) or isinstance(d, bool):
                errors.append(f"Connection {src} - {dst}: invalid distance {d!r}")
            elif d <= 0:
                errors.append(f"Connection {src} - {dst}: non-positive distance {d}")
            elif d < MIN_DISTANCE or d > MAX_DISTANCE:
                warnings.append(f"Connection {src} - {dst}: suspicious distance {d} m")
            for n in (src, dst):
                st_lines = self.graph.station_lines.get(n)
                if st_lines and line and line not in st_lines:
                    warnings.append(f"Connection {src} - {dst}: line '{line}' not listed for station '{n}'")
        isolated = sorted(names - connected)
        for n in isolated:
            warnings.append(f"Station '{n}' has no connections")

        # Connectivity
        components = self._component_count()
        if components > 1:
            warnings.append(f"Network has {components} disconnected components")

        stats = {
            "stations": len(names),
            "connections": len(conns),
            "lines": len(all_lines),
            "transfer_stations": len(self.transfer_stations()),
            "components": components,
        }
        return {"ok": not errors, "errors": errors, "warnings": warnings, "stats": stats}

    def _component_count(self):
        seen = set()
        count = 0
        for v in self.graph.vertices():
            if v in seen:
                continue
            count += 1
            stack = [v]
            seen.add(v)
            while stack:
                u = stack.pop()
                for e in self.graph.adj(u):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        stack.append(e.dest)
        return count

    # ------------------------------------------------------------------ #
    # Update mechanisms
    def add_line(self, line_name, station_names, distances):
        """Add a new line given ordered station names and inter-station distances."""
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        if any(d <= 0 for d in distances):
            raise ValueError("Distances must be positive")
        by_name = {st["name"]: st for st in self.data["stations"]}
        for name in station_names:
            if name in by_name:
                if line_name not in by_name[name].setdefault("lines", []):
                    by_name[name]["lines"].append(line_name)
            else:
                st = {"name": name, "lines": [line_name]}
                self.data["stations"].append(st)
                by_name[name] = st
        for i, d in enumerate(distances):
            self.data["connections"].append({
                "source": station_names[i], "dest": station_names[i + 1],
                "distance": int(d), "line": line_name,
            })
        self.build()

    def modify_station(self, name, new_name=None, lines=None):
        found = False
        for st in self.data["stations"]:
            if st["name"] == name:
                found = True
                if new_name:
                    st["name"] = new_name
                if lines is not None:
                    st["lines"] = list(lines)
        if not found:
            raise KeyError(f"Station '{name}' does not exist")
        if new_name:
            for c in self.data["connections"]:
                if c["source"] == name:
                    c["source"] = new_name
                if c["dest"] == name:
                    c["dest"] = new_name
        self.build()

    def update_distance(self, a, b, distance, line=None):
        if distance <= 0:
            raise ValueError("Distance must be positive")
        updated = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c.get("line") == line):
                c["distance"] = int(distance)
                updated = True
        if not updated:
            raise KeyError(f"No connection between '{a}' and '{b}'")
        self.build()

    def save(self, path=None):
        save_subway_data(self.data, path or self.path)

"""Path visualization and export (text and JSON)."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    """Meters for short distances, kilometres for long ones."""
    if meters < 1000:
        return f"{int(round(meters))} m"
    return f"{meters / 1000.0:.2f} km"


def build_route_info(manager, sp, start_name, end_name, config):
    """Collect all route information into a dict. Raises RuntimeError if no path."""
    target = manager.station_to_number[end_name]
    path, edges = sp.path_with_edges(target)
    if not path:
        raise RuntimeError(f"No route found from '{start_name}' to '{end_name}'"
                           + (" within the transfer limit" if config.max_transfers is not None else ""))
    names = [manager.number_to_station[n] for n in path]
    distance = sum(e.weight for e in edges)

    segments = []
    for e in edges:
        frm = manager.number_to_station[e.source]
        to = manager.number_to_station[e.dest]
        if segments and segments[-1]["line"] == e.line:
            seg = segments[-1]
            seg["stations"].append(to)
            seg["to"] = to
            seg["distance"] += e.weight
        else:
            segments.append({"line": e.line, "from": frm, "to": to,
                             "stations": [frm, to], "distance": e.weight})
    transfers = []
    for i in range(1, len(segments)):
        transfers.append({"station": segments[i]["from"],
                          "from_line": segments[i - 1]["line"],
                          "to_line": segments[i]["line"]})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    transfer_count = len(transfers)
    try:
        fare = Sum_money(distance)
    except Exception as exc:  # fare calculation exception handling
        raise RuntimeError(f"Fare calculation failed: {exc}") from exc
    est = calculate_estimated_time(distance, transfer_count)
    return {
        "start": start_name, "end": end_name,
        "mode": config.mode, "max_transfers": config.max_transfers,
        "line_preference": list(config.preferred_lines),
        "path": names, "segments": segments, "transfers": transfers,
        "lines": lines, "distance": distance, "transfer_count": transfer_count,
        "fare": fare, "estimated_time": est,
    }

def format_text_report(info):
    lines = []
    lines.append("=" * 50)
    lines.append(f"Route: {info['start']} -> {info['end']}  (mode: {info['mode']})")
    lines.append("=" * 50)
    lines.append(f"Stations ({len(info['path'])}): " + " -> ".join(info["path"]))
    lines.append("Line segments:")
    for i, seg in enumerate(info["segments"]):
        lines.append(f"  [{seg['line']}] {seg['from']} -> {seg['to']} "
                     f"({len(seg['stations']) - 1} stops, {format_distance(seg['distance'])})")
        if i < len(info["transfers"]):
            t = info["transfers"][i]
            lines.append(f"  >> Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")

"""Command line interface for the Beijing Subway fare calculation system.

Usage:
    python src/main.py [--format text|json] [--output_file FILE]
                       [--mode shortest_distance|least_transfer|comprehensive]
                       [--max_transfers N] [--line_preference "Line 1,Line 2"]
                       [--check_data]

Enter queries as "Starting Station; Ending Station".  Type 'quit' to exit.

import argparse
import json
import sys

from data_loader import DATA_FILE_PATH
from data_manager import SubwayDataManager
from pathfinding import PathfindingConfig, ShortestPath, VALID_MODES
from report import build_route_info, format_json_report, format_text_report, to_json_dict

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text", help="Output format")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                   help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="Preferred line(s), comma separated, e.g. 'Line 1'")
    p.add_argument("--check_data", action="store_true", help="Check data integrity and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    args = p.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        p.error("--max_transfers must be non-negative")
    return args


def parse_query(text):
    """Return (start, end) or raise ValueError with a user-facing message."""
    text = text.replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2:
        raise ValueError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise ValueError(FORMAT_ERROR)
    return start, end


def resolve_stations(manager, start, end):
    s = manager.find_station(start)
    if s is None:
        msg = f"Starting station '{start}' does not exist, please check the station name"
        sug = manager.suggest(start)
        if sug:
            msg += f" (did you mean: {', '.join(sug)}?)"
        raise ValueError(msg)
    e = manager.find_station(end)
    if e is None:
        msg = f"Ending station '{end}' does not exist, please check the station name"
        sug = manager.suggest(end)
        if sug:
            msg += f" (did you mean: {', '.join(sug)}?)"
        raise ValueError(msg)
    if s == e:
        raise ValueError("Starting station cannot be the same as the ending station")
    return s, e


def process_query(manager, config, text):
    start, end = parse_query(text)
    start, end = resolve_stations(manager, start, end)
    try:
        sp = ShortestPath(manager.graph, manager.station_to_number[start], config)
    except Exception as exc:
        raise RuntimeError(f"Path calculation error: {exc}") from exc
    return build_route_info(manager, sp, start, end, config)


def run_check(manager):
    report = manager.check_integrity()
    st = report["stats"]
    print("Data integrity check")
    print(f"  Stations: {st['stations']}, Connections: {st['connections']}, "
          f"Lines: {st['lines']}, Transfer stations: {st['transfer_stations']}, "
          f"Components: {st['components']}")
    for e in report["errors"]:
        print(f"  ERROR: {e}")
    for w in report["warnings"]:
        print(f"  WARNING: {w}")
    print("Result: " + ("PASSED" if report["ok"] else "FAILED"))
    return 0 if report["ok"] else 1


def write_output(path, fmt, results):
    with open(path, "w", encoding="utf-8") as f:
        if fmt == "json":
            data = [to_json_dict(r) for r in results]
            json.dump(data[0] if len(data) == 1 else data, f, ensure_ascii=False, indent=2)
        else:
            f.write("\n\n".join(format_text_report(r) for r in results) + "\n")


def main(argv=None):
    args = parse_args(argv)
    try:
        manager = SubwayDataManager(path=args.data_file)
    except Exception as exc:
        print(f"Failed to load subway data: {exc}", file=sys.stderr)
        return 1
    if args.check_data:
        return run_check(manager)

    preferred = []
    if args.line_preference:
        preferred = [l.strip() for l in args.line_preference.split(",") if l.strip()]
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                               preferred_lines=preferred)

    print("Beijing Subway Fare Calculator")
    print("Enter 'Starting Station; Ending Station' (type 'quit' to exit).")
    results = []
    while True:
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("quit", "exit", "q"):
            break
        try:
            info = process_query(manager, config, line)
        except ValueError as exc:
            print(str(exc))
            continue
        except RuntimeError as exc:
            print(str(exc))
            continue
        except Exception as exc:
            print(f"Unexpected error: {exc}")
            continue
        if args.format == "json":
            print(format_json_report(info))
        else:
            print(format_text_report(info))
        results.append(info)
        if args.output_file:
            try:
                write_output(args.output_file, args.format, results)
            except OSError as exc:
                print(f"Failed to write output file: {exc}")
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

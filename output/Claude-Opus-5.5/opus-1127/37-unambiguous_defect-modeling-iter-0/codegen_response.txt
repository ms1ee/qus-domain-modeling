"""Core data models for the lottery system."""

from __future__ import annotations

import re
from typing import List, Optional


class User:
    """A participant of the lottery."""

    def __init__(self, name, department):
        self.name = str(name).strip() if name is not None else ""
        self.department = str(department).strip() if department is not None else ""

    @property
    def is_valid(self) -> bool:
        return bool(self.name) and bool(self.department)

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self):
        return f"{self.name} ({self.department})"


class Prize:
    """A prize to be drawn in one lottery round."""

    def __init__(self, name, quantity, avoid_groups: Optional[List[str]] = None,
                 associated_drawer: Optional[str] = None):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups = [str(g).strip() for g in (avoid_groups or []) if str(g).strip()]
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else associated_drawer
        self.associated_drawer = drawer or None

    @property
    def level_rank(self) -> int:
        return prize_level_rank(self.name)

    def __eq__(self, other):
        if not isinstance(other, Prize):
            return NotImplemented
        return (self.name == other.name and self.quantity == other.quantity
                and self.avoid_groups == other.avoid_groups
                and self.associated_drawer == other.associated_drawer)

    def __hash__(self):
        return hash(self.name)

    def __repr__(self):
        return (f"Prize(name={self.name!r}, quantity={self.quantity}, "
                f"avoid_groups={self.avoid_groups!r}, associated_drawer={self.associated_drawer!r})")


_LEVEL_KEYWORDS = [
    (r"\b(special|grand)\b", 0),
    (r"\bfirst\b", 1),
    (r"\bsecond\b", 2),
    (r"\bthird\b", 3),
    (r"\bfourth\b", 4),
    (r"\bfifth\b", 5),
    (r"\b(consolation|participation|lucky)\b", 6),
]


def prize_level_rank(name: str) -> int:
    """Return a numeric level rank (smaller = higher level). Unknown -> 50."""
    lower = (name or "").lower()
    for pattern, rank in _LEVEL_KEYWORDS:
        if re.search(pattern, lower):
            return rank
    return 50

_SEPARATOR = re.compile(r"[,\t\uff0c]")


class ListManager:
    """Manages participant lists."""

    def __init__(self):
        self.users: List[User] = []
        self.source: str = ""

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        self.users = users
        self.source = "default"
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        users, errors = self._load_from_path(path)
        if users:
            self.users = users
            self.source = path
        return users, errors

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as fh:
                    lines = fh.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"File Error: cannot read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        users, errors = self._parse_lines(lines)
        if not users and not errors:
            errors.append("File Error: the list contains no participant data")
        return users, errors

    def _parse_lines(self, lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [p.strip() for p in _SEPARATOR.split(line)]
            if idx == 1 and len(parts) == 2 and parts[0].lower() == "name" \
                    and parts[1].lower() in ("department", "group"):
                continue  # header line
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' (expected format: Name,Department)")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # --------------------------------------------------------------- validation
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            if not user.name or not user.department:
                continue
            if user.name in seen:
                duplicates.append(user)
                continue
            seen.add(user.name)
            valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages prizes and their draw order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = str(name).strip() if name is not None else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        if isinstance(avoid_groups, str):
            avoid_groups = [g for g in avoid_groups.split(",")]
        prize = Prize(name, qty, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def get_prize(self, name) -> Optional[Prize]:
        for p in self.prizes:
            if p.name == name:
                return p
        return None

    def reorder_prizes(self, order) -> bool:
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        n = len(self.prizes)
        if n == 0 or len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

"""Lottery rule engine and draw execution."""

from __future__ import annotations

import random
from typing import Iterable, List, Optional

from lottery_system.models import Prize, User


def _norm(s) -> str:
    return " ".join(str(s or "").strip().lower().split())


class LotteryManager:
    RELAXED_DRAWER = "Li Bin"
    TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
    FALLBACK_DRAWER = "Ma Jiaqi"
    PROTECTED_GROUP = "Cheng Ru group"

    def __init__(self, users, rng=None):
        self.users: List[User] = list(users or [])
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self._rng = rng or random.SystemRandom()

    # ----------------------------------------------------------------- helpers
    @classmethod
    def is_protected_member(cls, user: User) -> bool:
        dep = _norm(user.department)
        target = _norm(cls.PROTECTED_GROUP)
        return dep == target or "cheng ru" in dep

    def _drawer_is(self, prize: Prize, person: str) -> bool:
        return _norm(prize.associated_drawer) == _norm(person)

    def is_relaxed(self, prize: Prize) -> bool:
        return self._drawer_is(prize, self.RELAXED_DRAWER)

    def is_fallback_active(self, prize: Prize) -> bool:
        if not self._drawer_is(prize, self.FALLBACK_DRAWER):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) >= 2 and not any(recent)

    def is_winner(self, user: User) -> bool:
        return user.name in self.winner_names

    # ------------------------------------------------------------- rule engine
    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        available = [u for u in self.users if not self.is_winner(u)]
        if self.is_fallback_active(prize):
            protected = [u for u in available if self.is_protected_member(u)]
            if protected:
                return protected
        if self.is_relaxed(prize):
            return available
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in available if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        if self.is_fallback_active(prize):
            parts.append(f"FALLBACK ACTIVE: pool limited to '{self.PROTECTED_GROUP}' members "
                         f"(Cheng Ru group protection)")
        elif self.is_relaxed(prize):
            parts.append(f"Relaxed rules for lottery person {self.RELAXED_DRAWER}: "
                         f"avoidance groups are ignored")
        elif prize.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")
        if prize.associated_drawer:
            parts.append(f"Lottery person: {prize.associated_drawer}")
        return "; ".join(parts)

    # ------------------------------------------------------------------ drawing
    def mark_winner(self, user: User) -> bool:
        if self.is_winner(user):
            return False
        self.winner_names.add(user.name)
        return True

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.mark_winner(user)
        return user

    def record_round_result(self, prize: Prize, winners: Iterable[User]) -> None:
        winners = list(winners)
        if any(self._drawer_is(prize, p) for p in self.TRIGGER_DRAWERS):
            self.last_two_draw_results.append(any(self.is_protected_member(u) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self._drawer_is(prize, self.FALLBACK_DRAWER):
            self.last_two_draw_results = []

    def reset(self) -> None:
        self.winner_names.clear()
        self.last_two_draw_results = []

"""Winner recording, summary, and export."""

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import prize_level_rank


class Winner:
    def __init__(self, user, prize_name: str, win_time: datetime, order: int):
        self.user = user
        self.prize_name = prize_name
        self.prize = prize_name
        self.win_time = win_time
        self.order = order

    def __repr__(self):
        return f"Winner(user={self.user!r}, prize={self.prize_name!r}, order={self.order})"


def _prize_name(prize) -> str:
    return prize.name if hasattr(prize, "name") else str(prize)


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self._counter = 0

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self._counter += 1
        self.winners.append(Winner(user, _prize_name(prize), datetime.now(), self._counter))
        return True

    def get_round_winners(self, prize) -> List[Winner]:
        name = _prize_name(prize)
        return [w for w in self.winners if w.prize_name == name]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        names = []
        for p in prizes or []:
            n = _prize_name(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        indexed = list(enumerate(names))
        indexed.sort(key=lambda t: (prize_level_rank(t[1]), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, n in indexed:
            ws = [w for w in self.winners if w.prize_name == n]
            ws.sort(key=lambda w: (w.win_time, w.order))
            result[n] = ws
        return result

    def clear(self) -> None:
        self.winners = []
        self._counter = 0

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as fh:
                writer = csv.writer(fh)
                writer.writerow(["No.", "Name", "Department", "Prize", "Winning Time"])
                idx = 0
                for prize_name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        idx += 1
                        writer.writerow([idx, w.user.name, w.user.department, prize_name,
                                         w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from __future__ import annotations

import sys
from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _input(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        raise SystemExit(0)


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotterySystem:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_manager = LotteryManager([])
        self.drawn_prizes = set()
        self.list_loading_mode = ""

    # ------------------------------------------------------------------ lists
    def _accept_users(self, users, errors, mode) -> None:
        for e in errors:
            print("  " + e)
        valid, dups = self.list_manager.validate_users(users)
        for d in dups:
            print(f"  Duplicate Name: {d.name} ({d.department})")
        if errors or dups:
            choice = _input("Invalid data detected. [r] remove invalid data and continue, "
                            "[t] terminate upload: ").strip().lower()
            if choice != "r":
                print("Upload terminated. Current list unchanged.")
                return
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        self.list_loading_mode = mode
        self.lottery_manager = LotteryManager(valid)
        self.result_manager.clear()
        self.drawn_prizes.clear()
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._accept_users(users, errors, "default")

    def load_custom(self):
        path = _input("Enter txt file path (format per line: Name,Department): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users:
            for e in errors:
                print("  " + e)
            print("No participants loaded.")
            return
        self._accept_users(users, errors, "custom")

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current valid groups:")
        for i, (dep, cnt) in enumerate(info.items(), 1):
            print(f"  {i}. {dep}: {cnt} people")

    # ----------------------------------------------------------------- prizes
    def load_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prize library loaded.")

    def add_prize(self):
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            print("Available groups: " + ", ".join(f"{i}.{g}" for i, g in enumerate(groups, 1)))
        raw = _input("Avoidance groups (numbers or names, comma-separated, empty for none): ").strip()
        avoid = []
        for tok in [t.strip() for t in raw.replace("\uff0c", ",").split(",") if t.strip()]:
            if tok.isdigit() and 1 <= int(tok) <= len(groups):
                avoid.append(groups[int(tok) - 1])
            else:
                avoid.append(tok)
        drawer = _input("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            status = " [drawn]" if p.name in self.drawn_prizes else ""
            print(f"  {i}. {p.name} x{p.quantity} | avoid: {', '.join(p.avoid_groups) or 'none'}"
                  f" | lottery person: {p.associated_drawer or 'none'}{status}")

    def reorder(self):
        self.view_prizes()
        raw = _input("Enter new order as serial numbers (e.g. 3,1,2): ").strip()
        try:
            order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.view_prizes()
        else:
            print("Invalid order. Must be a permutation of all serial numbers.")

    # ---------------------------------------------------------------- lottery
    def run_round(self, round_no, prize):
        lm = self.lottery_manager
        print("\n" + "=" * 60)
        print(f"Round {round_no}: {prize.name} | Quantity: {prize.quantity}")
        print(f"Rules: {lm.describe_rules(prize)}")
        print("Commands: [Enter] draw next, 'pause', 'continue', 'terminate'")
        round_winners = []
        paused = False
        terminated = False
        for i in range(prize.quantity):
            while True:
                cmd = _input("(paused) > " if paused else "> ").strip().lower()
                if cmd == "terminate":
                    terminated = True
                    break
                if paused:
                    if cmd == "continue":
                        paused = False
                        print("Lottery resumed.")
                        continue
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to end round.")
                    continue
                if cmd == "pause":
                    paused = True
                    print("Lottery paused.")
                    continue
                if cmd in ("", "continue", "draw"):
                    break
                print("Unknown command.")
            if terminated:
                print("Current round terminated.")
                break
            print(f"Currently drawing the {ordinal(i + 1)} winner...")
            user = lm.draw_one(prize)
            if user is None:
                print("No eligible participants remain for this prize.")
                break
            self.result_manager.add_winner(user, prize)
            round_winners.append(user)
            print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        lm.record_round_result(prize, round_winners)
        self.drawn_prizes.add(prize.name)
        print(f"--- Round {round_no} results ({prize.name}) ---")
        if not round_winners:
            print("  No winners.")
        for i, u in enumerate(round_winners, 1):
            print(f"  {i}. {u.name} ({u.department})")

    def start_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = [p for p in self.prize_manager.get_prizes() if p.name not in self.drawn_prizes]
        if not prizes:
            print("No pending prizes to draw.")
            return
        all_prizes = self.prize_manager.get_prizes()
        for p in prizes:
            self.run_round(all_prizes.index(p) + 1, p)
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for prize_name, ws in data.items():
            if not ws:
                continue
            print(f"[{prize_name}]")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {prize_name} "
                      f"@ {w.win_time:%H:%M:%S}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ------------------------------------------------------------------- menu
    def run(self):
        actions = {
            "1": ("Load default team building list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.view_groups),
            "4": ("Load preset prize library", self.load_prizes),
            "5": ("Add custom prize", self.add_prize),
            "6": ("View prizes", self.view_prizes),
            "7": ("Reorder prizes", self.reorder),
            "8": ("Start lottery", self.start_lottery),
            "9": ("View winner summary", self.show_summary),
            "10": ("Export results", self.export),
        }
        while True:
            print("\n===== Team Building Smart Lottery System =====")
            for k, (label, _) in actions.items():
                print(f" {k}. {label}")
            print(" 0. Exit")
            choice = _input("Select: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action[1]()
            else:
                print("Invalid choice.")


def main():
    try:
        LotterySystem().run()

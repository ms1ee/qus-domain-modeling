"""
Core data structures for the Beijing Subway fare calculation system.

Contains:
    * Edge   - a weighted, line-labelled undirected edge between two stations.
    * Graph  - a weighted undirected graph with adjacency lists.  Station
               numbers are assigned by the caller, not by the graph.
    * MinPQ  - a binary-heap based minimum priority queue used to speed up
               Dijkstra's algorithm to O((V + E) log V).
"""

from __future__ import annotations

from typing import Any, Dict, Iterable, List, Optional


class Edge:
    """An undirected weighted edge between two station numbers.

    Attributes:
        source:   station number of one end point.
        dest:     station number of the other end point.
        distance: distance in meters.
        weight:   same as ``distance`` (used by the path algorithms).
        line:     the name of the subway line this edge belongs to.
    """

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source: int, dest: int, distance: float, line: Optional[str] = None):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance cannot be negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self) -> int:
        """Return one end point of the edge."""
        return self.source

    def other(self, vertex: int) -> int:
        """Return the end point of the edge that is not ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex!r} is not an end point of this edge")

    def set_distance(self, distance: float) -> None:
        """Update the distance (and weight) of the edge."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance cannot be negative")
        self.distance = distance
        self.weight = distance

    def __repr__(self) -> str:
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller.  The graph additionally keeps
    ``station_lines`` (station name -> list of lines) and optional name/number
    mappings for convenience.
    """

    def __init__(self) -> None:
        self._adj: Dict[int, List[Edge]] = {}
        self._edges: List[Edge] = []
        self.station_lines: Dict[str, List[str]] = {}
        self.station_to_number: Dict[str, int] = {}
        self.number_to_station: Dict[int, str] = {}

    # ------------------------------------------------------------------ nodes
    def add_vertex(self, vertex: int) -> None:
        """Ensure ``vertex`` exists in the graph."""
        if vertex not in self._adj:
            self._adj[vertex] = []

    def add_station(self, name: str, number: int, lines: Optional[Iterable[str]] = None) -> None:
        """Register a station name with a caller-assigned number."""
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.add_vertex(number)
        if lines is not None:
            self.set_station_lines(name, lines)

    def set_station_lines(self, station_name: str, lines: Iterable[str]) -> None:
        """Set the list of lines serving ``station_name``."""
        self.station_lines[station_name] = list(lines)

    def is_transfer_station(self, station_name: str) -> bool:
        return len(self.station_lines.get(station_name, [])) > 1

    def has_vertex(self, vertex: int) -> bool:
        return vertex in self._adj

    def vertices(self) -> List[int]:
        return list(self._adj.keys())

    def V(self) -> int:
        return len(self._adj)

    # ------------------------------------------------------------------ edges
    def add_edge(self, edge: Edge) -> None:
        """Add an undirected edge."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self._adj[edge.dest].append(edge)
        self._edges.append(edge)

    def adj(self, vertex: int) -> List[Edge]:
        """Return edges incident to ``vertex``."""
        return self._adj.get(vertex, [])

    def edges(self) -> List[Edge]:
        return list(self._edges)

    def E(self) -> int:
        return len(self._edges)

    def find_edges(self, source: int, dest: int, line: Optional[str] = None) -> List[Edge]:
        result = []
        for e in self._adj.get(source, []):
            if (e.source == source and e.dest == dest) or (e.source == dest and e.dest == source):
                if line is None or e.line == line:
                    result.append(e)
        return result

    def update_edge_distance(self, source: int, dest: int, distance: float,
                             line: Optional[str] = None) -> int:
        """Update the distance of matching edges.  Returns number updated."""
        matches = self.find_edges(source, dest, line)
        for e in matches:
            e.set_distance(distance)
        return len(matches)

    def remove_edge(self, source: int, dest: int, line: Optional[str] = None) -> int:
        """Remove matching edges.  Returns number of removed edges."""
        matches = self.find_edges(source, dest, line)
        for e in matches:
            if e in self._adj.get(e.source, []):
                self._adj[e.source].remove(e)
            if e.dest != e.source and e in self._adj.get(e.dest, []):
                self._adj[e.dest].remove(e)
            self._edges.remove(e)
        return len(matches)

    def degree(self, vertex: int) -> int:
        return len(self._adj.get(vertex, []))

    def to_dict(self) -> Dict[int, Dict[int, float]]:
        """Return a plain dict adjacency representation {v: {w: min distance}}."""
        result: Dict[int, Dict[int, float]] = {v: {} for v in self._adj}
        for e in self._edges:
            for a, b in ((e.source, e.dest), (e.dest, e.source)):
                if b not in result[a] or e.weight < result[a][b]:
                    result[a][b] = e.weight
        return result

    def __repr__(self) -> str:
        return f"Graph(V={self.V()}, E={self.E()})"


class MinPQ:
    """Binary heap based minimum priority queue.

    Items must be mutually comparable (tuples are typical).
    """

    def __init__(self) -> None:
        self._heap: List[Any] = []

    def insert(self, item: Any) -> None:
        heap = self._heap
        heap.append(item)
        self._swim(len(heap) - 1)

    push = insert

    def min(self) -> Any:
        if not self._heap:
            raise IndexError("Priority queue underflow")
        return self._heap[0]

    def del_min(self) -> Any:
        heap = self._heap
        if not heap:
            raise IndexError("Priority queue underflow")
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sink(0)
        return top

    pop = del_min

    def is_empty(self) -> bool:
        return not self._heap

    def size(self) -> int:
        return len(self._heap)

    def __len__(self) -> int:
        return len(self._heap)

    def __bool__(self) -> bool:
        return bool(self._heap)

    def _swim(self, k: int) -> None:
        heap = self._heap
        item = heap[k]
        while k > 0:
            parent = (k - 1) >> 1
            if item < heap[parent]:
                heap[k] = heap[parent]
                k = parent
            else:
                break
        heap[k] = item

    def _sink(self, k: int) -> None:
        heap = self._heap
        n = len(heap)
        item = heap[k]
        while True:
            child = 2 * k + 1
            if child >= n:
                break
            if child + 1 < n and heap[child + 1] < heap[child]:
                child += 1
            if heap[child] < item:
                heap[k] = heap[child]
                k = child
            else:
                break
        heap[k] = item

"""
Path calculation module.

Implements Dijkstra's algorithm (priority-queue optimised) over an expanded
state space (station, current line[, transfers]) so that transfers at
multi-line interchange stations are handled correctly.

Supported modes:
    * shortest_distance - minimise total distance (ties -> fewer transfers)
    * least_transfer    - minimise transfers (ties -> shorter distance)
    * comprehensive     - minimise distance + transfers * transfer_penalty
"""

from __future__ import annotations

import itertools
from typing import Dict, List, Optional, Tuple

from models import Edge, Graph, MinPQ

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Equivalent distance (meters) charged per transfer in comprehensive mode.
# Models walking distance / waiting time at interchange stations.
DEFAULT_TRANSFER_PENALTY = 3000
# Weight multiplier applied (for routing score only) to non-preferred lines.
DEFAULT_NON_PREFERRED_FACTOR = 1.5

INF = float("inf")


class PathfindingConfig:
    """Configuration of a path query."""

    def __init__(self, mode: str = "shortest_distance",
                 max_transfers: Optional[int] = None,
                 preferred_lines: Optional[List[str]] = None,
                 transfer_penalty: float = DEFAULT_TRANSFER_PENALTY,
                 non_preferred_factor: float = DEFAULT_NON_PREFERRED_FACTOR):
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}', must be one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer")
            if max_transfers < 0:
                raise ValueError("max_transfers cannot be negative")
        if transfer_penalty < 0:
            raise ValueError("transfer_penalty cannot be negative")
        if non_preferred_factor < 1:
            raise ValueError("non_preferred_factor must be >= 1")
        self.mode = mode
        self.max_transfers = max_transfers
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.preferred_lines = [p.strip() for p in preferred_lines if p and p.strip()]
        self.transfer_penalty = transfer_penalty
        self.non_preferred_factor = non_preferred_factor

    def __repr__(self) -> str:
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


def _norm_line(name: Optional[str]) -> str:
    return " ".join(str(name).split()).lower() if name is not None else ""


class ShortestPath:
    """Dijkstra single-source shortest paths from ``source`` (station number)."""

    def __init__(self, graph: Graph, source: int, config: Optional[PathfindingConfig] = None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to: Dict[int, float] = {v: INF for v in graph.vertices()}
        self.dist_to[source] = 0
        self._transfers_to: Dict[int, int] = {source: 0}
        self._best_state: Dict[int, tuple] = {}
        self._parent: Dict[tuple, Optional[Tuple[tuple, Edge]]] = {}
        self._preferred = {_norm_line(p) for p in self.config.preferred_lines}
        self._run()

    # ------------------------------------------------------------ internals
    def _edge_score(self, edge: Edge) -> float:
        if self._preferred and _norm_line(edge.line) not in self._preferred:
            return edge.weight * self.config.non_preferred_factor
        return edge.weight

    def _cost(self, score_dist: float, transfers: int, dist: float) -> tuple:
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, score_dist, dist)
        if mode == "comprehensive":
            return (score_dist + transfers * self.config.transfer_penalty, transfers, dist)
        return (score_dist, transfers, dist)

    def _run(self) -> None:
        max_t = self.config.max_transfers
        track = max_t is not None
        start = (self.source, None, 0 if track else None)
        # info[state] = (distance, transfers, score_distance)
        info: Dict[tuple, Tuple[float, int, float]] = {start: (0, 0, 0)}
        best_cost: Dict[tuple, tuple] = {start: self._cost(0, 0, 0)}
        settled = set()
        self._parent[start] = None
        counter = itertools.count()
        pq = MinPQ()
        pq.insert((best_cost[start], next(counter), start))

        while pq:
            cost, _, state = pq.del_min()
            if state in settled or cost > best_cost[state]:
                continue
            settled.add(state)
            v, line, _t = state
            d, t, sd = info[state]
            for edge in self.graph.adj(v):
                w = edge.other(v)
                new_t = t + (1 if (line is not None and edge.line != line) else 0)
                if track and new_t > max_t:
                    continue
                new_d = d + edge.weight
                new_sd = sd + self._edge_score(edge)
                ns = (w, edge.line, new_t if track else None)
                if ns in settled:
                    continue
                nc = self._cost(new_sd, new_t, new_d)
                if ns not in best_cost or nc < best_cost[ns]:
                    best_cost[ns] = nc
                    info[ns] = (new_d, new_t, new_sd)
                    self._parent[ns] = (state, edge)
                    pq.insert((nc, next(counter), ns))

        # Pick, for every vertex, the best state reaching it.
        best_for_vertex: Dict[int, tuple] = {}
        for state in settled:
            v = state[0]
            if v not in best_for_vertex or best_cost[state] < best_cost[best_for_vertex[v]]:
                best_for_vertex[v] = state
        best_for_vertex[self.source] = start
        for v, state in best_for_vertex.items():
            self._best_state[v] = state
            self.dist_to[v] = info[state][0]
            self._transfers_to[v] = info[state][1]

    # ------------------------------------------------------------ queries
    def has_path_to(self, target: int) -> bool:
        return target in self._best_state

    def distance_to(self, target: int) -> float:
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target: int) -> Tuple[List[int], List[Edge]]:
        """Return (stations, edges) of the best path to ``target``.

        Returns ([], []) if ``target`` is unreachable.
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        stations: List[int] = [target]
        edges: List[Edge] = []
        while True:
            parent = self._parent.get(state)
            if parent is None:
                break
            prev_state, edge = parent
            edges.append(edge)
            stations.append(prev_state[0])
            state = prev_state
        stations.reverse()
        edges.reverse()
        return stations, edges

    def path_to(self, target: int) -> List[int]:
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target: int) -> int:
        """Number of transfers on the best path to ``target`` (-1 if unreachable)."""
        if target not in self._best_state:
            return -1
        return self._transfers_to[target]

    def get_transfers(self, target: int) -> List[Tuple[int, str, str]]:
        """List of (station number, from_line, to_line) transfers along the path."""
        stations, edges = self.path_with_edges(target)
        result = []
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                result.append((stations[i], edges[i - 1].line, edges[i].line))
        return result

    def get_lines(self, target: int) -> List[str]:
        """Ordered distinct list of lines used on the path."""
        _, edges = self.path_with_edges(target)
        lines: List[str] = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)
        return lines

"""
Fare calculation module implementing the Beijing Subway fare structure.

    0  - 6  km : 3 yuan
    6  - 12 km : 4 yuan
    12 - 22 km : 5 yuan
    22 - 32 km : 6 yuan
    > 32 km    : 6 yuan + 1 yuan per (started) additional 20 km
                 (32-52 km: 7 yuan, 52-72 km: 8 yuan, ...)

Intervals are left-closed and right-open, i.e. boundary values are charged
according to the next higher tier.  Fractional distances (meters) are
rounded up before evaluation.
"""

from __future__ import annotations

import math

# (lower bound inclusive, upper bound exclusive, fare) in meters / yuan
FARE_TIERS = [
    (0, 6000, 3),
    (6000, 12000, 4),
    (12000, 22000, 5),
    (22000, 32000, 6),
]
EXTENDED_START = 32000
EXTENDED_STEP = 20000
EXTENDED_BASE_FARE = 7

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5


def _validate_number(value, name: str) -> float:
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise TypeError(f"{name} must be a number")
    if math.isnan(value) or math.isinf(value):
        raise ValueError(f"{name} must be a finite number")
    if value < 0:
        raise ValueError(f"{name} cannot be negative")
    return value


def Sum_money(distance) -> int:
    """Return the fare (yuan) for a travel distance given in meters."""
    _validate_number(distance, "distance")
    meters = math.ceil(distance)
    for low, high, fare in FARE_TIERS:
        if low <= meters < high:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_START) // EXTENDED_STEP


def calculate_fare(distance) -> int:
    """Alias of :func:`Sum_money`."""
    return Sum_money(distance)


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated travel time in minutes.

    ~1.5 minutes per kilometer plus ~5 minutes per transfer.
    """
    _validate_number(distance, "distance")
    _validate_number(transfer_count, "transfer_count")
    return (distance / 1000.0) * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""
Data management module.

Loads the Beijing Subway network from ``subway_data.json``, builds the
station mapping and graph structures, performs data integrity checks and
supports data updates (new lines, station modification, distance updates).
"""

from __future__ import annotations

import json
import os
from collections import deque
from typing import Dict, List, Optional

from models import Edge, Graph

DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")

# Main lines that should be covered by the data set.
MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

# Representative stations (English / Chinese candidates) for main lines.
REPRESENTATIVE_STATIONS = {
    "Line 1": ["Tian'anmen East", "Tiananmen East", "Xidan", "天安门东", "西单"],
    "Line 2": ["Beijing Railway Station", "Jishuitan", "北京站", "积水潭"],
    "Line 4": ["Xidan", "Zhongguancun", "西单", "中关村"],
    "Line 5": ["Dongdan", "Yonghegong Lama Temple", "Yonghegong", "东单", "雍和宫"],
    "Line 10": ["Guomao", "Sanyuanqiao", "国贸", "三元桥"],
    "Line 13": ["Xizhimen", "Huilongguan", "西直门", "回龙观"],
}

# Module-level data structures (populated by init_global_data()).
station_to_number: Dict[str, int] = {}
graph: Dict[str, Dict[str, int]] = {}


def load_subway_data(path: str = DATA_FILE_PATH) -> dict:
    """Load and return the JSON content of the subway data file."""
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise ValueError("'stations' and 'connections' must be lists")
    return data


def build_station_mapping(data: dict) -> Dict[str, int]:
    """Map station names to numbers (in order of appearance)."""
    mapping: Dict[str, int] = {}
    for st in data.get("stations", []):
        name = st.get("name")
        if name is not None and name not in mapping:
            mapping[name] = len(mapping)
    for c in data.get("connections", []):
        for key in ("source", "dest"):
            name = c.get(key)
            if name is not None and name not in mapping:
                mapping[name] = len(mapping)
    return mapping


def build_graph_dict(data: dict) -> Dict[str, Dict[str, int]]:
    """Adjacency dict {station: {neighbor: distance}} (undirected, min distance)."""
    g: Dict[str, Dict[str, int]] = {}
    for st in data.get("stations", []):
        if st.get("name") is not None:
            g.setdefault(st["name"], {})
    for c in data.get("connections", []):
        s, d, dist = c.get("source"), c.get("dest"), c.get("distance")
        if s is None or d is None or not isinstance(dist, (int, float)):
            continue
        for a, b in ((s, d), (d, s)):
            g.setdefault(a, {})
            if b not in g[a] or dist < g[a][b]:
                g[a][b] = dist
    return g


def init_global_data(path: str = DATA_FILE_PATH) -> None:
    """Populate the module-level ``station_to_number`` and ``graph`` dicts."""
    data = load_subway_data(path)
    station_to_number.clear()
    station_to_number.update(build_station_mapping(data))
    graph.clear()
    graph.update(build_graph_dict(data))


class SubwayNetwork:
    """Container for the built network structures."""

    def __init__(self, graph_obj: Graph, station_to_number_map: Dict[str, int],
                 number_to_station_map: Dict[int, str], graph_dict: Dict[str, Dict[str, int]]):
        self.graph = graph_obj
        self.station_to_number = station_to_number_map
        self.number_to_station = number_to_station_map
        self.graph_dict = graph_dict

    def station_names(self) -> List[str]:
        return list(self.station_to_number.keys())


def build_network(data: dict) -> SubwayNetwork:
    """Build Graph and mappings from loaded data."""
    mapping = build_station_mapping(data)
    reverse = {v: k for k, v in mapping.items()}
    g = Graph()
    station_lines: Dict[str, List[str]] = {name: [] for name in mapping}
    for st in data.get("stations", []):
        name = st.get("name")
        if name is None:
            continue
        for line in st.get("lines", []) or []:
            if line not in station_lines[name]:
                station_lines[name].append(line)
    for c in data.get("connections", []):
        line = c.get("line")
        for key in ("source", "dest"):
            name = c.get(key)
            if name is not None and line and line not in station_lines[name]:
                station_lines[name].append(line)
    for name, num in mapping.items():
        g.add_station(name, num, station_lines[name])
    for c in data.get("connections", []):
        s, d, dist = c.get("source"), c.get("dest"), c.get("distance")
        if s is None or d is None or s == d:
            continue
        if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist < 0:
            continue
        g.add_edge(Edge(mapping[s], mapping[d], dist, c.get("line")))
    return SubwayNetwork(g, mapping, reverse, build_graph_dict(data))


class SubwayDataManager:
    """Manages subway data: integrity checks and updates."""

    def __init__(self, data: Optional[dict] = None, path: str = DATA_FILE_PATH):
        self.path = path
        self.data = data if data is not None else load_subway_data(path)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ---------------------------------------------------------- queries
    def _station(self, name: str) -> Optional[dict]:
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def station_names(self) -> List[str]:
        return [st.get("name") for st in self.data["stations"] if st.get("name") is not None]

    def get_lines(self) -> List[str]:
        lines: List[str] = []
        for st in self.data["stations"]:
            for line in st.get("lines", []) or []:
                if line not in lines:
                    lines.append(line)
        for c in self.data["connections"]:
            line = c.get("line")
            if line and line not in lines:
                lines.append(line)
        return lines

    def get_line_stations(self, line: str) -> List[str]:
        """Station sequence of a line derived from its connections."""
        adj: Dict[str, List[str]] = {}
        for c in self.data["connections"]:
            if c.get("line") != line:
                continue
            s, d = c.get("source"), c.get("dest")
            adj.setdefault(s, [])
            adj.setdefault(d, [])
            if d not in adj[s]:
                adj[s].append(d)
            if s not in adj[d]:
                adj[d].append(s)
        if not adj:
            return [st["name"] for st in self.data["stations"] if line in (st.get("lines") or [])]
        endpoints = [n for n, nb in adj.items() if len(nb) == 1]
        start = endpoints[0] if endpoints else next(iter(adj))
        order, seen = [], set()
        stack = [start]
        while stack:
            node = stack.pop()
            if node in seen:
                continue
            seen.add(node)
            order.append(node)
            for nb in reversed(adj[node]):
                if nb not in seen:
                    stack.append(nb)
        return order

    def get_transfer_stations(self) -> Dict[str, List[str]]:
        return {st["name"]: list(st.get("lines") or []) for st in self.data["stations"]
                if len(st.get("lines") or []) > 1}

    def build_network(self) -> SubwayNetwork:
        return build_network(self.data)

    # ---------------------------------------------------------- integrity
    def check_integrity(self) -> dict:
        errors: List[str] = []
        warnings: List[str] = []
        stations = self.data["stations"]
        connections = self.data["connections"]
        names = set()

        for i, st in enumerate(stations):
            name = st.get("name")
            if not name or not isinstance(name, str):
                errors.append(f"Station #{i} has no valid name")
                continue
            if name in names:
                warnings.append(f"Duplicate station entry: '{name}'")
            names.add(name)
            if not st.get("lines"):
                warnings.append(f"Station '{name}' has no line information")

        lines = self.get_lines()
        # Coverage of main lines
        norm_lines = {l.lower().replace(" ", ""): l for l in lines}
        for main in MAIN_LINES:
            if main.lower().replace(" ", "") not in norm_lines:
                warnings.append(f"Main line '{main}' not found in data")
        for line, candidates in REPRESENTATIVE_STATIONS.items():
            if line in lines and not any(c in names for c in candidates):
                warnings.append(f"No representative station found for {line}")

        # Connection integrity and distance accuracy
        seen_pairs = set()
        connected = set()
        station_line_map = {st.get("name"): set(st.get("lines") or []) for st in stations}
        for i, c in enumerate(connections):
            s, d, dist, line = c.get("source"), c.get("dest"), c.get("distance"), c.get("line")
            label = f"Connection #{i} ({s} - {d})"
            if s not in names:
                errors.append(f"{label}: source station '{s}' does not exist")
            if d not in names:
                errors.append(f"{label}: destination station '{d}' does not exist")
            if s == d:
                errors.append(f"{label}: connects a station to itself")
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                errors.append(f"{label}: invalid distance {dist!r}")
            elif dist <= 0:
                errors.append(f"{label}: distance must be positive (got {dist})")
            elif dist < 200 or dist > 30000:
                warnings.append(f"{label}: unusual distance {dist} meters")
            if not line:
                warnings.append(f"{label}: missing line information")
            else:
                for n in (s, d):
                    if n in station_line_map and line not in station_line_map[n]:
                        warnings.append(f"{label}: station '{n}' is not listed on {line}")
            key = (frozenset((s, d)), line)
            if key in seen_pairs:
                warnings.append(f"{label}: duplicate connection on {line}")
            seen_pairs.add(key)
            connected.add(s)
            connected.add(d)

        for n in names:
            if n not in connected:
                warnings.append(f"Station '{n}' has no connections")

        # Connectivity
        g = build_graph_dict(self.data)
        components = 0
        visited = set()
        for n in g:
            if n in visited:
                continue
            components += 1
            q = deque([n])
            visited.add(n)
            while q:
                x = q.popleft()
                for y in g[x]:
                    if y not in visited:
                        visited.add(y)
                        q.append(y)
        if components > 1:
            warnings.append(f"Network is not fully connected ({components} components)")

        return {
            "ok": not errors,
            "errors": errors,
            "warnings": warnings,
            "stats": {
                "stations": len(names),
                "connections": len(connections),
                "lines": len(lines),
                "transfer_stations": len(self.get_transfer_stations()),
                "components": components,
            },
        }

    # ---------------------------------------------------------- updates
    def add_station(self, name: str, lines: Optional[List[str]] = None) -> None:
        st = self._station(name)
        if st is None:
            self.data["stations"].append({"name": name, "lines": list(lines or [])})
        else:
            for line in lines or []:
                st.setdefault("lines", [])
                if line not in st["lines"]:
                    st["lines"].append(line)

    def add_line(self, line_name: str, stations: List[str], distances: List[int]) -> None:
        """Add a new line given its station sequence and inter-station distances."""
        if len(stations) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(stations) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        for dist in distances:
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                raise ValueError(f"Invalid distance: {dist!r}")
        for name in stations:
            self.add_station(name, [line_name])
        for i, dist in enumerate(distances):
            self.data["connections"].append({
                "source": stations[i], "dest": stations[i + 1],
                "distance": dist, "line": line_name,
            })

    def modify_station(self, name: str, new_name: Optional[str] = None,
                       lines: Optional[List[str]] = None) -> None:
        st = self._station(name)
        if st is None:
            raise KeyError(f"Station '{name}' does not exist")
        if new_name and new_name != name:
            if self._station(new_name) is not None:
                raise ValueError(f"Station '{new_name}' already exists")
            st["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            st["lines"] = list(lines)

    def update_distance(self, source: str, dest: str, distance: int,
                        line: Optional[str] = None) -> int:
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError(f"Invalid distance: {distance!r}")
        count = 0
        for c in self.data["connections"]:
            if {c.get("source"), c.get("dest")} == {source, dest} and (line is None or c.get("line") == line):
                c["distance"] = distance
                count += 1
        if count == 0:
            raise KeyError(f"No connection between '{source}' and '{dest}'")
        return count

    def save(self, path: Optional[str] = None) -> None:
        with open(path or self.path, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

"""
Path visualization and export module.

Builds route information dictionaries and renders them as console text or
JSON.
"""

from __future__ import annotations

import json
from typing import Dict, List, Optional

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters: float) -> str:
    """Show distance in meters for short trips and kilometers for long ones."""
    if meters < 1000:
        return f"{meters:g} meters ({meters / 1000:.2f} km)"
    return f"{meters / 1000:.2f} km ({meters:g} meters)"


def format_fare(fare: int) -> str:
    return f"Fare: {fare} yuan"


def build_route_info(path_names: List[str], edges, query: Optional[Dict] = None) -> Dict:
    """Build a route information dict from station names and edges."""
    distance = sum(e.distance for e in edges)
    segments: List[Dict] = []
    transfers: List[Dict] = []
    lines: List[str] = []
    for i, e in enumerate(edges):
        if not segments or segments[-1]["line"] != e.line:
            if segments:
                transfers.append({"station": path_names[i],
                                  "from_line": segments[-1]["line"], "to_line": e.line})
            segments.append({"line": e.line, "stations": [path_names[i]], "distance": 0})
        segments[-1]["stations"].append(path_names[i + 1])
        segments[-1]["distance"] += e.distance
        if e.line not in lines:
            lines.append(e.line)
    transfer_count = len(transfers)
    fare = Sum_money(distance)
    est = calculate_estimated_time(distance, transfer_count)
    return {
        "query": query or {},
        "path": list(path_names),
        "station_count": len(path_names),
        "distance": distance,
        "distance_km": round(distance / 1000.0, 3),
        "distance_display": format_distance(distance),
        "fare": fare,
        "fare_text": format_fare(fare),
        "estimated_time": round(est, 1),
        "transfer_count": transfer_count,
        "transfers": transfers,
        "lines": lines,
        "segments": segments,
    }


def build_summary(info: Dict) -> Dict:
    """Extract key information from route info."""
    return {
        "total_distance_km": info["distance_km"],
        "total_fare": info["fare"],
        "transfer_count": info["transfer_count"],
        "estimated_time_minutes": info["estimated_time"],
    }


def format_text_report(info: Dict) -> str:
    q = info.get("query", {})
    sep = "=" * 60
    lines = [sep]
    lines.append(f"Route: {q.get('start', info['path'][0] if info['path'] else '')} -> "
                 f"{q.get('end', info['path'][-1] if info['path'] else '')}")
    if q.get("mode"):
        lines.append(f"Mode: {q['mode']}")
    lines.append("-" * 60)
    lines.append(f"Stations ({info['station_count']}): " + " -> ".join(info["path"]))
    lines.append("Segments:")
    for i, seg in enumerate(info["segments"]):
        if i > 0:
            t = info["transfers"][i - 1]
            lines.append(f"  >> Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
        stops = len(seg["stations"]) - 1
        lines.append(f"  [{seg['line']}] " + " -> ".join(seg["stations"]) +
                     f"  ({stops} stop{'s' if stops != 1 else ''}, {format_distance(seg['distance'])})")
    lines.append(f"Total distance: {info['distance_display']}")
    if info["transfers"]:
        at = ", ".join(t["station"] for t in info["transfers"])
        lines.append(f"Transfers: {info['transfer_count']} (at {at})")
    else:
        lines.append("Transfers: 0")
    lines.append("Lines involved: " + ", ".join(str(l) for l in info["lines"]))
    lines.append(f"Estimated time: {info['estimated_time']:.1f} minutes")
    lines.append(info["fare_text"])
    lines.append(sep)
    return "\n".join(lines)


def to_json_dict(info: Dict) -> Dict:
    return {
        "query": info.get("query", {}),
        "path": info["path"],
        "distance": {"meters": info["distance"], "kilometers": info["distance_km"]},
        "fare": info["fare"],
        "estimated_time": info["estimated_time"],
        "transfer_count": info["transfer_count"],
        "transfers": info["transfers"],
        "lines": info["lines"],
        "segments": info["segments"],
    }


def format_json_report(info: Dict) -> str:
    return json.dumps(to_json_dict(info), ensure_ascii=False, indent=2)

"""
Command-line interface for the Beijing Subway fare calculation system.

Usage:
    python src/main.py [--output_file FILE] [--format text|json]
                       [--mode shortest_distance|least_transfer|comprehensive]
                       [--max_transfers N] [--line_preference "Line 1"]
                       [--check_data]

Enter queries as "Starting Station; Ending Station".  Type 'quit' to exit.
"""

from __future__ import annotations

import argparse
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, SubwayDataManager, load_subway_data  # noqa: E402
from pathfinding import VALID_MODES, PathfindingConfig, ShortestPath  # noqa: E402
from visualization import build_route_info, format_json_report, format_text_report, to_json_dict  # noqa: E402

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
EXIT_COMMANDS = {"quit", "exit", "q"}


class QueryError(Exception):
    """Raised for invalid user input."""


def _normalize(name: str) -> str:
    return " ".join(name.split()).lower()


class StationResolver:
    def __init__(self, names):
        self.names = set(names)
        self.norm = {}
        for n in names:
            self.norm.setdefault(_normalize(n), n)

    def resolve(self, raw: str):
        name = raw.strip()
        if name in self.names:
            return name
        key = _normalize(name)
        if key in self.norm:
            return self.norm[key]
        for suffix in (" station", "站"):
            if key.endswith(suffix) and key[: -len(suffix)].strip() in self.norm:
                return self.norm[key[: -len(suffix)].strip()]
        return None


def parse_query(text: str):
    text = text.replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2:
        raise QueryError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryError(FORMAT_ERROR)
    return start, end


class SubwayFareApp:
    def __init__(self, network, config, output_format="text", output_file=None):
        self.network = network
        self.config = config
        self.output_format = output_format
        self.output_file = output_file
        self.resolver = StationResolver(network.station_names())
        self._results = []
        self._file_started = False

    def query(self, text: str) -> dict:
        start_raw, end_raw = parse_query(text)
        start = self.resolver.resolve(start_raw)
        if start is None:
            raise QueryError(f"Starting station '{start_raw}' does not exist, please check the station name")
        end = self.resolver.resolve(end_raw)
        if end is None:
            raise QueryError(f"Ending station '{end_raw}' does not exist, please check the station name")
        if start == end:
            raise QueryError("Starting station cannot be the same as the ending station")
        s_num = self.network.station_to_number[start]
        e_num = self.network.station_to_number[end]
        try:
            sp = ShortestPath(self.network.graph, s_num, self.config)
            path, edges = sp.path_with_edges(e_num)
        except Exception as exc:  # path calculation exception handling
            raise QueryError(f"Path calculation error: {exc}")
        if not path:
            msg = f"No route found from '{start}' to '{end}'"
            if self.config.max_transfers is not None:
                msg += f" within {self.config.max_transfers} transfer(s)"
            raise QueryError(msg)
        names = [self.network.number_to_station[n] for n in path]
        query = {
            "start": start, "end": end, "mode": self.config.mode,
            "max_transfers": self.config.max_transfers,
            "line_preference": self.config.preferred_lines,
        }
        try:
            return build_route_info(names, edges, query)
        except Exception as exc:  # fare calculation exception handling
            raise QueryError(f"Fare calculation error: {exc}")

    def render(self, info: dict) -> str:
        return format_json_report(info) if self.output_format == "json" else format_text_report(info)

    def export(self, info: dict) -> None:
        if not self.output_file:
            return
        if self.output_format == "json":
            self._results.append(to_json_dict(info))
            content = self._results[0] if len(self._results) == 1 else self._results
            with open(self.output_file, "w", encoding="utf-8") as f:
                json.dump(content, f, ensure_ascii=False, indent=2)
        else:
            mode = "a" if self._file_started else "w"
            with open(self.output_file, mode, encoding="utf-8") as f:
                f.write(format_text_report(info) + "\n")
        self._file_started = True

    def handle(self, text: str) -> bool:
        """Process one input line; returns True on success."""
        try:
            info = self.query(text)
        except QueryError as exc:
            print(f"Error: {exc}")
            return False
        except Exception as exc:
            print(f"Error: unexpected failure: {exc}")
            return False
        print(self.render(info))
        try:
            self.export(info)
            if self.output_file:
                print(f"Result exported to {self.output_file}")
        except OSError as exc:
            print(f"Error: cannot write output file: {exc}")
        return True

    def run(self) -> None:
        print("Beijing Subway Fare Calculation System")
        print("Enter 'Starting Station; Ending Station' (type 'quit' to exit).")
        while True:
            try:
                line = input("Query> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            line = line.strip()
            if not line:
                continue
            if line.lower() in EXIT_COMMANDS:
                break
            self.handle(line)
        print("Goodbye!")


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", default="text", choices=["text", "json"], help="Output format")
    p.add_argument("--mode", default="shortest_distance", choices=list(VALID_MODES), help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None, help="Preferred line(s), comma separated")
    p.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    return p.parse_args(argv)


def print_check_report(report: dict) -> None:
    print("Data integrity check")
    for k, v in report["stats"].items():
        print(f"  {k}: {v}")
    print(f"Errors: {len(report['errors'])}")
    for e in report["errors"]:
        print(f"  [ERROR] {e}")
    print(f"Warnings: {len(report['warnings'])}")
    for w in report["warnings"]:
        print(f"  [WARNING] {w}")
    print("Result: " + ("PASSED" if report["ok"] else "FAILED"))


def main(argv=None) -> int:
    args = parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers cannot be negative")
        return 2
    try:
        data = load_subway_data(args.data_file)
    except (OSError, ValueError) as exc:
        print(f"Error: failed to load subway data: {exc}")
        return 1
    manager = SubwayDataManager(data, args.data_file)
    if args.check_data:
        report = manager.check_integrity()
        print_check_report(report)
        return 0 if report["ok"] else 1

    network = manager.build_network()
    preferred = []
    if args.line_preference:
        preferred = [p.strip() for p in args.line_preference.split(",") if p.strip()]
        known = {l.lower() for l in manager.get_lines()}
        for p in preferred:
            if p.lower() not in known:
                print(f"Warning: preferred line '{p}' not found in data")
    try:
        config = PathfindingConfig(args.mode, args.max_transfers, preferred)
    except ValueError as exc:
        print(f"Error: {exc}")
        return 2
    app = SubwayFareApp(network, config, args.format, args.output_file)
    app.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

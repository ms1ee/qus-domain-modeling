"""Global variable storage for the food delivery dispatch simulator.

The simulator keeps its shared runtime state (funds, riders, orders, the
simulated clock, ...) in a single module-level dictionary.  Access it with
:func:`set_value` and :func:`get_value`.

Well-known keys:

* ``"funds"``              -- int, system funds (may be negative)
* ``"riders"``             -- list[Rider]
* ``"rider_id_counter"``   -- int, last rider id handed out
* ``"running"``            -- bool, main loop flag
* ``"tick"``               -- int, current simulated time
* ``"orders"``             -- list[Order], order history
* ``"order_id_counter"``   -- int, last order id handed out
* ``"recruitment_cost"``   -- int, cost of recruiting one rider
* ``"max_riders"``         -- int, maximum number of riders
* ``"rider_speed"``        -- int, default rider speed (units/tick)
"""

_global_dict = {}


def _init():
    """(Re)initialise the global dictionary, discarding all stored values."""
    global _global_dict
    _global_dict = {}


def set_value(key, value):
    """Store ``value`` under ``key``."""
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` or ``default`` if missing."""
    return _global_dict.get(key, default)


def has_value(key):
    """Return True if ``key`` has been set."""
    return key in _global_dict


def all_values():
    """Return a shallow copy of the whole global dictionary."""
    return dict(_global_dict)

"""Grid based path planning.

The map is 800x800 units divided in 8x8 cells of 100x100 units.  Streets
run along the grid lines (x or y a multiple of 100).  Restaurants and
customers are located at cell centres (``(v // 100) * 100 + 50``).  A rider
serves a location from its *access point*, which is the middle of the
street segment just below the cell centre: ``(cx, cy - 50)``.
"""

import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE


def manhattan(a, b):
    """Return the Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class Route:
    """A planned route made of axis-aligned waypoints.

    Attributes:
        waypoints: list of (x, y) tuples, including the start point.
        length: total travelled distance in units.
        speed: rider speed used for the time estimate.
        estimated_time: ticks needed at ``speed`` (ceil).
    """

    def __init__(self, waypoints, speed=None):
        self.waypoints = [tuple(p) for p in waypoints]
        self.length = sum(manhattan(self.waypoints[i], self.waypoints[i + 1])
                          for i in range(len(self.waypoints) - 1))
        self.speed = speed
        if speed:
            self.estimated_time = int(math.ceil(self.length / float(speed)))
        else:
            self.estimated_time = 0

    def __repr__(self):
        return "Route(length=%d, waypoints=%r)" % (self.length, self.waypoints)


class PathPlanner:
    """Street-following path planner for the 8x8 grid."""

    @staticmethod
    def clamp(value, low=0, high=MAP_SIZE):
        """Clamp ``value`` to ``[low, high]``."""
        return max(low, min(high, int(value)))

    @staticmethod
    def to_grid_center_value(v):
        """Convert one input coordinate to the centre of its cell."""
        v = PathPlanner.clamp(v, 0, MAP_SIZE - 1)
        return (v // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2

    @staticmethod
    def to_grid_center(x, y):
        """Convert an input point to the centre of the grid cell it lies in.

        Boundary values (e.g. 800) are mapped into the last cell.
        """
        return (PathPlanner.to_grid_center_value(x),
                PathPlanner.to_grid_center_value(y))

    @staticmethod
    def cell_index(pos):
        """Return the (column, row) cell index of a point."""
        x = PathPlanner.clamp(pos[0], 0, MAP_SIZE - 1)
        y = PathPlanner.clamp(pos[1], 0, MAP_SIZE - 1)
        return x // CELL_SIZE, y // CELL_SIZE

    @staticmethod
    def access_point(pos):
        """Return the street point from which a cell-centre location is served."""
        cx, cy = PathPlanner.to_grid_center(pos[0], pos[1])
        return cx, cy - CELL_SIZE // 2

    @staticmethod
    def is_on_street(x, y):
        """Return True if the point lies on a street (a grid line)."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def nearest_street_point(x, y):
        """Return the closest point on a street (axis aligned move)."""
        if PathPlanner.is_on_street(x, y):
            return x, y
        nx = int(round(x / float(CELL_SIZE))) * CELL_SIZE
        ny = int(round(y / float(CELL_SIZE))) * CELL_SIZE
        nx = PathPlanner.clamp(nx)
        ny = PathPlanner.clamp(ny)
        if abs(nx - x) <= abs(ny - y):
            return nx, y
        return x, ny

    @staticmethod
    def _plan_horizontal(start, target):
        """Plan assuming ``target`` lies on a horizontal street."""
        special = False
        pts = [tuple(start)]
        x, y = start
        tx, ty = target
        if not PathPlanner.is_on_street(x, y):
            x, y = PathPlanner.nearest_street_point(x, y)
            pts.append((x, y))
        if (x, y) == (tx, ty):
            return pts, special
        if y == ty:
            pts.append((tx, ty))
        elif x % CELL_SIZE == 0:
            pts.append((x, ty))
            pts.append((tx, ty))
        else:
            # On a horizontal street, must reach a vertical street first.
            if x == tx:
                special = True  # directly above / below the target
            if tx % CELL_SIZE == 0:
                vx = tx
            else:
                lo = (tx // CELL_SIZE) * CELL_SIZE
                hi = lo + CELL_SIZE
                if x <= lo:
                    vx = lo
                elif x >= hi:
                    vx = hi
                else:
                    vx = lo if (x - lo) <= (hi - x) else hi
            pts.append((vx, y))
            pts.append((vx, ty))
            pts.append((tx, ty))
        return pts, special

    @staticmethod
    def plan(start, target):
        """Plan a street path from ``start`` to ``target``.

        Returns:
            (waypoints, special_case) where waypoints includes the start and
            the target, and special_case is True when a detour was needed
            because the rider was directly above/below the target.
        """
        start = (int(start[0]), int(start[1]))
        target = (int(target[0]), int(target[1]))
        if target[1] % CELL_SIZE != 0 and target[0] % CELL_SIZE == 0:
            # Target on a vertical street: solve the mirrored problem.
            pts, special = PathPlanner._plan_horizontal(
                (start[1], start[0]), (target[1], target[0]))
            pts = [(p[1], p[0]) for p in pts]
        elif target[1] % CELL_SIZE != 0:
            target = PathPlanner.nearest_street_point(*target)
            pts, special = PathPlanner.plan(start, target)
            return pts, special
        else:
            pts, special = PathPlanner._plan_horizontal(start, target)
        cleaned = []
        for p in pts:
            if not cleaned or cleaned[-1] != p:
                cleaned.append(p)
        return cleaned, special

    @staticmethod
    def plan_multi(start, targets, speed=None):
        """Plan a continuous route through several targets in order."""
        waypoints = [tuple(start)]
        current = tuple(start)
        for t in targets:
            pts, _ = PathPlanner.plan(current, t)
            for p in pts[1:]:
                if waypoints[-1] != p:
                    waypoints.append(p)
            current = tuple(t)
        return Route(waypoints, speed)

    @staticmethod
    def street_distance(a, b):
        """Return the street path length between two street points."""
        pts, _ = PathPlanner.plan(a, b)
        return Route(pts).length

"""Order data model."""

import globalvar
from path_planner import PathPlanner


class OrderStatus:
    """Possible order statuses."""
    PENDING = "pending"
    ASSIGNED = "assigned"
    PICKING_UP = "picking_up"
    DELIVERING = "delivering"
    COMPLETED = "completed"


def _validate_pos(pos, name):
    try:
        if len(pos) != 2:
            raise ValueError
        return int(pos[0]), int(pos[1])
    except (TypeError, ValueError):
        raise ValueError("%s must be a pair of integers" % name)


class Order:
    """A delivery order from a restaurant to a customer.

    Input coordinates are converted to grid cell centres.

    Attributes:
        id: unique order id.
        raw_res_pos / raw_cus_pos: coordinates as entered.
        res_pos / cus_pos: grid centre coordinates.
        restaurant_x, restaurant_y, customer_x, customer_y: same values.
        status: "pending", "assigned", "picking_up", "delivering", "completed".
        rider_id: id of the assigned rider (or None).
        start_time, assign_time, pickup_time, completion_time: ticks.
        distance: street distance restaurant -> customer.
        time_taken: completion_time - start_time once completed.
    """

    def __init__(self, res_pos, cus_pos, start_time=None):
        res = _validate_pos(res_pos, "res_pos")
        cus = _validate_pos(cus_pos, "cus_pos")
        self.raw_res_pos = res
        self.raw_cus_pos = cus
        self.res_pos = PathPlanner.to_grid_center(*res)
        self.cus_pos = PathPlanner.to_grid_center(*cus)
        self.restaurant_x, self.restaurant_y = self.res_pos
        self.customer_x, self.customer_y = self.cus_pos
        counter = globalvar.get_value("order_id_counter", 0) or 0
        counter += 1
        globalvar.set_value("order_id_counter", counter)
        self.id = counter
        self.status = OrderStatus.PENDING
        self.rider_id = None
        if start_time is None:
            start_time = globalvar.get_value("tick", 0) or 0
        self.start_time = start_time
        self.assign_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.distance = PathPlanner.street_distance(
            PathPlanner.access_point(self.res_pos),
            PathPlanner.access_point(self.cus_pos))
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(self)

    @property
    def completed(self):
        """True once the order has been delivered."""
        return self.status == OrderStatus.COMPLETED

    @property
    def res_access(self):
        """Street point where the rider picks up the order."""
        return PathPlanner.access_point(self.res_pos)

    @property
    def cus_access(self):
        """Street point where the rider hands over the order."""
        return PathPlanner.access_point(self.cus_pos)

    def assign_to(self, rider_id):
        """Assign the order to the rider with id ``rider_id``."""
        self.rider_id = rider_id
        if self.status == OrderStatus.PENDING:
            self.status = OrderStatus.ASSIGNED
        self.assign_time = globalvar.get_value("tick", 0) or 0

    def mark_picking_up(self):
        """Rider is heading to the restaurant."""
        if self.status != OrderStatus.COMPLETED:
            self.status = OrderStatus.PICKING_UP

    def mark_picked_up(self, time):
        """Rider picked the food up at the restaurant."""
        self.pickup_time = time
        self.status = OrderStatus.DELIVERING

    def complete(self, time):
        """Mark the order delivered at ``time``."""
        self.completion_time = time
        self.time_taken = time - (self.start_time or 0)
        self.status = OrderStatus.COMPLETED

    def to_dict(self):
        """Serialise to a JSON friendly dict."""
        return {
            "id": self.id,
            "raw_res_pos": list(self.raw_res_pos),
            "raw_cus_pos": list(self.raw_cus_pos),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "assign_time": self.assign_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from :meth:`to_dict` output (not registered)."""
        obj = cls.__new__(cls)
        obj.id = int(data["id"])
        obj.raw_res_pos = tuple(data.get("raw_res_pos", data["res_pos"]))
        obj.raw_cus_pos = tuple(data.get("raw_cus_pos", data["cus_pos"]))
        obj.res_pos = tuple(data["res_pos"])
        obj.cus_pos = tuple(data["cus_pos"])
        obj.restaurant_x, obj.restaurant_y = obj.res_pos
        obj.customer_x, obj.customer_y = obj.cus_pos
        obj.status = data.get("status", OrderStatus.PENDING)
        obj.rider_id = data.get("rider_id")
        obj.start_time = data.get("start_time", 0)
        obj.assign_time = data.get("assign_time")
        obj.pickup_time = data.get("pickup_time")
        obj.completion_time = data.get("completion_time")
        obj.time_taken = data.get("time_taken")
        obj.distance = data.get("distance", 0)
        return obj

    def __repr__(self):
        return "Order(id=%d, res=%r, cus=%r, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

"""Rider model: movement, state machine and status log."""

import logging

import globalvar
from path_planner import PathPlanner, Route, MAP_SIZE

logger = logging.getLogger(__name__)

DEFAULT_SPEED = 20


class RiderState:
    """Rider state enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2

    NAMES = {0: "idle", 1: "picking up", 2: "delivering"}

    @classmethod
    def name(cls, state):
        """Human readable name of a state."""
        return cls.NAMES.get(state, "unknown")


def _now():
    return globalvar.get_value("tick", 0) or 0


class Rider:
    """A delivery rider.

    Attributes:
        rider_id: unique id.
        x, y: current position (always on a street).
        speed: units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: assigned orders (served first-in first-out).
        special_case: True while a detour around the target is required.
        log: list of event dicts with keys "event", "order_id", "time",
            "x", "y", "state".
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = rider_id
        if x is None or y is None:
            x = (rider_id * 300) % MAP_SIZE
            y = (rider_id * 500) % MAP_SIZE
        x, y = PathPlanner.nearest_street_point(PathPlanner.clamp(x),
                                                PathPlanner.clamp(y))
        self.x = x
        self.y = y
        if speed is None:
            speed = globalvar.get_value("rider_speed", DEFAULT_SPEED) or DEFAULT_SPEED
        self.speed = int(speed)
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.delivered_count = 0
        self.distance_travelled = 0
        self._acknowledged = set()
        self._log_event("spawn", None)

    # ------------------------------------------------------------------ utils
    @property
    def position(self):
        """Current (x, y)."""
        return self.x, self.y

    def _log_event(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": _now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def current_order(self):
        """Order currently being served, or None."""
        return self.orders[0] if self.orders else None

    def current_target(self):
        """Street point the rider is currently heading to (or None)."""
        order = self.current_order()
        if order is None:
            return None
        if self.state == RiderState.DELIVERING:
            return PathPlanner.access_point(order.cus_pos)
        return PathPlanner.access_point(order.res_pos)

    def remaining_targets(self):
        """All remaining stops in order of service."""
        targets = []
        for i, order in enumerate(self.orders):
            if not (i == 0 and self.state == RiderState.DELIVERING):
                targets.append(PathPlanner.access_point(order.res_pos))
            targets.append(PathPlanner.access_point(order.cus_pos))
        return targets

    def planned_route(self):
        """Return the full :class:`Route` through all remaining stops."""
        return PathPlanner.plan_multi(self.position, self.remaining_targets(),
                                      self.speed)

    def final_position(self):
        """Position after finishing all assigned orders."""
        targets = self.remaining_targets()
        return targets[-1] if targets else self.position

    def remaining_work(self):
        """Remaining travel distance for all assigned orders."""
        return self.planned_route().length

    # --------------------------------------------------------------- movement
    def _advance(self, waypoints):
        remaining = self.speed
        for tx, ty in waypoints:
            if remaining <= 0:
                break
            dx = tx - self.x
            dy = ty - self.y
            dist = abs(dx) + abs(dy)
            if dist <= remaining:
                self.x, self.y = tx, ty
                remaining -= dist
                self.distance_travelled += dist
            else:
                if dx != 0:
                    self.x += remaining if dx > 0 else -remaining
                else:
                    self.y += remaining if dy > 0 else -remaining
                self.distance_travelled += remaining
                remaining = 0

    def move(self):
        """Advance the rider by one tick.

        Handles acknowledgement of new orders, state transitions, movement
        along streets and order completion, logging every key event.
        """
        # Drop orders completed elsewhere.
        for order in list(self.orders):
            if getattr(order, "status", None) == "completed":
                self.orders.remove(order)
                self._acknowledged.discard(order.id)

        for order in self.orders:
            if order.id not in self._acknowledged:
                self._acknowledged.add(order.id)
                if getattr(order, "rider_id", None) is None:
                    order.assign_to(self.rider_id)
                self._log_event("accepted_order", order.id)

        if not self.orders:
            self.state = RiderState.IDLE
            self.special_case = False
            return

        order = self.orders[0]
        if self.state == RiderState.IDLE:
            self.state = RiderState.PICKING_UP
            order.mark_picking_up()

        target = self.current_target()
        waypoints, special = PathPlanner.plan(self.position, target)
        if special:
            self.special_case = True
        self._advance(waypoints[1:])

        if self.position == tuple(target):
            now = _now()
            if self.state == RiderState.PICKING_UP:
                self.state = RiderState.DELIVERING
                self.special_case = False
                order.mark_picked_up(now)
                self._log_event("arrived_at_merchant", order.id)
            else:
                order.complete(now)
                self.orders.pop(0)
                self._acknowledged.discard(order.id)
                self.state = RiderState.IDLE
                self.special_case = False
                self.delivered_count += 1
                self._log_event("delivered_to_customer", order.id)

    # ---------------------------------------------------------- persistence
    def to_dict(self):
        """Serialise to a JSON friendly dict."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "log": list(self.log),
            "delivered_count": self.delivered_count,
            "distance_travelled": self.distance_travelled,
            "acknowledged": sorted(self._acknowledged),
        }

    @classmethod
    def from_dict(cls, data, orders_by_id=None):
        """Rebuild a rider from :meth:`to_dict` output."""
        orders_by_id = orders_by_id or {}
        obj = cls.__new__(cls)
        obj.rider_id = int(data["rider_id"])
        obj.x = int(data["x"])
        obj.y = int(data["y"])
        obj.speed = int(data.get("speed", DEFAULT_SPEED))
        obj.state = int(data.get("state", 0))
        obj.orders = [orders_by_id[i] for i in data.get("orders", [])
                      if i in orders_by_id]
        obj.special_case = bool(data.get("special_case", False))
        obj.log = list(data.get("log", []))
        obj.delivered_count = int(data.get("delivered_count", 0))
        obj.distance_travelled = int(data.get("distance_travelled", 0))
        obj._acknowledged = set(data.get("acknowledged", []))
        return obj

    def __repr__(self):
        return "Rider(id=%s, pos=(%d, %d), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, RiderState.name(self.state),
            len(self.orders))

"""System initialisation and resource management."""

import logging

import globalvar
from rider import Rider, RiderState

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruitment_cost": 100,
    "max_riders": 20,
    "rider_speed": 20,
    "ticks_per_command": 1,
}


def init_system(config=None):
    """Initialise all global runtime state.

    Args:
        config: optional dict overriding :data:`DEFAULT_CONFIG` values.

    Returns:
        The effective configuration dict.
    """
    cfg = dict(DEFAULT_CONFIG)
    if config:
        for key, value in config.items():
            if key in DEFAULT_CONFIG:
                try:
                    cfg[key] = int(value)
                except (TypeError, ValueError):
                    logger.warning("Ignoring invalid config value %s=%r", key, value)
    globalvar._init()
    globalvar.set_value("funds", int(cfg["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)
    globalvar.set_value("recruitment_cost", int(cfg["recruitment_cost"]))
    globalvar.set_value("max_riders", int(cfg["max_riders"]))
    globalvar.set_value("rider_speed", int(cfg["rider_speed"]))
    globalvar.set_value("ticks_per_command", max(0, int(cfg["ticks_per_command"])))
    return cfg


def get_riders():
    """Return the global rider list (creating it if needed)."""
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    return riders


def can_recruit():
    """True if funds and the rider limit allow a new rider."""
    funds = globalvar.get_value("funds", 0) or 0
    cost = globalvar.get_value("recruitment_cost", DEFAULT_CONFIG["recruitment_cost"])
    max_riders = globalvar.get_value("max_riders", DEFAULT_CONFIG["max_riders"])
    return funds >= cost and len(get_riders()) < max_riders


def recruit_rider(force=False):
    """Recruit a new rider, deducting the recruitment cost.

    Returns:
        The new Rider, or None if recruitment is not possible.
    """
    if not force and not can_recruit():
        return None
    riders = get_riders()
    counter = globalvar.get_value("rider_id_counter", 0) or 0
    existing = [r.rider_id for r in riders]
    if existing:
        counter = max(counter, max(existing))
    counter += 1
    rider = Rider(counter)
    riders.append(rider)
    cost = globalvar.get_value("recruitment_cost", DEFAULT_CONFIG["recruitment_cost"])
    globalvar.set_value("funds", (globalvar.get_value("funds", 0) or 0) - cost)
    globalvar.set_value("rider_id_counter", counter)
    logger.info("Recruited rider %d", counter)
    return rider


def advance_tick(n=1):
    """Advance the simulation ``n`` ticks, moving every rider."""
    for _ in range(max(0, int(n))):
        globalvar.set_value("tick", (globalvar.get_value("tick", 0) or 0) + 1)
        for rider in list(get_riders()):
            try:
                rider.move()
            except Exception:  # keep the simulation alive
                logger.exception("Rider %s failed to move", rider.rider_id)


def all_idle():
    """True when no rider has outstanding orders."""
    return all(not r.orders for r in get_riders())


def check_consistency():
    """Check and repair global state. Returns a list of repair messages."""
    fixes = []
    funds = globalvar.get_value("funds", 0)
    if not isinstance(funds, int):
        try:
            globalvar.set_value("funds", int(funds))
        except (TypeError, ValueError):
            globalvar.set_value("funds", 0)
        fixes.append("funds reset to integer")
    if not isinstance(globalvar.get_value("tick", 0), int):
        globalvar.set_value("tick", 0)

            fixes.append("rider %s set idle" % rider.rider_id)

        counter = globalvar.get_value("rider_id_counter", 0) or 0
        if counter < max(seen):

"""Order creation, assignment and completion handling."""

import logging

import globalvar
import system
from order_model import Order
from path_planner import PathPlanner, manhattan
from rider import RiderState

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Select the most suitable rider for ``order``.

    Strategy:
      1. The idle rider closest (Manhattan distance) to the restaurant.
      2. Otherwise, recruit a new rider if funds and limits allow.
      3. Otherwise, the busy rider able to reach the restaurant soonest
         (remaining work + distance from its final stop), which balances load.

    Returns:
        A Rider, or None if no rider exists and none can be recruited.
    """
    riders = system.get_riders()
    target = PathPlanner.access_point(order.res_pos)
    idle = [r for r in riders if r.state == RiderState.IDLE and not r.orders]
    if idle:
        return min(idle, key=lambda r: (manhattan(r.position, target), r.rider_id))
    new_rider = system.recruit_rider()
    if new_rider is not None:
        return new_rider
    if not riders:
        return None
    return min(riders, key=lambda r: (
        r.remaining_work() + manhattan(r.final_position(), target),
        len(r.orders), r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion queries."""

    @staticmethod
    def create_order(coords):
        """Create an order from ``[rx, ry, cx, cy]`` (validated 0-800)."""
        if len(coords) != 4:
            raise ValueError("exactly four coordinates are required")
        values = []
        for c in coords:
            v = int(c)
            if v < 0 or v > 800:
                raise ValueError("coordinate %d out of range 0-800" % v)
            values.append(v)
        return Order((values[0], values[1]), (values[2], values[3]))

    @staticmethod
    def assign_order(order):
        """Assign ``order`` to the best rider.

        Returns:
            (rider, recruited) -- recruited is True if a new rider was hired.
        """
        before = len(system.get_riders())
        rider = find_best_rider_for_order(order)
        recruited = len(system.get_riders()) > before
        if rider is None:
            logger.warning("No rider available for order %d", order.id)
            return None, recruited
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        return rider, recruited

    @staticmethod
    def all_orders():
        """All orders ever created."""
        return list(globalvar.get_value("orders", []) or [])

    @staticmethod
    def completed_orders():
        """Orders that have been delivered."""
        return [o for o in OrderManager.all_orders() if o.completed]

    @staticmethod
    def pending_orders():
        """Orders not yet delivered."""
        return [o for o in OrderManager.all_orders() if not o.completed]

    @staticmethod
    def efficiency_stats():
        """Return (completed_count, average_time, average_distance)."""
        done = OrderManager.completed_orders()
        if not done:
            return 0, 0.0, 0.0
        avg_t = sum(o.time_taken or 0 for o in done) / float(len(done))
        avg_d = sum(o.distance for o in done) / float(len(done))
        return len(done), avg_t, avg_d

import globalvar
from path_planner import PathPlanner, MAP_SIZE
from rider import RiderState

STEP = 50
SIZE = MAP_SIZE // STEP + 1  # 17 x 17 character grid

SYMBOLS = {
    "intersection": "+",
    "h_street": "-",
    "v_street": "|",
    "empty": ".",
    "restaurant": "R",
    "customer": "C",
    "route": "*",
    "multi_rider": "#",
}


def _cell(pos):
    col = int(pos[0] / float(STEP) + 0.5)
    row = SIZE - 1 - int(pos[1] / float(STEP) + 0.5)
    return max(0, min(SIZE - 1, row)), max(0, min(SIZE - 1, col))


class DisplayManager:
    """Builds textual views of the simulation state."""

    @staticmethod
    def _base_grid():
        grid = []
        for r in range(SIZE):
            row = []
            for c in range(SIZE):
                if r % 2 == 0 and c % 2 == 0:
                    row.append(SYMBOLS["intersection"])
                elif r % 2 == 0:
                    row.append(SYMBOLS["h_street"])
                elif c % 2 == 0:
                    row.append(SYMBOLS["v_street"])
                else:
                    row.append(SYMBOLS["empty"])
            grid.append(row)
        return grid

    @staticmethod
    def _mark_route(grid, waypoints):
        for i in range(len(waypoints) - 1):
            (x1, y1), (x2, y2) = waypoints[i], waypoints[i + 1]
            length = abs(x2 - x1) + abs(y2 - y1)
            steps = max(1, length // (STEP // 2))
            for s in range(steps + 1):
                x = x1 + (x2 - x1) * s / float(steps)
                y = y1 + (y2 - y1) * s / float(steps)
                r, c = _cell((x, y))
                if grid[r][c] in "+-|":
                    grid[r][c] = SYMBOLS["route"]

    @staticmethod
    def render_map(riders=None, orders=None, route=None):
        """Render the 8x8 grid as text.

        Riders are shown with the last digit of their id (``#`` if several
        share a spot), active restaurants as ``R``, customers as ``C`` and an
        optional route as ``*``.
        """
        if riders is None:
            riders = globalvar.get_value("riders", []) or []
        if orders is None:
            orders = [o for o in (globalvar.get_value("orders", []) or [])
                      if not o.completed]
        grid = DisplayManager._base_grid()
        if route is not None:
            DisplayManager._mark_route(grid, route.waypoints)
        for o in orders:
            r, c = _cell(o.res_pos)
            grid[r][c] = SYMBOLS["restaurant"]
            r, c = _cell(o.cus_pos)
            grid[r][c] = SYMBOLS["customer"] if grid[r][c] != "R" else "X"
        occupied = {}
        for rider in riders:
            cell = _cell(rider.position)
            occupied.setdefault(cell, []).append(rider)
        for (r, c), lst in occupied.items():
            grid[r][c] = (str(lst[0].rider_id % 10) if len(lst) == 1
                          else SYMBOLS["multi_rider"])
        lines = []
        for idx, row in enumerate(grid):
            y_val = (SIZE - 1 - idx) * STEP
            label = "%4d " % y_val if idx % 2 == 0 else "     "
            lines.append(label + " ".join(row))
        axis = "     " + " ".join(
            (str(c * STEP // 100) if c % 2 == 0 else " ") for c in range(SIZE))
        lines.append(axis + "   (x100)")
        lines.append("Legend: digit=rider  #=several riders  R=restaurant  "

        return "\n".join(lines)

"""Command parsing, shortcuts, history and input validation."""

try:  # optional line editing / arrow-key history
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - Windows without readline
    readline = None

COMMANDS = ("start", "status", "orders", "riders", "map", "tick", "run",
            "save", "load", "history", "help", "quit")

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "m": "map", "t": "tick", "ru": "run", "hi": "history",
    "h": "help", "?": "help", "q": "quit", "exit": "quit",
}


class InputManager:
    """Reads and validates user commands."""

    def __init__(self, input_func=input):
        self.command_history = []
        self._input = input_func

    def read_line(self, prompt="> "):
        """Read a line; EOF/interrupt becomes ``quit``."""
        try:
            return self._input(prompt)
        except (EOFError, KeyboardInterrupt):
            return "quit"

    def parse(self, line):
        """Parse a line into (command, args).

        Supports shortcuts and ``!n`` / ``!!`` history recall.  Returns
        (None, []) for an empty line and ("unknown", [word]) for unknown
        commands.
        """
        line = (line or "").strip()
        if not line:
            return None, []
        if line.startswith("!"):
            ref = line[1:]
            if not self.command_history:
                return "error", ["history is empty"]
            if ref == "!":
                line = self.command_history[-1]
            else:
                try:
                    idx = int(ref)
                    line = self.command_history[idx - 1]
                except (ValueError, IndexError):
                    return "error", ["invalid history reference: %s" % line]
        self.command_history.append(line)
        if len(self.command_history) > 1000:
            del self.command_history[:-1000]
        tokens = line.replace(",", " ").replace("[", " ").replace("]", " ").split()
        word = tokens[0].lower()
        cmd = SHORTCUTS.get(word, word)
        if cmd not in COMMANDS:
            return "unknown", [word]
        return cmd, tokens[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Validate four integer coordinates in range 0-800.

        Raises:
            ValueError with a user friendly message.
        """
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(tokens) != 4:
            raise ValueError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for t in tokens:
            try:
                v = int(t)
            except (TypeError, ValueError):
                raise ValueError("'%s' is not an integer" % t)
            if v < 0 or v > 800:
                raise ValueError("Coordinate %d out of range (0-800)" % v)
            values.append(v)
        return values

    @staticmethod
    def parse_positive_int(tokens, default=1, maximum=100000):
        """Parse an optional positive integer argument."""
        if not tokens:
            return default
        try:
            v = int(tokens[0])
        except ValueError:
            raise ValueError("'%s' is not an integer" % tokens[0])
        if v < 1 or v > maximum:
            raise ValueError("value must be between 1 and %d" % maximum)
        return v

    def history_text(self):
        """Numbered command history."""
        if not self.command_history:
            return "No history."
        return "\n".join("%3d  %s" % (i + 1, c)
                         for i, c in enumerate(self.command_history))

    @staticmethod
    def help_text():
        """Command overview."""
        return "\n".join([
            "Commands (shortcut):",
            "  start (s) [rx ry cx cy]  create an order (coordinates 0-800)",
            "  status (st)              funds and rider status",
            "  orders (o)               all order details",
            "  riders (r)               rider status change log",

            "  tick (t) [n]             advance simulation n ticks",
            "  run (ru)                 run until all orders delivered",
            "  save [file] / load [file] persist / restore state (JSON)",
            "  history (hi)             command history, !n or !! to repeat",
            "  help (h, ?)              this help",
            "  quit (q, exit)           leave the program",
        ])

class Configuration:
    """System parameters loaded from / saved to a JSON file."""

    def __init__(self, file_path="config.json"):
        self.file_path = file_path
        self.values = dict(DEFAULT_CONFIG)

    @property
    def initial_funds(self):
        return self.values["initial_funds"]

    @property
    def recruitment_cost(self):
        return self.values["recruitment_cost"]

    def load(self):
        """Load parameters if the file exists. Returns the values dict."""
        if self.file_path and os.path.exists(self.file_path):
            try:
                with open(self.file_path, "r", encoding="utf-8") as fh:
                    data = json.load(fh)
                params = data.get("system", data) if isinstance(data, dict) else {}
                for key in DEFAULT_CONFIG:
                    if key in params:
                        self.values[key] = int(params[key])
            except (OSError, ValueError, TypeError) as exc:
                logger.warning("Could not read config %s: %s", self.file_path, exc)
        return dict(self.values)

    def save(self):
        """Write parameters to the file."""
        with open(self.file_path, "w", encoding="utf-8") as fh:
            json.dump({"system": self.values}, fh, indent=2)

    def to_dict(self):
        return dict(self.values)

"""Command line entry point for the food delivery dispatch simulator."""

import argparse
import logging
import sys
import os

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import globalvar  # noqa: E402
import system  # noqa: E402
from config import Configuration, save_state, load_state  # noqa: E402
from display import DisplayManager  # noqa: E402
from input_manager import InputManager  # noqa: E402
from order import OrderManager  # noqa: E402

logger = logging.getLogger("dispatch")

NO_TICK_COMMANDS = ("help", "history", "quit", "save", "load", "tick", "run",
                    None, "unknown", "error")


class Application:
    """Main interactive loop; every command iteration is a time step."""

    def __init__(self, config_path="config.json", input_func=input, out=None):
        self.out = out or sys.stdout
        self.config = Configuration(config_path)
        self.manager = system.SystemManager(self.config.load())
        self.inputs = InputManager(input_func)

    def write(self, text=""):
        self.out.write(text + "\n")

    def run(self):
        self.write("Food Delivery Dispatch Simulator - type 'help' for commands.")
        while globalvar.get_value("running", False):
            line = self.inputs.read_line("[t=%s] > " % globalvar.get_value("tick", 0))
            self.handle(line)

    def handle(self, line):
        """Process a single command line."""
        cmd, args = self.inputs.parse(line)
        try:
            self.dispatch(cmd, args)
            if cmd not in NO_TICK_COMMANDS:
                system.advance_tick(globalvar.get_value("ticks_per_command", 1))
        except ValueError as exc:
            self.write("Error: %s" % exc)
        except Exception as exc:  # recovery
            logger.exception("Unexpected error")
            self.write("Unexpected error: %s. Recovering state..." % exc)
            fixes = system.check_consistency()
            if fixes:
                self.write("Repaired: " + "; ".join(fixes))

    def dispatch(self, cmd, args):
        if cmd is None:
            return
        if cmd == "unknown":
            self.write("Unknown command '%s'. Type 'help'." % args[0])
        elif cmd == "error":
            self.write("Error: %s" % args[0])
        elif cmd == "help":
            self.write(InputManager.help_text())
        elif cmd == "history":
            self.write(self.inputs.history_text())
        elif cmd == "quit":
            globalvar.set_value("running", False)
            self.write("Goodbye.")
        elif cmd == "start":
            self.cmd_start(args)
        elif cmd == "status":
            self.write(DisplayManager.status_text())
        elif cmd == "orders":
            self.write(DisplayManager.orders_text())
        elif cmd == "riders":
            self.write(DisplayManager.riders_log_text())
        elif cmd == "map":
            self.write(DisplayManager.render_map())
        elif cmd == "tick":
            n = InputManager.parse_positive_int(args)
            system.advance_tick(n)
            self.write("Advanced %d tick(s). Now t=%s." % (n, globalvar.get_value("tick")))
        elif cmd == "run":
            used = self.manager.run_until_idle()
            self.write("Ran %d tick(s); all orders delivered: %s." % (
                used, system.all_idle()))
        elif cmd == "save":
            path = args[0] if args else "state.json"
            save_state(path)
            self.write("State saved to %s." % path)
        elif cmd == "load":
            path = args[0] if args else "state.json"
            load_state(path)
            system.check_consistency()
            self.write("State loaded from %s." % path)

    def cmd_start(self, args):
        if not args:
            line = self.inputs.read_line(
                "Enter restaurant_x restaurant_y customer_x customer_y (0-800): ")
            args = line.replace(",", " ").replace("[", " ").replace("]", " ").split()
        coords = InputManager.parse_coordinates(args)
        order = OrderManager.create_order(coords)
        rider, recruited = OrderManager.assign_order(order)
        self.write(DisplayManager.order_assignment_text(order, rider, recruited))


def main(argv=None):
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default="config.json", help="JSON config file")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    opts = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if opts.debug else logging.WARNING,
                        format="%(asctime)s %(levelname)s %(name)s: %(message)s")
    Application(opts.config).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Data loading utilities for the Beijing Subway fare calculation system.

The subway network is stored in ``subway_data.json`` next to this module::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str,
                      "distance": int (meters), "line": str}]}
"""

import json
import os

# Absolute path of the bundled data file, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON file (defaults to ``DATA_FILE_PATH``)
    :return: dict with the keys ``stations`` and ``connections``
    :raises FileNotFoundError: if the file does not exist
    :raises ValueError: if the file is not valid JSON or has a wrong structure
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ValueError("Subway data file '%s' is not valid JSON: %s" % (path, exc)) from exc
    if not isinstance(data, dict):
        raise ValueError("Subway data file '%s' must contain a JSON object" % path)
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise ValueError("'stations' and 'connections' must be lists in '%s'" % path)
    return data

"""Core graph data structures: Edge, Graph, Station and Route."""

from dataclasses import dataclass, field
from typing import List, Optional


class Edge:
    """A weighted, line-labelled connection between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def other(self, v):
        """Return the endpoint of the edge that is not ``v``."""
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError("Vertex %r is not an endpoint of this edge" % (v,))

    def reversed(self):
        """Return a copy of the edge with the direction reversed."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller.  ``graph`` maps a station
    number to the list of outgoing edges (each undirected edge is stored
    once in each direction).
    """

    def __init__(self):
        self.graph = {}                 # station number -> list[Edge]
        self.station_lines = {}         # station name -> list[str]
        self.station_to_number = {}     # station name -> station number
        self.number_to_station = {}     # station number -> station name
        self._edge_count = 0

    # ----- vertices / stations -------------------------------------------------
    def add_vertex(self, v):
        self.graph.setdefault(v, [])

    def add_station(self, name, number, lines=None):
        """Register a station name with its caller-assigned number."""
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.add_vertex(number)
        if lines is not None:
            self.set_station_lines(name, lines)
        else:
            self.station_lines.setdefault(name, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines) if lines is not None else []

    def get_station_name(self, number):
        return self.number_to_station.get(number, str(number))

    def get_station_number(self, name):
        return self.station_to_number.get(name)

    def is_transfer_station(self, station_name):
        return len(set(self.station_lines.get(station_name, []))) > 1

    # ----- edges -------------------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        if edge.weight is None or edge.weight < 0:
            raise ValueError("Edge distance must be non-negative: %r" % (edge,))
        if edge.source == edge.dest:
            raise ValueError("Self-loop edges are not allowed: %r" % (edge,))
        self.graph.setdefault(edge.source, []).append(edge)
        self.graph.setdefault(edge.dest, []).append(edge.reversed())
        self._edge_count += 1

    def adj(self, v):
        return self.graph.get(v, [])

    def vertices(self):
        return list(self.graph.keys())

    def V(self):
        return len(self.graph)

    def E(self):
        return self._edge_count

    @property
    def vertex_count(self):
        return self.V()

    @property
    def edge_count(self):
        return self.E()

    def edges(self):
        """Return every undirected edge once."""
        seen = set()
        result = []
        for v, lst in self.graph.items():
            for e in lst:
                key = (min(e.source, e.dest, key=str), max(e.source, e.dest, key=str), e.line, e.distance)
                if key in seen:
                    continue
                seen.add(key)
                result.append(e)
        return result

    def lines_between(self, u, v):
        return [e.line for e in self.adj(u) if e.dest == v]

"""Shortest path computation (Dijkstra with a heap-based priority queue).

Three query modes are supported:

* ``shortest_distance`` - minimise the total distance
* ``least_transfer``    - minimise the number of transfers, then distance
* ``comprehensive``     - minimise distance plus a penalty per transfer

The search runs over (station, current line[, transfers]) states, which
correctly handles multi-line interchange stations and allows enforcing a
maximum transfer limit and line preferences.
"""

import math

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")


class MinPQ:
    """Binary min-heap priority queue (stable for equal keys)."""

    def __init__(self):
        self._heap = []
        self._counter = 0

    def insert(self, key, item):
        entry = (key, self._counter, item)
        self._counter += 1
        heap = self._heap
        heap.append(entry)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[i][:2] < heap[parent][:2]:
                heap[i], heap[parent] = heap[parent], heap[i]
                i = parent
            else:
                break

    push = insert

    def del_min(self):
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        heap = self._heap
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            i = 0
            n = len(heap)
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i
                if left < n and heap[left][:2] < heap[smallest][:2]:
                    smallest = left
                if right < n and heap[right][:2] < heap[smallest][:2]:
                    smallest = right
                if smallest == i:
                    break
                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest
        return top[0], top[2]

    pop = del_min

    def peek(self):
        if not self._heap:
            raise IndexError("peek from empty priority queue")
        return self._heap[0][0], self._heap[0][2]

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)


class PathfindingConfig:
    """Configuration of a path query."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode '%s', expected one of %s" % (mode, ", ".join(VALID_MODES)))
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        else:
            preferred_lines = list(preferred_lines)
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = preferred_lines

    def __repr__(self):
        return "PathfindingConfig(mode=%r, max_transfers=%r, preferred_lines=%r)" % (
            self.mode, self.max_transfers, self.preferred_lines)


class ShortestPath:
    """Dijkstra single-source shortest paths on a ``Graph``."""

    TRANSFER_PENALTY = 3000      # meters-equivalent penalty per transfer (comprehensive mode)
    PREFERENCE_FACTOR = 0.7      # weight multiplier for preferred lines

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self.edge_to = {}
        self._prev = {}
        self._best_state = {}
        self._transfers = {}
        self._run()

    # ------------------------------------------------------------------
    @staticmethod
    def _make_key(mode, dist, transfers, weighted, penalty):
        if mode == "least_transfer":
            return (transfers, weighted, dist)
        if mode == "comprehensive":
            return (weighted + transfers * penalty, transfers, dist)
        return (dist, transfers, weighted)

    def _run(self):
        cfg = self.config
        mode = getattr(cfg, "mode", "shortest_distance")
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode '%s'" % mode)
        limit = getattr(cfg, "max_transfers", None)
        preferred = set(getattr(cfg, "preferred_lines", None) or [])
        penalty = self.TRANSFER_PENALTY
        factor = self.PREFERENCE_FACTOR
        track_k = limit is not None

        for v in self.graph.vertices():
            self.dist_to[v] = math.inf

        src = self.source
        start = (src, None, 0 if track_k else None)
        comps = {start: (0, 0, 0.0)}
        keys = {start: self._make_key(mode, 0, 0, 0.0, penalty)}
        settled = []
        settled_set = set()
        pq = MinPQ()
        pq.insert(keys[start], start)

        while not pq.is_empty():
            key, s = pq.del_min()
            if s in settled_set or key != keys.get(s):
                continue
            settled_set.add(s)
            settled.append(s)
            node, line, _ = s
            d, t, w = comps[s]
            for e in self.graph.adj(node):
                changed = line is not None and e.line != line
                nt = t + 1 if changed else t
                if track_k and nt > limit:
                    continue
                ns = (e.dest, e.line, nt if track_k else None)
                if ns in settled_set:
                    continue
                nd = d + e.weight
                nw = w + e.weight * (factor if e.line in preferred else 1.0)
                nk = self._make_key(mode, nd, nt, nw, penalty)
                old = keys.get(ns)
                if old is None or nk < old:
                    keys[ns] = nk
                    comps[ns] = (nd, nt, nw)
                    self._prev[ns] = (s, e)
                    pq.insert(nk, ns)

        # choose the best state for every station
        best_key = {}
        for s in settled:
            node = s[0]
            k = keys[s]
            if node not in best_key or k < best_key[node]:
                best_key[node] = k
                self._best_state[node] = s
        for node, s in self._best_state.items():
            d, t, _ = comps[s]
            self.dist_to[node] = d
            self._transfers[node] = t
            prev = self._prev.get(s)
            if prev is not None:
                self.edge_to[node] = prev[1]

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target."""
        if target not in self._best_state:
            return [], []
        edges = []
        s = self._best_state[target]
        while s in self._prev:
            prev_state, e = self._prev[s]
            edges.append(e)
            s = prev_state
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes on the path to ``target`` (0 if unreachable)."""
        _, edges = self.path_with_edges(target)
        count = 0
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                count += 1
        return count

    def get_transfer_stations(self, target):
        """Station numbers where a line change happens."""
        nodes, edges = self.path_with_edges(target)
        return [nodes[i] for i in range(1, len(edges)) if edges[i].line != edges[i - 1].line]

    def get_lines(self, target):
        """Ordered list of distinct lines used on the path."""
        _, edges = self.path_with_edges(target)
        lines = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)
        return lines

"""Beijing Subway fare calculation.

Fare structure (distance in meters, boundaries belong to the higher tier):

* [0, 6000)       -> 3 yuan
* [6000, 12000)   -> 4 yuan
* [12000, 22000)  -> 5 yuan
* [22000, 32000)  -> 6 yuan
* >= 32000        -> 7 yuan plus 1 yuan for every further 20 km
"""

import math

# (min distance inclusive, max distance exclusive, fare)
FARE_TIERS = [
    (0, 6000, 3),
    (6000, 12000, 4),
    (12000, 22000, 5),
    (22000, 32000, 6),
]
EXTENDED_START = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_INCREMENT_DISTANCE = 20000

AVERAGE_SPEED_KMH = 35.0     # average operating speed including stops
TRANSFER_TIME_MINUTES = 5.0  # average time per transfer


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number, got %r" % (distance,))
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare in yuan for a distance given in meters.

    Fractional distances are rounded up to the next whole meter.
    """
    _validate_distance(distance)
    d = math.ceil(distance)
    for low, high, fare in FARE_TIERS:
        if low <= d < high:
            return fare
    return EXTENDED_BASE_FARE + (d - EXTENDED_START) // EXTENDED_INCREMENT_DISTANCE


def get_fare_tier_name(distance):
    """Human readable name of the tier for a distance in meters."""
    _validate_distance(distance)
    d = math.ceil(distance)
    if d < 6000:
        return "Basic Tier"
    if d < EXTENDED_START:
        return "Progressive Tier"
    return "Extended Tier"


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes for a distance in meters."""
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, (int, float)) or transfer_count < 0:
        raise ValueError("Transfer count must be a non-negative number")
    minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0 + transfer_count * TRANSFER_TIME_MINUTES
    return round(float(minutes), 2)

"""Subway network model: builds the graph and answers route queries."""

import math
import re

from data_loader import DATA_FILE_PATH, load_subway_data
from fare_calculator import Sum_money, calculate_estimated_time
from models import Edge, Graph, Route, Station
from pathfinding import PathfindingConfig, ShortestPath

INPUT_FORMAT_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class SubwayError(Exception):
    """Base class of all user facing errors."""


class InputFormatError(SubwayError):
    pass


class StationNotFoundError(SubwayError):
    pass


class SameStationError(SubwayError):
    pass


class RouteNotFoundError(SubwayError):
    pass


class PathCalculationError(SubwayError):
    pass


class FareCalculationError(SubwayError):
    pass


_SEPARATOR_RE = re.compile(r"\s*(?:;|；|,|，|->|→)\s*")


def parse_query_input(text):
    """Parse 'Start; End' style input. Returns (start, end)."""
    if text is None:
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    parts = _SEPARATOR_RE.split(text.strip())
    if len(parts) != 2:
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise InputFormatError(INPUT_FORMAT_MESSAGE)
    return start, end


def normalize_name(name):
    return re.sub(r"\s+", " ", str(name)).strip().casefold()


def build_graph(data):
    """Build a Graph from the raw subway data dict."""
    graph = Graph()
    number = 0
    for st in data.get("stations", []):
        name = st.get("name") if isinstance(st, dict) else None
        if not name or name in graph.station_to_number:
            continue
        graph.add_station(name, number, st.get("lines", []) or [])
        number += 1
    for conn in data.get("connections", []):
        if not isinstance(conn, dict):
            continue
        src, dst = conn.get("source"), conn.get("dest")
        dist = conn.get("distance")
        if not src or not dst or src == dst:
            continue
        if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist < 0:
            continue
        for name in (src, dst):
            if name not in graph.station_to_number:
                graph.add_station(name, number, [conn.get("line")] if conn.get("line") else [])
                number += 1
        graph.add_edge(Edge(graph.station_to_number[src], graph.station_to_number[dst],
                            dist, conn.get("line")))
    return graph


class SubwayNetwork:
    """The Beijing Subway network with routing and fare functionality."""

    def __init__(self, data=None, data_source=DATA_FILE_PATH):
        if data is None:
            data = load_subway_data(data_source)
        self.data_source = data_source
        self.data = data
        self.graph = build_graph(data)
        self.station_to_number = self.graph.station_to_number
        self.number_to_station = self.graph.number_to_station
        self._normalized = {}
        for name in self.station_to_number:
            self._normalized.setdefault(normalize_name(name), name)

    # ------------------------------------------------------------------
    def stations(self):
        return [Station(n, num, list(self.graph.station_lines.get(n, [])))
                for n, num in self.station_to_number.items()]

    def resolve_station(self, name):
        """Return the official station name for user input, or None."""
        if name is None:
            return None
        name = str(name).strip()
        if name in self.station_to_number:
            return name
        key = normalize_name(name)
        if key in self._normalized:
            return self._normalized[key]
        for suffix in (" station", "站"):
            if key.endswith(suffix) and key[: -len(suffix)].strip() in self._normalized:
                return self._normalized[key[: -len(suffix)].strip()]
        return None

    def validate_stations(self, start, end):
        s = self.resolve_station(start)
        if s is None:
            raise StationNotFoundError(
                "Starting station '%s' does not exist, please check the station name" % start)
        e = self.resolve_station(end)
        if e is None:
            raise StationNotFoundError(
                "Ending station '%s' does not exist, please check the station name" % end)
        if s == e:
            raise SameStationError("Starting station cannot be the same as the ending station")
        return s, e

    # ------------------------------------------------------------------
    def plan_route(self, start, end, config=None):
        """Compute a Route between two station names."""
        config = config if config is not None else PathfindingConfig()
        s_name, e_name = self.validate_stations(start, end)
        s_num = self.station_to_number[s_name]
        e_num = self.station_to_number[e_name]
        try:
            sp = ShortestPath(self.graph, s_num, config)
            nodes, edges = sp.path_with_edges(e_num)
        except SubwayError:
            raise
        except Exception as exc:  # pragma: no cover - defensive
            raise PathCalculationError("Path calculation failed: %s" % exc) from exc
        if not nodes or len(nodes) < 2:
            msg = "No route found from '%s' to '%s'" % (s_name, e_name)
            if config.max_transfers is not None:
                msg += " within %d transfer(s)" % config.max_transfers
            raise RouteNotFoundError(msg)

        names = [self.graph.get_station_name(n) for n in nodes]
        total = sum(e.weight for e in edges)
        if isinstance(total, float) and total.is_integer():
            total = int(total)
        transfer_stations = []
        segments = []
        for i, e in enumerate(edges):
            if i > 0 and e.line != edges[i - 1].line:
                transfer_stations.append(names[i])
            if not segments or segments[-1]["line"] != e.line:
                segments.append({"line": e.line, "from": names[i], "to": names[i + 1],
                                 "stations": [names[i], names[i + 1]], "stops": 1,
                                 "distance": e.weight})
            else:
                seg = segments[-1]
                seg["to"] = names[i + 1]
                seg["stations"].append(names[i + 1])
                seg["stops"] += 1
                seg["distance"] += e.weight
        lines = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)
        transfer_count = len(transfer_stations)
        try:
            fare = Sum_money(total)
        except (TypeError, ValueError) as exc:
            raise FareCalculationError("Fare calculation failed: %s" % exc) from exc
        try:
            est = calculate_estimated_time(total, transfer_count)
        except (TypeError, ValueError):
            est = math.nan
        return Route(start=s_name, end=e_name, mode=config.mode, stations=names,
                     edge_lines=[e.line for e in edges], edge_distances=[e.weight for e in edges],
                     total_distance=total, transfer_count=transfer_count,
                     transfer_stations=transfer_stations, lines=lines, segments=segments,
                     fare=fare, estimated_time=est, max_transfers=config.max_transfers,
                     line_preference=list(config.preferred_lines or []))

def format_distance(meters):
    """Return distance shown in both meters and kilometers."""
    km = meters / 1000.0

def _num(value):
    if isinstance(value, float) and value.is_integer():

    }


def format_text(report):
    """Console-formatted text report."""
    q = report["query"]
    out = []
    out.append("=" * 60)
    out.append("Route: %s -> %s" % (q["start"], q["end"]))
    out.append("Mode: %s" % q["mode"])
    out.append("-" * 60)
    out.append("Stations (%d): %s" % (report["station_count"], " -> ".join(report["path"])))
    out.append("Segments:")
    segs = report["segments"]
    for i, s in enumerate(segs, 1):
        out.append("  %d. [%s] %s -> %s (%d stop(s), %.2f km)"
                   % (i, s["line"], s["from"], s["to"], s["stops"], s["distance"] / 1000.0))
        if i < len(segs):
            out.append("     Transfer at %s: %s -> %s" % (s["to"], s["line"], segs[i]["line"]))
    out.append("Transfer stations: %s" % (", ".join(report["transfer_stations"]) or "None"))
    out.append("Lines involved: %s" % ", ".join(str(l) for l in report["lines"]))
    out.append("Total distance: %s" % format_distance(report["distance"]))
    out.append("Transfers: %d" % report["transfer_count"])
    out.append("Estimated time: %.1f minutes" % report["estimated_time"])
    out.append(report["fare_text"])
    out.append("=" * 60)
    return "\n".join(out)


def format_json(report):
    return json.dumps(report, ensure_ascii=False, indent=2)


def render(report, fmt="text"):
    return format_json(report) if fmt == "json" else format_text(report)


def write_reports(path, reports, fmt="text"):
    """Write reports to a file (a single JSON object or a list for several)."""
    if fmt == "json":
        payload = reports[0] if len(reports) == 1 else reports
        content = json.dumps(payload, ensure_ascii=False, indent=2)
    else:
        content = "\n\n".join(format_text(r) for r in reports)
    with open(path, "w", encoding="utf-8") as fh:
        fh.write(content)
        fh.write("\n")

def _norm(name):
    return "".join(str(name).split()).casefold()


@dataclass
class IntegrityReport:
    station_coverage_valid: bool = True
    connection_integrity_valid: bool = True
    distance_accuracy_valid: bool = True
    errors: List[str] = field(default_factory=list)
    warnings: List[str] = field(default_factory=list)
    stats: dict = field(default_factory=dict)

    @property
    def is_valid(self):
        return (self.station_coverage_valid and self.connection_integrity_valid
                and self.distance_accuracy_valid)

    def to_text(self):
        def mark(ok):
            return "PASS" if ok else "FAIL"
        lines = ["Data integrity check",
                 "  Stations: %(stations)d, Connections: %(connections)d, Lines: %(lines)d" % self.stats,
                 "  Station coverage:      %s" % mark(self.station_coverage_valid),
                 "  Connection integrity:  %s" % mark(self.connection_integrity_valid),
                 "  Distance accuracy:     %s" % mark(self.distance_accuracy_valid)]
        for e in self.errors:
            lines.append("  ERROR: %s" % e)
        for w in self.warnings:
            lines.append("  WARNING: %s" % w)
        lines.append("Result: %s" % ("Data is valid" if self.is_valid else "Data has errors"))
        return "\n".join(lines)


def check_data_integrity(data):
    """Run all integrity checks on raw subway data and return an IntegrityReport."""
    rep = IntegrityReport()
    stations = data.get("stations", []) or []
    connections = data.get("connections", []) or []

    # --- station coverage ---
    names = {}
    all_lines = set()
    if not stations:
        rep.station_coverage_valid = False
        rep.errors.append("No station data found")
    for st in stations:
        name = st.get("name") if isinstance(st, dict) else None
        if not name:
            rep.station_coverage_valid = False
            rep.errors.append("Station entry without a name: %r" % (st,))
            continue
        if name in names:
            rep.warnings.append("Duplicate station entry: %s" % name)
        lines = st.get("lines") or []
        if not lines:
            rep.station_coverage_valid = False
            rep.errors.append("Station '%s' has no line affiliation" % name)
        names[name] = list(lines)
        all_lines.update(lines)
    normalized = {_norm(n) for n in names}
    for line, candidates in REPRESENTATIVE_STATIONS.items():
        if not any(_norm(c) in normalized for c in candidates):
            rep.warnings.append("No representative station found for %s (%s)"
                                % (line, ", ".join(candidates)))

    # --- connection integrity & distance accuracy ---
    adjacency = {n: set() for n in names}
    seen = {}
    for i, c in enumerate(connections):
        if not isinstance(c, dict):
            rep.connection_integrity_valid = False
            rep.errors.append("Connection #%d is not an object" % i)
            continue
        s, d, line, dist = c.get("source"), c.get("dest"), c.get("line"), c.get("distance")
        if not s or not d:
            rep.connection_integrity_valid = False
            rep.errors.append("Connection #%d is missing source/dest" % i)
            continue
        if s == d:
            rep.connection_integrity_valid = False
            rep.errors.append("Connection #%d connects '%s' to itself" % (i, s))
        for n in (s, d):
            if n not in names:
                rep.connection_integrity_valid = False
                rep.errors.append("Connection #%d references unknown station '%s'" % (i, n))
        if line and s in names and line not in names[s]:
            rep.warnings.append("Station '%s' is not listed on %s" % (s, line))
        if line and d in names and line not in names[d]:
            rep.warnings.append("Station '%s' is not listed on %s" % (d, line))
        if isinstance(dist, bool) or not isinstance(dist, (int, float)):
            rep.distance_accuracy_valid = False
            rep.errors.append("Connection %s - %s has invalid distance %r" % (s, d, dist))
        elif dist <= 0 or dist > MAX_REASONABLE_DISTANCE:
            rep.distance_accuracy_valid = False
            rep.errors.append("Connection %s - %s has unreasonable distance %s m" % (s, d, dist))
        else:
            key = (frozenset((s, d)), line)
            if key in seen and seen[key] != dist:
                rep.warnings.append("Inconsistent distances for %s - %s on %s: %s vs %s"
                                    % (s, d, line, seen[key], dist))
            seen.setdefault(key, dist)
        adjacency.setdefault(s, set()).add(d)
        adjacency.setdefault(d, set()).add(s)
        if line:
            all_lines.add(line)

    isolated = [n for n in names if not adjacency.get(n)]
    for n in isolated:
        rep.warnings.append("Station '%s' has no connections" % n)
    if adjacency:
        start = next(iter(adjacency))
        visited = {start}
        dq = deque([start])
        while dq:
            u = dq.popleft()
            for v in adjacency.get(u, ()):
                if v not in visited:
                    visited.add(v)
                    dq.append(v)
        if len(visited) < len(adjacency):
            rep.warnings.append("Network is not fully connected (%d of %d stations reachable)"
                                % (len(visited), len(adjacency)))

    rep.stats = {"stations": len(names), "connections": len(connections), "lines": len(all_lines)}
    return rep


class SubwayDataManager:
    """Maintain subway data: add lines, modify stations, update distances."""

    def __init__(self, data=None, path=DATA_FILE_PATH):
        self.path = path
        self.data = copy.deepcopy(data) if data is not None else load_subway_data(path)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    def _find_station(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def add_station(self, name, lines):
        st = self._find_station(name)
        if st is None:
            st = {"name": name, "lines": []}
            self.data["stations"].append(st)
        for l in lines:
            if l not in st["lines"]:
                st["lines"].append(l)
        return st

    def add_line(self, line_name, station_names, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must be number of stations minus one")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for n in station_names:
            self.add_station(n, [line_name])
        for i, d in enumerate(distances):
            self.data["connections"].append({"source": station_names[i], "dest": station_names[i + 1],
                                             "distance": d, "line": line_name})

    def update_station(self, name, new_name=None, lines=None):
        st = self._find_station(name)
        if st is None:
            raise KeyError("Station '%s' does not exist" % name)
        if new_name and new_name != name:
            if self._find_station(new_name) is not None:
                raise ValueError("Station '%s' already exists" % new_name)
            st["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            st["lines"] = list(lines)
        return st

    def update_distance(self, source, dest, distance, line=None):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        count = 0
        for c in self.data["connections"]:
            if {c.get("source"), c.get("dest")} == {source, dest} and (line is None or c.get("line") == line):
                c["distance"] = distance
                count += 1
        if count == 0:
            raise KeyError("No connection between '%s' and '%s'" % (source, dest))
        return count

    def check_integrity(self):
        return check_data_integrity(self.data)

    def save(self, path=None):
        with open(path or self.path, "w", encoding="utf-8") as fh:
            json.dump(self.data, fh, ensure_ascii=False, indent=2)

Usage examples::

def _non_negative_int(value):
    try:
        v = int(value)
    except ValueError:
        raise argparse.ArgumentTypeError("max_transfers must be a non-negative integer")
    if v < 0:
        raise argparse.ArgumentTypeError("max_transfers must be a non-negative integer")
    return v


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", help="export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text", help="output format")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                   help="path query mode")
    p.add_argument("--max_transfers", type=_non_negative_int, default=None,
                   help="maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="preferred line(s), comma separated (e.g. 'Line 1')")
    p.add_argument("--check_data", action="store_true", help="run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help="subway data JSON file")
    p.add_argument("--start", help="starting station (non-interactive query)")
    p.add_argument("--end", help="ending station (non-interactive query)")
    return p


def run_single(network, start, end, config, fmt, reports):
    try:
        route = network.plan_route(start, end, config)
        report = build_report(route)
    except SubwayError as exc:
        print("Error: %s" % exc)
        return False
    except Exception as exc:  # unexpected failure, keep CLI alive
        print("Error: unexpected failure during query: %s" % exc)
        return False
    print(render(report, fmt))
    reports.append(report)
    return True


def main(argv=None):
    args = build_parser().parse_args(argv)

    try:
        data = load_subway_data(args.data_file)
    except (OSError, ValueError) as exc:
        print("Error: failed to load subway data: %s" % exc)
        return 1

    if args.check_data:
        rep = check_data_integrity(data)
        print(rep.to_text())
        return 0 if rep.is_valid else 1

    try:
        network = SubwayNetwork(data=data, data_source=args.data_file)
    except Exception as exc:
        print("Error: failed to build subway network: %s" % exc)
        return 1

    prefs = []
    if args.line_preference:
        prefs = [p.strip() for p in args.line_preference.split(",") if p.strip()]
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers, preferred_lines=prefs)

    reports = []
    status = 0
    if args.start or args.end:
        if not (args.start and args.end):
            print("Error: both --start and --end must be provided")
            return 1
        status = 0 if run_single(network, args.start, args.end, config, args.format, reports) else 1
    else:
        print("Beijing Subway Fare Calculator (%d stations loaded)" % len(network.station_to_number))
        print("Enter 'Starting Station; Ending Station' to query, or 'quit' to exit.")
        while True:
            try:
                line = input("> ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            text = line.strip()
            if not text:
                continue
            if text.casefold() in EXIT_COMMANDS:
                break
            if text.casefold() in ("help", "h", "?"):
                print("Format: Starting Station; Ending Station  (separators: ; , ->)")
                continue
            try:
                start, end = parse_query_input(text)
            except SubwayError as exc:
                print("Error: %s" % exc)
                continue
            run_single(network, start, end, config, args.format, reports)
        print("Goodbye!")

    if args.output_file and reports:
        try:
            write_reports(args.output_file, reports, args.format)
            print("Results exported to %s" % args.output_file)
        except OSError as exc:
            print("Error: failed to write output file: %s" % exc)
            return 1
    return status


if __name__ == "__main__":
    sys.exit(main())

"""Grid based path planning for riders.

The map is 800x800 units split into an 8x8 grid of 100x100 cells.
Grid lines (x or y multiple of 100) are treated as streets, and riders
prefer to travel along them.  Every path consists only of horizontal and
vertical segments (right-angle turns).

Riders do not block each other on the map, and every path is computed
deterministically from (start, end), so path conflicts and deadlocks
cannot occur.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE


class PathPlanner:
    """Collection of grid pathfinding helpers."""

    @staticmethod
    def clamp(v, low=0, high=MAP_SIZE):
        """Clamp ``v`` into [low, high]."""
        return max(low, min(high, int(v)))

    @staticmethod
    def to_grid_index(v):
        """Convert a raw coordinate (0-800) into a grid index (0-7)."""
        v = PathPlanner.clamp(v)
        return min(v, MAP_SIZE - 1) // CELL_SIZE

    @staticmethod
    def to_grid_center(v):
        """Convert a raw coordinate into the centre of its grid cell.

        ``center = (v // 100) * 100 + 50``; the boundary value 800 is
        mapped to the last cell (750).
        """
        return PathPlanner.to_grid_index(v) * CELL_SIZE + CELL_SIZE // 2

    @staticmethod
    def point_to_center(point):
        """Convert an (x, y) point into its cell-centre point."""
        return (PathPlanner.to_grid_center(point[0]),
                PathPlanner.to_grid_center(point[1]))

    @staticmethod
    def is_on_street(x, y):
        """Return True if the point lies on a street (grid line)."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def _street_toward(v, target):
        """Nearest grid line from ``v`` in the direction of ``target``."""
        if v % CELL_SIZE == 0:
            return v
        if target > v:
            return min(MAP_SIZE, (v // CELL_SIZE + 1) * CELL_SIZE)
        return (v // CELL_SIZE) * CELL_SIZE

    @staticmethod
    def is_special_case(start, end):
        """Return True when start and end share a column or row."""
        sx, sy = start
        ex, ey = end
        return (sx == ex) != (sy == ey)

    @staticmethod
    def find_path(start, end):
        """Return the list of waypoints from ``start`` to ``end``.

        The start point itself is not included; the end point is the last
        waypoint.  Returns an empty list when start == end.
        """
        sx, sy = int(start[0]), int(start[1])
        ex, ey = int(end[0]), int(end[1])
        if (sx, sy) == (ex, ey):
            return []
        if sx == ex:
            # Directly above/below: use the vertical street to the left.
            if sx % CELL_SIZE == 0:
                pts = [(ex, ey)]
            else:
                xs = (sx // CELL_SIZE) * CELL_SIZE
                pts = [(xs, sy), (xs, ey), (ex, ey)]
        elif sy == ey:
            if sy % CELL_SIZE == 0:
                pts = [(ex, ey)]
            else:
                ys = (sy // CELL_SIZE) * CELL_SIZE
                pts = [(sx, ys), (ex, ys), (ex, ey)]
        else:
            ys = PathPlanner._street_toward(sy, ey)
            pts = [(sx, ys), (ex, ys), (ex, ey)]
        result = []
        prev = (sx, sy)
        for p in pts:
            if p != prev:
                result.append(p)
                prev = p
        return result

    @staticmethod
    def find_multi_path(start, stops):
        """Continuous path visiting each stop in order (e.g. restaurant,
        then customer)."""
        path = []
        cur = tuple(start)
        for stop in stops:
            path.extend(PathPlanner.find_path(cur, stop))
            cur = tuple(stop)
        return path

    @staticmethod
    def path_length(start, waypoints):
        """Manhattan length of the path starting at ``start``."""
        total = 0
        cx, cy = start
        for x, y in waypoints:
            total += abs(x - cx) + abs(y - cy)
            cx, cy = x, y
        return total

    @staticmethod
    def manhattan(a, b):
        """Manhattan distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

import globalvar
from path_planner import PathPlanner


class Order:
    """A delivery order from a restaurant to a customer.

    Input coordinates (0-800) are converted to grid-cell centres.
    Status lifecycle: pending -> assigned -> picking_up -> delivering ->
    completed.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.raw_res_pos = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus_pos = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = PathPlanner.point_to_center(self.raw_res_pos)
        self.cus_pos = PathPlanner.point_to_center(self.raw_cus_pos)
        self.res_grid = (PathPlanner.to_grid_index(self.raw_res_pos[0]),
                         PathPlanner.to_grid_index(self.raw_res_pos[1]))
        self.cus_grid = (PathPlanner.to_grid_index(self.raw_cus_pos[0]),
                         PathPlanner.to_grid_index(self.raw_cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("time", 0) or 0
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.complete_time = None
        self.time_taken = None
        self.trip_distance = PathPlanner.path_length(
            self.res_pos, PathPlanner.find_path(self.res_pos, self.cus_pos))
        self.distance = None
        self.efficiency = None

    @classmethod
    def reset_counter(cls):
        """Reset the order id counter."""
        cls._next_id = 1

    def assign_to(self, rider_id):
        """Record that this order is assigned to ``rider_id``."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        self.assign_time = globalvar.get_value("time", 0) or 0

    def is_completed(self):
        """Return True if delivered."""
        return self.status == "completed"

    def to_dict(self):
        """Serialisable representation."""
        return {
            "id": self.id,
            "raw_res_pos": list(self.raw_res_pos),
            "raw_cus_pos": list(self.raw_cus_pos),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "create_time": self.create_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "efficiency": self.efficiency,
        }

    def __repr__(self):
        return "Order(id=%d, %s->%s, %s)" % (self.id, self.res_pos,
                                            self.cus_pos, self.status)

"""Rider model: movement, state machine and event log."""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {STATE_IDLE: "idle", STATE_PICKING_UP: "picking up",
               STATE_DELIVERING: "delivering"}

DEFAULT_SPEED = 50
DEFAULT_X = 400
DEFAULT_Y = 400
DEFAULT_ORDER_INCOME = 100


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``; ``move()``
    then picks them up in order, performing state transitions and logging.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        cfg = globalvar.get_value("config") or {}
        self.rider_id = rider_id
        self.x = int(x if x is not None else cfg.get("spawn_x", DEFAULT_X))
        self.y = int(y if y is not None else cfg.get("spawn_y", DEFAULT_Y))
        self.speed = int(speed if speed is not None
                         else cfg.get("rider_speed", DEFAULT_SPEED))
        if self.speed <= 0:
            self.speed = DEFAULT_SPEED
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.path = []
        self.ticks = 0
        self.total_distance = 0
        self.completed_count = 0
        self.log = []
        self._log("spawn", None)

    # ------------------------------------------------------------------
    def _now(self):
        t = globalvar.get_value("time")
        return t if t is not None else self.ticks

    def _log(self, event, order_id):
        entry = {"event": event, "order_id": order_id, "time": self._now(),
                 "x": self.x, "y": self.y, "state": self.state}
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)

    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    def active_orders(self):
        """Orders not yet completed."""
        return [o for o in self.orders if o.status != "completed"]

    def is_available(self):
        """True when idle with no queued orders."""
        return self.state == STATE_IDLE and not self.active_orders()

    def queue_end_position(self):
        """Position where the rider will be after all queued orders."""
        active = self.active_orders()
        if active:
            return tuple(active[-1].cus_pos)
        return (self.x, self.y)

    # ------------------------------------------------------------------
    def _plan(self, target):
        self.special_case = PathPlanner.is_special_case((self.x, self.y),
                                                        target)
        self.path = PathPlanner.find_path((self.x, self.y), target)

    def _start_next_order(self):
        for order in self.orders:
            if order.status != "completed":
                break
        else:
            return False
        self.current_order = order
        if order.rider_id is None:
            order.assign_to(self.rider_id)
        self.state = STATE_PICKING_UP
        order.status = "picking_up"
        order.accept_time = self._now()
        to_res = PathPlanner.path_length(
            (self.x, self.y),
            PathPlanner.find_path((self.x, self.y), order.res_pos))
        order.distance = to_res + order.trip_distance
        self._plan(order.res_pos)
        self._log("accepted_order", order.id)
        return True

    def _advance(self, budget):
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            if self.x != tx:
                step = min(budget, abs(tx - self.x))
                self.x += step if tx > self.x else -step
            else:
                step = min(budget, abs(ty - self.y))
                self.y += step if ty > self.y else -step
            budget -= step
            self.total_distance += step
            if (self.x, self.y) == (tx, ty):
                self.path.pop(0)
            if step == 0:
                self.path.pop(0)
        return budget

    def _handle_arrival(self):
        order = self.current_order
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.status = "delivering"
            order.pickup_time = self._now()
            self._log("arrived_at_merchant", order.id)
            self._plan(order.cus_pos)
        elif self.state == STATE_DELIVERING:
            order.status = "completed"
            order.complete_time = self._now()
            order.time_taken = order.complete_time - order.create_time
            if order.time_taken and order.time_taken > 0:
                order.efficiency = round((order.distance or 0)
                                         / order.time_taken, 2)
            self.state = STATE_IDLE
            self.special_case = False
            if order in self.orders:
                self.orders.remove(order)
            self.current_order = None
            self.completed_count += 1
            funds = globalvar.get_value("funds")
            if funds is not None:
                cfg = globalvar.get_value("config") or {}
                globalvar.set_value(
                    "funds",
                    funds + cfg.get("order_income", DEFAULT_ORDER_INCOME))
            self._log("delivered_to_customer", order.id)

    def move(self):
        """Advance the rider by one tick."""
        self.ticks += 1
        if self.state == STATE_IDLE or self.current_order is None:
            self.state = STATE_IDLE
            if not self._start_next_order():
                return
        budget = self.speed
        guard = 0
        while self.state != STATE_IDLE and guard < 10:
            guard += 1
            budget = self._advance(budget)
            if self.path:
                break
            self._handle_arrival()

    # ------------------------------------------------------------------
    def to_dict(self):
        """Serialisable representation."""
        return {
            "rider_id": self.rider_id, "x": self.x, "y": self.y,
            "speed": self.speed, "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "total_distance": self.total_distance,
            "completed_count": self.completed_count,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%d,%d), state=%s, orders=%s)" % (
            self.rider_id, self.x, self.y, self.state_name(),
            [o.id for o in self.orders])

"""Order management: creation, assignment, completion."""

import logging

import globalvar
from order_model import Order
from path_planner import PathPlanner

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Return the best rider for ``order`` (closest rider first).

    Idle riders are preferred, ordered by street distance to the
    restaurant.  If none is idle, the rider with the fewest queued orders
    (then shortest distance from its queue end) is chosen to balance load.
    Returns None when there are no riders.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None

    def dist(r, start):
        return PathPlanner.path_length(
            start, PathPlanner.find_path(start, order.res_pos))

    idle = [r for r in riders if r.is_available()]
    if idle:
        return min(idle, key=lambda r: (dist(r, (r.x, r.y)), r.rider_id))
    return min(riders, key=lambda r: (len(r.active_orders()),
                                      dist(r, r.queue_end_position()),
                                      r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion queries."""

    @staticmethod
    def _orders():
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create an order from ``[rx, ry, cx, cy]`` and record it."""
        if len(coords) != 4:
            raise ValueError("exactly four coordinates are required")
        vals = []
        for c in coords:
            v = int(c)
            if v < 0 or v > 800:
                raise ValueError("coordinate %d out of range 0-800" % v)
            vals.append(v)
        order = Order((vals[0], vals[1]), (vals[2], vals[3]))
        self._orders().append(order)
        logger.info("Created %r", order)
        return order

    def assign_order(self, order):
        """Assign ``order`` to the best rider; returns rider or None."""
        if order.rider_id is not None or order.status == "completed":
            return None
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Assigned order %d to rider %s", order.id,
                    rider.rider_id)
        return rider

    def assign_pending(self):
        """Try to assign all pending orders; returns list of (order, rider)."""
        result = []
        for order in self._orders():
            if order.status == "pending" and order.rider_id is None:
                rider = self.assign_order(order)
                if rider is not None:
                    result.append((order, rider))
        return result

    def all_orders(self):
        """All orders."""
        return list(self._orders())

    def completed_orders(self):
        """Delivered orders."""
        return [o for o in self._orders() if o.status == "completed"]

    def pending_orders(self):
        """Orders without a rider."""
        return [o for o in self._orders() if o.status == "pending"]

"""System initialisation and the SystemManager core class."""

import json
import logging
import os

import globalvar
from order import OrderManager
from order_model import Order
from rider import Rider

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 300,
    "recruit_threshold": 300,
    "rider_cost": 300,
    "order_income": 100,
    "rider_speed": 50,
    "spawn_x": 400,
    "spawn_y": 400,
    "max_riders": 50,
    "state_file": "dispatch_state.json",
}


def load_config(path=None):
    """Load a JSON configuration file merged over the defaults."""
    cfg = dict(DEFAULT_CONFIG)
    if path and os.path.exists(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                cfg.update({k: v for k, v in data.items() if k in cfg})
        except (OSError, ValueError) as exc:
            logger.warning("Failed to load config %s: %s", path, exc)
    return cfg


def init_system(config=None):
    """Initialise all global system parameters."""
    globalvar._init()
    cfg = dict(DEFAULT_CONFIG)
    if config:
        cfg.update(config)
    globalvar.set_value("config", cfg)
    globalvar.set_value("funds", int(cfg["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("orders", [])
    globalvar.set_value("time", 0)
    Order.reset_counter()


class SystemManager:
    """Manages funds, riders, simulated time and overall system state."""

    def __init__(self, config=None, config_path=None):
        cfg = load_config(config_path)
        if config:
            cfg.update(config)
        init_system(cfg)
        self.order_manager = OrderManager()

    # properties -------------------------------------------------------
    @property
    def config(self):
        return globalvar.get_value("config")

    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        return globalvar.get_value("riders", [])

    @property
    def time(self):
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        return globalvar.get_value("running", False)

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    # riders -----------------------------------------------------------
    def create_rider(self):
        """Create a rider with the next id and add it to the list."""
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        logger.info("Recruited rider %d", counter)
        return rider

    def remove_rider(self, rider_id):
        """Remove an idle rider; returns True on success."""
        for r in self.riders:
            if r.rider_id == rider_id and r.is_available():
                self.riders.remove(r)
                return True
        return False

    def get_rider(self, rider_id):
        """Find rider by id."""
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    def recruit_if_possible(self):
        """Recruit a rider when funds >= threshold; returns rider or None."""
        cfg = self.config
        if (self.funds >= cfg["recruit_threshold"]
                and len(self.riders) < cfg["max_riders"]):
            globalvar.set_value("funds", self.funds - cfg["rider_cost"])
            return self.create_rider()
        return None

    # simulation -------------------------------------------------------
    def tick(self, n=1):
        """Advance simulated time by ``n`` ticks."""
        for _ in range(max(0, int(n))):
            globalvar.set_value("time", self.time + 1)
            try:
                self.recruit_if_possible() if self.order_manager \
                    .pending_orders() else None
                self.order_manager.assign_pending()
                for rider in list(self.riders):
                    rider.move()
                self.check_consistency()
            except Exception as exc:  # error recovery
                logger.exception("Error during tick: %s", exc)
                self.check_consistency()

    def check_consistency(self):
        """Verify and repair system state; returns list of fixes made."""
        fixes = []
        rider_ids = {r.rider_id for r in self.riders}
        for order in self.order_manager.all_orders():
            if order.status == "completed":
                continue
            if order.rider_id is not None and order.rider_id not in rider_ids:
                order.rider_id = None
                order.status = "pending"
                fixes.append("order %d orphaned -> pending" % order.id)
            elif order.rider_id is not None:
                rider = self.get_rider(order.rider_id)
                if order not in rider.orders:
                    rider.orders.append(order)
                    fixes.append("order %d re-queued" % order.id)
        for r in self.riders:
            if r.state not in (0, 1, 2):
                r.state = 0
                r.current_order = None
                fixes.append("rider %d state reset" % r.rider_id)
            if r.state != 0 and r.current_order is None:
                r.state = 0
                fixes.append("rider %d state reset" % r.rider_id)
        for f in fixes:
            logger.warning("Consistency fix: %s", f)
        return fixes

    # persistence ------------------------------------------------------
    def to_dict(self):
        """Serialisable system snapshot."""
        return {
            "config": self.config,
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.all_orders()],
        }

    def save_state(self, path=None):
        """Write the snapshot to a JSON file; returns the path or None."""
        path = path or self.config.get("state_file", "dispatch_state.json")
        try:
            with open(path, "w", encoding="utf-8") as fh:
                json.dump(self.to_dict(), fh, indent=2)
            return path
        except OSError as exc:
            logger.error("Could not save state: %s", exc)
            return None

"""Status display and ASCII map visualisation."""

from path_planner import MAP_SIZE
from rider import STATE_NAMES

HALF = 50
SIZE = MAP_SIZE // HALF + 1  # 17 columns / rows


def _idx(v):
    return max(0, min(SIZE - 1, int(v) // HALF))


class DisplayManager:
    """Builds text output for the command line."""

    def render_map(self, riders, orders, path_start=None, path=None,
                   highlight=None):
        """Render the 8x8 grid map.

        '#' streets, '.' cell interior, '*' route, 'S' restaurant,
        'C' customer, 'R' rider (highlighted rider shown as '@' if given).
        """
        grid = [["#" if (r % 2 == 0 or c % 2 == 0) else "."
                 for c in range(SIZE)] for r in range(SIZE)]
        if path_start is not None and path:
            cx, cy = path_start
            for px, py in path:
                x0, x1 = sorted((_idx(cx), _idx(px)))
                y0, y1 = sorted((_idx(cy), _idx(py)))
                for row in range(y0, y1 + 1):
                    for col in range(x0, x1 + 1):
                        grid[row][col] = "*"
                cx, cy = px, py
        for o in orders:
            if o.status == "completed":
                continue
            grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = "S"

"""Command parsing and input validation."""

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "q": "quit", "exit": "quit", "h": "help", "?": "help",
    "m": "map", "t": "tick", "hist": "history", "sv": "save",
}
COMMANDS = {"start", "status", "orders", "riders", "quit", "help", "map",
            "tick", "history", "save"}


class InputManager:
    """Parses commands, validates coordinates and keeps history."""

    def __init__(self):
        self.history = []
        try:
            import readline  # noqa: F401  (enables arrow-key history)
            self._readline = readline
        except ImportError:
            self._readline = None

    def parse_command(self, line):
        """Return (command, args) or (None, []) for empty input."""
        parts = line.strip().split()
        if not parts:
            return None, []
        self.history.append(line.strip())
        if len(self.history) > 1000:
            self.history = self.history[-1000:]
        cmd = parts[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        return cmd, parts[1:]

    @staticmethod
    def validate_coordinates(tokens):
        """Validate four integer coordinates in 0-800.

        Returns (True, [ints]) or (False, error_message).
        """
        tokens = [t for tok in tokens for t in tok.replace(",", " ").split()]
        if len(tokens) != 4:
            return False, ("Please enter exactly 4 integers: "
                           "restaurant_x restaurant_y customer_x customer_y")
        vals = []
        for t in tokens:
            try:
                v = int(t)
            except ValueError:
                return False, "'%s' is not an integer." % t
            if v < 0 or v > 800:
                return False, "Coordinate %d is out of range (0-800)." % v
            vals.append(v)
        return True, vals

    @staticmethod
    def validate_positive_int(token, default=1):
        """Parse a positive int, returning default on failure."""
        try:
            v = int(token)
            return v if v > 0 else default
        except (TypeError, ValueError):
            return default

"""Command-line entry point for the food delivery dispatch simulator."""

import logging
import math
import sys

from display import DisplayManager
from input_manager import COMMANDS, InputManager
from path_planner import PathPlanner
from system import SystemManager

HELP = """Commands:
  start [rx ry cx cy]  (s)   place an order
  status               (st)  system snapshot
  orders               (o)   all orders
  riders               (r)   rider event log
  map                  (m)   show map
  tick [n]             (t)   advance n ticks
  history              (hist) command history
  save                 (sv)  save state to JSON
  help                 (h)
  quit / exit          (q)
Each command entered advances the simulation by one tick."""


class Application:
    """Main loop and command handlers."""

    def __init__(self, config_path=None):
        self.system = SystemManager(config_path=config_path)
        self.display = DisplayManager()
        self.input = InputManager()

    def cmd_start(self, args):
        if not args:
            try:
                args = input("Enter restaurant_x restaurant_y customer_x "
                             "customer_y: ").split()
            except EOFError:
                return
        ok, res = self.input.validate_coordinates(args)
        if not ok:
            print("Error:", res)
            return
        om = self.system.order_manager
        order = om.create_order(res)
        print("Order %d created." % order.id)
        print("  Restaurant grid: %s -> center %s" % (order.res_grid,
                                                       order.res_pos))
        print("  Customer   grid: %s -> center %s" % (order.cus_grid,
                                                       order.cus_pos))
        new_rider = self.system.recruit_if_possible()
        print("  New rider added: %s" % (
            "yes (rider %d)" % new_rider.rider_id if new_rider else "no"))
        start_pos = None
        rider = None
        from order import find_best_rider_for_order
        best = find_best_rider_for_order(order)
        if best is not None:
            start_pos = best.queue_end_position()
            rider = om.assign_order(order)
        if rider is None:
            print("  No rider available; order is pending.")
            return
        path = PathPlanner.find_multi_path(start_pos,
                                           [order.res_pos, order.cus_pos])
        dist = PathPlanner.path_length(start_pos, path)
        print("  Assigned to rider %d (speed %d units/tick)" % (
            rider.rider_id, rider.speed))
        print("  Route distance: %d, estimated time: %d ticks" % (
            dist, math.ceil(dist / rider.speed) if rider.speed else 0))
        print(self.display.render_map(self.system.riders, [order],
                                      start_pos, path))

    def handle(self, cmd, args):
        """Dispatch one command; returns False to stop."""
        if cmd == "quit":
            path = self.system.save_state()
            if path:
                print("State saved to %s" % path)
            self.system.stop()
            print("Bye.")
            return False
        if cmd == "start":
            self.cmd_start(args)
        elif cmd == "status":
            print(self.display.render_status(self.system.funds,
                                             self.system.time,
                                             self.system.riders))
        elif cmd == "orders":
            print(self.display.render_orders(
                self.system.order_manager.all_orders()))
        elif cmd == "riders":
            print(self.display.render_rider_logs(self.system.riders))
        elif cmd == "map":
            print(self.display.render_map(
                self.system.riders, self.system.order_manager.all_orders()))
        elif cmd == "tick":
            n = self.input.validate_positive_int(args[0] if args else 1)
            self.system.tick(n - 1)
            print("Advanced %d tick(s)." % n)
        elif cmd == "history":
            for i, h in enumerate(self.input.history, 1):
                print("%4d  %s" % (i, h))
        elif cmd == "save":
            path = self.system.save_state()
            print("Saved to %s" % path if path else "Save failed.")
        elif cmd == "help":
            print(HELP)
        return True

    def run(self):
        """Main loop: each iteration is one tick."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.running:
            try:
                line = input("[t=%d funds=%d]> " % (self.system.time,
                                                    self.system.funds))
            except (EOFError, KeyboardInterrupt):
                print()
                self.handle("quit", [])
                break
            cmd, args = self.input.parse_command(line)
            if cmd is None:
                continue
            if cmd not in COMMANDS:
                print("Unknown command '%s'. Type 'help'." % cmd)
                continue
            try:
                if not self.handle(cmd, args):
                    break
            except Exception as exc:  # graceful recovery
                logging.exception("Command failed")
                print("Error: %s (state recovered)" % exc)
                self.system.check_consistency()
            self.system.tick(1)


def main(argv=None):
    """Program entry."""
    argv = sys.argv[1:] if argv is None else argv
    level = logging.DEBUG if "--debug" in argv else logging.WARNING
    logging.basicConfig(level=level,
                        format="%(levelname)s %(name)s: %(message)s")
    cfg = None
    if "--config" in argv:
        i = argv.index("--config")
        if i + 1 < len(argv):
            cfg = argv[i + 1]
    Application(cfg).run()


if __name__ == "__main__":
    main()

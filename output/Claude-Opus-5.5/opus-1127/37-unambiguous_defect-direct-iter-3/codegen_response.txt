The program is a command-line lottery tool for team events. It loads a staff list, sets up prizes, runs draws round by round, and saves the winners to a file.

**How to run it**

1. From inside `src/`, run `python -m lottery_system`. Running `python main.py` also works.
2. The default staff list ships as `src/lottery_system/data/default_list.txt`. If that file is missing, the same list built into the code is used instead.
3. Custom lists are `.txt` files with one person per line, written as `Name,Department`.
   - Blank lines are skipped, as are lines starting with `#` and a `Name,Department` header line.
   - A tab or full-width comma also works as the separator.

**Behaviours the PRD left open**

- **Li Bin's relaxed rule:** a prize drawn by Li Bin ignores its avoidance groups. Previous winners are still excluded.
- **Cheng Ru group fallback:** only rounds drawn by Chen Qixian or Jin Duo are recorded in `last_two_draw_results`, and the last two are kept.
  - If both show no Cheng Ru group winner, Ma Jiaqi's next pool contains only Cheng Ru group members who have not won yet. This applies even if that group is on the prize's avoidance list.
  - If none are left, the pool is empty.
  - The record is cleared after Ma Jiaqi's round.
- **Prize level for the summary:** levels are worked out from words in the prize name. "Grand" or "Special" ranks highest, then "First", "Second", "Third" and so on. Prizes with none of these words come after.
- **Default prizes:** each of the three default prizes is linked to one of the named drawers, so the fallback can happen without extra setup. Third Prize goes with Chen Qixian, Second Prize with Jin Duo and First Prize with Ma Jiaqi.
- **Export with no winners:** `export_to_csv` returns `False` and writes no file.
- **Bad prize input:** `add_prize` raises `ValueError` for an empty name, a duplicate name, or a quantity that isn't a positive whole number.

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A participant in the lottery."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"

    def __hash__(self) -> int:

@dataclass
class Prize:
    """A prize with its configuration."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if self.associated_drawer is not None:
            self.associated_drawer = str(self.associated_drawer).strip() or None

    def rule_description(self) -> str:
        parts = []
        if self.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("Avoidance groups: none")
        if self.associated_drawer:
            parts.append(f"Lottery person: {self.associated_drawer}")
        else:
            parts.append("Lottery person: not specified")
        parts.append("Previous winners are excluded")
        return "; ".join(parts)

class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.default_path = DEFAULT_LIST_PATH

    # ------------------------------------------------------------------ load
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._load_from_path(self.default_path)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if path is None or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not path.lower().endswith(".txt"):
            return [], [f"File Error: only .txt files are supported ({path})"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                content = f.read()
        except FileNotFoundError:
            if os.path.abspath(str(path)) == os.path.abspath(self.default_path):
                content = DEFAULT_LIST_CONTENT
            else:
                return [], [f"File Error: file not found ({path})"]
        except UnicodeDecodeError:
            return [], [f"File Error: file is not valid UTF-8 text ({path})"]
        except OSError as exc:
            return [], [f"File Error: cannot read file ({path}): {exc}"]
        return self._parse_content(content)

    @staticmethod
    def _parse_content(content: str) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(content.splitlines(), start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            normalized = line.replace("，", ",").replace("\t", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) == 2 and parts[0].lower() == "name" and parts[1].lower() in (
                "department",
                "group",
            ):
                continue
            if len(parts) != 2:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department', got '{line}'"
                )
                continue
            name, dept = parts
            if not name or not dept:
                errors.append(
                    f"Line {idx}: Format Error - name and department must not be empty ('{line}')"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    # -------------------------------------------------------------- validate
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: "OrderedDict[str, int]" = OrderedDict()
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return dict(info)

"""Prize configuration management."""

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and the round order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name,
        quantity,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else ""
        if not name:
            raise ValueError("Prize name must not be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        prize = Prize(name, qty, groups, drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def set_avoid_groups(self, index: int, groups: List[str]) -> bool:
        if not 0 <= index < len(self.prizes):
            return False
        cleaned = []
        for g in groups:
            g = g.strip()
            if g and g not in cleaned:
                cleaned.append(g)
        self.prizes[index].avoid_groups = cleaned
        return True

    def reorder_prizes(self, order) -> bool:
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        n = len(self.prizes)
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
FALLBACK_WATCH_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(s: Optional[str]) -> str:
    return (s or "").strip().lower()


class LotteryManager:
    """Executes draws according to the rule engine."""

    def __init__(self, users, result_manager=None, rng=None) -> None:
        self.users: List[User] = list(users)
        self.result_manager = result_manager
        self.rng = rng or random.Random()
        self.winner_names = set()
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []
        self.current_round_winners: List[User] = []

    # ----------------------------------------------------------------- rules
    def _is_winner(self, user: User) -> bool:
        return user.name in self.winner_names

    def is_fallback_active(self, prize: Prize) -> bool:
        return (
            _norm(prize.associated_drawer) == _norm(FALLBACK_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def describe_rules(self, prize: Prize) -> str:
        desc = prize.rule_description()
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            desc += "; Special rule: relaxed filtering (avoidance groups ignored)"
        if self.is_fallback_active(prize):
            desc += (
                f"; Fallback active: pool limited to '{PROTECTED_GROUP}' members"
            )
        return desc

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = [u for u in self.users if not self._is_winner(u)]
        if self.is_fallback_active(prize):
            return [u for u in candidates if _norm(u.department) == _norm(PROTECTED_GROUP)]
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return candidates
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in candidates if _norm(u.department) not in avoid]

    # ----------------------------------------------------------------- draws
    def start_round(self, prize: Prize) -> None:
        self.current_round_winners = []

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winner_names.add(user.name)
        self.winners.append(user)
        self.current_round_winners.append(user)
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return user

    def finish_round(self, prize: Prize) -> List[User]:
        drawer = _norm(prize.associated_drawer)
        results = list(self.current_round_winners)
        if drawer in {_norm(d) for d in FALLBACK_WATCH_DRAWERS}:
            contains = any(_norm(u.department) == _norm(PROTECTED_GROUP) for u in results)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_DRAWER):
            self.last_two_draw_results = []
        self.current_round_winners = []
        return results

    def run_round(self, prize: Prize) -> List[User]:
        """Draw a whole round without interaction."""
        self.start_round(prize)
        for _ in range(prize.quantity):
            if self.draw_one(prize) is None:
                break
        return self.finish_round(prize)

"""Result management: collecting, ordering and exporting winners."""

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User

LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
    ("consolation", 8),
]


def prize_level(name: str) -> int:
    low = (name or "").lower()
    for kw, lvl in LEVEL_KEYWORDS:
        if kw in low:
            return lvl
    return 9


class Winner:
    """A winning record."""

    def __init__(self, user: User, prize: Prize, order: int, timestamp=None) -> None:
        self.user = user
        self.prize = prize
        self.order = order
        self.timestamp = timestamp or datetime.now()

    def __repr__(self) -> str:
        return f"Winner({self.user!r}, {self.prize.name!r}, #{self.order})"


class ResultManager:
    """Stores winners and exports them."""

    def __init__(self) -> None:
        self.winners: List[Winner] = []
        self._counter = 0

    def add_winner(self, user: User, prize: Prize) -> Winner:
        self._counter += 1
        w = Winner(user, prize, self._counter)
        self.winners.append(w)
        return w

    def get_round_winners(self, prize: Prize) -> List[Winner]:
        return [w for w in self.winners if w.prize.name == prize.name]

    def _sorted_prizes(self, prizes) -> List[Prize]:
        prizes = list(prizes)
        known = {p.name for p in prizes}
        for w in self.winners:
            if w.prize.name not in known:
                prizes.append(w.prize)
                known.add(w.prize.name)
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda t: (prize_level(t[1].name), t[0]))
        return [p for _, p in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in self._sorted_prizes(prizes):
            ws = [w for w in self.winners if w.prize.name == p.name]
            ws.sort(key=lambda w: (w.timestamp, w.order))
            result[p.name] = ws
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export."
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["No", "Name", "Department", "Prize Name", "Winning Time"])
                no = 0
                for prize_name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        no += 1
                        writer.writerow([
                            no,
                            w.user.name,
                            w.user.department,
                            prize_name,
                            w.timestamp.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ask(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        return "exit"


class LotteryCLI:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery = None

    # ---------------------------------------------------------------- lists
    def _accept_list(self, users, errors) -> None:
        valid, dups = self.list_manager.validate_users(users)
        if errors:
            print("Format errors detected:")
            for e in errors:
                print("  " + e)
        if dups:
            print("Duplicate names detected:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
        if errors or dups:
            choice = ask("Remove invalid data and continue (r) or terminate upload (t)? ").strip().lower()
            if choice != "r":
                print("Upload terminated. Current list unchanged.")
                return
        if not valid:
            print("No valid users loaded.")
            return
        self.users = valid
        self.lottery = None
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._accept_list(users, errors)

    def load_custom(self) -> None:
        path = ask("Enter txt file path (format per line: Name,Department): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            return
        self._accept_list(users, errors)

    def show_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current valid groups:")
        for dept, cnt in info.items():
            print(f"  {dept}: {cnt} people")

    # --------------------------------------------------------------- prizes
    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def _select_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = ask("Avoidance groups (numbers or names, comma separated, empty for none): ").strip()
        result = []
        for token in raw.replace("，", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                g = groups[int(token) - 1]
            else:
                g = token
            if g not in result:
                result.append(g)
        return result

    def add_prize(self) -> None:
        name = ask("Prize name: ").strip()
        qty = ask("Quantity: ").strip()
        avoid = self._select_groups()
        drawer = ask("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print("Prize added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def configure_avoid(self) -> None:
        self.show_prizes()
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            return
        raw = ask("Prize number to configure: ").strip()
        if not raw.isdigit():
            print("Invalid number.")
            return
        groups = self._select_groups()
        if self.prize_manager.set_avoid_groups(int(raw) - 1, groups):
            print("Avoidance groups updated.")
        else:
            print("Invalid prize number.")

    def reorder(self) -> None:
        self.show_prizes()
        raw = ask("Enter new order as serial numbers (e.g. 3,1,2): ").strip()
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Order updated.")
            self.show_prizes()
        else:
            print("Invalid order.")

    # -------------------------------------------------------------- lottery
    def run_lottery(self) -> None:
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users, self.result_manager)
        lm = self.lottery
        print("Commands during the draw: Enter = draw next, 'pause', 'continue', 'terminate'.")
        for idx, prize in enumerate(prizes, 1):
            already = len(self.result_manager.get_round_winners(prize))
            remaining = prize.quantity - already
            if remaining <= 0:
                continue
            print("\n" + "=" * 60)
            print(f"Round {idx}: {prize.name} | Quantity: {prize.quantity}")
            print("Rules: " + lm.describe_rules(prize))
            print("=" * 60)
            lm.start_round(prize)
            paused = False
            drawn = 0
            terminated = False
            while drawn < remaining:
                if paused:
                    cmd = ask("[Paused] Enter 'continue' to resume or 'terminate' to skip: ").strip().lower()
                else:
                    cmd = ask(f"Currently drawing the {already + drawn + 1}th winner (Enter to draw): ").strip().lower()
                if cmd == "exit":
                    terminated = True
                    break
                if cmd == "pause":
                    paused = True
                    print("Lottery paused.")
                    continue
                if cmd == "continue":
                    if paused:
                        print("Lottery resumed.")
                    paused = False
                    continue
                if cmd == "terminate":
                    print("Current round terminated and skipped.")
                    terminated = True
                    break
                if paused:
                    continue
                user = lm.draw_one(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                drawn += 1
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            lm.finish_round(prize)
            print(f"\nResults of {prize.name}:")
            for i, w in enumerate(self.result_manager.get_round_winners(prize), 1):
                print(f"  {i}. {w.user.name} ({w.user.department})")
            if terminated and idx < len(prizes):
                nxt = ask("Continue to next round? (y/n): ").strip().lower()
                if nxt not in ("y", "yes", ""):
                    print("Lottery stopped.")
                    return
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not any(grouped.values()):
            print("No winners yet.")
            return
        print("\nComplete winner list:")
        for prize_name, ws in grouped.items():
            if not ws:
                continue
            print(f"[{prize_name}]")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {prize_name}")

    def run(self) -> None:
        actions = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.show_groups),
            "4": ("Load preset prizes", self._load_default_prizes),
            "5": ("Add custom prize", self.add_prize),
            "6": ("Configure avoidance groups", self.configure_avoid),
            "7": ("Reorder prizes", self.reorder),
            "8": ("View prizes", self.show_prizes),
            "9": ("Start lottery", self.run_lottery),
            "10": ("View results", self.show_summary),
            "11": ("Export results", self.export),
            "0": ("Exit", None),
        }
        print("Company Team Building Event Smart Lottery System")
        while True:
            print()
            for k, (label, _) in actions.items():
                print(f"  {k}. {label}")
            choice = ask("Select: ").strip().lower()
            if choice in ("0", "exit", "quit"):
                print("Bye.")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid choice.")
                continue
            action[1]()

        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        self.show_prizes()

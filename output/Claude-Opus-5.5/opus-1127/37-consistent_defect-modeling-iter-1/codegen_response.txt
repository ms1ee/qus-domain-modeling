from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import List, Optional


def normalize_name(value: Optional[str]) -> str:
    """Normalize a name for comparison: trimmed, single-spaced, lowercase."""
    if value is None:
        return ""
    return " ".join(str(value).split()).lower()


@dataclass
class User:
    """A lottery participant."""

    name: str
    department: str

    def __post_init__(self) -> None:
        self.name = " ".join(str(self.name).split())
        self.department = " ".join(str(self.department).split())

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        self.name = " ".join(str(self.name).split())
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [
                " ".join(str(g).split()) for g in self.avoid_groups if str(g).strip()
            ]
        if self.associated_drawer is not None:
            drawer = " ".join(str(self.associated_drawer).split())
            self.associated_drawer = drawer or None

    @property
    def level_rank(self) -> int:
        return prize_level_rank(self.name)

    def __str__(self) -> str:
        return f"{self.name} x{self.quantity}"


_LEVEL_PATTERNS = [
    (r"\b(special|grand|super)\b", 0),
    (r"\bfirst\b|\b1st\b", 1),
    (r"\bsecond\b|\b2nd\b", 2),
    (r"\bthird\b|\b3rd\b", 3),
    (r"\bfourth\b|\b4th\b", 4),
    (r"\bfifth\b|\b5th\b", 5),
    (r"\b(participation|consolation|lucky)\b", 50),
]


def prize_level_rank(prize_name: str) -> int:
    """Return a rank for the prize level (smaller = higher level)."""
    lowered = (prize_name or "").lower()
    for pattern, rank in _LEVEL_PATTERNS:
        if re.search(pattern, lowered):
            return rank
    return 99

_SEPARATOR_RE = re.compile(r"[,;，；]")


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.users: List[User] = []
        self.load_mode: str = ""
        self.source_path: str = ""

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        self.load_mode = "default"
        if os.path.isfile(DEFAULT_LIST_PATH):
            self.source_path = DEFAULT_LIST_PATH
            return self._load_from_path(DEFAULT_LIST_PATH)
        self.source_path = "<built-in>"
        return self._parse_lines(BUILTIN_DEFAULT_LINES)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        self.load_mode = "custom"
        self.source_path = str(path) if path is not None else ""
        if path is None or not str(path).strip():
            return [], ["File Error: no file path provided"]
        return self._load_from_path(str(path).strip())

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not os.path.exists(path):
            return [], [f"File Error: file not found '{path}'"]
        if not os.path.isfile(path):
            return [], [f"File Error: '{path}' is not a file"]
        try:
            with open(path, "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as fh:
                    lines = fh.read().splitlines()
            except (OSError, UnicodeDecodeError) as exc:
                return [], [f"File Error: cannot read file '{path}': {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file '{path}': {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for index, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in _SEPARATOR_RE.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {index}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # --------------------------------------------------------------- validation
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users or []:
            key = normalize_name(user.name)
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users or []:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize, normalize_name

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Holds the ordered list of prizes (lottery round sequence)."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []
        self.is_preset: bool = False

    def _find(self, name: str) -> Optional[Prize]:
        key = normalize_name(name)
        for prize in self.prizes:
            if normalize_name(prize.name) == key:
                return prize
        return None

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            self.prizes.append(Prize(name, qty, [], None))
        self.is_preset = True
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Prize quantity must be a positive integer")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError("Prize quantity must be a positive integer")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        if self._find(str(name)) is not None:
            raise ValueError(f"Prize '{str(name).strip()}' already exists")
        prize = Prize(str(name), qty, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            indexes = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(indexes) != n or sorted(indexes) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in indexes]
        return True

"""Winner records, summaries and export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime
from typing import Dict, List, Optional

from lottery_system.models import Prize, User, normalize_name, prize_level_rank


@dataclass
class Winner:
    user: User
    prize_name: str
    prize: Optional[Prize] = None
    win_time: datetime = field(default_factory=datetime.now)
    draw_order: int = 0

    def __str__(self) -> str:
        return (
            f"Congratulations {self.user.name} ({self.user.department}) "
            f"for winning {self.prize_name}"
        )


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []

    @property
    def total_winners(self) -> int:
        return len(self.winners)

    def has_won(self, user) -> bool:
        key = normalize_name(user.name)
        return any(normalize_name(w.user.name) == key for w in self.winners)

    def add_winner(self, user, prize) -> bool:
        if self.has_won(user):
            return False
        if isinstance(prize, Prize):
            name, prize_obj = prize.name, prize
        else:
            name, prize_obj = str(prize), None
        self.winners.append(
            Winner(user, name, prize_obj, datetime.now(), len(self.winners) + 1)
        )
        return True

    def _ordered_prize_names(self, prizes) -> List[str]:
        names: List[str] = []
        for p in prizes or []:
            n = p.name if isinstance(p, Prize) else str(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        indexed = list(enumerate(names))
        indexed.sort(key=lambda item: (prize_level_rank(item[1]), item[0]))
        return [n for _, n in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for name in self._ordered_prize_names(prizes):
            group = [w for w in self.winners if w.prize_name == name]
            group.sort(key=lambda w: (w.win_time, w.draw_order))
            result[name] = group
        return result

    def get_sorted_winners(self, prizes) -> List[Winner]:
        out: List[Winner] = []
        for group in self.get_winners_by_prize(prizes).values():
            out.extend(group)
        return out

    def export_to_csv(self, prizes, directory: Optional[str] = None):
        if not self.winners:
            return False, "No winners to export"
        directory = directory or os.getcwd()
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(directory, filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for w in self.get_sorted_winners(prizes):
                    writer.writerow(
                        [
                            w.user.name,
                            w.user.department,
                            w.prize_name,
                            w.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ]
                    )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and round process control."""

from __future__ import annotations

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User, normalize_name
from lottery_system.services.result_manager import ResultManager, Winner


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryManager:
    SPECIAL_DRAWER = "Li Bin"
    TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
    FALLBACK_DRAWER = "Ma Jiaqi"
    PROTECTED_GROUP = "Cheng Ru Group"

    def __init__(self, users, result_manager: Optional[ResultManager] = None) -> None:
        self.users: List[User] = list(users or [])
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winner_names = {normalize_name(w.user.name) for w in self.result_manager.winners}
        self.last_two_draw_results: List[bool] = []
        self.fallback_applied = False
        self.rng = random.Random()

    # ----------------------------------------------------------------- helpers
    def _is_winner(self, user: User) -> bool:
        return normalize_name(user.name) in self.winner_names

    @classmethod
    def _is_protected(cls, user: User) -> bool:
        dept = normalize_name(user.department).replace(" ", "")
        return dept == "chengrugroup" or dept == "chengru" or dept.startswith("chengru")

    @staticmethod
    def _in_avoid(user: User, prize: Prize) -> bool:
        avoid = {normalize_name(g) for g in (prize.avoid_groups or [])}
        return normalize_name(user.department) in avoid

    def _is_special(self, prize: Prize) -> bool:
        return normalize_name(prize.associated_drawer) == normalize_name(self.SPECIAL_DRAWER)

    def _fallback_triggered(self, prize: Prize) -> bool:
        if normalize_name(prize.associated_drawer) != normalize_name(self.FALLBACK_DRAWER):
            return False
        hist = self.last_two_draw_results
        return len(hist) >= 2 and not any(hist[-2:])

    # ----------------------------------------------------------------- engine
    def _get_eligible_pool(self, prize) -> List[User]:
        available = [u for u in self.users if not self._is_winner(u)]
        if self._fallback_triggered(prize):
            protected = [u for u in available if self._is_protected(u)]
            if protected:
                return protected
        if self._is_special(prize):
            return available
        return [u for u in available if not self._in_avoid(u, prize)]

    def _get_secondary_pool(self, prize) -> List[User]:
        """Avoidance-group members: eligible only after all other members."""
        if self._is_special(prize):
            return []
        return [u for u in self.users if not self._is_winner(u) and self._in_avoid(u, prize)]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        drawer = prize.associated_drawer
        if self._is_special(prize):
            parts.append(f"Special lottery person '{drawer}': avoidance groups not applied")
        elif prize.avoid_groups:
            parts.append(
                "Avoidance groups (drawn only after other members): "
                + ", ".join(prize.avoid_groups)
            )
        else:
            parts.append("No avoidance groups")
        if self._fallback_triggered(prize):
            parts.append(
                f"Fallback active: prize pool limited to '{self.PROTECTED_GROUP}' members"
            )
        if drawer:
            parts.insert(0, f"Lottery person: {drawer}")
        return "; ".join(parts)

    def draw_winner(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = self._get_secondary_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winner_names.add(normalize_name(user.name))
        self.result_manager.add_winner(user, prize)
        return user

    def _update_history(self, prize: Prize, round_users: List[User]) -> None:
        drawer = normalize_name(prize.associated_drawer)
        triggers = {normalize_name(t) for t in self.TRIGGER_DRAWERS}
        if drawer in triggers:
            self.last_two_draw_results.append(any(self._is_protected(u) for u in round_users))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == normalize_name(self.FALLBACK_DRAWER):
            self.last_two_draw_results = []
        elif drawer:
            self.last_two_draw_results = []

    # ----------------------------------------------------------------- rounds
    def run_round(
        self,
        prize: Prize,
        round_number: int = 1,
        input_func: Callable[[str], str] = input,
        output: Callable[[str], None] = print,
        interactive: bool = True,
    ) -> List[Winner]:
        self.fallback_applied = self._fallback_triggered(prize)
        if self.fallback_applied:
            output(
                f"[Fallback] Neither {' nor '.join(self.TRIGGER_DRAWERS)} drew a "
                f"'{self.PROTECTED_GROUP}' member; this round is limited to that group."
            )
        round_users: List[User] = []
        start = len(self.result_manager.winners)
        terminated = False
        while len(round_users) < prize.quantity and not terminated:
            idx = len(round_users) + 1
            if interactive:
                try:
                    cmd = input_func(
                        f"[Round {round_number}] Press Enter to draw the {ordinal(idx)} winner "
                        "(commands: pause / continue / terminate): "
                    ).strip().lower()
                except EOFError:
                    cmd = ""
                if cmd == "pause":
                    output("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                    while True:
                        try:
                            c2 = input_func("(paused) > ").strip().lower()
                        except EOFError:
                            c2 = "terminate"
                        if c2 == "continue":
                            output("Lottery resumed.")
                            break
                        if c2 == "terminate":
                            terminated = True
                            break
                        output("Lottery is paused. Type 'continue' or 'terminate'.")
                elif cmd == "terminate":
                    terminated = True
                elif cmd == "continue":
                    output("Lottery is not paused, drawing continues.")
                elif cmd:
                    output(f"Unknown command '{cmd}', drawing continues.")
                if terminated:
                    output(f"Round {round_number} terminated; skipping remaining draws.")
                    break
            output(f"Currently drawing the {ordinal(idx)} winner...")
            user = self.draw_winner(prize)
            if user is None:
                output("No eligible participants remaining for this prize.")
                break
            round_users.append(user)
            output(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        self._update_history(prize, round_users)
        return self.result_manager.winners[start:]

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import User, normalize_name
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class LotteryApp:
    def __init__(self) -> None:
        self.event_name = "Company Team Building Event"
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery: Optional[LotteryManager] = None
        self.drawn_prizes = set()

    # ---------------------------------------------------------------- utils
    @staticmethod
    def ask(prompt: str) -> str:
        try:
            return input(prompt)
        except EOFError:
            return "0"

    def _set_users(self, users: List[User]) -> None:
        self.users = users
        self.lottery = None

    def _get_lottery(self) -> LotteryManager:
        if self.lottery is None:
            self.lottery = LotteryManager(self.users, self.result_manager)
        return self.lottery

    def _handle_loaded(self, users, errors) -> None:
        valid, dups = self.list_manager.validate_users(users)
        if errors:
            print("Problems detected:")
            for e in errors:
                print("  " + e)
        if dups:
            print("Duplicate names detected:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
        if not valid:
            print("No valid participants loaded.")
            return
        if errors or dups:
            choice = self.ask(
                "1. Remove invalid data and continue  2. Terminate upload\nChoose: "
            ).strip()
            if choice != "1":
                print("Upload terminated. Current list unchanged.")
                return
        self._set_users(valid)
        print(f"Loaded {len(valid)} valid participants.")

    # ---------------------------------------------------------------- actions
    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._handle_loaded(users, errors)

    def load_custom(self) -> None:
        path = self.ask("Enter txt file path (each line 'Name,Department Group'): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print(e)
            return
        self._handle_loaded(users, errors)

    def view_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print(f"Valid groups ({len(info)}):")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    def remove_participant(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        name = self.ask("Enter name to remove: ").strip()
        before = len(self.users)
        self._set_users([u for u in self.users if normalize_name(u.name) != normalize_name(name)])
        print("Removed." if len(self.users) < before else "Name not found.")

    def view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            avoid = ", ".join(p.avoid_groups) if p.avoid_groups else "None"
            drawer = p.associated_drawer or "None"
            status = " [drawn]" if p.name in self.drawn_prizes else ""
            print(f"  {i}. {p.name} | Qty: {p.quantity} | Avoid: {avoid} | Drawer: {drawer}{status}")

    def add_prize(self) -> None:
        name = self.ask("Prize name: ").strip()
        qty = self.ask("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid: List[str] = []
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            sel = self.ask("Avoidance groups (numbers separated by comma, blank for none): ").strip()
            for part in sel.replace("，", ",").split(","):
                part = part.strip()
                if part.isdigit() and 1 <= int(part) <= len(groups):
                    if groups[int(part) - 1] not in avoid:
                        avoid.append(groups[int(part) - 1])
                elif part:
                    print(f"Ignored invalid selection '{part}'")
        else:
            sel = self.ask("Avoidance groups (names separated by comma, blank for none): ")
            avoid = [g.strip() for g in sel.split(",") if g.strip()]
        drawer = self.ask("Associated lottery person (blank for none): ").strip() or None
        try:
            p = self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize '{p.name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def reorder(self) -> None:
        self.view_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = self.ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.view_prizes()
        else:
            print("Invalid order: must contain each serial number exactly once.")

    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        lottery = self._get_lottery()
        for rnd, prize in enumerate(prizes, 1):
            if prize.name in self.drawn_prizes:
                print(f"Round {rnd}: {prize.name} already drawn, skipped.")
                continue
            print("=" * 60)
            print(f"Round {rnd}: {prize.name}")
            print(f"Quantity: {prize.quantity}")
            print(f"Rules: {lottery.describe_rules(prize)}")
            print("=" * 60)
            winners = lottery.run_round(prize, rnd, self.ask, print, True)
            self.drawn_prizes.add(prize.name)
            print(f"--- Round {rnd} results ({prize.name}) ---")
            if not winners:
                print("  No winners in this round.")
            for i, w in enumerate(winners, 1):
                print(f"  {i}. {w.user.name} ({w.user.department}) - {w.prize_name}")
        print("All rounds completed.")
        self.show_summary()

    def show_summary(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("=" * 60)
        print("Complete Winner List")
        if not self.result_manager.winners:
            print("  No winners yet.")
            return
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            print(f"[{prize_name}]")
            for w in winners:
                print(
                    f"  {w.user.name} ({w.user.department}) - {w.prize_name} "
                    f"@ {w.win_time.strftime('%H:%M:%S')}"
                )

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ---------------------------------------------------------------- loop
    def run(self) -> None:
        print(f"=== {self.event_name} Smart Lottery System ===")
        actions = {
            "1": self.load_default,
            "2": self.load_custom,
            "3": self.view_groups,
            "4": self.remove_participant,
            "5": lambda: [print(e) for e in self.prize_manager.load_default_prizes()]
            or print("Preset prizes loaded."),
            "6": self.add_prize,
            "7": self.view_prizes,
            "8": self.reorder,
            "9": self.start_lottery,
            "10": self.show_summary,
            "11": self.export,
        }
        while True:
            print(
                "\n1. Load default list   2. Upload custom list   3. View groups\n"
                "4. Remove participant  5. Load preset prizes   6. Add custom prize\n"
                "7. View prizes         8. Reorder prizes       9. Start lottery\n"
                "10. View results       11. Export results      0. Exit"
            )
            choice = self.ask("Choose: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid choice.")
                continue
            action()


def main() -> None:
    LotteryApp().run()


if __name__ == "__main__":
    main()

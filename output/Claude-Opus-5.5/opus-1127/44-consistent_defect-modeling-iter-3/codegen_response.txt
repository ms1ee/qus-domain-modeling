"""Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
Its structure is::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str,
                      "distance": int (meters), "line": str}]}
"""

import json
import os

# Absolute path of the bundled data file, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                              "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway data file cannot be loaded."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path to the JSON data file (defaults to ``DATA_FILE_PATH``)
    :return: the parsed JSON content (a dict)
    :raises DataLoadError: if the file is missing or malformed
    """
    if path is None:
        path = DATA_FILE_PATH
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Data file is not valid JSON: {path} ({exc})") from exc
    if not isinstance(data, dict):
        raise DataLoadError("Data file root must be a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file."""
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, ensure_ascii=False, indent=2)


def build_graph(data):
    """Build a directed weighted graph from the loaded data.

    Station numbers are assigned here (in order of appearance) and passed to
    the ``Graph``.

    :return: tuple ``(graph, station_to_number, number_to_station)``
    """
    from models import Edge, Graph

    graph = Graph()
    station_to_number = {}
    number_to_station = {}

    def _number_of(name):
        if name not in station_to_number:
            num = len(station_to_number)
            station_to_number[name] = num
            number_to_station[num] = name
            graph.add_vertex(num)
        return station_to_number[name]

    for station in data.get("stations", []):
        name = station.get("name")
        if not name:
            continue
        _number_of(name)
        graph.set_station_lines(name, list(station.get("lines", [])))

    for conn in data.get("connections", []):
        src = conn.get("source")
        dst = conn.get("dest")
        if not src or not dst:
            continue
        try:
            distance = int(conn.get("distance", 0))
        except (TypeError, ValueError):
            continue
        s = _number_of(src)
        d = _number_of(dst)
        line = conn.get("line")
        graph.add_edge(Edge(s, d, distance, line))
        # Make sure every endpoint has a line list (transfer detection).
        for nm in (src, dst):
            lines = graph.station_lines.setdefault(nm, [])
            if line and line not in lines:
                lines.append(line)

    return graph, station_to_number, number_to_station

"""Core data structures: stations, lines, edges and the network graph."""

from dataclasses import dataclass, field
from typing import Dict, List, Optional


@dataclass
class Station:
    """A subway station."""
    name: str
    number: int
    lines: List[str] = field(default_factory=list)

    @property
    def is_transfer_station(self):
        return len(self.lines) > 1


@dataclass
class Line:
    """A subway line with its ordered station sequence."""
    name: str
    station_sequence: List[str] = field(default_factory=list)


class Edge:
    """A directed weighted edge between two station numbers."""

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def from_(self):
        return self.source

    def to(self):
        return self.dest

    def __repr__(self):
        return (f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, "
                f"{self.line!r})")

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.distance, self.line) == \
            (other.source, other.dest, other.distance, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.distance, self.line))


class Graph:
    """Weighted directed graph with adjacency lists.

    Station numbers are assigned by the caller. Each edge is usable only in
    its listed direction.
    """

    def __init__(self):
        self.adj: Dict[int, List[Edge]] = {}
        self.station_lines: Dict[str, List[str]] = {}
        self._edge_count = 0

    # ----- vertices -----
    def add_vertex(self, v):
        self.adj.setdefault(v, [])

    def vertices(self):
        return list(self.adj.keys())

    def V(self):
        return len(self.adj)

    def E(self):
        return self._edge_count

    # ----- station metadata -----
    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    # ----- edges -----
    def add_edge(self, edge):
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, [])
        self._edge_count += 1

    def edges_from(self, v):
        return self.adj.get(v, [])

    def edges(self):
        for lst in self.adj.values():
            for e in lst:
                yield e

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance of an existing edge. Returns True on success."""
        updated = False
        for e in self.adj.get(source, []):
            if e.dest == dest and (line is None or e.line == line):
                e.distance = distance
                e.weight = distance
                updated = True
        return updated

"""Dijkstra based path computation over the subway graph."""

import heapq
import math

MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Penalty (in meters) equivalent to one transfer for comprehensive scoring:
# 5 minutes per transfer at 1.5 minutes per kilometer.
TRANSFER_PENALTY_METERS = 5.0 / 1.5 * 1000.0
# Weight multiplier applied to edges on non-preferred lines.
NON_PREFERRED_FACTOR = 1.5


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None,
                 preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Binary-heap min priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = 0

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, self._counter, item))
        self._counter += 1

    insert = push

    def pop(self):
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def del_min(self):
        return self.pop()

    def is_empty(self):
        return not self._heap

    @property
    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)


def _norm(line):
    return "".join(str(line).split()).lower() if line is not None else ""


class ShortestPath:
    """Dijkstra's algorithm with line-aware states (for transfer handling)."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        if self.config.mode not in MODES:
            raise ValueError(f"Unknown path mode: {self.config.mode}")
        if self.config.max_transfers is not None and self.config.max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        self._preferred = {_norm(l) for l in self.config.preferred_lines if l}
        self.dist_to = {v: math.inf for v in graph.vertices()}
        self.dist_to[source] = 0
        self._best_state = {}
        self._info = {}  # key -> [cost, eff, dist, transfers, parent, edge, line]
        self._run()

    # ------------------------------------------------------------------
    def _key(self, v, line, transfers):
        if self.config.max_transfers is not None:
            return (v, line, transfers)
        return (v, line, None)

    def _cost(self, eff, transfers):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, eff)
        if mode == "comprehensive":
            return (eff + transfers * TRANSFER_PENALTY_METERS, transfers)
        return (eff, transfers)

    def _edge_weight(self, edge):
        if not self._preferred or _norm(edge.line) in self._preferred:
            return edge.weight
        return edge.weight * NON_PREFERRED_FACTOR

    def _run(self):
        pq = MinPQ()
        start = self._key(self.source, None, 0)
        self._info[start] = [self._cost(0, 0), 0, 0, 0, None, None, None]
        pq.push(self._info[start][0], start)
        settled = set()
        max_t = self.config.max_transfers
        while not pq.is_empty():
            cost, key = pq.pop()
            if key in settled:
                continue
            info = self._info[key]
            if cost != info[0]:
                continue
            settled.add(key)
            v = key[0]
            _, eff, dist, transfers, _, _, line = info
            if v not in self._best_state:
                self._best_state[v] = key
                self.dist_to[v] = dist
            for edge in self.graph.edges_from(v):
                new_line = edge.line if edge.line is not None else line
                nt = transfers
                if line is not None and new_line != line:
                    nt += 1
                if max_t is not None and nt > max_t:
                    continue
                neff = eff + self._edge_weight(edge)
                nd = dist + edge.weight
                nkey = self._key(edge.dest, new_line, nt)
                if nkey in settled:
                    continue
                ncost = self._cost(neff, nt)
                old = self._info.get(nkey)
                if old is None or ncost < old[0]:
                    self._info[nkey] = [ncost, neff, nd, nt, key, edge, new_line]
                    pq.push(ncost, nkey)

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return (station numbers from source to target, edges used)."""
        if target not in self._best_state:
            return [], []
        edges = []
        key = self._best_state[target]
        while key is not None:
            info = self._info[key]
            if info[5] is not None:
                edges.append(info[5])
            key = info[4]
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._best_state:
            return 0
        return self._info[self._best_state[target]][3]

"""Beijing Subway fare calculation.

Fare structure (distances inclusive of upper bound):
  0-6 km: 3 yuan, 6-12 km: 4 yuan, 12-22 km: 5 yuan, 22-32 km: 6 yuan,
  over 32 km: +1 yuan for every additional 20 km (rounded up).
"""

import math
import numbers

FARE_TIERS = [
    # (max distance in meters inclusive, fare)
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_BASE_DISTANCE = 32000
EXTENDED_BASE_FARE = 6
EXTENDED_INCREMENT_DISTANCE = 20000
EXTENDED_INCREMENT_FARE = 1

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, numbers.Real):
        raise TypeError("distance must be a number (meters)")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("distance must be finite")
    if distance < 0:
        raise ValueError("distance must be non-negative")


def Sum_money(distance):
    """Return the fare (yuan) for a travel distance in meters."""
    _validate_distance(distance)
    for max_dist, fare in FARE_TIERS:
        if distance <= max_dist:
            return fare
    extra = distance - EXTENDED_BASE_DISTANCE
    steps = math.ceil(extra / EXTENDED_INCREMENT_DISTANCE)
    return EXTENDED_BASE_FARE + steps * EXTENDED_INCREMENT_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes (1.5 min/km, 5 min/transfer)."""
    _validate_distance(distance)
    if transfer_count is None or transfer_count < 0:
        transfer_count = 0
    return distance / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

from data_loader import (DATA_FILE_PATH, build_graph, load_subway_data,

class SubwayNetwork:
    """Manages stations, connections and lines; supports data updates."""

    def __init__(self, data=None, data_source=DATA_FILE_PATH):
        self.data_source = data_source
        self.data = data if data is not None else load_subway_data(data_source)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ----- queries -----
    def station_names(self):
        return [s["name"] for s in self.data["stations"]]

    def find_station(self, name):
        for s in self.data["stations"]:
            if s["name"] == name:
                return s
        return None

    def lines(self):
        """Return dict line name -> ordered station sequence."""
        result = {}
        for s in self.data["stations"]:
            for l in s.get("lines", []):
                result.setdefault(l, [])
        for c in self.data["connections"]:
            seq = result.setdefault(c.get("line"), [])
            for nm in (c["source"], c["dest"]):
                if nm not in seq:
                    seq.append(nm)
        return result

    def transfer_stations(self):
        return [s["name"] for s in self.data["stations"] if len(s.get("lines", [])) > 1]

    def build_graph(self):
        return build_graph(self.data)

    # ----- updates -----
    def add_station(self, name, lines):
        st = self.find_station(name)
        if st is None:
            self.data["stations"].append({"name": name, "lines": list(lines)})
        else:
            for l in lines:
                if l not in st["lines"]:
                    st["lines"].append(l)

    def add_line(self, line_name, stations, distances, bidirectional=True):
        """Add a new line: ``stations`` ordered, ``distances`` between neighbours."""
        if len(distances) != max(len(stations) - 1, 0):
            raise ValueError("distances must have len(stations) - 1 entries")
        for nm in stations:
            self.add_station(nm, [line_name])
        for a, b, d in zip(stations, stations[1:], distances):
            if d <= 0:
                raise ValueError("distance must be positive")
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": int(d), "line": line_name})
            if bidirectional:
                self.data["connections"].append(
                    {"source": b, "dest": a, "distance": int(d), "line": line_name})

    def update_station(self, name, new_name=None, lines=None):
        st = self.find_station(name)
        if st is None:
            raise KeyError(f"Station '{name}' does not exist")
        if lines is not None:
            st["lines"] = list(lines)
        if new_name and new_name != name:
            st["name"] = new_name
            for c in self.data["connections"]:
                if c["source"] == name:
                    c["source"] = new_name
                if c["dest"] == name:
                    c["dest"] = new_name

    def update_distance(self, source, dest, distance, line=None, both_directions=True):
        if distance <= 0:
            raise ValueError("distance must be positive")
        count = 0
        for c in self.data["connections"]:
            if line is not None and c.get("line") != line:
                continue
            if (c["source"], c["dest"]) == (source, dest) or \
                    (both_directions and (c["source"], c["dest"]) == (dest, source)):
                c["distance"] = int(distance)
                count += 1
        if count == 0:
            raise KeyError(f"No connection between '{source}' and '{dest}'")
        return count

    def save(self, path=None):
        save_subway_data(self.data, path or self.data_source)

"""Data integrity checking for the subway network data."""

MAIN_LINES = ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8",
              "Line 10", "Line 13", "Line 14", "Line 15", "Changping Line",
              "Yizhuang Line", "Fangshan Line", "Airport Line"]

_CN = {"Line 1": "1号线", "Line 2": "2号线", "Line 4": "4号线", "Line 5": "5号线",
       "Line 6": "6号线", "Line 8": "8号线", "Line 10": "10号线",
       "Line 13": "13号线", "Line 14": "14号线", "Line 15": "15号线",
       "Changping Line": "昌平线", "Yizhuang Line": "亦庄线",
       "Fangshan Line": "房山线", "Airport Line": "首都机场线"}

MAX_REASONABLE_DISTANCE = 50000


def _norm(s):
    return "".join(str(s).split()).lower()


class DataIntegrityChecker:
    """Checks station coverage, connection integrity and distance accuracy."""

    def __init__(self, data):
        self.data = data
        self.station_coverage_valid = False
        self.connection_integrity_valid = False
        self.distance_accuracy_valid = False
        self.errors = []
        self.warnings = []

    def check_station_coverage(self):
        all_lines = set()
        for s in self.data.get("stations", []):
            for l in s.get("lines", []):
                all_lines.add(_norm(l))
        for c in self.data.get("connections", []):
            if c.get("line"):
                all_lines.add(_norm(c["line"]))
        missing = []
        for line in MAIN_LINES:
            cands = {_norm(line), _norm(_CN.get(line, line))}
            if not any(any(cd == l or cd in l for l in all_lines) for cd in cands):
                missing.append(line)
        if not self.data.get("stations"):
            self.errors.append("No station data present")
            self.station_coverage_valid = False
        else:
            self.station_coverage_valid = True
        for m in missing:
            self.warnings.append(f"Main line '{m}' not found in data")
        return self.station_coverage_valid

    def check_connection_integrity(self):
        ok = True
        names = {}
        for s in self.data.get("stations", []):
            nm = s.get("name")
            if not nm:
                self.errors.append("Station without name")
                ok = False
                continue
            if nm in names:
                self.warnings.append(f"Duplicate station '{nm}'")
            names[nm] = set(s.get("lines", []))
        connected = set()
        for i, c in enumerate(self.data.get("connections", [])):
            src, dst = c.get("source"), c.get("dest")
            for nm in (src, dst):
                if nm not in names:
                    self.errors.append(f"Connection #{i} references unknown station '{nm}'")
                    ok = False
            if src == dst:
                self.errors.append(f"Connection #{i} is a self-loop at '{src}'")
                ok = False
            line = c.get("line")
            for nm in (src, dst):
                if nm in names and line and line not in names[nm]:
                    self.warnings.append(
                        f"Station '{nm}' not listed on line '{line}' used by connection #{i}")
            connected.add(src)
            connected.add(dst)
        for nm in names:
            if nm not in connected:
                self.warnings.append(f"Station '{nm}' has no connections")
        self.connection_integrity_valid = ok
        return ok

    def check_distance_accuracy(self):
        ok = True
        for i, c in enumerate(self.data.get("connections", [])):
            d = c.get("distance")
            if isinstance(d, bool) or not isinstance(d, (int, float)):
                self.errors.append(f"Connection #{i} has invalid distance {d!r}")
                ok = False
            elif d <= 0:
                self.errors.append(f"Connection #{i} has non-positive distance {d}")
                ok = False
            elif d > MAX_REASONABLE_DISTANCE:
                self.errors.append(f"Connection #{i} has unrealistic distance {d}")
                ok = False
        self.distance_accuracy_valid = ok
        return ok

    def run(self):
        self.check_station_coverage()
        self.check_connection_integrity()
        self.check_distance_accuracy()
        return self.is_valid()

    def is_valid(self):
        return (self.station_coverage_valid and self.connection_integrity_valid
                and self.distance_accuracy_valid)

    def report(self):
        lines = ["Data integrity check:",
                 f"  Stations: {len(self.data.get('stations', []))}",
                 f"  Connections: {len(self.data.get('connections', []))}",
                 f"  Station coverage: {'OK' if self.station_coverage_valid else 'FAILED'}",
                 f"  Connection integrity: {'OK' if self.connection_integrity_valid else 'FAILED'}",
                 f"  Distance accuracy: {'OK' if self.distance_accuracy_valid else 'FAILED'}"]
        for e in self.errors:
            lines.append(f"  ERROR: {e}")
        for w in self.warnings:
            lines.append(f"  WARNING: {w}")
        lines.append("Result: " + ("PASSED" if self.is_valid() else "FAILED"))
        return "\n".join(lines)

"""Path visualization, summary generation and export."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def build_route_info(start_name, end_name, mode, stations, edges,
                     number_to_station, transfer_count):
    """Build a dict describing a route (used for text and JSON output)."""
    names = [number_to_station[s] for s in stations]
    total = sum(e.distance for e in edges)
    segments = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            seg = segments[-1]
            seg["stations"].append(number_to_station[e.dest])
            seg["distance"] += e.distance
        else:
            segments.append({"line": e.line,
                             "stations": [number_to_station[e.source],
                                          number_to_station[e.dest]],
                             "distance": e.distance})
    transfers = []
    for a, b in zip(segments, segments[1:]):
        transfers.append({"station": b["stations"][0],
                          "from_line": a["line"], "to_line": b["line"]})
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    fare = Sum_money(total)
    est = calculate_estimated_time(total, transfer_count)
    return {
        "query": {"start": start_name, "end": end_name, "mode": mode},
        "path": names,
        "distance": total,
        "distance_km": round(total / 1000.0, 2),
        "fare": fare,
        "estimated_time": round(est, 1),
        "transfer_count": transfer_count,
        "lines": lines,
        "segments": segments,
        "transfers": transfers,
    }


def generate_summary(info):
    """Key information summary (distance in meters)."""
    return {"total_distance": info["distance"],
            "total_fare": info["fare"],
            "transfer_count": info["transfer_count"],
            "estimated_time": info["estimated_time"]}


def format_distance(meters):
    return f"{meters} meters ({meters / 1000.0:.2f} km)"


def format_text(info):
    out = []
    q = info["query"]
    out.append("=" * 50)
    out.append(f"Route: {q['start']} -> {q['end']} (mode: {q['mode']})")
    out.append("-" * 50)
    out.append("Stations: " + " -> ".join(info["path"]))
    out.append(f"Number of stations: {len(info['path'])}")
    out.append("Segments:")
    for seg in info["segments"]:
        out.append(f"  [{seg['line']}] " + " -> ".join(seg["stations"])
                   + f" ({format_distance(seg['distance'])})")
    if info["transfers"]:
        out.append("Transfers:")
        for t in info["transfers"]:
            out.append(f"  Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
    else:
        out.append("Transfers: none")
    out.append("Lines involved: " + ", ".join(str(l) for l in info["lines"]))
    out.append(f"Total distance: {format_distance(info['distance'])}")

            fh.write(json.dumps(payload, ensure_ascii=False, indent=2))
        else:
            fh.write("\n\n".join(format_text(r) for r in results))
        fh.write("\n")

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import difflib
import re
import sys

from data_checker import DataIntegrityChecker
from data_loader import DATA_FILE_PATH, DataLoadError, build_graph, load_subway_data
from pathfinding import MODES, PathfindingConfig, ShortestPath
from report import build_route_info, export_results, format_json, format_text

QUIT_WORDS = {"q", "quit", "exit"}
FORMAT_ERROR = ("Input format incorrect, please use "
                "'Starting Station; Ending Station' format")


class ValidationError(Exception):
    def __init__(self, error_type, message):
        super().__init__(message)
        self.error_type = error_type
        self.message = message


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    p.add_argument("--data_file", default=DATA_FILE_PATH)
    args = p.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        p.error("--max_transfers must be non-negative")
    return args


class CommandLineInterface:
    def __init__(self, args, data):
        self.args = args
        self.data = data
        self.graph, self.station_to_number, self.number_to_station = build_graph(data)
        prefs = []
        if args.line_preference:
            prefs = [s.strip() for s in re.split(r"[,，]", args.line_preference) if s.strip()]
        self.config = PathfindingConfig(args.mode, args.max_transfers, prefs)
        self.results = []
        self.is_running = False
        self._lower = {n.lower(): n for n in self.station_to_number}

    def resolve_station(self, name, role):
        name = name.strip()
        if name in self.station_to_number:
            return name
        cands = [name.lower()]
        for suffix in ("站", " station"):
            if name.lower().endswith(suffix):
                cands.append(name.lower()[:-len(suffix)].strip())
        for c in cands:
            if c in self._lower:
                return self._lower[c]
        msg = f"{role} station '{name}' does not exist, please check the station name"
        close = difflib.get_close_matches(name, list(self.station_to_number), n=3)
        if close:
            msg += " (did you mean: " + ", ".join(close) + "?)"
        raise ValidationError("station_not_found", msg)

    def parse_query(self, line):
        parts = re.split(r"[;,；，]", line)
        if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
            raise ValidationError("format", FORMAT_ERROR)
        start = self.resolve_station(parts[0], "Starting")
        end = self.resolve_station(parts[1], "Ending")
        if start == end:
            raise ValidationError("logic",
                                  "Starting station cannot be the same as the ending station")
        return start, end

    def process_query(self, line):
        """Process one query line; returns output text."""
        try:
            start, end = self.parse_query(line)
        except ValidationError as e:
            return e.message
        try:
            s = self.station_to_number[start]
            t = self.station_to_number[end]
            sp = ShortestPath(self.graph, s, self.config)
            if not sp.has_path_to(t):
                msg = f"No path found from '{start}' to '{end}'"
                if self.config.max_transfers is not None:
                    msg += f" within {self.config.max_transfers} transfer(s)"
                return msg
            stations, edges = sp.path_with_edges(t)
            transfers = sp.get_transfer_count(t)
        except Exception as e:  # path calculation exception
            return f"Path calculation error: {e}"
        try:
            info = build_route_info(start, end, self.config.mode, stations, edges,
                                    self.number_to_station, transfers)
        except Exception as e:  # fare calculation exception
            return f"Fare calculation error: {e}"
        self.results.append(info)
        if self.args.output_file:
            try:
                export_results(self.results, self.args.output_file, self.args.format)
            except OSError as e:
                print(f"Failed to write output file: {e}", file=sys.stderr)
        return format_json(info) if self.args.format == "json" else format_text(info)

    def run(self, stdin=None):
        stdin = stdin or sys.stdin
        self.is_running = True
        print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' "
              "(or 'quit' to exit).")
        while self.is_running:
            print("> ", end="", flush=True)
            line = stdin.readline()
            if not line:
                break
            line = line.strip()
            if not line:
                continue
            if line.lower() in QUIT_WORDS:
                break
            print(self.process_query(line), flush=True)
        self.is_running = False
        print("Goodbye.")


def main(argv=None):
    args = parse_args(argv)
    try:
        data = load_subway_data(args.data_file)
    except DataLoadError as e:
        print(f"Error: {e}", file=sys.stderr)
        return 1
    if args.check_data:
        checker = DataIntegrityChecker(data)
        ok = checker.run()
        print(checker.report())
        return 0 if ok else 1
    CommandLineInterface(args, data).run()
    return 0

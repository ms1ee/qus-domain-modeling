"""Global variable storage shared by all modules of the dispatch system.

The store is a plain dictionary hidden behind small accessor functions so
that every module reads and writes shared state the same way.

Well known keys:
    funds             -- int, current system funds (may become negative)
    riders            -- list[Rider], all riders in the system
    rider_id_counter  -- int, last rider id handed out
    running           -- bool, controls the main loop
    time              -- int, current simulated tick
    orders            -- list[Order], every order ever created
"""

_global_dict = {}


def _init():
    """(Re)initialise the global dictionary to an empty state."""
    global _global_dict
    _global_dict = {}


def set_value(key, value):
    """Store ``value`` under ``key``."""
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` or ``default`` when missing."""
    return _global_dict.get(key, default)


def has_key(key):
    """Return True when ``key`` is present in the store."""
    return key in _global_dict


def all_values():
    """Return a shallow copy of the whole store (for debugging)."""
    return dict(_global_dict)

"""System configuration: defaults and JSON configuration file loading."""

import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "rider_cost": 300,
    "rider_start_x": 435,
    "rider_start_y": 385,
    "rider_speed": 10,
    "map_size": 800,
    "cell_size": 100,
    "ticks_per_command": 1,
    "max_run_ticks": 100000,
    "state_file": "dispatch_state.json",
}

_active = dict(DEFAULT_CONFIG)


def get(key, default=None):
    """Return an active configuration value."""
    return _active.get(key, DEFAULT_CONFIG.get(key, default))


def get_all():
    """Return a copy of the active configuration."""
    return dict(_active)


def set_active(cfg):
    """Replace the active configuration (missing keys fall back to defaults)."""
    global _active
    merged = dict(DEFAULT_CONFIG)
    if cfg:
        merged.update(cfg)
    _active = merged


def reset():
    """Reset the active configuration to defaults."""
    set_active(None)


def load_config(path=None):
    """Load a JSON configuration file and merge it with the defaults.

    Invalid or missing files are reported in the log and ignored.
    Returns the merged configuration dictionary.
    """
    cfg = dict(DEFAULT_CONFIG)
    if path and os.path.isfile(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                params = data.get("system", data)
                for key, value in params.items():
                    if key in DEFAULT_CONFIG:
                        cfg[key] = value
        except (OSError, ValueError) as exc:
            logger.warning("Could not load config %s: %s", path, exc)
    return cfg

"""Grid based path planning for riders.

The map is an 800x800 coordinate system split into an 8x8 grid of
100x100 cells.  Restaurants and customers are located at cell centres.
Riders move horizontally first and then vertically (right-angle turns);
when a rider is directly above or below its target it moves straight
vertically (the "special case").
"""

import math

CELL_SIZE = 100
MAP_SIZE = 800
GRID_DIM = MAP_SIZE // CELL_SIZE


class PathPlanner:
    """Collection of path planning helpers (all static/class methods)."""

    @staticmethod
    def to_grid_center_value(value):
        """Convert one coordinate to the centre of its grid cell.

        ``center = (v // 100) * 100 + 50``; values outside 0-800 are clamped
        and 800 is treated as belonging to the last cell.
        """
        v = int(value)
        if v < 0:
            v = 0
        if v >= MAP_SIZE:
            v = MAP_SIZE - 1
        return (v // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2

    @classmethod
    def to_grid_center(cls, pos):
        """Convert an (x, y) pair to its grid cell centre."""
        return (cls.to_grid_center_value(pos[0]),
                cls.to_grid_center_value(pos[1]))

    @staticmethod
    def grid_index(pos):
        """Return the (column, row) grid cell index of a position."""
        def idx(v):
            i = int(v) // CELL_SIZE
            return max(0, min(GRID_DIM - 1, i))
        return idx(pos[0]), idx(pos[1])

    @staticmethod
    def manhattan(a, b):
        """Manhattan (street) distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def euclidean(a, b):
        """Euclidean distance between two points."""
        return math.hypot(a[0] - b[0], a[1] - b[1])

    @staticmethod
    def is_on_street(pos):
        """True when the position lies on a grid line (a street)."""
        return pos[0] % CELL_SIZE == 0 or pos[1] % CELL_SIZE == 0

    @staticmethod
    def is_special_case(x, y, tx, ty):
        """True when the rider is directly above or below the target."""
        return x == tx and y != ty

    @staticmethod
    def plan_route(start, target):
        """Return the waypoints of the right-angle route start -> target."""
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        points = [(sx, sy)]
        if sx != tx and sy != ty:
            points.append((tx, sy))
        if points[-1] != (tx, ty):
            points.append((tx, ty))
        return points

    @classmethod
    def plan_multi_route(cls, start, stops):
        """Plan a continuous route through several stops in order."""
        points = [(int(start[0]), int(start[1]))]
        for stop in stops:
            segment = cls.plan_route(points[-1], stop)
            points.extend(segment[1:])
        return points

    @classmethod
    def route_length(cls, points):
        """Total length of a polyline made of axis aligned segments."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += cls.manhattan(a, b)
        return total

    @staticmethod
    def next_step(x, y, tx, ty, speed):
        """Advance one tick from (x, y) towards (tx, ty).

        Horizontal movement comes first; leftover movement budget is used
        to turn the corner vertically.  The rider never overshoots the
        target.  A non-positive speed is treated as 1 so that a rider
        always makes progress (deadlock avoidance).
        """
        budget = int(speed) if speed and int(speed) > 0 else 1
        x, y, tx, ty = int(x), int(y), int(tx), int(ty)
        if x != tx:
            d = min(budget, abs(tx - x))
            x += d if tx > x else -d
            budget -= d
        if budget > 0 and y != ty:
            d = min(budget, abs(ty - y))
            y += d if ty > y else -d
        return x, y

"""Order data model."""

import globalvar
from path_planner import PathPlanner

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def _now():
    t = globalvar.get_value("time")
    return t if isinstance(t, int) else 0


class Order:
    """A delivery order from a restaurant to a customer.

    Coordinates are converted to the centre of their grid cell.

    Attributes:
        id: unique auto-incrementing order id.
        res_pos: (x, y) restaurant position (grid centre).
        cus_pos: (x, y) customer position (grid centre).
        status: "pending", "assigned", "picking_up", "delivering" or
            "completed".
        rider_id: id of the assigned rider or None.
        created_time / assigned_time / pickup_time / completed_time: ticks.
        distance: street distance restaurant -> customer.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.raw = [int(res_pos[0]), int(res_pos[1]),
                    int(cus_pos[0]), int(cus_pos[1])]
        self.res_pos = PathPlanner.to_grid_center(res_pos)
        self.cus_pos = PathPlanner.to_grid_center(cus_pos)
        self.status = STATUS_PENDING
        self.rider_id = None
        self.created_time = _now()
        self.assigned_time = None
        self.accepted_time = None
        self.pickup_time = None
        self.completed_time = None
        self.distance = PathPlanner.manhattan(self.res_pos, self.cus_pos)
        self.travel_distance = 0

    @classmethod
    def reset_counter(cls, value=1):
        """Reset the id counter (used when the system is reinitialised)."""
        cls._next_id = value

    def assign_to(self, rider_id):
        """Record that this order is assigned to ``rider_id``."""
        self.rider_id = rider_id
        self.status = STATUS_ASSIGNED
        self.assigned_time = _now()

    @property
    def time_taken(self):
        """Ticks from creation to completion, or None while open."""
        if self.completed_time is None:
            return None
        return self.completed_time - self.created_time

    @property
    def efficiency(self):
        """Delivery distance per tick taken (None while open)."""
        tt = self.time_taken
        if not tt:
            return None
        return round(self.distance / tt, 2)

    def is_completed(self):
        """True when the order has been delivered."""
        return self.status == STATUS_COMPLETED

    def to_list(self):
        """Return [restaurant_x, restaurant_y, customer_x, customer_y]."""
        return [self.res_pos[0], self.res_pos[1],
                self.cus_pos[0], self.cus_pos[1]]

    def to_dict(self):
        """Serialise the order to a JSON friendly dictionary."""
        return {
            "id": self.id,
            "raw": self.raw,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_time": self.created_time,
            "assigned_time": self.assigned_time,
            "accepted_time": self.accepted_time,
            "pickup_time": self.pickup_time,
            "completed_time": self.completed_time,
            "distance": self.distance,
            "travel_distance": self.travel_distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from :meth:`to_dict` output."""
        order = cls(data["res_pos"], data["cus_pos"])
        order.id = int(data["id"])
        order.raw = list(data.get("raw", order.raw))
        order.status = data.get("status", STATUS_PENDING)
        order.rider_id = data.get("rider_id")
        for key in ("created_time", "assigned_time", "accepted_time",
                    "pickup_time", "completed_time", "travel_distance"):
            if key in data:
                setattr(order, key, data[key])
        if order.created_time is None:
            order.created_time = 0
        return order

    def __repr__(self):
        return ("Order(id=%d, res=%s, cus=%s, status=%s, rider=%s)"
                % (self.id, self.res_pos, self.cus_pos, self.status,
                   self.rider_id))

"""Rider model: movement, state machine and event log."""

import logging

import config
import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``.  Each call
    to :meth:`move` advances the rider by one tick and performs the state
    transitions idle -> picking up -> delivering -> idle, recording each
    transition in ``rider.log``.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = rider_id
        self.x = int(config.get("rider_start_x") if x is None else x)
        self.y = int(config.get("rider_start_y") if y is None else y)
        self.speed = int(config.get("rider_speed") if speed is None else speed)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.distance_traveled = 0
        self.delivered_count = 0
        self._ticks = 0
        self._log_event("spawn", None)

    # ------------------------------------------------------------------ util
    def _now(self):
        t = globalvar.get_value("time")
        return t if isinstance(t, int) else self._ticks

    def _log_event(self, event, order_id):
        entry = {
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)

    @property
    def position(self):
        """Current (x, y) position."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def current_order(self):
        """The order currently being handled (or None)."""
        if self.state != IDLE and self.orders:
            return self.orders[0]
        return None

    def open_orders(self):
        """Orders on the rider that are not yet completed."""
        return [o for o in self.orders if getattr(o, "status", None) != "completed"]

    def is_idle(self):
        """True when the rider is idle and has no queued orders."""
        return self.state == IDLE and not self.open_orders()

    def remaining_stops(self):
        """Ordered list of remaining target positions for all orders."""
        stops = []
        for i, order in enumerate(self.open_orders()):
            if i == 0 and self.state == DELIVERING:
                stops.append(tuple(order.cus_pos))
            else:
                stops.append(tuple(order.res_pos))
                stops.append(tuple(order.cus_pos))
        return stops

    def planned_route(self):
        """Waypoints of the continuous route through all remaining stops."""
        return PathPlanner.plan_multi_route(self.position, self.remaining_stops())

    # -------------------------------------------------------------- movement
    def _step_towards(self, target):
        tx, ty = int(target[0]), int(target[1])
        self.special_case = PathPlanner.is_special_case(self.x, self.y, tx, ty)
        nx, ny = PathPlanner.next_step(self.x, self.y, tx, ty, self.speed)
        moved = abs(nx - self.x) + abs(ny - self.y)
        self.distance_traveled += moved
        order = self.current_order
        if order is not None and hasattr(order, "travel_distance"):
            order.travel_distance += moved
        self.x, self.y = nx, ny
        return (self.x, self.y) == (tx, ty)

    def _accept(self, order):
        self.state = PICKING_UP
        order.status = "picking_up"
        if getattr(order, "rider_id", None) is None:
            order.rider_id = self.rider_id
        if hasattr(order, "accepted_time"):
            order.accepted_time = self._now()
        self._log_event("accepted_order", order.id)

    def _deliver(self, order):
        order.status = "completed"
        order.completed_time = self._now()
        if order in self.orders:
            self.orders.remove(order)
        self.delivered_count += 1
        self.state = IDLE
        self.special_case = False
        self._log_event("delivered_to_customer", order.id)

    def move(self):
        """Advance the rider by one tick.

        * idle with queued orders -> accept the first order (picking up)
        * picking up -> move toward restaurant; on arrival -> delivering
        * delivering -> move toward customer; on arrival -> idle, order
          completed and removed from ``orders``.
        """
        self._ticks += 1
        # drop any orders already completed elsewhere (in place)
        self.orders[:] = [o for o in self.orders
                          if getattr(o, "status", None) != "completed"]

        if self.state not in (IDLE, PICKING_UP, DELIVERING):
            self.state = IDLE
        if self.state != IDLE and not self.orders:
            # inconsistent state: recover to idle
            self.state = IDLE
            self.special_case = False
            return
        if self.state == IDLE:
            if not self.orders:
                self.special_case = False
                return
            self._accept(self.orders[0])

        order = self.orders[0]
        if self.state == PICKING_UP:
            if self._step_towards(order.res_pos):
                self.state = DELIVERING
                order.status = "delivering"
                if hasattr(order, "pickup_time"):
                    order.pickup_time = self._now()
                self._log_event("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            if self._step_towards(order.cus_pos):
                self._deliver(order)

    # ----------------------------------------------------------- persistence
    def to_dict(self):
        """Serialise the rider to a JSON friendly dictionary."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "log": list(self.log),
            "distance_traveled": self.distance_traveled,
            "delivered_count": self.delivered_count,
        }

    @classmethod
    def from_dict(cls, data, orders_by_id=None):
        """Rebuild a rider from :meth:`to_dict` output."""
        rider = cls(data["rider_id"], data.get("x"), data.get("y"),
                    data.get("speed"))
        rider.state = int(data.get("state", IDLE))
        rider.special_case = bool(data.get("special_case", False))
        rider.log = list(data.get("log", rider.log))
        rider.distance_traveled = data.get("distance_traveled", 0)
        rider.delivered_count = data.get("delivered_count", 0)
        orders_by_id = orders_by_id or {}
        rider.orders = [orders_by_id[i] for i in data.get("orders", [])
                        if i in orders_by_id]
        return rider

    def __repr__(self):
        return ("Rider(id=%s, pos=(%d, %d), state=%s, orders=%d)"
                % (self.rider_id, self.x, self.y, self.state_name,
                   len(self.orders)))

"""Order management: creation, assignment and completion tracking."""

import logging

import globalvar
from order_model import Order
from path_planner import PathPlanner

logger = logging.getLogger(__name__)


def _rider_distance(rider, order):
    """Euclidean distance between a rider and the order's restaurant."""
    rx, ry = order.res_pos
    return ((rider.x - rx) ** 2 + (rider.y - ry) ** 2) ** 0.5


def _is_idle(rider):
    if hasattr(rider, "is_idle"):
        return rider.is_idle()
    return rider.state == 0 and not rider.orders


def find_best_rider_for_order(order, riders=None):
    """Return the best rider for ``order`` (closest rider first).

    Idle riders are preferred; if none are idle, the closest rider overall
    is chosen (ties broken by fewer assigned orders, then by id).
    Returns None when there are no riders.
    """
    if riders is None:
        riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    idle = [r for r in riders if _is_idle(r)]
    candidates = idle if idle else list(riders)
    return min(candidates,
               key=lambda r: (_rider_distance(r, order), len(r.orders),
                              r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion bookkeeping."""

    def __init__(self, system=None):
        self.system = system

    @staticmethod
    def all_orders():
        """List of every order created."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create an order from [rx, ry, cx, cy] and register it."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        self.all_orders().append(order)
        logger.info("Created %r", order)
        return order

    def pending_orders(self):
        """Orders not yet assigned to any rider."""
        return [o for o in self.all_orders() if o.status == "pending"]

    def completed_orders(self):
        """Orders that have been delivered."""
        return [o for o in self.all_orders() if o.status == "completed"]

    @staticmethod
    def assign_order(order, riders=None):
        """Assign ``order`` to the best rider; returns the rider or None."""
        rider = find_best_rider_for_order(order, riders)
        if rider is None:
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        logger.info("Order %d assigned to rider %s", order.id, rider.rider_id)
        return rider

    def assign_pending(self):
        """Try to assign every pending order; returns list of (order, rider)."""
        result = []
        for order in self.pending_orders():
            rider = self.assign_order(order)
            if rider is None:
                break
            result.append((order, rider))
        return result

    def dispatch(self, coords):
        """Full order intake: create, auto-recruit, assign.

        Returns a dict with keys ``order``, ``rider`` (or None),
        ``new_riders`` (list) and ``route`` (waypoints or []).
        """
        order = self.create_order(coords)
        new_riders = []
        if self.system is not None:
            new_riders = self.system.auto_recruit()
        rider = self.assign_order(order)
        route = rider.planned_route() if rider is not None else []
        return {"order": order, "rider": rider, "new_riders": new_riders,
                "route": route,
                "route_length": PathPlanner.route_length(route) if route else 0}

"""System manager: funds, riders, simulated time and persistence."""

import json
import logging

import config
import globalvar
from order import OrderManager
from order_model import Order
from rider import Rider, IDLE, PICKING_UP, DELIVERING

logger = logging.getLogger(__name__)


class SystemManager:
    """Core class managing funds, riders, time and system state.

    All state lives in :mod:`globalvar` so that other modules can access it.
    """

    def __init__(self):
        self.order_manager = OrderManager(self)

    # ------------------------------------------------------------ properties
    @property
    def funds(self):
        """Current system funds."""
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def riders(self):
        """List of all riders."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def time(self):
        """Current simulated tick."""
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        """Main loop running flag."""
        return bool(globalvar.get_value("running", False))

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    # ---------------------------------------------------------------- riders
    def idle_riders(self):
        """Riders that are idle with no queued orders."""
        return [r for r in self.riders if r.is_idle()]

    def get_rider(self, rider_id):
        """Return the rider with ``rider_id`` or None."""
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    def recruit_rider(self):
        """Recruit a rider if funds allow; returns the rider or None."""
        cost = int(config.get("rider_cost"))
        if self.funds < cost:
            return None
        self.funds = self.funds - cost
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        logger.info("Recruited rider %d, funds now %d", counter, self.funds)
        return rider

    def auto_recruit(self):
        """Recruit riders while funds >= cost and pending > idle riders."""
        new = []
        cost = int(config.get("rider_cost"))
        while (self.funds >= cost and
               len(self.order_manager.pending_orders()) > len(self.idle_riders())):
            rider = self.recruit_rider()
            if rider is None:
                break
            new.append(rider)
        return new

    # ------------------------------------------------------------ simulation
    def tick(self, n=1):
        """Advance the simulation ``n`` ticks; returns completed orders."""
        completed = []
        for _ in range(max(0, int(n))):
            globalvar.set_value("time", self.time + 1)
            self.auto_recruit()
            self.order_manager.assign_pending()
            for rider in list(self.riders):
                before = set(o.id for o in rider.orders)
                try:
                    rider.move()
                except Exception as exc:  # error recovery
                    logger.exception("Rider %s failed to move: %s",
                                     rider.rider_id, exc)
                    rider.state = IDLE
                after = set(o.id for o in rider.orders)
                for oid in before - after:
                    for o in self.order_manager.all_orders():
                        if o.id == oid and o.status == "completed":
                            completed.append(o)
            self.check_consistency()
        return completed

    def has_active_work(self):
        """True when any order is not yet completed."""
        return any(o.status != "completed"
                   for o in self.order_manager.all_orders())

    def run_until_idle(self, limit=None):
        """Tick until all orders are completed (or ``limit`` ticks)."""
        limit = int(limit or config.get("max_run_ticks"))
        ticks = 0
        while self.has_active_work() and ticks < limit:
            if not self.riders and self.funds < int(config.get("rider_cost")):
                break
            self.tick(1)
            ticks += 1
        return ticks

    def check_consistency(self):
        """Repair inconsistent state; returns a list of fixes applied."""
        fixes = []
        size = int(config.get("map_size"))
        rider_ids = {r.rider_id for r in self.riders}
        for r in self.riders:
            if r.state not in (IDLE, PICKING_UP, DELIVERING):
                r.state = IDLE
                fixes.append("rider %s invalid state reset" % r.rider_id)
            if r.state != IDLE and not r.orders:
                r.state = IDLE
                fixes.append("rider %s busy without orders reset" % r.rider_id)
            cx = max(0, min(size, r.x))
            cy = max(0, min(size, r.y))
            if (cx, cy) != (r.x, r.y):
                r.x, r.y = cx, cy
                fixes.append("rider %s position clamped" % r.rider_id)
        held = {id(o) for r in self.riders for o in r.orders}
        for o in self.order_manager.all_orders():
            if o.status in ("assigned", "picking_up", "delivering"):
                if o.rider_id not in rider_ids or id(o) not in held:
                    o.status = "pending"
                    o.rider_id = None
                    fixes.append("order %d returned to pending" % o.id)
        for fix in fixes:
            logger.warning("Consistency fix: %s", fix)
        return fixes

    # ----------------------------------------------------------- persistence
    def to_dict(self):
        """Serialise the whole system state."""
        return {
            "system": config.get_all(),
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.all_orders()],
        }

    def save(self, path=None):
        """Save the state to a JSON file; returns the path used."""
        path = path or config.get("state_file")
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, indent=2)
        return path

    def load(self, path=None):
        """Load state from a JSON file written by :meth:`save`."""
        path = path or config.get("state_file")
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        if isinstance(data.get("system"), dict):
            config.set_active(data["system"])
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        by_id = {o.id: o for o in orders}
        riders = [Rider.from_dict(d, by_id) for d in data.get("riders", [])]
        globalvar.set_value("funds", int(data.get("funds", self.funds)))
        globalvar.set_value("time", int(data.get("time", 0)))
        globalvar.set_value("rider_id_counter",
                            int(data.get("rider_id_counter", len(riders))))
        globalvar.set_value("orders", orders)
        globalvar.set_value("riders", riders)
        Order.reset_counter(max(by_id) + 1 if by_id else 1)
        self.check_consistency()
        return path


def init_system(config_path=None):
    """Initialise global state and return a :class:`SystemManager`.

    Funds = 1000, riders = [], rider id counter = 0, running = True,
    time = 0, orders = [].
    """
    cfg = config.load_config(config_path) if config_path else None
    config.set_active(cfg)
    globalvar._init()
    globalvar.set_value("funds", int(config.get("initial_funds")))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    Order.reset_counter(1)
    manager = SystemManager()
    globalvar.set_value("system", manager)
    return manager

"""Status displays and ASCII map visualisation."""

from path_planner import PathPlanner, MAP_SIZE
from rider import STATE_NAMES

HALF = 50
DIM = MAP_SIZE // HALF + 1  # 17 characters: streets on even indices


def _idx(v):
    return max(0, min(DIM - 1, int(round(v / HALF))))


class DisplayManager:
    """Builds text for status, order, rider-log and map displays."""

    @staticmethod
    def _base_grid():
        grid = []
        for row in range(DIM):
            line = []
            for col in range(DIM):
                line.append("#" if row % 2 == 0 or col % 2 == 0 else ".")
            grid.append(line)
        return grid

    @staticmethod
    def _draw_route(grid, points):
        for a, b in zip(points, points[1:]):
            ax, ay, bx, by = _idx(a[0]), _idx(a[1]), _idx(b[0]), _idx(b[1])
            step = 1 if bx >= ax else -1
            for c in range(ax, bx + step, step):
                grid[ay][c] = "*"
            step = 1 if by >= ay else -1
            for r in range(ay, by + step, step):
                grid[r][bx] = "*"

    def render_map(self, riders=(), orders=(), routes=()):
        """Render the 8x8 grid (streets '#', cells '.', route '*').

        'S' marks restaurants and 'C' customers of open orders, 'R' riders.
        """
        grid = self._base_grid()
        for route in routes:
            if route:
                self._draw_route(grid, route)
        for o in orders:
            if o.status == "completed":
                continue
            if o.status in ("pending", "assigned", "picking_up"):
                grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = "S"
            grid[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = "C"
        for r in riders:
            grid[_idx(r.y)][_idx(r.x)] = "R"
        lines = ["     " + " ".join(str((c * HALF) // 100 % 10) if c % 2 == 0
                                    else " " for c in range(DIM))]
        for row in range(DIM):
            label = "%3d " % (row * HALF) if row % 2 == 0 else "    "
            lines.append(label + " " + " ".join(grid[row]))
        lines.append("Legend: R=rider S=restaurant C=customer *=route "
                     "#=street .=empty (x axis in hundreds, y grows downward)")
        return "\n".join(lines)

    def format_dispatch(self, result):
        """Describe the outcome of a ``start`` command."""
        order = result["order"]
        lines = ["Order #%d created." % order.id,
                 "  Restaurant grid position: (%d, %d)" % order.res_pos,
                 "  Customer grid position:   (%d, %d)" % order.cus_pos]
        new = result.get("new_riders") or []
        if new:
            lines.append("  New rider added: yes (ID %s)"
                         % ", ".join(str(r.rider_id) for r in new))
        else:
            lines.append("  New rider added: no")
        rider = result.get("rider")
        if rider is None:
            lines.append("  No rider available - order is pending.")
            return "\n".join(lines)
        length = result.get("route_length", 0)
        speed = rider.speed if rider.speed > 0 else 1
        eta = -(-length // speed)
        lines.append("  Assigned rider: %d at (%d, %d)" % (rider.rider_id,
                                                           rider.x, rider.y))
        lines.append("  Route: " + " -> ".join("(%d,%d)" % p
                                               for p in result["route"]))
        lines.append("  Speed: %d units/tick, distance: %d units, "
                     "estimated time: %d ticks" % (speed, length, eta))
        lines.append(self.render_map([rider], [order] + rider.open_orders(),
                                     [result["route"]]))
        return "\n".join(lines)

    @staticmethod
    def format_status(system):
        """Snapshot of funds, time and every rider."""
        riders = system.riders
        lines = ["Time: %d   Funds: %d   Riders: %d"
                 % (system.time, system.funds, len(riders))]
        if not riders:
            lines.append("  (no riders)")
        for r in riders:
            lines.append("  Rider %d: pos=(%d, %d) state=%s orders=%d %s%s"
                         % (r.rider_id, r.x, r.y,
                            STATE_NAMES.get(r.state, "?"), len(r.orders),
                            [o.id for o in r.orders],
                            " [special case]" if r.special_case else ""))
        pending = system.order_manager.pending_orders()
        if pending:
            lines.append("  Pending (unassigned) orders: %s"
                         % [o.id for o in pending])
        return "\n".join(lines)

    @staticmethod
    def format_orders(orders):
        """Table with details of every order."""
        if not orders:
            return "No orders."
        header = ("%-4s %-7s %-12s %-12s %-6s %-6s %-6s %-11s"
                  % ("ID", "Start", "Restaurant", "Customer", "Dist",
                     "Taken", "Rider", "Status"))
        lines = [header, "-" * len(header)]
        for o in orders:
            taken = o.time_taken
            lines.append("%-4d %-7d %-12s %-12s %-6d %-6s %-6s %-11s"
                         % (o.id, o.created_time,
                            "(%d,%d)" % tuple(o.res_pos),
                            "(%d,%d)" % tuple(o.cus_pos), o.distance,
                            "-" if taken is None else taken,
                            "-" if o.rider_id is None else o.rider_id,
                            o.status))
        return "\n".join(lines)

    @staticmethod
    def format_rider_logs(riders):
        """Every recorded status change of every rider."""
        if not riders:
            return "No riders."
        lines = []
        for r in riders:
            lines.append("Rider %d (delivered %d, travelled %d units):"
                         % (r.rider_id, r.delivered_count, r.distance_traveled))
            for e in r.log:
                lines.append("  t=%-5s %-22s pos=(%s, %s) order=%s"
                             % (e.get("time"), e.get("event"), e.get("x"),
                                e.get("y"),
                                "-" if e.get("order_id") is None
                                else e.get("order_id")))
        return "\n".join(lines)

try:  # optional line editing / arrow-key history

SHORTCUTS = {

    "st": "status", "status": "status",

    "r": "riders", "riders": "riders",

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy] (s)  create an order; prompts if coordinates omitted
  status (st)              funds and rider states
  orders (o)               details of every order
  riders (r)               every status change of every rider
  map (m)                  ASCII map of riders and open orders
  tick [n] (t)             advance the simulation n ticks (default 1)
  run                      run until all orders are delivered
  save [file] / load [file] persist state as JSON
  history (hist)           show command history; !n repeats entry n, !! last
  help (h, ?)              this help
  quit / exit (q)          exit the program
Every command also advances the simulation by the configured ticks."""


class InputError(ValueError):
    """Raised for invalid user input."""


class InputManager:
    """Parses commands and validates coordinates."""

    def __init__(self, min_coord=0, max_coord=800):
        self.min_coord = min_coord
        self.max_coord = max_coord
        self.history = []

    def expand_history(self, line):
        """Expand ``!!`` and ``!n`` references to earlier commands."""
        line = line.strip()
        if line == "!!":
            if not self.history:
                raise InputError("No command history yet.")
            return self.history[-1]
        if line.startswith("!") and line[1:].isdigit():
            n = int(line[1:])
            if not 1 <= n <= len(self.history):
                raise InputError("No history entry %d." % n)
            return self.history[n - 1]
        return line

    def parse(self, line):
        """Return (command, args).  Unknown commands keep their raw name."""
        line = self.expand_history(line)
        tokens = line.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if not tokens:
            return "", []
        self.history.append(line)
        if len(self.history) > 1000:
            del self.history[:-1000]
        cmd = tokens[0].lower()
        return SHORTCUTS.get(cmd, cmd), tokens[1:]

    def validate_coordinates(self, tokens):
        """Validate four integer coordinates in range; return list[int]."""
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ").replace("]", " ").split()
        tokens = list(tokens)
        if len(tokens) != 4:
            raise InputError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y (got %d)." % len(tokens))
        values = []
        for tok in tokens:
            try:
                v = int(str(tok).strip())
            except ValueError:
                raise InputError("'%s' is not an integer." % tok) from None
            if not self.min_coord <= v <= self.max_coord:
                raise InputError("Coordinate %d out of range %d-%d."
                                 % (v, self.min_coord, self.max_coord))
            values.append(v)
        return values

    @staticmethod
    def parse_positive_int(tokens, default=1):
        """Parse an optional positive integer argument."""
        if not tokens:
            return default
        try:
            v = int(tokens[0])
        except ValueError:
            raise InputError("'%s' is not an integer." % tokens[0]) from None
        if v <= 0:
            raise InputError("Value must be positive.")
        return v

    def format_history(self):
        """Numbered command history."""
        if not self.history:
            return "No history."
        return "\n".join("%4d  %s" % (i + 1, c) for i, c in enumerate(self.history))

"""Command-line entry point of the food delivery dispatch simulator.

Run with:  python src/main.py [--config FILE] [--load FILE] [--debug]
"""

import argparse
import logging
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import config  # noqa: E402
import globalvar  # noqa: E402
from display import DisplayManager  # noqa: E402
from input_manager import InputManager, InputError, HELP_TEXT  # noqa: E402
from system import init_system  # noqa: E402

logger = logging.getLogger("dispatch")


def handle_command(system, im, dm, cmd, args, read=input):
    """Execute one command; returns the text to print."""
    if cmd == "start":
        if not args:
            raw = read("Enter restaurant_x restaurant_y customer_x customer_y: ")
            args = raw.replace(",", " ").replace("[", " ").replace("]", " ").split()
        coords = im.validate_coordinates(args)
        result = system.order_manager.dispatch(coords)
        return dm.format_dispatch(result)
    if cmd == "status":
        return dm.format_status(system)
    if cmd == "orders":
        return dm.format_orders(system.order_manager.all_orders())
    if cmd == "riders":
        return dm.format_rider_logs(system.riders)
    if cmd == "map":
        routes = [r.planned_route() for r in system.riders if r.orders]
        return dm.render_map(system.riders, system.order_manager.all_orders(),
                             routes)
    if cmd == "tick":
        n = im.parse_positive_int(args, 1)
        done = system.tick(n)
        msg = "Advanced %d tick(s). Time is now %d." % (n, system.time)
        if done:
            msg += " Delivered orders: %s" % [o.id for o in done]
        return msg
    if cmd == "run":
        ticks = system.run_until_idle()
        return "Ran %d tick(s). Time is now %d.\n%s" % (
            ticks, system.time, dm.format_status(system))
    if cmd == "save":
        return "State saved to %s" % system.save(args[0] if args else None)
    if cmd == "load":
        return "State loaded from %s" % system.load(args[0] if args else None)
    if cmd == "history":
        return im.format_history()
    if cmd == "help":
        return HELP_TEXT
    if cmd == "quit":
        system.stop()
        return "Goodbye."
    raise InputError("Unknown command '%s'. Type 'help' for commands." % cmd)


def main(argv=None):
    """Run the interactive main loop."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", help="JSON configuration file")
    parser.add_argument("--load", help="load a saved JSON state file")
    parser.add_argument("--debug", action="store_true", help="debug logging")
    opts = parser.parse_args(argv)

    logging.basicConfig(level=logging.DEBUG if opts.debug else logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")
    system = init_system(opts.config)
    if opts.load:
        try:
            system.load(opts.load)
        except (OSError, ValueError, KeyError) as exc:
            print("Could not load state: %s" % exc)

    im = InputManager(0, int(config.get("map_size")))
    dm = DisplayManager()
    print("Food Delivery Dispatch System - funds %d. Type 'help' for commands."
          % system.funds)

    while globalvar.get_value("running", False):
        try:
            line = input("dispatch> ")
        except (EOFError, KeyboardInterrupt):
            print()
            system.stop()
            break
        try:
            cmd, args = im.parse(line)
            if not cmd:
                continue
            output = handle_command(system, im, dm, cmd, args)
            print(output)
            if cmd not in ("quit", "help", "history"):
                system.tick(int(config.get("ticks_per_command")))
        except InputError as exc:
            print("Error: %s" % exc)
        except (EOFError, KeyboardInterrupt):
            print("\nInput cancelled.")
        except Exception as exc:  # graceful recovery
            logger.exception("Unexpected error")
            print("Unexpected error: %s (system state checked and recovered)" % exc)
            system.check_consistency()
    return 0


if __name__ == "__main__":
    sys.exit(main())

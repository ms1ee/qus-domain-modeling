"""Grid based path planning for riders.

The map is an 800x800 coordinate plane divided into 100x100 cells.  Lines
at multiples of 100 are treated as streets; riders prefer to travel along
them, making right-angle turns where necessary.
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = MAP_SIZE // CELL_SIZE


class GridMap:
    """Description of the map geometry."""

    def __init__(self, width=MAP_SIZE, height=MAP_SIZE, cell_size=CELL_SIZE):
        self.width = width
        self.height = height
        self.cell_size = cell_size
        self.grid_dimension = width // cell_size

    def clamp(self, value):
        """Clamp a coordinate to the map range."""
        return max(0, min(self.width, int(value)))

    def to_grid_corner(self, x, y):
        """Convert raw coordinates to the top-left corner of their cell."""
        x = self.clamp(x)
        y = self.clamp(y)
        return (x // self.cell_size) * self.cell_size, \
            (y // self.cell_size) * self.cell_size

    def to_grid_center(self, x, y):
        """Convert raw coordinates to the center of their cell."""
        x = min(self.clamp(x), self.width - 1)
        y = min(self.clamp(y), self.height - 1)
        return (x // self.cell_size) * self.cell_size + self.cell_size // 2, \
            (y // self.cell_size) * self.cell_size + self.cell_size // 2

    def is_street(self, x, y):
        """Return True if the point lies on a street line."""
        return x % self.cell_size == 0 or y % self.cell_size == 0


def manhattan(a, b):
    """Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def euclidean(a, b):
    """Euclidean distance between two points."""
    return ((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2) ** 0.5


class Route:
    """A sequence of waypoints with an estimated travel time."""

    def __init__(self, points, speed=20):
        self.points = list(points)
        self.speed = speed
        self.length = sum(manhattan(self.points[i], self.points[i + 1])
                          for i in range(len(self.points) - 1))
        if speed > 0:
            self.estimated_time = -(-self.length // speed)
        else:
            self.estimated_time = 0

    def extend(self, other):
        """Append another route (skipping the duplicated joint point)."""
        pts = list(other.points)
        if self.points and pts and tuple(self.points[-1]) == tuple(pts[0]):
            pts = pts[1:]
        return Route(self.points + pts, self.speed)


class PathPlanner:
    """Plans right-angle paths preferring streets."""

    def __init__(self, grid_map=None):
        self.grid_map = grid_map or GridMap()

    def plan_points(self, start, end):
        """Return a list of waypoints (including start and end).

        Returns a tuple ``(points, special_case)`` where ``special_case`` is
        True when the rider is directly above/below or beside the target.
        """
        sx, sy = int(start[0]), int(start[1])
        ex, ey = int(end[0]), int(end[1])
        if (sx, sy) == (ex, ey):
            return [(sx, sy)], False
        if sx == ex or sy == ey:
            # Directly above/below (or on the same row): straight line.
            return [(sx, sy), (ex, ey)], True
        cs = self.grid_map.cell_size
        # Prefer the first leg that lies on a street.
        if sy % cs == 0:
            corner = (ex, sy)          # horizontal first along street
        elif sx % cs == 0:
            corner = (sx, ey)          # vertical first along street
        elif ey % cs == 0:
            corner = (sx, ey)          # second leg horizontal on street
        else:
            corner = (ex, sy)
        return [(sx, sy), corner, (ex, ey)], False

    def plan_route(self, start, end, speed=20):
        """Plan a Route from start to end."""
        points, _ = self.plan_points(start, end)
        return Route(points, speed)

    def plan_multi(self, start, destinations, speed=20):
        """Plan a continuous route through several destinations."""
        route = Route([tuple(start)], speed)
        current = tuple(start)
        for dest in destinations:
            route = route.extend(self.plan_route(current, dest, speed))
            current = tuple(dest)
        return route

    @staticmethod
    def advance_along(x, y, waypoints, budget):
        """Move from (x, y) along ``waypoints`` using at most ``budget`` units.

        ``waypoints`` is consumed in place as points are reached.
        Returns the new position and the distance travelled.
        """
        travelled = 0
        while waypoints and budget > 0:
            tx, ty = waypoints[0]
            dist = abs(tx - x) + abs(ty - y)
            if dist == 0:
                waypoints.pop(0)
                continue
            step = min(budget, dist)
            if tx != x:
                dx = min(step, abs(tx - x))
                x += dx if tx > x else -dx
                step_used = dx
            else:
                dy = min(step, abs(ty - y))
                y += dy if ty > y else -dy
                step_used = dy
            budget -= step_used
            travelled += step_used
            if (x, y) == (tx, ty):
                waypoints.pop(0)
        return x, y, travelled

"""Rider model: movement, state machine and status history."""

import globalvar
from path_planner import PathPlanner


class RiderState:
    """Rider status enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2

    NAMES = {0: "idle", 1: "picking up", 2: "delivering"}

    @classmethod
    def name(cls, state):
        """Human readable state name."""
        return cls.NAMES.get(state, "unknown")


DEFAULT_SPEED = 20
DEFAULT_START = (400, 400)

_planner = PathPlanner()


def _current_time(fallback):
    value = globalvar.get_value("current_tick")
    return fallback if value is None else value


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``.  Calling
    :meth:`move` advances the rider by one simulation tick.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        cfg = globalvar.get_value("config") or {}
        self.rider_id = rider_id
        start = cfg.get("rider_start", DEFAULT_START)
        self.x = int(start[0]) if x is None else int(x)
        self.y = int(start[1]) if y is None else int(y)
        self.speed = int(cfg.get("rider_speed", DEFAULT_SPEED)) \
            if speed is None else int(speed)
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.waypoints = []
        self.total_distance = 0
        self._ticks = 0
        self.log = []
        self._record("spawn", None)

    # ------------------------------------------------------------------
    def _record(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": _current_time(self._ticks),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def active_orders(self):
        """Return unique, not yet completed orders."""
        seen = set()
        result = []
        for order in self.orders:
            if id(order) in seen:
                continue
            seen.add(id(order))
            if getattr(order, "status", None) != "completed":
                result.append(order)
        return result

    def is_idle(self):
        """True when the rider has nothing to do."""
        return self.state == RiderState.IDLE and not self.active_orders()

    def target(self):
        """Current target coordinates, or None."""
        order = self.current_order
        if order is None:
            return None
        if self.state == RiderState.PICKING_UP:
            return tuple(order.res_pos)
        if self.state == RiderState.DELIVERING:
            return tuple(order.cus_pos)
        return None

    def _plan(self):
        tgt = self.target()
        if tgt is None:
            self.waypoints = []
            return
        points, special = _planner.plan_points((self.x, self.y), tgt)
        self.special_case = special
        self.waypoints = points[1:]

    def _start_next_order(self):
        pending = self.active_orders()
        if not pending:
            self.current_order = None
            self.state = RiderState.IDLE
            self.waypoints = []
            return False
        order = pending[0]
        self.current_order = order
        if getattr(order, "rider_id", None) is None:
            order.rider_id = self.rider_id
        if getattr(order, "status", None) == "pending":
            order.status = "assigned"
        if getattr(order, "accept_time", None) is None:
            try:
                order.accept_time = _current_time(self._ticks)
            except AttributeError:
                pass
        self.state = RiderState.PICKING_UP
        self._record("accepted_order", order.id)
        self._plan()
        return True

    def move(self):
        """Advance the rider by one tick, handling state transitions."""
        self._ticks += 1
        if self.current_order is None or \
                getattr(self.current_order, "status", None) == "completed":
            self.current_order = None
            if not self._start_next_order():
                return
        if not self.waypoints and (self.x, self.y) != self.target():
            self._plan()
        self.x, self.y, travelled = PathPlanner.advance_along(
            self.x, self.y, self.waypoints, self.speed)
        self.total_distance += travelled
        if (self.x, self.y) != self.target():
            return
        order = self.current_order
        if self.state == RiderState.PICKING_UP:
            self.state = RiderState.DELIVERING
            order.status = "picked_up"
            self._record("arrived_at_merchant", order.id)
            self._plan()
        elif self.state == RiderState.DELIVERING:
            now = _current_time(self._ticks)
            if hasattr(order, "complete"):
                order.complete(now)
            else:
                order.status = "completed"
            self._record("delivered_to_customer", order.id)
            self.current_order = None
            self.waypoints = []
            self.special_case = False
            if not self._start_next_order():
                self.state = RiderState.IDLE

    # ------------------------------------------------------------------
    def to_dict(self):
        """Serialise rider to a dictionary."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [o.id for o in self.orders],
            "current_order_id": self.current_order.id
            if self.current_order else None,
            "waypoints": [list(p) for p in self.waypoints],
            "total_distance": self.total_distance,
            "ticks": self._ticks,
            "log": list(self.log),
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        """Rebuild a rider from :meth:`to_dict` output."""
        rider = cls.__new__(cls)
        rider.rider_id = data["rider_id"]
        rider.x = data["x"]
        rider.y = data["y"]
        rider.speed = data["speed"]
        rider.state = data["state"]
        rider.special_case = data.get("special_case", False)
        rider.orders = [orders_by_id[i] for i in data.get("order_ids", [])
                        if i in orders_by_id]
        cid = data.get("current_order_id")
        rider.current_order = orders_by_id.get(cid) if cid is not None \
            else None
        rider.waypoints = [tuple(p) for p in data.get("waypoints", [])]
        rider.total_distance = data.get("total_distance", 0)
        rider._ticks = data.get("ticks", 0)
        rider.log = list(data.get("log", []))
        return rider

    def __repr__(self):
        return "Rider(id=%s, pos=(%s, %s), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, RiderState.name(self.state),
            len(self.orders))

"""Order data model."""

import globalvar
from path_planner import GridMap, manhattan

_grid = GridMap()


def _next_order_id():
    counter = globalvar.get_value("order_id_counter", 0) or 0
    counter += 1
    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A delivery order from a restaurant to a customer.

    Coordinates are converted to the top-left corner of their grid cell.
    Status values: "pending", "assigned", "picked_up", "completed".
    """

    def __init__(self, res_pos, cus_pos):
        self.id = _next_order_id()
        self.raw_res_pos = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus_pos = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = _grid.to_grid_corner(*self.raw_res_pos)
        self.cus_pos = _grid.to_grid_corner(*self.raw_cus_pos)
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("current_tick", 0) or 0
        self.accept_time = None
        self.completion_time = None
        self.distance = manhattan(self.res_pos, self.cus_pos)
        self.time_taken = None
        self.efficiency = None
        orders = globalvar.get_value("orders")
        if isinstance(orders, list):
            orders.append(self)

    @property
    def completed(self):
        """True once the order was delivered."""
        return self.status == "completed"

    def assign_to(self, rider_id):
        """Assign the order to the rider with ``rider_id``.

        The order is appended to that rider's order list (if the rider is
        registered and does not already hold this order).
        """
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        for rider in globalvar.get_value("riders") or []:
            if getattr(rider, "rider_id", None) == rider_id:
                if not any(o is self for o in rider.orders):
                    rider.orders.append(self)
                break

    def complete(self, time):
        """Mark the order completed at ``time``."""
        self.status = "completed"
        self.completion_time = time
        self.time_taken = time - self.start_time
        if self.time_taken and self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)

    def to_dict(self):
        """Serialise to a dict."""
        return {
            "id": self.id,
            "raw_res_pos": list(self.raw_res_pos),
            "raw_cus_pos": list(self.raw_cus_pos),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "accept_time": self.accept_time,
            "completion_time": self.completion_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
            "efficiency": self.efficiency,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild from :meth:`to_dict` output (not re-registered)."""
        order = cls.__new__(cls)
        order.id = data["id"]
        order.raw_res_pos = tuple(data.get("raw_res_pos", data["res_pos"]))
        order.raw_cus_pos = tuple(data.get("raw_cus_pos", data["cus_pos"]))
        order.res_pos = tuple(data["res_pos"])
        order.cus_pos = tuple(data["cus_pos"])
        order.status = data["status"]
        order.rider_id = data.get("rider_id")
        order.start_time = data.get("start_time", 0)
        order.accept_time = data.get("accept_time")
        order.completion_time = data.get("completion_time")
        order.distance = data.get("distance", 0)
        order.time_taken = data.get("time_taken")
        order.efficiency = data.get("efficiency")
        return order

    def __repr__(self):
        return "Order(id=%s, res=%s, cus=%s, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

"""System initialisation and resource management."""

import json
import logging
import os

import globalvar
from rider import Rider, RiderState

logger = logging.getLogger("dispatch")

DEFAULT_CONFIG = {
    "funds": 1000,
    "recruitment_cost": 500,
    "recruitment_threshold": 300,
    "rider_speed": 20,
    "rider_start": [400, 400],
}

CONFIG_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                           "config.json")


def load_config(path=CONFIG_FILE):
    """Load configuration overrides from a JSON file if present."""
    cfg = dict(DEFAULT_CONFIG)
    try:
        if path and os.path.exists(path):
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                cfg.update({k: v for k, v in data.items()
                            if k in DEFAULT_CONFIG})
    except (OSError, ValueError) as exc:
        logger.warning("Could not load config %s: %s", path, exc)
    return cfg


def init_system(config=None):
    """Initialise all global system parameters."""
    globalvar._init()
    cfg = dict(DEFAULT_CONFIG)
    if config:
        cfg.update(config)
    globalvar.set_value("config", cfg)
    globalvar.set_value("funds", int(cfg["funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("current_tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)


def _ensure_init():
    if globalvar.get_value("riders") is None:
        riders = []
        globalvar.set_value("riders", riders)
    if globalvar.get_value("funds") is None:
        globalvar.set_value("funds", DEFAULT_CONFIG["funds"])
    if globalvar.get_value("rider_id_counter") is None:
        globalvar.set_value("rider_id_counter", 0)
    if globalvar.get_value("config") is None:
        globalvar.set_value("config", dict(DEFAULT_CONFIG))


def recruit_rider():
    """Create a new rider, deducting the recruitment cost."""
    _ensure_init()
    cfg = globalvar.get_value("config")
    riders = globalvar.get_value("riders")
    counter = globalvar.get_value("rider_id_counter") + 1
    used = {r.rider_id for r in riders}
    while counter in used:
        counter += 1
    globalvar.set_value("rider_id_counter", counter)
    rider = Rider(counter)
    riders.append(rider)
    globalvar.set_value("funds", globalvar.get_value("funds")
                        - int(cfg.get("recruitment_cost", 500)))
    logger.info("Recruited rider %s", counter)
    return rider


def pending_order_count():
    """Number of orders not yet assigned to a rider."""
    return sum(1 for o in globalvar.get_value("orders") or []
               if o.status == "pending")


def idle_rider_count():
    """Number of riders that are idle."""
    return sum(1 for r in globalvar.get_value("riders") or [] if r.is_idle())


def should_recruit():
    """Check the automatic recruitment trigger."""
    _ensure_init()
    cfg = globalvar.get_value("config")
    return (globalvar.get_value("funds")
            >= int(cfg.get("recruitment_threshold", 300))
            and pending_order_count() > idle_rider_count())


def check_and_recruit():
    """Recruit a rider if the trigger condition holds; return it or None."""
    if should_recruit():
        return recruit_rider()
    return None


def advance_tick(n=1):
    """Advance the simulation by ``n`` ticks, moving all riders."""
    for _ in range(max(0, int(n))):
        globalvar.set_value("current_tick",
                            (globalvar.get_value("current_tick") or 0) + 1)
        for rider in list(globalvar.get_value("riders") or []):
            try:
                rider.move()
            except Exception as exc:  # recover from unexpected errors
                logger.exception("Rider %s failed to move: %s",
                                 rider.rider_id, exc)
                rider.waypoints = []


class SystemManager:
    """Facade managing funds, riders and system state."""

    def __init__(self, config=None, init=True):
        if init:
            init_system(config)

    @property
    def funds(self):
        return globalvar.get_value("funds")

    @property
    def riders(self):
        return globalvar.get_value("riders")

    @property
    def orders(self):
        return globalvar.get_value("orders")

    @property
    def running(self):
        return globalvar.get_value("running")

    @running.setter
    def running(self, value):
        globalvar.set_value("running", bool(value))

    @property
    def current_tick(self):
        return globalvar.get_value("current_tick")

    def recruit_rider(self):
        return recruit_rider()

    def check_and_recruit(self):
        return check_and_recruit()

    def tick(self, n=1):
        advance_tick(n)

    def check_consistency(self):
        """Repair inconsistent state; return a list of fixed issues."""
        issues = []
        for rider in self.riders:
            if rider.state not in (RiderState.IDLE, RiderState.PICKING_UP,
                                   RiderState.DELIVERING):
                rider.state = RiderState.IDLE
                issues.append("rider %s had invalid state" % rider.rider_id)
            if rider.state == RiderState.IDLE and rider.current_order:
                rider.current_order = None
                issues.append("rider %s idle with order" % rider.rider_id)
        return issues

"""Order management: creation, assignment and completion."""

import globalvar
import system
from order_model import Order
from path_planner import euclidean


def find_best_rider_for_order(order):
    """Return the best rider for ``order``.

    Riders with the fewest active orders are preferred; ties are broken by
    the straight-line distance to the restaurant, then by rider id.
    If no rider exists, one is recruited when funds allow.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        cfg = globalvar.get_value("config") or system.DEFAULT_CONFIG
        funds = globalvar.get_value("funds")
        if funds is None or funds >= int(cfg.get("recruitment_threshold",
                                                 300)):
            system.recruit_rider()
        riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    res = tuple(order.res_pos)
    return min(riders, key=lambda r: (len(r.active_orders()),
                                      euclidean((r.x, r.y), res),
                                      r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion queries."""

    def create_order(self, res_pos, cus_pos):
        """Create and assign an order.

        Returns a dict with keys ``order``, ``rider`` and ``recruited``
        (the new Rider or None).
        """
        order = Order(res_pos, cus_pos)
        recruited = system.check_and_recruit()
        rider = find_best_rider_for_order(order)
        if rider is not None:
            order.assign_to(rider.rider_id)
            if not any(o is order for o in rider.orders):
                rider.orders.append(order)
        return {"order": order, "rider": rider, "recruited": recruited}

    @staticmethod
    def all_orders():

"""Status displays and ASCII map visualisation."""

import globalvar
from rider import RiderState
from path_planner import MAP_SIZE

MAP_CHARS = 16


class DisplayManager:
    """Formats system state into text."""

    def __init__(self, map_size=MAP_CHARS):
        self.map_size = map_size
        self.unit = MAP_SIZE // map_size

    def _idx(self, v):
        return max(0, min(self.map_size - 1, int(v) // self.unit))

    def render_map(self, route_points=None, highlight_orders=None):
        """Return the ASCII map as a string."""
        n = self.map_size
        grid = []
        for r in range(n):
            row = []
            for c in range(n):
                row.append("#" if r % 2 == 0 or c % 2 == 0 else ".")
            grid.append(row)
        if route_points and len(route_points) > 1:
            for (x1, y1), (x2, y2) in zip(route_points, route_points[1:]):
                steps = max(abs(x2 - x1), abs(y2 - y1)) // self.unit + 1
                for i in range(steps + 1):
                    t = i / steps if steps else 0
                    x = x1 + (x2 - x1) * t
                    y = y1 + (y2 - y1) * t
                    grid[self._idx(y)][self._idx(x)] = "*"
        orders = highlight_orders
        if orders is None:
            orders = [o for o in globalvar.get_value("orders") or []
                      if not o.completed]
        for o in orders:
            grid[self._idx(o.res_pos[1])][self._idx(o.res_pos[0])] = "S"
            grid[self._idx(o.cus_pos[1])][self._idx(o.cus_pos[0])] = "C"
        for rider in globalvar.get_value("riders") or []:
            grid[self._idx(rider.y)][self._idx(rider.x)] = "R"
        header = "    " + "".join("%-2d" % (c % 100) if c % 2 == 0 else ""
                                  for c in range(n))
        lines = [header.rstrip()]
        for r in range(n):
            lines.append("%4d " % (r * self.unit) + " ".join(grid[r]))
        lines.append("Legend: R=rider S=restaurant C=customer *=route "
                     "#=street .=empty (1 char = %d units)" % self.unit)
        return "\n".join(lines)

    def status_text(self):

        riders = globalvar.get_value("riders") or []
        lines = ["Tick: %s" % globalvar.get_value("current_tick", 0),

                 "Riders: %d" % len(riders)]
        for r in riders:

            lines.append("  Rider %d at (%d, %d) speed=%d state=%s "

                             RiderState.name(r.state),

    def orders_text(self):

        lines = ["%-4s %-6s %-12s %-12s %-8s %-6s %-6s %-10s" % (

"""Command parsing and input validation."""

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "q": "quit", "h": "help", "?": "help", "m": "map", "t": "tick",
    "hist": "history",
}

COORD_MIN = 0
COORD_MAX = 1000


class InputManager:
    """Parses commands and validates coordinates; keeps a history."""

    def __init__(self):
        self.command_history = []
        try:
            import readline  # noqa: F401  (enables arrow-key history)
            self._readline = readline
        except ImportError:
            self._readline = None

    def parse_command(self, line):
        """Return (command, args) with shortcuts expanded."""
        line = (line or "").strip()
        if not line:
            return "", []
        self.command_history.append(line)
        parts = line.replace(",", " ").replace("[", " ") \
            .replace("]", " ").split()
        cmd = parts[0].lower()
        return SHORTCUTS.get(cmd, cmd), parts[1:]

    @staticmethod
    def validate_coordinates(tokens):
        """Validate four integer coordinates; raise ValueError if invalid."""
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ") \
                .replace("]", " ").split()
        tokens = list(tokens)
        if len(tokens) != 4:
            raise ValueError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y")
        values = []
        for t in tokens:
            try:
                v = int(str(t))
            except ValueError:
                raise ValueError("'%s' is not an integer" % t)
            if not COORD_MIN <= v <= COORD_MAX:
                raise ValueError("Coordinate %d out of range %d-%d"
                                 % (v, COORD_MIN, COORD_MAX))
            values.append(v)
        return values

"""Command-line entry point for the food delivery dispatch simulator."""

import logging
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import globalvar  # noqa: E402
import system  # noqa: E402
from display import DisplayManager  # noqa: E402
from input_manager import InputManager  # noqa: E402
from order import OrderManager  # noqa: E402
from path_planner import PathPlanner  # noqa: E402
import persistence  # noqa: E402

HELP = """Commands:
  start [rx ry cx cy] (s)  create an order (restaurant + customer coords)
  status (st)              system funds and rider status
  orders (o)               all order details
  riders (r)               rider status change history
  map (m)                  show the ASCII map
  tick [n] (t)             advance simulation n ticks (default 1)
  save [file] / load [file] persist state as JSON (default state.json)
  history (hist)           show command history
  help (h)                 this help
  quit / exit (q)          exit
Each command also advances the simulation by one tick."""


def _route_for(rider, order, planner):
    """Planned route from the rider's queue end through the order."""
    start = (rider.x, rider.y)
    others = [o for o in rider.active_orders() if o is not order]
    if others:
        start = tuple(others[-1].cus_pos)
    return planner.plan_multi(start, [order.res_pos, order.cus_pos],
                              rider.speed)


def handle_start(args, im, om, dm, planner):
    if not args:
        raw = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
        args = raw.replace(",", " ").replace("[", " ").replace("]", " ") \
            .split()
    try:
        rx, ry, cx, cy = im.validate_coordinates(args)
    except ValueError as exc:
        print("Error: %s" % exc)
        return
    result = om.create_order((rx, ry), (cx, cy))
    order, rider = result["order"], result["rider"]
    print("Order %d created." % order.id)
    print("Restaurant grid coordinates: (%d, %d)" % order.res_pos)
    print("Customer grid coordinates:   (%d, %d)" % order.cus_pos)
    if result["recruited"]:
        print("New rider added: Rider %d (funds now %s)" % (
            result["recruited"].rider_id, globalvar.get_value("funds")))
    else:
        print("No new rider added.")
    if rider is None:
        print("No rider available; order is pending.")
        return
    route = _route_for(rider, order, planner)
    print("Assigned to Rider %d (speed %d units/tick)" % (
        rider.rider_id, rider.speed))
    print("Route: " + " -> ".join("(%d,%d)" % p for p in route.points))
    print("Distance: %d units, estimated time: %d ticks" % (
        route.length, route.estimated_time))
    print(dm.render_map(route.points, [order]))


def main():
    logging.basicConfig(level=logging.WARNING,
                        format="%(levelname)s %(message)s")
    system.init_system(system.load_config())
    im = InputManager()
    om = OrderManager()
    dm = DisplayManager()
    planner = PathPlanner()
    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        try:
            cmd, args = im.parse_command(line)
            if not cmd:
                continue
            advance = True
            if cmd == "start":
                handle_start(args, im, om, dm, planner)
            elif cmd == "status":
                print(dm.status_text())
            elif cmd == "orders":
                print(dm.orders_text())
            elif cmd == "riders":
                print(dm.riders_text())
            elif cmd == "map":
                print(dm.render_map())
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                system.advance_tick(n)
                advance = False
                print("Advanced to tick %s" % globalvar.get_value(
                    "current_tick"))
            elif cmd == "save":
                persistence.save_state(args[0] if args else "state.json")
                print("Saved.")
            elif cmd == "load":
                persistence.load_state(args[0] if args else "state.json")
                print("Loaded.")
            elif cmd == "history":
                for i, h in enumerate(im.command_history, 1):
                    print("%3d  %s" % (i, h))
            elif cmd == "help":
                print(HELP)
            elif cmd in ("quit", "exit"):
                globalvar.set_value("running", False)
                advance = False
            else:
                print("Unknown command '%s'. Type 'help'." % cmd)
                advance = False
            if advance:
                system.advance_tick(1)
        except Exception as exc:  # graceful recovery
            print("Error: %s" % exc)
            system.SystemManager(init=False).check_consistency()
    print("Goodbye.")


if __name__ == "__main__":
    main()

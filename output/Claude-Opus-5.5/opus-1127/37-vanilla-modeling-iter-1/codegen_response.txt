from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import List, Optional

CHENG_RU_GROUP = "Cheng Ru Group"
SPECIAL_DRAWER = "Li Bin"
MONITORED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"


def normalize_name(text: Optional[str]) -> str:
    """Normalize a name for comparison (case/space insensitive)."""
    if text is None:
        return ""
    return re.sub(r"\s+", "", str(text)).lower()


def is_cheng_ru_group(department: Optional[str]) -> bool:
    """Return True if the department refers to the Cheng Ru group."""
    norm = normalize_name(department)
    return norm in ("chengrugroup", "chengru")


@dataclass
class User:
    """A participant of the lottery."""

    name: str
    department: str

    def __post_init__(self) -> None:
        self.name = str(self.name).strip()
        self.department = str(self.department).strip()

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


_LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
    ("participation", 50),
    ("consolation", 50),
]


def infer_prize_level(name: str) -> int:
    """Infer a prize level from its name (smaller value = higher prize)."""
    low = (name or "").lower()
    for keyword, level in _LEVEL_KEYWORDS:
        if keyword in low:
            return level
    return 100


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None
    level: Optional[int] = None

    def __post_init__(self) -> None:
        self.name = str(self.name).strip()
        try:
            self.quantity = int(self.quantity)
        except (TypeError, ValueError):
            self.quantity = 0
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None
        if self.level is None:
            self.level = infer_prize_level(self.name)

    def __str__(self) -> str:
        parts = [f"{self.name} x{self.quantity}"]
        if self.avoid_groups:
            parts.append(f"avoid: {', '.join(self.avoid_groups)}")
        if self.associated_drawer:
            parts.append(f"drawer: {self.associated_drawer}")
        return " | ".join(parts)

    ("Qin Zhu", CHENG_RU_GROUP),
    ("You Ju", CHENG_RU_GROUP),
    ("Xu Feng", "Operations Department"),
    ("He Yu", "Operations Department"),

class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.load_mode: str = ""
        self.source_file_path: str = ""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        self.load_mode = "default"
        self.source_file_path = ""
        return [User(n, d) for n, d in DEFAULT_LIST], []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        self.load_mode = "custom"
        self.source_file_path = str(path)
        if not path or not os.path.isfile(str(path)):
            return [], [f"File Error: file not found '{path}'"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        try:
            with open(str(path), "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]

        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users or []:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = OrderedDict()
        for user in users or []:
            info[user.department] = info.get(user.department, 0) + 1
        return dict(info)

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Manages the prize library and draw order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self.prizes}
        for name, qty in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            self.prizes.append(Prize(name, qty, [], None))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if name is None or not str(name).strip():
            return False
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            return False
        if qty <= 0:
            return False
        prize = Prize(str(name).strip(), qty, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

    def find_prize(self, name: str) -> Optional[Prize]:
        for p in self.prizes:
            if p.name == name:
                return p
        return None

"""Lottery rule engine and draw execution."""

from __future__ import annotations

import random
from typing import List, Optional

from lottery_system.models import (
    FALLBACK_TARGET_DRAWER,
    MONITORED_DRAWERS,
    SPECIAL_DRAWER,
    Prize,
    User,
    is_cheng_ru_group,
    normalize_name,
)


class LotteryManager:
    """Applies the rule engine and performs draws."""

    def __init__(self, users, result_manager=None) -> None:
        self.users: List[User] = list(users or [])
        self.result_manager = result_manager
        self.winner_names = set()
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []
        self._rng = random.SystemRandom()

    # ----- rule helpers -----
    @staticmethod
    def _is_drawer(prize: Prize, target: str) -> bool:
        return normalize_name(prize.associated_drawer) == normalize_name(target)

    def is_fallback_active(self, prize: Prize) -> bool:
        if not self._is_drawer(prize, FALLBACK_TARGET_DRAWER):
            return False
        return len(self.last_two_draw_results) >= 2 and not any(self.last_two_draw_results[-2:])

    def describe_rule(self, prize: Prize) -> str:
        if self.is_fallback_active(prize):
            return ("Fallback mechanism active (Cheng Ru group protection): "
                    "prize pool limited to Cheng Ru group members, excluding previous winners")
        if self._is_drawer(prize, SPECIAL_DRAWER):
            return (f"Special drawer {SPECIAL_DRAWER}: only previous winners are excluded, "
                    "avoidance groups not applied")
        desc = "Excludes previous winners"
        if prize.avoid_groups:
            desc += f"; avoids groups: {', '.join(prize.avoid_groups)}"
        return desc

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        not_won = [u for u in self.users if u.name not in self.winner_names]
        if self.is_fallback_active(prize):
            return [u for u in not_won if is_cheng_ru_group(u.department)]
        if self._is_drawer(prize, SPECIAL_DRAWER):
            return not_won
        avoid = {normalize_name(g) for g in (prize.avoid_groups or [])}
        return [u for u in not_won if normalize_name(u.department) not in avoid]

    # ----- drawing -----
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.winner_names.add(user.name)
        self.winners.append(user)
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return user

    def finish_round(self, prize: Prize, round_winners) -> None:
        """Update fallback tracking after a round is finished."""
        drawer = normalize_name(prize.associated_drawer)
        if drawer and drawer in {normalize_name(d) for d in MONITORED_DRAWERS}:
            has_cr = any(is_cheng_ru_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(has_cr)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self._is_drawer(prize, FALLBACK_TARGET_DRAWER):
            self.last_two_draw_results = []

    def draw(self, prize: Prize) -> List[User]:
        """Non-interactive draw of a full round."""
        result: List[User] = []
        for _ in range(prize.quantity):
            user = self.draw_one(prize)
            if user is None:
                break
            result.append(user)
        self.finish_round(prize, result)
        return result

"""Result storage, summarizing and export."""

from __future__ import annotations

import os
from datetime import datetime
from typing import Dict, List, Tuple

from lottery_system.models import Prize, User


class Winner:
    """A winning record."""

    def __init__(self, user: User, prize: Prize, order: int, win_time: datetime = None) -> None:
        self.user = user
        self.prize = prize
        self.order = order
        self.win_time = win_time or datetime.now()

    @property
    def prize_name(self) -> str:
        return self.prize.name if isinstance(self.prize, Prize) else str(self.prize)

    def __repr__(self) -> str:
        return f"Winner({self.user.name}, {self.user.department}, {self.prize_name})"


class ResultManager:
    """Maintains the winner list."""

    def __init__(self) -> None:
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> Winner:
        winner = Winner(user, prize, len(self.winners) + 1)
        self.winners.append(winner)
        return winner

    @staticmethod
    def _sorted_prizes(prizes) -> List[Prize]:
        indexed = list(enumerate(prizes or []))
        indexed.sort(key=lambda t: (getattr(t[1], "level", 100) or 0, t[0]))
        return [p for _, p in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for prize in self._sorted_prizes(prizes):
            result.setdefault(prize.name, [])
        for w in sorted(self.winners, key=lambda w: (w.win_time, w.order)):
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8", newline="") as fh:
                fh.write("Name,Department,Prize Name,Winning Time\n")
                for _, lst in self.get_winners_by_prize(prizes).items():
                    for w in lst:
                        fh.write(
                            f"{w.user.name},{w.user.department},{w.prize_name},"
                            f"{w.win_time.strftime('%Y-%m-%d %H:%M:%S')}\n"
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from __future__ import annotations

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryCLI:
    def __init__(self) -> None:
        self.status = "idle"
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_manager = None

    def _input(self, prompt: str) -> str:
        try:
            return input(prompt).strip()
        except EOFError:
            return "0"

    # ----- list -----
    def _set_users(self, users):
        self.users = users
        self.result_manager = ResultManager()
        self.lottery_manager = None
        print(f"{len(users)} valid participants loaded.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        valid, _ = self.list_manager.validate_users(users)
        for e in errors:
            print(e)
        self._set_users(valid)

    def load_custom(self):
        path = self._input("Enter txt file path: ")
        users, errors = self.list_manager.load_custom_list(path)
        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            print("Validation issues found:")
            for e in errors:
                print("  " + e)
            for d in duplicates:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            if not valid:
                print("No valid data. Upload terminated.")
                return
            choice = self._input("Enter 'r' to remove invalid data and continue, 't' to terminate upload: ").lower()
            if choice != "r":
                print("Upload terminated.")
                return
            print("Invalid data removed.")
        if not valid:
            print("No valid data loaded.")
            return
        self._set_users(valid)

    def view_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        print("Department groups:")
        for dept, cnt in info.items():
            print(f"  {dept}: {cnt} people")

    # ----- prizes -----
    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")

    def add_prize(self):
        name = self._input("Prize name: ")
        qty = self._input("Quantity: ")
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = self._input("Avoidance groups (numbers or names, comma separated, blank for none): ")
        avoid = []
        for tok in raw.replace("，", ",").split(","):
            tok = tok.strip()
            if not tok:
                continue
            if tok.isdigit() and 1 <= int(tok) <= len(groups):
                avoid.append(groups[int(tok) - 1])
            else:
                avoid.append(tok)
        drawer = self._input("Associated lottery person (blank for none): ") or None
        if self.prize_manager.add_prize(name, qty, avoid, drawer):
            print("Prize added.")
        else:
            print("Invalid prize (name must be non-empty and quantity a positive integer).")

    def reorder(self):
        self.view_prizes()
        raw = self._input("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        print("Order updated." if self.prize_manager.reorder_prizes(order) else "Invalid order.")

    # ----- lottery -----
    def _control(self) -> str:
        """Return 'draw' or 'terminate'."""
        while True:
            cmd = self._input("[Enter]=draw, 'pause', 'terminate': ").lower()
            if cmd in ("", "draw", "continue"):
                return "draw"
            if cmd == "terminate":
                return "terminate"
            if cmd == "pause":
                print("Lottery paused. Type 'continue' to resume or 'terminate' to skip this round.")
                while True:
                    c = self._input("> ").lower()
                    if c == "continue":
                        print("Resumed.")
                        return "draw"
                    if c == "terminate":
                        return "terminate"
                    print("Paused. Type 'continue' or 'terminate'.")
            print("Unknown command.")

    def start_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery_manager is None:
            self.lottery_manager = LotteryManager(self.users, self.result_manager)
        lm = self.lottery_manager
        self.status = "running"
        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 50)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            if prize.associated_drawer:
                print(f"Lottery person: {prize.associated_drawer}")
            print(f"Rule: {lm.describe_rule(prize)}")
            round_winners = []
            for i in range(1, prize.quantity + 1):
                if self._control() == "terminate":
                    print("Current round terminated.")
                    break
                print(f"Currently drawing the {ordinal(i)} winner...")
                user = lm.draw_one(prize)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            lm.finish_round(prize, round_winners)
            print(f"Round {rnd} results:")
            if not round_winners:
                print("  (no winners)")
            for j, u in enumerate(round_winners, 1):
                print(f"  {j}. {u.name} ({u.department}) - {prize.name}")
        self.status = "finished"
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for pname, winners in data.items():
            if not winners:
                continue
            print(f"[{pname}]")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {pname} - "
                      f"{w.win_time.strftime('%H:%M:%S')}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": lambda: [print(e) for e in self.prize_manager.load_default_prizes()] and None,
            "5": self.add_prize, "6": self.view_prizes, "7": self.reorder,
            "8": self.start_lottery, "9": self.show_summary, "10": self.export,
        }
        while True:
            print("\n===== Team Building Smart Lottery =====")
            print("1. Load default list      2. Upload custom list    3. View groups")
            print("4. Load preset prizes     5. Add custom prize      6. View prizes")
            print("7. Reorder prizes         8. Start lottery         9. View results")
            print("10. Export results        0. Exit")
            choice = self._input("Choose: ")
            if choice == "0":
                print("Bye.")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid choice.")


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

"""Data loading utilities for the Beijing Subway fare calculation system.

This module locates and reads ``subway_data.json`` and converts the raw JSON
content into the graph data structures used by the path-finding module.

The JSON format is::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str,
                      "distance": int (meters), "line": str}]}
"""

import json
import os

from models import Edge, Graph

# Absolute path of the bundled data file; valid from any working directory.
DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


def load_subway_data(path=DATA_FILE_PATH) -> dict:
    """Load and return the JSON content of the subway data file.

    Raises:
        FileNotFoundError: if the file does not exist.
        ValueError: if the file is not valid JSON or has the wrong shape.
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        data = json.load(fh)
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    return data


def build_graph(data):
    """Build the weighted undirected graph from raw subway data.

    Returns:
        (graph, station_to_number, number_to_station)
        ``station_to_number`` maps station names to integer numbers;
        ``number_to_station`` is the reverse mapping.
    """
    graph = Graph()
    station_to_number = {}
    number_to_station = {}

    def register(name, lines=None):
        if name not in station_to_number:
            number = len(station_to_number)
            station_to_number[name] = number
            number_to_station[number] = name
            graph.add_vertex(number)
            graph.set_station_lines(name, list(lines or []))
        elif lines:
            merged = list(graph.station_lines.get(name, []))
            for line in lines:
                if line not in merged:
                    merged.append(line)
            graph.set_station_lines(name, merged)
        return station_to_number[name]

    for station in data.get("stations", []) or []:
        name = station.get("name")
        if not name:
            continue
        register(name, station.get("lines", []) or [])

    for conn in data.get("connections", []) or []:
        src_name = conn.get("source")
        dst_name = conn.get("dest")
        if not src_name or not dst_name:
            continue
        line = conn.get("line", "")
        src = register(src_name, [line] if line else None)
        dst = register(dst_name, [line] if line else None)
        try:
            distance = int(conn.get("distance", 0))
        except (TypeError, ValueError):
            continue
        if distance < 0:
            continue
        graph.add_edge(Edge(src, dst, distance, line))

    return graph, station_to_number, number_to_station

"""Core data structures: stations, lines, connections, edges and the graph."""

from dataclasses import dataclass, field
from typing import List


@dataclass
class Station:
    """A subway station."""
    name: str
    number: int
    lines: List[str] = field(default_factory=list)

    @property
    def is_transfer_station(self) -> bool:
        return len(self.lines) > 1


@dataclass
class Line:
    """A subway line with its ordered station sequence."""
    name: str
    stations: List[str] = field(default_factory=list)


@dataclass
class Connection:
    """A bidirectional connection between two adjacent stations."""
    source: str
    dest: str
    distance: int
    line: str
    bidirectional: bool = True


@dataclass
class TransferDetail:
    """A transfer made at a station between two lines."""
    station_name: str
    from_line: str
    to_line: str


class Edge:
    """A weighted edge between two station numbers on a given line."""

    def __init__(self, source, dest, distance, line):
        if distance is None or distance < 0:
            raise ValueError("Edge distance must be a non-negative number")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        return self.source

    def other(self, vertex):
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError("Vertex is not an endpoint of this edge")

    def reversed(self):
        """Return the same connection traversed in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.distance, self.line) == (
            other.source, other.dest, other.distance, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.distance, self.line))

    def __repr__(self):
        return "Edge({!r}, {!r}, {!r}, {!r})".format(
            self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Station numbers are assigned by the caller. Every undirected edge is
    stored once in each endpoint's adjacency list, oriented away from that
    endpoint (so ``edge.source`` is always the current vertex).
    """

    def __init__(self):
        self.adj = {}             # station number -> list[Edge]
        self.station_lines = {}   # station name -> list[str]
        self._edges = []

    # ----- vertices -------------------------------------------------------
    def add_vertex(self, v):
        self.adj.setdefault(v, [])

    def vertices(self):
        return list(self.adj.keys())

    @property
    def V(self):
        return len(self.adj)

    vertex_count = V

    # ----- station info ---------------------------------------------------
    def set_station_lines(self, station_name, lines):
        unique = []
        for line in lines or []:
            if line not in unique:
                unique.append(line)
        self.station_lines[station_name] = unique

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    # ----- edges ----------------------------------------------------------
    def add_edge(self, edge):
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.source != edge.dest:
            self.adj[edge.dest].append(edge.reversed())
        self._edges.append(edge)

    def adjacent(self, v):
        return list(self.adj.get(v, []))

    def edges(self):
        return list(self._edges)

    @property
    def E(self):
        return len(self._edges)

    edge_count = E

    @property
    def graph(self):
        """The adjacency-list dictionary representation."""
        return self.adj

    def __repr__(self):
        return "Graph(V={}, E={})".format(self.V, self.E)

"""Path calculation: priority queue and Dijkstra-based shortest path search.

The search runs over states ``(station, arrival line[, transfers])`` so that
transfers at multi-line interchange stations are handled exactly. Costs are
tuples compared lexicographically, which lets the same algorithm serve the
three query modes:

* ``shortest_distance`` – minimise distance, then transfers
* ``least_transfer``    – minimise transfers, then distance
* ``comprehensive``     – minimise distance + transfer penalty
"""

import math

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance (meters) of one transfer in comprehensive mode
# (~5 minutes of transfer time at ~1.5 min/km).
TRANSFER_PENALTY_METERS = 3500
# Weight multiplier applied to non-preferred lines when a preference is set.
NON_PREFERRED_LINE_FACTOR = 1.3


class MinPQ:
    """Binary-heap based minimum priority queue.

    Items are stored with a key; ties are broken by insertion order so the
    items themselves never need to be comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = 0

    def insert(self, key, item=None):
        entry = (key, self._counter, item)
        self._counter += 1
        self._heap.append(entry)
        self._swim(len(self._heap) - 1)

    push = insert

    def min(self):
        if not self._heap:
            raise IndexError("Priority queue underflow")
        key, _, item = self._heap[0]
        return key, item

    def del_min(self):
        if not self._heap:
            raise IndexError("Priority queue underflow")
        top = self._heap[0]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._sink(0)
        return top[0], top[2]

    pop = del_min

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def _swim(self, k):
        heap = self._heap
        while k > 0:
            parent = (k - 1) // 2
            if heap[k] < heap[parent]:
                heap[k], heap[parent] = heap[parent], heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        heap = self._heap
        n = len(heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and heap[right] < heap[left]:
                child = right
            if heap[child] < heap[k]:
                heap[k], heap[child] = heap[child], heap[k]
                k = child
            else:
                break


class PathfindingConfig:
    """Configuration for a path query."""

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None, transfer_penalty=TRANSFER_PENALTY_METERS,
                 non_preferred_factor=NON_PREFERRED_LINE_FACTOR):
        self.mode = mode
        self.max_transfers = max_transfers
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.preferred_lines = list(preferred_lines)
        self.transfer_penalty = transfer_penalty
        self.non_preferred_factor = non_preferred_factor

    def __repr__(self):
        return ("PathfindingConfig(mode={!r}, max_transfers={!r}, "
                "preferred_lines={!r})").format(
                    self.mode, self.max_transfers, self.preferred_lines)


class ShortestPath:
    """Dijkstra shortest path from a single source station number."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        mode = self.config.mode or MODE_SHORTEST_DISTANCE
        if mode not in VALID_MODES:
            raise ValueError("Unknown path query mode: {!r}".format(mode))
        self._mode = mode
        max_t = self.config.max_transfers
        if max_t is not None and max_t < 0:
            raise ValueError("max_transfers must be non-negative")
        self._max_transfers = max_t
        self._preferred = set(self.config.preferred_lines or [])

        self.dist_to = {}       # station number -> actual distance (meters)
        self._best_state = {}   # station number -> best state
        self._parent = {}       # state -> (previous state, Edge)
        self._state_info = {}   # state -> (effective dist, transfers, real dist)
        self._run()

    # ----- internal -------------------------------------------------------
    def _make_state(self, v, line, transfers):
        if self._max_transfers is not None:
            return (v, line, transfers)
        return (v, line, -1)

    def _cost(self, eff, transfers, real):
        if self._mode == MODE_LEAST_TRANSFER:
            return (transfers, eff, real)
        if self._mode == MODE_COMPREHENSIVE:
            return (eff + self.config.transfer_penalty * transfers, transfers, real)
        return (eff, transfers, real)

    def _edge_factor(self, line):
        if self._preferred and line not in self._preferred:
            return self.config.non_preferred_factor
        return 1

    def _run(self):
        pq = MinPQ()
        best_cost = {}
        settled = set()

        start = self._make_state(self.source, None, 0)
        self._state_info[start] = (0, 0, 0)
        start_cost = self._cost(0, 0, 0)
        best_cost[start] = start_cost
        pq.insert(start_cost, start)
        adj = self.graph.adj

        while not pq.is_empty():
            cost, state = pq.del_min()
            if state in settled or cost != best_cost.get(state):
                continue
            settled.add(state)
            v, line = state[0], state[1]
            eff, transfers, real = self._state_info[state]
            if v not in self._best_state:
                self._best_state[v] = state
                self.dist_to[v] = real

            for edge in adj.get(v, ()):
                new_t = transfers + (1 if line is not None and edge.line != line else 0)
                if self._max_transfers is not None and new_t > self._max_transfers:
                    continue
                new_state = self._make_state(edge.dest, edge.line, new_t)
                if new_state in settled:
                    continue
                new_eff = eff + edge.weight * self._edge_factor(edge.line)
                new_real = real + edge.weight
                new_cost = self._cost(new_eff, new_t, new_real)
                old = best_cost.get(new_state)
                if old is None or new_cost < old:
                    best_cost[new_state] = new_cost
                    self._state_info[new_state] = (new_eff, new_t, new_real)
                    self._parent[new_state] = (state, edge)
                    pq.insert(new_cost, new_state)

    # ----- queries --------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return (station numbers, edges) from source to target.

        Returns ([], []) when the target is unreachable.
        """
        state = self._best_state.get(target)
        if state is None:
            return [], []
        edges = []
        while state in self._parent:
            prev, edge = self._parent[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        path = [self.source] + [e.dest for e in edges]
        return path, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                count += 1
        return count

    def get_lines(self, target):
        _, edges = self.path_with_edges(target)
        lines = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)
        return lines

    def get_transfer_points(self, target):
        """List of (station number, from_line, to_line) for each transfer."""
        path, edges = self.path_with_edges(target)
        points = []
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                points.append((path[i], edges[i - 1].line, edges[i].line))
        return points

"""Fare calculation according to the Beijing Subway fare structure.

Tiers (left-closed, right-open intervals, distance in meters):
    [0, 6000)       -> 3 yuan
    [6000, 12000)   -> 4 yuan
    [12000, 22000)  -> 5 yuan
    [22000, 32000)  -> 6 yuan
    >= 32000        -> 7 yuan + 1 yuan per additional full 20 km
"""

import math
import numbers
from dataclasses import dataclass


class FareCalculationError(ValueError):
    """Raised when a fare cannot be calculated."""


@dataclass(frozen=True)
class FareTier:
    min_distance: int
    max_distance: int
    fare: int
    tier_type: str


FARE_TIERS = [
    FareTier(0, 6000, 3, "basic"),
    FareTier(6000, 12000, 4, "progressive"),
    FareTier(12000, 22000, 5, "progressive"),
    FareTier(22000, 32000, 6, "progressive"),
]

EXTENDED_THRESHOLD = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_INCREMENT_DISTANCE = 20000
EXTENDED_INCREMENT_FARE = 1

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, numbers.Real):
        raise FareCalculationError("Distance must be a number, got {!r}".format(distance))
    if math.isnan(distance) or math.isinf(distance):
        raise FareCalculationError("Distance must be a finite number")
    if distance < 0:
        raise FareCalculationError("Distance cannot be negative")


def Sum_money(distance) -> int:
    """Return the fare (yuan) for a distance given in meters.

    Fractional distances are rounded up to the next whole meter.
    """
    _validate_distance(distance)
    meters = math.ceil(distance)
    for tier in FARE_TIERS:
        if tier.min_distance <= meters < tier.max_distance:
            return tier.fare
    extra = (meters - EXTENDED_THRESHOLD) // EXTENDED_INCREMENT_DISTANCE
    return int(EXTENDED_BASE_FARE + extra * EXTENDED_INCREMENT_FARE)


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated travel time in minutes (1.5 min/km, 5 min per transfer)."""
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, numbers.Real) \
            or transfer_count < 0:
        raise FareCalculationError("Transfer count must be a non-negative number")
    return float(distance / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER)


class FareCalculator:
    """Object-oriented wrapper around the fare functions."""

    tiers = FARE_TIERS

    def calculate(self, distance) -> int:
        return Sum_money(distance)

    def estimated_time(self, distance, transfer_count) -> float:
        return calculate_estimated_time(distance, transfer_count)

"""Data management: network model, updates and integrity checks."""

import copy
import json
from collections import defaultdict, deque

from data_loader import DATA_FILE_PATH, build_graph, load_subway_data
from models import Station, Line, Connection


def normalize_name(name):
    """Normalise a station name for lenient matching."""
    if name is None:
        return ""
    text = str(name).strip().casefold()
    for suffix in (" station", "站"):
        if text.endswith(suffix) and len(text) > len(suffix):
            text = text[: -len(suffix)]
    for ch in " '’‘`-_·.":
        text = text.replace(ch, "")
    return text


class SubwayNetwork:
    """The Beijing Subway network: stations, lines, connections and graph."""

    def __init__(self, data_source_file=DATA_FILE_PATH, data=None):
        self.data_source_file = data_source_file
        if data is None:
            data = load_subway_data(data_source_file)
        self.data = copy.deepcopy(data)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])
        self.rebuild()

    # ----- building -------------------------------------------------------
    def rebuild(self):
        self.graph, self.station_to_number, self.number_to_station = build_graph(self.data)
        self._lookup = {}
        for name in self.station_to_number:
            key = normalize_name(name)
            self._lookup.setdefault(key, name)

    # ----- queries --------------------------------------------------------
    @property
    def stations(self):
        return [Station(name, num, self.graph.get_station_lines(name))
                for name, num in self.station_to_number.items()]

    @property
    def connections(self):
        return [Connection(c["source"], c["dest"], c["distance"], c.get("line", ""))
                for c in self.data["connections"]]

    def station_exists(self, name):
        return name in self.station_to_number

    def resolve_station(self, name):
        """Return the official station name matching ``name`` or None."""
        if name is None:
            return None
        stripped = str(name).strip()
        if stripped in self.station_to_number:
            return stripped
        return self._lookup.get(normalize_name(stripped))

    def get_station_lines(self, name):
        return self.graph.get_station_lines(name)

    def is_transfer_station(self, name):
        return self.graph.is_transfer_station(name)

    def line_names(self):
        names = []
        for lines in self.graph.station_lines.values():
            for line in lines:
                if line not in names:
                    names.append(line)
        for c in self.data["connections"]:
            line = c.get("line")
            if line and line not in names:
                names.append(line)
        return names

    def resolve_line(self, name):
        target = normalize_name(name)
        for line in self.line_names():
            if line == name or normalize_name(line) == target:
                return line
        return None

    def line_stations(self, line):
        """Ordered station sequence of a line, derived from connections."""
        adj = defaultdict(list)
        for c in self.data["connections"]:
            if c.get("line") == line:
                adj[c["source"]].append(c["dest"])
                adj[c["dest"]].append(c["source"])
        if not adj:
            return [n for n, ls in self.graph.station_lines.items() if line in ls]
        ends = [s for s in adj if len(set(adj[s])) == 1]
        start = ends[0] if ends else next(iter(adj))
        order, visited, stack = [], set(), [start]
        while stack:
            v = stack.pop()
            if v in visited:
                continue
            visited.add(v)
            order.append(v)
            for n in reversed(adj[v]):
                if n not in visited:
                    stack.append(n)
        return order

    @property
    def lines(self):
        return [Line(name, self.line_stations(name)) for name in self.line_names()]

    def transfer_stations(self):
        return [n for n in self.station_to_number if self.graph.is_transfer_station(n)]

    # ----- updates --------------------------------------------------------
    def _station_record(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def _ensure_station(self, name, line=None):
        rec = self._station_record(name)
        if rec is None:
            rec = {"name": name, "lines": []}
            self.data["stations"].append(rec)
        rec.setdefault("lines", [])
        if line and line not in rec["lines"]:
            rec["lines"].append(line)
        return rec

    def add_line(self, line_name, stations, distances):
        """Add a new line given its ordered stations and segment distances."""
        if not line_name:
            raise ValueError("Line name cannot be empty")
        if len(stations) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(stations) - 1:
            raise ValueError("Number of distances must equal number of stations - 1")
        for d in distances:
            if not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for name in stations:
            self._ensure_station(name, line_name)
        for i, d in enumerate(distances):
            self.data["connections"].append({
                "source": stations[i], "dest": stations[i + 1],
                "distance": int(d), "line": line_name})
        self.rebuild()

    def add_connection(self, source, dest, distance, line):
        if distance <= 0:
            raise ValueError("Distance must be positive")
        self._ensure_station(source, line)
        self._ensure_station(dest, line)
        self.data["connections"].append(
            {"source": source, "dest": dest, "distance": int(distance), "line": line})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        rec = self._station_record(name)
        if rec is None:
            raise KeyError("Station '{}' does not exist".format(name))
        if new_name and new_name != name:
            if self._station_record(new_name) is not None:
                raise ValueError("Station '{}' already exists".format(new_name))
            rec["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            rec["lines"] = list(lines)
        self.rebuild()

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance of the connection(s) between two stations."""
        if not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        updated = 0
        for c in self.data["connections"]:
            pair = {c.get("source"), c.get("dest")}
            if pair == {source, dest} and (line is None or c.get("line") == line):
                c["distance"] = int(distance)
                updated += 1
        if not updated:
            raise KeyError("No connection between '{}' and '{}'".format(source, dest))
        self.rebuild()
        return updated

    def save(self, path=None):
        target = path or self.data_source_file
        with open(target, "w", encoding="utf-8") as fh:
            json.dump(self.data, fh, ensure_ascii=False, indent=2)


# Main lines with name aliases and a representative station (with aliases).

class DataIntegrityChecker:
    """Checks station coverage, connection integrity and distance accuracy."""

    def __init__(self, network):
        self.network = network
        self.station_coverage_valid = False
        self.connection_integrity_valid = False
        self.distance_accuracy_valid = False
        self.errors = []
        self.warnings = []
        self.info = []

    def check_station_coverage(self):
        net = self.network
        line_keys = {normalize_name(l): l for l in net.line_names()}
        missing_lines = []
        for main, (aliases, rep_aliases) in MAIN_LINES.items():
            found = None
            for alias in aliases:
                if normalize_name(alias) in line_keys:
                    found = line_keys[normalize_name(alias)]
                    break
            if found is None:
                missing_lines.append(main)
                continue
            count = len(net.line_stations(found))
            self.info.append("{}: {} stations".format(found, count))
            if not any(net.resolve_station(a) for a in rep_aliases):
                self.warnings.append("Representative station '{}' of {} not found".format(
                    rep_aliases[0], main))
        if missing_lines:
            self.errors.append("Main lines missing: " + ", ".join(missing_lines))
        if not net.station_to_number:
            self.errors.append("No stations in data")
        self.station_coverage_valid = not missing_lines and bool(net.station_to_number)
        return self.station_coverage_valid

    def check_connection_integrity(self):
        net = self.network
        valid = True
        declared = {st.get("name") for st in net.data["stations"]}
        connected = set()
        for i, c in enumerate(net.data["connections"]):
            s, d, line = c.get("source"), c.get("dest"), c.get("line")
            if not s or not d:
                self.errors.append("Connection #{} lacks source/dest".format(i))
                valid = False
                continue
            for name in (s, d):
                if name not in declared:
                    self.errors.append("Connection #{} references undeclared station '{}'"
                                       .format(i, name))
                    valid = False
                elif line and line not in (net._station_record(name) or {}).get("lines", []):
                    self.warnings.append("Station '{}' not listed on line '{}'".format(name, line))
            if s == d:
                self.errors.append("Connection #{} is a self-loop at '{}'".format(i, s))
                valid = False
            connected.update((s, d))
        for name in declared:
            if name not in connected:
                self.warnings.append("Station '{}' has no connections".format(name))
        # connectivity
        graph = net.graph
        verts = graph.vertices()
        if verts:
            seen = {verts[0]}
            queue = deque([verts[0]])
            while queue:
                v = queue.popleft()
                for e in graph.adj.get(v, ()):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        queue.append(e.dest)
            unreachable = [net.number_to_station[v] for v in verts if v not in seen]
            if unreachable:
                self.warnings.append("{} station(s) unreachable from '{}'".format(
                    len(unreachable), net.number_to_station[verts[0]]))
        self.connection_integrity_valid = valid
        return valid

    def check_distance_accuracy(self):
        valid = True
        seen = {}
        for i, c in enumerate(self.network.data["connections"]):
            dist = c.get("distance")
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                self.errors.append("Connection #{} has invalid distance {!r}".format(i, dist))
                valid = False
                continue
            if not MIN_PLAUSIBLE_DISTANCE <= dist <= MAX_PLAUSIBLE_DISTANCE:
                self.warnings.append("Connection {} - {} distance {} m looks implausible"
                                     .format(c.get("source"), c.get("dest"), dist))
            key = (frozenset((c.get("source"), c.get("dest"))), c.get("line"))
            if key in seen and seen[key] != dist:
                self.errors.append("Inconsistent distances for {} - {} on {}: {} vs {}".format(
                    c.get("source"), c.get("dest"), c.get("line"), seen[key], dist))
                valid = False
            seen.setdefault(key, dist)
        self.distance_accuracy_valid = valid
        return valid

    def run(self):
        self.errors, self.warnings, self.info = [], [], []
        self.check_station_coverage()
        self.check_connection_integrity()
        self.check_distance_accuracy()
        return {
            "station_coverage_valid": self.station_coverage_valid,
            "connection_integrity_valid": self.connection_integrity_valid,
            "distance_accuracy_valid": self.distance_accuracy_valid,
            "valid": self.is_valid(),
            "errors": list(self.errors),
            "warnings": list(self.warnings),
            "info": list(self.info),
        }

    def is_valid(self):
        return (self.station_coverage_valid and self.connection_integrity_valid
                and self.distance_accuracy_valid)

    def format_report(self):
        net = self.network
        ok = lambda b: "PASS" if b else "FAIL"
        lines = [
            "===== Data Integrity Check =====",
            "Stations: {}  Connections: {}  Lines: {}".format(
                len(net.station_to_number), len(net.data["connections"]),
                len(net.line_names())),
            "Station coverage:      " + ok(self.station_coverage_valid),
            "Connection integrity:  " + ok(self.connection_integrity_valid),
            "Distance accuracy:     " + ok(self.distance_accuracy_valid),
        ]
        for msg in self.info:
            lines.append("  [INFO] " + msg)
        for msg in self.errors:
            lines.append("  [ERROR] " + msg)
        for msg in self.warnings:
            lines.append("  [WARNING] " + msg)
        lines.append("Overall: " + ("Data integrity check passed"
                                    if self.is_valid() else "Data integrity check found problems"))
        return "\n".join(lines)

"""Query validation and route planning (combines path search and fares)."""

from fare_calculator import Sum_money, calculate_estimated_time, FareCalculationError
from pathfinding import PathfindingConfig, ShortestPath

FORMAT_ERROR_MSG = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class QueryValidationError(ValueError):
    """Raised when user input is invalid."""


class PathCalculationError(RuntimeError):
    """Raised when no route can be calculated."""


def split_query(text):
    """Split 'Start; End' input into two stripped parts."""
    if text is None:
        raise QueryValidationError(FORMAT_ERROR_MSG)
    normalized = text.replace("；", ";")
    parts = normalized.split(";")
    if len(parts) != 2:
        raise QueryValidationError(FORMAT_ERROR_MSG)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryValidationError(FORMAT_ERROR_MSG)
    return start, end


class RoutePlanner:
    """Validates queries and computes route results."""

    def __init__(self, network, config=None):
        self.network = network
        self.config = config if config is not None else PathfindingConfig()

    def parse_query(self, text):
        """Validate input text and return official (start, end) names."""
        start_raw, end_raw = split_query(text)
        start = self.network.resolve_station(start_raw)
        if start is None:
            raise QueryValidationError(
                "Starting station '{}' does not exist, please check the station name"
                .format(start_raw))
        end = self.network.resolve_station(end_raw)
        if end is None:
            raise QueryValidationError(
                "Ending station '{}' does not exist, please check the station name"
                .format(end_raw))
        if start == end:
            raise QueryValidationError("Starting station cannot be the same as the ending station")
        return start, end

    def plan(self, start, end):
        """Compute the route result dictionary between two official names."""
        net = self.network
        s_num = net.station_to_number[start]
        e_num = net.station_to_number[end]
        try:
            sp = ShortestPath(net.graph, s_num, self.config)
            path, edges = sp.path_with_edges(e_num)
        except (ValueError, KeyError, IndexError) as exc:
            raise PathCalculationError("Path calculation failed: {}".format(exc)) from exc
        if not path:
            msg = "No route found from '{}' to '{}'".format(start, end)
            if self.config.max_transfers is not None:
                msg += " within {} transfer(s)".format(self.config.max_transfers)
            raise PathCalculationError(msg)

        names = [net.number_to_station[n] for n in path]
        distance = sum(e.weight for e in edges)
        transfer_count = sp.get_transfer_count(e_num)

        lines = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)

        transfers = []
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                transfers.append({"station": names[i],
                                  "from_line": edges[i - 1].line,
                                  "to_line": edges[i].line})

        segments = []
        for i, e in enumerate(edges):
            if segments and segments[-1]["line"] == e.line:
                seg = segments[-1]
                seg["stations"].append(names[i + 1])
                seg["to"] = names[i + 1]
                seg["distance"] += e.weight
            else:
                segments.append({"line": e.line, "from": names[i], "to": names[i + 1],
                                 "stations": [names[i], names[i + 1]],
                                 "distance": e.weight})

        try:
            fare = Sum_money(distance)
            est_time = calculate_estimated_time(distance, transfer_count)
        except FareCalculationError as exc:
            raise FareCalculationError("Fare calculation failed: {}".format(exc)) from exc

        return {
            "query": {
                "start": start,
                "end": end,
                "mode": self.config.mode,
                "max_transfers": self.config.max_transfers,
                "line_preference": list(self.config.preferred_lines),
            },
            "path": names,
            "station_count": len(names),
            "distance": {"meters": distance, "kilometers": round(distance / 1000.0, 2)},
            "fare": fare,
            "estimated_time": round(est_time, 1),
            "transfer_count": transfer_count,
            "lines": lines,
            "transfers": transfers,
            "segments": segments,
        }

    def query(self, text):
        start, end = self.parse_query(text)
        return self.plan(start, end)

"""Path visualisation and report export (text and JSON)."""

import json


def format_distance(meters):
    """Human friendly distance: meters for short, kilometers for long."""
    if meters < 1000:
        return "{} meters ({:.2f} km)".format(meters, meters / 1000.0)
    return "{:.2f} km ({} meters)".format(meters / 1000.0, meters)


def build_summary(result):
    return {
        "total_distance_km": round(result["distance"]["meters"] / 1000.0, 2),
        "fare": result["fare"],
        "transfer_count": result["transfer_count"],
        "estimated_time": result["estimated_time"],
    }


def format_text(result):
    q = result["query"]
    lines = []
    lines.append("=" * 50)
    lines.append("Route: {} -> {}".format(q["start"], q["end"]))
    lines.append("Mode: {}".format(q["mode"]))
    if q.get("max_transfers") is not None:
        lines.append("Max transfers: {}".format(q["max_transfers"]))
    if q.get("line_preference"):
        lines.append("Line preference: {}".format(", ".join(q["line_preference"])))
    lines.append("-" * 50)
    lines.append("Stations ({}): {}".format(len(result["path"]), " -> ".join(result["path"])))
    lines.append("Segments:")
    for i, seg in enumerate(result["segments"]):
        if i > 0:
            prev = result["segments"][i - 1]
            lines.append("  >> Transfer at {}: {} -> {}".format(
                seg["from"], prev["line"], seg["line"]))
        lines.append("  [{}] {} -> {} ({} stops, {})".format(
            seg["line"], seg["from"], seg["to"], len(seg["stations"]) - 1,
            format_distance(seg["distance"])))
        lines.append("      " + " -> ".join(seg["stations"]))
    lines.append("-" * 50)
    lines.append("Total distance: {}".format(format_distance(result["distance"]["meters"])))
    if result["transfers"]:
        lines.append("Transfers: {} (at {})".format(
            result["transfer_count"], ", ".join(t["station"] for t in result["transfers"])))
    else:
        lines.append("Transfers: 0")
    lines.append("Lines involved: {}".format(", ".join(result["lines"])))
    lines.append("Estimated time: {} minutes".format(result["estimated_time"]))
    lines.append("Fare: {} yuan".format(result["fare"]))
    lines.append("=" * 50)
    return "\n".join(lines)


def format_json(result):
    data = dict(result)
    data["summary"] = build_summary(result)
    return json.dumps(data, ensure_ascii=False, indent=2)


def render(result, fmt="text"):
    if fmt == "json":
        return format_json(result)
    return format_text(result)


def export_results(results, path, fmt="text"):
    """Write all query results to a file in the chosen format."""
    with open(path, "w", encoding="utf-8") as fh:
        if fmt == "json":
            items = [dict(r, summary=build_summary(r)) for r in results]
            payload = items[0] if len(items) == 1 else items
            json.dump(payload, fh, ensure_ascii=False, indent=2)
            fh.write("\n")
        else:
            fh.write("\n\n".join(format_text(r) for r in results))
            fh.write("\n")

"""Command-line interface for the Beijing Subway fare calculation system.

Usage:
    python main.py [--format text|json] [--output_file PATH]
                   [--mode shortest_distance|least_transfer|comprehensive]
                   [--max_transfers N] [--line_preference "Line 1"]
                   [--check_data]

Enter queries as 'Starting Station; Ending Station'. Type 'quit' to exit.
"""

import argparse
import sys

from data_loader import DATA_FILE_PATH
from data_manager import DataIntegrityChecker, SubwayNetwork
from fare_calculator import FareCalculationError
from pathfinding import PathfindingConfig, VALID_MODES
from report import export_results, render
from route_planner import PathCalculationError, QueryValidationError, RoutePlanner

EXIT_COMMANDS = {"q", "quit", "exit"}
PROMPT = "Please enter 'Starting Station; Ending Station' (or 'quit' to exit): "


def parse_args(argv=None):
    parser = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", "--output-file", dest="output_file", default=None,
                        help="Export query results to this file")
    parser.add_argument("--format", dest="format", choices=["text", "json"], default="text",
                        help="Output format")
    parser.add_argument("--mode", dest="mode", choices=list(VALID_MODES),
                        default="shortest_distance", help="Path query mode")
    parser.add_argument("--max_transfers", "--max-transfers", dest="max_transfers",
                        type=int, default=None, help="Maximum number of transfers")
    parser.add_argument("--line_preference", "--line-preference", dest="line_preference",
                        default=None, help="Preferred line(s), comma separated")
    parser.add_argument("--check_data", "--check-data", dest="check_data",
                        action="store_true", help="Run data integrity check and exit")
    parser.add_argument("--data_file", dest="data_file", default=DATA_FILE_PATH,
                        help=argparse.SUPPRESS)
    return parser.parse_args(argv)


def build_config(args, network, out):
    preferred = []
    if args.line_preference:
        for raw in args.line_preference.replace("，", ",").split(","):
            raw = raw.strip()
            if not raw:
                continue
            line = network.resolve_line(raw)
            if line is None:
                print("Warning: line '{}' does not exist, preference ignored".format(raw),
                      file=out)
            else:
                preferred.append(line)
    return PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                             preferred_lines=preferred)


def run_interactive(planner, args, inp=None, out=None):
    inp = inp if inp is not None else sys.stdin
    out = out if out is not None else sys.stdout
    results = []
    while True:
        out.write(PROMPT)
        out.flush()
        try:
            line = inp.readline()
        except KeyboardInterrupt:
            out.write("\n")
            break
        if not line:
            out.write("\n")
            break
        text = line.strip()
        if not text:
            continue
        if text.lower() in EXIT_COMMANDS:
            break
        try:
            result = planner.query(text)
        except QueryValidationError as exc:
            print("Error: {}".format(exc), file=out)
            continue
        except PathCalculationError as exc:
            print("Path calculation error: {}".format(exc), file=out)
            continue
        except FareCalculationError as exc:
            print("Fare calculation error: {}".format(exc), file=out)
            continue
        except Exception as exc:  # defensive: keep the CLI alive
            print("Unexpected error: {}".format(exc), file=out)
            continue
        print(render(result, args.format), file=out)
        results.append(result)
        if args.output_file:
            try:
                export_results(results, args.output_file, args.format)
                print("Result exported to {}".format(args.output_file), file=out)
            except OSError as exc:
                print("Error: cannot write output file: {}".format(exc), file=out)
    print("Goodbye!", file=out)
    return results


def main(argv=None):
    args = parse_args(argv)
    out = sys.stdout
    try:
        network = SubwayNetwork(args.data_file)
    except (OSError, ValueError) as exc:
        print("Error: failed to load subway data: {}".format(exc), file=out)
        return 1

    if args.check_data:
        checker = DataIntegrityChecker(network)
        checker.run()
        print(checker.format_report(), file=out)
        return 0

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer", file=out)
        return 1

    config = build_config(args, network, out)
    planner = RoutePlanner(network, config)
    print("Beijing Subway Fare Calculation System ({} stations loaded)".format(
        len(network.station_to_number)), file=out)
    run_interactive(planner, args, sys.stdin, out)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Core data models for the lottery system."""

import re
from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A participant of the lottery."""

    name: str
    department: str

    def __post_init__(self):
        self.name = (self.name or "").strip()
        self.department = (self.department or "").strip()

    def __hash__(self):
        return hash((self.name, self.department))

    def __str__(self):
        return f"{self.name} ({self.department})"


_LEVEL_KEYWORDS = [
    (r"\bspecial\b", 0),
    (r"\bgrand\b", 0),
    (r"\bfirst\b", 1),
    (r"\bsecond\b", 2),
    (r"\bthird\b", 3),
    (r"\bfourth\b", 4),
    (r"\bfifth\b", 5),
    (r"\bsixth\b", 6),
]


@dataclass
class Prize:
    """A prize configured for one lottery round."""

    name: str
    quantity: int
    avoid_groups: Optional[List[str]] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        self.name = (self.name or "").strip()
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if isinstance(self.associated_drawer, str):
            self.associated_drawer = self.associated_drawer.strip() or None

    @property
    def level(self) -> int:
        """Numeric level: smaller means higher prize. Unknown levels are lowest."""
        lowered = self.name.lower()
        for pattern, lvl in _LEVEL_KEYWORDS:
            if re.search(pattern, lowered):
                return lvl
        return 99

    def __str__(self):
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "None"
        return (f"{self.name} | Quantity: {self.quantity} | "
                f"Avoidance groups: {avoid} | Lottery person: {drawer}")

_SEPARATORS = re.compile(r"[,;，；]")


class ListManager:
    """Loads, validates and analyses participant lists."""

    def __init__(self):
        self.load_mode = None
        self.source_file_path = None

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        self.load_mode = "default"
        self.source_file_path = None
        return [User(n, d) for n, d in DEFAULT_LIST], []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        self.load_mode = "custom"
        self.source_file_path = path
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        if not path or not os.path.isfile(path):
            return users, [f"File not found: {path}"]
        content = None
        for enc in ("utf-8-sig", "gbk", "latin-1"):
            try:
                with open(path, "r", encoding=enc) as f:
                    content = f.read()
                break
            except (UnicodeDecodeError, LookupError):
                continue
            except OSError as exc:
                return users, [f"Unable to read file {path}: {exc}"]
        if content is None:
            return users, [f"Unable to decode file: {path}"]

        for idx, raw in enumerate(content.splitlines(), start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in _SEPARATORS.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - '{line}' "
                    f"(expected 'Name,Department Group')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            key = u.name.strip()
            if key in seen:
                duplicates.append(u)
            else:
                seen.add(key)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

"""Prize library management."""

from typing import List

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], None),
    ("Second Prize - Shopping Cards", 3, [], None),
    ("First Prize - 1 Day Annual Leave", 1, [], None),
]


class PrizeManager:
    """Manages the prize library and lottery round sequence."""

    def __init__(self):
        self.prizes: List[Prize] = []
        self.is_preset = False

    def load_default_prizes(self) -> List[str]:
        errors = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            if self._find(name) is not None:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            if not self.add_prize(name, qty, list(avoid), drawer):
                errors.append(f"Failed to add preset prize '{name}'")
        self.is_preset = True
        return errors

    def _find(self, name):
        for p in self.prizes:
            if p.name == name:
                return p
        return None

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> bool:
        if not isinstance(name, str) or not name.strip():
            return False
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            return False
        if qty <= 0:
            return False
        if self._find(name.strip()) is not None:
            return False
        self.prizes.append(Prize(name.strip(), qty, list(avoid_groups or []), associated_drawer))
        return True

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def set_avoid_groups(self, index, groups) -> bool:
        if not 0 <= index < len(self.prizes):
            return False
        self.prizes[index].avoid_groups = [g.strip() for g in groups if g and g.strip()]
        return True

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

"""Lottery rule engine and draw execution."""

import random
from typing import Iterable, List, Optional

from lottery_system.models import Prize, User

SPECIAL_PERSON = "Li Bin"
FALLBACK_TRIGGER_PERSONS = ["Chen Qixian", "Jin Duo"]
FALLBACK_TARGET_PERSON = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def _norm_person(name) -> str:
    return " ".join((name or "").strip().lower().split())


def _norm_group(name) -> str:
    return (name or "").strip().lower()


def _is_protected_group(dept) -> bool:
    key = "".join((dept or "").lower().split())
    if key.endswith("group"):
        key = key[:-5]
    return key == "chengru"


class LotteryManager:
    """Applies the rule engine and draws winners."""

    def __init__(self, users, existing_winners: Optional[Iterable[User]] = None):
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []
        self.rng = random.Random()
        for u in existing_winners or []:
            self.mark_winner(u)

    # ---- rule helpers ----
    def _not_won(self) -> List[User]:
        return [u for u in self.users if u.name not in self.winner_names]

    def is_special_person(self, drawer) -> bool:
        return _norm_person(drawer) == _norm_person(SPECIAL_PERSON)

    def is_fallback_active(self, prize: Prize) -> bool:
        if _norm_person(prize.associated_drawer) != _norm_person(FALLBACK_TARGET_PERSON):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def _avoided(self, user: User, prize: Prize) -> bool:
        groups = {_norm_group(g) for g in (prize.avoid_groups or [])}
        return _norm_group(user.department) in groups

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = self._not_won()
        if self.is_fallback_active(prize):
            protected = [u for u in candidates if _is_protected_group(u.department)]
            if protected:
                return protected
        if self.is_special_person(prize.associated_drawer):
            return candidates
        return [u for u in candidates if not self._avoided(u, prize)]

    def _get_deferred_pool(self, prize: Prize) -> List[User]:
        """Avoidance-group members: still eligible but drawn after others."""
        return [u for u in self._not_won() if self._avoided(u, prize)]

    def describe_rules(self, prize: Prize) -> str:
        rules = ["Previous winners are excluded"]
        if self.is_fallback_active(prize):
            rules.append(f"Fallback active: pool limited to '{PROTECTED_GROUP}' members "
                         f"(avoidance groups ignored)")
        elif self.is_special_person(prize.associated_drawer):
            rules.append(f"Special lottery person {SPECIAL_PERSON}: avoidance groups not applied")
        elif prize.avoid_groups:
            rules.append("Avoidance groups (drawn only after other members): "
                         + ", ".join(prize.avoid_groups))
        else:
            rules.append("No avoidance groups")
        if prize.associated_drawer:
            rules.append(f"Lottery person: {prize.associated_drawer}")
        return "; ".join(rules)

    # ---- draw ----
    def mark_winner(self, user: User):
        if user.name not in self.winner_names:
            self.winner_names.add(user.name)
            self.winners.append(user)

    def draw_winner(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool and not self.is_special_person(prize.associated_drawer):
            pool = self._get_deferred_pool(prize)
        pool = [u for u in pool if u.name not in self.winner_names]
        if not pool:
            return None
        winner = self.rng.choice(pool)
        self.mark_winner(winner)
        return winner

    def finish_round(self, prize: Prize, round_winners: List[User]):
        drawer = _norm_person(prize.associated_drawer)
        if drawer in {_norm_person(p) for p in FALLBACK_TRIGGER_PERSONS}:
            has_protected = any(_is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(has_protected)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm_person(FALLBACK_TARGET_PERSON):
            self.last_two_draw_results = []

"""Winner records, summaries and export."""

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize


class Winner:
    def __init__(self, user, prize, time=None, order=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.order = order

    @property
    def prize_name(self) -> str:
        return self.prize.name if isinstance(self.prize, Prize) else str(self.prize)

    def __repr__(self):
        return f"Winner({self.user!r}, {self.prize_name!r})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    @property
    def total_winners(self) -> int:
        return len(self.winners)

    def add_winner(self, user, prize):
        if any(w.user.name == user.name for w in self.winners):
            return None
        w = Winner(user, prize, datetime.now(), len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        indexed = list(enumerate(prizes or []))
        indexed.sort(key=lambda t: (t[1].level, t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, p in indexed:
            result.setdefault(p.name, [])
        for w in sorted(self.winners, key=lambda x: (x.time, x.order)):
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winning results to export"
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for pname, winners in self.get_winners_by_prize(prizes).items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, pname,
                                         w.time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

import sys

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ask(prompt=""):
    try:
        return input(prompt).strip()
    except EOFError:
        print("\nInput closed. Exiting.")
        sys.exit(0)


class LotteryCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None
        self.drawn_count = {}
        self.status = "idle"

    # ---------- list ----------
    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        self._set_users(valid)
        print(f"Default list loaded: {len(valid)} participants.")

    def load_custom(self):
        path = ask("Enter txt file path: ")
        users, errors = self.list_manager.load_custom_list(path)
        valid, dups = self.list_manager.validate_users(users)
        if errors:
            print("Problems detected:")
            for e in errors:
                print("  " + e)
        if dups:
            print("Duplicate names detected:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
        if not valid:
            print("No valid participants found. Upload terminated.")
            return
        if errors or dups:
            choice = ask("Type 'r' to remove invalid data and continue, 't' to terminate upload: ").lower()
            if choice != "r":
                print("Upload terminated.")
                return
        self._set_users(valid)
        print(f"Custom list loaded: {len(valid)} participants.")

    def _set_users(self, users):
        self.users = users
        prior = [w.user for w in self.result_manager.winners]
        self.lottery = LotteryManager(users, existing_winners=prior)

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        print("Valid groups:")
        for g, c in self.list_manager.get_group_info(self.users).items():
            print(f"  {g}: {c} people")

    # ---------- prizes ----------
    def load_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prize library loaded.")

    def _choose_groups(self):
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if not groups:
            raw = ask("Enter avoidance groups separated by ',' (blank for none): ")
            return [g.strip() for g in raw.split(",") if g.strip()]
        for i, g in enumerate(groups, 1):
            print(f"  {i}. {g}")
        raw = ask("Select avoidance groups by number, separated by ',' (blank for none): ")
        selected = []
        for part in raw.split(","):
            part = part.strip()
            if part.isdigit() and 1 <= int(part) <= len(groups):
                if groups[int(part) - 1] not in selected:
                    selected.append(groups[int(part) - 1])
            elif part:
                print(f"  Ignored invalid selection: {part}")
        return selected

    def add_prize(self):
        name = ask("Prize name: ")
        qty = ask("Quantity: ")
        drawer = ask("Associated lottery person (blank for none): ") or None
        groups = self._choose_groups()
        if self.prize_manager.add_prize(name, qty, groups, drawer):
            print("Prize added.")
        else:
            print("Failed to add prize (empty/duplicate name or invalid quantity).")

    def configure_avoidance(self):
        prizes = self.view_prizes()
        if not prizes:
            return
        idx = ask("Select prize number: ")
        if not idx.isdigit() or not 1 <= int(idx) <= len(prizes):
            print("Invalid selection.")
            return
        groups = self._choose_groups()
        self.prize_manager.set_avoid_groups(int(idx) - 1, groups)
        print("Avoidance groups updated.")

    def reorder(self):
        prizes = self.view_prizes()
        if not prizes:
            return
        raw = ask("Enter new order of serial numbers separated by ',' (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace(" ", "").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.view_prizes()
        else:
            print("Invalid order.")

    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")
        return prizes

    # ---------- lottery ----------
    def _control(self):
        """Return 'go' or 'terminate'."""
        cmd = ask("[Enter]=draw | pause | terminate > ").lower()
        if cmd == "terminate":
            return "terminate"
        if cmd == "pause":
            self.status = "paused"
            print("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
            while True:
                c = ask("> ").lower()
                if c == "continue":
                    self.status = "running"
                    print("Resumed.")
                    return "go"
                if c == "terminate":
                    return "terminate"
        return "go"

    def run_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        self.status = "running"
        for rnd, prize in enumerate(prizes, 1):
            done = self.drawn_count.get(prize.name, 0)
            remaining = prize.quantity - done
            if remaining <= 0:
                continue
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {remaining}")
            print("Rules: " + self.lottery.describe_rules(prize))
            print("=" * 60)
            round_winners = []
            for i in range(remaining):
                if self._control() == "terminate":
                    print("Current round terminated, skipping.")
                    break
                print(f"Currently drawing the {done + i + 1}th winner...")
                winner = self.lottery.draw_winner(prize)
                if winner is None:
                    print("No eligible participants left for this prize.")
                    break
                self.result_manager.add_winner(winner, prize)
                round_winners.append(winner)
                self.drawn_count[prize.name] = self.drawn_count.get(prize.name, 0) + 1
                print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}")
            self.lottery.finish_round(prize, round_winners)
            print(f"\nRound {rnd} results ({prize.name}):")
            if not round_winners:
                print("  (none)")
            for n, w in enumerate(round_winners, 1):
                print(f"  {n}. {w.name} ({w.department})")
        self.status = "finished"
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for pname, winners in data.items():
            if not winners:
                continue
            print(f"[{pname}]")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {pname} - "
                      f"{w.time.strftime('%Y-%m-%d %H:%M:%S')}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_prizes, "5": self.add_prize, "6": self.configure_avoidance,
            "7": self.reorder, "8": self.view_prizes, "9": self.run_lottery,
            "10": self.show_summary, "11": self.export,
        }
        while True:
            print("""
===== Team Building Smart Lottery System =====
 1. Load default list          2. Upload custom txt list
 3. View groups                4. Load preset prizes
 5. Add custom prize           6. Configure avoidance groups
 7. Reorder prizes             8. View prizes
 9. Start lottery             10. View winner summary
11. Export results             0. Exit""")
            choice = ask("Select: ")
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid option.")


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

"""
Data loading utilities for the Beijing Subway fare calculation system.

The network data lives in ``subway_data.json`` next to this module and has
the following structure::

    {
        "stations":    [{"name": str, "lines": [str, ...]}, ...],
        "connections": [{"source": str, "dest": str,
                         "distance": int (meters), "line": str}, ...]
    }
"""

import json
import os

# Absolute path of the bundled data file; valid from any working directory.
DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON data file (defaults to DATA_FILE_PATH)
    :return: the parsed JSON content (a dict)
    :raises FileNotFoundError: if the file does not exist
    :raises ValueError: if the file is not valid JSON
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid JSON in data file '{path}': {exc}") from exc
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file (used by the data update mechanism)."""
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, ensure_ascii=False, indent=2)

"""
Core graph data structures: Edge, Graph and the MinPQ priority queue.
"""


class Edge:
    """A weighted, line-labelled connection between two stations.

    ``source`` and ``dest`` are station numbers; ``weight`` equals the
    distance in meters.
    """

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        """Return the same connection in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def other(self, v):
        """Return the endpoint that is not ``v``."""
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError(f"Vertex {v} is not an endpoint of this edge")

    def __repr__(self):
        return (f"Edge({self.source!r}, {self.dest!r}, "
                f"{self.distance!r}, {self.line!r})")


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    ``graph`` is a dictionary mapping station number -> list of outgoing
    Edge objects. Each undirected connection is stored in both directions.
    Station numbers are assigned by the caller.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self._edges = []

    # ---- vertices -------------------------------------------------------
    def add_vertex(self, v):
        self.graph.setdefault(v, [])

    def vertices(self):
        return list(self.graph.keys())

    @property
    def V(self):
        return len(self.graph)

    @property
    def E(self):
        return len(self._edges)

    # ---- station metadata ----------------------------------------------
    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    # ---- edges ----------------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        if edge.source != edge.dest:
            self.graph[edge.dest].append(edge.reversed())
        self._edges.append(edge)

    def adj(self, v):
        return self.graph.get(v, [])

    def edges(self):
        return list(self._edges)

    def remove_edge(self, a, b, line=None):
        """Remove the connection(s) between a and b (optionally on one line)."""
        def match(e, x, y):
            return e.source == x and e.dest == y and (line is None or e.line == line)

        removed = 0
        for v in (a, b):
            if v in self.graph:
                before = len(self.graph[v])
                self.graph[v] = [e for e in self.graph[v]
                                 if not (match(e, a, b) or match(e, b, a))]
                removed += before - len(self.graph[v])
        self._edges = [e for e in self._edges
                       if not (match(e, a, b) or match(e, b, a))]
        return removed > 0

    def update_edge_distance(self, a, b, distance, line=None):
        """Update the distance of connection(s) between a and b."""
        updated = False
        for v in (a, b):
            for e in self.graph.get(v, []):
                if {e.source, e.dest} == {a, b} and (line is None or e.line == line):
                    e.distance = distance
                    e.weight = distance
                    updated = True
        for e in self._edges:
            if {e.source, e.dest} == {a, b} and (line is None or e.line == line):
                e.distance = distance
                e.weight = distance
        return updated


class MinPQ:
    """Binary-heap minimum priority queue.

    Items are (key, value) pairs; ties are broken by insertion order so that
    values never need to be comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = 0

    def __len__(self):
        return len(self._heap)

    def size(self):
        return len(self._heap)

    def is_empty(self):
        return not self._heap

    def insert(self, key, value):
        self._heap.append((key, self._counter, value))
        self._counter += 1
        self._swim(len(self._heap) - 1)

    def min(self):
        if not self._heap:
            raise IndexError("Priority queue underflow")
        key, _, value = self._heap[0]
        return key, value

    def del_min(self):
        if not self._heap:
            raise IndexError("Priority queue underflow")
        top = self._heap[0]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._sink(0)
        return top[0], top[2]

    def _less(self, i, j):
        a, b = self._heap[i], self._heap[j]
        return (a[0], a[1]) < (b[0], b[1])

    def _swim(self, k):
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._heap[k], self._heap[parent] = self._heap[parent], self._heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        n = len(self._heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and self._less(right, left):
                child = right
            if self._less(child, k):
                self._heap[k], self._heap[child] = self._heap[child], self._heap[k]
                k = child
            else:
                break

"""
Dijkstra-based path computation with transfer awareness.

The search runs over states (station, current line[, transfers]) so that
transfers at multi-line interchange stations are counted correctly, which
enables the ``least_transfer``/``comprehensive`` modes, the maximum transfer
limit and line preferences. With the default configuration the result is
the plain shortest-distance path.
"""

from models import MinPQ

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Penalty (in meters equivalent) applied per transfer in comprehensive mode.
TRANSFER_PENALTY_METERS = 3000
# Edges on preferred lines are weighted by this factor during the search.
PREFERRED_LINE_FACTOR = 0.7

INF = float("inf")


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None,
                 preferred_lines=None, transfer_penalty=TRANSFER_PENALTY_METERS):
        if mode not in VALID_MODES:
            raise ValueError(
                f"Invalid mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise TypeError("max_transfers must be an integer")
            if max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []
        self.transfer_penalty = transfer_penalty


class ShortestPath:
    """Single-source Dijkstra over the subway graph.

    Attributes:
        dist_to: station number -> travelled distance (meters) of the chosen
                 path; ``inf`` for unreachable stations.
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: INF for v in graph.vertices()}
        self.dist_to[source] = 0
        self._info = {}    # state -> (key, eff, dist, transfers, parent_state, edge)
        self._final = {}   # station -> best settled state
        self._preferred = {p.strip().lower() for p in self.config.preferred_lines
                           if p and p.strip()}
        self._run()

    # ------------------------------------------------------------------
    def _key(self, eff, transfers):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, eff)
        if mode == "comprehensive":
            return (eff + transfers * self.config.transfer_penalty, transfers)
        return (eff, transfers)

    def _edge_cost(self, edge):
        if self._preferred and str(edge.line).strip().lower() in self._preferred:
            return edge.weight * PREFERRED_LINE_FACTOR
        return edge.weight

    def _run(self):
        cfg = self.config
        track_t = cfg.max_transfers is not None
        start = (self.source, None, 0 if track_t else None)
        key0 = self._key(0, 0)
        self._info[start] = (key0, 0, 0, 0, None, None)
        pq = MinPQ()
        pq.insert(key0, start)
        settled = set()

        while not pq.is_empty():
            key, state = pq.del_min()
            if state in settled:
                continue
            rec = self._info[state]
            if key != rec[0]:
                continue
            settled.add(state)
            v, line, _ = state
            if v not in self._final:
                self._final[v] = state
                self.dist_to[v] = rec[2]

            for e in self.graph.adj(v):
                transfers = rec[3] + (1 if (line is not None and e.line != line) else 0)
                if track_t and transfers > cfg.max_transfers:
                    continue
                ns = (e.dest, e.line, transfers if track_t else None)
                if ns in settled:
                    continue
                eff = rec[1] + self._edge_cost(e)
                nk = self._key(eff, transfers)
                old = self._info.get(ns)
                if old is None or nk < old[0]:
                    self._info[ns] = (nk, eff, rec[2] + e.weight, transfers, state, e)
                    pq.insert(nk, ns)

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._final

    def path_with_edges(self, target):
        """Return (stations, edges) from source to target; ([], []) if unreachable."""
        if target not in self._final:
            return [], []
        edges = []
        state = self._final[target]
        while True:
            rec = self._info[state]
            if rec[4] is None:
                break
            edges.append(rec[5])
            state = rec[4]
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes on the path to target (0 if unreachable)."""
        if target not in self._final:
            return 0
        return self._info[self._final[target]][3]

"""
Beijing Subway fare calculation.

Fare structure:
    0-6 km: 3 yuan, 6-12 km: 4 yuan, 12-22 km: 5 yuan, 22-32 km: 6 yuan,
    above 32 km: +1 yuan for every (started) 20 km.
Distances are rounded up to the next whole kilometer.
"""

import math

# (upper bound in km inclusive, fare in yuan)
FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]
BASE_EXTENDED_FARE = 6
EXTENDED_THRESHOLD_KM = 32
EXTENDED_STEP_KM = 20

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number (meters)")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def distance_to_km(distance):
    """Convert meters to whole kilometers, rounding up."""
    _validate_distance(distance)
    if isinstance(distance, int):
        return -(-distance // 1000)
    return math.ceil(distance / 1000.0)


def Sum_money(distance):
    """Return the fare in yuan for a distance given in meters."""
    km = distance_to_km(distance)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    extra = math.ceil((km - EXTENDED_THRESHOLD_KM) / EXTENDED_STEP_KM)
    return BASE_EXTENDED_FARE + extra


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes for a distance in meters."""
    _validate_distance(distance)
    if transfer_count is None:
        transfer_count = 0
    if transfer_count < 0:
        raise ValueError("Transfer count cannot be negative")
    minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    minutes += transfer_count * TRANSFER_TIME_MINUTES
    return round(float(minutes), 1)


def format_fare(fare):
    return f"Fare: {fare} yuan"

def _norm(text):
    text = unicodedata.normalize("NFKC", str(text)).strip().lower()
    text = text.replace("’", "'").replace("‘", "'")
    return re.sub(r"\s+", " ", text)


def _compact(text):
    return re.sub(r"[\s'\-·.]", "", _norm(text))


def _strip_suffix(text):
    t = _norm(text)
    if t.endswith("站") and len(t) > 1:
        t = t[:-1]
    elif t.endswith(" station") and len(t) > len(" station"):
        t = t[: -len(" station")]
    return t


class NetworkError(Exception):
    """Raised on invalid data update or query operations."""


class SubwayNetwork:
    """Beijing Subway network model."""

    def __init__(self, data=None, data_path=None):
        self.data_path = data_path or DATA_FILE_PATH
        if data is None:
            data = load_subway_data(self.data_path)
        if not isinstance(data, dict):
            raise ValueError("Subway data must be a JSON object")
        self.data = copy.deepcopy(data)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])
        self._rebuild()

    # ------------------------------------------------------------------
    # Model construction
    # ------------------------------------------------------------------
    def _rebuild(self):
        stations = self.data.get("stations", [])
        connections = self.data.get("connections", [])

        self.station_to_number = {}
        self.number_to_station = {}
        self.station_info = {}

        def register(name, lines):
            if name in self.station_to_number:
                existing = self.station_info[name]["lines"]
                for ln in lines:
                    if ln not in existing:
                        existing.append(ln)
                return
            number = len(self.station_to_number)
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.station_info[name] = {"name": name, "number": number,
                                       "lines": list(lines)}

        for st in stations:
            if isinstance(st, dict) and st.get("name"):
                register(st["name"], st.get("lines", []) or [])

        for c in connections:
            for key in ("source", "dest"):
                name = c.get(key)
                if name and name not in self.station_to_number:
                    register(name, [c["line"]] if c.get("line") else [])
                elif name and c.get("line") and not self.station_info[name]["lines"]:
                    self.station_info[name]["lines"].append(c["line"])

        # Name-based adjacency dictionary (weighted undirected graph).
        self.graph = {name: [] for name in self.station_to_number}
        self.network = Graph()
        for name, info in self.station_info.items():
            self.network.add_vertex(info["number"])
            self.network.set_station_lines(name, info["lines"])

        for c in connections:
            s, d = c.get("source"), c.get("dest")
            dist = c.get("distance")
            if not s or not d or s == d:
                continue
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist < 0:
                continue
            line = c.get("line", "")
            self.network.add_edge(Edge(self.station_to_number[s],
                                       self.station_to_number[d], dist, line))
            self.graph[s].append({"station": d, "distance": dist, "line": line})
            self.graph[d].append({"station": s, "distance": dist, "line": line})

        self.lines = self._build_line_sequences()
        self._build_name_index()

    def _build_line_sequences(self):
        per_line = defaultdict(dict)
        for c in self.data.get("connections", []):
            s, d, line = c.get("source"), c.get("dest"), c.get("line")
            if not s or not d or not line or s == d:
                continue
            per_line[line].setdefault(s, [])
            per_line[line].setdefault(d, [])
            if d not in per_line[line][s]:
                per_line[line][s].append(d)
            if s not in per_line[line][d]:
                per_line[line][d].append(s)

        lines = {}
        for line, adj in per_line.items():
            ends = [s for s in adj if len(adj[s]) == 1]
            is_loop = not ends and all(len(n) == 2 for n in adj.values())
            order, visited = [], set()
            starts = ends + [s for s in adj if s not in ends]
            for start in starts:
                if start in visited:
                    continue
                stack = [start]
                while stack:
                    v = stack.pop()
                    if v in visited:
                        continue
                    visited.add(v)
                    order.append(v)
                    for w in reversed(adj[v]):
                        if w not in visited:
                            stack.append(w)
            transfers = [s for s in order
                         if len(self.station_info.get(s, {}).get("lines", [])) > 1]
            lines[line] = {"name": line, "stations": order, "is_loop": is_loop,
                           "transfer_stations": transfers}
        return lines

    def _build_name_index(self):
        self._exact = {}
        self._compact_index = {}
        for name in self.station_to_number:
            self._exact.setdefault(_norm(name), name)
        for name in self.station_to_number:
            for key in (_strip_suffix(name),):
                self._exact.setdefault(key, name)
            self._compact_index.setdefault(_compact(name), name)
            self._compact_index.setdefault(_compact(_strip_suffix(name)), name)

    # ------------------------------------------------------------------
    # Lookup
    # ------------------------------------------------------------------
    def resolve_station(self, name):
        """Return the official station name matching ``name`` or None."""
        if name is None:
            return None
        if name in self.station_to_number:
            return name
        for key in (_norm(name), _strip_suffix(name)):
            if key in self._exact:
                return self._exact[key]
        for key in (_compact(name), _compact(_strip_suffix(name))):
            if key and key in self._compact_index:
                return self._compact_index[key]
        return None

    def suggest_stations(self, name, n=3):
        return difflib.get_close_matches(str(name).strip(),
                                         list(self.station_to_number), n=n, cutoff=0.6)

    def all_line_names(self):
        names = set(self.lines)
        for info in self.station_info.values():
            names.update(info["lines"])
        return sorted(names)

    def resolve_line(self, line):
        target = _compact(line)
        for name in self.all_line_names():
            if _compact(name) == target:
                return name
        return None

    def transfer_stations(self):
        return [n for n, i in self.station_info.items() if len(i["lines"]) > 1]

    # ------------------------------------------------------------------
    # Route queries
    # ------------------------------------------------------------------
    def find_route(self, start, end, config=None):
        """Compute a route. Returns dict with station names, edges, distance,
        transfer count; or None if no route satisfies the constraints."""
        if start not in self.station_to_number:
            raise NetworkError(f"Unknown station '{start}'")
        if end not in self.station_to_number:
            raise NetworkError(f"Unknown station '{end}'")
        src = self.station_to_number[start]
        dst = self.station_to_number[end]
        sp = ShortestPath(self.network, src, config or PathfindingConfig())
        if not sp.has_path_to(dst):
            return None
        numbers, edges = sp.path_with_edges(dst)
        return {
            "stations": [self.number_to_station[n] for n in numbers],
            "edges": [{"source": self.number_to_station[e.source],
                       "dest": self.number_to_station[e.dest],
                       "distance": e.distance, "line": e.line} for e in edges],
            "distance": sp.dist_to[dst],
            "transfer_count": sp.get_transfer_count(dst),
        }

    # ------------------------------------------------------------------
    # Integrity check
    # ------------------------------------------------------------------
    def check_data_integrity(self):
        errors, warnings = [], []
        raw_stations = self.data.get("stations", [])
        raw_connections = self.data.get("connections", [])

        declared = {}
        for i, st in enumerate(raw_stations):
            if not isinstance(st, dict) or not st.get("name"):
                errors.append(f"Station entry #{i} has no name")
                continue
            if st["name"] in declared:
                warnings.append(f"Duplicate station entry '{st['name']}'")
            declared[st["name"]] = st.get("lines", []) or []
            if not declared[st["name"]]:
                warnings.append(f"Station '{st['name']}' has no line information")

        seen_pairs = {}
        for i, c in enumerate(raw_connections):
            if not isinstance(c, dict):
                errors.append(f"Connection #{i} is not an object")
                continue
            s, d, dist, line = c.get("source"), c.get("dest"), c.get("distance"), c.get("line")
            label = f"Connection #{i} ({s} - {d}, {line})"
            if not s or not d:
                errors.append(f"{label}: missing source or destination")
                continue
            for name in (s, d):
                if name not in declared:
                    errors.append(f"{label}: station '{name}' is not in the station list")
            if s == d:
                errors.append(f"{label}: connects a station to itself")
            if not line:
                warnings.append(f"{label}: missing line name")
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                errors.append(f"{label}: distance is not a number")
                continue
            if dist <= 0:
                errors.append(f"{label}: distance must be positive")
            elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                warnings.append(f"{label}: unusual distance {dist} meters")
            if line:
                for name in (s, d):
                    if name in declared and declared[name] and line not in declared[name]:
                        warnings.append(
                            f"{label}: station '{name}' is not marked as on line '{line}'")
            key = (frozenset((s, d)), line)
            if key in seen_pairs:
                if seen_pairs[key] != dist:
                    errors.append(f"{label}: conflicting distances "
                                  f"{seen_pairs[key]} and {dist}")
                else:
                    warnings.append(f"{label}: duplicate connection")
            else:
                seen_pairs[key] = dist

        for name, neighbors in self.graph.items():
            if not neighbors:
                errors.append(f"Station '{name}' has no connections")

        components = self._count_components()
        if components > 1:
            warnings.append(f"Network is not fully connected ({components} components)")

        # Main line coverage
        line_norms = {_compact(n): n for n in self.all_line_names()}
        missing_lines = []
        for main, aliases in MAIN_LINES.items():
            found = any(_compact(a) in line_norms for a in aliases)
            if not found:
                found = any(any(_compact(a) in ln for a in aliases if len(_compact(a)) > 4)
                            for ln in line_norms)
            if not found:
                missing_lines.append(main)
        for ln in missing_lines:
            warnings.append(f"Main line '{ln}' not found in data")

        missing_stations = []
        checked = 0
        for line, reps in REPRESENTATIVE_STATIONS.items():
            for cn, en in reps:
                checked += 1
                if not (self.resolve_station(cn) or self.resolve_station(en)):
                    missing_stations.append(f"{en} ({cn}) [{line}]")
        for s in missing_stations:
            warnings.append(f"Representative station {s} not found")

        stats = {
            "stations": len(self.station_to_number),
            "connections": len(raw_connections),
            "lines": len(self.all_line_names()),
            "transfer_stations": len(self.transfer_stations()),
            "components": components,
            "representative_stations_checked": checked,
            "representative_stations_found": checked - len(missing_stations),
            "main_lines_found": len(MAIN_LINES) - len(missing_lines),
            "main_lines_total": len(MAIN_LINES),
        }
        return {"ok": not errors, "errors": errors, "warnings": warnings, "stats": stats}

    def _count_components(self):
        seen, count = set(), 0
        for start in self.graph:
            if start in seen:
                continue
            count += 1
            queue = deque([start])
            seen.add(start)
            while queue:
                v = queue.popleft()
                for nb in self.graph[v]:
                    if nb["station"] not in seen:
                        seen.add(nb["station"])
                        queue.append(nb["station"])
        return count

    # ------------------------------------------------------------------
    # Data updates
    # ------------------------------------------------------------------
    def _find_station_entry(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def add_station(self, name, lines):
        if not name or not str(name).strip():
            raise NetworkError("Station name cannot be empty")
        entry = self._find_station_entry(name)
        if entry is None:
            self.data["stations"].append({"name": name, "lines": list(lines)})
        else:
            for ln in lines:
                if ln not in entry.setdefault("lines", []):
                    entry["lines"].append(ln)
        self._rebuild()

    def add_line(self, line_name, stations, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if not line_name:
            raise NetworkError("Line name cannot be empty")
        if len(stations) < 2:
            raise NetworkError("A line needs at least two stations")
        if len(distances) != len(stations) - 1:
            raise NetworkError("Number of distances must equal number of stations - 1")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise NetworkError(f"Invalid distance {d!r}")
        for name in stations:
            entry = self._find_station_entry(name)
            if entry is None:
                self.data["stations"].append({"name": name, "lines": [line_name]})
            elif line_name not in entry.setdefault("lines", []):
                entry["lines"].append(line_name)
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": d, "line": line_name})
        self._rebuild()

    def update_station(self, name, new_name=None, lines=None):
        entry = self._find_station_entry(name)
        if entry is None:
            raise NetworkError(f"Station '{name}' does not exist")
        if new_name and new_name != name:
            if self._find_station_entry(new_name) is not None:
                raise NetworkError(f"Station '{new_name}' already exists")
            entry["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            entry["lines"] = list(lines)
        self._rebuild()

    def update_distance(self, source, dest, distance, line=None):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise NetworkError(f"Invalid distance {distance!r}")
        updated = 0
        for c in self.data["connections"]:
            if {c.get("source"), c.get("dest")} == {source, dest} and \
                    (line is None or c.get("line") == line):
                c["distance"] = distance
                updated += 1
        if not updated:
            raise NetworkError(f"No connection between '{source}' and '{dest}'")
        self._rebuild()
        return updated

    def save(self, path=None):
        save_subway_data(self.data, path or self.data_path)

"""
Path visualization and export: builds route reports and renders them as
console text or JSON.
"""

import json

from fare_calculator import Sum_money, calculate_estimated_time, format_fare


def format_distance(meters):
    """Meters for short distances, kilometers for long ones (both shown)."""
    km = meters / 1000.0
    if meters < 1000:
        return f"{int(round(meters))} meters ({km:.2f} km)"
    return f"{km:.2f} km ({int(round(meters))} meters)"


def build_segments(edges):
    segments = []
    for e in edges:
        if segments and segments[-1]["line"] == e["line"]:
            seg = segments[-1]
            seg["to"] = e["dest"]
            seg["stations"].append(e["dest"])
            seg["distance"] += e["distance"]
        else:
            segments.append({"line": e["line"], "from": e["source"], "to": e["dest"],
                             "stations": [e["source"], e["dest"]],
                             "distance": e["distance"]})
    return segments


class FareError(Exception):
    pass


def build_route_report(start, end, route, config):
    """Create the structured report dict from a route computed by SubwayNetwork."""
    distance = route["distance"]
    segments = build_segments(route["edges"])
    transfer_stations = [seg["to"] for seg in segments[:-1]]
    transfer_count = len(transfer_stations)
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    try:
        fare = Sum_money(distance)
    except (TypeError, ValueError) as exc:
        raise FareError(str(exc)) from exc
    est = calculate_estimated_time(distance, transfer_count)
    return {
        "query": {
            "start": start,
            "end": end,
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "line_preference": list(config.preferred_lines),
        },
        "path": list(route["stations"]),
        "station_count": len(route["stations"]),
        "distance": {"meters": distance, "kilometers": round(distance / 1000.0, 2)},
        "fare": fare,
        "estimated_time": est,
        "estimated_time_unit": "minutes",
        "transfer_count": transfer_count,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": [
            {"line": s["line"], "from": s["from"], "to": s["to"],
             "stations": s["stations"], "stops": len(s["stations"]) - 1,
             "distance": s["distance"]}
            for s in segments
        ],
    }


def summarize(report):
    """Key information summary of a report."""
    return {
        "total_distance_km": report["distance"]["kilometers"],
        "total_fare": report["fare"],
        "transfer_count": report["transfer_count"],
        "estimated_time_minutes": report["estimated_time"],
    }


def format_text_report(report):
    q = report["query"]
    sep = "=" * 60
    out = [sep, f"Route: {q['start']} -> {q['end']}", f"Mode: {q['mode']}"]
    if q["max_transfers"] is not None:
        out.append(f"Max transfers: {q['max_transfers']}")
    if q["line_preference"]:
        out.append(f"Line preference: {', '.join(q['line_preference'])}")
    out.append("-" * 60)
    out.append(f"Station sequence ({report['station_count']} stations):")
    out.append("  " + " -> ".join(report["path"]))
    out.append("Line segments:")
    segs = report["segments"]
    for i, s in enumerate(segs, 1):
        out.append(f"  {i}. [{s['line']}] {s['from']} -> {s['to']} "
                   f"({s['stops']} stops, {format_distance(s['distance'])})")
        if i < len(segs):
            out.append(f"     >> Transfer at {s['to']}: {s['line']} -> {segs[i]['line']}")
    ts = report["transfer_stations"]
    out.append(f"Transfer stations: {', '.join(ts) if ts else 'None'}")
    out.append(f"Lines involved: {', '.join(report['lines'])}")
    out.append(f"Number of transfers: {report['transfer_count']}")
    out.append(f"Total distance: {format_distance(report['distance']['meters'])}")
    out.append(f"Estimated time: {report['estimated_time']} minutes")
    out.append(format_fare(report["fare"]))
    out.append(sep)
    return "\n".join(out)


def format_json_report(report):
    return json.dumps(report, ensure_ascii=False, indent=2)

"""
Command-line interface of the Beijing Subway fare calculation system.

Usage:
    python src/main.py [--format text|json] [--output_file PATH]
                       [--mode shortest_distance|least_transfer|comprehensive]
                       [--max_transfers N] [--line_preference "Line 1"]
                       [--check_data]

Enter queries as "Starting Station; Ending Station". Type 'quit' to exit.
"""

import argparse
import json
import sys

from data_loader import DATA_FILE_PATH
from pathfinding import VALID_MODES, PathfindingConfig
from report import FareError, build_route_report, format_json_report, format_text_report
from subway_network import SubwayNetwork

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
EXIT_COMMANDS = {"quit", "exit", "q"}


class QueryError(Exception):
    """User-facing query error."""


def parse_query(raw):
    text = (raw or "").strip().replace("；", ";")
    if text.count(";") != 1:
        raise QueryError(FORMAT_ERROR)
    start, end = (part.strip() for part in text.split(";"))
    if not start or not end:
        raise QueryError(FORMAT_ERROR)
    return start, end


def process_query(network, raw, config):
    start_raw, end_raw = parse_query(raw)
    start = network.resolve_station(start_raw)
    if start is None:
        msg = f"Starting station '{start_raw}' does not exist, please check the station name"
        sugg = network.suggest_stations(start_raw)
        if sugg:
            msg += f" (did you mean: {', '.join(sugg)}?)"
        raise QueryError(msg)
    end = network.resolve_station(end_raw)
    if end is None:
        msg = f"Ending station '{end_raw}' does not exist, please check the station name"
        sugg = network.suggest_stations(end_raw)
        if sugg:
            msg += f" (did you mean: {', '.join(sugg)}?)"
        raise QueryError(msg)
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")
    try:
        route = network.find_route(start, end, config)
    except Exception as exc:  # path calculation exception handling
        raise QueryError(f"Path calculation error: {exc}") from exc
    if route is None:
        msg = f"No route found from '{start}' to '{end}'"
        if config.max_transfers is not None:
            msg += f" within {config.max_transfers} transfer(s)"
        raise QueryError(msg)
    try:
        return build_route_report(start, end, route, config)
    except FareError as exc:
        raise QueryError(f"Fare calculation error: {exc}") from exc


def run_check(network):
    result = network.check_data_integrity()
    print("Data integrity check")
    print("-" * 40)
    for k, v in result["stats"].items():
        print(f"  {k}: {v}")
    print(f"Errors: {len(result['errors'])}")
    for e in result["errors"]:
        print(f"  [ERROR] {e}")
    print(f"Warnings: {len(result['warnings'])}")
    for w in result["warnings"]:
        print(f"  [WARNING] {w}")
    print("Result: " + ("PASSED" if result["ok"] else "FAILED"))
    return 0 if result["ok"] else 1


def write_output(path, fmt, reports):
    with open(path, "w", encoding="utf-8") as fh:
        if fmt == "json":
            payload = reports[0] if len(reports) == 1 else reports
            json.dump(payload, fh, ensure_ascii=False, indent=2)
        else:
            fh.write("\n\n".join(format_text_report(r) for r in reports))
        fh.write("\n")


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text", help="Output format")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                   help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="Preferred line(s), comma separated, e.g. 'Line 1'")
    p.add_argument("--check_data", action="store_true",
                   help="Execute data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    return p


def main(argv=None):
    args = build_parser().parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer", file=sys.stderr)
        return 2

    try:
        network = SubwayNetwork(data_path=args.data_file)
    except (OSError, ValueError) as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        return run_check(network)

    preferred = []
    if args.line_preference:
        for item in args.line_preference.replace("，", ",").split(","):
            item = item.strip()
            if not item:
                continue
            resolved = network.resolve_line(item)
            if resolved is None:
                print(f"Warning: line '{item}' not found in network data, preference ignored")
            else:
                preferred.append(resolved)

    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                               preferred_lines=preferred)

    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' to query, 'quit' to exit.")
    reports = []
    while True:
        try:
            raw = input("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if not raw.strip():
            continue
        if raw.strip().lower() in EXIT_COMMANDS:
            break
        try:
            report = process_query(network, raw, config)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # last-resort safety net
            print(f"Error: unexpected failure: {exc}")
            continue
        if args.format == "json":
            print(format_json_report(report))
        else:
            print(format_text_report(report))
        if args.output_file:
            reports.append(report)
            try:
                write_output(args.output_file, args.format, reports)
                print(f"Result exported to {args.output_file}")
            except OSError as exc:
                print(f"Error: cannot write output file: {exc}")
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

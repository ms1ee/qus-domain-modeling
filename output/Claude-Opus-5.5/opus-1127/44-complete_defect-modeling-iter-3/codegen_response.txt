"""Loading of the Beijing Subway network data file.

The data file ``subway_data.json`` lives next to this module inside ``src/``.
``DATA_FILE_PATH`` is an absolute path, so it is valid from any working
directory.

Expected file layout::

    {
      "stations":    [{"name": str, "lines": [str, ...]}, ...],
      "connections": [{"source": str, "dest": str,
                       "distance": int (meters), "line": str}, ...]
    }
"""

import json
import os

# Absolute path of src/subway_data.json (independent of the current directory).
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway data file cannot be read or is malformed."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway network JSON file and return its content as a dict.

    :param path: path of the JSON data file (defaults to ``DATA_FILE_PATH``).
    :return: the parsed JSON content (a dict).
    :raises DataLoadError: if the file is missing, unreadable or malformed.
    """
    if path is None:
        path = DATA_FILE_PATH
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Subway data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Subway data file is not valid JSON: {path} ({exc})") from exc
    except OSError as exc:
        raise DataLoadError(f"Unable to read subway data file: {path} ({exc})") from exc

    if not isinstance(data, dict):
        raise DataLoadError("Subway data file must contain a JSON object at top level")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise DataLoadError("'stations' and 'connections' must be JSON arrays")
    return data

class Edge:
    """A directed, weighted edge between two station numbers.

    ``weight`` equals the inter-station distance in meters; ``line`` is the
    name of the subway line the connection belongs to.
    """

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source, dest, distance, line):
        if distance is None or distance < 0:
            raise ValueError(f"Edge distance must be non-negative, got {distance!r}")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        """Return one endpoint of the edge (the source)."""
        return self.source

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex} is not an endpoint of this edge")

    def reversed(self):
        """Return the same connection travelled in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source!r} -> {self.dest!r}, {self.weight}m, {self.line!r})"

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.weight, self.line) == (
            other.source, other.dest, other.weight, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.weight, self.line))


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Vertices are station numbers assigned by the caller.  Each undirected
    connection is stored as two directed ``Edge`` objects (one per direction)
    so that path reconstruction always yields correctly oriented edges.
    Multiple edges between the same pair of stations (on different lines)
    are allowed; this is how multi-line interchanges are represented.
    """

    def __init__(self):
        self.adj: Dict[int, List[Edge]] = {}
        self.station_lines: Dict[str, List[str]] = {}
        self.edges: List[Edge] = []  # original (undirected) edges as added

    # ------------------------------------------------------------------ nodes
    def add_vertex(self, vertex):
        """Add a vertex if not already present."""
        self.adj.setdefault(vertex, [])

    def has_vertex(self, vertex):
        return vertex in self.adj

    def vertices(self):
        return list(self.adj.keys())

    @property
    def V(self):
        """Number of vertices."""
        return len(self.adj)

    @property
    def E(self):
        """Number of undirected edges."""
        return len(self.edges)

    vertex_count = V
    edge_count = E

    def set_station_lines(self, station_name, lines):
        """Record the lines served by a station (by station name)."""
        self.station_lines[station_name] = list(lines or [])

    # ------------------------------------------------------------------ edges
    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        if not isinstance(edge, Edge):
            raise TypeError("add_edge expects an Edge instance")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.source != edge.dest:
            self.adj[edge.dest].append(edge.reversed())
        self.edges.append(edge)

    def adjacent(self, vertex):
        """Return the outgoing edges of ``vertex``."""
        return self.adj.get(vertex, [])

    def degree(self, vertex):
        return len(self.adj.get(vertex, []))

    def to_dict(self):
        """Return the adjacency lists as a plain dictionary.

        Format: ``{vertex: [(neighbour, distance, line), ...]}``
        """
        return {v: [(e.dest, e.weight, e.line) for e in edges] for v, edges in self.adj.items()}

    def __repr__(self):
        return f"Graph(V={self.V}, E={self.E})"

"""Shortest path computation (Dijkstra with a heap-based priority queue).

The search runs over states ``(station, current line[, transfers])`` so that
transfers can be counted exactly, limited (``max_transfers``) and penalised
(``least_transfer`` / ``comprehensive`` modes).  Complexity is
O((V+E) log V) in the number of states, which is a small constant multiple
of the number of stations.
"""

import math

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Equivalent distance penalty of one transfer in comprehensive mode (meters).
TRANSFER_PENALTY_METERS = 3000
# Weight multiplier applied to edges on preferred lines.
PREFERENCE_FACTOR = 0.8


class PathfindingConfig:
    """Configuration of a path query."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None,
                 transfer_penalty=TRANSFER_PENALTY_METERS, preference_factor=PREFERENCE_FACTOR):
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class MinPQ:
    """Binary min-heap priority queue.

    Items are stored with a monotonically increasing counter so that ties on
    the key never require comparing the items themselves.
    """

    def __init__(self):
        self._heap = []
        self._counter = 0

    def insert(self, key, item):
        entry = (key, self._counter, item)
        self._counter += 1
        heap = self._heap
        heap.append(entry)
        # sift up
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[i][:2] < heap[parent][:2]:
                heap[i], heap[parent] = heap[parent], heap[i]
                i = parent
            else:
                break

    push = insert

    def min(self):
        if not self._heap:
            raise IndexError("Priority queue is empty")
        key, _, item = self._heap[0]
        return key, item

    def del_min(self):
        """Remove and return ``(key, item)`` with the smallest key."""
        heap = self._heap
        if not heap:
            raise IndexError("Priority queue is empty")
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            # sift down
            n = len(heap)
            i = 0
            while True:
                left = 2 * i + 1
                if left >= n:
                    break
                smallest = left
                right = left + 1
                if right < n and heap[right][:2] < heap[left][:2]:
                    smallest = right
                if heap[smallest][:2] < heap[i][:2]:
                    heap[i], heap[smallest] = heap[smallest], heap[i]
                    i = smallest
                else:
                    break
        return top[0], top[2]

    pop = del_min

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)


class ShortestPath:
    """Single-source optimal paths using Dijkstra's algorithm.

    :param graph: a ``models.Graph``
    :param source: source station number
    :param config: optional ``PathfindingConfig``
    """

    def __init__(self, graph, source, config=None):
        if config is None:
            config = PathfindingConfig()
        if not graph.has_vertex(source):
            raise ValueError(f"Source station {source!r} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config
        self.dist_to = {v: math.inf for v in graph.vertices()}
        self.edge_to = {}
        self._best_state = {}
        self._parent = {}
        self._state_dist = {}
        self._state_wdist = {}
        self._state_transfers = {}
        self._run()

    # ------------------------------------------------------------------ core
    def _cost(self, weighted_distance, transfers):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, weighted_distance)
        if mode == "comprehensive":
            return (weighted_distance + transfers * self.config.transfer_penalty, transfers)
        return (weighted_distance, transfers)

    def _run(self):
        cfg = self.config
        limited = cfg.max_transfers is not None
        preferred = set(cfg.preferred_lines)
        factor = cfg.preference_factor

        start = (self.source, None, 0 if limited else None)
        self._state_dist[start] = 0
        self._state_wdist[start] = 0.0
        self._state_transfers[start] = 0
        best_cost = {start: self._cost(0.0, 0)}

        pq = MinPQ()
        pq.insert(best_cost[start], start)
        settled = set()

        while not pq.is_empty():
            cost, state = pq.del_min()
            if state in settled or cost > best_cost[state]:
                continue
            settled.add(state)
            node, line_in, _ = state
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = self._state_dist[state]
                if state in self._parent:
                    self.edge_to[node] = self._parent[state][1]

            t = self._state_transfers[state]
            wd = self._state_wdist[state]
            d = self._state_dist[state]
            for edge in self.graph.adjacent(node):
                nt = t + (1 if (line_in is not None and edge.line != line_in) else 0)
                if limited and nt > cfg.max_transfers:
                    continue
                ns = (edge.dest, edge.line, nt if limited else None)
                if ns in settled:
                    continue
                w = edge.weight * factor if (preferred and edge.line in preferred) else edge.weight
                nwd = wd + w
                nc = self._cost(nwd, nt)
                if ns not in best_cost or nc < best_cost[ns]:
                    best_cost[ns] = nc
                    self._state_wdist[ns] = nwd
                    self._state_dist[ns] = d + edge.weight
                    self._state_transfers[ns] = nt
                    self._parent[ns] = (state, edge)
                    pq.insert(nc, ns)

    # --------------------------------------------------------------- queries
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return ``(stations, edges)`` from the source to ``target``.

        ``stations`` is the ordered list of station numbers (including both
        endpoints); ``edges`` the ordered list of traversed edges.  Both lists
        are empty if ``target`` is unreachable.
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while state in self._parent:
            prev, edge = self._parent[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes on the path to ``target`` (-1 if unreachable)."""
        if target not in self._best_state:
            return -1
        return self._state_transfers[self._best_state[target]]

"""Beijing Subway fare calculation.

Fare structure (distance rounded up to whole kilometers):
  * 0 - 6 km   : 3 yuan
  * 6 - 12 km  : 4 yuan
  * 12 - 22 km : 5 yuan
  * 22 - 32 km : 6 yuan
  * over 32 km : +1 yuan for every (started) 20 km
"""

import math

from models import Fare, FareTier

FARE_TIERS = [
    FareTier(0, 6, 3),
    FareTier(6, 12, 4),
    FareTier(12, 22, 5),
    FareTier(22, 32, 6),
]
EXTENDED_BASE_KM = 32
EXTENDED_BASE_FARE = 6
EXTENDED_INCREMENT_KM = 20
EXTENDED_INCREMENT_FARE = 1

# Fare structure as a dictionary: (min_km, max_km) -> fare
FARE_STRUCTURE = {(t.min_distance_km, t.max_distance_km): t.fare for t in FARE_TIERS}

AVERAGE_SPEED_KMH = 35.0     # average operating speed including stops
TRANSFER_TIME_MINUTES = 5.0  # average time per transfer


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError(f"Distance must be a number (meters), got {type(distance).__name__}")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def distance_to_km(distance):
    """Convert meters to kilometers, rounding fractional kilometers up."""
    _validate_distance(distance)
    if isinstance(distance, int):
        return -(-distance // 1000)
    return int(math.ceil(distance / 1000.0))


def Sum_money(distance):
    """Return the fare in yuan for a trip of ``distance`` meters."""
    km = distance_to_km(distance)
    for tier in FARE_TIERS:
        if km <= tier.max_distance_km:
            return tier.fare
    extra = km - EXTENDED_BASE_KM
    increments = -(-extra // EXTENDED_INCREMENT_KM)
    return EXTENDED_BASE_FARE + increments * EXTENDED_INCREMENT_FARE


def calculate_fare(distance):
    """Return a ``Fare`` object for ``distance`` meters."""
    return Fare(amount=Sum_money(distance), distance_km=round(distance / 1000.0, 3))


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes for ``distance`` meters and transfers."""
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise ValueError("transfer_count must be a non-negative integer")
    minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0 + transfer_count * TRANSFER_TIME_MINUTES
    return float(round(minutes, 1))

class DataError(Exception):
    """Raised for invalid network data or invalid update operations."""


def normalize_station_key(name):
    """Normalized key for tolerant station name comparison."""
    return re.sub(r"[\s'’`\-_.]", "", str(name)).lower()


def normalize_line_key(name):
    """Normalized key for tolerant line name comparison ('Line 1' == '1号线')."""
    key = re.sub(r"[\s_\-]", "", str(name)).lower()
    for token in ("subway", "地铁", "号线", "line", "线"):
        key = key.replace(token, "")
    if key.isdigit():
        key = str(int(key))
    return key


class SubwayNetwork:
    """In-memory model of the Beijing Subway network."""

    def __init__(self, data=None, data_source=None):
        self.data_source = data_source or DATA_FILE_PATH
        if data is None:
            data = load_subway_data(self.data_source)
        if not isinstance(data, dict):
            raise DataError("Network data must be a dictionary")
        self.stations = {}            # name -> Station
        self.station_to_number = {}   # name -> number
        self.number_to_station = {}   # number -> name
        self.connections = []         # list[Connection]
        self.graph = Graph()
        self._next_number = 0
        self._load(data)
        self.rebuild_graph()

    # ----------------------------------------------------------- loading
    def _register_station(self, name, lines=None):
        name = str(name).strip()
        if not name:
            raise DataError("Station name cannot be empty")
        if name in self.stations:
            station = self.stations[name]
            for line in lines or []:
                if line not in station.lines:
                    station.lines.append(line)
            return station
        number = self._next_number
        self._next_number += 1
        station = Station(name=name, number=number, lines=list(lines or []))
        self.stations[name] = station
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        return station

    def _load(self, data):
        for item in data.get("stations", []):
            if not isinstance(item, dict) or "name" not in item:
                raise DataError(f"Invalid station entry: {item!r}")
            self._register_station(item["name"], item.get("lines", []))
        for item in data.get("connections", []):
            try:
                src = str(item["source"]).strip()
                dst = str(item["dest"]).strip()
                dist = item["distance"]
            except (KeyError, TypeError) as exc:
                raise DataError(f"Invalid connection entry: {item!r}") from exc
            line = item.get("line", "") or ""
            for name in (src, dst):
                if name not in self.stations:
                    self._register_station(name, [line] if line else [])
            self.connections.append(Connection(src, dst, dist, line))

    def rebuild_graph(self):
        """(Re)build the weighted graph from stations and connections."""
        graph = Graph()
        for name, station in self.stations.items():
            graph.add_vertex(station.number)
            graph.set_station_lines(name, station.lines)
        for conn in self.connections:
            if conn.source not in self.station_to_number or conn.dest not in self.station_to_number:
                continue
            if not isinstance(conn.distance, (int, float)) or conn.distance < 0:
                continue
            graph.add_edge(Edge(self.station_to_number[conn.source],
                                self.station_to_number[conn.dest], conn.distance, conn.line))
        self.graph = graph
        return graph

    # ----------------------------------------------------------- queries
    def adjacency_dict(self):
        """Weighted undirected graph as ``{station: [(neighbour, distance, line)]}``."""
        result = {name: [] for name in self.stations}
        for v, edges in self.graph.adj.items():
            name = self.number_to_station[v]
            result[name] = [(self.number_to_station[e.dest], e.weight, e.line) for e in edges]
        return result

    def get_station_number(self, name):
        return self.station_to_number.get(name)

    def get_station_name(self, number):
        return self.number_to_station.get(number)

    def is_transfer_station(self, name):
        station = self.stations.get(name)
        if station is None:
            return False
        lines = set(station.lines)
        lines.update(c.line for c in self.connections if name in (c.source, c.dest) and c.line)
        return len(lines) > 1

    def resolve_station_name(self, raw):
        """Return the official station name matching ``raw`` or None."""
        if raw is None:
            return None
        cleaned = re.sub(r"\s+", " ", str(raw)).strip()
        if cleaned in self.stations:
            return cleaned
        key = normalize_station_key(cleaned)
        if not key:
            return None
        for name in self.stations:
            if normalize_station_key(name) == key:
                return name
        return None

    def suggest_stations(self, raw, n=3):
        return difflib.get_close_matches(str(raw).strip(), list(self.stations), n=n, cutoff=0.6)

    def all_line_names(self):
        names = set()
        for station in self.stations.values():
            names.update(l for l in station.lines if l)
        names.update(c.line for c in self.connections if c.line)
        return sorted(names)

    def resolve_line_names(self, raw):
        """Return the data line names matching the user supplied ``raw`` name."""
        if raw is None:
            return []
        raw = str(raw).strip()
        lines = self.all_line_names()
        if raw in lines:
            return [raw]
        lowered = [l for l in lines if l.lower() == raw.lower()]
        if lowered:
            return lowered
        key = normalize_line_key(raw)
        return [l for l in lines if normalize_line_key(l) == key]

    def _derive_line_sequence(self, line):
        adj = defaultdict(list)
        order = []
        for conn in self.connections:
            if conn.line != line:
                continue
            for a, b in ((conn.source, conn.dest), (conn.dest, conn.source)):
                if b not in adj[a]:
                    adj[a].append(b)
            for s in (conn.source, conn.dest):
                if s not in order:
                    order.append(s)
        if not adj:
            return [n for n, s in self.stations.items() if line in s.lines]
        ends = [s for s in order if len(adj[s]) == 1]
        sequence, visited = [], set()
        for start in ends + order:
            if start in visited:
                continue
            stack = [start]
            while stack:
                node = stack.pop()
                if node in visited:
                    continue
                visited.add(node)
                sequence.append(node)
                for nxt in reversed(adj[node]):
                    if nxt not in visited:
                        stack.append(nxt)
        return sequence

    @property
    def lines(self):
        """Dictionary ``line name -> Line`` (station sequence and transfers)."""
        result = {}
        for line in self.all_line_names():
            seq = self._derive_line_sequence(line)
            transfers = [s for s in seq if self.is_transfer_station(s)]
            result[line] = Line(name=line, station_sequence=seq, transfer_stations=transfers)
        return result

    # ----------------------------------------------------------- updates
    def add_station(self, name, lines=None):
        station = self._register_station(name, lines)
        self.rebuild_graph()
        return station

    def add_connection(self, source, dest, distance, line):
        if source == dest:
            raise DataError("A connection must link two different stations")
        if not isinstance(distance, (int, float)) or isinstance(distance, bool) or distance <= 0:
            raise DataError("Distance must be a positive number of meters")
        self._register_station(source, [line] if line else [])
        self._register_station(dest, [line] if line else [])
        self.connections.append(Connection(source, dest, distance, line))
        self.rebuild_graph()

    def add_line(self, line_name, station_names, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if not line_name:
            raise DataError("Line name cannot be empty")
        if len(station_names) < 2:
            raise DataError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise DataError("Number of distances must equal number of stations minus one")
        for d in distances:
            if not isinstance(d, (int, float)) or isinstance(d, bool) or d <= 0:
                raise DataError("Distances must be positive numbers of meters")
        for name in station_names:
            self._register_station(name, [line_name])
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.connections.append(Connection(a, b, d, line_name))
        self.rebuild_graph()

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        if name not in self.stations:
            raise DataError(f"Station '{name}' does not exist")
        station = self.stations[name]
        if new_name is not None and new_name != name:
            new_name = new_name.strip()
            if not new_name:
                raise DataError("New station name cannot be empty")
            if new_name in self.stations:
                raise DataError(f"Station '{new_name}' already exists")
            del self.stations[name]
            del self.station_to_number[name]
            station.name = new_name
            self.stations[new_name] = station
            self.station_to_number[new_name] = station.number
            self.number_to_station[station.number] = new_name
            for conn in self.connections:
                if conn.source == name:
                    conn.source = new_name
                if conn.dest == name:
                    conn.dest = new_name
        if lines is not None:
            station.lines = list(lines)
        self.rebuild_graph()
        return station

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance of the connection(s) between two stations."""
        if not isinstance(distance, (int, float)) or isinstance(distance, bool) or distance <= 0:
            raise DataError("Distance must be a positive number of meters")
        updated = 0
        for conn in self.connections:
            if {conn.source, conn.dest} == {source, dest} and (line is None or conn.line == line):
                conn.distance = distance
                updated += 1
        if not updated:
            raise DataError(f"No connection between '{source}' and '{dest}'")
        self.rebuild_graph()
        return updated

    def to_dict(self):
        return {
            "stations": [{"name": s.name, "lines": list(s.lines)}
                         for s in sorted(self.stations.values(), key=lambda s: s.number)],
            "connections": [{"source": c.source, "dest": c.dest, "distance": c.distance, "line": c.line}
                            for c in self.connections],
        }

    def save(self, path=None):
        with open(path or self.data_source, "w", encoding="utf-8") as handle:
            json.dump(self.to_dict(), handle, ensure_ascii=False, indent=2)


class DataIntegrityChecker:
    """Checks coverage, connection integrity and distance accuracy."""

    def __init__(self, network):
        self.network = network
        self.coverage_valid = None
        self.connection_valid = None
        self.distance_valid = None
        self.errors = []
        self.warnings = []
        self.info = []

    def check_coverage(self):
        net = self.network
        lines = net.all_line_names()
        line_keys = {normalize_line_key(l): l for l in lines}
        line_stations = net.lines
        ok = True
        for main, aliases in MAIN_LINES.items():
            found = next((line_keys[normalize_line_key(a)] for a in aliases
                          if normalize_line_key(a) in line_keys), None)
            if found is None:
                ok = False
                self.errors.append(f"Main line missing: {main}")
            else:
                count = len(line_stations[found].station_sequence)
                if count < 2:
                    ok = False
                    self.errors.append(f"Line '{found}' has fewer than 2 stations")
                else:
                    self.info.append(f"Line '{found}': {count} stations")
        station_keys = {normalize_station_key(s) for s in net.stations}
        for line, aliases in REPRESENTATIVE_STATIONS.items():
            if not any(normalize_station_key(a) in station_keys for a in aliases):
                ok = False
                self.errors.append(f"Representative station of {line} missing: {aliases[0]}")
        self.coverage_valid = ok
        return ok

    def check_connections(self):
        net = self.network
        ok = True
        seen = {}
        connected = set()
        for conn in net.connections:
            if conn.source not in net.stations or conn.dest not in net.stations:
                ok = False
                self.errors.append(f"Connection references unknown station: {conn.source} - {conn.dest}")
                continue
            if conn.source == conn.dest:
                ok = False
                self.errors.append(f"Self-loop connection at station '{conn.source}'")
                continue
            if not conn.line:
                self.warnings.append(f"Connection {conn.source} - {conn.dest} has no line")
            else:
                for s in (conn.source, conn.dest):
                    if net.stations[s].lines and conn.line not in net.stations[s].lines:
                        self.warnings.append(f"Station '{s}' is not listed on line '{conn.line}'")
            key = (frozenset((conn.source, conn.dest)), conn.line)
            if key in seen:
                self.warnings.append(f"Duplicate connection {conn.source} - {conn.dest} ({conn.line})")
            seen[key] = conn
            connected.update((conn.source, conn.dest))
        for name in net.stations:
            if name not in connected:
                self.warnings.append(f"Station '{name}' has no connections")
        # Connectivity of the whole network
        vertices = net.graph.vertices()
        if vertices:
            start = vertices[0]
            visited = {start}
            queue = deque([start])
            while queue:
                v = queue.popleft()
                for e in net.graph.adjacent(v):
                    if e.dest not in visited:
                        visited.add(e.dest)
                        queue.append(e.dest)
            unreachable = len(vertices) - len(visited)
            if unreachable:
                self.warnings.append(f"Network is not fully connected: {unreachable} stations unreachable "
                                     f"from '{net.number_to_station[start]}'")
        self.connection_valid = ok
        return ok

    def check_distances(self):
        ok = True
        pair_distances = defaultdict(set)
        for conn in self.network.connections:
            d = conn.distance
            if isinstance(d, bool) or not isinstance(d, (int, float)):
                ok = False
                self.errors.append(f"Non-numeric distance {conn.source} - {conn.dest}: {d!r}")
                continue
            if d < MIN_VALID_DISTANCE or d > MAX_VALID_DISTANCE:
                ok = False
                self.errors.append(f"Distance out of range {conn.source} - {conn.dest}: {d} m")
                continue
            if d < SUSPICIOUS_SHORT_DISTANCE:
                self.warnings.append(f"Unusually short distance {conn.source} - {conn.dest}: {d} m")
            elif d > SUSPICIOUS_LONG_DISTANCE:
                self.warnings.append(f"Unusually long distance {conn.source} - {conn.dest}: {d} m")
            pair_distances[(frozenset((conn.source, conn.dest)), conn.line)].add(d)
        for (pair, line), values in pair_distances.items():
            if len(values) > 1:
                self.warnings.append(f"Inconsistent distances for {' - '.join(sorted(pair))} ({line}): "
                                     f"{sorted(values)}")
        self.distance_valid = ok
        return ok

    def run_all(self):
        self.errors, self.warnings, self.info = [], [], []
        self.check_coverage()
        self.check_connections()
        self.check_distances()
        return {
            "coverage_valid": self.coverage_valid,
            "connection_valid": self.connection_valid,
            "distance_valid": self.distance_valid,
            "valid": bool(self.coverage_valid and self.connection_valid and self.distance_valid),
            "errors": list(self.errors),
            "warnings": list(self.warnings),
            "station_count": len(self.network.stations),
            "connection_count": len(self.network.connections),
            "line_count": len(self.network.all_line_names()),
        }

    def format_report(self, result=None):
        result = result or self.run_all()
        status = lambda ok: "PASS" if ok else "FAIL"
        lines = [
            "Data Integrity Check Report",
            "=" * 40,
            f"Stations: {result['station_count']}, Connections: {result['connection_count']}, "
            f"Lines: {result['line_count']}",
            f"Station coverage check:   {status(result['coverage_valid'])}",
            f"Connection integrity:     {status(result['connection_valid'])}",
            f"Distance accuracy check:  {status(result['distance_valid'])}",
        ]
        if result["errors"]:
            lines.append("Errors:")
            lines.extend(f"  - {e}" for e in result["errors"])
        if result["warnings"]:
            lines.append(f"Warnings ({len(result['warnings'])}):")
            lines.extend(f"  - {w}" for w in result["warnings"][:50])
            if len(result["warnings"]) > 50:
                lines.append(f"  ... {len(result['warnings']) - 50} more")
        lines.append(f"Overall result: {status(result['valid'])}")
        return "\n".join(lines)

import json

def build_route(network, stations, edges):
    """Build a ``Route`` from station numbers and edges produced by ShortestPath."""
    names = [network.number_to_station[n] for n in stations]
    segments = []
    for edge in edges:
        src = network.number_to_station[edge.source]
        dst = network.number_to_station[edge.dest]
        if segments and segments[-1]["line"] == edge.line:
            segments[-1]["stations"].append(dst)
            segments[-1]["distance"] += edge.weight
        else:
            segments.append({"line": edge.line, "stations": [src, dst], "distance": edge.weight})
    transfer_stations = [seg["stations"][0] for seg in segments[1:]]
    involved = []
    for seg in segments:
        if seg["line"] not in involved:
            involved.append(seg["line"])
    total = sum(e.weight for e in edges)
    transfers = max(len(segments) - 1, 0)
    return Route(
        station_sequence=names,
        total_distance=total,
        transfer_count=transfers,
        involved_lines=involved,
        estimated_time=calculate_estimated_time(total, transfers),
        fare=Sum_money(total),
        segments=segments,
        transfer_stations=transfer_stations,
    )


def generate_summary(route):
    return PathSummary(
        total_distance_km=round(route.total_distance / 1000.0, 2),
        total_fare=route.fare,
        transfer_count=route.transfer_count,
        estimated_time=route.estimated_time,
    )


def format_distance(meters):
    """Meters for short distances, kilometers for long ones."""
    if meters < 1000:
        return f"{meters:g} m"
    return f"{meters / 1000.0:.2f} km"


def format_text_report(route, summary=None, query=None):
    summary = summary or generate_summary(route)
    query = query or {}
    start = route.station_sequence[0] if route.station_sequence else ""
    end = route.station_sequence[-1] if route.station_sequence else ""
    out = ["=" * 60, f"Route: {start} -> {end}"]
    if query.get("mode"):
        out.append(f"Mode: {query['mode']}")
    out.append("-" * 60)
    out.append(f"Station sequence ({len(route.station_sequence)} stations):")
    out.append("  " + " -> ".join(route.station_sequence))
    out.append("Line segments:")
    for i, seg in enumerate(route.segments, 1):
        if i > 1:
            prev = route.segments[i - 2]["line"]
            out.append(f"  ** Transfer at {seg['stations'][0]}: {prev} -> {seg['line']}")
        out.append(f"  [{i}] {seg['line']}: {' -> '.join(seg['stations'])} "
                   f"({len(seg['stations']) - 1} stops, {format_distance(seg['distance'])})")
    out.append("Transfer stations: " + (", ".join(route.transfer_stations) or "None"))
    out.append("Lines involved: " + (", ".join(route.involved_lines) or "None"))
    out.append(f"Number of transfers: {summary.transfer_count}")
    out.append(f"Total distance: {format_distance(route.total_distance)} "
               f"({route.total_distance} meters / {summary.total_distance_km:.2f} km)")
    out.append(f"Estimated time: {summary.estimated_time:.1f} minutes")
    out.append(f"Fare: {summary.total_fare} yuan")
    out.append("=" * 60)
    return "\n".join(out)


def route_to_dict(route, summary=None, query=None):
    summary = summary or generate_summary(route)
    return {
        "query": dict(query or {}),
        "path": list(route.station_sequence),
        "segments": [{"line": s["line"], "stations": list(s["stations"]), "distance": s["distance"]}
                     for s in route.segments],
        "distance": {"meters": route.total_distance, "kilometers": summary.total_distance_km,
                     "display": format_distance(route.total_distance)},
        "fare": summary.total_fare,
        "fare_display": f"Fare: {summary.total_fare} yuan",
        "estimated_time": summary.estimated_time,
        "transfer_count": summary.transfer_count,
        "transfer_stations": list(route.transfer_stations),
        "lines": list(route.involved_lines),
    }


def format_json_report(route, summary=None, query=None):
    return json.dumps(route_to_dict(route, summary, query), ensure_ascii=False, indent=2)

"""Command-line interface of the Beijing Subway fare calculation system.

Usage::

    python main.py [--output_file FILE] [--format text|json]
                   [--mode shortest_distance|least_transfer|comprehensive]
                   [--max_transfers N] [--line_preference "Line 1"] [--check_data]

Enter queries as ``Starting Station; Ending Station``; type ``q`` to quit.
"""

import argparse
import json
import sys

from data_loader import DATA_FILE_PATH, DataLoadError
from data_manager import DataError, DataIntegrityChecker, SubwayNetwork
from pathfinding import VALID_MODES, PathfindingConfig, ShortestPath
from report import build_route, format_json_report, format_text_report, generate_summary, route_to_dict

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
SAME_STATION_ERROR = "Starting station cannot be the same as the ending station"
QUIT_COMMANDS = {"q", "quit", "exit"}


class QueryError(Exception):
    """User-facing query error."""


def parse_args(argv=None):
    parser = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", default=None, help="Export query results to this file")
    parser.add_argument("--format", default="text", choices=["text", "json"], help="Output format")
    parser.add_argument("--mode", default="shortest_distance", choices=list(VALID_MODES),
                        help="Path query mode")
    parser.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line(s), e.g. 'Line 1' (comma separated for several)")
    parser.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    parser.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")
    return args


def parse_query_input(raw):
    """Split 'Start; End' into two station names (raises QueryError)."""
    if raw is None:
        raise QueryError(FORMAT_ERROR)
    text = raw.replace("；", ";").strip()
    parts = text.split(";")
    if len(parts) != 2:
        raise QueryError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryError(FORMAT_ERROR)
    return start, end


def validate_stations(network, start, end):
    """Return the official names of start and end stations (raises QueryError)."""
    resolved_start = network.resolve_station_name(start)
    if resolved_start is None:
        msg = f"Starting station '{start}' does not exist, please check the station name"
        hints = network.suggest_stations(start)
        if hints:
            msg += f" (did you mean: {', '.join(hints)}?)"
        raise QueryError(msg)
    resolved_end = network.resolve_station_name(end)
    if resolved_end is None:
        msg = f"Ending station '{end}' does not exist, please check the station name"
        hints = network.suggest_stations(end)
        if hints:
            msg += f" (did you mean: {', '.join(hints)}?)"
        raise QueryError(msg)
    if resolved_start == resolved_end:
        raise QueryError(SAME_STATION_ERROR)
    return resolved_start, resolved_end


def resolve_preferred_lines(network, line_preference, err=sys.stderr):
    if not line_preference:
        return []
    result = []
    for item in line_preference.split(","):
        item = item.strip()
        if not item:
            continue
        matches = network.resolve_line_names(item)
        if not matches:
            print(f"Warning: preferred line '{item}' not found in network data, ignored", file=err)
        result.extend(m for m in matches if m not in result)
    return result


def process_query(network, start, end, config):
    """Compute the route for two official station names. Returns (route, summary)."""
    try:
        sp = ShortestPath(network.graph, network.station_to_number[start], config)
        stations, edges = sp.path_with_edges(network.station_to_number[end])
    except Exception as exc:  # path calculation exception handling
        raise QueryError(f"Path calculation error: {exc}") from exc
    if not stations:
        if config.max_transfers is not None:
            raise QueryError(f"No route found from '{start}' to '{end}' within "
                             f"{config.max_transfers} transfer(s)")
        raise QueryError(f"No route found from '{start}' to '{end}'")
    try:
        route = build_route(network, stations, edges)
        summary = generate_summary(route)
    except Exception as exc:  # fare calculation exception handling
        raise QueryError(f"Fare calculation error: {exc}") from exc
    return route, summary


def _write_output(args, text_report, json_results):
    try:
        if args.format == "json":
            payload = json_results[0] if len(json_results) == 1 else json_results
            with open(args.output_file, "w", encoding="utf-8") as handle:
                json.dump(payload, handle, ensure_ascii=False, indent=2)
        else:
            with open(args.output_file, "a", encoding="utf-8") as handle:
                handle.write(text_report + "\n\n")
        return True
    except OSError as exc:
        print(f"Error: unable to write output file '{args.output_file}': {exc}")
        return False


def run_interactive(network, args, config):
    print("Beijing Subway Fare Calculation System")
    print(f"Loaded {len(network.stations)} stations, {len(network.connections)} connections, "
          f"{len(network.all_line_names())} lines.")
    print("Enter 'Starting Station; Ending Station' to query, or 'q' to quit.")
    json_results = []
    if args.output_file and args.format == "text":
        try:
            open(args.output_file, "w", encoding="utf-8").close()
        except OSError as exc:
            print(f"Error: unable to write output file '{args.output_file}': {exc}")
    while True:
        try:
            raw = input("\nQuery> ")
        except (EOFError, KeyboardInterrupt):
            print("\nGoodbye!")
            break
        raw = raw.strip()
        if not raw:
            continue
        if raw.lower() in QUIT_COMMANDS:
            print("Goodbye!")
            break
        try:
            start, end = parse_query_input(raw)
            start, end = validate_stations(network, start, end)
            route, summary = process_query(network, start, end, config)
        except QueryError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # never crash the query loop
            print(f"Error: unexpected failure: {exc}")
            continue
        query_info = {
            "start": start,
            "end": end,
            "raw_input": raw,
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "line_preference": config.preferred_lines,
        }
        text_report = format_text_report(route, summary, query_info)
        if args.format == "json":
            print(format_json_report(route, summary, query_info))
        else:
            print(text_report)
        if args.output_file:
            json_results.append(route_to_dict(route, summary, query_info))
            if _write_output(args, text_report, json_results):
                print(f"Result exported to {args.output_file}")
    return 0


def main(argv=None):
    args = parse_args(argv)
    try:
        network = SubwayNetwork(data_source=args.data_file)
    except (DataLoadError, DataError) as exc:
        print(f"Error: {exc}")
        return 1

    if args.check_data:
        checker = DataIntegrityChecker(network)
        result = checker.run_all()
        print(checker.format_report(result))
        return 0 if result["valid"] else 1

    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=resolve_preferred_lines(network, args.line_preference,
                                                                           sys.stdout))
    except ValueError as exc:
        print(f"Error: {exc}")
        return 1
    return run_interactive(network, args, config)


if __name__ == "__main__":
    sys.exit(main())

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass(frozen=True)
class User:
    """A participant in the lottery."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize configuration."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if isinstance(self.associated_drawer, str):
            drawer = self.associated_drawer.strip()
            self.associated_drawer = drawer or None

import os
from typing import Dict, Iterable, List, Tuple

from lottery_system.models import User

DEFAULT_LIST_LINES = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Wang Wu,Technology Department",
    "Zhao Liu,Technology Department",
    "Qian Qi,Product Department",
    "Sun Ba,Product Department",
    "Zhou Jiu,Product Department",
    "Wu Shi,Marketing Department",
    "Zheng Yi,Marketing Department",
    "Feng Er,Marketing Department",
    "Chen Yu,Human Resources Department",
    "Chu Wei,Human Resources Department",
    "Wei Xiang,Finance Department",
    "Jiang Lan,Finance Department",
    "Cheng Ru,Cheng Ru group",
    "Shen Yue,Cheng Ru group",
    "Han Mei,Cheng Ru group",
    "Yang Fan,Cheng Ru group",
    "Zhu Lin,Operations Department",
    "Qin Feng,Operations Department",
]


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.users: List[User] = []

    # ------------------------------------------------------------------ loading
    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().replace("\uff0c", ",")
            if not line or line.startswith("#"):
                continue
            parts = line.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Format Error: Line {idx}: '{raw.strip()}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            name, dept = parts[0].strip(), parts[1].strip()
            if not name or not dept:
                errors.append(
                    f"Format Error: Line {idx}: '{raw.strip()}' "
                    f"(name and department group must not be empty)"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.users = users
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File Error: file not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as fh:
                    lines = fh.read().splitlines()
            except (OSError, UnicodeDecodeError) as exc:
                return [], [f"File Error: cannot read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        users, errors = self._parse_lines(lines)
        self.users = users
        return users, errors

    # --------------------------------------------------------------- validation
    @staticmethod
    def validate_users(users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    @staticmethod
    def get_group_info(users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize configuration management."""

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Manages the prize library and the draw order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else name
        if not name:
            raise ValueError("Prize name must not be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError(f"Prize quantity must be positive: {quantity}")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize already exists: {name}")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def set_avoid_groups(self, index: int, groups) -> bool:
        """Set avoidance groups for prize at 1-based index."""
        if not 1 <= index <= len(self.prizes):
            return False
        self.prizes[index - 1].avoid_groups = [g for g in groups if g]
        return True

    def remove_prize(self, index: int) -> Optional[Prize]:
        if not 1 <= index <= len(self.prizes):
            return None
        return self.prizes.pop(index - 1)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idxs = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idxs) != n or sorted(idxs) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idxs]
        return True

"""Result storage, summary and export."""

import csv
import os
from datetime import datetime
from typing import Dict, List

LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
    ("consolation", 6),
    ("participation", 7),
]
UNKNOWN_LEVEL = 100


class Winner:
    """A single winning record."""

    def __init__(self, user, prize, time=None) -> None:
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()

    @property
    def prize_name(self) -> str:
        return self.prize.name if hasattr(self.prize, "name") else str(self.prize)

    def __repr__(self) -> str:
        return f"Winner({self.user!r}, {self.prize_name!r}, {self.time:%Y-%m-%d %H:%M:%S})"


def prize_level(name: str) -> int:
    lower = name.lower()
    for kw, level in LEVEL_KEYWORDS:
        if kw in lower:
            return level
    return UNKNOWN_LEVEL


class ResultManager:
    """Keeps winning records."""

    def __init__(self) -> None:
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> Winner:
        winner = Winner(user, prize)
        self.winners.append(winner)
        return winner

    def _sorted_prize_names(self, prizes) -> List[str]:
        names: List[str] = []
        for p in prizes:
            n = p.name if hasattr(p, "name") else str(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        indexed = list(enumerate(names))
        indexed.sort(
            key=lambda t: (
                prize_level(t[1]),
                -t[0] if prize_level(t[1]) == UNKNOWN_LEVEL else t[0],
            )
        )
        return [n for _, n in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for name in self._sorted_prize_names(prizes):
            ws = [w for w in self.winners if w.prize_name == name]
            ws.sort(key=lambda w: w.time)  # stable: keeps insertion order on ties
            result[name] = ws
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        writer.writerow(
                            [w.user.name, w.user.department, name,
                             w.time.strftime("%Y-%m-%d %H:%M:%S")]
                        )
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and round execution."""

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(text) -> str:
    return " ".join(str(text or "").strip().lower().split())


def _is_protected(user: User) -> bool:
    return _norm(user.department) == _norm(PROTECTED_GROUP)


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryManager:
    """Executes draws according to the rule engine."""

    def __init__(self, users, result_manager: Optional[ResultManager] = None) -> None:
        self.users: List[User] = list(users)
        self.result_manager = result_manager or ResultManager()
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []

    # ------------------------------------------------------------- rule engine
    def _fallback_active(self, prize: Prize) -> bool:
        return (
            _norm(prize.associated_drawer) == _norm(FALLBACK_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize) -> List[User]:
        base = [u for u in self.users if u.name not in self.winner_names]
        if self._fallback_active(prize):
            protected = [u for u in base if _is_protected(u)]
            if protected:
                return protected
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return base
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in base if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        drawer = prize.associated_drawer or "N/A"
        if self._fallback_active(prize):
            return (f"Drawer: {drawer}. Fallback mechanism active: pool limited to "
                    f"'{PROTECTED_GROUP}' members (avoidance groups ignored).")
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return (f"Drawer: {drawer}. Special rule: only previous winners are "
                    f"excluded; avoidance groups are not applied.")
        groups = ", ".join(prize.avoid_groups) if prize.avoid_groups else "None"
        return f"Drawer: {drawer}. Excludes previous winners; avoidance groups: {groups}."

    def _record_drawer_result(self, prize: Prize, winners: List[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in WATCHED_DRAWERS}:
            self.last_two_draw_results.append(any(_is_protected(u) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_DRAWER):
            self.last_two_draw_results = []

    # --------------------------------------------------------------- execution
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winner_names.add(user.name)
        self.result_manager.add_winner(user, prize)
        return user

    def run_round(
        self,
        prize: Prize,
        command_reader: Optional[Callable[[str], str]] = None,
        printer: Callable[[str], None] = print,
    ) -> List[User]:
        """Run one prize round. command_reader=None draws automatically."""
        results: List[User] = []
        while len(results) < prize.quantity:
            n = len(results) + 1
            if command_reader is not None:
                cmd = command_reader(
                    f"[Enter] draw {ordinal(n)} winner / 'pause' / 'terminate': "
                ).strip().lower()
                if cmd == "pause":
                    printer("Lottery paused.")
                    terminated = False
                    while True:
                        c = command_reader("Paused. Type 'continue' or 'terminate': ").strip().lower()
                        if c == "continue":
                            printer("Lottery resumed.")
                            break
                        if c == "terminate":
                            terminated = True
                            break
                        printer("Invalid command while paused.")
                    if terminated:
                        printer(f"Round '{prize.name}' terminated.")
                        break
                    continue
                if cmd == "terminate":
                    printer(f"Round '{prize.name}' terminated.")
                    break
                if cmd not in ("", "continue", "draw"):
                    printer("Unknown command. Use Enter, 'pause', 'continue' or 'terminate'.")
                    continue
            printer(f"Currently drawing the {ordinal(n)} winner...")
            user = self.draw_one(prize)
            if user is None:
                printer("No eligible participants remain for this prize. Round ends.")
                break
            results.append(user)
            printer(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        self._record_drawer_result(prize, results)
        return results

"""Command-line interface for the lottery system."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class App:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery = None

    # ----------------------------------------------------------------- lists
    def _accept_list(self, users, errors) -> None:
        for e in errors:
            print("  " + e)
        valid, dups = self.list_manager.validate_users(users)
        for d in dups:
            print(f"  Duplicate name detected: {d.name} ({d.department})")
        if errors or dups:
            while True:
                c = input("Invalid data found. 'r' remove invalid data and continue / "
                          "'t' terminate upload: ").strip().lower()
                if c == "r":
                    break
                if c == "t":
                    print("Upload terminated. List unchanged.")
                    return
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        self.lottery = None
        print(f"Loaded {len(valid)} valid participants.")

    def menu_list(self) -> None:
        while True:
            print("\n-- List Management --\n1. Load default list\n2. Upload custom txt list\n"
                  "3. View groups\n4. View participants\n5. Remove a participant\n0. Back")
            c = input("> ").strip()
            if c == "1":
                self._accept_list(*self.list_manager.load_default_list())
            elif c == "2":
                path = input("Path of txt file (each line 'Name,Department Group'): ").strip()
                self._accept_list(*self.list_manager.load_custom_list(path))
            elif c == "3":
                info = self.list_manager.get_group_info(self.users)
                if not info:
                    print("No list loaded.")
                for i, (g, n) in enumerate(info.items(), 1):
                    print(f"  {i}. {g}: {n} people")
            elif c == "4":
                for i, u in enumerate(self.users, 1):
                    print(f"  {i}. {u.name} ({u.department})")
            elif c == "5":
                name = input("Name to remove: ").strip()
                before = len(self.users)
                self.users = [u for u in self.users if u.name != name]
                print("Removed." if len(self.users) < before else "Not found.")
                self.lottery = None
            elif c == "0":
                return

    # ---------------------------------------------------------------- prizes
    def _select_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            raw = input("Avoidance groups (numbers or names, comma-separated, empty=none): ")
        else:
            raw = input("Avoidance groups (names, comma-separated, empty=none): ")
        result = []
        for part in raw.replace("\uff0c", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                part = groups[int(part) - 1]
            if part not in result:
                result.append(part)
        return result

    def _show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | drawer: {p.associated_drawer or 'N/A'} | "
                  f"avoid: {', '.join(p.avoid_groups) or 'None'}")

    def menu_prize(self) -> None:
        while True:
            print("\n-- Prize Configuration --\n1. Load preset prizes\n2. Add custom prize\n"
                  "3. Set avoidance groups\n4. Reorder prizes\n5. View prizes\n"
                  "6. Remove prize\n0. Back")
            c = input("> ").strip()
            try:
                if c == "1":
                    for e in self.prize_manager.load_default_prizes():
                        print("  " + e)
                    self._show_prizes()
                elif c == "2":
                    name = input("Prize name: ")
                    qty = input("Quantity: ")
                    drawer = input("Associated lottery person (empty=none): ").strip() or None
                    groups = self._select_groups()
                    self.prize_manager.add_prize(name, qty, groups, drawer)
                    print("Prize added.")
                elif c == "3":
                    self._show_prizes()
                    idx = int(input("Prize number: "))
                    ok = self.prize_manager.set_avoid_groups(idx, self._select_groups())
                    print("Updated." if ok else "Invalid prize number.")
                elif c == "4":
                    self._show_prizes()
                    raw = input("New order (e.g. 3,1,2): ").replace(" ", ",")
                    order = [int(x) for x in raw.split(",") if x.strip()]
                    print("Reordered." if self.prize_manager.reorder_prizes(order)
                          else "Invalid order.")
                    self._show_prizes()
                elif c == "5":
                    self._show_prizes()
                elif c == "6":
                    self._show_prizes()
                    p = self.prize_manager.remove_prize(int(input("Prize number: ")))
                    print(f"Removed {p.name}." if p else "Invalid prize number.")
                elif c == "0":
                    return
            except ValueError as exc:
                print(f"Error: {exc}")

    # --------------------------------------------------------------- lottery
    def run_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users, self.result_manager)
        auto = input("Manual control mode? (y/n, default y): ").strip().lower() == "n"
        reader = None if auto else input
        for rnd, prize in enumerate(prizes, 1):
            print(f"\n===== Round {rnd}: {prize.name} (quantity: {prize.quantity}) =====")
            print("Rules: " + self.lottery.describe_rules(prize))
            results = self.lottery.run_round(prize, reader)
            print(f"--- Results of {prize.name} ---")
            for i, u in enumerate(results, 1):
                print(f"  {i}. {u.name} ({u.department})")
            if not results:
                print("  No winners.")
        self.show_summary()

    def show_summary(self) -> None:
        print("\n===== Complete Winner List =====")
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not any(data.values()):
            print("No winners yet.")
            return
        for name, ws in data.items():
            if not ws:
                continue
            print(f"[{name}]")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {name} - "
                      f"{w.time:%Y-%m-%d %H:%M:%S}")

    def run(self) -> None:
        print("=== Company Team Building Event Smart Lottery System ===")
        while True:
            print("\n1. List management\n2. Prize configuration\n3. Start lottery\n"
                  "4. View results\n5. Export results\n0. Exit")
            c = input("> ").strip()
            if c == "1":
                self.menu_list()
            elif c == "2":
                self.menu_prize()
            elif c == "3":
                self.run_lottery()
            elif c == "4":
                self.show_summary()
            elif c == "5":
                ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
                print(msg)
            elif c == "0":
                print("Goodbye!")
                return


def main() -> None:
    try:
        App().run()
    except (KeyboardInterrupt, EOFError):
        print("\nExited.")


if __name__ == "__main__":
    main()

"""Core data models for the lottery system."""


class User:
    """A participant in the lottery."""

    def __init__(self, name, department):
        self.name = str(name).strip()
        self.department = str(department).strip()

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))


class Prize:
    """A prize with its quantity, avoidance groups and associated lottery person."""

    def __init__(self, name, quantity, avoid_groups=None, associated_drawer=None):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups = list(avoid_groups) if avoid_groups else []
        if associated_drawer is not None:
            associated_drawer = str(associated_drawer).strip() or None
        self.associated_drawer = associated_drawer

    def __repr__(self):
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
            f"avoid_groups={self.avoid_groups!r}, "
            f"associated_drawer={self.associated_drawer!r})"
        )

class ListManager:
    """Loads and validates participant lists."""

    def __init__(self):
        self.users = []

    # ------------------------------------------------------------------ load
    def load_default_list(self):
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        return users, errors

    def load_custom_list(self, path):
        if path is None or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path):
        try:
            try:
                with open(path, "r", encoding="utf-8-sig") as f:
                    content = f.read()
            except UnicodeDecodeError:
                with open(path, "r", encoding="gbk", errors="replace") as f:
                    content = f.read()
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        return self._parse_lines(content.splitlines())

    @staticmethod
    def _parse_lines(lines):
        users = []
        errors = []
        for index, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = normalized.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Format Error: Line {index}: '{raw.rstrip()}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            name, dept = parts[0].strip(), parts[1].strip()
            if not name or not dept:
                errors.append(
                    f"Format Error: Line {index}: '{raw.rstrip()}' "
                    f"(name and department group must not be empty)"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    # -------------------------------------------------------------- validate
    def validate_users(self, users):
        valid = []
        duplicates = []
        seen = set()
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    # ---------------------------------------------------------------- groups
    def get_group_info(self, users):
        info = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),

class PrizeManager:
    """Manages the prize library and the lottery round order."""

    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def _find(self, name):
        for p in self.prizes:
            if p.name == name:
                return p
        return None

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = str(name).strip() if name is not None else ""
        if not name:
            raise ValueError("Prize name must not be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if self._find(name) is not None:
            raise ValueError(f"Prize '{name}' already exists")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def set_avoid_groups(self, index, groups):
        if not 0 <= index < len(self.prizes):
            return False
        cleaned = []
        for g in groups or []:
            g = str(g).strip()
            if g and g not in cleaned:
                cleaned.append(g)
        self.prizes[index].avoid_groups = cleaned
        return True

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        n = len(self.prizes)
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

"""Lottery rule engine."""

import random

LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
CHENGRU_GROUP = "Cheng Ru group"


def _norm(text):
    return "".join(str(text or "").lower().split())


def is_chengru(department):
    d = _norm(department).replace("_", "").replace("-", "")
    return d in ("chengrugroup", "chengru")


def _same_person(a, b):
    return _norm(a) == _norm(b)


class LotteryManager:
    """Implements filtering, special rules, fallback and anti-duplication."""

    def __init__(self, users, result_manager=None, rng=None):
        self.users = list(users)
        self.result_manager = result_manager
        self.rng = rng or random.SystemRandom()
        self.winner_names = set()
        self.winners = []
        self.last_two_draw_results = []

    # ------------------------------------------------------------- rules
    def _fallback_active(self, prize):
        return (
            _same_person(prize.associated_drawer, MA_JIAQI)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize):
        candidates = [u for u in self.users if u.name not in self.winner_names]
        if self._fallback_active(prize):
            return [u for u in candidates if is_chengru(u.department)]
        if _same_person(prize.associated_drawer, LI_BIN):
            return candidates
        avoid = {_norm(g) for g in prize.avoid_groups}
        return [u for u in candidates if _norm(u.department) not in avoid]

    def describe_rule(self, prize):
        parts = []
        drawer = prize.associated_drawer or "N/A"
        parts.append(f"Lottery person: {drawer}")
        if self._fallback_active(prize):
            parts.append(
                "Fallback mechanism active: prize pool limited to "
                f"'{CHENGRU_GROUP}' members (avoidance groups ignored)"
            )
        elif _same_person(prize.associated_drawer, LI_BIN):
            parts.append(
                "Special rule (Li Bin): only previous winners are excluded, "
                "avoidance groups are not applied"
            )
        else:
            if prize.avoid_groups:
                parts.append("Avoidance groups: " + ", ".join(prize.avoid_groups))
            else:
                parts.append("Avoidance groups: none")
            parts.append("Previous winners are excluded")
        return "; ".join(parts)

    # -------------------------------------------------------------- draw
    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        if user.name in self.winner_names:  # defensive anti-duplication
            return None
        self.winner_names.add(user.name)
        self.winners.append((user, prize))
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return user

    def complete_round(self, prize, round_winners):
        """Update the fallback tracking after a round has ended."""
        drawer = prize.associated_drawer
        if _same_person(drawer, CHEN_QIXIAN) or _same_person(drawer, JIN_DUO):
            hit = any(is_chengru(u.department) for u in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif _same_person(drawer, MA_JIAQI):
            self.last_two_draw_results = []

"""Result storage, summary and export."""

import csv
import os
import re
from datetime import datetime

_LEVEL_WORDS = [
    (r"\b(special|grand)\b", 0),
    (r"\bfirst\b", 1),
    (r"\bsecond\b", 2),
    (r"\bthird\b", 3),
    (r"\bfourth\b", 4),
    (r"\bfifth\b", 5),
    (r"\bsixth\b", 6),
    (r"\b(consolation|participation|lucky)\b", 50),
]


def _prize_level(name):
    lower = str(name).lower()
    for pattern, level in _LEVEL_WORDS:
        if re.search(pattern, lower):
            return level
    return None


class Winner:
    def __init__(self, user, prize_name, time, seq):
        self.user = user
        self.prize_name = prize_name
        self.time = time
        self.seq = seq

    def __repr__(self):
        return f"Winner(user={self.user!r}, prize={self.prize_name!r})"


class ResultManager:
    def __init__(self):
        self.winners = []
        self._seq = 0

    def add_winner(self, user, prize):
        prize_name = getattr(prize, "name", prize)
        self._seq += 1
        w = Winner(user, str(prize_name), datetime.now(), self._seq)
        self.winners.append(w)
        return w

    def _ordered_prize_names(self, prizes):
        names = []
        for p in prizes or []:
            n = getattr(p, "name", p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        count = len(names)
        keyed = []
        for idx, n in enumerate(names):
            level = _prize_level(n)
            if level is not None:
                keyed.append(((0, level, idx), n))
            else:
                # Unknown level: later rounds are considered higher prizes.
                keyed.append(((1, count - idx, idx), n))
        keyed.sort(key=lambda x: x[0])
        return [n for _, n in keyed]

    def get_winners_by_prize(self, prizes):
        result = {}
        for name in self._ordered_prize_names(prizes):
            ws = [w for w in self.winners if w.prize_name == name]
            ws.sort(key=lambda w: (w.time, w.seq))
            result[name] = ws
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winning results to export."
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        writer.writerow([
                            w.user.name,
                            w.user.department,
                            name,
                            w.time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

import time

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class _Exit(Exception):
    pass


def _input(prompt):
    try:
        return input(prompt)
    except EOFError:
        raise _Exit()


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None

    # ------------------------------------------------------------ list
    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        for e in errors:
            print("  " + e)
        for d in dups:
            print(f"  Duplicate name removed: {d.name} ({d.department})")
        self.users = valid
        self.lottery = None
        print(f"Default list loaded: {len(valid)} participants.")

    def load_custom(self):
        path = _input("Enter path of txt list file: ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if errors and not users:
            for e in errors:
                print("  " + e)
            print("No valid data loaded.")
            return
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Problems detected in the list:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate name: {d.name} ({d.department})")
            while True:
                c = _input("Input 'r' to remove invalid data and continue, "
                           "'t' to terminate upload: ").strip().lower()
                if c == "t":
                    print("Upload terminated.")
                    return
                if c == "r":
                    break
        if not valid:
            print("No valid participants in the list.")
            return
        self.users = valid
        self.lottery = None
        print(f"Custom list loaded: {len(valid)} participants.")

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current valid groups:")
        for dept, cnt in info.items():
            print(f"  {dept}: {cnt} people")

    # ----------------------------------------------------------- prizes
    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            avoid = ", ".join(p.avoid_groups) if p.avoid_groups else "none"
            print(f"  {i}. {p.name} x{p.quantity} | drawer: "
                  f"{p.associated_drawer or 'N/A'} | avoid: {avoid}")

    def load_default_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prize library loaded.")

    def _select_groups(self):
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = _input("Enter avoidance groups (numbers or names, comma separated, "
                     "empty for none): ").strip()
        result = []
        for token in raw.replace("，", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                token = groups[int(token) - 1]
            if token not in result:
                result.append(token)
        return result

    def add_prize(self):
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        drawer = _input("Associated lottery person (empty for none): ").strip() or None
        groups = self._select_groups()
        try:
            self.prize_manager.add_prize(name, qty, groups, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def set_avoid(self):
        self.view_prizes()
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            return
        raw = _input("Prize number: ").strip()
        if not raw.isdigit() or not 1 <= int(raw) <= len(prizes):
            print("Invalid prize number.")
            return
        groups = self._select_groups()
        self.prize_manager.set_avoid_groups(int(raw) - 1, groups)
        print("Avoidance groups updated.")

    def reorder(self):
        self.view_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = _input("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.view_prizes()
        else:
            print("Invalid order: must be a permutation of all serial numbers.")

    # ---------------------------------------------------------- lottery
    def start_lottery(self):
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users, self.result_manager)
        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            print("Rule: " + self.lottery.describe_rule(prize))
            print("Commands: [Enter]=draw, pause, continue, terminate")
            round_winners = []
            paused = False
            terminated = False
            while len(round_winners) < prize.quantity:
                cmd = _input("> ").strip().lower()
                if cmd == "pause":
                    paused = True
                    print("Lottery paused. Input 'continue' to resume.")
                    continue
                if cmd == "continue":
                    if paused:
                        print("Lottery resumed.")
                    paused = False
                    continue
                if cmd == "terminate":
                    terminated = True
                    print("Current round terminated.")
                    break
                if paused:
                    print("Lottery is paused. Input 'continue' to resume "
                          "or 'terminate' to skip this round.")
                    continue
                n = len(round_winners) + 1
                print(f"Currently drawing the {n}th winner...")
                time.sleep(0.2)
                user = self.lottery.draw_one(prize)
                if user is None:
                    print("No eligible candidates remain for this prize.")
                    break
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) "
                      f"for winning {prize.name}")
            self.lottery.complete_round(prize, round_winners)
            print(f"--- Round {rnd} results ({prize.name}) ---")
            if round_winners:
                for i, u in enumerate(round_winners, 1):
                    print(f"  {i}. {u.name} ({u.department})")
            else:
                print("  No winners.")
            if terminated:
                continue
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        if not any(data.values()):
            print("No winners yet.")
            return
        for name, ws in data.items():
            print(f"[{name}]")
            if not ws:
                print("  (none)")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {name} - "
                      f"{w.time.strftime('%H:%M:%S')}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # -------------------------------------------------------------- run
    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_default_prizes, "5": self.add_prize, "6": self.set_avoid,
            "7": self.reorder, "8": self.view_prizes, "9": self.start_lottery,
            "10": self.show_summary, "11": self.export,
        }
        menu = """
====== Team Building Smart Lottery System ======
 1. Load default list          2. Upload custom txt list
 3. View groups                4. Load preset prizes
 5. Add custom prize           6. Set prize avoidance groups
 7. Reorder prizes             8. View prizes
 9. Start lottery             10. View result summary
11. Export results             0. Exit"""
        try:
            while True:
                print(menu)
                choice = _input("Select: ").strip()
                if choice == "0":
                    break
                action = actions.get(choice)
                if action:
                    action()
                else:
                    print("Invalid choice.")
        except (_Exit, KeyboardInterrupt):
            print()
        print("Goodbye!")


def main():
    LotteryApp().run()


if __name__ == "__main__":
    main()

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if isinstance(self.associated_drawer, str):
            self.associated_drawer = self.associated_drawer.strip() or None

    def rule_description(self) -> str:
        parts = ["Previous winners are excluded"]
        drawer = (self.associated_drawer or "").strip()
        if drawer.lower() == "li bin":
            parts.append("special drawer Li Bin: avoidance groups are NOT applied")
        elif self.avoid_groups:
            parts.append("avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("no avoidance groups")
        if drawer.lower() == "ma jiaqi":
            parts.append("Cheng Ru group protection may apply")
        return "; ".join(parts)

    ("Zhao Liu", "Technology Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),

    ("Zheng Shi", "Product Department"),
    ("Qian Yi", "Marketing Department"),
    ("Feng Er", "Marketing Department"),
    ("Chen San", "Marketing Department"),
    ("Chu Si", "Marketing Department"),
    ("Wei Wu", "Human Resources Department"),
    ("Jiang Liu", "Human Resources Department"),
    ("Shen Qi", "Finance Department"),
    ("Han Ba", "Finance Department"),
    ("Yang Jiu", "Cheng Ru Group"),
    ("Zhu Shi", "Cheng Ru Group"),
    ("Qin Yi", "Cheng Ru Group"),
    ("You Er", "Cheng Ru Group"),
    ("Xu San", "Cheng Ru Group"),

class ListManager:
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return [User(n, d) for n, d in DEFAULT_LIST], []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File Error: file not found '{path}'"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.readlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in line.replace("，", ",").split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            key = u.name.strip()
            if key in seen:
                duplicates.append(u)
            else:
                seen.add(key)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

"""Prize configuration management."""

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, groups, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(groups), drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None,
                  associated_drawer: Optional[str] = None) -> Prize:
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be positive")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "li bin"
WATCHED_DRAWERS = ("chen qixian", "jin duo")
PROTECTED_DRAWER = "ma jiaqi"
PROTECTED_GROUP = "cheng ru group"


def _norm(s) -> str:
    return (s or "").strip().lower()


def is_protected_member(user: User) -> bool:
    return _norm(user.department) == PROTECTED_GROUP


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []

    def _available(self) -> List[User]:
        return [u for u in self.users if u.name not in self.winner_names]

    def is_fallback_active(self, prize: Prize) -> bool:
        return (_norm(prize.associated_drawer) == PROTECTED_DRAWER
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        available = self._available()
        if self.is_fallback_active(prize):
            return [u for u in available if is_protected_member(u)]
        if _norm(prize.associated_drawer) == SPECIAL_DRAWER:
            return available
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in available if _norm(u.department) not in avoid]

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.mark_winner(user)
        return user

    def mark_winner(self, user: User) -> bool:
        if user.name in self.winner_names:
            return False
        self.winner_names.add(user.name)
        self.winners.append(user)
        return True

    def finish_round(self, prize: Prize, round_winners) -> None:
        if _norm(prize.associated_drawer) in WATCHED_DRAWERS:
            contains = any(is_protected_member(u) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

"""Result storage, display and export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime
from typing import Any, Dict, List, Tuple

from lottery_system.models import User


def _prize_name(prize) -> str:
    return prize.name if hasattr(prize, "name") else str(prize)


@dataclass
class Winner:
    user: User
    prize: Any

    @property
    def prize_name(self) -> str:
        return _prize_name(self.prize)


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self.winners.append(Winner(user, prize))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes or []:
            result.setdefault(_prize_name(p), [])
        for w in self.winners:
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes, path=None) -> Tuple[bool, str]:
        if path is None:
            fname = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
            path = os.path.join(os.getcwd(), fname)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for pname, ws in grouped.items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, pname])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _input(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        return "terminate"


class LotteryCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_done = False

    # ---------- list ----------
    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._accept_users(users, errors)

    def load_custom(self):
        path = _input("Enter txt list path: ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            print("Upload failed.")
            return
        self._accept_users(users, errors)

    def _accept_users(self, users, errors):
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Invalid data detected:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            choice = _input("Type 'r' to remove invalid data and continue, "
                            "'t' to terminate upload: ").strip().lower()
            if choice != "r":
                print("Upload terminated.")
                return
        self.users = valid
        print(f"Loaded {len(valid)} valid participants.")

    def view_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        print("Current valid groups:")
        for g, c in info.items():
            print(f"  {g}: {c} people")

    # ---------- prizes ----------
    def _choose_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            raw = _input("Select avoidance groups by number (comma separated) "
                         "or type names, blank for none: ").strip()
        else:
            raw = _input("Enter avoidance group names (comma separated), "
                         "blank for none: ").strip()
        result = []
        for part in raw.replace("，", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                g = groups[int(part) - 1]
            else:
                g = part
            if g not in result:
                result.append(g)
        return result

    def add_prize(self):
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        groups = self._choose_groups()
        drawer = _input("Associated lottery person (blank for none): ").strip()
        try:
            self.prize_manager.add_prize(name, qty, groups, drawer or None)
            print("Prize added.")
        except ValueError as e:
            print(f"Error: {e}")

    def load_default_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prizes loaded.")

    def set_avoid_groups(self):
        prizes = self.prize_manager.get_prizes()
        if not self.list_prizes():
            return
        raw = _input("Select prize number: ").strip()
        if not raw.isdigit() or not 1 <= int(raw) <= len(prizes):
            print("Invalid selection.")
            return
        prizes[int(raw) - 1].avoid_groups = self._choose_groups()
        print("Avoidance groups updated.")

    def list_prizes(self) -> bool:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return False
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | drawer: "
                  f"{p.associated_drawer or '-'} | avoid: "
                  f"{', '.join(p.avoid_groups) or '-'}")
        return True

    def reorder(self):
        if not self.list_prizes():
            return
        raw = _input("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Order updated.")
            self.list_prizes()
        else:
            print("Invalid order.")

    # ---------- lottery ----------
    def run_lottery(self):
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        self.result_manager = ResultManager()
        engine = LotteryManager(self.users)
        print("Commands during lottery: Enter = draw next, 'pause', "
              "'continue', 'terminate' (end current round).")
        for rnd, prize in enumerate(prizes, 1):
            print(f"\n=== Round {rnd}: {prize.name} | Quantity: {prize.quantity}"
                  f" | Drawer: {prize.associated_drawer or '-'} ===")
            print(f"Rules: {prize.rule_description()}")
            if engine.is_fallback_active(prize):
                print("Fallback activated: prize pool limited to Cheng Ru group.")
            round_winners = []
            paused = False
            terminated = False
            i = 0
            while i < prize.quantity:
                cmd = _input("[paused] > " if paused else
                             f"(draw {i + 1}/{prize.quantity}) > ").strip().lower()
                if cmd == "terminate":
                    print("Current round terminated.")
                    terminated = True
                    break
                if cmd == "pause":
                    paused = True
                    print("Lottery paused. Type 'continue' to resume.")
                    continue
                if cmd == "continue":
                    if paused:
                        print("Lottery resumed.")
                    paused = False
                    continue
                if paused:
                    print("Lottery is paused. Type 'continue' or 'terminate'.")
                    continue
                print(f"Currently drawing the {i + 1}th winner...")
                user = engine.draw_one(prize)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) "
                      f"for winning {prize.name}")
                i += 1
            engine.finish_round(prize, round_winners)
            print(f"--- Round {rnd} results ({prize.name}) ---")
            if round_winners:
                for n, u in enumerate(round_winners, 1):
                    print(f"  {n}. {u.name} ({u.department})")
            else:
                print("  No winners.")
            if terminated:
                continue
        self.lottery_done = True
        self.show_summary()

    def show_summary(self):
        grouped = self.result_manager.get_winners_by_prize(
            self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        any_w = False
        for pname, ws in grouped.items():
            for w in ws:
                any_w = True
                print(f"  {w.user.name}, {w.user.department}, {pname}")
        if not any_w:
            print("  No winners yet.")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ---------- main loop ----------
    def run(self):
        actions = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.view_groups),
            "4": ("Load preset prizes", self.load_default_prizes),
            "5": ("Add custom prize", self.add_prize),
            "6": ("Set prize avoidance groups", self.set_avoid_groups),
            "7": ("View prizes", self.list_prizes),
            "8": ("Reorder prizes", self.reorder),
            "9": ("Start lottery", self.run_lottery),
            "10": ("View winner list", self.show_summary),
            "11": ("Export results", self.export),
        }
        while True:
            print("\n==== Team Building Smart Lottery ====")
            for k, (label, _) in actions.items():
                print(f" {k}. {label}")
            print(" 0. Exit")
            choice = _input("Select: ").strip()
            if choice in ("0", "terminate", "exit", "quit"):
                print("Bye.")
                return
            action = actions.get(choice)
            if action:
                action[1]()
            else:
                print("Invalid option.")


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

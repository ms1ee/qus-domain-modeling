"""Grid-based path planning on an 800x800 map made of 100x100 cells.

Streets are the grid lines (x or y being a multiple of 100). Restaurants and
customers live at cell centres. Riders leave a cell centre onto the nearest
useful street, travel along streets (straight lines and right-angle turns)
and finally enter the destination cell centre.
"""

GRID_SIZE = 100
GRID_COUNT = 8
MAP_SIZE = GRID_SIZE * GRID_COUNT


def _manhattan(a, b):
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class PathPlanner:
    """Collection of path-planning helpers (all static)."""

    @staticmethod
    def validate_coordinate(value):
        """Validate a single raw coordinate (integer in 0..800)."""
        if isinstance(value, bool) or not isinstance(value, int):
            raise ValueError("coordinate must be an integer: %r" % (value,))
        if value < 0 or value > MAP_SIZE:
            raise ValueError("coordinate out of range 0-%d: %d"
                             % (MAP_SIZE, value))
        return value

    @staticmethod
    def to_grid_index(value):
        """Convert a raw coordinate into a grid index 0..7 (800 -> 7)."""
        PathPlanner.validate_coordinate(value)
        return min(value // GRID_SIZE, GRID_COUNT - 1)

    @staticmethod
    def to_grid_center(x, y):
        """Convert raw coordinates into the centre of their grid cell."""
        gx = PathPlanner.to_grid_index(x)
        gy = PathPlanner.to_grid_index(y)
        return (gx * GRID_SIZE + GRID_SIZE // 2,
                gy * GRID_SIZE + GRID_SIZE // 2)

    @staticmethod
    def is_on_vertical_street(p):
        return p[0] % GRID_SIZE == 0

    @staticmethod
    def is_on_horizontal_street(p):
        return p[1] % GRID_SIZE == 0

    @staticmethod
    def is_on_street(p):
        """Return True if the point lies on a street line."""
        return (PathPlanner.is_on_vertical_street(p)
                or PathPlanner.is_on_horizontal_street(p))

    @staticmethod
    def same_cell(a, b):
        """Return True if two off-street points are inside the same cell."""
        return (a[0] // GRID_SIZE == b[0] // GRID_SIZE
                and a[1] // GRID_SIZE == b[1] // GRID_SIZE)

    @staticmethod
    def is_vertically_aligned(a, b):
        """Special case: target directly above or below (same column)."""
        return (a[0] == b[0] and a[1] != b[1]
                and not PathPlanner.is_on_vertical_street(a))

    @staticmethod
    def _projections(p):
        """Candidate street entry points for point ``p``."""
        x, y = int(p[0]), int(p[1])
        if PathPlanner.is_on_street((x, y)):
            return [(x, y)]
        opts = []
        fx = (x // GRID_SIZE) * GRID_SIZE
        fy = (y // GRID_SIZE) * GRID_SIZE
        for vx in (fx, fx + GRID_SIZE):
            if 0 <= vx <= MAP_SIZE:
                opts.append((vx, y))
        for vy in (fy, fy + GRID_SIZE):
            if 0 <= vy <= MAP_SIZE:
                opts.append((x, vy))
        return opts

    @staticmethod
    def _best_line(p, q):
        """Pick a street line (multiple of 100) minimising |p-h|+|q-h|."""
        cands = set()
        for v in (p, q):
            f = (v // GRID_SIZE) * GRID_SIZE
            for c in (f, f + GRID_SIZE):
                if 0 <= c <= MAP_SIZE:
                    cands.add(c)
        return min(sorted(cands), key=lambda h: (abs(p - h) + abs(q - h),
                                                 abs(p - h)))

    @staticmethod
    def _dedup(start, points):
        out = []
        prev = start
        for pt in points:
            pt = (int(pt[0]), int(pt[1]))
            if pt != prev:
                out.append(pt)
                prev = pt
        return out

    @staticmethod
    def _street_route(a, b):
        """Route between two street points along streets only."""
        if a == b:
            return []
        on_v = PathPlanner.is_on_vertical_street
        on_h = PathPlanner.is_on_horizontal_street
        if on_v(a) and on_v(b) and a[0] == b[0]:
            return [b]
        if on_h(a) and on_h(b) and a[1] == b[1]:
            return [b]
        if on_v(a) and on_h(b):
            return PathPlanner._dedup(a, [(a[0], b[1]), b])
        if on_h(a) and on_v(b):
            return PathPlanner._dedup(a, [(b[0], a[1]), b])
        if on_v(a) and on_v(b):
            h = PathPlanner._best_line(a[1], b[1])
            return PathPlanner._dedup(a, [(a[0], h), (b[0], h), b])
        v = PathPlanner._best_line(a[0], b[0])
        return PathPlanner._dedup(a, [(v, a[1]), (v, b[1]), b])

    @staticmethod
    def route_length(start, waypoints):
        """Total Manhattan length of ``start`` followed by ``waypoints``."""
        total = 0
        prev = start
        for pt in waypoints:
            total += _manhattan(prev, pt)
            prev = pt
        return total

    @staticmethod
    def plan(start, end):
        """Plan a street-following route.

        Returns a list of waypoints (excluding ``start``, including ``end``).
        Every segment is horizontal or vertical.
        """
        start = (int(start[0]), int(start[1]))
        end = (int(end[0]), int(end[1]))
        if start == end:
            return []
        if (not PathPlanner.is_on_street(start)
                and not PathPlanner.is_on_street(end)
                and PathPlanner.same_cell(start, end)):
            return PathPlanner._dedup(start, [(end[0], start[1]), end])
        best = None
        best_len = None
        for s1 in PathPlanner._projections(start):
            for t1 in PathPlanner._projections(end):
                route = PathPlanner._dedup(
                    start, [s1] + PathPlanner._street_route(s1, t1) + [end])
                length = PathPlanner.route_length(start, route)
                if best is None or length < best_len:
                    best, best_len = route, length
        return best or [end]

    @staticmethod
    def path_length(start, end):
        """Length of the planned route between two points."""
        return PathPlanner.route_length(
            (int(start[0]), int(start[1])), PathPlanner.plan(start, end))

"""Rider model: movement, state machine and event log."""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_START = (400, 400)
DEFAULT_SPEED = 100


class Rider:
    """A delivery rider.

    Orders are assigned by appending ``Order`` objects to ``rider.orders``.
    Each call to :meth:`move` advances the rider by one tick.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        cfg = globalvar.get_value("config", {}) or {}
        start = cfg.get("rider_start", DEFAULT_START)
        self.rider_id = rider_id
        self.x = int(start[0] if x is None else x)
        self.y = int(start[1] if y is None else y)
        if speed is None:
            speed = globalvar.get_value("rider_speed", DEFAULT_SPEED)
        self.speed = max(1, int(speed))
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.path = []
        self.current_order = None
        self.log = []
        self.total_distance = 0
        self.completed_orders = 0
        self._log("spawn", None)

    # ------------------------------------------------------------------ log
    def _log(self, event, order_id):
        entry = {
            "time": globalvar.get_value("time", 0),
            "rider_id": self.rider_id,
            "event": event,
            "order_id": order_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("rider %s: %s", self.rider_id, entry)

    # ------------------------------------------------------------ properties
    @property
    def position(self):
        return (self.x, self.y)

    @property
    def state_name(self):
        return STATE_NAMES.get(self.state, "unknown")

    # -------------------------------------------------------------- movement
    def _set_path_to(self, target):
        self.special_case = PathPlanner.is_vertically_aligned(
            self.position, target)
        self.path = PathPlanner.plan(self.position, target)

    def _advance(self):
        """Move up to ``speed`` units along the current path."""
        remaining = self.speed
        while remaining > 0 and self.path:
            tx, ty = self.path[0]
            dx, dy = tx - self.x, ty - self.y
            dist = abs(dx) + abs(dy)
            if dist <= remaining:
                self.x, self.y = tx, ty
                remaining -= dist
                self.total_distance += dist
                self.path.pop(0)
                continue
            if dx != 0:
                step = min(remaining, abs(dx))
                self.x += step if dx > 0 else -step
            else:
                step = min(remaining, abs(dy))
                self.y += step if dy > 0 else -step
            remaining -= step
            self.total_distance += step

    def move(self):
        """Advance one tick, performing state transitions and logging."""
        try:
            self._move()
        except Exception:  # error recovery: keep the simulation alive
            logger.exception("rider %s failed to move; recovering",
                             self.rider_id)
            self._recover()

    def _recover(self):
        if self.current_order is not None and \
                self.current_order not in self.orders:
            self.current_order = None
        if self.current_order is None:
            self.state = STATE_IDLE
            self.path = []

    def _move(self):
        if self.state == STATE_IDLE:
            if not self.orders:
                return
            order = self.orders[0]
            self.current_order = order
            if hasattr(order, "mark_accepted"):
                order.mark_accepted(self.rider_id)
            self.state = STATE_PICKING_UP
            self._set_path_to(order.res_pos)
            self._log("accepted_order", order.id)
            return

        order = self.current_order
        if order is None:
            # inconsistent state: recover to idle
            self.state = STATE_IDLE
            self.path = []
            return

        self._advance()
        if self.path:
            return

        if self.state == STATE_PICKING_UP:
            if hasattr(order, "mark_picked_up"):
                order.mark_picked_up()
            self.state = STATE_DELIVERING
            self._log("arrived_at_merchant", order.id)
            self._set_path_to(order.cus_pos)
        elif self.state == STATE_DELIVERING:
            self._complete(order)

    def _complete(self, order):
        now = globalvar.get_value("time", 0)
        if hasattr(order, "mark_completed"):
            order.mark_completed(now)
        else:
            order.status = "completed"
        fee = getattr(order, "fee", 0)
        globalvar.set_value("funds", globalvar.get_value("funds", 0) + fee)
        if order in self.orders:
            self.orders.remove(order)
        self.current_order = None
        self.path = []
        self.special_case = False
        self.state = STATE_IDLE
        self.completed_orders += 1
        self._log("delivered_to_customer", order.id)

    # ------------------------------------------------------------- planning
    def full_route(self):
        """All remaining waypoints (starting at current position)."""
        pos = self.position
        pts = [pos]
        if self.state != STATE_IDLE and self.current_order is not None:
            pts.extend(self.path)
            if self.path:
                pos = self.path[-1]
            if self.state == STATE_PICKING_UP:
                seg = PathPlanner.plan(pos, self.current_order.cus_pos)
                pts.extend(seg)
                pos = self.current_order.cus_pos
            queued = [o for o in self.orders if o is not self.current_order]
        else:
            queued = list(self.orders)
        for o in queued:
            pts.extend(PathPlanner.plan(pos, o.res_pos))
            pts.extend(PathPlanner.plan(o.res_pos, o.cus_pos))
            pos = o.cus_pos
        return pts

    def workload(self):
        """Return (remaining distance, end position) of queued work."""
        pts = self.full_route()
        return PathPlanner.route_length(pts[0], pts[1:]), pts[-1]

    def estimate_cost(self, target):
        """Distance this rider must cover before reaching ``target``."""
        dist, end = self.workload()
        return dist + PathPlanner.path_length(end, target)

    def to_dict(self):
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "total_distance": self.total_distance,
            "completed_orders": self.completed_orders,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%s,%s), state=%s, orders=%s)" % (
            self.rider_id, self.x, self.y, self.state,
            [o.id for o in self.orders])

import globalvar
from path_planner import PathPlanner

BASE_FEE = 50


def _validate_pos(pos, name):
    if not isinstance(pos, (list, tuple)) or len(pos) != 2:
        raise ValueError("%s must be a pair of coordinates" % name)
    for v in pos:
        PathPlanner.validate_coordinate(v)
    return (int(pos[0]), int(pos[1]))


class Order:
    """A delivery order from a restaurant to a customer.

    Raw coordinates are converted to grid cell centres.
    Status lifecycle: pending -> assigned -> picking_up -> delivering
    -> completed.
    """

    def __init__(self, res_pos, cus_pos):
        raw_res = _validate_pos(res_pos, "restaurant position")
        raw_cus = _validate_pos(cus_pos, "customer position")
        self.raw_res = raw_res
        self.raw_cus = raw_cus
        self.res_pos = PathPlanner.to_grid_center(*raw_res)
        self.cus_pos = PathPlanner.to_grid_center(*raw_cus)
        self.res_grid = (PathPlanner.to_grid_index(raw_res[0]),
                         PathPlanner.to_grid_index(raw_res[1]))
        self.cus_grid = (PathPlanner.to_grid_index(raw_cus[0]),
                         PathPlanner.to_grid_index(raw_cus[1]))
        new_id = globalvar.get_value("order_id_counter", 0) + 1
        globalvar.set_value("order_id_counter", new_id)
        self.id = new_id
        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("time", 0)
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.complete_time = None
        self.time_taken = None
        self.distance = PathPlanner.path_length(self.res_pos, self.cus_pos)
        self.fee = BASE_FEE + self.distance // 10

    def assign_to(self, rider_id):
        """Mark this order as assigned to ``rider_id``."""
        self.rider_id = rider_id
        self.status = "assigned"
        self.assign_time = globalvar.get_value("time", 0)

    def mark_accepted(self, rider_id=None):
        if rider_id is not None:
            self.rider_id = rider_id
        self.status = "picking_up"
        self.accept_time = globalvar.get_value("time", 0)

    def mark_picked_up(self):
        self.status = "delivering"
        self.pickup_time = globalvar.get_value("time", 0)

    def mark_completed(self, now=None):
        if now is None:
            now = globalvar.get_value("time", 0)
        self.status = "completed"
        self.complete_time = now
        self.time_taken = now - self.create_time

    @property
    def efficiency(self):
        """Distance per tick of total order time (None until completed)."""
        if not self.time_taken:
            return None
        return round(self.distance / self.time_taken, 2)

    def to_list(self):
        return [self.res_pos[0], self.res_pos[1],
                self.cus_pos[0], self.cus_pos[1]]

    def to_dict(self):
        return {
            "id": self.id,
            "raw": [self.raw_res[0], self.raw_res[1],
                    self.raw_cus[0], self.raw_cus[1]],
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "create_time": self.create_time,
            "accept_time": self.accept_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "fee": self.fee,
        }

    def __repr__(self):
        return "Order(id=%s, res=%s, cus=%s, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

import json
import logging
import os

import globalvar
from rider import Rider

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 300,
    "hire_cost": 300,
    "rider_speed": 100,
    "rider_start": [400, 400],
    "max_riders": 50,
    "state_file": "dispatch_state.json",
}


def load_config(path=None):
    """Load a JSON configuration file, merged over the defaults."""
    cfg = dict(DEFAULT_CONFIG)
    if path and os.path.isfile(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                cfg.update({k: v for k, v in data.items() if k in cfg})
        except (OSError, ValueError) as exc:
            logger.warning("could not load config %s: %s", path, exc)
    return cfg


def init_system(config_path=None):
    """Initialise all global system parameters."""
    cfg = load_config(config_path)
    globalvar._init()
    globalvar.set_value("config", cfg)
    globalvar.set_value("funds", int(cfg["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)
    globalvar.set_value("rider_speed", int(cfg["rider_speed"]))
    return cfg


def _ensure_init():
    if globalvar.get_value("riders") is None:
        init_system()


def recruit_rider(charge=True):
    """Create a new rider, optionally charging the hire cost."""
    _ensure_init()
    cfg = globalvar.get_value("config", DEFAULT_CONFIG)
    new_id = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", new_id)
    rider = Rider(new_id)
    globalvar.get_value("riders").append(rider)
    if charge:
        globalvar.set_value("funds", globalvar.get_value("funds", 0)
                            - int(cfg.get("hire_cost", 300)))
    logger.info("recruited rider %s", new_id)
    return rider


def auto_recruit():
    """Recruit a rider automatically if funds >= hire cost.

    Returns the new Rider or None.
    """
    _ensure_init()
    cfg = globalvar.get_value("config", DEFAULT_CONFIG)
    cost = int(cfg.get("hire_cost", 300))
    riders = globalvar.get_value("riders")
    if globalvar.get_value("funds", 0) >= cost and \
            len(riders) < int(cfg.get("max_riders", 50)):
        return recruit_rider(charge=True)
    return None


def tick(n=1):
    """Advance the simulation ``n`` ticks."""
    _ensure_init()
    for _ in range(max(0, int(n))):
        globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
        for rider in list(globalvar.get_value("riders")):
            rider.move()
    check_consistency()


def check_consistency():
    """Repair inconsistent state where possible. Returns list of issues."""
    issues = []
    for rider in globalvar.get_value("riders", []):
        if rider.state not in (0, 1, 2):
            issues.append("rider %s invalid state" % rider.rider_id)
            rider.state = 0
            rider.current_order = None
            rider.path = []
        if rider.state != 0 and rider.current_order is None:
            issues.append("rider %s busy without order" % rider.rider_id)
            rider.state = 0
            rider.path = []
    for msg in issues:
        logger.warning("consistency fix: %s", msg)
    return issues


def snapshot():
    """Return a JSON-serialisable snapshot of the system."""
    return {
        "config": globalvar.get_value("config", DEFAULT_CONFIG),
        "funds": globalvar.get_value("funds", 0),
        "time": globalvar.get_value("time", 0),
        "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
        "riders": [r.to_dict() for r in globalvar.get_value("riders", [])],
        "orders": [o.to_dict() for o in globalvar.get_value("orders", [])],
    }


def save_state(path=None):
    """Persist the system state as JSON. Returns the path or None."""
    cfg = globalvar.get_value("config", DEFAULT_CONFIG)
    path = path or cfg.get("state_file", "dispatch_state.json")
    try:
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(snapshot(), fh, indent=2)
        return path
    except (OSError, TypeError, ValueError) as exc:
        logger.error("could not save state: %s", exc)
        return None

"""Order creation, assignment and completion management."""

import math

import globalvar
import system
from order_model import Order
from path_planner import PathPlanner


def find_best_rider_for_order(order):
    """Return the rider that can reach the restaurant soonest.

    The cost of a rider is the distance of all its queued work plus the
    distance from where that work ends to the order's restaurant, so idle
    nearby riders win and busy riders are naturally load-balanced. If there
    are no riders at all, one is recruited (funds may go negative).
    """
    riders = globalvar.get_value("riders")
    if riders is None:
        system.init_system()
        riders = globalvar.get_value("riders")
    if not riders:
        system.recruit_rider(charge=True)
        riders = globalvar.get_value("riders")
    best = None
    best_key = None
    for rider in riders:
        cost = rider.estimate_cost(order.res_pos)
        key = (cost, len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best, best_key = rider, key
    return best

"""Status display and ASCII map visualisation."""

import globalvar
from path_planner import MAP_SIZE
from rider import STATE_NAMES

CELL = 50
N = MAP_SIZE // CELL + 1  # 17 characters per side


def _idx(v):
    return max(0, min(N - 1, int(round(v / CELL))))


class DisplayManager:
    """Formats system information for the command line."""

    def render_map(self, riders=None, orders=None, routes=None):
        """Return an ASCII map string.

        Each character represents 50 units; '#' are streets, '.' blocks,
        'S' restaurants, 'C' customers, 'R' riders and '-', '|', '+' routes.
        """
        canvas = [['#' if r % 2 == 0 or c % 2 == 0 else '.'
                   for c in range(N)] for r in range(N)]
        for route in routes or []:
            for a, b in zip(route, route[1:]):
                self._draw_segment(canvas, a, b)
            for p in route[1:-1]:
                canvas[_idx(p[1])][_idx(p[0])] = '+'
        for o in orders or []:
            if o.status == "completed":
                continue
            canvas[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = 'S'
            canvas[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = 'C'
        for r in riders or []:
            canvas[_idx(r.y)][_idx(r.x)] = 'R'
        header = "     " + " ".join(
            str(c // 2) if c % 2 == 1 else " " for c in range(N))
        lines = [header]
        for r, row in enumerate(canvas):
            label = "%3d " % (r * CELL) if r % 2 == 0 else "  %d " % (r // 2)
            lines.append(label + " " + " ".join(row))
        lines.append("Legend: R=rider S=restaurant C=customer #=street "
                     ".=block, route: - | +  (1 char = 50 units)")
        return "\n".join(lines)

    @staticmethod
    def _draw_segment(canvas, a, b):
        r1, c1 = _idx(a[1]), _idx(a[0])
        r2, c2 = _idx(b[1]), _idx(b[0])
        if r1 == r2:
            for c in range(min(c1, c2), max(c1, c2) + 1):
                canvas[r1][c] = '-'
        elif c1 == c2:
            for r in range(min(r1, r2), max(r1, r2) + 1):
                canvas[r][c1] = '|'
        else:
            for c in range(min(c1, c2), max(c1, c2) + 1):
                canvas[r1][c] = '-'
            for r in range(min(r1, r2), max(r1, r2) + 1):
                canvas[r][c2] = '|'

    def current_map(self):
        riders = globalvar.get_value("riders", [])
        orders = globalvar.get_value("orders", [])
        routes = [r.full_route() for r in riders if r.orders]
        return self.render_map(riders, orders, routes)

    def order_result(self, result):
        o = result["order"]
        r = result["rider"]
        lines = [
            "Order #%d created at t=%d" % (o.id, o.create_time),
            "  Restaurant: input %s -> grid %s, centre %s"

            "  Customer:   input %s -> grid %s, centre %s"
            % (o.raw_cus, o.cus_grid, o.cus_pos),
            "  New rider added: %s" % ("yes" if result["new_rider"] else "no"),
            "  Assigned rider: #%d at (%d,%d), speed %d units/tick"

"""Command parsing and input validation."""

from path_planner import MAP_SIZE

try:  # command history / line editing where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

SHORTCUTS = {
    "s": "start", "start": "start",
    "st": "status", "status": "status",
    "o": "orders", "orders": "orders",
    "r": "riders", "riders": "riders",
    "m": "map", "map": "map",
    "t": "tick", "tick": "tick",
    "hi": "history", "history": "history",
    "save": "save",
    "h": "help", "help": "help", "?": "help",
    "q": "quit", "quit": "quit", "exit": "quit",
}


class InputManager:
    """Parses commands and validates user input."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        """Return (command, args). Unknown commands yield ('unknown', ...)."""
        line = (line or "").strip()
        if not line:
            return "", []
        self.history.append(line)
        if len(self.history) > 1000:
            self.history = self.history[-1000:]
        tokens = line.replace(",", " ").replace("[", " ") \
            .replace("]", " ").split()
        cmd = SHORTCUTS.get(tokens[0].lower(), "unknown")
        return cmd, tokens[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Validate four integer coordinates in 0..800.

        Raises ValueError with a friendly message on bad input.
        """
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ") \
                .replace("]", " ").split()
        if len(tokens) != 4:
            raise ValueError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for t in tokens:
            try:
                v = int(t)
            except (TypeError, ValueError):
                raise ValueError("'%s' is not an integer" % t)
            if v < 0 or v > MAP_SIZE:
                raise ValueError("coordinate %d out of range 0-%d"
                                 % (v, MAP_SIZE))
            values.append(v)
        return values

    @staticmethod
    def parse_tick_count(tokens):
        if not tokens:
            return 1
        try:
            n = int(tokens[0])
        except ValueError:
            raise ValueError("tick count must be an integer")
        if n < 1 or n > 10000:
            raise ValueError("tick count must be between 1 and 10000")
        return n

"""Command-line entry point for the food delivery dispatch simulator."""

import logging
import sys

import globalvar
import system
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager


def handle(cmd, args, inp, disp, om):
    """Execute a command. Returns the number of ticks to advance."""
    if cmd == "":
        return 1
    if cmd == "start":
        if not args:
            raw = input("Enter restaurant_x restaurant_y customer_x "
                        "customer_y (0-800): ")
            values = inp.parse_coordinates(raw)
        else:
            values = inp.parse_coordinates(args)
        result = om.create_order(values)
        print(disp.order_result(result))
        return 1
    if cmd == "status":
        print(disp.status())
        return 1
    if cmd == "orders":
        print(disp.orders())
        return 1
    if cmd == "riders":
        print(disp.rider_logs())
        return 1
    if cmd == "map":
        print(disp.current_map())
        return 1
    if cmd == "tick":
        n = inp.parse_tick_count(args)
        print("Advancing %d tick(s)." % n)
        return n
    if cmd == "history":
        for i, h in enumerate(inp.history, 1):
            print("%4d  %s" % (i, h))
        return 1
    if cmd == "save":
        path = system.save_state()
        print("State saved to %s" % path if path else "Save failed.")
        return 1
    if cmd == "help":
        print(disp.help())
        return 0
    if cmd == "quit":
        globalvar.set_value("running", False)
        return 0
    print("Unknown command. Type 'help' for the list of commands.")
    return 0


def main(argv=None):
    argv = list(sys.argv[1:] if argv is None else argv)
    level = logging.DEBUG if "--debug" in argv else logging.WARNING
    logging.basicConfig(level=level,
                        format="%(levelname)s %(name)s: %(message)s")
    config_path = "config.json"
    if "--config" in argv:
        i = argv.index("--config")
        if i + 1 < len(argv):
            config_path = argv[i + 1]
    system.init_system(config_path)
    inp = InputManager()
    disp = DisplayManager()
    om = OrderManager()
    print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
    print(disp.help())
    while globalvar.get_value("running", False):
        try:
            line = input("[t=%d funds=%d] > " % (
                globalvar.get_value("time", 0),
                globalvar.get_value("funds", 0)))
        except (EOFError, KeyboardInterrupt):
            print()
            globalvar.set_value("running", False)
            break
        try:
            cmd, args = inp.parse_command(line)
            ticks = handle(cmd, args, inp, disp, om)
            if ticks:
                system.tick(ticks)
        except ValueError as exc:
            print("Input error: %s" % exc)
        except (EOFError, KeyboardInterrupt):
            print("\nCancelled.")
        except Exception as exc:  # recover and keep running
            logging.exception("unexpected error")
            print("Unexpected error: %s (system state recovered)" % exc)
            system.check_consistency()
    path = system.save_state()
    if path:
        print("State saved to %s" % path)
    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Data models for the lottery system."""
from dataclasses import dataclass, field
from typing import List, Optional


class User:
    """A participant of the lottery."""

    def __init__(self, name: str, department: str):
        self.name = str(name).strip()
        self.department = str(department).strip()

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self):
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        self.name = str(self.name).strip()
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None

    def __str__(self):
        parts = [f"{self.name} x{self.quantity}"]
        if self.associated_drawer:
            parts.append(f"drawer: {self.associated_drawer}")
        if self.avoid_groups:
            parts.append(f"avoid: {', '.join(self.avoid_groups)}")
        return " | ".join(parts)

class ListManager:
    """Loads, validates and analyses participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            return [], [f"Unable to decode file (UTF-8 required): {path}"]
        except OSError as e:
            return [], [f"Unable to read file {path}: {e}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in line.replace("，", ",").split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            if u.name in seen:
                duplicates.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and the round order."""

    def __init__(self):
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = str(name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists")
        prize = Prize(name, qty, list(avoid_groups or []), associated_drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def get_prize(self, name) -> Optional[Prize]:
        for p in self._prizes:
            if p.name == name:
                return p
        return None

    def remove_prize(self, index: int) -> bool:
        if 1 <= index <= len(self._prizes):
            del self._prizes[index - 1]
            return True
        return False

    def set_avoid_groups(self, name, groups) -> bool:
        prize = self.get_prize(name)
        if prize is None:
            return False
        prize.avoid_groups = [str(g).strip() for g in (groups or []) if str(g).strip()]
        return True

    def reorder_prizes(self, order) -> bool:
        n = len(self._prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""
import random
from typing import Iterable, List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
MONITORED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"


def _norm(text) -> str:
    return " ".join(str(text or "").split()).casefold()


def is_cheng_ru(department) -> bool:
    d = _norm(department)
    if d.endswith(" group"):
        d = d[: -len(" group")]
    return d.replace(" ", "") == "chengru"


class LotteryManager:
    """Applies lottery rules and draws winners."""

    def __init__(self, users, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self._winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()

    # ---- rule helpers ----
    @staticmethod
    def _drawer_is(prize: Prize, drawer: str) -> bool:
        return _norm(prize.associated_drawer) == _norm(drawer)

    def is_fallback_triggered(self, prize: Prize) -> bool:
        if not self._drawer_is(prize, FALLBACK_DRAWER):
            return False
        last = self.last_two_draw_results[-2:]
        return len(last) == 2 and not any(last)

    def has_won(self, user: User) -> bool:
        return user.name in self._winner_names

    def describe_rules(self, prize: Prize) -> str:
        if self.is_fallback_triggered(prize):
            return ("Fallback mechanism active: prize pool limited to 'Cheng Ru group' members "
                    "(avoidance groups ignored); previous winners excluded")
        if self._drawer_is(prize, SPECIAL_DRAWER):
            return "Special drawer Li Bin: only previous winners are excluded (no avoidance groups)"
        if prize.avoid_groups:
            return ("Previous winners excluded; avoidance groups: " + ", ".join(prize.avoid_groups))
        return "Previous winners excluded; no avoidance groups"

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        available = [u for u in self.users if not self.has_won(u)]
        if self.is_fallback_triggered(prize):
            cheng_ru = [u for u in available if is_cheng_ru(u.department)]
            if cheng_ru:
                return cheng_ru
        if self._drawer_is(prize, SPECIAL_DRAWER):
            return available
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in available if _norm(u.department) not in avoid]

    # ---- drawing ----
    def register_winner(self, user: User) -> bool:
        if self.has_won(user):
            return False
        self._winner_names.add(user.name)
        self.winners.append(user)
        return True

    def draw_winner(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.register_winner(user)
        return user

    def record_round_result(self, prize: Prize, round_winners: Iterable[User]) -> None:
        round_winners = list(round_winners)
        if any(self._drawer_is(prize, d) for d in MONITORED_DRAWERS):
            self.last_two_draw_results.append(any(is_cheng_ru(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self._drawer_is(prize, FALLBACK_DRAWER):
            self.last_two_draw_results = []

"""Result storage, ordering and export."""
import csv
import os
import re
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User

TIME_FORMAT = "%Y-%m-%d %H:%M:%S"

_LEVEL_KEYWORDS = [
    (r"\bspecial\b", 0), (r"\bgrand\b", 0),
    (r"\bfirst\b", 1), (r"\bsecond\b", 2), (r"\bthird\b", 3),
    (r"\bfourth\b", 4), (r"\bfifth\b", 5),
    (r"\bconsolation\b", 50), (r"\bparticipation\b", 50),
]


def prize_level(name: str) -> int:
    lower = str(name).lower()
    for pattern, level in _LEVEL_KEYWORDS:
        if re.search(pattern, lower):
            return level
    return 100


class Winner:
    def __init__(self, user: User, prize: Prize, time: datetime = None):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()

    @property
    def time_str(self) -> str:
        return self.time.strftime(TIME_FORMAT)

    def __repr__(self):
        return f"Winner({self.user!r}, {self.prize.name!r}, {self.time_str})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user: User, prize: Prize):
        if any(w.user.name == user.name for w in self.winners):
            return None
        winner = Winner(user, prize)
        self.winners.append(winner)
        return winner

    def clear(self):
        self.winners = []

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        prizes = list(prizes or [])
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda t: (prize_level(t[1].name), -t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, p in indexed:
            result.setdefault(p.name, [])
        for w in self.winners:  # winners are stored in winning time order
            result.setdefault(w.prize.name, []).append(w)
        for name in result:
            result[name].sort(key=lambda w: w.time)
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winning results to export"
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in self.get_winners_by_prize(prizes).items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, prize_name, w.time_str])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""
from typing import List, Optional

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _input(prompt: str) -> Optional[str]:
    try:
        return input(prompt)
    except EOFError:
        return None


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.running = True

    # ---------- list ----------
    def load_default_list(self):
        users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        self.users = valid
        for e in errors:
            print("  " + e)
        print(f"Default list loaded: {len(valid)} participants.")

    def load_custom_list(self):
        path = _input("Enter txt list path: ")
        if path is None:
            return
        users, errors = self.list_manager.load_custom_list(path.strip().strip('"'))
        valid, dups = self.list_manager.validate_users(users)
        if errors:
            print("Problems found:")
            for e in errors:
                print("  " + e)
        if dups:
            print("Duplicate names found:")
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
        if not valid:
            print("No valid participants loaded. Upload terminated.")
            return
        if errors or dups:
            while True:
                choice = _input("[1] Remove invalid data and continue  [2] Terminate upload: ")
                if choice is None or choice.strip() == "2":
                    print("Upload terminated. Current list unchanged.")
                    return
                if choice.strip() == "1":
                    break
                print("Invalid choice.")
        self.users = valid
        print(f"Custom list loaded: {len(valid)} participants.")

    def show_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print(f"Valid groups ({len(info)}):")
        for dept, count in info.items():
            print(f"  {dept}: {count} people")
        print(f"Total: {len(self.users)} people")

    def remove_participant(self):
        if not self.users:
            print("No list loaded.")
            return
        for i, u in enumerate(self.users, 1):
            print(f"  {i}. {u}")
        s = _input("Enter number(s) to remove (comma separated): ")
        if not s:
            return
        try:
            idx = sorted({int(x) for x in s.replace("，", ",").split(",") if x.strip()}, reverse=True)
        except ValueError:
            print("Invalid input.")
            return
        for i in idx:
            if 1 <= i <= len(self.users):
                print(f"Removed {self.users.pop(i - 1)}")

    # ---------- prizes ----------
    def _groups(self) -> List[str]:
        return list(self.list_manager.get_group_info(self.users).keys())

    def _select_groups(self) -> List[str]:
        groups = self._groups()
        if not groups:
            s = _input("Enter avoidance groups (comma separated, empty for none): ") or ""
            return [g.strip() for g in s.replace("，", ",").split(",") if g.strip()]
        for i, g in enumerate(groups, 1):
            print(f"  {i}. {g}")
        s = _input("Select avoidance groups by number (comma separated, empty for none): ") or ""
        selected = []
        for x in s.replace("，", ",").split(","):
            x = x.strip()
            if not x:
                continue
            if x.isdigit() and 1 <= int(x) <= len(groups):
                if groups[int(x) - 1] not in selected:
                    selected.append(groups[int(x) - 1])
            else:
                print(f"  Ignored invalid selection: {x}")
        return selected

    def load_default_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prizes loaded.")
        self.show_prizes()

    def add_prize(self):
        name = _input("Prize name: ")
        if not name:
            print("Cancelled.")
            return
        qty = _input("Quantity: ") or ""
        drawer = (_input("Associated lottery person (empty for none): ") or "").strip() or None
        groups = self._select_groups()
        try:
            self.prize_manager.add_prize(name, qty.strip(), groups, drawer)
            print("Prize added.")
        except ValueError as e:
            print(f"Error: {e}")

    def configure_avoidance(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self.show_prizes()
        s = (_input("Select prize number: ") or "").strip()
        if not s.isdigit() or not 1 <= int(s) <= len(prizes):
            print("Invalid selection.")
            return
        prize = prizes[int(s) - 1]
        groups = self._select_groups()
        self.prize_manager.set_avoid_groups(prize.name, groups)
        print(f"Avoidance groups for '{prize.name}': {', '.join(groups) or 'none'}")

    def reorder_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self.show_prizes()
        s = _input("Enter new order as serial numbers (e.g. 3,1,2): ") or ""
        try:
            order = [int(x) for x in s.replace("，", ",").replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.show_prizes()
        else:
            print("Invalid order: must contain each serial number exactly once.")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Prize rounds:")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")

    # ---------- lottery ----------
    @staticmethod
    def _wait_command(position: int) -> str:
        paused = False
        while True:
            if paused:
                prompt = "[Paused] Input 'continue' to resume or 'terminate' to skip round: "
            else:
                prompt = (f"Currently drawing the {position}th winner - press Enter to draw "
                          "('pause' / 'terminate' / 'auto'): ")
            cmd = _input(prompt)
            if cmd is None:
                return "terminate"
            cmd = cmd.strip().lower()
            if cmd == "terminate":
                return "terminate"
            if paused:
                if cmd == "continue":
                    paused = False
                    print("Lottery resumed.")
                else:
                    print("Lottery is paused.")
                continue
            if cmd in ("", "draw", "d"):
                return "draw"
            if cmd == "auto":
                return "auto"
            if cmd == "pause":
                paused = True
                print("Lottery paused.")
            elif cmd == "continue":
                print("Lottery is not paused.")
            else:
                print("Unknown command.")

    def start_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.result_manager.winners:
            c = (_input("Existing results will be cleared. Continue? (y/n): ") or "").strip().lower()
            if c != "y":
                return
        self.result_manager.clear()
        lm = LotteryManager(self.users)
        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}"
                  + (f" | Lottery person: {prize.associated_drawer}" if prize.associated_drawer else ""))
            print(f"Rules: {lm.describe_rules(prize)}")
            print("=" * 60)
            round_winners = []
            auto = False
            for i in range(1, prize.quantity + 1):
                if not lm._get_eligible_pool(prize):
                    print("No eligible participants remain for this prize. Round ended.")
                    break
                if auto:
                    print(f"Currently drawing the {i}th winner...")
                else:
                    cmd = self._wait_command(i)
                    if cmd == "terminate":
                        print("Current round terminated.")
                        break
                    if cmd == "auto":
                        auto = True
                user = lm.draw_winner(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            lm.record_round_result(prize, round_winners)
            print(f"\nRound {rnd} results ({prize.name}):")
            if round_winners:
                for i, u in enumerate(round_winners, 1):
                    print(f"  {i}. {u.name} ({u.department})")
            else:
                print("  No winners.")
        print("\nAll rounds completed.")
        self.show_results()

    def show_results(self):
        if not self.result_manager.winners:
            print("No results yet.")
            return
        print("\n===== Complete Winner List =====")
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        for name, winners in grouped.items():
            print(f"[{name}]")
            if not winners:
                print("  (none)")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {name} - {w.time_str}")

    def export_results(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ---------- loop ----------
    def run(self):
        actions = {
            "1": ("Load default list", self.load_default_list),
            "2": ("Upload custom txt list", self.load_custom_list),
            "3": ("View groups", self.show_groups),
            "4": ("Remove participants", self.remove_participant),
            "5": ("Load preset prizes", self.load_default_prizes),
            "6": ("Add custom prize", self.add_prize),
            "7": ("Configure avoidance groups", self.configure_avoidance),
            "8": ("Reorder prizes", self.reorder_prizes),
            "9": ("View prizes", self.show_prizes),

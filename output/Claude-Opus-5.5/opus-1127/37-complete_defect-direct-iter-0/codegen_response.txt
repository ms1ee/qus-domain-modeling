"""Core data models for the lottery system."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass(frozen=True)
class User:
    """A lottery participant."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize configured for one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None

    def __str__(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "None"
        return (f"{self.name} | Quantity: {self.quantity} | "
                f"Avoidance groups: {avoid} | Drawer: {drawer}")

    "Yang Ba,Cheng Ru group",

    "You Yi,Cheng Ru group",
    "Xu Er,Cheng Ru group",

class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if path is None or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not path.lower().endswith(".txt"):
            return [], [f"File Error: only .txt files are supported ({path})"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except (OSError, UnicodeDecodeError) as exc:
                return [], [f"File Error: cannot read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("\uff0c", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users or []:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users or []:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Holds the ordered prize library."""

    def __init__(self) -> None:
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None,
                  associated_drawer: Optional[str] = None) -> Prize:
        name = str(name).strip() if name is not None else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be positive")
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups: List[str] = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self._prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in idx]
        return True

    def clear(self) -> None:
        self._prizes = []

"""Lottery rule engine."""

import random
from typing import Iterable, List, Optional, Set

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(text) -> str:
    return " ".join(str(text or "").split()).lower()


class LotteryManager:
    """Selects winners while applying all lottery rules."""

    def __init__(self, users) -> None:
        self.users: List[User] = []
        seen = set()
        for u in users or []:
            if u.name not in seen:
                seen.add(u.name)
                self.users.append(u)
        self.winner_names: Set[str] = set()
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []
        self._rng = random.SystemRandom()

    # ---- rule helpers -------------------------------------------------
    @staticmethod
    def _is_protected(user: User) -> bool:
        return _norm(user.department) == _norm(PROTECTED_GROUP)

    def is_fallback_active(self, prize: Prize) -> bool:
        return (_norm(prize.associated_drawer) == _norm(FALLBACK_DRAWER)
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if u.name not in self.winner_names]
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return base
        if self.is_fallback_active(prize):
            protected = [u for u in base if self._is_protected(u)]
            if protected:
                return protected
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in base if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded."]
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            parts.append(f"Special rule: drawer {SPECIAL_DRAWER} - only previous "
                         "winners are filtered, avoidance groups are NOT applied.")
        elif self.is_fallback_active(prize):
            parts.append(f"Fallback rule: previous drawers {WATCHED_DRAWERS[0]} and "
                         f"{WATCHED_DRAWERS[1]} drew no '{PROTECTED_GROUP}' members, "
                         f"prize pool is limited to '{PROTECTED_GROUP}' members.")
        elif prize.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(prize.avoid_groups) + ".")
        else:
            parts.append("No avoidance groups.")
        if prize.associated_drawer:
            parts.append(f"Drawer: {prize.associated_drawer}.")
        return " ".join(parts)

    # ---- drawing ------------------------------------------------------
    def draw_next(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.winner_names.add(user.name)
        self.winners.append(user)
        return user

    def complete_round(self, prize: Prize, round_winners: Iterable[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in WATCHED_DRAWERS}:
            has_protected = any(self._is_protected(u) for u in round_winners)
            self.last_two_draw_results.append(has_protected)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            # Consecutive chain broken (or fallback consumed).
            self.last_two_draw_results = []

    def draw_round(self, prize: Prize) -> List[User]:
        """Draw a whole round without interaction."""
        result: List[User] = []
        for _ in range(prize.quantity):
            u = self.draw_next(prize)
            if u is None:
                break
            result.append(u)
        self.complete_round(prize, result)
        return result

    def is_winner(self, name: str) -> bool:
        return name in self.winner_names

    def reset(self) -> None:
        self.winner_names.clear()
        self.winners.clear()
        self.last_two_draw_results = []

"""Result management and export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime
from typing import Dict, List, Optional

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize_name: str
    order: int

    def __str__(self) -> str:
        return (f"Congratulations {self.user.name} ({self.user.department}) "
                f"for winning {self.prize_name}")


def _prize_name(prize) -> str:
    return prize.name if isinstance(prize, Prize) or hasattr(prize, "name") else str(prize)


class ResultManager:
    """Keeps the list of winners and exports it."""

    def __init__(self) -> None:
        self.winners: List[Winner] = []
        self._names = set()

    def add_winner(self, user: User, prize) -> Optional[Winner]:
        if user.name in self._names:
            return None
        w = Winner(user, _prize_name(prize), len(self.winners) + 1)
        self.winners.append(w)
        self._names.add(user.name)
        return w

    def has_winner(self, name: str) -> bool:
        return name in self._names

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes or []:
            result[_prize_name(p)] = []
        for w in self.winners:
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for prize_name, winners in grouped.items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

    def clear(self) -> None:
        self.winners = []
        self._names = set()

"""Command-line interface for the lottery system."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def ask(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        raise SystemExit(0)


class LotteryCLI:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []

    # ---- list -------------------------------------------------------
    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        for e in errors:
            print("  " + e)
        self.users = valid
        print(f"Default list loaded: {len(valid)} valid participants.")

    def load_custom(self) -> None:
        path = ask("Enter txt file path (format per line: Name,Department Group): ")
        users, errors = self.list_manager.load_custom_list(path)
        if errors and not users and any(e.startswith("File Error") for e in errors):
            for e in errors:
                print("  " + e)
            return
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Invalid data detected:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            choice = ask("Remove invalid data and continue (y) or terminate upload (n)? ").strip().lower()
            if choice not in ("y", "yes"):
                print("Upload terminated.")
                return
        if not valid:
            print("No valid participants in file. Upload terminated.")
            return
        self.users = valid
        print(f"Custom list loaded: {len(valid)} valid participants.")

    def show_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current valid groups:")
        for i, (dept, cnt) in enumerate(info.items(), 1):
            print(f"  {i}. {dept}: {cnt} people")
        print(f"Total: {len(self.users)} people")

    # ---- prizes -----------------------------------------------------
    def load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Default prizes loaded.")

    def add_custom_prize(self) -> None:
        name = ask("Prize name: ").strip()
        qty = ask("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid: List[str] = []
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            sel = ask("Avoidance groups (numbers separated by commas, empty for none): ").strip()
            if sel:
                for part in sel.replace("\uff0c", ",").split(","):
                    part = part.strip()
                    if part.isdigit() and 1 <= int(part) <= len(groups):
                        if groups[int(part) - 1] not in avoid:
                            avoid.append(groups[int(part) - 1])
                    elif part:
                        print(f"  Ignored invalid selection: {part}")
        else:
            sel = ask("Avoidance groups (names separated by commas, empty for none): ").strip()
            avoid = [s.strip() for s in sel.split(",") if s.strip()]
        drawer = ask("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")

    def reorder(self) -> None:
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("\uff0c", ",").replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered:")
            self.show_prizes()
        else:
            print("Invalid order: must contain each serial number exactly once.")

    # ---- lottery ----------------------------------------------------
    def run_lottery(self) -> None:
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.result_manager.winners:
            c = ask("Previous results exist. Clear them and restart? (y/n): ").strip().lower()
            if c not in ("y", "yes"):
                return
        self.result_manager.clear()
        lm = LotteryManager(self.users)
        print("Commands during drawing: [Enter] draw next, 'pause', 'continue', 'terminate'.")
        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            print("Rules: " + lm.describe_rules(prize))
            print("=" * 60)
            round_winners: List[User] = []
            terminated = False
            i = 1
            while i <= prize.quantity:
                cmd = ask(f"[{ordinal(i)} winner] command: ").strip().lower()
                if cmd == "pause":
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to skip round.")
                    while True:
                        c2 = ask("(paused) > ").strip().lower()
                        if c2 == "continue":
                            print("Lottery resumed.")
                            break
                        if c2 == "terminate":
                            terminated = True
                            break
                        print("Lottery is paused. Type 'continue' or 'terminate'.")
                    if terminated:
                        break
                    continue
                if cmd == "continue":
                    print("Lottery is not paused.")
                    continue
                if cmd == "terminate":
                    terminated = True
                    break
                if cmd not in ("", "draw", "d"):
                    print("Unknown command.")
                    continue
                print(f"Currently drawing the {ordinal(i)} winner...")
                user = lm.draw_next(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                w = self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(str(w))
                i += 1
            if terminated:
                print(f"Round '{prize.name}' terminated and skipped.")
            lm.complete_round(prize, round_winners)
            print(f"--- Results of round {rnd}: {prize.name} ---")
            if round_winners:
                for n, u in enumerate(round_winners, 1):
                    print(f"  {n}. {u.name} ({u.department})")
            else:
                print("  No winners.")
        print("\nAll rounds completed.")
        self.show_results()

    def show_results(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        print("\n===== Complete Winner List =====")
        print(f"{'Name':<20}{'Department':<30}Prize")
        for prize_name, winners in grouped.items():
            for w in winners:
                print(f"{w.user.name:<20}{w.user.department:<30}{prize_name}")

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self) -> None:
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.show_groups,
            "4": self.load_default_prizes, "5": self.add_custom_prize,
            "6": self.show_prizes, "7": self.reorder, "8": self.run_lottery,
            "9": self.show_results, "10": self.export,
        }
        while True:
            print("\n===== Team Building Smart Lottery System =====")
            print(" 1. Load default list        2. Upload custom txt list")
            print(" 3. View groups              4. Load preset prizes")
            print(" 5. Add custom prize         6. View prizes")
            print(" 7. Reorder prizes           8. Start lottery")
            print(" 9. View all results        10. Export results")
            print(" 0. Exit")
            choice = ask("Select: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid option.")
                continue
            action()


def main() -> None:
    try:
        LotteryCLI().run()
    except KeyboardInterrupt:
        print("\nExited.")


if __name__ == "__main__":
    main()

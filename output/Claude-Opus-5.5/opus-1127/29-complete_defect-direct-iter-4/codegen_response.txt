"""Grid based path planning.

The map is 800x800 units divided into 8x8 cells of 100x100 units.  Streets
are the grid lines (x or y is a multiple of 100).  Riders move along streets
using right-angle turns, and only leave a street for the final short hop
into the destination cell (cell centres are at 50-unit offsets).
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE


def _clamp(v, lo, hi):
    return max(lo, min(hi, v))


class PathPlanner:
    """Grid-based path planning helpers (all static)."""

    @staticmethod
    def to_grid_index(v):
        """Convert a raw coordinate (0-800) to a grid index (0-7)."""
        v = _clamp(int(v), 0, MAP_SIZE - 1)
        return v // CELL_SIZE

    @staticmethod
    def to_grid_center(v):
        """Convert a raw coordinate to the centre of its grid cell.

        ``center = (v // 100) * 100 + 50``; the boundary value 800 is mapped
        into the last cell (750).
        """
        return PathPlanner.to_grid_index(v) * CELL_SIZE + CELL_SIZE // 2

    @staticmethod
    def to_center_pos(pos):
        """Convert an (x, y) pair to its grid cell centre."""
        return (PathPlanner.to_grid_center(pos[0]),
                PathPlanner.to_grid_center(pos[1]))

    @staticmethod
    def is_on_street(x, y):
        """Return True if (x, y) lies on a street (grid line)."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def is_special(start, target):
        """Return True when the rider is directly above/below the target
        while not standing on a vertical street (requires a detour)."""
        x, y = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        return x == tx and y != ty and x % CELL_SIZE != 0

    @staticmethod
    def _nearest_street(v, toward):
        """Return the street line nearest to ``v`` in the direction of
        ``toward`` (v itself if already on a street line)."""
        if v % CELL_SIZE == 0:
            return v
        lower = (v // CELL_SIZE) * CELL_SIZE
        upper = lower + CELL_SIZE
        s = upper if toward > v else lower
        return _clamp(s, 0, MAP_SIZE)

    @staticmethod
    def _clean(points):
        """Remove consecutive duplicates and collinear middle points."""
        out = []
        for p in points:
            if not out or out[-1] != p:
                out.append(p)
        changed = True
        while changed and len(out) > 2:
            changed = False
            for i in range(1, len(out) - 1):
                a, b, c = out[i - 1], out[i], out[i + 1]
                if (a[0] == b[0] == c[0]) or (a[1] == b[1] == c[1]):
                    # Only remove if b lies between a and c (no reversal)
                    if (min(a[0], c[0]) <= b[0] <= max(a[0], c[0]) and
                            min(a[1], c[1]) <= b[1] <= max(a[1], c[1])):
                        del out[i]
                        changed = True
                        break
        return out

    @staticmethod
    def plan(start, target):
        """Plan a street-following path.

        Returns a list of waypoints (tuples) starting with ``start`` and
        ending with ``target``; every segment is axis aligned.
        """
        x, y = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        pts = [(x, y)]
        if (x, y) == (tx, ty):
            return pts

        # Step onto the nearest street (vertical move) if inside a cell.
        if not PathPlanner.is_on_street(x, y):
            y = PathPlanner._nearest_street(y, ty)
            pts.append((x, y))

        # Horizontal street used to approach the target.
        if ty % CELL_SIZE == 0:
            ey = ty
        else:
            ey = PathPlanner._nearest_street(ty, y)

        if y == ey:
            pts.append((tx, ey))
        else:
            if x % CELL_SIZE == 0:
                vx = x
            elif tx % CELL_SIZE == 0:
                vx = tx
            else:
                lower = (tx // CELL_SIZE) * CELL_SIZE
                upper = min(lower + CELL_SIZE, MAP_SIZE)
                vx = min((lower, upper),
                         key=lambda c: (abs(x - c) + abs(c - tx), c))
            pts.append((vx, y))
            pts.append((vx, ey))
            pts.append((tx, ey))
        pts.append((tx, ty))
        return PathPlanner._clean(pts)

    @staticmethod
    def plan_multi(start, targets):
        """Plan a continuous path through several destinations."""
        path = [(int(start[0]), int(start[1]))]
        cur = path[0]
        for t in targets:
            seg = PathPlanner.plan(cur, t)
            path.extend(seg[1:])
            cur = (int(t[0]), int(t[1]))
        return PathPlanner._clean(path)

    @staticmethod
    def path_length(points):
        """Return the Manhattan length of a waypoint path."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    @staticmethod
    def distance(a, b):
        """Street distance between two points (length of planned path)."""
        return PathPlanner.path_length(PathPlanner.plan(a, b))

"""Order data model."""

import globalvar
from path_planner import PathPlanner


class Order:
    """A delivery order from a restaurant to a customer.

    Coordinates are converted to the centre of their grid cell.

    Attributes:
        id: unique order id.
        res_pos / cus_pos: grid-centre positions of restaurant / customer.
        raw_res / raw_cus: raw user input positions.
        status: "pending", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: id of the assigned rider (or None).
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.raw_res = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = PathPlanner.to_center_pos(self.raw_res)
        self.cus_pos = PathPlanner.to_center_pos(self.raw_cus)
        self.res_grid = (PathPlanner.to_grid_index(self.raw_res[0]),
                         PathPlanner.to_grid_index(self.raw_res[1]))
        self.cus_grid = (PathPlanner.to_grid_index(self.raw_cus[0]),
                         PathPlanner.to_grid_index(self.raw_cus[1]))
        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("time", 0)
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.complete_time = None
        self.time_taken = None
        self.travel_distance = None
        self.efficiency = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)
        cfg = globalvar.get_value("config") or {}
        base = cfg.get("order_base_fee", 30)
        per = cfg.get("order_fee_per_100", 10)
        self.fee = int(base + per * self.distance // 100)

    @classmethod
    def reset_counter(cls, value=1):
        """Reset the order id counter."""
        cls._next_id = value

    def assign_to(self, rider_id):
        """Mark this order as assigned to ``rider_id``."""
        self.rider_id = rider_id
        if self.status in ("pending", "assigned"):
            self.status = "assigned"
        self.assign_time = globalvar.get_value("time", 0)

    def complete(self, travel_distance=None):
        """Mark the order as completed and record efficiency data."""
        self.status = "completed"
        self.complete_time = globalvar.get_value("time", 0)
        self.time_taken = self.complete_time - self.create_time
        self.travel_distance = travel_distance
        if self.time_taken and self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = None

    def to_dict(self):
        """Serialise to a JSON compatible dict."""
        d = dict(self.__dict__)
        for k in ("raw_res", "raw_cus", "res_pos", "cus_pos",
                  "res_grid", "cus_grid"):
            d[k] = list(d[k])
        return d

    @classmethod
    def from_dict(cls, d):
        """Rebuild an order from :meth:`to_dict` output."""
        obj = cls.__new__(cls)
        for k, v in d.items():
            if k in ("raw_res", "raw_cus", "res_pos", "cus_pos",
                     "res_grid", "cus_grid"):
                v = tuple(v)
            setattr(obj, k, v)
        return obj

    def __repr__(self):
        return "Order(id=%s, res=%s, cus=%s, status=%s, rider=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status, self.rider_id)

"""Rider model: movement, state and order handling."""

import globalvar
from path_planner import PathPlanner

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``; ``move()``
    advances the rider one tick and performs state transitions.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        cfg = globalvar.get_value("config") or {}
        self.rider_id = rider_id
        self.x = int(x if x is not None else cfg.get("spawn_x", 400))
        self.y = int(y if y is not None else cfg.get("spawn_y", 400))
        self.speed = int(speed if speed is not None
                         else cfg.get("rider_speed", 50))
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.path = []
        self.route = []
        self.current_order = None
        self.total_distance = 0
        self._order_distance = 0
        self.log = []
        self._log("spawn", None)

    # ------------------------------------------------------------------
    def _log(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("time", 0),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def state_name(self):
        """Human-readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    def _plan_to(self, target):
        start = (self.x, self.y)
        self.special_case = PathPlanner.is_special(start, target)
        full = PathPlanner.plan(start, target)
        self.route = full
        self.path = list(full[1:])

    def remaining_distance(self):
        """Distance left on the current planned path."""
        pts = [(self.x, self.y)] + list(self.path)
        return PathPlanner.path_length(pts)

    def _advance(self):
        budget = self.speed
        while budget > 0 and self.path:
            nx, ny = self.path[0]
            dx, dy = nx - self.x, ny - self.y
            d = abs(dx) + abs(dy)
            if d <= budget:
                self.x, self.y = nx, ny
                budget -= d
                moved = d
                self.path.pop(0)
            else:
                moved = budget
                if dx != 0:
                    step = min(budget, abs(dx))
                    self.x += step if dx > 0 else -step
                    budget -= step
                if budget > 0 and dy != 0:
                    step = min(budget, abs(dy))
                    self.y += step if dy > 0 else -step
                    budget -= step
                budget = 0
            self.total_distance += moved
            self._order_distance += moved

    def move(self):
        """Advance one tick: pick up new orders, move, and transition."""
        if self.state == STATE_IDLE:
            self.orders[:] = [o for o in self.orders
                              if getattr(o, "status", None) != "completed"]
            if not self.orders:
                self.special_case = False
                return
            order = self.orders[0]
            self.current_order = order
            if getattr(order, "rider_id", None) is None:
                order.assign_to(self.rider_id)
            order.status = "picking_up"
            order.accept_time = globalvar.get_value("time", 0)
            self.state = STATE_PICKING_UP
            self._order_distance = 0
            self._log("accepted_order", order.id)
            self._plan_to(order.res_pos)

        if self.current_order is None:
            # Inconsistent state: recover to idle.
            self.state = STATE_IDLE
            self.path = []
            return

        self._advance()

        if not self.path:
            order = self.current_order
            if self.state == STATE_PICKING_UP:
                self.state = STATE_DELIVERING
                order.status = "delivering"
                order.pickup_time = globalvar.get_value("time", 0)
                self._log("arrived_at_merchant", order.id)
                self._plan_to(order.cus_pos)
            elif self.state == STATE_DELIVERING:
                order.complete(self._order_distance)
                if order in self.orders:
                    self.orders.remove(order)
                money = globalvar.get_value("money", 0) or 0
                globalvar.set_value("money", money + getattr(order, "fee", 0))
                self.state = STATE_IDLE
                self.current_order = None
                self.path = []
                self.special_case = False
                self._log("delivered_to_customer", order.id)

    # ------------------------------------------------------------------
    def to_dict(self):
        """Serialise to a JSON compatible dict (orders as ids)."""
        return {
            "rider_id": self.rider_id, "x": self.x, "y": self.y,
            "speed": self.speed, "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "path": [list(p) for p in self.path],
            "route": [list(p) for p in self.route],
            "current_order": (self.current_order.id
                              if self.current_order else None),
            "total_distance": self.total_distance,
            "order_distance": self._order_distance,
            "log": list(self.log),
        }

    @classmethod
    def from_dict(cls, d, order_map):
        """Rebuild a rider; ``order_map`` maps order id -> Order."""
        r = cls.__new__(cls)
        r.rider_id = d["rider_id"]
        r.x, r.y = int(d["x"]), int(d["y"])
        r.speed = int(d["speed"])
        r.state = int(d["state"])
        r.orders = [order_map[i] for i in d.get("orders", [])
                    if i in order_map]
        r.special_case = bool(d.get("special_case", False))
        r.path = [tuple(p) for p in d.get("path", [])]
        r.route = [tuple(p) for p in d.get("route", [])]
        cid = d.get("current_order")
        r.current_order = order_map.get(cid) if cid is not None else None
        r.total_distance = d.get("total_distance", 0)
        r._order_distance = d.get("order_distance", 0)
        r.log = list(d.get("log", []))
        return r

    def __repr__(self):
        return "Rider(id=%s, pos=(%s,%s), state=%s, orders=%s)" % (
            self.rider_id, self.x, self.y, self.state,
            [o.id for o in self.orders])

"""Order management: creation, assignment and completion helpers."""

import logging
import math

import globalvar
from order_model import Order
from path_planner import PathPlanner

logger = logging.getLogger("dispatch.order")


def _manhattan(a, b):
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def projected_state(rider, exclude=None):
    """Return (cost, position) after the rider finishes its backlog.

    ``exclude`` is an order that should not be counted (e.g. the order
    currently being evaluated).
    """
    pos = (rider.x, rider.y)
    cost = 0
    for i, o in enumerate(rider.orders):
        if o is exclude or getattr(o, "status", None) == "completed":
            continue
        if o is rider.current_order and rider.state != 0:
            if rider.state == 1:
                cost += rider.remaining_distance() + o.distance
            else:
                cost += rider.remaining_distance()
        else:
            cost += PathPlanner.distance(pos, o.res_pos) + o.distance
        pos = o.cus_pos
    return cost, pos


def estimate_cost(rider, order):
    """Estimated distance before ``rider`` could reach the restaurant."""
    cost, pos = projected_state(rider, exclude=order)
    return cost + PathPlanner.distance(pos, order.res_pos)


def find_best_rider_for_order(order, riders=None):
    """Return the best rider for ``order`` (closest first, load balanced).

    Idle riders are ranked by their distance to the restaurant; busy riders
    include their remaining backlog distance, so work spreads across riders.
    Returns None if there are no riders.
    """
    if riders is None:
        riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    return min(riders, key=lambda r: (estimate_cost(r, order),
                                      len(r.orders), r.rider_id))


class OrderManager:
    """Handles order creation, assignment and querying."""

    def __init__(self, system_manager=None):
        self.system_manager = system_manager

    @staticmethod
    def all_orders():
        """Return list of all orders."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    @staticmethod
    def validate_coordinates(coords):
        """Validate four coordinates; raise ValueError on error."""
        if len(coords) != 4:
            raise ValueError("Exactly four coordinates are required.")
        out = []
        for c in coords:
            if isinstance(c, bool) or not isinstance(c, int):
                raise ValueError("Coordinates must be integers.")
            if c < 0 or c > 800:
                raise ValueError("Coordinate %d out of range (0-800)." % c)
            out.append(c)
        return out

    def assign(self, order, rider):
        """Assign ``order`` to ``rider``."""
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)

    def create_order(self, rx, ry, cx, cy):
        """Create, store and assign an order.

        Returns a dict with keys: order, rider, new_rider, route, eta, cost.
        """
        self.validate_coordinates([rx, ry, cx, cy])
        order = Order((rx, ry), (cx, cy))
        self.all_orders().append(order)
        new_rider = None
        if self.system_manager is not None:
            new_rider = self.system_manager.auto_recruit()
        rider = find_best_rider_for_order(order)
        result = {"order": order, "rider": rider, "new_rider": new_rider,
                  "route": [], "eta": None, "cost": None}
        if rider is None:
            logger.warning("No rider available for order %s", order.id)
            return result
        backlog, start = projected_state(rider, exclude=order)
        route = PathPlanner.plan_multi(start, [order.res_pos, order.cus_pos])
        total = backlog + PathPlanner.path_length(route)
        self.assign(order, rider)
        result["route"] = route
        result["cost"] = total
        result["eta"] = int(math.ceil(total / float(max(rider.speed, 1))))
        return result

"""System initialisation and the SystemManager core class."""

import json
import logging
import os

import globalvar
from order import find_best_rider_for_order
from order_model import Order
from rider import Rider

logger = logging.getLogger("dispatch.system")

DEFAULT_CONFIG = {
    "initial_funds": 500,
    "recruit_threshold": 300,
    "recruit_cost": 300,
    "rider_speed": 50,
    "spawn_x": 400,
    "spawn_y": 400,
    "order_base_fee": 30,
    "order_fee_per_100": 10,
    "ticks_per_command": 1,
    "max_run_ticks": 10000,
    "save_file": "dispatch_state.json",
}


def load_config(path=None):
    """Load configuration from a JSON file merged over defaults."""
    cfg = dict(DEFAULT_CONFIG)
    if path and os.path.exists(path):
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
            if isinstance(data, dict):
                cfg.update({k: v for k, v in data.items() if k in cfg})
        except (OSError, ValueError) as exc:
            logger.error("Failed to load config %s: %s", path, exc)
    return cfg


def init_system(config=None):
    """Initialise global state and return a SystemManager."""
    globalvar._init()
    cfg = dict(DEFAULT_CONFIG)
    if config:
        cfg.update(config)
    globalvar.set_value("config", cfg)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("money", int(cfg["initial_funds"]))
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("map_size", 800)
    globalvar.set_value("cell_size", 100)
    globalvar.set_value("grid_dim", 8)
    Order.reset_counter(1)
    return SystemManager()


class SystemManager:
    """Manages funds, riders, simulated time and system state."""

    @property
    def config(self):
        return globalvar.get_value("config") or dict(DEFAULT_CONFIG)

    @property
    def funds(self):
        return globalvar.get_value("money", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("money", int(value))

    @property
    def riders(self):
        r = globalvar.get_value("riders")
        if r is None:
            r = []
            globalvar.set_value("riders", r)
        return r

    @property
    def orders(self):
        o = globalvar.get_value("orders")
        if o is None:
            o = []
            globalvar.set_value("orders", o)
        return o

    @property
    def time(self):
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        return globalvar.get_value("running", True)

    @running.setter
    def running(self, value):
        globalvar.set_value("running", bool(value))

    # ------------------------------------------------------------------
    def create_rider(self):
        """Create a rider with the next id and add it to the list."""
        rid = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", rid)
        rider = Rider(rid)
        self.riders.append(rider)
        logger.info("Rider %s created", rid)
        return rider

    def remove_rider(self, rider_id):
        """Remove an idle rider without orders; return True on success."""
        for r in self.riders:
            if r.rider_id == rider_id and r.state == 0 and not r.orders:
                self.riders.remove(r)
                return True
        return False

    def get_rider(self, rider_id):
        """Return the rider with the given id or None."""
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    def auto_recruit(self):
        """Recruit a rider if funds >= threshold; return it or None."""
        cfg = self.config
        if self.funds >= cfg["recruit_threshold"]:
            self.funds = self.funds - cfg["recruit_cost"]
            return self.create_rider()
        return None

    def dispatch_pending(self):
        """Assign any pending, unassigned orders."""
        for o in self.orders:
            if o.status == "pending" and o.rider_id is None:
                best = find_best_rider_for_order(o, self.riders)
                if best is None:
                    break
                o.assign_to(best.rider_id)
                best.orders.append(o)

    def tick(self, n=1):
        """Advance the simulation by ``n`` ticks."""
        for _ in range(max(0, int(n))):
            globalvar.set_value("time", self.time + 1)
            self.dispatch_pending()
            for r in self.riders:
                try:
                    r.move()
                except Exception as exc:  # recovery
                    logger.exception("Rider %s failed: %s", r.rider_id, exc)
                    self.consistency_check()

    def active_orders(self):
        """Return orders not yet completed."""
        return [o for o in self.orders if o.status != "completed"]

    def run_until_complete(self):
        """Tick until all orders are complete; return ticks used."""
        limit = int(self.config.get("max_run_ticks", 10000))
        used = 0
        while self.active_orders() and used < limit:
            if not self.riders:
                break
            self.tick(1)
            used += 1
        return used

    def consistency_check(self):
        """Repair inconsistent state; return a list of fix descriptions."""
        fixes = []
        for r in self.riders:
            before = len(r.orders)
            r.orders[:] = [o for o in r.orders if o.status != "completed"]
            if len(r.orders) != before:
                fixes.append("rider %s: removed completed orders" % r.rider_id)
            if r.state not in (0, 1, 2):
                r.state = 0
                fixes.append("rider %s: invalid state reset" % r.rider_id)
            if r.state != 0 and (r.current_order is None or
                                 r.current_order not in r.orders):
                r.state = 0
                r.current_order = None
                r.path = []
                fixes.append("rider %s: reset to idle" % r.rider_id)
            cx = max(0, min(800, r.x))
            cy = max(0, min(800, r.y))
            if (cx, cy) != (r.x, r.y):
                r.x, r.y = cx, cy
                fixes.append("rider %s: position clamped" % r.rider_id)
        ids = {r.rider_id for r in self.riders}
        for o in self.orders:
            if o.status != "completed" and o.rider_id is not None \
                    and o.rider_id not in ids:
                o.rider_id = None
                o.status = "pending"
                fixes.append("order %s: requeued" % o.id)
        return fixes

    # ------------------------------------------------------------------
    def save_state(self, path=None):
        """Save system state to a JSON file; return the path."""
        path = path or self.config.get("save_file", "dispatch_state.json")
        data = {
            "config": self.config,
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "next_order_id": Order._next_id,
            "orders": [o.to_dict() for o in self.orders],
            "riders": [r.to_dict() for r in self.riders],
        }
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)
        return path

    def load_state(self, path=None):
        """Load system state from a JSON file."""
        path = path or self.config.get("save_file", "dispatch_state.json")
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        cfg = dict(DEFAULT_CONFIG)
        cfg.update(data.get("config", {}))
        globalvar.set_value("config", cfg)
        globalvar.set_value("money", int(data.get("funds", 0)))
        globalvar.set_value("time", int(data.get("time", 0)))
        globalvar.set_value("rider_id_counter",
                            int(data.get("rider_id_counter", 0)))
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        omap = {o.id: o for o in orders}
        riders = [Rider.from_dict(d, omap) for d in data.get("riders", [])]
        globalvar.set_value("orders", orders)
        globalvar.set_value("riders", riders)
        Order.reset_counter(int(data.get("next_order_id",
                                         len(orders) + 1)))
        self.consistency_check()
        return path

"""Status display and ASCII map visualisation."""

from rider import STATE_NAMES

GRID_CHARS = 17  # 8 cells + 9 street lines


def _idx(v):
    return max(0, min(GRID_CHARS - 1, int(round(v / 50.0))))


class DisplayManager:
    """Formats status information and renders the ASCII map."""

    @staticmethod
    def render_map(riders, orders, route=None):
        """Return an ASCII map.

        '#' street, '.' empty cell, '*' route, 'S' restaurant,
        'C' customer, 'R' rider.
        """
        grid = [['#' if (i % 2 == 0 or j % 2 == 0) else '.'
                 for j in range(GRID_CHARS)] for i in range(GRID_CHARS)]
        if route:
            for a, b in zip(route, route[1:]):
                r1, c1 = _idx(a[1]), _idx(a[0])
                r2, c2 = _idx(b[1]), _idx(b[0])
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        if r in (r1, r2) or c in (c1, c2):
                            grid[r][c] = '*'
            if len(route) == 1:
                grid[_idx(route[0][1])][_idx(route[0][0])] = '*'
        for o in orders:
            if o.status == "completed":
                continue
            grid[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = 'C'
            grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = 'S'
        for r in riders:
            grid[_idx(r.y)][_idx(r.x)] = 'R'
        header = "    " + " ".join(str(j // 2) if j % 2 else ' '
                                   for j in range(GRID_CHARS))
        lines = [header]
        for i, row in enumerate(grid):
            label = str(i // 2) if i % 2 else ' '
            lines.append(" %s  %s" % (label, " ".join(row)))
        lines.append("Legend: R=Rider S=Restaurant C=Customer "
                     "*=Route #=Street .=Empty")
        return "\n".join(lines)

    @staticmethod
    def status_text(sm):
        """Return a snapshot of the system state."""
        lines = ["=== System Status (time %d) ===" % sm.time,
                 "Funds: %d" % sm.funds,
                 "Riders: %d" % len(sm.riders)]
        for r in sm.riders:
            lines.append(
                "  Rider %d at (%d,%d) speed=%d state=%s%s orders=%s" % (
                    r.rider_id, r.x, r.y, r.speed,
                    STATE_NAMES.get(r.state, "?"),
                    " [special]" if r.special_case else "",
                    [o.id for o in r.orders]))
        pending = [o.id for o in sm.orders if o.status == "pending"]
        if pending:
            lines.append("Unassigned orders: %s" % pending)
        return "\n".join(lines)

    @staticmethod
    def orders_text(orders):
        """Return a table of all orders."""
        if not orders:
            return "No orders yet."
        lines = ["%-4s %-6s %-11s %-11s %-6s %-6s %-6s %-11s" % (
            "ID", "Start", "Restaurant", "Customer", "Dist", "Time",
            "Rider", "Status")]
        for o in orders:
            lines.append("%-4s %-6s %-11s %-11s %-6s %-6s %-6s %-11s" % (
                o.id, o.create_time,
                "(%d,%d)" % tuple(o.res_pos), "(%d,%d)" % tuple(o.cus_pos),
                o.distance,
                o.time_taken if o.time_taken is not None else "-",
                o.rider_id if o.rider_id is not None else "-",
                o.status))
        return "\n".join(lines)

    @staticmethod
    def riders_log_text(riders):
        """Return all rider status-change records."""
        entries = []
        for r in riders:
            for e in r.log:
                entries.append((e.get("time", 0), r.rider_id, e))
        if not entries:
            return "No rider records."
        entries.sort(key=lambda t: (t[0], t[1]))
        lines = []
        for t, rid, e in entries:
            oid = e.get("order_id")
            lines.append("t=%-5s rider %-3s %-22s order=%-4s at (%s,%s)" % (
                t, rid, e.get("event"), oid if oid is not None else "-",
                e.get("x"), e.get("y")))
        return "\n".join(lines)

    @staticmethod
    def start_result_text(result, riders, orders):
        """Return the output of the start command."""
        o = result["order"]
        lines = [
            "Order %d created." % o.id,
            "Restaurant grid: (%d,%d) center (%d,%d)" % (
                o.res_grid + o.res_pos),
            "Customer grid:   (%d,%d) center (%d,%d)" % (
                o.cus_grid + o.cus_pos),
            "New rider added: %s" % (
                "yes (rider %d)" % result["new_rider"].rider_id
                if result["new_rider"] else "no"),
        ]
        rider = result["rider"]
        if rider is None:
            lines.append("No rider available; order queued.")
            return "\n".join(lines)
        lines.append("Assigned rider: %d  speed=%d units/tick  "
                     "estimated time=%s ticks" % (
                         rider.rider_id, rider.speed, result["eta"]))
        lines.append("Route: " + " -> ".join(
            "(%d,%d)" % p for p in result["route"]))
        lines.append(DisplayManager.render_map(riders, orders,
                                               result["route"]))
        return "\n".join(lines)

"""Command parsing and input validation."""

try:  # command history support where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover

ALIASES = {
    "s": "start", "start": "start",
    "st": "status", "status": "status",
    "o": "orders", "orders": "orders",
    "r": "riders", "riders": "riders",
    "m": "map", "map": "map",
    "t": "tick", "tick": "tick",
    "run": "run",
    "save": "save", "load": "load",
    "hist": "history", "history": "history",
    "h": "help", "help": "help", "?": "help",
    "q": "quit", "quit": "quit", "exit": "quit",
}

HELP_TEXT = """Commands:
  start (s) [rx ry cx cy]  create an order (coordinates 0-800)
  status (st)              system snapshot
  orders (o)               list all orders
  riders (r)               rider status-change records
  map (m)                  show the ASCII map
  tick (t) [n]             advance simulation n ticks
  run                      run until all orders are delivered
  save [file] / load [file] persist / restore state (JSON)
  history (hist)           show command history
  help (h)                 show this help
  quit / exit (q)          exit"""


class InputManager:
    """Parses commands and validates user input."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        """Return (command, args). Unknown commands return (None, args)."""
        line = (line or "").strip()
        if not line:
            return "", []
        self.history.append(line)
        parts = line.split()
        cmd = ALIASES.get(parts[0].lower())
        return cmd, parts[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Parse four integer coordinates in 0-800; raise ValueError."""
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ") \
                .replace("]", " ").split()
        else:
            joined = " ".join(str(t) for t in tokens)
            tokens = joined.replace(",", " ").replace("[", " ") \
                .replace("]", " ").split()
        if len(tokens) != 4:
            raise ValueError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for t in tokens:
            try:
                v = int(t)
            except ValueError:
                raise ValueError("'%s' is not an integer." % t)
            if v < 0 or v > 800:
                raise ValueError("Coordinate %d out of range (0-800)." % v)
            values.append(v)
        return values

    @staticmethod
    def parse_positive_int(token, default=1):
        """Parse a positive integer argument."""
        if token is None:
            return default
        v = int(token)
        if v <= 0:
            raise ValueError("Value must be positive.")
        return v

"""Command-line entry point for the food delivery dispatch simulator."""

import argparse
import logging
import sys

import globalvar
from display import DisplayManager
from input_manager import HELP_TEXT, InputManager
from order import OrderManager
from system import init_system, load_config

logger = logging.getLogger("dispatch")


def handle_command(cmd, args, sm, om, im, input_func=input):
    """Execute a command; return output text. Sets running False on quit."""
    if cmd == "quit":
        sm.running = False
        return "Goodbye."
    if cmd == "help":
        return HELP_TEXT
    if cmd == "start":
        if args:
            coords = im.parse_coordinates(args)
        else:
            raw = input_func("Enter restaurant_x restaurant_y "
                             "customer_x customer_y: ")
            coords = im.parse_coordinates(raw)
        result = om.create_order(*coords)
        return DisplayManager.start_result_text(result, sm.riders, sm.orders)
    if cmd == "status":
        return DisplayManager.status_text(sm)
    if cmd == "orders":
        return DisplayManager.orders_text(sm.orders)
    if cmd == "riders":
        return DisplayManager.riders_log_text(sm.riders)
    if cmd == "map":
        return DisplayManager.render_map(sm.riders, sm.orders)
    if cmd == "tick":
        n = im.parse_positive_int(args[0] if args else None, 1)
        sm.tick(n)
        return "Advanced %d tick(s). Time is now %d." % (n, sm.time)
    if cmd == "run":
        used = sm.run_until_complete()
        return "Ran %d tick(s). Time is now %d." % (used, sm.time)
    if cmd == "save":
        return "State saved to %s" % sm.save_state(args[0] if args else None)
    if cmd == "load":
        return "State loaded from %s" % sm.load_state(args[0] if args else None)
    if cmd == "history":
        return "\n".join("%3d  %s" % (i + 1, h)
                         for i, h in enumerate(im.history))
    return "Unknown command. Type 'help' for a list of commands."


def main(argv=None):
    """Run the interactive main loop (one tick per command)."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch")
    parser.add_argument("--config", help="JSON configuration file")
    parser.add_argument("--load", help="JSON state file to load")
    parser.add_argument("--debug", action="store_true", help="debug logging")
    opts = parser.parse_args(argv)

    logging.basicConfig(level=logging.DEBUG if opts.debug else logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")

    sm = init_system(load_config(opts.config))
    om = OrderManager(sm)
    im = InputManager()
    if opts.load:
        try:
            sm.load_state(opts.load)
        except (OSError, ValueError, KeyError) as exc:
            print("Failed to load state: %s" % exc)

    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
    while globalvar.get_value("running", True):
        try:
            line = input("[t=%d] > " % sm.time)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        cmd, args = im.parse_command(line)
        if cmd == "":
            continue
        try:
            out = handle_command(cmd, args, sm, om, im)
            if out:
                print(out)
        except ValueError as exc:
            print("Error: %s" % exc)
        except (OSError, KeyError) as exc:
            print("Error: %s" % exc)
        except Exception as exc:  # graceful recovery
            logger.exception("Unexpected error")
            print("Unexpected error: %s. Recovering state." % exc)
            fixes = sm.consistency_check()
            if fixes:
                print("Repairs: " + "; ".join(fixes))
        if cmd != "quit" and cmd is not None and sm.running:
            try:
                sm.tick(int(sm.config.get("ticks_per_command", 1)))
            except Exception:
                logger.exception("Tick failed")
                sm.consistency_check()
    return 0


if __name__ == "__main__":
    sys.exit(main())

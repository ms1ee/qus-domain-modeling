"""Core data models."""
import re
from typing import List, Optional

PROTECTED_GROUP = "Cheng Ru group"
SPECIAL_DRAWER = "Li Bin"
FALLBACK_TRIGGER_PERSONS = ["Chen Qixian", "Jin Duo"]
FALLBACK_TARGET_PERSON = "Ma Jiaqi"


class User:
    """A participant: name plus department group."""

    def __init__(self, name: str, department: str):
        self.name = name.strip()
        self.department = department.strip()

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self):
        return f"{self.name} ({self.department})"


class Prize:
    """A prize with quantity, avoidance groups and an optional associated drawer."""

    def __init__(self, name: str, quantity: int, avoid_groups: Optional[List[str]] = None,
                 associated_drawer: Optional[str] = None):
        self.name = name.strip()
        self.quantity = int(quantity)
        self.avoid_groups = list(avoid_groups) if avoid_groups else []
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        self.associated_drawer = drawer or None

    @property
    def level(self) -> int:
        return prize_level(self.name)

    def __repr__(self):
        return (f"Prize(name={self.name!r}, quantity={self.quantity}, "
                f"avoid_groups={self.avoid_groups!r}, associated_drawer={self.associated_drawer!r})")


_LEVEL_WORDS = [
    (("grand", "special", "super"), 0),
    (("first", "1st"), 1),
    (("second", "2nd"), 2),
    (("third", "3rd"), 3),
    (("fourth", "4th"), 4),
    (("fifth", "5th"), 5),
    (("sixth", "6th"), 6),
    (("participation", "consolation", "lucky"), 50),
]

UNKNOWN_LEVEL = 100


def prize_level(name: str) -> int:
    """Return numeric level for a prize name; smaller is higher."""
    lower = name.lower()
    for words, level in _LEVEL_WORDS:
        for w in words:
            if re.search(r"\b" + re.escape(w) + r"\b", lower):
                return level
    return UNKNOWN_LEVEL

import os
from collections import OrderedDict
from typing import Dict, List, Tuple

from lottery_system.models import User, PROTECTED_GROUP

DEFAULT_LIST_LINES = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Wang Wu,Technology Department",
    "Zhao Liu,Technology Department",
    "Sun Qi,Product Department",
    "Zhou Ba,Product Department",
    "Wu Jiu,Product Department",
    "Zheng Shi,Marketing Department",
    "Feng Yi,Marketing Department",
    "Chen Er,Marketing Department",
    "Chu Hong,HR Department",
    "Wei Lan,HR Department",
    f"Cheng Ru,{PROTECTED_GROUP}",
    f"Jiang Hao,{PROTECTED_GROUP}",
    f"Shen Yue,{PROTECTED_GROUP}",
    f"Han Mei,{PROTECTED_GROUP}",
    "Yang Fan,Finance Department",
    "Zhu Lin,Finance Department",
]


class ListManager:
    def __init__(self):
        self.users: List[User] = []
        self.source = None

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.source = "default"
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File Error: file not found: {path}"]
        if not str(path).lower().endswith(".txt"):
            # still try to load, but warn
            users, errors = self._load_from_path(path)
            return users, [f"Warning: file is not a .txt file: {path}"] + errors
        self.source = str(path)
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        lines = None
        last_exc = None
        for enc in ("utf-8-sig", "gbk", "latin-1"):
            try:
                with open(path, "r", encoding=enc) as f:
                    lines = f.read().splitlines()
                break
            except UnicodeDecodeError as e:
                last_exc = e
            except OSError as e:
                return [], [f"File Error: cannot read file {path}: {e}"]
        if lines is None:
            return [], [f"File Error: cannot decode file {path}: {last_exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("\uff0c", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' (expected 'Name,Department Group')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        duplicates: List[User] = []
        for u in users:
            if u.name in seen:
                duplicates.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

"""Prize library management."""
from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self.prizes}
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        groups = []
        for g in (avoid_groups or []):
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine and round execution."""
import random
from typing import Callable, List, Optional

from lottery_system.models import (
    User, Prize, PROTECTED_GROUP, SPECIAL_DRAWER,
    FALLBACK_TRIGGER_PERSONS, FALLBACK_TARGET_PERSON,
)


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


def _same_group(a: str, b: str) -> bool:
    return a.strip().lower() == b.strip().lower()


class LotteryManager:
    def __init__(self, users, result_manager=None):
        self.users: List[User] = list(users)
        self.winners = set()  # names of winners
        self.result_manager = result_manager
        if result_manager is not None:
            for w in result_manager.winners:
                self.winners.add(w.user.name)
        # results of the most recent trigger persons' rounds: True if contained protected group
        self.last_two_draw_results: List[bool] = []
        self.fallback_triggered = False
        self._rng = random.SystemRandom()

    # ---------- rule engine ----------
    def is_fallback_active(self, prize: Prize) -> bool:
        return (prize.associated_drawer == FALLBACK_TARGET_PERSON
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = [u for u in self.users if u.name not in self.winners]
        if self.is_fallback_active(prize):
            self.fallback_triggered = True
            return [u for u in candidates if _same_group(u.department, PROTECTED_GROUP)]
        if prize.associated_drawer == SPECIAL_DRAWER:
            return candidates
        avoid = [g.strip().lower() for g in prize.avoid_groups]
        return [u for u in candidates if u.department.strip().lower() not in avoid]

    def get_rule_description(self, prize: Prize) -> str:
        if self.is_fallback_active(prize):
            return (f"Fallback active: {', '.join(FALLBACK_TRIGGER_PERSONS)} drew no "
                    f"'{PROTECTED_GROUP}' members, so the pool is limited to '{PROTECTED_GROUP}' "
                    f"members (avoidance groups ignored); previous winners excluded")
        if prize.associated_drawer == SPECIAL_DRAWER:
            return (f"Special drawer {SPECIAL_DRAWER}: only previous winners are excluded, "
                    f"avoidance groups are not applied")
        desc = "Previous winners excluded"
        if prize.avoid_groups:
            desc += f"; avoidance groups excluded: {', '.join(prize.avoid_groups)}"
        else:
            desc += "; no avoidance groups"
        return desc

    def record_round(self, prize: Prize, round_winners: List[User]) -> None:
        drawer = prize.associated_drawer
        if drawer in FALLBACK_TRIGGER_PERSONS:
            contains = any(_same_group(u.department, PROTECTED_GROUP) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == FALLBACK_TARGET_PERSON:
            self.last_two_draw_results = []

    # ---------- drawing ----------
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.winners.add(user.name)
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return user

    def run_round(self, prize: Prize, input_func: Callable[[str], str] = input,
                  output_func: Callable[[str], None] = print) -> List[User]:
        round_winners: List[User] = []
        terminated = False
        for i in range(prize.quantity):
            # command handling
            while True:
                cmd = self._read(input_func,
                                 f"[{prize.name}] Press Enter to draw, or type pause/terminate: ")
                if cmd in ("", "draw", "continue"):
                    break
                if cmd == "pause":
                    output_func("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                    while True:
                        c2 = self._read(input_func, "(paused) > ")
                        if c2 in ("continue", "terminate"):
                            break
                        output_func("Lottery is paused. Type 'continue' or 'terminate'.")
                    if c2 == "continue":
                        output_func("Lottery resumed.")
                        break
                    cmd = "terminate"
                if cmd == "terminate":
                    terminated = True
                    break
                output_func("Unknown command. Use Enter, 'pause', 'continue' or 'terminate'.")
            if terminated:
                output_func(f"Round for '{prize.name}' terminated; skipping remaining draws.")
                break
            output_func(f"Currently drawing the {_ordinal(i + 1)} winner...")
            user = self.draw_one(prize)
            if user is None:
                output_func("No eligible participants remain for this prize. Round ends.")
                break
            round_winners.append(user)
            output_func(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        self.record_round(prize, round_winners)
        return round_winners

    @staticmethod
    def _read(input_func, prompt) -> str:
        try:
            return input_func(prompt).strip().lower()
        except EOFError:
            return "terminate"

"""Winner records, summary and export."""
import csv
import os
from datetime import datetime
from typing import Dict, List, Tuple

from lottery_system.models import prize_level, UNKNOWN_LEVEL

TIME_FMT = "%Y-%m-%d %H:%M:%S"


class Winner:
    def __init__(self, user, prize, time=None, order=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.order = order

    @property
    def prize_name(self):
        return self.prize.name if hasattr(self.prize, "name") else str(self.prize)

    def __repr__(self):
        return f"Winner(user={self.user!r}, prize={self.prize_name!r}, order={self.order})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self._counter = 0

    def has_won(self, user) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def add_winner(self, user, prize):
        if self.has_won(user):
            return None
        self._counter += 1
        w = Winner(user, prize, datetime.now(), self._counter)
        self.winners.append(w)
        return w

    def _sorted_prizes(self, prizes):
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda t: (prize_level(t[1].name), t[0]))
        return [p for _, p in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in self._sorted_prizes(prizes):
            result[p.name] = sorted(
                [w for w in self.winners if w.prize_name == p.name],
                key=lambda w: (w.time, w.order))
        # winners for prizes not in the list
        known = set(result)
        for w in sorted(self.winners, key=lambda w: (w.time, w.order)):
            if w.prize_name not in known:
                result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, ws in grouped.items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, prize_name,
                                         w.time.strftime(TIME_FMT)])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface."""
from lottery_system.models import SPECIAL_DRAWER
from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager, TIME_FMT

MENU = """
========== Team Building Smart Lottery ==========
 1. Load default participant list
 2. Upload custom participant list (txt)
 3. View department groups
 4. Load preset prizes
 5. Add custom prize
 6. View prizes
 7. Reorder prizes
 8. Start lottery
 9. View winner summary
10. Export results
 0. Exit
=================================================="""


def _input(prompt=""):
    try:
        return input(prompt)
    except EOFError:
        raise SystemExit(0)


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.lottery_manager = None
        self.users = []
        self.status = "idle"

    # ----- list -----
    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._accept_users(users, errors, allow_terminate=False)

    def load_custom(self):
        path = _input("Enter txt file path: ").strip().strip('"')
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print(e)
            print("No valid data loaded.")
            return
        self._accept_users(users, errors, allow_terminate=True)

    def _accept_users(self, users, errors, allow_terminate):
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Problems detected in the list:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            if allow_terminate:
                while True:
                    c = _input("Type 'r' to remove invalid data and continue, 't' to terminate upload: ").strip().lower()
                    if c in ("r", "t"):
                        break
                if c == "t":
                    print("Upload terminated. Current list unchanged.")
                    return
            print("Invalid data removed.")
        self.users = valid
        self.lottery_manager = None
        print(f"Loaded {len(valid)} valid participants.")

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for g, c in info.items():
            print(f"  {g}: {c} people")

    # ----- prizes -----
    def load_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print(e)
        print("Preset prizes loaded.")

    def add_prize(self):
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys()) if self.users else []
        avoid = []
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = _input("Avoidance groups (numbers or names, comma-separated, empty for none): ").strip()
        for item in raw.replace("\uff0c", ",").split(","):
            item = item.strip()
            if not item:
                continue
            if item.isdigit() and 1 <= int(item) <= len(groups):
                avoid.append(groups[int(item) - 1])
            else:
                avoid.append(item)
        drawer = _input("Associated lottery person (empty for none): ").strip() or None
        try:
            p = self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize '{p.name}' added.")
        except ValueError as e:
            print(f"Error: {e}")

    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | avoid: {', '.join(p.avoid_groups) or 'none'}"
                  f" | drawer: {p.associated_drawer or 'none'}")

    def reorder(self):
        self.view_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = _input("Enter new order as serial numbers (e.g. 3,1,2): ").strip()
        try:
            order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.view_prizes()
        else:
            print("Invalid order; must be a permutation of current serial numbers.")

    # ----- lottery -----
    def start_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery_manager is None:
            self.lottery_manager = LotteryManager(self.users, self.result_manager)
        self.status = "running"
        print("Commands during lottery: Enter = draw, 'pause', 'continue', 'terminate' (skip round).")
        for idx, prize in enumerate(prizes, 1):
            drawn = sum(1 for w in self.result_manager.winners if w.prize_name == prize.name)
            if drawn >= prize.quantity:
                continue
            print(f"\n----- Round {idx}: {prize.name} -----")
            print(f"Quantity: {prize.quantity}")
            print(f"Lottery person: {prize.associated_drawer or 'none'}"
                  + (" (special)" if prize.associated_drawer == SPECIAL_DRAWER else ""))
            print(f"Rules: {self.lottery_manager.get_rule_description(prize)}")
            winners = self.lottery_manager.run_round(prize, input, print)
            print(f"Round {idx} results:")
            if not winners:
                print("  (no winners)")
            for i, u in enumerate(winners, 1):
                print(f"  {i}. {u.name} ({u.department}) - {prize.name}")
        self.status = "finished"
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n========== Winner Summary ==========")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for pname, ws in grouped.items():
            print(f"[{pname}]")
            if not ws:
                print("  (none)")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {pname} - {w.time.strftime(TIME_FMT)}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_prizes, "5": self.add_prize, "6": self.view_prizes,
            "7": self.reorder, "8": self.start_lottery, "9": self.show_summary,
            "10": self.export,
        }
        while True:
            print(MENU)
            choice = _input("Select: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid choice.")
                continue
            action()


def main():
    LotteryApp().run()


if __name__ == "__main__":
    main()

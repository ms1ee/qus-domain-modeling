"""Helpers for matching subway line names written in different styles.

For example "Line 1", "line1", "1号线" and "地铁1号线" all map to the same
canonical key ``line1``.
"""

import re
import unicodedata

# Ordered list: more specific keywords must come first.
_NAMED_LINES = [
    ("daxingairport", ("大兴机场", "daxingairport")),
    ("airport", ("首都机场", "机场线", "airport")),
    ("changping", ("昌平", "changping")),
    ("yizhuang", ("亦庄", "yizhuang")),
    ("fangshan", ("房山", "fangshan")),
    ("batong", ("八通", "batong")),
    ("xijiao", ("西郊", "xijiao")),
    ("yanfang", ("燕房", "yanfang")),
]

_cache = {}


def canonical_line_key(name):
    """Return a canonical key for a line name (used for loose matching)."""
    if name is None:
        return ""
    if name in _cache:
        return _cache[name]
    s = unicodedata.normalize("NFKC", str(name)).strip().lower()
    s = re.sub(r"[\s_\-]+", "", s)
    for noise in ("beijingsubway", "subway", "地铁", "metro"):
        s = s.replace(noise, "")
    key = s
    match = re.fullmatch(r"(?:line)?(\d+)(?:号线|line|号)?", s)
    if match:
        key = "line" + str(int(match.group(1)))
    else:
        for canonical, keywords in _NAMED_LINES:
            if any(k in s for k in keywords):
                key = canonical
                break
    _cache[name] = key
    return key


def lines_match(a, b):
    """Return True when two line names refer to the same line."""
    return canonical_line_key(a) == canonical_line_key(b)

"""Graph data structures for subway network modeling.

The network is a weighted undirected graph. Vertices are station numbers
(assigned by the caller); every edge carries the inter-station distance in
meters as weight together with the line it belongs to.
"""

from dataclasses import dataclass, field


class Edge:
    """A weighted edge between two stations on a given line."""

    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    # Convenience aliases following the domain model naming.
    @property
    def from_(self):
        return self.source

    @property
    def to(self):
        return self.dest

    def other(self, vertex):
        """Return the endpoint that is not ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError("Vertex %r is not an endpoint of this edge" % (vertex,))

    def reversed(self):
        """Return the same edge in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.weight, self.line) == (
            other.source, other.dest, other.weight, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.weight, self.line))

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.weight, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists.

    Attributes:
        graph: dict mapping station number -> list of outgoing ``Edge``.
        station_lines: dict mapping station name -> list of line names.
        station_to_number: dict mapping station name -> station number.
        number_to_station: dict mapping station number -> station name.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self._edge_count = 0

    # ----- vertices -------------------------------------------------
    def add_vertex(self, number):
        self.graph.setdefault(number, [])

    def add_station(self, name, number, lines=None):
        """Register a station name with its (caller-assigned) number."""
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.add_vertex(number)
        if lines is not None:
            self.set_station_lines(name, lines)
        else:
            self.station_lines.setdefault(name, [])

    def set_station_lines(self, station_name, lines):
        """Set the lines a station belongs to."""
        if isinstance(lines, str):
            lines = [lines]
        unique = []
        for line in lines or []:
            if line not in unique:
                unique.append(line)
        self.station_lines[station_name] = unique

    def has_vertex(self, number):
        return number in self.graph or number in self.number_to_station

    def vertices(self):
        result = set(self.graph.keys())
        result.update(self.number_to_station.keys())
        return result

    @property
    def vertex_count(self):
        return len(self.vertices())

    @property
    def edge_count(self):
        return self._edge_count

    # ----- edges ----------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge (stored in both adjacency lists)."""
        if edge.weight is None or edge.weight < 0:
            raise ValueError("Edge weight must be a non-negative number")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        if edge.source != edge.dest:
            self.graph[edge.dest].append(edge.reversed())
        self._edge_count += 1

    def adj(self, number):
        """Return the edges leaving the given vertex."""
        return self.graph.get(number, [])

    def edges(self):
        """Return every undirected edge once."""
        seen = set()
        result = []
        for v, edges in self.graph.items():
            for e in edges:
                key = (min(e.source, e.dest), max(e.source, e.dest), e.weight, e.line)
                if key not in seen:
                    seen.add(key)
                    result.append(e)
        return result

    # ----- stations -------------------------------------------------
    def get_station_name(self, number):
        return self.number_to_station.get(number, str(number))

    def get_station_number(self, name):
        return self.station_to_number.get(name)

    def is_transfer_station(self, name):
        return len(self.station_lines.get(name, [])) > 1

    def __repr__(self):
        return "Graph(V=%d, E=%d)" % (self.vertex_count, self.edge_count)

"""Path calculation using Dijkstra's algorithm with a heap based priority queue.

The search runs over (station, current line[, transfers]) states so that
transfers at interchange stations can be counted and penalised exactly.
Complexity is O((V+E) log V) in the number of states.
"""

import heapq
import itertools
import math

from line_utils import canonical_line_key

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Distance-equivalent cost (meters) of one transfer in comprehensive mode.
DEFAULT_TRANSFER_PENALTY = 4000
# Weight multiplier applied to edges of preferred lines.
DEFAULT_PREFERENCE_FACTOR = 0.8


class PathfindingConfig:
    """Configuration for path queries.

    Attributes:
        mode: one of ``VALID_MODES``.
        max_transfers: maximum allowed transfers (None = unlimited).
        preferred_lines: list of preferred line names.
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None,
                 transfer_penalty=DEFAULT_TRANSFER_PENALTY,
                 preference_factor=DEFAULT_PREFERENCE_FACTOR):
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode %r, expected one of %s" % (mode, ", ".join(VALID_MODES)))
        if max_transfers is not None:
            max_transfers = int(max_transfers)
            if max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        else:
            preferred_lines = list(preferred_lines)
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = preferred_lines
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def __repr__(self):
        return "PathfindingConfig(mode=%r, max_transfers=%r, preferred_lines=%r)" % (
            self.mode, self.max_transfers, self.preferred_lines)


class MinPQ:
    """Minimum priority queue based on a binary heap (``heapq``)."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    insert = push

    def pop(self):
        """Remove and return ``(priority, item)`` with the smallest priority."""
        if not self._heap:
            raise IndexError("pop from an empty priority queue")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    del_min = pop

    def peek(self):
        if not self._heap:
            raise IndexError("peek from an empty priority queue")
        priority, _, item = self._heap[0]
        return priority, item

    def is_empty(self):
        return not self._heap

    @property
    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)


class ShortestPath:
    """Single-source path computation with Dijkstra's algorithm.

    Attributes:
        source: source station number.
        dist_to: dict station number -> actual distance (meters) of the best
                 route under the configured mode (``math.inf`` if unreachable).
    """

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError("Source station %r is not in the graph" % (source,))
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: math.inf for v in graph.vertices()}
        self.dist_to[source] = 0
        self._transfer_to = {}
        self._final_state = {}
        self._edge_to = {}
        self._run()

    # ----- configuration helpers -----------------------------------
    def _preferred_keys(self):
        prefs = getattr(self.config, "preferred_lines", None) or []
        if isinstance(prefs, str):
            prefs = [p for p in prefs.split(",")]
        return {canonical_line_key(p) for p in prefs if p and str(p).strip()}

    def _key(self, dist, wdist, transfers):
        mode = getattr(self.config, "mode", MODE_SHORTEST_DISTANCE)
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, wdist, dist)
        if mode == MODE_COMPREHENSIVE:
            penalty = getattr(self.config, "transfer_penalty", DEFAULT_TRANSFER_PENALTY)
            return (wdist + transfers * penalty, transfers, dist)
        return (dist, transfers, wdist)

    # ----- Dijkstra -------------------------------------------------
    def _run(self):
        max_transfers = getattr(self.config, "max_transfers", None)
        limited = max_transfers is not None
        factor = getattr(self.config, "preference_factor", DEFAULT_PREFERENCE_FACTOR)
        preferred = self._preferred_keys()

        start = (self.source, None, 0 if limited else None)
        best = {start: self._key(0, 0, 0)}
        settled = set()
        pq = MinPQ()
        pq.push(best[start], (start, 0, 0, 0))

        while pq:
            key, (state, dist, wdist, transfers) = pq.pop()
            if state in settled:
                continue
            settled.add(state)
            v, cur_line = state[0], state[1]
            if v not in self._final_state:
                # First settled state of a station is the best one for it.
                self._final_state[v] = state
                self.dist_to[v] = dist
                self._transfer_to[v] = transfers
            for edge in self.graph.adj(v):
                line = edge.line
                is_transfer = cur_line is not None and line is not None and line != cur_line
                nt = transfers + (1 if is_transfer else 0)
                if limited and nt > max_transfers:
                    continue
                new_line = line if line is not None else cur_line
                ns = (edge.dest, new_line, nt if limited else None)
                if ns in settled:
                    continue
                w = edge.weight
                if preferred and line is not None and canonical_line_key(line) in preferred:
                    ww = w * factor
                else:
                    ww = w
                nd = dist + w
                nw = wdist + ww
                nk = self._key(nd, nw, nt)
                if ns not in best or nk < best[ns]:
                    best[ns] = nk
                    self._edge_to[ns] = (state, edge)
                    pq.push(nk, (ns, nd, nw, nt))

    # ----- queries --------------------------------------------------
    def has_path_to(self, target):
        return target in self._final_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return ``(stations, edges)`` from source to target.

        Both lists are empty when the target is unreachable.
        """
        if target not in self._final_state:
            return [], []
        state = self._final_state[target]
        edges = []
        while state in self._edge_to:
            prev, edge = self._edge_to[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        path = [self.source] + [e.dest for e in edges]
        return path, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of transfers on the route to target (-1 if unreachable)."""
        if target not in self._final_state:
            return -1
        return self._transfer_to[target]

"""Fare calculation according to the Beijing Subway fare structure.

* 0-6 km: 3 yuan, 6-12 km: 4 yuan, 12-22 km: 5 yuan, 22-32 km: 6 yuan
* over 32 km: 7 yuan, plus 1 yuan for every further 20 km
Boundary values belong to the higher tier (e.g. exactly 6 km costs 4 yuan).
Fractional distances are rounded up to whole meters.
"""

import math

# (min distance inclusive, max distance exclusive, fare, tier type), meters
FARE_TIERS = [
    (0, 6000, 3, "basic"),
    (6000, 12000, 4, "progressive"),
    (12000, 22000, 5, "progressive"),
    (22000, 32000, 6, "progressive"),
]
EXTENDED_START_DISTANCE = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_INCREMENT_DISTANCE = 20000

# Parameters of the travel time estimate.
AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


def _validate_number(value, name):
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise TypeError("%s must be a number" % name)
    if math.isnan(value) or math.isinf(value):
        raise ValueError("%s must be finite" % name)
    if value < 0:
        raise ValueError("%s must be non-negative" % name)


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters."""
    _validate_number(distance, "distance")
    d = math.ceil(distance)
    for low, high, fare, _tier in FARE_TIERS:
        if low <= d < high:
            return fare
    return EXTENDED_BASE_FARE + (d - EXTENDED_START_DISTANCE) // EXTENDED_INCREMENT_DISTANCE


def get_fare_tier(distance):
    """Return the tier type name ("basic", "progressive", "extended")."""
    _validate_number(distance, "distance")
    d = math.ceil(distance)
    for low, high, _fare, tier in FARE_TIERS:
        if low <= d < high:
            return tier
    return "extended"


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes from distance (meters) and transfers."""
    _validate_number(distance, "distance")
    _validate_number(transfer_count, "transfer_count")
    riding = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    return round(riding + transfer_count * TRANSFER_TIME_MINUTES, 1)


def format_fare(fare):
    return "Fare: %d yuan" % fare

def build_graph(data):
    """Build a ``Graph`` from subway data; station numbers follow file order."""
    graph = Graph()
    number = 0
    for station in data.get("stations", []) or []:
        name = station.get("name") if isinstance(station, dict) else None
        if not name or name in graph.station_to_number:
            continue
        graph.add_station(name, number, list(station.get("lines", []) or []))
        number += 1
    for conn in data.get("connections", []) or []:
        if not isinstance(conn, dict):
            continue
        src, dst = conn.get("source"), conn.get("dest")
        dist = conn.get("distance")
        line = conn.get("line")
        if not src or not dst or src == dst:
            continue
        if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
            continue
        for name in (src, dst):
            if name not in graph.station_to_number:
                graph.add_station(name, number, [])
                number += 1
            if line and line not in graph.station_lines.get(name, []):
                graph.station_lines.setdefault(name, []).append(line)
        graph.add_edge(Edge(graph.station_to_number[src], graph.station_to_number[dst], dist, line))
    return graph

class SubwayDataManager:
    """Manages subway data (stations, connections, lines) and updates."""

    def __init__(self, data):
        if not isinstance(data, dict):
            raise ValueError("Subway data must be a dict")
        self.data = copy.deepcopy(data)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ----- accessors ------------------------------------------------
    @property
    def stations(self):
        return self.data["stations"]

    @property
    def connections(self):
        return self.data["connections"]

    def station_names(self):
        return [s.get("name") for s in self.stations if isinstance(s, dict)]

    def _find_station(self, name):
        for s in self.stations:
            if isinstance(s, dict) and s.get("name") == name:
                return s
        return None

    def build_graph(self):
        return build_graph(self.data)

    def get_lines(self):
        """Return dict line name -> ordered list of station names."""
        adjacency = defaultdict(lambda: defaultdict(list))
        for c in self.connections:
            line = c.get("line")
            if not line:
                continue
            a, b = c.get("source"), c.get("dest")
            adjacency[line][a].append(b)
            adjacency[line][b].append(a)
        for s in self.stations:
            for line in s.get("lines", []) or []:
                adjacency[line].setdefault(s.get("name"), [])
        result = {}
        for line, adj in adjacency.items():
            order, seen = [], set()
            starts = [n for n in adj if len(adj[n]) == 1] + list(adj.keys())
            for start in starts:
                if start in seen:
                    continue
                stack = [start]
                while stack:
                    node = stack.pop()
                    if node in seen:
                        continue
                    seen.add(node)
                    order.append(node)
                    for nxt in reversed(adj[node]):
                        if nxt not in seen:
                            stack.append(nxt)
            result[line] = order
        return result

    def get_transfer_stations(self):
        return [s["name"] for s in self.stations if len(s.get("lines", []) or []) > 1]

    # ----- updates --------------------------------------------------
    def add_station(self, name, lines=None):
        if not name:
            raise ValueError("Station name must not be empty")
        if self._find_station(name) is not None:
            raise ValueError("Station '%s' already exists" % name)
        self.stations.append({"name": name, "lines": list(lines or [])})

    def add_line(self, line_name, station_sequence, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if not line_name:
            raise ValueError("Line name must not be empty")
        if len(station_sequence) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_sequence) - 1:
            raise ValueError("distances must have len(station_sequence) - 1 entries")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for name in station_sequence:
            st = self._find_station(name)
            if st is None:
                self.stations.append({"name": name, "lines": [line_name]})
            elif line_name not in st.setdefault("lines", []):
                st["lines"].append(line_name)
        for i, d in enumerate(distances):
            self.connections.append({"source": station_sequence[i], "dest": station_sequence[i + 1],
                                     "distance": d, "line": line_name})

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        st = self._find_station(name)
        if st is None:
            raise ValueError("Station '%s' does not exist" % name)
        if new_name and new_name != name:
            if self._find_station(new_name) is not None:
                raise ValueError("Station '%s' already exists" % new_name)
            st["name"] = new_name
            for c in self.connections:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            st["lines"] = list(lines)

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance between two adjacent stations (adds if missing and line given)."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        updated = 0
        for c in self.connections:
            pair = {c.get("source"), c.get("dest")}
            if pair == {source, dest} and (line is None or c.get("line") == line):
                c["distance"] = distance
                updated += 1
        if updated == 0:
            if line is None:
                raise ValueError("No connection between '%s' and '%s'" % (source, dest))
            for name in (source, dest):
                if self._find_station(name) is None:
                    raise ValueError("Station '%s' does not exist" % name)
            self.connections.append({"source": source, "dest": dest, "distance": distance, "line": line})
            updated = 1
        return updated

    def save(self, path):
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(self.data, handle, ensure_ascii=False, indent=2)

    # ----- integrity check -----------------------------------------
    def check_integrity(self):
        report = DataIntegrityReport()
        names = []
        station_lines = {}
        for i, s in enumerate(self.stations):
            if not isinstance(s, dict) or not s.get("name"):
                report.errors.append("Station entry #%d has no name" % i)
                report.station_coverage_valid = False
                continue
            name = s["name"]
            if name in station_lines:
                report.warnings.append("Duplicate station entry '%s'" % name)
            lines = s.get("lines", []) or []
            if not lines:
                report.warnings.append("Station '%s' has no line information" % name)
            station_lines[name] = lines
            names.append(name)

        # Coverage of main lines and representative stations.
        line_members = defaultdict(set)
        for name, lines in station_lines.items():
            for line in lines:
                line_members[canonical_line_key(line)].add(name)
        for c in self.connections:
            if isinstance(c, dict) and c.get("line"):
                key = canonical_line_key(c["line"])
                line_members[key].update(x for x in (c.get("source"), c.get("dest")) if x)
        for key, (label, reps) in MAIN_LINES.items():
            members = line_members.get(key)
            if not members:
                report.station_coverage_valid = False
                report.warnings.append("Main line '%s' not found in data" % label)
                continue
            if len(members) < 2:
                report.station_coverage_valid = False
                report.warnings.append("Main line '%s' has fewer than 2 stations" % label)
            lowered = {m.casefold() for m in members}
            if not any(r.casefold() in lowered or (r + "站").casefold() in lowered for r in reps):
                report.warnings.append("No representative station found for '%s'" % label)

        # Connections.
        adjacency = defaultdict(set)
        seen_pairs = {}
        for i, c in enumerate(self.connections):
            if not isinstance(c, dict):
                report.errors.append("Connection #%d is malformed" % i)
                report.connection_integrity_valid = False
                continue
            src, dst, line, dist = c.get("source"), c.get("dest"), c.get("line"), c.get("distance")
            label = "%s - %s (%s)" % (src, dst, line)
            if src not in station_lines or dst not in station_lines:
                missing = [x for x in (src, dst) if x not in station_lines]
                report.errors.append("Connection %s references unknown station(s): %s"
                                     % (label, ", ".join(str(m) for m in missing)))
                report.connection_integrity_valid = False
            if src == dst:
                report.errors.append("Connection %s is a self loop" % label)
                report.connection_integrity_valid = False
            if not line:
                report.warnings.append("Connection %s has no line" % label)
            else:
                for n in (src, dst):
                    if n in station_lines and line not in station_lines[n]:
                        report.warnings.append("Station '%s' is not marked as on '%s'" % (n, line))
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                report.errors.append("Connection %s has invalid distance %r" % (label, dist))
                report.distance_accuracy_valid = False
            else:
                if dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                    report.warnings.append("Connection %s has unusual distance %s m" % (label, dist))
                key = (frozenset((src, dst)), line)
                if key in seen_pairs and seen_pairs[key] != dist:
                    report.warnings.append("Connection %s has inconsistent distances %s and %s"
                                           % (label, seen_pairs[key], dist))
                    report.distance_accuracy_valid = False
                seen_pairs[key] = dist
            if src and dst:
                adjacency[src].add(dst)
                adjacency[dst].add(src)

        isolated = [n for n in names if n not in adjacency]
        for n in isolated:
            report.warnings.append("Station '%s' has no connections" % n)
        # Connectivity.
        components = 0
        visited = set()
        for n in adjacency:
            if n in visited:
                continue
            components += 1
            queue = deque([n])
            visited.add(n)
            while queue:
                cur = queue.popleft()
                for nxt in adjacency[cur]:
                    if nxt not in visited:
                        visited.add(nxt)
                        queue.append(nxt)
        if components > 1:
            report.warnings.append("Network has %d disconnected components" % components)

        report.stats = {
            "Stations": len(names),
            "Connections": len(self.connections),
            "Lines": len({l for ls in station_lines.values() for l in ls}
                         | {c.get("line") for c in self.connections if isinstance(c, dict) and c.get("line")}),
            "Transfer stations": sum(1 for ls in station_lines.values() if len(ls) > 1),
        }
        return report

"""High level subway route service: station validation, routing and fares."""

import math
import re
import unicodedata
from dataclasses import dataclass, field

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import SubwayDataManager
from exceptions import (FareCalculationError, InvalidQueryError, PathCalculationError,
                        StationNotFoundError)
from fare_calculator import Sum_money, calculate_estimated_time
from line_utils import canonical_line_key
from pathfinding import PathfindingConfig, ShortestPath


def _normalize(name):
    text = unicodedata.normalize("NFKC", str(name)).strip()
    return re.sub(r"\s+", " ", text).casefold()


def _compact(name):
    return re.sub(r"[\s'’`\-·.]", "", _normalize(name))


@dataclass
class RouteSegment:
    """A part of a route travelled on a single line."""
    order: int
    line: str
    stations: list
    distance: float = 0

    @property
    def stop_count(self):
        return len(self.stations) - 1


@dataclass
class Route:
    """A computed route with distance, fare and transfer information."""
    start: str
    end: str
    stations: list
    segments: list
    total_distance: float
    transfer_count: int
    transfer_stations: list
    lines: list
    fare: int
    estimated_time: float
    query: dict = field(default_factory=dict)

    @property
    def total_distance_km(self):
        return self.total_distance / 1000.0


class SubwaySystem:
    """Facade combining data management, path finding and fare calculation."""

    def __init__(self, data=None, data_path=None):
        if data is None:
            data = load_subway_data(data_path or DATA_FILE_PATH)
        self.manager = SubwayDataManager(data)
        self.rebuild()

    def rebuild(self):
        """(Re)build the graph after data updates."""
        self.graph = self.manager.build_graph()
        self.station_to_number = self.graph.station_to_number
        self._index = {}
        self._compact_index = {}
        for name in self.station_to_number:
            self._index.setdefault(_normalize(name), name)
            self._compact_index.setdefault(_compact(name), name)

    # ----- validation ----------------------------------------------
    def resolve_station(self, name):
        """Return the official station name for user input, or None."""
        if name is None:
            return None
        raw = str(name).strip()
        if not raw:
            return None
        if raw in self.station_to_number:
            return raw
        key = _normalize(raw)
        candidates = [key]
        for suffix in ("站", " station"):
            if key.endswith(suffix) and len(key) > len(suffix):
                candidates.append(key[:-len(suffix)].strip())
        candidates.append(key + "站")
        candidates.append(key + " station")
        for c in candidates:
            if c in self._index:
                return self._index[c]
        for c in candidates:
            comp = _compact(c)
            if comp in self._compact_index:
                return self._compact_index[comp]
        return None

    def validate_query(self, start, end):
        s = self.resolve_station(start)
        if s is None:
            raise StationNotFoundError(
                "Starting station '%s' does not exist, please check the station name" % str(start).strip())
        e = self.resolve_station(end)
        if e is None:
            raise StationNotFoundError(
                "Ending station '%s' does not exist, please check the station name" % str(end).strip())
        if s == e:
            raise InvalidQueryError("Starting station cannot be the same as the ending station")
        return s, e

    def resolve_lines(self, preferred):
        """Map preferred line names to line names present in the data.

        Returns (resolved, unknown).
        """
        known = {}
        for lines in self.graph.station_lines.values():
            for line in lines:
                known.setdefault(canonical_line_key(line), line)
        resolved, unknown = [], []
        for p in preferred or []:
            key = canonical_line_key(p)
            if key in known:
                resolved.append(known[key])
            else:
                unknown.append(p)
        return resolved, unknown

    # ----- routing --------------------------------------------------
    def find_route(self, start, end, config=None):
        config = config or PathfindingConfig()
        s, e = self.validate_query(start, end)
        src, dst = self.station_to_number[s], self.station_to_number[e]
        try:
            sp = ShortestPath(self.graph, src, config)
            path, edges = sp.path_with_edges(dst)
        except Exception as exc:  # pragma: no cover - defensive
            raise PathCalculationError("Path calculation failed: %s" % exc)
        if not path:
            msg = "No route found from '%s' to '%s'" % (s, e)
            if config.max_transfers is not None:
                msg += " within %d transfer(s)" % config.max_transfers
            raise PathCalculationError(msg)

        names = [self.graph.get_station_name(n) for n in path]
        segments = []
        for edge in edges:
            a = self.graph.get_station_name(edge.source)
            b = self.graph.get_station_name(edge.dest)
            line = edge.line
            if segments and (line is None or line == segments[-1].line):
                segments[-1].stations.append(b)
                segments[-1].distance += edge.weight
            else:
                segments.append(RouteSegment(len(segments) + 1, line if line else "Unknown line",
                                             [a, b], edge.weight))
        total = sum(edge.weight for edge in edges)
        transfer_stations = [seg.stations[-1] for seg in segments[:-1]]
        lines = []
        for seg in segments:
            if seg.line not in lines:
                lines.append(seg.line)
        transfer_count = len(segments) - 1 if segments else 0

        try:
            fare = Sum_money(total)
        except (TypeError, ValueError) as exc:
            raise FareCalculationError("Fare calculation failed: %s" % exc)
        try:
            est = calculate_estimated_time(total, transfer_count)
        except (TypeError, ValueError):
            est = 0.0
        if isinstance(total, float) and total.is_integer():
            total = int(total)

        query = {
            "start": s,
            "end": e,
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "line_preference": list(config.preferred_lines or []),
        }
        return Route(s, e, names, segments, total, transfer_count, transfer_stations, lines,
                     fare, est, query)

def format_distance(meters):
    """Format a distance using an appropriate unit."""
    if meters < 1000:
        return "%d m" % round(meters)
    return "%.2f km" % (meters / 1000.0)


def summarize_route(route):
    """Extract key information from a route."""
    return {
        "total_distance_m": route.total_distance,
        "total_distance_km": round(route.total_distance / 1000.0, 2),
        "fare": route.fare,
        "transfer_count": route.transfer_count,
        "estimated_time_min": route.estimated_time,
    }


def format_text_report(route):
    """Return a console-formatted report for a route."""
    out = ["=" * 56]
    out.append("Route: %s -> %s" % (route.start, route.end))
    mode = route.query.get("mode")
    if mode:
        out.append("Query mode: %s" % MODE_LABELS.get(mode, mode))
    out.append("-" * 56)
    out.append("Station sequence (%d stations):" % len(route.stations))
    out.append("  " + " -> ".join(route.stations))
    out.append("")
    out.append("Route details:")
    for i, seg in enumerate(route.segments):
        out.append("  %d. [%s] %s -> %s (%d stop%s, %s)" % (
            seg.order, seg.line, seg.stations[0], seg.stations[-1], seg.stop_count,
            "" if seg.stop_count == 1 else "s", format_distance(seg.distance)))
        out.append("     " + " -> ".join(seg.stations))
        if i < len(route.segments) - 1:
            out.append("     >> Transfer at %s: %s -> %s" % (
                seg.stations[-1], seg.line, route.segments[i + 1].line))
    out.append("")
    out.append("Total distance: %d meters (%.2f km)" % (round(route.total_distance),
                                                       route.total_distance / 1000.0))
    if route.transfer_count:
        out.append("Transfers: %d (at %s)" % (route.transfer_count, ", ".join(route.transfer_stations)))
    else:
        out.append("Transfers: 0 (direct)")
    out.append("Lines involved: %s" % ", ".join(route.lines))
    out.append("Estimated time: %.1f minutes" % route.estimated_time)
    out.append(format_fare(route.fare))
    out.append("=" * 56)
    return "\n".join(out)


def build_json_report(route):
    """Return a JSON-serializable dict describing a route."""
    return {
        "query": dict(route.query),
        "path": list(route.stations),
        "segments": [{"order": s.order, "line": s.line, "stations": list(s.stations),
                      "distance": s.distance, "stops": s.stop_count} for s in route.segments],
        "distance": route.total_distance,
        "distance_km": round(route.total_distance / 1000.0, 2),
        "fare": route.fare,
        "fare_text": format_fare(route.fare),
        "estimated_time": route.estimated_time,
        "transfer_count": route.transfer_count,
        "transfer_stations": list(route.transfer_stations),
        "lines": list(route.lines),
    }


def format_json_report(route):
    return json.dumps(build_json_report(route), ensure_ascii=False, indent=2)


def generate_report(route, fmt="text"):
    """Generate a report in the requested format ("text" or "json")."""
    if fmt == "json":
        return format_json_report(route)
    return format_text_report(route)

    """Write one or more route reports to a file."""

        if fmt == "json":
            items = [build_json_report(r) for r in routes]

            json.dump(payload, handle, ensure_ascii=False, indent=2)
            handle.write("\n")
        else:
            handle.write("\n\n".join(format_text_report(r) for r in routes))
            handle.write("\n")

"""Command-line interface of the Beijing Subway fare calculation system.

Usage examples:
    python main.py
    python main.py --mode least_transfer --format json --output_file out.json
    python main.py --check_data
Enter queries as "Starting Station; Ending Station" (q to quit).
"""

import argparse
import re
import sys
import unicodedata

from data_loader import DATA_FILE_PATH
from data_manager import SubwayDataManager
from exceptions import InputFormatError, SubwayError
from pathfinding import PathfindingConfig, VALID_MODES
from report import export_reports, generate_report

FORMAT_ERROR_MESSAGE = "Input format incorrect, please use 'Starting Station; Ending Station' format"
PROMPT = "Please enter 'Starting Station; Ending Station' (q to quit): "
QUIT_WORDS = {"q", "quit", "exit"}


def build_arg_parser():
    parser = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", default=None, help="export query results to this file")
    parser.add_argument("--format", default="text", choices=["text", "json"], help="output format")
    parser.add_argument("--mode", default="shortest_distance", choices=list(VALID_MODES),
                        help="path query mode")
    parser.add_argument("--max_transfers", type=int, default=None, help="maximum number of transfers")
    parser.add_argument("--line_preference", default=None,
                        help="preferred line(s), comma separated (e.g. 'Line 1')")
    parser.add_argument("--check_data", action="store_true", help="run data integrity check and exit")
    parser.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    parser.add_argument("--query", default=None, help="run a single query 'Start; End' and exit")
    return parser


def parse_query_input(text):
    """Parse 'Start; End' input. Raises InputFormatError on bad format."""
    if text is None:
        raise InputFormatError(FORMAT_ERROR_MESSAGE)
    normalized = unicodedata.normalize("NFKC", str(text)).strip()
    parts = re.split(r"\s*(?:;|,|->|→)\s*", normalized)
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise InputFormatError(FORMAT_ERROR_MESSAGE)
    return parts[0].strip(), parts[1].strip()


def _handle_query(system, text, config, args, results, out):
    try:
        start, end = parse_query_input(text)
        route = system.find_route(start, end, config)
    except SubwayError as exc:
        print("Error: %s" % exc, file=out)
        return False
    except Exception as exc:  # unexpected failure, keep CLI alive
        print("Error: unexpected failure during query: %s" % exc, file=out)
        return False
    print(generate_report(route, args.format), file=out)
    results.append(route)
    if args.output_file:
        try:
            export_reports(results, args.output_file, args.format)
            print("Result exported to %s" % args.output_file, file=out)
        except OSError as exc:
            print("Error: cannot write output file: %s" % exc, file=out)
    return True


def run(argv=None, input_stream=None, output_stream=None):
    """Run the CLI; returns the process exit code."""
    inp = input_stream if input_stream is not None else sys.stdin
    out = output_stream if output_stream is not None else sys.stdout
    parser = build_arg_parser()
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer", file=out)
        return 2

    # Import here so that a failing data file is reported nicely.
    from data_loader import load_subway_data
    from subway_system import SubwaySystem
    try:
        data = load_subway_data(args.data_file)
    except (OSError, ValueError) as exc:
        print("Error: failed to load subway data: %s" % exc, file=out)
        return 1

    if args.check_data:
        try:
            report = SubwayDataManager(data).check_integrity()
        except Exception as exc:
            print("Error: data check failed: %s" % exc, file=out)
            return 1
        print(report.format(), file=out)
        return 0 if report.is_valid else 1

    try:
        system = SubwaySystem(data=data)
    except Exception as exc:
        print("Error: failed to build subway network: %s" % exc, file=out)
        return 1

    preferred = []
    if args.line_preference:
        raw = [p.strip() for p in re.split(r"[,，]", args.line_preference) if p.strip()]
        resolved, unknown = system.resolve_lines(raw)
        for u in unknown:
            print("Warning: preferred line '%s' does not exist and will be ignored" % u, file=out)
        preferred = resolved
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                               preferred_lines=preferred)

    results = []
    if args.query is not None:
        ok = _handle_query(system, args.query, config, args, results, out)
        return 0 if ok else 1

    print("Beijing Subway Fare Calculation System", file=out)
    while True:
        out.write(PROMPT)
        out.flush()
        line = inp.readline()
        if not line:
            out.write("\n")
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in QUIT_WORDS:
            break
        _handle_query(system, line, config, args, results, out)
    print("Goodbye!", file=out)
    return 0


def main():
    try:
        if hasattr(sys.stdout, "reconfigure"):
            sys.stdout.reconfigure(encoding="utf-8")
    except Exception:
        pass
    try:
        code = run()
    except KeyboardInterrupt:
        print("\nGoodbye!")
        code = 0
    sys.exit(code)


if __name__ == "__main__":
    main()

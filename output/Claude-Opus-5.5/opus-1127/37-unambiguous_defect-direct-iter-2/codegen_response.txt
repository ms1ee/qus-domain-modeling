"""Core data models of the lottery system."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A participant of the lottery."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if isinstance(self.associated_drawer, str):
            self.associated_drawer = self.associated_drawer.strip() or None

    def rule_description(self) -> str:
        parts = ["Previous winners are excluded"]
        if self.avoid_groups:
            parts.append("avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("no avoidance groups")
        if self.associated_drawer:
            parts.append(f"drawer: {self.associated_drawer}")
        else:
            parts.append("drawer: not specified")
        return "; ".join(parts)

    def __str__(self) -> str:
        return f"{self.name} x{self.quantity}"

_SEPARATOR_RE = re.compile(r"[,，\t]")


class ListManager:
    """Loads and verifies participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        path = str(path)
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"Unable to read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"Unable to read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [p.strip() for p in _SEPARATOR_RE.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department', got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        """Split users into valid (first occurrence) and duplicates by name."""
        seen = set()
        valid: List[User] = []
        duplicates: List[User] = []
        for user in users:
            key = user.name.strip().lower()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = OrderedDict()
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return dict(info)

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and round order."""

    def __init__(self) -> None:
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name cannot be empty")
        name = name.strip()
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        if any(p.name == name for p in self._prizes):
            raise ValueError(f"Prize already exists: {name}")
        groups: List[str] = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer: Optional[str] = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        prize = Prize(name, qty, groups, drawer or None)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        n = len(self._prizes)
        if n == 0 or len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in idx]
        return True

"""Result recording, summarizing and export."""

import csv
import os
import re
from datetime import datetime
from typing import Dict, List, Optional

from lottery_system.models import Prize, User

_LEVEL_PATTERNS = [
    (re.compile(r"\b(special|grand)\b", re.I), 0),
    (re.compile(r"\bfirst\b", re.I), 1),
    (re.compile(r"\bsecond\b", re.I), 2),
    (re.compile(r"\bthird\b", re.I), 3),
    (re.compile(r"\bfourth\b", re.I), 4),
    (re.compile(r"\bfifth\b", re.I), 5),
    (re.compile(r"\b(consolation|participation)\b", re.I), 50),
]


def prize_level(name: str) -> int:
    """Lower value means higher prize level."""
    for pattern, level in _LEVEL_PATTERNS:
        if pattern.search(name):
            return level
    return 40


class Winner:
    def __init__(self, user: User, prize_name: str, order: int, prize: Optional[Prize] = None):
        self.user = user
        self.prize_name = prize_name
        self.prize = prize
        self.order = order
        self.time = datetime.now()

    def __repr__(self) -> str:
        return f"Winner({self.user!r}, {self.prize_name!r}, order={self.order})"


def _prize_name(prize) -> str:
    return prize.name if isinstance(prize, Prize) else str(prize)


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []
        self._counter = 0

    def has_won(self, user: User) -> bool:
        key = user.name.strip().lower()
        return any(w.user.name.strip().lower() == key for w in self.winners)

    def add_winner(self, user, prize) -> bool:
        if self.has_won(user):
            return False
        self._counter += 1
        self.winners.append(
            Winner(user, _prize_name(prize), self._counter, prize if isinstance(prize, Prize) else None)
        )
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        names: List[str] = []
        for p in prizes or []:
            n = _prize_name(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        ordered = sorted(enumerate(names), key=lambda t: (prize_level(t[1]), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, n in ordered:
            result[n] = sorted((w for w in self.winners if w.prize_name == n), key=lambda w: w.order)
        return result

    def export_to_csv(self, prizes, output_dir: Optional[str] = None):
        directory = output_dir or os.getcwd()
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(directory, filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["No.", "Name", "Department", "Prize", "Win Time"])
                no = 0
                for prize_name, winners in grouped.items():
                    for w in winners:
                        no += 1
                        writer.writerow([
                            no, w.user.name, w.user.department, prize_name,
                            w.time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and process control."""

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

SPECIAL_DRAWER = "Li Bin"
FALLBACK_WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm_name(s: Optional[str]) -> str:
    return " ".join((s or "").strip().lower().split())


def _norm_group(s: Optional[str]) -> str:
    g = _norm_name(s)
    if g.endswith(" group"):
        g = g[: -len(" group")].strip()
    return g


def is_protected_group(department: str) -> bool:
    return _norm_group(department) == _norm_group(PROTECTED_GROUP)


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryManager:
    def __init__(self, users, result_manager: Optional[ResultManager] = None, rng=None):
        self.users: List[User] = list(users)
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.rng = rng or random.Random()
        self.winner_names = set()
        for w in self.result_manager.winners:
            self.winner_names.add(_norm_name(w.user.name))
        self.last_two_draw_results: List[bool] = []

    # ---------------- rule engine ----------------
    def _is_winner(self, user: User) -> bool:
        return _norm_name(user.name) in self.winner_names

    def fallback_active(self, prize: Prize) -> bool:
        return (
            _norm_name(prize.associated_drawer) == _norm_name(FALLBACK_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize) -> List[User]:
        candidates = [u for u in self.users if not self._is_winner(u)]
        if self.fallback_active(prize):
            protected = [u for u in candidates if is_protected_group(u.department)]
            if protected:
                return protected
        if _norm_name(prize.associated_drawer) == _norm_name(SPECIAL_DRAWER):
            return candidates
        avoid = {_norm_group(g) for g in (prize.avoid_groups or [])}
        return [u for u in candidates if _norm_group(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        desc = prize.rule_description()
        if _norm_name(prize.associated_drawer) == _norm_name(SPECIAL_DRAWER):
            desc += "; special rule: relaxed filtering (avoidance groups ignored)"
        if self.fallback_active(prize):
            desc += f"; fallback active: pool limited to '{PROTECTED_GROUP}' members"
        return desc

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        while pool:
            user = self.rng.choice(pool)
            if self.result_manager.add_winner(user, prize):
                self.winner_names.add(_norm_name(user.name))
                return user
            self.winner_names.add(_norm_name(user.name))
            pool = [u for u in pool if u is not user and not self._is_winner(u)]
        return None

    def record_round(self, prize: Prize, round_winners: List[User]) -> None:
        drawer = _norm_name(prize.associated_drawer)
        watched = {_norm_name(d) for d in FALLBACK_WATCHED_DRAWERS}
        if drawer in watched:
            self.last_two_draw_results.append(any(is_protected_group(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            self.last_two_draw_results = []

    # ---------------- process control ----------------
    def run_round(self, prize: Prize, round_no: int = 1, total_rounds: int = 1,
                  input_func: Callable[[str], str] = input,
                  output: Callable[[str], None] = print) -> List[User]:
        output("")
        output("=" * 60)
        output(f"Round {round_no}/{total_rounds}: {prize.name}")
        output(f"Quantity: {prize.quantity}")
        output(f"Rules: {self.describe_rules(prize)}")
        output("=" * 60)
        round_winners: List[User] = []
        auto = False
        k = 0
        while k < prize.quantity:
            if not auto:
                try:
                    cmd = input_func(
                        "Press Enter to draw (commands: pause / continue / terminate / auto): "
                    ).strip().lower()
                except EOFError:
                    cmd = "auto"
                if cmd == "pause":
                    output("Lottery paused. Input 'continue' to resume or 'terminate' to skip this round.")
                    while True:
                        try:
                            c = input_func("[Paused] > ").strip().lower()
                        except EOFError:
                            c = "continue"
                        if c in ("continue", "terminate"):
                            break
                        output("Lottery is paused. Input 'continue' or 'terminate'.")
                    if c == "terminate":
                        output(f"Round '{prize.name}' terminated.")
                        break
                    output("Lottery resumed.")
                    continue
                if cmd == "terminate":
                    output(f"Round '{prize.name}' terminated.")
                    break
                if cmd == "auto":
                    auto = True
                elif cmd not in ("", "continue"):
                    output(f"Unknown command: {cmd}")
                    continue
            output(f"Currently drawing the {ordinal(k + 1)} winner...")
            user = self.draw_one(prize)
            if user is None:
                output("No eligible participants remain for this prize. Round ends early.")
                break
            round_winners.append(user)
            output(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            k += 1
        self.record_round(prize, round_winners)
        output(f"--- Results of round '{prize.name}' ---")
        if round_winners:
            for i, u in enumerate(round_winners, 1):
                output(f"{i}. {u.name} ({u.department})")
        else:
            output("No winners in this round.")
        return round_winners

    def run_all(self, prizes, input_func=input, output=print):
        prizes = list(prizes)
        results = {}
        for i, prize in enumerate(prizes, 1):
            results[prize.name] = self.run_round(prize, i, len(prizes), input_func, output)
        return results

"""Command-line interface."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class LotteryCLI:
    def __init__(self, input_func=input, output=print):
        self.input = input_func
        self.out = output
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_done = False

    def ask(self, prompt: str) -> str:
        try:
            return self.input(prompt)
        except EOFError:
            return "0"

    def run(self):
        actions = {
            "1": self.load_default_list, "2": self.upload_list, "3": self.view_groups,
            "4": self.load_default_prizes, "5": self.add_prize, "6": self.view_prizes,
            "7": self.reorder, "8": self.start_lottery, "9": self.show_results,
            "10": self.export,
        }
        while True:
            self.out("\n===== Team Building Smart Lottery System =====")
            self.out("1. Load default list        2. Upload custom txt list")
            self.out("3. View groups              4. Load preset prizes")
            self.out("5. Add custom prize         6. View prizes")
            self.out("7. Reorder prizes           8. Start lottery")
            self.out("9. Show results             10. Export results")
            self.out("0. Exit")
            choice = self.ask("Select: ").strip()
            if choice == "0":
                self.out("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                self.out("Invalid choice.")

    def _apply_list(self, users, errors) -> None:
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            for e in errors:
                self.out(e)
            for d in dups:
                self.out(f"Duplicate name: {d.name} ({d.department})")
            while True:
                c = self.ask("Remove invalid data and continue (r) or terminate upload (t)? ").strip().lower()
                if c == "r":
                    break
                if c == "t":
                    self.out("Upload terminated.")
                    return
        if not valid:
            self.out("No valid participants loaded.")
            return
        self.users = valid
        self.out(f"Loaded {len(valid)} valid participants.")

    def load_default_list(self):
        self._apply_list(*self.list_manager.load_default_list())

    def upload_list(self):
        path = self.ask("Path of txt list (format per line: Name,Department): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors and errors[0].startswith(("File not found", "Unable")):
            self.out(errors[0])
            return
        self._apply_list(users, errors)

    def view_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            self.out("No list loaded.")
            return
        for i, (g, c) in enumerate(info.items(), 1):
            self.out(f"{i}. {g}: {c} people")

    def load_default_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            self.out(e)
        self.view_prizes()

    def add_prize(self):
        name = self.ask("Prize name: ").strip()
        qty = self.ask("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                self.out(f"{i}. {g}")
        raw = self.ask("Avoidance groups (numbers or names, comma-separated, blank for none): ").strip()
        avoid = []
        for part in raw.replace("，", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                avoid.append(groups[int(part) - 1])
            else:
                avoid.append(part)
        drawer = self.ask("Associated lottery person (blank for none): ").strip() or None
        try:
            p = self.prize_manager.add_prize(name, qty, avoid, drawer)
            self.out(f"Prize added: {p.name} x{p.quantity}")
        except ValueError as exc:
            self.out(f"Error: {exc}")

    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            self.out(f"{i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def reorder(self):
        self.view_prizes()
        raw = self.ask("New order (serial numbers separated by space/comma): ")
        try:
            order = [int(x) for x in raw.replace(",", " ").split()]
        except ValueError:
            self.out("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.out("Prizes reordered.")
            self.view_prizes()
        else:
            self.out("Invalid order; it must be a permutation of current serial numbers.")

    def start_lottery(self):
        if not self.users:
            self.out("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("Please configure prizes first.")
            return
        self.result_manager = ResultManager()
        lm = LotteryManager(self.users, self.result_manager)
        lm.run_all(prizes, self.ask, self.out)
        self.lottery_done = True
        self.out("\nAll rounds completed.")
        self.show_results()

    def show_results(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not self.result_manager.winners:
            self.out("No winners yet.")
            return
        self.out("===== Complete Winner List =====")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            self.out(f"[{prize_name}]")
            for w in winners:
                self.out(f"  {w.user.name} ({w.user.department}) - {prize_name}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self.out(msg)


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

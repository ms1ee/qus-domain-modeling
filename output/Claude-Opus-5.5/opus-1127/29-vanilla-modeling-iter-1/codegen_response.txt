"""Configuration handling (JSON based) for the dispatch simulator."""

import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG_PATH = "dispatch_config.json"


class Configuration:
    """System parameters, optionally loaded from a JSON file."""

    def __init__(self, file_path=None, initial_funds=1000, recruitment_cost=300,
                 initial_rider_x=435, initial_rider_y=385, default_speed=10):
        self.file_path = file_path
        self.initial_funds = initial_funds
        self.recruitment_cost = recruitment_cost
        self.initial_rider_x = initial_rider_x
        self.initial_rider_y = initial_rider_y
        self.default_speed = default_speed

    def to_dict(self):
        """Return the configuration as a plain dictionary."""
        return {
            "initial_funds": self.initial_funds,
            "recruitment_cost": self.recruitment_cost,
            "initial_rider_x": self.initial_rider_x,
            "initial_rider_y": self.initial_rider_y,
            "default_speed": self.default_speed,
        }

    @classmethod
    def from_dict(cls, data, file_path=None):
        """Build a configuration from a dictionary, ignoring unknown keys."""
        cfg = cls(file_path=file_path)
        for key in cfg.to_dict():
            if key in data:
                try:
                    setattr(cfg, key, int(data[key]))
                except (TypeError, ValueError):
                    logger.warning("Invalid config value for %s: %r", key, data[key])
        return cfg

    @classmethod
    def load(cls, file_path=None):
        """Load configuration from ``file_path``; fall back to defaults."""
        if not file_path or not os.path.exists(file_path):
            return cls(file_path=file_path)
        try:
            with open(file_path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict) and isinstance(data.get("config"), dict):
                data = data["config"]
            return cls.from_dict(data if isinstance(data, dict) else {}, file_path)
        except (OSError, ValueError) as exc:
            logger.warning("Could not load config %s: %s", file_path, exc)
            return cls(file_path=file_path)

    def save(self, file_path=None):
        """Write the configuration to a JSON file."""
        path = file_path or self.file_path or DEFAULT_CONFIG_PATH
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, indent=2)
        return path

"""Grid based path planning.

The map is 800x800 units, split into an 8x8 grid of 100x100 cells.
Lines at multiples of 100 are treated as streets.  Restaurants and
customers sit at cell centres (``x*100 + 50``).
"""

import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = 8


class GridMap:
    """Static description of the map geometry."""

    def __init__(self, map_size=MAP_SIZE, cell_size=CELL_SIZE):
        self.map_size = map_size
        self.cell_size = cell_size
        self.grid_dimension = map_size // cell_size

    def cell_of(self, x, y):
        """Return the (col, row) cell index containing point (x, y)."""
        return (_clamp_cell(x), _clamp_cell(y))


def _clamp_cell(v):
    """Return the cell index for coordinate ``v`` (clamped to 0..7)."""
    return min(max(int(v) // CELL_SIZE, 0), GRID_DIM - 1)


def to_grid_center(value):
    """Convert a raw coordinate (0-800) to the centre of its grid cell.

    ``center = (v // 100) * 100 + 50``; the boundary value 800 maps to the
    last cell (750).

    Raises:
        ValueError: if value is not an integer in range 0..800.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError("Coordinate must be an integer: %r" % (value,))
    if value < 0 or value > MAP_SIZE:
        raise ValueError("Coordinate out of range 0-800: %d" % value)
    return _clamp_cell(value) * CELL_SIZE + CELL_SIZE // 2


class Route:
    """A planned route: list of points (including start), speed, time."""

    def __init__(self, points, speed):
        self.points = list(points)
        self.speed = speed
        self.length = PathPlanner.path_length(self.points)
        self.estimated_time = (int(math.ceil(self.length / float(speed)))
                               if speed > 0 else 0)

    def __repr__(self):
        return "Route(points=%r, speed=%r, time=%r)" % (
            self.points, self.speed, self.estimated_time)


class PathPlanner:
    """Street-preferring Manhattan path planner."""

    grid_map = GridMap()

    @staticmethod
    def nearest_street(v):
        """Return the street line (multiple of 100) closest to ``v``."""
        s = ((int(v) + CELL_SIZE // 2) // CELL_SIZE) * CELL_SIZE
        return min(max(s, 0), MAP_SIZE)

    @staticmethod
    def is_special_case(start, target):
        """True when the rider is directly above/below the target.

        That is: same grid column (or identical x) but different y.
        """
        sx, sy = start
        tx, ty = target
        if sy == ty:
            return False
        return sx == tx or _clamp_cell(sx) == _clamp_cell(tx)

    @classmethod
    def plan_route(cls, start, target):
        """Return the waypoints (excluding start) from ``start`` to ``target``.

        Strategy:
          * already aligned on x or y -> straight line;
          * directly above/below (same column) -> vertical then horizontal;
          * otherwise -> go to nearest horizontal street, follow it to the
            target column, then go vertically to the target.
        """
        start = (int(start[0]), int(start[1]))
        target = (int(target[0]), int(target[1]))
        if start == target:
            return []
        sx, sy = start
        tx, ty = target
        if sx == tx or sy == ty:
            return [target]
        if cls.is_special_case(start, target):
            raw = [(sx, ty), (tx, ty)]
        else:
            street_y = cls.nearest_street(sy)
            raw = [(sx, street_y), (tx, street_y), (tx, ty)]
        points = []
        prev = start
        for p in raw:
            if p != prev:
                points.append(p)
                prev = p
        return points

    @classmethod
    def plan_multi(cls, start, targets, speed):
        """Plan a continuous route through several destinations.

        Returns:
            Route whose points include the start point.
        """
        points = [(int(start[0]), int(start[1]))]
        current = points[0]
        for t in targets:
            seg = cls.plan_route(current, t)
            points.extend(seg)
            current = (int(t[0]), int(t[1]))
        return Route(points, speed)

    @staticmethod
    def path_length(points):
        """Manhattan length of a polyline."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    @staticmethod
    def cells_on_path(points):
        """Return the set of (col, row) cells touched by the polyline."""
        cells = set()
        if not points:
            return cells
        cells.add((_clamp_cell(points[0][0]), _clamp_cell(points[0][1])))
        for (ax, ay), (bx, by) in zip(points, points[1:]):
            steps = max(abs(bx - ax), abs(by - ay))
            n = max(steps // 10, 1)
            for i in range(n + 1):
                x = ax + (bx - ax) * i / n
                y = ay + (by - ay) * i / n
                cells.add((_clamp_cell(x), _clamp_cell(y)))
        return cells

"""Order data model."""

import globalvar
from path_planner import PathPlanner, to_grid_center


class Order:
    """A delivery order from a restaurant to a customer.

    Coordinates are converted to grid cell centres on creation.

    Attributes:
        id: unique order id.
        res_pos / cus_pos: (x, y) tuples of grid-cell centres.
        status: "pending", "assigned", "picking_up", "delivering", "completed".
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        if len(res_pos) != 2 or len(cus_pos) != 2:
            raise ValueError("Positions must be (x, y) pairs")
        self.raw = [res_pos[0], res_pos[1], cus_pos[0], cus_pos[1]]
        self.res_pos = (to_grid_center(res_pos[0]), to_grid_center(res_pos[1]))
        self.cus_pos = (to_grid_center(cus_pos[0]), to_grid_center(cus_pos[1]))
        self.id = Order._next_id
        Order._next_id += 1
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.assign_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.distance = PathPlanner.path_length(
            [self.res_pos] + PathPlanner.plan_route(self.res_pos, self.cus_pos))
        self.efficiency = None

    @classmethod
    def reset_ids(cls, start=1):
        """Reset the id counter (used on system initialisation)."""
        cls._next_id = start

    @property
    def order_id(self):
        """Alias for ``id``."""
        return self.id

    @property
    def is_completed(self):
        """True once the order has been delivered."""
        return self.status == "completed"

    def as_list(self):
        """Return ``[restaurant_x, restaurant_y, customer_x, customer_y]``."""
        return [self.res_pos[0], self.res_pos[1], self.cus_pos[0], self.cus_pos[1]]

    def assign_to(self, rider_id):
        """Mark the order as assigned to ``rider_id``."""
        self.rider_id = rider_id
        self.status = "assigned"
        self.assign_time = globalvar.get_value("tick", 0)

    def mark_picked_up(self):
        """Record that the rider reached the restaurant."""
        self.status = "delivering"
        self.pickup_time = globalvar.get_value("tick", 0)

    def mark_completed(self):
        """Record delivery completion and efficiency data."""
        self.status = "completed"
        self.completion_time = globalvar.get_value("tick", 0)
        self.time_taken = self.completion_time - (self.start_time or 0)
        if self.time_taken and self.time_taken > 0:
            self.efficiency = round(self.distance / float(self.time_taken), 2)
        else:
            self.efficiency = None

    def to_dict(self):
        """Serialise the order to a dictionary."""
        return {
            "id": self.id, "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos), "status": self.status,
            "rider_id": self.rider_id, "start_time": self.start_time,
            "assign_time": self.assign_time, "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken, "distance": self.distance,
            "efficiency": self.efficiency,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from ``to_dict`` output (id counter untouched)."""
        saved_next = cls._next_id
        o = cls(tuple(data["res_pos"]), tuple(data["cus_pos"]))
        cls._next_id = saved_next
        for key in ("id", "status", "rider_id", "start_time", "assign_time",
                    "pickup_time", "completion_time", "time_taken",
                    "distance", "efficiency"):
            if key in data:
                setattr(o, key, data[key])
        return o

    def __repr__(self):
        return "Order(id=%d, res=%r, cus=%r, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

"""Rider model: movement, state machine and status log."""

import logging

import globalvar
from path_planner import PathPlanner, to_grid_center

logger = logging.getLogger(__name__)


class RiderState:
    """Rider state enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2
    NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


DEFAULT_X = 435
DEFAULT_Y = 385
DEFAULT_SPEED = 10


def _order_positions(order):
    """Return (restaurant_pos, customer_pos) for an Order or a 4-list."""
    if hasattr(order, "res_pos") and hasattr(order, "cus_pos"):
        return tuple(order.res_pos), tuple(order.cus_pos)
    rx, ry, cx, cy = order
    return ((to_grid_center(rx), to_grid_center(ry)),
            (to_grid_center(cx), to_grid_center(cy)))


def _order_id(order):
    return getattr(order, "id", None)


class Rider:
    """A delivery rider.

    Attributes:
        rider_id, x, y, speed, state (0/1/2), orders (list), special_case,
        log (list of dicts with keys time, event, order_id, x, y, state).
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        cfg = globalvar.get_value("config")
        self.rider_id = rider_id
        self.x = int(x if x is not None else getattr(cfg, "initial_rider_x", DEFAULT_X))
        self.y = int(y if y is not None else getattr(cfg, "initial_rider_y", DEFAULT_Y))
        self.speed = int(speed if speed is not None
                         else getattr(cfg, "default_speed", DEFAULT_SPEED))
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.route_points = []
        self._route_target = None
        self._record("spawn", None)

    # ------------------------------------------------------------------ helpers
    @property
    def position(self):
        """Current (x, y) tuple."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human readable state."""
        return RiderState.NAMES.get(self.state, str(self.state))

    def is_idle(self):
        """True when idle with no queued orders."""
        return self.state == RiderState.IDLE and not self.orders

    def _record(self, event, order):
        entry = {
            "time": globalvar.get_value("tick", 0),
            "event": event,
            "order_id": _order_id(order) if order is not None else None,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)

    def _reset_route(self):
        self.route_points = []
        self._route_target = None
        self.special_case = False

    def _advance(self, target):
        """Move up to ``speed`` units along the planned path to target."""
        if self._route_target != target or not self.route_points:
            self.route_points = PathPlanner.plan_route(self.position, target)
            self.special_case = PathPlanner.is_special_case(self.position, target)
            self._route_target = target
        remaining = self.speed
        while remaining > 0 and self.route_points:
            wx, wy = self.route_points[0]
            dx = wx - self.x
            dy = wy - self.y
            if dx:
                step = min(remaining, abs(dx))
                self.x += step if dx > 0 else -step
                remaining -= step
            elif dy:
                step = min(remaining, abs(dy))
                self.y += step if dy > 0 else -step
                remaining -= step
            if (self.x, self.y) == (wx, wy):
                self.route_points.pop(0)

    # --------------------------------------------------------------------- tick
    def move(self):
        """Advance the rider by one tick.

        Handles the state machine:
          idle -> picking up (order accepted),
          picking up -> delivering (arrived at merchant),
          delivering -> idle (delivered to customer).
        """
        if self.state == RiderState.IDLE:
            if not self.orders:
                return
            order = self.orders[0]
            self.state = RiderState.PICKING_UP
            if hasattr(order, "status"):
                order.status = "picking_up"
            self._reset_route()
            self._record("accepted_order", order)

        if not self.orders:
            self.state = RiderState.IDLE
            self._reset_route()
            return

        order = self.orders[0]
        res_pos, cus_pos = _order_positions(order)
        target = res_pos if self.state == RiderState.PICKING_UP else cus_pos

        if self.position != target:
            self._advance(target)

        if self.position == target:
            if self.state == RiderState.PICKING_UP:
                self.state = RiderState.DELIVERING
                if hasattr(order, "mark_picked_up"):
                    order.mark_picked_up()
                self._reset_route()
                self._record("arrived_at_merchant", order)
            else:
                self.orders.pop(0)
                if hasattr(order, "mark_completed"):
                    order.mark_completed()
                elif hasattr(order, "status"):
                    order.status = "completed"
                self.state = RiderState.IDLE
                self._reset_route()
                self._record("delivered_to_customer", order)

    # ------------------------------------------------------------ persistence
    def to_dict(self):
        """Serialise the rider."""
        return {
            "rider_id": self.rider_id, "x": self.x, "y": self.y,
            "speed": self.speed, "state": self.state,
            "special_case": self.special_case,
            "orders": [_order_id(o) for o in self.orders],
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%d,%d), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state, len(self.orders))

"""Order management: creation, assignment and bookkeeping."""

import logging
import math

import globalvar
from order_model import Order
from rider import RiderState, _order_positions

logger = logging.getLogger(__name__)


def _distance(rider, pos):
    return math.sqrt((rider.x - pos[0]) ** 2 + (rider.y - pos[1]) ** 2)


def find_best_rider_for_order(order):
    """Return the rider closest (Euclidean) to the order's restaurant.

    Idle riders are preferred; if none are idle, the closest rider overall
    is chosen.  Returns None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    res_pos, _ = _order_positions(order)
    idle = [r for r in riders if r.state == RiderState.IDLE and not r.orders]
    pool = idle or list(riders)
    best = None
    best_dist = None
    for r in pool:
        d = _distance(r, res_pos)
        if best is None or d < best_dist:
            best, best_dist = r, d
    return best


class OrderManager:
    """Handles order creation, assignment and completion queries."""

    def __init__(self, system=None):
        self.system = system
        if globalvar.get_value("orders") is None:
            globalvar.set_value("orders", [])

    @property
    def orders(self):
        """All orders."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create an order from ``[rx, ry, cx, cy]`` and register it."""
        if len(coords) != 4:
            raise ValueError("Exactly four coordinates are required")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        self.orders.append(order)
        logger.info("Created %r", order)
        return order

    def pending_orders(self):
        """Orders not yet assigned to any rider."""
        return [o for o in self.orders if o.status == "pending"]

    def completed_orders(self):
        """Delivered orders."""
        return [o for o in self.orders if o.status == "completed"]

    def assign_order(self, order):
        """Assign ``order`` to the best rider; return the rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

    def assign_pending(self):
        """Try to assign every pending order. Returns list of (order, rider)."""
        result = []
        for order in self.pending_orders():
            rider = self.assign_order(order)
            if rider is None:
                break
            result.append((order, rider))
        return result

    def get_order(self, order_id):
        """Return the order with ``order_id`` or None."""
        for o in self.orders:
            if o.id == order_id:
                return o
        return None

"""System manager and initialisation."""

import json
import logging

import globalvar
from config import Configuration
from order import OrderManager
from order_model import Order
from path_planner import GridMap, PathPlanner
from rider import Rider, RiderState

logger = logging.getLogger(__name__)


class SystemManager:
    """Core manager of funds, riders, time and orders (state in globalvar)."""

    def __init__(self, config=None):
        self.config = config or globalvar.get_value("config") or Configuration()
        self.recruitment_cost = self.config.recruitment_cost
        self.grid_map = GridMap()
        self.order_manager = OrderManager(self)

    # -------------------------------------------------------------- properties
    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", value)

    @property
    def riders(self):
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def rider_id_counter(self):
        return globalvar.get_value("rider_id_counter", 0)

    @property
    def is_running(self):
        return globalvar.get_value("running", False)

    @is_running.setter
    def is_running(self, value):
        globalvar.set_value("running", bool(value))

    @property
    def current_tick(self):
        return globalvar.get_value("tick", 0)

    # ----------------------------------------------------------------- riders
    def recruit_rider(self):
        """Hire a rider if funds allow. Returns the rider or None."""
        if self.funds < self.recruitment_cost:
            return None
        self.funds = self.funds - self.recruitment_cost
        new_id = self.rider_id_counter + 1
        globalvar.set_value("rider_id_counter", new_id)
        rider = Rider(new_id, self.config.initial_rider_x,
                      self.config.initial_rider_y, self.config.default_speed)
        self.riders.append(rider)
        logger.info("Recruited rider %d", new_id)
        return rider

    def idle_rider_count(self):
        """Number of idle riders with no queued orders."""
        return sum(1 for r in self.riders if r.is_idle())

    def check_recruitment(self):
        """Recruit while funds >= cost and pending orders > idle riders."""
        added = []
        while (self.funds >= self.recruitment_cost and
               len(self.order_manager.pending_orders()) > self.idle_rider_count()):
            rider = self.recruit_rider()
            if rider is None:
                break
            added.append(rider)
        return added

    def get_rider(self, rider_id):
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    # ----------------------------------------------------------------- orders
    def place_order(self, coords):
        """Create and assign an order.

        Returns:
            dict with keys order, new_riders, rider, route.
        """
        order = self.order_manager.create_order(coords)
        new_riders = self.check_recruitment()
        rider = self.order_manager.assign_order(order)
        route = None
        if rider is not None:
            route = PathPlanner.plan_multi(rider.position,
                                           [order.res_pos, order.cus_pos],
                                           rider.speed)
        return {"order": order, "new_riders": new_riders,
                "rider": rider, "route": route}

    # ------------------------------------------------------------------- time
    def tick(self, n=1):
        """Advance the simulation by ``n`` ticks."""
        for _ in range(max(int(n), 0)):
            globalvar.set_value("tick", self.current_tick + 1)
            try:
                self.check_recruitment()
                self.order_manager.assign_pending()
                for r in list(self.riders):
                    r.move()
            except Exception as exc:  # error recovery
                logger.exception("Error during tick: %s", exc)
            self.consistency_check()

    def consistency_check(self):
        """Repair obviously inconsistent rider states."""
        for r in self.riders:
            if r.state != RiderState.IDLE and not r.orders:
                logger.warning("Rider %s inconsistent, resetting", r.rider_id)
                r.state = RiderState.IDLE
            r.x = min(max(r.x, 0), 800)
            r.y = min(max(r.y, 0), 800)

    def stop(self):
        self.is_running = False

    # ------------------------------------------------------------ persistence
    def save_state(self, path):
        """Save config, riders and orders to a JSON file."""
        data = {
            "config": self.config.to_dict(),
            "funds": self.funds,
            "tick": self.current_tick,
            "rider_id_counter": self.rider_id_counter,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.orders],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

    def load_state(self, path):
        """Load state written by ``save_state``."""
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        self.config = Configuration.from_dict(data.get("config", {}), path)
        self.recruitment_cost = self.config.recruitment_cost
        globalvar.set_value("config", self.config)
        globalvar.set_value("funds", int(data.get("funds", 0)))
        globalvar.set_value("tick", int(data.get("tick", 0)))
        globalvar.set_value("rider_id_counter", int(data.get("rider_id_counter", 0)))
        orders = [Order.from_dict(o) for o in data.get("orders", [])]
        Order.reset_ids(max([o.id for o in orders] + [0]) + 1)
        globalvar.set_value("orders", orders)
        by_id = {o.id: o for o in orders}
        riders = []
        for rd in data.get("riders", []):
            r = Rider(rd["rider_id"], rd["x"], rd["y"], rd["speed"])
            r.state = int(rd.get("state", 0))
            r.special_case = bool(rd.get("special_case", False))
            r.orders = [by_id[i] for i in rd.get("orders", []) if i in by_id]
            r.log = list(rd.get("log", []))
            riders.append(r)
        globalvar.set_value("riders", riders)


def init_system(config_path=None):
    """Initialise global state and return a SystemManager.

    Funds 1000, empty rider list, rider id counter 0, running True, tick 0.
    """
    globalvar._init()
    config = Configuration.load(config_path)
    globalvar.set_value("config", config)
    globalvar.set_value("funds", config.initial_funds)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    Order.reset_ids()
    manager = SystemManager(config)
    globalvar.set_value("system", manager)
    return manager

class DisplayManager:

    def render_map(self, riders=None, orders=None, route_points=None):
        """Render the 8x8 map.

        Symbols: R rider, S restaurant, C customer, # route/street, . empty.
        """
        riders = self.system.riders if riders is None else riders
        if orders is None:
            orders = [o for o in self.system.order_manager.orders
                      if o.status != "completed"]
        grid = [["." for _ in range(GRID_DIM)] for _ in range(GRID_DIM)]
        if route_points:
            for (c, r) in PathPlanner.cells_on_path(route_points):
                grid[r][c] = "#"
        for o in orders:
            grid[_clamp_cell(o.cus_pos[1])][_clamp_cell(o.cus_pos[0])] = "C"
        for o in orders:
            grid[_clamp_cell(o.res_pos[1])][_clamp_cell(o.res_pos[0])] = "S"
        for rd in riders:
            grid[_clamp_cell(rd.y)][_clamp_cell(rd.x)] = "R"
        lines = ["    " + " ".join(str(i) for i in range(GRID_DIM))]
        for idx, row in enumerate(grid):
            lines.append("%3d " % (idx * 100) + " ".join(row))
        lines.append("Legend: R=Rider S=Restaurant C=Customer #=Route .=Empty")
        return "\n".join(lines)

    def format_order_result(self, result):
        """Format output of SystemManager.place_order."""
        order = result["order"]
        lines = [
            "Order #%d created" % order.id,
            "  Restaurant grid position: %s" % self._fmt_pos(order.res_pos),
            "  Customer grid position:   %s" % self._fmt_pos(order.cus_pos),
        ]
        if result["new_riders"]:
            ids = ", ".join(str(r.rider_id) for r in result["new_riders"])
            lines.append("  New rider added: yes (id %s), funds now %d"
                         % (ids, self.system.funds))
        else:
            lines.append("  New rider added: no")
        rider = result["rider"]
        if rider is None:
            lines.append("  No rider available; order is pending.")
            return "\n".join(lines)
        route = result["route"]
        lines.append("  Assigned rider: %d at %s (%s, %d queued orders)"
                     % (rider.rider_id, self._fmt_pos(rider.position),
                        rider.state_name, len(rider.orders)))
        lines.append("  Route: " + " -> ".join(self._fmt_pos(p) for p in route.points))
        lines.append("  Speed: %d units/tick, Distance: %d, Estimated time: %d ticks"

"""Command parsing and input validation."""

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "q": "quit", "exit": "quit", "h": "help", "?": "help",
    "m": "map", "t": "tick", "hist": "history",
}

COMMANDS = ("start", "status", "orders", "riders", "quit", "help",
            "map", "tick", "history", "save", "load")

HELP_TEXT = """Commands:
  start (s) [rx ry cx cy]  create an order (coordinates 0-800)
  status (st)              funds, riders and map
  orders (o)               all order details
  riders (r)               rider status-change logs
  map (m)                  show map
  tick (t) [n]             advance simulation n ticks (default 1)
  history (hist)           show command history
  save <file> / load <file> persist state as JSON
  quit / exit (q)          exit"""


class InputManager:
    """Parses commands, keeps history and validates coordinates."""

    def __init__(self):
        self.command_history = []
        try:  # optional line editing / history support
            import readline  # noqa: F401
        except ImportError:
            pass

    def parse_command(self, line):
        """Return (command, args). Unknown commands are returned as-is."""
        line = (line or "").strip()
        if not line:
            return "", []
        self.command_history.append(line)
        parts = line.split(None, 1)
        cmd = parts[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        rest = parts[1] if len(parts) > 1 else ""
        return cmd, self.tokenize(rest)

    @staticmethod
    def tokenize(text):
        """Split text on spaces/commas, ignoring brackets."""
        for ch in "[](),":
            text = text.replace(ch, " ")
        return text.split()

    @staticmethod
    def validate_coordinates(tokens):
        """Validate four integer coordinates in 0..800.

        Raises:
            ValueError: with a user-friendly message.
        """
        if len(tokens) != 4:
            raise ValueError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for t in tokens:
            if isinstance(t, bool):
                raise ValueError("Invalid coordinate: %r" % (t,))
            if isinstance(t, int):
                v = t
            else:
                try:
                    v = int(str(t).strip())
                except ValueError:
                    raise ValueError("'%s' is not an integer" % t)
            if v < 0 or v > 800:
                raise ValueError("Coordinate %d out of range 0-800" % v)
            values.append(v)
        return values

"""Command-line entry point for the food delivery dispatch simulator."""

import logging
import sys

from display import DisplayManager
from input_manager import HELP_TEXT, InputManager
from system import init_system


def run(config_path=None, stream_in=None):
    """Run the interactive main loop. Each loop iteration is one tick."""
    system = init_system(config_path)
    display = DisplayManager(system)
    inputs = InputManager()
    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
    while system.is_running:
        try:
            if stream_in is not None:
                line = stream_in.readline()
                if not line:
                    break
            else:
                line = input("[t=%d] > " % system.current_tick)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        try:
            cmd, args = inputs.parse_command(line)
            if cmd == "":
                pass
            elif cmd == "quit":
                system.stop()
                print("Bye.")
                break
            elif cmd == "help":
                print(HELP_TEXT)
            elif cmd == "start":
                if not args:
                    raw = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
                    args = inputs.tokenize(raw)
                coords = inputs.validate_coordinates(args)
                print(display.format_order_result(system.place_order(coords)))
            elif cmd == "status":
                print(display.format_status())
            elif cmd == "orders":
                print(display.format_orders())
            elif cmd == "riders":
                print(display.format_rider_logs())
            elif cmd == "map":
                print(display.render_map())
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                if n < 0:
                    raise ValueError("Tick count must be non-negative")
                system.tick(n - 1 if n > 0 else 0)
                print("Advanced %d tick(s)." % n)
            elif cmd == "history":
                for i, h in enumerate(inputs.command_history, 1):
                    print("%3d  %s" % (i, h))
            elif cmd == "save":
                path = args[0] if args else "dispatch_state.json"
                print("Saved to %s" % system.save_state(path))
            elif cmd == "load":
                path = args[0] if args else "dispatch_state.json"
                system.load_state(path)
                print("Loaded %s" % path)
            else:
                print("Unknown command '%s'. Type 'help'." % cmd)
        except ValueError as exc:
            print("Error: %s" % exc)
        except (OSError, KeyError) as exc:
            print("Error: %s" % exc)
        except Exception as exc:  # graceful recovery
            logging.exception("Unexpected error")
            print("Unexpected error: %s (system recovered)" % exc)
        if system.is_running:
            system.tick(1)
    return 0


def main():
    """Program entry."""
    logging.basicConfig(level=logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")
    config_path = sys.argv[1] if len(sys.argv) > 1 else None
    return run(config_path)


if __name__ == "__main__":
    sys.exit(main())

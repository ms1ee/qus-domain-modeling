"""Core data models for the lottery system."""

from __future__ import annotations

from typing import List, Optional


class User:
    """A participant of the lottery (an employee)."""

    def __init__(self, name: str, department: str):
        self.name = str(name).strip()
        self.department = str(department).strip()

    def __eq__(self, other) -> bool:
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self) -> int:
        return hash((self.name, self.department))

    def __repr__(self) -> str:
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


class Prize:
    """A prize to be drawn in one lottery round."""

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups: List[str] = [
            str(g).strip() for g in (avoid_groups or []) if str(g).strip()
        ]
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        self.associated_drawer: Optional[str] = drawer or None

    def __repr__(self) -> str:
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
            f"avoid_groups={self.avoid_groups!r}, "
            f"associated_drawer={self.associated_drawer!r})"
        )

    def __str__(self) -> str:
        parts = [f"{self.name} x{self.quantity}"]
        if self.avoid_groups:
            parts.append("avoids: " + ", ".join(self.avoid_groups))
        if self.associated_drawer:
            parts.append("drawer: " + self.associated_drawer)
        return " | ".join(parts)

_SPLIT_RE = re.compile(r"\s*[,\t，]\s*")


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self):
        self.users: List[User] = []
        self.errors: List[str] = []

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        self.users, self.errors = users, errors
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        if not path.lower().endswith(".txt"):
            # still try to load, but txt is the expected format
            pass
        users, errors = self._load_from_path(path)
        self.users, self.errors = users, errors
        return users, errors

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as fh:
                    lines = fh.read().splitlines()
            except (OSError, UnicodeDecodeError) as exc:
                return [], [f"File Error: cannot read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        header_checked = False
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = _SPLIT_RE.split(line)
            if not header_checked:
                header_checked = True
                if len(parts) == 2 and parts[0].lower() == "name" and parts[1].lower() in (
                    "department",
                    "group",
                    "dept",
                ):
                    continue
            if len(parts) != 2:
                errors.append(
                    f"Format Error: Line {idx}: expected 'Name,Department' but got '{line}'"
                )
                continue
            name, dept = parts[0].strip(), parts[1].strip()
            if not name or not dept:
                errors.append(
                    f"Format Error: Line {idx}: name and department must not be empty ('{line}')"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    # --------------------------------------------------------------- validation
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    # ------------------------------------------------------------------- groups
    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and lottery round order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name,
        quantity,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ) -> Prize:
        if name is None or not str(name).strip():
            raise ValueError("Prize name must not be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError(f"Prize quantity must be a positive integer: {quantity}")
        name = str(name).strip()
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        if isinstance(avoid_groups, str):
            avoid_groups = [g for g in avoid_groups.split(",")]
        prize = Prize(name, qty, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List, Optional

from lottery_system.models import Prize, User

_LEVEL_KEYWORDS = [
    ("special", 0),
    ("grand", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
    ("consolation", 90),
    ("participation", 91),
]


def prize_level_rank(name: str) -> int:
    lower = name.lower()
    for kw, rank in _LEVEL_KEYWORDS:
        if kw in lower:
            return rank
    return 50


def sort_prizes_by_level(prizes) -> List:
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda t: (prize_level_rank(_prize_name(t[1])), -t[0]))
    return [p for _, p in indexed]


def _prize_name(prize) -> str:
    return prize.name if hasattr(prize, "name") else str(prize)


class Winner:
    """A winning record."""

    def __init__(self, user: User, prize, order: int, win_time: Optional[datetime] = None):
        self.user = user
        self.prize = prize if isinstance(prize, Prize) else None
        self.prize_name = _prize_name(prize)
        self.order = order
        self.win_time = win_time or datetime.now()

    def __repr__(self) -> str:
        return f"Winner(user={self.user!r}, prize={self.prize_name!r}, order={self.order})"


class ResultManager:
    """Maintains the winner list in real time."""

    def __init__(self):
        self.winners: List[Winner] = []
        self._names = set()

    def has_won(self, name: str) -> bool:
        return name in self._names

    def add_winner(self, user, prize) -> bool:
        if user.name in self._names:
            return False
        self.winners.append(Winner(user, prize, len(self.winners) + 1))
        self._names.add(user.name)
        return True

    def get_winners(self) -> List[Winner]:
        return list(self.winners)

    def clear(self) -> None:
        self.winners.clear()
        self._names.clear()

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for prize in sort_prizes_by_level(list(prizes or [])):
            result[_prize_name(prize)] = []
        extra: Dict[str, List[Winner]] = {}
        for w in sorted(self.winners, key=lambda x: (x.win_time, x.order)):
            if w.prize_name in result:
                result[w.prize_name].append(w)
            else:
                extra.setdefault(w.prize_name, []).append(w)
        result.update(extra)
        return result

    def export_to_csv(self, prizes, directory: Optional[str] = None):
        if not self.winners:
            return False, "No winners to export"
        directory = directory or os.getcwd()
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(directory, filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", encoding="utf-8", newline="") as fh:
                writer = csv.writer(fh)
                writer.writerow(["No", "Name", "Department", "Prize", "Winning Order", "Winning Time"])
                no = 1
                for prize_name, winners in grouped.items():
                    for w in winners:
                        writer.writerow([
                            no,
                            w.user.name,
                            w.user.department,
                            prize_name,
                            w.order,
                            w.win_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
                        no += 1
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery process control and rule engine."""

from __future__ import annotations

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager


def _norm(text: Optional[str]) -> str:
    return " ".join(str(text or "").lower().split())


def _norm_group(text: Optional[str]) -> str:
    t = _norm(text)
    if t.endswith(" group"):
        t = t[: -len(" group")].strip()
    return t


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryManager:
    """Rule engine + round execution."""

    PROTECTED_GROUP = "Cheng Ru group"
    RELAXED_DRAWER = "Li Bin"
    FALLBACK_TRIGGERS = ("Chen Qixian", "Jin Duo")
    FALLBACK_TARGET = "Ma Jiaqi"

    def __init__(self, users, result_manager: Optional[ResultManager] = None, rng=None):
        self.users: List[User] = list(users or [])
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winner_names = {w.user.name for w in self.result_manager.get_winners()}
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.SystemRandom()
        self.round_number = 0

    # ------------------------------------------------------------ rule helpers
    def is_protected_member(self, user: User) -> bool:
        return _norm_group(user.department) == _norm_group(self.PROTECTED_GROUP)

    def is_relaxed(self, prize: Prize) -> bool:
        return _norm(prize.associated_drawer) == _norm(self.RELAXED_DRAWER)

    def is_fallback_active(self, prize: Prize) -> bool:
        if _norm(prize.associated_drawer) != _norm(self.FALLBACK_TARGET):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) >= 2 and not any(recent)

    def _has_won(self, user: User) -> bool:
        return user.name in self.winner_names or self.result_manager.has_won(user.name)

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if not self._has_won(u)]
        if self.is_fallback_active(prize):
            protected = [u for u in base if self.is_protected_member(u)]
            if protected:
                return protected
        if self.is_relaxed(prize):
            return base
        avoid = {_norm_group(g) for g in prize.avoid_groups}
        return [u for u in base if _norm_group(u.department) not in avoid]

    def get_rule_description(self, prize: Prize) -> str:
        rules = ["previous winners excluded"]
        if self.is_fallback_active(prize):
            rules.append(
                f"FALLBACK ACTIVE: pool limited to '{self.PROTECTED_GROUP}' members "
                f"(Cheng Ru group protection)"
            )
        elif self.is_relaxed(prize):
            rules.append(f"relaxed rules for drawer {self.RELAXED_DRAWER}: avoidance groups ignored")
        elif prize.avoid_groups:
            rules.append("avoidance groups: " + ", ".join(prize.avoid_groups))
        else:
            rules.append("no avoidance groups")
        if prize.associated_drawer:
            rules.append(f"drawer: {prize.associated_drawer}")
        return "; ".join(rules)

    # ---------------------------------------------------------------- drawing
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self.rng.choice(pool)
        if not self.result_manager.add_winner(winner, prize):
            # Should not happen: pool excludes winners. Guard anyway.
            self.winner_names.add(winner.name)
            return None
        self.winner_names.add(winner.name)
        return winner

    def finish_round(self, prize: Prize, round_winners: List[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        triggers = {_norm(t) for t in self.FALLBACK_TRIGGERS}
        if drawer in triggers:
            self.last_two_draw_results.append(
                any(self.is_protected_member(u) for u in round_winners)
            )
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            self.last_two_draw_results = []

    def run_round(
        self,
        prize: Prize,
        input_func: Callable[[str], str] = input,
        output_func: Callable[[str], None] = print,
    ) -> List[User]:
        self.round_number += 1
        output_func("")
        output_func("=" * 60)
        output_func(f"Round {self.round_number}: {prize.name}")
        output_func(f"Quantity: {prize.quantity}")
        output_func(f"Rules: {self.get_rule_description(prize)}")
        output_func("=" * 60)
        output_func("Commands: [Enter]=draw next, 'auto'=draw all remaining, "
                    "'pause', 'continue', 'terminate'")
        round_winners: List[User] = []
        auto = False
        paused = False
        terminated = False
        i = 0
        while i < prize.quantity:
            if not auto:
                try:
                    cmd = _norm(input_func(
                        "(paused) > " if paused
                        else f"Draw {ordinal(i + 1)} winner > "
                    ))
                except EOFError:
                    cmd = "terminate"
                if cmd == "terminate":
                    terminated = True
                    output_func("Current round terminated and skipped.")
                    break
                if cmd == "pause":
                    paused = True
                    output_func("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                    continue
                if paused:
                    if cmd == "continue":
                        paused = False
                        output_func("Lottery resumed.")
                    else:
                        output_func("Lottery is paused. Type 'continue' or 'terminate'.")
                    continue
                if cmd == "continue":
                    output_func("Lottery is not paused.")
                    continue
                if cmd == "auto":
                    auto = True
                elif cmd != "":
                    output_func(f"Unknown command: {cmd}")
                    continue
            output_func(f"Currently drawing the {ordinal(i + 1)} winner...")
            winner = self.draw_one(prize)
            if winner is None:
                output_func("No eligible participants remain for this prize. Round ends early.")
                break
            round_winners.append(winner)
            output_func(
                f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}"
            )
            i += 1
        self.finish_round(prize, round_winners)
        output_func("-" * 60)
        status = "terminated" if terminated else "completed"
        output_func(f"Round {self.round_number} {status}. Results ({len(round_winners)}/{prize.quantity}):")
        if round_winners:
            for n, u in enumerate(round_winners, start=1):
                output_func(f"  {n}. {u.name} ({u.department}) - {prize.name}")
        else:
            output_func("  No winners in this round.")
        return round_winners

"""Command-line interface for the lottery system."""

from __future__ import annotations

from typing import Callable, List

from lottery_system.models import Prize, User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager

MENU = """
========== Team Building Smart Lottery ==========
 1. Load default participant list
 2. Upload custom participant list (txt, 'Name,Department' per line)
 3. View groups
 4. Load preset prize library
 5. Add custom prize
 6. View prizes
 7. Reorder prizes
 8. Start lottery
 9. View result summary
10. Export results
 0. Exit
================================================="""


class LotteryApp:
    def __init__(self, input_func: Callable[[str], str] = input,
                 output_func: Callable[[str], None] = print):
        self.inp = input_func
        self.out = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_manager: LotteryManager | None = None
        self.drawn_prizes = set()

    def ask(self, prompt: str) -> str:
        return self.inp(prompt).strip()

    # ------------------------------------------------------------------ lists
    def _set_users(self, users: List[User]) -> None:
        if self.result_manager.get_winners():
            self.out("Note: a new list was loaded; previous lottery results are cleared.")
        self.users = users
        self.result_manager.clear()
        self.drawn_prizes.clear()
        self.lottery_manager = LotteryManager(users, self.result_manager)
        self.out(f"{len(users)} valid participants loaded.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        for e in errors:
            self.out(e)
        valid, dups = self.list_manager.validate_users(users)
        self._set_users(valid)

    def load_custom(self) -> None:
        path = self.ask("Enter txt file path: ")
        users, errors = self.list_manager.load_custom_list(path)
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            self.out("Validation issues detected:")
            for e in errors:
                self.out("  " + e)
            for d in dups:
                self.out(f"  Duplicate Name: {d.name} ({d.department})")
            if not valid:
                self.out("No valid data. Upload terminated.")
                return
            choice = self.ask("Type 'r' to remove invalid data and continue, 't' to terminate upload: ").lower()
            if choice != "r":
                self.out("Upload terminated.")
                return
        if not valid:
            self.out("No valid participants found.")
            return
        self._set_users(valid)

    def view_groups(self) -> List[str]:
        info = self.list_manager.get_group_info(self.users)
        if not info:
            self.out("No participants loaded.")
            return []
        self.out("Current groups:")
        names = list(info.keys())
        for i, name in enumerate(names, start=1):
            self.out(f"  {i}. {name}: {info[name]} people")
        return names

    # ----------------------------------------------------------------- prizes
    def load_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            self.out(e)
        self.out("Preset prize library loaded.")

    def add_prize(self) -> None:
        name = self.ask("Prize name: ")
        qty = self.ask("Quantity: ")
        groups = self.view_groups()
        avoid: List[str] = []
        raw = self.ask("Avoidance groups (numbers or names, comma separated, blank for none): ")
        if raw:
            for tok in raw.replace("，", ",").split(","):
                tok = tok.strip()
                if not tok:
                    continue
                if tok.isdigit() and 1 <= int(tok) <= len(groups):
                    avoid.append(groups[int(tok) - 1])
                else:
                    avoid.append(tok)
        drawer = self.ask("Associated lottery person (blank for none): ") or None
        try:
            prize = self.prize_manager.add_prize(name, qty, avoid, drawer)
            self.out(f"Prize added: {prize}")
        except ValueError as exc:
            self.out(f"Error: {exc}")

    def view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("No prizes configured.")
            return
        for i, p in enumerate(prizes, start=1):
            mark = " [drawn]" if p.name in self.drawn_prizes else ""
            self.out(f"  {i}. {p}{mark}")

    def reorder(self) -> None:
        self.view_prizes()
        raw = self.ask("Enter new order as serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            self.out("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.out("Prize order updated.")
            self.view_prizes()
        else:
            self.out("Invalid order: must be a permutation of all serial numbers.")

    # ---------------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.users or self.lottery_manager is None:
            self.out("Please load a participant list first.")
            return
        prizes = [p for p in self.prize_manager.get_prizes() if p.name not in self.drawn_prizes]
        if not prizes:
            self.out("No prizes left to draw. Please configure prizes.")
            return
        for prize in prizes:
            self.lottery_manager.run_round(prize, self.inp, self.out)
            self.drawn_prizes.add(prize.name)
        self.out("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not self.result_manager.get_winners():
            self.out("No winners yet.")
            return
        self.out("\n========== Complete Winner List ==========")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            self.out(f"[{prize_name}]")
            for w in winners:
                self.out(f"  {w.user.name} ({w.user.department}) - {prize_name} "
                         f"@ {w.win_time.strftime('%H:%M:%S')}")

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self.out(msg)

    # ------------------------------------------------------------------- loop
    def run(self) -> None:
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_prizes, "5": self.add_prize, "6": self.view_prizes,
            "7": self.reorder, "8": self.start_lottery, "9": self.show_summary,
            "10": self.export,
        }
        while True:
            self.out(MENU)
            try:
                choice = self.ask("Select: ")
            except (EOFError, KeyboardInterrupt):
                self.out("\nBye.")
                return
            if choice == "0":
                self.out("Bye.")
                return
            action = actions.get(choice)
            if action is None:
                self.out("Invalid option.")
                continue
            try:
                action()
            except (EOFError, KeyboardInterrupt):
                self.out("\nBye.")
                return


def main() -> None:
    LotteryApp().run()


if __name__ == "__main__":
    main()

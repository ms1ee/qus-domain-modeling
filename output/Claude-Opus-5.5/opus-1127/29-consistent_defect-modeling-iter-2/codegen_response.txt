"""Grid map definition and street-based path planning."""

from dataclasses import dataclass, field
from typing import List, Optional, Tuple

CELL_SIZE = 100
MAP_SIZE = 800
GRID_DIMENSION = MAP_SIZE // CELL_SIZE

Point = Tuple[int, int]


class GridMap:
    """An 8x8 grid of 100x100 cells covering an 800x800 coordinate space.

    Grid lines (x or y a multiple of the cell size) are streets; cell
    centres host restaurants and customers.
    """

    def __init__(self, width=MAP_SIZE, height=MAP_SIZE, cell_size=CELL_SIZE):
        self.width = width
        self.height = height
        self.cell_size = cell_size
        self.grid_dimension = width // cell_size

    def clamp(self, value, upper=None):
        """Clamp a coordinate into the map range."""
        upper = self.width if upper is None else upper
        return max(0, min(int(value), upper))

    def to_cell_center(self, value):
        """Convert a raw coordinate into the centre of its grid cell."""
        v = max(0, min(int(value), self.width - 1))
        return (v // self.cell_size) * self.cell_size + self.cell_size // 2

    def cell_of(self, x, y):
        """Return (column, row) of the cell containing the point."""
        cx = max(0, min(int(x), self.width - 1)) // self.cell_size
        cy = max(0, min(int(y), self.height - 1)) // self.cell_size
        return cx, cy

    def is_street(self, x, y):
        """True if the point lies on a street (grid line)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0


@dataclass
class Waypoint:
    """A single point on a route."""
    x: int
    y: int
    sequence: int


@dataclass
class Route:
    """A planned route made of axis-aligned segments."""
    waypoints: List[Waypoint] = field(default_factory=list)
    speed: int = 20
    length: int = 0
    estimated_time: int = 0

    def points(self) -> List[Point]:
        """Return the route as a list of (x, y) tuples."""
        return [(w.x, w.y) for w in self.waypoints]


def manhattan(a: Point, b: Point) -> int:
    """Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class PathPlanner:
    """Plans routes that travel along streets with right-angle turns."""

    def __init__(self, cell_size=CELL_SIZE, map_size=MAP_SIZE):
        self.cell_size = cell_size
        self.map_size = map_size

    # ------------------------------------------------------------------ utils
    def is_street(self, x, y):
        """True if (x, y) is on a grid line."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    @staticmethod
    def is_special_case(sx, sy, tx, ty):
        """Rider directly above or below its target (same column)."""
        return sx == tx and sy != ty

    def _best_line(self, a, b):
        """Choose a grid line minimising |a-l| + |l-b|."""
        cs = self.cell_size
        cands = set()
        for v in (a, b):
            cands.add((v // cs) * cs)
            cands.add(-(-v // cs) * cs)
        cands = {max(0, min(c, self.map_size)) for c in cands}
        return min(cands, key=lambda l: (abs(a - l) + abs(l - b), abs(a - l), l))

    @staticmethod
    def _dedupe(points, start):
        result = []
        prev = start
        for p in points:
            if p != prev:
                result.append(p)
                prev = p
        return result

    def _street_route(self, x, y, ex, ey):
        cs = self.cell_size
        if (x, y) == (ex, ey):
            return []
        on_h, on_v = y % cs == 0, x % cs == 0
        e_h, e_v = ey % cs == 0, ex % cs == 0
        if on_h and e_h and y == ey:
            return [(ex, ey)]
        if on_v and e_v and x == ex:
            return [(ex, ey)]
        if on_v and e_h:
            return [(x, ey), (ex, ey)]
        if on_h and e_v:
            return [(ex, y), (ex, ey)]
        if on_v:
            yh = self._best_line(y, ey)
            return [(x, yh), (ex, yh), (ex, ey)]
        xv = self._best_line(x, ex)
        return [(xv, y), (xv, ey), (ex, ey)]

    # ---------------------------------------------------------------- planning
    def plan(self, sx, sy, tx, ty) -> List[Point]:
        """Plan a path from (sx, sy) to (tx, ty).

        Returns the list of waypoints (excluding the start). Consecutive
        waypoints are always axis-aligned.
        """
        cs = self.cell_size
        sx, sy, tx, ty = int(sx), int(sy), int(tx), int(ty)
        if (sx, sy) == (tx, ty):
            return []
        # Both inside the same cell: move directly inside the cell.
        if (not self.is_street(sx, sy) and not self.is_street(tx, ty)
                and sx // cs == tx // cs and sy // cs == ty // cs):
            return self._dedupe([(sx, ty), (tx, ty)], (sx, sy))
        points = []
        x, y = sx, sy
        if not self.is_street(x, y):
            # Exit the cell to the street on its top edge.
            y = (y // cs) * cs
            points.append((x, y))
        if self.is_street(tx, ty):
            entry, final = (tx, ty), None
        else:
            entry, final = (tx, (ty // cs) * cs), (tx, ty)
        points.extend(self._street_route(x, y, entry[0], entry[1]))
        if final is not None:
            points.append(final)
        return self._dedupe(points, (sx, sy))

    def path_length(self, start: Point, points: List[Point]) -> int:
        """Total Manhattan length of a path."""
        total, prev = 0, start
        for p in points:
            total += manhattan(prev, p)
            prev = p
        return total

    def plan_multi(self, start: Point, targets: List[Point]):
        """Plan a continuous path visiting several destinations in order.

        Returns (points, length).
        """
        points, pos = [], start
        for t in targets:
            seg = self.plan(pos[0], pos[1], t[0], t[1])
            points.extend(seg)
            pos = t
        return points, self.path_length(start, points)

    def build_route(self, start: Point, points: List[Point], speed: int) -> Route:
        """Wrap a list of points into a Route object."""
        all_pts = [start] + list(points)
        wps = [Waypoint(p[0], p[1], i) for i, p in enumerate(all_pts)]
        length = self.path_length(start, points)
        est = -(-length // speed) if speed > 0 else 0
        return Route(waypoints=wps, speed=speed, length=length,
                     estimated_time=est)


def first_axis_step(cur: Point, target: Point) -> Optional[Point]:
    """Helper: direction vector toward target (x first, then y)."""
    dx, dy = target[0] - cur[0], target[1] - cur[1]
    if dx:
        return (1 if dx > 0 else -1, 0)
    if dy:
        return (0, 1 if dy > 0 else -1)
    return None

import globalvar

def to_grid_point(pos):
    """Convert a raw (x, y) position into the centre of its grid cell."""
    if pos is None or len(pos) != 2:
        raise ValueError("position must contain exactly two coordinates")
    return (_GRID.to_cell_center(pos[0]), _GRID.to_cell_center(pos[1]))


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique order id.
        res_pos / cus_pos: grid-converted restaurant / customer positions.
        status: "pending", "assigned", "picking_up", "delivering",
            "completed".
    """

    def __init__(self, res_pos, cus_pos):
        new_id = (globalvar.get_value("order_id_counter", 0) or 0) + 1
        globalvar.set_value("order_id_counter", new_id)
        self.id = new_id
        self.raw = [int(res_pos[0]), int(res_pos[1]),
                    int(cus_pos[0]), int(cus_pos[1])]
        self.res_pos = to_grid_point(res_pos)
        self.cus_pos = to_grid_point(cus_pos)

        self.assigned_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.rider_id = None

    @property

        return self.cus_pos[0]

    @property
    def customer_y(self):
        return self.cus_pos[1]

        """Return [restaurant_x, restaurant_y, customer_x, customer_y]."""
        return [self.res_pos[0], self.res_pos[1],

        return self.status == "completed"

            "time_taken": self.time_taken, "rider_id": self.rider_id,

        obj.raw = list(data.get("raw", []))

"""Rider model: movement, state machine and status log."""

import logging
from enum import IntEnum

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

DEFAULT_SPEED = 20
START_POSITION = (400, 400)


class RiderState(IntEnum):
    """Rider status enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2


STATE_NAMES = {
    RiderState.IDLE: "idle",
    RiderState.PICKING_UP: "picking up",
    RiderState.DELIVERING: "delivering",
}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``. Each call to
    :meth:`move` advances the rider by one tick.
    """

    def __init__(self, rider_id, x=START_POSITION[0], y=START_POSITION[1],
                 speed=None):
        self.rider_id = rider_id
        self.x = int(x)
        self.y = int(y)
        if speed is None:
            speed = globalvar.get_value("rider_speed", DEFAULT_SPEED) or DEFAULT_SPEED
        self.speed = int(speed)
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.path = []
        self.log = []
        self.distance_travelled = 0
        self._planner = PathPlanner()
        self._record("spawn", None)

    # ---------------------------------------------------------------- helpers
    @property
    def id(self):
        return self.rider_id

    @property
    def position(self):
        return (self.x, self.y)

    @property
    def state_name(self):
        return STATE_NAMES.get(RiderState(self.state), str(self.state))

    @property
    def active_orders(self):
        return [o for o in self.orders if o.status != "completed"]

    @property
    def active_order_count(self):
        return len(self.active_orders)

    def is_idle(self):
        return self.state == RiderState.IDLE and self.active_order_count == 0

    def _tick(self):
        return globalvar.get_value("tick", 0) or 0

    def _record(self, event, order_id):
        self.log.append({
            "time": self._tick(), "event": event, "order_id": order_id,
            "x": self.x, "y": self.y, "state": int(self.state),
        })
        logger.debug("Rider %s: %s (order %s) at (%s,%s)",
                     self.rider_id, event, order_id, self.x, self.y)

    def _next_pending_order(self):
        for o in self.orders:
            if o.status != "completed" and o is not self.current_order:
                return o
        return None

    def _plan_to(self, target):
        self.special_case = PathPlanner.is_special_case(
            self.x, self.y, target[0], target[1])
        self.path = self._planner.plan(self.x, self.y, target[0], target[1])

    # ------------------------------------------------------------ transitions
    def _accept(self, order):
        if order.rider_id != self.rider_id:
            order.assign_to(self.rider_id)
        self.current_order = order
        self.state = RiderState.PICKING_UP
        order.status = "picking_up"
        self._plan_to(order.res_pos)
        self._record("accepted_order", order.id)

    def _arrive(self):
        order = self.current_order
        if order is None:
            return
        if self.state == RiderState.PICKING_UP:
            self.state = RiderState.DELIVERING
            order.status = "delivering"
            order.pickup_time = self._tick()
            self._record("arrived_at_merchant", order.id)
            self._plan_to(order.cus_pos)
        elif self.state == RiderState.DELIVERING:
            self._complete()
        else:
            # Inconsistent state: restart the pickup leg.
            self.state = RiderState.PICKING_UP
            self._plan_to(order.res_pos)

    def _complete(self):
        order = self.current_order
        order.mark_completed(self._tick())
        self._record("delivered_to_customer", order.id)
        self.current_order = None
        self.path = []
        nxt = self._next_pending_order()
        if nxt is not None:
            self._accept(nxt)
        else:
            self.state = RiderState.IDLE
            self.special_case = False

    # --------------------------------------------------------------- movement
    def move(self):
        """Advance the rider by one tick."""
        if self.current_order is not None and self.current_order.status == "completed":
            self.current_order = None
            self.path = []
        if self.current_order is None:
            nxt = self._next_pending_order()
            if nxt is None:
                self.state = RiderState.IDLE
                return
            self._accept(nxt)
        budget = self.speed
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            dx, dy = tx - self.x, ty - self.y
            if dx:
                step = min(abs(dx), budget)
                self.x += step if dx > 0 else -step
            elif dy:
                step = min(abs(dy), budget)
                self.y += step if dy > 0 else -step
            else:
                step = 0
            budget -= step
            self.distance_travelled += step
            if (self.x, self.y) == (tx, ty):
                self.path.pop(0)
        if not self.path:
            self._arrive()

    def remaining_route(self):
        """Return (points, length) of the full remaining planned route."""
        pos = (self.x, self.y)
        points = list(self.path)
        end = points[-1] if points else pos
        targets = []
        if self.current_order is not None and self.state == RiderState.PICKING_UP:
            targets.append(self.current_order.cus_pos)
        for o in self.orders:
            if o.status != "completed" and o is not self.current_order:
                targets.extend([o.res_pos, o.cus_pos])
        more, _ = self._planner.plan_multi(end, targets)
        points.extend(more)
        return points, self._planner.path_length(pos, points)

    # ------------------------------------------------------------ persistence
    def to_dict(self):
        return {
            "rider_id": self.rider_id, "x": self.x, "y": self.y,
            "speed": self.speed, "state": int(self.state),
            "special_case": self.special_case,
            "orders": [o.id for o in self.orders],
            "current_order": self.current_order.id if self.current_order else None,
            "path": [list(p) for p in self.path],
            "log": self.log, "distance_travelled": self.distance_travelled,
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        obj = cls.__new__(cls)
        obj.rider_id = data["rider_id"]
        obj.x, obj.y = int(data["x"]), int(data["y"])
        obj.speed = int(data.get("speed", DEFAULT_SPEED))
        obj.state = RiderState(int(data.get("state", 0)))
        obj.special_case = bool(data.get("special_case", False))
        obj.orders = [orders_by_id[i] for i in data.get("orders", [])
                      if i in orders_by_id]
        cur = data.get("current_order")
        obj.current_order = orders_by_id.get(cur) if cur is not None else None
        obj.path = [tuple(p) for p in data.get("path", [])]
        obj.log = list(data.get("log", []))
        obj.distance_travelled = data.get("distance_travelled", 0)
        obj._planner = PathPlanner()
        return obj

    def __repr__(self):
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state_name}, orders={len(self.orders)})")


def recruit_rider(cost=None):
    """Create a new rider, deduct the recruitment cost and register it."""
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    counter = globalvar.get_value("rider_id", 0) or 0
    existing = max([r.rider_id for r in riders], default=0)
    new_id = max(counter, existing) + 1
    globalvar.set_value("rider_id", new_id)
    if cost is None:
        cost = globalvar.get_value("recruitment_cost", 500)
        if cost is None:
            cost = 500
    globalvar.set_value("funds", (globalvar.get_value("funds", 0) or 0) - cost)
    rider = Rider(new_id)
    riders.append(rider)
    logger.info("Recruited rider %s (cost %s)", new_id, cost)
    return rider

"""Order management: creation, assignment and queries."""

import logging
import math

import globalvar
from order_model import Order
from rider import recruit_rider

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Return the best rider for ``order``.

    Strategy: the rider with the fewest current (uncompleted) orders wins;
    ties are broken by straight-line distance to the restaurant
    (sqrt((rx - x)^2 + (ry - y)^2)) and then by rider id. If no riders
    exist, one is recruited. The order is not appended to the rider.
    """
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    if not riders:
        recruit_rider()
        riders = globalvar.get_value("riders")
    rx, ry = order.res_pos

    def key(r):
        return (r.active_order_count, math.hypot(r.x - rx, r.y - ry), r.rider_id)

    return min(riders, key=key)


class OrderManager:
    """Handles order creation, assignment and completion queries."""

    def _orders(self):
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    @property
    def orders(self):
        return self._orders()

    def create_order(self, coords):
        """Create an order from [rx, ry, cx, cy] and register it."""
        if len(coords) != 4:
            raise ValueError("exactly four coordinates are required")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        self._orders().append(order)
        return order

    def assign_order(self, order):
        """Assign ``order`` to the best rider and return that rider."""
        rider = find_best_rider_for_order(order)
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider

    def pending_orders(self):
        return [o for o in self._orders() if o.status != "completed"]

    def completed_orders(self):
        return [o for o in self._orders() if o.status == "completed"]

    def get(self, order_id):
        for o in self._orders():
            if o.id == order_id:
                return o
        return None

"""Status display and ASCII map visualisation."""

from path_planner import MAP_SIZE
from rider import STATE_NAMES, RiderState


class DisplayManager:
    """Renders status tables and a 16x16 ASCII map (50 units per char)."""

    def __init__(self, map_display_size=16, map_size=MAP_SIZE):
        self.map_display_size = map_display_size
        self.map_size = map_size
        self.unit = map_size // map_display_size

    def _idx(self, v):
        return max(0, min(self.map_display_size - 1, int(v) // self.unit))

    def render_map(self, riders, orders, route=None):
        """Render the map.

        Symbols: R rider, S restaurant, C customer, * route,
        # street, . empty space.
        """
        n = self.map_display_size
        grid = [['#' if (r % 2 == 0 or c % 2 == 0) else '.' for c in range(n)]
                for r in range(n)]
        if route and len(route) > 1:
            for a, b in zip(route, route[1:]):
                length = abs(b[0] - a[0]) + abs(b[1] - a[1])
                steps = max(1, length // (self.unit // 2))
                for i in range(steps + 1):
                    x = a[0] + (b[0] - a[0]) * i // steps
                    y = a[1] + (b[1] - a[1]) * i // steps
                    grid[self._idx(y)][self._idx(x)] = '*'
        for o in orders:
            grid[self._idx(o.cus_pos[1])][self._idx(o.cus_pos[0])] = 'C'
        for o in orders:
            grid[self._idx(o.res_pos[1])][self._idx(o.res_pos[0])] = 'S'
        for r in riders:

"""Command parsing, input validation and command history."""

import re

try:  # Optional line editing / history shortcuts (not on all platforms).
    import readline  # noqa: F401
except Exception:  # pragma: no cover
    readline = None

ALIASES = {
    "s": "start", "n": "start", "new": "start",
    "st": "status", "o": "orders", "r": "riders",
    "q": "quit", "exit": "quit", "h": "help", "?": "help",
    "t": "tick", "m": "map", "hist": "history",
}

COMMANDS = {"start", "status", "orders", "riders", "quit", "help", "tick",
            "run", "map", "history", "save", "load"}


class InputManager:
    """Parses commands and validates coordinate input."""

    def __init__(self, min_coord=0, max_coord=1000):
        self.min_coord = min_coord
        self.max_coord = max_coord
        self.command_history = []

    def read(self, prompt="> "):
        """Read a line from the user, supporting '!!' to repeat."""
        line = input(prompt).strip()
        if line == "!!":
            if not self.command_history:
                raise ValueError("no command history")
            line = self.command_history[-1]
            print(line)
        if line:
            self.command_history.append(line)
        return line

    @staticmethod
    def parse_command(line):
        """Split a line into (command, args) and resolve aliases."""
        parts = line.strip().split(None, 1)
        if not parts:
            return "", ""
        cmd = parts[0].lower()
        cmd = ALIASES.get(cmd, cmd)
        args = parts[1] if len(parts) > 1 else ""
        return cmd, args

    def parse_coordinates(self, text):
        """Parse four integer coordinates; raise ValueError on bad input."""
        if isinstance(text, (list, tuple)):
            tokens = [str(t) for t in text]
        else:
            tokens = [t for t in re.split(r"[\s,\[\]()]+", str(text)) if t]
        if len(tokens) != 4:
            raise ValueError("please enter exactly four integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for t in tokens:
            try:
                v = int(t)
            except ValueError:
                raise ValueError(f"'{t}' is not an integer") from None
            if not self.min_coord <= v <= self.max_coord:
                raise ValueError(f"coordinate {v} out of range "
                                 f"[{self.min_coord}, {self.max_coord}]")
            values.append(v)
        return values

    @staticmethod
    def parse_positive_int(text, default=1):
        text = (text or "").strip()
        if not text:
            return default
        v = int(text)
        if v <= 0:
            raise ValueError("value must be positive")
        return v

"""System manager: initialisation, recruitment, ticking and command loop."""

import json
import logging
import os

import globalvar
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from order_model import Order
from path_planner import GridMap, PathPlanner
from rider import Rider, RiderState, recruit_rider

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "funds": 1000,
    "recruitment_cost": 500,
    "recruitment_threshold": 300,
    "rider_speed": 20,
    "state_file": "dispatch_state.json",
}

HELP_TEXT = """Commands:
  start [rx ry cx cy]  (s/n)  create an order (restaurant + customer coords)
  status               (st)   system funds and rider status
  orders               (o)    all order details
  riders               (r)    rider status-change history
  map                  (m)    show the ASCII map
  tick [n]             (t)    advance n ticks (default 1)
  run                         advance until all orders are delivered
  history              (hist) show command history ('!!' repeats last)
  save [file] / load [file]   persist / restore state as JSON
  quit / exit          (q)    exit the program
Each command entered advances the simulation by one tick."""


def init_system(config=None):
    """Initialise the global system parameters."""
    cfg = dict(DEFAULT_CONFIG)
    if config:
        cfg.update(config)
    globalvar._init()
    globalvar.set_value("funds", int(cfg["funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)
    globalvar.set_value("recruitment_cost", int(cfg["recruitment_cost"]))
    globalvar.set_value("recruitment_threshold", int(cfg["recruitment_threshold"]))
    globalvar.set_value("rider_speed", int(cfg["rider_speed"]))
    globalvar.set_value("state_file", cfg["state_file"])


def load_config(path):
    """Load a JSON configuration file; returns {} if missing/invalid."""
    if not path or not os.path.exists(path):
        return {}
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        return {k: v for k, v in data.items() if k in DEFAULT_CONFIG}
    except (OSError, ValueError) as exc:
        logger.warning("Could not read config %s: %s", path, exc)
        return {}


class SystemManager:
    """Core manager for funds, riders, orders and the main loop."""

    def __init__(self, config=None, config_path=None):
        cfg = load_config(config_path) if config_path else {}
        if config:
            cfg.update(config)
        init_system(cfg)
        self.grid_map = GridMap()
        self.path_planner = PathPlanner()
        self.order_manager = OrderManager()
        self.display_manager = DisplayManager()
        self.input_manager = InputManager()

    # ------------------------------------------------------------- properties
    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        return globalvar.get_value("riders", [])

    @property
    def running(self):
        return globalvar.get_value("running", False)

    @running.setter
    def running(self, value):
        globalvar.set_value("running", bool(value))

    @property
    def current_tick(self):
        return globalvar.get_value("tick", 0)

    @property
    def recruitment_cost(self):
        return globalvar.get_value("recruitment_cost", 500)

    @property
    def recruitment_threshold(self):
        return globalvar.get_value("recruitment_threshold", 300)

    # ------------------------------------------------------------ operations
    def idle_riders(self):
        return [r for r in self.riders if r.is_idle()]

    def maybe_recruit(self):
        """Recruit a rider if funds allow and orders outnumber idle riders."""
        pending = len(self.order_manager.pending_orders())
        if self.funds >= self.recruitment_threshold and pending > len(self.idle_riders()):
            return recruit_rider(self.recruitment_cost)
        return None

    def tick(self, n=1):
        """Advance the simulation by ``n`` ticks."""
        for _ in range(n):
            globalvar.set_value("tick", self.current_tick + 1)
            for r in self.riders:
                try:
                    r.move()
                except Exception:  # error recovery for a single rider
                    logger.exception("Rider %s failed to move", r.rider_id)
                    self._recover_rider(r)
            self.check_consistency()

    def _recover_rider(self, rider):
        rider.path = []
        if rider.current_order is not None and rider.current_order.status != "completed":
            rider.state = RiderState.PICKING_UP
            rider.path = rider._planner.plan(rider.x, rider.y, *rider.current_order.res_pos)
        else:
            rider.current_order = None
            rider.state = RiderState.IDLE

    def check_consistency(self):
        """Fix obvious inconsistencies between riders and orders."""
        for r in self.riders:
            if r.current_order is None and r.active_order_count == 0 \
                    and r.state != RiderState.IDLE:
                r.state = RiderState.IDLE
                r.path = []
            for o in r.orders:
                if o.rider_id is None:
                    o.rider_id = r.rider_id

    def run_until_done(self, limit=100000):
        count = 0
        while self.order_manager.pending_orders() and count < limit:
            self.tick()
            count += 1
        return count

    # ----------------------------------------------------------- persistence
    def save_state(self, path=None):
        path = path or globalvar.get_value("state_file", DEFAULT_CONFIG["state_file"])
        data = {
            "config": {
                "funds": self.funds,
                "recruitment_cost": self.recruitment_cost,
                "recruitment_threshold": self.recruitment_threshold,
                "rider_speed": globalvar.get_value("rider_speed", 20),
            },
            "tick": self.current_tick,
            "rider_id": globalvar.get_value("rider_id", 0),
            "order_id_counter": globalvar.get_value("order_id_counter", 0),
            "orders": [o.to_dict() for o in self.order_manager.orders],
            "riders": [r.to_dict() for r in self.riders],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

    def load_state(self, path=None):
        path = path or globalvar.get_value("state_file", DEFAULT_CONFIG["state_file"])
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        cfg = dict(data.get("config", {}))
        cfg["state_file"] = path
        init_system(cfg)
        globalvar.set_value("tick", data.get("tick", 0))
        globalvar.set_value("rider_id", data.get("rider_id", 0))
        globalvar.set_value("order_id_counter", data.get("order_id_counter", 0))
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        globalvar.set_value("orders", orders)
        by_id = {o.id: o for o in orders}
        riders = [Rider.from_dict(d, by_id) for d in data.get("riders", [])]
        globalvar.set_value("riders", riders)
        return path

    # -------------------------------------------------------------- commands
    def handle_start(self, coords):
        values = self.input_manager.parse_coordinates(coords)
        order = self.order_manager.create_order(values)
        recruited = self.maybe_recruit()
        rider = self.order_manager.assign_order(order)
        points, length = rider.remaining_route()
        start = (rider.x, rider.y)
        route = self.path_planner.build_route(start, points, rider.speed)
        lines = [
            f"Order #{order.id} created at tick {order.start_time}",
            f"Restaurant grid position: {order.res_pos} "
            f"(cell {self.grid_map.cell_of(*order.res_pos)})",
            f"Customer grid position:   {order.cus_pos} "
            f"(cell {self.grid_map.cell_of(*order.cus_pos)})",
        ]
        if recruited is not None:
            lines.append(f"New rider added: Yes (Rider #{recruited.rider_id}, "
                         f"cost {self.recruitment_cost}, funds now {self.funds})")
        else:
            lines.append(f"New rider added: No (funds {self.funds})")
        lines.append(f"Assigned to Rider #{rider.rider_id} at ({rider.x},{rider.y}), "
                     f"active orders: {rider.active_order_count}")
        lines.append(f"Rider speed: {rider.speed} units/tick, route length: "
                     f"{route.length} units, estimated time: {route.estimated_time} ticks")
        lines.append("Route: " + " -> ".join(f"({p[0]},{p[1]})" for p in route.points()))
        shown = [o for o in rider.orders if o.status != "completed"]
        lines.append(self.display_manager.render_map([rider], shown, route.points()))
        return "\n".join(lines)

    def handle_command(self, line):
        """Execute a command line. Returns (output, advance_tick)."""
        cmd, args = self.input_manager.parse_command(line)
        if not cmd:
            return "", False
        if cmd == "start":
            if not args.strip():
                args = self.input_manager.read(
                    "Enter restaurant_x restaurant_y customer_x customer_y: ")
            return self.handle_start(args), True
        if cmd == "status":
            return self.display_manager.format_status(
                self.funds, self.current_tick, self.riders), True
        if cmd == "orders":
            return self.display_manager.format_orders(self.order_manager.orders), True
        if cmd == "riders":
            return self.display_manager.format_rider_logs(self.riders), True
        if cmd == "map":
            return self.display_manager.render_map(
                self.riders, self.order_manager.pending_orders()), True
        if cmd == "tick":
            n = self.input_manager.parse_positive_int(args, 1)
            self.tick(n)
            return f"Advanced {n} tick(s). Now at tick {self.current_tick}.", False
        if cmd == "run":
            n = self.run_until_done()
            return f"Advanced {n} tick(s); all orders handled. Tick {self.current_tick}.", False
        if cmd == "history":
            return "\n".join(f"{i + 1}: {c}" for i, c in
                             enumerate(self.input_manager.command_history)), False
        if cmd == "save":
            return f"State saved to {self.save_state(args.strip() or None)}", False
        if cmd == "load":
            return f"State loaded from {self.load_state(args.strip() or None)}", False
        if cmd == "help":
            return HELP_TEXT, False
        if cmd == "quit":
            self.running = False
            return "Goodbye.", False
        return f"Unknown command '{cmd}'. Type 'help' for a list of commands.", False

    def run(self):
        """Main loop: each iteration is one simulated tick."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.running:
            try:
                line = self.input_manager.read(f"[t={self.current_tick}] > ")
                output, advance = self.handle_command(line)
                if output:
                    print(output)
                if advance and self.running:
                    self.tick()
            except (EOFError, KeyboardInterrupt):
                print("\nExiting.")
                self.running = False
            except ValueError as exc:
                print(f"Input error: {exc}")
            except Exception as exc:  # graceful recovery
                logger.exception("Unexpected error")
                print(f"Error: {exc}. System state recovered.")
                self.check_consistency()

"""Entry point for the food delivery dispatch simulator."""

import argparse
import logging
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from system import SystemManager  # noqa: E402


def main(argv=None):
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", default="config.json", help="JSON config file")
    parser.add_argument("--load", default=None, help="load saved state JSON")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if args.debug else logging.WARNING,
                        format="%(asctime)s %(levelname)s %(name)s: %(message)s")
    manager = SystemManager(config_path=args.config)
    if args.load:
        try:
            manager.load_state(args.load)
        except (OSError, ValueError) as exc:
            print(f"Could not load state: {exc}")
    manager.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

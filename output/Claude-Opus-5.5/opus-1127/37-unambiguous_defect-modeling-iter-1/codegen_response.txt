from __future__ import annotations

import re
from typing import List, Optional


def normalize(text: Optional[str]) -> str:
    """Normalize a name / group string for comparisons."""
    if text is None:
        return ""
    return " ".join(str(text).strip().split()).casefold()


class User:
    """A participant of the lottery (name + department group)."""

    def __init__(self, name: str, department: str):
        self.name = str(name).strip()
        self.department = str(department).strip()
        self.is_valid = True

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self):
        return f"{self.name} ({self.department})"


_LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("1st", 1),
    ("second", 2),
    ("2nd", 2),
    ("third", 3),
    ("3rd", 3),
    ("fourth", 4),
    ("4th", 4),
    ("fifth", 5),
    ("5th", 5),
]

UNKNOWN_LEVEL_RANK = 100


def prize_level_rank(name: str) -> int:
    """Return the prize level rank derived from its name (lower = higher level)."""
    low = normalize(name)
    for keyword, rank in _LEVEL_KEYWORDS:
        if re.search(r"\b" + re.escape(keyword) + r"\b", low):
            return rank
    return UNKNOWN_LEVEL_RANK


class Prize:
    """A prize to be drawn in one lottery round."""

    def __init__(self, name: str, quantity: int, avoid_groups: Optional[List[str]] = None,
                 associated_drawer: Optional[str] = None):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups = list(avoid_groups) if avoid_groups else []
        if associated_drawer is not None:
            associated_drawer = str(associated_drawer).strip() or None
        self.associated_drawer = associated_drawer

    @property
    def level(self) -> int:
        return prize_level_rank(self.name)

    def avoids(self, department: str) -> bool:
        dep = normalize(department)
        return any(normalize(g) == dep for g in self.avoid_groups)

    def rule_description(self) -> str:
        parts = []
        if self.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("No avoidance groups")
        parts.append("Drawer: " + (self.associated_drawer or "N/A"))
        parts.append("Previous winners excluded")
        return "; ".join(parts)

    def __repr__(self):
        return (f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
                f"avoid_groups={self.avoid_groups!r}, associated_drawer={self.associated_drawer!r})")

_SEPARATORS = [",", "，", "\t", ";", "|"]


class ListManager:
    """Loads, validates and analyses participant lists."""

    def __init__(self):
        self.default_path = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
                                         "data", "default_list.txt")

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        if os.path.isfile(self.default_path):
            users, errors = self._load_from_path(self.default_path)
            if users:
                return users, errors
        return [User(n, d) for n, d in DEFAULT_LIST], []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File Error: file not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"File Error: cannot read {path}: {exc}"]
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            sep = next((s for s in _SEPARATORS if s in line), None)
            if sep is None:
                errors.append(f"Format Error: Line {idx}: '{line}' (expected 'name,department')")
                continue
            parts = [p.strip() for p in line.split(sep)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {idx}: '{line}' (expected 'name,department')")
                continue
            if idx == 1 and parts[0].lower() == "name" and parts[1].lower() in ("department", "group"):
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            key = u.name.strip()
            if key in seen:
                duplicates.append(u)
            else:
                seen.add(key)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and the lottery order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer: Optional[str] = None) -> Prize:
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be positive")
        if any(p.name == str(name).strip() for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = [str(g).strip() for g in (avoid_groups or []) if str(g).strip()]
        prize = Prize(str(name).strip(), qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        n = len(self.prizes)
        if n == 0 or len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""

from __future__ import annotations

import random
from typing import Iterable, List, Optional

from lottery_system.models import Prize, User, normalize

SPECIAL_DRAWER = "Li Bin"
FALLBACK_MONITORED = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def is_cheng_ru(department: str) -> bool:
    d = normalize(department).replace("_", " ").replace("-", " ")
    d = d.replace("group", "").replace("team", "")
    d = "".join(d.split())
    return d == "chengru"


class LotteryManager:
    """Selects winners according to the configured rules."""

    def __init__(self, users, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users)
        self.winners = set()
        self.winner_order: List[User] = []
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()

    # ---------- helpers ----------
    def _has_won(self, user: User) -> bool:
        if user in self.winners:
            return True
        names = {w.name if isinstance(w, User) else str(w) for w in self.winners}
        return user.name in names

    @staticmethod
    def _drawer_is(prize: Prize, person: str) -> bool:
        return normalize(prize.associated_drawer) == normalize(person)

    def is_fallback_active(self, prize: Prize) -> bool:
        return (self._drawer_is(prize, FALLBACK_TARGET)
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def get_rule_description(self, prize: Prize) -> str:
        if self.is_fallback_active(prize):
            return ("Fallback rule active: pool limited to Cheng Ru group members "
                    "(avoidance overridden); previous winners excluded")
        if self._drawer_is(prize, SPECIAL_DRAWER):
            return "Special drawer Li Bin: relaxed filtering (avoidance ignored); previous winners excluded"
        return prize.rule_description()

    # ---------- core ----------
    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = [u for u in self.users if not self._has_won(u)]
        if self.is_fallback_active(prize):
            protected = [u for u in candidates if is_cheng_ru(u.department)]
            if protected:
                return protected
        if self._drawer_is(prize, SPECIAL_DRAWER):
            return candidates
        return [u for u in candidates if not prize.avoids(u.department)]

    def mark_winner(self, user: User) -> None:
        self.winners.add(user)
        self.winner_order.append(user)

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.mark_winner(user)
        return user

    def finish_round(self, prize: Prize, round_winners: Iterable[User]) -> None:
        """Update fallback tracking after a round completes (or is terminated)."""
        winners = list(round_winners)
        if any(self._drawer_is(prize, p) for p in FALLBACK_MONITORED):
            self.last_two_draw_results.append(any(is_cheng_ru(u.department) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self._drawer_is(prize, FALLBACK_TARGET):
            self.last_two_draw_results = []

"""Result management and export."""

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import prize_level_rank


class Winner:
    def __init__(self, user, prize, order: int, win_time: datetime = None):
        self.user = user
        self.prize = prize
        self.prize_name = getattr(prize, "name", str(prize))
        self.order = order
        self.win_time = win_time or datetime.now()

    def __repr__(self):
        return f"Winner(user={self.user!r}, prize={self.prize_name!r}, order={self.order})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self.winners.append(Winner(user, prize, len(self.winners) + 1))
        return True

    def _sorted_prizes(self, prizes):
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda t: (prize_level_rank(getattr(t[1], "name", str(t[1]))), t[0]))
        return [p for _, p in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in self._sorted_prizes(prizes):
            name = getattr(p, "name", str(p))
            ws = [w for w in self.winners if w.prize_name == name]
            ws.sort(key=lambda w: (w.win_time, w.order))
            result[name] = ws
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Prize Name", "Name", "Department", "Win Order", "Win Time"])
                grouped = self.get_winners_by_prize(prizes)
                listed = set()
                for pname, ws in grouped.items():
                    for w in ws:
                        listed.add(id(w))
                        writer.writerow([pname, w.user.name, w.user.department, w.order,
                                         w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
                for w in self.winners:
                    if id(w) not in listed:
                        writer.writerow([w.prize_name, w.user.name, w.user.department, w.order,
                                         w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command line interface of the lottery system."""

from __future__ import annotations

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _input(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        return "q"


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []

    # ---------- list ----------
    def load_list(self):
        mode = _input("1) Default list  2) Custom txt list (name,department per line): ").strip()
        if mode == "2":
            path = _input("Path to txt file: ").strip()
            users, errors = self.list_manager.load_custom_list(path)
        else:
            users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            choice = _input("Remove invalid data and continue (r) or terminate upload (t)? ").strip().lower()
            if choice != "r":
                print("Upload terminated.")
                return
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        print(f"Loaded {len(valid)} valid participants.")

    def show_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
        for g, c in info.items():
            print(f"  {g}: {c}")

    # ---------- prizes ----------
    def add_prize(self):
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        groups = _input("Avoidance groups (comma separated, blank for none): ").strip()
        drawer = _input("Associated lottery person (blank for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, [g for g in groups.split(",") if g.strip()], drawer)
            print("Prize added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def reorder(self):
        self.show_prizes()
        raw = _input("New order (e.g. 3 1 2): ").replace(",", " ").split()
        print("Reordered." if self.prize_manager.reorder_prizes(raw) else "Invalid order.")

    # ---------- lottery ----------
    def run_lottery(self):
        prizes = self.prize_manager.get_prizes()
        if not self.users or not prizes:
            print("Please load a list and configure prizes first.")
            return
        manager = LotteryManager(self.users)
        for w in self.result_manager.winners:
            manager.mark_winner(w.user)
        for rnd, prize in enumerate(prizes, 1):
            print(f"\n=== Round {rnd}: {prize.name} | Quantity: {prize.quantity} ===")
            print(f"Rule: {manager.get_rule_description(prize)}")
            round_winners = []
            i = 0
            while i < prize.quantity:
                cmd = _input(f"[Enter]=draw #{i + 1}, 'pause', 'terminate': ").strip().lower()
                if cmd == "q":
                    cmd = "terminate"
                if cmd == "pause":
                    print("Lottery paused. Type 'continue' to resume.")
                    while True:
                        c = _input("> ").strip().lower()
                        if c == "continue":
                            print("Resumed.")
                            break
                        if c in ("terminate", "q"):
                            cmd = "terminate"
                            break
                    if cmd != "terminate":
                        continue
                if cmd == "terminate":
                    print("Round terminated.")
                    break
                print(f"Currently drawing the {i + 1}th winner...")
                user = manager.draw_one(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
                i += 1
            manager.finish_round(prize, round_winners)
            print(f"--- Round {rnd} results ---")
            for n, u in enumerate(round_winners, 1):
                print(f"  {n}. {u.name} ({u.department})")
        self.show_summary()

    def show_summary(self):
        print("\n=== Complete Winner List ===")
        for pname, ws in self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes()).items():
            print(f"{pname}:")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {pname}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": self.load_list, "2": self.show_groups,
            "3": lambda: print("\n".join(self.prize_manager.load_default_prizes()) or "Preset prizes loaded."),
            "4": self.add_prize, "5": self.show_prizes, "6": self.reorder,
            "7": self.run_lottery, "8": self.show_summary, "9": self.export,
        }
        while True:
            print("\n1) Load list 2) View groups 3) Load preset prizes 4) Add prize 5) View prizes "
                  "6) Reorder prizes 7) Start lottery 8) Show results 9) Export 0) Exit")
            c = _input("Choice: ").strip()
            if c in ("0", "q"):
                break
            action = actions.get(c)
            if action:
                action()
            else:
                print("Invalid choice.")


def main():
    LotteryApp().run()


if __name__ == "__main__":
    main()

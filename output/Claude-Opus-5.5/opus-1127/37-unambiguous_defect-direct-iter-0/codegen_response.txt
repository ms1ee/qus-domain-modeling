from dataclasses import dataclass, field
from typing import List, Optional


CHENG_RU_GROUP = "Cheng Ru group"
SPECIAL_DRAWER_RELAXED = "Li Bin"
FALLBACK_WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"


def normalize(text: Optional[str]) -> str:
    """Normalize a string for case/space-insensitive comparisons."""
    if text is None:
        return ""
    return " ".join(str(text).split()).lower()


def is_cheng_ru_group(department: Optional[str]) -> bool:
    """Return True if the department is the Cheng Ru group."""
    norm = normalize(department)
    return norm == normalize(CHENG_RU_GROUP) or norm == "cheng ru"


@dataclass
class User:
    name: str
    department: str

    def __post_init__(self):
        self.name = " ".join(str(self.name).split())
        self.department = " ".join(str(self.department).split())

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if self.associated_drawer is not None:
            drawer = " ".join(str(self.associated_drawer).split())
            self.associated_drawer = drawer or None

    def rule_description(self) -> str:
        parts = []
        if self.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("Avoidance groups: none")
        if self.associated_drawer:
            parts.append(f"Lottery person: {self.associated_drawer}")
            if normalize(self.associated_drawer) == normalize(SPECIAL_DRAWER_RELAXED):
                parts.append("Relaxed filtering rules apply (avoidance groups ignored)")
            if normalize(self.associated_drawer) == normalize(FALLBACK_TARGET_DRAWER):
                parts.append(
                    "Cheng Ru group protection may apply if previous two draws "
                    "(Chen Qixian, Jin Duo) had no Cheng Ru group winners"
                )
        else:
            parts.append("Lottery person: not specified")
        parts.append("Previous winners are excluded")
        return "; ".join(parts)


_LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("1st", 1),
    ("second", 2),
    ("2nd", 2),
    ("third", 3),
    ("3rd", 3),
    ("fourth", 4),
    ("4th", 4),
    ("fifth", 5),
    ("5th", 5),
]


def prize_level(prize_name: str) -> int:
    """Return a numeric level for a prize (lower is higher-level)."""
    norm = normalize(prize_name)
    best = None
    for keyword, level in _LEVEL_KEYWORDS:
        if norm.startswith(keyword):
            return level
        if keyword in norm.split() or f"{keyword} prize" in norm:
            if best is None or level < best:
                best = level
    return best if best is not None else 100

    "Luo Mei,Operations Department",

class ListManager:
    """Handles participant list loading and validation.

    Line format: ``Name,Department`` (one participant per line).
    Blank lines and lines starting with ``#`` are ignored.
    """

    def __init__(self):
        self.users: List[User] = []

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff")
            if not line or line.startswith("#"):
                continue
            # Accept comma (ASCII or full-width) or tab as separator
            normalized = line.replace("\uff0c", ",")
            if "," not in normalized and "\t" in normalized:
                normalized = normalized.replace("\t", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department', got '{line}'"
                )
                continue
            name, dept = parts
            if not name or not dept:
                errors.append(
                    f"Line {idx}: Format Error - name and department must not be empty: '{line}'"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File not found: {path}"]
        lines = None
        for encoding in ("utf-8-sig", "gbk", "latin-1"):
            try:
                with open(path, "r", encoding=encoding) as f:
                    lines = f.read().splitlines()
                break
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"Failed to read file {path}: {exc}"]
        if lines is None:
            return [], [f"Failed to decode file: {path}"]
        return self._parse_lines(lines)

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.lower()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize


DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    def __init__(self):
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = " ".join(str(name if name is not None else "").split())
        if not name:
            raise ValueError("Prize name must not be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name.lower() == name.lower() for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups: List[str] = []
        for g in avoid_groups or []:
            g = " ".join(str(g).split())
            if g and g not in groups:
                groups.append(g)
        drawer: Optional[str] = associated_drawer if associated_drawer else None
        prize = Prize(name, qty, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        try:
            positions = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        n = len(self._prizes)
        if len(positions) != n or sorted(positions) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in positions]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import (
    FALLBACK_TARGET_DRAWER,
    FALLBACK_WATCHED_DRAWERS,
    SPECIAL_DRAWER_RELAXED,
    Prize,
    User,
    is_cheng_ru_group,
    normalize,
)


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.winners: List[User] = []
        # Results (contains Cheng Ru group member?) of the most recent
        # draws by the watched drawers (Chen Qixian, Jin Duo); max 2 items.
        self.last_two_draw_results: List[bool] = []
        self.rng = random.Random()

    # ---------- rule helpers ----------
    def _is_winner(self, user: User) -> bool:
        return user.name.lower() in self.winner_names

    def _fallback_active(self, prize: Prize) -> bool:
        if normalize(prize.associated_drawer) != normalize(FALLBACK_TARGET_DRAWER):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        available = [u for u in self.users if not self._is_winner(u)]
        if self._fallback_active(prize):
            pool = [u for u in available if is_cheng_ru_group(u.department)]
            if pool:
                return pool
        if normalize(prize.associated_drawer) == normalize(SPECIAL_DRAWER_RELAXED):
            return available
        avoid = {normalize(g) for g in (prize.avoid_groups or [])}
        return [u for u in available if normalize(u.department) not in avoid]

    def is_fallback_active(self, prize: Prize) -> bool:
        return self._fallback_active(prize)

    # ---------- drawing ----------
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winner_names.add(user.name.lower())
        self.winners.append(user)
        return user

    def record_round(self, prize: Prize, round_winners: List[User]) -> None:
        drawer = normalize(prize.associated_drawer)
        watched = {normalize(d) for d in FALLBACK_WATCHED_DRAWERS}
        if drawer in watched:
            contains = any(is_cheng_ru_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == normalize(FALLBACK_TARGET_DRAWER):
            self.last_two_draw_results = []

"""Result management and export."""

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User, prize_level


class Winner:
    def __init__(self, user: User, prize: Prize, order: int, time: datetime):
        self.user = user
        self.prize = prize
        self.order = order
        self.time = time

    def __repr__(self):
        return f"Winner({self.user.name!r}, {self.prize.name!r}, #{self.order})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self._counter = 0

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name.lower() == user.name.lower() for w in self.winners):
            return False
        self._counter += 1
        self.winners.append(Winner(user, prize, self._counter, datetime.now()))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        prize_list = list(prizes or [])
        names = [p.name for p in prize_list]
        for w in self.winners:
            if w.prize.name not in names:
                names.append(w.prize.name)
        indexed = list(enumerate(names))
        indexed.sort(key=lambda t: (prize_level(t[1]), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, name in indexed:
            result[name] = sorted(
                [w for w in self.winners if w.prize.name == name], key=lambda w: w.order
            )
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Order", "Winning Time"])
                for prize_name, winners in grouped.items():
                    for w in winners:
                        writer.writerow([
                            w.user.name,
                            w.user.department,
                            prize_name,
                            w.order,
                            w.time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        return "q"


class LotteryCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery = None

    # ---------- list ----------
    def _accept_list(self, users, errors) -> None:
        if errors:
            print("\nProblems detected:")
            for e in errors:
                print("  " + e)
            if not users:
                print("No valid data loaded.")
                return
            c = ask("Remove invalid data and continue (r) or terminate upload (t)? [r/t]: ").lower()
            if c != "r":
                print("Upload terminated.")
                return
        valid, dups = self.list_manager.validate_users(users)
        if dups:
            print("\nDuplicate names detected:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
            c = ask("Remove duplicates and continue (r) or terminate upload (t)? [r/t]: ").lower()
            if c != "r":
                print("Upload terminated.")
                return
        self.users = valid
        self.lottery = None
        print(f"List loaded: {len(valid)} valid participants.")

    def menu_list(self):
        while True:
            print("\n--- List Management ---")
            print("1. Load default list\n2. Upload custom txt list (Name,Department per line)")
            print("3. View groups\n4. View participants\n0. Back")
            c = ask("Choice: ")
            if c == "1":
                self._accept_list(*self.list_manager.load_default_list())
            elif c == "2":
                path = ask("File path: ")
                self._accept_list(*self.list_manager.load_custom_list(path))
            elif c == "3":
                info = self.list_manager.get_group_info(self.users)
                if not info:
                    print("No list loaded.")
                for dept, cnt in info.items():
                    print(f"  {dept}: {cnt}")
            elif c == "4":
                for i, u in enumerate(self.users, 1):
                    print(f"  {i}. {u.name} ({u.department})")
                if not self.users:
                    print("No list loaded.")
            elif c in ("0", "q"):
                return

    # ---------- prizes ----------
    def _show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def _select_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = ask("Avoidance groups (numbers or names, comma-separated, empty for none): ")
        result = []
        for token in raw.replace("\uff0c", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                result.append(groups[int(token) - 1])
            else:
                result.append(token)
        return result

    def menu_prizes(self):
        while True:
            print("\n--- Prize Configuration ---")
            print("1. Load preset prizes\n2. Add custom prize\n3. View prizes")
            print("4. Reorder prizes\n0. Back")
            c = ask("Choice: ")
            if c == "1":
                for e in self.prize_manager.load_default_prizes():
                    print("  " + e)
                self._show_prizes()
            elif c == "2":
                name = ask("Prize name: ")
                qty = ask("Quantity: ")
                groups = self._select_groups()
                drawer = ask("Associated lottery person (empty for none): ") or None
                try:
                    self.prize_manager.add_prize(name, qty, groups, drawer)
                    print("Prize added.")
                except ValueError as exc:
                    print(f"Error: {exc}")
            elif c == "3":
                self._show_prizes()
            elif c == "4":
                self._show_prizes()
                raw = ask("New order (serial numbers separated by spaces/commas): ")
                tokens = raw.replace(",", " ").split()
                if self.prize_manager.reorder_prizes(tokens):
                    print("Prizes reordered.")
                    self._show_prizes()
                else:
                    print("Invalid order.")
            elif c in ("0", "q"):
                return

    # ---------- lottery ----------
    def run_lottery(self):
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)
            for w in self.result_manager.winners:
                self.lottery.winner_names.add(w.user.name.lower())
        print("\nCommands during draw: [Enter]=draw next, pause, continue, terminate")
        for rnd, prize in enumerate(prizes, 1):
            print(f"\n===== Round {rnd}: {prize.name} (Quantity: {prize.quantity}) =====")
            print("Rules: " + prize.rule_description())
            if self.lottery.is_fallback_active(prize):
                print("Fallback active: prize pool limited to Cheng Ru group members.")
            round_winners = []
            terminated = False
            for i in range(1, prize.quantity + 1):
                cmd = ask(f"Ready to draw winner {i} > ").lower()
                while cmd == "pause":
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to skip round.")
                    cmd = ask("(paused) > ").lower()
                    if cmd not in ("continue", "terminate"):
                        cmd = "pause"
                if cmd in ("terminate", "q"):
                    print("Current round terminated.")
                    terminated = True
                    break
                print(f"Currently drawing the {i}th winner...")
                user = self.lottery.draw_one(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            self.lottery.record_round(prize, round_winners)
            print(f"--- Round {rnd} results ---")
            for idx, u in enumerate(round_winners, 1):
                print(f"  {idx}. {u.name} ({u.department}) - {prize.name}")
            if not round_winners:
                print("  No winners." + (" (terminated)" if terminated else ""))
        self.show_summary()

    def show_summary(self):
        print("\n===== Complete Winner List =====")
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        any_w = False
        for name, winners in grouped.items():
            if not winners:
                continue
            any_w = True
            print(f"[{name}]")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {name}")
        if not any_w:
            print("No winners yet.")

    def run(self):
        print("=== Company Team Building Smart Lottery System ===")
        while True:
            print("\n1. List management\n2. Prize configuration\n3. Start lottery")
            print("4. Show results\n5. Export results\n0. Exit")
            c = ask("Choice: ")
            if c == "1":
                self.menu_list()
            elif c == "2":
                self.menu_prizes()
            elif c == "3":
                self.run_lottery()
            elif c == "4":
                self.show_summary()
            elif c == "5":
                ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
                print(msg)
            elif c in ("0", "q"):
                print("Goodbye.")
                return


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

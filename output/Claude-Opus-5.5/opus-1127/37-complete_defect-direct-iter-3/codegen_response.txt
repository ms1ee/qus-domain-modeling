    avoid_groups: List[str] = field(default_factory=list)

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if isinstance(self.associated_drawer, str):
            self.associated_drawer = self.associated_drawer.strip() or None

import os
from typing import Dict, Iterable, List, Tuple

from lottery_system.models import User

DEFAULT_LIST_LINES = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Wang Wu,Technology Department",
    "Zhao Liu,Technology Department",
    "Sun Qi,Product Department",
    "Zhou Ba,Product Department",
    "Wu Jiu,Product Department",
    "Zheng Shi,Marketing Department",
    "Feng Yi,Marketing Department",
    "Chen Er,Marketing Department",
    "Chu Wei,Human Resources Department",
    "Wei Lan,Human Resources Department",
    "Jiang Hai,Finance Department",
    "Shen Yue,Finance Department",
    "Han Mei,Cheng Ru Group",
    "Yang Fan,Cheng Ru Group",
    "Zhu Lin,Cheng Ru Group",
    "Qin Feng,Cheng Ru Group",
    "You Xin,Operations Department",
    "Xu Ming,Operations Department",
]


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not path.lower().endswith(".txt"):
            return [], [f"File Error: only .txt files are supported ({path})"]
        if not os.path.isfile(path):
            return [], [f"File Error: file not found ({path})"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except FileNotFoundError:
            return [], [f"File Error: file not found ({path})"]
        except UnicodeDecodeError:
            return [], [f"File Error: file is not valid UTF-8 text ({path})"]
        except OSError as e:
            return [], [f"File Error: cannot read file ({path}): {e}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("\uff0c", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users or []:
            key = u.name.strip()
            if key in seen:
                duplicates.append(u)
            else:
                seen.add(key)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users or []:
            info[u.department] = info.get(u.department, 0) + 1
        return info

"""Prize configuration management."""
from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], None),
    ("Second Prize - Shopping Cards", 2, [], None),
    ("First Prize - 1 Day Annual Leave", 1, [], None),
]


class PrizeManager:
    def __init__(self):
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self._prizes}
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            try:
                self.add_prize(name, qty, avoid, drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        groups: List[str] = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer: Optional[str] = None
        if associated_drawer is not None and str(associated_drawer).strip():
            drawer = str(associated_drawer).strip()
        prize = Prize(str(name).strip(), qty, groups, drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self._prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""
import random
from typing import List, Optional, Set

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(s: Optional[str]) -> str:
    return " ".join((s or "").split()).lower()


def is_protected_group(department: str) -> bool:
    return _norm(department) == _norm(PROTECTED_GROUP)


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users or [])
        self.winner_names: Set[str] = set()
        self.last_two_draw_results: List[bool] = []
        self._rng = random.SystemRandom()

    # ---------- rule helpers ----------
    def is_fallback_active(self, prize: Prize) -> bool:
        return (
            _norm(prize.associated_drawer) == _norm(FALLBACK_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def describe_rules(self, prize: Prize) -> str:
        rules = ["Previous winners are excluded"]
        drawer = prize.associated_drawer
        if _norm(drawer) == _norm(SPECIAL_DRAWER):
            rules.append(f"Special drawer {SPECIAL_DRAWER}: avoidance groups are NOT applied")
        elif self.is_fallback_active(prize):
            rules.append(
                f"Fallback active: pool limited to '{PROTECTED_GROUP}' members "
                f"(avoidance groups ignored)"
            )
        elif prize.avoid_groups:
            rules.append("Avoidance groups: " + ", ".join(prize.avoid_groups))
        else:
            rules.append("No avoidance groups")
        if drawer:
            rules.append(f"Drawer: {drawer}")
        return "; ".join(rules)

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if u.name not in self.winner_names]
        drawer = _norm(prize.associated_drawer)
        if drawer == _norm(SPECIAL_DRAWER):
            return base
        if self.is_fallback_active(prize):
            return [u for u in base if is_protected_group(u.department)]
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in base if _norm(u.department) not in avoid]

    # ---------- drawing ----------
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.winner_names.add(user.name)
        return user

    def finish_round(self, prize: Prize, round_winners: List[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in WATCHED_DRAWERS}:
            contains = any(is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_DRAWER):
            self.last_two_draw_results = []

"""Result storage and export."""
import csv
import os
from dataclasses import dataclass
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize: Prize
    order: int = 0


def _prize_name(p) -> str:
    return p.name if hasattr(p, "name") else str(p)


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self.winners.append(Winner(user, prize, len(self.winners) + 1))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes or []:
            result[_prize_name(p)] = []
        for w in self.winners:
            result.setdefault(_prize_name(w.prize), []).append(w)
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for pname, ws in grouped.items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, pname])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""
from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        return "0"


class App:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []

    # ---------------- list ----------------
    def _accept_list(self, users, errors):
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("\nInvalid data detected:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            choice = ask("Remove invalid data and continue (r) or terminate upload (t)? ").lower()
            if choice != "r":
                print("Upload terminated. Current list unchanged.")
                return
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._accept_list(users, errors)

    def load_custom(self):
        path = ask("Enter txt file path (each line 'Name,Department Group'): ")
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            return
        self._accept_list(users, errors)

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("\nValid groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")
        print(f"Total: {len(self.users)} people")

    # ---------------- prizes ----------------
    def load_preset_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prizes loaded.")

    def add_custom_prize(self):
        name = ask("Prize name: ")
        qty = ask("Quantity: ")
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid: List[str] = []
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = ask("Avoidance groups (numbers or names, comma-separated, blank for none): ")
        for token in raw.replace("\uff0c", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                avoid.append(groups[int(token) - 1])
            else:
                avoid.append(token)
        drawer = ask("Associated lottery person (blank for none): ")
        try:
            p = self.prize_manager.add_prize(name, qty, avoid, drawer or None)
            print(f"Prize '{p.name}' added.")
        except ValueError as e:
            print(f"Error: {e}")

    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("\nPrize order:")
        for i, p in enumerate(prizes, 1):
            avoid = ", ".join(p.avoid_groups) if p.avoid_groups else "None"
            drawer = p.associated_drawer or "None"
            print(f"  {i}. {p.name} x{p.quantity} | Avoid: {avoid} | Drawer: {drawer}")

    def reorder(self):
        self.view_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.view_prizes()
        else:
            print("Invalid order. Must be a permutation of all serial numbers.")

    # ---------------- lottery ----------------
    def start_lottery(self):
        prizes = self.prize_manager.get_prizes()
        if not self.users:
            print("Please load a participant list first.")
            return
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.result_manager.winners:
            if ask("Previous results exist and will be cleared. Continue? (y/n): ").lower() != "y":
                return
        self.result_manager = ResultManager()
        engine = LotteryManager(self.users)

        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            print("Rules: " + engine.describe_rules(prize))
            print("=" * 60)
            round_winners: List[User] = []
            terminated = False
            x = 1
            while x <= prize.quantity:
                cmd = ask(f"[Enter]=draw winner #{x}, 'pause', 'terminate': ").lower()
                if cmd == "pause":
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to skip round.")
                    while True:
                        c = ask("> ").lower()
                        if c == "continue":
                            print("Lottery resumed.")
                            break
                        if c == "terminate":
                            terminated = True
                            break
                    if terminated:
                        break
                    continue
                if cmd == "terminate":
                    terminated = True
                    break
                if cmd == "0":
                    terminated = True
                    break
                print(f"Currently drawing the {x}th winner...")
                user = engine.draw_one(prize)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
                x += 1
            if terminated:
                print("Current round terminated and skipped.")
            engine.finish_round(prize, round_winners)
            print(f"\nRound {rnd} results ({prize.name}):")
            if round_winners:
                for i, u in enumerate(round_winners, 1):
                    print(f"  {i}. {u.name} ({u.department})")
            else:
                print("  No winners.")
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("\n===== Complete Winner List =====")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for pname, ws in grouped.items():
            for w in ws:
                print(f"  {w.user.name} | {w.user.department} | {pname}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": self.load_default,
            "2": self.load_custom,
            "3": self.view_groups,
            "4": self.load_preset_prizes,
            "5": self.add_custom_prize,
            "6": self.view_prizes,
            "7": self.reorder,
            "8": self.start_lottery,
            "9": self.show_summary,
            "10": self.export,
        }
        while True:
            print(
                "\n=== Team Building Smart Lottery ===\n"
                " 1. Load default list\n 2. Upload custom list\n 3. View groups\n"
                " 4. Load preset prizes\n 5. Add custom prize\n 6. View prizes\n"
                " 7. Reorder prizes\n 8. Start lottery\n 9. View winner list\n"
                "10. Export results\n 0. Exit"
            )
            choice = ask("Select: ")
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid option.")


def main():
    App().run()


if __name__ == "__main__":
    main()

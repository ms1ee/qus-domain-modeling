"""Configuration parameters for the food delivery dispatch simulator.

Values can be overridden through a JSON configuration file (see
:func:`load_config`).  Keys in the JSON file are case-insensitive and map
to the module level constants defined below.
"""

import json
import logging
import os

logger = logging.getLogger(__name__)

# Map / grid
GRID_SIZE = 100          # size of one grid cell in units
MAP_SIZE = 800           # map size in units
GRID_CELLS = 8           # number of cells per side

# Economy
INITIAL_FUNDS = 300      # funds available at system start
RIDER_COST = 300         # cost of recruiting a rider (trigger threshold)
ORDER_FEE = 100          # income credited when an order is delivered
MAX_RIDERS = 50          # safety cap on automatic recruitment

# Riders
RIDER_SPEED = 50         # units per tick
RIDER_START_X = 400      # spawn position (street intersection)
RIDER_START_Y = 400

# Files
STATE_FILE = "dispatch_state.json"
CONFIG_FILE = "config.json"

_CONFIGURABLE = (
    "INITIAL_FUNDS", "RIDER_COST", "ORDER_FEE", "MAX_RIDERS",
    "RIDER_SPEED", "RIDER_START_X", "RIDER_START_Y", "STATE_FILE",
)


def load_config(path=None):
    """Load configuration overrides from a JSON file.

    :param path: path of the JSON file (defaults to ``CONFIG_FILE``).
    :return: True if a file was loaded, False otherwise.
    """
    path = path or CONFIG_FILE
    if not os.path.exists(path):
        return False
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except (OSError, ValueError) as exc:
        logger.warning("Could not read config file %s: %s", path, exc)
        return False
    if not isinstance(data, dict):
        return False
    params = data.get("system", data)
    if not isinstance(params, dict):
        return False
    for key, value in params.items():
        name = str(key).upper()
        if name in _CONFIGURABLE:
            current = globals()[name]
            try:
                globals()[name] = type(current)(value)
            except (TypeError, ValueError):
                logger.warning("Invalid value for %s: %r", name, value)
    return True

"""Grid based path planning.

The map is 800x800 units split into 8x8 cells of 100x100.  Grid lines
(coordinates that are multiples of 100) are *streets*.  Riders prefer to
travel along streets: they leave their current block to the nearest
street, travel along horizontal and vertical streets with right-angle
turns, and finally enter the destination block.
"""


class PathPlanner:
    """Path planning helper for grid based movement."""

    GRID = 100
    SIZE = 800

    @staticmethod
    def to_grid_center(c):
        """Convert a raw coordinate (0-800) to its grid cell centre."""
        c = int(c)
        c = max(0, min(PathPlanner.SIZE - 1, c))
        return (c // PathPlanner.GRID) * PathPlanner.GRID + PathPlanner.GRID // 2

    @staticmethod
    def to_grid_index(c):
        """Return the grid index (0-7) of a coordinate."""
        c = max(0, int(c))
        return min(PathPlanner.SIZE // PathPlanner.GRID - 1, c // PathPlanner.GRID)

    @classmethod
    def cell_of(cls, x, y):
        """Return the (gx, gy) grid cell containing a point."""
        return cls.to_grid_index(x), cls.to_grid_index(y)

    @staticmethod
    def is_on_street(x, y):
        """True if the point lies on a street (grid line)."""
        return int(x) % PathPlanner.GRID == 0 or int(y) % PathPlanner.GRID == 0

    @staticmethod
    def nearest_street(c, toward):
        """Nearest street coordinate to ``c``; ties broken toward ``toward``."""
        g = PathPlanner.GRID
        if c % g == 0:
            return c
        lower = (c // g) * g
        upper = lower + g
        dl, du = c - lower, upper - c
        if dl < du:
            s = lower
        elif du < dl:
            s = upper
        else:
            s = upper if toward > c else lower
        return max(0, min(PathPlanner.SIZE, s))

    @staticmethod
    def _simplify(start, pts):
        """Remove duplicate and collinear waypoints."""
        full = [tuple(start)]
        for p in pts:
            p = (int(p[0]), int(p[1]))
            if p == full[-1]:
                continue
            if len(full) >= 2:
                a, b = full[-2], full[-1]
                if (a[0] == b[0] == p[0]) or (a[1] == b[1] == p[1]):
                    full[-1] = p
                    if len(full) >= 2 and full[-1] == full[-2]:
                        full.pop()
                    continue
            full.append(p)
        return full[1:]

    def plan(self, start, target):
        """Plan a path from ``start`` to ``target``.

        :return: tuple ``(waypoints, special_case)``; waypoints exclude the
                 start point.  ``special_case`` is True when the target is
                 directly above or below the start in another cell, in which
                 case the rider detours onto a parallel street.
        """
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        if (sx, sy) == (tx, ty):
            return [], False
        if self.cell_of(sx, sy) == self.cell_of(tx, ty):
            pts = [(tx, sy), (tx, ty)]
            special = False
        else:
            hy = self.nearest_street(sy, ty)
            vx = self.nearest_street(tx, sx)
            pts = [(sx, hy), (vx, hy), (vx, ty), (tx, ty)]
            special = sx == tx
        return self._simplify((sx, sy), pts), special

    def plan_route(self, points):
        """Plan a continuous route through several destinations.

        :param points: list of points; the first is the start.
        :return: list of waypoints including the start point.
        """
        if not points:
            return []
        route = [(int(points[0][0]), int(points[0][1]))]
        for target in points[1:]:
            wps, _ = self.plan(route[-1], target)
            route.extend(wps)
        return route

    @staticmethod
    def path_length(points):
        """Total Manhattan length of a polyline."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    def distance(self, start, target):
        """Length of the planned path between two points."""
        wps, _ = self.plan(start, target)
        return self.path_length([tuple(start)] + wps)

"""Order data model."""

import globalvar
import config
from path_planner import PathPlanner


def _validate_pos(pos, name):
    """Validate a coordinate pair and return it as a tuple of ints."""
    try:
        x, y = pos
    except (TypeError, ValueError):
        raise ValueError("%s must contain two coordinates" % name)
    result = []
    for v in (x, y):
        if isinstance(v, bool):
            raise ValueError("%s coordinates must be integers" % name)
        try:
            iv = int(v)
        except (TypeError, ValueError):
            raise ValueError("%s coordinates must be integers" % name)
        if iv != v and not isinstance(v, str):
            raise ValueError("%s coordinates must be integers" % name)
        if iv < 0 or iv > config.MAP_SIZE:
            raise ValueError("%s coordinates must be within 0-%d"
                             % (name, config.MAP_SIZE))
        result.append(iv)
    return tuple(result)


class Order:
    """A delivery order from a restaurant to a customer.

    Statuses: ``pending`` -> ``assigned`` -> ``picking_up`` ->
    ``delivering`` -> ``completed``.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        """Create an order; raw coordinates are converted to cell centres."""
        self.raw_res = _validate_pos(res_pos, "Restaurant")
        self.raw_cus = _validate_pos(cus_pos, "Customer")
        self.res_pos = (PathPlanner.to_grid_center(self.raw_res[0]),
                        PathPlanner.to_grid_center(self.raw_res[1]))
        self.cus_pos = (PathPlanner.to_grid_center(self.raw_cus[0]),
                        PathPlanner.to_grid_center(self.raw_cus[1]))
        self.id = Order._next_id
        Order._next_id += 1
        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("tick", 0) or 0
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.complete_time = None
        self.fee = config.ORDER_FEE
        self.distance = PathPlanner().distance(self.res_pos, self.cus_pos)

    @classmethod
    def reset_counter(cls, value=1):
        """Reset the order id counter."""
        cls._next_id = value

    @property
    def res_grid(self):
        """Grid cell of the restaurant."""
        return PathPlanner.cell_of(*self.res_pos)

    @property
    def cus_grid(self):
        """Grid cell of the customer."""
        return PathPlanner.cell_of(*self.cus_pos)

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        self.assign_time = globalvar.get_value("tick", 0) or 0

    def complete(self, time=None):
        """Mark the order as completed."""
        self.status = "completed"
        self.complete_time = (globalvar.get_value("tick", 0) or 0
                              if time is None else time)

    @property
    def time_taken(self):
        """Ticks from creation to completion (None if not completed)."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.create_time

    @property
    def efficiency(self):
        """Delivery efficiency in distance units per tick."""
        t = self.time_taken
        if not t:
            return None
        return round(self.distance / t, 2)

    def to_dict(self):
        """Serialise to a JSON compatible dict."""
        return {
            "id": self.id, "raw_res": list(self.raw_res),
            "raw_cus": list(self.raw_cus), "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos), "status": self.status,
            "rider_id": self.rider_id, "create_time": self.create_time,
            "assign_time": self.assign_time, "accept_time": self.accept_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time, "fee": self.fee,
            "distance": self.distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from :meth:`to_dict` output."""
        obj = cls.__new__(cls)
        obj.id = int(data["id"])
        obj.raw_res = tuple(data["raw_res"])
        obj.raw_cus = tuple(data["raw_cus"])
        obj.res_pos = tuple(data["res_pos"])
        obj.cus_pos = tuple(data["cus_pos"])
        obj.status = data.get("status", "pending")
        obj.rider_id = data.get("rider_id")
        obj.create_time = data.get("create_time", 0)
        obj.assign_time = data.get("assign_time")
        obj.accept_time = data.get("accept_time")
        obj.pickup_time = data.get("pickup_time")
        obj.complete_time = data.get("complete_time")
        obj.fee = data.get("fee", config.ORDER_FEE)
        obj.distance = data.get("distance", 0)
        if cls._next_id <= obj.id:
            cls._next_id = obj.id + 1
        return obj

    def __repr__(self):
        return "Order(id=%d, %s->%s, %s)" % (self.id, self.res_pos,
                                             self.cus_pos, self.status)

"""Rider model: movement, state machine and event log."""

import logging

import config
import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {STATE_IDLE: "idle", STATE_PICKING_UP: "picking up",
               STATE_DELIVERING: "delivering"}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to :attr:`orders`.  Each call to
    :meth:`move` advances the rider by one tick.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = int(rider_id)
        self.x = int(config.RIDER_START_X if x is None else x)
        self.y = int(config.RIDER_START_Y if y is None else y)
        self.speed = int(config.RIDER_SPEED if speed is None else speed)
        if self.speed <= 0:
            self.speed = 1
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.path = []
        self.log = []
        self.total_distance = 0
        self.completed_count = 0
        self.planner = PathPlanner()
        self._record("spawn", None)

    # ------------------------------------------------------------------ utils
    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    def _record(self, event, order_id):
        """Append an entry to the rider's event log."""
        self.log.append({
            "event": event, "order_id": order_id,
            "time": globalvar.get_value("tick", 0) or 0,
            "x": self.x, "y": self.y, "state": self.state,
        })

    def active_orders(self):
        """Orders that are not completed yet."""
        return [o for o in self.orders if o.status != "completed"]

    def current_order(self):
        """The order currently being worked on (FIFO)."""
        active = self.active_orders()
        return active[0] if active else None

    def expected_free_position(self, exclude=None):
        """Position where the rider will be once current work is done."""
        active = [o for o in self.active_orders() if o is not exclude]
        if active:
            return tuple(active[-1].cus_pos)
        return (self.x, self.y)

    # --------------------------------------------------------------- movement
    def _advance(self, budget):
        """Move along the path using up to ``budget`` units.

        :return: True if the path has been fully travelled.
        """
        while budget > 0 and self.path:
            wx, wy = self.path[0]
            dx, dy = wx - self.x, wy - self.y
            if dx != 0:
                step = min(abs(dx), budget)
                self.x += step if dx > 0 else -step
                budget -= step
                self.total_distance += step
            elif dy != 0:
                step = min(abs(dy), budget)
                self.y += step if dy > 0 else -step
                budget -= step
                self.total_distance += step
            if (self.x, self.y) == (wx, wy):
                self.path.pop(0)
        return not self.path

    def _set_path(self, target):
        """Plan a path from the current position to ``target``."""
        self.path, self.special_case = self.planner.plan((self.x, self.y),
                                                          target)

    def _accept(self, order):
        """Idle -> picking up."""
        self.state = STATE_PICKING_UP
        if order.rider_id is None:
            order.rider_id = self.rider_id
        order.status = "picking_up"
        order.accept_time = globalvar.get_value("tick", 0) or 0
        self._set_path(order.res_pos)
        self._record("accepted_order", order.id)

    def _arrive_merchant(self, order):
        """Picking up -> delivering."""
        self.state = STATE_DELIVERING
        order.status = "delivering"
        order.pickup_time = globalvar.get_value("tick", 0) or 0
        self._record("arrived_at_merchant", order.id)
        self._set_path(order.cus_pos)

    def _deliver(self, order):
        """Delivering -> idle; completes the order and credits funds."""
        order.complete()
        if order in self.orders:
            self.orders.remove(order)
        self.completed_count += 1
        self.state = STATE_IDLE
        self.path = []
        self.special_case = False
        self._record("delivered_to_customer", order.id)
        funds = globalvar.get_value("funds")
        if funds is not None:
            globalvar.set_value("funds", funds + getattr(order, "fee", 0))

    def move(self):
        """Advance one tick: accept, travel and perform state transitions."""
        order = self.current_order()
        if self.state == STATE_IDLE:
            if order is None:
                return
            self._accept(order)
        elif order is None:
            # Consistency recovery: busy without an order.
            logger.warning("Rider %d busy without order; resetting",
                           self.rider_id)
            self.state = STATE_IDLE
            self.path = []
            self.special_case = False
            return
        elif self.state == STATE_PICKING_UP and order.status == "delivering":
            self.state = STATE_DELIVERING
        arrived = self._advance(self.speed)
        if not arrived:
            return
        if self.state == STATE_PICKING_UP:
            self._arrive_merchant(order)
        elif self.state == STATE_DELIVERING:
            self._deliver(order)

    # ------------------------------------------------------------ persistence
    def to_dict(self):
        """Serialise to a JSON compatible dict."""
        return {
            "rider_id": self.rider_id, "x": self.x, "y": self.y,
            "speed": self.speed, "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "path": [list(p) for p in self.path], "log": self.log,
            "total_distance": self.total_distance,
            "completed_count": self.completed_count,
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        """Rebuild a rider from :meth:`to_dict` output."""
        r = cls(data["rider_id"], data.get("x"), data.get("y"),
                data.get("speed"))
        r.state = int(data.get("state", 0))
        r.orders = [orders_by_id[i] for i in data.get("orders", [])
                    if i in orders_by_id]
        r.special_case = bool(data.get("special_case", False))
        r.path = [tuple(p) for p in data.get("path", [])]
        r.log = list(data.get("log", r.log))
        r.total_distance = data.get("total_distance", 0)
        r.completed_count = data.get("completed_count", 0)
        return r

    def __repr__(self):
        return "Rider(id=%d, pos=(%d,%d), state=%s, orders=%s)" % (
            self.rider_id, self.x, self.y, self.state_name,
            [o.id for o in self.orders])

"""Order creation, assignment and completion handling."""

import math

import globalvar
from order_model import Order
from path_planner import PathPlanner


def _rider_score(rider, order):
    """Sort key: load first (balancing), then distance, then id."""
    active = [o for o in rider.orders if o.status != "completed"]
    ex, ey = rider.expected_free_position()
    rx, ry = order.res_pos
    dist = abs(ex - rx) + abs(ey - ry)
    return (len(active), dist, rider.rider_id)


def find_best_rider_for_order(order):
    """Return the best rider for ``order`` (closest available first).

    Idle riders (no active orders) are preferred; among riders with equal
    load the one closest to the restaurant wins.  Returns None when there
    are no riders.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    return min(riders, key=lambda r: _rider_score(r, order))


class OrderManager:
    """Handles order creation, dispatching and reporting."""

    def __init__(self, system_manager=None):
        self.system = system_manager
        self.planner = PathPlanner()

    @staticmethod
    def orders():
        """All orders known to the system."""
        lst = globalvar.get_value("orders")
        if lst is None:
            lst = []
            globalvar.set_value("orders", lst)
        return lst

    def create_order(self, rx, ry, cx, cy):
        """Create and register an order from raw coordinates."""
        order = Order((rx, ry), (cx, cy))
        self.orders().append(order)
        return order

    def dispatch(self, order):
        """Assign ``order`` to the best rider, recruiting first if possible.

        :return: tuple ``(rider or None, recruited: bool)``.
        """
        recruited = False
        if self.system is not None:
            recruited = self.system.check_recruitment() is not None
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None, recruited
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider, recruited

    def assign_pending(self):
        """Try to assign all pending orders; returns number assigned."""
        count = 0
        for order in self.orders():
            if order.status == "pending" and order.rider_id is None:
                rider = find_best_rider_for_order(order)
                if rider is None:
                    break
                order.assign_to(rider.rider_id)
                rider.orders.append(order)
                count += 1
        return count

    def planned_route(self, rider, order):
        """Planned route for ``rider`` to fulfil ``order``.

        :return: tuple ``(route_points, distance, eta_ticks)``.
        """
        start = rider.expected_free_position(exclude=order)
        route = self.planner.plan_route([start, order.res_pos, order.cus_pos])
        dist = self.planner.path_length(route)
        eta = math.ceil(dist / rider.speed) if rider.speed else 0
        return route, dist, eta

    def completed_orders(self):
        """List of completed orders."""
        return [o for o in self.orders() if o.status == "completed"]

"""System manager: funds, riders, time and consistency."""

import logging

import config
import globalvar
from order_model import Order
from rider import Rider, STATE_IDLE

logger = logging.getLogger(__name__)


class SystemManager:
    """Core manager for funds, riders and the simulation clock."""

    @property
    def funds(self):
        """Current system funds."""
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def riders(self):
        """List of riders."""
        lst = globalvar.get_value("riders")
        if lst is None:
            lst = []
            globalvar.set_value("riders", lst)
        return lst

    @property
    def tick(self):
        """Current simulated time."""
        return globalvar.get_value("tick", 0)

    @property
    def running(self):
        """Whether the main loop should keep running."""
        return bool(globalvar.get_value("running", False))

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    def add_rider(self):
        """Create a new rider with an auto-incremented ID."""
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        return rider

    def remove_rider(self, rider_id):
        """Remove an idle rider by id; returns True on success."""
        for r in self.riders:
            if r.rider_id == rider_id and not r.active_orders():
                self.riders.remove(r)
                return True
        return False

    def get_rider(self, rider_id):
        """Find a rider by id."""
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    def check_recruitment(self):
        """Recruit a rider if funds allow; returns the rider or None."""
        if self.funds >= config.RIDER_COST and \
                len(self.riders) < config.MAX_RIDERS:
            self.funds = self.funds - config.RIDER_COST
            rider = self.add_rider()
            logger.info("Recruited rider %d", rider.rider_id)
            return rider
        return None

    def advance_tick(self):
        """Advance the simulation by one tick."""
        globalvar.set_value("tick", self.tick + 1)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception as exc:  # recovery mechanism
                logger.error("Rider %d error: %s", rider.rider_id, exc)
                self.consistency_check()
        from order import OrderManager
        OrderManager(self).assign_pending()
        self.check_recruitment()

    def consistency_check(self):
        """Repair inconsistent state; returns list of fixes performed."""
        fixes = []
        for r in self.riders:
            if r.state != STATE_IDLE and not r.active_orders():
                r.state = STATE_IDLE
                r.path = []

"""Status display and ASCII map visualisation."""

from path_planner import PathPlanner
from rider import STATE_NAMES

N = 17  # characters per side: 9 street lines + 8 cells


def _cell(x, y):
    col = min(N - 1, max(0, int(x) // 50))
    row = min(N - 1, max(0, int(y) // 50))
    return row, col


class DisplayManager:
    """Builds textual representations of the system state."""

    def _base(self):
        return [['#' if (r % 2 == 0 or c % 2 == 0) else '.'
                 for c in range(N)] for r in range(N)]

    @staticmethod
    def _mark(grid, r, c, ch):
        cur = grid[r][c]
        if (cur == '-' and ch == '|') or (cur == '|' and ch == '-') \
                or cur == '+':
            grid[r][c] = '+'
        else:
            grid[r][c] = ch

    def _draw_route(self, grid, route):
        for a, b in zip(route, route[1:]):
            r1, c1 = _cell(*a)
            r2, c2 = _cell(*b)
            if r1 == r2:
                for c in range(min(c1, c2), max(c1, c2) + 1):
                    self._mark(grid, r1, c, '-')
            else:
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    self._mark(grid, r, c1, '|')
        for p in route[1:-1]:
            r, c = _cell(*p)
            grid[r][c] = '+'

    def render_map(self, riders, orders=None, route=None, title=None):
        """Render the 8x8 grid as ASCII art."""
        grid = self._base()
        if route:
            self._draw_route(grid, route)
        for o in orders or []:
            if o.status == "completed":
                continue
            r, c = _cell(*o.res_pos)
            grid[r][c] = 'S'
            r, c = _cell(*o.cus_pos)
            grid[r][c] = 'C'
        for rd in riders or []:
            r, c = _cell(rd.x, rd.y)
            grid[r][c] = 'R'
        lines = []
        if title:
            lines.append(title)
        header = [str(c // 2) if c % 2 else ' ' for c in range(N)]
        lines.append("    " + " ".join(header))
        for r in range(N):
            label = str(r // 2) if r % 2 else ' '
            lines.append("%3s " % label + " ".join(grid[r]))
        lines.append("Legend: R=Rider S=Restaurant C=Customer "
                     "#=Street .=Block -|+=Route")
        return "\n".join(lines)

    def show_route(self, rider, order, route, distance, eta):
        """Render the route map for an assigned rider."""
        title = ("Route for rider %d (speed %d units/tick): distance %d, "
                 "estimated time %d ticks"
                 % (rider.rider_id, rider.speed, distance, eta))
        pts = " -> ".join("(%d,%d)" % p for p in route)
        return "\n".join([self.render_map([rider], [order], route, title),
                          "Waypoints: " + pts])

    def show_status(self, funds, tick, riders, orders):
        """Snapshot of the system state."""
        pending = [o.id for o in orders if o.status == "pending"]
        lines = ["=== System Status (tick %d) ===" % tick,
                 "Funds: %d" % funds,
                 "Riders: %d" % len(riders),
                 "Orders: %d total, %d completed, pending: %s" % (
                     len(orders),
                     len([o for o in orders if o.status == "completed"]),
                     pending or "none")]
        for r in riders:
            lines.append(
                "  Rider %d: pos=(%d,%d) grid=%s speed=%d state=%s "
                "orders=%s special=%s delivered=%d" % (
                    r.rider_id, r.x, r.y, PathPlanner.cell_of(r.x, r.y),
                    r.speed, STATE_NAMES.get(r.state, "?"),
                    [o.id for o in r.orders], r.special_case,
                    r.completed_count))
        return "\n".join(lines)

    def show_orders(self, orders):
        """Table with details of all orders."""
        if not orders:
            return "No orders yet."
        lines = ["%-4s %-6s %-20s %-20s %-6s %-8s %-6s %-11s" % (
            "ID", "Start", "Restaurant", "Customer", "Dist", "Taken",
            "Rider", "Status")]
        for o in orders:
            taken = o.time_taken
            lines.append("%-4d %-6d %-20s %-20s %-6d %-8s %-6s %-11s" % (
                o.id, o.create_time,
                "%s->%s" % (tuple(o.raw_res), tuple(o.res_pos)),
                "%s->%s" % (tuple(o.raw_cus), tuple(o.cus_pos)),
                o.distance, "-" if taken is None else str(taken),
                "-" if o.rider_id is None else str(o.rider_id), o.status))
        return "\n".join(lines)

    def show_rider_logs(self, riders):
        """All recorded rider status changes."""
        if not riders:
            return "No riders yet."
        lines = []
        for r in riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%-5s %-22s pos=(%s,%s) order=%s" % (
                    e.get("time"), e.get("event"), e.get("x"), e.get("y"),
                    e.get("order_id") if e.get("order_id") is not None
                    else "-"))
        return "\n".join(lines)

"""Command parsing and input validation."""

import config

try:  # command line editing/history where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover - Windows
    readline = None


class InputManager:
    """Parses commands, supports shortcuts and history, validates input."""

    SHORTCUTS = {
        "s": "start", "st": "status", "o": "orders", "r": "riders",
        "m": "map", "t": "tick", "h": "help", "?": "help",
        "q": "quit", "e": "exit", "hist": "history",
    }
    COMMANDS = {
        "start": "Input an order: start [rx ry cx cy] (0-800)",
        "status": "Show funds and rider states",
        "orders": "Show details of all orders",
        "riders": "Show rider status change log",
        "map": "Show the ASCII map",
        "tick": "Advance time: tick [n]",
        "history": "Show command history (!n repeats command n)",
        "save": "Save state to JSON: save [file]",
        "load": "Load state from JSON: load [file]",
        "help": "Show this help",
        "quit": "Exit the program (also: exit)",
        "exit": "Exit the program",
    }

    def __init__(self):
        self.history = []

    @staticmethod
    def tokenize(line):
        """Split a line into tokens, tolerating commas and brackets."""
        for ch in ",[]()":
            line = line.replace(ch, " ")
        return line.split()

    def parse(self, line):
        """Parse a command line into ``(command, args)``."""
        line = (line or "").strip()
        if not line:
            return "", []
        if line.startswith("!") and line[1:].isdigit():
            idx = int(line[1:])
            if 1 <= idx <= len(self.history):
                line = self.history[idx - 1]
            else:
                return "invalid_history", []
        self.history.append(line)
        if len(self.history) > 1000:
            self.history = self.history[-1000:]
        tokens = self.tokenize(line)
        if not tokens:
            return "", []
        cmd = tokens[0].lower()
        cmd = self.SHORTCUTS.get(cmd, cmd)
        return cmd, tokens[1:]

    def read_command(self, prompt="> "):
        """Read and parse a command from stdin."""
        return self.parse(input(prompt))

    @staticmethod
    def validate_coordinates(tokens):
        """Validate four integer coordinates in 0-800.

        :raises ValueError: with a user friendly message.
        """
        if len(tokens) != 4:
            raise ValueError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y (got %d)" % len(tokens))
        values = []
        for t in tokens:
            try:
                v = int(str(t))
            except ValueError:
                raise ValueError("'%s' is not an integer" % t)
            if v < 0 or v > config.MAP_SIZE:
                raise ValueError("Coordinate %d out of range 0-%d"
                                 % (v, config.MAP_SIZE))
            values.append(v)
        return values

    def help_text(self):
        """Formatted help text."""
        lines = ["Commands:"]
        for cmd, desc in self.COMMANDS.items():
            short = [k for k, v in self.SHORTCUTS.items() if v == cmd]
            lines.append("  %-8s %-10s %s" % (cmd, ",".join(short), desc))
        return "\n".join(lines)

        "riders": [r.to_dict() for r in globalvar.get_value("riders") or []],
    }
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, indent=2)
    return path


def load_state(path=None):
    """Load system state from a JSON file previously written by save."""
    path = path or config.STATE_FILE
    with open(path, "r", encoding="utf-8") as fh:
        data = json.load(fh)
    sysd = data.get("system", {})
    Order.reset_counter()
    orders = [Order.from_dict(d) for d in data.get("orders", [])]
    by_id = {o.id: o for o in orders}
    globalvar.set_value("funds", int(sysd.get("funds", 0)))
    globalvar.set_value("tick", int(sysd.get("tick", 0)))
    riders = [Rider.from_dict(d, by_id) for d in data.get("riders", [])]
    counter = int(sysd.get("rider_id_counter", 0))
    if riders:
        counter = max(counter, max(r.rider_id for r in riders))
    globalvar.set_value("rider_id_counter", counter)
    globalvar.set_value("orders", orders)
    globalvar.set_value("riders", riders)
    globalvar.set_value("running", True)
    return path

"""Command line entry point for the food delivery dispatch simulator."""

import argparse
import logging
import sys

import config
import globalvar
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from path_planner import PathPlanner
from persistence import load_state, save_state
from system import init_system


class App:
    """Main application tying all managers together."""

    def __init__(self):
        globalvar._init()
        self.system = init_system()
        self.orders = OrderManager(self.system)
        self.display = DisplayManager()
        self.input = InputManager()

    # ---------------------------------------------------------------- commands
    def cmd_start(self, args):
        if not args:
            raw = input("Enter restaurant_x restaurant_y customer_x "
                        "customer_y (0-800): ")
            args = self.input.tokenize(raw)
        rx, ry, cx, cy = self.input.validate_coordinates(args)
        order = self.orders.create_order(rx, ry, cx, cy)
        rider, recruited = self.orders.dispatch(order)
        print("Order #%d created." % order.id)
        print("Restaurant: (%d,%d) -> grid %s centre %s"
              % (rx, ry, order.res_grid, order.res_pos))
        print("Customer:   (%d,%d) -> grid %s centre %s"
              % (cx, cy, order.cus_grid, order.cus_pos))
        print("New rider added: %s" % ("yes" if recruited else "no"))
        if rider is None:
            print("No rider available; order is pending (funds: %d)."
                  % self.system.funds)
            return
        route, dist, eta = self.orders.planned_route(rider, order)
        queued = len(rider.active_orders()) - 1
        print("Assigned to rider %d%s." % (
            rider.rider_id,
            "" if queued <= 0 else " (queued after %d order(s))" % queued))
        print(self.display.show_route(rider, order, route, dist, eta))

    def cmd_tick(self, args):
        n = 1
        if args:
            try:
                n = int(args[0])
            except ValueError:
                raise ValueError("tick expects an integer")
            if n < 1 or n > 100000:
                raise ValueError("tick count must be within 1-100000")
        # One tick is consumed by the main loop itself.
        for _ in range(n - 1):
            self.system.advance_tick()
        print("Advancing %d tick(s)." % n)

    def handle(self, cmd, args):
        """Dispatch a parsed command. Returns False if no tick should pass."""
        g = globalvar
        if cmd == "":
            return True
        if cmd == "start":
            self.cmd_start(args)
        elif cmd == "status":
            print(self.display.show_status(self.system.funds,
                                           self.system.tick,
                                           self.system.riders,
                                           self.orders.orders()))
        elif cmd == "orders":
            print(self.display.show_orders(self.orders.orders()))
        elif cmd == "riders":
            print(self.display.show_rider_logs(self.system.riders))
        elif cmd == "map":
            print(self.display.render_map(self.system.riders,
                                          self.orders.orders(),
                                          title="Map at tick %d"
                                          % self.system.tick))
        elif cmd == "tick":
            self.cmd_tick(args)
        elif cmd == "history":
            for i, h in enumerate(self.input.history, 1):
                print("%4d  %s" % (i, h))
            return False
        elif cmd == "save":
            print("State saved to %s" % save_state(args[0] if args else None))
            return False
        elif cmd == "load":
            print("State loaded from %s"
                  % load_state(args[0] if args else None))
            return False
        elif cmd == "help":
            print(self.input.help_text())
            return False
        elif cmd in ("quit", "exit"):
            g.set_value("running", False)
            return False
        elif cmd == "invalid_history":
            print("No such history entry.")
            return False
        else:
            print("Unknown command '%s'. Type 'help' for commands." % cmd)
            return False
        return True

    def run(self):
        """Main loop: each iteration represents one tick."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Funds: %d. Type 'help' for commands." % self.system.funds)
        while self.system.running:
            try:
                cmd, args = self.input.read_command(
                    "[t=%d] > " % self.system.tick)
            except (EOFError, KeyboardInterrupt):
                print()
                break
            try:
                advance = self.handle(cmd, args)
            except ValueError as exc:
                print("Input error: %s" % exc)
                continue
            except (OSError, KeyError) as exc:
                print("Error: %s" % exc)
                continue
            except Exception as exc:  # recovery mechanism
                logging.exception("Unexpected error")
                print("Unexpected error: %s. Recovering..." % exc)
                for fix in self.system.consistency_check():
                    print("  fixed: %s" % fix)
                continue
            if advance and self.system.running:
                self.system.advance_tick()
        print("Goodbye.")


def main(argv=None):
    """Program entry point."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch")
    parser.add_argument("--config", default=None, help="JSON config file")
    parser.add_argument("--debug", action="store_true", help="debug logs")
    ns = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if ns.debug else logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")
    config.load_config(ns.config)
    PathPlanner.SIZE = config.MAP_SIZE
    App().run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A lottery participant."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None

class ListManager:
    """Manages participant lists."""

    def _parse_lines(self, lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().replace("\ufeff", "")
            if not line:
                continue
            line = line.replace("\uff0c", ",")  # full-width comma
            parts = line.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            name, dept = parts[0].strip(), parts[1].strip()
            if not name or not dept:
                errors.append(
                    f"Line {idx}: Format Error - name or department is empty in '{line}'"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not str(path).strip():
            return [], ["File Error - no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error - file not found: {path}"]
        if not path.lower().endswith(".txt"):
            return [], [f"File Error - only .txt files are supported: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except (OSError, UnicodeDecodeError) as exc:
                return [], [f"File Error - cannot read file: {exc}"]
        except OSError as exc:
            return [], [f"File Error - cannot read file: {exc}"]
        return self._parse_lines(lines)

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            if user.name in seen:
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
    ("Technology Department Exclusive Coupon", 2, ["Product Department"], "Li Bin"),
]


class PrizeManager:
    """Manages the prize library and the drawing order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else name
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists, skipped")
        groups: List[str] = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def get_prize(self, index: int) -> Optional[Prize]:
        if 0 <= index < len(self.prizes):
            return self.prizes[index]
        return None

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional, Set

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def _norm(text) -> str:
    return " ".join(str(text or "").split()).lower()


def is_protected_group(department) -> bool:
    return _norm(department) == _norm(PROTECTED_GROUP)


class LotteryManager:
    """Applies the lottery rules and performs draws."""

    def __init__(self, users) -> None:
        self.users: List[User] = list(users)
        self.winner_names: Set[str] = set()
        self.last_two_draw_results: List[bool] = []
        self._rng = random.SystemRandom()

    # ---- rules -------------------------------------------------------
    def _not_won(self) -> List[User]:
        return [u for u in self.users if u.name not in self.winner_names]

    def fallback_active(self, prize: Prize) -> bool:
        return (
            _norm(prize.associated_drawer) == _norm(FALLBACK_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize) -> List[User]:
        base = self._not_won()
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return base
        if self.fallback_active(prize):
            protected = [u for u in base if is_protected_group(u.department)]
            if protected:
                return protected
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in base if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        rules = ["Previous winners are excluded"]
        drawer = prize.associated_drawer
        if _norm(drawer) == _norm(SPECIAL_DRAWER):
            rules.append(f"Special drawer {SPECIAL_DRAWER}: avoidance groups are NOT applied")
        elif self.fallback_active(prize):
            rules.append(
                f"Fallback triggered: pool limited to '{PROTECTED_GROUP}' members "
                "(avoidance groups ignored)"
            )
        elif prize.avoid_groups:
            rules.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        else:
            rules.append("No avoidance groups")
        if drawer:
            rules.insert(0, f"Drawer: {drawer}")
        return "; ".join(rules)

    # ---- drawing -----------------------------------------------------
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.winner_names.add(user.name)
        return user

    def mark_winner(self, user: User) -> None:
        self.winner_names.add(user.name)

    def record_round(self, prize: Prize, winners) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in WATCHED_DRAWERS}:
            has_protected = any(is_protected_group(u.department) for u in winners)
            self.last_two_draw_results.append(has_protected)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            self.last_two_draw_results = []

"""Result storage, display and export."""

import csv
import os
from dataclasses import dataclass
from datetime import datetime
from typing import Any, Dict, List, Tuple

from lottery_system.models import User


def _prize_name(prize) -> str:
    return prize if isinstance(prize, str) else getattr(prize, "name", str(prize))


@dataclass
class Winner:
    user: User
    prize: Any

    @property
    def prize_name(self) -> str:
        return _prize_name(self.prize)


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self.winners.append(Winner(user, prize))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes:
            result[_prize_name(p)] = []
        for w in self.winners:
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for pname, ws in grouped.items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, pname])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


def ask(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        return "0"


class App:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.drawn_prizes = []

    # ---- list -------------------------------------------------------
    def _accept_users(self, users, errors) -> None:
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Problems detected in the list:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            while True:
                c = ask("Type 'r' to remove invalid data and continue, 't' to terminate upload: ").strip().lower()
                if c == "r":
                    break
                if c == "t":
                    print("Upload terminated. Current list unchanged.")
                    return
        if not valid:
            print("No valid users loaded.")
            return
        self.users = valid
        print(f"Loaded {len(valid)} valid users.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._accept_users(users, errors)

    def load_custom(self) -> None:
        path = ask("Enter txt file path (each line 'Name,Department Group'): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            print("Nothing loaded.")
            return
        self._accept_users(users, errors)

    def view_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")
        print(f"Total: {len(self.users)} people")

    # ---- prizes -----------------------------------------------------
    def load_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prize library loaded.")
        self.show_prizes()

    def _choose_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = ask("Avoidance groups (numbers or names, comma separated, blank for none): ").strip()
        result: List[str] = []
        for part in raw.replace("\uff0c", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                g = groups[int(part) - 1]
            else:
                g = part
            if g not in result:
                result.append(g)
        return result

    def add_custom_prize(self) -> None:
        name = ask("Prize name: ").strip()
        qty = ask("Quantity: ").strip()
        drawer = ask("Associated lottery person (blank for none): ").strip() or None
        avoid = self._choose_groups()
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        self.show_prizes()
        raw = ask("Select prize number: ").strip()
        if not raw.isdigit() or not 1 <= int(raw) <= len(prizes):
            print("Invalid selection.")
            return
        prize = prizes[int(raw) - 1]
        prize.avoid_groups = self._choose_groups()
        print(f"Avoidance groups for '{prize.name}': {', '.join(prize.avoid_groups) or 'None'}")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        print("Prize order:")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | drawer: {p.associated_drawer or '-'} "
                  f"| avoid: {', '.join(p.avoid_groups) or 'None'}")

    def reorder(self) -> None:
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = ask("Enter new order as serial numbers (e.g. 3,1,2): ").strip()
        try:
            order = [int(x) for x in raw.replace("\uff0c", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Order updated.")
            self.show_prizes()
        else:
            print("Invalid order: must contain every serial number exactly once.")

    # ---- lottery ----------------------------------------------------
    def _control(self) -> str:
        """Return 'draw' or 'terminate'."""
        paused = False
        while True:
            prompt = ("[Paused] Type 'continue' or 'terminate': " if paused
                      else "Press Enter to draw ('pause' / 'terminate'): ")
            cmd = ask(prompt).strip().lower()
            if cmd == "terminate":
                return "terminate"
            if paused:
                if cmd == "continue":
                    paused = False
                    print("Lottery resumed.")
                continue
            if cmd == "pause":
                paused = True
                print("Lottery paused.")
                continue
            if cmd in ("", "continue", "draw"):
                return "draw"
            print("Unknown command.")

    def start_lottery(self) -> None:
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        lottery = LotteryManager(self.users)
        self.result_manager = ResultManager()
        self.drawn_prizes = prizes
        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            print(f"Rules: {lottery.describe_rules(prize)}")
            print("=" * 60)
            round_winners: List[User] = []
            for i in range(1, prize.quantity + 1):
                if self._control() == "terminate":
                    print("Current round terminated, skipping.")
                    break
                print(f"Currently drawing the {ordinal(i)} winner...")
                user = lottery.draw_one(prize)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                if not self.result_manager.add_winner(user, prize):
                    continue
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            lottery.record_round(prize, round_winners)
            print(f"\nRound {rnd} results ({prize.name}):")
            if round_winners:
                for i, u in enumerate(round_winners, 1):
                    print(f"  {i}. {u.name} ({u.department})")
            else:
                print("  No winners.")
        print("\nAll rounds completed.")
        self.show_results()

    def show_results(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.drawn_prizes)
        if not any(grouped.values()):
            print("No winners yet.")
            return
        print("\nComplete winner list:")
        print(f"  {'Name':<20}{'Department':<30}Prize")
        for pname, ws in grouped.items():
            for w in ws:
                print(f"  {w.user.name:<20}{w.user.department:<30}{pname}")

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.drawn_prizes)
        print(msg)

    # ---- main loop --------------------------------------------------
    def run(self) -> None:
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_prizes, "5": self.add_custom_prize, "6": self.set_avoid_groups,
            "7": self.reorder, "8": self.start_lottery, "9": self.show_results,
            "10": self.export,
        }
        while True:
            print("\n===== Team Building Smart Lottery System =====")
            print(" 1. Load default list      2. Upload custom txt list")
            print(" 3. View groups            4. Load preset prizes")
            print(" 5. Add custom prize       6. Set prize avoidance groups")
            print(" 7. View / reorder prizes  8. Start lottery")
            print(" 9. View all results      10. Export results")
            print(" 0. Exit")
            choice = ask("Select: ").strip()
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid option.")


def main() -> None:
    App().run()


if __name__ == "__main__":
    main()

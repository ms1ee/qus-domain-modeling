from __future__ import annotations

from typing import List, Optional


class User:
    """A participant of the team building event."""

    def __init__(self, name: str, department: str):
        self.name = name
        self.department = department
        self.is_valid = True
        self.has_won = False

    def __repr__(self) -> str:
        return f"User(name={self.name!r}, department={self.department!r})"

    def __eq__(self, other) -> bool:
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self) -> int:
        return hash((self.name, self.department))


class Prize:
    """A prize to be drawn in one lottery round."""

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ):
        self.name = name
        self.quantity = quantity
        self.avoid_groups: List[str] = list(avoid_groups) if avoid_groups else []
        self.associated_drawer: Optional[str] = associated_drawer or None

    @property
    def rule_description(self) -> str:
        parts = ["Previous winners are excluded"]
        drawer = self.associated_drawer
        if drawer:
            parts.append(f"Drawer: {drawer}")
        if drawer and drawer.strip().lower() == "li bin":
            parts.append("Special drawer: avoidance groups are NOT applied")
        elif self.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("No avoidance groups")
        return "; ".join(parts)

    def __repr__(self) -> str:
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
            f"avoid_groups={self.avoid_groups!r}, "
            f"associated_drawer={self.associated_drawer!r})"
        )

class ListManager:
    """Loads, validates and analyses participant lists."""

    def __init__(self):
        self.load_mode = None
        self.source_file_path = None

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        self.load_mode = "default"
        self.source_file_path = None
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        self.load_mode = "custom"
        self.source_file_path = path
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except (OSError, UnicodeDecodeError) as exc:
                return [], [f"Unable to read file: {path} ({exc})"]
        except OSError as exc:
            return [], [f"Unable to read file: {path} ({exc})"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # --------------------------------------------------------------- validation
    @staticmethod
    def validate_users(users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            if user.name in seen:
                user.is_valid = False
                duplicates.append(user)
            else:
                seen.add(user.name)
                user.is_valid = True
                valid.append(user)
        return valid, duplicates

    @staticmethod
    def get_group_info(users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the configured prizes and their draw order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, avoid, drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(
        self,
        name,
        quantity,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ) -> Prize:
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        prize = Prize(name.strip(), qty, groups, drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine and draw execution."""

from __future__ import annotations

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
PROTECTED_GROUP = "Cheng Ru Group"
TRIGGER_DRAWERS = ["Chen Qixian", "Jin Duo"]
TARGET_DRAWER = "Ma Jiaqi"


def _norm_name(name) -> str:
    return " ".join(str(name or "").split()).lower()


def _norm_group(group) -> str:
    g = _norm_name(group)
    if g.endswith(" group"):
        g = g[: -len(" group")].strip()
    return g


def is_protected_group(department) -> bool:
    return _norm_group(department) == _norm_group(PROTECTED_GROUP)


class LotteryManager:
    """Applies lottery rules and draws winners."""

    def __init__(self, users, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()
        self.fallback_applied = False

    # ------------------------------------------------------------ rule engine
    def _fallback_active(self, prize: Prize) -> bool:
        return (
            _norm_name(prize.associated_drawer) == _norm_name(TARGET_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if u.name not in self.winner_names and not u.has_won]
        self.fallback_applied = False
        if self._fallback_active(prize):
            protected = [u for u in base if is_protected_group(u.department)]
            if protected:
                self.fallback_applied = True
                return protected
        if _norm_name(prize.associated_drawer) == _norm_name(SPECIAL_DRAWER):
            return base
        avoid = {_norm_group(g) for g in prize.avoid_groups}
        if not avoid:
            return base
        return [u for u in base if _norm_group(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        desc = prize.rule_description
        if self._fallback_active(prize):
            desc += f"; FALLBACK: pool limited to {PROTECTED_GROUP} members"
        return desc

    # ---------------------------------------------------------------- drawing
    def mark_winner(self, user: User) -> None:
        self.winner_names.add(user.name)
        user.has_won = True

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self.rng.choice(pool)
        self.mark_winner(winner)
        return winner

    def record_round_result(self, prize: Prize, winners: List[User]) -> None:
        drawer = _norm_name(prize.associated_drawer)
        if drawer in {_norm_name(t) for t in TRIGGER_DRAWERS}:
            contains = any(is_protected_group(w.department) for w in winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            self.last_two_draw_results = []

    def run_round(
        self,
        prize: Prize,
        control: Optional[Callable[[int], str]] = None,
        on_draw: Optional[Callable[[int, Optional[User]], None]] = None,
    ) -> List[User]:
        """Run a full round. control(i) returns 'draw' or 'terminate'."""
        winners: List[User] = []
        for i in range(1, prize.quantity + 1):
            if control is not None and control(i) == "terminate":
                break
            user = self.draw_one(prize)
            if on_draw:
                on_draw(i, user)
            if user is None:
                break
            winners.append(user)
        self.record_round_result(prize, winners)
        return winners

"""Result management and export."""

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List


class Winner:
    def __init__(self, user, prize, order: int):
        self.user = user
        self.prize = prize
        self.win_order = order
        self.timestamp = datetime.now()

    def __repr__(self) -> str:
        return f"Winner({self.user.name!r}, {self.prize.name!r}, #{self.win_order})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self.export_format = "csv"
        self.export_path = None

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self.winners.append(Winner(user, prize, len(self.winners) + 1))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes:
            result.setdefault(p.name, [])
        for w in self.winners:
            result.setdefault(w.prize.name, []).append(w)
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for prize_name, winners in grouped.items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, prize_name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        self.export_path = path
        return True, f"Results successfully exported to {path}"

"""Command-line interface."""

from __future__ import annotations

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryCLI:
    def __init__(self, input_func=input, output_func=print):
        self.input = input_func
        self.print = output_func
        self.event_name = "Company Team Building Event"
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery_done = False

    def ask(self, prompt: str) -> str:
        try:
            return self.input(prompt).strip()
        except EOFError:
            return "0"

    # ------------------------------------------------------------- list
    def _accept_list(self, users, errors):
        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            self.print("Validation issues detected:")
            for e in errors:
                self.print("  " + e)
            for d in duplicates:
                self.print(f"  Duplicate Name: {d.name} ({d.department})")
            choice = self.ask("Type 'r' to remove invalid data and continue, 't' to terminate upload: ").lower()
            if choice != "r":
                self.print("Upload terminated.")
                return
        if not valid:
            self.print("No valid participants loaded.")
            return
        self.users = valid
        self.print(f"Loaded {len(valid)} valid participants.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._accept_list(users, errors)

    def load_custom(self):
        path = self.ask("Enter txt file path: ")
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors and errors[0].startswith(("File not found", "Unable")):
            self.print(errors[0])
            return
        self._accept_list(users, errors)

    def view_groups(self):
        if not self.users:
            self.print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self.print("Department groups:")
        for g, c in info.items():
            self.print(f"  {g}: {c} people")

    # ------------------------------------------------------------- prizes
    def load_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            self.print(e)
        self.print("Preset prizes loaded.")

    def add_prize(self):
        name = self.ask("Prize name: ")
        qty = self.ask("Quantity: ")
        groups_info = list(self.list_manager.get_group_info(self.users).keys())
        if groups_info:
            self.print("Available groups: " + ", ".join(f"{i}.{g}" for i, g in enumerate(groups_info, 1)))
        raw = self.ask("Avoidance groups (numbers or names, comma separated, blank for none): ")
        groups = []
        for item in raw.replace("，", ",").split(","):
            item = item.strip()
            if not item:
                continue
            if item.isdigit() and 1 <= int(item) <= len(groups_info):
                groups.append(groups_info[int(item) - 1])
            else:
                groups.append(item)
        drawer = self.ask("Associated lottery person (blank for none): ")
        try:
            p = self.prize_manager.add_prize(name, qty, groups, drawer or None)
            self.print(f"Prize added: {p.name} x{p.quantity}")
        except ValueError as exc:
            self.print(f"Error: {exc}")

    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            self.print(f"  {i}. {p.name} x{p.quantity} | {p.rule_description}")

    def reorder(self):
        self.view_prizes()
        raw = self.ask("Enter new order as serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            self.print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.print("Prize order updated.")
            self.view_prizes()
        else:
            self.print("Invalid order.")

    # ------------------------------------------------------------- lottery
    def start_lottery(self):
        if not self.users:
            self.print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.print("Please configure prizes first.")
            return
        manager = LotteryManager(self.users)
        for w in self.result_manager.winners:
            manager.winner_names.add(w.user.name)
        for rnd, prize in enumerate(prizes, 1):
            self.print("=" * 50)
            self.print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            self.print(f"Rules: {manager.describe_rules(prize)}")
            state = {"terminated": False}

            def control(i, state=state):
                while True:
                    cmd = self.ask(f"[Enter]=draw {ordinal(i)} winner, 'pause', 'terminate': ").lower()
                    if cmd == "pause":
                        self.print("Lottery paused. Type 'continue' to resume.")
                        while True:
                            c = self.ask("> ").lower()
                            if c == "continue":
                                self.print("Lottery resumed.")
                                break
                            if c in ("terminate", "0"):
                                state["terminated"] = True
                                return "terminate"
                        continue
                    if cmd in ("terminate", "0"):
                        state["terminated"] = True
                        return "terminate"
                    return "draw"

            def on_draw(i, user, prize=prize):
                self.print(f"Currently drawing the {ordinal(i)} winner...")
                if user is None:
                    self.print("No eligible participants remain for this prize.")
                    return
                if manager.fallback_applied:
                    self.print("(Fallback rule applied: Cheng Ru Group protection)")
                self.result_manager.add_winner(user, prize)
                self.print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")

            winners = manager.run_round(prize, control, on_draw)
            if state["terminated"]:
                self.print("Current round terminated and skipped.")
            self.print(f"Round {rnd} results:")
            if not winners:
                self.print("  (no winners)")
            for i, w in enumerate(winners, 1):
                self.print(f"  {i}. {w.name} ({w.department})")
        self.lottery_done = True
        self.show_results()

    def show_results(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        self.print("=" * 50)
        self.print("Complete winner list:")
        if not self.result_manager.winners:
            self.print("  (no winners yet)")
            return
        for prize_name, winners in grouped.items():
            for w in winners:
                self.print(f"  {w.user.name}, {w.user.department}, {prize_name}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self.print(msg)

    # ------------------------------------------------------------- main
    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_prizes, "5": self.add_prize, "6": self.view_prizes,
            "7": self.reorder, "8": self.start_lottery, "9": self.show_results,
            "10": self.export,
        }
        self.print(f"Welcome to the {self.event_name} Smart Lottery System")
        while True:
            self.print(
                "\n1. Load default list\n2. Upload custom list\n3. View groups\n"
                "4. Load preset prizes\n5. Add custom prize\n6. View prizes\n"
                "7. Reorder prizes\n8. Start lottery\n9. View winner list\n"
                "10. Export results\n0. Exit"
            )
            choice = self.ask("Select: ")
            if choice == "0":
                self.print("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                self.print("Invalid option.")


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

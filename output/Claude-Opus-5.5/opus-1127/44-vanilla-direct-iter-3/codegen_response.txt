"""Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
Its layout is::

    {
      "stations":    [{"name": str, "lines": [str, ...]}, ...],
      "connections": [{"source": str, "dest": str,
                       "distance": int (meters), "line": str}, ...]
    }
"""

import json
import os

from models import Edge, Graph

# Absolute path, so it is valid regardless of the current working directory.
DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON file (defaults to ``DATA_FILE_PATH``).
    :raises FileNotFoundError: if the file does not exist.
    :raises json.JSONDecodeError: if the file is not valid JSON.
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        return json.load(fh)


def build_graph(data):
    """Build a :class:`Graph` from loaded subway data.

    Station numbers are assigned in the order stations appear in the data.
    Stations that are only referenced by connections are appended afterwards.
    The returned graph exposes ``station_to_number`` and
    ``number_to_station`` dictionaries.
    """
    graph = Graph()
    stations = data.get("stations", []) if isinstance(data, dict) else []
    connections = data.get("connections", []) if isinstance(data, dict) else []

    for record in stations:
        name = record.get("name")
        if not name or name in graph.station_to_number:
            if name in graph.station_to_number:
                # merge lines for duplicated records
                merged = list(graph.station_lines.get(name, []))
                for line in record.get("lines", []) or []:
                    if line not in merged:
                        merged.append(line)
                graph.set_station_lines(name, merged)
            continue
        graph.add_station(name, len(graph.station_to_number), record.get("lines", []) or [])

    for conn in connections:
        src = conn.get("source")
        dst = conn.get("dest")
        if not src or not dst:
            continue
        line = conn.get("line")
        for name in (src, dst):
            if name not in graph.station_to_number:
                graph.add_station(name, len(graph.station_to_number), [line] if line else [])
        distance = conn.get("distance", 0)
        try:
            edge = Edge(graph.station_to_number[src], graph.station_to_number[dst],
                        distance, line)
        except (TypeError, ValueError):
            continue
        graph.add_edge(edge)
    return graph

"""Core data structures: Edge, Graph (weighted undirected) and MinPQ."""

import heapq
import itertools
import math


class Edge:
    """A weighted edge between two stations (identified by numbers)."""

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("Edge distance must be a number")
        if math.isnan(distance) or distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        return self.source

    def other(self, vertex):
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError("Vertex is not an endpoint of this edge")

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r -> %r, %r m, %r)" % (self.source, self.dest, self.weight, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists.

    ``graph`` maps a station number to the list of outgoing :class:`Edge`
    objects. Every undirected connection is stored in both directions.
    Station numbers are assigned by the caller.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self._edges = []

    # ---- stations -------------------------------------------------------
    def add_station(self, station_name, number, lines=None):
        self.station_to_number[station_name] = number
        self.number_to_station[number] = station_name
        self.graph.setdefault(number, [])
        if lines is not None:
            self.set_station_lines(station_name, lines)

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines or [])

    def add_vertex(self, v):
        self.graph.setdefault(v, [])

    def is_transfer_station(self, station_name):
        return len(set(self.station_lines.get(station_name, []))) > 1

    def transfer_stations(self):
        return [n for n in self.station_lines if self.is_transfer_station(n)]

    def get_station_name(self, number):
        return self.number_to_station.get(number, str(number))

    def get_station_number(self, name):
        return self.station_to_number.get(name)

    # ---- edges ----------------------------------------------------------
    def add_edge(self, edge):
        self.graph.setdefault(edge.source, []).append(edge)
        if edge.source != edge.dest:
            self.graph.setdefault(edge.dest, []).append(edge.reversed())
        self._edges.append(edge)

    def adj(self, v):
        return list(self.graph.get(v, []))

    def edges(self):
        return list(self._edges)

    def vertices(self):
        return list(self.graph.keys())

    @property
    def V(self):
        return len(self.graph)

    @property
    def E(self):
        return len(self._edges)

    def __contains__(self, v):
        return v in self.graph


class MinPQ:
    """Heap-based minimum priority queue (stable for equal priorities)."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def del_min(self):
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def min(self):
        if not self._heap:
            raise IndexError("min from empty priority queue")
        priority, _, item = self._heap[0]
        return priority, item

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

"""Dijkstra-based path finding with transfer awareness.

The search runs over states ``(station, current line, transfers)`` so that
transfers at multi-line interchange stations are counted correctly and
transfer limits / line preferences can be honoured.
"""

from models import MinPQ

INF = float("inf")

MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Equivalent distance cost (meters) of one transfer in comprehensive mode
# (about 5 minutes of walking/waiting ~ 3.3 km of riding at 1.5 min/km).
TRANSFER_PENALTY_METERS = 3000
# Weight multiplier applied to edges on preferred lines.
PREFERENCE_FACTOR = 0.7


class PathfindingConfig:
    """Configuration of a path query."""

    def __init__(self, mode="shortest_distance", max_transfers=None,
                 preferred_lines=None, transfer_penalty=TRANSFER_PENALTY_METERS,
                 preference_factor=PREFERENCE_FACTOR):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def validate(self):
        if self.mode not in MODES:
            raise ValueError("Unknown mode '%s', expected one of %s" % (self.mode, ", ".join(MODES)))
        if self.max_transfers is not None:
            if isinstance(self.max_transfers, bool) or not isinstance(self.max_transfers, int):
                raise ValueError("max_transfers must be an integer")
            if self.max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        if self.preferred_lines is None:
            self.preferred_lines = []


class ShortestPath:
    """Single-source optimal paths from ``source`` (a station number)."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.config.validate()
        self._preferred = set(self.config.preferred_lines or [])
        self.dist_to = {v: INF for v in graph.vertices()}
        self.dist_to[source] = 0
        self._info = {}      # state -> (cost, distance, transfers, adjusted)
        self._parent = {}    # state -> (prev_state, edge)
        self._best = {}      # station -> best state
        self._run()

    # ------------------------------------------------------------------
    def _factor(self, line):
        if self._preferred and line in self._preferred:
            return self.config.preference_factor
        return 1.0

    def _cost(self, adjusted, transfers, distance):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, adjusted, distance)
        if mode == "comprehensive":
            return (adjusted + transfers * self.config.transfer_penalty, transfers, distance)
        return (adjusted, transfers, distance)

    def _run(self):
        max_t = self.config.max_transfers
        pq = MinPQ()
        start = (self.source, None, 0)
        start_cost = self._cost(0, 0, 0)
        self._info[start] = (start_cost, 0, 0, 0)
        pq.insert(start_cost, start)
        done = set()
        while not pq.is_empty():
            cost, state = pq.del_min()
            if state in done:
                continue
            info = self._info[state]
            if cost > info[0]:
                continue
            done.add(state)
            v, line, _ = state
            _, d, t, a = info
            for e in self.graph.graph.get(v, []):
                nt = t + (1 if line is not None and e.line != line else 0)
                if max_t is not None and nt > max_t:
                    continue
                nd = d + e.weight
                na = a + e.weight * self._factor(e.line)
                ncost = self._cost(na, nt, nd)
                ns = (e.dest, e.line, nt if max_t is not None else 0)
                if ns in done:
                    continue
                old = self._info.get(ns)
                if old is None or ncost < old[0]:
                    self._info[ns] = (ncost, nd, nt, na)
                    self._parent[ns] = (state, e)
                    pq.insert(ncost, ns)

        for state, info in self._info.items():
            v = state[0]
            cur = self._best.get(v)
            if cur is None or info[0] < self._info[cur][0]:
                self._best[v] = state
        for v, state in self._best.items():
            self.dist_to[v] = self._info[state][1]

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target."""
        if target not in self._best:
            return [], []
        edges = []
        state = self._best[target]
        while state in self._parent:
            prev, edge = self._parent[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._best:
            return -1
        return self._info[self._best[target]][2]

    def get_transfer_stations(self, target):
        _, edges = self.path_with_edges(target)
        result = []
        for prev, nxt in zip(edges, edges[1:]):
            if prev.line != nxt.line:
                result.append(prev.dest)
        return result

    def get_lines(self, target):
        _, edges = self.path_with_edges(target)
        lines = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)
        return lines

"""Beijing Subway fare structure (distance based, left-closed intervals)."""

import math

# (lower bound inclusive, upper bound exclusive, fare) in meters / yuan
FARE_TIERS = [
    (0, 6000, 3),
    (6000, 12000, 4),
    (12000, 22000, 5),
    (22000, 32000, 6),
]
EXTENDED_START = 32000
EXTENDED_STEP = 20000
EXTENDED_BASE_FARE = 7

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5


def _check_number(value, name):
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise TypeError("%s must be a number" % name)
    if math.isnan(value) or math.isinf(value):
        raise ValueError("%s must be a finite number" % name)
    if value < 0:
        raise ValueError("%s must be non-negative" % name)


def Sum_money(distance):
    """Return the fare (yuan) for a travel distance given in meters.

    Fractional distances are rounded up to the next whole meter; tier
    boundaries belong to the higher tier (6000 m -> 4 yuan, ...).
    """
    _check_number(distance, "distance")
    d = math.ceil(distance)
    for low, high, fare in FARE_TIERS:
        if low <= d < high:
            return fare
    return EXTENDED_BASE_FARE + (d - EXTENDED_START) // EXTENDED_STEP


calculate_fare = Sum_money


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes (1.5 min/km + 5 min per transfer)."""
    _check_number(distance, "distance")
    _check_number(transfer_count, "transfer_count")
    return float(distance / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER)

import json

MODE_LABELS = {
    "shortest_distance": "Shortest distance",
    "least_transfer": "Least transfers",

def format_distance(meters):
    """Meters for short distances, kilometers for long ones (both shown)."""
    km = meters / 1000.0
    m = int(round(meters))
    if meters < 1000:
        return "%d meters (%.2f km)" % (m, km)
    return "%.2f km (%d meters)" % (km, m)


def build_segments(edges, name_of):
    segments = []
    for e in edges:
        if not segments or segments[-1]["line"] != e.line:
            segments.append({"line": e.line, "stations": [name_of(e.source)], "distance": 0})
        segments[-1]["stations"].append(name_of(e.dest))
        segments[-1]["distance"] += e.weight
    return segments


def build_route_info(start, end, mode, station_numbers, edges, distance, fare,
                     estimated_time, transfer_count, graph, config=None):
    name_of = graph.get_station_name
    stations = [name_of(n) for n in station_numbers]
    segments = build_segments(edges, name_of)
    transfers = []
    for a, b in zip(segments, segments[1:]):
        transfers.append({"station": b["stations"][0], "from_line": a["line"], "to_line": b["line"]})
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    return {
        "query": {
            "start": start,
            "end": end,
            "mode": mode,
            "max_transfers": getattr(config, "max_transfers", None),
            "line_preference": list(getattr(config, "preferred_lines", []) or []),
        },
        "path": stations,
        "station_count": len(stations),
        "distance": {"meters": distance, "kilometers": round(distance / 1000.0, 3)},
        "fare": fare,
        "estimated_time_minutes": round(estimated_time, 1),
        "transfer_count": transfer_count,
        "transfers": transfers,
        "lines": lines,
        "segments": segments,
    }


def format_text(info):
    q = info["query"]
    out = []
    out.append("=" * 60)
    out.append("Route: %s -> %s  [%s]" % (q["start"], q["end"], MODE_LABELS.get(q["mode"], q["mode"])))
    out.append("-" * 60)
    out.append("Stations (%d): %s" % (info["station_count"], " -> ".join(info["path"])))
    out.append("Line segments:")
    for i, seg in enumerate(info["segments"]):
        out.append("  [%s] %s  (%d stops, %s)" % (
            seg["line"], " -> ".join(seg["stations"]), len(seg["stations"]) - 1,
            format_distance(seg["distance"])))
        if i < len(info["transfers"]):
            t = info["transfers"][i]
            out.append("  >> Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    out.append("Total distance: %s" % format_distance(info["distance"]["meters"]))
    if info["transfers"]:
        out.append("Transfers: %d (at %s)" % (
            info["transfer_count"], ", ".join(t["station"] for t in info["transfers"])))
    else:
        out.append("Transfers: 0")
    out.append("Lines involved: %s" % ", ".join(str(l) for l in info["lines"]))
    out.append("Estimated time: %.1f minutes" % info["estimated_time_minutes"])
    out.append("Fare: %d yuan" % info["fare"])
    out.append("=" * 60)
    return "\n".join(out)

class SubwayDataManager:
    """Manage stations, connections and lines of the subway network."""

    def __init__(self, data=None, path=DATA_FILE_PATH):
        self.path = path
        if data is None:
            data = load_subway_data(path)
        self.data = data
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    @property
    def stations(self):
        return self.data["stations"]

    @property
    def connections(self):
        return self.data["connections"]

    # ---- queries --------------------------------------------------------
    def get_station(self, name):
        for s in self.stations:
            if s.get("name") == name:
                return s
        return None

    def station_names(self):
        return [s.get("name") for s in self.stations]

    def get_lines(self):
        """Map line name -> list of station names (sequence order when possible)."""
        names = []
        for s in self.stations:
            for line in s.get("lines", []) or []:
                if line not in names:
                    names.append(line)
        for c in self.connections:
            if c.get("line") not in names and c.get("line") is not None:
                names.append(c.get("line"))
        return {line: self.get_line_sequence(line) for line in names}

    def get_line_sequence(self, line):
        adj = {}
        order = []
        for c in self.connections:
            if c.get("line") != line:
                continue
            a, b = c.get("source"), c.get("dest")
            for x in (a, b):
                if x not in adj:
                    adj[x] = []
                    order.append(x)
            if b not in adj[a]:
                adj[a].append(b)
            if a not in adj[b]:
                adj[b].append(a)
        if not adj:
            return [s["name"] for s in self.stations if line in (s.get("lines") or [])]
        ends = [x for x in order if len(adj[x]) == 1]
        start = ends[0] if ends else order[0]
        seq, seen, stack = [], set(), [start]
        while stack:
            v = stack.pop()
            if v in seen:
                continue
            seen.add(v)
            seq.append(v)
            for w in reversed(adj[v]):
                if w not in seen:
                    stack.append(w)
        for x in order:
            if x not in seen:
                seq.append(x)
        return seq

    def build_graph(self):
        return build_graph(self.data)

    # ---- updates --------------------------------------------------------
    def add_station(self, name, lines):
        if not name:
            raise ValueError("Station name cannot be empty")
        st = self.get_station(name)
        if st is None:
            self.stations.append({"name": name, "lines": list(lines)})
        else:
            for line in lines:
                if line not in st.setdefault("lines", []):
                    st["lines"].append(line)

    def add_line(self, line_name, station_names, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if not line_name:
            raise ValueError("Line name cannot be empty")
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must be number of stations minus one")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for name in station_names:
            self.add_station(name, [line_name])
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.connections.append({"source": a, "dest": b, "distance": int(d), "line": line_name})

    def modify_station(self, name, new_name=None, lines=None):
        st = self.get_station(name)
        if st is None:
            raise KeyError("Station '%s' does not exist" % name)
        if new_name and new_name != name:
            if self.get_station(new_name) is not None:
                raise ValueError("Station '%s' already exists" % new_name)
            st["name"] = new_name
            for c in self.connections:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            st["lines"] = list(lines)

    def update_distance(self, station_a, station_b, distance, line=None):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        count = 0
        for c in self.connections:
            pair = {c.get("source"), c.get("dest")}
            if pair == {station_a, station_b} and (line is None or c.get("line") == line):
                c["distance"] = int(distance)
                count += 1
        if count == 0:
            raise KeyError("No connection between '%s' and '%s'" % (station_a, station_b))
        return count

    def save(self, path=None):
        with open(path or self.path, "w", encoding="utf-8") as fh:
            json.dump(self.data, fh, ensure_ascii=False, indent=2)

    # ---- integrity ------------------------------------------------------
    def check_integrity(self):
        rep = IntegrityReport()
        names = {}
        for i, s in enumerate(self.stations):
            name = s.get("name")
            if not name:
                rep.errors.append("Station record #%d has no name" % i)
                continue
            if name in names:
                rep.warnings.append("Duplicate station record: '%s'" % name)
            names[name] = s.get("lines", []) or []
            if not names[name]:
                rep.warnings.append("Station '%s' has no line information" % name)
        rep.info.append("Stations: %d, connections: %d" % (len(names), len(self.connections)))

        # connections
        seen_pairs = set()
        adjacency = {n: set() for n in names}
        for i, c in enumerate(self.connections):
            a, b, d, line = c.get("source"), c.get("dest"), c.get("distance"), c.get("line")
            label = "Connection #%d (%s - %s, %s)" % (i, a, b, line)
            if a not in names or b not in names:
                rep.errors.append("%s references an unknown station" % label)
                continue
            if a == b:
                rep.errors.append("%s is a self-loop" % label)
                continue
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                rep.errors.append("%s has invalid distance %r" % (label, d))
            elif d < MIN_REASONABLE_DISTANCE or d > MAX_REASONABLE_DISTANCE:
                rep.warnings.append("%s has suspicious distance %r m" % (label, d))
            key = (frozenset((a, b)), line)
            if key in seen_pairs:
                rep.warnings.append("%s is duplicated" % label)
            seen_pairs.add(key)
            for x in (a, b):
                if line is not None and names.get(x) and line not in names[x]:
                    rep.warnings.append("%s: station '%s' is not listed on line '%s'" % (label, x, line))
            adjacency[a].add(b)
            adjacency[b].add(a)

        isolated = [n for n, nb in adjacency.items() if not nb]
        for n in isolated:
            rep.errors.append("Station '%s' has no connections" % n)

        # connectivity
        if names:
            start = next(iter(names))
            seen = {start}
            dq = deque([start])
            while dq:
                v = dq.popleft()
                for w in adjacency.get(v, ()):
                    if w not in seen:
                        seen.add(w)
                        dq.append(w)
            if len(seen) < len(names):
                rep.warnings.append("Network is not fully connected: %d of %d stations reachable from '%s'"
                                    % (len(seen), len(names), start))
            else:
                rep.info.append("Network is fully connected")

        # line coverage
        lines = self.get_lines()
        norm_lines = {_norm_line(l): l for l in lines}
        for main in MAIN_LINES:
            actual = norm_lines.get(_norm_line(main))
            if actual is None:
                rep.warnings.append("Main line '%s' not found in data" % main)
                continue
            seq = lines[actual]
            total = sum(c.get("distance", 0) or 0 for c in self.connections
                        if c.get("line") == actual and isinstance(c.get("distance"), (int, float)))
            rep.info.append("%s: %d stations, %.2f km" % (actual, len(seq), total / 1000.0))
            reps = REPRESENTATIVE_STATIONS.get(main, [])
            found = [r for r in reps if r in names]
            if reps and not found:
                rep.warnings.append("No representative station found for '%s'" % main)
        return rep

"""High-level service combining station resolution, routing and fares."""

import difflib

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath
from report import build_route_info

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class InputValidationError(Exception):
    pass


class PathCalculationError(Exception):
    pass


class FareCalculationError(Exception):
    pass


def _normalize(name):
    s = str(name).strip().casefold()
    for suffix in (" station", "站"):
        if s.endswith(suffix) and len(s) > len(suffix):
            s = s[: -len(suffix)]
    for ch in " \t'’`-_·":
        s = s.replace(ch, "")
    return s


class SubwayFareSystem:
    def __init__(self, graph, config=None):
        self.graph = graph
        self.config = config or PathfindingConfig()
        self._norm_map = {}
        for name in graph.station_to_number:
            self._norm_map.setdefault(_normalize(name), name)

    # ---- input ----------------------------------------------------------
    def resolve_station(self, name):
        name = " ".join(str(name).split())
        if name in self.graph.station_to_number:
            return name
        return self._norm_map.get(_normalize(name))

    def suggest(self, name, n=3):
        return difflib.get_close_matches(name, list(self.graph.station_to_number), n=n, cutoff=0.6)

    def resolve_lines(self, lines):
        known = set()
        for ls in self.graph.station_lines.values():
            known.update(ls)
        for e in self.graph.edges():
            known.add(e.line)
        lookup = {"".join(str(l).lower().split()): l for l in known if l is not None}
        resolved, unknown = [], []
        for l in lines:
            r = lookup.get("".join(l.lower().split()))
            if r is None:
                unknown.append(l)
            elif r not in resolved:
                resolved.append(r)
        return resolved, unknown

    def parse_query(self, raw):
        text = (raw or "").replace("；", ";").strip()
        parts = text.split(";")
        if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
            raise InputValidationError(FORMAT_ERROR)
        start_raw, end_raw = parts[0].strip(), parts[1].strip()
        start = self.resolve_station(start_raw)
        if start is None:
            raise InputValidationError(self._missing_msg("Starting", start_raw))
        end = self.resolve_station(end_raw)
        if end is None:
            raise InputValidationError(self._missing_msg("Ending", end_raw))
        if start == end:
            raise InputValidationError("Starting station cannot be the same as the ending station")
        return start, end

    def _missing_msg(self, kind, name):
        msg = "%s station '%s' does not exist, please check the station name" % (kind, name)
        sugg = self.suggest(name)
        if sugg:
            msg += " (did you mean: %s?)" % ", ".join(sugg)
        return msg

    # ---- routing --------------------------------------------------------
    def find_route(self, start, end):
        s = self.graph.station_to_number[start]
        t = self.graph.station_to_number[end]
        try:
            sp = ShortestPath(self.graph, s, self.config)
        except Exception as exc:
            raise PathCalculationError("Path calculation failed: %s" % exc) from exc
        if not sp.has_path_to(t):
            msg = "No route found from '%s' to '%s'" % (start, end)
            if self.config.max_transfers is not None:
                msg += " within a maximum of %d transfer(s)" % self.config.max_transfers
            raise PathCalculationError(msg)
        try:
            stations, edges = sp.path_with_edges(t)
            distance = sp.dist_to[t]
            transfers = sp.get_transfer_count(t)
        except Exception as exc:
            raise PathCalculationError("Path calculation failed: %s" % exc) from exc
        try:
            fare = Sum_money(distance)
            est = calculate_estimated_time(distance, transfers)
        except (TypeError, ValueError) as exc:
            raise FareCalculationError("Fare calculation failed: %s" % exc) from exc
        return build_route_info(start, end, self.config.mode, stations, edges, distance,
                                fare, est, transfers, self.graph, self.config)

    def query(self, raw):
        start, end = self.parse_query(raw)
        return self.find_route(start, end)

def non_negative_int(value):
    try:
        n = int(value)
    except ValueError:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    if n < 0:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    return n


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", help="export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text", help="output format")
    p.add_argument("--mode", choices=list(MODES), default="shortest_distance", help="path query mode")
    p.add_argument("--max_transfers", type=non_negative_int, default=None, help="maximum transfers")
    p.add_argument("--line_preference", default=None,

class ResultWriter:
    def __init__(self, path, fmt):
        self.path = path
        self.fmt = fmt
        self.results = []
        if path:
            # start with a fresh file
            with open(path, "w", encoding="utf-8") as fh:
                fh.write("[]\n" if fmt == "json" else "")

    def write(self, info, text):
        if not self.path:
            return
        if self.fmt == "json":
            self.results.append(info)
            with open(self.path, "w", encoding="utf-8") as fh:
                json.dump(self.results, fh, ensure_ascii=False, indent=2)
        else:
            with open(self.path, "a", encoding="utf-8") as fh:
                fh.write(text + "\n\n")


def run_check(path):
    try:
        manager = SubwayDataManager(path=path)
    except (OSError, ValueError) as exc:
        print("Failed to load subway data: %s" % exc)
        return 1
    report = manager.check_integrity()
    print(report.format())
    return 0 if report.ok else 1


def main(argv=None):
    args = parse_args(argv)
    if args.check_data:
        return run_check(args.data_file)

    try:
        graph = build_graph(load_subway_data(args.data_file))
    except (OSError, ValueError) as exc:
        print("Failed to load subway data: %s" % exc)
        return 1

    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers)
    system = SubwayFareSystem(graph, config)
    if args.line_preference:
        wanted = [x.strip() for x in args.line_preference.replace("，", ",").split(",") if x.strip()]
        resolved, unknown = system.resolve_lines(wanted)
        for u in unknown:
            print("Warning: line '%s' does not exist and will be ignored" % u)
        config.preferred_lines = resolved

    try:
        writer = ResultWriter(args.output_file, args.format)
    except OSError as exc:
        print("Cannot open output file: %s" % exc)
        return 1

    print("Beijing Subway Fare Calculation System (%d stations loaded)" % len(graph.station_to_number))
    while True:
        try:
            raw = input(PROMPT)
        except EOFError:
            print()
            break
        except KeyboardInterrupt:
            print()
            break
        text = raw.strip()
        if not text:
            continue
        if text.lower() in ("quit", "exit", "q"):
            print("Goodbye!")
            break
        try:
            info = system.query(text)
        except InputValidationError as exc:
            print(exc)
            continue
        except PathCalculationError as exc:
            print("Path calculation error: %s" % exc)
            continue
        except FareCalculationError as exc:
            print("Fare calculation error: %s" % exc)
            continue
        except Exception as exc:  # last-resort safety net
            print("Unexpected error: %s" % exc)
            continue
        output = format_json(info) if args.format == "json" else format_text(info)
        print(output)
        try:
            writer.write(info, output)
        except OSError as exc:
            print("Failed to write output file: %s" % exc)
    return 0


if __name__ == "__main__":
    sys.exit(main())

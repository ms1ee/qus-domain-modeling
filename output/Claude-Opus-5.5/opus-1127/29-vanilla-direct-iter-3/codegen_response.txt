"""Grid-based path planning for riders.

The map is an 800x800 coordinate system divided into an 8x8 grid of
100x100 cells.  Riders move only horizontally or vertically (right-angle
turns).  The planner produces axis-aligned waypoint lists.  Because every
plan is computed independently and riders never block each other, the
planner is free of path conflicts and deadlocks.
"""

import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_SIZE = MAP_SIZE // CELL_SIZE


class PathPlanner:
    """Collection of path-finding helpers (all static)."""

    @staticmethod
    def clamp(value):
        """Clamp a coordinate into the valid map range 0..800."""
        return max(0, min(MAP_SIZE, int(value)))

    @staticmethod
    def to_grid_center(x, y):
        """Convert raw coordinates into the centre of their grid cell.

        ``center = (v // 100) * 100 + 50``.  The boundary value 800 belongs
        to the last cell, so it maps to 750.
        """
        def conv(v):
            v = max(0, min(MAP_SIZE - 1, int(v)))
            return (v // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2
        return conv(x), conv(y)

    @staticmethod
    def grid_cell(x, y):
        """Return the (column, row) grid cell index of a coordinate."""
        col = min(GRID_SIZE - 1, max(0, int(x) // CELL_SIZE))
        row = min(GRID_SIZE - 1, max(0, int(y) // CELL_SIZE))
        return col, row

    @staticmethod
    def is_street(x, y):
        """Return True if the point is a road point.

        Points where both coordinates are integer hundreds are
        restaurant/customer locations; everything else is road.
        """
        return not (int(x) % CELL_SIZE == 0 and int(y) % CELL_SIZE == 0)

    @staticmethod
    def is_special_case(start, target):
        """Return True when the rider is directly above or below the target.

        That is, both points lie in the same grid column but are not on
        the same horizontal line.
        """
        sx, sy = start
        tx, ty = target
        if sy == ty:
            return False
        return PathPlanner.grid_cell(sx, sy)[0] == PathPlanner.grid_cell(tx, ty)[0]

    @staticmethod
    def plan_path(start, target):
        """Plan an axis-aligned path from ``start`` to ``target``.

        Returns a list of waypoints (excluding ``start``) ending in
        ``target``.  Returns an empty list when already at the target.

        * Aligned points: straight line.
        * Directly above/below (same grid column): vertical first, then a
          short horizontal adjustment.
        * Otherwise: horizontal along the street first, then vertical.
        """
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        if (sx, sy) == (tx, ty):
            return []
        if sx == tx or sy == ty:
            return [(tx, ty)]
        if PathPlanner.is_special_case((sx, sy), (tx, ty)):
            return [(sx, ty), (tx, ty)]
        return [(tx, sy), (tx, ty)]

    @staticmethod
    def plan_route(start, destinations):
        """Plan a continuous multi-destination route.

        Returns the full polyline including ``start``.
        """
        points = [(int(start[0]), int(start[1]))]
        current = points[0]
        for dest in destinations:
            for wp in PathPlanner.plan_path(current, dest):
                points.append(wp)
            current = (int(dest[0]), int(dest[1]))
        return points

    @staticmethod
    def path_length(points):
        """Total Manhattan length of a polyline."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    @staticmethod
    def euclidean(a, b):
        """Euclidean distance between two points."""
        return math.sqrt((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2)

"""Order data model."""

import globalvar as gv
from path_planner import PathPlanner


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique order id.
        res_pos: (x, y) restaurant position.
        cus_pos: (x, y) customer position.
        status: "pending", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: id of the assigned rider or None.
        start_time, assigned_time, pickup_time, complete_time: tick times.
        distance: Manhattan route length restaurant -> customer.
    """

    def __init__(self, res_pos, cus_pos):
        counter = gv.get_value("order_id_counter", 0) or 0
        counter += 1
        gv.set_value("order_id_counter", counter)
        self.id = counter
        self.res_pos = (int(res_pos[0]), int(res_pos[1]))
        self.cus_pos = (int(cus_pos[0]), int(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = gv.get_value("time", 0)
        self.assigned_time = None
        self.pickup_time = None
        self.complete_time = None
        self.assign_distance = None
        self.distance = PathPlanner.path_length(
            PathPlanner.plan_route(self.res_pos, [self.cus_pos]))

    def assign_to(self, rider_id):
        """Assign this order to the rider with ``rider_id``."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        self.assigned_time = gv.get_value("time", 0)

    def mark_picking_up(self):
        """Rider accepted the order and heads to the restaurant."""
        self.status = "picking_up"

    def mark_picked_up(self):
        """Rider arrived at the restaurant."""
        self.status = "delivering"
        self.pickup_time = gv.get_value("time", 0)

    def mark_completed(self):
        """Rider delivered the order."""
        self.status = "completed"
        self.complete_time = gv.get_value("time", 0)

    @property
    def time_taken(self):
        """Ticks from creation to completion, or None if unfinished."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.start_time

    def as_list(self):
        """Return ``[restaurant_x, restaurant_y, customer_x, customer_y]``."""
        return [self.res_pos[0], self.res_pos[1], self.cus_pos[0], self.cus_pos[1]]

    def to_dict(self):
        """Serialise to a JSON-friendly dict."""
        return {
            "id": self.id, "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos), "status": self.status,
            "rider_id": self.rider_id, "start_time": self.start_time,
            "assigned_time": self.assigned_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "assign_distance": self.assign_distance,
            "distance": self.distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from :meth:`to_dict` output."""
        obj = cls.__new__(cls)
        obj.id = int(data["id"])
        obj.res_pos = tuple(data["res_pos"])
        obj.cus_pos = tuple(data["cus_pos"])
        obj.status = data.get("status", "pending")
        obj.rider_id = data.get("rider_id")
        obj.start_time = data.get("start_time", 0)
        obj.assigned_time = data.get("assigned_time")
        obj.pickup_time = data.get("pickup_time")
        obj.complete_time = data.get("complete_time")
        obj.assign_distance = data.get("assign_distance")
        obj.distance = data.get("distance", 0)
        return obj

    def __repr__(self):
        return "Order(id=%d, res=%s, cus=%s, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

"""Rider model: movement, state machine and order handling."""

import globalvar as gv
from path_planner import PathPlanner

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2
STATE_NAMES = {STATE_IDLE: "idle", STATE_PICKING_UP: "picking up",
               STATE_DELIVERING: "delivering"}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``; the rider
    picks them up in FIFO order during :meth:`move`.
    """

    def __init__(self, rider_id):
        cfg = gv.get_value("config") or {}
        self.rider_id = rider_id
        self.x = int(cfg.get("spawn_x", 435))
        self.y = int(cfg.get("spawn_y", 385))
        self.speed = int(cfg.get("rider_speed", 10))
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.path = []
        self.log = []
        self.completed_orders = []
        self.distance_travelled = 0
        self._record("spawn", None)

    # ------------------------------------------------------------------
    def _record(self, event, order_id):
        """Append a status-change entry to the rider log."""
        self.log.append({
            "event": event, "order_id": order_id,
            "time": gv.get_value("time", 0), "x": self.x, "y": self.y,
            "state": self.state,
        })

    def _accept_next_order(self):
        """Take the next unfinished order if idle (state 0 -> 1)."""
        if self.current_order is not None:
            return
        for order in self.orders:
            if getattr(order, "status", None) == "completed":
                continue
            self.current_order = order
            if getattr(order, "rider_id", None) != self.rider_id:
                order.assign_to(self.rider_id)
            if hasattr(order, "mark_picking_up"):
                order.mark_picking_up()
            self.state = STATE_PICKING_UP
            self.path = []
            self._record("accepted_order", order.id)
            return

    def current_target(self):
        """Return the current destination or None."""
        if self.current_order is None:
            return None
        if self.state == STATE_PICKING_UP:
            return tuple(self.current_order.res_pos)
        return tuple(self.current_order.cus_pos)

    def _arrive(self):
        """Handle arrival at the current target."""
        order = self.current_order
        self.path = []
        self.special_case = False
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.mark_picked_up()
            self._record("arrived_at_merchant", order.id)
        else:
            order.mark_completed()
            if order in self.orders:
                self.orders.remove(order)
            self.completed_orders.append(order.id)
            self.current_order = None
            self.state = STATE_IDLE
            self._record("delivered_to_customer", order.id)
            self._accept_next_order()

    def move(self):
        """Advance the rider by one tick.

        Moves up to ``speed`` units along the planned axis-aligned path and
        performs any state transitions (with log entries) on the way.
        """
        self.orders = [o for o in self.orders
                       if getattr(o, "status", None) != "completed"
                       or o is self.current_order]
        self._accept_next_order()
        budget = self.speed
        guard = 0
        while self.current_order is not None and guard < 1000:
            guard += 1
            target = self.current_target()
            if (self.x, self.y) == target:
                self._arrive()
                continue
            if budget <= 0:
                break
            if not self.path:
                self.special_case = PathPlanner.is_special_case(
                    (self.x, self.y), target)
                self.path = PathPlanner.plan_path((self.x, self.y), target)
                if not self.path:
                    continue
            wx, wy = self.path[0]
            if (self.x, self.y) == (wx, wy):
                self.path.pop(0)
                continue
            if self.x != wx:
                delta = wx - self.x
                step = min(budget, abs(delta))
                self.x += step if delta > 0 else -step
            else:
                delta = wy - self.y
                step = min(budget, abs(delta))
                self.y += step if delta > 0 else -step
            budget -= step
            self.distance_travelled += step
            if (self.x, self.y) == (wx, wy):
                self.path.pop(0)

    # ------------------------------------------------------------------
    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, str(self.state))

    def remaining_route(self):
        """Polyline covering all outstanding orders of this rider."""
        dests = []
        for order in self.orders:
            if order.status == "completed":
                continue
            if order is self.current_order and self.state == STATE_DELIVERING:
                dests.append(tuple(order.cus_pos))
            else:
                dests.append(tuple(order.res_pos))
                dests.append(tuple(order.cus_pos))
        return PathPlanner.plan_route((self.x, self.y), dests)

    def to_dict(self):
        """Serialise to a JSON-friendly dict (orders by id)."""
        return {
            "rider_id": self.rider_id, "x": self.x, "y": self.y,
            "speed": self.speed, "state": self.state,
            "orders": [o.id for o in self.orders],
            "current_order": (self.current_order.id
                              if self.current_order else None),
            "special_case": self.special_case, "log": self.log,
            "completed_orders": self.completed_orders,
            "distance_travelled": self.distance_travelled,
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        """Rebuild a rider from :meth:`to_dict` output."""
        obj = cls.__new__(cls)
        obj.rider_id = data["rider_id"]
        obj.x = int(data["x"])
        obj.y = int(data["y"])
        obj.speed = int(data.get("speed", 10))
        obj.state = int(data.get("state", 0))
        obj.orders = [orders_by_id[i] for i in data.get("orders", [])
                      if i in orders_by_id]
        cur = data.get("current_order")
        obj.current_order = orders_by_id.get(cur) if cur is not None else None
        obj.special_case = bool(data.get("special_case", False))
        obj.path = []
        obj.log = list(data.get("log", []))
        obj.completed_orders = list(data.get("completed_orders", []))
        obj.distance_travelled = data.get("distance_travelled", 0)
        return obj

    def __repr__(self):
        return "Rider(id=%s, pos=(%d,%d), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state_name, len(self.orders))

"""Order creation, assignment and completion handling."""

import globalvar as gv
from order_model import Order
from path_planner import PathPlanner


def find_best_rider_for_order(order):
    """Return the rider closest (Euclidean) to the order's restaurant.

    Distance: sqrt((rider_x - restaurant_x)^2 + (rider_y - restaurant_y)^2).
    Ties are broken by fewer assigned orders, then by lower id (load
    balancing).  Returns None if there are no riders.
    """
    riders = gv.get_value("riders") or []
    best = None
    best_key = None
    for rider in riders:
        dist = PathPlanner.euclidean((rider.x, rider.y), order.res_pos)
        key = (dist, len(rider.orders), rider.rider_id)
        if best_key is None or key < best_key:
            best_key = key
            best = rider
    return best


class OrderManager:
    """Handles order creation, dispatch and completion bookkeeping."""

    def __init__(self, system_manager):
        self.system = system_manager

    @staticmethod
    def all_orders():
        """Return the global order list (creating it if needed)."""
        orders = gv.get_value("orders")
        if orders is None:
            orders = []
            gv.set_value("orders", orders)
        return orders

    @staticmethod
    def pending_orders():
        """Orders that are not yet assigned to a rider."""
        return [o for o in OrderManager.all_orders() if o.status == "pending"]

    def create_order(self, raw):
        """Create an order from ``[rx, ry, cx, cy]`` raw coordinates."""
        res = PathPlanner.to_grid_center(raw[0], raw[1])
        cus = PathPlanner.to_grid_center(raw[2], raw[3])
        order = Order(res, cus)
        self.all_orders().append(order)
        return order

    @staticmethod
    def assign(order, rider):
        """Assign ``order`` to ``rider``."""
        order.assign_distance = round(
            PathPlanner.euclidean((rider.x, rider.y), order.res_pos), 2)
        rider.orders.append(order)
        order.assign_to(rider.rider_id)

    def dispatch(self, raw):
        """Create an order, recruit if necessary and assign it.

        Returns (order, new_rider_or_None, assigned_rider_or_None).
        """
        order = self.create_order(raw)
        new_rider = self.system.maybe_recruit()
        rider = find_best_rider_for_order(order)
        if rider is not None:
            self.assign(order, rider)
        return order, new_rider, rider

    def assign_pending(self):
        """Try to assign any orders still waiting for a rider."""
        for order in self.pending_orders():
            self.system.maybe_recruit()
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            self.assign(order, rider)

    @staticmethod
    def completed_orders():
        """Return all completed orders."""
        return [o for o in OrderManager.all_orders() if o.status == "completed"]

    @staticmethod
    def efficiency_stats():
        """Return (completed count, average time taken or None)."""
        done = OrderManager.completed_orders()
        if not done:
            return 0, None
        avg = sum(o.time_taken for o in done) / len(done)
        return len(done), avg

"""System initialisation and the SystemManager core class."""

import json
import logging
import os

import globalvar as gv
from order_model import Order
from rider import Rider, STATE_IDLE

logger = logging.getLogger("dispatch")

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "rider_cost": 300,
    "rider_speed": 10,
    "spawn_x": 435,
    "spawn_y": 385,
}


def init_system(config=None):
    """Initialise all global system parameters.

    funds=1000, riders=[], rider_id_counter=0, running=True, plus the
    order list, order counter and simulated time.
    """
    cfg = dict(DEFAULT_CONFIG)
    if config:
        cfg.update(config)
    gv.set_value("config", cfg)
    gv.set_value("funds", int(cfg["initial_funds"]))
    gv.set_value("riders", [])
    gv.set_value("rider_id_counter", 0)
    gv.set_value("running", True)
    gv.set_value("orders", [])
    gv.set_value("order_id_counter", 0)
    gv.set_value("time", 0)


def load_config(path):
    """Load a JSON config file; returns {} if missing or invalid."""
    if not path or not os.path.exists(path):
        return {}
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        return {k: data[k] for k in DEFAULT_CONFIG if k in data}
    except (OSError, ValueError) as exc:
        logger.warning("Could not load config %s: %s", path, exc)
        return {}


class SystemManager:
    """Manages funds, riders, simulated time and system state."""

    @property
    def funds(self):
        """Current fund balance."""
        return gv.get_value("funds", 0)

    @property
    def riders(self):
        """List of riders."""
        riders = gv.get_value("riders")
        if riders is None:
            riders = []
            gv.set_value("riders", riders)
        return riders

    @property
    def time(self):
        """Current simulated tick."""
        return gv.get_value("time", 0)

    def create_rider(self):
        """Create a rider, deducting the recruitment cost."""
        cfg = gv.get_value("config") or DEFAULT_CONFIG
        counter = (gv.get_value("rider_id_counter", 0) or 0) + 1
        gv.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        gv.set_value("funds", self.funds - int(cfg.get("rider_cost", 300)))
        logger.info("Recruited rider %d", counter)
        return rider

    def maybe_recruit(self):
        """Recruit a rider if funds >= cost and pending > idle riders.

        Returns the new rider or None.
        """
        cfg = gv.get_value("config") or DEFAULT_CONFIG
        cost = int(cfg.get("rider_cost", 300))
        orders = gv.get_value("orders") or []
        pending = sum(1 for o in orders if o.status == "pending")
        idle = sum(1 for r in self.riders
                   if r.state == STATE_IDLE and not r.orders)
        if self.funds >= cost and pending > idle:
            return self.create_rider()
        return None

    def tick(self, order_manager=None):
        """Advance the simulation by one tick."""
        gv.set_value("time", self.time + 1)
        if order_manager is not None:
            order_manager.assign_pending()
        for rider in self.riders:
            rider.move()

    def all_idle(self):
        """True if no orders are outstanding."""
        orders = gv.get_value("orders") or []
        return all(o.status == "completed" for o in orders)

    def consistency_check(self):
        """Repair inconsistent state; returns list of fixes applied."""
        fixes = []
        for rider in self.riders:
            rider.x = max(0, min(800, int(rider.x)))
            rider.y = max(0, min(800, int(rider.y)))
            if rider.state not in (0, 1, 2):
                rider.state = STATE_IDLE
                fixes.append("rider %s invalid state reset" % rider.rider_id)
            if rider.current_order is None and rider.state != STATE_IDLE:
                rider.state = STATE_IDLE
                rider.path = []
                fixes.append("rider %s state reset to idle" % rider.rider_id)
            if (rider.current_order is not None
                    and rider.current_order.status == "completed"):
                rider.current_order = None
                rider.state = STATE_IDLE
                rider.path = []
                fixes.append("rider %s cleared finished order" % rider.rider_id)
            before = len(rider.orders)
            rider.orders = [o for o in rider.orders if o.status != "completed"]
            if len(rider.orders) != before:
                fixes.append("rider %s removed completed orders" % rider.rider_id)
        if gv.get_value("running") is None:
            gv.set_value("running", True)
        for fix in fixes:
            logger.warning("Consistency fix: %s", fix)
        return fixes

    # ---------------------------------------------------------- persistence
    def save_state(self, path):
        """Save config, riders and order history to a JSON file."""
        data = {
            "config": gv.get_value("config"),
            "funds": self.funds,
            "rider_id_counter": gv.get_value("rider_id_counter", 0),
            "order_id_counter": gv.get_value("order_id_counter", 0),
            "time": self.time,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in (gv.get_value("orders") or [])],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)

    def load_state(self, path):
        """Load state saved by :meth:`save_state`."""
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        cfg = dict(DEFAULT_CONFIG)
        cfg.update(data.get("config") or {})
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        by_id = {o.id: o for o in orders}
        riders = [Rider.from_dict(d, by_id) for d in data.get("riders", [])]
        gv.set_value("config", cfg)
        gv.set_value("funds", int(data.get("funds", cfg["initial_funds"])))
        gv.set_value("rider_id_counter", int(data.get("rider_id_counter", 0)))
        gv.set_value("order_id_counter", int(data.get("order_id_counter", 0)))
        gv.set_value("time", int(data.get("time", 0)))

"""Status display and ASCII map visualisation."""

import globalvar as gv
from rider import STATE_NAMES

CHAR_DIM = 17  # 8 cells + 9 street lines


def _idx(v):
    """Map a coordinate (0..800) to a character index (0..16)."""
    return max(0, min(CHAR_DIM - 1, int(round(v / 50.0))))


class DisplayManager:
    """Formats status information and ASCII maps."""

    @staticmethod
    def render_map(riders, orders, route=None):
        """Return an ASCII map string.

        '#' streets, '.' empty cell, 'S' restaurant, 'C' customer,
        'R' rider, '*' planned route.
        """
        grid = [["#" if (r % 2 == 0 or c % 2 == 0) else "."
                 for c in range(CHAR_DIM)] for r in range(CHAR_DIM)]
        if route and len(route) > 1:
            for a, b in zip(route, route[1:]):
                c0, r0 = _idx(a[0]), _idx(a[1])
                c1, r1 = _idx(b[0]), _idx(b[1])
                if c0 == c1:
                    step = 1 if r1 >= r0 else -1
                    for r in range(r0, r1 + step, step):
                        grid[r][c0] = "*"
                else:
                    step = 1 if c1 >= c0 else -1
                    for c in range(c0, c1 + step, step):
                        grid[r0][c] = "*"
        for order in orders:
            if order.status == "completed":
                continue
            grid[_idx(order.cus_pos[1])][_idx(order.cus_pos[0])] = "C"
        for order in orders:
            if order.status in ("completed", "delivering"):
                continue
            grid[_idx(order.res_pos[1])][_idx(order.res_pos[0])] = "S"
        for rider in riders:
            grid[_idx(rider.y)][_idx(rider.x)] = "R"
        header = "    " + "".join(str((c // 2) % 10) if c % 2 else " "
                                  for c in range(CHAR_DIM))
        lines = [header]
        for r in range(CHAR_DIM):
            label = str(r // 2) if r % 2 else " "
            lines.append("  %s %s" % (label, "".join(grid[r])))
        lines.append("Legend: R=rider S=restaurant C=customer "
                     "#=street .=empty *=route")
        return "\n".join(lines)

    @staticmethod
    def format_status(riders):
        """Snapshot of funds and rider states."""
        lines = ["Time: %d   Funds: %d   Riders: %d" % (
            gv.get_value("time", 0), gv.get_value("funds", 0), len(riders))]
        for r in riders:
            lines.append(
                "  Rider %d  pos=(%d,%d)  state=%s  speed=%d  orders=%s%s" % (
                    r.rider_id, r.x, r.y, STATE_NAMES.get(r.state, r.state),
                    r.speed, [o.id for o in r.orders],
                    "  [special]" if r.special_case else ""))
        orders = gv.get_value("orders") or []
        waiting = [o.id for o in orders if o.status == "pending"]
        if waiting:
            lines.append("  Unassigned orders: %s" % waiting)
        return "\n".join(lines)

    @staticmethod
    def format_orders(orders):
        """Table of all orders."""
        if not orders:
            return "No orders yet."
        lines = ["%-4s %-6s %-12s %-12s %-6s %-6s %-6s %s" % (
            "ID", "Start", "Restaurant", "Customer", "Dist", "Taken",
            "Rider", "Status")]
        for o in orders:
            taken = o.time_taken
            lines.append("%-4d %-6d %-12s %-12s %-6d %-6s %-6s %s" % (
                o.id, o.start_time, "(%d,%d)" % o.res_pos,
                "(%d,%d)" % o.cus_pos, o.distance,
                "-" if taken is None else str(taken),
                "-" if o.rider_id is None else str(o.rider_id), o.status))
        done = [o for o in orders if o.status == "completed"]
        if done:
            avg = sum(o.time_taken for o in done) / len(done)
            lines.append("Completed: %d   Avg delivery time: %.1f ticks" % (
                len(done), avg))
        return "\n".join(lines)

    @staticmethod
    def format_rider_logs(riders):
        """Every status change of every rider."""
        if not riders:
            return "No riders yet."
        lines = []
        for r in riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%-5s %-22s pos=(%s,%s) order=%s" % (
                    e.get("time"), e.get("event"), e.get("x"), e.get("y"),
                    "-" if e.get("order_id") is None else e.get("order_id")))
        return "\n".join(lines)

"""Command parsing and input validation."""

import re

COMMANDS = {
    "start": "start", "s": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "run": "run", "ru": "run",
    "save": "save", "load": "load",
    "history": "history", "hi": "history",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "q": "quit", "exit": "quit",
}


class InputManager:
    """Parses commands and validates order coordinates."""

    def __init__(self):
        self.history = []
        try:
            import readline  # noqa: F401  (enables arrow-key history)
            self.readline = readline
        except ImportError:
            self.readline = None

    def parse_command(self, line):
        """Return (command, args) where command may be None/'unknown'."""
        line = (line or "").strip()
        if not line:
            return None, []
        self.history.append(line)
        if len(self.history) > 1000:
            self.history = self.history[-1000:]
        parts = line.split(None, 1)
        cmd = COMMANDS.get(parts[0].lower(), "unknown:" + parts[0])
        rest = parts[1] if len(parts) > 1 else ""
        return cmd, rest

    @staticmethod
    def parse_order_input(text):
        """Parse four integer coordinates in range 0..800.

        Accepts '[1, 2, 3, 4]', '1,2,3,4' or '1 2 3 4'.
        Raises ValueError with a friendly message on bad input.
        """
        tokens = [t for t in re.split(r"[\s,\[\]()]+", text or "") if t]
        if len(tokens) != 4:
            raise ValueError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y")
        values = []
        for t in tokens:
            try:
                v = int(t)
            except ValueError:
                raise ValueError("'%s' is not an integer" % t)
            if v < 0 or v > 800:
                raise ValueError("Coordinate %d out of range 0-800" % v)
            values.append(v)
        return values

    @staticmethod
    def parse_positive_int(text, default=1):
        """Parse a positive integer argument with a default."""
        text = (text or "").strip()
        if not text:
            return default
        v = int(text)
        if v <= 0:
            raise ValueError("Value must be positive")
        return v

"""Command-line entry point for the food delivery dispatch simulator.

Each iteration of the main loop represents one simulated tick.
"""

import logging
import math
import os
import sys

import globalvar as gv
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from system import SystemManager, init_system, load_config

HERE = os.path.dirname(os.path.abspath(__file__))
CONFIG_PATH = os.path.join(HERE, "config.json")
DEFAULT_STATE = "dispatch_state.json"

HELP = """Commands (shortcut):
  start (s) [rx ry cx cy]  create an order (coords 0-800)
  status (st)              system funds and rider states
  orders (o)               all order details
  riders (r)               rider status-change log
  map (m)                  show ASCII map
  tick (t) [n]             advance n ticks
  run (ru)                 run until all orders delivered
  save [file] / load [file]  persist state as JSON
  history (hi)             command history
  help (h)                 this help
  quit/exit (q)            exit"""


class App:
    """Glue between the managers and the command line."""

    def __init__(self):
        gv._init()
        init_system(load_config(CONFIG_PATH))
        self.system = SystemManager()
        self.orders = OrderManager(self.system)
        self.display = DisplayManager()
        self.input = InputManager()

    def cmd_start(self, args):
        if not args.strip():
            args = input("Enter [restaurant_x, restaurant_y, customer_x, "
                         "customer_y]: ")
        raw = self.input.parse_order_input(args)
        order, new_rider, rider = self.orders.dispatch(raw)
        print("Order %d created." % order.id)
        print("  Restaurant grid position: (%d, %d)" % order.res_pos)
        print("  Customer grid position:   (%d, %d)" % order.cus_pos)
        if new_rider is not None:
            print("  New rider added: yes (rider %d, funds now %d)" % (
                new_rider.rider_id, self.system.funds))
        else:
            print("  New rider added: no")
        if rider is None:
            print("  No rider available; order is waiting for assignment.")
            return
        route = rider.remaining_route()
        from path_planner import PathPlanner
        length = PathPlanner.path_length(route)
        print("  Assigned to rider %d at (%d, %d), speed %d units/tick" % (
            rider.rider_id, rider.x, rider.y, rider.speed))
        print("  Route length %d units, estimated time %d ticks" % (
            length, math.ceil(length / float(rider.speed or 1))))
        print(self.display.render_map(self.system.riders,
                                      gv.get_value("orders") or [], route))

    def cmd_tick(self, args):
        n = self.input.parse_positive_int(args, 1)
        for _ in range(n):
            self.system.tick(self.orders)
        print("Advanced %d tick(s). Time is now %d." % (n, self.system.time))

    def cmd_run(self, _args):
        steps = 0
        while not self.system.all_idle() and steps < 100000:
            self.system.tick(self.orders)
            steps += 1
        print("Ran %d tick(s). Time is now %d." % (steps, self.system.time))
        if not self.system.all_idle():
            print("Some orders could not be completed (no riders available).")

    def handle(self, cmd, args):
        """Dispatch a parsed command."""
        if cmd == "start":
            self.cmd_start(args)
        elif cmd == "status":
            print(self.display.format_status(self.system.riders))
        elif cmd == "orders":
            print(self.display.format_orders(gv.get_value("orders") or []))
        elif cmd == "riders":
            print(self.display.format_rider_logs(self.system.riders))
        elif cmd == "map":
            print(self.display.render_map(self.system.riders,
                                          gv.get_value("orders") or []))
        elif cmd == "tick":
            self.cmd_tick(args)
        elif cmd == "run":
            self.cmd_run(args)
        elif cmd == "save":
            path = args.strip() or DEFAULT_STATE
            self.system.save_state(path)
            print("State saved to %s" % path)
        elif cmd == "load":
            path = args.strip() or DEFAULT_STATE
            self.system.load_state(path)
            print("State loaded from %s" % path)
        elif cmd == "history":
            for i, h in enumerate(self.input.history, 1):
                print("%4d  %s" % (i, h))
        elif cmd == "help":
            print(HELP)
        elif cmd == "quit":
            gv.set_value("running", False)
            print("Goodbye.")
        elif cmd and cmd.startswith("unknown:"):
            print("Unknown command '%s'. Type 'help'." % cmd[8:])

    def run(self):
        """Main loop: one iteration == one tick."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while gv.get_value("running"):
            try:
                line = input("dispatch[t=%d]> " % self.system.time)
            except EOFError:
                print()
                break
            except KeyboardInterrupt:
                print("\nInterrupted. Type 'quit' to exit.")
                continue
            cmd, args = self.input.parse_command(line)
            try:
                self.handle(cmd, args)
            except ValueError as exc:
                print("Input error: %s" % exc)
            except (OSError, KeyError) as exc:
                print("Error: %s" % exc)
            except Exception as exc:  # recover from unexpected failures
                logging.getLogger("dispatch").exception("Unexpected error")
                print("Unexpected error: %s (state recovered)" % exc)
                self.system.consistency_check()
            if gv.get_value("running"):
                self.system.tick(self.orders)


def main():
    """Program entry point."""
    level = logging.DEBUG if "--debug" in sys.argv else logging.WARNING
    logging.basicConfig(level=level, format="%(levelname)s: %(message)s")
    App().run()


if __name__ == "__main__":
    main()

"""Global variable management for the dispatch simulator.

All shared simulation state (funds, riders, orders, tick, ...) is stored in
a single module-level dictionary so every module sees the same values.
"""

_global_dict = None


def _init():
    """Initialise (or reset) the global dictionary."""
    global _global_dict
    _global_dict = {}


def _ensure():
    """Lazily initialise the global dictionary if needed."""
    if _global_dict is None:
        _init()


def set_value(key, value):
    """Store ``value`` under ``key``."""
    _ensure()
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` (or ``default``)."""
    _ensure()
    return _global_dict.get(key, default)


def has_value(key):
    """Return True if ``key`` has been set."""
    _ensure()
    return key in _global_dict

"""JSON configuration loading and state persistence."""

import json
import logging
import os

import globalvar

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 300,
    "recruitment_threshold": 300,
    "rider_cost": 300,
    "order_fee": 100,
    "rider_speed": 100,
    "spawn_x": 400,
    "spawn_y": 400,
    "map_size": 800,
    "cell_size": 100,
    "state_file": "dispatch_state.json",
    "log_level": "WARNING",
}

DEFAULT_CONFIG_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                                   "config.json")


class Configuration:
    """Handles the JSON configuration file and state snapshots."""

    def __init__(self, file_path=None):
        self.file_path = file_path or DEFAULT_CONFIG_PATH
        self.format = "json"

    def load(self):
        """Load configuration parameters, falling back to defaults.

        :return: dict of parameters.
        """
        params = dict(DEFAULT_CONFIG)
        if self.file_path and os.path.exists(self.file_path):
            try:
                with open(self.file_path, "r", encoding="utf-8") as fh:
                    data = json.load(fh)
                if isinstance(data, dict):
                    for key, value in data.items():
                        if key in params and isinstance(value, type(params[key])):
                            params[key] = value
            except (OSError, ValueError) as exc:
                logger.warning("Could not read config %s: %s", self.file_path, exc)
        return params

    @staticmethod
    def build_state():
        """Build a serialisable snapshot of the simulation state."""
        riders = globalvar.get_value("riders", []) or []
        orders = globalvar.get_value("orders", []) or []
        return {
            "system": {
                "funds": globalvar.get_value("funds", 0),
                "tick": globalvar.get_value("tick", 0),
                "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
                "order_id_counter": globalvar.get_value("order_id_counter", 0),
            },
            "riders": [r.to_dict() for r in riders],
            "orders": [o.to_dict() for o in orders],
        }

    def save_state(self, path=None):
        """Write the state snapshot to a JSON file.

        :return: path written, or None on failure.
        """
        path = path or globalvar.get_value("state_file", DEFAULT_CONFIG["state_file"])
        try:
            with open(path, "w", encoding="utf-8") as fh:
                json.dump(self.build_state(), fh, indent=2)
            return path
        except OSError as exc:
            logger.error("Failed to save state to %s: %s", path, exc)
            return None

"""Grid-based path planning.

The map is 800x800 units divided into 8x8 cells of 100x100.  Streets run
along the grid lines (coordinates that are multiples of 100).  Restaurants
and customers sit at cell centres (multiples of 100 plus 50).
"""

import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = MAP_SIZE // CELL_SIZE


def clamp(value, low=0, high=MAP_SIZE):
    """Clamp ``value`` into [low, high]."""
    return max(low, min(high, value))


def to_grid_index(value):
    """Convert a coordinate (0-800) to a grid index (0-7)."""
    value = clamp(int(value), 0, MAP_SIZE - 1)
    return value // CELL_SIZE


def to_grid_center(value):
    """Convert a coordinate to the centre of the containing cell.

    ``center = (v // 100) * 100 + 50``; the boundary value 800 is mapped
    into the last cell.
    """
    return to_grid_index(value) * CELL_SIZE + CELL_SIZE // 2


def manhattan(a, b):
    """Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class Location:
    """Simple 2D point."""

    def __init__(self, x, y):
        self.x = int(x)
        self.y = int(y)

    def as_tuple(self):
        return (self.x, self.y)

    def __repr__(self):
        return "Location({}, {})".format(self.x, self.y)


class Route:
    """A planned polyline route made of axis-aligned segments."""

    def __init__(self, waypoints, speed=100, special_case=False):
        self.waypoints = [tuple(p) for p in waypoints]
        self.speed = max(1, int(speed))
        self.special_case = special_case

    @property
    def length(self):
        total = 0
        for a, b in zip(self.waypoints, self.waypoints[1:]):
            total += manhattan(a, b)
        return total

    @property
    def estimated_time(self):
        return int(math.ceil(self.length / float(self.speed)))

    def __repr__(self):
        return "Route({}, length={})".format(self.waypoints, self.length)


class PathPlanner:
    """Plans street-first routes on the grid."""

    def __init__(self, cell_size=CELL_SIZE, map_size=MAP_SIZE):
        self.cell_size = cell_size
        self.map_size = map_size

    def on_street(self, value):
        """True if the coordinate lies on a grid line (street)."""
        return value % self.cell_size == 0

    def _street_toward(self, start, end):
        """Choose the horizontal street line to use from ``start``."""
        if self.on_street(start):
            return start
        low = (start // self.cell_size) * self.cell_size
        high = low + self.cell_size
        if end > start:
            street = high
        elif end < start:
            street = low
        else:
            street = low if start - low <= high - start else high
        return clamp(street, 0, self.map_size)

    @staticmethod
    def _simplify(points):
        """Remove duplicate and collinear intermediate points."""
        result = []
        for p in points:
            if result and result[-1] == p:
                continue
            if len(result) >= 2:
                a, b = result[-2], result[-1]
                if (a[0] == b[0] == p[0]) or (a[1] == b[1] == p[1]):
                    result[-1] = p
                    continue
            result.append(p)
        return result

    def plan(self, start, end, speed=100):
        """Plan a route from ``start`` to ``end``.

        Strategy: move vertically to the nearest horizontal street toward
        the target, travel along it, then turn into the destination.
        Special case: when the start is directly above/below the target
        (same x) the rider goes straight.

        :return: Route
        """
        sx, sy = int(start[0]), int(start[1])
        ex, ey = int(end[0]), int(end[1])
        special = False
        if (sx, sy) == (ex, ey):
            points = [(sx, sy)]
        elif sx == ex:
            special = True
            points = [(sx, sy), (ex, ey)]
        elif sy == ey and self.on_street(sy):
            points = [(sx, sy), (ex, ey)]
        else:
            street_y = self._street_toward(sy, ey)
            points = [(sx, sy), (sx, street_y), (ex, street_y), (ex, ey)]
        return Route(self._simplify(points), speed=speed, special_case=special)

    def plan_multi(self, start, stops, speed=100):
        """Plan a continuous route through several destinations."""
        points = [tuple(start)]
        special = False
        current = tuple(start)
        for stop in stops:
            leg = self.plan(current, stop, speed)
            special = special or leg.special_case
            points.extend(leg.waypoints[1:])
            current = tuple(stop)
        # Keep stops as explicit waypoints; only drop exact duplicates.
        cleaned = []
        for p in points:
            if not cleaned or cleaned[-1] != p:
                cleaned.append(p)
        return Route(cleaned, speed=speed, special_case=special)

"""Order data model."""

import globalvar
from path_planner import PathPlanner, to_grid_center, to_grid_index, MAP_SIZE

ORDER_PENDING = "pending"
ORDER_ASSIGNED = "assigned"
ORDER_PICKING_UP = "picking_up"
ORDER_DELIVERING = "delivering"
ORDER_COMPLETED = "completed"


def _validate_coordinate(value, label):
    """Validate a single coordinate and return it as int."""
    if isinstance(value, bool):
        raise ValueError("{} must be an integer".format(label))
    if isinstance(value, int):
        result = value
    elif isinstance(value, str) and value.strip().lstrip("-").isdigit():
        result = int(value.strip())
    elif isinstance(value, float) and value.is_integer():
        result = int(value)
    else:
        raise ValueError("{} must be an integer, got {!r}".format(label, value))
    if result < 0 or result > MAP_SIZE:
        raise ValueError("{} must be in range 0-{}, got {}".format(
            label, MAP_SIZE, result))
    return result


def _validate_pos(pos, label):
    """Validate a 2-element position."""
    try:
        items = list(pos)
    except TypeError:
        raise ValueError("{} position must be a pair of integers".format(label))
    if len(items) != 2:
        raise ValueError("{} position must have 2 coordinates".format(label))
    return (_validate_coordinate(items[0], label + " x"),
            _validate_coordinate(items[1], label + " y"))


class Order:
    """A delivery order from a restaurant to a customer."""

    def __init__(self, res_pos, cus_pos):
        rx, ry = _validate_pos(res_pos, "restaurant")
        cx, cy = _validate_pos(cus_pos, "customer")
        self.raw_restaurant = (rx, ry)
        self.raw_customer = (cx, cy)
        self.res_pos = (to_grid_center(rx), to_grid_center(ry))
        self.cus_pos = (to_grid_center(cx), to_grid_center(cy))
        counter = globalvar.get_value("order_id_counter", 0) or 0
        self.id = counter + 1
        globalvar.set_value("order_id_counter", self.id)
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.status = ORDER_PENDING
        self.rider_id = None
        self.distance = PathPlanner().plan(self.res_pos, self.cus_pos).length
        self.travel_distance = 0

    @property
    def restaurant_x(self):
        return self.res_pos[0]

    @property
    def restaurant_y(self):
        return self.res_pos[1]

    @property
    def customer_x(self):
        return self.cus_pos[0]

    @property
    def customer_y(self):
        return self.cus_pos[1]

    @property
    def restaurant_cell(self):
        return (to_grid_index(self.res_pos[0]), to_grid_index(self.res_pos[1]))

    @property
    def customer_cell(self):
        return (to_grid_index(self.cus_pos[0]), to_grid_index(self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Assign the order to a rider."""
        self.rider_id = rider_id
        if self.status == ORDER_PENDING:
            self.status = ORDER_ASSIGNED

    def is_completed(self):
        return self.status == ORDER_COMPLETED

    def to_dict(self):
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "raw_restaurant": list(self.raw_restaurant),
            "raw_customer": list(self.raw_customer),
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "travel_distance": self.travel_distance,
            "status": self.status,
            "rider_id": self.rider_id,
        }

    def __repr__(self):
        return "Order(id={}, {}->{}, status={})".format(
            self.id, self.res_pos, self.cus_pos, self.status)

"""Rider model: movement, state machine and status log."""

import logging

import globalvar
from path_planner import PathPlanner, clamp
from order_model import (ORDER_COMPLETED, ORDER_PICKING_UP, ORDER_DELIVERING)

logger = logging.getLogger(__name__)


class RiderState:
    """Rider state enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2

    NAMES = {0: "idle", 1: "picking up", 2: "delivering"}

    @classmethod
    def name(cls, state):
        return cls.NAMES.get(state, "unknown")


class Rider:
    """A delivery rider moving on the grid."""

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = rider_id
        self.x = int(x if x is not None else globalvar.get_value("spawn_x", 400))
        self.y = int(y if y is not None else globalvar.get_value("spawn_y", 400))
        self.speed = int(speed if speed is not None
                         else globalvar.get_value("rider_speed", 100))
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.path = []
        self.current_order = None
        self.completed_orders = []
        self.distance_travelled = 0
        self._planner = PathPlanner()
        self._log_event("spawn", None)

    # ------------------------------------------------------------------ helpers
    def _log_event(self, event, order_id):
        entry = {
            "time": globalvar.get_value("tick", 0) or 0,
            "x": self.x,
            "y": self.y,
            "event": event,
            "order_id": order_id,
            "state": self.state,
        }
        self.log.append(entry)
        logger.info("Rider %s: %s (order %s) at %s", self.rider_id, event,
                    order_id, (self.x, self.y))

    def pending_orders(self):
        """Orders assigned to this rider that are not completed."""
        return [o for o in self.orders if o.status != ORDER_COMPLETED]

    def has_pending_orders(self):
        return bool(self.pending_orders())

    def final_position(self):
        """Position where the rider will be after finishing all orders."""
        pending = self.pending_orders()
        if pending:
            return pending[-1].cus_pos
        return (self.x, self.y)

    def _plan_to(self, target):
        route = self._planner.plan((self.x, self.y), target, self.speed)
        self.path = list(route.waypoints[1:])
        self.special_case = route.special_case

    def _advance(self, target):
        """Move up to ``speed`` units along the planned path."""
        if (self.x, self.y) != tuple(target) and (
                not self.path or self.path[-1] != tuple(target)):
            self._plan_to(target)
        remaining = max(0, self.speed)
        moved = 0
        while remaining > 0 and self.path:
            tx, ty = self.path[0]
            dx, dy = tx - self.x, ty - self.y
            if dx != 0 and dy != 0:
                # Should not happen (segments are axis-aligned); fix by
                # inserting a corner point.
                self.path.insert(0, (tx, self.y))
                continue
            dist = abs(dx) + abs(dy)
            if dist <= remaining:
                self.x, self.y = tx, ty
                remaining -= dist
                moved += dist
                self.path.pop(0)
            else:
                if dx:
                    self.x += remaining if dx > 0 else -remaining
                else:
                    self.y += remaining if dy > 0 else -remaining
                moved += remaining
                remaining = 0
        self.x = clamp(self.x)
        self.y = clamp(self.y)
        self.distance_travelled += moved
        if self.current_order is not None:
            self.current_order.travel_distance += moved

    def _reset_idle(self):
        self.state = RiderState.IDLE
        self.current_order = None
        self.path = []
        self.special_case = False

    # ------------------------------------------------------------------- tick
    def move(self):
        """Advance the rider by one tick.

        * Idle with a pending order -> accept it (state 1).
        * Picking up -> move toward restaurant; on arrival -> state 2.
        * Delivering -> move toward customer; on arrival -> complete, state 0.
        """
        tick = globalvar.get_value("tick", 0) or 0
        if self.state == RiderState.IDLE:
            pending = self.pending_orders()
            if not pending:
                self.special_case = False
                return
            order = pending[0]
            self.current_order = order
            if order.rider_id != self.rider_id:
                order.assign_to(self.rider_id)
            order.status = ORDER_PICKING_UP
            self.state = RiderState.PICKING_UP
            self._plan_to(order.res_pos)
            self._log_event("accepted_order", order.id)
            return

        order = self.current_order
        if order is None or order not in self.orders or order.is_completed():
            # Inconsistent state: recover to idle.
            logger.warning("Rider %s recovered from inconsistent state",
                           self.rider_id)
            self._reset_idle()
            return

        if self.state == RiderState.PICKING_UP:
            self._advance(order.res_pos)
            if (self.x, self.y) == tuple(order.res_pos):
                self.state = RiderState.DELIVERING
                order.status = ORDER_DELIVERING
                order.pickup_time = tick
                self._log_event("arrived_at_merchant", order.id)
                self._plan_to(order.cus_pos)
        elif self.state == RiderState.DELIVERING:
            self._advance(order.cus_pos)
            if (self.x, self.y) == tuple(order.cus_pos):
                order.status = ORDER_COMPLETED
                order.completion_time = tick
                order.time_taken = tick - order.start_time
                self._log_event("delivered_to_customer", order.id)
                self.orders.remove(order)
                self.completed_orders.append(order)
                self._reset_idle()
        else:
            self._reset_idle()

    # ------------------------------------------------------------- reporting
    def to_dict(self):
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "completed_orders": [o.id for o in self.completed_orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id={}, pos=({}, {}), state={})".format(
            self.rider_id, self.x, self.y, RiderState.name(self.state))

"""Order management: creation, assignment (nearest rider first)."""

import logging

import globalvar
from order_model import Order, ORDER_PENDING, ORDER_COMPLETED
from path_planner import PathPlanner, manhattan
from rider import RiderState

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Return the best rider for ``order`` (closest rider first).

    Idle riders without pending orders are preferred and the one closest
    (Manhattan distance) to the restaurant wins.  If every rider is busy,
    the rider with the fewest pending orders whose final position is
    closest is chosen (load balancing).  Returns None if no riders exist.
    """
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None
    target = order.res_pos
    idle = [r for r in riders
            if r.state == RiderState.IDLE and not r.has_pending_orders()]
    if idle:
        return min(idle, key=lambda r: (manhattan((r.x, r.y), target),
                                         r.rider_id))
    return min(riders, key=lambda r: (len(r.pending_orders()),
                                      manhattan(r.final_position(), target),
                                      r.rider_id))


class OrderManager:
    """Handles order creation, assignment and pending queue."""

    def __init__(self, planner=None):
        self.assignment_strategy = "nearest"
        self.planner = planner or PathPlanner()

    @staticmethod
    def all_orders():
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    @staticmethod
    def pending_queue():
        queue = globalvar.get_value("pending_orders")
        if queue is None:
            queue = []
            globalvar.set_value("pending_orders", queue)
        return queue

    def create_order(self, coords):
        """Create an order from ``[rx, ry, cx, cy]`` and register it."""
        if coords is None or len(coords) != 4:
            raise ValueError("an order needs exactly 4 coordinates")
        order = Order(coords[0:2], coords[2:4])
        self.all_orders().append(order)
        fee = globalvar.get_value("order_fee", 100) or 0
        globalvar.set_value("funds", (globalvar.get_value("funds", 0) or 0) + fee)
        return order

    def assign_order(self, order):
        """Assign an order to the best rider, or queue it if none exists.

        :return: (rider or None, preview Route or None)
        """
        rider = find_best_rider_for_order(order)
        if rider is None:
            if order not in self.pending_queue():
                self.pending_queue().append(order)
            return None, None
        start = rider.final_position()
        route = self.planner.plan_multi(start, [order.res_pos, order.cus_pos],
                                        rider.speed)
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        queue = self.pending_queue()
        if order in queue:
            queue.remove(order)
        return rider, route

    def dispatch_pending(self):
        """Try to assign queued orders. Returns number assigned."""
        assigned = 0
        for order in list(self.pending_queue()):
            if order.status != ORDER_PENDING:
                self.pending_queue().remove(order)
                continue
            rider, _ = self.assign_order(order)
            if rider is None:
                break
            assigned += 1
        return assigned

    def completed_orders(self):
        return [o for o in self.all_orders() if o.status == ORDER_COMPLETED]

"""System initialisation and core management (funds, riders, ticks)."""

import logging

import globalvar
from config import Configuration
from order import OrderManager
from order_model import ORDER_COMPLETED
from path_planner import clamp, GridMapInfo if False else clamp  # noqa: F401
from rider import Rider, RiderState

class SystemManager:
    """Core class managing funds, riders and simulated time.

    All state is stored in :mod:`globalvar`; this class offers operations.
    """

    def __init__(self, configuration=None):
        self.configuration = configuration or Configuration()
        self.order_manager = OrderManager()

    # -------------------------------------------------------------- props
    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def riders(self):
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def running(self):
        return bool(globalvar.get_value("running", False))

    @property
    def current_tick(self):
        return globalvar.get_value("tick", 0)

    @property
    def recruitment_threshold(self):
        return globalvar.get_value("recruitment_threshold", 300)

    # ------------------------------------------------------------ riders
    def recruit_rider(self):
        """Create a new rider with an auto-incremented ID."""
        rider_id = globalvar.get_value("rider_id_counter", 0) or 0
        rider = Rider(rider_id)
        globalvar.set_value("rider_id_counter", rider_id + 1)
        self.riders.append(rider)
        return rider

    def remove_rider(self, rider_id):
        """Remove an idle rider. Returns True on success."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                if rider.has_pending_orders():
                    return False
                self.riders.remove(rider)
                return True
        return False

    def get_rider(self, rider_id):
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def check_recruitment(self):
        """Recruit riders while funds >= threshold. Returns new riders."""
        new_riders = []
        cost = globalvar.get_value("rider_cost", 300) or 0
        threshold = self.recruitment_threshold
        guard = 0
        while self.funds >= threshold and guard < 100:
            guard += 1
            new_riders.append(self.recruit_rider())
            self.funds = self.funds - cost
            if cost <= 0:
                break
        return new_riders

    # ------------------------------------------------------------ orders
    def place_order(self, coords):
        """Handle the ``start`` command.

        :return: dict with keys order, rider, route, new_riders.
        """
        new_riders = self.check_recruitment()
        order = self.order_manager.create_order(coords)
        rider, route = self.order_manager.assign_order(order)
        return {"order": order, "rider": rider, "route": route,
                "new_riders": new_riders}

    # -------------------------------------------------------------- time
    def advance_tick(self, count=1):
        """Advance the simulation ``count`` ticks."""
        for _ in range(max(0, int(count))):
            globalvar.set_value("tick", (self.current_tick or 0) + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception as exc:  # pragma: no cover - recovery
                    logger.exception("Rider %s failed to move: %s",
                                     rider.rider_id, exc)
                    self.check_consistency()
            self.check_recruitment()
            self.order_manager.dispatch_pending()

    def stop(self):
        globalvar.set_value("running", False)

    # -------------------------------------------------------- consistency
    def check_consistency(self):
        """Repair inconsistent rider states. Returns number of fixes."""
        fixes = 0
        for rider in self.riders:
            before = len(rider.orders)
            rider.orders[:] = [o for o in rider.orders
                               if o.status != ORDER_COMPLETED]
            fixes += before - len(rider.orders)
            rider.x, rider.y = clamp(rider.x), clamp(rider.y)
            if rider.state not in (RiderState.IDLE, RiderState.PICKING_UP,
                                   RiderState.DELIVERING):
                rider.state = RiderState.IDLE
                fixes += 1
            if rider.state != RiderState.IDLE and (
                    rider.current_order is None
                    or rider.current_order not in rider.orders):
                rider.state = RiderState.IDLE
                rider.current_order = None
                rider.path = []
                fixes += 1
        if not isinstance(self.funds, int):
            self.funds = int(self.funds or 0)
            fixes += 1
        return fixes

    def save(self, path=None):
        return self.configuration.save_state(path)

def _fine_index(value):
    idx = int(round(value / 50.0))

    # -------------------------------------------------------------- maps
    @staticmethod
    def render_grid():
        """Render the 8x8 cell map: R rider, S restaurant, C customer,
        '#' cells on active routes, '.' empty."""
        grid = [["." for _ in range(GRID_DIMENSION)]
                for _ in range(GRID_DIMENSION)]
        riders = globalvar.get_value("riders", []) or []
        for rider in riders:
            prev = (rider.x, rider.y)
            for point in rider.path:
                for cx, cy in DisplayManager._cells_on_segment(prev, point):
                    grid[cy][cx] = "#"
                prev = point
        for rider in riders:
            for order in rider.pending_orders():
                cx, cy = order.customer_cell
                grid[cy][cx] = "C"
        for rider in riders:
            for order in rider.pending_orders():
                rx, ry = order.restaurant_cell
                grid[ry][rx] = "S"
        for rider in riders:
            grid[to_grid_index(rider.y)][to_grid_index(rider.x)] = "R"
        lines = ["    " + " ".join(str(i) for i in range(GRID_DIMENSION))]
        for y in range(GRID_DIMENSION):
            lines.append("{:>3} ".format(y) + " ".join(grid[y]))
        lines.append("Legend: R=Rider S=Restaurant C=Customer #=Route .=Empty")

    @staticmethod
    def _cells_on_segment(a, b):
        cells = []
        x0, y0 = to_grid_index(a[0]), to_grid_index(a[1])
        x1, y1 = to_grid_index(b[0]), to_grid_index(b[1])
        for x in range(min(x0, x1), max(x0, x1) + 1):
            for y in range(min(y0, y1), max(y0, y1) + 1):

        return cells

            row = []
            for c in range(FINE_SIZE):
                if r % 2 == 0 and c % 2 == 0:
                    row.append("+")
                elif r % 2 == 1 and c % 2 == 1:
                    row.append(".")
                else:
                    row.append(" ")

        for a, b in zip(points, points[1:]):
            ax, ay = _fine_index(a[0]), _fine_index(a[1])
            bx, by = _fine_index(b[0]), _fine_index(b[1])
            for x in range(min(ax, bx), max(ax, bx) + 1):
                grid[ay][x] = "#"
            for y in range(min(ay, by), max(ay, by) + 1):
                grid[y][bx] = "#"
        if customer:
            grid[_fine_index(customer[1])][_fine_index(customer[0])] = "C"
        if restaurant:
            grid[_fine_index(restaurant[1])][_fine_index(restaurant[0])] = "S"
        if points:
            grid[_fine_index(points[0][1])][_fine_index(points[0][0])] = "R"
        header = "      " + " ".join(
            str(c * 50 // 100) if c % 2 == 0 else " " for c in range(FINE_SIZE))

"""Command parsing, shortcuts, history and input validation."""

import re

from path_planner import MAP_SIZE

try:  # optional line editing / arrow-key history
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None


class Command:
    """A command with its shortcut and description."""

    def __init__(self, name, shortcut, description):
        self.name = name
        self.shortcut = shortcut
        self.description = description


COMMANDS = [
    Command("start", "s", "Input an order: rx ry cx cy (0-800)"),
    Command("status", "st", "Show funds, riders and map"),
    Command("orders", "o", "Show all orders"),
    Command("riders", "r", "Show rider status change logs"),
    Command("map", "m", "Show the 8x8 map"),
    Command("tick", "t", "Advance simulation: tick [n]"),
    Command("history", "hi", "Show command history (!! repeat, !n run n)"),
    Command("save", "sv", "Save state to JSON"),
    Command("help", "h", "Show this help"),
    Command("quit", "q", "Exit the program"),
    Command("exit", "x", "Exit the program"),
]


class InputManager:
    """Parses commands and validates user input."""

    def __init__(self):
        self.command_history = []
        self._lookup = {}
        for cmd in COMMANDS:
            self._lookup[cmd.name] = cmd.name
            self._lookup[cmd.shortcut] = cmd.name
        self._lookup["?"] = "help"

    def help_text(self):
        lines = ["Commands:"]
        for cmd in COMMANDS:
            lines.append("  {:<8} ({:<2})  {}".format(cmd.name, cmd.shortcut,
                                                    cmd.description))
        return "\n".join(lines)

    def resolve_history(self, line):
        """Expand ``!!`` and ``!n`` references. Raises ValueError."""
        line = line.strip()
        if line == "!!":
            if not self.command_history:
                raise ValueError("no command history")
            return self.command_history[-1]
        if line.startswith("!") and line[1:].isdigit():
            idx = int(line[1:])
            if idx < 1 or idx > len(self.command_history):
                raise ValueError("history index out of range")
            return self.command_history[idx - 1]
        return line

    def parse(self, line):
        """Parse a line into (command_name, args).

        Unknown commands return (None, [word]).
        """
        line = self.resolve_history(line or "")
        if not line:
            return "", []
        self.command_history.append(line)
        if len(self.command_history) > 1000:
            del self.command_history[:-1000]
        parts = line.split(None, 1)
        word = parts[0].lower()
        rest = parts[1] if len(parts) > 1 else ""
        name = self._lookup.get(word)
        args = [a for a in re.split(r"[\s,\[\]()]+", rest) if a]
        if name is None:
            return None, [word]
        return name, args

    @staticmethod
    def validate_coordinates(args):
        """Validate four integer coordinates in range 0-800."""
        if isinstance(args, str):
            args = [a for a in re.split(r"[\s,\[\]()]+", args) if a]
        if len(args) != 4:
            raise ValueError("expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y, got {}".format(len(args)))
        values = []
        for a in args:
            try:
                v = int(str(a).strip())
            except ValueError:
                raise ValueError("'{}' is not an integer".format(a))
            if v < 0 or v > MAP_SIZE:
                raise ValueError("coordinate {} out of range 0-{}".format(
                    v, MAP_SIZE))
            values.append(v)
        return values

    @staticmethod
    def read_line(prompt):
        """Read a line, converting EOF/interrupt to 'quit'."""
        try:
            return input(prompt)
        except (EOFError, KeyboardInterrupt):
            print()
            return "quit"

"""Command-line entry point for the Food Delivery Order Dispatch System."""

import logging
import sys

import globalvar
from display import DisplayManager
from input_manager import InputManager
from system import init_system


class Application:
    """Main loop: each loop iteration represents one tick."""

    def __init__(self, config_path=None):
        self.system = init_system(config_path)
        level = getattr(logging, str(globalvar.get_value("log_level",
                                                         "WARNING")).upper(),
                        logging.WARNING)
        logging.basicConfig(level=level,
                            format="%(levelname)s %(name)s: %(message)s")
        self.display = DisplayManager()
        self.inputs = InputManager()

    def handle(self, name, args):
        """Execute a command. Returns True if a tick should be advanced."""
        if name == "":
            return True
        if name is None:
            print("Unknown command '{}'. Type 'help' for a list.".format(args[0]))
            return False
        if name in ("quit", "exit"):
            self.system.stop()
            path = self.system.save()
            if path:
                print("State saved to {}".format(path))
            print("Bye.")
            return False
        if name == "help":
            print(self.inputs.help_text())
            return False
        if name == "history":
            for i, cmd in enumerate(self.inputs.command_history, 1):
                print("{:>4}  {}".format(i, cmd))
            return False
        if name == "start":
            if not args:
                line = self.inputs.read_line(
                    "restaurant_x restaurant_y customer_x customer_y: ")
                args = line
            try:
                coords = self.inputs.validate_coordinates(args)
            except ValueError as exc:
                print("Invalid input: {}".format(exc))
                return False
            result = self.system.place_order(coords)
            print(self.display.format_start_result(result))
            return True
        if name == "status":
            print(self.display.format_status())
            return True
        if name == "map":
            print(self.display.render_grid())
            return True
        if name == "orders":
            print(self.display.format_orders())
            return True
        if name == "riders":
            print(self.display.format_rider_logs())
            return True
        if name == "tick":
            count = 1
            if args:
                try:
                    count = int(args[0])
                except ValueError:
                    print("Invalid tick count '{}'".format(args[0]))
                    return False
                if count < 1 or count > 10000:
                    print("Tick count must be 1-10000")
                    return False
            self.system.advance_tick(count)
            print("Advanced {} tick(s); now at tick {}.".format(
                count, self.system.current_tick))
            return False
        if name == "save":
            path = self.system.save()
            print("Saved to {}".format(path) if path else "Save failed.")
            return False
        return False

    def run(self):
        print("Food Delivery Order Dispatch System (8x8 grid, 800x800)")
        print(self.inputs.help_text())
        while self.system.running:
            line = self.inputs.read_line("[t={} funds={}]> ".format(
                self.system.current_tick, self.system.funds))
            try:
                name, args = self.inputs.parse(line)
                if self.handle(name, args) and self.system.running:
                    self.system.advance_tick(1)
            except ValueError as exc:
                print("Error: {}".format(exc))
            except Exception as exc:  # recovery
                logging.getLogger(__name__).exception("Unexpected error")
                print("Unexpected error: {} - recovering.".format(exc))
                self.system.check_consistency()
        return 0


def main(argv=None):
    argv = sys.argv[1:] if argv is None else argv
    config_path = argv[0] if argv else None
    return Application(config_path).run()


if __name__ == "__main__":
    sys.exit(main())

"""Core data models of the lottery system."""

import re
from dataclasses import dataclass, field
from typing import List, Optional


@dataclass(frozen=True)
class User:
    """A participant of the lottery."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize configuration (one lottery round)."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [g for g in self.avoid_groups if g and str(g).strip()]
        if isinstance(self.associated_drawer, str):
            drawer = self.associated_drawer.strip()
            self.associated_drawer = drawer or None


_LEVEL_PATTERNS = [
    (r"\b(grand|special|super)\b", 0),
    (r"\bfirst\b|\b1st\b", 1),
    (r"\bsecond\b|\b2nd\b", 2),
    (r"\bthird\b|\b3rd\b", 3),
    (r"\bfourth\b|\b4th\b", 4),
    (r"\bfifth\b|\b5th\b", 5),
    (r"\b(consolation|participation|lucky)\b", 50),
]


def prize_rank(name: str) -> int:
    """Return a rank for a prize name; smaller means higher prize level."""
    lowered = (name or "").lower()
    for pattern, rank in _LEVEL_PATTERNS:
        if re.search(pattern, lowered):
            return rank
    return 99


def normalize(text: Optional[str]) -> str:
    """Normalize a text for case/space-insensitive comparison."""
    if not text:
        return ""
    return " ".join(str(text).split()).lower()

    "Li Si,Technology Department",
    "Wang Wu,Technology Department",
    "Zhao Liu,Technology Department",
    "Sun Qi,Product Department",
    "Zhou Ba,Product Department",
    "Wu Jiu,Product Department",
    "Zheng Shi,Product Department",
    "Liu Yang,Marketing Department",
    "Chen Jing,Marketing Department",
    "Huang Lei,Marketing Department",
    "Yang Fan,HR Department",
    "Xu Li,HR Department",
    "Gao Feng,Finance Department",
    "Lin Na,Finance Department",
    "He Ming,Cheng Ru Group",
    "Guo Tao,Cheng Ru Group",
    "Luo Xue,Cheng Ru Group",
    "Song Yu,Cheng Ru Group",
    "Tang Wei,Operations Department",

_SEPARATOR_RE = re.compile(r"[,;，；]")


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.users: List[User] = []

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.users = users
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error: file '{path}' does not exist"]
        users, errors = self._load_from_path(path)
        self.users = users
        return users, errors

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        lines = None
        last_error = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=encoding) as f:
                    lines = f.read().splitlines()
                break
            except UnicodeDecodeError as exc:
                last_error = exc
                continue
            except OSError as exc:
                return [], [f"File Error: cannot read '{path}': {exc}"]
        if lines is None:
            return [], [f"File Error: cannot decode '{path}': {last_error}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = _SEPARATOR_RE.split(line)
            if len(parts) != 2:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{raw}'"
                )
                continue
            name, dept = parts[0].strip(), parts[1].strip()
            if not name or not dept:
                errors.append(
                    f"Line {idx}: Format Error - name or department group is empty in '{raw}'"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    # --------------------------------------------------------------- validation
    @staticmethod
    def validate_users(users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    @staticmethod
    def get_group_info(users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Holds the ordered prize list (the lottery round sequence)."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity of prize '{name}' must be a positive integer")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def get_prize(self, name) -> Optional[Prize]:
        for p in self.prizes:
            if p.name == name:
                return p
        return None

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User, normalize

CHENG_RU_GROUP = "Cheng Ru Group"
SPECIAL_DRAWER = "Li Bin"
FALLBACK_TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"


def is_cheng_ru(department: str) -> bool:
    key = (department or "").replace(" ", "").lower()
    return key in ("chengrugroup", "chengru")


class LotteryManager:
    """Applies the filtering rules and draws winners."""

    def __init__(self, users, rng=None) -> None:
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self._winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.SystemRandom()

    # ------------------------------------------------------------- rule helpers
    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    @staticmethod
    def _drawer(prize: Prize) -> str:
        return normalize(prize.associated_drawer)

    def is_special_drawer(self, prize: Prize) -> bool:
        return self._drawer(prize) == normalize(SPECIAL_DRAWER)

    def fallback_active(self, prize: Prize) -> bool:
        if self._drawer(prize) != normalize(FALLBACK_TARGET_DRAWER):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def _not_won(self) -> List[User]:
        return [u for u in self.users if not self.is_winner(u)]

    def _is_avoided(self, user: User, prize: Prize) -> bool:
        avoid = {normalize(g) for g in prize.avoid_groups}
        return normalize(user.department) in avoid

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = self._not_won()
        if self.fallback_active(prize):
            forced = [u for u in candidates if is_cheng_ru(u.department)]
            if forced:
                return forced
        if self.is_special_drawer(prize):
            return candidates
        return [u for u in candidates if not self._is_avoided(u, prize)]

    def _get_secondary_pool(self, prize: Prize) -> List[User]:
        """Avoidance group members, drawn only after all other members."""
        if self.is_special_drawer(prize):
            return []
        if self.fallback_active(prize) and any(
            is_cheng_ru(u.department) for u in self._not_won()
        ):
            return []
        return [u for u in self._not_won() if self._is_avoided(u, prize)]

    # ------------------------------------------------------------------ drawing
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = self._get_secondary_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        if self.is_winner(user):  # anti-duplication safety net
            return None
        self.winners.append(user)
        self._winner_names.add(user.name)
        return user

    def draw_round(self, prize: Prize) -> List[User]:
        result = []
        for _ in range(prize.quantity):
            user = self.draw_one(prize)
            if user is None:
                break
            result.append(user)
        self.record_round_result(prize, result)
        return result

    def record_round_result(self, prize: Prize, round_winners) -> None:
        drawer = self._drawer(prize)
        if drawer in {normalize(d) for d in FALLBACK_TRIGGER_DRAWERS}:
            self.last_two_draw_results.append(
                any(is_cheng_ru(u.department) for u in round_winners)
            )
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer:
            self.last_two_draw_results = []

    def describe_rules(self, prize: Prize) -> List[str]:
        lines = []
        drawer = prize.associated_drawer or "(not specified)"
        lines.append(f"Lottery person: {drawer}")
        if self.fallback_active(prize):
            lines.append(
                f"Fallback rule: previous drawers ({', '.join(FALLBACK_TRIGGER_DRAWERS)}) "
                f"drew no '{CHENG_RU_GROUP}' members; pool is limited to '{CHENG_RU_GROUP}' members"
            )
        elif self.is_special_drawer(prize):
            lines.append(
                f"Special rule: drawer is {SPECIAL_DRAWER}, only previous winners are excluded "
                "(avoidance groups not applied)"
            )
        else:
            if prize.avoid_groups:
                lines.append(
                    "Avoidance groups: " + ", ".join(prize.avoid_groups)
                    + " (drawn only after all other members)"
                )
            else:
                lines.append("Avoidance groups: none")
        lines.append("Previous winners are excluded")
        return lines

"""Result recording, summarizing and exporting."""

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime
from typing import Dict, List, Tuple

from lottery_system.models import Prize, User, prize_rank

TIME_FORMAT = "%Y-%m-%d %H:%M:%S"


@dataclass
class Winner:
    user: User
    prize_name: str
    win_time: datetime = field(default_factory=datetime.now)

    def __str__(self) -> str:
        return (f"Congratulations {self.user.name} ({self.user.department}) "
                f"for winning {self.prize_name}")


def _prize_name(prize) -> str:
    return prize.name if isinstance(prize, Prize) else str(prize)


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []

    def add_winner(self, user, prize):
        if any(w.user.name == user.name for w in self.winners):
            return None
        winner = Winner(user, _prize_name(prize))
        self.winners.append(winner)
        return winner

    def get_round_winners(self, prize) -> List[Winner]:
        name = _prize_name(prize)
        return [w for w in self.winners if w.prize_name == name]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda t: (prize_rank(_prize_name(t[1])), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, prize in indexed:
            name = _prize_name(prize)
            items = [w for w in self.winners if w.prize_name == name]
            items.sort(key=lambda w: w.win_time)  # stable: keeps insertion order
            result[name] = items
        return result

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for name, items in self.get_winners_by_prize(prizes).items():
                    for w in items:
                        writer.writerow([w.user.name, w.user.department, name,
                                         w.win_time.strftime(TIME_FORMAT)])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        raise SystemExit(0)


class LotteryApp:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None
        self.finished_rounds = set()

    # ------------------------------------------------------------------- lists
    def _set_users(self, users) -> None:
        self.users = list(users)
        self.lottery = LotteryManager(self.users)
        self.result_manager = ResultManager()
        self.finished_rounds = set()
        print(f"{len(self.users)} valid participants loaded.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        for e in errors:
            print("  " + e)
        valid, dups = self.list_manager.validate_users(users)
        self._set_users(valid)

    def upload_custom(self) -> None:
        path = ask("Enter txt list path (each line 'Name,Department Group'): ")
        users, errors = self.list_manager.load_custom_list(path)
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Problems detected in the list:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate name: {d.name} ({d.department})")
            if not valid:
                print("No valid data. Upload terminated.")
                return
            choice = ask("Input 'r' to remove invalid data and continue, 't' to terminate upload: ").lower()
            if choice != "r":
                print("Upload terminated.")
                return
        if not valid:
            print("No valid data found. Upload terminated.")
            return
        self._set_users(valid)

    def show_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Valid groups:")
        for i, (dept, cnt) in enumerate(info.items(), 1):
            print(f"  {i}. {dept}: {cnt} people")

    # ------------------------------------------------------------------ prizes
    def load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prizes loaded.")

    def add_custom_prize(self) -> None:
        name = ask("Prize name: ")
        qty = ask("Quantity: ")
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid = []
        if groups:
            print("Groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = ask("Avoidance groups (numbers or names separated by commas, empty for none): ")
        for part in raw.replace("，", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                avoid.append(groups[int(part) - 1])
            else:
                avoid.append(part)
        drawer = ask("Associated lottery person (empty for none): ")
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer or None)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            avoid = ", ".join(p.avoid_groups) or "none"
            status = " [done]" if p.name in self.finished_rounds else ""
            print(f"  {i}. {p.name} x{p.quantity} | avoid: {avoid} | "
                  f"drawer: {p.associated_drawer or '-'}{status}")

    def reorder(self) -> None:
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.show_prizes()
        else:
            print("Invalid order: must be a permutation of all serial numbers.")

    # ----------------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.users or self.lottery is None:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        pending = [p for p in prizes if p.name not in self.finished_rounds]
        if not pending:
            print("All rounds have been completed.")
        for prize in pending:
            self._run_round(prize)
        if all(p.name in self.finished_rounds for p in prizes):
            print("\nAll rounds completed.")
            self.show_summary()

    def _run_round(self, prize) -> None:
        print("\n" + "=" * 60)
        print(f"Round: {prize.name} | Quantity: {prize.quantity}")
        for line in self.lottery.describe_rules(prize):
            print("  Rule: " + line)
        print("Commands: Enter=draw next, 'auto'=draw remaining, 'pause', 'continue', 'terminate'")
        round_winners = []
        idx = 1
        paused = False
        auto = False
        while idx <= prize.quantity:
            if paused:
                cmd = ask("[Paused] input 'continue' to resume or 'terminate' to end this round: ").lower()
                if cmd == "continue":
                    paused = False
                    print("Lottery resumed.")
                elif cmd == "terminate":
                    print("Round terminated.")
                    break
                else:
                    print("Lottery is paused.")
                continue
            if not auto:
                cmd = ask(f"Next: {ordinal(idx)} winner > ").lower()
                if cmd == "pause":
                    paused = True
                    print("Lottery paused.")
                    continue
                if cmd == "terminate":
                    print("Round terminated.")
                    break
                if cmd == "auto":
                    auto = True
                elif cmd not in ("", "draw", "next", "continue"):
                    print("Unknown command.")
                    continue
            print(f"Currently drawing the {ordinal(idx)} winner...")
            user = self.lottery.draw_one(prize)
            if user is None:
                print("No eligible participants left for this prize.")
                break
            winner = self.result_manager.add_winner(user, prize)
            if winner is None:
                print("Duplicate winner detected, skipped.")
                continue
            print(str(winner))
            round_winners.append(user)
            idx += 1
        self.lottery.record_round_result(prize, round_winners)
        self.finished_rounds.add(prize.name)
        print(f"--- Results of {prize.name} ---")
        if not round_winners:
            print("  (no winners)")
        for i, u in enumerate(round_winners, 1):
            print(f"  {i}. {u.name} ({u.department})")

    def show_summary(self) -> None:
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        if not any(data.values()):
            print("No winners yet.")
            return
        for name, items in data.items():
            print(f"[{name}]")
            for w in items:
                print(f"  {w.user.name} ({w.user.department}) - {name} - "
                      f"{w.win_time.strftime('%Y-%m-%d %H:%M:%S')}")

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # --------------------------------------------------------------------- run
    def run(self) -> None:
        actions = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.upload_custom),
            "3": ("View groups", self.show_groups),
            "4": ("Load preset prizes", self.load_default_prizes),
            "5": ("Add custom prize", self.add_custom_prize),
            "6": ("View prizes", self.show_prizes),
            "7": ("Reorder prizes", self.reorder),
            "8": ("Start lottery", self.start_lottery),
            "9": ("View winner summary", self.show_summary),
            "10": ("Export results", self.export),
        }
        print("=== Team Building Smart Lottery System ===")
        while True:
            print()
            for key, (label, _) in actions.items():
                print(f"{key}. {label}")
            print("0. Exit")
            choice = ask("Select: ")
            if choice == "0":
                print("Bye.")
                return
            action = actions.get(choice)
            if action:
                action[1]()
            else:
                print("Invalid choice.")


def main() -> None:
    LotteryApp().run()


if __name__ == "__main__":
    main()

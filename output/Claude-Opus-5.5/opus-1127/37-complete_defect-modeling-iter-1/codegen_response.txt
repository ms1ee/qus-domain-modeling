"""Core data models for the lottery system."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A participant of the lottery."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if isinstance(self.associated_drawer, str):
            drawer = self.associated_drawer.strip()
            self.associated_drawer = drawer if drawer else None

    def __str__(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "None"
        return (f"{self.name} | Quantity: {self.quantity} | "
                f"Avoid groups: {avoid} | Drawer: {drawer}")

import os
from typing import Dict, List, Tuple

from lottery_system.models import User

DEFAULT_LIST_DATA = """Zhang San,Technology Department
Li Si,Technology Department
Wang Wu,Technology Department
Zhao Liu,Product Department
Qian Qi,Product Department
Sun Ba,Product Department
Zhou Jiu,Marketing Department
Wu Shi,Marketing Department
Zheng Yi,Marketing Department
Feng Er,Human Resources Department
Chen San,Human Resources Department
Chu Si,Finance Department
Wei Wu,Finance Department
Jiang Liu,Cheng Ru group
Shen Qi,Cheng Ru group
Han Ba,Cheng Ru group
Yang Jiu,Cheng Ru group
Zhu Shi,Operations Department
Qin Yi,Operations Department
You Er,Operations Department"""


class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_DATA.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        lines = None
        for enc in ("utf-8-sig", "gbk", "latin-1"):
            try:
                with open(path, "r", encoding=enc) as f:
                    lines = f.read().splitlines()
                break
            except (UnicodeDecodeError, LookupError):
                continue
            except OSError as e:
                return [], [f"File Error: cannot read file {path}: {e}"]
        if lines is None:
            return [], [f"File Error: cannot decode file {path}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            if u.name in seen:
                duplicates.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

"""Prize library management."""

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], None),
    ("Second Prize - Shopping Cards", 2, [], None),
    ("First Prize - 1 Day Annual Leave", 1, [], None),
]


class PrizeManager:
    """Manages the prize library and the round order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name cannot be empty")
        name = name.strip()
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be positive")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups: List[str] = []
        for g in (avoid_groups or []):
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine and drawing logic."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User

SPECIAL_PERSON = "Li Bin"
PROTECTED_GROUP = "Cheng Ru group"
TRIGGER_PERSONS = ("Chen Qixian", "Jin Duo")
TARGET_PERSON = "Ma Jiaqi"


def _norm(s: Optional[str]) -> str:
    return " ".join((s or "").strip().lower().split())


def is_protected_group(department: str) -> bool:
    return "cheng ru" in _norm(department)


class LotteryManager:
    """Applies lottery rules and draws winners."""

    def __init__(self, users) -> None:
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []

    # ---- rule engine -------------------------------------------------
    def _fallback_active(self, prize: Prize) -> bool:
        return (_norm(prize.associated_drawer) == _norm(TARGET_PERSON)
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if u.name not in self.winner_names]
        if self._fallback_active(prize):
            protected = [u for u in base if is_protected_group(u.department)]
            if protected:
                return protected
        if _norm(prize.associated_drawer) == _norm(SPECIAL_PERSON):
            return base
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in base if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        drawer = prize.associated_drawer
        if self._fallback_active(prize):
            parts.append(f"Fallback active: pool limited to '{PROTECTED_GROUP}' members")
        elif _norm(drawer) == _norm(SPECIAL_PERSON):
            parts.append(f"Special drawer {SPECIAL_PERSON}: avoidance groups not applied")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        if drawer:
            parts.append(f"Drawer: {drawer}")
        return "; ".join(parts)

    # ---- drawing -----------------------------------------------------
    def draw_winner(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self.mark_winner(winner)
        return winner

    def mark_winner(self, user: User) -> None:
        if user.name not in self.winner_names:
            self.winner_names.add(user.name)
            self.winners.append(user)

    def record_round_result(self, prize: Prize, round_winners) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(p) for p in TRIGGER_PERSONS}:
            contains = any(is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(TARGET_PERSON):
            self.last_two_draw_results = []

    def draw_round(self, prize: Prize) -> List[User]:
        """Non-interactive round draw."""
        result: List[User] = []
        for _ in range(prize.quantity):
            w = self.draw_winner(prize)
            if w is None:
                break
            result.append(w)
        self.record_round_result(prize, result)
        return result

"""Result storage, display grouping and export."""

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User


class Winner:
    def __init__(self, user: User, prize: Prize, order: int) -> None:
        self.user = user
        self.prize = prize
        self.order = order

    def __repr__(self) -> str:
        return f"Winner({self.order}, {self.user.name}, {self.user.department}, {self.prize.name})"


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> Winner:
        w = Winner(user, prize, len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {p.name: [] for p in prizes}
        for w in self.winners:
            result.setdefault(w.prize.name, []).append(w)
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for prize_name, ws in grouped.items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, prize_name])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _input(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        return "0"


class LotteryCLI:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None

    # ---------- list ----------
    def _accept_users(self, users, errors) -> None:
        if errors:
            print("Detected invalid data:")
            for e in errors:
                print("  " + e)
        valid, dups = self.list_manager.validate_users(users)
        if dups:
            print("Detected duplicate names:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
        if errors or dups:
            choice = _input("Remove invalid data and continue (r) or terminate upload (t)? ").strip().lower()
            if choice != "r":
                print("Upload terminated.")
                return
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        self.lottery = LotteryManager(self.users)
        self.result_manager = ResultManager()
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._accept_users(users, errors)

    def load_custom(self) -> None:
        path = _input("Enter txt file path: ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            return
        self._accept_users(users, errors)

    def view_groups(self) -> None:
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        print("Groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    # ---------- prizes ----------
    def load_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prizes loaded.")

    def add_prize(self) -> None:
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = _input("Avoid groups (numbers or names, comma separated, empty for none): ").strip()
        avoid = []
        for token in raw.replace("，", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                avoid.append(groups[int(token) - 1])
            else:
                avoid.append(token)
        drawer = _input("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print("Prize added.")
        except ValueError as e:
            print(f"Error: {e}")

    def view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")

    def reorder(self) -> None:
        self.view_prizes()
        raw = _input("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        print("Order updated." if self.prize_manager.reorder_prizes(order) else "Invalid order.")

    # ---------- lottery ----------
    def start_lottery(self) -> None:
        if not self.lottery:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        for rnd, prize in enumerate(prizes, 1):
            print(f"\n===== Round {rnd}: {prize.name} (Quantity: {prize.quantity}) =====")
            print("Rules: " + self.lottery.describe_rules(prize))
            round_winners = []
            terminated = False
            for i in range(1, prize.quantity + 1):
                cmd = _input("Press Enter to draw, or type pause/terminate: ").strip().lower()
                while cmd == "pause":
                    print("Lottery paused.")
                    cmd = _input("Type 'continue' to resume or 'terminate' to skip round: ").strip().lower()
                    if cmd == "continue":
                        print("Lottery resumed.")
                        cmd = ""
                    elif cmd != "terminate":
                        cmd = "pause"
                if cmd == "terminate":
                    print("Current round terminated.")
                    terminated = True
                    break
                print(f"Currently drawing the {i}th winner...")
                w = self.lottery.draw_winner(prize)
                if w is None:
                    print("No eligible participants left for this prize.")
                    break
                self.result_manager.add_winner(w, prize)
                round_winners.append(w)
                print(f"Congratulations {w.name} ({w.department}) for winning {prize.name}")
            self.lottery.record_round_result(prize, round_winners)
            print(f"--- Round {rnd} results{' (terminated)' if terminated else ''} ---")
            for k, w in enumerate(round_winners, 1):
                print(f"  {k}. {w.name} ({w.department})")
        self.show_summary()

    def show_summary(self) -> None:
        print("\n===== Complete Winner List =====")
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        for prize_name, ws in grouped.items():
            for w in ws:
                print(f"  {w.user.name} | {w.user.department} | {prize_name}")
        if not self.result_manager.winners:
            print("  No winners yet.")

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self) -> None:
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_prizes, "5": self.add_prize, "6": self.view_prizes,
            "7": self.reorder, "8": self.start_lottery, "9": self.show_summary,
            "10": self.export,
        }
        while True:
            print("\n==== Team Building Smart Lottery System ====")
            print("1. Load default list   2. Upload custom list   3. View groups")
            print("4. Load preset prizes  5. Add custom prize     6. View prizes")
            print("7. Reorder prizes      8. Start lottery        9. View winners")
            print("10. Export results     0. Exit")
            choice = _input("Select: ").strip()
            if choice == "0":
                print("Goodbye.")
                break
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid choice.")


def main() -> None:
    LotteryCLI().run()


if __name__ == "__main__":
    main()

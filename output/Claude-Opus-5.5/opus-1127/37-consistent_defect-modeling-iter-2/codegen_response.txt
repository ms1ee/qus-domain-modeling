"""Core data models."""
from dataclasses import dataclass, field
from typing import List, Optional

SPECIAL_DRAWER = "Li Bin"
PROTECTED_GROUP = "Cheng Ru group"
MONITORED_DRAWERS = ["Chen Qixian", "Jin Duo"]
FALLBACK_DRAWER = "Ma Jiaqi"


def normalize(text: Optional[str]) -> str:
    return (text or "").strip().lower()


@dataclass
class User:
    name: str
    department: str

    def __post_init__(self):
        self.name = (self.name or "").strip()
        self.department = (self.department or "").strip()

    def __hash__(self):
        return hash((self.name, self.department))

    def __str__(self):
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        self.name = (self.name or "").strip()
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [g.strip() for g in self.avoid_groups if g and g.strip()]
        if self.associated_drawer is not None:
            self.associated_drawer = self.associated_drawer.strip() or None

    def rule_description(self) -> str:
        parts = ["Previous winners are excluded"]
        drawer = self.associated_drawer
        if drawer and normalize(drawer) == normalize(SPECIAL_DRAWER):
            parts.append(f"lottery person '{drawer}' is special: avoidance groups are not applied")
        elif self.avoid_groups:
            parts.append("avoidance groups: " + ", ".join(self.avoid_groups)
                         + " (drawn only after other members)")
        else:
            parts.append("no avoidance groups")
        if drawer and normalize(drawer) == normalize(FALLBACK_DRAWER):
            parts.append(f"'{PROTECTED_GROUP}' protection may apply if "
                         f"{' and '.join(MONITORED_DRAWERS)} drew no '{PROTECTED_GROUP}' members")
        parts.append(f"lottery person: {drawer if drawer else 'not specified'}")
        return "; ".join(parts)

import os
import re
from collections import OrderedDict
from typing import Dict, List, Tuple

from lottery_system.models import User

DEFAULT_LIST = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Qian Yi", "Human Resources Department"),
    ("Feng Er", "Human Resources Department"),
    ("Chen Yu", "Cheng Ru group"),
    ("Lin Xiao", "Cheng Ru group"),
    ("He Ming", "Cheng Ru group"),
    ("Xu Lei", "Finance Department"),
    ("Gao Fei", "Finance Department"),
]

_SEP = re.compile(r"[,;，；]")


class ListManager:
    def __init__(self):
        self.users: List[User] = []

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users = [User(n, d) for n, d in DEFAULT_LIST]
        return users, []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        if not path or not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"Unable to read file {path}: {exc}"]
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in _SEP.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {idx}: '{raw}' "
                              f"(expected 'Name,Department Group')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users or []:
            if not u.name or not u.department:
                continue
            if u.name in seen:
                duplicates.append(u)
                continue
            seen.add(u.name)
            valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = OrderedDict()
        for u in users or []:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

"""Prize library management."""
from typing import List, Optional

from lottery_system.models import Prize

PRESET_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors = []
        for name, qty in PRESET_PRIZES:
            if any(p.name == name for p in self.prizes):
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            self.prizes.append(Prize(name, qty, [], None))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Prize quantity must be an integer")
        if qty <= 0:
            raise ValueError("Prize quantity must be positive")
        prize = Prize(str(name).strip(), qty, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Result management and export."""
import csv
import os
from datetime import datetime
from typing import Dict, List, Optional


def prize_level(name: str) -> int:
    n = (name or "").lower()
    for key, lvl in (("special", 0), ("grand", 0), ("first", 1), ("second", 2),
                     ("third", 3), ("fourth", 4), ("fifth", 5)):
        if key in n:
            return lvl
    return 99


class Winner:
    def __init__(self, user, prize, time=None, order=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.order = order

    @property
    def prize_name(self):
        return getattr(self.prize, "name", str(self.prize))

    def __repr__(self):
        return f"Winner({self.user.name}, {self.user.department}, {self.prize_name})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def has_winner(self, user) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def add_winner(self, user, prize) -> bool:
        if self.has_winner(user):
            return False
        self.winners.append(Winner(user, prize, datetime.now(), len(self.winners) + 1))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        names = []
        for p in prizes or []:
            nm = getattr(p, "name", str(p))
            if nm not in names:
                names.append(nm)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        ordered = sorted(enumerate(names), key=lambda t: (prize_level(t[1]), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, nm in ordered:
            ws = [w for w in self.winners if w.prize_name == nm]
            ws.sort(key=lambda w: (w.time, w.order))
            result[nm] = ws
        return result

    def export_to_csv(self, prizes, directory: Optional[str] = None):
        directory = directory or os.getcwd()
        path = os.path.join(directory, f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for nm, ws in grouped.items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, nm,
                                         w.time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and process control."""
import random
from typing import Callable, List, Optional

from lottery_system.models import (FALLBACK_DRAWER, MONITORED_DRAWERS, PROTECTED_GROUP,
                                   SPECIAL_DRAWER, Prize, User, normalize)
from lottery_system.services.result_manager import ResultManager


class LotteryManager:
    def __init__(self, users, result_manager: Optional[ResultManager] = None):
        self.users: List[User] = list(users or [])
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winners = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = random.Random()

    # ---- rule engine ----
    def _is_winner(self, user) -> bool:
        return user.name in self.winners or self.result_manager.has_winner(user)

    def _non_winners(self) -> List[User]:
        return [u for u in self.users if not self._is_winner(u)]

    def _fallback_active(self, prize) -> bool:
        return (normalize(prize.associated_drawer) == normalize(FALLBACK_DRAWER)
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize) -> List[User]:
        candidates = self._non_winners()
        if self._fallback_active(prize):
            protected = [u for u in candidates
                         if normalize(u.department) == normalize(PROTECTED_GROUP)]
            if protected:
                return protected
        if normalize(prize.associated_drawer) == normalize(SPECIAL_DRAWER):
            return candidates
        avoid = {normalize(g) for g in prize.avoid_groups}
        return [u for u in candidates if normalize(u.department) not in avoid]

    def _get_deferred_pool(self, prize) -> List[User]:
        """Avoidance-group members, drawn only after all other members are exhausted."""
        avoid = {normalize(g) for g in prize.avoid_groups}
        return [u for u in self._non_winners() if normalize(u.department) in avoid]

    def get_rule_mode(self, prize) -> str:
        if self._fallback_active(prize):
            return f"FALLBACK: pool limited to '{PROTECTED_GROUP}' members"
        return prize.rule_description()

    # ---- drawing ----
    def draw_one(self, prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = self._get_deferred_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winners.add(user.name)
        self.result_manager.add_winner(user, prize)
        return user

    def _finish_round(self, prize, round_winners):
        drawer = normalize(prize.associated_drawer)
        if drawer in {normalize(d) for d in MONITORED_DRAWERS}:
            hit = any(normalize(u.department) == normalize(PROTECTED_GROUP)
                      for u in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == normalize(FALLBACK_DRAWER):
            self.last_two_draw_results = []

    def draw_prize(self, prize) -> List[User]:
        round_winners = []
        for _ in range(prize.quantity):
            u = self.draw_one(prize)
            if u is None:
                break
            round_winners.append(u)
        self._finish_round(prize, round_winners)
        return round_winners

    def run_round(self, prize, round_number: int = 1,
                  input_func: Callable[[str], str] = input,
                  output: Callable[[str], None] = print) -> List[User]:
        output(f"\n===== Round {round_number}: {prize.name} =====")
        output(f"Quantity: {prize.quantity}")
        output(f"Rules: {self.get_rule_mode(prize)}")
        output("Commands: [Enter] draw next, 'pause', 'continue', 'terminate'")
        round_winners: List[User] = []
        i = 0
        paused = False
        while i < prize.quantity:
            try:
                cmd = input_func("Paused> " if paused else
                                 f"Draw #{i + 1} (Enter/pause/terminate)> ").strip().lower()
            except EOFError:
                cmd = "terminate"
            if cmd == "terminate":
                output("Current round terminated, skipping remaining draws.")
                break
            if cmd == "pause":
                paused = True
                output("Lottery paused. Type 'continue' to resume or 'terminate' to end round.")
                continue
            if cmd == "continue":
                if paused:
                    paused = False
                    output("Lottery resumed.")
                continue
            if paused:
                output("Lottery is paused. Type 'continue' or 'terminate'.")
                continue
            if cmd not in ("", "draw", "d"):
                output("Unknown command.")
                continue
            output(f"Currently drawing the {i + 1}th winner...")
            user = self.draw_one(prize)
            if user is None:
                output("No eligible participants left for this prize.")
                break
            round_winners.append(user)
            output(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            i += 1
        self._finish_round(prize, round_winners)
        output(f"--- Round {round_number} results ({prize.name}) ---")
        if not round_winners:
            output("(no winners)")
        for n, u in enumerate(round_winners, 1):
            output(f"{n}. {u.name} ({u.department})")
        return round_winners

"""Command-line interface."""
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager

MENU = """
======== Team Building Smart Lottery ========
1. Load default participant list
2. Upload custom participant list (txt)
3. View department groups
4. Load preset prize library
5. Add custom prize
6. View / reorder prizes
7. Start lottery
8. Show summary of winners
9. Export results
0. Exit
"""


class App:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None

    def _set_users(self, users):
        self.users = users
        self.result_manager = ResultManager()
        self.lottery = LotteryManager(users, self.result_manager)

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        self._set_users(valid)
        print(f"Default list loaded: {len(valid)} participants.")

    def load_custom(self):
        path = input("Enter txt file path: ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        valid, dups = self.list_manager.validate_users(users)
        for e in errors:
            print("  " + e)
        for d in dups:
            print(f"  Duplicate name: {d.name} ({d.department})")
        if not valid:
            print("No valid participants loaded.")
            return
        if errors or dups:
            ch = input("Invalid data found. 'r' remove invalid data and continue, "
                       "'t' terminate upload: ").strip().lower()
            if ch != "r":
                print("Upload terminated.")
                return
        self._set_users(valid)
        print(f"Custom list loaded: {len(valid)} participants.")

    def view_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"{i}. {g}: {c} people")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            print(f"{i}. {p.name} x{p.quantity} | avoid: {', '.join(p.avoid_groups) or '-'} "
                  f"| drawer: {p.associated_drawer or '-'}")
        return prizes

    def add_prize(self):
        name = input("Prize name: ").strip()
        qty = input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid = []
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            sel = input("Avoidance groups (numbers separated by comma, empty for none): ").strip()
            for s in sel.replace("，", ",").split(","):
                s = s.strip()
                if s.isdigit() and 1 <= int(s) <= len(groups):
                    if groups[int(s) - 1] not in avoid:
                        avoid.append(groups[int(s) - 1])
                elif s:
                    print(f"  Ignored invalid selection: {s}")
        drawer = input("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print("Prize added.")
        except ValueError as e:
            print(f"Error: {e}")

    def reorder(self):
        prizes = self.show_prizes()
        if not prizes:
            return
        s = input("New order (serial numbers separated by space/comma, empty to keep): ").strip()
        if not s:
            return
        parts = [p for p in s.replace(",", " ").split() if p]
        if self.prize_manager.reorder_prizes(parts):
            print("Prizes reordered.")
            self.show_prizes()
        else:
            print("Invalid order.")

    def start(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        for i, p in enumerate(prizes, 1):
            self.lottery.run_round(p, i)
        print("\nAll rounds completed.")
        self.summary()

    def summary(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n======== Winner Summary ========")
        any_w = False
        for nm, ws in grouped.items():
            for w in ws:
                any_w = True
                print(f"{w.user.name},{w.user.department},{nm},"
                      f"{w.time.strftime('%Y-%m-%d %H:%M:%S')}")
        if not any_w:
            print("(no winners)")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {"1": self.load_default, "2": self.load_custom, "3": self.view_groups,
                   "4": self._load_prizes, "5": self.add_prize, "6": self.reorder,
                   "7": self.start, "8": self.summary, "9": self.export}
        while True:
            print(MENU)
            try:
                ch = input("Choose: ").strip()
            except EOFError:
                break
            if ch == "0":
                break
            act = actions.get(ch)
            if act:
                try:
                    act()
                except EOFError:
                    break
            else:
                print("Invalid choice.")
        print("Bye.")

    def _load_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print(e)
        print("Preset prizes loaded.")


def main():
    App().run()


if __name__ == "__main__":
    main()

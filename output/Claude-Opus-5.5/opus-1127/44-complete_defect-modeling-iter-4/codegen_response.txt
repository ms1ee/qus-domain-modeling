"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
The file has the following structure::

    {
        "stations": [{"name": str, "lines": [str, ...]}, ...],
        "connections": [{"source": str, "dest": str,
                         "distance": int (meters), "line": str}, ...]
    }
"""

import json
import os

from models import Edge, Graph

# Absolute path of the bundled data file; valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    Raises:
        FileNotFoundError: if the file does not exist.
        ValueError: if the file content is not valid JSON or lacks the
            required top-level keys.
    """
    if path is None:
        path = DATA_FILE_PATH
    if not os.path.isfile(path):
        raise FileNotFoundError(f"Subway data file not found: {path}")
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Subway data file is not valid JSON: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise ValueError("'stations' and 'connections' must be lists")
    return data


def build_graph(data):
    """Build a weighted undirected graph from raw subway data.

    Returns:
        (graph, station_to_number, number_to_station)
        - graph: :class:`models.Graph`
        - station_to_number: dict mapping station name -> station number
        - number_to_station: list where index is the station number
    """
    graph = Graph()
    station_to_number = {}
    number_to_station = []

    def number_of(name):
        if name not in station_to_number:
            station_to_number[name] = len(number_to_station)
            number_to_station.append(name)
            graph.add_vertex(station_to_number[name])
        return station_to_number[name]

    declared_lines = {}
    for station in data.get("stations", []):
        name = station.get("name")
        if not name:
            continue
        number_of(name)
        declared_lines[name] = list(station.get("lines") or [])

    connection_lines = {}
    for conn in data.get("connections", []):
        src, dst = conn.get("source"), conn.get("dest")
        dist, line = conn.get("distance"), conn.get("line")
        if not src or not dst:
            continue
        if not isinstance(dist, (int, float)) or isinstance(dist, bool) or dist < 0:
            continue
        s, d = number_of(src), number_of(dst)
        graph.add_edge(Edge(s, d, dist, line))
        for n in (src, dst):
            connection_lines.setdefault(n, [])
            if line is not None and line not in connection_lines[n]:
                connection_lines[n].append(line)

    for name in number_to_station:
        lines = declared_lines.get(name) or connection_lines.get(name, [])
        graph.set_station_lines(name, lines)

    return graph, station_to_number, number_to_station

"""
Core data structures: Edge, Graph and simple domain records.
"""

from dataclasses import dataclass, field
from typing import List


class Edge:
    """A directed view of a connection between two stations.

    Attributes:
        source: station number of the start of the edge
        dest: station number of the end of the edge
        distance: distance in meters
        weight: same as distance (used by the shortest path algorithm)
        line: name of the line this connection belongs to
    """

    def __init__(self, source, dest, distance, line):
        if distance is None or distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    @property
    def from_(self):
        return self.source

    @property
    def to(self):
        return self.dest

    def other(self, v):
        """Return the endpoint that is not ``v``."""
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError(f"Vertex {v} is not an endpoint of this edge")

    def reversed(self):
        """Return a new edge in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.distance, self.line) == (
            other.source, other.dest, other.distance, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.distance, self.line))

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}m, {self.line})"


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller. ``adj`` (alias ``graph``)
    maps a station number to the list of outgoing :class:`Edge` objects.
    """

    def __init__(self):
        self.adj = {}
        self.station_lines = {}
        self._edges = []

    @property
    def graph(self):
        return self.adj

    def add_vertex(self, v):
        self.adj.setdefault(v, [])

    def has_vertex(self, v):
        return v in self.adj

    def set_station_lines(self, station_name, lines):
        """Record the lines a station belongs to (duplicates removed)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines or []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.source != edge.dest:
            self.adj[edge.dest].append(edge.reversed())
        self._edges.append(edge)

    def adjacent(self, v):
        return self.adj.get(v, [])

    neighbors = adjacent

    def vertices(self):
        return list(self.adj.keys())

    def edges(self):
        """Return each undirected edge once (as originally added)."""
        return list(self._edges)

    @property
    def vertex_count(self):
        return len(self.adj)

    @property
    def edge_count(self):
        return len(self._edges)

    V = vertex_count
    E = edge_count

"""
Shortest path computation (Dijkstra with a heap-based priority queue).

The search runs over (station, current line[, transfers]) states so that
transfers at multi-line interchange stations are detected correctly and
modes such as "least_transfer" or a maximum transfer limit can be honoured.
Complexity is O((V+E) log V) per line-state layer.
"""

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Penalty (in meters) applied per transfer in "comprehensive" mode.
TRANSFER_PENALTY_METERS = 2000
# Weight factor applied to edges on preferred lines.
PREFERRED_LINE_FACTOR = 0.8


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Binary min-heap priority queue.

    Entries are (priority, insertion counter, item); the counter guarantees
    a stable order and avoids comparing items.
    """

    def __init__(self):
        self._heap = []
        self._counter = 0

    def __len__(self):
        return len(self._heap)

    def size(self):
        return len(self._heap)

    def is_empty(self):
        return not self._heap

    def insert(self, priority, item):
        self._heap.append((priority, self._counter, item))
        self._counter += 1
        self._swim(len(self._heap) - 1)

    def min(self):
        if not self._heap:
            raise IndexError("Priority queue is empty")
        p, _, item = self._heap[0]
        return p, item

    def del_min(self):
        if not self._heap:
            raise IndexError("Priority queue is empty")
        heap = self._heap
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sink(0)
        return top[0], top[2]

    def _swim(self, k):
        heap = self._heap
        while k > 0:
            parent = (k - 1) // 2
            if heap[k][:2] < heap[parent][:2]:
                heap[k], heap[parent] = heap[parent], heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        heap = self._heap
        n = len(heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and heap[right][:2] < heap[left][:2]:
                child = right
            if heap[child][:2] < heap[k][:2]:
                heap[k], heap[child] = heap[child], heap[k]
                k = child
            else:
                break


class ShortestPath:
    """Dijkstra shortest path from a single source station.

    Attributes:
        dist_to: dict station number -> actual distance (meters) of the best
            path found under the configured mode.
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        if self.config.mode not in VALID_MODES:
            raise ValueError(f"Unknown path query mode: {self.config.mode}")
        if self.config.max_transfers is not None and self.config.max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        if not graph.has_vertex(source):
            raise ValueError(f"Source station {source} is not in the graph")
        self.dist_to = {}
        self.edge_to = {}          # state -> (previous state, edge)
        self._best_state = {}      # station -> settled state
        self._transfers_to = {}
        self._run()

    # ------------------------------------------------------------------
    def _key(self, weighted, transfers):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, weighted)
        if mode == "comprehensive":
            return (weighted + TRANSFER_PENALTY_METERS * transfers, transfers)
        return (weighted, transfers)

    def _edge_cost(self, edge):
        preferred = self.config.preferred_lines
        if preferred and edge.line in preferred:
            return edge.weight * PREFERRED_LINE_FACTOR
        return edge.weight

    def _run(self):
        max_t = self.config.max_transfers
        track = max_t is not None
        start = (self.source, None, 0)
        # state -> (key, weighted distance, actual distance, transfers)
        info = {start: (self._key(0, 0), 0, 0, 0)}
        pq = MinPQ()
        pq.insert(info[start][0], start)
        settled = set()

        while not pq.is_empty():
            key, state = pq.del_min()
            if state in settled or info[state][0] != key:
                continue
            settled.add(state)
            v, line, _ = state
            _, wd, actual, t = info[state]
            if v not in self.dist_to:
                self.dist_to[v] = actual
                self._best_state[v] = state
                self._transfers_to[v] = t
            for e in self.graph.adjacent(v):
                nt = t + (1 if line is not None and e.line != line else 0)
                if track and nt > max_t:
                    continue
                nstate = (e.dest, e.line, nt if track else 0)
                if nstate in settled:
                    continue
                nwd = wd + self._edge_cost(e)
                nkey = self._key(nwd, nt)
                old = info.get(nstate)
                if old is None or nkey < old[0]:
                    info[nstate] = (nkey, nwd, actual + e.weight, nt)
                    self.edge_to[nstate] = (state, e)
                    pq.insert(nkey, nstate)

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, float("inf"))

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target.

        Returns ([], []) when the target is unreachable.
        """
        if target not in self._best_state:
            return [], []
        edges = []
        state = self._best_state[target]
        while state in self.edge_to:
            prev, edge = self.edge_to[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        path = [self.source] + [e.dest for e in edges]
        return path, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the path to target (0 if none/unreachable)."""
        _, edges = self.path_with_edges(target)
        count = 0
        for prev, cur in zip(edges, edges[1:]):
            if prev.line != cur.line:
                count += 1
        return count

"""
Beijing Subway fare calculation.

Fare structure (distance rounded up to whole kilometers):
    0-6 km   : 3 yuan
    6-12 km  : 4 yuan
    12-22 km : 5 yuan
    22-32 km : 6 yuan
    > 32 km  : +1 yuan for every additional 20 km (or part thereof)
"""

import math

# (max distance in km inclusive, fare in yuan)
FARE_TIERS = [
    (6, 3),
    (12, 4),
    (22, 5),
    (32, 6),
]
EXTENDED_BASE_KM = 32
EXTENDED_BASE_FARE = 6
EXTENDED_INCREMENT_KM = 20
EXTENDED_INCREMENT_FARE = 1

# Estimated time parameters
AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number (meters)")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be finite")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def distance_to_km_ceil(distance):
    """Convert meters to kilometers rounding up to a whole kilometer."""
    _validate_distance(distance)
    return int(math.ceil(distance / 1000.0))


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters."""
    km = distance_to_km_ceil(distance)
    for max_km, fare in FARE_TIERS:
        if km <= max_km:
            return fare
    extra = km - EXTENDED_BASE_KM
    increments = int(math.ceil(extra / EXTENDED_INCREMENT_KM))
    return EXTENDED_BASE_FARE + increments * EXTENDED_INCREMENT_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes for a distance (meters) and transfers."""
    _validate_distance(distance)
    if transfer_count is None or transfer_count < 0:
        raise ValueError("Transfer count cannot be negative")
    ride = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    return round(ride + transfer_count * TRANSFER_TIME_MINUTES, 1)


class FareCalculator:
    """Object-oriented wrapper around the fare structure."""

    rounding_mode = "ceil"

    def calculate(self, distance):
        return Sum_money(distance)

    def format_fare(self, distance):
        return f"Fare: {Sum_money(distance)} yuan"

"""
SubwayNetwork: data management (stations, connections, lines) on top of
the raw JSON data, including data update mechanisms.
"""

import difflib
import json
import re

from data_loader import DATA_FILE_PATH, build_graph, load_subway_data


def _normalize(name):
    return re.sub(r"\s+", " ", str(name).strip()).casefold()


def _natural_key(s):
    return [int(p) if p.isdigit() else p.lower() for p in re.split(r"(\d+)", str(s))]


class SubwayNetwork:
    """The Beijing Subway network, modelled as a weighted graph."""

    def __init__(self, data=None, data_source=DATA_FILE_PATH):
        self.data_source = data_source
        if data is None:
            data = load_subway_data(data_source)
        self.stations = {}  # name -> list of lines (ordered)
        for st in data.get("stations", []):
            name = st.get("name")
            if name:
                self.stations.setdefault(name, [])
                for ln in st.get("lines") or []:
                    if ln not in self.stations[name]:
                        self.stations[name].append(ln)
        self.connections = [dict(c) for c in data.get("connections", [])]
        self.rebuild()

    # ------------------------------------------------------------------
    def to_dict(self):
        return {
            "stations": [{"name": n, "lines": list(l)} for n, l in self.stations.items()],
            "connections": [dict(c) for c in self.connections],
        }

    def rebuild(self):
        """(Re)build the graph and station mapping from current data."""
        self.graph, self.station_to_number, self.number_to_station = build_graph(self.to_dict())
        self._normalized = {_normalize(n): n for n in self.station_to_number}

    def save(self, path=None):
        path = path or self.data_source
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, ensure_ascii=False, indent=2)

    # ------------------------------------------------------------------
    def resolve_station(self, name):
        """Return the official station name for user input, or None."""
        if name is None:
            return None
        name = str(name).strip()
        if name in self.station_to_number:
            return name
        return self._normalized.get(_normalize(name))

    def suggest_stations(self, name, n=3):
        return difflib.get_close_matches(str(name).strip(), list(self.station_to_number), n=n, cutoff=0.6)

    def station_number(self, name):
        return self.station_to_number[name]

    def station_name(self, number):
        return self.number_to_station[number]

    def lines_of(self, station_name):
        return list(self.graph.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return self.graph.is_transfer_station(station_name)

    def transfer_stations(self):
        return [n for n in self.number_to_station if self.is_transfer_station(n)]

    def line_names(self):
        names = set()
        for lines in self.stations.values():
            names.update(lines)
        for c in self.connections:
            if c.get("line"):
                names.add(c["line"])
        return sorted(names, key=_natural_key)

    def resolve_line(self, name):
        if not name:
            return None
        name = str(name).strip()
        lines = self.line_names()
        if name in lines:
            return name
        norm = _normalize(name)
        for ln in lines:
            if _normalize(ln) == norm:
                return ln
        if name.isdigit():
            for cand in (f"Line {name}", f"{name}号线"):
                if cand in lines:
                    return cand
        return None

    def get_line_sequence(self, line):
        """Return the ordered station sequence of a line (best effort)."""
        adj = {}
        for c in self.connections:
            if c.get("line") != line:
                continue
            adj.setdefault(c["source"], [])
            adj.setdefault(c["dest"], [])
            if c["dest"] not in adj[c["source"]]:
                adj[c["source"]].append(c["dest"])
            if c["source"] not in adj[c["dest"]]:
                adj[c["dest"]].append(c["source"])
        if not adj:
            return [n for n, ls in self.stations.items() if line in ls]
        ends = [n for n, nb in adj.items() if len(nb) == 1]
        start = ends[0] if ends else next(iter(adj))
        seq, seen, stack = [], set(), [start]
        while stack:
            v = stack.pop()
            if v in seen:
                continue
            seen.add(v)
            seq.append(v)
            for w in reversed(adj[v]):
                if w not in seen:
                    stack.append(w)
        return seq

    # ------------------------------------------------------------------
    # Data update mechanisms
    # ------------------------------------------------------------------
    def add_line(self, line_name, station_sequence, distances):
        """Add a new line given its station sequence and segment distances (m)."""
        if not line_name:
            raise ValueError("Line name cannot be empty")
        if len(station_sequence) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_sequence) - 1:
            raise ValueError("Number of distances must be number of stations - 1")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for name in station_sequence:
            lines = self.stations.setdefault(name, [])
            if line_name not in lines:
                lines.append(line_name)
        for a, b, d in zip(station_sequence, station_sequence[1:], distances):
            self.connections.append({"source": a, "dest": b, "distance": int(d), "line": line_name})
        self.rebuild()

    def modify_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        if name not in self.stations:
            raise KeyError(f"Station '{name}' does not exist")
        if new_name and new_name != name:
            if new_name in self.stations:
                raise ValueError(f"Station '{new_name}' already exists")
            self.stations = {(new_name if k == name else k): v for k, v in self.stations.items()}
            for c in self.connections:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
            name = new_name
        if lines is not None:
            self.stations[name] = list(dict.fromkeys(lines))
        self.rebuild()

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance between two adjacent stations."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        updated = 0
        for c in self.connections:
            pair = {c.get("source"), c.get("dest")}
            if pair == {station_a, station_b} and (line is None or c.get("line") == line):
                c["distance"] = int(distance)
                updated += 1
        if not updated:
            raise KeyError(f"No connection between '{station_a}' and '{station_b}'")
        self.rebuild()
        return updated

class DataIntegrityChecker:

    def __init__(self, network, representative_stations=None):

        station_names = set(net.stations)
        norm_stations = {_norm(n) for n in net.station_to_number}
        lines = net.line_names()
        norm_lines = {_norm(l) for l in lines}

        # --- coverage ---------------------------------------------------
        missing_lines = [alts[0] for alts in MAIN_LINES if not any(_norm(a) in norm_lines for a in alts)]
        for ln in missing_lines:
            warnings.append(f"Main line '{ln}' not found in data")
        found_rep, missing_rep = [], []
        for alts in self.representative_stations:
            if any(_norm(a) in norm_stations for a in alts):
                found_rep.append(alts[0])
            else:
                missing_rep.append(alts[0])
        for st in missing_rep:
            warnings.append(f"Representative station '{st}' not found in data")
        if not station_names:
            errors.append("No stations defined")

        # --- connections ------------------------------------------------
        seen = Counter()
        dist_by_key = {}
        degree = Counter()
        for i, c in enumerate(net.connections):
            src, dst, dist, line = c.get("source"), c.get("dest"), c.get("distance"), c.get("line")
            label = f"connection #{i} ({src} - {dst})"
            if not src or not dst:
                errors.append(f"{label}: missing source or destination")
                continue
            for s in (src, dst):
                if s not in station_names:
                    errors.append(f"{label}: station '{s}' is not defined in station list")
            if src == dst:
                errors.append(f"{label}: connects a station to itself")
            if not line:
                errors.append(f"{label}: missing line")
            else:
                for s in (src, dst):
                    if s in net.stations and net.stations[s] and line not in net.stations[s]:
                        warnings.append(f"{label}: station '{s}' is not listed on line '{line}'")
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                errors.append(f"{label}: distance is not a number")
            elif dist <= 0:
                errors.append(f"{label}: distance must be positive (got {dist})")
            elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                warnings.append(f"{label}: suspicious distance {dist} m")
            key = (frozenset((src, dst)), line)
            seen[key] += 1
            if seen[key] > 1:
                if dist_by_key.get(key) != dist:
                    errors.append(f"{label}: duplicate connection with inconsistent distance")
                else:
                    warnings.append(f"{label}: duplicate connection")
            dist_by_key.setdefault(key, dist)
            degree[src] += 1
            degree[dst] += 1

        for name in net.stations:
            if not net.stations[name]:
                warnings.append(f"Station '{name}' has no line information")
            if degree[name] == 0:
                warnings.append(f"Station '{name}' has no connections")

        # --- connectivity -----------------------------------------------
        components = self._count_components()
        if components > 1:
            warnings.append(f"Network is split into {components} disconnected components")

        stats = {
            "station_count": len(net.station_to_number),
            "connection_count": len(net.connections),
            "line_count": len(lines),
            "transfer_station_count": len(net.transfer_stations()),
            "representative_stations_found": len(found_rep),
            "representative_stations_total": len(self.representative_stations),
            "components": components,
        }
        self.result = {"ok": not errors, "errors": errors, "warnings": warnings, "stats": stats}
        return self.result

    def _count_components(self):
        g = self.network.graph
        seen, count = set(), 0
        for v in g.vertices():
            if v in seen:
                continue
            count += 1
            stack = [v]
            seen.add(v)
            while stack:
                u = stack.pop()
                for e in g.adjacent(u):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        stack.append(e.dest)
        return count


def format_check_report(result):
    s = result["stats"]
    lines = [
        "===== Data Integrity Check =====",
        f"Stations: {s['station_count']}",
        f"Connections: {s['connection_count']}",
        f"Lines: {s['line_count']}",
        f"Transfer stations: {s['transfer_station_count']}",
        f"Representative stations found: {s['representative_stations_found']}/"
        f"{s['representative_stations_total']}",
        f"Connected components: {s['components']}",
        f"Errors: {len(result['errors'])}",
    ]
    lines += [f"  [ERROR] {e}" for e in result["errors"]]
    lines.append(f"Warnings: {len(result['warnings'])}")
    lines += [f"  [WARN] {w}" for w in result["warnings"]]
    lines.append("Result: " + ("PASSED" if result["ok"] else "FAILED"))
    return "\n".join(lines)

"""
Route construction, path visualization and export (text / JSON).
"""

import json

from fare_calculator import Sum_money, calculate_estimated_time
from models import Route, Transfer


def build_route(network, sp, target):
    """Build a Route from a ShortestPath result. Returns None if unreachable."""
    path, edges = sp.path_with_edges(target)
    if not path:
        return None
    names = [network.station_name(n) for n in path]
    segments, transfers, involved = [], [], []
    for i, e in enumerate(edges):
        if not segments or segments[-1]["line"] != e.line:
            if segments:
                transfers.append(Transfer(names[i], segments[-1]["line"], e.line))
            segments.append({"line": e.line, "stations": [names[i]], "distance": 0})
        segments[-1]["stations"].append(names[i + 1])
        segments[-1]["distance"] += e.distance
        if e.line not in involved:
            involved.append(e.line)
    total = sum(e.distance for e in edges)
    tcount = len(transfers)
    return Route(
        station_sequence=names,
        total_distance=total,
        transfer_count=tcount,
        estimated_time=calculate_estimated_time(total, tcount),
        involved_lines=involved,
        segments=segments,
        transfers=transfers,
    )


def generate_summary(route):
    """Key information: distance (km), fare, transfers, estimated time."""
    return {
        "total_distance_m": route.total_distance,
        "total_distance_km": round(route.total_distance / 1000.0, 2),
        "fare": Sum_money(route.total_distance),
        "transfer_count": route.transfer_count,
        "estimated_time_minutes": route.estimated_time,
    }


def format_distance(meters):
    km = meters / 1000.0
    if meters < 1000:
        return f"{meters} m ({km:.2f} km)"
    return f"{km:.2f} km ({meters} m)"


def build_result(route, query_info, fare=None):
    """Structured result dictionary (used for JSON and text output)."""
    if fare is None:
        fare = Sum_money(route.total_distance)
    return {
        "query": dict(query_info),
        "path": list(route.station_sequence),
        "station_count": len(route.station_sequence),
        "segments": [
            {"line": s["line"], "from": s["stations"][0], "to": s["stations"][-1],
             "stations": list(s["stations"]), "distance": s["distance"]}
            for s in route.segments
        ],
        "distance": {
            "meters": route.total_distance,
            "kilometers": round(route.total_distance / 1000.0, 2),
        },
        "fare": fare,
        "fare_text": f"Fare: {fare} yuan",
        "estimated_time": route.estimated_time,
        "estimated_time_unit": "minutes",
        "transfer_count": route.transfer_count,
        "transfers": [
            {"station": t.station_name, "from_line": t.from_line, "to_line": t.to_line}
            for t in route.transfers
        ],
        "lines": list(route.involved_lines),
    }


def format_text(result):
    q = result["query"]
    out = [
        "=" * 50,
        f"Route: {q.get('start')} -> {q.get('end')}",
        f"Mode: {q.get('mode', 'shortest_distance')}",
        "-" * 50,
        f"Stations ({result['station_count']}): " + " -> ".join(result["path"]),
        "Segments:",
    ]
    transfers = result["transfers"]
    for i, seg in enumerate(result["segments"]):
        stops = len(seg["stations"]) - 1
        out.append(f"  [{seg['line']}] " + " -> ".join(seg["stations"])
                   + f" ({stops} stop{'s' if stops != 1 else ''}, {format_distance(seg['distance'])})")
        if i < len(transfers):
            t = transfers[i]
            out.append(f"  ** Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
    out.append(f"Total distance: {format_distance(result['distance']['meters'])}")
    if transfers:
        out.append(f"Transfers: {result['transfer_count']} (at "
                   + ", ".join(t["station"] for t in transfers) + ")")
    else:
        out.append("Transfers: 0")
    out.append("Lines involved: " + ", ".join(str(l) for l in result["lines"]))
    out.append(f"Estimated time: {result['estimated_time']} min")
    out.append(result["fare_text"])
    out.append("=" * 50)
    return "\n".join(out)


def format_json(result):
    return json.dumps(result, ensure_ascii=False, indent=2)


def render(result, fmt="text"):
    return format_json(result) if fmt == "json" else format_text(result)

"""
Command-line interface for the Beijing Subway fare calculation system.

Usage:
    python main.py [--output_file FILE] [--format text|json]
                   [--mode shortest_distance|least_transfer|comprehensive]
                   [--max_transfers N] [--line_preference "Line 1"]
                   [--check_data]

Enter queries as "Starting Station; Ending Station". Type 'quit' to exit.
"""

import argparse
import json
import sys

from data_checker import DataIntegrityChecker, format_check_report
from data_loader import DATA_FILE_PATH
from fare_calculator import Sum_money
from network import SubwayNetwork
from pathfinding import VALID_MODES, PathfindingConfig, ShortestPath
from report import build_result, build_route, render

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
QUIT_WORDS = {"quit", "exit", "q"}


def parse_args(argv=None):
    parser = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", default=None, help="Export query results to this file")
    parser.add_argument("--format", default="text", choices=["text", "json"], help="Output format")
    parser.add_argument("--mode", default="shortest_distance", choices=list(VALID_MODES),
                        help="Path query mode")
    parser.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line(s), comma separated (e.g. 'Line 1')")
    parser.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    parser.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be a non-negative integer")
    return args


def parse_query(raw):
    """Split 'Start; End' input. Returns (start, end) or raises ValueError."""
    if raw is None:
        raise ValueError(FORMAT_ERROR)
    text = raw.replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2:
        raise ValueError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise ValueError(FORMAT_ERROR)
    return start, end


def _unknown_station_msg(network, kind, name):
    msg = f"{kind} station '{name}' does not exist, please check the station name"
    sugg = network.suggest_stations(name)
    if sugg:
        msg += " (did you mean: " + ", ".join(sugg) + "?)"
    return msg


def process_query(network, raw, config=None):
    """Process one raw query. Returns (True, result_dict) or (False, error message)."""
    config = config or PathfindingConfig()
    try:
        start_in, end_in = parse_query(raw)
    except ValueError as exc:
        return False, str(exc)

    start = network.resolve_station(start_in)
    if start is None:
        return False, _unknown_station_msg(network, "Starting", start_in)
    end = network.resolve_station(end_in)
    if end is None:
        return False, _unknown_station_msg(network, "Ending", end_in)
    if start == end:
        return False, "Starting station cannot be the same as the ending station"

    try:
        sp = ShortestPath(network.graph, network.station_number(start), config)
        route = build_route(network, sp, network.station_number(end))
    except Exception as exc:  # path calculation exception handling
        return False, f"Path calculation error: {exc}"
    if route is None:
        if config.max_transfers is not None:
            return False, (f"No route found from '{start}' to '{end}' "
                           f"within {config.max_transfers} transfer(s)")
        return False, f"No route found from '{start}' to '{end}'"

    try:
        fare = Sum_money(route.total_distance)
    except Exception as exc:  # fare calculation exception handling
        return False, f"Fare calculation error: {exc}"

    query_info = {
        "start": start,
        "end": end,
        "raw_input": raw.strip(),
        "mode": config.mode,
        "max_transfers": config.max_transfers,
        "line_preference": list(config.preferred_lines),
    }
    return True, build_result(route, query_info, fare)


def export_results(path, results, fmt):
    """Write all results so far to the output file."""
    with open(path, "w", encoding="utf-8") as fh:
        if fmt == "json":
            payload = results[0] if len(results) == 1 else results
            json.dump(payload, fh, ensure_ascii=False, indent=2)
        else:
            fh.write("\n\n".join(render(r, "text") for r in results))
        fh.write("\n")


def build_config(network, args):
    preferred = []
    if args.line_preference:
        for item in args.line_preference.split(","):
            item = item.strip()
            if not item:
                continue
            resolved = network.resolve_line(item)
            if resolved is None:
                print(f"Warning: line '{item}' does not exist, preference ignored")
            else:
                preferred.append(resolved)
    return PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                             preferred_lines=preferred)


def main(argv=None):
    args = parse_args(argv)
    try:
        network = SubwayNetwork(data_source=args.data_file)
    except (OSError, ValueError) as exc:
        print(f"Error loading subway data: {exc}")
        return 1

    if args.check_data:
        result = DataIntegrityChecker(network).run()
        print(format_check_report(result))
        return 0 if result["ok"] else 1

    config = build_config(network, args)
    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' to query, or 'quit' to exit.")

    results = []
    while True:
        try:
            raw = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        raw = raw.strip()
        if not raw:
            continue
        if raw.lower() in QUIT_WORDS:
            break
        ok, result = process_query(network, raw, config)
        if not ok:
            print(result)
            continue
        print(render(result, args.format))
        if args.output_file:
            results.append(result)
            try:
                export_results(args.output_file, results, args.format)
                print(f"Result exported to {args.output_file}")
            except OSError as exc:
                print(f"Failed to write output file: {exc}")
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

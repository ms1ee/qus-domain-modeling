"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
``DATA_FILE_PATH`` is an absolute path, so it is valid regardless of the
current working directory.
"""

import json
import os

#: Absolute path of the bundled subway data file (src/subway_data.json).
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway data file cannot be read or is malformed."""


def load_subway_data(path=DATA_FILE_PATH):
    """
    Load the subway network JSON file and return its content as a dict.

    Expected structure::

        {"stations": [{"name": str, "lines": [str]}],
         "connections": [{"source": str, "dest": str,
                          "distance": int, "line": str}]}

    :param path: path of the JSON file (defaults to ``DATA_FILE_PATH``)
    :return: the parsed JSON content (dict)
    :raises FileNotFoundError: if the file does not exist
    :raises DataLoadError: if the file content is not valid subway data
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise DataLoadError(f"Invalid JSON in data file '{path}': {exc}") from exc

    if not isinstance(data, dict):
        raise DataLoadError(f"Data file '{path}' must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise DataLoadError(f"Data file '{path}' has invalid 'stations'/'connections' fields")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file (used by the data update mechanism)."""
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, ensure_ascii=False, indent=2)

"""
Core data structures: Edge, Graph and simple domain objects
(Station, Line, Connection, Transfer, Route).
"""

import math
from dataclasses import dataclass, field
from typing import Dict, List, Optional


class Edge:
    """
    A weighted edge between two stations (identified by station numbers).

    Attributes:
        source:   station number of the origin
        dest:     station number of the destination
        distance: distance in meters
        weight:   same as distance (used by the shortest path algorithm)
        line:     name of the subway line this edge belongs to
    """

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError(f"Edge distance must be numeric, got {distance!r}")
        if math.isnan(distance) or distance < 0:
            raise ValueError(f"Edge distance must be non-negative, got {distance!r}")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def other(self, v):
        """Return the endpoint opposite to ``v``."""
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError(f"Vertex {v} is not an endpoint of this edge")

    def reversed(self):
        """Return a new edge going in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.distance, self.line) == (
            other.source, other.dest, other.distance, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.distance, self.line))


class Graph:
    """
    Weighted undirected graph stored as adjacency lists.

    Station numbers are assigned by the caller. Each undirected connection is
    stored once in ``edges`` and twice in ``adjacency`` (once per direction, so
    that every edge in ``adjacency[v]`` has ``edge.source == v``).
    """

    def __init__(self):
        self.adjacency: Dict[int, List[Edge]] = {}
        self.station_lines: Dict[str, List[str]] = {}
        self.edges: List[Edge] = []

    # ---- vertices -------------------------------------------------------
    def add_vertex(self, v):
        if v not in self.adjacency:
            self.adjacency[v] = []

    @property
    def vertices(self):
        return list(self.adjacency.keys())

    @property
    def vertex_count(self):
        return len(self.adjacency)

    @property
    def edge_count(self):
        return len(self.edges)

    def V(self):
        return self.vertex_count

    def E(self):
        return self.edge_count

    # ---- station lines --------------------------------------------------
    def set_station_lines(self, station_name, lines):
        """Record the lines serving a station (duplicates removed, order kept)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines or []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    # ---- edges ----------------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adjacency[edge.source].append(edge)
        self.adjacency[edge.dest].append(edge.reversed())
        self.edges.append(edge)

    def adj(self, v):
        """Edges leaving vertex ``v`` (each has ``source == v``)."""
        return self.adjacency.get(v, [])

    def degree(self, v):
        return len(self.adjacency.get(v, []))

    def __repr__(self):
        return f"Graph(V={self.vertex_count}, E={self.edge_count})"

"""
Path calculation: priority queue (MinPQ) and Dijkstra's shortest path
(ShortestPath) with support for several query modes, maximum transfer limits
and line preferences.

The search runs over states (station, current line[, transfers used]) so that
transfers at multi-line interchange stations are modeled explicitly.
Complexity is O((V+E) log V) in the number of states/edges.
"""

import itertools
import math

try:
    from models import Edge, Graph  # noqa: F401
except ImportError:  # pragma: no cover - package-style import
    from .models import Edge, Graph  # noqa: F401

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

#: Transfer cost (in equivalent meters) used by the comprehensive mode.
TRANSFER_PENALTY_METERS = 3000
#: Weight multiplier applied to edges on preferred lines.
PREFERRED_LINE_FACTOR = 0.8


class MinPQ:
    """Binary-heap based minimum priority queue of (key, item) pairs."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        self._heap.append((key, next(self._counter), item))
        self._swim(len(self._heap) - 1)

    def del_min(self):
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        heap = self._heap
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sink(0)
        return top[0], top[2]

    def min(self):
        if not self._heap:
            raise IndexError("min from empty priority queue")
        return self._heap[0][0], self._heap[0][2]

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    # ---- heap helpers ---------------------------------------------------
    def _less(self, i, j):
        a, b = self._heap[i], self._heap[j]
        return (a[0], a[1]) < (b[0], b[1])

    def _swim(self, k):
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self._heap[k], self._heap[parent] = self._heap[parent], self._heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        n = len(self._heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and self._less(right, left):
                child = right
            if self._less(child, k):
                self._heap[k], self._heap[child] = self._heap[child], self._heap[k]
                k = child
            else:
                break


class PathfindingConfig:
    """
    Configuration of a path query.

    Attributes:
        mode:            'shortest_distance', 'least_transfer' or 'comprehensive'
        max_transfers:   maximum number of transfers allowed (None = unlimited)
        preferred_lines: list of line names to favour
    """

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None,
                 transfer_penalty=TRANSFER_PENALTY_METERS, preference_factor=PREFERRED_LINE_FACTOR):
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = [p for p in preferred_lines if p]
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class ShortestPath:
    """
    Dijkstra's algorithm from a single source station.

    Attributes:
        source:       source station number
        dist_to:      dict station number -> distance (meters) of the chosen
                      route (math.inf when unreachable)
        edge_to:      dict station number -> last Edge on the chosen route
        transfers_to: dict station number -> number of transfers on that route
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: math.inf for v in graph.vertices}
        self.dist_to[source] = 0
        self.edge_to = {}
        self.transfers_to = {source: 0}
        self._best_state = {}
        self._parent = {}
        self._preferred = set(self.config.preferred_lines)
        self._run()

    # ---- key according to mode -------------------------------------------
    def _key(self, eff, transfers):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, eff)
        if mode == "comprehensive":
            return (eff + transfers * self.config.transfer_penalty, transfers)
        return (eff, transfers)

    def _run(self):
        max_t = self.config.max_transfers
        use_t = max_t is not None
        factor = self.config.preference_factor

        start = (self.source, None, 0) if use_t else (self.source, None)
        labels = {start: (0.0, 0, 0)}  # state -> (effective cost, real distance, transfers)
        self._parent[start] = None
        settled = set()
        pq = MinPQ()
        pq.insert(self._key(0.0, 0), start)

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)
            v, cur_line = state[0], state[1]
            eff, real, t = labels[state]

            if v not in self._best_state:
                self._best_state[v] = state
                self.dist_to[v] = real
                self.transfers_to[v] = t
                parent = self._parent[state]
                if parent is not None:
                    self.edge_to[v] = parent[1]

            for e in self.graph.adj(v):
                nt = t + (1 if cur_line is not None and e.line != cur_line else 0)
                if use_t and nt > max_t:
                    continue
                ns = (e.dest, e.line, nt) if use_t else (e.dest, e.line)
                if ns in settled:
                    continue
                w = e.weight * (factor if e.line in self._preferred else 1.0)
                neff = eff + w
                nk = self._key(neff, nt)
                old = labels.get(ns)
                if old is None or nk < self._key(old[0], old[2]):
                    labels[ns] = (neff, real + e.weight, nt)
                    self._parent[ns] = (state, e)
                    pq.insert(nk, ns)

        if self.source not in self._best_state:
            self._best_state[self.source] = start

    # ---- queries ---------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """
        Return (list of station numbers, list of Edge) from source to target.
        Both lists are empty when target is unreachable.
        """
        if target not in self._best_state:
            return [], []
        edges = []
        state = self._best_state[target]
        while True:
            parent = self._parent.get(state)
            if parent is None:
                break
            state, edge = parent
            edges.append(edge)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the chosen path to ``target``."""
        _, edges = self.path_with_edges(target)
        count = 0
        for prev, cur in zip(edges, edges[1:]):
            if cur.line != prev.line:
                count += 1
        return count

"""
Fare calculation based on the Beijing Subway fare structure.

    0  - 6 km   : 3 yuan
    6  - 12 km  : 4 yuan
    12 - 22 km  : 5 yuan
    22 - 32 km  : 6 yuan
    >= 32 km    : 7 yuan, +1 yuan for every additional 20 km

Boundary values belong to the upper tier (6000 m -> 4 yuan, 12000 -> 5,
22000 -> 6, 32000 -> 7). Fractional distances are rounded up to whole meters.
"""

import math


class FareCalculationError(ValueError, TypeError):
    """Raised for invalid distance input."""


#: (tier name, min meters inclusive, max meters exclusive, fare)
FARE_TIERS = [
    ("Basic", 0, 6000, 3),
    ("Progressive", 6000, 12000, 4),
    ("Progressive", 12000, 22000, 5),
    ("Progressive", 22000, 32000, 6),
]
EXTENDED_THRESHOLD_METERS = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_INCREMENT_METERS = 20000

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


class FareStructure:
    """Container exposing the fare tiers (domain model helper)."""

    tiers = FARE_TIERS
    extended_threshold_km = EXTENDED_THRESHOLD_METERS // 1000
    extended_increment_km = EXTENDED_INCREMENT_METERS // 1000

    def fare(self, distance):
        return Sum_money(distance)


def _normalize_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise FareCalculationError(f"Distance must be a number, got {distance!r}")
    if math.isnan(distance) or math.isinf(distance):
        raise FareCalculationError(f"Distance must be finite, got {distance!r}")
    if distance < 0:
        raise FareCalculationError(f"Distance cannot be negative, got {distance!r}")
    return math.ceil(distance)


def Sum_money(distance):
    """
    Calculate the fare (yuan) for a distance in meters.

    :raises FareCalculationError: (a ValueError/TypeError) for invalid input
    """
    d = _normalize_distance(distance)
    for _name, low, high, fare in FARE_TIERS:
        if low <= d < high:
            return fare
    extra = (d - EXTENDED_THRESHOLD_METERS) // EXTENDED_INCREMENT_METERS
    return EXTENDED_BASE_FARE + int(extra)


def get_fare_tier(distance):
    """Return the tier name ('Basic', 'Progressive' or 'Extended')."""
    d = _normalize_distance(distance)
    for name, low, high, _fare in FARE_TIERS:
        if low <= d < high:
            return name
    return "Extended"


def calculate_estimated_time(distance, transfer_count):
    """
    Estimate travel time in minutes from distance (meters) and transfer count.
    Uses an average speed of 35 km/h plus 5 minutes per transfer.
    """
    if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance < 0 \
            or math.isnan(distance):
        raise FareCalculationError(f"Invalid distance {distance!r}")
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise FareCalculationError(f"Invalid transfer count {transfer_count!r}")
    ride = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    return round(ride + transfer_count * TRANSFER_TIME_MINUTES, 1)

def normalize_name(name):
    """Normalization used for tolerant name matching."""
    return re.sub(r"[\s'’`\-_·.]", "", str(name)).lower()


def build_station_to_number(data):
    """Map station names to numbers (order of appearance, starting at 0)."""
    station_to_number = {}
    for st in data.get("stations", []):
        name = st.get("name") if isinstance(st, dict) else None
        if name and name not in station_to_number:
            station_to_number[name] = len(station_to_number)
    for conn in data.get("connections", []):
        for key in ("source", "dest"):
            name = conn.get(key)
            if name and name not in station_to_number:
                station_to_number[name] = len(station_to_number)
    return station_to_number


def build_graph_dict(data, station_to_number):
    """Adjacency-list dict: number -> list of (neighbor number, distance, line)."""
    graph = {n: [] for n in station_to_number.values()}
    for conn in data.get("connections", []):
        s, d = conn.get("source"), conn.get("dest")
        dist = conn.get("distance")
        if s not in station_to_number or d not in station_to_number:
            continue
        if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
            continue
        a, b = station_to_number[s], station_to_number[d]
        graph[a].append((b, dist, conn.get("line")))
        graph[b].append((a, dist, conn.get("line")))
    return graph

class SubwayNetwork:
    """The Beijing Subway network built from subway_data.json."""

    def __init__(self, data=None, data_file=None):
        self.data_file = data_file or DATA_FILE_PATH
        self.data = data if data is not None else load_subway_data(self.data_file)
        self.rebuild()

    # ---- construction ------------------------------------------------------
    def rebuild(self):
        """(Re)build station numbering and graph structures from self.data."""
        data = self.data
        self.station_to_number = build_station_to_number(data)
        self.number_to_station = {n: s for s, n in self.station_to_number.items()}
        lines_of = defaultdict(list)
        for st in data.get("stations", []):
            if isinstance(st, dict) and st.get("name"):
                for ln in st.get("lines") or []:
                    if ln not in lines_of[st["name"]]:
                        lines_of[st["name"]].append(ln)
        for conn in data.get("connections", []):
            ln = conn.get("line")
            for key in ("source", "dest"):
                name = conn.get(key)
                if name and ln and ln not in lines_of[name]:
                    lines_of[name].append(ln)
        self.station_lines = {name: list(lines_of.get(name, [])) for name in self.station_to_number}

        g = Graph()
        for name, num in self.station_to_number.items():
            g.add_vertex(num)
            g.set_station_lines(name, self.station_lines[name])
        for conn in data.get("connections", []):
            s, d, dist = conn.get("source"), conn.get("dest"), conn.get("distance")
            if s not in self.station_to_number or d not in self.station_to_number or s == d:
                continue
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                continue
            g.add_edge(Edge(self.station_to_number[s], self.station_to_number[d], dist, conn.get("line")))
        self.graph_obj = g
        self.graph = build_graph_dict(data, self.station_to_number)
        self._norm_index = {normalize_name(n): n for n in self.station_to_number}
        self._line_norm_index = {normalize_name(l): l for l in self.all_lines()}

    # ---- lookup ----------------------------------------------------------
    def all_lines(self):
        lines = []
        for ls in self.station_lines.values():
            for l in ls:
                if l not in lines:
                    lines.append(l)
        return lines

    def get_station(self, name):
        if name not in self.station_to_number:
            return None
        return Station(name, self.station_to_number[name], list(self.station_lines.get(name, [])))

    def is_transfer_station(self, name):
        return len(self.station_lines.get(name, [])) > 1

    def transfer_stations(self):
        return [n for n in self.station_to_number if self.is_transfer_station(n)]

    def resolve_station(self, raw):
        """Standardize a user-entered station name; return canonical name or None."""
        if raw is None:
            return None
        name = " ".join(str(raw).strip().split())
        if not name:
            return None
        if name in self.station_to_number:
            return name
        n = normalize_name(name)
        if n in self._norm_index:
            return self._norm_index[n]
        for suffix in ("站", "station", "subwaystation"):
            if n.endswith(suffix) and n[: -len(suffix)] in self._norm_index:
                return self._norm_index[n[: -len(suffix)]]
        return None

    def suggest_stations(self, raw, n=3):
        return difflib.get_close_matches(str(raw).strip(), list(self.station_to_number), n=n, cutoff=0.6)

    def resolve_line(self, raw):
        if raw is None:
            return None
        n = normalize_name(raw)
        if n in self._line_norm_index:
            return self._line_norm_index[n]
        if n.isdigit() and normalize_name(f"Line {n}") in self._line_norm_index:
            return self._line_norm_index[normalize_name(f"Line {n}")]
        return None

    def get_line_stations(self, line):
        """Return the ordered station sequence of a line (derived from connections)."""
        adj = defaultdict(list)
        for conn in self.data.get("connections", []):
            if conn.get("line") == line:
                s, d = conn.get("source"), conn.get("dest")
                if d not in adj[s]:
                    adj[s].append(d)
                if s not in adj[d]:
                    adj[d].append(s)
        if not adj:
            return [s for s, ls in self.station_lines.items() if line in ls]
        ends = [s for s, nb in adj.items() if len(nb) == 1]
        start = ends[0] if ends else next(iter(adj))
        seq, seen = [], set()
        stack = [start]
        while stack:
            cur = stack.pop()
            if cur in seen:
                continue
            seen.add(cur)
            seq.append(cur)
            for nb in reversed(adj[cur]):
                if nb not in seen:
                    stack.append(nb)
        return seq

    def get_lines(self):
        return [Line(l, self.get_line_stations(l)) for l in self.all_lines()]

    # ---- integrity check -------------------------------------------------
    def check_data(self):
        r = DataIntegrityReport()
        data = self.data
        stations = data.get("stations", [])
        conns = data.get("connections", [])
        declared = set()
        r.stats = {
            "Stations": len(self.station_to_number),
            "Connections": len(conns),
            "Lines": len(self.all_lines()),
            "Transfer stations": len(self.transfer_stations()),
        }

        # Station coverage
        if not stations:
            r.station_coverage_valid = False
            r.errors.append("No station data found")
        for st in stations:
            if not isinstance(st, dict) or not st.get("name"):
                r.station_coverage_valid = False
                r.errors.append(f"Invalid station entry: {st!r}")
                continue
            if st["name"] in declared:
                r.warnings.append(f"Duplicate station entry '{st['name']}'")
            declared.add(st["name"])
            if not st.get("lines"):
                r.warnings.append(f"Station '{st['name']}' has no line information")
        line_counts = defaultdict(int)
        for ls in self.station_lines.values():
            for l in ls:
                line_counts[normalize_name(l)] += 1
        for main, aliases in MAIN_LINES.items():
            found = 0
            for norm_line, cnt in line_counts.items():
                for a in [normalize_name(main)] + [normalize_name(x) for x in aliases]:
                    if norm_line == a or (norm_line.startswith(a) and not norm_line[len(a):len(a) + 1].isdigit()):
                        found = max(found, cnt)
            if found < 2:
                r.station_coverage_valid = False
                r.errors.append(f"Main line '{main}' is missing or has fewer than 2 stations")
        norm_names = set(self._norm_index)
        for en, zh in REPRESENTATIVE_STATIONS:
            if normalize_name(en) not in norm_names and normalize_name(zh) not in norm_names:
                r.warnings.append(f"Representative station '{en}' ({zh}) not found")

        # Connection integrity
        seen_pairs = {}
        for i, c in enumerate(conns):
            s, d, line = c.get("source"), c.get("dest"), c.get("line")
            if not s or not d:
                r.connection_integrity_valid = False
                r.errors.append(f"Connection #{i} missing source/dest")
                continue
            for name in (s, d):
                if name not in declared:
                    r.connection_integrity_valid = False
                    r.errors.append(f"Connection #{i} references unknown station '{name}'")
            if s == d:
                r.connection_integrity_valid = False
                r.errors.append(f"Connection #{i} is a self-loop at '{s}'")
            if not line:
                r.connection_integrity_valid = False
                r.errors.append(f"Connection #{i} ({s} - {d}) has no line")
            else:
                for st in stations:
                    if isinstance(st, dict) and st.get("name") in (s, d) and line not in (st.get("lines") or []):
                        r.warnings.append(f"Station '{st['name']}' does not list line '{line}'")
            key = (frozenset((s, d)), line)
            dist = c.get("distance")
            if key in seen_pairs:
                if seen_pairs[key] != dist:
                    r.distance_accuracy_valid = False
                    r.errors.append(f"Inconsistent distances for {s} - {d} on {line}: {seen_pairs[key]} vs {dist}")
                else:
                    r.warnings.append(f"Duplicate connection {s} - {d} on {line}")
            else:
                seen_pairs[key] = dist

            # Distance accuracy
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                r.distance_accuracy_valid = False
                r.errors.append(f"Connection {s} - {d} has non-numeric distance {dist!r}")
            elif dist <= 0:
                r.distance_accuracy_valid = False
                r.errors.append(f"Connection {s} - {d} has non-positive distance {dist}")
            elif not (MIN_PLAUSIBLE_DISTANCE <= dist <= MAX_PLAUSIBLE_DISTANCE):
                r.warnings.append(f"Connection {s} - {d} has unusual distance {dist} m")

        # Isolated stations / connectivity
        g = self.graph_obj
        isolated = [self.number_to_station[v] for v in g.vertices if g.degree(v) == 0]
        if isolated:
            r.connection_integrity_valid = False
            r.errors.append(f"Stations without connections: {', '.join(isolated[:20])}")
        if g.vertices:
            start = g.vertices[0]
            seen = {start}
            q = deque([start])
            while q:
                v = q.popleft()
                for e in g.adj(v):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        q.append(e.dest)
            unreached = len(g.vertices) - len(seen)
            if unreached and not isolated:
                r.connection_integrity_valid = False
                r.errors.append(f"Network is not connected: {unreached} stations unreachable")
        return r

    # ---- data updates ------------------------------------------------------
    def _station_entry(self, name):
        for st in self.data["stations"]:
            if isinstance(st, dict) and st.get("name") == name:
                return st
        return None

    def add_station(self, name, lines=None):
        entry = self._station_entry(name)
        if entry is None:
            self.data["stations"].append({"name": name, "lines": list(lines or [])})
        else:
            for l in lines or []:
                entry.setdefault("lines", [])
                if l not in entry["lines"]:
                    entry["lines"].append(l)
        self.rebuild()

    def add_line(self, line_name, stations, distances):
        """Add a new line given its ordered stations and inter-station distances (m)."""
        if len(stations) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have exactly len(stations) - 1 entries")
        for dist in distances:
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                raise ValueError(f"Invalid distance {dist!r}")
        for name in stations:
            entry = self._station_entry(name)
            if entry is None:
                self.data["stations"].append({"name": name, "lines": [line_name]})
            else:
                entry.setdefault("lines", [])
                if line_name not in entry["lines"]:
                    entry["lines"].append(line_name)
        for a, b, dist in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": dist, "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        entry = self._station_entry(name)
        if entry is None:
            raise KeyError(f"Station '{name}' does not exist")
        if new_name and new_name != name:
            if self._station_entry(new_name) is not None:
                raise ValueError(f"Station '{new_name}' already exists")
            entry["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            entry["lines"] = list(lines)
        self.rebuild()

    def update_distance(self, source, dest, distance, line=None):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError(f"Invalid distance {distance!r}")
        updated = 0
        for c in self.data["connections"]:
            if {c.get("source"), c.get("dest")} == {source, dest} and (line is None or c.get("line") == line):
                c["distance"] = distance
                updated += 1
        if not updated:
            raise KeyError(f"No connection between '{source}' and '{dest}'")
        self.rebuild()
        return updated

    def save(self, path=None):
        save_subway_data(self.data, path or self.data_file)

def build_segments(network, nodes, edges):
    """Group consecutive edges on the same line into segments."""
    segments = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = network.number_to_station[e.dest]
            segments[-1]["stops"] += 1
            segments[-1]["distance"] += e.weight
            segments[-1]["stations"].append(network.number_to_station[e.dest])
        else:
            segments.append({
                "line": e.line,
                "from": network.number_to_station[e.source],
                "to": network.number_to_station[e.dest],
                "stops": 1,
                "distance": e.weight,
                "stations": [network.number_to_station[e.source], network.number_to_station[e.dest]],
            })
    return segments


def plan_route(network, start, end, config=None):
    """Compute the route between two canonical station names."""
    config = config or PathfindingConfig()
    if start not in network.station_to_number:
        raise StationNotFoundError(f"Starting station '{start}' does not exist, please check the station name")
    if end not in network.station_to_number:
        raise StationNotFoundError(f"Ending station '{end}' does not exist, please check the station name")
    s = network.station_to_number[start]
    t = network.station_to_number[end]
    try:
        sp = ShortestPath(network.graph_obj, s, config)
        nodes, edges = sp.path_with_edges(t)
    except Exception as exc:  # defensive
        raise PathCalculationError(f"Path calculation failed: {exc}") from exc
    if not nodes:
        extra = ""
        if config.max_transfers is not None:
            extra = f" within {config.max_transfers} transfer(s)"
        raise PathNotFoundError(f"No path found from '{start}' to '{end}'{extra}")

    distance = sum(e.weight for e in edges)
    transfers = []
    for prev, cur in zip(edges, edges[1:]):
        if cur.line != prev.line:
            transfers.append(Transfer(network.number_to_station[cur.source], prev.line, cur.line))
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    try:
        fare = Sum_money(distance)
        est = calculate_estimated_time(distance, len(transfers))
    except (ValueError, TypeError) as exc:
        raise FareError(f"Fare calculation failed: {exc}") from exc

    return Route(
        start=start,
        end=end,
        mode=config.mode,
        station_sequence=[network.number_to_station[n] for n in nodes],
        total_distance_meters=distance,
        transfer_count=len(transfers),
        estimated_time_minutes=est,
        involved_lines=lines,
        transfers=transfers,
        segments=build_segments(network, nodes, edges),
        fare=fare,
        max_transfers=config.max_transfers,
        line_preference=list(config.preferred_lines),
    )

"""

import json


def format_distance(meters):
    if meters >= 1000:
        return f"{meters:g} meters ({meters / 1000:.2f} km)"
    return f"{meters:g} meters ({meters / 1000:.3f} km)"


def generate_summary(route):
    """Key information of a route."""
    return {
        "total_distance_km": route.total_distance_km,
        "total_distance_meters": route.total_distance_meters,
        "fare": route.fare,
        "transfer_count": route.transfer_count,
        "estimated_time_minutes": route.estimated_time_minutes,
    }


def format_text(route, network=None):
    transfer_stations = {t.station_name for t in route.transfers}
    seq = []
    for name in route.station_sequence:
        if name in transfer_stations:
            seq.append(f"{name} [transfer]")
        else:
            seq.append(name)
    out = ["=" * 50,
           f"Route: {route.start} -> {route.end}",
           f"Mode: {route.mode}",
           "-" * 50,
           f"Stations ({len(route.station_sequence)}):",
           "  " + " -> ".join(seq),
           "Line segments:"]
    for i, seg in enumerate(route.segments, 1):
        out.append(f"  {i}. {seg['line']}: {seg['from']} -> {seg['to']} "
                   f"({seg['stops']} stop(s), {format_distance(seg['distance'])})")
    if route.transfers:
        out.append(f"Transfers ({len(route.transfers)}):")
        for t in route.transfers:
            out.append(f"  - At {t.station_name}: {t.from_line} -> {t.to_line}")
    else:
        out.append("Transfers: none")
    out.append("Lines involved: " + ", ".join(str(l) for l in route.involved_lines))
    out.append(f"Total distance: {format_distance(route.total_distance_meters)}")
    out.append(f"Transfer count: {route.transfer_count}")
    out.append(f"Estimated time: {route.estimated_time_minutes} minutes")
    out.append(f"Fare: {route.fare} yuan")
    out.append("=" * 50)
    return "\n".join(out)


def format_json(route):
    return json.dumps(route.to_dict(), ensure_ascii=False, indent=2)


def render(route, fmt="text"):
    return format_json(route) if fmt == "json" else format_text(route)


def export_routes(routes, path, fmt="text"):
    """Write all routes to a file in the given format."""
    with open(path, "w", encoding="utf-8") as fh:
        if fmt == "json":
            payload = routes[0].to_dict() if len(routes) == 1 else [r.to_dict() for r in routes]
            json.dump(payload, fh, ensure_ascii=False, indent=2)
        else:
            fh.write("\n\n".join(format_text(r) for r in routes))

"""
Command-line interface of the Beijing Subway fare calculation system.

Usage examples:
    python main.py
    python main.py --mode least_transfer --format json --output_file out.json
    python main.py --check_data
"""

import argparse
import re
import sys

try:
    from data_manager import SubwayNetwork
    from pathfinding import PathfindingConfig, VALID_MODES
    from route_service import (plan_route, StationNotFoundError, PathNotFoundError,
                               PathCalculationError, FareError)
    from report import render, export_routes
except ImportError:  # pragma: no cover
    from .data_manager import SubwayNetwork
    from .pathfinding import PathfindingConfig, VALID_MODES
    from .route_service import (plan_route, StationNotFoundError, PathNotFoundError,
                                PathCalculationError, FareError)
    from .report import render, export_routes

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
SEPARATORS = r"[;；,，]"
EXIT_COMMANDS = {"q", "quit", "exit"}


class InputFormatError(Exception):
    pass


class QueryValidationError(Exception):
    pass


def parse_query_input(text):
    """Split 'Start; End' input into two raw station names."""
    if text is None:
        raise InputFormatError(FORMAT_ERROR)
    parts = re.split(SEPARATORS, text.strip())
    if len(parts) != 2:
        raise InputFormatError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise InputFormatError(FORMAT_ERROR)
    return start, end


def validate_query(network, start_raw, end_raw):
    """Return canonical (start, end) or raise QueryValidationError."""
    start = network.resolve_station(start_raw)
    if start is None:
        msg = f"Starting station '{start_raw}' does not exist, please check the station name"
        sug = network.suggest_stations(start_raw)
        if sug:
            msg += f" (did you mean: {', '.join(sug)}?)"
        raise QueryValidationError(msg)
    end = network.resolve_station(end_raw)
    if end is None:
        msg = f"Ending station '{end_raw}' does not exist, please check the station name"
        sug = network.suggest_stations(end_raw)
        if sug:
            msg += f" (did you mean: {', '.join(sug)}?)"
        raise QueryValidationError(msg)
    if start == end:
        raise QueryValidationError("Starting station cannot be the same as the ending station")
    return start, end


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", help="Export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text", help="Output format")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance", help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="Preferred line(s), comma separated (e.g. 'Line 1')")
    p.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    p.add_argument("--data_file", default=None, help="Alternative subway data file")
    p.add_argument("--start", default=None, help="Starting station (single query mode)")
    p.add_argument("--end", default=None, help="Ending station (single query mode)")
    return p


def run_query(network, config, text, fmt):
    """Process one input line; return (Route or None, message to print)."""
    start_raw, end_raw = parse_query_input(text)
    start, end = validate_query(network, start_raw, end_raw)
    route = plan_route(network, start, end, config)
    return route, render(route, fmt)


def main(argv=None):
    args = build_parser().parse_args(argv)

    try:
        network = SubwayNetwork(data_file=args.data_file)
    except FileNotFoundError as exc:
        print(f"Error: data file not found: {exc}", file=sys.stderr)
        return 1
    except Exception as exc:
        print(f"Error: failed to load subway data: {exc}", file=sys.stderr)
        return 1

    if args.check_data:
        report = network.check_data()
        print(report.format_text())
        return 0 if report.is_valid else 1

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer", file=sys.stderr)
        return 2

    preferred = []
    if args.line_preference:
        for raw in re.split(r"[,，;；]", args.line_preference):
            raw = raw.strip()
            if not raw:
                continue
            line = network.resolve_line(raw)
            if line is None:
                print(f"Warning: preferred line '{raw}' not found in network data, ignored")
            else:
                preferred.append(line)
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers, preferred_lines=preferred)

    routes = []

    def handle(text):
        try:
            route, output = run_query(network, config, text, args.format)
        except (InputFormatError, QueryValidationError, StationNotFoundError) as exc:
            print(f"Error: {exc}")
            return False
        except PathNotFoundError as exc:
            print(f"Error: {exc}")
            return False
        except PathCalculationError as exc:
            print(f"Path calculation error: {exc}")
            return False
        except FareError as exc:
            print(f"Fare calculation error: {exc}")
            return False
        except Exception as exc:  # last-resort safety net
            print(f"Unexpected error: {exc}")
            return False
        print(output)
        routes.append(route)
        if args.output_file:
            try:
                export_routes(routes, args.output_file, args.format)
                print(f"Result exported to {args.output_file}")
            except OSError as exc:
                print(f"Error: cannot write output file '{args.output_file}': {exc}")
        return True

    if args.start or args.end:
        if not (args.start and args.end):
            print(f"Error: {FORMAT_ERROR}")
            return 2
        return 0 if handle(f"{args.start};{args.end}") else 1

    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' to query, 'help' for help, 'quit' to exit.")
    while True:
        try:
            text = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        cmd = text.strip()
        if not cmd:
            continue
        if cmd.lower() in EXIT_COMMANDS:
            break
        if cmd.lower() == "help":
            print("Input format: Starting Station; Ending Station  (separators: ';' or ',')")
            print("Commands: help, lines, quit")
            continue
        if cmd.lower() == "lines":
            print(", ".join(str(l) for l in network.all_lines()))
            continue
        handle(cmd)
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

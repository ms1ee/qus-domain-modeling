"""Global variable registry for the food delivery dispatch simulator.

The registry is a simple key/value store shared by every module.  Well-known
keys:

* ``"funds"``             -- int, system funds (may become negative)
* ``"riders"``            -- list[Rider], all riders in the system
* ``"rider_id_counter"``  -- int, last rider id handed out
* ``"running"``           -- bool, main loop flag
* ``"tick"``              -- int, current simulated time
* ``"orders"``            -- list[Order], every order ever created
* ``"order_id_counter"``  -- int, last order id handed out
"""

_global_dict = None


def _init():
    """(Re)initialise the global registry to an empty dictionary."""
    global _global_dict
    _global_dict = {}


def set_value(key, value):
    """Store ``value`` under ``key`` (initialises the registry if needed)."""
    global _global_dict
    if _global_dict is None:
        _init()
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` or ``default`` if missing."""
    if _global_dict is None:
        return default
    return _global_dict.get(key, default)


def has_key(key):
    """Return True if ``key`` is present in the registry."""
    return _global_dict is not None and key in _global_dict

import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "rider_speed": 20,
    "recruitment_cost": 500,
    "recruitment_threshold": 300,
    "state_file": "dispatch_state.json",
}


class Configuration:
    """System parameters, optionally loaded from a JSON file."""

    def __init__(self, file_path="config.json"):
        self.file_path = file_path
        self.values = dict(DEFAULT_CONFIG)

    @property
    def initial_funds(self):
        """Initial system funds."""
        return self.values["initial_funds"]

    @property
    def rider_speed(self):
        """Rider speed in units per tick."""
        return self.values["rider_speed"]

    def load(self):
        """Load configuration from ``file_path`` if it exists.

        Invalid files are ignored (defaults are kept).
        """
        if not self.file_path or not os.path.exists(self.file_path):
            return self.values
        try:
            with open(self.file_path, "r", encoding="utf-8") as handle:
                data = json.load(handle)
            if isinstance(data, dict):
                for key, value in data.items():
                    if key in DEFAULT_CONFIG and isinstance(
                            value, type(DEFAULT_CONFIG[key])):
                        self.values[key] = value
        except (OSError, ValueError) as exc:
            logger.warning("Could not load config %s: %s", self.file_path, exc)
        return self.values

    def save(self):
        """Write current configuration to ``file_path``."""
        try:
            with open(self.file_path, "w", encoding="utf-8") as handle:
                json.dump(self.values, handle, indent=2)
            return True
        except OSError as exc:
            logger.warning("Could not save config: %s", exc)
            return False


def save_state(path, data):
    """Persist ``data`` (a JSON-serialisable dict) to ``path``."""
    try:
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(data, handle, indent=2)
        return True
    except (OSError, TypeError, ValueError) as exc:
        logger.warning("Could not save state: %s", exc)
        return False

class GridMap:
    """Describes the coordinate system of the simulated city."""

    def __init__(self, width=MAP_SIZE, height=MAP_SIZE, cell_size=CELL_SIZE):
        self.width = width
        self.height = height
        self.cell_size = cell_size
        self.rows = height // cell_size
        self.columns = width // cell_size

    def clamp(self, value, upper=None):
        """Clamp a coordinate into the map range."""
        upper = self.width if upper is None else upper
        return max(0, min(upper, int(value)))

    def to_cell_corner(self, value):
        """Convert a raw coordinate to the top-left corner of its cell."""
        value = self.clamp(value)
        return (value // self.cell_size) * self.cell_size

    def to_cell_center(self, value):
        """Return the centre of the cell containing ``value``."""
        value = self.clamp(value)
        center = (value // self.cell_size) * self.cell_size + self.cell_size // 2
        return min(center, self.width)

    def is_location(self, x, y):
        """True if (x, y) is a restaurant/customer location point."""
        return x % self.cell_size == 0 and y % self.cell_size == 0

    def is_street(self, x, y):
        """True if (x, y) lies on a street line."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

"""Grid based path planning."""

from grid_map import CELL_SIZE


class Path:
    """A planned path consisting of axis-aligned waypoints."""

    def __init__(self, waypoints):
        self.waypoints = list(waypoints)
        self.length = sum(
            abs(b[0] - a[0]) + abs(b[1] - a[1])
            for a, b in zip(self.waypoints, self.waypoints[1:]))

    def __repr__(self):
        return "Path(%r, length=%d)" % (self.waypoints, self.length)


class PathPlanner:
    """Plans street-following routes (right angle turns only)."""

    def __init__(self, cell_size=CELL_SIZE):
        self.cell_size = cell_size

    def plan(self, start, end):
        """Return a :class:`Path` from ``start`` to ``end``.

        Strategy:
        * same point -> single waypoint;
        * straight line (incl. rider directly above/below target) -> direct;
        * otherwise move along a street: horizontal first if the current
          row is a street, vertical first if only the column is a street,
          and if on neither, snap vertically to the nearest street first.
        """
        sx, sy = int(start[0]), int(start[1])
        ex, ey = int(end[0]), int(end[1])
        if (sx, sy) == (ex, ey):
            return Path([(sx, sy)])
        if sx == ex or sy == ey:
            return Path([(sx, sy), (ex, ey)])
        c = self.cell_size
        on_h_street = sy % c == 0
        on_v_street = sx % c == 0
        if on_h_street:
            points = [(sx, sy), (ex, sy), (ex, ey)]
        elif on_v_street:
            points = [(sx, sy), (sx, ey), (ex, ey)]
        else:
            low = (sy // c) * c
            high = low + c
            snap = low if abs(low - ey) <= abs(high - ey) else high
            points = [(sx, sy), (sx, snap), (ex, snap), (ex, ey)]
        cleaned = []
        for p in points:
            if not cleaned or cleaned[-1] != p:
                cleaned.append(p)
        return Path(cleaned)

    def plan_multi(self, start, stops):
        """Plan a continuous path visiting ``stops`` in order."""
        waypoints = [tuple(start)]
        current = tuple(start)
        for stop in stops:
            segment = self.plan(current, stop).waypoints
            for p in segment[1:]:
                if waypoints[-1] != p:
                    waypoints.append(p)
            current = tuple(stop)
        return Path(waypoints)

"""Rider model: movement, state machine and status-change log."""

import globalvar
from path_planner import PathPlanner

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

DEFAULT_SPEED = 20
START_X = 400
START_Y = 400


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``; each call to
    :meth:`move` advances the rider by one tick.
    """

    def __init__(self, rider_id, x=START_X, y=START_Y, speed=None):
        self.rider_id = rider_id
        self.x = int(x)
        self.y = int(y)
        if speed is None:
            speed = globalvar.get_value("rider_speed") or DEFAULT_SPEED
        self.speed = int(speed)
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.path = []
        self._path_target = None
        self.planner = PathPlanner()
        self.total_distance = 0
        self.log = []
        self._record("spawn", None, None, IDLE)

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _now():
        tick = globalvar.get_value("tick")
        return tick if isinstance(tick, int) else 0

    def _record(self, event, order, from_state, to_state):
        self.log.append({
            "event": event,
            "order_id": getattr(order, "id", None) if order is not None else None,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "from_state": from_state,
            "to_state": to_state,
        })

    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, str(self.state))

    def pending_orders(self):
        """Orders assigned but not yet started."""
        return [o for o in self.orders
                if getattr(o, "status", "pending") != "completed"
                and o is not self.current_order]

    def active_order_count(self):
        """Number of uncompleted orders held by this rider."""
        return sum(1 for o in self.orders
                   if getattr(o, "status", "pending") != "completed")

    def is_idle(self):
        """True if the rider has no work at all."""
        return self.state == IDLE and self.active_order_count() == 0

    # ---------------------------------------------------------------- movement
    def _accept_next(self):
        pending = self.pending_orders()
        if not pending:
            self.current_order = None
            return False
        order = pending[0]
        if getattr(order, "rider_id", None) is None and hasattr(order, "assign_to"):
            order.assign_to(self.rider_id)
        self.current_order = order
        old = self.state
        self.state = PICKING_UP
        if hasattr(order, "mark_accepted"):
            order.mark_accepted(self._now())
        self._path_target = None
        self._record("accepted_order", order, old, PICKING_UP)
        return True

    def _advance_towards(self, tx, ty, budget):
        """Move up to ``budget`` units along a planned path to (tx, ty)."""
        if (self.x, self.y) == (tx, ty):
            return budget
        if self._path_target != (tx, ty) or not self.path:
            self.special_case = self.x == tx and self.y != ty
            self.path = self.planner.plan((self.x, self.y), (tx, ty)).waypoints[1:]
            self._path_target = (tx, ty)
        while budget > 0 and self.path:
            wx, wy = self.path[0]
            dist = abs(wx - self.x) + abs(wy - self.y)
            if dist <= budget:
                self.x, self.y = wx, wy
                budget -= dist
                self.total_distance += dist
                self.path.pop(0)
            else:
                if wx != self.x:
                    self.x += budget if wx > self.x else -budget
                else:
                    self.y += budget if wy > self.y else -budget
                self.total_distance += budget
                budget = 0
        return budget

    def move(self):
        """Advance the rider by one tick, performing state transitions."""
        budget = self.speed
        if self.state == IDLE or self.current_order is None:
            if not self._accept_next():
                self.state = IDLE
                return
        while self.current_order is not None:
            order = self.current_order
            if self.state == PICKING_UP:
                target = (order.res_x, order.res_y)
                budget = self._advance_towards(target[0], target[1], budget)
                if (self.x, self.y) == target:
                    self.state = DELIVERING
                    if hasattr(order, "mark_picked_up"):
                        order.mark_picked_up(self._now())
                    self._path_target = None
                    self._record("arrived_at_merchant", order, PICKING_UP, DELIVERING)
                    continue
                break
            if self.state == DELIVERING:
                target = (order.cus_x, order.cus_y)
                budget = self._advance_towards(target[0], target[1], budget)
                if (self.x, self.y) == target:
                    if hasattr(order, "mark_completed"):
                        order.mark_completed(self._now())
                    else:
                        order.status = "completed"
                    self.state = IDLE
                    self.current_order = None
                    self.special_case = False
                    self._path_target = None
                    self._record("delivered_to_customer", order, DELIVERING, IDLE)
                    if budget > 0 and self._accept_next():
                        continue
                    if budget == 0 and self.pending_orders():
                        self._accept_next()
                    break
                break
            break

    def remaining_stops(self):
        """Ordered list of stops still to visit."""
        stops = []
        order = self.current_order
        if order is not None:
            if self.state == PICKING_UP:
                stops.append((order.res_x, order.res_y))
            stops.append((order.cus_x, order.cus_y))
        for o in self.pending_orders():
            stops.append((o.res_x, o.res_y))
            stops.append((o.cus_x, o.cus_y))
        return stops

    def to_dict(self):
        """Serialisable representation."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [getattr(o, "id", None) for o in self.orders],
            "total_distance": self.total_distance,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%d,%d), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state_name, len(self.orders))

import globalvar
from grid_map import DEFAULT_GRID


def to_grid(value):
    """Convert a raw coordinate to the top-left corner of its grid cell."""
    return DEFAULT_GRID.to_cell_corner(value)


def _next_order_id():
    counter = globalvar.get_value("order_id_counter")
    if not isinstance(counter, int):
        counter = 0
    counter += 1
    globalvar.set_value("order_id_counter", counter)
    return counter


def _now():
    tick = globalvar.get_value("tick")
    return tick if isinstance(tick, int) else 0


class Order:
    """A delivery order from a restaurant to a customer."""

    def __init__(self, res_pos, cus_pos):
        if len(res_pos) != 2 or len(cus_pos) != 2:
            raise ValueError("positions must be (x, y) pairs")
        self.id = _next_order_id()
        self.raw = [int(res_pos[0]), int(res_pos[1]),
                    int(cus_pos[0]), int(cus_pos[1])]
        self.res_x = to_grid(res_pos[0])
        self.res_y = to_grid(res_pos[1])
        self.cus_x = to_grid(cus_pos[0])
        self.cus_y = to_grid(cus_pos[1])
        self.res_pos = (self.res_x, self.res_y)
        self.cus_pos = (self.cus_x, self.cus_y)
        self.status = "pending"
        self.rider_id = None
        self.start_time = _now()
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.distance = abs(self.res_x - self.cus_x) + abs(self.res_y - self.cus_y)

    def assign_to(self, rider_id):
        """Record that this order is assigned to ``rider_id``."""
        self.rider_id = rider_id
        self.assign_time = _now()
        if self.status == "pending":
            self.status = "assigned"

    def mark_accepted(self, time):
        """Rider started heading to the restaurant."""
        self.accept_time = time
        if self.status != "completed":
            self.status = "picking_up"

    def mark_picked_up(self, time):
        """Rider arrived at the restaurant."""
        self.pickup_time = time
        if self.status != "completed":
            self.status = "delivering"

    def mark_completed(self, time):
        """Rider delivered the order."""
        self.completion_time = time
        self.time_taken = time - self.start_time
        self.status = "completed"

    def efficiency(self):
        """Distance delivered per tick (None if not completed)."""
        if not self.time_taken:
            return None
        return round(self.distance / self.time_taken, 2)

    def to_list(self):
        """Return ``[restaurant_x, restaurant_y, customer_x, customer_y]``."""
        return [self.res_x, self.res_y, self.cus_x, self.cus_y]

    def to_dict(self):
        """Serialisable representation."""
        return {
            "id": self.id,
            "raw": self.raw,
            "order": self.to_list(),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
        }

    def __repr__(self):
        return "Order(id=%d, %s, status=%s)" % (self.id, self.to_list(), self.status)

"""Order management: creation, assignment and completion tracking."""

import math

import globalvar
from order_model import Order


def _euclid(rider, order):
    return math.sqrt((rider.x - order.res_x) ** 2 + (rider.y - order.res_y) ** 2)


def find_best_rider_for_order(order):
    """Return the best rider for ``order``.

    Riders with the fewest current (uncompleted) orders are preferred; ties
    are broken by the straight-line distance to the restaurant, then by id.
    If no rider exists, one is recruited.
    """
    riders = globalvar.get_value("riders")
    if not riders:
        from system import create_rider
        return create_rider()
    return min(riders, key=lambda r: (r.active_order_count(),
                                      _euclid(r, order), r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion queries."""

    def __init__(self, system_manager=None):
        self.assignment_strategy = "fewest-orders-then-closest"
        self.system_manager = system_manager

    @staticmethod
    def all_orders():
        """All orders ever created."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, res_pos, cus_pos):
        """Create and register an order."""
        order = Order(res_pos, cus_pos)
        self.all_orders().append(order)
        return order

    def pending_orders(self):
        """Orders not yet assigned to a rider."""
        return [o for o in self.all_orders() if o.status == "pending"]

    def completed_orders(self):
        """Completed orders."""
        return [o for o in self.all_orders() if o.status == "completed"]

    def dispatch(self, order):
        """Run recruitment check and assign ``order``.

        Returns ``(rider, new_rider_or_None)``.
        """
        recruited = None
        if self.system_manager is not None:
            recruited = self.system_manager.check_recruitment()
        before = len(globalvar.get_value("riders") or [])
        rider = find_best_rider_for_order(order)
        if recruited is None and len(globalvar.get_value("riders") or []) > before:
            recruited = rider
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider, recruited

"""System initialisation and the SystemManager core class."""

import logging

import globalvar
from rider import Rider, DEFAULT_SPEED

logger = logging.getLogger(__name__)

DEFAULT_FUNDS = 1000
RECRUITMENT_COST = 500
RECRUITMENT_THRESHOLD = 300


def init_system(config=None):
    """Initialise all global system parameters.

    ``config`` may be a dict of overrides (see :mod:`config`).
    Returns a :class:`SystemManager`.
    """
    config = config or {}
    globalvar._init()
    globalvar.set_value("funds", int(config.get("initial_funds", DEFAULT_FUNDS)))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)
    globalvar.set_value("rider_speed", int(config.get("rider_speed", DEFAULT_SPEED)))
    globalvar.set_value("recruitment_cost",
                        int(config.get("recruitment_cost", RECRUITMENT_COST)))
    globalvar.set_value("recruitment_threshold",
                        int(config.get("recruitment_threshold", RECRUITMENT_THRESHOLD)))
    return SystemManager()


def create_rider(charge=True):
    """Create a rider, register it and (optionally) deduct the cost."""
    counter = globalvar.get_value("rider_id_counter")
    counter = (counter if isinstance(counter, int) else 0) + 1
    globalvar.set_value("rider_id_counter", counter)
    rider = Rider(counter)
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    riders.append(rider)
    if charge:
        funds = globalvar.get_value("funds")
        if not isinstance(funds, int):
            funds = DEFAULT_FUNDS
        cost = globalvar.get_value("recruitment_cost") or RECRUITMENT_COST
        globalvar.set_value("funds", funds - cost)
    logger.info("Recruited rider %d", rider.rider_id)
    return rider


class SystemManager:
    """Manages funds, riders, time and system state."""

    @property
    def funds(self):
        """Current funds."""
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        """All riders."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def rider_id_counter(self):
        """Last assigned rider id."""
        return globalvar.get_value("rider_id_counter", 0)

    @property
    def running(self):
        """Main loop flag."""
        return bool(globalvar.get_value("running", False))

    @running.setter
    def running(self, value):
        globalvar.set_value("running", bool(value))

    @property
    def current_tick(self):
        """Current simulated time."""
        return globalvar.get_value("tick", 0)

    @property
    def recruitment_cost(self):
        """Cost of one rider."""
        return globalvar.get_value("recruitment_cost", RECRUITMENT_COST)

    @property
    def recruitment_threshold(self):
        """Minimum funds to recruit."""
        return globalvar.get_value("recruitment_threshold", RECRUITMENT_THRESHOLD)

    def idle_riders(self):
        """Riders without any work."""
        return [r for r in self.riders if r.is_idle()]

    def pending_order_count(self):
        """Number of unassigned orders."""
        return sum(1 for o in (globalvar.get_value("orders") or [])
                   if o.status == "pending")

    def check_recruitment(self):
        """Recruit a rider if funds >= threshold and pending > idle riders.

        Returns the new rider or None.
        """
        if (self.funds >= self.recruitment_threshold
                and self.pending_order_count() > len(self.idle_riders())):
            return create_rider()
        return None

    def advance(self, ticks=1):
        """Advance the simulation by ``ticks`` time units."""
        for _ in range(max(0, int(ticks))):
            globalvar.set_value("tick", self.current_tick + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception:  # keep simulation alive
                    logger.exception("Rider %s failed to move; resetting",
                                     rider.rider_id)
                    rider.path = []
                    rider._path_target = None

    def consistency_check(self):
        """Return a list of detected inconsistencies (and repair them)."""
        problems = []
        ids = [r.rider_id for r in self.riders]
        if len(ids) != len(set(ids)):
            problems.append("duplicate rider ids")
        for r in self.riders:
            if r.state not in (0, 1, 2):
                problems.append("rider %s invalid state %s" % (r.rider_id, r.state))
                r.state = 0
                r.current_order = None
        return problems

    def snapshot(self):
        """Serialisable state of the whole system."""
        return {
            "funds": self.funds,
            "tick": self.current_tick,
            "rider_id_counter": self.rider_id_counter,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in (globalvar.get_value("orders") or [])],
        }

MAP_CELLS = 16
UNIT = MAP_SIZE // MAP_CELLS  # 50 units per character


def _cell(v):
    return max(0, min(MAP_CELLS - 1, int(v) // UNIT))


class DisplayManager:
    """Builds text output for the command line interface."""

    def __init__(self, map_size=MAP_CELLS):
        self.map_size = map_size

    def render_map(self, riders, orders=(), route=None):
        """Render a 16x16 ASCII map.

        '#' street, '.' empty, '*' route, 'S' restaurant, 'C' customer,
        'R' rider.
        """
        n = self.map_size
        grid = [["#" if (r % 2 == 0 or c % 2 == 0) else "." for c in range(n)]
                for r in range(n)]
        if route:
            for (ax, ay), (bx, by) in zip(route, route[1:]):
                steps = max(abs(bx - ax), abs(by - ay)) // (UNIT // 2) or 1
                for i in range(steps + 1):
                    x = ax + (bx - ax) * i / steps
                    y = ay + (by - ay) * i / steps
                    grid[_cell(y)][_cell(x)] = "*"
        for o in orders:
            if o.status == "completed":
                continue
            grid[_cell(o.res_y)][_cell(o.res_x)] = "S"
            grid[_cell(o.cus_y)][_cell(o.cus_x)] = "C"
        for r in riders:
            grid[_cell(r.y)][_cell(r.x)] = "R"
        header = "    " + "".join("%-2d" % (c % 100) if c % 2 == 0 else ""
                                    for c in range(n))
        lines = ["   x→ (each char = %d units)" % UNIT]
        lines.append("    " + "".join(str((c * UNIT) // 100) if c % 2 == 0 else " "
                                      for c in range(n)))
        del header
        for r in range(n):
            label = str((r * UNIT) // 100) if r % 2 == 0 else " "
            lines.append("%3s " % label + "".join(grid[r]))
        lines.append("Legend: R=rider S=restaurant C=customer *=route #=street .=empty")
        return "\n".join(lines)

    @staticmethod
    def format_status(manager):
        """System status snapshot text."""
        lines = ["=== System Status (tick %d) ===" % manager.current_tick,
                 "Funds: %d" % manager.funds,
                 "Riders: %d" % len(manager.riders)]
        for r in manager.riders:
            active = [o.id for o in r.orders if o.status != "completed"]
            done = [o.id for o in r.orders if o.status == "completed"]
            lines.append("  Rider %d @ (%d,%d) speed=%d state=%s orders=%d "
                         "active=%s done=%s" % (r.rider_id, r.x, r.y, r.speed,
                                                r.state_name, len(r.orders),
                                                active, done))
        return "\n".join(lines)

    @staticmethod
    def format_orders(orders):
        """Order details table."""
        if not orders:
            return "No orders yet."
        lines = ["%-4s %-6s %-11s %-11s %-6s %-11s %-6s %-5s" % (
            "ID", "Start", "Restaurant", "Customer", "Dist", "Status",
            "Taken", "Rider")]
        for o in orders:
            lines.append("%-4d %-6d %-11s %-11s %-6d %-11s %-6s %-5s" % (
                o.id, o.start_time, "(%d,%d)" % o.res_pos, "(%d,%d)" % o.cus_pos,
                o.distance, o.status,
                "-" if o.time_taken is None else o.time_taken,
                "-" if o.rider_id is None else o.rider_id))
        return "\n".join(lines)

    @staticmethod
    def format_rider_logs(riders):
        """All rider status changes."""
        if not riders:
            return "No riders yet."
        lines = []
        for r in riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                frm = e.get("from_state")
                to = e.get("to_state")
                lines.append("  t=%-5s %-22s order=%-4s at (%d,%d) %s -> %s" % (
                    e.get("time"), e["event"],
                    "-" if e["order_id"] is None else e["order_id"],
                    e.get("x", 0), e.get("y", 0),
                    "-" if frm is None else STATE_NAMES.get(frm, frm),
                    STATE_NAMES.get(to, to)))
        return "\n".join(lines)

"""Command parsing and input validation."""

COMMANDS = {
    "start": ("s", "1"),
    "status": ("st", "2"),
    "orders": ("o", "3"),
    "riders": ("r", "4"),
    "tick": ("t",),
    "map": ("m",),
    "history": ("hist",),
    "help": ("h", "?"),
    "quit": ("q", "exit"),
}


class Command:
    """A named command with shortcuts."""

    def __init__(self, name, shortcuts):
        self.name = name
        self.shortcuts = tuple(shortcuts)
        self.shortcut = self.shortcuts[0] if self.shortcuts else ""


class InputManager:
    """Parses commands, keeps history and validates coordinates."""

    def __init__(self, min_coordinate=0, max_coordinate=1000):
        self.command_history = []
        self.min_coordinate = min_coordinate
        self.max_coordinate = max_coordinate
        self.commands = {n: Command(n, s) for n, s in COMMANDS.items()}
        self._lookup = {}
        for name, cmd in self.commands.items():
            self._lookup[name] = name
            for s in cmd.shortcuts:
                self._lookup[s] = name

    def parse(self, line):
        """Return ``(command_name or None, args)`` for an input line.

        ``!N`` re-runs history entry N (1-based), ``!!`` the last one.
        """
        line = (line or "").strip()
        if not line:
            return None, []
        if line.startswith("!") and self.command_history:
            ref = line[1:]
            if ref == "!":
                line = self.command_history[-1]
            elif ref.isdigit() and 1 <= int(ref) <= len(self.command_history):
                line = self.command_history[int(ref) - 1]
            else:
                return None, []
        self.command_history.append(line)
        if len(self.command_history) > 1000:
            del self.command_history[:-1000]
        tokens = line.split(None, 1)
        name = self._lookup.get(tokens[0].lower())
        rest = tokens[1] if len(tokens) > 1 else ""
        return name, self.tokenize(rest)

    @staticmethod
    def tokenize(text):
        """Split free text containing brackets/commas into tokens."""
        for ch in "[](),":
            text = text.replace(ch, " ")
        return text.split()

    def parse_coordinates(self, tokens):
        """Validate four integer coordinates and return them.

        Raises ValueError with a friendly message on failure.
        """
        if isinstance(tokens, str):
            tokens = self.tokenize(tokens)
        if len(tokens) != 4:
            raise ValueError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y (got %d values)" % len(tokens))
        values = []
        for t in tokens:
            try:
                v = int(t)
            except (TypeError, ValueError):
                raise ValueError("'%s' is not an integer" % t) from None
            if not self.min_coordinate <= v <= self.max_coordinate:
                raise ValueError("Coordinate %d out of range %d-%d" % (
                    v, self.min_coordinate, self.max_coordinate))
            values.append(v)
        return values

    def help_text(self):
        """Help listing."""
        lines = ["Commands (shortcuts in brackets):"]
        desc = {
            "start": "create an order: start rx ry cx cy",
            "status": "show funds and riders",
            "orders": "show all orders",
            "riders": "show rider status change history",
            "tick": "advance simulation: tick [n]",
            "map": "show the ASCII map",
            "history": "show command history (!N re-runs, !! last)",
            "help": "show this help",
            "quit": "save and exit",
        }
        for name, cmd in self.commands.items():
            lines.append("  %-8s [%s]  %s" % (name, ",".join(cmd.shortcuts),
                                              desc[name]))
        lines.append("Each command advances the simulation by one tick.")
        return "\n".join(lines)

"""Command line entry point for the food delivery dispatch simulator."""

import logging
import math
import sys

import globalvar
from config import Configuration, save_state
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from system import init_system

try:  # command history / arrow keys where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

logging.basicConfig(level=logging.WARNING,
                    format="%(levelname)s %(name)s: %(message)s")
logger = logging.getLogger("dispatch")


class DispatchApp:
    """Ties the managers together and runs the main loop."""

    def __init__(self, config_path="config.json"):
        self.config = Configuration(config_path)
        self.config.load()
        self.system = init_system(self.config.values)
        self.orders = OrderManager(self.system)
        self.display = DisplayManager()
        self.input = InputManager()

    def cmd_start(self, args):
        """Create and dispatch an order."""
        if not args:
            args = self.input.tokenize(
                input("Enter restaurant_x restaurant_y customer_x customer_y: "))
        coords = self.input.parse_coordinates(args)
        order = self.orders.create_order(coords[0:2], coords[2:4])
        rider, recruited = self.orders.dispatch(order)
        start = (rider.x, rider.y)
        path = rider.planner.plan_multi(start, rider.remaining_stops())
        ticks = math.ceil(path.length / rider.speed) if rider.speed else 0
        out = ["Order %d created." % order.id,
               "Restaurant grid: (%d, %d)   Customer grid: (%d, %d)" % (
                   order.res_x, order.res_y, order.cus_x, order.cus_y),
               "New rider added: %s" % (
                   "yes (rider %d, funds now %d)" % (recruited.rider_id,
                                                     self.system.funds)
                   if recruited else "no"),
               "Assigned rider: %d at (%d, %d), speed %d units/tick" % (
                   rider.rider_id, rider.x, rider.y, rider.speed),
               "Route: %s" % " -> ".join("(%d,%d)" % p for p in path.waypoints),
               "Route length: %d units, estimated time: %d ticks" % (
                   path.length, ticks),
               self.display.render_map([rider], [order], path.waypoints)]
        return "\n".join(out)

    def execute(self, name, args):
        """Execute a parsed command and return output text."""
        if name == "start":
            return self.cmd_start(args)
        if name == "status":
            return self.display.format_status(self.system)
        if name == "orders":
            return self.display.format_orders(self.orders.all_orders())
        if name == "riders":
            return self.display.format_rider_logs(self.system.riders)
        if name == "map":
            return self.display.render_map(self.system.riders,
                                           self.orders.all_orders())
        if name == "tick":
            n = int(args[0]) if args else 1
            if n < 0 or n > 100000:
                raise ValueError("tick count must be 0-100000")
            self.system.advance(n - 1 if n > 0 else 0)
            return "Advanced to tick %d." % (self.system.current_tick + 1)
        if name == "history":
            return "\n".join("%3d  %s" % (i + 1, c) for i, c in
                             enumerate(self.input.command_history)) or "(empty)"
        if name == "help":
            return self.input.help_text()
        if name == "quit":
            self.system.running = False
            return "Bye."
        return "Unknown command. Type 'help'."

    def save(self):
        """Persist system state to JSON."""
        data = self.system.snapshot()
        data["config"] = self.config.values
        save_state(self.config.values.get("state_file", "dispatch_state.json"), data)

    def run(self):
        """Main loop: each iteration is one tick."""
        print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while self.system.running:
            try:
                line = input("[t=%d] > " % self.system.current_tick)
            except (EOFError, KeyboardInterrupt):
                print()
                self.system.running = False
                break
            name, args = self.input.parse(line)
            if name is None:
                if line.strip():
                    print("Unknown command. Type 'help'.")
                continue
            try:
                print(self.execute(name, args))
            except ValueError as exc:
                print("Error: %s" % exc)
            except Exception as exc:  # recover and continue
                logger.exception("Unexpected error")
                print("Unexpected error: %s (state recovered)" % exc)
                self.system.consistency_check()
            if self.system.running:
                self.system.advance(1)
        self.save()


def main(argv=None):
    """Program entry point."""
    argv = sys.argv[1:] if argv is None else argv
    app = DispatchApp(argv[0] if argv else "config.json")
    app.run()
    globalvar.set_value("running", False)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
``DATA_FILE_PATH`` is an absolute path, so it is valid from any working
directory.

File layout::

    {
      "stations":    [{"name": str, "lines": [str, ...]}, ...],
      "connections": [{"source": str, "dest": str,
                       "distance": int (meters), "line": str}, ...]
    }
"""

import json
import os

#: Absolute path of the bundled subway data file (src/subway_data.json).
DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


class DataLoadError(Exception):
    """Raised when the subway data file cannot be read or parsed."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway data JSON file and return its content as a dict.

    :param path: path of the JSON data file (defaults to ``DATA_FILE_PATH``)
    :return: the parsed JSON content (a dict)
    :raises DataLoadError: if the file is missing, unreadable or malformed
    """
    if path is None:
        path = DATA_FILE_PATH
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Data file is not valid JSON: {path} ({exc})") from exc
    except OSError as exc:
        raise DataLoadError(f"Cannot read data file: {path} ({exc})") from exc

    if not isinstance(data, dict):
        raise DataLoadError(f"Data file must contain a JSON object: {path}")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data

"""
Core data structures: weighted edges, an undirected weighted graph and a
heap-based minimum priority queue.
"""

import heapq
import itertools


class Edge:
    """A weighted, undirected edge between two station numbers.

    Attributes:
        source:   station number of one endpoint
        dest:     station number of the other endpoint
        distance: distance in meters
        weight:   same as distance (used by the shortest path algorithm)
        line:     name of the subway line this connection belongs to
    """

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source, dest, distance, line=None):
        if distance is None:
            raise ValueError("Edge distance cannot be None")
        if distance < 0:
            raise ValueError(f"Edge distance cannot be negative: {distance}")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        """Return one endpoint of the edge."""
        return self.source

    def other(self, vertex):
        """Return the endpoint of the edge different from ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex} is not an endpoint of {self!r}")

    def __repr__(self):
        return (f"Edge({self.source!r}, {self.dest!r}, "
                f"{self.distance!r}, {self.line!r})")


class Graph:
    """Weighted undirected graph represented with adjacency lists.

    Station numbers are assigned by the caller; the graph only stores
    whatever vertex identifiers appear in the added edges.

    Attributes:
        graph:         dict[vertex, list[Edge]] adjacency lists
        station_lines: dict[station name, list[str]] line affiliation
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self._edges = []

    # ----------------------------------------------------------- vertices
    def add_vertex(self, v):
        """Ensure the vertex exists (possibly without edges)."""
        if v not in self.graph:
            self.graph[v] = []

    def has_vertex(self, v):
        return v in self.graph

    def vertices(self):
        return list(self.graph.keys())

    @property
    def vertex_count(self):
        return len(self.graph)

    def V(self):
        return self.vertex_count

    # -------------------------------------------------------------- edges
    def add_edge(self, edge):
        """Add an undirected edge (stored in both endpoint adjacency lists)."""
        if not isinstance(edge, Edge):
            raise TypeError("add_edge expects an Edge instance")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        if edge.dest != edge.source:
            self.graph[edge.dest].append(edge)
        self._edges.append(edge)

    def adj(self, v):
        """Edges incident to vertex ``v``."""
        return self.graph.get(v, [])

    def edges(self):
        return list(self._edges)

    @property
    def edge_count(self):
        return len(self._edges)

    def E(self):
        return self.edge_count

    def degree(self, v):
        return len(self.graph.get(v, []))

    # ----------------------------------------------------- station lines
    def set_station_lines(self, station_name, lines):
        """Record which lines a station belongs to."""
        self.station_lines[station_name] = list(lines or [])

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return len(set(self.station_lines.get(station_name, []))) > 1

    def __repr__(self):
        return f"Graph(V={self.vertex_count}, E={self.edge_count})"


class MinPQ:
    """Minimum priority queue backed by a binary heap (``heapq``).

    Items with equal priority are returned in insertion order; items
    themselves never need to be comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    push = insert

    def del_min(self):
        """Remove and return ``(priority, item)`` with the smallest priority."""
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    pop = del_min

    def min(self):
        if not self._heap:
            raise IndexError("min of empty priority queue")
        priority, _, item = self._heap[0]
        return priority, item

    def is_empty(self):
        return not self._heap

    @property
    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

"""
Dijkstra-based path computation on the subway graph.

The search runs on an expanded state space ``(station, current line[,
transfers])`` so that line changes (transfers) at interchange stations can be
counted and optimised exactly.  Three modes are supported:

* ``shortest_distance`` - minimise distance (ties broken by fewer transfers)
* ``least_transfer``    - minimise transfers, then distance
* ``comprehensive``     - minimise distance + a per-transfer penalty

A maximum number of transfers and a list of preferred lines can also be
configured.  Complexity is O((V+E) log V) on the expanded graph thanks to the
heap-based ``MinPQ``.
"""

from models import MinPQ, Edge, Graph  # noqa: F401  (re-exported)

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

#: Equivalent distance (meters) of one transfer in comprehensive mode.
DEFAULT_TRANSFER_PENALTY = 4000
#: Weight multiplier applied to edges on preferred lines.
DEFAULT_PREFERENCE_FACTOR = 0.7

INF = float("inf")


class PathfindingConfig:
    """Configuration for a path query.

    Attributes:
        mode:            one of ``VALID_MODES``
        max_transfers:   maximum allowed transfers or ``None`` (unlimited)
        preferred_lines: list of line names to favour
    """

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None,
                 transfer_penalty=DEFAULT_TRANSFER_PENALTY,
                 preference_factor=DEFAULT_PREFERENCE_FACTOR):
        if mode not in VALID_MODES:
            raise ValueError(
                f"Invalid mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            max_transfers = int(max_transfers)
            if max_transfers < 0:
                raise ValueError("max_transfers cannot be negative")
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines)
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, "
                f"max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class ShortestPath:
    """Single-source shortest paths (Dijkstra) from ``source``.

    Attributes:
        source:  source station number
        dist_to: dict[station number, distance in meters of the chosen path]
                 (``inf`` for unreachable stations)
    """

    def __init__(self, graph, source, config=None):
        if config is None:
            config = PathfindingConfig()
        self.graph = graph
        self.source = source
        self.config = config
        self._preferred = set(config.preferred_lines or [])
        self._track_transfers = config.max_transfers is not None

        self.dist_to = {v: INF for v in graph.vertices()}
        self._edge_to = {}          # state -> (previous state, Edge)
        self._station_state = {}    # station -> best settled state

        if not graph.has_vertex(source):
            raise ValueError(f"Source station {source!r} is not in the graph")
        self._run()

    # ------------------------------------------------------------ helpers
    def _key(self, eff, transfers, dist):
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, eff, dist)
        if mode == MODE_COMPREHENSIVE:
            return (eff + self.config.transfer_penalty * transfers, transfers, dist)
        return (eff, transfers, dist)

    def _effective_weight(self, edge):
        if self._preferred and edge.line in self._preferred:
            return edge.weight * self.config.preference_factor
        return edge.weight

    # --------------------------------------------------------- algorithm
    def _run(self):
        start = (self.source, None, 0 if self._track_transfers else None)
        info = {start: (0, 0, 0)}  # state -> (effective, actual distance, transfers)
        best_key = {start: self._key(0, 0, 0)}
        settled = set()
        pq = MinPQ()
        pq.insert(best_key[start], start)
        max_t = self.config.max_transfers

        while not pq.is_empty():
            key, state = pq.del_min()
            if state in settled or key != best_key.get(state):
                continue
            settled.add(state)
            v, line, _ = state
            eff, dist, transfers = info[state]
            if v not in self._station_state:
                self._station_state[v] = state
                self.dist_to[v] = dist

            for edge in self.graph.adj(v):
                w = edge.other(v)
                changed = line is not None and edge.line != line
                new_t = transfers + (1 if changed else 0)
                if max_t is not None and new_t > max_t:
                    continue
                new_state = (w, edge.line, new_t if self._track_transfers else None)
                if new_state in settled:
                    continue
                new_eff = eff + self._effective_weight(edge)
                new_dist = dist + edge.weight
                new_key = self._key(new_eff, new_t, new_dist)
                old = best_key.get(new_state)
                if old is None or new_key < old:
                    best_key[new_state] = new_key
                    info[new_state] = (new_eff, new_dist, new_t)
                    self._edge_to[new_state] = (state, edge)
                    pq.insert(new_key, new_state)

    # ------------------------------------------------------------ queries
    def has_path_to(self, target):
        return target in self._station_state

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return ``(stations, edges)`` of the chosen path to ``target``.

        ``stations`` is the ordered list of station numbers from the source to
        the target; ``edges`` the list of traversed ``Edge`` objects.  Both are
        empty if the target is unreachable.
        """
        if target not in self._station_state:
            return [], []
        state = self._station_state[target]
        stations = [state[0]]
        edges = []
        while state in self._edge_to:
            prev, edge = self._edge_to[state]
            edges.append(edge)
            stations.append(prev[0])
            state = prev
        stations.reverse()
        edges.reverse()
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    @staticmethod
    def count_transfers(edges):
        count = 0
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                count += 1
        return count

    def get_transfer_count(self, target):
        """Number of line changes along the path to ``target`` (0 if none)."""
        _, edges = self.path_with_edges(target)
        return self.count_transfers(edges)

    def get_transfer_stations(self, target):
        """Station numbers where a line change happens along the path."""
        stations, edges = self.path_with_edges(target)
        result = []
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                result.append(stations[i])
        return result

    def get_lines(self, target):
        """Ordered list of distinct lines used along the path."""
        _, edges = self.path_with_edges(target)
        lines = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)
        return lines

"""
Beijing Subway fare calculation.

Fare structure (distance in meters, lower bound inclusive):

* [0, 6000)       -> 3 yuan  (basic tier)
* [6000, 12000)   -> 4 yuan
* [12000, 22000)  -> 5 yuan
* [22000, 32000)  -> 6 yuan
* [32000, ...)    -> 7 yuan + 1 yuan for every further 20 km (extended tier)

Fractional distances are rounded up to the next whole meter.
"""

import math

#: (upper bound exclusive in meters, fare in yuan, tier name)
FARE_TIERS = [
    (6000, 3, "Basic Tier (0-6 km)"),
    (12000, 4, "Progressive Tier (6-12 km)"),
    (22000, 5, "Progressive Tier (12-22 km)"),
    (32000, 6, "Progressive Tier (22-32 km)"),
]
EXTENDED_START_DISTANCE = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_INCREMENT_DISTANCE = 20000

#: Average operating speed (km/h) used for time estimation.
AVERAGE_SPEED_KMH = 35.0
#: Minutes added per transfer.
TRANSFER_TIME_MINUTES = 5.0
#: Minutes for entering/leaving the stations.
BASE_TIME_MINUTES = 2.0


class FareCalculationError(ValueError):
    """Raised for invalid fare calculation input."""


def _normalize_distance(distance):
    if isinstance(distance, bool):
        raise FareCalculationError("Distance must be a number")
    if not isinstance(distance, (int, float)):
        try:
            distance = float(distance)
        except (TypeError, ValueError):
            raise FareCalculationError(f"Distance must be a number: {distance!r}")
    if math.isnan(distance) or math.isinf(distance):
        raise FareCalculationError(f"Distance must be finite: {distance!r}")
    if distance < 0:
        raise FareCalculationError(f"Distance cannot be negative: {distance!r}")
    return int(math.ceil(distance))


def Sum_money(distance):
    """Return the fare (yuan, int) for a trip of ``distance`` meters."""
    d = _normalize_distance(distance)
    for upper, fare, _ in FARE_TIERS:
        if d < upper:
            return fare
    return EXTENDED_BASE_FARE + (d - EXTENDED_START_DISTANCE) // EXTENDED_INCREMENT_DISTANCE


def get_fare_tier(distance):
    """Return the human-readable tier name for ``distance`` meters."""
    d = _normalize_distance(distance)
    for upper, _, name in FARE_TIERS:
        if d < upper:
            return name
    return "Extended Tier (over 32 km)"


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes from distance (meters) and transfers."""
    d = _normalize_distance(distance)
    if transfer_count is None:
        transfer_count = 0
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, (int, float)):
        raise ValueError("transfer_count must be a number")
    if transfer_count < 0:
        raise ValueError("transfer_count cannot be negative")
    if d == 0:
        return 0.0
    ride = d / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    total = ride + transfer_count * TRANSFER_TIME_MINUTES + BASE_TIME_MINUTES
    return round(total, 1)

"""
Subway network data management: building the graph from the JSON data,
station name lookup, line information, data updates and integrity checks.
"""

import difflib
import json
import re
import unicodedata
from collections import defaultdict, deque

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph
from pathfinding import ShortestPath, PathfindingConfig


def normalize_name(name):
    """Normalise a station/line name for tolerant matching."""
    s = unicodedata.normalize("NFKC", str(name)).strip().lower()
    s = re.sub(r"[\s'’`\-_.·]", "", s)
    for suffix in ("station", "站"):
        if s.endswith(suffix) and len(s) > len(suffix):
            s = s[: -len(suffix)]
    return s


def normalize_line(name):
    s = unicodedata.normalize("NFKC", str(name)).strip().lower()
    return re.sub(r"[\s'’`\-_.]", "", s)


class SubwayNetwork:
    """The Beijing Subway network.

    Attributes:
        data_source_file:  path of the data file
        stations:          dict[name, {"name", "number", "lines"}]
        station_to_number: dict[name, int]
        number_to_station: dict[int, name]
        connections:       list of {"source", "dest", "distance", "line"}
        graph:             dict[number, list[(neighbor, distance, line)]]
        graph_obj:         ``models.Graph`` instance used for path finding
    """

    def __init__(self, data=None, data_source_file=DATA_FILE_PATH):
        self.data_source_file = data_source_file
        if data is None:
            data = load_subway_data(data_source_file)
        self.stations = {}
        self.connections = []
        self.load_issues = []
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = {}
        self.graph_obj = Graph()
        self._lookup = {}
        self._next_number = 0
        self._load(data)

    # ------------------------------------------------------------ loading
    def _register_station(self, name, lines):
        if name in self.stations:
            for ln in lines:
                if ln not in self.stations[name]["lines"]:
                    self.stations[name]["lines"].append(ln)
            return
        self.stations[name] = {"name": name, "number": self._next_number,
                               "lines": list(lines)}
        self._next_number += 1

    def _load(self, data):
        for st in data.get("stations", []) or []:
            if not isinstance(st, dict) or not st.get("name"):
                self.load_issues.append(f"Invalid station record: {st!r}")
                continue
            name = str(st["name"]).strip()
            lines = [str(x) for x in (st.get("lines") or [])]
            if name in self.stations:
                self.load_issues.append(f"Duplicate station record: '{name}'")
            self._register_station(name, lines)

        for c in data.get("connections", []) or []:
            if not isinstance(c, dict):
                self.load_issues.append(f"Invalid connection record: {c!r}")
                continue
            src, dst = c.get("source"), c.get("dest")
            if not src or not dst:
                self.load_issues.append(f"Connection missing endpoint: {c!r}")
                continue
            src, dst = str(src).strip(), str(dst).strip()
            line = c.get("line")
            for endpoint in (src, dst):
                if endpoint not in self.stations:
                    self.load_issues.append(
                        f"Connection references unknown station '{endpoint}'")
                    self._register_station(endpoint, [line] if line else [])
            self.connections.append({"source": src, "dest": dst,
                                     "distance": c.get("distance"), "line": line})
        self.rebuild()

    @staticmethod
    def _valid_distance(d):
        return (isinstance(d, (int, float)) and not isinstance(d, bool)
                and d == d and d >= 0 and d != float("inf"))

    def rebuild(self):
        """(Re)build the station mapping, adjacency dict and Graph object."""
        self.station_to_number = {n: s["number"] for n, s in self.stations.items()}
        self.number_to_station = {num: n for n, num in self.station_to_number.items()}
        self.graph = {num: [] for num in self.number_to_station}
        self.graph_obj = Graph()
        for num in self.number_to_station:
            self.graph_obj.add_vertex(num)

        extra_lines = defaultdict(list)
        for c in self.connections:
            if not self._valid_distance(c["distance"]):
                continue
            if c["source"] not in self.station_to_number or \
                    c["dest"] not in self.station_to_number:
                continue
            u = self.station_to_number[c["source"]]
            v = self.station_to_number[c["dest"]]
            self.graph_obj.add_edge(Edge(u, v, c["distance"], c["line"]))
            self.graph[u].append((v, c["distance"], c["line"]))
            if u != v:
                self.graph[v].append((u, c["distance"], c["line"]))
            for name in (c["source"], c["dest"]):
                if c["line"] and c["line"] not in extra_lines[name]:
                    extra_lines[name].append(c["line"])

        for name, st in self.stations.items():
            lines = list(st["lines"])
            for ln in extra_lines.get(name, []):
                if ln not in lines:
                    lines.append(ln)
            self.graph_obj.set_station_lines(name, lines)

        self._lookup = {}
        for name in self.stations:
            self._lookup.setdefault(normalize_name(name), name)

    # ------------------------------------------------------------ lookups
    def find_station(self, name):
        """Return the official station name matching ``name`` or ``None``."""
        if name is None:
            return None
        raw = str(name).strip()
        if raw in self.stations:
            return raw
        return self._lookup.get(normalize_name(raw))

    def suggest_stations(self, name, n=3):
        keys = list(self._lookup.keys())
        matches = difflib.get_close_matches(normalize_name(name), keys, n=n, cutoff=0.6)
        return [self._lookup[k] for k in matches]

    def get_station_lines(self, name):
        return self.graph_obj.get_station_lines(name)

    def is_transfer_station(self, name):
        return self.graph_obj.is_transfer_station(name)

    def transfer_stations(self):
        return [n for n in self.stations if self.is_transfer_station(n)]

    def get_line_names(self):
        lines = []
        for st in self.stations.values():
            for ln in st["lines"]:
                if ln not in lines:
                    lines.append(ln)
        for c in self.connections:
            if c["line"] and c["line"] not in lines:
                lines.append(c["line"])
        return lines

    def find_line(self, name):
        """Return the official line name matching ``name`` or ``None``."""
        if not name:
            return None
        lines = self.get_line_names()
        if name in lines:
            return name
        key = normalize_line(name)
        for ln in lines:
            if normalize_line(ln) == key:
                return ln
        # allow "1" -> "Line 1"
        for ln in lines:
            if normalize_line(ln) in (f"line{key}", f"{key}号线"):
                return ln
        return None

    def get_line_stations(self, line):
        """Station sequence of a line, derived from its connections."""
        adj = defaultdict(list)
        first = None
        for c in self.connections:
            if c["line"] != line:
                continue
            if first is None:
                first = c["source"]
            if c["dest"] not in adj[c["source"]]:
                adj[c["source"]].append(c["dest"])
            if c["source"] not in adj[c["dest"]]:
                adj[c["dest"]].append(c["source"])
        if not adj:
            return []
        ends = [s for s in adj if len(adj[s]) == 1]
        start = ends[0] if ends else first
        order, seen, stack = [], set(), [start]
        while stack:
            s = stack.pop()
            if s in seen:
                continue
            seen.add(s)
            order.append(s)
            for nb in reversed(adj[s]):
                if nb not in seen:
                    stack.append(nb)
        for s in adj:  # disconnected parts of a line
            if s not in seen:
                order.append(s)
                seen.add(s)
        return order

    def get_line_info(self):
        info = {}
        for ln in self.get_line_names():
            seq = self.get_line_stations(ln)
            info[ln] = {
                "name": ln,
                "stations": seq,
                "transfer_stations": [s for s in seq if self.is_transfer_station(s)],
            }
        return info

    # ------------------------------------------------------- path finding
    def shortest_path(self, start, end, config=None):
        """Compute a route; return ``(ShortestPath, station names, edges)``."""
        s = self.find_station(start)
        e = self.find_station(end)
        if s is None:
            raise KeyError(f"Unknown station '{start}'")
        if e is None:
            raise KeyError(f"Unknown station '{end}'")
        sp = ShortestPath(self.graph_obj, self.station_to_number[s],
                          config or PathfindingConfig())
        nums, edges = sp.path_with_edges(self.station_to_number[e])
        return sp, [self.number_to_station[n] for n in nums], edges

    # -------------------------------------------------------- data update
    def add_line(self, line_name, station_names, distances):
        """Add a new line given its ordered stations and segment distances."""
        if not line_name:
            raise ValueError("Line name cannot be empty")
        if len(station_names) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_names) - 1:
            raise ValueError("Number of distances must equal number of stations - 1")
        for d in distances:
            if not self._valid_distance(d) or d <= 0:
                raise ValueError(f"Invalid distance: {d!r}")
        names = []
        for n in station_names:
            official = self.find_station(n) or str(n).strip()
            names.append(official)
            self._register_station(official, [line_name])
        for i, d in enumerate(distances):
            self.connections.append({"source": names[i], "dest": names[i + 1],
                                     "distance": d, "line": line_name})
        self.rebuild()

    def modify_station(self, name, new_name=None, lines=None):
        """Rename a station and/or change its line affiliation."""
        official = self.find_station(name)
        if official is None:
            raise KeyError(f"Unknown station '{name}'")
        st = self.stations[official]
        if lines is not None:
            st["lines"] = list(lines)
        if new_name and new_name != official:
            new_name = str(new_name).strip()
            if new_name in self.stations:
                raise ValueError(f"Station '{new_name}' already exists")
            new_stations = {}
            for n, data in self.stations.items():
                if n == official:
                    data["name"] = new_name
                    new_stations[new_name] = data
                else:
                    new_stations[n] = data
            self.stations = new_stations
            for c in self.connections:
                if c["source"] == official:
                    c["source"] = new_name
                if c["dest"] == official:
                    c["dest"] = new_name
        self.rebuild()

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update distance between adjacent stations; return #updated records."""
        if not self._valid_distance(distance) or distance <= 0:
            raise ValueError(f"Invalid distance: {distance!r}")
        a, b = self.find_station(station_a), self.find_station(station_b)
        if a is None or b is None:
            raise KeyError("Unknown station")
        count = 0
        for c in self.connections:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                count += 1
        if count == 0:
            raise KeyError(f"No connection between '{a}' and '{b}'")
        self.rebuild()
        return count

    def to_dict(self):
        return {
            "stations": [{"name": s["name"], "lines": list(s["lines"])}
                         for s in self.stations.values()],
            "connections": [dict(c) for c in self.connections],
        }

    def save(self, path=None):
        path = path or self.data_source_file
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, ensure_ascii=False, indent=2)

class DataIntegrityCheck:
    """Runs integrity checks on a ``SubwayNetwork``."""

    def __init__(self, network):
        self.network = network
        self.station_coverage_valid = False
        self.connection_integrity_valid = False
        self.distance_accuracy_valid = False
        self.errors = []
        self.warnings = []
        self.info = []

    # aliases following the domain model naming
    @property
    def stationCoverageValid(self):
        return self.station_coverage_valid

    @property
    def connectionIntegrityValid(self):
        return self.connection_integrity_valid

    @property
    def distanceAccuracyValid(self):
        return self.distance_accuracy_valid

    @property
    def is_valid(self):
        return self.connection_integrity_valid and self.distance_accuracy_valid

    def run(self):
        self._check_coverage()
        self._check_connections()
        self._check_distances()
        return self

    def _check_coverage(self):
        net = self.network
        line_keys = {normalize_line(ln) for ln in net.get_line_names()}
        missing_lines = []
        for main, aliases in MAIN_LINES.items():
            if not any(normalize_line(a) in line_keys for a in aliases):
                missing_lines.append(main)
        missing_st = []
        for rep, aliases in REPRESENTATIVE_STATIONS.items():
            if not any(net.find_station(a) for a in aliases):
                missing_st.append(rep)
        self.info.append(f"Stations: {len(net.stations)}, connections: "
                         f"{len(net.connections)}, lines: {len(net.get_line_names())}, "
                         f"transfer stations: {len(net.transfer_stations())}")
        if missing_lines:
            self.warnings.append("Main lines not found: " + ", ".join(missing_lines))
        if missing_st:
            self.warnings.append("Representative stations not found: "
                                 + ", ".join(missing_st))
        self.station_coverage_valid = bool(net.stations) and not missing_lines \
            and not missing_st

    def _check_connections(self):
        net = self.network
        errors = list(net.load_issues)
        if not net.connections:
            errors.append("No connections defined")
        connected = set()
        for c in net.connections:
            if c["source"] == c["dest"]:
                errors.append(f"Self-loop connection at '{c['source']}'")
            if not c["line"]:
                self.warnings.append(
                    f"Connection {c['source']} - {c['dest']} has no line")
            for end in (c["source"], c["dest"]):
                connected.add(end)
                if c["line"] and c["line"] not in net.stations.get(end, {}).get("lines", []):
                    self.warnings.append(
                        f"Station '{end}' is not declared on line '{c['line']}'")
        isolated = [n for n in net.stations if n not in connected]
        if isolated:
            errors.append("Stations without connections: " + ", ".join(isolated[:20])
                          + (" ..." if len(isolated) > 20 else ""))
        # connectivity
        if net.graph:
            start = next(iter(net.graph))
            seen = {start}
            dq = deque([start])
            while dq:
                u = dq.popleft()
                for v, _, _ in net.graph[u]:
                    if v not in seen:
                        seen.add(v)
                        dq.append(v)
            unreachable = [net.number_to_station[n] for n in net.graph if n not in seen]
            if unreachable:
                self.warnings.append(
                    f"{len(unreachable)} station(s) not reachable from "
                    f"'{net.number_to_station[start]}': "
                    + ", ".join(unreachable[:10]) + (" ..." if len(unreachable) > 10 else ""))
        self.errors.extend(errors)
        self.connection_integrity_valid = not errors

    def _check_distances(self):
        net = self.network
        errors = []
        seen = {}
        for c in net.connections:
            d = c["distance"]
            label = f"{c['source']} - {c['dest']} ({c['line']})"
            if not net._valid_distance(d) or d <= 0:
                errors.append(f"Invalid distance {d!r} for {label}")
                continue
            if d < MIN_REASONABLE_DISTANCE or d > MAX_REASONABLE_DISTANCE:
                self.warnings.append(f"Unusual distance {d} m for {label}")
            key = (frozenset((c["source"], c["dest"])), c["line"])
            if key in seen and seen[key] != d:
                self.warnings.append(
                    f"Inconsistent duplicate distances for {label}: {seen[key]} vs {d}")
            seen.setdefault(key, d)
        self.errors.extend(errors)
        self.distance_accuracy_valid = not errors

    def report(self):
        def flag(v):
            return "PASS" if v else "FAIL"
        lines = ["Data Integrity Check Report", "=" * 40]
        lines.extend(self.info)
        lines.append(f"Station coverage:       {flag(self.station_coverage_valid)}")
        lines.append(f"Connection integrity:   {flag(self.connection_integrity_valid)}")
        lines.append(f"Distance accuracy:      {flag(self.distance_accuracy_valid)}")
        if self.errors:
            lines.append("Errors:")
            lines.extend(f"  - {e}" for e in self.errors)
        if self.warnings:
            lines.append("Warnings:")
            lines.extend(f"  - {w}" for w in self.warnings)
        lines.append("Overall: " + ("OK" if self.is_valid else "PROBLEMS FOUND"))
        return "\n".join(lines)


def check_data_integrity(network):
    return DataIntegrityCheck(network).run()

import json

def build_segments(station_names, edges):
    """Group consecutive edges on the same line into route segments."""
    segments = []
    for i, edge in enumerate(edges):
        if segments and segments[-1]["line"] == edge.line:
            seg = segments[-1]
            seg["to"] = station_names[i + 1]
            seg["stations"].append(station_names[i + 1])
            seg["distance"] += edge.weight
        else:
            segments.append({
                "sequence": len(segments) + 1,
                "line": edge.line,
                "from": station_names[i],
                "to": station_names[i + 1],
                "stations": [station_names[i], station_names[i + 1]],
                "distance": edge.weight,
            })
    for seg in segments:
        seg["stops"] = len(seg["stations"]) - 1
    return segments


def build_route_result(start, end, station_names, edges, mode="shortest_distance",
                       config=None):
    """Build the structured route summary dict."""
    distance = sum(e.weight for e in edges)
    if isinstance(distance, float) and distance.is_integer():
        distance = int(distance)
    segments = build_segments(station_names, edges)
    transfer_count = ShortestPath.count_transfers(edges)
    transfers = []
    for i in range(1, len(segments)):
        transfers.append({"station": segments[i]["from"],
                          "from_line": segments[i - 1]["line"],
                          "to_line": segments[i]["line"]})
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    fare = Sum_money(distance)
    return {
        "query": {
            "start": start,
            "end": end,
            "mode": mode,
            "max_transfers": getattr(config, "max_transfers", None),
            "line_preference": list(getattr(config, "preferred_lines", []) or []),
        },
        "path": list(station_names),
        "station_count": len(station_names),
        "segments": segments,
        "distance": {"meters": distance, "kilometers": round(distance / 1000.0, 2)},
        "fare": fare,
        "fare_tier": get_fare_tier(distance),
        "estimated_time": calculate_estimated_time(distance, transfer_count),
        "transfer_count": transfer_count,
        "transfer_stations": [t["station"] for t in transfers],
        "transfers": transfers,
        "lines": lines,
    }


def format_distance(meters):
    km = meters / 1000.0
    if meters < 1000:
        return f"{meters} meters ({km:.2f} km)"
    return f"{km:.2f} km ({meters} meters)"


def format_text(result):
    """Console-formatted route report."""
    q = result["query"]
    out = ["=" * 50,
           f"Route: {q['start']} -> {q['end']}  (mode: {q['mode']})",
           "-" * 50,
           f"Stations ({result['station_count']}): " + " -> ".join(result["path"]),
           "Segments:"]
    for i, seg in enumerate(result["segments"]):
        out.append(f"  {seg['sequence']}. [{seg['line']}] {seg['from']} -> {seg['to']} "
                   f"({seg['stops']} stop(s), {seg['distance'] / 1000.0:.2f} km)")
        out.append("       " + " -> ".join(seg["stations"]))
        if i < len(result["transfers"]):
            t = result["transfers"][i]
            out.append(f"     * Transfer at {t['station']}: "
                       f"{t['from_line']} -> {t['to_line']}")
    d = result["distance"]
    out.append(f"Total distance: {d['meters']} meters ({d['kilometers']:.2f} km)")
    if result["transfer_count"]:
        out.append(f"Transfers: {result['transfer_count']} "
                   f"(at {', '.join(result['transfer_stations'])})")
    else:
        out.append("Transfers: 0 (direct)")
    out.append("Lines involved: " + ", ".join(str(x) for x in result["lines"]))
    out.append(f"Estimated time: {result['estimated_time']} minutes")
    out.append(f"Fare tier: {result['fare_tier']}")
    out.append(f"Fare: {result['fare']} yuan")
    out.append("=" * 50)
    return "\n".join(out)

    return json.dumps(result, ensure_ascii=False, indent=2)

"""
Command-line interface of the Beijing Subway fare calculation system.

Usage::

    python main.py [--mode MODE] [--format text|json] [--output_file PATH]
                   [--max_transfers N] [--line_preference "Line 1"]
                   [--check_data] [--data_file PATH]

Enter queries as ``Starting Station; Ending Station`` (``,`` / ``->`` also
accepted).  Type ``quit`` to exit, ``help`` for help, ``lines`` to list lines.
"""

import argparse
import json
import re
import sys

from data_loader import DATA_FILE_PATH, DataLoadError
from data_manager import SubwayNetwork, check_data_integrity
from fare_calculator import FareCalculationError
from pathfinding import PathfindingConfig, VALID_MODES
from report import build_route_result, format_result

FORMAT_ERROR = ("Input format incorrect, please use "
                "'Starting Station; Ending Station' format")
SEPARATOR_RE = re.compile(r"\s*(?:;|；|,|，|->|→|\t)\s*")
EXIT_COMMANDS = {"quit", "exit", "q"}
HELP_TEXT = ("Enter 'Starting Station; Ending Station' to query a route and fare.\n"
             "Commands: 'lines' (list lines), 'help', 'quit'.")


class QueryError(Exception):
    """User-facing query error."""


def parse_query(text):
    """Split an input line into (start, end); raise QueryError on bad format."""
    if text is None:
        raise QueryError(FORMAT_ERROR)
    parts = SEPARATOR_RE.split(text.strip())
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise QueryError(FORMAT_ERROR)
    return parts[0].strip(), parts[1].strip()


def process_query(network, text, config):
    """Validate and run a query; return the result dict or raise QueryError."""
    start_raw, end_raw = parse_query(text)
    start = network.find_station(start_raw)
    if start is None:
        msg = f"Starting station '{start_raw}' does not exist, please check the station name"
        sugg = network.suggest_stations(start_raw)
        if sugg:
            msg += f" (did you mean: {', '.join(sugg)}?)"
        raise QueryError(msg)
    end = network.find_station(end_raw)
    if end is None:
        msg = f"Ending station '{end_raw}' does not exist, please check the station name"
        sugg = network.suggest_stations(end_raw)
        if sugg:
            msg += f" (did you mean: {', '.join(sugg)}?)"
        raise QueryError(msg)
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")

    try:
        _, names, edges = network.shortest_path(start, end, config)
    except Exception as exc:  # path calculation exception
        raise QueryError(f"Path calculation error: {exc}")
    if not edges:
        extra = ""
        if config.max_transfers is not None:
            extra = f" with at most {config.max_transfers} transfer(s)"
        raise QueryError(f"No route found from '{start}' to '{end}'{extra}")
    try:
        return build_route_result(start, end, names, edges, config.mode, config)
    except FareCalculationError as exc:
        raise QueryError(f"Fare calculation error: {exc}")


def write_output(path, results, fmt):
    """Export results to file (single result as object, several as list)."""
    try:
        with open(path, "w", encoding="utf-8") as fh:
            if fmt == "json":
                payload = results[0] if len(results) == 1 else results
                json.dump(payload, fh, ensure_ascii=False, indent=2)
                fh.write("\n")
            else:
                fh.write("\n\n".join(format_result(r, "text") for r in results))
                fh.write("\n")
    except OSError as exc:
        print(f"Error: cannot write output file '{path}': {exc}")


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", help="export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text",
                   help="output format")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                   help="path query mode")
    p.add_argument("--max_transfers", type=int, default=None,
                   help="maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="preferred line(s), e.g. 'Line 1' (comma separated)")
    p.add_argument("--check_data", action="store_true",
                   help="run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    return p


def resolve_preferences(network, pref):
    if not pref:
        return []
    resolved = []
    candidates = [pref] if network.find_line(pref) else re.split(r"[,，;；]", pref)
    for item in candidates:
        item = item.strip()
        if not item:
            continue
        line = network.find_line(item)
        if line is None:
            print(f"Warning: line '{item}' not found, preference ignored")
        elif line not in resolved:
            resolved.append(line)
    return resolved


def main(argv=None):
    args = build_parser().parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers cannot be negative")
        return 2

    try:
        network = SubwayNetwork(data_source_file=args.data_file)
    except DataLoadError as exc:
        print(f"Error: {exc}")
        return 2

    if args.check_data:
        check = check_data_integrity(network)
        print(check.report())
        return 0 if check.is_valid else 1

    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                               preferred_lines=resolve_preferences(network,
                                                                   args.line_preference))
    interactive = sys.stdin.isatty()
    if interactive:
        print("Beijing Subway Fare Calculation System")
        print(HELP_TEXT)

    results = []
    while True:
        try:
            line = input("Enter 'Starting Station; Ending Station': " if interactive else "")
        except (EOFError, KeyboardInterrupt):
            if interactive:
                print()
            break
        text = line.strip()
        if not text:
            continue
        cmd = text.lower()
        if cmd in EXIT_COMMANDS:
            break
        if cmd == "help":
            print(HELP_TEXT)
            continue
        if cmd == "lines":
            print("\n".join(network.get_line_names()))
            continue
        try:
            result = process_query(network, text, config)
        except QueryError as exc:
            print(exc)
            continue
        except Exception as exc:  # last-resort safety net
            print(f"Unexpected error: {exc}")
            continue
        print(format_result(result, args.format))
        results.append(result)
        if args.output_file:
            write_output(args.output_file, results, args.format)
    if interactive:
        print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

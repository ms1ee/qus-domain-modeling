from __future__ import annotations

from typing import List, Optional

SPECIAL_DRAWER = "Li Bin"
PROTECTED_GROUP = "Cheng Ru group"
FALLBACK_TRIGGER_DRAWERS = ["Chen Qixian", "Jin Duo"]
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"


def normalize_name(value: Optional[str]) -> str:
    """Normalize a person name for comparison."""
    if value is None:
        return ""
    return " ".join(str(value).strip().split()).lower()


def is_protected_group(department: Optional[str]) -> bool:
    """Return True if the department is the protected 'Cheng Ru group'."""
    norm = normalize_name(department)
    if norm.endswith(" group"):
        norm = norm[: -len(" group")].strip()
    return norm == "cheng ru"


class User:
    """A lottery participant."""

    def __init__(self, name: str, department: str):
        self.name = str(name).strip()
        self.department = str(department).strip()

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self):
        return f"{self.name} ({self.department})"


class Prize:
    """A prize with quantity, avoidance groups and associated drawer."""

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups: List[str] = list(avoid_groups) if avoid_groups else []
        if associated_drawer is not None:
            associated_drawer = str(associated_drawer).strip() or None
        self.associated_drawer: Optional[str] = associated_drawer

    def rule_description(self) -> str:
        parts = ["Previous winners are excluded"]
        drawer = self.associated_drawer
        if drawer and normalize_name(drawer) == normalize_name(SPECIAL_DRAWER):
            parts.append(f"special drawer {SPECIAL_DRAWER}: avoidance groups are NOT applied")
        elif self.avoid_groups:
            parts.append("avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("no avoidance groups")
        if drawer and normalize_name(drawer) == normalize_name(FALLBACK_TARGET_DRAWER):
            parts.append(
                f"fallback: if {' and '.join(FALLBACK_TRIGGER_DRAWERS)} drew no "
                f"'{PROTECTED_GROUP}' member, pool is limited to '{PROTECTED_GROUP}'"
            )
        return "; ".join(parts)

    def __repr__(self):
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
            f"avoid_groups={self.avoid_groups!r}, associated_drawer={self.associated_drawer!r})"
        )

    def __str__(self):
        drawer = self.associated_drawer or "-"
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "-"
        return f"{self.name} x{self.quantity} | drawer: {drawer} | avoid: {avoid}"

class ListManager:
    """Manages participant lists."""

    def __init__(self):
        self.users: List[User] = []
        self.source: str = ""

    # ------------------------------------------------------------------ load
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._load_from_path(DEFAULT_LIST_PATH)
        self.source = "default"
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File Error: file '{path}' not found"]
        users, errors = self._load_from_path(path)
        self.source = str(path)
        return users, errors

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        lines: List[str]
        if path == DEFAULT_LIST_PATH and not os.path.isfile(path):
            lines = list(BUILTIN_DEFAULT_LINES)
        else:
            try:
                with open(path, "r", encoding="utf-8-sig") as f:
                    lines = f.read().splitlines()
            except (OSError, UnicodeDecodeError) as exc:
                return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: List[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - '{line}' (expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # -------------------------------------------------------------- validate
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        duplicates: List[User] = []
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    # ---------------------------------------------------------------- groups
    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, ["Cheng Ru group"], "Ma Jiaqi"),
    ("Special Prize - Lucky Red Envelope", 1, ["Technology Department"], "Li Bin"),
]


class PrizeManager:
    """Manages prizes and their drawing order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        name = str(name).strip()
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def get_prize(self, name: str) -> Optional[Prize]:
        for p in self.prizes:
            if p.name == name:
                return p
        return None

    def set_avoid_groups(self, index: int, groups: List[str]) -> bool:
        if index < 0 or index >= len(self.prizes):
            return False
        cleaned = []
        for g in groups:
            g = str(g).strip()
            if g and g not in cleaned:
                cleaned.append(g)
        self.prizes[index].avoid_groups = cleaned
        return True

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Result recording, display and export."""

from __future__ import annotations

import csv
import datetime
import os
from typing import Dict, List, Tuple


class Winner:
    """A winning record."""

    def __init__(self, user, prize, order: int = 0):
        self.user = user
        self.prize = prize
        self.prize_name = getattr(prize, "name", str(prize))
        self.order = order

    def __repr__(self):
        return f"Winner(user={self.user!r}, prize={self.prize_name!r}, order={self.order})"

    def __str__(self):
        return (
            f"Congratulations {self.user.name} ({self.user.department}) "
            f"for winning {self.prize_name}"
        )


class ResultManager:
    """Maintains the winner list."""

    def __init__(self):
        self.winners: List[Winner] = []

    def has_won(self, user) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def add_winner(self, user, prize) -> bool:
        if self.has_won(user):
            return False
        self.winners.append(Winner(user, prize, len(self.winners) + 1))
        return True

    def get_all_winners(self) -> List[Winner]:
        return list(self.winners)

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes:
            pname = getattr(p, "name", str(p))
            result[pname] = []
        for w in self.winners:
            if w.prize_name in result:
                result[w.prize_name].append(w)
            else:
                result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes, path=None) -> Tuple[bool, str]:
        if not self.winners:
            return False, "No winners to export"
        if path is None:
            date_str = datetime.datetime.now().strftime("%Y%m%d")
            path = os.path.join(os.getcwd(), f"awards_result_{date_str}.txt")
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for pname, winners in grouped.items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, pname])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery process control and rule engine."""

from __future__ import annotations

import random
from typing import Callable, List, Optional

from lottery_system.models import (
    FALLBACK_TARGET_DRAWER,
    FALLBACK_TRIGGER_DRAWERS,
    PROTECTED_GROUP,
    SPECIAL_DRAWER,
    User,
    is_protected_group,
    normalize_name,
)
from lottery_system.services.result_manager import ResultManager


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryManager:
    """Executes lottery rounds applying all rules."""

    def __init__(self, users, result_manager: Optional[ResultManager] = None):
        self.users: List[User] = list(users)
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winner_names = set(w.user.name for w in self.result_manager.winners)
        self.last_two_draw_results: List[bool] = []
        self._rng = random.SystemRandom()

    # ------------------------------------------------------------ rule engine
    @staticmethod
    def _is_drawer(prize, name: str) -> bool:
        return normalize_name(getattr(prize, "associated_drawer", None)) == normalize_name(name)

    def _fallback_triggered(self, prize) -> bool:
        if not self._is_drawer(prize, FALLBACK_TARGET_DRAWER):
            return False
        if len(self.last_two_draw_results) < 2:
            return False
        return not any(self.last_two_draw_results[-2:])

    def _available_users(self) -> List[User]:
        return [u for u in self.users if u.name not in self.winner_names]

    def _get_eligible_pool(self, prize) -> List[User]:
        base = self._available_users()
        if self._is_drawer(prize, SPECIAL_DRAWER):
            return base
        if self._fallback_triggered(prize):
            protected = [u for u in base if is_protected_group(u.department)]
            if protected:
                return protected
        avoid = set(g.strip() for g in (prize.avoid_groups or []))
        return [u for u in base if u.department not in avoid]

    def describe_pool_rule(self, prize) -> str:
        if self._is_drawer(prize, SPECIAL_DRAWER):
            return f"Special drawer {SPECIAL_DRAWER}: only previous winners are excluded."
        if self._fallback_triggered(prize):
            return (
                f"Fallback triggered: pool limited to '{PROTECTED_GROUP}' members "
                f"(avoidance groups overridden)."
            )
        return prize.rule_description()

    def _update_fallback_state(self, prize, round_winners: List[User]) -> None:
        if any(self._is_drawer(prize, n) for n in FALLBACK_TRIGGER_DRAWERS):
            self.last_two_draw_results.append(
                any(is_protected_group(u.department) for u in round_winners)
            )
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif self._is_drawer(prize, FALLBACK_TARGET_DRAWER):
            self.last_two_draw_results = []

    # ---------------------------------------------------------------- drawing
    def draw_one(self, prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        if winner.name in self.winner_names:
            return None
        self.winner_names.add(winner.name)
        self.result_manager.add_winner(winner, prize)
        return winner

    def run_round(
        self,
        prize,
        round_number: int = 1,
        input_func: Callable[[str], str] = input,
        output_func: Callable[[str], None] = print,
        interactive: bool = True,
    ) -> List[User]:
        output_func("")
        output_func(f"===== Round {round_number}: {prize.name} =====")
        output_func(f"Quantity: {prize.quantity}")
        output_func(f"Drawer: {prize.associated_drawer or '-'}")
        output_func(f"Rules: {self.describe_pool_rule(prize)}")

        round_winners: List[User] = []
        terminated = False
        for i in range(prize.quantity):
            if interactive:
                cmd = self._read(input_func,
                                 "Press Enter to draw, or type 'pause' / 'terminate': ")
                if cmd == "pause":
                    output_func("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                    while True:
                        c = self._read(input_func, "(paused) > ")
                        if c == "continue":
                            output_func("Lottery resumed.")
                            break
                        if c == "terminate":
                            terminated = True
                            break
                        output_func("Unknown command. Type 'continue' or 'terminate'.")
                elif cmd == "terminate":
                    terminated = True
            if terminated:
                output_func(f"Round terminated. Skipping remaining draws for {prize.name}.")
                break
            output_func(f"Currently drawing the {_ordinal(i + 1)} winner...")
            winner = self.draw_one(prize)
            if winner is None:
                output_func("No eligible participants remain for this prize. Round ends.")
                break
            round_winners.append(winner)
            output_func(
                f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}"
            )

        self._update_fallback_state(prize, round_winners)
        output_func(f"--- Round {round_number} results ({prize.name}) ---")
        if round_winners:
            for idx, u in enumerate(round_winners, 1):
                output_func(f"{idx}. {u.name} ({u.department})")
        else:
            output_func("No winners in this round.")
        return round_winners

    @staticmethod
    def _read(input_func, prompt: str) -> str:
        try:
            return (input_func(prompt) or "").strip().lower()
        except EOFError:
            return ""

"""Command-line interface for the lottery system."""

from __future__ import annotations

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        return "0"


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_manager = None

    # ------------------------------------------------------------- list
    def _handle_loaded(self, users, errors) -> None:
        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            print("Validation problems detected:")
            for e in errors:
                print("  " + e)
            for d in duplicates:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            choice = _ask("Remove invalid data and continue (r) or terminate upload (t)? ").lower()
            if choice != "r":
                print("Upload terminated. Current list unchanged.")
                return
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        self.result_manager = ResultManager()
        self.lottery_manager = None
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._handle_loaded(users, errors)

    def load_custom(self):
        path = _ask("Enter txt file path (each line 'Name,Department Group'): ")
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print(e)
            return
        self._handle_loaded(users, errors)

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    # ------------------------------------------------------------- prizes
    def _select_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            raw = _ask("Select avoidance groups by number or name (comma separated, empty for none): ")
        else:
            raw = _ask("Enter avoidance group names (comma separated, empty for none): ")
        selected = []
        for part in raw.replace("，", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                selected.append(groups[int(part) - 1])
            else:
                selected.append(part)
        return selected

    def load_default_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prize library loaded.")

    def add_prize(self):
        name = _ask("Prize name: ")
        qty = _ask("Quantity: ")
        drawer = _ask("Associated lottery person (empty for none): ") or None
        groups = self._select_groups()
        try:
            self.prize_manager.add_prize(name, qty, groups, drawer)
            print("Prize added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def list_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return False
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")
        return True

    def set_avoid_groups(self):
        if not self.list_prizes():
            return
        idx = _ask("Select prize number: ")
        if not idx.isdigit():
            print("Invalid number.")
            return
        groups = self._select_groups()
        if self.prize_manager.set_avoid_groups(int(idx) - 1, groups):
            print("Avoidance groups updated.")
        else:
            print("Invalid prize number.")

    def reorder(self):
        if not self.list_prizes():
            return
        raw = _ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated:")
            self.list_prizes()
        else:
            print("Invalid order; must be a permutation of all serial numbers.")

    # ------------------------------------------------------------- lottery
    def start_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery_manager is None:
            self.lottery_manager = LotteryManager(self.users, self.result_manager)
        for n, prize in enumerate(prizes, 1):
            self.lottery_manager.run_round(prize, n)
        print("\nAll rounds completed.")
        self.show_results()

    def show_results(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("===== Complete Winner List =====")
        any_w = False
        for pname, winners in grouped.items():
            for w in winners:
                any_w = True
                print(f"  {w.user.name} | {w.user.department} | {pname}")
        if not any_w:
            print("  No winners yet.")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ------------------------------------------------------------- loop
    def run(self):
        actions = {
            "1": self.load_default,
            "2": self.load_custom,
            "3": self.view_groups,
            "4": self.load_default_prizes,
            "5": self.add_prize,
            "6": self.set_avoid_groups,
            "7": self.list_prizes,
            "8": self.reorder,
            "9": self.start_lottery,
            "10": self.show_results,
            "11": self.export,
        }

"""Core data models of the lottery system."""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import List, Optional

# Prize level keywords, ordered from the highest level to the lowest.
_LEVEL_KEYWORDS = [
    (0, "Special Prize", ("special", "grand", "特等")),
    (1, "First Prize", ("first", "一等")),
    (2, "Second Prize", ("second", "二等")),
    (3, "Third Prize", ("third", "三等")),
    (4, "Fourth Prize", ("fourth", "四等")),
    (5, "Fifth Prize", ("fifth", "五等")),
]
CUSTOM_LEVEL_RANK = 99


def normalize_group(name: Optional[str]) -> str:
    """Normalize a department/group name for comparison."""
    if not name:
        return ""
    text = " ".join(str(name).strip().lower().split())
    if text.endswith(" group"):
        text = text[: -len(" group")].strip()
    return text


def level_rank_from_name(name: str) -> int:
    lowered = (name or "").lower()
    for rank, _label, keys in _LEVEL_KEYWORDS:
        for key in keys:
            if key in lowered:
                return rank
    return CUSTOM_LEVEL_RANK


def level_label_from_rank(rank: int) -> str:
    for r, label, _keys in _LEVEL_KEYWORDS:
        if r == rank:
            return label
    return "Custom Prize"


@dataclass
class User:
    """A lottery participant."""

    name: str
    department: str

    def __post_init__(self) -> None:
        self.name = str(self.name).strip()
        self.department = str(self.department).strip()

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None
    level: Optional[str] = None

    def __post_init__(self) -> None:
        self.name = str(self.name).strip()
        self.quantity = int(self.quantity)
        self.avoid_groups = [str(g).strip() for g in (self.avoid_groups or []) if str(g).strip()]
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None
        if not self.level:
            self.level = level_label_from_rank(level_rank_from_name(self.name))

    @property
    def level_rank(self) -> int:
        """Lower value means higher prize level."""
        rank = level_rank_from_name(self.level or "")
        if rank == CUSTOM_LEVEL_RANK:
            rank = level_rank_from_name(self.name)
        return rank

    def __str__(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "None"
        return (f"{self.name} | Quantity: {self.quantity} | Avoid groups: {avoid} "
                f"| Lottery person: {drawer}")

_SEPARATOR = re.compile(r"[,\t，]")


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.users: List[User] = []
        self.source: str = ""
        self.file_path: str = ""

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_TEXT.splitlines())
        self.source = "default"
        self.file_path = ""
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        path = str(path).strip().strip('"').strip("'") if path is not None else ""
        if not path:
            return [], ["No file path provided"]
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        if not path.lower().endswith(".txt"):
            return [], [f"Unsupported file type (only .txt is supported): {path}"]
        users, errors = self._load_from_path(path)
        self.source = "custom"
        self.file_path = path
        return users, errors

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        content = None
        last_error = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=encoding) as fh:
                    content = fh.read()
                break
            except UnicodeDecodeError as exc:
                last_error = exc
            except OSError as exc:
                return [], [f"Unable to read file {path}: {exc}"]
        if content is None:
            return [], [f"Unable to decode file {path}: {last_error}"]
        return self._parse_lines(content.splitlines())

    def _parse_lines(self, lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for index, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [p.strip() for p in _SEPARATOR.split(line)]
            if len(parts) != 2:
                errors.append(f"Line {index}: Format Error - expected 'Name,Department', got '{line}'")
                continue
            name, department = parts
            if not name or not department:
                errors.append(f"Line {index}: Format Error - empty name or department in '{line}'")
                continue
            users.append(User(name, department))
        return users, errors

    # --------------------------------------------------------------- validation
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users or []:
            key = user.name.strip().lower()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    @staticmethod
    def remove_user(users: List[User], name: str) -> List[User]:
        target = (name or "").strip().lower()
        return [u for u in users if u.name.strip().lower() != target]

    # ------------------------------------------------------------------- groups
    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users or []:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

PRESET_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], None),
    ("Second Prize - Shopping Cards", 3, [], None),
    ("First Prize - 1 Day Annual Leave", 1, [], None),
]


class PrizeManager:
    """Holds prizes in lottery-round order (default: addition order)."""

    def __init__(self) -> None:
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in PRESET_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = str(name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name.lower() == name.lower() for p in self._prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups: List[str] = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer: Optional[str] = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        prize = Prize(name, qty, groups, drawer or None)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        try:
            positions = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        n = len(self._prizes)
        if len(positions) != n or sorted(positions) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in positions]
        return True

"""Lottery rule engine and drawing logic."""

from __future__ import annotations

import random
from typing import List, Optional, Set

from lottery_system.models import Prize, User, normalize_group

RELAXED_DRAWER = "Li Bin"
MONITORED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _same_person(a: Optional[str], b: str) -> bool:
    return a is not None and " ".join(a.split()).lower() == b.lower()


class LotteryManager:
    """Applies filtering rules, the fallback mechanism and anti-duplication."""

    def __init__(self, users, rng: Optional[random.Random] = None) -> None:
        self.users: List[User] = list(users or [])
        self.winner_names: Set[str] = set()
        self.last_two_draw_results: List[bool] = []
        self.fallback_triggered: bool = False
        self._rng = rng or random.SystemRandom()

    # ------------------------------------------------------------------ helpers
    @staticmethod
    def _key(user: User) -> str:
        return user.name.strip().lower()

    def has_won(self, user: User) -> bool:
        return self._key(user) in self.winner_names

    def mark_winner(self, user: User) -> bool:
        key = self._key(user)
        if key in self.winner_names:
            return False
        self.winner_names.add(key)
        return True

    @staticmethod
    def is_protected_member(user: User) -> bool:
        return normalize_group(user.department) == normalize_group(PROTECTED_GROUP)

    def is_fallback_active(self, prize: Prize) -> bool:
        if not _same_person(prize.associated_drawer, FALLBACK_TARGET_DRAWER):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    # ---------------------------------------------------------------- rule core
    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        not_won = [u for u in self.users if not self.has_won(u)]

        if self.is_fallback_active(prize):
            protected = [u for u in not_won if self.is_protected_member(u)]
            if protected:
                self.fallback_triggered = True
                return protected
            # No protected member available: fall through to normal rules.

        if _same_person(prize.associated_drawer, RELAXED_DRAWER):
            # Relaxed rules: avoidance groups are not applied,
            # only previous winners are excluded.
            return not_won

        avoid = {normalize_group(g) for g in prize.avoid_groups}
        return [u for u in not_won if normalize_group(u.department) not in avoid]

    def get_rule_description(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        drawer = prize.associated_drawer
        if self.is_fallback_active(prize):
            parts.append(f"FALLBACK ACTIVE: pool limited to '{PROTECTED_GROUP}' members "
                         f"(overrides avoidance groups)")
        elif _same_person(drawer, RELAXED_DRAWER):
            parts.append(f"Relaxed rules for lottery person {RELAXED_DRAWER}: avoidance groups ignored")
        if prize.avoid_groups and not _same_person(drawer, RELAXED_DRAWER):
            parts.append("Avoidance groups: " + ", ".join(prize.avoid_groups))
        elif not prize.avoid_groups:
            parts.append("No avoidance groups")
        parts.append(f"Lottery person: {drawer or 'None'}")
        return "; ".join(parts)

    # ------------------------------------------------------------------ drawing
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self._rng.choice(pool)
        self.mark_winner(winner)
        return winner

    def draw(self, prize: Prize, count: Optional[int] = None) -> List[User]:
        count = prize.quantity if count is None else count
        winners: List[User] = []
        for _ in range(count):
            w = self.draw_one(prize)
            if w is None:
                break
            winners.append(w)
        self.record_round_result(prize, winners)
        return winners

    def record_round_result(self, prize: Prize, winners: List[User]) -> None:
        """Update fallback monitoring state after a round has finished."""
        drawer = prize.associated_drawer
        if any(_same_person(drawer, m) for m in MONITORED_DRAWERS):
            self.last_two_draw_results.append(any(self.is_protected_member(w) for w in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            # Consecutiveness broken or fallback consumed by the target drawer.
            self.last_two_draw_results = []

"""Winner records, summary and export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime
from typing import Dict, List, Optional

from lottery_system.models import Prize, User


@dataclass
class Winner:
    user: User
    prize: Prize
    draw_order: int
    win_time: datetime = field(default_factory=datetime.now)


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []
        self._counter = 0

    @property
    def total_winners(self) -> int:
        return len(self.winners)

    def has_won(self, user: User) -> bool:
        key = user.name.strip().lower()
        return any(w.user.name.strip().lower() == key for w in self.winners)

    def add_winner(self, user, prize) -> Optional[Winner]:
        if self.has_won(user):
            return None
        self._counter += 1
        winner = Winner(user=user, prize=prize, draw_order=self._counter)
        self.winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        prizes = list(prizes or [])
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda item: (item[1].level_rank, item[0]))
        result: Dict[str, List[Winner]] = {}
        for _, prize in indexed:
            items = [w for w in self.winners if w.prize.name == prize.name]
            items.sort(key=lambda w: (w.win_time, w.draw_order))
            result[prize.name] = items
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        grouped = self.get_winners_by_prize(prizes)
        # Include winners of prizes not in the given list, too.
        listed = {name for name in grouped}
        extra = [w for w in self.winners if w.prize.name not in listed]
        try:
            with open(path, "w", encoding="utf-8", newline="") as fh:
                writer = csv.writer(fh)
                writer.writerow(["No.", "Name", "Department", "Prize Name", "Prize Level", "Win Time"])
                no = 0
                for prize_name, winners in grouped.items():
                    for w in winners:
                        no += 1
                        writer.writerow([no, w.user.name, w.user.department, prize_name,
                                         w.prize.level, w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
                for w in extra:
                    no += 1
                    writer.writerow([no, w.user.name, w.user.department, w.prize.name,
                                     w.prize.level, w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""

from __future__ import annotations

from typing import List, Optional, Set

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class _Exit(Exception):
    pass


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


def ask(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        raise _Exit()


class LotteryCLI:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery: Optional[LotteryManager] = None
        self.completed: Set[str] = set()

    # ------------------------------------------------------------------- lists
    def _accept_list(self, users: List[User], errors: List[str]) -> None:
        valid, duplicates = self.list_manager.validate_users(users)
        if errors:
            print("Format errors detected:")
            for e in errors:
                print("  " + e)
        if duplicates:
            print("Duplicate names detected:")
            for d in duplicates:
                print(f"  {d.name} ({d.department})")
        if errors or duplicates:
            choice = ask("Input 'r' to remove invalid data and continue, 't' to terminate upload: ").strip().lower()
            if choice != "r":
                print("Upload terminated. Current list unchanged.")
                return
        while True:
            name = ask("Input a name to remove manually (Enter to finish): ").strip()
            if not name:
                break
            before = len(valid)
            valid = self.list_manager.remove_user(valid, name)
            print("Removed." if len(valid) < before else "Name not found.")
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        self.lottery = LotteryManager(self.users)
        self.result_manager = ResultManager()
        self.completed = set()
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._accept_list(users, errors)

    def load_custom(self) -> None:
        path = ask("Input txt file path (one 'Name,Department' per line): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            print("No participants loaded.")
            return
        self._accept_list(users, errors)

    def view_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Current valid groups:")
        for i, (dept, cnt) in enumerate(info.items(), 1):
            print(f"  {i}. {dept}: {cnt} people")

    # ------------------------------------------------------------------ prizes
    def load_preset_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prizes loaded.")

    def add_custom_prize(self) -> None:
        name = ask("Prize name: ").strip()
        qty = ask("Quantity: ").strip()
        avoid: List[str] = []
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            sel = ask("Avoidance groups (serial numbers separated by spaces/commas, Enter for none): ")
            for token in sel.replace(",", " ").split():
                if token.isdigit() and 1 <= int(token) <= len(groups):
                    if groups[int(token) - 1] not in avoid:
                        avoid.append(groups[int(token) - 1])
                else:
                    print(f"  Ignored invalid selection: {token}")
        else:
            sel = ask("Avoidance groups (names separated by commas, Enter for none): ")
            avoid = [s.strip() for s in sel.split(",") if s.strip()]
        drawer = ask("Associated lottery person (Enter for none): ").strip() or None
        try:
            prize = self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Added: {prize}")
        except ValueError as exc:
            print(f"Error: {exc}")

    def view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            status = " [done]" if p.name in self.completed else ""
            print(f"  {i}. {p}{status}")

    def reorder(self) -> None:
        self.view_prizes()
        if not self.prize_manager.get_prizes():
            return
        text = ask("Input new order of serial numbers (e.g. 3 1 2): ")
        order = text.replace(",", " ").split()
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.view_prizes()
        else:
            print("Invalid order. Must be a permutation of all serial numbers.")

    # ----------------------------------------------------------------- lottery
    def _control(self) -> str:
        """Return 'draw' or 'terminate'."""
        while True:
            cmd = ask("[Enter]=draw | pause | terminate > ").strip().lower()
            if cmd in ("", "draw", "continue"):
                return "draw"
            if cmd == "terminate":
                return "terminate"
            if cmd == "pause":
                print("Lottery paused. Input 'continue' to resume or 'terminate' to end this round.")
                while True:
                    c = ask("(paused) > ").strip().lower()
                    if c == "continue":
                        print("Lottery resumed.")
                        return "draw"
                    if c == "terminate":
                        return "terminate"
                    print("Lottery is paused. Input 'continue' or 'terminate'.")
            print("Unknown command.")

    def run_lottery(self) -> None:
        if not self.users or self.lottery is None:
            print("Please load a participant list first.")
            return
        prizes = [p for p in self.prize_manager.get_prizes() if p.name not in self.completed]
        if not prizes:
            print("No pending prizes to draw.")
            return
        for round_no, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {round_no}: {prize.name}")
            print(f"Quantity: {prize.quantity}")
            print(f"Rules: {self.lottery.get_rule_description(prize)}")
            print("=" * 60)
            round_winners: List[User] = []
            for i in range(1, prize.quantity + 1):
                if self._control() == "terminate":
                    print("Current round terminated, skipping.")
                    break
                print(f"Currently drawing the {ordinal(i)} winner...")
                winner = self.lottery.draw_one(prize)
                if winner is None:
                    print("No eligible participants left for this prize.")
                    break
                self.result_manager.add_winner(winner, prize)
                round_winners.append(winner)
                print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}")
            self.lottery.record_round_result(prize, round_winners)
            self.completed.add(prize.name)
            print(f"\nRound {round_no} results ({prize.name}):")
            if round_winners:
                for i, w in enumerate(round_winners, 1):
                    print(f"  {i}. {w.name} ({w.department})")
            else:
                print("  No winners.")
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            print(f"{prize_name}:")
            for w in winners:

            "3": ("View groups and member counts", self.view_groups),
            "4": ("Load preset prize library", self.load_preset_prizes),
            "5": ("Add custom prize", self.add_custom_prize),
            "6": ("View prizes", self.view_prizes),
            "7": ("Reorder prizes", self.reorder),
            "8": ("Start lottery", self.run_lottery),
            "9": ("View winner summary", self.show_summary),
            "10": ("Export results", self.export),
        }
        try:
            while True:
                print("\n===== Team Building Smart Lottery System =====")
                for key, (label, _) in actions.items():
                    print(f"{key}. {label}")
                print("0. Exit")
                choice = ask("Select: ").strip()
                if choice == "0":
                    break
                if choice in actions:
                    actions[choice][1]()
                else:
                    print("Invalid selection.")
        except (_Exit, KeyboardInterrupt):
            print()
        print("Goodbye.")

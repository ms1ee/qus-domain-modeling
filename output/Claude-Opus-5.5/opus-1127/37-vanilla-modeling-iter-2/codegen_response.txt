"""Core data models for the lottery system."""


class User:
    """A participant of the lottery."""

    def __init__(self, name, department):
        self.name = str(name).strip()
        self.department = str(department).strip()

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self):
        return f"{self.name} ({self.department})"


class Prize:
    """A prize to be drawn in one lottery round."""

    def __init__(self, name, quantity, avoid_groups=None, associated_drawer=None):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups = [str(g).strip() for g in (avoid_groups or []) if str(g).strip()]
        if associated_drawer is not None:
            associated_drawer = str(associated_drawer).strip() or None
        self.associated_drawer = associated_drawer

    def __repr__(self):
        return (f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
                f"avoid_groups={self.avoid_groups!r}, "
                f"associated_drawer={self.associated_drawer!r})")

    def __str__(self):
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "None"
        return (f"{self.name} x{self.quantity} | Avoid groups: {avoid} | "
                f"Lottery person: {drawer}")

class ListManager:
    """Loads, validates and analyses participant lists."""

    def __init__(self):
        self.load_mode = None
        self.source_file_path = None

    # ------------------------------------------------------------------ loading
    def load_default_list(self):
        """Load the built-in default list. Returns (users, errors)."""
        self.load_mode = "default"
        if os.path.isfile(DEFAULT_LIST_PATH):
            self.source_file_path = DEFAULT_LIST_PATH
            return self._load_from_path(DEFAULT_LIST_PATH)
        self.source_file_path = None
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path):
        """Load a custom txt list. Returns (users, errors)."""
        self.load_mode = "custom"
        self.source_file_path = path
        return self._load_from_path(path)

    def _load_from_path(self, path):
        if not path or not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        lines = None
        for enc in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=enc) as f:
                    lines = f.read().splitlines()
                break
            except UnicodeDecodeError:
                continue
            except OSError as e:
                return [], [f"Failed to read file {path}: {e}"]
        if lines is None:
            return [], [f"Failed to decode file {path}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines):
        users, errors = [], []
        for i, raw in enumerate(lines, 1):
            line = raw.strip().lstrip("\ufeff")
            if not line:
                continue
            line = line.replace("\uff0c", ",")
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {i}: Format Error - expected 'Name,Department Group', got '{raw.strip()}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # --------------------------------------------------------------- validation
    def validate_users(self, users):
        """Split users into (valid_users, duplicates) by name; first occurrence is kept."""
        seen = set()
        valid, duplicates = [], []
        for u in users:
            if u.name in seen:
                duplicates.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users):
        """Return dict of department -> member count (in first-seen order)."""
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),

    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),

class PrizeManager:
    """Manages the prize library and the drawing sequence."""

    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = str(name).strip() if name is not None else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        if isinstance(avoid_groups, str):
            avoid_groups = [g for g in avoid_groups.split(",")]
        prize = Prize(name, qty, avoid_groups or [], associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            idx = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery process control and rule engine."""

import random

SPECIAL_DRAWER = "Li Bin"
FALLBACK_TRIGGERS = ["Chen Qixian", "Jin Duo"]
FALLBACK_TARGET = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def _norm_group(name):
    s = str(name or "").strip().lower().replace(" ", "")
    if s.endswith("group"):
        s = s[: -len("group")]
    return s


def _same_group(a, b):
    return _norm_group(a) == _norm_group(b)


def _norm_person(name):
    return " ".join(str(name or "").split()).lower()


def ordinal(n):
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryManager:
    """Executes draws while applying the rule engine."""

    def __init__(self, users, result_manager=None, rng=None):
        self.users = list(users)
        self.result_manager = result_manager
        self.rng = rng or random.Random()
        self.winner_names = set()
        self.winners = []  # list of (user, prize)
        self.last_two_draw_results = []

    # -------------------------------------------------------------- rule engine
    def is_winner(self, user):
        return user.name in self.winner_names

    def is_fallback_active(self, prize):
        if _norm_person(prize.associated_drawer) != _norm_person(FALLBACK_TARGET):
            return False
        last = self.last_two_draw_results[-2:]
        return len(last) >= 2 and not any(last)

    def _get_eligible_pool(self, prize):
        not_won = [u for u in self.users if not self.is_winner(u)]
        if self.is_fallback_active(prize):
            return [u for u in not_won if _same_group(u.department, PROTECTED_GROUP)]
        if _norm_person(prize.associated_drawer) == _norm_person(SPECIAL_DRAWER):
            return not_won
        avoid = [_norm_group(g) for g in prize.avoid_groups]
        return [u for u in not_won if _norm_group(u.department) not in avoid]

    def describe_rules(self, prize):
        drawer = prize.associated_drawer or "None"
        parts = [f"Lottery person: {drawer}", "Previous winners excluded"]
        if self.is_fallback_active(prize):
            parts.append(
                f"FALLBACK ACTIVE: pool limited to '{PROTECTED_GROUP}' members "
                f"(previous rounds by {' and '.join(FALLBACK_TRIGGERS)} had no {PROTECTED_GROUP} winners)"
            )
        elif _norm_person(prize.associated_drawer) == _norm_person(SPECIAL_DRAWER):
            parts.append(f"Special lottery person {SPECIAL_DRAWER}: avoidance groups not applied")
        elif prize.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")
        return "; ".join(parts)

    def record_round_result(self, prize, round_winners):
        drawer = _norm_person(prize.associated_drawer)
        if drawer in [_norm_person(p) for p in FALLBACK_TRIGGERS]:
            hit = any(_same_group(u.department, PROTECTED_GROUP) for u in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm_person(FALLBACK_TARGET):
            self.last_two_draw_results = []

    # ----------------------------------------------------------------- drawing
    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winner_names.add(user.name)
        self.winners.append((user, prize))
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return user

    def _prompt_control(self, input_func, output_func):
        """Returns 'draw' or 'terminate'."""
        paused = False
        while True:
            prompt = ("[Paused] type 'continue' to resume or 'terminate' to end round: "
                      if paused else
                      "Press Enter to draw (or type 'pause' / 'terminate'): ")
            try:
                cmd = input_func(prompt).strip().lower()
            except EOFError:
                return "draw"
            if cmd == "terminate":
                return "terminate"
            if paused:
                if cmd == "continue":
                    output_func("Lottery resumed.")
                    return "draw"
                output_func("Lottery is paused.")
                continue
            if cmd == "pause":
                paused = True
                output_func("Lottery paused.")
                continue
            if cmd in ("", "continue", "draw"):
                return "draw"
            output_func("Unknown command.")

    def run_round(self, prize, round_number=1, input_func=input, output_func=print,
                  interactive=True):
        output_func("=" * 60)
        output_func(f"Round {round_number}: {prize.name} | Quantity: {prize.quantity}")
        output_func(f"Rules: {self.describe_rules(prize)}")
        output_func("=" * 60)
        round_winners = []
        for i in range(prize.quantity):
            if interactive:
                if self._prompt_control(input_func, output_func) == "terminate":
                    output_func("Current round terminated and skipped.")
                    break
            output_func(f"Currently drawing the {ordinal(i + 1)} winner...")
            user = self.draw_one(prize)
            if user is None:
                output_func("No eligible participants remain for this prize. Round ended.")
                break
            output_func(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            round_winners.append(user)
        self.record_round_result(prize, round_winners)
        output_func(f"--- Round {round_number} results ({prize.name}) ---")
        if round_winners:
            for n, u in enumerate(round_winners, 1):
                output_func(f"  {n}. {u.name} ({u.department})")
        else:
            output_func("  No winners in this round.")
        return round_winners

"""Result management and export."""

import csv
import os
import re
from datetime import datetime

_LEVEL_WORDS = [
    ("grand", 0), ("special", 0), ("first", 1), ("1st", 1), ("second", 2), ("2nd", 2),
    ("third", 3), ("3rd", 3), ("fourth", 4), ("4th", 4), ("fifth", 5), ("5th", 5),
]


def prize_level(name):
    """Smaller value = higher prize. Unknown levels return None."""
    low = str(name).lower()
    for word, lvl in _LEVEL_WORDS:
        if re.search(r"\b" + re.escape(word) + r"\b", low):
            return lvl
    return None


class Winner:
    def __init__(self, user, prize, time=None, order=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.order = order

    @property
    def prize_name(self):
        return self.prize.name if hasattr(self.prize, "name") else str(self.prize)

    def __repr__(self):
        return f"Winner(user={self.user!r}, prize={self.prize_name!r}, order={self.order})"


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize, datetime.now(), len(self.winners) + 1)
        self.winners.append(w)
        return w

    def _sorted_prize_names(self, prizes):
        names = []
        for p in prizes or []:
            n = p.name if hasattr(p, "name") else str(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        indexed = list(enumerate(names))

        def key(item):
            idx, n = item
            lvl = prize_level(n)
            return (lvl if lvl is not None else 100, -idx)

        return [n for _, n in sorted(indexed, key=key)]

    def get_winners_by_prize(self, prizes):
        result = {}
        for n in self._sorted_prize_names(prizes):
            result[n] = sorted(
                [w for w in self.winners if w.prize_name == n],
                key=lambda w: (w.time, w.order),
            )
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export."
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, name,
                                         w.time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


class LotterySystem:
    def __init__(self, input_func=input, output_func=print):
        self.inp = input_func
        self.out = output_func
        self.status = "idle"
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery_manager = None

    def ask(self, prompt):
        try:
            return self.inp(prompt).strip()
        except EOFError:
            return "0"

    # ----------------------------------------------------------- list handling
    def _accept_list(self, users, errors):
        for e in errors:
            self.out(f"  [!] {e}")
        valid, dups = self.list_manager.validate_users(users)
        for d in dups:
            self.out(f"  [!] Duplicate name: {d.name} ({d.department})")
        if errors or dups:
            choice = self.ask("Invalid data found. (r)emove invalid data and continue, "
                              "or (t)erminate upload? [r/t]: ").lower()
            if choice.startswith("t"):
                self.out("Upload terminated. Current list unchanged.")
                return
        if not valid:
            self.out("No valid participants loaded.")
            return
        self.users = valid
        self.lottery_manager = None
        self.out(f"Loaded {len(valid)} valid participants.")

    def load_default(self):
        self._accept_list(*self.list_manager.load_default_list())

    def load_custom(self):
        path = self.ask("Enter txt file path: ")
        self._accept_list(*self.list_manager.load_custom_list(path))

    def show_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            self.out("No list loaded.")
            return
        self.out("Department groups:")
        for g, c in info.items():
            self.out(f"  {g}: {c} people")

    # ---------------------------------------------------------- prize handling
    def load_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            self.out(f"  [!] {e}")
        self.out("Preset prizes loaded.")

    def add_prize(self):
        name = self.ask("Prize name: ")
        qty = self.ask("Quantity: ")
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            self.out("Available groups: " + ", ".join(f"{i}.{g}" for i, g in enumerate(groups, 1)))
        raw = self.ask("Avoidance groups (numbers or names, comma separated, blank for none): ")
        avoid = []
        for item in raw.replace("\uff0c", ",").split(","):
            item = item.strip()
            if not item:
                continue
            if item.isdigit() and 1 <= int(item) <= len(groups):
                avoid.append(groups[int(item) - 1])
            else:
                avoid.append(item)
        drawer = self.ask("Associated lottery person (blank for none): ") or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            self.out("Prize added.")
        except ValueError as e:
            self.out(f"  [!] {e}")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            self.out(f"  {i}. {p}")

    def reorder(self):
        self.show_prizes()
        raw = self.ask("Enter new order as serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            self.out("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.out("Prizes reordered.")
            self.show_prizes()
        else:
            self.out("Invalid order.")

    # ----------------------------------------------------------------- lottery
    def start_lottery(self):
        if not self.users:
            self.out("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("Please configure prizes first.")
            return
        if self.lottery_manager is None:
            self.lottery_manager = LotteryManager(self.users, self.result_manager)
        self.status = "running"
        for i, p in enumerate(prizes, 1):
            self.lottery_manager.run_round(p, i, self.inp, self.out)
        self.status = "finished"
        self.out("All rounds completed.")
        self.show_summary()

    def show_summary(self):
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        self.out("====== Complete Winner List ======")
        any_w = False
        for name, ws in data.items():
            if not ws:
                continue
            any_w = True
            self.out(f"[{name}]")
            for w in ws:
                self.out(f"  {w.user.name} ({w.user.department}) - {name} - "
                         f"{w.time.strftime('%Y-%m-%d %H:%M:%S')}")
        if not any_w:
            self.out("No winners yet.")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self.out(msg)

    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.show_groups,
            "4": self.load_prizes, "5": self.add_prize, "6": self.show_prizes,
            "7": self.reorder, "8": self.start_lottery, "9": self.show_summary,
            "10": self.export,
        }
        while True:
            self.out("\n===== Team Building Smart Lottery =====")
            self.out("1. Load default list      2. Upload custom list   3. View groups")
            self.out("4. Load preset prizes     5. Add custom prize     6. View prizes")
            self.out("7. Reorder prizes         8. Start lottery        9. Show results")
            self.out("10. Export results        0. Exit")
            c = self.ask("Choose: ")
            if c == "0":
                self.out("Goodbye.")
                break
            fn = actions.get(c)
            if fn:
                fn()
            else:
                self.out("Invalid choice.")


def main():
    LotterySystem().run()


if __name__ == "__main__":
    main()

"""Grid based path planning.

The map is an 800x800 coordinate system split into 100x100 cells.  Lines
where ``x`` or ``y`` is a multiple of 100 are streets.  Paths produced by
:class:`PathPlanner` are made of axis aligned segments (straight lines and
right angle turns) that prefer to run along streets.
"""

MAP_SIZE = 800
CELL_SIZE = 100


def _clamp(value, low=0, high=MAP_SIZE):
    """Clamp ``value`` into ``[low, high]``."""
    return max(low, min(high, value))


def nearest_street(value):
    """Return the street coordinate (multiple of 100) closest to ``value``."""
    return _clamp(((int(value) + CELL_SIZE // 2) // CELL_SIZE) * CELL_SIZE)


class PathPlanner:
    """Path finding logic for riders."""

    @staticmethod
    def is_on_street(x, y):
        """Return True if the point lies on a street line."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def is_special_case(start, end):
        """Return True if the rider is directly above/below (or level with)
        the target while not standing on a street that connects them."""
        sx, sy = int(start[0]), int(start[1])
        ex, ey = int(end[0]), int(end[1])
        if (sx, sy) == (ex, ey):
            return False
        if sx == ex and sx % CELL_SIZE != 0:
            return True
        if sy == ey and sy % CELL_SIZE != 0:
            return True
        return False

    @staticmethod
    def _cleanup(start, points):
        """Remove consecutive duplicates and points equal to the start."""
        result = []
        prev = (int(start[0]), int(start[1]))
        for p in points:
            p = (int(p[0]), int(p[1]))
            if p != prev:
                result.append(p)
                prev = p
        return result

    @staticmethod
    def plan_path(start, end):
        """Plan a street-preferring path from ``start`` to ``end``.

        Returns a list of waypoints (tuples), excluding the start point and
        ending with ``end``.  Consecutive waypoints are always axis aligned.
        An empty list is returned when start equals end.
        """
        sx, sy = int(start[0]), int(start[1])
        ex, ey = int(end[0]), int(end[1])
        if (sx, sy) == (ex, ey):
            return []

        # Special case: directly above / below the target.
        if sx == ex:
            if sx % CELL_SIZE == 0:
                return [(ex, ey)]
            vx = nearest_street(sx)
            return PathPlanner._cleanup(start, [(vx, sy), (vx, ey), (ex, ey)])
        # Special case: directly left / right of the target.
        if sy == ey:
            if sy % CELL_SIZE == 0:
                return [(ex, ey)]
            hy = nearest_street(sy)
            return PathPlanner._cleanup(start, [(sx, hy), (ex, hy), (ex, ey)])

        # Standard right-angle paths along streets.
        if sy % CELL_SIZE == 0 and ex % CELL_SIZE == 0:
            return PathPlanner._cleanup(start, [(ex, sy), (ex, ey)])
        if sx % CELL_SIZE == 0 and ey % CELL_SIZE == 0:
            return PathPlanner._cleanup(start, [(sx, ey), (ex, ey)])

        # General case: reach the nearest horizontal street, ride along it
        # to the vertical street closest to the target, then approach.
        hy = nearest_street(sy)
        vx = nearest_street(ex)
        points = [(sx, hy), (vx, hy), (vx, ey), (ex, ey)]
        return PathPlanner._cleanup(start, points)

    @staticmethod
    def plan_multi(start, stops):
        """Plan a continuous path visiting every point in ``stops``."""
        path = []
        current = (int(start[0]), int(start[1]))
        for stop in stops:
            segment = PathPlanner.plan_path(current, stop)
            path.extend(segment)
            current = (int(stop[0]), int(stop[1]))
        return path

    @staticmethod
    def path_length(start, path):
        """Return the Manhattan length of a path starting at ``start``."""
        total = 0
        cx, cy = int(start[0]), int(start[1])
        for x, y in path:
            total += abs(x - cx) + abs(y - cy)
            cx, cy = x, y
        return total

"""Order data model."""

import itertools

import globalvar

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def _pos_tuple(pos):
    """Return ``pos`` as an ``(int, int)`` tuple."""
    return int(pos[0]), int(pos[1])


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique auto-incremented order id.
        res_pos: restaurant position.
        cus_pos: customer position.
        status: one of pending/assigned/picking_up/delivering/completed.
        rider_id: id of the assigned rider (or None).
    """

    _id_counter = itertools.count(1)

    def __init__(self, res_pos, cus_pos):
        if len(res_pos) != 2 or len(cus_pos) != 2:
            raise ValueError("Positions must contain exactly two coordinates")
        self.id = next(Order._id_counter)
        self.res_pos = res_pos
        self.cus_pos = cus_pos
        self.status = STATUS_PENDING
        self.rider_id = None
        self.created_time = globalvar.get_value("current_time", 0) or 0
        self.assigned_time = None
        self.accepted_time = None
        self.pickup_time = None
        self.completed_time = None
        rx, ry = _pos_tuple(res_pos)
        cx, cy = _pos_tuple(cus_pos)
        self.distance = abs(rx - cx) + abs(ry - cy)

    def assign_to(self, rider_id):
        """Record the assignment of this order to ``rider_id``."""
        self.rider_id = rider_id
        if self.status == STATUS_PENDING:
            self.status = STATUS_ASSIGNED
        self.assigned_time = globalvar.get_value("current_time", 0) or 0

    @property
    def is_completed(self):
        """True once the order has been delivered."""
        return self.status == STATUS_COMPLETED

    @property
    def time_taken(self):
        """Ticks from creation to completion, or None if not completed."""
        if self.completed_time is None:
            return None
        return self.completed_time - self.created_time

    @property
    def efficiency(self):
        """Delivery efficiency: restaurant->customer distance per tick."""
        taken = self.time_taken
        if taken is None:
            return None
        if taken <= 0:
            return float(self.distance)
        return round(self.distance / taken, 2)

    def to_dict(self):
        """Serialise the order to a JSON compatible dict."""
        return {
            "id": self.id,
            "res_pos": list(_pos_tuple(self.res_pos)),
            "cus_pos": list(_pos_tuple(self.cus_pos)),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_time": self.created_time,
            "assigned_time": self.assigned_time,
            "accepted_time": self.accepted_time,
            "pickup_time": self.pickup_time,
            "completed_time": self.completed_time,
            "distance": self.distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from :meth:`to_dict` output."""
        order = cls(tuple(data["res_pos"]), tuple(data["cus_pos"]))
        order.id = int(data["id"])
        for key in ("status", "rider_id", "created_time", "assigned_time",
                    "accepted_time", "pickup_time", "completed_time"):
            if key in data:
                setattr(order, key, data[key])
        cls.bump_counter(order.id)
        return order

    @classmethod
    def bump_counter(cls, used_id):
        """Make sure future ids are greater than ``used_id``."""
        current = next(cls._id_counter)
        cls._id_counter = itertools.count(max(current, used_id + 1))

    def __repr__(self):
        return ("Order(id={}, res={}, cus={}, status={}, rider={})"
                .format(self.id, _pos_tuple(self.res_pos),
                        _pos_tuple(self.cus_pos), self.status, self.rider_id))

"""Rider model: movement, state machine and order handling."""

import logging

import globalvar
from path_planner import PathPlanner, MAP_SIZE

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_SPEED = 20
DEFAULT_START = (400, 400)


def _pos(pos):
    """Return a position as an ``(int, int)`` tuple."""
    return int(pos[0]), int(pos[1])


def _now():
    """Current simulated time."""
    return globalvar.get_value("current_time", 0) or 0


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to :attr:`orders`.  Calling
    :meth:`move` advances the rider by one tick, performing state
    transitions and logging events.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        start = globalvar.get_value("rider_start", DEFAULT_START) or DEFAULT_START
        self.rider_id = rider_id
        self.x = int(start[0]) if x is None else int(x)
        self.y = int(start[1]) if y is None else int(y)
        if speed is None:
            speed = globalvar.get_value("rider_speed", DEFAULT_SPEED) or DEFAULT_SPEED
        self.speed = int(speed)
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.path = []
        self.total_distance = 0
        self.log = []
        self._record("spawn", None)

    # ------------------------------------------------------------------ #
    # Helpers
    # ------------------------------------------------------------------ #
    def _record(self, event, order_id):
        """Append an event entry to the rider log."""
        entry = {
            "event": event,
            "order_id": order_id,
            "time": _now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)

    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, str(self.state))

    @property
    def position(self):
        """Current position tuple."""
        return self.x, self.y

    def active_orders(self):
        """Orders that are not yet completed."""
        return [o for o in self.orders
                if getattr(o, "status", None) != "completed"]

    def active_order_count(self):
        """Number of not yet completed orders."""
        return len(self.active_orders())

    def is_idle(self):
        """True if the rider has nothing to do."""
        return self.state == STATE_IDLE and not self.active_orders()

    def _next_pending_order(self):
        for order in self.orders:
            if order is self.current_order:
                continue
            if getattr(order, "status", None) != "completed":
                return order
        return None

    def _current_target(self):
        if self.current_order is None:
            return None
        if self.state == STATE_DELIVERING:
            return _pos(self.current_order.cus_pos)
        return _pos(self.current_order.res_pos)

    def _plan(self, target):
        self.special_case = PathPlanner.is_special_case(self.position, target)
        self.path = PathPlanner.plan_path(self.position, target)
        if not self.path and self.position != target:
            self.path = [target]

    def _accept(self, order):
        self.current_order = order
        if getattr(order, "rider_id", None) is None:
            try:
                order.rider_id = self.rider_id
            except AttributeError:
                pass
        order.status = "picking_up"
        if hasattr(order, "accepted_time"):
            order.accepted_time = _now()
        self.state = STATE_PICKING_UP
        self._record("accepted_order", order.id)
        self._plan(_pos(order.res_pos))

    def _advance(self, budget):
        """Move along the planned path using up to ``budget`` units."""
        while budget > 0 and self.path:
            wx, wy = self.path[0]
            dx = wx - self.x
            dy = wy - self.y
            dist = abs(dx) + abs(dy)
            if dist <= budget:
                self.x, self.y = wx, wy
                budget -= dist
                self.total_distance += dist
                self.path.pop(0)
                continue
            # Partial move, x axis first then y axis.
            step_x = min(abs(dx), budget)
            self.x += step_x if dx > 0 else -step_x
            budget -= step_x
            self.total_distance += step_x
            if budget > 0:
                step_y = min(abs(dy), budget)
                self.y += step_y if dy > 0 else -step_y
                budget -= step_y
                self.total_distance += step_y
        self.x = max(0, min(MAP_SIZE, self.x))
        self.y = max(0, min(MAP_SIZE, self.y))
        return budget

    def _handle_arrival(self):
        """Handle reaching the current target.

        Returns True if the tick should end (an order was delivered).
        """
        order = self.current_order
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.status = "delivering"
            if hasattr(order, "pickup_time"):
                order.pickup_time = _now()
            self._record("arrived_at_merchant", order.id)
            self.path = []
            return False
        # Delivering -> delivered.
        order.status = "completed"
        if hasattr(order, "completed_time"):
            order.completed_time = _now()
        self._record("delivered_to_customer", order.id)
        self.current_order = None
        self.path = []
        self.special_case = False
        nxt = self._next_pending_order()
        if nxt is not None:
            self._accept(nxt)
        else:
            self.state = STATE_IDLE
        return True

    # ------------------------------------------------------------------ #
    # Public API
    # ------------------------------------------------------------------ #
    def move(self):
        """Advance the rider by one tick."""
        if self.current_order is not None and (
                self.current_order not in self.orders
                or self.current_order.status == "completed"):
            self.current_order = None
            self.path = []
        if self.current_order is None:
            nxt = self._next_pending_order()
            if nxt is None:
                self.state = STATE_IDLE
                self.path = []
                return
            self._accept(nxt)

        budget = self.speed
        guard = 0
        while self.current_order is not None and guard < 32:
            guard += 1
            target = self._current_target()
            if self.position != target and not self.path:
                self._plan(target)
            budget = self._advance(budget)
            if self.position == target:
                if self._handle_arrival():
                    break
                continue
            if budget <= 0:
                break

    def planned_route(self):
        """Full list of waypoints for the remaining work, starting with the
        current position (current target then customer, then queued orders).
        """
        points = [self.position]
        if self.current_order is None and not self.active_orders():
            return points
        stops = []
        if self.current_order is not None:
            if self.state == STATE_PICKING_UP:
                stops.append(_pos(self.current_order.res_pos))
            stops.append(_pos(self.current_order.cus_pos))
        for order in self.active_orders():
            if order is self.current_order:
                continue
            stops.append(_pos(order.res_pos))
            stops.append(_pos(order.cus_pos))
        points.extend(PathPlanner.plan_multi(self.position, stops))
        return points

    def estimated_ticks(self):
        """Estimated ticks to finish all remaining work."""
        route = self.planned_route()
        length = PathPlanner.path_length(route[0], route[1:])
        if self.speed <= 0:
            return None
        return -(-length // self.speed)

    def to_dict(self):
        """Serialise to a JSON compatible dict."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "current_order": self.current_order.id if self.current_order else None,
            "special_case": self.special_case,
            "total_distance": self.total_distance,
            "log": list(self.log),
        }

    def __repr__(self):
        return ("Rider(id={}, pos=({}, {}), state={}, orders={})"
                .format(self.rider_id, self.x, self.y, self.state_name,
                        len(self.orders)))

"""System initialisation and resource management."""

import logging

import globalvar
from rider import Rider

logger = logging.getLogger(__name__)

DEFAULT_FUNDS = 1000
RECRUIT_COST = 500
RECRUIT_THRESHOLD = 300


def init_system():
    """Initialise all global system parameters."""
    globalvar._init()
    globalvar.set_value("funds", DEFAULT_FUNDS)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("current_time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("recruit_cost", RECRUIT_COST)
    globalvar.set_value("recruit_threshold", RECRUIT_THRESHOLD)
    logger.info("System initialised")


def _riders():
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    return riders


def _orders():
    orders = globalvar.get_value("orders")
    if orders is None:
        orders = []
        globalvar.set_value("orders", orders)
    return orders


def recruit_rider():
    """Recruit a new rider if funds allow. Returns the rider or None."""
    funds = globalvar.get_value("funds", DEFAULT_FUNDS)
    threshold = globalvar.get_value("recruit_threshold", RECRUIT_THRESHOLD)
    cost = globalvar.get_value("recruit_cost", RECRUIT_COST)
    if funds is None or funds < threshold:
        return None
    riders = _riders()
    counter = globalvar.get_value("rider_id_counter", 0) or 0
    existing = [r.rider_id for r in riders if isinstance(r.rider_id, int)]
    if existing:
        counter = max(counter, max(existing))
    counter += 1
    rider = Rider(counter)
    riders.append(rider)
    globalvar.set_value("rider_id_counter", counter)
    globalvar.set_value("funds", funds - cost)
    logger.info("Recruited rider %s (funds now %s)", counter, funds - cost)
    return rider


def pending_order_count():
    """Number of orders not yet assigned to a rider."""
    return sum(1 for o in _orders() if o.status == "pending")


def idle_rider_count():
    """Number of idle riders."""
    return sum(1 for r in _riders() if r.is_idle())


def auto_recruit():
    """Recruit a rider when funds >= threshold and pending orders exceed
    idle riders. Returns the new rider or None."""
    funds = globalvar.get_value("funds", DEFAULT_FUNDS)
    threshold = globalvar.get_value("recruit_threshold", RECRUIT_THRESHOLD)
    if funds is not None and funds >= threshold and \
            pending_order_count() > idle_rider_count():
        return recruit_rider()
    return None


def tick(count=1):
    """Advance the simulation by ``count`` ticks."""
    for _ in range(max(0, int(count))):
        globalvar.set_value("current_time",
                            (globalvar.get_value("current_time", 0) or 0) + 1)
        for rider in list(_riders()):
            try:
                rider.move()
            except Exception:  # pragma: no cover - recovery path
                logger.exception("Rider %s failed to move; resetting",
                                 getattr(rider, "rider_id", "?"))
                rider.path = []
                rider.current_order = None


def check_consistency():
    """Run state consistency checks; returns a list of problem strings."""
    problems = []
    ids = [r.rider_id for r in _riders()]
    if len(ids) != len(set(ids)):
        problems.append("Duplicate rider ids")
    for rider in _riders():
        if not (0 <= rider.x <= 800 and 0 <= rider.y <= 800):
            problems.append("Rider {} out of map".format(rider.rider_id))
        if rider.state not in (0, 1, 2):
            problems.append("Rider {} invalid state".format(rider.rider_id))
    return problems


class SystemManager:
    """Object oriented facade over the global system state."""

    def __init__(self, initialise=False):
        if initialise or globalvar.get_value("riders") is None:
            init_system()

    @property
    def funds(self):
        return globalvar.get_value("funds", DEFAULT_FUNDS)

    @property
    def riders(self):
        return _riders()

    @property
    def orders(self):
        return _orders()

    @property
    def current_time(self):
        return globalvar.get_value("current_time", 0) or 0

    @property
    def running(self):
        return bool(globalvar.get_value("running", True))

    def stop(self):
        globalvar.set_value("running", False)

    def recruit(self):
        return recruit_rider()

    def auto_recruit(self):
        return auto_recruit()

    def tick(self, count=1):
        tick(count)

    def get_rider(self, rider_id):
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def remove_rider(self, rider_id):
        rider = self.get_rider(rider_id)
        if rider is not None and not rider.active_orders():
            self.riders.remove(rider)
            return True
        return False

"""Order creation, coordinate conversion and assignment."""

import logging
import math

import globalvar
import system
from order_model import Order

logger = logging.getLogger(__name__)

MAP_SIZE = 800
CELL_SIZE = 100


def convert_coordinate(value):
    """Convert an input coordinate to the top-left corner of its grid cell."""
    value = max(0, min(MAP_SIZE, int(value)))
    return (value // CELL_SIZE) * CELL_SIZE


def grid_center(value):
    """Return the centre of the grid cell containing ``value``."""
    value = max(0, min(MAP_SIZE - 1, int(value)))
    return (value // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2


def convert_order_input(coords):
    """Convert ``[rx, ry, cx, cy]`` into restaurant/customer grid points."""
    if len(coords) != 4:
        raise ValueError("Exactly four coordinates are required")
    rx, ry, cx, cy = (convert_coordinate(c) for c in coords)
    return (rx, ry), (cx, cy)


def _distance(rider, pos):
    return math.sqrt((rider.x - int(pos[0])) ** 2 + (rider.y - int(pos[1])) ** 2)


def find_best_rider_for_order(order):
    """Return the best rider for ``order``.

    Riders with the fewest current (uncompleted) orders are preferred; among
    those the closest rider to the restaurant (Euclidean distance) wins, and
    ties are broken by rider id.  If no rider exists one is recruited when
    funds permit; otherwise None is returned.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return system.recruit_rider()
    return min(riders, key=lambda r: (r.active_order_count(),
                                      _distance(r, order.res_pos),
                                      r.rider_id))


class OrderManager:
    """Handles order creation, assignment and queries."""

    @staticmethod
    def all_orders():
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    @staticmethod
    def create_order(coords):
        """Create and assign an order.

        Returns ``(order, rider, new_rider_added)``.
        """
        res_pos, cus_pos = convert_order_input(coords)
        order = Order(res_pos, cus_pos)
        OrderManager.all_orders().append(order)
        before = len(globalvar.get_value("riders") or [])
        system.auto_recruit()
        rider = find_best_rider_for_order(order)
        after = len(globalvar.get_value("riders") or [])
        if rider is not None:
            OrderManager.assign(order, rider)
        else:
            logger.warning("No rider available for order %s", order.id)
        return order, rider, after > before

    def assign(order, rider):
        """Assign ``order`` to ``rider``."""

        rider.orders.append(order)

                if rider is not None:
                    OrderManager.assign(order, rider)

        for order in OrderManager.all_orders():

"""Status display and ASCII map visualisation."""

from path_planner import MAP_SIZE
from rider import STATE_NAMES

MAP_CELLS = 16
CHAR_CELL = MAP_SIZE // MAP_CELLS  # 50 units per character cell


def _cell(value):
    return max(0, min(MAP_CELLS - 1, int(value) // CHAR_CELL))


class DisplayManager:
    """Builds textual views of the system state."""

    @staticmethod
    def render_map(riders=None, orders=None, route=None):
        """Render a 16x16 ASCII map.

        '#' street, '.' block, '*' route, 'S' restaurant, 'C' customer,
        'R' rider.
        """
        grid = [["#" if (r % 2 == 0 or c % 2 == 0) else "."
                 for c in range(MAP_CELLS)] for r in range(MAP_CELLS)]
        if route and len(route) > 1:
            for (x1, y1), (x2, y2) in zip(route, route[1:]):
                c1, r1, c2, r2 = _cell(x1), _cell(y1), _cell(x2), _cell(y2)
                for c in range(min(c1, c2), max(c1, c2) + 1):
                    grid[r1][c] = "*"
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    grid[r][c2] = "*"
        for order in orders or []:
            if order.status == "completed":
                continue
            grid[_cell(order.res_pos[1])][_cell(order.res_pos[0])] = "S"
            grid[_cell(order.cus_pos[1])][_cell(order.cus_pos[0])] = "C"
        for rider in riders or []:
            grid[_cell(rider.y)][_cell(rider.x)] = "R"
        header = "   " + "".join("{:X}".format(i) for i in range(MAP_CELLS))
        lines = [header]
        for idx, row in enumerate(grid):
            lines.append("{:2X} ".format(idx) + "".join(row))
        lines.append("Legend: R=rider S=restaurant C=customer "
                     "#=street .=block *=route (1 char = 50 units)")
        return "\n".join(lines)

    @staticmethod
    def format_route(rider, orders=None):
        """Route map plus speed and estimated time for a rider."""
        route = rider.planned_route()
        lines = ["Rider {} route: {}".format(
            rider.rider_id, " -> ".join("({},{})".format(x, y) for x, y in route))]
        lines.append("Speed: {} units/tick, estimated time: {} ticks".format(
            rider.speed, rider.estimated_ticks()))
        active = [o for o in rider.orders if o.status != "completed"]
        lines.append(DisplayManager.render_map([rider], orders or active, route))
        return "\n".join(lines)

    @staticmethod
    def format_status(funds, riders, current_time=0):
        """System funds and rider overview."""
        lines = ["Time: {}  Funds: {}  Riders: {}".format(
            current_time, funds, len(riders))]
        for r in riders:
            order_ids = [o.id for o in r.orders]
            lines.append(
                "  Rider {:>3} at ({:>3},{:>3}) state={:<11} active={} "
                "orders={}".format(r.rider_id, r.x, r.y,
                                   STATE_NAMES.get(r.state, r.state),
                                   r.active_order_count(), order_ids))
        if not riders:
            lines.append("  (no riders)")
        return "\n".join(lines)

    @staticmethod
    def format_orders(orders):
        """Detailed order table."""
        if not orders:
            return "No orders."
        lines = ["{:>4} {:>6} {:>11} {:>11} {:>6} {:>6} {:>6} {:<11}".format(
            "ID", "Start", "Restaurant", "Customer", "Dist", "Time",
            "Rider", "Status")]
        for o in orders:
            taken = o.time_taken
            lines.append("{:>4} {:>6} {:>11} {:>11} {:>6} {:>6} {:>6} {:<11}".format(
                o.id, o.created_time,
                "({},{})".format(*o.res_pos), "({},{})".format(*o.cus_pos),
                o.distance, "-" if taken is None else taken,
                "-" if o.rider_id is None else o.rider_id, o.status))
        return "\n".join(lines)

    @staticmethod
    def format_rider_logs(riders):
        """Every recorded status change for every rider."""
        if not riders:
            return "No riders."
        lines = []
        for r in riders:
            lines.append("Rider {}:".format(r.rider_id))
            for e in r.log:
                lines.append("  t={:<5} ({:>3},{:>3}) {:<22} order={}".format(
                    e.get("time", "-"), e.get("x", "-"), e.get("y", "-"),
                    e["event"], "-" if e["order_id"] is None else e["order_id"]))
        return "\n".join(lines)

"""Command parsing and input validation."""

import re

try:  # Command history / line editing where available.
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

COMMANDS = {
    "start": "start", "s": "start", "new": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "save": "save", "load": "load",
    "history": "history", "hist": "history",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "exit": "quit", "q": "quit",
}

MIN_COORD = 0
MAX_COORD = 1000


class InputManager:
    """Parses commands and validates user input."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        """Return ``(command, args)``; command is None for unknown input.

        ``!!`` repeats the last command, ``!n`` repeats history entry n.
        """
        line = (line or "").strip()
        if not line:
            return "", []
        if line.startswith("!"):
            ref = line[1:]
            if ref == "!" and self.history:
                line = self.history[-1]
            elif ref.isdigit() and 1 <= int(ref) <= len(self.history):
                line = self.history[int(ref) - 1]
            else:
                return None, [line]
        parts = line.split()
        cmd = COMMANDS.get(parts[0].lower())
        if cmd is not None:
            self.history.append(line)
            if len(self.history) > 1000:
                self.history = self.history[-1000:]
        return cmd, parts[1:]

    @staticmethod
    def parse_coordinates(text):
        """Parse four integer coordinates; raises ValueError when invalid."""
        if isinstance(text, (list, tuple)):
            tokens = [str(t) for t in text]
        else:
            cleaned = re.sub(r"[\[\]()]", " ", str(text))
            tokens = [t for t in re.split(r"[,\s]+", cleaned) if t]
        if len(tokens) != 4:
            raise ValueError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for tok in tokens:
            try:
                val = int(tok)
            except ValueError:
                raise ValueError("'{}' is not an integer".format(tok))
            if not MIN_COORD <= val <= MAX_COORD:
                raise ValueError("Coordinate {} out of range {}-{}".format(
                    val, MIN_COORD, MAX_COORD))
            values.append(val)
        return values

    @staticmethod
    def help_text():
        return ("Commands:\n"
                "  start|s [rx ry cx cy]  create an order\n"
                "  status|st              system snapshot\n"
                "  orders|o               all orders\n"
                "  riders|r               rider event logs\n"
                "  map|m                  ASCII map\n"
                "  tick|t [n]             advance n ticks\n"
                "  save|load [file]       persist / restore state (JSON)\n"
                "  history|hist, !!, !n   command history\n"
                "  quit|exit|q            exit")

"""JSON persistence of configuration and system state."""

import json
import os

import globalvar
from order_model import Order
from rider import Rider

CONFIG_KEYS = ("funds", "rider_speed", "recruit_cost", "recruit_threshold",
               "rider_start")


def load_config(path):
    """Apply system parameters from a JSON config file if it exists."""
    if not os.path.exists(path):
        return False
    with open(path, "r", encoding="utf-8") as fh:
        data = json.load(fh)
    for key in CONFIG_KEYS:
        if key in data:
            value = data[key]
            if key == "rider_start":
                value = tuple(int(v) for v in value)
            globalvar.set_value(key, value)
    return True


def save_state(path):
    """Write funds, riders and order history to a JSON file."""
    data = {
        "funds": globalvar.get_value("funds"),
        "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
        "current_time": globalvar.get_value("current_time", 0),
        "riders": [r.to_dict() for r in globalvar.get_value("riders") or []],
        "orders": [o.to_dict() for o in globalvar.get_value("orders") or []],
    }
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, indent=2)


def load_state(path):
    """Restore state written by :func:`save_state`."""
    with open(path, "r", encoding="utf-8") as fh:
        data = json.load(fh)
    orders = [Order.from_dict(o) for o in data.get("orders", [])]
    by_id = {o.id: o for o in orders}
    riders = []
    for rd in data.get("riders", []):
        rider = Rider(rd["rider_id"], rd["x"], rd["y"], rd.get("speed"))
        rider.state = rd.get("state", 0)
        rider.special_case = rd.get("special_case", False)
        rider.total_distance = rd.get("total_distance", 0)
        rider.orders = [by_id[i] for i in rd.get("orders", []) if i in by_id]
        cur = rd.get("current_order")
        rider.current_order = by_id.get(cur) if cur is not None else None
        rider.log = rd.get("log", rider.log)
        riders.append(rider)
    globalvar.set_value("funds", data.get("funds", 1000))
    globalvar.set_value("rider_id_counter", data.get("rider_id_counter", 0))
    globalvar.set_value("current_time", data.get("current_time", 0))
    globalvar.set_value("riders", riders)
    globalvar.set_value("orders", orders)

"""Command line entry point for the food delivery dispatch simulator."""

import logging
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import globalvar  # noqa: E402
import persistence  # noqa: E402
import system  # noqa: E402
from display import DisplayManager  # noqa: E402
from input_manager import InputManager  # noqa: E402
from order import OrderManager  # noqa: E402

CONFIG_FILE = "config.json"
STATE_FILE = "state.json"


def handle_start(args, inputs):
    if args:
        raw = " ".join(args)
    else:
        raw = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
    try:
        coords = inputs.parse_coordinates(raw)
    except ValueError as exc:
        print("Invalid input: {}".format(exc))
        return
    order, rider, new_rider = OrderManager.create_order(coords)
    print("Order {} created.".format(order.id))
    print("Restaurant grid: ({}, {})  Customer grid: ({}, {})".format(
        order.res_pos[0], order.res_pos[1], order.cus_pos[0], order.cus_pos[1]))
    print("New rider added: {}".format("yes" if new_rider else "no"))
    if rider is None:
        print("No rider available (insufficient funds); order is pending.")
        return
    print("Assigned to rider {}".format(rider.rider_id))
    print(DisplayManager.format_route(rider))


def execute(cmd, args, inputs, manager):
    """Execute a parsed command. Returns False when the loop should stop."""
    if cmd == "start":
        handle_start(args, inputs)
    elif cmd == "status":
        print(DisplayManager.format_status(manager.funds, manager.riders,
                                           manager.current_time))
        problems = system.check_consistency()
        for p in problems:
            print("WARNING: " + p)
    elif cmd == "orders":
        print(DisplayManager.format_orders(manager.orders))
    elif cmd == "riders":
        print(DisplayManager.format_rider_logs(manager.riders))
    elif cmd == "map":
        print(DisplayManager.render_map(manager.riders, manager.orders))
    elif cmd == "tick":
        n = 1
        if args:
            try:
                n = max(1, int(args[0]))
            except ValueError:
                print("tick expects an integer")
                return True
        manager.tick(n - 1)  # the loop itself advances one tick
        print("Advancing {} tick(s).".format(n))
    elif cmd == "save":
        path = args[0] if args else STATE_FILE
        persistence.save_state(path)
        print("State saved to {}".format(path))
    elif cmd == "load":
        path = args[0] if args else STATE_FILE
        try:
            persistence.load_state(path)
            print("State loaded from {}".format(path))
        except (OSError, ValueError, KeyError) as exc:
            print("Could not load state: {}".format(exc))
    elif cmd == "history":
        for i, line in enumerate(inputs.history, 1):
            print("{:>4}  {}".format(i, line))
    elif cmd == "help":
        print(inputs.help_text())
    elif cmd == "quit":
        manager.stop()
        return False
    return True


def main():
    logging.basicConfig(level=os.environ.get("DISPATCH_LOG", "WARNING"),
                        format="%(levelname)s %(name)s: %(message)s")
    system.init_system()
    try:
        persistence.load_config(CONFIG_FILE)
    except (OSError, ValueError) as exc:
        print("Config ignored: {}".format(exc))
    manager = system.SystemManager()
    inputs = InputManager()
    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
    while globalvar.get_value("running", True):
        try:
            line = input("[t={}] > ".format(manager.current_time))
        except (EOFError, KeyboardInterrupt):
            print()
            break
        cmd, args = inputs.parse_command(line)
        if cmd is None:
            print("Unknown command. Type 'help' for the command list.")
        elif cmd:
            try:
                if not execute(cmd, args, inputs, manager):
                    break
            except (EOFError, KeyboardInterrupt):
                print()
                break
            except Exception as exc:  # graceful recovery
                logging.exception("Command failed")
                print("Error: {}. System state preserved.".format(exc))
        manager.tick(1)
        OrderManager.assign_pending()
    print("Goodbye.")


if __name__ == "__main__":
    main()

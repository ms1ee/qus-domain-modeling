"""Global variable storage for the dispatch system.

Simple key/value store shared between modules.  Typical keys:

* ``funds``             -- int, system funds (may become negative)
* ``riders``            -- list[Rider]
* ``rider_id_counter``  -- int, last issued rider id
* ``running``           -- bool, main loop control
* ``time``              -- int, simulated tick counter
* ``orders``            -- list[Order], every order ever created
* ``order_counter``     -- int, last issued order id
* ``config``            -- dict, system parameters
"""

_global_dict = None


def _init():
    """Initialise (or reset) the global dictionary."""
    global _global_dict
    _global_dict = {}


def _ensure():
    """Lazily initialise the store if ``_init`` was never called."""
    if _global_dict is None:
        _init()


def set_value(key, value):
    """Store ``value`` under ``key``."""
    _ensure()
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` (or ``default``)."""
    _ensure()
    return _global_dict.get(key, default)


def has_key(key):
    """Return True if ``key`` is present in the store."""
    _ensure()
    return key in _global_dict

"""System parameter configuration (JSON based)."""

import json
import logging
import os

logger = logging.getLogger("dispatch.config")

DEFAULT_CONFIG_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                                   "dispatch_config.json")

DEFAULT_CONFIG = {
    "initial_funds": 10000,
    "rider_cost": 1000,
    "rider_speed": 50,
    "rider_start_x": 400,
    "rider_start_y": 400,
    "max_riders": 20,
    "map_size": 800,
    "cell_size": 100,
}


def load_config(path=None):
    """Load configuration from a JSON file merged over the defaults.

    If ``path`` is None the default file next to this module is used if it
    exists.  Invalid files are ignored (defaults are used instead).
    """
    cfg = dict(DEFAULT_CONFIG)
    if path is None:
        path = DEFAULT_CONFIG_FILE
    if path and os.path.isfile(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                params = data.get("system", data)
                for key in DEFAULT_CONFIG:
                    if key in params and isinstance(params[key], int):
                        cfg[key] = params[key]
        except (OSError, ValueError) as exc:
            logger.warning("Could not read config %s: %s", path, exc)
    return cfg


def get_config_value(key):
    """Return a configuration value from the active config or defaults."""
    import globalvar
    cfg = globalvar.get_value("config")
    if isinstance(cfg, dict) and key in cfg:
        return cfg[key]
    return DEFAULT_CONFIG[key]

"""Grid-based path planning.

The map is 800x800 units split into 8x8 cells of 100x100 units.  Streets
run along the grid lines (x or y multiple of 100).  Restaurants and
customers are located at cell centres (``(v // 100) * 100 + 50``).

A route always consists of axis-aligned segments that follow streets and
finally step 50 units into the target cell centre.  Riders never block each
other (streets have unlimited capacity), so path conflicts and deadlocks
cannot occur.
"""

CELL = 100
HALF = 50
MAP_SIZE = 800


class PathPlanner:
    """Path planning helper (all methods are static / class methods)."""

    CELL = CELL
    MAP_SIZE = MAP_SIZE

    @staticmethod
    def to_center(value):
        """Convert a raw coordinate (0-800) to its grid cell centre."""
        v = int(value)
        v = max(0, min(v, MAP_SIZE - 1))
        return (v // CELL) * CELL + HALF

    @classmethod
    def to_grid_center(cls, pos):
        """Convert an ``(x, y)`` pair to the centre of its grid cell."""
        return (cls.to_center(pos[0]), cls.to_center(pos[1]))

    @staticmethod
    def is_on_street(x, y):
        """True if the point lies on a street (grid line)."""
        return x % CELL == 0 or y % CELL == 0

    @classmethod
    def plan_with_info(cls, start, target):
        """Plan a street route from ``start`` to ``target``.

        Returns ``(waypoints, special_case)`` where ``waypoints`` excludes the
        start point and ends at the target, and ``special_case`` is True if
        the rider was directly above/below the target and had to detour to a
        side street.
        """
        x, y = int(round(start[0])), int(round(start[1]))
        tx, ty = int(round(target[0])), int(round(target[1]))
        special = False
        if (x, y) == (tx, ty):
            return [], False
        pts = []

        on_h = y % CELL == 0
        on_v = x % CELL == 0
        if not on_h and not on_v:
            if (x // CELL, y // CELL) == (tx // CELL, ty // CELL):
                # Same cell interior: simple L-shaped move.
                pts = [(tx, y), (tx, ty)]
                return cls._clean((x, y), pts), False
            # Step out of the cell onto the nearest horizontal street
            # in the direction of the target.
            if ty > y:
                ny = (y // CELL + 1) * CELL
            else:
                ny = (y // CELL) * CELL
            pts.append((x, ny))
            y = ny

        # Horizontal street adjacent to the target from which we enter it.
        ey = ty - HALF if y <= ty else ty + HALF

        if y == ey:
            pts.append((tx, ey))
            pts.append((tx, ty))
        else:
            if x % CELL != 0:
                if tx > x:
                    nx = (x // CELL + 1) * CELL
                elif tx < x:
                    nx = (x // CELL) * CELL
                else:
                    # Directly above/below the target: detour via side street.
                    special = True
                    nx = (x // CELL) * CELL
                pts.append((nx, y))
                x = nx
            pts.append((x, ey))
            pts.append((tx, ey))
            pts.append((tx, ty))
        return cls._clean(start, pts), special

    @classmethod
    def plan(cls, start, target):
        """Return only the waypoints of :meth:`plan_with_info`."""
        return cls.plan_with_info(start, target)[0]

    @staticmethod
    def _clean(start, pts):
        """Remove duplicate consecutive points (and points equal to start)."""
        result = []
        prev = (int(round(start[0])), int(round(start[1])))
        for p in pts:
            p = (int(p[0]), int(p[1]))
            if p != prev:
                result.append(p)
                prev = p
        return result

    @staticmethod
    def polyline_length(points):
        """Total Manhattan length of a polyline."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    @classmethod
    def path_length(cls, start, target):
        """Length of the planned route from ``start`` to ``target``."""
        pts = [tuple(start)] + cls.plan(start, target)
        return cls.polyline_length(pts)

    @staticmethod
    def sample_polyline(points, step=HALF):
        """Return points sampled every ``step`` units along the polyline."""
        if not points:
            return []
        samples = [tuple(points[0])]
        for a, b in zip(points, points[1:]):
            ax, ay = a
            bx, by = b
            length = abs(bx - ax) + abs(by - ay)
            if length == 0:
                continue
            n = max(1, length // step)
            for i in range(1, n + 1):
                t = i / n
                samples.append((ax + (bx - ax) * t, ay + (by - ay) * t))
        return samples

"""Order data model."""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger("dispatch.order")


class Order:
    """A single delivery order.

    Attributes:
        id: unique order id (auto-incrementing, starts at 1).
        raw: original ``[restaurant_x, restaurant_y, customer_x, customer_y]``.
        res_pos / cus_pos: grid-centre coordinates of restaurant / customer.
        status: "pending", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: id of the assigned rider (or None).
        create_time, assign_time, pickup_time, complete_time: ticks.
        distance: route length from restaurant to customer.
    """

    def __init__(self, res_pos, cus_pos):
        rx, ry = int(res_pos[0]), int(res_pos[1])
        cx, cy = int(cus_pos[0]), int(cus_pos[1])
        self.raw = [rx, ry, cx, cy]
        self.res_pos = PathPlanner.to_grid_center((rx, ry))
        self.cus_pos = PathPlanner.to_grid_center((cx, cy))

        counter = globalvar.get_value("order_counter", 0) or 0
        counter += 1
        globalvar.set_value("order_counter", counter)
        self.id = counter

        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("time", 0) or 0
        self.assign_time = None
        self.pickup_time = None
        self.complete_time = None
        self.distance = PathPlanner.path_length(self.res_pos, self.cus_pos)

        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(self)
        logger.debug("Created order %s %s -> %s", self.id, self.res_pos,
                     self.cus_pos)

    def assign_to(self, rider_id):
        """Mark the order as assigned to ``rider_id``."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        self.assign_time = globalvar.get_value("time", 0) or 0

    def mark_completed(self, now):
        """Record completion of the order at tick ``now``."""
        self.status = "completed"
        self.complete_time = now

    @property
    def time_taken(self):
        """Ticks from creation to completion (None if not completed)."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.create_time

    @property
    def efficiency(self):
        """Delivery distance per tick (None if not completed)."""
        taken = self.time_taken
        if taken is None:
            return None
        return round(self.distance / taken, 2) if taken > 0 else None

    def to_dict(self):
        """Serialisable representation."""
        return {
            "id": self.id,
            "raw": list(self.raw),
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "create_time": self.create_time,
            "assign_time": self.assign_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
            "time_taken": self.time_taken,
        }

    def __repr__(self):
        return "Order(id=%s, res=%s, cus=%s, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

"""Rider model: movement, state machine and event log."""

import logging

import globalvar
from config import get_config_value
from path_planner import PathPlanner

logger = logging.getLogger("dispatch.rider")

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up",
               DELIVERING: "delivering"}


class Rider:
    """A delivery rider.

    Attributes:
        rider_id: unique id.
        x, y: current coordinates.
        speed: movement units per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: assigned (not yet completed) orders.  Assign an order by
            appending it to this list.
        special_case: True while following a special-case detour route.
        log: list of event dicts (keys include "event" and "order_id").
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = int(rider_id)
        self.x = int(x if x is not None else get_config_value("rider_start_x"))
        self.y = int(y if y is not None else get_config_value("rider_start_y"))
        self.speed = int(speed if speed is not None
                         else get_config_value("rider_speed"))
        if self.speed <= 0:
            self.speed = 1
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.path = []
        self.current_order = None
        self.completed_orders = []
        self.distance_travelled = 0
        self.log = []
        self._log_event("spawn", None)

    # ------------------------------------------------------------------ #
    # helpers
    # ------------------------------------------------------------------ #
    @staticmethod
    def _now():
        return globalvar.get_value("time", 0) or 0

    def _log_event(self, event, order):
        entry = {
            "event": event,
            "order_id": getattr(order, "id", None) if order is not None
            else None,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)

    def _next_order(self):
        for o in self.orders:
            if getattr(o, "status", None) != "completed":
                return o
        return None

    def pending_orders(self):
        """Orders not yet completed."""
        return [o for o in self.orders
                if getattr(o, "status", None) != "completed"]

    def is_available(self):
        """True if idle with no pending work."""
        return self.state == IDLE and not self.pending_orders()

    @property
    def state_name(self):
        return STATE_NAMES.get(self.state, "unknown")

    def _set_path(self, target):
        self.path, self.special_case = PathPlanner.plan_with_info(
            (self.x, self.y), target)

    def _advance(self):
        """Move along the current path by ``speed``; True if path finished."""
        budget = self.speed
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            d = abs(tx - self.x) + abs(ty - self.y)
            if d <= budget:
                self.x, self.y = tx, ty
                self.path.pop(0)
                budget -= d
                self.distance_travelled += d
            else:
                if tx != self.x:
                    self.x += budget if tx > self.x else -budget
                else:
                    self.y += budget if ty > self.y else -budget
                self.distance_travelled += budget
                budget = 0
        return not self.path

    # ------------------------------------------------------------------ #
    # simulation
    # ------------------------------------------------------------------ #
    def move(self):
        """Advance the rider by one tick.

        Handles state transitions (idle -> picking up -> delivering -> idle)
        and logs the corresponding events.
        """
        if self.state == IDLE:
            order = self._next_order()
            if order is None:
                return
            self.current_order = order
            self.state = PICKING_UP
            if getattr(order, "rider_id", None) is None and hasattr(
                    order, "assign_to"):
                order.assign_to(self.rider_id)
            order.status = "picking_up"
            self._log_event("accepted_order", order)
            self._set_path(order.res_pos)

        order = self.current_order
        if order is None:
            # Inconsistent state: recover.
            self.state = IDLE
            self.path = []
            return

        if self.state == PICKING_UP:
            if self._advance():
                self.state = DELIVERING
                order.status = "delivering"
                order.pickup_time = self._now()
                self._log_event("arrived_at_merchant", order)
                self._set_path(order.cus_pos)
            return

        if self.state == DELIVERING:
            if self._advance():
                now = self._now()
                if hasattr(order, "mark_completed"):
                    order.mark_completed(now)
                else:
                    order.status = "completed"
                self.state = IDLE
                self._log_event("delivered_to_customer", order)
                if order in self.orders:
                    self.orders.remove(order)
                self.completed_orders.append(order)
                self.current_order = None
                self.special_case = False
                self.path = []

    # ------------------------------------------------------------------ #
    # estimates
    # ------------------------------------------------------------------ #
    def planned_route(self, extra_order=None):
        """Polyline of the rider's full planned route (optionally with an
        extra order appended at the end)."""
        pts = [(self.x, self.y)]
        cur = (self.x, self.y)

        def extend(target):
            nonlocal cur
            pts.extend(PathPlanner.plan(cur, target))
            cur = tuple(target)

        if self.current_order is not None and self.state != IDLE:
            pts.extend(self.path)
            if self.path:
                cur = tuple(self.path[-1])
            if self.state == PICKING_UP:
                extend(self.current_order.cus_pos)
        pending = [o for o in self.pending_orders()
                   if o is not self.current_order]
        if extra_order is not None and all(o is not extra_order
                                           for o in pending):
            pending.append(extra_order)
        for o in pending:
            extend(o.res_pos)
            extend(o.cus_pos)
        return pts

    def estimated_workload(self, extra_order=None):
        """Remaining route length including an optional extra order."""
        return PathPlanner.polyline_length(self.planned_route(extra_order))

    def to_dict(self):
        """Serialisable representation."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [getattr(o, "id", None) for o in self.orders],
            "completed_orders": [getattr(o, "id", None)
                                 for o in self.completed_orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%s,%s), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state, len(self.orders))

"""System initialisation and resource management."""

import json
import logging

import globalvar
from config import load_config, get_config_value
from rider import Rider, IDLE

logger = logging.getLogger("dispatch.system")


def init_system(config_path=None):
    """Initialise all global system parameters.

    Returns a :class:`SystemManager` bound to the global state.
    """
    cfg = load_config(config_path)
    globalvar.set_value("config", cfg)
    globalvar.set_value("funds", int(cfg["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_counter", 0)
    logger.info("System initialised with funds %s", cfg["initial_funds"])
    return SystemManager()


def _ensure_initialised():
    if globalvar.get_value("riders") is None:
        init_system()


def can_recruit():
    """True if funds and rider limit allow a new recruit."""
    _ensure_initialised()
    funds = globalvar.get_value("funds", 0)
    riders = globalvar.get_value("riders")
    return (funds >= get_config_value("rider_cost")
            and len(riders) < get_config_value("max_riders"))


def recruit_rider(force=False):
    """Recruit a new rider, deducting the cost from funds.

    Returns the new Rider, or None if not allowed (unless ``force``).
    """
    _ensure_initialised()
    if not force and not can_recruit():
        return None
    counter = (globalvar.get_value("rider_id_counter", 0) or 0) + 1
    globalvar.set_value("rider_id_counter", counter)
    globalvar.set_value("funds", globalvar.get_value("funds", 0)
                        - get_config_value("rider_cost"))
    rider = Rider(counter)
    globalvar.get_value("riders").append(rider)
    logger.info("Recruited rider %s", counter)
    return rider


class SystemManager:
    """Facade over the global system state."""

    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        _ensure_initialised()
        return globalvar.get_value("riders")

    @property
    def orders(self):
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    @property
    def time(self):
        return globalvar.get_value("time", 0) or 0

    @property
    def running(self):
        return bool(globalvar.get_value("running", False))

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    def can_recruit(self):
        return can_recruit()

    def recruit_rider(self, force=False):
        return recruit_rider(force)

    def get_rider(self, rider_id):
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    def tick(self, n=1):
        """Advance the simulation by ``n`` ticks."""
        for _ in range(max(0, int(n))):
            globalvar.set_value("time", self.time + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception as exc:  # pragma: no cover - recovery
                    logger.error("Rider %s failed to move: %s",
                                 rider.rider_id, exc)
                    self._recover_rider(rider)

    @staticmethod
    def _recover_rider(rider):
        rider.path = []
        if rider.current_order is not None and \
                rider.current_order.status != "completed":
            rider.current_order.status = "assigned"
        rider.current_order = None
        rider.state = IDLE

    def check_consistency(self):
        """Repair obvious inconsistencies; return a list of issues fixed."""
        issues = []
        if not isinstance(globalvar.get_value("funds"), int):
            try:
                globalvar.set_value("funds", int(globalvar.get_value("funds")))
            except (TypeError, ValueError):
                globalvar.set_value("funds", 0)
            issues.append("funds type repaired")
        seen = set()
        for rider in self.riders:
            if rider.rider_id in seen:
                issues.append("duplicate rider id %s" % rider.rider_id)
            seen.add(rider.rider_id)
            if rider.state not in (0, 1, 2):
                self._recover_rider(rider)
                issues.append("rider %s invalid state" % rider.rider_id)
            elif rider.state != IDLE and rider.current_order is None:
                self._recover_rider(rider)
                issues.append("rider %s busy without order" % rider.rider_id)
            rider.x = max(0, min(800, rider.x))
            rider.y = max(0, min(800, rider.y))
        return issues

    def to_dict(self):
        return {
            "system": dict(globalvar.get_value("config") or {}),
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.orders],
        }

    def save_state(self, path):
        """Persist parameters, riders and order history as JSON."""
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, indent=2)
        return path

"""Order assignment and management."""

import logging

import globalvar
import system
from order_model import Order
from path_planner import PathPlanner

logger = logging.getLogger("dispatch.order")


def find_best_rider_for_order(order):
    """Select the best rider for ``order``.

    Strategy (distance first):
    1. If idle riders exist, pick the one with the shortest street route to
       the restaurant (ties: fewer orders, lower id).
    2. Otherwise recruit a new rider if funds allow.
    3. Otherwise pick the busy rider whose total remaining route including
       this order is shortest (load balancing).
    4. With no riders at all and no funds, a rider is recruited anyway
       (funds may go negative).

    The order is NOT appended to the rider's order list here.
    """
    riders = globalvar.get_value("riders")
    if riders is None:
        system.init_system()
        riders = globalvar.get_value("riders")

    idle = [r for r in riders if r.is_available()]
    if idle:
        return min(idle, key=lambda r: (
            PathPlanner.path_length((r.x, r.y), order.res_pos),
            len(r.orders), r.rider_id))

    new_rider = system.recruit_rider()
    if new_rider is not None:
        return new_rider

    if riders:
        return min(riders, key=lambda r: (
            r.estimated_workload(order), len(r.orders), r.rider_id))

    return system.recruit_rider(force=True)

    "q": "quit", "quit": "quit", "exit": "quit",

class InputError(ValueError):
    """Raised on invalid user input."""


class InputManager:
    """Parses commands and validates coordinates."""

    MIN_COORD = 0
    MAX_COORD = 800

    def __init__(self):
        self.history = []
        try:
            import readline  # noqa: F401  (enables arrow-key history)
            self._readline = readline
        except ImportError:  # pragma: no cover - Windows
            self._readline = None

    def parse_command(self, line):
        """Return ``(command, args)``; command is None for empty input,
        "unknown" for unrecognised commands."""
        line = (line or "").strip()
        if not line:
            return None, []
        self.history.append(line)
        parts = line.split()
        cmd = SHORTCUTS.get(parts[0].lower(), "unknown")
        return cmd, parts[1:]

    def parse_coordinates(self, tokens):
        """Validate four integer coordinates in range 0-800."""
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ") \
                .replace("]", " ").split()
        else:
            joined = " ".join(str(t) for t in tokens)
            tokens = joined.replace(",", " ").replace("[", " ") \
                .replace("]", " ").split()
        if len(tokens) != 4:
            raise InputError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for tok in tokens:
            try:
                v = int(tok)
            except ValueError:
                raise InputError("'%s' is not an integer" % tok)
            if not self.MIN_COORD <= v <= self.MAX_COORD:
                raise InputError("Coordinate %d out of range (%d-%d)"
                                 % (v, self.MIN_COORD, self.MAX_COORD))
            values.append(v)
        return values

    def parse_positive_int(self, tokens, default=1, maximum=10000):
        """Parse an optional positive integer argument."""
        if not tokens:
            return default
        try:
            v = int(tokens[0])
        except ValueError:
            raise InputError("'%s' is not an integer" % tokens[0])
        if v < 1 or v > maximum:
            raise InputError("Value must be between 1 and %d" % maximum)
        return v

from path_planner import PathPlanner

SCALE = 50
N = 800 // SCALE + 1  # 17 character positions per axis

EVENT_NAMES = {
    "spawn": "spawned",
    "accepted_order": "accepted order",
    "arrived_at_merchant": "arrived at restaurant",
    "delivered_to_customer": "delivered to customer",
}


def _idx(v):
    return max(0, min(N - 1, int(round(v / SCALE))))


class DisplayManager:
    """Formats status information and maps as text."""

    LEGEND = ("Legend: '+' intersection  '-' '|' street  '.' empty cell  "
              "'R' restaurant  'C' customer  '*' route  digit/'@' rider")

    def render_map(self, riders, orders=(), route=None, highlight=None):
        """Return an ASCII map string.

        ``route`` is an optional polyline drawn with '*'.
        """
        grid = []
        for r in range(N):
            row = []
            for c in range(N):
                if r % 2 == 0 and c % 2 == 0:
                    row.append("+")
                elif r % 2 == 0:
                    row.append("-")
                elif c % 2 == 0:
                    row.append("|")
                else:
                    row.append(".")
            grid.append(row)
        if route:
            for x, y in PathPlanner.sample_polyline(route):
                grid[_idx(y)][_idx(x)] = "*"
        for o in orders:
            if o.status == "completed":
                continue
            rx, ry = o.res_pos
            cx, cy = o.cus_pos
            grid[_idx(ry)][_idx(rx)] = "R"
            grid[_idx(cy)][_idx(cx)] = "C"
        for rd in riders:
            if highlight is not None and rd.rider_id == highlight:
                sym = "@"
            else:
                sym = str(rd.rider_id % 10)
            grid[_idx(rd.y)][_idx(rd.x)] = sym
        lines = ["     " + "".join(("%-4d" % (c * SCALE)) if c % 2 == 0
                                   else "" for c in range(N))]
        for r, row in enumerate(grid):
            label = "%4d " % (r * SCALE) if r % 2 == 0 else "     "
            lines.append(label + " ".join(row))
        lines.append(self.LEGEND)
        return "\n".join(lines)

    @staticmethod
    def format_status(manager):
        lines = ["=== System status (tick %d) ===" % manager.time,
                 "Funds: %d" % manager.funds,
                 "Riders: %d" % len(manager.riders)]
        for r in manager.riders:
            ids = [o.id for o in r.orders]
            lines.append("  Rider #%d at (%d,%d) speed=%d state=%s orders=%d %s"
                         % (r.rider_id, r.x, r.y, r.speed,
                            STATE_NAMES.get(r.state, "?"), len(r.orders),
                            ids))
        return "\n".join(lines)

    @staticmethod
    def format_orders(orders):
        if not orders:
            return "No orders yet."
        lines = ["%-4s %-6s %-11s %-11s %-6s %-6s %-11s %-5s" % (
            "ID", "Start", "Restaurant", "Customer", "Dist", "Time",
            "Status", "Rider")]
        for o in orders:
            taken = o.time_taken
            lines.append("%-4d %-6d %-11s %-11s %-6d %-6s %-11s %-5s" % (
                o.id, o.create_time, "(%d,%d)" % o.res_pos,
                "(%d,%d)" % o.cus_pos, o.distance,
                "-" if taken is None else str(taken), o.status,
                "-" if o.rider_id is None else str(o.rider_id)))
        return "\n".join(lines)

    @staticmethod
    def format_rider_logs(riders):
        if not riders:
            return "No riders yet."
        lines = []
        for r in riders:
            lines.append("Rider #%d:" % r.rider_id)
            for e in r.log:
                desc = EVENT_NAMES.get(e["event"], e["event"])
                order_txt = ("" if e.get("order_id") is None
                             else " order #%s" % e["order_id"])
                lines.append("  t=%-5s %-22s%s at (%s,%s)" % (
                    e.get("time"), desc, order_txt, e.get("x"), e.get("y")))
        return "\n".join(lines)

    @staticmethod
    def format_help():
        return "\n".join([
            "Commands:",
            "  start|s [rx ry cx cy]  create an order (coordinates 0-800)",
            "  status|st              system funds and rider states",

            "  history                show command history",
            "  save [file]            save state to JSON",
            "  help|h                 this help",
            "  quit|exit|q            exit",
            "Every command advances the simulation by one tick.",
        ])

"""Command-line entry point for the food delivery dispatch simulator."""

import logging
import math
import sys

from display import DisplayManager
from input_manager import InputManager, InputError
from order import OrderManager
from system import init_system

logger = logging.getLogger("dispatch")


class Application:
    """Main interactive loop."""

    def __init__(self, manager, out=None):
        self.manager = manager
        self.input = InputManager()
        self.display = DisplayManager()
        self.orders = OrderManager()
        self.out = out or sys.stdout

    def write(self, text=""):
        self.out.write(str(text) + "\n")

    def cmd_start(self, args):
        if args:
            coords = self.input.parse_coordinates(args)
        else:
            raw = input("Enter restaurant_x restaurant_y customer_x "
                        "customer_y: ")
            coords = self.input.parse_coordinates(raw)
        result = self.orders.create_order(*coords)
        order, rider = result["order"], result["rider"]
        length = self.display_route_info(order, rider, result)
        return length

    def display_route_info(self, order, rider, result):
        route = result["route"]
        from path_planner import PathPlanner
        length = PathPlanner.polyline_length(route)
        self.write("Order #%d created." % order.id)
        self.write("  Restaurant grid position: (%d, %d)" % order.res_pos)
        self.write("  Customer grid position:   (%d, %d)" % order.cus_pos)
        self.write("  New rider added: %s" % ("yes" if result["new_rider"]
                                              else "no"))
        self.write("  Assigned rider #%d at (%d, %d), speed %d units/tick"
                   % (rider.rider_id, rider.x, rider.y, rider.speed))
        self.write("  Route length: %d units, estimated time: %d ticks"
                   % (length, math.ceil(length / rider.speed)))
        self.write(self.display.render_map(self.manager.riders,
                                           self.manager.orders, route,
                                           highlight=rider.rider_id))
        return length

    def handle(self, line):
        """Process one command line; returns False when quitting."""
        cmd, args = self.input.parse_command(line)
        if cmd is None:
            return True
        try:
            if cmd == "start":
                self.cmd_start(args)
            elif cmd == "status":
                self.write(self.display.format_status(self.manager))
            elif cmd == "orders":
                self.write(self.display.format_orders(self.manager.orders))
            elif cmd == "riders":
                self.write(self.display.format_rider_logs(
                    self.manager.riders))
            elif cmd == "map":
                self.write(self.display.render_map(self.manager.riders,
                                                   self.manager.orders))
            elif cmd == "tick":
                n = self.input.parse_positive_int(args)
                self.manager.tick(n - 1)
                self.write("Advanced %d tick(s)." % n)
            elif cmd == "history":
                for i, h in enumerate(self.input.history, 1):
                    self.write("%3d  %s" % (i, h))
            elif cmd == "save":
                path = args[0] if args else "dispatch_state.json"
                self.manager.save_state(path)
                self.write("State saved to %s" % path)
            elif cmd == "help":
                self.write(self.display.format_help())
            elif cmd == "quit":
                self.manager.stop()
                self.write("Goodbye.")
                return False
            else:
                self.write("Unknown command '%s'. Type 'help'." % line.strip())
        except InputError as exc:
            self.write("Input error: %s" % exc)
        except (EOFError, KeyboardInterrupt):
            self.write("\nInput cancelled.")
        except Exception as exc:  # recovery from unexpected errors
            logger.exception("Unexpected error")
            self.write("Error: %s (system state recovered)" % exc)
            self.manager.check_consistency()
        return True

    def run(self):
        self.write("Food Delivery Dispatch Simulator - type 'help'.")
        while self.manager.running:
            try:
                line = input("[t=%d] > " % self.manager.time)
            except (EOFError, KeyboardInterrupt):
                self.write("")
                self.manager.stop()
                break
            if not self.handle(line):
                break
            self.manager.tick(1)
            issues = self.manager.check_consistency()
            for issue in issues:
                logger.warning("Consistency fix: %s", issue)


def main(argv=None):
    argv = sys.argv[1:] if argv is None else argv
    level = logging.DEBUG if "--debug" in argv else logging.WARNING
    logging.basicConfig(level=level,
                        format="%(asctime)s %(name)s %(levelname)s "
                               "%(message)s")
    manager = init_system()
    Application(manager).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

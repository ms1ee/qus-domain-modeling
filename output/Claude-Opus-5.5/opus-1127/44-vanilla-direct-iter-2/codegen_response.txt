"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
``DATA_FILE_PATH`` is computed from this file's own location, so it is valid
no matter which working directory the program is started from.
"""

import json
import os

DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


class DataFormatError(ValueError):
    """Raised when the subway data file does not have the expected structure."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway JSON data file and return its content as a dict.

    Expected structure::

        {"stations": [{"name": str, "lines": [str]}],
         "connections": [{"source": str, "dest": str,
                          "distance": int, "line": str}]}

    Raises:
        FileNotFoundError: if the file does not exist.
        json.JSONDecodeError: if the file is not valid JSON.
        DataFormatError: if the top-level structure is wrong.
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    if not isinstance(data, dict):
        raise DataFormatError("Subway data must be a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise DataFormatError("'stations' and 'connections' must be lists")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file (used by the data update mechanism)."""
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

"""
Core data structures: Edge, Graph (weighted undirected graph) and MinPQ
(binary-heap priority queue used by Dijkstra's algorithm).
"""

import numbers


class Edge:
    """A weighted undirected edge between two station numbers on one line."""

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, numbers.Real):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        return self.source

    def other(self, vertex):
        """Return the endpoint of the edge different from ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError("Vertex %r is not an endpoint of this edge" % (vertex,))

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists.

    Station numbers are assigned by the caller. ``adj`` maps a station number
    to the list of incident edges. ``station_lines`` maps station names to the
    lines that serve them (used for transfer station detection).
    """

    def __init__(self):
        self.adj = {}
        self.station_lines = {}
        self._edges = []

    def add_vertex(self, v):
        if v not in self.adj:
            self.adj[v] = []

    def has_vertex(self, v):
        return v in self.adj

    def set_station_lines(self, station_name, lines):
        seen = []
        for line in lines or []:
            if line not in seen:
                seen.append(line)
        self.station_lines[station_name] = seen

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    def add_edge(self, edge):
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self.adj[edge.dest].append(edge)
        self._edges.append(edge)

    def remove_edge(self, source, dest, line=None):
        """Remove edges between source and dest (optionally only on ``line``)."""
        removed = 0
        keep = []
        for e in self._edges:
            match = {e.source, e.dest} == {source, dest} and (line is None or e.line == line)
            if match:
                removed += 1
                for v in (e.source, e.dest):
                    if v in self.adj and e in self.adj[v]:
                        self.adj[v].remove(e)
            else:
                keep.append(e)
        self._edges = keep
        return removed

    def neighbors(self, v):
        return self.adj.get(v, [])

    def vertices(self):
        return list(self.adj.keys())

    def edges(self):
        return list(self._edges)

    def V(self):
        return len(self.adj)

    def E(self):
        return len(self._edges)


class MinPQ:
    """Binary min-heap priority queue.

    Items are stored with a comparable key; ties are broken by insertion
    order so that the items themselves never need to be compared.
    """

    def __init__(self):
        self._heap = []
        self._counter = 0

    def insert(self, key, item=None):
        entry = (key, self._counter, item)
        self._counter += 1
        self._heap.append(entry)
        self._swim(len(self._heap) - 1)

    def min(self):
        if not self._heap:
            raise IndexError("Priority queue underflow")
        key, _, item = self._heap[0]
        return key, item

    def del_min(self):
        if not self._heap:
            raise IndexError("Priority queue underflow")
        heap = self._heap
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sink(0)
        return top[0], top[2]

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)

    def _less(self, i, j):
        a, b = self._heap[i], self._heap[j]
        return (a[0], a[1]) < (b[0], b[1])

    def _swim(self, k):
        heap = self._heap
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                heap[k], heap[parent] = heap[parent], heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        heap = self._heap
        n = len(heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            right = left + 1
            if right < n and self._less(right, left):
                child = right
            if self._less(child, k):
                heap[k], heap[child] = heap[child], heap[k]
                k = child
            else:
                break

"""
Path calculation using Dijkstra's algorithm with a heap-based priority queue.

The search runs over a state space (station, current line[, transfers]) so
that transfers between lines can be counted and constrained correctly at
multi-line interchange stations.

Modes:
  * shortest_distance - minimise total distance (ties: preferred lines, fewer transfers)
  * least_transfer    - minimise number of transfers, then distance
  * comprehensive     - minimise distance + transfer penalty (composite score)
"""

from models import MinPQ

INF = float("inf")

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance cost (meters) of one transfer in comprehensive mode.
# A transfer takes about 5 minutes; at ~1.5 min/km that is ~3.3 km.
DEFAULT_TRANSFER_PENALTY = 3000
# Weight multiplier applied to edges on preferred lines (comprehensive / least_transfer).
DEFAULT_PREFERENCE_FACTOR = 0.8


class PathfindingConfig:
    """Configuration of a path query."""

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None,
                 transfer_penalty=DEFAULT_TRANSFER_PENALTY,
                 preference_factor=DEFAULT_PREFERENCE_FACTOR):
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode %r, expected one of %s" % (mode, ", ".join(VALID_MODES)))
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise TypeError("max_transfers must be an integer or None")
            if max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = [l for l in preferred_lines if l]
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def __repr__(self):
        return ("PathfindingConfig(mode=%r, max_transfers=%r, preferred_lines=%r)"
                % (self.mode, self.max_transfers, self.preferred_lines))


class ShortestPath:
    """Dijkstra shortest path from a single source station number."""

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError("Source station %r is not in the graph" % (source,))
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: INF for v in graph.vertices()}
        self._best_state = {}
        self._parent = {}
        self._state_info = {}
        self._run()

    # ------------------------------------------------------------------
    def _cost(self, dist, transfers, weighted, nonpref):
        mode = self.config.mode
        if mode == MODE_SHORTEST_DISTANCE:
            return (dist, nonpref, transfers)
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, weighted, dist)
        return (weighted + self.config.transfer_penalty * transfers, transfers, dist)

    def _run(self):
        cfg = self.config
        preferred = set(cfg.preferred_lines)
        track_transfers = cfg.max_transfers is not None
        start = (self.source, None, 0)
        # state -> (dist, transfers, weighted, nonpref)
        self._state_info[start] = (0, 0, 0, 0)
        best_cost = {start: self._cost(0, 0, 0, 0)}
        pq = MinPQ()
        pq.insert(best_cost[start], start)
        settled = set()

        while not pq.is_empty():
            cost, state = pq.del_min()
            if state in settled or cost != best_cost.get(state):
                continue
            settled.add(state)
            node, line, _ = state
            dist, transfers, weighted, nonpref = self._state_info[state]
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = dist

            for edge in self.graph.neighbors(node):
                w = edge.other(node)
                is_transfer = line is not None and edge.line != line
                new_t = transfers + (1 if is_transfer else 0)
                if track_transfers and new_t > cfg.max_transfers:
                    continue
                new_state = (w, edge.line, new_t if track_transfers else 0)
                if new_state in settled:
                    continue
                on_pref = edge.line in preferred
                new_dist = dist + edge.weight
                new_weighted = weighted + (edge.weight * cfg.preference_factor if on_pref else edge.weight)
                new_nonpref = nonpref + (0 if on_pref or not preferred else edge.weight)
                new_cost = self._cost(new_dist, new_t, new_weighted, new_nonpref)
                old = best_cost.get(new_state)
                if old is None or new_cost < old:
                    best_cost[new_state] = new_cost
                    self._state_info[new_state] = (new_dist, new_t, new_weighted, new_nonpref)
                    self._parent[new_state] = (state, edge)
                    pq.insert(new_cost, new_state)

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return (station number list, edge list) from source to target.

        Returns ([], []) when the target is unreachable.
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        nodes = [state[0]]
        edges = []
        while state in self._parent:
            prev, edge = self._parent[state]
            edges.append(edge)
            nodes.append(prev[0])
            state = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the path to target (0 if unreachable)."""
        _, edges = self.path_with_edges(target)
        return count_transfers(edges)


def count_transfers(edges):
    count = 0
    for prev, cur in zip(edges, edges[1:]):
        if prev.line != cur.line:
            count += 1
    return count

"""
Beijing Subway fare calculation.

Fare tiers (left-closed, right-open intervals):
  [0, 6) km   -> 3 yuan
  [6, 12) km  -> 4 yuan
  [12, 22) km -> 5 yuan
  [22, 32) km -> 6 yuan
  >= 32 km    -> 7 yuan + 1 yuan per additional full 20 km
"""

import math
import numbers

# (upper bound in meters (exclusive), fare in yuan)
FARE_TIERS = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_BASE_DISTANCE = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP = 20000

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5


class FareCalculationError(ValueError):
    """Raised when a fare cannot be computed for the given distance."""


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, numbers.Real):
        raise FareCalculationError("Distance must be a number, got %r" % (distance,))
    if math.isnan(distance) or math.isinf(distance):
        raise FareCalculationError("Distance must be finite")
    if distance < 0:
        raise FareCalculationError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare (yuan, int) for a travel distance in meters.

    Fractional meters are rounded up before tier lookup.
    """
    _validate_distance(distance)
    meters = int(math.ceil(distance))
    for upper, fare in FARE_TIERS:
        if meters < upper:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_BASE_DISTANCE) // EXTENDED_STEP


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes: 1.5 min/km + 5 min per transfer."""
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, numbers.Real) or transfer_count < 0:
        raise FareCalculationError("Transfer count must be a non-negative number")
    return float(distance) / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""
Subway network modeling and data management.

Builds the station_to_number mapping, the adjacency-list ``graph`` dict and a
``Graph`` object from the JSON data, and provides integrity checks and data
update operations (adding lines, modifying stations, updating distances).
"""

import copy
from collections import deque

from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
from models import Edge, Graph
from pathfinding import ShortestPath, PathfindingConfig

REQUIRED_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

# Representative stations of main lines (English / Chinese alternatives).
REPRESENTATIVE_STATIONS = {
    "Line 1": ["Xidan", "西单"],
    "Line 2": ["Xizhimen", "西直门"],
    "Line 4": ["Zhongguancun", "中关村"],
    "Line 5": ["Dongdan", "东单"],
    "Line 6": ["Chaoyangmen", "朝阳门"],
    "Line 8": ["Gulou Dajie", "鼓楼大街"],
    "Line 10": ["Guomao", "国贸"],
    "Line 13": ["Huilongguan", "回龙观"],
    "Line 14": ["Wangjing", "望京"],
    "Line 15": ["Wangjing East", "望京东"],
    "Changping Line": ["Shahe", "沙河"],
    "Yizhuang Line": ["Songjiazhuang", "宋家庄"],
    "Fangshan Line": ["Guogongzhuang", "郭公庄"],
    "Airport Line": ["Sanyuanqiao", "三元桥"],
}

MIN_REASONABLE_DISTANCE = 100       # meters
MAX_REASONABLE_DISTANCE = 30000     # meters


def normalize_name(name):
    """Normalize a station/line name for tolerant matching."""
    if name is None:
        return ""
    s = str(name).strip().lower()
    for ch in (" ", "\t", "'", "’", "`", "-", "_", "·", "　"):
        s = s.replace(ch, "")
    return s


def _strip_suffix(norm):
    for suffix in ("station", "站"):
        if norm.endswith(suffix) and len(norm) > len(suffix):
            return norm[: -len(suffix)]
    return norm


def normalize_line(name):
    s = normalize_name(name)
    if s.endswith("号线"):
        s = "line" + s[:-2]
    return s


class SubwayNetwork:
    """The Beijing Subway network model."""

    def __init__(self, data=None, data_path=DATA_FILE_PATH):
        self.data_path = data_path
        if data is None:
            data = load_subway_data(data_path)
        self.data = copy.deepcopy(data)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])
        self.build()

    # ------------------------------------------------------------------
    # Model construction
    # ------------------------------------------------------------------
    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = {}                 # number -> [(neighbor, distance, line)]
        self.subway_graph = Graph()
        self.lines = {}                 # line -> set of station names
        self._lookup = {}
        self._line_lookup = {}
        self.load_warnings = []

        station_lines = {}
        for st in self.data["stations"]:
            name = st.get("name") if isinstance(st, dict) else None
            if not name:
                self.load_warnings.append("Station entry without name: %r" % (st,))
                continue
            self._add_station_number(name)
            lst = station_lines.setdefault(name, [])
            for line in st.get("lines", []) or []:
                if line not in lst:
                    lst.append(line)

        for conn in self.data["connections"]:
            try:
                src, dst = conn["source"], conn["dest"]
                dist, line = conn["distance"], conn.get("line", "")
            except (KeyError, TypeError):
                self.load_warnings.append("Malformed connection: %r" % (conn,))
                continue
            for name in (src, dst):
                if name not in self.station_to_number:
                    self.load_warnings.append("Station '%s' used in connection but not declared" % name)
                    self._add_station_number(name)
                lst = station_lines.setdefault(name, [])
                if line and line not in lst:
                    lst.append(line)
            try:
                edge = Edge(self.station_to_number[src], self.station_to_number[dst], dist, line)
            except (TypeError, ValueError) as exc:
                self.load_warnings.append("Invalid connection %s-%s: %s" % (src, dst, exc))
                continue
            self.subway_graph.add_edge(edge)
            a, b = edge.source, edge.dest
            self.graph[a].append((b, dist, line))
            if a != b:
                self.graph[b].append((a, dist, line))

        for name, lines in station_lines.items():
            self.subway_graph.set_station_lines(name, lines)
            for line in lines:
                self.lines.setdefault(line, set()).add(name)
        for line in self.lines:
            self._line_lookup[normalize_line(line)] = line

    def _add_station_number(self, name):
        if name in self.station_to_number:
            return self.station_to_number[name]
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name
        self.graph[num] = []
        self.subway_graph.add_vertex(num)
        norm = normalize_name(name)
        self._lookup.setdefault(norm, name)
        self._lookup.setdefault(_strip_suffix(norm), name)
        return num

    # ------------------------------------------------------------------
    # Queries
    # ------------------------------------------------------------------
    @property
    def station_lines(self):
        return self.subway_graph.station_lines

    def find_station(self, name):
        """Return the official station name matching ``name`` or None."""
        if name is None:
            return None
        name = str(name).strip()
        if name in self.station_to_number:
            return name
        norm = normalize_name(name)
        if not norm:
            return None
        if norm in self._lookup:
            return self._lookup[norm]
        return self._lookup.get(_strip_suffix(norm))

    def find_line(self, name):
        if name in self.lines:
            return name
        norm = normalize_line(name)
        if norm in self._line_lookup:
            return self._line_lookup[norm]
        if norm.isdigit():
            return self._line_lookup.get("line" + norm)
        return None

    def is_transfer_station(self, name):
        return self.subway_graph.is_transfer_station(name)

    def get_line_stations(self, line):
        """Return the ordered station sequence of a line (derived from connections)."""
        adj = {}
        for e in self.subway_graph.edges():
            if e.line != line:
                continue
            adj.setdefault(e.source, set()).add(e.dest)
            adj.setdefault(e.dest, set()).add(e.source)
        if not adj:
            return []
        ends = sorted(v for v, n in adj.items() if len(n) == 1)
        start = ends[0] if ends else min(adj)
        order, visited, stack = [], set(), [start]
        while stack:
            v = stack.pop()
            if v in visited:
                continue
            visited.add(v)
            order.append(self.number_to_station[v])
            stack.extend(sorted(adj[v] - visited, reverse=True))
        return order

    def find_route(self, start, end, config=None):
        """Compute a route. Returns (station names, edges) or ([], [])."""
        config = config or PathfindingConfig()
        s = self.station_to_number[start]
        t = self.station_to_number[end]
        sp = ShortestPath(self.subway_graph, s, config)
        nodes, edges = sp.path_with_edges(t)
        return [self.number_to_station[n] for n in nodes], edges

    # ------------------------------------------------------------------
    # Integrity check
    # ------------------------------------------------------------------
    def check_integrity(self):
        errors, warnings = [], list(self.load_warnings)
        stats = {
            "stations": len(self.station_to_number),
            "connections": self.subway_graph.E(),
            "lines": len(self.lines),
        }

        # Station coverage
        present_lines = {normalize_line(l) for l in self.lines}
        missing_lines = [l for l in REQUIRED_LINES if normalize_line(l) not in present_lines]
        if missing_lines:
            warnings.append("Main lines missing from data: " + ", ".join(missing_lines))
        missing_rep = []
        for line, candidates in REPRESENTATIVE_STATIONS.items():
            if not any(self.find_station(c) for c in candidates):
                missing_rep.append("%s (%s)" % (candidates[0], line))
        if missing_rep:
            warnings.append("Representative stations not found: " + ", ".join(missing_rep))

        # Station declarations
        names_seen = set()
        for st in self.data["stations"]:
            if not isinstance(st, dict) or not st.get("name"):
                errors.append("Invalid station entry: %r" % (st,))
                continue
            if st["name"] in names_seen:
                warnings.append("Duplicate station entry: %s" % st["name"])
            names_seen.add(st["name"])
            if not st.get("lines"):
                warnings.append("Station '%s' has no line information" % st["name"])

        # Connections
        pairs = set()
        for conn in self.data["connections"]:
            if not isinstance(conn, dict) or any(k not in conn for k in ("source", "dest", "distance")):
                errors.append("Malformed connection: %r" % (conn,))
                continue
            src, dst, dist = conn["source"], conn["dest"], conn["distance"]
            line = conn.get("line")
            if src not in names_seen:
                errors.append("Connection references undeclared station '%s'" % src)
            if dst not in names_seen:
                errors.append("Connection references undeclared station '%s'" % dst)
            if src == dst:
                errors.append("Self-loop connection at '%s'" % src)
            if not line:
                warnings.append("Connection %s-%s has no line" % (src, dst))
            key = (frozenset((src, dst)), line)
            if key in pairs:
                warnings.append("Duplicate connection %s-%s on %s" % (src, dst, line))
            pairs.add(key)
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                errors.append("Non-numeric distance for %s-%s" % (src, dst))
            elif dist <= 0:
                errors.append("Non-positive distance %s for %s-%s" % (dist, src, dst))
            elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                warnings.append("Suspicious distance %s m for %s-%s" % (dist, src, dst))
            for name in (src, dst):
                if line and name in names_seen:
                    declared = next((s.get("lines", []) for s in self.data["stations"]
                                     if isinstance(s, dict) and s.get("name") == name), [])
                    if declared and line not in declared:
                        warnings.append("Station '%s' is on %s per connections but not declared"
                                        % (name, line))

        # Isolated stations and connectivity
        isolated = [self.number_to_station[v] for v, adj in self.graph.items() if not adj]
        if isolated:
            warnings.append("Isolated stations: " + ", ".join(sorted(isolated)))
        if self.graph:
            start = next(iter(self.graph))
            seen = {start}
            q = deque([start])
            while q:
                v = q.popleft()
                for w, _, _ in self.graph[v]:
                    if w not in seen:
                        seen.add(w)
                        q.append(w)
            unreachable = len(self.graph) - len(seen)
            stats["unreachable_from_first_station"] = unreachable
            if unreachable:
                warnings.append("%d stations are not connected to the main network" % unreachable)

        return {"passed": not errors, "errors": errors, "warnings": warnings, "stats": stats}

    # ------------------------------------------------------------------
    # Data update mechanisms
    # ------------------------------------------------------------------
    def add_line(self, line_name, stations, distances):
        """Add a new line given ordered station names and inter-station distances."""
        if len(stations) < 2 or len(distances) != len(stations) - 1:
            raise ValueError("A line needs >= 2 stations and len(stations)-1 distances")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        existing = {s["name"]: s for s in self.data["stations"]}
        for name in stations:
            if name in existing:
                if line_name not in existing[name].setdefault("lines", []):
                    existing[name]["lines"].append(line_name)
            else:
                entry = {"name": name, "lines": [line_name]}
                self.data["stations"].append(entry)
                existing[name] = entry
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.build()

    def modify_station(self, old_name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        if old_name not in self.station_to_number:
            raise KeyError("Station '%s' does not exist" % old_name)
        if new_name and new_name != old_name and new_name in self.station_to_number:
            raise ValueError("Station '%s' already exists" % new_name)
        for st in self.data["stations"]:
            if st.get("name") == old_name:
                if new_name:
                    st["name"] = new_name
                if lines is not None:
                    st["lines"] = list(lines)
        if new_name:
            for conn in self.data["connections"]:
                if conn.get("source") == old_name:
                    conn["source"] = new_name
                if conn.get("dest") == old_name:
                    conn["dest"] = new_name
        self.build()

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance of an existing connection. Returns number updated."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        updated = 0
        for conn in self.data["connections"]:
            if {conn.get("source"), conn.get("dest")} == {station_a, station_b} and \
                    (line is None or conn.get("line") == line):
                conn["distance"] = distance
                updated += 1
        if not updated:
            raise KeyError("No connection between '%s' and '%s'" % (station_a, station_b))
        self.build()
        return updated

    def save(self, path=None):
        save_subway_data(self.data, path or self.data_path)

"""
Route summary generation, visualization and export (text / JSON).
"""

import json

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import count_transfers


def format_distance(meters):
    """Meters for short distances, kilometers for long ones."""
    if meters < 1000:
        return "%d m" % round(meters)
    return "%.2f km" % (meters / 1000.0)


def build_segments(stations, edges):
    """Group consecutive edges on the same line into ride segments."""
    segments = []
    for i, edge in enumerate(edges):
        a, b = stations[i], stations[i + 1]
        if segments and segments[-1]["line"] == edge.line:
            seg = segments[-1]
            seg["to"] = b
            seg["stations"].append(b)
            seg["distance"] += edge.weight
        else:
            segments.append({"line": edge.line, "from": a, "to": b,
                             "stations": [a, b], "distance": edge.weight})
    for seg in segments:
        seg["stops"] = len(seg["stations"]) - 1
    return segments


def build_route_info(stations, edges, query, fare_func=Sum_money):
    """Build the structured route summary dict."""
    distance = sum(e.weight for e in edges)
    transfers = count_transfers(edges)
    segments = build_segments(stations, edges)
    transfer_stations = [seg["from"] for seg in segments[1:]]
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    fare = fare_func(distance)
    est = calculate_estimated_time(distance, transfers)
    return {
        "query": dict(query),
        "path": list(stations),
        "station_count": len(stations),
        "segments": segments,
        "distance_meters": distance,
        "distance_km": round(distance / 1000.0, 3),
        "fare": fare,
        "estimated_time_minutes": round(est, 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
    }


def format_text(info):
    sep = "=" * 50
    q = info["query"]
    out = [sep, "Route: %s -> %s" % (q.get("start"), q.get("end"))]
    if q.get("mode"):
        out.append("Mode: %s" % q["mode"])
    out.append("-" * 50)
    out.append("Station sequence (%d stations):" % info["station_count"])
    out.append("  " + " -> ".join(info["path"]))
    out.append("Line segments:")
    for i, seg in enumerate(info["segments"]):
        if i > 0:
            prev = info["segments"][i - 1]
            out.append("  >> Transfer at %s: %s -> %s" % (seg["from"], prev["line"], seg["line"]))
        out.append("  [%s] %s -> %s (%d stops, %s)" % (
            seg["line"], seg["from"], seg["to"], seg["stops"], format_distance(seg["distance"])))
    out.append("Transfer details:")
    out.append("  Transfers: %d" % info["transfer_count"])
    out.append("  Transfer stations: %s" % (", ".join(info["transfer_stations"]) or "None"))
    out.append("  Lines involved: %s" % (", ".join(info["lines"]) or "None"))
    d = info["distance_meters"]
    out.append("Total distance: %s (%d meters, %.2f kilometers)" % (format_distance(d), round(d), d / 1000.0))
    out.append("Estimated time: %.1f minutes" % info["estimated_time_minutes"])
    out.append("Fare: %d yuan" % info["fare"])
    out.append(sep)
    return "\n".join(out)


def format_json(info, indent=2):
    return json.dumps(info, ensure_ascii=False, indent=indent)


def format_report(info, fmt="text"):
    return format_json(info) if fmt == "json" else format_text(info)

"""
Command-line interface for the Beijing Subway fare calculation system.

Usage:
    python main.py [--format text|json] [--output_file PATH]
                   [--mode shortest_distance|least_transfer|comprehensive]
                   [--max_transfers N] [--line_preference "Line 1[,Line 2]"]
                   [--check_data]

Enter queries as "Starting Station; Ending Station". Type 'quit' or 'exit' to leave.
"""

import argparse
import json
import sys

from data_loader import DATA_FILE_PATH
from fare_calculator import FareCalculationError
from pathfinding import PathfindingConfig, VALID_MODES
from report import build_route_info, format_report
from subway_network import SubwayNetwork

EXIT_COMMANDS = {"quit", "exit", "q"}
FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class QueryError(Exception):
    """User-facing query error."""


def parse_query(text):
    text = (text or "").strip().replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2:
        raise QueryError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryError(FORMAT_ERROR)
    return start, end


def process_query(network, text, config):
    """Validate and run one query. Returns the route info dict; raises QueryError."""
    start_raw, end_raw = parse_query(text)
    start = network.find_station(start_raw)
    if start is None:
        raise QueryError("Starting station '%s' does not exist, please check the station name" % start_raw)
    end = network.find_station(end_raw)
    if end is None:
        raise QueryError("Ending station '%s' does not exist, please check the station name" % end_raw)
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")
    try:
        stations, edges = network.find_route(start, end, config)
    except Exception as exc:  # path calculation exception handling
        raise QueryError("Path calculation error: %s" % exc)
    if not stations:
        if config.max_transfers is not None:
            raise QueryError("No path found from '%s' to '%s' within %d transfer(s)"
                             % (start, end, config.max_transfers))
        raise QueryError("No path found from '%s' to '%s'" % (start, end))
    query = {
        "start": start,
        "end": end,
        "mode": config.mode,
        "max_transfers": config.max_transfers,
        "line_preference": list(config.preferred_lines),
    }
    try:
        return build_route_info(stations, edges, query)
    except FareCalculationError as exc:
        raise QueryError("Fare calculation error: %s" % exc)
    except (ValueError, TypeError) as exc:
        raise QueryError("Fare calculation error: %s" % exc)


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text", help="Output format")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                   help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="Preferred line(s), comma separated (e.g. 'Line 1')")
    p.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    return p


def print_integrity_report(report):
    print("Data integrity check")
    print("-" * 50)
    for k, v in report["stats"].items():
        print("  %s: %s" % (k, v))
    print("Errors: %d" % len(report["errors"]))
    for e in report["errors"]:
        print("  [ERROR] %s" % e)
    print("Warnings: %d" % len(report["warnings"]))
    for w in report["warnings"]:
        print("  [WARNING] %s" % w)
    print("Result: %s" % ("PASSED" if report["passed"] else "FAILED"))


def write_output(path, fmt, results):
    with open(path, "w", encoding="utf-8") as f:
        if fmt == "json":
            payload = results[0] if len(results) == 1 else results
            json.dump(payload, f, ensure_ascii=False, indent=2)
            f.write("\n")
        else:
            f.write("\n\n".join(format_report(r, "text") for r in results))
            f.write("\n")


def main(argv=None):
    args = build_parser().parse_args(argv)

    if args.max_transfers is not None and args.max_transfers < 0:
        print("Error: --max_transfers must be a non-negative integer")
        return 2

    try:
        network = SubwayNetwork(data_path=args.data_file)
    except FileNotFoundError:
        print("Error: subway data file not found: %s" % args.data_file)
        return 1
    except (ValueError, OSError) as exc:
        print("Error: failed to load subway data: %s" % exc)
        return 1

    if args.check_data:
        report = network.check_integrity()
        print_integrity_report(report)
        return 0 if report["passed"] else 1

    preferred = []
    if args.line_preference:
        for raw in args.line_preference.replace("，", ",").split(","):
            raw = raw.strip()
            if not raw:
                continue
            line = network.find_line(raw)
            if line is None:
                print("Warning: line preference '%s' does not match any known line" % raw)
                preferred.append(raw)
            else:
                preferred.append(line)

    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=preferred)
    except (ValueError, TypeError) as exc:
        print("Error: %s" % exc)
        return 2

    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' to query, 'quit' to exit.")
    results = []
    while True:
        try:
            text = input("Query> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if text.strip().lower() in EXIT_COMMANDS:
            break
        if not text.strip():
            print(FORMAT_ERROR)
            continue
        try:
            info = process_query(network, text, config)
        except QueryError as exc:
            print(str(exc))
            continue
        except Exception as exc:  # last-resort safety net
            print("Unexpected error: %s" % exc)
            continue
        print(format_report(info, args.format))
        if args.output_file:
            results.append(info)
            try:
                write_output(args.output_file, args.format, results)
                print("Result exported to %s" % args.output_file)
            except OSError as exc:
                print("Error: failed to write output file: %s" % exc)
        sys.stdout.flush()
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

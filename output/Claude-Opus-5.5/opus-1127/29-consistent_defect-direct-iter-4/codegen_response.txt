"""Global variable storage shared across the dispatch system modules.

The module keeps a single dictionary that is used as a light-weight
registry for system wide values such as funds, the rider list, the
rider id counter, the simulated time and the order list.
"""

_global_dict = None


def _init():
    """Initialise (or reset) the global dictionary."""
    global _global_dict
    _global_dict = {}


def _ensure():
    """Make sure the global dictionary exists (lazy initialisation)."""
    global _global_dict
    if _global_dict is None:
        _global_dict = {}


def set_value(key, value):
    """Store ``value`` under ``key`` in the global dictionary."""
    _ensure()
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` or ``default`` if missing."""
    _ensure()
    return _global_dict.get(key, default)


def has_value(key):
    """Return True if ``key`` exists in the global dictionary."""
    _ensure()
    return key in _global_dict

"""Grid based path planning for riders.

The map is an 800x800 coordinate system split into 8x8 cells of
100x100 units.  Lines whose coordinate is a multiple of 100 are
streets.  Restaurants and customers are located at cell centres.
"""

import math

GRID_SIZE = 100
MAP_SIZE = 800
GRID_COUNT = MAP_SIZE // GRID_SIZE


def to_grid_cell(value):
    """Return the grid cell index (0-7) for a raw coordinate.

    Values outside the map are clamped onto the boundary cells.
    """
    try:
        v = int(value)
    except (TypeError, ValueError):
        v = 0
    v = max(0, min(v, MAP_SIZE))
    return min(v // GRID_SIZE, GRID_COUNT - 1)


def to_grid_center(value):
    """Convert a raw coordinate into the centre of its grid cell.

    ``center = (value // 100) * 100 + 50`` with boundary clamping.
    """
    return to_grid_cell(value) * GRID_SIZE + GRID_SIZE // 2


def to_grid_corner(value):
    """Convert a raw coordinate into the top-left corner of its grid cell."""
    return to_grid_cell(value) * GRID_SIZE


class PathPlanner:
    """Grid based pathfinding helper (L-shaped routes along streets)."""

    @staticmethod
    def is_vertical_special(start, target):
        """Return True if the start is directly above or below the target."""
        return start[0] == target[0] and start[1] != target[1]

    @staticmethod
    def plan_route(start, target):
        """Plan a list of waypoints from ``start`` to ``target``.

        Movement is horizontal and vertical only, with at most one
        right-angle turn.  If the rider is currently on a horizontal
        street it travels horizontally first; if it is only on a
        vertical street it travels vertically first.
        """
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        if (sx, sy) == (tx, ty):
            return []
        if sx == tx or sy == ty:
            return [(tx, ty)]
        on_horizontal = sy % GRID_SIZE == 0
        on_vertical = sx % GRID_SIZE == 0
        if on_horizontal or not on_vertical:
            corner = (tx, sy)
        else:
            corner = (sx, ty)
        return [corner, (tx, ty)]

    @staticmethod
    def plan_full_route(start, restaurant, customer):
        """Plan the full multi-destination route start->restaurant->customer.

        Returns a list of points including the start point.
        """
        points = [(int(start[0]), int(start[1]))]
        first = PathPlanner.plan_route(start, restaurant)
        points.extend(first)
        second = PathPlanner.plan_route(restaurant, customer)
        points.extend(second)
        return points

    @staticmethod
    def route_length(points):
        """Return the total Manhattan length of a list of points."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    @staticmethod
    def advance(x, y, route, budget):
        """Move along ``route`` by at most ``budget`` units.

        Returns ``(x, y, remaining_route, moved_distance)``.
        """
        route = list(route)
        moved = 0
        while route and budget > 0:
            wx, wy = route[0]
            dist = abs(wx - x) + abs(wy - y)
            if dist <= budget:
                x, y = wx, wy
                budget -= dist
                moved += dist
                route.pop(0)
                continue
            if wx != x:
                step = budget if wx > x else -budget
                x += step
            else:
                step = budget if wy > y else -budget
                y += step
            moved += budget
            budget = 0
        return x, y, route, moved

    @staticmethod
    def euclidean(a, b):
        """Euclidean distance between two points."""
        return math.sqrt((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2)

    @staticmethod
    def manhattan(a, b):
        """Manhattan distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Order data model."""

import globalvar
from path_planner import to_grid_center, PathPlanner


class Order:
    """A delivery order from a restaurant to a customer.

    Coordinates are automatically converted to grid cell centres.
    Status lifecycle: pending -> assigned -> picking_up -> delivering
    -> completed.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.raw_res_pos = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus_pos = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = (to_grid_center(res_pos[0]), to_grid_center(res_pos[1]))
        self.cus_pos = (to_grid_center(cus_pos[0]), to_grid_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.created_time = globalvar.get_value("time", 0)
        self.assigned_time = None
        self.accepted_time = None
        self.picked_time = None
        self.completed_time = None
        self.distance = 0

    @property
    def order_id(self):
        """Alias of ``id``."""
        return self.id

    def assign_to(self, rider_id):
        """Record that this order is assigned to rider ``rider_id``."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"
        self.assigned_time = globalvar.get_value("time", 0)

    def is_completed(self):
        """Return True if the order has been delivered."""
        return self.status == "completed"

    def direct_distance(self):
        """Manhattan distance from restaurant to customer."""
        return PathPlanner.manhattan(self.res_pos, self.cus_pos)

    def time_taken(self):
        """Ticks from creation to completion (None if not completed)."""
        if self.completed_time is None:
            return None
        return self.completed_time - self.created_time

    def to_dict(self):
        """Serialise the order to a JSON compatible dict."""
        return {
            "id": self.id,
            "raw_res_pos": list(self.raw_res_pos),
            "raw_cus_pos": list(self.raw_cus_pos),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_time": self.created_time,
            "assigned_time": self.assigned_time,
            "accepted_time": self.accepted_time,
            "picked_time": self.picked_time,
            "completed_time": self.completed_time,
            "distance": self.distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from a dict produced by ``to_dict``."""
        order = cls.__new__(cls)
        order.id = int(data["id"])
        order.raw_res_pos = tuple(data.get("raw_res_pos", data["res_pos"]))
        order.raw_cus_pos = tuple(data.get("raw_cus_pos", data["cus_pos"]))
        order.res_pos = tuple(data["res_pos"])
        order.cus_pos = tuple(data["cus_pos"])
        order.status = data.get("status", "pending")
        order.rider_id = data.get("rider_id")
        order.created_time = data.get("created_time", 0)
        order.assigned_time = data.get("assigned_time")
        order.accepted_time = data.get("accepted_time")
        order.picked_time = data.get("picked_time")
        order.completed_time = data.get("completed_time")
        order.distance = data.get("distance", 0)
        if cls._next_id <= order.id:
            cls._next_id = order.id + 1
        return order

    def __repr__(self):
        return "Order(id=%d, res=%s, cus=%s, status=%s, rider=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status, self.rider_id)

"""Rider model: movement, state machine and order handling."""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}

DEFAULT_START = (400, 400)
DEFAULT_SPEED = 20


class Rider:
    """A delivery rider.

    Attributes:
        rider_id: unique identifier.
        x, y: current position.
        speed: units moved per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: all orders assigned to the rider (including history).
        special_case: True while moving straight up/down to a target.
        log: list of event dicts with keys "event" and "order_id".
    """

    def __init__(self, rider_id):
        self.rider_id = rider_id
        cfg = globalvar.get_value("config") or {}
        start = cfg.get("rider_start", DEFAULT_START)
        self.x = int(start[0])
        self.y = int(start[1])
        self.speed = int(cfg.get("rider_speed", DEFAULT_SPEED))
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.route = []
        self.total_distance = 0
        self.log = []
        self._log("spawn", None)

    # ------------------------------------------------------------------
    def _log(self, event, order_id):
        entry = {
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("time", 0),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)

    @property
    def position(self):
        """Current (x, y) position."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    def current_order(self):
        """Return the first not-yet-completed order, or None."""
        for order in self.orders:
            if getattr(order, "status", None) != "completed":
                return order
        return None

    def active_orders(self):
        """Return all orders that are not completed."""
        return [o for o in self.orders
                if getattr(o, "status", None) != "completed"]

    def is_idle(self):
        """True if the rider is idle with no outstanding work."""
        return self.state == STATE_IDLE and self.current_order() is None

    def _plan(self, target):
        self.route = PathPlanner.plan_route((self.x, self.y), target)
        self.special_case = PathPlanner.is_vertical_special(
            (self.x, self.y), target)

    def _accept(self, order):
        now = globalvar.get_value("time", 0)
        self.state = STATE_PICKING_UP
        if getattr(order, "rider_id", None) is None:
            order.rider_id = self.rider_id
        order.status = "picking_up"
        order.accepted_time = now
        self._log("accepted_order", order.id)
        self._plan(order.res_pos)

    # ------------------------------------------------------------------
    def move(self):
        """Advance the rider by one tick.

        Handles accepting new orders, moving along the planned route,
        and state transitions on arrival at the restaurant/customer.
        """
        order = self.current_order()
        if order is None:
            if self.state != STATE_IDLE:
                self.state = STATE_IDLE
            self.route = []
            self.special_case = False
            return

        if self.state == STATE_IDLE:
            self._accept(order)

        target = tuple(order.res_pos) if self.state == STATE_PICKING_UP \
            else tuple(order.cus_pos)
        if not self.route or tuple(self.route[-1]) != target:
            if (self.x, self.y) != target:
                self._plan(target)

        self.x, self.y, self.route, moved = PathPlanner.advance(
            self.x, self.y, self.route, self.speed)
        self.total_distance += moved
        order.distance = getattr(order, "distance", 0) + moved

        if (self.x, self.y) != target:
            return

        now = globalvar.get_value("time", 0)
        self.special_case = False
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.status = "delivering"
            order.picked_time = now
            self._log("arrived_at_merchant", order.id)
            self._plan(tuple(order.cus_pos))
        elif self.state == STATE_DELIVERING:
            order.status = "completed"
            order.completed_time = now
            self._log("delivered_to_customer", order.id)
            self.route = []
            next_order = self.current_order()
            if next_order is not None:
                self._accept(next_order)
            else:
                self.state = STATE_IDLE

    # ------------------------------------------------------------------
    def to_dict(self):
        """Serialise rider data to a JSON compatible dict."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "route": [list(p) for p in self.route],
            "total_distance": self.total_distance,
            "order_ids": [o.id for o in self.orders],
            "log": list(self.log),
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        """Rebuild a rider from serialised data."""
        rider = cls.__new__(cls)
        rider.rider_id = data["rider_id"]
        rider.x = int(data["x"])
        rider.y = int(data["y"])
        rider.speed = int(data.get("speed", DEFAULT_SPEED))
        rider.state = int(data.get("state", STATE_IDLE))
        rider.special_case = bool(data.get("special_case", False))
        rider.route = [tuple(p) for p in data.get("route", [])]
        rider.total_distance = data.get("total_distance", 0)
        rider.orders = [orders_by_id[i] for i in data.get("order_ids", [])
                        if i in orders_by_id]
        rider.log = list(data.get("log", []))
        return rider

    def __repr__(self):
        return "Rider(id=%s, pos=(%s, %s), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state_name, len(self.orders))

"""System manager: funds, riders, time, recruitment and persistence."""

import json
import logging
import os

import globalvar
from rider import Rider, STATE_IDLE

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruit_cost": 500,
    "recruit_threshold": 300,
    "rider_speed": 20,
    "rider_start": [400, 400],
}


def load_config(path):
    """Load a JSON configuration file merged over the defaults."""
    cfg = dict(DEFAULT_CONFIG)
    if path and os.path.exists(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                for key in DEFAULT_CONFIG:
                    if key in data:
                        cfg[key] = data[key]
        except (OSError, ValueError) as exc:
            logger.warning("Failed to load config %s: %s", path, exc)
    return cfg


class SystemManager:
    """Manage funds, riders, orders and simulated time."""

    def __init__(self, config=None):
        self.config = dict(DEFAULT_CONFIG)
        if config:
            self.config.update(config)

    def init(self):
        """Initialise global state."""
        globalvar._init()
        globalvar.set_value("config", self.config)
        globalvar.set_value("funds", int(self.config["initial_funds"]))
        globalvar.set_value("riders", [])
        globalvar.set_value("rider_id_counter", 0)
        globalvar.set_value("running", True)
        globalvar.set_value("orders", [])
        globalvar.set_value("time", 0)
        globalvar.set_value("system", self)

    # ------------------------------------------------------------------
    @property
    def funds(self):
        """Current system funds."""
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def riders(self):
        """List of riders."""
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def orders(self):
        """List of all orders."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    @property
    def time(self):
        """Current simulated tick."""
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        """System running flag."""
        return globalvar.get_value("running", True)

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    # ------------------------------------------------------------------
    def recruit_rider(self):
        """Create a new rider and deduct the recruitment cost."""
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        self.funds = self.funds - int(self.config["recruit_cost"])
        logger.info("Recruited rider %d", counter)
        return rider

    def get_rider(self, rider_id):
        """Return the rider with the given id or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    def idle_rider_count(self):
        """Number of riders that are idle with no outstanding orders."""
        return sum(1 for r in self.riders
                   if r.state == STATE_IDLE and r.current_order() is None)

    def pending_order_count(self):
        """Number of orders not yet assigned to any rider."""
        return sum(1 for o in self.orders if o.status == "pending")

    def auto_recruit(self):
        """Recruit riders while funds allow and pending > idle riders."""
        recruited = []
        threshold = int(self.config["recruit_threshold"])
        while (self.funds >= threshold
               and self.pending_order_count() > self.idle_rider_count()):
            recruited.append(self.recruit_rider())
        return recruited

    # ------------------------------------------------------------------
    def tick(self, count=1):
        """Advance the simulation by ``count`` ticks."""
        for _ in range(max(0, int(count))):
            globalvar.set_value("time", self.time + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception as exc:  # recover from rider errors
                    logger.error("Rider %s failed to move: %s",
                                 rider.rider_id, exc)
                    rider.route = []

    def check_consistency(self):
        """Repair simple inconsistencies; return list of fixes applied."""
        fixes = []
        for rider in self.riders:
            if rider.current_order() is None and rider.state != STATE_IDLE:
                rider.state = STATE_IDLE
                fixes.append("rider %s reset to idle" % rider.rider_id)
        for order in self.orders:
            if order.status == "pending" and order.rider_id is not None:
                order.status = "assigned"
                fixes.append("order %s marked assigned" % order.id)
        return fixes

    # ------------------------------------------------------------------
    def save_state(self, path):
        """Persist system parameters, riders and orders to JSON."""
        data = {
            "config": self.config,
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "time": self.time,
            "orders": [o.to_dict() for o in self.orders],
            "riders": [r.to_dict() for r in self.riders],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)

    def load_state(self, path):
        """Load state previously written by ``save_state``."""
        from order_model import Order
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        cfg = dict(DEFAULT_CONFIG)
        cfg.update(data.get("config", {}))
        self.config = cfg
        globalvar.set_value("config", cfg)
        globalvar.set_value("funds", int(data.get("funds", 0)))
        globalvar.set_value("rider_id_counter",
                            int(data.get("rider_id_counter", 0)))
        globalvar.set_value("time", int(data.get("time", 0)))
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        by_id = {o.id: o for o in orders}
        riders = [Rider.from_dict(d, by_id) for d in data.get("riders", [])]
        globalvar.set_value("orders", orders)
        globalvar.set_value("riders", riders)
        globalvar.set_value("running", True)


def init_system(config=None):
    """Initialise the global system state and return the SystemManager."""
    manager = SystemManager(config)
    manager.init()
    return manager


def get_system():
    """Return the active SystemManager, creating one if necessary."""
    manager = globalvar.get_value("system")
    if manager is None:
        manager = SystemManager(globalvar.get_value("config"))
        if globalvar.get_value("riders") is None:
            manager.init()
        else:
            globalvar.set_value("system", manager)
    return manager

"""Order management: creation, assignment and completion."""

import logging

import globalvar
from order_model import Order
from path_planner import PathPlanner

logger = logging.getLogger(__name__)


def _rider_key(rider, order):
    active = len(rider.active_orders()) if hasattr(rider, "active_orders") \
        else len(rider.orders)
    dist = PathPlanner.euclidean((rider.x, rider.y), order.res_pos)
    return (active, dist, rider.rider_id)


def find_best_rider_for_order(order):
    """Return the best rider for ``order``.

    Riders with the fewest outstanding orders are preferred; ties are
    broken by the straight-line distance to the restaurant, then by id.
    If no riders exist, one is recruited (if funds allow).
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        from system import get_system
        manager = get_system()
        if manager.funds >= int(manager.config["recruit_threshold"]):
            manager.recruit_rider()
        riders = globalvar.get_value("riders") or []
        if not riders:
            return None
    return min(riders, key=lambda r: _rider_key(r, order))


class OrderManager:
    """Handle order creation, assignment and completion queries."""

    def __init__(self, system_manager):
        self.system = system_manager

    def create_order(self, rx, ry, cx, cy):
        """Create, register and assign an order.

        Returns ``(order, rider, recruited_riders)``.
        """
        order = Order((rx, ry), (cx, cy))
        self.system.orders.append(order)
        recruited = self.system.auto_recruit()
        rider = self.assign(order)
        return order, rider, recruited

    def assign(self, order):
        """Assign ``order`` to the best rider; return the rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            logger.warning("No rider available for order %s", order.id)
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider

    def assign_pending(self):
        """Try to assign any orders still pending; return assigned count."""
        count = 0
        for order in self.system.orders:
            if order.status == "pending":
                if self.assign(order) is not None:
                    count += 1
        return count

    def completed_orders(self):
        """Return all completed orders."""
        return [o for o in self.system.orders if o.status == "completed"]

"""Status display and ASCII map visualisation."""

from path_planner import PathPlanner, MAP_SIZE
from rider import STATE_NAMES

MAP_CHARS = 16
CELL = MAP_SIZE // MAP_CHARS  # 50 units per character


def _idx(value):
    return max(0, min(int(value) // CELL, MAP_CHARS - 1))


class DisplayManager:
    """Render system status, orders, rider logs and the ASCII map."""

    def __init__(self, system_manager):
        self.system = system_manager

    # ------------------------------------------------------------------
    @staticmethod
    def _base_grid():
        grid = []
        for row in range(MAP_CHARS):
            line = []
            for col in range(MAP_CHARS):
                line.append("#" if row % 2 == 0 or col % 2 == 0 else ".")
            grid.append(line)
        return grid

    @staticmethod
    def _draw_route(grid, points):
        for a, b in zip(points, points[1:]):
            dist = max(abs(b[0] - a[0]), abs(b[1] - a[1]))
            steps = max(1, dist // (CELL // 2))
            for i in range(steps + 1):
                x = a[0] + (b[0] - a[0]) * i / steps
                y = a[1] + (b[1] - a[1]) * i / steps
                grid[_idx(y)][_idx(x)] = "*"

    def render_map(self, route=None, orders=None, riders=None):
        """Return a 16x16 ASCII map as a string."""
        grid = self._base_grid()
        if route:
            self._draw_route(grid, route)
        if orders is None:
            orders = [o for o in self.system.orders
                      if o.status != "completed"]
        for order in orders:
            grid[_idx(order.res_pos[1])][_idx(order.res_pos[0])] = "S"
            grid[_idx(order.cus_pos[1])][_idx(order.cus_pos[0])] = "C"
        if riders is None:
            riders = self.system.riders
        for rider in riders:
            grid[_idx(rider.y)][_idx(rider.x)] = "R"
        header = "    " + "".join("%d" % (i % 10) for i in range(MAP_CHARS))
        lines = [header]
        for i, row in enumerate(grid):
            lines.append("%3d " % (i * CELL) + "".join(row))
        lines.append("Legend: R=Rider S=Restaurant C=Customer "
                     "#=Street .=Empty *=Route (1 char = %d units)" % CELL)
        return "\n".join(lines)

    def render_route(self, rider, order):
        """Render the planned route for ``rider`` delivering ``order``."""
        points = PathPlanner.plan_full_route(
            (rider.x, rider.y), order.res_pos, order.cus_pos)
        length = PathPlanner.route_length(points)
        est = -(-length // rider.speed) if rider.speed > 0 else 0
        lines = [
            "Route for rider %s: %s" % (
                rider.rider_id, " -> ".join("(%d,%d)" % p for p in points)),
            "Speed: %d units/tick, distance: %d units, "
            "estimated time: %d ticks" % (rider.speed, length, est),
            self.render_map(route=points, orders=[order], riders=[rider]),
        ]
        return "\n".join(lines)

    # ------------------------------------------------------------------
    def render_status(self):
        """Return a snapshot of funds and rider states."""
        lines = ["=== System Status (tick %d) ===" % self.system.time,
                 "Funds: %d" % self.system.funds,
                 "Riders: %d" % len(self.system.riders)]
        for r in self.system.riders:
            ids = ", ".join(
                "#%d(%s)" % (o.id, o.status) for o in r.orders) or "none"
            lines.append("  Rider %d at (%d,%d) state=%s orders=%d [%s]" % (
                r.rider_id, r.x, r.y, STATE_NAMES.get(r.state, "?"),
                len(r.orders), ids))
        lines.append(self.render_map())
        return "\n".join(lines)

    def render_orders(self):
        """Return details for all orders."""
        orders = self.system.orders
        if not orders:
            return "No orders."
        lines = ["%-4s %-6s %-12s %-12s %-8s %-6s %-6s %-11s" % (
            "ID", "Start", "Restaurant", "Customer", "Distance", "Time",
            "Rider", "Status")]
        for o in orders:
            taken = o.time_taken()
            lines.append("%-4d %-6d %-12s %-12s %-8d %-6s %-6s %-11s" % (
                o.id, o.created_time,
                "(%d,%d)" % tuple(o.res_pos), "(%d,%d)" % tuple(o.cus_pos),
                o.distance, "-" if taken is None else str(taken),
                "-" if o.rider_id is None else str(o.rider_id), o.status))
        return "\n".join(lines)

    def render_rider_logs(self):
        """Return every logged status change of every rider."""
        if not self.system.riders:
            return "No riders."
        lines = []
        for r in self.system.riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%-5s %-22s order=%-5s at (%s,%s)" % (
                    e.get("time"), e.get("event"),
                    "-" if e.get("order_id") is None else e.get("order_id"),
                    e.get("x"), e.get("y")))
        return "\n".join(lines)

"""Command parsing and input validation."""

import re

COMMANDS = ("start", "status", "orders", "riders", "tick", "save", "load",
            "map", "history", "help", "quit", "exit")

SHORTCUTS = {
    "s": "start", "n": "start", "new": "start",
    "st": "status", "o": "orders", "r": "riders",
    "t": "tick", "m": "map", "h": "help", "?": "help",
    "q": "quit", "hist": "history",
}

MIN_COORD = 0
MAX_COORD = 1000


class InputManager:
    """Parse commands and validate coordinate input."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        """Return ``(command, args)``; command is None for empty input."""
        line = (line or "").strip()
        if not line:
            return None, []
        self.history.append(line)
        parts = line.split(None, 1)
        cmd = parts[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        args = parts[1] if len(parts) > 1 else ""
        return cmd, args

    @staticmethod
    def parse_coordinates(text):
        """Parse four integer coordinates.

        Accepts "100 200 300 400", "100,200,300,400" or
        "[100, 200, 300, 400]".  Raises ValueError with a friendly
        message on invalid input.
        """
        cleaned = re.sub(r"[\[\]\(\),]", " ", text or "")
        tokens = cleaned.split()
        if len(tokens) != 4:
            raise ValueError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for tok in tokens:
            try:
                values.append(int(tok))
            except ValueError:
                raise ValueError("'%s' is not an integer" % tok) from None
        for v in values:
            if v < MIN_COORD or v > MAX_COORD:
                raise ValueError("Coordinate %d out of range (%d-%d)"
                                 % (v, MIN_COORD, MAX_COORD))
        return values

"""Command line entry point for the food delivery dispatch simulator."""

import argparse
import logging
import sys

from system import init_system, load_config
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager
from path_planner import to_grid_cell

try:  # command history support where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

HELP = """Commands:
  start  (s)  [rx ry cx cy]  create an order (restaurant + customer coords)
  status (st)               show funds, riders and map
  orders (o)                show all orders
  riders (r)                show rider status change logs
  tick   (t)  [n]           advance the simulation n ticks (default 1)
  map    (m)                show the map
  save <file> / load <file> persist / restore state as JSON
  history                   show command history
  help   (h)                show this help
  quit / exit (q)           leave the program
Each command loop iteration advances the simulation by one tick."""


def handle_start(args, system, orders, display, inputs):
    """Process the start command."""
    text = args
    if not text:
        text = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
    coords = inputs.parse_coordinates(text)
    before = len(system.riders)
    order, rider, recruited = orders.create_order(*coords)
    added = len(system.riders) > before
    print("Order #%d created." % order.id)
    print("Restaurant grid: cell (%d,%d) -> point %s" % (
        to_grid_cell(coords[0]), to_grid_cell(coords[1]), order.res_pos))
    print("Customer grid:   cell (%d,%d) -> point %s" % (
        to_grid_cell(coords[2]), to_grid_cell(coords[3]), order.cus_pos))
    print("New rider added: %s%s" % (
        "yes" if added else "no",
        (" (ids %s)" % ", ".join(str(r.rider_id) for r in system.riders[before:]))
        if added else ""))
    print("Funds: %d" % system.funds)
    if rider is None:
        print("No rider available; order remains pending.")
    else:
        print("Assigned to rider %d." % rider.rider_id)
        print(display.render_route(rider, order))


def main(argv=None):
    """Run the interactive command loop."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch")
    parser.add_argument("--config", default="config.json")
    parser.add_argument("--load", default=None)
    parser.add_argument("--debug", action="store_true")
    opts = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if opts.debug else logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")

    system = init_system(load_config(opts.config))
    if opts.load:
        try:
            system.load_state(opts.load)
        except (OSError, ValueError, KeyError) as exc:
            print("Could not load state: %s" % exc)
    orders = OrderManager(system)
    display = DisplayManager(system)
    inputs = InputManager()
    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")

    while system.running:
        try:
            line = input("[t=%d] > " % system.time)
        except (EOFError, KeyboardInterrupt):
            print()
            break
        try:
            cmd, args = inputs.parse_command(line)
            if cmd is None:
                pass
            elif cmd in ("quit", "exit"):
                system.stop()
                break
            elif cmd == "start":
                handle_start(args, system, orders, display, inputs)
            elif cmd == "status":
                print(display.render_status())
            elif cmd == "orders":
                print(display.render_orders())
            elif cmd == "riders":
                print(display.render_rider_logs())
            elif cmd == "map":
                print(display.render_map())
            elif cmd == "tick":
                n = int(args) if args.strip() else 1
                system.tick(max(0, n - 1))
            elif cmd == "save":
                system.save_state(args.strip() or "state.json")
                print("State saved.")
            elif cmd == "load":
                system.load_state(args.strip() or "state.json")
                print("State loaded.")
            elif cmd == "history":
                for i, h in enumerate(inputs.history, 1):
                    print("%3d  %s" % (i, h))
            elif cmd == "help":
                print(HELP)
            else:
                print("Unknown command '%s'. Type 'help'." % cmd)
        except ValueError as exc:
            print("Input error: %s" % exc)
        except OSError as exc:
            print("File error: %s" % exc)
        except Exception as exc:  # graceful recovery
            logging.exception("Unexpected error")
            print("Error: %s (system state recovered)" % exc)
            system.check_consistency()
        orders.assign_pending()
        system.tick(1)
    print("Goodbye.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

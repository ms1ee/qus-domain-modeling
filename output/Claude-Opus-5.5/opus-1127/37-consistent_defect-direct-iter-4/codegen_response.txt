import os
import re
from typing import Dict, List, Tuple

from lottery_system.models import User

_SEPARATOR = re.compile(r"[,;，；]")

DEFAULT_LIST_PATH = os.path.join(
    os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "data", "default_list.txt"
)

_EMBEDDED_DEFAULT = """Zhang San,Technology Department
Li Si,Technology Department
Wang Wu,Technology Department
Zhao Liu,Technology Department
Sun Qi,Product Department
Zhou Ba,Product Department
Wu Jiu,Product Department
Zheng Shi,Marketing Department
Qian Yi,Marketing Department
Feng Er,Marketing Department
Chen Mei,Human Resources Department
Chu Lan,Human Resources Department
Wei Zhu,Finance Department
Jiang Ju,Finance Department
Shen Hao,Cheng Ru Group
Han Xue,Cheng Ru Group
Yang Fan,Cheng Ru Group
Li Bin,Operations Department
Chen Qixian,Operations Department
Jin Duo,Operations Department
Ma Jiaqi,Operations Department
"""


class ListManager:
    """Loads, parses and validates participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        if os.path.isfile(DEFAULT_LIST_PATH):
            return self._load_from_path(DEFAULT_LIST_PATH)
        return self._parse_lines(_EMBEDDED_DEFAULT.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if path is None or not str(path).strip():
            return [], ["Error: No file path provided."]
        return self._load_from_path(str(path).strip())

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if not os.path.isfile(path):
            return [], [f"Error: File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"Error: Unable to read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"Error: Unable to read file {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in _SEPARATOR.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Manages prize library and round order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self.prizes}
        for name, qty in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize '{name}' already exists, skipped.")
                continue
            self.prizes.append(Prize(name, qty, [], None))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty.")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Prize quantity must be a positive integer.")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer.")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer: Optional[str] = None
        if associated_drawer is not None and str(associated_drawer).strip():
            drawer = str(associated_drawer).strip()
        prize = Prize(str(name).strip(), qty, groups, drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

import csv
import os
import re
from datetime import datetime
from typing import Dict, List

_LEVELS = [
    (r"\b(grand|special|super)\b", 0),
    (r"\bfirst\b", 1),
    (r"\bsecond\b", 2),
    (r"\bthird\b", 3),
    (r"\bfourth\b", 4),
    (r"\bfifth\b", 5),
    (r"\bsixth\b", 6),
    (r"\bconsolation\b|\bparticipation\b|\blucky\b", 50),
]


def prize_level(name: str) -> int:
    lower = name.lower()
    for pattern, level in _LEVELS:
        if re.search(pattern, lower):
            return level
    return 100


def sort_prizes_by_level(prizes) -> list:
    indexed = list(enumerate(prizes))
    indexed.sort(key=lambda t: (prize_level(t[1].name), t[0]))
    return [p for _, p in indexed]


class Winner:
    def __init__(self, user, prize, time=None, seq=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.seq = seq

    @property
    def prize_name(self):
        return self.prize.name if hasattr(self.prize, "name") else str(self.prize)

    def __repr__(self):
        return f"Winner({self.user!r}, {self.prize_name!r}, {self.time})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self._seq = 0

    def has_won(self, user) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def add_winner(self, user, prize):
        if self.has_won(user):
            return False
        self._seq += 1
        self.winners.append(Winner(user, prize, datetime.now(), self._seq))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes:
            result[p.name] = []
        for w in sorted(self.winners, key=lambda x: (x.time, x.seq)):
            result.setdefault(w.prize_name, []).append(w)
        return result

    def get_summary(self, prizes):
        """Ordered list of (prize_name, [Winner]) by prize level high->low."""
        by_prize = self.get_winners_by_prize(prizes)
        ordered = []
        for p in sort_prizes_by_level(prizes):
            ordered.append((p.name, by_prize.get(p.name, [])))
        known = {p.name for p in prizes}
        for name, ws in by_prize.items():
            if name not in known:
                ordered.append((name, ws))
        return ordered

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export."
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, ws in self.get_summary(prizes):
                    for w in ws:
                        writer.writerow([
                            w.user.name,
                            w.user.department,
                            prize_name,
                            w.time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User
from lottery_system.services.result_manager import ResultManager

SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"


def is_cheng_ru(department: str) -> bool:
    return "chengru" in department.lower().replace(" ", "").replace("_", "").replace("-", "")


def _norm(name: Optional[str]) -> str:
    return (name or "").strip().lower()


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryManager:
    def __init__(self, users, result_manager: Optional[ResultManager] = None):
        self.users: List[User] = list(users)
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self._rng = random.SystemRandom()

    # ---------- rules ----------
    def is_fallback_active(self, prize: Prize) -> bool:
        return (
            _norm(prize.associated_drawer) == _norm(PROTECTED_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = [u for u in self.users if u.name not in self.winner_names]
        if self.is_fallback_active(prize):
            cheng = [u for u in candidates if is_cheng_ru(u.department)]
            if cheng:
                return cheng
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return candidates
        avoid = {g.strip() for g in (prize.avoid_groups or [])}
        non_avoid = [u for u in candidates if u.department.strip() not in avoid]
        if non_avoid:
            return non_avoid
        # avoided members remain eligible, drawn after all others
        return [u for u in candidates if u.department.strip() in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        drawer = prize.associated_drawer
        if self.is_fallback_active(prize):
            parts.append(
                f"Cheng Ru group protection active: prize pool limited to Cheng Ru group members "
                f"(drawer {drawer})"
            )
        elif _norm(drawer) == _norm(SPECIAL_DRAWER):
            parts.append(f"Special drawer {drawer}: avoidance groups not applied")
        elif prize.avoid_groups:
            parts.append(
                "Avoidance groups (drawn only after all other members): "
                + ", ".join(prize.avoid_groups)
            )
        else:
            parts.append("No avoidance groups")
        if drawer:
            parts.append(f"Lottery person: {drawer}")
        return "; ".join(parts)

    # ---------- drawing ----------
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self._rng.choice(pool)
        self.winner_names.add(user.name)
        self.result_manager.add_winner(user, prize)
        return user

    def _update_tracking(self, prize: Prize, winners: List[User]):
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in WATCHED_DRAWERS}:
            self.last_two_draw_results.append(any(is_cheng_ru(u.department) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(PROTECTED_DRAWER):
            self.last_two_draw_results = []

    def run_round(self, prize: Prize, input_func: Callable = input,
                  output_func: Callable = print) -> List[User]:
        output_func("=" * 50)
        output_func(f"Current prize: {prize.name}")
        output_func(f"Quantity: {prize.quantity}")
        output_func(f"Rules: {self.describe_rules(prize)}")
        output_func("Commands: [Enter]=draw, 'pause', 'continue', 'terminate'")
        output_func("=" * 50)
        winners: List[User] = []
        terminated = False
        for i in range(1, prize.quantity + 1):
            while True:
                try:
                    cmd = input_func(f"Press Enter to draw the {ordinal(i)} winner > ").strip().lower()
                except EOFError:
                    cmd = ""
                if cmd in ("", "draw", "continue"):
                    break
                if cmd == "terminate":
                    terminated = True
                    break
                if cmd == "pause":
                    output_func("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
                    while True:
                        try:
                            c = input_func("(paused) > ").strip().lower()
                        except EOFError:
                            c = "terminate"
                        if c == "continue":
                            output_func("Lottery resumed.")
                            break
                        if c == "terminate":
                            terminated = True
                            break
                        output_func("Lottery is paused. Type 'continue' or 'terminate'.")
                    break
                output_func("Unknown command. Use Enter, 'pause', 'continue' or 'terminate'.")
            if terminated:
                output_func(f"Round '{prize.name}' terminated and skipped.")
                break
            output_func(f"Currently drawing the {ordinal(i)} winner...")
            user = self.draw_one(prize)
            if user is None:
                output_func("No eligible participants remain for this prize.")
                break
            winners.append(user)
            output_func(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        self._update_tracking(prize, winners)
        output_func("-" * 50)
        output_func(f"Round results for {prize.name}:")
        if winners:
            for n, u in enumerate(winners, 1):
                output_func(f"  {n}. {u.name} ({u.department})")
        else:
            output_func("  No winners in this round.")
        return winners

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.lottery = None

    def _set_users(self, users):
        if self.lottery and self.lottery.result_manager.winners:
            ans = input("Loading a new list resets lottery results. Continue? (y/n): ").strip().lower()
            if ans != "y":
                print("Load cancelled.")
                return
        self.users = users
        self.lottery = LotteryManager(users)
        print(f"{len(users)} participants loaded.")

    def _handle_loaded(self, users, errors):
        if errors:
            print("Problems detected:")
            for e in errors:
                print("  " + e)
        valid, dups = self.list_manager.validate_users(users)
        if dups:
            print("Duplicate names detected:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
        if errors or dups:
            while True:
                c = input("Type 'r' to remove invalid data and continue, 't' to terminate upload: ").strip().lower()
                if c == "r":
                    break
                if c == "t":
                    print("Upload terminated.")
                    return
        if not valid:
            print("No valid participants found.")
            return
        self._set_users(valid)

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._handle_loaded(users, errors)

    def load_custom(self):
        path = input("Enter txt file path: ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print(e)
            return
        self._handle_loaded(users, errors)

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    def load_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print(e)
        print("Preset prizes loaded.")

    def add_prize(self):
        name = input("Prize name: ").strip()
        qty = input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid = []
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            sel = input("Avoidance groups (numbers separated by commas, blank for none): ").strip()
            for s in sel.replace("，", ",").split(","):
                s = s.strip()
                if not s:
                    continue
                if s.isdigit() and 1 <= int(s) <= len(groups):
                    avoid.append(groups[int(s) - 1])
                else:
                    print(f"Ignored invalid selection: {s}")
        drawer = input("Associated lottery person (blank for none): ").strip() or None
        try:
            p = self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize '{p.name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def list_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            av = ", ".join(p.avoid_groups) or "None"
            print(f"  {i}. {p.name} x{p.quantity} | avoid: {av} | drawer: {p.associated_drawer or 'None'}")

    def reorder(self):
        self.list_prizes()
        if not self.prize_manager.get_prizes():
            return
        s = input("Enter new order as serial numbers (e.g. 3,1,2): ").strip()
        try:
            order = [int(x) for x in s.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.list_prizes()
        else:
            print("Invalid order.")

    def start(self):
        if not self.users or self.lottery is None:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        for p in prizes:
            self.lottery.run_round(p)
        print("All rounds completed.")
        self.summary()

    def summary(self):
        if self.lottery is None or not self.lottery.result_manager.winners:
            print("No winners yet.")
            return
        print("=== Complete Winner List ===")
        for name, ws in self.lottery.result_manager.get_summary(self.prize_manager.get_prizes()):
            if not ws:
                continue
            print(f"[{name}]")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {w.time.strftime('%Y-%m-%d %H:%M:%S')}")

    def export(self):
        if self.lottery is None:
            print("No winners to export.")
            return
        ok, msg = self.lottery.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_prizes, "5": self.add_prize, "6": self.list_prizes,
            "7": self.reorder, "8": self.start, "9": self.summary, "10": self.export,
        }
        while True:
            print("\n==== Team Building Smart Lottery ====")
            print("1. Load default list      2. Upload custom list")
            print("3. View groups            4. Load preset prizes")
            print("5. Add custom prize       6. View prizes")
            print("7. Reorder prizes         8. Start lottery")
            print("9. View summary          10. Export results")
            print("0. Exit")
            try:
                c = input("Choose: ").strip()
            except EOFError:
                break
            if c == "0":
                break
            act = actions.get(c)
            if act:
                try:
                    act()
                except EOFError:
                    break
            else:
                print("Invalid choice.")
        print("Goodbye!")


def main():
    LotteryApp().run()


if __name__ == "__main__":
    main()

"""Data loading utilities for the Beijing Subway fare calculation system.

The network data lives in ``subway_data.json`` next to this module.  Its
layout is::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str,
                      "distance": int, "line": str}]}
"""

import json
import os

# Absolute path of the bundled data file, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                              "subway_data.json")


class DataLoadError(Exception):
    """Raised when the subway data file cannot be read or is malformed."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON data file (defaults to DATA_FILE_PATH)
    :return: dict with "stations" and "connections" keys
    :raises DataLoadError: if the file is missing or not valid JSON
    """
    if path is None:
        path = DATA_FILE_PATH
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except FileNotFoundError as exc:
        raise DataLoadError("Data file not found: %s" % path) from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError("Data file is not valid JSON: %s (%s)" % (path, exc)) from exc
    except OSError as exc:
        raise DataLoadError("Cannot read data file %s: %s" % (path, exc)) from exc
    if not isinstance(data, dict):
        raise DataLoadError("Data file root must be a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file."""
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, ensure_ascii=False, indent=2)

"""Core data structures: Edge, Graph (weighted undirected) and MinPQ."""


class Edge:
    """A weighted, line-labelled edge between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        return self.source

    def other(self, v):
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError("Vertex %r is not an endpoint of this edge" % (v,))

    def reversed(self):
        """Return the same connection travelled in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.weight, self.line) == \
               (other.source, other.dest, other.weight, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.weight, self.line))

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.weight, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller.  ``adj`` (also exposed as
    ``graph``) maps a station number to the list of outgoing edges.
    """

    def __init__(self):
        self.adj = {}
        self.graph = self.adj
        self.station_lines = {}
        self.station_names = {}   # number -> name (optional, set by caller)
        self._edges = []

    # ---- nodes -------------------------------------------------------
    def add_vertex(self, v):
        self.adj.setdefault(v, [])

    def add_station(self, number, name, lines=None):
        self.add_vertex(number)
        self.station_names[number] = name
        if lines is not None:
            self.set_station_lines(name, lines)

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def lines_of(self, number):
        name = self.station_names.get(number)
        return list(self.station_lines.get(name, [])) if name is not None else []

    def vertices(self):
        return list(self.adj.keys())

    def V(self):
        return len(self.adj)

    def has_vertex(self, v):
        return v in self.adj

    # ---- edges -------------------------------------------------------
    def add_edge(self, edge):
        if edge.weight is None or edge.weight < 0:
            raise ValueError("Edge distance must be a non-negative number")
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, []).append(edge.reversed())
        self._edges.append(edge)

    def adjacent(self, v):
        return self.adj.get(v, [])

    def edges(self):
        return list(self._edges)

    def E(self):
        return len(self._edges)


class MinPQ:
    """Binary-heap based min priority queue of (key, value) pairs.

    Ties on key are broken by insertion order so values never need to be
    comparable.
    """

    def __init__(self):
        self._heap = []
        self._counter = 0

    def __len__(self):
        return len(self._heap)

    def is_empty(self):
        return not self._heap

    def push(self, key, value):
        self._heap.append((key, self._counter, value))
        self._counter += 1
        self._swim(len(self._heap) - 1)

    insert = push

    def peek(self):
        if not self._heap:
            raise IndexError("peek from empty priority queue")
        key, _, value = self._heap[0]
        return key, value

    def pop(self):
        if not self._heap:
            raise IndexError("pop from empty priority queue")
        heap = self._heap
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sink(0)
        return top[0], top[2]

    del_min = pop

    def _less(self, i, j):
        a, b = self._heap[i], self._heap[j]
        return (a[0], a[1]) < (b[0], b[1])

    def _swim(self, k):
        heap = self._heap
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                heap[k], heap[parent] = heap[parent], heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        heap = self._heap
        n = len(heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            if left + 1 < n and self._less(left + 1, left):
                child = left + 1
            if self._less(child, k):
                heap[k], heap[child] = heap[child], heap[k]
                k = child
            else:
                break

"""Dijkstra-based path finding with several query modes.

Modes:
  * shortest_distance - minimise total distance
  * least_transfer    - minimise transfers, then distance
  * comprehensive     - minimise distance + transfers * TRANSFER_PENALTY
Optional constraints: maximum number of transfers and preferred lines.
"""

from models import MinPQ

INF = float("inf")

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance (meters) of one transfer in comprehensive mode;
# models walking distance/time spent changing lines.
TRANSFER_PENALTY = 3000
# Cost multiplier applied to edges on non-preferred lines.
NON_PREFERRED_FACTOR = 1.3


class PathfindingConfig:
    """Configuration for a path query."""

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None, transfer_penalty=TRANSFER_PENALTY,
                 non_preferred_factor=NON_PREFERRED_FACTOR):
        if mode not in MODES:
            raise ValueError("Unknown mode %r, expected one of %s" % (mode, ", ".join(MODES)))
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []
        self.transfer_penalty = transfer_penalty
        self.non_preferred_factor = non_preferred_factor


def count_transfers(edges):
    """Number of line changes along a sequence of edges."""
    count = 0
    for prev, cur in zip(edges, edges[1:]):
        if prev.line != cur.line:
            count += 1
    return count


class ShortestPath:
    """Single-source path computation using Dijkstra with a MinPQ."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: INF for v in graph.vertices()}
        self.dist_to[source] = 0
        self._edge_to = {}        # plain mode: node -> Edge
        self._final_state = {}    # state mode: node -> state
        self._prev = {}           # state mode: state -> (prev_state, Edge)
        cfg = self.config
        self._state_mode = not (cfg.mode == MODE_SHORTEST_DISTANCE
                                and cfg.max_transfers is None
                                and not cfg.preferred_lines)
        if self._state_mode:
            self._run_state_dijkstra()
        else:
            self._run_plain_dijkstra()

    # ------------------------------------------------------------------
    def _run_plain_dijkstra(self):
        dist = self.dist_to
        pq = MinPQ()
        pq.push(0, self.source)
        while not pq.is_empty():
            d, v = pq.pop()
            if d > dist.get(v, INF):
                continue
            for e in self.graph.adjacent(v):
                nd = d + e.weight
                if nd < dist.get(e.dest, INF):
                    dist[e.dest] = nd
                    self._edge_to[e.dest] = e
                    pq.push(nd, e.dest)

    def _edge_factor(self, line):
        prefs = self.config.preferred_lines
        if prefs and line not in prefs:
            return self.config.non_preferred_factor
        return 1.0

    def _cost(self, weighted, transfers):
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, weighted)
        if mode == MODE_COMPREHENSIVE:
            return (weighted + transfers * self.config.transfer_penalty, transfers)
        return (weighted, transfers)

    def _run_state_dijkstra(self):
        cfg = self.config
        track = cfg.max_transfers is not None
        start = (self.source, None, 0) if track else (self.source, None)
        best = {start: self._cost(0, 0)}
        real_dist = {start: 0}
        weighted = {start: 0.0}
        transfers = {start: 0}
        self._prev[start] = (None, None)
        settled = set()
        pq = MinPQ()
        pq.push(best[start], start)
        while not pq.is_empty():
            cost, state = pq.pop()
            if state in settled or cost != best.get(state):
                continue
            settled.add(state)
            v, line = state[0], state[1]
            d, w, tr = real_dist[state], weighted[state], transfers[state]
            if v not in self._final_state:
                self._final_state[v] = state
                self.dist_to[v] = d
            for e in self.graph.adjacent(v):
                ntr = tr + (1 if line is not None and e.line != line else 0)
                if track and ntr > cfg.max_transfers:
                    continue
                nw = w + e.weight * self._edge_factor(e.line)
                ncost = self._cost(nw, ntr)
                nstate = (e.dest, e.line, ntr) if track else (e.dest, e.line)
                if nstate in settled:
                    continue
                old = best.get(nstate)
                if old is None or ncost < old:
                    best[nstate] = ncost
                    real_dist[nstate] = d + e.weight
                    weighted[nstate] = nw
                    transfers[nstate] = ntr
                    self._prev[nstate] = (state, e)
                    pq.push(ncost, nstate)

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return self.dist_to.get(target, INF) < INF

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return (station numbers from source to target, edges travelled).

        Returns ([], []) when the target is unreachable.
        """
        if not self.has_path_to(target):
            return [], []
        if target == self.source:
            return [self.source], []
        edges = []
        if self._state_mode:
            state = self._final_state[target]
            while True:
                prev_state, e = self._prev[state]
                if e is None:
                    break
                edges.append(e)
                state = prev_state
        else:
            v = target
            while v != self.source:
                e = self._edge_to[v]
                edges.append(e)
                v = e.source
        edges.reverse()
        path = [self.source] + [e.dest for e in edges]
        return path, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return count_transfers(edges)

"""Beijing Subway fare calculation.

Tiers are left-closed / right-open intervals in meters:
  [0, 6000) -> 3, [6000, 12000) -> 4, [12000, 22000) -> 5,
  [22000, 32000) -> 6, then +1 yuan per started 20 km above 32 km
  ([32000, 52000) -> 7, [52000, 72000) -> 8, ...).
Fractional distances are rounded up to the next whole meter.
"""

import math

# (upper bound in meters exclusive, fare in yuan)
FARE_STRUCTURE = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_START = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP = 20000

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5


class FareCalculationError(ValueError):
    """Raised when a fare cannot be computed for the given distance."""


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise FareCalculationError("Distance must be a number, got %r" % (distance,))
    if math.isnan(distance) or math.isinf(distance):
        raise FareCalculationError("Distance must be finite")
    if distance < 0:
        raise FareCalculationError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare (yuan) for a travel distance given in meters."""
    _validate_distance(distance)
    meters = int(math.ceil(distance))
    for upper, fare in FARE_STRUCTURE:
        if meters < upper:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_START) // EXTENDED_STEP


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes (1.5 min/km + 5 min/transfer)."""
    _validate_distance(distance)
    if transfer_count is None or transfer_count < 0:
        raise ValueError("transfer_count must be non-negative")
    return distance / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""Builds the subway network graph from raw data and answers route queries."""

import re

from fare_calculator import Sum_money, calculate_estimated_time
from models import Edge, Graph
from pathfinding import PathfindingConfig, ShortestPath, count_transfers


class StationNotFoundError(KeyError):
    pass


class PathNotFoundError(Exception):
    pass


def _norm(name):
    s = str(name).strip().lower()
    s = re.sub(r"\s+", " ", s)
    return s


def _loose(name):
    s = _norm(name)
    for suffix in (" station", "站"):
        if s.endswith(suffix) and len(s) > len(suffix):
            s = s[: -len(suffix)]
    return re.sub(r"[\s'’\-]", "", s)


def build_station_to_number(data):
    """Map every station name (stations + connection endpoints) to a number."""
    mapping = {}
    for st in data.get("stations", []):
        name = st.get("name")
        if name is not None and name not in mapping:
            mapping[name] = len(mapping)
    for c in data.get("connections", []):
        for key in ("source", "dest"):
            name = c.get(key)
            if name is not None and name not in mapping:
                mapping[name] = len(mapping)
    return mapping


class SubwayNetwork:
    """Subway network: station mapping, Graph object and adjacency dict."""

    def __init__(self, data):
        self.data = data
        self.station_to_number = build_station_to_number(data)
        self.number_to_station = {n: s for s, n in self.station_to_number.items()}
        self.graph = Graph()
        # graph dict: station name -> list of (neighbor name, distance, line)
        self.graph_dict = {name: [] for name in self.station_to_number}
        lines_map = {}
        for st in data.get("stations", []):
            lines_map.setdefault(st["name"], [])
            for ln in st.get("lines", []) or []:
                if ln not in lines_map[st["name"]]:
                    lines_map[st["name"]].append(ln)
        for c in data.get("connections", []):
            for key in ("source", "dest"):
                lst = lines_map.setdefault(c["source"] if key == "source" else c["dest"], [])
                if c.get("line") and c["line"] not in lst:
                    lst.append(c["line"])
        for name, num in self.station_to_number.items():
            self.graph.add_station(num, name, lines_map.get(name, []))
        for c in data.get("connections", []):
            try:
                s, d = c["source"], c["dest"]
                dist = c["distance"]
            except KeyError:
                continue
            if s == d or not isinstance(dist, (int, float)) or dist < 0:
                continue
            self.graph.add_edge(Edge(self.station_to_number[s], self.station_to_number[d],
                                     dist, c.get("line")))
            self.graph_dict[s].append((d, dist, c.get("line")))
            self.graph_dict[d].append((s, dist, c.get("line")))
        self._exact = {_norm(n): n for n in self.station_to_number}
        self._loose = {}
        for n in self.station_to_number:
            self._loose.setdefault(_loose(n), n)

    # ------------------------------------------------------------------
    @property
    def station_lines(self):
        return self.graph.station_lines

    def all_lines(self):
        lines = []
        for lst in self.graph.station_lines.values():
            for ln in lst:
                if ln not in lines:
                    lines.append(ln)
        return lines

    def resolve_station(self, name):
        """Standardize a user-entered station name; None if unknown."""
        if name is None:
            return None
        raw = str(name).strip()
        if raw in self.station_to_number:
            return raw
        n = _norm(raw)
        if n in self._exact:
            return self._exact[n]
        return self._loose.get(_loose(raw))

    def query(self, start, end, config=None):
        """Compute a route between two (canonical) station names.

        :return: dict describing the route
        """
        if start not in self.station_to_number:
            raise StationNotFoundError(start)
        if end not in self.station_to_number:
            raise StationNotFoundError(end)
        config = config or PathfindingConfig()
        src = self.station_to_number[start]
        dst = self.station_to_number[end]
        sp = ShortestPath(self.graph, src, config)
        path, edges = sp.path_with_edges(dst)
        if not path:
            msg = "No route found from '%s' to '%s'" % (start, end)
            if config.max_transfers is not None:
                msg += " within %d transfer(s)" % config.max_transfers
            raise PathNotFoundError(msg)
        names = [self.number_to_station[v] for v in path]
        segments = []
        for i, e in enumerate(edges):
            if not segments or segments[-1]["line"] != e.line:
                segments.append({"line": e.line, "stations": [names[i]], "distance": 0})
            segments[-1]["stations"].append(names[i + 1])
            segments[-1]["distance"] += e.weight
        lines = []
        for seg in segments:
            if seg["line"] not in lines:
                lines.append(seg["line"])
        transfer_stations = [seg["stations"][0] for seg in segments[1:]]
        distance = sp.dist_to[dst]
        transfers = count_transfers(edges)
        fare = Sum_money(distance)
        est = calculate_estimated_time(distance, transfers)
        return {
            "start": start,
            "end": end,
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "preferred_lines": list(config.preferred_lines),
            "path": names,
            "distance": distance,
            "transfer_count": transfers,
            "transfer_stations": transfer_stations,
            "lines": lines,
            "segments": segments,
            "fare": fare,
            "estimated_time": est,
        }

"""Route visualization and export (console text and JSON)."""

import json


def format_distance(meters):
    """Meters for short distances (<1 km), kilometers otherwise."""
    if meters < 1000:
        return "%d meters" % round(meters)
    return "%.2f km" % (meters / 1000.0)


def format_fare(fare):
    return "Fare: %d yuan" % fare


def build_summary(route):
    return {
        "total_distance_km": round(route["distance"] / 1000.0, 3),
        "total_fare": route["fare"],
        "transfer_count": route["transfer_count"],
        "estimated_time_minutes": round(route["estimated_time"], 1),
    }


def to_json_dict(route):
    return {
        "query": {
            "start": route["start"],
            "end": route["end"],
            "mode": route["mode"],
            "max_transfers": route.get("max_transfers"),
            "line_preference": route.get("preferred_lines", []),
        },
        "path": route["path"],
        "station_count": len(route["path"]),
        "distance": {
            "meters": route["distance"],
            "kilometers": round(route["distance"] / 1000.0, 3),
        },
        "fare": route["fare"],
        "estimated_time": round(route["estimated_time"], 1),
        "transfer_count": route["transfer_count"],
        "transfer_stations": route["transfer_stations"],
        "lines": route["lines"],
        "segments": [
            {"line": s["line"], "stations": s["stations"], "distance": s["distance"]}
            for s in route["segments"]
        ],
        "summary": build_summary(route),
    }


def format_json(route):
    return json.dumps(to_json_dict(route), ensure_ascii=False, indent=2)


def format_text(route):
    lines = []
    lines.append("=" * 50)
    lines.append("Route: %s -> %s" % (route["start"], route["end"]))
    lines.append("Mode: %s" % route["mode"])
    lines.append("-" * 50)
    lines.append("Stations (%d): %s" % (len(route["path"]), " -> ".join(route["path"])))
    lines.append("Segments:")
    segs = route["segments"]
    for i, seg in enumerate(segs):
        lines.append("  [%s] %s (%d stops, %s)" % (
            seg["line"], " -> ".join(seg["stations"]),
            len(seg["stations"]) - 1, format_distance(seg["distance"])))
        if i + 1 < len(segs):
            lines.append("  >> Transfer at %s: %s -> %s" % (
                segs[i + 1]["stations"][0], seg["line"], segs[i + 1]["line"]))
    d = route["distance"]
    lines.append("Total distance: %s (%d meters, %.2f km)" % (
        format_distance(d), round(d), d / 1000.0))
    lines.append("Transfers: %d" % route["transfer_count"])
    if route["transfer_stations"]:
        lines.append("Transfer stations: %s" % ", ".join(route["transfer_stations"]))
    lines.append("Lines involved: %s" % ", ".join(str(x) for x in route["lines"]))
    lines.append("Estimated time: %.1f minutes" % route["estimated_time"])
    lines.append(format_fare(route["fare"]))
    lines.append("=" * 50)
    return "\n".join(lines)


def render(route, fmt="text"):
    return format_json(route) if fmt == "json" else format_text(route)

def _loose(s):
    s = str(s).strip().lower()
    for suffix in (" station", "站"):
        if s.endswith(suffix) and len(s) > len(suffix):
            s = s[: -len(suffix)]
    return re.sub(r"[\s'’\-]", "", s)


class SubwayDataManager:
    """Wraps raw subway data with check and update operations."""

    def __init__(self, data):
        self.data = copy.deepcopy(data)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ---- lookup helpers ---------------------------------------------
    def _station(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def _ensure_station(self, name, line):
        st = self._station(name)
        if st is None:
            st = {"name": name, "lines": []}
            self.data["stations"].append(st)
        st.setdefault("lines", [])
        if line and line not in st["lines"]:
            st["lines"].append(line)
        return st

    def line_names(self):
        names = []
        for st in self.data["stations"]:
            for ln in st.get("lines", []) or []:
                if ln not in names:
                    names.append(ln)
        for c in self.data["connections"]:
            if c.get("line") and c["line"] not in names:
                names.append(c["line"])
        return names

    def get_line_stations(self, line):
        """Ordered station sequence of a line, derived from connections."""
        adj = {}
        for c in self.data["connections"]:
            if c.get("line") == line:
                adj.setdefault(c["source"], []).append(c["dest"])
                adj.setdefault(c["dest"], []).append(c["source"])
        if not adj:
            return []
        start = next((s for s, n in adj.items() if len(set(n)) == 1), next(iter(adj)))
        order, seen, cur = [start], {start}, start
        while True:
            nxt = next((n for n in adj[cur] if n not in seen), None)
            if nxt is None:
                break
            order.append(nxt)
            seen.add(nxt)
            cur = nxt
        for s in adj:  # remaining stations of branched/loop lines
            if s not in seen:
                order.append(s)
                seen.add(s)
        return order

    def get_transfer_stations(self):
        return sorted(st["name"] for st in self.data["stations"]
                      if len(st.get("lines", []) or []) > 1)

    # ---- integrity check --------------------------------------------
    def check_integrity(self):
        errors, warnings = [], []
        stations = self.data["stations"]
        names = set()
        for i, st in enumerate(stations):
            name = st.get("name")
            if not name:
                errors.append("Station #%d has no name" % i)
                continue
            if name in names:
                errors.append("Duplicate station name: %s" % name)
            names.add(name)
            if not st.get("lines"):
                warnings.append("Station '%s' has no line information" % name)

        # Coverage
        loose_lines = {_loose(l) for l in self.line_names()}
        missing_lines = []
        for canonical, aliases in MAIN_LINES.items():
            if not any(_loose(a) in loose_lines for a in aliases + [canonical]):
                missing_lines.append(canonical)
        for ml in missing_lines:
            warnings.append("Main line not found in data: %s" % ml)
        loose_names = {_loose(n) for n in names}
        found_rep = 0
        missing_rep = []
        for candidates in REPRESENTATIVE_STATIONS:
            if any(_loose(c) in loose_names for c in candidates):
                found_rep += 1
            else:
                missing_rep.append(candidates[0])
        if missing_rep:
            warnings.append("Representative stations not found: %s" % ", ".join(missing_rep))

        # Connections
        seen_pairs = {}
        connected = set()
        adj = {}
        bad_distance = 0
        for i, c in enumerate(self.data["connections"]):
            s, d, dist, line = c.get("source"), c.get("dest"), c.get("distance"), c.get("line")
            if s is None or d is None:
                errors.append("Connection #%d missing source/dest" % i)
                continue
            if s not in names:
                errors.append("Connection #%d references unknown station '%s'" % (i, s))
            if d not in names:
                errors.append("Connection #%d references unknown station '%s'" % (i, d))
            if s == d:
                errors.append("Connection #%d is a self loop at '%s'" % (i, s))
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                errors.append("Connection %s - %s has invalid distance %r" % (s, d, dist))
                bad_distance += 1
            elif not (MIN_REASONABLE_DISTANCE <= dist <= MAX_REASONABLE_DISTANCE):
                warnings.append("Connection %s - %s has unusual distance %s m" % (s, d, dist))
            if not line:
                warnings.append("Connection %s - %s has no line" % (s, d))
            else:
                for n in (s, d):
                    st = self._station(n)
                    if st is not None and line not in (st.get("lines") or []):
                        warnings.append("Station '%s' is on connection of '%s' but does not "
                                        "list that line" % (n, line))
            key = (frozenset((s, d)), line)
            if key in seen_pairs and seen_pairs[key] != dist:
                errors.append("Inconsistent distances for %s - %s on %s: %s vs %s"
                              % (s, d, line, seen_pairs[key], dist))
            seen_pairs[key] = dist
            connected.update((s, d))
            adj.setdefault(s, set()).add(d)
            adj.setdefault(d, set()).add(s)

        isolated = sorted(names - connected)
        for n in isolated:
            warnings.append("Station '%s' has no connections" % n)
        components = 0
        visited = set()
        for n in adj:
            if n in visited:
                continue
            components += 1
            stack = [n]
            visited.add(n)
            while stack:
                v = stack.pop()
                for w in adj[v]:
                    if w not in visited:
                        visited.add(w)
                        stack.append(w)
        if components > 1:
            warnings.append("Network is split into %d disconnected components" % components)

        return {
            "ok": not errors,
            "station_count": len(names),
            "connection_count": len(self.data["connections"]),
            "line_count": len(self.line_names()),
            "transfer_station_count": len(self.get_transfer_stations()),
            "representative_coverage": "%d/%d" % (found_rep, len(REPRESENTATIVE_STATIONS)),
            "missing_main_lines": missing_lines,
            "components": components,
            "errors": errors,
            "warnings": warnings,
        }

    # ---- updates ----------------------------------------------------
    def add_line(self, line_name, stations, distances):
        """Add a new line given station sequence and inter-station distances."""
        if len(stations) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(stations) - 1:
            raise ValueError("Need exactly len(stations) - 1 distances")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for name in stations:
            self._ensure_station(name, line_name)
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": int(d), "line": line_name})

    def modify_station(self, name, new_name=None, lines=None):
        st = self._station(name)
        if st is None:
            raise KeyError("Station '%s' does not exist" % name)
        if new_name and new_name != name:
            if self._station(new_name) is not None:
                raise ValueError("Station '%s' already exists" % new_name)
            st["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            st["lines"] = list(lines)

    def update_distance(self, source, dest, distance, line=None):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        updated = 0
        for c in self.data["connections"]:
            if {c.get("source"), c.get("dest")} == {source, dest} and \
                    (line is None or c.get("line") == line):
                c["distance"] = int(distance)
                updated += 1
        if not updated:
            raise KeyError("No connection between '%s' and '%s'" % (source, dest))
        return updated

    def save(self, path=DATA_FILE_PATH):
        save_subway_data(self.data, path)


def format_integrity_report(report):
    out = ["Data integrity check",
           "Stations: %d" % report["station_count"],
           "Connections: %d" % report["connection_count"],
           "Lines: %d" % report["line_count"],
           "Transfer stations: %d" % report["transfer_station_count"],
           "Representative station coverage: %s" % report["representative_coverage"],
           "Connected components: %d" % report["components"]]
    if report["errors"]:
        out.append("Errors (%d):" % len(report["errors"]))
        out.extend("  - " + e for e in report["errors"])
    if report["warnings"]:
        out.append("Warnings (%d):" % len(report["warnings"]))
        out.extend("  - " + w for w in report["warnings"])
    out.append("Result: %s" % ("PASSED" if report["ok"] else "FAILED"))
    return "\n".join(out)

"""Command-line interface for the Beijing Subway fare calculation system.

Usage:
    python main.py [--mode MODE] [--format text|json] [--output_file PATH]
                   [--max_transfers N] [--line_preference "Line 1,Line 2"]
                   [--check_data]

Enter queries as "Starting Station; Ending Station". Type 'q' to quit.

import argparse

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
QUIT_WORDS = {"q", "quit", "exit"}


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", default="text", choices=["text", "json"], help="Output format")
    p.add_argument("--mode", default="shortest_distance", choices=list(MODES),
                   help="Path query mode")
    p.add_argument("--max_transfers", type=int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="Preferred line(s), comma separated (e.g. 'Line 1')")
    p.add_argument("--check_data", action="store_true",
                   help="Run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    args = p.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        p.error("--max_transfers must be non-negative")
    return args


def parse_query(line):
    """Split 'A; B' into (A, B); raise ValueError on bad format."""
    text = line.replace("；", ";").strip()
    parts = text.split(";")
    if len(parts) != 2:
        raise ValueError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise ValueError(FORMAT_ERROR)
    return start, end


def process_query(network, line, config, fmt="text"):
    """Process one input line. Returns (success, output text, route or None)."""
    try:
        start_raw, end_raw = parse_query(line)
    except ValueError as exc:
        return False, str(exc), None
    start = network.resolve_station(start_raw)
    if start is None:
        return False, "Starting station '%s' does not exist, please check the station name" \
            % start_raw, None
    end = network.resolve_station(end_raw)
    if end is None:
        return False, "Ending station '%s' does not exist, please check the station name" \
            % end_raw, None
    if start == end:
        return False, "Starting station cannot be the same as the ending station", None
    try:
        route = network.query(start, end, config)
    except PathNotFoundError as exc:
        return False, "Path calculation error: %s" % exc, None
    except FareCalculationError as exc:
        return False, "Fare calculation error: %s" % exc, None
    except Exception as exc:  # pragma: no cover - defensive
        return False, "Path calculation error: %s" % exc, None
    try:
        output = render(route, fmt)
    except Exception as exc:  # pragma: no cover - defensive
        return False, "Fare calculation error: %s" % exc, None
    return True, output, route


class ResultWriter:
    """Exports query results to a file (text appended, JSON kept valid)."""

    def __init__(self, path, fmt):
        self.path = path
        self.fmt = fmt
        self.results = []
        self.initialized = False

    def write(self, route, text):
        if not self.path:
            return
        try:
            if self.fmt == "json":
                self.results.append(to_json_dict(route))
                payload = self.results[0] if len(self.results) == 1 else self.results
                with open(self.path, "w", encoding="utf-8") as fh:
                    json.dump(payload, fh, ensure_ascii=False, indent=2)
            else:
                mode = "a" if self.initialized else "w"
                with open(self.path, mode, encoding="utf-8") as fh:
                    fh.write(text + "\n")
            self.initialized = True
        except OSError as exc:
            print("Failed to write output file: %s" % exc)


def main(argv=None):
    args = parse_args(argv)
    try:
        data = load_subway_data(args.data_file)
    except DataLoadError as exc:
        print("Error: %s" % exc)
        return 1

    if args.check_data:
        report = SubwayDataManager(data).check_integrity()
        print(format_integrity_report(report))
        return 0 if report["ok"] else 1

    network = SubwayNetwork(data)
    preferred = []
    if args.line_preference:
        preferred = [s.strip() for s in args.line_preference.replace("，", ",").split(",")
                     if s.strip()]
        unknown = [l for l in preferred if l not in network.all_lines()]
        if unknown:
            print("Warning: unknown line(s) in preference: %s" % ", ".join(unknown))
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                               preferred_lines=preferred)
    writer = ResultWriter(args.output_file, args.format)

    print("Beijing Subway Fare Calculator (%d stations loaded)" % len(network.station_to_number))
    print("Enter 'Starting Station; Ending Station' to query, 'q' to quit.")
    while True:
        try:
            print("> ", end="", flush=True)
            line = sys.stdin.readline()
        except KeyboardInterrupt:
            print()
            break
        if not line:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in QUIT_WORDS:
            break
        ok, output, route = process_query(network, line, config, args.format)
        print(output)
        if ok:
            writer.write(route, output)
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

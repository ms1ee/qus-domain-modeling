"""Grid based path planning for riders."""

MAP_MIN = 0
MAP_MAX = 800
CELL = 100


def clamp(value, low=MAP_MIN, high=MAP_MAX):
    """Clamp ``value`` to [low, high]."""
    return max(low, min(high, value))


def to_grid_center(value):
    """Convert a raw coordinate (0-800) to the centre of its grid cell.

    ``center = (v // 100) * 100 + 50``; 800 is mapped into the last cell.
    """
    value = clamp(int(value))
    cell = min(value // CELL, (MAP_MAX // CELL) - 1)
    return cell * CELL + CELL // 2


class PathPlanner:
    """Grid based path finding (L-shaped routes, horizontal first)."""

    @staticmethod
    def manhattan(a, b):
        """Manhattan distance between two points."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def euclidean(a, b):
        """Euclidean distance between two points."""
        return ((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2) ** 0.5

    @staticmethod
    def is_vertical_special_case(start, target):
        """True when the target is directly above or below the start."""
        return start[0] == target[0] and start[1] != target[1]

    @staticmethod
    def compute_path(start, target):
        """Return list of waypoints from ``start`` to ``target``.

        Movement is horizontal first, then vertical (right-angle turn).
        When the target is directly above/below only a vertical segment
        is produced.
        """
        start = (int(start[0]), int(start[1]))
        target = (int(target[0]), int(target[1]))
        path = [start]
        if start == target:
            return path
        if start[0] != target[0] and start[1] != target[1]:
            path.append((target[0], start[1]))
        path.append(target)
        return path

    @classmethod
    def compute_multi_path(cls, points):
        """Continuous path through several destinations."""
        if not points:
            return []
        path = [tuple(points[0])]
        for nxt in points[1:]:
            segment = cls.compute_path(path[-1], nxt)
            path.extend(segment[1:])
        return path

    @classmethod
    def path_length(cls, path):
        """Total length of a waypoint path."""
        return sum(cls.manhattan(path[i], path[i + 1])
                   for i in range(len(path) - 1))

    @staticmethod
    def step(pos, target, speed):
        """Advance from ``pos`` towards ``target`` by at most ``speed``.

        Horizontal movement is consumed first, then vertical.  Never
        overshoots the target.  Returns the new (x, y).
        """
        x, y = int(pos[0]), int(pos[1])
        tx, ty = int(target[0]), int(target[1])
        budget = max(0, int(speed))
        dx = tx - x
        if dx != 0 and budget > 0:
            move = min(abs(dx), budget)
            x += move if dx > 0 else -move
            budget -= move
        dy = ty - y
        if dy != 0 and budget > 0:
            move = min(abs(dy), budget)
            y += move if dy > 0 else -move
        return clamp(x), clamp(y)

"""Order data model."""

import globalvar
from path_planner import to_grid_center, PathPlanner


def _now():
    t = globalvar.get_value("time")
    return t if isinstance(t, int) else 0


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique order id.
        res_pos: (x, y) restaurant location (grid cell centre).
        cus_pos: (x, y) customer location (grid cell centre).
        status: "pending", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: id of assigned rider or None.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.raw_res = (int(res_pos[0]), int(res_pos[1]))
        self.raw_cus = (int(cus_pos[0]), int(cus_pos[1]))
        self.res_pos = (to_grid_center(res_pos[0]), to_grid_center(res_pos[1]))
        self.cus_pos = (to_grid_center(cus_pos[0]), to_grid_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = _now()
        self.assign_time = None
        self.pickup_time = None
        self.complete_time = None
        self.distance = PathPlanner.manhattan(self.res_pos, self.cus_pos)

    def assign_to(self, rider_id):
        """Record assignment of this order to rider ``rider_id``."""
        self.rider_id = rider_id
        self.assign_time = _now()
        if self.status == "pending":
            self.status = "assigned"

    @property
    def time_taken(self):
        """Ticks from creation to completion (None if not completed)."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.start_time

    def to_dict(self):
        """Serialise to a JSON friendly dict."""
        return {
            "id": self.id,
            "raw_res": list(self.raw_res),
            "raw_cus": list(self.raw_cus),
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "assign_time": self.assign_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from ``to_dict`` output."""
        order = cls(data["res_pos"], data["cus_pos"])
        order.id = data["id"]
        order.raw_res = tuple(data.get("raw_res", data["res_pos"]))
        order.raw_cus = tuple(data.get("raw_cus", data["cus_pos"]))
        order.res_pos = tuple(data["res_pos"])
        order.cus_pos = tuple(data["cus_pos"])
        order.status = data["status"]
        order.rider_id = data["rider_id"]
        order.start_time = data["start_time"]
        order.assign_time = data.get("assign_time")
        order.pickup_time = data.get("pickup_time")
        order.complete_time = data.get("complete_time")
        order.distance = data.get("distance", order.distance)
        cls._next_id = max(cls._next_id, order.id + 1)
        return order

    def __repr__(self):
        return "Order(id=%d, res=%s, cus=%s, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

"""Rider model: movement, state and orders."""

import logging

import globalvar
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

START_X = 435
START_Y = 385
DEFAULT_SPEED = 10


def _now():
    t = globalvar.get_value("time")
    return t if isinstance(t, int) else 0


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``; each call
    to :meth:`move` advances one tick and performs state transitions.
    """

    def __init__(self, rider_id, x=START_X, y=START_Y, speed=DEFAULT_SPEED):
        self.rider_id = rider_id
        self.x = x
        self.y = y
        self.speed = speed
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.completed_orders = []
        self.total_distance = 0
        self.log = []
        self._log_event("spawn", None)

    # ------------------------------------------------------------------
    def _log_event(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": _now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })
        logger.debug("rider %s %s order=%s at (%s,%s)",
                     self.rider_id, event, order_id, self.x, self.y)

    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def position(self):
        """Current (x, y)."""
        return (self.x, self.y)

    @property
    def current_order(self):
        """Order currently being handled (or None)."""
        return self.orders[0] if self.orders else None

    def _step_toward(self, target):
        self.special_case = PathPlanner.is_vertical_special_case(
            self.position, target)
        nx, ny = PathPlanner.step(self.position, target, self.speed)
        self.total_distance += abs(nx - self.x) + abs(ny - self.y)
        self.x, self.y = nx, ny
        return (self.x, self.y) == tuple(target)

    def _accept(self, order):
        if getattr(order, "rider_id", None) is None:
            try:
                order.assign_to(self.rider_id)
            except AttributeError:
                pass
        order.status = "picking_up"
        self.state = PICKING_UP
        self._log_event("accepted_order", order.id)

    def move(self):
        """Advance one tick of movement and state transitions."""
        if self.state == IDLE:
            if not self.orders:
                self.special_case = False
                return
            self._accept(self.orders[0])

        order = self.current_order
        if order is None:
            self.state = IDLE
            return

        if self.state == PICKING_UP:
            if self._step_toward(order.res_pos):
                self.state = DELIVERING
                order.status = "delivering"
                order.pickup_time = _now()
                self._log_event("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            if self._step_toward(order.cus_pos):
                order.status = "completed"
                order.complete_time = _now()
                self.orders.remove(order)
                self.completed_orders.append(order.id)
                self.state = IDLE
                self.special_case = False
                self._log_event("delivered_to_customer", order.id)

    def remaining_path(self):
        """Waypoints the rider still needs to travel for all orders."""
        points = [self.position]
        for idx, order in enumerate(self.orders):
            if idx == 0 and self.state == DELIVERING:
                points.append(order.cus_pos)
            else:
                points.extend([order.res_pos, order.cus_pos])
        return PathPlanner.compute_multi_path(points)

    def to_dict(self):
        """Serialise rider to a JSON friendly dict."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "completed_orders": list(self.completed_orders),
            "total_distance": self.total_distance,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%s,%s), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state, len(self.orders))

"""System initialisation and resource management."""

import json
import logging
import os

import globalvar
from rider import Rider, IDLE
from order_model import Order

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruit_cost": 300,
    "rider_speed": 10,
    "rider_start": [435, 385],
}

CONFIG = dict(DEFAULT_CONFIG)


def load_config(path):
    """Load configuration overrides from a JSON file (if it exists)."""
    if path and os.path.exists(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            for key in DEFAULT_CONFIG:
                if key in data:
                    CONFIG[key] = data[key]
        except (OSError, ValueError) as exc:
            logger.warning("Could not load config %s: %s", path, exc)
    return CONFIG


def init_system(config_path=None):
    """Initialise global system state."""
    if config_path:
        load_config(config_path)
    globalvar.set_value("funds", CONFIG["initial_funds"])
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("pending_orders", [])


class SystemManager:
    """Manages funds, riders, simulated time and persistence."""

    @staticmethod
    def ensure_initialized():
        """Initialise the system if globals are missing (recovery)."""
        if globalvar.get_value("riders") is None:
            init_system()
        for key, default in (("funds", CONFIG["initial_funds"]),
                             ("rider_id_counter", 0), ("running", True),
                             ("time", 0), ("orders", []),
                             ("pending_orders", [])):
            if globalvar.get_value(key) is None:
                globalvar.set_value(key, default)

    @staticmethod
    def funds():
        return globalvar.get_value("funds", 0)

    @staticmethod
    def riders():
        return globalvar.get_value("riders") or []

    @staticmethod
    def idle_riders():
        return [r for r in SystemManager.riders()
                if r.state == IDLE and not r.orders]

    @staticmethod
    def recruit_rider():
        """Create a new rider, deducting the recruitment cost."""
        SystemManager.ensure_initialized()
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        sx, sy = CONFIG["rider_start"]
        rider = Rider(counter, sx, sy, CONFIG["rider_speed"])
        globalvar.get_value("riders").append(rider)
        globalvar.set_value("funds", SystemManager.funds() - CONFIG["recruit_cost"])
        logger.info("Recruited rider %d", counter)
        return rider

    @staticmethod
    def auto_recruit(pending_count):
        """Recruit if funds allow and pending orders exceed idle riders."""
        if (SystemManager.funds() >= CONFIG["recruit_cost"]
                and pending_count > len(SystemManager.idle_riders())):
            return SystemManager.recruit_rider()
        return None

    @staticmethod
    def tick():
        """Advance simulated time by one tick and move all riders."""
        SystemManager.ensure_initialized()
        globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
        for rider in SystemManager.riders():
            try:
                rider.move()
            except Exception as exc:  # recovery: keep simulation alive
                logger.exception("Rider %s move failed: %s", rider.rider_id, exc)

    @staticmethod
    def check_consistency():
        """Return list of consistency problems (empty when consistent)."""
        problems = []
        ids = [r.rider_id for r in SystemManager.riders()]
        if len(ids) != len(set(ids)):
            problems.append("duplicate rider ids")
        for r in SystemManager.riders():
            if r.state not in (0, 1, 2):
                problems.append("rider %s invalid state" % r.rider_id)
            if r.state != IDLE and not r.orders:
                problems.append("rider %s busy without orders" % r.rider_id)
                r.state = IDLE
            if not (0 <= r.x <= 800 and 0 <= r.y <= 800):
                problems.append("rider %s out of map" % r.rider_id)
        return problems

    @staticmethod
    def save_state(path):
        """Persist system state to a JSON file."""
        data = {
            "config": CONFIG,
            "funds": SystemManager.funds(),
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "time": globalvar.get_value("time", 0),
            "riders": [r.to_dict() for r in SystemManager.riders()],
            "orders": [o.to_dict() for o in globalvar.get_value("orders") or []],
            "pending_orders": [o.id for o in
                               globalvar.get_value("pending_orders") or []],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)

    @staticmethod
    def load_state(path):
        """Restore system state from a JSON file."""
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        init_system()
        orders = [Order.from_dict(o) for o in data.get("orders", [])]
        by_id = {o.id: o for o in orders}
        riders = []
        for rd in data.get("riders", []):
            r = Rider(rd["rider_id"], rd["x"], rd["y"], rd["speed"])
            r.state = rd["state"]
            r.orders = [by_id[i] for i in rd.get("orders", []) if i in by_id]
            r.special_case = rd.get("special_case", False)
            r.completed_orders = rd.get("completed_orders", [])
            r.total_distance = rd.get("total_distance", 0)
            r.log = rd.get("log", r.log)
            riders.append(r)
        globalvar.set_value("funds", data.get("funds", CONFIG["initial_funds"]))
        globalvar.set_value("rider_id_counter", data.get("rider_id_counter", 0))
        globalvar.set_value("time", data.get("time", 0))
        globalvar.set_value("riders", riders)
        globalvar.set_value("orders", orders)
        globalvar.set_value("pending_orders",
                            [by_id[i] for i in data.get("pending_orders", [])
                             if i in by_id])

"""Order creation, assignment and completion handling."""

import logging

import globalvar
from order_model import Order
from path_planner import PathPlanner, MAP_MIN, MAP_MAX
from rider import IDLE

logger = logging.getLogger(__name__)


def _distance(rider, order):
    return PathPlanner.euclidean((rider.x, rider.y), order.res_pos)


def find_best_rider_for_order(order):
    """Return the rider closest (Euclidean) to the order's restaurant.

    Idle riders are preferred; if none is idle the closest rider overall
    (with the fewest orders as tie breaker) is returned.  Returns None if
    there are no riders.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    idle = [r for r in riders if r.state == IDLE and not r.orders]
    candidates = idle if idle else riders
    return min(candidates,
               key=lambda r: (_distance(r, order), len(r.orders), r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion tracking."""

    @staticmethod
    def validate_coords(coords):
        """Validate 4 integer coordinates in range 0-800."""
        if len(coords) != 4:
            raise ValueError("exactly 4 coordinates required")
        result = []
        for c in coords:
            if isinstance(c, bool) or not isinstance(c, int):
                raise ValueError("coordinates must be integers")
            if not MAP_MIN <= c <= MAP_MAX:
                raise ValueError("coordinates must be within 0-800")
            result.append(c)
        return result

    @staticmethod
    def assign(order):
        """Assign ``order`` to the best rider; returns rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        return rider

    @staticmethod
    def create_order(coords):
        """Create and assign an order; returns a result dict."""
        from system import SystemManager
        SystemManager.ensure_initialized()
        rx, ry, cx, cy = OrderManager.validate_coords(list(coords))
        order = Order((rx, ry), (cx, cy))
        globalvar.get_value("orders").append(order)
        pending = globalvar.get_value("pending_orders")
        new_rider = SystemManager.auto_recruit(len(pending) + 1)
        rider = OrderManager.assign(order)
        if rider is None:
            pending.append(order)
        return {"order": order, "rider": rider, "new_rider": new_rider}

    @staticmethod
    def process_pending():
        """Try to assign queued orders (and recruit if needed)."""
        from system import SystemManager
        pending = globalvar.get_value("pending_orders") or []
        if not pending:
            return []
        SystemManager.auto_recruit(len(pending))
        assigned = []
        for order in list(pending):
            rider = OrderManager.assign(order)
            if rider is None:
                break
            pending.remove(order)
            assigned.append(order)
        return assigned

    @staticmethod
    def all_orders():
        return globalvar.get_value("orders") or []

    @staticmethod
    def completed_orders():
        return [o for o in OrderManager.all_orders() if o.status == "completed"]

    @staticmethod
    def efficiency():
        """Average delivery time of completed orders (None if none)."""
        done = OrderManager.completed_orders()
        if not done:
            return None
        return sum(o.time_taken for o in done) / len(done)

"""Status display and ASCII map visualisation."""

import math

import globalvar
from path_planner import PathPlanner
from rider import STATE_NAMES

SIZE = 17  # 9 street lines + 8 cell rows


def _idx(v):
    return max(0, min(SIZE - 1, int(round(v / 50.0))))


class DisplayManager:
    """Builds textual displays of the simulation."""

    @staticmethod
    def _blank_map():
        grid = []
        for r in range(SIZE):
            row = []
            for c in range(SIZE):
                row.append("#" if (r % 2 == 0 or c % 2 == 0) else ".")
            grid.append(row)
        return grid

    @staticmethod
    def _draw_path(grid, path):
        for i in range(len(path) - 1):
            (x1, y1), (x2, y2) = path[i], path[i + 1]
            length = abs(x2 - x1) + abs(y2 - y1)
            steps = max(1, int(length // 25))
            for s in range(steps + 1):
                x = x1 + (x2 - x1) * s / steps
                y = y1 + (y2 - y1) * s / steps
                grid[_idx(y)][_idx(x)] = "*"

    @staticmethod
    def render_map(route=None, orders=None):
        """Return the ASCII map as a string."""
        grid = DisplayManager._blank_map()
        if route:
            DisplayManager._draw_path(grid, route)
        if orders is None:
            orders = [o for o in globalvar.get_value("orders") or []
                      if o.status != "completed"]
        for o in orders:
            grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = "S"
            grid[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = "C"
        for r in globalvar.get_value("riders") or []:
            grid[_idx(r.y)][_idx(r.x)] = "R"
        header = "    " + "".join(str((c // 2) % 10) if c % 2 else " "
                                  for c in range(SIZE))
        lines = [header]
        for i, row in enumerate(grid):
            label = str(i // 2) if i % 2 else " "
            lines.append(" %s  %s" % (label, "".join(row)))
        lines.append("Legend: R=rider S=restaurant C=customer *=route "
                     "#=street .=empty")
        return "\n".join(lines)

    @staticmethod
    def order_result(result):
        """Text for the result of a ``start`` command."""
        order, rider, new_rider = result["order"], result["rider"], result["new_rider"]
        lines = [
            "Order #%d created." % order.id,
            "Restaurant grid position: (%d, %d)" % order.res_pos,
            "Customer grid position:   (%d, %d)" % order.cus_pos,
            "New rider added: %s" % (
                "yes (rider #%d)" % new_rider.rider_id if new_rider else "no"),
        ]
        if rider is None:
            lines.append("No rider available; order queued as pending.")
            lines.append(DisplayManager.render_map())
            return "\n".join(lines)
        path = PathPlanner.compute_multi_path(
            [rider.position, order.res_pos, order.cus_pos])
        if rider.orders and rider.orders[0] is not order:
            path = rider.remaining_path()
        dist = PathPlanner.path_length(path)
        eta = int(math.ceil(dist / float(rider.speed))) if rider.speed else 0
        lines.append("Assigned rider: #%d at (%d, %d)" % (

"""Command parsing and input validation."""

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "q": "quit", "exit": "quit", "h": "help", "?": "help",
    "t": "tick", "hist": "history",
}

COMMANDS = ("start", "status", "orders", "riders", "quit", "help",
            "tick", "run", "history", "save", "load", "map")

HELP_TEXT = """Commands:
  start [rx ry cx cy] (s)  create an order (coordinates 0-800)
  status (st)              system snapshot
  orders (o)               all order details
  riders (r)               rider status change history
  map                      show the map
  tick [n] (t)             advance n ticks (default 1)
  run                      advance until all orders delivered
  save [file] / load [file] persist / restore state (JSON)
  history (hist)           show command history
  help (h)                 this help
  quit / exit (q)          leave the program"""


class InputManager:
    """Parses commands and validates user input."""

    def __init__(self):
        self.history = []

    def parse(self, line):
        """Return (command, args) for an input line."""
        line = (line or "").strip()
        if not line:
            return "", []
        self.history.append(line)
        if len(self.history) > 500:
            self.history = self.history[-500:]
        parts = line.replace(",", " ").replace("[", " ").replace("]", " ").split()
        cmd = parts[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        return cmd, parts[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Parse 4 integer coordinates in range 0-800.

        Returns (coords, error_message); one of them is None.
        """
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ").replace(
                "]", " ").split()
        if len(tokens) != 4:
            return None, "Please enter exactly 4 integers: rx ry cx cy."
        coords = []
        for t in tokens:
            try:
                v = int(t)
            except (TypeError, ValueError):
                return None, "Invalid value '%s': coordinates must be integers." % t
            if not 0 <= v <= 800:
                return None, "Coordinate %d out of range (0-800)." % v
            coords.append(v)
        return coords, None

    @staticmethod
    def parse_positive_int(tokens, default=1):
        """Parse an optional positive integer argument."""
        if not tokens:
            return default, None
        try:
            v = int(tokens[0])
        except ValueError:
            return None, "Expected an integer."
        if v <= 0:
            return None, "Expected a positive integer."
        return v, None

"""Command-line entry point of the food delivery dispatch simulator."""

import logging
import sys

import globalvar
from system import init_system, SystemManager
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager, HELP_TEXT

try:  # command history / line editing where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

DEFAULT_SAVE = "dispatch_state.json"
MAX_RUN_TICKS = 100000


def advance(n=1):
    """Advance the simulation ``n`` ticks."""
    for _ in range(n):
        SystemManager.tick()
        OrderManager.process_pending()


def handle(cmd, args, inp, out=print):
    """Handle a single command. Returns False when the program should exit."""
    if cmd == "":
        return True
    if cmd == "quit":
        globalvar.set_value("running", False)
        out("Goodbye.")
        return False
    if cmd == "help":
        out(HELP_TEXT)
    elif cmd == "start":
        tokens = args
        if not tokens:
            try:
                tokens = input("Enter restaurant_x restaurant_y customer_x "
                               "customer_y: ")
            except EOFError:
                return True
        coords, err = inp.parse_coordinates(tokens)
        if err:
            out("Error: " + err)
            return True
        result = OrderManager.create_order(coords)
        out(DisplayManager.order_result(result))
    elif cmd == "status":
        out(DisplayManager.status())
    elif cmd == "orders":
        out(DisplayManager.orders())
    elif cmd == "riders":
        out(DisplayManager.riders_log())
    elif cmd == "map":
        out(DisplayManager.render_map())
    elif cmd == "tick":
        n, err = inp.parse_positive_int(args)
        if err:
            out("Error: " + err)
            return True
        advance(n)
        out("Advanced %d tick(s). Time is now %d." % (
            n, globalvar.get_value("time", 0)))
    elif cmd == "run":
        count = 0
        while count < MAX_RUN_TICKS and any(
                o.status != "completed" for o in OrderManager.all_orders()):
            advance(1)
            count += 1
        out("Ran %d tick(s). Time is now %d." % (
            count, globalvar.get_value("time", 0)))
    elif cmd == "history":
        for i, h in enumerate(inp.history, 1):
            out("%4d  %s" % (i, h))
    elif cmd == "save":
        path = args[0] if args else DEFAULT_SAVE
        try:
            SystemManager.save_state(path)
            out("State saved to %s" % path)
        except OSError as exc:
            out("Error saving: %s" % exc)
    elif cmd == "load":
        path = args[0] if args else DEFAULT_SAVE
        try:
            SystemManager.load_state(path)
            out("State loaded from %s" % path)
        except (OSError, ValueError, KeyError) as exc:
            out("Error loading: %s" % exc)
    else:
        out("Unknown command '%s'. Type 'help' for a list." % cmd)
    return True


def main(argv=None):
    """Main loop: each iteration is one simulated tick."""
    argv = sys.argv[1:] if argv is None else argv
    level = logging.DEBUG if "--debug" in argv else logging.WARNING
    logging.basicConfig(level=level, format="%(levelname)s %(name)s: %(message)s")
    config = None
    for a in argv:
        if a.startswith("--config="):
            config = a.split("=", 1)[1]
    globalvar._init()
    init_system(config)
    inp = InputManager()
    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
    while globalvar.get_value("running", True):
        try:
            line = input("[t=%d] > " % globalvar.get_value("time", 0))
        except (EOFError, KeyboardInterrupt):
            print()
            break
        cmd, args = inp.parse(line)
        try:
            if not handle(cmd, args, inp):
                break
            if cmd not in ("tick", "run", "help", "history", ""):
                advance(1)
            problems = SystemManager.check_consistency()
            for p in problems:
                logging.warning("Consistency issue fixed/detected: %s", p)
        except Exception as exc:  # graceful recovery
            logging.exception("Unexpected error")
            print("Error: %s (system state preserved)" % exc)
            SystemManager.ensure_initialized()
    return 0


if __name__ == "__main__":
    sys.exit(main())

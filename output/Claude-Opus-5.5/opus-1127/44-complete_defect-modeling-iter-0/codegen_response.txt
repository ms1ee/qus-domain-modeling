"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network is stored in ``subway_data.json`` next to this module::

    {
        "stations": [{"name": str, "lines": [str, ...]}, ...],
        "connections": [{"source": str, "dest": str,
                         "distance": int (meters), "line": str}, ...]
    }
"""

import json
import os

# Absolute path of the bundled data file, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON data file (defaults to ``DATA_FILE_PATH``)
    :return: dict with the keys ``stations`` and ``connections``
    :raises FileNotFoundError: if the file does not exist
    :raises ValueError: if the file is not valid JSON or has a wrong structure
    """
    if path is None:
        path = DATA_FILE_PATH
    if not os.path.isfile(path):
        raise FileNotFoundError(f"Subway data file not found: {path}")
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Subway data file is not valid JSON: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise ValueError("'stations' and 'connections' must be lists")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data back to a JSON file (used by the data update mechanism)."""
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, ensure_ascii=False, indent=2)

from dataclasses import dataclass, field
from typing import Dict, List, Optional


class Edge:
    """A weighted, line-labelled edge between two station numbers."""

    def __init__(self, source, dest, distance, line=None):
        if distance is None or isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        """Return the same connection in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def other(self, vertex):
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex} is not an endpoint of this edge")

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.weight, self.line) == (
            other.source, other.dest, other.weight, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.weight, self.line))

    def __repr__(self):
        return f"Edge({self.source!r}, {self.dest!r}, {self.weight!r}, {self.line!r})"


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller; the graph just stores them.
    """

    def __init__(self):
        self.adj: Dict[int, List[Edge]] = {}
        self.station_lines: Dict[str, List[str]] = {}
        self.station_to_number: Dict[str, int] = {}
        self.number_to_station: Dict[int, str] = {}
        self._edges: List[Edge] = []

    # ---- vertices -------------------------------------------------------
    def add_vertex(self, v):
        self.adj.setdefault(v, [])

    def add_station(self, name, number, lines=None):
        """Register a named station with a caller-assigned number."""
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.add_vertex(number)
        if lines is not None:
            self.set_station_lines(name, lines)

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(dict.fromkeys(lines or []))

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    def vertices(self):
        return list(self.adj.keys())

    @property
    def V(self):
        return len(self.adj)

    @property
    def vertex_count(self):
        return self.V

    # ---- edges ----------------------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge (stored in both adjacency lists)."""
        if not isinstance(edge, Edge):
            raise TypeError("add_edge expects an Edge instance")
        self.adj.setdefault(edge.source, []).append(edge)
        if edge.source != edge.dest:
            self.adj.setdefault(edge.dest, []).append(edge.reversed())
        self._edges.append(edge)

    def adjacent(self, v):
        return self.adj.get(v, [])

    def edges(self):
        return list(self._edges)

    @property
    def E(self):
        return len(self._edges)

    @property
    def edge_count(self):
        return self.E

    def __repr__(self):
        return f"Graph(V={self.V}, E={self.E})"

"""
Path calculation: heap-based priority queue and Dijkstra shortest path.

The search runs on (station, current line[, transfers]) states, which makes
transfers at multi-line interchange stations explicit and allows supporting
transfer limits, least-transfer mode and line preferences while keeping the
O((V+E) log V) behaviour of Dijkstra's algorithm.
"""

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance penalty (meters) of one transfer in comprehensive mode.
TRANSFER_PENALTY_METERS = 3000
# Weight factor applied to edges of preferred lines.
PREFERRED_LINE_FACTOR = 0.8

INF = float("inf")


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self.heap = []

    @property
    def size(self):
        return len(self.heap)

    def __len__(self):
        return len(self.heap)

    def is_empty(self):
        return not self.heap

    def insert(self, item):
        self.heap.append(item)
        self._swim(len(self.heap) - 1)

    def min(self):
        if not self.heap:
            raise IndexError("Priority queue underflow")
        return self.heap[0]

    def del_min(self):
        if not self.heap:
            raise IndexError("Priority queue underflow")
        top = self.heap[0]
        last = self.heap.pop()
        if self.heap:
            self.heap[0] = last
            self._sink(0)
        return top

    def _swim(self, k):
        heap = self.heap
        while k > 0:
            parent = (k - 1) // 2
            if heap[k] < heap[parent]:
                heap[k], heap[parent] = heap[parent], heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        heap = self.heap
        n = len(heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            if left + 1 < n and heap[left + 1] < heap[left]:
                child = left + 1
            if heap[child] < heap[k]:
                heap[k], heap[child] = heap[child], heap[k]
                k = child
            else:
                break


class PathfindingConfig:
    """Configuration of a path query."""

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        if isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.preferred_lines = list(preferred_lines or [])

    def __repr__(self):
        return (f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
                f"preferred_lines={self.preferred_lines!r})")


class ShortestPath:
    """Dijkstra shortest path from a source station number."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        if self.config.mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{self.config.mode}'")
        self._preferred = set(self.config.preferred_lines or [])
        self.dist_to = {v: INF for v in graph.vertices()}
        self.edge_to = {}
        self._keys = {}
        self._info = {}          # state -> (weighted cost, distance, transfers)
        self._state_edge_to = {}  # state -> (previous state, edge)
        self._best_state = {}
        self._run()

    # ------------------------------------------------------------------
    def _weighted(self, edge):
        if self._preferred and edge.line in self._preferred:
            return edge.weight * PREFERRED_LINE_FACTOR
        return edge.weight

    def _key(self, wdist, transfers, dist):
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, wdist, dist)
        if mode == MODE_COMPREHENSIVE:
            return (wdist + transfers * TRANSFER_PENALTY_METERS, transfers, dist)
        return (wdist, transfers, dist)

    def _run(self):
        limit = self.config.max_transfers
        start = (self.source, None, 0 if limit is not None else None)
        self._info[start] = (0, 0, 0)
        self._keys[start] = self._key(0, 0, 0)
        pq = MinPQ()
        counter = 0
        pq.insert((self._keys[start], counter, start))
        while not pq.is_empty():
            key, _, state = pq.del_min()
            if key > self._keys[state]:
                continue  # stale entry (lazy deletion)
            node, line, _ = state
            wd, d, t = self._info[state]
            for e in self.graph.adjacent(node):
                if e.dest == node:
                    continue
                switching = line is not None and e.line is not None and e.line != line
                nt = t + (1 if switching else 0)
                if limit is not None and nt > limit:
                    continue
                new_line = e.line if e.line is not None else line
                ns = (e.dest, new_line, nt if limit is not None else None)
                nwd = wd + self._weighted(e)
                nd = d + e.weight
                nk = self._key(nwd, nt, nd)
                old = self._keys.get(ns)
                if old is None or nk < old:
                    self._keys[ns] = nk
                    self._info[ns] = (nwd, nd, nt)
                    self._state_edge_to[ns] = (state, e)
                    counter += 1
                    pq.insert((nk, counter, ns))

        for state, key in self._keys.items():
            node = state[0]
            best = self._best_state.get(node)
            if best is None or key < self._keys[best]:
                self._best_state[node] = state
        for node, state in self._best_state.items():
            self.dist_to[node] = self._info[state][1]
            if state in self._state_edge_to:
                self.edge_to[node] = self._state_edge_to[state][1]

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target.

        Both lists are empty if the target is unreachable.
        """
        state = self._best_state.get(target)
        if state is None:
            return [], []
        edges = []
        while state in self._state_edge_to:
            prev, e = self._state_edge_to[state]
            edges.append(e)
            state = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes on the chosen path (-1 if unreachable)."""
        state = self._best_state.get(target)
        if state is None:
            return -1
        return self._info[state][2]

"""
Beijing Subway fare calculation.

Fare structure (distance rounded up to whole kilometres):
    0-6 km: 3 yuan, 6-12 km: 4 yuan, 12-22 km: 5 yuan, 22-32 km: 6 yuan,
    above 32 km: +1 yuan for every (started) 20 km.
"""

import math
from dataclasses import dataclass, field
from typing import List

# (min_km exclusive, max_km inclusive, fare yuan)
FARE_TIERS = [(0, 6, 3), (6, 12, 4), (12, 22, 5), (22, 32, 6)]
EXTENDED_INCREMENT_KM = 20
EXTENDED_INCREMENT_YUAN = 1

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


def _validate_number(value, name):
    if value is None or isinstance(value, bool) or not isinstance(value, (int, float)):
        raise TypeError(f"{name} must be a number")
    if isinstance(value, float) and math.isnan(value):
        raise ValueError(f"{name} must be a valid number")
    if value < 0:
        raise ValueError(f"{name} must be non-negative")


def distance_to_km_ceiling(distance):
    """Convert meters to whole kilometres, rounding up."""
    _validate_number(distance, "distance")
    if isinstance(distance, int):
        return -(-distance // 1000)
    return int(math.ceil(distance / 1000.0))


@dataclass
class FareTier:
    min_km: int
    max_km: int
    fare_yuan: int


@dataclass
class FareStructure:
    tiers: List[FareTier] = field(default_factory=lambda: [FareTier(*t) for t in FARE_TIERS])
    extended_increment_km: int = EXTENDED_INCREMENT_KM
    extended_increment_yuan: int = EXTENDED_INCREMENT_YUAN


class FareCalculator:
    """Distance-to-fare conversion based on a FareStructure."""

    def __init__(self, structure=None, round_up_distance=True):
        self.structure = structure or FareStructure()
        self.round_up_distance = round_up_distance

    def calculate(self, distance):
        _validate_number(distance, "distance")
        km = distance_to_km_ceiling(distance) if self.round_up_distance else distance / 1000.0
        tiers = self.structure.tiers
        for tier in tiers:
            if km <= tier.max_km:
                return tier.fare_yuan
        last = tiers[-1]
        extra = km - last.max_km
        steps = int(math.ceil(extra / float(self.structure.extended_increment_km)))
        return last.fare_yuan + steps * self.structure.extended_increment_yuan


_DEFAULT_CALCULATOR = FareCalculator()


def Sum_money(distance):
    """Return the fare in yuan for a distance given in meters."""
    return _DEFAULT_CALCULATOR.calculate(distance)


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes for a distance (meters) and transfers."""
    _validate_number(distance, "distance")
    _validate_number(transfer_count, "transfer_count")
    riding = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    return float(riding + transfer_count * TRANSFER_TIME_MINUTES)

"""
Subway network data management: modelling, querying, updating and checking.
"""

import difflib
import re
from collections import deque

try:
    from data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
    from models import Edge, Graph, Line, Route, Transfer
    from pathfinding import ShortestPath, PathfindingConfig
    from fare_calculator import Sum_money, calculate_estimated_time
except ImportError:  # pragma: no cover - package-style import
    from .data_loader import DATA_FILE_PATH, load_subway_data, save_subway_data
    from .models import Edge, Graph, Line, Route, Transfer
    from .pathfinding import ShortestPath, PathfindingConfig
    from .fare_calculator import Sum_money, calculate_estimated_time


def _normalize_name(name):
    n = str(name).strip().casefold()
    n = re.sub(r"[\s'’`\-_.·]", "", n)
    if n.endswith("station") and len(n) > len("station"):
        n = n[: -len("station")]
    if n.endswith("站") and len(n) > 1:
        n = n[:-1]
    return n


class SubwayNetwork:
    """Holds stations, connections, lines and the graph model of the network."""

    def __init__(self, data_file=DATA_FILE_PATH, data=None):
        self.data_source_file = data_file
        if data is None:
            data = load_subway_data(data_file)
        self.stations = [dict(s) for s in data.get("stations", [])]
        self.connections = [dict(c) for c in data.get("connections", [])]
        self._build()

    # ------------------------------------------------------------------
    def _build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        station_lines = {}

        def register(name):
            if name not in self.station_to_number:
                num = len(self.station_to_number)
                self.station_to_number[name] = num
                self.number_to_station[num] = name
                station_lines[name] = []

        for st in self.stations:
            name = st.get("name")
            if not name:
                continue
            register(name)
            for ln in st.get("lines", []) or []:
                if ln not in station_lines[name]:
                    station_lines[name].append(ln)

        for c in self.connections:
            for key in ("source", "dest"):
                name = c.get(key)
                if not name:
                    continue
                register(name)
                ln = c.get("line")
                if ln and ln not in station_lines[name]:
                    station_lines[name].append(ln)

        self.station_lines = station_lines
        self.graph = {name: [] for name in self.station_to_number}
        self.network_graph = Graph()
        for name, num in self.station_to_number.items():
            self.network_graph.add_station(name, num, station_lines[name])

        for c in self.connections:
            s, d = c.get("source"), c.get("dest")
            dist = c.get("distance")
            if not s or not d or s == d:
                continue
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist < 0:
                continue
            line = c.get("line")
            self.network_graph.add_edge(Edge(self.station_to_number[s], self.station_to_number[d], dist, line))
            self.graph[s].append({"station": d, "distance": dist, "line": line})
            self.graph[d].append({"station": s, "distance": dist, "line": line})

        self._normalized = {}
        for name in self.station_to_number:
            self._normalized.setdefault(_normalize_name(name), name)

    # ---- queries -----------------------------------------------------
    def station_names(self):
        return list(self.station_to_number.keys())

    def line_names(self):
        names = []
        for lines in self.station_lines.values():
            for ln in lines:
                if ln not in names:
                    names.append(ln)
        return names

    def transfer_stations(self):
        return [n for n, lines in self.station_lines.items() if len(lines) > 1]

    def resolve_station(self, name):
        """Return the official station name matching the input, or None."""
        if name is None:
            return None
        n = str(name).strip()
        if n in self.station_to_number:
            return n
        return self._normalized.get(_normalize_name(n))

    def suggest_stations(self, name, n=3):
        return difflib.get_close_matches(str(name).strip(), self.station_names(), n=n, cutoff=0.6)

    def get_line_stations(self, line):
        """Ordered station sequence of a line, derived from its connections."""
        adj = {}
        order = []
        for c in self.connections:
            if c.get("line") != line:
                continue
            s, d = c.get("source"), c.get("dest")
            for a, b in ((s, d), (d, s)):
                if a not in adj:
                    adj[a] = []
                    order.append(a)
                if b not in adj[a]:
                    adj[a].append(b)
        if not adj:
            return [n for n, lines in self.station_lines.items() if line in lines]
        start = next((n for n in order if len(adj[n]) == 1), order[0])
        result, visited = [], set()
        stack = [start]
        while stack:
            node = stack.pop()
            if node in visited:
                continue
            visited.add(node)
            result.append(node)
            for nb in reversed(adj[node]):
                if nb not in visited:
                    stack.append(nb)
        for n in order:
            if n not in visited:
                result.append(n)
        return result

    def get_lines(self):
        lines = []
        for ln in self.line_names():
            seq = self.get_line_stations(ln)
            transfers = [s for s in seq if len(self.station_lines.get(s, [])) > 1]
            lines.append(Line(ln, seq, ", ".join(transfers)))
        return lines

    # ---- route finding ------------------------------------------------
    def find_route(self, start, end, config=None):
        """Compute a Route between two (official) station names, or None."""
        config = config or PathfindingConfig()
        if start not in self.station_to_number:
            raise KeyError(f"Unknown station: {start}")
        if end not in self.station_to_number:
            raise KeyError(f"Unknown station: {end}")
        sp = ShortestPath(self.network_graph, self.station_to_number[start], config)
        target = self.station_to_number[end]
        nodes, edges = sp.path_with_edges(target)
        if not nodes:
            return None
        names = [self.number_to_station[n] for n in nodes]

        segments = []
        for e in edges:
            a, b = self.number_to_station[e.source], self.number_to_station[e.dest]
            if segments and segments[-1]["line"] == e.line:
                seg = segments[-1]
                seg["stations"].append(b)
                seg["distance"] += e.weight
            else:
                segments.append({"line": e.line, "stations": [a, b], "distance": e.weight})

        transfers = []
        for prev, nxt in zip(segments, segments[1:]):
            transfers.append(Transfer(prev["stations"][-1], prev["line"], nxt["line"]))

        involved = []
        for seg in segments:
            if seg["line"] is not None and seg["line"] not in involved:
                involved.append(seg["line"])

        total = sum(e.weight for e in edges)
        if isinstance(total, float) and total.is_integer():
            total = int(total)
        transfer_count = len(transfers)
        return Route(
            start=start,
            end=end,
            station_sequence=names,
            total_distance_meters=total,
            transfer_count=transfer_count,
            estimated_time=calculate_estimated_time(total, transfer_count),
            involved_lines=involved,
            transfers=transfers,
            segments=segments,
            fare=Sum_money(total),
            mode=config.mode,
        )

    # ---- data update mechanisms ---------------------------------------
    def add_line(self, line_name, station_sequence, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if len(station_sequence) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_sequence) - 1:
            raise ValueError("distances must have exactly len(stations) - 1 items")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        existing = {s["name"]: s for s in self.stations}
        for name in station_sequence:
            if name in existing:
                lines = existing[name].setdefault("lines", [])
                if line_name not in lines:
                    lines.append(line_name)
            else:
                st = {"name": name, "lines": [line_name]}
                self.stations.append(st)
                existing[name] = st
        for a, b, d in zip(station_sequence, station_sequence[1:], distances):
            self.connections.append({"source": a, "dest": b, "distance": d, "line": line_name})
        self._build()

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        if name not in self.station_to_number:
            raise KeyError(f"Unknown station: {name}")
        if new_name and new_name != name and new_name in self.station_to_number:
            raise ValueError(f"Station '{new_name}' already exists")
        found = False
        for st in self.stations:
            if st.get("name") == name:
                found = True
                if new_name:
                    st["name"] = new_name
                if lines is not None:
                    st["lines"] = list(lines)
        if not found:
            self.stations.append({"name": new_name or name, "lines": list(lines or self.station_lines[name])})
        if new_name:
            for c in self.connections:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        self._build()

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance of the connection between two adjacent stations."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        updated = 0
        for c in self.connections:
            if {c.get("source"), c.get("dest")} == {station_a, station_b} and (line is None or c.get("line") == line):
                c["distance"] = distance
                updated += 1
        if not updated:
            raise KeyError(f"No connection between '{station_a}' and '{station_b}'")
        self._build()
        return updated

    def to_dict(self):
        return {"stations": self.stations, "connections": self.connections}

    def save(self, path=None):
        save_subway_data(self.to_dict(), path or self.data_source_file)

def _line_matches(main_alias, data_line):
    a, b = _normalize_name(main_alias), _normalize_name(data_line)
    if a == b:
        return True
    return b.startswith(a) and not b[len(a):len(a) + 1].isdigit()


class DataIntegrityChecker:
    """Checks coverage, connection integrity and distance accuracy of the data."""

    def __init__(self, network, representative_stations=None):
        self.network = network
        self.representative_stations = representative_stations or REPRESENTATIVE_STATIONS
        self.result = None
        self.errors = []
        self.warnings = []
        self.info = []

    def run(self):
        self.errors, self.warnings, self.info = [], [], []
        self._check_coverage()
        self._check_connections()
        self._check_distances()
        self._check_connectivity()
        self.result = not self.errors
        return self.result

    def _check_coverage(self):
        net = self.network
        data_lines = net.line_names()
        self.info.append(f"Stations: {len(net.station_to_number)}, connections: {len(net.connections)}, "
                         f"lines: {len(data_lines)}, transfer stations: {len(net.transfer_stations())}")
        for main, aliases in MAIN_LINES.items():
            if not any(_line_matches(a, dl) for a in aliases for dl in data_lines):
                self.warnings.append(f"Main line '{main}' not found in data")
        for line, candidates in self.representative_stations.items():
            if not any(net.resolve_station(c) for c in candidates):
                self.warnings.append(f"Representative station of {line} ({candidates[0]}) not found")
        declared = {s.get("name") for s in net.stations}
        if not net.stations:
            self.errors.append("No station data present")
        for s in net.stations:
            if not s.get("name"):
                self.errors.append("Station entry without a name")
            elif not s.get("lines"):
                self.warnings.append(f"Station '{s.get('name')}' has no line information")
        names = [s.get("name") for s in net.stations]
        dups = {n for n in names if names.count(n) > 1}
        for d in sorted(x for x in dups if x):
            self.warnings.append(f"Duplicate station entry '{d}'")
        self._declared = declared

    def _check_connections(self):
        net = self.network
        declared = self._declared
        seen = {}
        connected = set()
        for i, c in enumerate(net.connections):
            s, d, line = c.get("source"), c.get("dest"), c.get("line")
            if not s or not d:
                self.errors.append(f"Connection #{i} is missing source or destination")
                continue
            if s == d:
                self.errors.append(f"Connection #{i} connects '{s}' to itself")
            for name in (s, d):
                if name not in declared:
                    self.errors.append(f"Connection #{i} references undeclared station '{name}'")
            if not line:
                self.warnings.append(f"Connection {s} - {d} has no line")
            else:
                for name in (s, d):
                    st = next((x for x in net.stations if x.get("name") == name), None)
                    if st and line not in (st.get("lines") or []):
                        self.warnings.append(f"Station '{name}' is not declared on {line} but has a {line} connection")
            key = (frozenset((s, d)), line)
            dist = c.get("distance")
            if key in seen and seen[key] != dist:
                self.errors.append(f"Conflicting distances for {s} - {d} on {line}: {seen[key]} vs {dist}")
            seen[key] = dist
            connected.update((s, d))
        for name in declared:
            if name and name not in connected:
                self.warnings.append(f"Station '{name}' has no connections")

    def _check_distances(self):
        for c in self.network.connections:
            dist = c.get("distance")
            label = f"{c.get('source')} - {c.get('dest')}"
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                self.errors.append(f"Distance of {label} is not a number")
            elif dist <= 0:
                self.errors.append(f"Distance of {label} must be positive (got {dist})")
            elif dist > MAX_REASONABLE_DISTANCE:
                self.errors.append(f"Distance of {label} is unrealistically large ({dist} m)")
            elif dist > WARN_DISTANCE:
                self.warnings.append(f"Distance of {label} is unusually large ({dist} m)")
            elif dist < MIN_REASONABLE_DISTANCE:
                self.warnings.append(f"Distance of {label} is unusually small ({dist} m)")

    def _check_connectivity(self):
        graph = self.network.graph
        if not graph:
            return
        start = next(iter(graph))
        seen = {start}
        q = deque([start])
        while q:
            n = q.popleft()
            for nb in graph[n]:
                if nb["station"] not in seen:
                    seen.add(nb["station"])
                    q.append(nb["station"])
        unreachable = [n for n in graph if n not in seen]
        if unreachable:
            sample = ", ".join(unreachable[:10])
            self.warnings.append(f"{len(unreachable)} station(s) not connected to the main network: {sample}")

    def format_report(self):
        if self.result is None:
            self.run()
        lines = ["=== Data Integrity Check ==="]
        lines += [f"[INFO] {m}" for m in self.info]
        lines += [f"[ERROR] {m}" for m in self.errors]
        lines += [f"[WARNING] {m}" for m in self.warnings]
        lines.append(f"Result: {'PASSED' if self.result else 'FAILED'} "
                     f"({len(self.errors)} error(s), {len(self.warnings)} warning(s))")
        return "\n".join(lines)

def format_distance(meters):
    """Meters for short distances, kilometres for long ones."""
    if meters < 1000:
        return f"{int(round(meters))} meters"
    return f"{meters / 1000.0:.2f} km"

def format_text_report(route):
    summary = build_summary(route)
    out = [f"=== Route: {route.start} -> {route.end} ===",
           f"Mode: {route.mode}",
           f"Stations ({len(route.station_sequence)}): " + " -> ".join(route.station_sequence),
           "Segments:"]
    for i, seg in enumerate(route.segments):
        stops = len(seg["stations"]) - 1
        out.append(f"  [{seg['line'] or 'Unknown line'}] {seg['stations'][0]} -> {seg['stations'][-1]} "
                   f"({stops} stop{'s' if stops != 1 else ''}, {format_distance(seg['distance'])})")
        if i < len(route.transfers):
            t = route.transfers[i]
            out.append(f"    Transfer at {t.station_name}: {t.from_line} -> {t.to_line}")

    out.append("Transfer stations: " + (", ".join(transfer_names) if transfer_names else "None"))
    out.append("Lines involved: " + ", ".join(route.involved_lines))
    out.append(f"Total distance: {format_distance(route.total_distance_meters)} "
               f"({route.total_distance_meters} meters / {summary.total_distance_km:.2f} km)")
    out.append(f"Transfers: {summary.transfer_count}")
    out.append(f"Estimated time: {summary.estimated_time:.1f} minutes")
    out.append(f"Fare: {summary.total_fare} yuan")

Command-line interface of the Beijing Subway fare calculation system.

Usage:
    python src/main.py [--format text|json] [--output_file FILE]
                       [--mode shortest_distance|least_transfer|comprehensive]
                       [--max_transfers N] [--line_preference "Line 1"] [--check_data]

Enter queries as "Starting Station; Ending Station"; enter 'q' to quit.

import argparse
import json
import sys
from dataclasses import dataclass, field
from typing import List, Optional

try:
    from data_loader import DATA_FILE_PATH
    from network import SubwayNetwork, DataIntegrityChecker
    from pathfinding import PathfindingConfig, VALID_MODES
    from report import format_text_report, build_json_report

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
SAME_STATION_ERROR = "Starting station cannot be the same as the ending station"
QUIT_WORDS = {"q", "quit", "exit"}


class QueryError(Exception):
    """User-facing query error."""


@dataclass
class QueryConfiguration:
    output_file: Optional[str] = None
    format: str = "text"
    mode: str = "shortest_distance"
    max_transfers: Optional[int] = None
    line_preference: List[str] = field(default_factory=list)
    check_data: bool = False
    data_file: str = DATA_FILE_PATH


def _non_negative_int(value):
    try:
        n = int(value)
    except ValueError:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    if n < 0:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    return n


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="export query results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text", help="output format")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance", help="path query mode")
    p.add_argument("--max_transfers", type=_non_negative_int, default=None, help="maximum transfers")
    p.add_argument("--line_preference", default=None, help="preferred line(s), comma separated")
    p.add_argument("--check_data", action="store_true", help="run data integrity check and exit")
    p.add_argument("--data_file", default=DATA_FILE_PATH, help=argparse.SUPPRESS)
    a = p.parse_args(argv)
    prefs = [x.strip() for x in a.line_preference.split(",") if x.strip()] if a.line_preference else []
    return QueryConfiguration(a.output_file, a.format, a.mode, a.max_transfers, prefs, a.check_data, a.data_file)


class CLI:
    input_separator = ";"

    def __init__(self, network, config, out=sys.stdout):
        self.network = network
        self.config = config
        self.out = out
        self.continuous_mode = True
        self.json_results = []
        self.path_config = PathfindingConfig(config.mode, config.max_transfers, config.line_preference)

    def _print(self, text=""):
        print(text, file=self.out)

    def parse_query(self, raw):
        """Validate input and return (start, end) official station names."""
        if raw is None:
            raise QueryError(FORMAT_ERROR)
        text = raw.replace("；", ";")
        parts = text.split(self.input_separator)
        if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
            raise QueryError(FORMAT_ERROR)
        start_raw, end_raw = parts[0].strip(), parts[1].strip()
        start = self.network.resolve_station(start_raw)
        if start is None:
            raise QueryError(self._missing("Starting", start_raw))
        end = self.network.resolve_station(end_raw)
        if end is None:
            raise QueryError(self._missing("Ending", end_raw))
        if start == end:
            raise QueryError(SAME_STATION_ERROR)
        return start, end

    def _missing(self, kind, name):
        msg = f"{kind} station '{name}' does not exist, please check the station name"
        sugg = self.network.suggest_stations(name)
        if sugg:
            msg += f" (did you mean: {', '.join(sugg)}?)"
        return msg

    def process_query(self, raw):
        """Process one query; returns the rendered output or raises QueryError."""
        start, end = self.parse_query(raw)
        try:
            route = self.network.find_route(start, end, self.path_config)
        except (ValueError, TypeError, ArithmeticError) as exc:
            if "fare" in str(exc).lower() or "distance" in str(exc).lower():
                raise QueryError(f"Fare calculation error: {exc}")
            raise QueryError(f"Path calculation error: {exc}")
        except Exception as exc:
            raise QueryError(f"Path calculation error: {exc}")
        if route is None:
            constraint = " under the current constraints" if self.config.max_transfers is not None else ""
            raise QueryError(f"No route found from '{start}' to '{end}'{constraint}")
        if self.config.format == "json":
            data = build_json_report(route, self.path_config)
            self.json_results.append(data)
            rendered = json.dumps(data, ensure_ascii=False, indent=2)
        else:
            rendered = format_text_report(route)
        self._export(rendered)
        return rendered

    def _export(self, rendered):
        path = self.config.output_file
        if not path:
            return
        try:
            if self.config.format == "json":
                payload = self.json_results[0] if len(self.json_results) == 1 else self.json_results
                with open(path, "w", encoding="utf-8") as fh:
                    json.dump(payload, fh, ensure_ascii=False, indent=2)
            else:
                with open(path, "a", encoding="utf-8") as fh:
                    fh.write(rendered + "\n\n")
        except OSError as exc:
            self._print(f"Failed to write output file '{path}': {exc}")

    def run(self, input_func=input):
        self._print("Beijing Subway Fare Calculation System")
        self._print("Enter 'Starting Station; Ending Station' (or 'q' to quit).")
        while self.continuous_mode:
            try:
                raw = input_func("> ")
            except (EOFError, KeyboardInterrupt):
                self._print()
                break
            raw = raw.strip()
            if not raw:
                continue
            if raw.lower() in QUIT_WORDS:
                break
            try:
                self._print(self.process_query(raw))
            except QueryError as exc:
                self._print(f"Error: {exc}")
            except Exception as exc:  # last-resort safety net
                self._print(f"Unexpected error: {exc}")
            self._print()
        self._print("Goodbye!")


def main(argv=None):
    config = parse_args(argv)
    try:
        network = SubwayNetwork(config.data_file)
    except (OSError, ValueError) as exc:
        print(f"Failed to load subway data: {exc}", file=sys.stderr)
        return 1
    if config.check_data:
        checker = DataIntegrityChecker(network)
        ok = checker.run()
        print(checker.format_report())
        return 0 if ok else 1
    CLI(network, config).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Core graph data structures: Edge, Graph and MinPQ."""

import heapq
import itertools


class Edge:
    """A weighted, directed connection between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return f"Edge({self.source!r} -> {self.dest!r}, {self.distance}m, {self.line!r})"


class Graph:
    """Weighted directed graph. Station numbers are assigned by the caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}
        self.edges = []

    def add_vertex(self, v):
        self.adj.setdefault(v, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, [])
        self.edges.append(edge)

    def remove_edge(self, source, dest, line=None):
        """Remove edges source->dest (optionally only of a given line)."""
        removed = 0
        lst = self.adj.get(source, [])
        keep = []
        for e in lst:
            if e.dest == dest and (line is None or e.line == line):
                removed += 1
            else:
                keep.append(e)
        self.adj[source] = keep
        self.edges = [e for e in self.edges
                      if not (e.source == source and e.dest == dest and (line is None or e.line == line))]
        return removed

    def edges_from(self, v):
        return list(self.adj.get(v, []))

    def vertices(self):
        return list(self.adj.keys())

    @property
    def V(self):
        return len(self.adj)

    @property
    def E(self):
        return len(self.edges)


class MinPQ:
    """Heap-based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, key, item):
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def pop(self):
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)

"""Dijkstra-based path computation with transfer-aware modes."""

import math

from models import MinPQ

MODES = ("shortest_distance", "least_transfer", "comprehensive")
# Equivalent distance penalty (meters) of one transfer in comprehensive mode.
TRANSFER_PENALTY_METERS = 3000
# Weight multiplier applied to non-preferred lines when preferences exist.
NON_PREFERRED_FACTOR = 1.3


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in MODES:
            raise ValueError(f"Unknown mode '{mode}', expected one of {MODES}")
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class ShortestPath:
    """Single-source shortest paths over a Graph from a station number."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: math.inf for v in graph.vertices()}
        self.dist_to[source] = 0
        self._final = {}
        self._parent = {}
        self._run()

    def _initial_key(self):
        return (0, 0) if self.config.mode == "least_transfer" else 0

    def _next_key(self, prev, edge, is_transfer):
        w = edge.weight
        prefs = self.config.preferred_lines
        if prefs and edge.line not in prefs:
            w = w * NON_PREFERRED_FACTOR
        mode = self.config.mode
        if mode == "least_transfer":
            return (prev[0] + (1 if is_transfer else 0), prev[1] + w)
        if mode == "comprehensive":
            return prev + w + (TRANSFER_PENALTY_METERS if is_transfer else 0)
        return prev + w

    def _run(self):
        max_t = self.config.max_transfers
        track_t = max_t is not None
        start = (self.source, None, 0)
        best = {start: self._initial_key()}
        actual = {start: 0}
        parent = {start: (None, None)}
        settled = set()
        pq = MinPQ()
        pq.push(best[start], start)
        while not pq.is_empty():
            key, state = pq.pop()
            if state in settled:
                continue
            settled.add(state)
            node, line, t = state
            for e in self.graph.adj.get(node, []):
                is_transfer = line is not None and e.line != line
                nt = t + (1 if is_transfer else 0)
                if track_t and nt > max_t:
                    continue
                ns = (e.dest, e.line, nt if track_t else 0)
                if ns in settled:
                    continue
                nkey = self._next_key(key, e, is_transfer)
                if ns not in best or nkey < best[ns]:
                    best[ns] = nkey
                    actual[ns] = actual[state] + e.weight
                    parent[ns] = (state, e)
                    pq.push(nkey, ns)

        node_best = {}
        for state in settled:
            node = state[0]
            if node not in node_best or best[state] < best[node_best[node]]:
                node_best[node] = state
        # Source itself is always reached with zero cost.
        node_best[self.source] = start
        self._final = node_best
        self._parent = parent
        for node, state in node_best.items():
            self.dist_to[node] = actual[state]

    def has_path_to(self, target):
        return target in self._final

    def path_with_edges(self, target):
        """Return (station number list, edge list) from source to target."""
        if target not in self._final:
            return [], []
        edges = []
        state = self._final[target]
        while True:
            prev, edge = self._parent[state]
            if prev is None:
                break
            edges.append(edge)
            state = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                count += 1
        return count

"""Beijing Subway fare calculation."""

import math

# (upper bound in meters, fare, inclusive upper bound?)
FARE_TABLE = [
    (6000, 3, True),
    (12000, 4, True),
    (22000, 5, False),
    (32000, 6, True),
]
EXTENDED_BASE_DISTANCE = 32000
EXTENDED_BASE_FARE = 6
EXTENDED_STEP = 20000


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters."""
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("Distance must be a number")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be finite")
    if distance < 0:
        raise ValueError("Distance cannot be negative")
    d = math.ceil(distance)
    for upper, fare, inclusive in FARE_TABLE:
        if d < upper or (inclusive and d == upper):
            return fare
    extra = d - EXTENDED_BASE_DISTANCE
    return EXTENDED_BASE_FARE + math.ceil(extra / EXTENDED_STEP)


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes: 1.5 min/km plus 5 min per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("Distance and transfer count must be non-negative")
    return float(distance) / 1000.0 * 1.5 + transfer_count * 5.0

"""Subway network data management: graph building, integrity checks, updates."""

from models import Edge, Graph

REPRESENTATIVE_STATIONS = [
    "Xidan", "Guomao", "Dongzhimen", "Xizhimen", "Beijing Railway Station",
    "Haidian Huangzhuang", "Songjiazhuang", "Zhichunlu",
]

MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]


class SubwayNetwork:
    """Holds station mapping (station_to_number) and graph."""

    def __init__(self, data=None):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.lines = {}
        if data:
            self.load(data)

    def _ensure_station(self, name, lines=None):
        if name not in self.station_to_number:
            num = len(self.station_to_number)
            self.station_to_number[name] = num
            self.number_to_station[num] = name
            self.graph.add_vertex(num)
            self.graph.set_station_lines(name, lines or [])
        elif lines:
            existing = self.graph.station_lines.get(name, [])
            for ln in lines:
                if ln not in existing:
                    existing.append(ln)
            self.graph.set_station_lines(name, existing)
        return self.station_to_number[name]

    def load(self, data):
        for st in data.get("stations", []):
            self._ensure_station(st["name"], st.get("lines", []))
            for ln in st.get("lines", []):
                self.lines.setdefault(ln, [])
                if st["name"] not in self.lines[ln]:
                    self.lines[ln].append(st["name"])
        for c in data.get("connections", []):
            self.add_connection(c["source"], c["dest"], c["distance"], c.get("line", ""))

    def add_connection(self, source, dest, distance, line):
        s = self._ensure_station(source, [line] if line else None)
        d = self._ensure_station(dest, [line] if line else None)
        self.graph.add_edge(Edge(s, d, distance, line))
        if line:
            seq = self.lines.setdefault(line, [])
            for n in (source, dest):
                if n not in seq:
                    seq.append(n)

    # ---- data update mechanisms ----
    def add_line(self, line_name, stations, distances, bidirectional=True):
        """Add a new line: stations in order and distances between consecutive ones."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have exactly len(stations)-1 entries")
        for a, b, dist in zip(stations, stations[1:], distances):
            self.add_connection(a, b, dist, line_name)
            if bidirectional:
                self.add_connection(b, a, dist, line_name)

    def modify_station(self, old_name, new_name=None, lines=None):
        if old_name not in self.station_to_number:
            raise KeyError(f"Station '{old_name}' does not exist")
        num = self.station_to_number[old_name]
        cur_lines = self.graph.station_lines.pop(old_name, [])
        if new_name and new_name != old_name:
            if new_name in self.station_to_number:
                raise ValueError(f"Station '{new_name}' already exists")
            del self.station_to_number[old_name]
            self.station_to_number[new_name] = num
            self.number_to_station[num] = new_name
            for seq in self.lines.values():
                for i, n in enumerate(seq):
                    if n == old_name:
                        seq[i] = new_name
        name = self.number_to_station[num]
        self.graph.set_station_lines(name, lines if lines is not None else cur_lines)

    def update_distance(self, source, dest, distance, line=None):
        if distance <= 0:
            raise ValueError("Distance must be positive")
        s = self.station_to_number.get(source)
        d = self.station_to_number.get(dest)
        if s is None or d is None:
            raise KeyError("Unknown station")
        updated = 0
        for e in self.graph.adj.get(s, []):
            if e.dest == d and (line is None or e.line == line):
                e.distance = distance
                e.weight = distance
                updated += 1
        if updated == 0:
            raise KeyError(f"No connection from '{source}' to '{dest}'")
        return updated

    # ---- integrity check ----
    def check_integrity(self, data=None):
        """Return (ok, list of messages)."""
        msgs = []
        ok = True
        names = set(self.station_to_number)
        present_lines = set(self.lines)
        missing_lines = [ln for ln in MAIN_LINES if ln not in present_lines]
        if missing_lines:
            msgs.append("Warning: main lines missing: " + ", ".join(missing_lines))
        missing_st = [s for s in REPRESENTATIVE_STATIONS if s not in names]
        if missing_st:
            msgs.append("Warning: representative stations missing: " + ", ".join(missing_st))
        else:
            msgs.append("Representative station coverage: OK")
        if data is not None:
            declared = {s["name"] for s in data.get("stations", [])}
            for c in data.get("connections", []):
                for key in ("source", "dest"):
                    if c.get(key) not in declared:
                        ok = False
                        msgs.append(f"Error: connection references undeclared station '{c.get(key)}'")
                dist = c.get("distance")
                if not isinstance(dist, (int, float)) or isinstance(dist, bool) or dist <= 0:
                    ok = False
                    msgs.append(f"Error: invalid distance {dist!r} for {c.get('source')} -> {c.get('dest')}")
                elif dist > 30000:
                    msgs.append(f"Warning: unusually long distance {dist} m for {c.get('source')} -> {c.get('dest')}")
        isolated = [self.number_to_station[v] for v in self.graph.vertices()
                    if not self.graph.adj[v] and not any(e.dest == v for e in self.graph.edges)]
        if isolated:
            ok = False
            msgs.append("Error: isolated stations: " + ", ".join(sorted(isolated)))
        else:
            msgs.append("Connection integrity: OK")
        msgs.append(f"Stations: {len(names)}, connections: {self.graph.E}, lines: {len(present_lines)}")
        return ok, msgs

"""Route summary, text and JSON report generation."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return f"{meters} meters ({meters / 1000:.2f} km)"
    return f"{meters / 1000:.2f} km ({meters} meters)"


def build_summary(network, sp, start_name, end_name, mode):
    target = network.station_to_number[end_name]
    nodes, edges = sp.path_with_edges(target)
    if not nodes:
        raise RuntimeError(f"No path found from '{start_name}' to '{end_name}'")
    distance = sum(e.distance for e in edges)
    transfers = sp.get_transfer_count(target)
    try:
        fare = Sum_money(distance)
    except Exception as exc:
        raise RuntimeError(f"Fare calculation failed: {exc}") from exc
    stations = [network.number_to_station[n] for n in nodes]
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    transfer_stations = []
    for a, b in zip(edges, edges[1:]):
        if a.line != b.line:
            transfer_stations.append({"station": network.number_to_station[a.dest],
                                      "from_line": a.line, "to_line": b.line})
    segments = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["stations"].append(network.number_to_station[e.dest])
        else:
            segments.append({"line": e.line,
                             "stations": [network.number_to_station[e.source],
                                          network.number_to_station[e.dest]]})
    return {
        "query": {"start": start_name, "end": end_name, "mode": mode},
        "path": stations,
        "distance": distance,
        "fare": fare,
        "estimated_time": round(calculate_estimated_time(distance, transfers), 1),
        "transfer_count": transfers,
        "lines": lines,
        "transfer_stations": transfer_stations,
        "segments": segments,
    }


def format_text(summary):
    out = []
    q = summary["query"]
    out.append("=" * 50)
    out.append(f"Route: {q['start']} -> {q['end']} (mode: {q['mode']})")
    out.append("-" * 50)
    out.append("Stations: " + " -> ".join(summary["path"]))
    out.append(f"Number of stations: {len(summary['path'])}")
    out.append("Segments:")
    for seg in summary["segments"]:
        out.append(f"  [{seg['line']}] " + " -> ".join(seg["stations"]))
    if summary["transfer_stations"]:
        out.append("Transfers:")
        for t in summary["transfer_stations"]:
            out.append(f"  At {t['station']}: {t['from_line']} -> {t['to_line']}")
    out.append(f"Transfer count: {summary['transfer_count']}")
    out.append("Lines involved: " + ", ".join(summary["lines"]))
    out.append(f"Total distance: {format_distance(summary['distance'])}")
    out.append(f"Estimated time: {summary['estimated_time']} minutes")
    out.append(f"Fare: {summary['fare']} yuan")
    out.append("=" * 50)
    return "\n".join(out)


def format_json(summary):
    data = {k: summary[k] for k in ("query", "path", "distance", "fare", "estimated_time",
                                    "transfer_count", "lines", "transfer_stations")}
    return json.dumps(data, ensure_ascii=False, indent=2)

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import difflib
import re
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import SubwayNetwork
from pathfinding import MODES, PathfindingConfig, ShortestPath
from report import build_summary, format_json, format_text

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", action="append", default=None)
    p.add_argument("--check_data", action="store_true")
    p.add_argument("--data_file", default=DATA_FILE_PATH)
    return p.parse_args(argv)


def resolve_station(network, name, role):
    """Return (resolved_name, message or None). Raises ValueError if not found."""
    if name in network.station_to_number:
        return name, None
    lower = {n.lower(): n for n in network.station_to_number}
    if name.lower() in lower:
        return lower[name.lower()], None
    match = difflib.get_close_matches(name.lower(), list(lower), n=1, cutoff=0.6)
    if match:
        real = lower[match[0]]
        return real, f"{role} station '{name}' not found, using nearest match '{real}'"
    raise ValueError(f"{role} station '{name}' does not exist, please check the station name")


def parse_query(line):
    parts = [x.strip() for x in re.split(r"[;,；，]", line)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERROR)
    return parts[0], parts[1]


def handle_query(network, line, config, fmt):
    start, end = parse_query(line)
    start, m1 = resolve_station(network, start, "Starting")
    end, m2 = resolve_station(network, end, "Ending")
    notes = [m for m in (m1, m2) if m]
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    try:
        sp = ShortestPath(network.graph, network.station_to_number[start], config)
    except Exception as exc:
        raise RuntimeError(f"Path calculation error: {exc}") from exc
    summary = build_summary(network, sp, start, end, config.mode)
    text = format_json(summary) if fmt == "json" else format_text(summary)
    return notes, text


def main(argv=None):
    args = parse_args(argv)
    try:
        data = load_subway_data(args.data_file)
    except (OSError, ValueError) as exc:
        print(f"Failed to load subway data: {exc}", file=sys.stderr)
        return 1
    network = SubwayNetwork(data)

    if args.check_data:
        ok, msgs = network.check_integrity(data)
        for m in msgs:
            print(m)
        print("Data integrity check passed" if ok else "Data integrity check failed")
        return 0 if ok else 1

    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=args.line_preference)
    except ValueError as exc:
        print(f"Configuration error: {exc}", file=sys.stderr)
        return 1

    interactive = sys.stdin.isatty()
    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (or 'quit' to exit).")
    while True:
        if interactive:
            print("> ", end="", flush=True)
        line = sys.stdin.readline()
        if not line:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("quit", "exit", "q"):
            break
        try:
            notes, text = handle_query(network, line, config, args.format)
        except (ValueError, RuntimeError) as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # unexpected
            print(f"Error: unexpected failure: {exc}")
            continue
        for n in notes:
            print(f"Note: {n}")
        print(text)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as fh:
                    fh.write(text + "\n")
            except OSError as exc:
                print(f"Error: could not write output file: {exc}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

    """Initialise (or reset) the global variable dictionary."""
    global _global_dict
    _global_dict = {}

    _global_dict[key] = value

"""Configuration handling (JSON based)."""

import json
import logging
import os

logger = logging.getLogger(__name__)


class Configuration:
    """System parameters, optionally loaded from / saved to a JSON file."""

    DEFAULTS = {
        "initial_funds": 500,
        "rider_recruit_cost": 300,
        "recruit_threshold": 300,
        "order_income": 100,
        "delivery_bonus": 0,
        "default_speed": 50,
        "max_riders": 50,
        "spawn_x": 400,
        "spawn_y": 400,
        "data_file": "dispatch_data.json",
    }

    def __init__(self, file_path=None, **overrides):
        self.__dict__["values"] = dict(self.DEFAULTS)
        self.__dict__["file_path"] = file_path
        if file_path and os.path.exists(file_path):
            self.load(file_path)
        self.values.update(overrides)

    def __getattr__(self, name):
        values = self.__dict__.get("values", {})
        if name in values:
            return values[name]
        raise AttributeError(name)

    def __setattr__(self, name, value):
        if name in ("file_path",):
            self.__dict__[name] = value
        else:
            self.values[name] = value

    def get(self, key, default=None):
        """Return a configuration value."""
        return self.values.get(key, default)

    def load(self, file_path=None):
        """Load values from a JSON file; invalid files are ignored with a warning."""
        path = file_path or self.file_path
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                for key, value in data.items():
                    if key in self.DEFAULTS and isinstance(
                            value, type(self.DEFAULTS[key])):
                        self.values[key] = value
            return True
        except (OSError, ValueError) as exc:
            logger.warning("Could not load configuration %s: %s", path, exc)
            return False

    def save(self, file_path=None):
        """Save values to a JSON file."""
        path = file_path or self.file_path or "config.json"
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.values, fh, indent=2)
        return path

    def to_dict(self):
        """Return a copy of the configuration values."""
        return dict(self.values)

"""Grid map and coordinate conversion helpers (8x8 grid, 100x100 cells)."""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE


def validate_coordinate(value, name="coordinate"):
    """Validate an input coordinate: integer within 0..MAP_SIZE.

    Raises ValueError on invalid input.
    """
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError("%s must be an integer, got %r" % (name, value))
    if value < 0 or value > MAP_SIZE:
        raise ValueError("%s must be between 0 and %d, got %d"
                         % (name, MAP_SIZE, value))
    return value


def to_grid_index(value):
    """Convert a coordinate to its grid cell index (0..7). 800 maps to 7."""
    validate_coordinate(value)
    return min(value // CELL_SIZE, GRID_DIM - 1)


def to_cell_center(value):
    """Convert a coordinate to the center of its grid cell.

    ``center = (v // 100) * 100 + 50`` with the boundary 800 clamped to 750.
    """
    return to_grid_index(value) * CELL_SIZE + CELL_SIZE // 2


def is_on_street(x, y):
    """Return True if the point lies on a street (grid line)."""
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


class GridCell:
    """A single 100x100 cell of the map."""

    def __init__(self, grid_x, grid_y, symbol="."):
        self.grid_x = grid_x
        self.grid_y = grid_y
        self.center_x = grid_x * CELL_SIZE + CELL_SIZE // 2
        self.center_y = grid_y * CELL_SIZE + CELL_SIZE // 2
        self.symbol = symbol

    def __repr__(self):
        return "GridCell(%d, %d)" % (self.grid_x, self.grid_y)


class GridMap:
    """The 8x8 grid map containing 64 cells."""

    def __init__(self, map_size=MAP_SIZE, cell_size=CELL_SIZE):
        self.map_size = map_size
        self.cell_size = cell_size
        self.grid_dimension = map_size // cell_size
        self.cells = {(gx, gy): GridCell(gx, gy)
                      for gx in range(self.grid_dimension)
                      for gy in range(self.grid_dimension)}

    def cell_at(self, x, y):
        """Return the cell containing coordinate (x, y)."""
        return self.cells[(to_grid_index(x), to_grid_index(y))]

"""Grid-based path planning that prefers moving along streets."""

import math

from grid_map import CELL_SIZE, MAP_SIZE

DEFAULT_SPEED = 50


def manhattan(a, b):
    """Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class RoutePoint:
    """A waypoint in a route."""

    def __init__(self, x, y, sequence):
        self.x = x
        self.y = y
        self.sequence = sequence

    def as_tuple(self):
        """Return (x, y)."""
        return (self.x, self.y)

    def __repr__(self):
        return "RoutePoint(%d, %d, #%d)" % (self.x, self.y, self.sequence)


class Route:
    """A sequence of axis-aligned waypoints with distance/time estimates."""

    def __init__(self, points, speed=DEFAULT_SPEED, special_case=False):
        self.points = [RoutePoint(p[0], p[1], i) for i, p in enumerate(points)]
        self.total_distance = sum(
            manhattan(points[i], points[i + 1]) for i in range(len(points) - 1))
        self.speed = speed
        self.estimated_time = (int(math.ceil(self.total_distance / speed))
                               if speed and speed > 0 else 0)
        self.special_case = special_case

    def waypoints(self):
        """Return the waypoints as a list of (x, y) tuples."""
        return [p.as_tuple() for p in self.points]


def _dedupe(points):
    result = []
    for p in points:
        p = (int(p[0]), int(p[1]))
        if not result or result[-1] != p:
            result.append(p)
    return result


class PathPlanner:
    """Plans paths on the 8x8 grid.

    Streets are the grid lines (x or y a multiple of 100). A rider leaves its
    current position onto the nearest street, travels along streets and
    finally enters the target cell. Riders do not block each other (streets
    have unlimited capacity), so no path conflicts or deadlocks can occur.
    """

    def __init__(self, grid_map=None):
        self.grid_map = grid_map

    @staticmethod
    def nearest_street(value, toward=None):
        """Return the street line coordinate nearest to ``value``.

        If ``toward`` is given, the street in the direction of ``toward`` is
        chosen. Values already on a street are returned unchanged.
        """
        if value % CELL_SIZE == 0:
            return value
        low = (value // CELL_SIZE) * CELL_SIZE
        high = min(low + CELL_SIZE, MAP_SIZE)
        if toward is not None and toward != value:
            return high if toward > value else low
        return low if value - low <= high - value else high

    def _waypoints(self, start, end):
        """Return (waypoints, special_case) from start to end."""
        sx, sy = start
        tx, ty = end
        if (sx, sy) == (tx, ty):
            return [start], False
        special = False
        if sx == tx:
            # Special case: target directly above or below.
            if sx % CELL_SIZE == 0:
                pts = [start, end]
            else:
                xs = self.nearest_street(sx)
                pts = [start, (xs, sy), (xs, ty), end]
                special = True
        elif sy == ty:
            if sy % CELL_SIZE == 0:
                pts = [start, end]
            else:
                ys = self.nearest_street(sy)
                pts = [start, (sx, ys), (tx, ys), end]
        else:
            ys = self.nearest_street(sy, ty)
            pts = [start, (sx, ys), (tx, ys), end]
        return _dedupe(pts), special

    def plan(self, start, end, speed=DEFAULT_SPEED):
        """Plan a route from ``start`` to ``end``."""
        pts, special = self._waypoints(tuple(start), tuple(end))
        return Route(pts, speed, special)

    def plan_multi(self, start, destinations, speed=DEFAULT_SPEED):
        """Plan a continuous route through several destinations in order."""
        pts = [tuple(start)]
        special = False
        current = tuple(start)
        for dest in destinations:
            seg, sp = self._waypoints(current, tuple(dest))
            pts.extend(seg[1:])
            special = special or sp
            current = tuple(dest)
        return Route(_dedupe(pts), speed, special)

"""Order data model."""

import globalvar
from grid_map import to_cell_center, to_grid_index, validate_coordinate
from path_planner import PathPlanner


class OrderStatus:
    """Order lifecycle status values."""
    PENDING = "pending"
    ASSIGNED = "assigned"
    PICKING_UP = "picking_up"
    DELIVERING = "delivering"
    COMPLETED = "completed"


def _current_time():
    t = globalvar.get_value("time")
    return t if isinstance(t, int) else 0


def _validate_pos(pos, name):
    if not isinstance(pos, (list, tuple)) or len(pos) != 2:
        raise ValueError("%s must be a pair of coordinates" % name)
    return (validate_coordinate(pos[0], name + " x"),
            validate_coordinate(pos[1], name + " y"))


class Order:
    """A delivery order from a restaurant to a customer.

    Input coordinates are converted to grid cell centers.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos, start_time=None, order_id=None):
        rx, ry = _validate_pos(res_pos, "restaurant")
        cx, cy = _validate_pos(cus_pos, "customer")
        if order_id is None:
            order_id = Order._next_id
        Order._next_id = max(Order._next_id, order_id + 1)
        self.id = order_id
        self.raw_res_pos = (rx, ry)
        self.raw_cus_pos = (cx, cy)
        self.res_pos = (to_cell_center(rx), to_cell_center(ry))
        self.cus_pos = (to_cell_center(cx), to_cell_center(cy))
        self.res_grid = (to_grid_index(rx), to_grid_index(ry))
        self.cus_grid = (to_grid_index(cx), to_grid_index(cy))
        self.start_time = _current_time() if start_time is None else start_time
        self.status = OrderStatus.PENDING
        self.rider_id = None
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.distance = PathPlanner().plan(self.res_pos,
                                           self.cus_pos).total_distance
        self.travel_distance = 0

    # Convenience attributes -------------------------------------------
    @property
    def restaurant_x(self):
        return self.res_pos[0]

    @property
    def restaurant_y(self):
        return self.res_pos[1]

    @property
    def customer_x(self):
        return self.cus_pos[0]

    @property
    def customer_y(self):
        return self.cus_pos[1]

    @classmethod
    def reset_counter(cls, start=1):
        """Reset the order id counter."""
        cls._next_id = start

    # Lifecycle ---------------------------------------------------------
    def assign_to(self, rider_id):
        """Record the assignment of this order to rider ``rider_id``."""
        self.rider_id = rider_id
        if self.status in (OrderStatus.PENDING, OrderStatus.ASSIGNED):
            self.status = OrderStatus.ASSIGNED
        self.assign_time = _current_time()

    def mark_picking_up(self, time):
        """Rider accepted the order and is heading to the restaurant."""
        self.status = OrderStatus.PICKING_UP
        self.accept_time = time

    def mark_picked_up(self, time):
        """Rider picked the food up at the restaurant."""
        self.status = OrderStatus.DELIVERING
        self.pickup_time = time

    def complete(self, time):
        """Mark the order completed at ``time``."""
        self.status = OrderStatus.COMPLETED
        self.completion_time = time
        self.time_taken = time - self.start_time

    @property
    def efficiency(self):
        """Distance (restaurant to customer) per tick of total time taken."""
        if not self.time_taken:
            return None
        return round(self.distance / self.time_taken, 2)

    # Persistence -------------------------------------------------------
    def to_dict(self):
        return {
            "id": self.id,
            "raw_res_pos": list(self.raw_res_pos),
            "raw_cus_pos": list(self.raw_cus_pos),
            "start_time": self.start_time,
            "status": self.status,
            "rider_id": self.rider_id,
            "assign_time": self.assign_time,
            "accept_time": self.accept_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "travel_distance": self.travel_distance,
        }

    @classmethod
    def from_dict(cls, data):
        order = cls(list(data["raw_res_pos"]), list(data["raw_cus_pos"]),
                    start_time=data.get("start_time", 0),
                    order_id=data["id"])
        for key in ("status", "rider_id", "assign_time", "accept_time",
                    "pickup_time", "completion_time", "time_taken",
                    "travel_distance"):
            if key in data:
                setattr(order, key, data[key])
        return order

    def __repr__(self):
        return "Order(#%d %s->%s %s)" % (self.id, self.res_pos, self.cus_pos,
                                         self.status)

"""Rider model: movement, state transitions and status log."""

import logging

import globalvar
from order_model import OrderStatus
from path_planner import DEFAULT_SPEED, PathPlanner, manhattan

logger = logging.getLogger(__name__)

DEFAULT_SPAWN = (400, 400)


class RiderState:
    """Rider state enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2
    NAMES = {0: "idle", 1: "picking up", 2: "delivering"}

    @classmethod
    def name(cls, state):
        return cls.NAMES.get(state, "unknown")


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``. Each call to
    :meth:`move` advances the rider by one tick, performing state transitions
    and recording them in ``rider.log``.
    """

    _planner = PathPlanner()

    def __init__(self, rider_id, x=None, y=None, speed=None):
        spawn = globalvar.get_value("spawn_position") or DEFAULT_SPAWN
        self.rider_id = rider_id
        self.x = int(spawn[0] if x is None else x)
        self.y = int(spawn[1] if y is None else y)
        if speed is None:
            speed = globalvar.get_value("default_speed") or DEFAULT_SPEED
        self.speed = int(speed)
        self.state = RiderState.IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.path = []
        self.route = None
        self.current_order = None
        self.total_distance = 0
        self.completed_count = 0
        self._ticks = 0
        self._record("spawn", None)

    # Helpers -----------------------------------------------------------
    @property
    def position(self):
        return (self.x, self.y)

    def _now(self):
        t = globalvar.get_value("time")
        return t if isinstance(t, int) else self._ticks

    def _record(self, event, order_id):
        self.log.append({
            "time": self._now(),
            "event": event,
            "order_id": order_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "state_name": RiderState.name(self.state),
        })

    def _plan_to(self, target):
        route = self._planner.plan(self.position, target, self.speed)
        self.route = route
        self.path = route.waypoints()[1:]
        self.special_case = route.special_case

    def pending_orders(self):
        """Orders assigned to this rider and not yet completed."""
        return [o for o in self.orders if o.status != OrderStatus.COMPLETED]

    def has_pending_orders(self):
        return bool(self.pending_orders())

    def remaining_work(self):
        """Return (remaining_distance, end_position) for all assigned work."""
        total = 0
        pos = self.position
        cur = self.current_order
        if cur is not None and cur.status != OrderStatus.COMPLETED:
            if self.state == RiderState.PICKING_UP:
                total += manhattan(pos, cur.res_pos) + manhattan(cur.res_pos,
                                                                 cur.cus_pos)
                pos = cur.cus_pos
            elif self.state == RiderState.DELIVERING:
                total += manhattan(pos, cur.cus_pos)
                pos = cur.cus_pos
        for o in self.pending_orders():
            if o is cur:
                continue
            total += manhattan(pos, o.res_pos) + manhattan(o.res_pos, o.cus_pos)
            pos = o.cus_pos
        return total, pos

    def reset_to_idle(self):
        """Release current work and return to the idle state."""
        self.state = RiderState.IDLE
        self.current_order = None
        self.path = []
        self.route = None
        self.special_case = False

    # Movement ----------------------------------------------------------
    def _accept_next_order(self):
        order = None
        for o in list(self.orders):
            if o.status == OrderStatus.COMPLETED:
                self.orders.remove(o)
                continue
            order = o
            break
        if order is None:
            return False
        self.current_order = order
        if order.rider_id != self.rider_id:
            order.assign_to(self.rider_id)
        order.mark_picking_up(self._now())
        self.state = RiderState.PICKING_UP
        self._plan_to(order.res_pos)
        self._record("accepted_order", order.id)
        return True

    def _advance(self, budget):
        moved = 0
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            if (self.x, self.y) == (tx, ty):
                self.path.pop(0)
                continue
            if self.x != tx:
                step = min(budget, abs(tx - self.x))
                self.x += step if tx > self.x else -step
            else:
                step = min(budget, abs(ty - self.y))
                self.y += step if ty > self.y else -step
            budget -= step
            moved += step
            if (self.x, self.y) == (tx, ty):
                self.path.pop(0)
        self.total_distance += moved
        if self.current_order is not None:
            self.current_order.travel_distance += moved

    def _deliver(self, order):
        order.complete(self._now())
        if order in self.orders:
            self.orders.remove(order)
        self.completed_count += 1
        self.reset_to_idle()
        self._record("delivered_to_customer", order.id)
        funds = globalvar.get_value("funds")
        bonus = globalvar.get_value("delivery_bonus") or 0
        if isinstance(funds, int) and bonus:
            globalvar.set_value("funds", funds + bonus)

    def _process_arrivals(self):
        for _ in range(3):
            order = self.current_order
            if order is None:
                return
            if self.state == RiderState.PICKING_UP:
                if self.position == order.res_pos:
                    order.mark_picked_up(self._now())
                    self.state = RiderState.DELIVERING
                    self._plan_to(order.cus_pos)
                    self._record("arrived_at_merchant", order.id)
                    continue
                if not self.path:
                    self._plan_to(order.res_pos)
                return
            if self.state == RiderState.DELIVERING:
                if self.position == order.cus_pos:
                    self._deliver(order)
                    return
                if not self.path:
                    self._plan_to(order.cus_pos)
                return
            return

    def move(self):
        """Advance one tick: accept orders, move, and handle arrivals."""
        self._ticks += 1
        if self.state == RiderState.IDLE:
            if not self._accept_next_order():
                return
        order = self.current_order
        if order is None or order.status == OrderStatus.COMPLETED:
            self.reset_to_idle()
            return
        self._process_arrivals()
        if self.state == RiderState.IDLE:
            return
        self._advance(self.speed)
        self._process_arrivals()

    # Persistence -------------------------------------------------------
    def to_dict(self):
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "order_ids": [o.id for o in self.orders],
            "current_order_id": (self.current_order.id
                                 if self.current_order else None),
            "log": self.log,
            "total_distance": self.total_distance,
            "completed_count": self.completed_count,
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        rider = cls(data["rider_id"], data["x"], data["y"], data["speed"])
        rider.state = data.get("state", RiderState.IDLE)
        rider.special_case = data.get("special_case", False)
        rider.orders = [orders_by_id[i] for i in data.get("order_ids", [])
                        if i in orders_by_id]
        cur = data.get("current_order_id")
        rider.current_order = orders_by_id.get(cur) if cur is not None else None
        rider.log = list(data.get("log", []))
        rider.total_distance = data.get("total_distance", 0)
        rider.completed_count = data.get("completed_count", 0)
        if rider.current_order is None:
            rider.state = RiderState.IDLE
        return rider

    def __repr__(self):
        return "Rider(#%d at (%d,%d) %s)" % (self.rider_id, self.x, self.y,
                                            RiderState.name(self.state))

"""Order management: creation, closest-rider assignment and completion."""

import logging

import globalvar
from order_model import Order, OrderStatus
from path_planner import manhattan
from rider import RiderState

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order, riders=None):
    """Return the best rider for ``order`` (closest rider first).

    Idle riders without pending work are preferred, ranked by distance to the
    restaurant. If no rider is idle, riders compete by the total distance of
    their remaining work plus the distance from their final drop-off to the
    restaurant (load balancing). Ties are broken by fewer pending orders and
    then lower rider id. Returns None if no rider exists.
    """
    if riders is None:
        riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    res = order.res_pos
    idle = [r for r in riders
            if r.state == RiderState.IDLE and not r.has_pending_orders()]
    if idle:
        return min(idle, key=lambda r: (manhattan(r.position, res),
                                        len(r.pending_orders()), r.rider_id))

    def score(r):
        remaining, end = r.remaining_work()
        return (remaining + manhattan(end, res), len(r.pending_orders()),
                r.rider_id)

    return min(riders, key=score)


class OrderManager:
    """Handles order creation, assignment and bookkeeping."""

    assignment_strategy = "closest_rider_first"

    @property
    def orders(self):
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create an order from ``[res_x, res_y, cus_x, cus_y]``."""
        if not isinstance(coords, (list, tuple)) or len(coords) != 4:
            raise ValueError("Order requires exactly four coordinates")
        order = Order([coords[0], coords[1]], [coords[2], coords[3]])
        self.orders.append(order)
        logger.info("Created %r", order)
        return order

    @staticmethod
    def assign_to_rider(order, rider):
        """Assign ``order`` to ``rider``."""
        if order not in rider.orders:
            rider.orders.append(order)
        order.assign_to(rider.rider_id)

    def assign_order(self, order):
        """Assign an order to the best rider; returns the rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is not None:
            self.assign_to_rider(order, rider)
            logger.info("Assigned order #%d to rider #%d", order.id,
                        rider.rider_id)
        return rider

    def pending_orders(self):
        return [o for o in self.orders if o.status == OrderStatus.PENDING]

    def completed_orders(self):
        return [o for o in self.orders if o.status == OrderStatus.COMPLETED]

    def active_orders(self):
        return [o for o in self.orders if o.status != OrderStatus.COMPLETED]

    def dispatch_pending(self):
        """Try to assign every pending order. Returns [(order, rider)]."""
        result = []
        for order in self.pending_orders():
            rider = self.assign_order(order)
            if rider is None:
                break
            result.append((order, rider))
        return result

"""System management: funds, riders, simulated time and persistence."""

import json
import logging

import globalvar
from config import Configuration
from grid_map import GridMap
from order import OrderManager, find_best_rider_for_order
from order_model import Order, OrderStatus
from path_planner import PathPlanner
from rider import Rider, RiderState

logger = logging.getLogger(__name__)


class SystemManager:
    """Core manager for funds, riders and the simulation clock.

    State is stored in :mod:`globalvar` so all modules share it.
    """

    def __init__(self, config=None):
        self.config = config or Configuration()
        self.order_manager = OrderManager()
        self.grid_map = GridMap()
        self.planner = PathPlanner(self.grid_map)

    # Shared state properties ------------------------------------------
    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def riders(self):
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def rider_id_counter(self):
        return globalvar.get_value("rider_id_counter", 0)

    @property
    def running(self):
        return globalvar.get_value("running", False)

    @running.setter
    def running(self, value):
        globalvar.set_value("running", bool(value))

    @property
    def current_time(self):
        return globalvar.get_value("time", 0)

    @property
    def orders(self):
        return self.order_manager.orders

    # Riders -------------------------------------------------------------
    def add_rider(self, x=None, y=None, speed=None):
        """Create a new rider with an auto-incremented id."""
        rider_id = self.rider_id_counter + 1
        globalvar.set_value("rider_id_counter", rider_id)
        rider = Rider(rider_id, x, y, speed)
        self.riders.append(rider)
        logger.info("Rider #%d created", rider_id)
        return rider

    def remove_rider(self, rider_id):
        """Remove an idle rider without pending orders. Returns True on success."""
        for r in self.riders:
            if r.rider_id == rider_id:
                if r.state != RiderState.IDLE or r.has_pending_orders():
                    return False
                self.riders.remove(r)
                return True
        return False

    def get_rider(self, rider_id):
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    def recruit_if_possible(self):
        """Recruit a rider when funds reach the threshold. Returns it or None."""
        if (self.funds >= self.config.recruit_threshold
                and self.funds >= self.config.rider_recruit_cost
                and len(self.riders) < self.config.max_riders):
            self.funds = self.funds - self.config.rider_recruit_cost
            return self.add_rider()
        return None

    # Orders -------------------------------------------------------------
    def place_order(self, coords):
        """Create and dispatch an order.

        Returns a dict with keys ``order``, ``rider``, ``new_rider`` and
        ``route``.
        """
        order = self.order_manager.create_order(coords)
        self.funds = self.funds + self.config.order_income
        new_rider = self.recruit_if_possible()
        rider = find_best_rider_for_order(order)
        route = None
        if rider is not None:
            _, start = rider.remaining_work()
            self.order_manager.assign_to_rider(order, rider)
            route = self.planner.plan_multi(start, [order.res_pos,
                                                    order.cus_pos],
                                            rider.speed)
        return {"order": order, "rider": rider, "new_rider": new_rider,
                "route": route}

    # Simulation ---------------------------------------------------------
    def tick(self, count=1):
        """Advance the simulation ``count`` ticks; returns event descriptions."""
        events = []
        for _ in range(max(0, int(count))):
            globalvar.set_value("time", self.current_time + 1)
            new_rider = self.recruit_if_possible()
            if new_rider is not None:
                events.append("t=%d: new rider #%d recruited"
                              % (self.current_time, new_rider.rider_id))
            for order, rider in self.order_manager.dispatch_pending():
                events.append("t=%d: pending order #%d assigned to rider #%d"
                              % (self.current_time, order.id, rider.rider_id))
            for rider in list(self.riders):
                before = len(rider.log)
                try:
                    rider.move()
                except Exception:  # pragma: no cover - recovery path
                    logger.exception("Rider #%d failed to move", rider.rider_id)
                    self.check_consistency(repair=True)
                for entry in rider.log[before:]:
                    events.append("t=%d: rider #%d %s (order #%s) at (%d,%d)"
                                  % (entry["time"], rider.rider_id,
                                     entry["event"], entry["order_id"],
                                     entry["x"], entry["y"]))
        return events

    def check_consistency(self, repair=True):
        """Check (and optionally repair) system state. Returns issue list."""
        issues = []
        if not isinstance(self.funds, int):
            issues.append("funds not an integer")
            if repair:
                try:
                    self.funds = int(self.funds)
                except (TypeError, ValueError):
                    self.funds = 0
        rider_ids = set()
        for r in self.riders:
            rider_ids.add(r.rider_id)
            if r.state not in (0, 1, 2):
                issues.append("rider #%d invalid state" % r.rider_id)
                if repair:
                    r.reset_to_idle()
            if r.state != RiderState.IDLE and (
                    r.current_order is None
                    or r.current_order.status == OrderStatus.COMPLETED):
                issues.append("rider #%d busy without active order"
                              % r.rider_id)
                if repair:
                    r.reset_to_idle()
            done = [o for o in r.orders if o.status == OrderStatus.COMPLETED]
            if done:
                issues.append("rider #%d holds completed orders" % r.rider_id)
                if repair:
                    for o in done:
                        r.orders.remove(o)
        held = {id(o) for r in self.riders for o in r.orders}
        for o in self.orders:
            if o.status == OrderStatus.COMPLETED:
                continue
            if o.rider_id is not None and (o.rider_id not in rider_ids
                                           or id(o) not in held):
                issues.append("order #%d assigned to missing rider" % o.id)
                if repair:
                    o.rider_id = None
                    o.status = OrderStatus.PENDING
        return issues

    # Persistence --------------------------------------------------------
    def save_state(self, path=None):
        """Save configuration, riders and order history to JSON."""
        path = path or self.config.data_file
        data = {
            "config": self.config.to_dict(),
            "funds": self.funds,
            "time": self.current_time,
            "rider_id_counter": self.rider_id_counter,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.orders],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

    def load_state(self, path=None):
        """Load state from a JSON file previously written by save_state."""
        path = path or self.config.data_file
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        for key, value in data.get("config", {}).items():
            if key in Configuration.DEFAULTS:
                self.config.values[key] = value
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        by_id = {o.id: o for o in orders}
        globalvar.set_value("orders", orders)
        globalvar.set_value("funds", int(data.get("funds", 0)))
        globalvar.set_value("time", int(data.get("time", 0)))
        globalvar.set_value("rider_id_counter",
                            int(data.get("rider_id_counter", 0)))
        globalvar.set_value("riders", [Rider.from_dict(d, by_id)
                                       for d in data.get("riders", [])])
        Order.reset_counter(max(by_id) + 1 if by_id else 1)
        self.check_consistency(repair=True)
        return path


def init_system(config=None):
    """Initialise global system state and return a SystemManager.

    Sets: riders=[], rider_id_counter=0, running=True, funds, time=0,
    orders=[].
    """
    if config is None:
        config = Configuration()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", int(config.initial_funds))
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("config", config)
    globalvar.set_value("default_speed", int(config.default_speed))
    globalvar.set_value("spawn_position", (int(config.spawn_x),
                                           int(config.spawn_y)))
    globalvar.set_value("delivery_bonus", int(config.delivery_bonus))
    Order.reset_counter()
    manager = SystemManager(config)
    globalvar.set_value("system", manager)
    return manager

        """Return the ASCII map.

        '#' streets, '.' blocks, 'S' restaurant, 'C' customer, 'R' rider,
        '-', '|', '+' route segments.

        if orders is None:
            orders = [o for o in self.system.orders
                      if o.status != OrderStatus.COMPLETED]
        if riders is None:
            riders = self.system.riders
        grid = [["#" if (r % 2 == 0 or c % 2 == 0) else "."
                 for c in range(CANVAS)] for r in range(CANVAS)]
        if route_points:
            pts = [(_idx(x), _idx(y)) for x, y in route_points]
            for (c1, r1), (c2, r2) in zip(pts, pts[1:]):
                if r1 == r2:
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        grid[r1][c] = "-"
                elif c1 == c2:
                    for r in range(min(r1, r2), max(r1, r2) + 1):
                        grid[r][c1] = "|"
            for c, r in pts:
                grid[r][c] = "+"
        for o in orders:
            grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = "S"
            grid[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = "C"

"""Command parsing, shortcuts, history and input validation."""

import re

from grid_map import MAP_SIZE

SHORTCUTS = {
    "s": "start", "n": "start", "new": "start", "start": "start",
    "st": "status", "status": "status",
    "o": "orders", "orders": "orders",
    "r": "riders", "riders": "riders",
    "m": "map", "map": "map",
    "t": "tick", "tick": "tick",
    "h": "help", "?": "help", "help": "help",
    "hist": "history", "history": "history",
    "c": "check", "check": "check",
    "save": "save", "load": "load",
    "q": "quit", "quit": "quit", "exit": "quit",
}

HELP_TEXT = """Commands (shortcut in brackets):
  start [rx ry cx cy] (s)  place an order: restaurant x,y + customer x,y (0-800)
  status (st)              system funds and rider states
  orders (o)               all order details
  riders (r)               rider status-change history
  map (m)                  ASCII map of riders and active orders
  tick [n] (t)             advance the simulation n ticks (empty line = 1 tick)
  check (c)                run a system consistency check
  save [file] / load [file]  persist / restore state as JSON
  history (hist), !n       show command history / repeat command n
  help (h, ?)              this help
  quit / exit (q)          leave the program"""


class InputManager:
    """Parses commands and validates user input."""

    def __init__(self):
        self.command_history = []

    def parse(self, line):
        """Return (command, args). Empty input returns (None, [])."""
        line = (line or "").strip()
        if not line:
            return None, []
        if line.startswith("!"):
            try:
                n = int(line[1:])
                line = self.command_history[n - 1]
            except (ValueError, IndexError):
                raise ValueError("No such history entry: %s" % line)
        self.command_history.append(line)
        if len(self.command_history) > 1000:
            del self.command_history[:-1000]
        parts = line.split(None, 1)
        word = parts[0].lower()
        if word not in SHORTCUTS:
            raise ValueError("Unknown command '%s'. Type 'help'." % word)
        rest = parts[1] if len(parts) > 1 else ""
        return SHORTCUTS[word], rest.split() if rest else []

    @staticmethod
    def parse_coordinates(text):
        """Parse four integer coordinates (0-800) from text.

        Accepts space or comma separated values, optionally in brackets.
        """
        cleaned = re.sub(r"[\[\]()]", " ", text or "")
        tokens = [t for t in re.split(r"[\s,]+", cleaned) if t]
        if len(tokens) != 4:
            raise ValueError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y")
        values = []
        for tok in tokens:
            try:
                v = int(tok)
            except ValueError:
                raise ValueError("'%s' is not an integer" % tok)
            if v < 0 or v > MAP_SIZE:
                raise ValueError("Coordinate %d out of range 0-%d"
                                 % (v, MAP_SIZE))
            values.append(v)
        return values

    @staticmethod
    def parse_count(args, default=1):
        """Parse an optional positive integer count."""
        if not args:
            return default
        try:
            n = int(args[0])
        except ValueError:
            raise ValueError("Tick count must be an integer")
        if n < 1 or n > 100000:
            raise ValueError("Tick count must be between 1 and 100000")
        return n

    def history_text(self):
        if not self.command_history:
            return "History is empty."
        return "\n".join("%3d  %s" % (i + 1, c)
                         for i, c in enumerate(self.command_history))

"""Command-line entry point for the food delivery dispatch simulator."""

import argparse
import logging
import os
import sys

import globalvar
from config import Configuration
from display import DisplayManager
from input_manager import HELP_TEXT, InputManager
from system import init_system

try:  # command history / line editing where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

logger = logging.getLogger("dispatch")


class Application:
    """Main loop: each iteration is one simulated tick."""

    def __init__(self, config_path=None, load_path=None):
        config = Configuration(config_path if config_path and
                               os.path.exists(config_path) else None)
        globalvar._init()
        self.system = init_system(config)
        self.display = DisplayManager(self.system)
        self.inputs = InputManager()
        if load_path:
            self.system.load_state(load_path)
            print("State loaded from %s" % load_path)

    def _start(self, args):
        text = " ".join(args) if args else input(
            "Enter restaurant_x restaurant_y customer_x customer_y (0-800): ")
        coords = self.inputs.parse_coordinates(text)
        result = self.system.place_order(coords)
        print(self.display.start_report(result))
        return 1

    def handle(self, line):
        """Execute one command line; returns number of ticks to advance."""
        cmd, args = self.inputs.parse(line)
        if cmd is None:
            return 1
        if cmd == "start":
            return self._start(args)
        if cmd == "status":
            print(self.display.status_text())
        elif cmd == "orders":
            print(self.display.orders_text())
        elif cmd == "riders":
            print(self.display.riders_text())
        elif cmd == "map":
            print(self.display.render_map())
        elif cmd == "tick":
            return self.inputs.parse_count(args)
        elif cmd == "help":
            print(HELP_TEXT)
        elif cmd == "history":
            print(self.inputs.history_text())
        elif cmd == "check":
            issues = self.system.check_consistency(repair=True)
            print("System consistent." if not issues
                  else "Repaired: " + "; ".join(issues))
        elif cmd == "save":
            print("Saved to %s" % self.system.save_state(
                args[0] if args else None))
        elif cmd == "load":
            print("Loaded from %s" % self.system.load_state(
                args[0] if args else None))
            return 0
        elif cmd == "quit":
            self.system.running = False
            return 0
        return 1

    def run(self):
        print("Food Delivery Order Dispatch Simulator (8x8 grid, 800x800)")
        print(HELP_TEXT)
        if not self.system.riders:
            rider = self.system.recruit_if_possible()
            if rider:
                print("Rider #%d recruited (funds now %d)"
                      % (rider.rider_id, self.system.funds))
        while self.system.running:
            try:
                line = input("[t=%d funds=%d] > " % (self.system.current_time,
                                                     self.system.funds))
            except (EOFError, KeyboardInterrupt):
                print()
                line = "quit"
            try:
                ticks = self.handle(line)
            except ValueError as exc:
                print("Error: %s" % exc)
                ticks = 0
            except (OSError, KeyError) as exc:
                print("Error: %s" % exc)
                ticks = 0
            except Exception as exc:  # recovery mechanism
                logger.exception("Unexpected error")
                print("Unexpected error: %s (system state repaired)" % exc)
                self.system.check_consistency(repair=True)
                ticks = 0
            if self.system.running and ticks:
                for event in self.system.tick(ticks):
                    print("  " + event)
        print("Goodbye.")


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--config", default="config.json",
                        help="JSON configuration file")
    parser.add_argument("--load", default=None, help="load saved state")
    parser.add_argument("--debug", action="store_true", help="debug logging")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if args.debug else logging.WARNING,
                        format="%(asctime)s %(levelname)s %(name)s: %(message)s")
    Application(args.config, args.load).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

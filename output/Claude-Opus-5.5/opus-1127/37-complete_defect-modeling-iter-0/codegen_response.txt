"""Core data models for the lottery system."""

from __future__ import annotations

from typing import List, Optional


class User:
    """A lottery participant."""

    def __init__(self, name: str, department: str):
        self.name = (name or "").strip()
        self.department = (department or "").strip()
        self.is_valid = True

    def __repr__(self) -> str:
        return f"User(name={self.name!r}, department={self.department!r})"

    def __eq__(self, other) -> bool:
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self) -> int:
        return hash((self.name, self.department))


class Prize:
    """A prize to be drawn in one lottery round."""

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ):
        self.name = (name or "").strip()
        self.quantity = int(quantity)
        self.avoid_groups: List[str] = list(avoid_groups) if avoid_groups else []
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        self.associated_drawer: Optional[str] = drawer or None

    @property
    def level(self) -> str:
        if " - " in self.name:
            return self.name.split(" - ", 1)[0].strip()
        return ""

    def __repr__(self) -> str:
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity}, "
            f"avoid_groups={self.avoid_groups!r}, associated_drawer={self.associated_drawer!r})"
        )

class ListManager:
    """Loads and validates participant lists."""

    def __init__(self):
        self.users: List[User] = []
        self.load_mode: str = ""
        self.source_file_path: str = ""

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        self.load_mode = "default"
        self.source_file_path = ""
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        users, errors = self._load_from_path(path)
        self.load_mode = "custom"
        self.source_file_path = str(path)
        return users, errors

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        path = str(path).strip().strip('"').strip("'")
        if not path:
            return [], ["File path is empty"]
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"Failed to read file: {exc}"]
        except OSError as exc:
            return [], [f"Failed to read file: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # --------------------------------------------------------------- validation
    @staticmethod
    def validate_users(users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users or []:
            if user.name in seen:
                user.is_valid = False
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    @staticmethod
    def get_group_info(users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users or []:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and round sequence."""

    def __init__(self):
        self.prizes: List[Prize] = []
        self.is_preset = False

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        self.is_preset = True
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if isinstance(avoid_groups, str):
            avoid_groups = [g.strip() for g in avoid_groups.replace("，", ",").split(",")]
        groups: List[str] = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer: Optional[str] = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        prize = Prize(name, qty, groups, drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def set_avoid_groups(self, index: int, groups: List[str]) -> bool:
        if not 0 <= index < len(self.prizes):
            return False
        cleaned: List[str] = []
        for g in groups:
            g = g.strip()
            if g and g not in cleaned:
                cleaned.append(g)
        self.prizes[index].avoid_groups = cleaned
        return True

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self.prizes)
        try:
            positions = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(positions) != n or sorted(positions) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[p - 1] for p in positions]
        return True

"""Lottery rule engine and drawing logic."""

from __future__ import annotations

import random
from typing import List, Optional, Set

from lottery_system.models import Prize, User

SPECIAL_PERSON = "Li Bin"
MONITORED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(text: Optional[str]) -> str:
    return " ".join((text or "").strip().lower().split())


def is_protected_group(department: str) -> bool:
    d = _norm(department).replace("_", " ").replace("-", " ")
    d = " ".join(d.split())
    if d.endswith(" group"):
        d = d[: -len(" group")].strip()
    return d in ("cheng ru", "chengru")


class LotteryManager:
    """Applies rules and performs draws."""

    def __init__(self, users, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users or [])
        self.rng = rng or random.Random()
        self.winner_names: Set[str] = set()
        self.last_two_draw_results: List[bool] = []
        self.fallback_used_last = False

    # ------------------------------------------------------------- rule checks
    @staticmethod
    def is_special_drawer(prize: Prize) -> bool:
        return _norm(prize.associated_drawer) == _norm(SPECIAL_PERSON)

    def is_fallback_triggered(self, prize: Prize) -> bool:
        if _norm(prize.associated_drawer) != _norm(FALLBACK_TARGET):
            return False
        if len(self.last_two_draw_results) < 2:
            return False
        return not any(self.last_two_draw_results[-2:])

    def get_rule_description(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        if self.is_fallback_triggered(prize):
            parts.append(
                f"FALLBACK: pool limited to '{PROTECTED_GROUP}' members "
                f"(previous two draws by {' and '.join(MONITORED_DRAWERS)} had no such winners)"
            )
        elif self.is_special_drawer(prize):
            parts.append(f"Special drawer {SPECIAL_PERSON}: avoidance groups are NOT applied")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")
        if prize.associated_drawer:
            parts.append(f"Drawer: {prize.associated_drawer}")
        return "; ".join(parts)

    # ------------------------------------------------------------------- pools
    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if u.name not in self.winner_names]
        if self.is_fallback_triggered(prize):
            protected = [u for u in base if is_protected_group(u.department)]
            if protected:
                return protected
        if self.is_special_drawer(prize):
            return base
        avoid = {_norm(g) for g in prize.avoid_groups}
        return [u for u in base if _norm(u.department) not in avoid]

    # ------------------------------------------------------------------- draws
    def draw_winner(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = self.rng.choice(pool)
        self.winner_names.add(winner.name)
        return winner

    def draw_round(self, prize: Prize) -> List[User]:
        winners = []
        for _ in range(prize.quantity):
            w = self.draw_winner(prize)
            if w is None:
                break
            winners.append(w)
        self.record_round_result(prize, winners)
        return winners

    def record_round_result(self, prize: Prize, round_winners) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in MONITORED_DRAWERS}:
            contains = any(is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_TARGET):
            self.fallback_used_last = not any(self.last_two_draw_results[-2:]) and len(
                self.last_two_draw_results
            ) >= 2
            self.last_two_draw_results = []
        else:
            self.last_two_draw_results = []

"""Winner records, summaries and export."""

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List, Optional, Tuple


class Winner:
    def __init__(self, user, prize, order: int):
        self.user = user
        self.prize = prize
        self.order = order
        self.timestamp = datetime.now()

    @property
    def prize_name(self) -> str:
        return getattr(self.prize, "name", str(self.prize))

    def __repr__(self) -> str:
        return f"Winner({self.user.name!r}, {self.prize_name!r})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    @property
    def total_winners(self) -> int:
        return len(self.winners)

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self.winners.append(Winner(user, prize, len(self.winners) + 1))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes or []:
            result.setdefault(getattr(p, "name", str(p)), [])
        for w in self.winners:
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes, directory: Optional[str] = None) -> Tuple[bool, str]:
        if not self.winners:
            return False, "No winners to export"
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(directory or os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from __future__ import annotations

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ask(prompt: str, default: str = "") -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        return default


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users: List[User] = []
        self.result_manager = ResultManager()
        self.state = "IDLE"
        self.current_round_index = 0

    # ---------------------------------------------------------------- lists
    def _accept_list(self, users, errors):
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Validation issues detected:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate name: {d.name} ({d.department})")
            choice = ask("Remove invalid data and continue (r) or terminate upload (t)? [r/t]: ", "t").lower()
            if choice != "r":
                print("Upload terminated. List unchanged.")
                return
        if not valid:
            print("No valid participants loaded.")
            return
        self.users = valid
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._accept_list(users, errors)

    def load_custom(self):
        path = ask("Enter txt file path (each line 'Name,Department Group'): ")
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            print("No valid data loaded.")
            return
        self._accept_list(users, errors)

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print(f"Valid groups ({len(info)}):")
        for g, c in info.items():
            print(f"  {g}: {c} people")

    # --------------------------------------------------------------- prizes
    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            avoid = ", ".join(p.avoid_groups) or "None"
            print(f"  {i}. {p.name} x{p.quantity} | Drawer: {p.associated_drawer or 'None'} | Avoid: {avoid}")

    def load_preset_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        self.show_prizes()

    def add_custom_prize(self):
        name = ask("Prize name: ")
        qty = ask("Quantity: ")
        drawer = ask("Associated lottery person (optional): ")
        avoid = self._select_groups()
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer or None)
            print("Prize added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def _select_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            raw = ask("Select avoidance groups by number (comma separated, blank for none): ")
        else:
            raw = ask("Enter avoidance group names (comma separated, blank for none): ")
        result = []
        for token in raw.replace("，", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and groups and 1 <= int(token) <= len(groups):
                result.append(groups[int(token) - 1])
            elif not token.isdigit():
                result.append(token)
            else:
                print(f"Ignored invalid selection: {token}")
        return result

    def configure_avoidance(self):
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        idx = ask("Select prize number: ")
        if not idx.isdigit():
            print("Invalid number.")
            return
        groups = self._select_groups()
        if self.prize_manager.set_avoid_groups(int(idx) - 1, groups):
            print("Avoidance groups updated.")
        else:
            print("Invalid prize number.")

    def reorder(self):
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = ask("Enter new order as serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Order updated.")
            self.show_prizes()
        else:
            print("Invalid order.")

    # -------------------------------------------------------------- lottery
    def start_lottery(self):
        prizes = self.prize_manager.get_prizes()
        if not self.users:
            print("Please load a participant list first.")
            return
        if not prizes:
            print("Please configure prizes first.")
            return
        engine = LotteryManager(self.users)
        self.result_manager = ResultManager()
        self.state = "RUNNING"
        print("Commands during draw: Enter=draw next, 'auto'=draw remaining, "
              "'pause', 'continue', 'terminate'.")
        for r, prize in enumerate(prizes, 1):
            self.current_round_index = r
            print(f"\n=== Round {r}: {prize.name} | Quantity: {prize.quantity} ===")
            print("Rules: " + engine.get_rule_description(prize))
            round_winners = []
            auto = False
            terminated = False
            i = 0
            while i < prize.quantity:
                if not auto:
                    cmd = ask(f"[{ordinal(i + 1)} draw] > ", "auto").lower()
                    if cmd == "pause":
                        self.state = "PAUSED"
                        print("Lottery paused. Type 'continue' to resume or 'terminate' to skip round.")
                        while True:
                            c = ask("(paused) > ", "terminate").lower()
                            if c == "continue":
                                self.state = "RUNNING"
                                print("Lottery resumed.")
                                break
                            if c == "terminate":
                                cmd = "terminate"
                                break
                            print("Lottery is paused.")
                        if cmd != "terminate":
                            continue
                    if cmd == "terminate":
                        self.state = "RUNNING"
                        print("Current round terminated and skipped.")
                        terminated = True
                        break
                    if cmd == "auto":
                        auto = True
                    elif cmd not in ("", "continue", "draw"):
                        print("Unknown command.")
                        continue
                print(f"Currently drawing the {ordinal(i + 1)} winner...")
                winner = engine.draw_winner(prize)
                if winner is None:
                    print("No eligible participants left for this prize.")
                    break
                round_winners.append(winner)
                self.result_manager.add_winner(winner, prize)
                print(f"Congratulations {winner.name} ({winner.department}) for winning {prize.name}")
                i += 1
            engine.record_round_result(prize, round_winners)
            print(f"--- Round {r} results{' (terminated)' if terminated else ''} ---")
            if not round_winners:
                print("  No winners.")
            for n, w in enumerate(round_winners, 1):
                print(f"  {n}. {w.name} ({w.department})")
        self.state = "FINISHED"
        print("\nAll rounds completed.")
        self.show_results()

    def show_results(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        print("===== Complete Winner List =====")
        for name, ws in grouped.items():
            for w in ws:
                print(f"  {w.user.name} | {w.user.department} | {name}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ----------------------------------------------------------------- menu
    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.view_groups,
            "4": self.load_preset_prizes, "5": self.add_custom_prize,
            "6": self.configure_avoidance, "7": self.reorder, "8": self.show_prizes,
            "9": self.start_lottery, "10": self.show_results, "11": self.export,
        }
        while True:
            print("\n===== Team Building Smart Lottery System =====")
            print(" 1. Load default list        2. Upload custom txt list")
            print(" 3. View groups              4. Load preset prizes")
            print(" 5. Add custom prize         6. Configure avoidance groups")
            print(" 7. Reorder prizes           8. View prizes")
            print(" 9. Start lottery           10. View all results")
            print("11. Export results           0. Exit")
            choice = ask("Select: ", "0")
            if choice == "0":
                print("Goodbye!")
                return
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid choice.")


def main():
    LotteryApp().run()


if __name__ == "__main__":
    main()

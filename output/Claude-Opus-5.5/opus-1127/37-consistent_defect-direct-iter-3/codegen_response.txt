from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)

    def __str__(self) -> str:
        return f"{self.name} x{self.quantity}"

import os
import re
from collections import OrderedDict
from typing import Dict, List, Tuple

from lottery_system.models import User

DEFAULT_LIST: List[Tuple[str, str]] = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Chen Qixian", "Marketing Department"),
    ("Jin Duo", "Operations Department"),
    ("Ma Jiaqi", "Operations Department"),
    ("Li Bin", "Operations Department"),
    ("Liu Yang", "Cheng Ru Group"),
    ("He Ping", "Cheng Ru Group"),
    ("Song Yu", "Cheng Ru Group"),
    ("Tang Ning", "Human Resources Department"),
    ("Feng Lei", "Human Resources Department"),
    ("Xu Jing", "Finance Department"),
]

_SEPARATOR = re.compile(r"[,;，；]")


class ListManager:
    """Manages the participant list."""

    def __init__(self):
        self.users: List[User] = []

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users = [User(name, dept) for name, dept in DEFAULT_LIST]
        return users, []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        if not path or not os.path.isfile(path):
            errors.append(f"File Error: file not found '{path}'")
            return users, errors
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.readlines()
        except (OSError, UnicodeDecodeError) as e:
            errors.append(f"File Error: cannot read '{path}': {e}")
            return users, errors

        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in _SEPARATOR.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            if u.name in seen:
                duplicates.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages prizes and their draw order."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self.prizes}
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Prize quantity must be an integer")
        if qty <= 0:
            raise ValueError("Prize quantity must be positive")
        groups = [g.strip() for g in (avoid_groups or []) if g and g.strip()]
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        if drawer == "":
            drawer = None
        prize = Prize(name.strip(), qty, groups, drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
FALLBACK_WATCHED = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"


def is_cheng_ru(department: str) -> bool:
    return "cheng ru" in (department or "").lower()


class LotteryManager:
    """Applies filtering rules and draws winners."""

    def __init__(self, users, result_manager=None):
        self.users: List[User] = list(users)
        self.result_manager = result_manager
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []

    # ---------- rules ----------
    def _is_winner(self, user: User) -> bool:
        if user.name in self.winner_names:
            return True
        if self.result_manager is not None and self.result_manager.has_won(user):
            return True
        return False

    def _is_forced(self, prize: Prize) -> bool:
        return (
            prize.associated_drawer == FALLBACK_DRAWER
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _normal_pool(self, prize: Prize) -> List[User]:
        avoid = set(prize.avoid_groups or [])
        apply_avoid = prize.associated_drawer != SPECIAL_DRAWER
        return [
            u for u in self.users
            if not self._is_winner(u) and not (apply_avoid and u.department in avoid)
        ]

    def _get_eligible_pool(self, prize) -> List[User]:
        if self._is_forced(prize):
            return [u for u in self.users if not self._is_winner(u) and is_cheng_ru(u.department)]
        return self._normal_pool(prize)

    def _candidate_pool(self, prize: Prize) -> List[User]:
        pool = self._get_eligible_pool(prize)
        if pool:
            return pool
        if self._is_forced(prize):
            pool = self._normal_pool(prize)
            if pool:
                return pool
        # Avoided groups remain eligible but are drawn after the others.
        return [u for u in self.users if not self._is_winner(u)]

    def rule_description(self, prize: Prize) -> str:
        parts = []
        if prize.associated_drawer:
            parts.append(f"Drawer: {prize.associated_drawer}")
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("Special rule: only previous winners are excluded (avoidance groups ignored)")
        else:
            parts.append("Previous winners excluded")
            if prize.avoid_groups:
                parts.append("Avoidance groups (drawn last): " + ", ".join(prize.avoid_groups))
        if self._is_forced(prize):
            parts.append("Fallback active: pool limited to Cheng Ru group members")
        return "; ".join(parts)

    # ---------- drawing ----------
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._candidate_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winner_names.add(user.name)
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return user

    def finish_round(self, prize: Prize, winners: List[User]) -> None:
        drawer = prize.associated_drawer
        if drawer in FALLBACK_WATCHED:
            self.last_two_draw_results.append(any(is_cheng_ru(u.department) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == FALLBACK_DRAWER:
            self.last_two_draw_results = []

    def draw_round(self, prize: Prize) -> List[User]:
        winners = []
        for _ in range(prize.quantity):
            u = self.draw_one(prize)
            if u is None:
                break
            winners.append(u)
        self.finish_round(prize, winners)
        return winners

"""Result management and export."""

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User

_LEVEL_KEYWORDS = [
    ("grand", 0), ("special", 0), ("first", 1), ("second", 2),
    ("third", 3), ("fourth", 4), ("fifth", 5), ("sixth", 6),
]


def prize_level(prize_name: str) -> int:
    lower = prize_name.lower()
    for kw, rank in _LEVEL_KEYWORDS:
        if kw in lower:
            return rank
    return 100


class Winner:
    def __init__(self, user: User, prize: Prize, time: datetime = None):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()

    def __repr__(self):
        return f"Winner({self.user.name}, {self.prize.name}, {self.time:%H:%M:%S})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def has_won(self, user: User) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def add_winner(self, user, prize) -> bool:
        if self.has_won(user):
            return False
        self.winners.append(Winner(user, prize))
        return True

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        indexed = list(enumerate(prizes))
        indexed.sort(key=lambda t: (prize_level(t[1].name), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, p in indexed:
            if p.name in result:
                continue
            ws = [w for w in self.winners if w.prize.name == p.name]
            ws.sort(key=lambda w: w.time)
            result[p.name] = ws
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for pname, ws in grouped.items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, pname,
                                         w.time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None

    def _set_users(self, users, errors):
        for e in errors:
            print("  " + e)
        valid, dups = self.list_manager.validate_users(users)
        for d in dups:
            print(f"  Duplicate name: {d.name} ({d.department})")
        if errors or dups:
            ans = input("Invalid data found. [r]emove invalid data and continue / [t]erminate upload: ").strip().lower()
            if ans.startswith("t"):
                print("Upload terminated.")
                return
        self.users = valid
        self.list_manager.users = valid
        self.lottery = None
        print(f"Loaded {len(valid)} valid participants.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._set_users(users, errors)

    def load_custom(self):
        path = input("Path to txt list: ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            return
        self._set_users(users, errors)

    def view_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        for dept, cnt in info.items():
            print(f"  {dept}: {cnt}")

    def load_prizes(self):
        for e in self.prize_manager.load_default_prizes():
            print("  " + e)
        self.show_prizes()

    def add_prize(self):
        name = input("Prize name: ").strip()
        qty = input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = input("Avoidance groups (numbers or names, comma separated, empty for none): ").strip()
        avoid = []
        for tok in [t.strip() for t in raw.split(",") if t.strip()]:
            if tok.isdigit() and 1 <= int(tok) <= len(groups):
                avoid.append(groups[int(tok) - 1])
            else:
                avoid.append(tok)
        drawer = input("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print("Prize added.")
        except ValueError as e:
            print(f"Error: {e}")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            avoid = ", ".join(p.avoid_groups) if p.avoid_groups else "None"
            print(f"  {i}. {p.name} x{p.quantity} | avoid: {avoid} | drawer: {p.associated_drawer or 'None'}")

    def reorder(self):
        self.show_prizes()
        raw = input("New order (e.g. 3,1,2): ").strip()
        try:
            order = [int(x) for x in raw.replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Order updated.")
            self.show_prizes()
        else:
            print("Invalid order.")

    def run_lottery(self):
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users, self.result_manager)
        lm = self.lottery
        for rnd, prize in enumerate(prizes, 1):
            if self.result_manager.get_winners_by_prize([prize])[prize.name]:
                print(f"Round {rnd} ({prize.name}) already drawn, skipped.")
                continue
            print(f"\n=== Round {rnd}: {prize.name} | Quantity: {prize.quantity} ===")
            print("Rules: " + lm.rule_description(prize))
            winners = []
            terminated = False
            for i in range(prize.quantity):
                print(f"Currently drawing the {ordinal(i + 1)} winner")
                while True:
                    cmd = input("[Enter]=draw, pause, terminate: ").strip().lower()
                    if cmd == "pause":
                        print("Lottery paused. Type 'continue' to resume or 'terminate' to end round.")
                        while True:
                            c2 = input("> ").strip().lower()
                            if c2 == "continue":
                                print("Resumed.")
                                break
                            if c2 == "terminate":
                                terminated = True
                                break
                        if terminated:
                            break
                        continue
                    if cmd == "terminate":
                        terminated = True
                    break
                if terminated:
                    print("Current round terminated.")
                    break
                user = lm.draw_one(prize)
                if user is None:
                    print("No eligible participants left.")
                    break
                winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            lm.finish_round(prize, winners)
            print(f"--- Round {rnd} results ---")
            for i, u in enumerate(winners, 1):
                print(f"  {i}. {u.name} ({u.department}) - {prize.name}")
            if not winners:
                print("  (no winners)")
        self.show_summary()

    def show_summary(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Winner Summary =====")
        for pname, ws in grouped.items():
            print(f"[{pname}]")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {pname} @ {w.time:%H:%M:%S}")
            if not ws:
                print("  (none)")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.view_groups),
            "4": ("Load preset prizes", self.load_prizes),
            "5": ("Add custom prize", self.add_prize),
            "6": ("View prizes", self.show_prizes),
            "7": ("Reorder prizes", self.reorder),
            "8": ("Start lottery", self.run_lottery),
            "9": ("Show summary", self.show_summary),
            "10": ("Export results", self.export),
        }
        while True:
            print("\n==== Team Building Smart Lottery ====")
            for k, (label, _) in actions.items():
                print(f" {k}. {label}")
            print(" 0. Exit")
            try:
                choice = input("Choose: ").strip()
            except EOFError:
                break
            if choice == "0":
                break
            act = actions.get(choice)
            if act:
                try:
                    act[1]()
                except EOFError:
                    break
            else:
                print("Invalid choice.")


def main():
    LotteryApp().run()


if __name__ == "__main__":
    main()

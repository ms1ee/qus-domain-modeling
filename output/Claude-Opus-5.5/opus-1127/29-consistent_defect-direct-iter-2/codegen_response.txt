"""System initialisation and rider recruitment."""

import globalvar

INITIAL_FUNDS = 1000
RECRUIT_COST = 500
RECRUIT_THRESHOLD = 300


def init_system(funds=INITIAL_FUNDS):
    """Initialise all core system parameters."""
    globalvar._init()
    globalvar.set_value("money", funds)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("orders", [])
    globalvar.set_value("time", 0)


def _ensure_init():
    if globalvar.get_value("riders") is None:
        globalvar.set_value("riders", [])
    if globalvar.get_value("money") is None:
        globalvar.set_value("money", INITIAL_FUNDS)
    if globalvar.get_value("rider_id_counter") is None:
        globalvar.set_value("rider_id_counter", 0)
    if globalvar.get_value("orders") is None:
        globalvar.set_value("orders", [])
    if globalvar.get_value("time") is None:
        globalvar.set_value("time", 0)


def recruit_rider(force=False):
    """Recruit a new rider if funds allow (>= 300). Returns the rider or None."""
    from rider import Rider
    _ensure_init()
    money = globalvar.get_value("money")
    if not force and money < RECRUIT_THRESHOLD:
        return None
    counter = globalvar.get_value("rider_id_counter") + 1
    globalvar.set_value("rider_id_counter", counter)
    globalvar.set_value("money", money - RECRUIT_COST)
    rider = Rider(counter)
    globalvar.get_value("riders").append(rider)
    return rider


def pending_order_count():
    """Number of orders not yet accepted/started by any rider."""
    _ensure_init()
    count = 0
    for o in globalvar.get_value("orders"):
        if o.status in ("pending", "assigned"):
            count += 1
    return count


def idle_rider_count():
    """Number of idle riders with no outstanding work."""
    _ensure_init()
    return sum(1 for r in globalvar.get_value("riders")
               if r.state == 0 and not r.active_orders())


def auto_recruit():
    """Recruit a rider if funds >= 300 and pending orders exceed idle riders."""
    _ensure_init()
    if (globalvar.get_value("money") >= RECRUIT_THRESHOLD
            and pending_order_count() > idle_rider_count()):
        return recruit_rider()
    return None

"""Grid based path planning along streets (grid lines every 100 units)."""

GRID = 100


class PathPlanner:
    """Street-preferring Manhattan path planner."""

    @staticmethod
    def next_axis(x, y, tx, ty):
        """Choose which axis to move along next, or None if arrived."""
        if x == tx and y == ty:
            return None
        # Special case: directly above/below target -> vertical only.
        if x == tx:
            return "y"
        if y == ty:
            return "x"
        # On a horizontal street -> move horizontally first.
        if y % GRID == 0:
            return "x"
        return "y"

    @classmethod
    def step(cls, x, y, tx, ty, budget):
        """Move from (x, y) toward (tx, ty) using up to ``budget`` units.

        Returns (new_x, new_y, remaining_budget).
        """
        while budget > 0:
            axis = cls.next_axis(x, y, tx, ty)
            if axis is None:
                break
            if axis == "x":
                d = tx - x
                mv = min(abs(d), budget)
                x += mv if d > 0 else -mv
            else:
                d = ty - y
                mv = min(abs(d), budget)
                y += mv if d > 0 else -mv
            budget -= mv
        return x, y, budget

    @classmethod
    def plan(cls, start, target):
        """Return list of waypoints (corners) from start to target."""
        x, y = start
        tx, ty = target
        points = [(x, y)]
        guard = 0
        while (x, y) != (tx, ty) and guard < 10:
            axis = cls.next_axis(x, y, tx, ty)
            if axis == "x":
                x = tx
            else:
                y = ty
            points.append((x, y))
            guard += 1
        return points

    @classmethod
    def plan_multi(cls, start, targets):
        """Plan a continuous route through multiple destinations."""
        route = [tuple(start)]
        cur = tuple(start)
        for t in targets:
            seg = cls.plan(cur, tuple(t))
            route.extend(seg[1:])
            cur = tuple(t)
        return route

    @staticmethod
    def route_length(points):
        """Manhattan length of a waypoint route."""
        total = 0
        for (x1, y1), (x2, y2) in zip(points, points[1:]):
            total += abs(x2 - x1) + abs(y2 - y1)
        return total

"""Rider model: movement, state and order handling."""

import globalvar
from path_planner import PathPlanner

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``. Completed
    orders remain in the list as delivery history.
    """

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.current_order = None
        self.log = []
        self._log("spawn", None)

    # ------------------------------------------------------------------
    def _now(self):
        t = globalvar.get_value("time")
        return t if t is not None else 0

    def _log(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
        })

    def active_orders(self):
        """Orders that are not completed yet."""
        return [o for o in self.orders
                if getattr(o, "status", None) != "completed"]

    def _next_pending(self):
        for o in self.orders:
            if getattr(o, "status", None) != "completed" and o is not self.current_order:
                return o
        return None

    def _accept(self, order):
        self.current_order = order
        self.state = PICKING_UP
        if getattr(order, "rider_id", None) is None and hasattr(order, "assign_to"):
            order.assign_to(self.rider_id)
        order.status = "picking_up"
        order.accept_time = self._now()
        self._log("accepted_order", order.id)

    # ------------------------------------------------------------------
    def move(self):
        """Advance the rider by one tick."""
        if self.state == IDLE:
            nxt = self._next_pending()
            if nxt is None:
                return
            self._accept(nxt)

        order = self.current_order
        if order is None:
            self.state = IDLE
            return

        if self.state == PICKING_UP:
            tx, ty = order.res_pos
        else:
            tx, ty = order.cus_pos
        self.special_case = (self.x == tx and self.y != ty)
        self.x, self.y, _ = PathPlanner.step(self.x, self.y, tx, ty, self.speed)

        if (self.x, self.y) != (tx, ty):
            return

        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.status = "delivering"
            order.pickup_time = self._now()
            self._log("arrived_at_merchant", order.id)
        else:
            order.status = "completed"
            order.complete_time = self._now()
            self._log("delivered_to_customer", order.id)
            self.current_order = None
            nxt = self._next_pending()
            if nxt is not None:
                self._accept(nxt)
            else:
                self.state = IDLE

    def to_dict(self):
        """Serialisable representation."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "log": self.log,
        }

"""Order data model."""

import globalvar

MAP_SIZE = 800
CELL = 100


def to_grid_corner(v):
    """Convert a coordinate to the top-left corner of its grid cell."""
    v = max(0, min(MAP_SIZE, int(v)))
    return min((v // CELL) * CELL, MAP_SIZE - CELL)


def to_grid_center(v):
    """Convert a coordinate to its grid cell center."""
    return to_grid_corner(v) + CELL // 2


class Order:
    """A delivery order from a restaurant to a customer."""

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.res_pos = tuple(int(v) for v in res_pos)
        self.cus_pos = tuple(int(v) for v in cus_pos)
        self.res_grid = (to_grid_corner(self.res_pos[0]), to_grid_corner(self.res_pos[1]))
        self.cus_grid = (to_grid_corner(self.cus_pos[0]), to_grid_corner(self.cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        t = globalvar.get_value("time")
        self.create_time = t if t is not None else 0
        self.accept_time = None
        self.pickup_time = None
        self.complete_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Record the rider this order is assigned to."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def time_taken(self):
        """Ticks from creation to completion, or None."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.create_time

    def to_dict(self):
        """Serialisable representation."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "create_time": self.create_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
            "time_taken": self.time_taken(),
        }

"""Order assignment logic and OrderManager."""

import math

import globalvar
import system
from order_model import Order


def _distance(rider, order):
    rx, ry = order.res_pos
    return math.sqrt((rider.x - rx) ** 2 + (rider.y - ry) ** 2)


def find_best_rider_for_order(order):
    """Return the best rider: fewest current orders, then closest, then id.

    Recruits a rider if none exist and funds allow. Returns None if no rider
    is available.
    """
    riders = globalvar.get_value("riders")
    if not riders:
        system.recruit_rider()
        riders = globalvar.get_value("riders")
    if not riders:
        return None
    return min(riders, key=lambda r: (len(r.active_orders()),
                                      _distance(r, order), r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion queries."""

    def create_order(self, coords):
        """Create and assign an order from [rx, ry, cx, cy].

        Returns (order, rider, recruited_flag).
        """
        order = Order(coords[0:2], coords[2:4])
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        before = len(globalvar.get_value("riders") or [])
        system.auto_recruit()
        rider = find_best_rider_for_order(order)
        recruited = len(globalvar.get_value("riders") or []) > before
        if rider is not None:
            rider.orders.append(order)
            order.assign_to(rider.rider_id)
        return order, rider, recruited

    @staticmethod
    def completed_orders():
        """List of completed orders."""
        return [o for o in globalvar.get_value("orders") or []
                if o.status == "completed"]

"""Status display and ASCII map visualisation."""

import globalvar
from rider import STATE_NAMES

SIZE = 16
UNIT = 50


def _idx(v):
    return max(0, min(SIZE - 1, int(v) // UNIT))


class DisplayManager:
    """Builds textual views of the system."""

    def base_map(self):
        grid = []
        for r in range(SIZE):
            row = []
            for c in range(SIZE):
                row.append("#" if (r % 2 == 0 or c % 2 == 0) else ".")
            grid.append(row)
        return grid

    def draw_route(self, grid, points):
        for (x1, y1), (x2, y2) in zip(points, points[1:]):
            c1, r1, c2, r2 = _idx(x1), _idx(y1), _idx(x2), _idx(y2)
            for c in range(min(c1, c2), max(c1, c2) + 1):
                grid[r1][c] = "*"
            for r in range(min(r1, r2), max(r1, r2) + 1):
                grid[r][c2] = "*"

    def render(self, route=None, orders=None):
        """Render the map with riders, restaurants, customers and route."""
        grid = self.base_map()
        if route:
            self.draw_route(grid, route)
        orders = orders if orders is not None else [
            o for o in globalvar.get_value("orders") or []
            if o.status != "completed"]
        for o in orders:
            grid[_idx(o.res_pos[1])][_idx(o.res_pos[0])] = "S"
            grid[_idx(o.cus_pos[1])][_idx(o.cus_pos[0])] = "C"
        for r in globalvar.get_value("riders") or []:
            grid[_idx(r.y)][_idx(r.x)] = "R"
        return "\n".join(" ".join(row) for row in grid)

    def status(self):
        lines = ["Time: %s" % globalvar.get_value("time"),
                 "Funds: %s" % globalvar.get_value("money")]
        riders = globalvar.get_value("riders") or []
        lines.append("Riders: %d" % len(riders))
        for r in riders:
            lines.append("  Rider %d at (%d, %d) state=%s orders=%s active=%d" % (
                r.rider_id, r.x, r.y, STATE_NAMES.get(r.state, r.state),
                [o.id for o in r.orders], len(r.active_orders())))
        lines.append(self.render())
        return "\n".join(lines)

    def orders(self):
        lines = []
        for o in globalvar.get_value("orders") or []:
            lines.append(
                "Order %d: start=%s restaurant=%s customer=%s distance=%d "
                "time_taken=%s rider=%s status=%s" % (
                    o.id, o.create_time, o.res_pos, o.cus_pos, o.distance,
                    o.time_taken(), o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    def riders(self):
        lines = []
        for r in globalvar.get_value("riders") or []:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%s %s order=%s at (%s, %s)" % (
                    e.get("time"), e["event"], e["order_id"], e.get("x"), e.get("y")))
        return "\n".join(lines) or "No riders."

"""Command parsing and input validation."""

SHORTCUTS = {"s": "start", "st": "status", "o": "orders", "r": "riders",
             "q": "quit", "exit": "quit", "h": "help"}
COMMANDS = {"start", "status", "orders", "riders", "quit", "help", "history"}


class InputManager:
    """Parses commands and validates coordinates."""

    def __init__(self):
        self.history = []
        try:
            import readline  # noqa: F401  (enables arrow-key history)
        except ImportError:
            pass

    def parse_command(self, text):
        cmd = text.strip().lower()
        if cmd:
            self.history.append(cmd)
        cmd = SHORTCUTS.get(cmd, cmd)
        return cmd if cmd in COMMANDS else None

    @staticmethod
    def parse_coords(text):
        """Parse four integers in range 0-800. Raises ValueError on error."""
        parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(parts) != 4:
            raise ValueError("Expected exactly 4 integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        for v in vals:
            if v < 0 or v > 800:
                raise ValueError("Coordinates must be in range 0-800.")
        return vals

"""SystemManager: funds, riders, simulation ticks and persistence."""

import json

import globalvar
import system


class SystemManager:
    """Core system controller."""

    def __init__(self, config_path=None):
        funds = system.INITIAL_FUNDS
        if config_path:
            try:
                with open(config_path) as f:
                    funds = int(json.load(f).get("funds", funds))
            except (OSError, ValueError):
                pass
        system.init_system(funds)

    def tick(self):
        """Advance simulation one tick."""
        globalvar.set_value("time", globalvar.get_value("time") + 1)
        for r in globalvar.get_value("riders"):
            try:
                r.move()
            except Exception as exc:  # error recovery
                print("Rider %d error: %s" % (r.rider_id, exc))

    def run_until_complete(self, order, limit=10000):
        n = 0
        while order.status != "completed" and n < limit:
            self.tick()
            n += 1
        return n

    def save(self, path="dispatch_data.json"):
        data = {
            "funds": globalvar.get_value("money"),
            "time": globalvar.get_value("time"),
            "riders": [r.to_dict() for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        try:
            with open(path, "w") as f:
                json.dump(data, f, indent=2)
        except OSError as exc:
            print("Could not save data: %s" % exc)

"""Command-line entry point for the food delivery dispatch simulator."""

import math

import globalvar
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from path_planner import PathPlanner
from system_manager import SystemManager

HELP = ("Commands: start(s), status(st), orders(o), riders(r), "
        "history, help(h), quit(q)/exit")


def main():
    sm = SystemManager("config.json")
    om = OrderManager()
    dm = DisplayManager()
    im = InputManager()
    print("Food Delivery Dispatch Simulator. " + HELP)
    while globalvar.get_value("running"):
        try:
            text = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        cmd = im.parse_command(text)
        try:
            if cmd is None:
                print("Unknown command. " + HELP)
            elif cmd == "help":
                print(HELP)
            elif cmd == "history":
                print("\n".join(im.history))
            elif cmd == "quit":
                globalvar.set_value("running", False)
            elif cmd == "start":
                raw = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
                coords = im.parse_coords(raw)
                order, rider, recruited = om.create_order(coords)
                print("Restaurant grid: %s  Customer grid: %s" % (order.res_grid, order.cus_grid))
                print("New rider added: %s" % ("yes" if recruited else "no"))
                if rider is None:
                    print("No rider available; order pending.")
                    continue
                route = PathPlanner.plan_multi((rider.x, rider.y),
                                               [order.res_pos, order.cus_pos])
                length = PathPlanner.route_length(route)
                print("Assigned rider %d speed=%d estimated time=%d ticks" % (
                    rider.rider_id, rider.speed, math.ceil(length / rider.speed)))
                print(dm.render(route=route, orders=[order]))
                ticks = sm.run_until_complete(order)
                print("Order %d completed in %d ticks." % (order.id, ticks))
            elif cmd == "status":
                print(dm.status())
            elif cmd == "orders":
                print(dm.orders())
            elif cmd == "riders":
                print(dm.riders())
        except ValueError as exc:
            print("Input error: %s" % exc)
        except Exception as exc:
            print("Error: %s" % exc)
    sm.save()
    print("Goodbye.")


if __name__ == "__main__":
    main()

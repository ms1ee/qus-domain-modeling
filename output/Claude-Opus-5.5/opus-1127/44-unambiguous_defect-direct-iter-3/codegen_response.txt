"""Data loading utilities for the Beijing Subway fare calculation system.

The subway network data lives in ``subway_data.json`` next to this module.
``DATA_FILE_PATH`` is an absolute path, so it is valid from any working
directory.

The JSON file has this structure::

    {
      "stations":    [{"name": str, "lines": [str, ...]}, ...],
      "connections": [{"source": str, "dest": str,
                       "distance": int, "line": str}, ...]
    }
"""

import json
import os

DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


class DataLoadError(Exception):
    """Raised when the subway data file cannot be loaded or is malformed."""


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway data JSON file and return its content as a dict.

    :param path: path of the JSON data file (defaults to ``DATA_FILE_PATH``).
    :return: the parsed JSON content (a dict).
    :raises DataLoadError: if the file is missing, unreadable or malformed.
    """
    if path is None:
        path = DATA_FILE_PATH
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise DataLoadError(f"Data file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise DataLoadError(f"Data file is not valid JSON: {path} ({exc})") from exc
    except OSError as exc:
        raise DataLoadError(f"Cannot read data file {path}: {exc}") from exc

    if not isinstance(data, dict):
        raise DataLoadError("Data file must contain a JSON object at top level")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(
        data["connections"], list
    ):
        raise DataLoadError("'stations' and 'connections' must be JSON arrays")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data (dict with stations/connections) to a JSON file."""
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)

"""Graph data structures used to model the subway network.

The network is a weighted undirected graph. Vertices are station numbers
(assigned by the caller, e.g. via a ``station_to_number`` dictionary) and
edges carry the inter-station distance in meters and the line name.
"""


class Edge:
    """A weighted undirected edge between two stations on a given line."""

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise ValueError(f"Edge distance must be numeric, got {distance!r}")
        if distance < 0:
            raise ValueError(f"Edge distance must be non-negative, got {distance}")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def either(self):
        """Return one endpoint of the edge."""
        return self.source

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError(f"Vertex {vertex!r} is not an endpoint of {self!r}")

    def set_distance(self, distance):
        """Update the edge distance (keeps ``weight`` in sync)."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise ValueError(f"Edge distance must be numeric, got {distance!r}")
        if distance < 0:
            raise ValueError("Edge distance must be non-negative")
        self.distance = distance
        self.weight = distance

    def __repr__(self):
        return (
            f"Edge({self.source!r}, {self.dest!r}, {self.distance!r}, {self.line!r})"
        )


class Graph:
    """Weighted undirected graph with adjacency lists.

    ``graph`` maps a station number to the list of incident edges.
    ``station_lines`` maps a station name to the list of lines it belongs to.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self._edge_count = 0

    @property
    def adj(self):
        return self.graph

    def add_vertex(self, vertex):
        """Ensure ``vertex`` exists in the graph."""
        if vertex not in self.graph:
            self.graph[vertex] = []

    def add_edge(self, edge):
        """Add an undirected edge."""
        if not isinstance(edge, Edge):
            raise TypeError("add_edge expects an Edge instance")
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        if edge.dest != edge.source:
            self.graph[edge.dest].append(edge)
        self._edge_count += 1

    def remove_edge(self, source, dest, line=None):
        """Remove edges between source and dest (optionally on a given line).

        :return: number of edges removed.
        """
        removed = []
        for edge in list(self.graph.get(source, [])):
            if {edge.source, edge.dest} == {source, dest} and (
                line is None or edge.line == line
            ):
                removed.append(edge)
        for edge in removed:
            for vertex in {edge.source, edge.dest}:
                if edge in self.graph.get(vertex, []):
                    self.graph[vertex].remove(edge)
            self._edge_count -= 1
        return len(removed)

    def adjacent(self, vertex):
        """Return the edges incident to ``vertex``."""
        return self.graph.get(vertex, [])

    def vertices(self):
        return list(self.graph.keys())

    def edges(self):
        """Return each edge exactly once."""
        seen = set()
        result = []
        for edge_list in self.graph.values():
            for edge in edge_list:
                if id(edge) not in seen:
                    seen.add(id(edge))
                    result.append(edge)
        return result

    def V(self):
        return len(self.graph)

    def E(self):
        return self._edge_count

    def set_station_lines(self, station_name, lines):
        """Record the lines that a station belongs to."""
        self.station_lines[station_name] = list(lines) if lines else []

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return len(set(self.station_lines.get(station_name, []))) > 1

    def __repr__(self):
        return f"Graph(V={self.V()}, E={self.E()})"

"""Path calculation: priority queue and Dijkstra shortest path.

Dijkstra runs over states ``(station, arriving line, transfers)`` so that line
changes (transfers) at interchange stations are tracked exactly. This lets the
algorithm support several modes (shortest distance, least transfers,
comprehensive), a maximum transfer limit and line preferences, while running in
O((V+E) log V) time with a binary-heap priority queue.
"""

import heapq
import itertools

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# In comprehensive mode one transfer "costs" as much as riding this distance.
TRANSFER_PENALTY_METERS = 3000
# Extra relative cost of riding a non-preferred line when preferences are set.
PREFERENCE_PENALTY_FACTOR = 0.3

INF = float("inf")


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None,
                 preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(
                f"Invalid mode {mode!r}, expected one of {', '.join(VALID_MODES)}"
            )
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int):
                raise ValueError("max_transfers must be an integer")
            if max_transfers < 0:
                raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        if preferred_lines is None:
            preferred_lines = []
        elif isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.preferred_lines = list(preferred_lines)

    def __repr__(self):
        return (
            f"PathfindingConfig(mode={self.mode!r}, max_transfers={self.max_transfers!r}, "
            f"preferred_lines={self.preferred_lines!r})"
        )


class MinPQ:
    """Heap-based minimum priority queue supporting decrease-key.

    Items are hashable keys; priorities are any comparable values.
    """

    _REMOVED = object()

    def __init__(self):
        self._heap = []
        self._entries = {}
        self._counter = itertools.count()

    def __len__(self):
        return len(self._entries)

    def size(self):
        return len(self._entries)

    def is_empty(self):
        return not self._entries

    def contains(self, item):
        return item in self._entries

    def __contains__(self, item):
        return item in self._entries

    def insert(self, item, priority):
        """Insert an item (replacing its priority if already present)."""
        if item in self._entries:
            self._remove(item)
        entry = [priority, next(self._counter), item]
        self._entries[item] = entry
        heapq.heappush(self._heap, entry)

    def decrease_key(self, item, priority):
        """Lower the priority of item (inserts it if absent)."""
        entry = self._entries.get(item)
        if entry is None or priority < entry[0]:
            self.insert(item, priority)

    def _remove(self, item):
        entry = self._entries.pop(item)
        entry[2] = MinPQ._REMOVED

    def min_priority(self):
        while self._heap and self._heap[0][2] is MinPQ._REMOVED:
            heapq.heappop(self._heap)
        if not self._heap:
            raise IndexError("min_priority from empty priority queue")
        return self._heap[0][0]

    def del_min(self):
        """Remove and return ``(item, priority)`` with the smallest priority."""
        while self._heap:
            priority, _, item = heapq.heappop(self._heap)
            if item is not MinPQ._REMOVED:
                del self._entries[item]
                return item, priority
        raise IndexError("del_min from empty priority queue")


class ShortestPath:
    """Dijkstra shortest path from a single source station.

    ``dist_to`` maps every vertex to the real distance (meters) of the best
    path found under the configured mode (``inf`` if unreachable).
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self._preferred = set(self.config.preferred_lines)
        self._track_transfers = self.config.max_transfers is not None

        self.dist_to = {v: INF for v in graph.vertices()}
        self.dist_to[source] = 0
        self._state_info = {}
        self._edge_to = {}
        self._best_state = {}
        self._run()

    # ------------------------------------------------------------------ cost
    def _cost(self, distance, penalty, transfers):
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, distance + penalty)
        if mode == MODE_COMPREHENSIVE:
            return (distance + penalty + transfers * TRANSFER_PENALTY_METERS,
                    transfers)
        return (distance + penalty, transfers)

    def _edge_penalty(self, edge):
        if self._preferred and edge.line not in self._preferred:
            return edge.weight * PREFERENCE_PENALTY_FACTOR
        return 0

    # ------------------------------------------------------------- algorithm
    def _run(self):
        start = (self.source, None, 0)
        start_cost = self._cost(0, 0, 0)
        # state -> (cost, real distance, penalty, transfers)
        self._state_info[start] = (start_cost, 0, 0, 0)
        self._edge_to[start] = None
        settled = set()
        pq = MinPQ()
        pq.insert(start, start_cost)
        max_t = self.config.max_transfers

        while not pq.is_empty():
            state, _ = pq.del_min()
            settled.add(state)
            u, line, _ = state
            _, dist, penalty, transfers = self._state_info[state]
            for edge in self.graph.adjacent(u):
                w = edge.other(u)
                new_transfers = transfers + (
                    1 if line is not None and edge.line != line else 0
                )
                if max_t is not None and new_transfers > max_t:
                    continue
                new_state = (w, edge.line,
                             new_transfers if self._track_transfers else 0)
                if new_state in settled:
                    continue
                new_dist = dist + edge.weight
                new_penalty = penalty + self._edge_penalty(edge)
                new_cost = self._cost(new_dist, new_penalty, new_transfers)
                old = self._state_info.get(new_state)
                if old is None or new_cost < old[0]:
                    self._state_info[new_state] = (
                        new_cost, new_dist, new_penalty, new_transfers)
                    self._edge_to[new_state] = (state, edge)
                    pq.decrease_key(new_state, new_cost)

        for state, info in self._state_info.items():
            node = state[0]
            best = self._best_state.get(node)
            if best is None or info[0] < self._state_info[best][0]:
                self._best_state[node] = state
        for node, state in self._best_state.items():
            self.dist_to[node] = self._state_info[state][1]

    # --------------------------------------------------------------- queries
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return ``(stations, edges)`` from source to target.

        Stations are station numbers in travel order; edges are the Edge
        objects traversed in order. Returns ``([], [])`` if unreachable.
        """
        if not self.has_path_to(target):
            return [], []
        nodes = []
        edges = []
        state = self._best_state[target]
        while state is not None:
            nodes.append(state[0])
            link = self._edge_to.get(state)
            if link is None:
                break
            prev_state, edge = link
            edges.append(edge)
            state = prev_state
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes on the path to target (0 if unreachable)."""
        if not self.has_path_to(target):
            return 0
        return self._state_info[self._best_state[target]][3]

    def get_lines(self, target):
        """Ordered list of distinct lines used on the path to target."""
        lines = []
        for edge in self.path_with_edges(target)[1]:
            if not lines or lines[-1] != edge.line:
                if edge.line not in lines:
                    lines.append(edge.line)
        return lines

    def get_transfer_stations(self, target):
        """Station numbers where a transfer happens on the path to target."""
        nodes, edges = self.path_with_edges(target)
        result = []
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                result.append(nodes[i])
        return result

"""Beijing Subway fare calculation.

Fare structure (distance in kilometers):
    [0, 6)   -> 3 yuan
    [6, 12)  -> 4 yuan
    [12, 22) -> 5 yuan
    [22, 32) -> 6 yuan
    >= 32    -> 7 yuan, plus 1 yuan for every further 20 km
Boundary values move into the higher tier (6 km = 4 yuan, 32 km = 7 yuan).
Fractional distances (meters) are rounded up.
"""

import math

# (exclusive upper bound in meters, fare in yuan)
FARE_TIERS = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_START = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP = 20000

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


class FareCalculationError(Exception):
    """Raised when a fare cannot be computed."""


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError(f"Distance must be a number, got {distance!r}")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError(f"Distance must be finite, got {distance!r}")
    if distance < 0:
        raise ValueError(f"Distance must be non-negative, got {distance!r}")


def Sum_money(distance):
    """Return the fare (yuan) for a trip of ``distance`` meters."""
    _validate_distance(distance)
    meters = math.ceil(distance)
    for upper, fare in FARE_TIERS:
        if meters < upper:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_START) // EXTENDED_STEP


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes.

    Based on an average running speed (including stops) of 35 km/h plus
    5 minutes per transfer.
    """
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, (int, float)):
        raise TypeError("transfer_count must be a number")
    if transfer_count < 0:
        raise ValueError("transfer_count must be non-negative")
    minutes = (distance / 1000.0) / AVERAGE_SPEED_KMH * 60.0
    minutes += transfer_count * TRANSFER_TIME_MINUTES
    return round(float(minutes), 1)


def fare_table():
    """Return a human-readable description of the fare structure."""
    rows = []
    lower = 0
    for upper, fare in FARE_TIERS:
        rows.append(f"{lower // 1000}-{upper // 1000} km: {fare} yuan")
        lower = upper
    rows.append(
        f"Over {EXTENDED_START // 1000} km: {EXTENDED_BASE_FARE} yuan, "
        f"+1 yuan per additional {EXTENDED_STEP // 1000} km"
    )
    return rows

def normalize_station_name(name):
    """Normalize a station name for tolerant lookup."""
    text = unicodedata.normalize("NFKC", str(name)).strip()
    text = text.replace("\u2019", "'").replace("\u2018", "'").replace("`", "'")
    text = re.sub(r"\s+", " ", text).lower()
    for suffix in (" station", "站"):
        if text.endswith(suffix) and len(text) > len(suffix):
            text = text[: -len(suffix)].strip()
    return text


def _compact(text):
    return re.sub(r"[\s'\-_.·]", "", text)


def normalize_line_name(name):
    text = unicodedata.normalize("NFKC", str(name)).strip().lower()
    return re.sub(r"[\s\-_]", "", text)


class SubwayNetwork:
    """In-memory model of the subway network."""

    def __init__(self, data=None, data_path=None):
        self.data_path = data_path or DATA_FILE_PATH
        if data is None:
            data = load_subway_data(self.data_path)
        self.stations = []
        for st in data.get("stations", []):
            if isinstance(st, dict) and "name" in st:
                self.stations.append(
                    {"name": str(st["name"]), "lines": list(st.get("lines") or [])}
                )
        self.connections = []
        for c in data.get("connections", []):
            if isinstance(c, dict):
                self.connections.append(dict(c))
        self.build()

    # ---------------------------------------------------------------- build
    def build(self):
        """(Re)build station mapping and graph from stations/connections."""
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.load_warnings = []

        def register(name, lines):
            if name in self.station_to_number:
                return self.station_to_number[name]
            number = len(self.station_to_number)
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
            self.graph.set_station_lines(name, lines)
            return number

        for st in self.stations:
            if st["name"] in self.station_to_number:
                self.load_warnings.append(f"Duplicate station '{st['name']}' ignored")
                continue
            register(st["name"], st["lines"])

        for c in self.connections:
            src, dst = c.get("source"), c.get("dest")
            line = c.get("line", "")
            dist = c.get("distance")
            if src is None or dst is None:
                self.load_warnings.append(f"Connection missing endpoint: {c}")
                continue
            for name in (src, dst):
                if name not in self.station_to_number:
                    self.load_warnings.append(
                        f"Station '{name}' referenced by a connection but not defined"
                    )
                    register(name, [line] if line else [])
            try:
                edge = Edge(self.station_to_number[src], self.station_to_number[dst],
                            dist, line)
            except (ValueError, TypeError) as exc:
                self.load_warnings.append(f"Invalid connection {src}-{dst}: {exc}")
                continue
            self.graph.add_edge(edge)

        self.lines = self._build_line_sequences()
        self._build_name_index()

    def _build_name_index(self):
        self._name_index = {}
        compact_map = defaultdict(set)
        for name in self.station_to_number:
            norm = normalize_station_name(name)
            self._name_index.setdefault(norm, name)
            compact_map[_compact(norm)].add(name)
        self._compact_index = {
            key: next(iter(names)) for key, names in compact_map.items()
            if len(names) == 1
        }
        self._line_index = {}
        for line in self.all_lines():
            self._line_index[normalize_line_name(line)] = line

    def _build_line_sequences(self):
        """Derive ordered station sequences for each line from connections."""
        adjacency = defaultdict(lambda: defaultdict(list))
        for c in self.connections:
            line = c.get("line", "")
            src, dst = c.get("source"), c.get("dest")
            if src is None or dst is None:
                continue
            adjacency[line][src].append(dst)
            adjacency[line][dst].append(src)
        sequences = {}
        for line, adj in adjacency.items():
            visited = set()
            order = []
            # Prefer starting from terminals (degree 1), then any station.
            starts = [s for s in adj if len(adj[s]) == 1] + list(adj.keys())
            for start in starts:
                if start in visited:
                    continue
                stack = [start]
                while stack:
                    node = stack.pop()
                    if node in visited:
                        continue
                    visited.add(node)
                    order.append(node)
                    for nxt in reversed(adj[node]):
                        if nxt not in visited:
                            stack.append(nxt)
            sequences[line] = order
        for st in self.stations:
            for line in st["lines"]:
                sequences.setdefault(line, [])
                if st["name"] not in sequences[line]:
                    sequences[line].append(st["name"])
        return sequences

    # --------------------------------------------------------------- lookup
    def all_lines(self):
        lines = set(self.lines.keys()) if hasattr(self, "lines") else set()
        for names in self.graph.station_lines.values():
            lines.update(names)
        lines.discard("")
        return sorted(lines, key=_line_sort_key)

    def resolve_station(self, name):
        """Return the canonical station name for user input, or None."""
        if name is None:
            return None
        if name in self.station_to_number:
            return name
        norm = normalize_station_name(name)
        if not norm:
            return None
        if norm in self._name_index:
            return self._name_index[norm]
        return self._compact_index.get(_compact(norm))

    def resolve_line(self, name):
        """Return the canonical line name for user input, or None."""
        if name in self._line_index.values():
            return name
        key = normalize_line_name(name)
        if key in self._line_index:
            return self._line_index[key]
        if key.isdigit():
            return self._line_index.get("line" + key) or self._line_index.get(
                key + "号线")
        return None

    # ---------------------------------------------------------------- route
    def plan_route(self, start, end, config=None):
        """Plan a route and compute all summary information.

        :return: dict with query, path, segments, distance, fare, etc.
        :raises RouteError subclasses on failure.
        """
        config = config or PathfindingConfig()
        start_name = self.resolve_station(start)
        if start_name is None:
            raise StationNotFoundError(
                f"Starting station '{start}' does not exist, please check the station name"
            )
        end_name = self.resolve_station(end)
        if end_name is None:
            raise StationNotFoundError(
                f"Ending station '{end}' does not exist, please check the station name"
            )
        if start_name == end_name:
            raise SameStationError(
                "Starting station cannot be the same as the ending station"
            )

        source = self.station_to_number[start_name]
        target = self.station_to_number[end_name]
        try:
            sp = ShortestPath(self.graph, source, config)
        except Exception as exc:  # defensive: report as path error
            raise PathCalculationError(f"Path calculation failed: {exc}") from exc
        if not sp.has_path_to(target):
            if config.max_transfers is not None:
                raise PathCalculationError(
                    f"No route from '{start_name}' to '{end_name}' within "
                    f"{config.max_transfers} transfer(s)"
                )
            raise PathCalculationError(
                f"No route found from '{start_name}' to '{end_name}'"
            )

        nodes, edges = sp.path_with_edges(target)
        names = [self.number_to_station[n] for n in nodes]
        distance = sp.dist_to[target]
        transfer_count = sp.get_transfer_count(target)
        transfer_stations = [self.number_to_station[n]
                             for n in sp.get_transfer_stations(target)]

        segments = []
        for i, edge in enumerate(edges):
            frm, to = names[i], names[i + 1]
            if segments and segments[-1]["line"] == edge.line:
                seg = segments[-1]
                seg["to"] = to
                seg["stations"].append(to)
                seg["stops"] += 1
                seg["distance"] += edge.weight
            else:
                segments.append({"line": edge.line, "from": frm, "to": to,
                                 "stations": [frm, to], "stops": 1,
                                 "distance": edge.weight})
        lines = []
        for seg in segments:
            if seg["line"] not in lines:
                lines.append(seg["line"])

        try:
            fare = Sum_money(distance)
        except (TypeError, ValueError) as exc:
            raise FareError(f"Fare calculation failed: {exc}") from exc
        try:
            est_time = calculate_estimated_time(distance, transfer_count)
        except (TypeError, ValueError):
            est_time = None

        return {
            "query": {
                "start": start_name,
                "end": end_name,
                "mode": config.mode,
                "max_transfers": config.max_transfers,
                "line_preference": list(config.preferred_lines),
            },
            "path": names,
            "station_count": len(names),
            "segments": segments,
            "distance": {"meters": distance,
                         "kilometers": round(distance / 1000.0, 2)},
            "fare": fare,
            "fare_text": f"Fare: {fare} yuan",
            "estimated_time_minutes": est_time,
            "transfer_count": transfer_count,
            "transfer_stations": transfer_stations,
            "lines": lines,
        }

    # ------------------------------------------------------------ integrity
    def check_integrity(self):
        """Run data integrity checks.

        :return: dict with keys ok, errors, warnings, info.
        """
        errors, warnings, info = [], [], []
        names = [st["name"] for st in self.stations]
        defined = set(names)
        seen = set()
        for name in names:
            if name in seen:
                errors.append(f"Duplicate station definition: '{name}'")
            seen.add(name)
        for st in self.stations:
            if not st["lines"]:
                warnings.append(f"Station '{st['name']}' has no line affiliation")

        pair_seen = {}
        connected = set()
        line_has_conn = defaultdict(set)
        for idx, c in enumerate(self.connections):
            src, dst = c.get("source"), c.get("dest")
            line, dist = c.get("line"), c.get("distance")
            label = f"Connection #{idx} ({src} - {dst}, {line})"
            if src not in defined:
                errors.append(f"{label}: source station not defined")
            if dst not in defined:
                errors.append(f"{label}: destination station not defined")
            if src == dst:
                errors.append(f"{label}: connects a station to itself")
            if not line:
                warnings.append(f"{label}: missing line name")
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                errors.append(f"{label}: distance is not a number ({dist!r})")
            elif dist <= 0:
                errors.append(f"{label}: distance must be positive ({dist})")
            elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                warnings.append(f"{label}: suspicious distance {dist} m")
            for name in (src, dst):
                if name in defined and line:
                    lines = self.graph.station_lines.get(name, [])
                    if line not in lines:
                        warnings.append(
                            f"{label}: station '{name}' is not listed on line '{line}'"
                        )
            if src is not None and dst is not None:
                connected.update((src, dst))
                line_has_conn[line].update((src, dst))
                key = (frozenset((src, dst)), line)
                if key in pair_seen:
                    if pair_seen[key] != dist:
                        errors.append(
                            f"{label}: conflicting duplicate distances "
                            f"({pair_seen[key]} vs {dist})")
                    else:
                        warnings.append(f"{label}: duplicate connection")
                else:
                    pair_seen[key] = dist

        for name in names:
            if name not in connected:
                warnings.append(f"Station '{name}' has no connections")
        for st in self.stations:
            for line in st["lines"]:
                if st["name"] in connected and st["name"] not in line_has_conn.get(line, ()):
                    warnings.append(
                        f"Station '{st['name']}' lists line '{line}' but has no "
                        f"connection on that line")

        # Connectivity
        components = self._components()
        if len(components) > 1:
            sizes = sorted((len(c) for c in components), reverse=True)
            warnings.append(
                f"Network has {len(components)} disconnected components (sizes: "
                f"{', '.join(map(str, sizes[:10]))})")

        # Main line coverage
        data_lines = {normalize_line_name(l): l for l in self.all_lines()}
        for main_line, aliases in MAIN_LINES.items():
            keys = {normalize_line_name(main_line)} | {normalize_line_name(a) for a in aliases}
            found = [data_lines[k] for k in keys if k in data_lines]
            if not found:
                warnings.append(f"Main line '{main_line}' not found in data")
                continue
            candidates = REPRESENTATIVE_STATIONS.get(main_line, [])
            if candidates and not any(self.resolve_station(c) for c in candidates):
                warnings.append(
                    f"No representative station found for '{main_line}' "
                    f"(checked: {', '.join(candidates)})")
            else:
                info.append(f"Main line '{main_line}' present as '{found[0]}'")

        for w in self.load_warnings:
            if w not in warnings:
                warnings.append(w)

        info.insert(0, f"Stations: {len(self.station_to_number)}")
        info.insert(1, f"Connections: {len(self.connections)}")
        info.insert(2, f"Lines: {len(self.all_lines())}")
        transfer_count = sum(1 for n in self.station_to_number
                             if self.graph.is_transfer_station(n))
        info.insert(3, f"Transfer stations: {transfer_count}")
        return {"ok": not errors, "errors": errors, "warnings": warnings, "info": info}

    def _components(self):
        seen = set()
        comps = []
        for v in self.graph.vertices():
            if v in seen:
                continue
            comp = []
            queue = deque([v])
            seen.add(v)
            while queue:
                u = queue.popleft()
                comp.append(u)
                for e in self.graph.adjacent(u):
                    w = e.other(u)
                    if w not in seen:
                        seen.add(w)
                        queue.append(w)
            comps.append(comp)
        return comps

    # --------------------------------------------------------------- update
    def _find_station_record(self, name):
        for st in self.stations:
            if st["name"] == name:
                return st
        return None

    def add_station(self, name, lines=None):
        if self._find_station_record(name) is not None:
            raise ValueError(f"Station '{name}' already exists")
        self.stations.append({"name": name, "lines": list(lines or [])})
        self.build()

    def add_line(self, line_name, station_sequence, distances):
        """Add a new line given an ordered station list and segment distances."""
        if len(station_sequence) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_sequence) - 1:
            raise ValueError("distances must have len(station_sequence) - 1 items")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError(f"Invalid distance {d!r}")
        for name in station_sequence:
            record = self._find_station_record(name)
            if record is None:
                self.stations.append({"name": name, "lines": [line_name]})
            elif line_name not in record["lines"]:
                record["lines"].append(line_name)
        for i, d in enumerate(distances):
            self.connections.append({"source": station_sequence[i],
                                     "dest": station_sequence[i + 1],
                                     "distance": d, "line": line_name})
        self.build()

    def modify_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        record = self._find_station_record(name)
        if record is None:
            raise ValueError(f"Station '{name}' does not exist")
        if new_name and new_name != name:
            if self._find_station_record(new_name) is not None:
                raise ValueError(f"Station '{new_name}' already exists")
            record["name"] = new_name
            for c in self.connections:
                if c.get("source") == name:
                    c["source"] = new_name
                if c.get("dest") == name:
                    c["dest"] = new_name
        if lines is not None:
            record["lines"] = list(lines)
        self.build()

    def update_distance(self, source, dest, distance, line=None):
        """Update distance of connections between two stations.

        :return: number of connections updated.
        """
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) \
                or distance <= 0:
            raise ValueError(f"Invalid distance {distance!r}")
        count = 0
        for c in self.connections:
            if {c.get("source"), c.get("dest")} == {source, dest} and (
                    line is None or c.get("line") == line):
                c["distance"] = distance
                count += 1
        if count == 0:
            raise ValueError(f"No connection between '{source}' and '{dest}'")
        self.build()
        return count

    def to_dict(self):
        return {"stations": [dict(s) for s in self.stations],
                "connections": [dict(c) for c in self.connections]}

    def save(self, path=None):
        save_subway_data(self.to_dict(), path or self.data_path)


def _line_sort_key(name):
    match = re.search(r"\d+", name)
    return (0, int(match.group()), name) if match else (1, 0, name)


def format_integrity_report(report):
    lines = ["Data integrity check", "=" * 50]
    for item in report["info"]:
        lines.append(f"[INFO] {item}")
    for item in report["warnings"]:
        lines.append(f"[WARN] {item}")
    for item in report["errors"]:
        lines.append(f"[ERROR] {item}")
    lines.append("-" * 50)
    lines.append(
        f"Result: {'PASSED' if report['ok'] else 'FAILED'} "
        f"({len(report['errors'])} error(s), {len(report['warnings'])} warning(s))")
    return "\n".join(lines)

"""Route visualization and export (console text and JSON)."""

import json


def format_distance(meters):
    """Format a distance in both meters and kilometers."""
    km = meters / 1000.0
    if float(meters).is_integer():
        meters_text = f"{int(meters)} m"
    else:
        meters_text = f"{meters:.1f} m"
    return f"{meters_text} ({km:.2f} km)"


def format_text_report(route):
    """Return a console-formatted text report for a planned route."""
    query = route["query"]
    width = 60
    out = ["=" * width,
           f"Route: {query['start']} -> {query['end']}",
           f"Mode: {query['mode']}"]
    if query.get("max_transfers") is not None:
        out.append(f"Max transfers: {query['max_transfers']}")
    if query.get("line_preference"):
        out.append(f"Line preference: {', '.join(query['line_preference'])}")
    out.append("-" * width)
    out.append(f"Stations ({route['station_count']}):")
    out.append("  " + " -> ".join(route["path"]))
    out.append("")
    out.append("Segments:")
    segments = route["segments"]
    for i, seg in enumerate(segments):
        out.append(
            f"  [{seg['line']}] {seg['from']} -> {seg['to']} "
            f"({seg['stops']} stop(s), {format_distance(seg['distance'])})")
        out.append("      " + " - ".join(seg["stations"]))
        if i + 1 < len(segments):
            out.append(
                f"  >> Transfer at {seg['to']}: {seg['line']} -> {segments[i + 1]['line']}")
    out.append("-" * width)
    out.append(f"Total distance: {format_distance(route['distance']['meters'])}")
    transfers = route["transfer_count"]
    if transfers:
        out.append(f"Transfers: {transfers} (at {', '.join(route['transfer_stations'])})")
    else:
        out.append("Transfers: 0 (direct)")
    out.append(f"Lines involved: {', '.join(route['lines'])}")
    if route.get("estimated_time_minutes") is not None:
        out.append(f"Estimated time: {route['estimated_time_minutes']} min")
    out.append(route.get("fare_text") or f"Fare: {route['fare']} yuan")
    out.append("=" * width)
    return "\n".join(out)


def format_json_report(route_or_routes):
    """Return a JSON string for one route dict or a list of them."""
    return json.dumps(route_or_routes, ensure_ascii=False, indent=2)


def render(route, fmt="text"):
    if fmt == "json":
        return format_json_report(route)
    return format_text_report(route)

"""Command-line interface of the Beijing Subway fare calculation system.

Usage examples:
    python src/main.py
    python src/main.py --mode least_transfer --format json --output_file out.json
    python src/main.py --check_data

Enter queries as "Starting Station; Ending Station" (separators ';', ',',
'->' and their full-width forms are accepted). Type 'q' to quit.
"""

import argparse
import re
import sys

from data_loader import DataLoadError
from data_manager import (
    RouteError,
    SubwayNetwork,
    format_integrity_report,
)
from fare_calculator import fare_table
from pathfinding import PathfindingConfig, VALID_MODES
from visualization import format_json_report, render

QUIT_COMMANDS = {"q", "quit", "exit", "退出"}
HELP_COMMANDS = {"h", "help", "?"}
SEPARATOR_RE = re.compile(r"\s*(?:;|；|,|，|->|→)\s*")
FORMAT_ERROR = ("Input format incorrect, please use "
                "'Starting Station; Ending Station' format")


class InputFormatError(ValueError):
    pass


def parse_query_input(text):
    """Split "Start; End" input into (start, end)."""
    if text is None:
        raise InputFormatError(FORMAT_ERROR)
    parts = SEPARATOR_RE.split(text.strip())
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise InputFormatError(FORMAT_ERROR)
    return parts[0].strip(), parts[1].strip()


def build_arg_parser():
    parser = argparse.ArgumentParser(
        description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", default=None,
                        help="Export query results to this file")
    parser.add_argument("--format", choices=["text", "json"], default="text",
                        help="Output format (text or json)")
    parser.add_argument("--mode", choices=list(VALID_MODES),
                        default="shortest_distance", help="Path query mode")
    parser.add_argument("--max_transfers", type=int, default=None,
                        help="Maximum number of transfers")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line(s), e.g. \"Line 1\" "
                             "(several separated by commas)")
    parser.add_argument("--check_data", action="store_true",
                        help="Run data integrity check and exit")
    return parser


def write_output(path, results, fmt):
    if fmt == "json":
        payload = results[0] if len(results) == 1 else results
        content = format_json_report(payload)
    else:
        content = "\n\n".join(render(r, "text") for r in results)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(content + "\n")


def print_help(network):
    print("Enter: Starting Station; Ending Station   (e.g. 'Xidan; Guomao')")
    print("Commands: 'help' shows this message, 'lines' lists lines, 'q' quits.")
    print("Fare structure:")
    for row in fare_table():
        print(f"  {row}")


def main(argv=None):
    parser = build_arg_parser()
    args = parser.parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        parser.error("--max_transfers must be non-negative")

    try:
        network = SubwayNetwork()
    except DataLoadError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2

    if args.check_data:
        report = network.check_integrity()
        print(format_integrity_report(report))
        return 0 if report["ok"] else 1

    preferred = []
    if args.line_preference:
        for raw in re.split(r"[,，;；]", args.line_preference):
            raw = raw.strip()
            if not raw:
                continue
            line = network.resolve_line(raw)
            if line is None:
                print(f"Warning: line '{raw}' not found, preference ignored")
            else:
                preferred.append(line)

    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=preferred)
    except ValueError as exc:
        parser.error(str(exc))
        return 2

    print("Beijing Subway Fare Calculation System")
    print(f"Loaded {len(network.station_to_number)} stations, "
          f"{len(network.all_lines())} lines. Type 'help' for help, 'q' to quit.")

    results = []
    while True:
        try:
            text = input("\nEnter query (Starting Station; Ending Station): ")
        except EOFError:
            print()
            break
        except KeyboardInterrupt:
            print()
            break
        text = text.strip()
        if not text:
            continue
        low = text.lower()
        if low in QUIT_COMMANDS:
            break
        if low in HELP_COMMANDS:
            print_help(network)
            continue
        if low == "lines":
            print(", ".join(network.all_lines()))
            continue
        try:
            start, end = parse_query_input(text)
            route = network.plan_route(start, end, config)
        except InputFormatError as exc:
            print(f"Error: {exc}")
            continue
        except RouteError as exc:
            print(f"Error: {exc}")
            continue
        except Exception as exc:  # last-resort handler keeps the CLI alive
            print(f"Error: unexpected failure while processing query: {exc}")
            continue

        print(render(route, args.format))
        results.append(route)
        if args.output_file:
            try:
                write_output(args.output_file, results, args.format)
                print(f"Result exported to {args.output_file}")
            except OSError as exc:
                print(f"Error: cannot write output file: {exc}")

    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

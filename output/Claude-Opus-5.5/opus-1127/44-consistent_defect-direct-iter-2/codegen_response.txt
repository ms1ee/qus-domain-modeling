"""Data loading utilities for the Beijing Subway fare calculation system.

Loads ``subway_data.json`` and builds the graph model used by the
path-finding module.
"""

import json
import os

from models import Edge, Graph

# Absolute path of the bundled data file, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file."""
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        data = json.load(fh)
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data


def build_graph(data):
    """Build a Graph from loaded data.

    Returns (graph, station_to_number, number_to_station).
    Station numbers are assigned here (by the caller of Graph), in order of
    appearance in the station list, then any connection endpoints not listed.
    """
    graph = Graph()
    station_to_number = {}
    number_to_station = {}

    def ensure(name):
        if name not in station_to_number:
            num = len(station_to_number)
            station_to_number[name] = num
            number_to_station[num] = name
            graph.add_vertex(num)
        return station_to_number[name]

    for st in data.get("stations", []):
        name = st.get("name")
        if not name:
            continue
        ensure(name)
        graph.set_station_lines(name, list(st.get("lines", [])))

    for conn in data.get("connections", []):
        src = conn.get("source")
        dst = conn.get("dest")
        if not src or not dst:
            continue
        try:
            dist = int(conn.get("distance", 0))
        except (TypeError, ValueError):
            continue
        line = conn.get("line", "")
        s = ensure(src)
        d = ensure(dst)
        for name in (src, dst):
            lines = graph.station_lines.setdefault(name, [])
            if line and line not in lines:
                lines.append(line)
        graph.add_edge(Edge(s, d, dist, line))

    graph.number_to_station = dict(number_to_station)
    return graph, station_to_number, number_to_station

"""Graph data structures: Edge, Graph and a heap-based MinPQ."""

import heapq
import itertools


class Edge:
    """A weighted connection between two stations (by station number)."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.weight, self.line)


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Station numbers are assigned by the caller. Every added edge can be
    travelled in both directions.
    """

    def __init__(self):
        self.adj = {}
        self.graph = self.adj  # adjacency-list dictionary alias
        self.station_lines = {}
        self.number_to_station = {}
        self._edges = []

    def add_vertex(self, v):
        self.adj.setdefault(v, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self._edges.append(edge)
        self.adj[edge.source].append(edge)
        if edge.source != edge.dest:
            self.adj[edge.dest].append(edge.reversed())

    def adjacent(self, v):
        return self.adj.get(v, [])

    def vertices(self):
        return list(self.adj.keys())

    def edges(self):
        return list(self._edges)

    def V(self):
        return len(self.adj)

    def E(self):
        return len(self._edges)


class MinPQ:
    """Min priority queue backed by a binary heap (heapq)."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)

"""Dijkstra-based shortest path computation with multiple query modes."""

from models import MinPQ

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = 3000      # meters-equivalent cost of one transfer (comprehensive mode)
PREFERENCE_FACTOR = 0.8      # weight multiplier for preferred lines
INF = float("inf")


def _norm_line(name):
    return "".join(str(name).lower().split())


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode '%s'; expected one of %s" % (mode, ", ".join(VALID_MODES)))
        if max_transfers is not None and max_transfers < 0:
            raise ValueError("max_transfers must be non-negative")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class ShortestPath:
    """Computes optimal paths from a source station using Dijkstra's algorithm.

    The search runs over states (station, current line[, transfers]) so that
    transfers at interchange stations are handled correctly.
    """

    def __init__(self, graph, source, config=None):
        self._graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: INF for v in graph.vertices()}
        self.dist_to[source] = 0
        self._best_state = {}
        self._edge_to = {}
        self._info = {}
        self._pref = {_norm_line(l) for l in self.config.preferred_lines}
        self._run()

    def _key(self, adj, transfers):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, adj)
        if mode == "comprehensive":
            return (adj + transfers * TRANSFER_PENALTY, transfers)
        return (adj, transfers)

    def _run(self):
        track = self.config.max_transfers is not None
        maxt = self.config.max_transfers
        start = (self.source, None, 0) if track else (self.source, None)
        self._info[start] = (0.0, 0, 0)  # adjusted cost, real distance, transfers
        pq = MinPQ()
        pq.push(self._key(0.0, 0), start)
        best_key = {start: self._key(0.0, 0)}
        settled = set()
        while not pq.is_empty():
            key, state = pq.pop()
            if state in settled:
                continue
            settled.add(state)
            node, line = state[0], state[1]
            adj, real, t = self._info[state]
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = real
            for e in self._graph.adjacent(node):
                nt = t + (1 if (line is not None and e.line != line) else 0)
                if track and nt > maxt:
                    continue
                w = e.weight
                if self._pref and _norm_line(e.line) in self._pref:
                    w = w * PREFERENCE_FACTOR
                nadj = adj + w
                nreal = real + e.weight
                ns = (e.dest, e.line, nt) if track else (e.dest, e.line)
                if ns in settled:
                    continue
                nkey = self._key(nadj, nt)
                if ns not in best_key or nkey < best_key[ns]:
                    best_key[ns] = nkey
                    self._info[ns] = (nadj, nreal, nt)
                    self._edge_to[ns] = (state, e)
                    pq.push(nkey, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target."""
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while state in self._edge_to:
            prev, e = self._edge_to[state]
            edges.append(e)
            state = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                count += 1
        return count

"""Beijing Subway fare calculation.

Fare structure (distance rounded up to whole kilometers):
  up to 6 km: 3 yuan; up to 12 km: 4 yuan; up to 22 km: 5 yuan;
  up to 32 km: 6 yuan; beyond 32 km: +1 yuan per (started) 20 km.
"""

import math

# (upper bound in km inclusive, fare)
FARE_TABLE = [(6, 3), (12, 4), (22, 5), (32, 6)]
EXTENDED_BASE_KM = 32
EXTENDED_BASE_FARE = 6
EXTENDED_STEP_KM = 20

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError("Distance must be a number")
    if distance != distance or distance in (float("inf"), float("-inf")):
        raise ValueError("Distance must be finite")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters."""
    _validate_distance(distance)
    km = math.ceil(distance / 1000.0)
    for upper, fare in FARE_TABLE:
        if km <= upper:
            return fare
    extra = km - EXTENDED_BASE_KM
    return EXTENDED_BASE_FARE + int(math.ceil(extra / float(EXTENDED_STEP_KM)))


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes: 1.5 min/km plus 5 min per transfer."""
    _validate_distance(distance)
    if transfer_count is None or transfer_count < 0:
        raise ValueError("Transfer count must be non-negative")
    return float(distance / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER)

"""Data management: integrity checks and update operations."""

import json

from data_loader import DATA_FILE_PATH, load_subway_data

MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]
MAX_REASONABLE_DISTANCE = 50000


def _norm(s):
    return "".join(str(s).lower().split())


class SubwayDataManager:
    """Manage stations, connections and lines of the subway dataset."""

    def __init__(self, data=None, path=DATA_FILE_PATH):
        self.path = path
        self.data = data if data is not None else load_subway_data(path)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ---- lookup helpers ----
    def _station(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def station_names(self):
        return [s.get("name") for s in self.data["stations"]]

    def lines(self):
        result = []
        for st in self.data["stations"]:
            for l in st.get("lines", []):
                if l not in result:
                    result.append(l)
        for c in self.data["connections"]:
            l = c.get("line")
            if l and l not in result:
                result.append(l)
        return result

    def line_stations(self, line):
        """Station sequence of a line (ordered following connections)."""
        conns = [c for c in self.data["connections"] if c.get("line") == line]
        if not conns:
            return []
        adj = {}
        for c in conns:
            adj.setdefault(c["source"], []).append(c["dest"])
            adj.setdefault(c["dest"], []).append(c["source"])
        start = next((n for n, v in adj.items() if len(v) == 1), conns[0]["source"])
        seq, seen, cur = [], set(), start
        while cur is not None and cur not in seen:
            seq.append(cur)
            seen.add(cur)
            cur = next((n for n in adj[cur] if n not in seen), None)
        return seq

    def transfer_stations(self):
        return [s["name"] for s in self.data["stations"] if len(s.get("lines", [])) > 1]

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        if len(stations) < 2 or len(distances) != len(stations) - 1:
            raise ValueError("A line needs >= 2 stations and len(stations)-1 distances")
        for d in distances:
            if not isinstance(d, int) or d <= 0:
                raise ValueError("Distances must be positive integers (meters)")
        for name in stations:
            st = self._station(name)
            if st is None:
                self.data["stations"].append({"name": name, "lines": [line_name]})
            elif line_name not in st.setdefault("lines", []):
                st["lines"].append(line_name)
        for i, d in enumerate(distances):
            self.data["connections"].append(
                {"source": stations[i], "dest": stations[i + 1], "distance": d, "line": line_name})

    def update_station(self, old_name, new_name=None, lines=None):
        st = self._station(old_name)
        if st is None:
            raise KeyError("Station '%s' does not exist" % old_name)
        if lines is not None:
            st["lines"] = list(lines)
        if new_name and new_name != old_name:
            if self._station(new_name) is not None:
                raise ValueError("Station '%s' already exists" % new_name)
            st["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == old_name:
                    c["source"] = new_name
                if c.get("dest") == old_name:
                    c["dest"] = new_name

    def update_distance(self, a, b, distance, line=None):
        if not isinstance(distance, int) or distance <= 0:
            raise ValueError("Distance must be a positive integer (meters)")
        updated = 0
        for c in self.data["connections"]:
            if {c.get("source"), c.get("dest")} == {a, b} and (line is None or c.get("line") == line):
                c["distance"] = distance
                updated += 1
        if not updated:
            raise KeyError("No connection between '%s' and '%s'" % (a, b))
        return updated

    def save(self, path=None):
        with open(path or self.path, "w", encoding="utf-8") as fh:
            json.dump(self.data, fh, ensure_ascii=False, indent=2)

    # ---- integrity ----
    def check_integrity(self):
        errors, warnings = [], []
        names = set()
        for st in self.data["stations"]:
            n = st.get("name")
            if not n:
                errors.append("Station entry without name: %r" % (st,))
                continue
            if n in names:
                warnings.append("Duplicate station name: %s" % n)
            names.add(n)
            if not st.get("lines"):
                warnings.append("Station '%s' has no line affiliation" % n)
        connected = set()
        seen_pairs = set()
        for c in self.data["connections"]:
            s, d, dist = c.get("source"), c.get("dest"), c.get("distance")
            for end in (s, d):
                if end not in names:
                    errors.append("Connection references unknown station '%s'" % end)
            if s == d:
                errors.append("Self-loop connection at '%s'" % s)
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist <= 0:
                errors.append("Invalid distance %r between '%s' and '%s'" % (dist, s, d))
            elif dist > MAX_REASONABLE_DISTANCE:
                warnings.append("Unusually large distance %s m between '%s' and '%s'" % (dist, s, d))
            key = (frozenset((s, d)), c.get("line"))
            if key in seen_pairs:
                warnings.append("Duplicate connection '%s'-'%s' on %s" % (s, d, c.get("line")))
            seen_pairs.add(key)
            connected.add(s)
            connected.add(d)
        for n in names - connected:
            warnings.append("Station '%s' has no connections" % n)
        known = {_norm(l) for l in self.lines()}
        for line in MAIN_LINES:
            if _norm(line) not in known:
                warnings.append("Main line '%s' not found in data" % line)
        return {
            "errors": errors,
            "warnings": warnings,
            "station_count": len(names),
            "connection_count": len(self.data["connections"]),
            "line_count": len(self.lines()),
        }

"""Route report generation (text and JSON)."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return "%d meters (%.2f km)" % (meters, meters / 1000.0)
    return "%.2f km (%d meters)" % (meters / 1000.0, meters)


def build_route_info(start, end, mode, nodes, edges, number_to_station):
    names = [number_to_station[n] for n in nodes]
    distance = int(sum(e.weight for e in edges))
    segments = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            seg = segments[-1]
            seg["to"] = names[i + 1]
            seg["stops"] += 1
            seg["distance"] += e.weight
        else:
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1],
                             "stops": 1, "distance": e.weight})
    transfers = []
    for i in range(1, len(segments)):
        transfers.append({"station": segments[i]["from"],
                          "from_line": segments[i - 1]["line"],
                          "to_line": segments[i]["line"]})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    tcount = len(transfers)
    fare = Sum_money(distance)
    est = calculate_estimated_time(distance, tcount)
    return {
        "query": {"start": start, "end": end, "mode": mode},
        "path": names,
        "station_count": len(names),
        "distance": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time": round(est, 1),
        "transfer_count": tcount,
        "transfers": transfers,
        "lines": lines,
        "segments": segments,
    }


def format_text(info):
    out = []
    q = info["query"]
    out.append("=" * 50)
    out.append("Route: %s -> %s  (mode: %s)" % (q["start"], q["end"], q["mode"]))
    out.append("Stations (%d): %s" % (info["station_count"], " -> ".join(info["path"])))
    out.append("Segments:")
    for s in info["segments"]:
        out.append("  [%s] %s -> %s (%d stops, %d m)" % (s["line"], s["from"], s["to"], s["stops"], s["distance"]))
    if info["transfers"]:
        out.append("Transfer stations:")
        for t in info["transfers"]:
            out.append("  %s (%s -> %s)" % (t["station"], t["from_line"], t["to_line"]))
    else:
        out.append("Transfer stations: none")
    out.append("Lines involved: %s" % (", ".join(info["lines"]) or "-"))
    out.append("Total distance: %s" % format_distance(info["distance"]))
    out.append("Transfers: %d" % info["transfer_count"])
    out.append("Estimated time: %.1f minutes" % info["estimated_time"])
    out.append("Fare: %d yuan" % info["fare"])
    out.append("=" * 50)
    return "\n".join(out)


def format_json(info):
    return json.dumps(info, ensure_ascii=False, indent=2)

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import difflib
import json
import re
import sys

from data_loader import DATA_FILE_PATH, build_graph, load_subway_data
from data_manager import SubwayDataManager
from pathfinding import PathfindingConfig, ShortestPath, VALID_MODES
from report import build_route_info, format_json, format_text

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_query(line):
    parts = [p.strip() for p in re.split(r"[;,；，]", line)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERROR)
    return parts[0], parts[1]


def resolve_station(name, station_to_number):
    name = " ".join(name.split())
    if name in station_to_number:
        return name
    candidates = [name]
    for suffix in ("站", " Station", " station"):
        if name.endswith(suffix):
            candidates.append(name[: -len(suffix)].strip())
    lower = {k.lower(): k for k in station_to_number}
    for c in candidates:
        if c in station_to_number:
            return c
        if c.lower() in lower:
            return lower[c.lower()]
    return None


def _missing_msg(role, name, station_to_number):
    msg = "%s station '%s' does not exist, please check the station name" % (role, name)
    sugg = difflib.get_close_matches(name, list(station_to_number), n=3, cutoff=0.6)
    if sugg:
        msg += " (did you mean: %s?)" % ", ".join(sugg)
    return msg


def run_query(line, graph, station_to_number, number_to_station, config):
    start_raw, end_raw = parse_query(line)
    start = resolve_station(start_raw, station_to_number)
    if start is None:
        raise LookupError(_missing_msg("Starting", start_raw, station_to_number))
    end = resolve_station(end_raw, station_to_number)
    if end is None:
        raise LookupError(_missing_msg("Ending", end_raw, station_to_number))
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    try:
        sp = ShortestPath(graph, station_to_number[start], config)
        nodes, edges = sp.path_with_edges(station_to_number[end])
    except Exception as exc:
        raise RuntimeError("Path calculation error: %s" % exc)
    if not nodes:
        raise RuntimeError("No route found from '%s' to '%s' under current settings" % (start, end))
    try:
        return build_route_info(start, end, config.mode, nodes, edges, number_to_station)
    except Exception as exc:
        raise RuntimeError("Fare calculation error: %s" % exc)


def check_data(path):
    try:
        mgr = SubwayDataManager(path=path)
    except Exception as exc:
        print("Failed to load data: %s" % exc)
        return 1
    res = mgr.check_integrity()
    print("Stations: %d, Connections: %d, Lines: %d"
          % (res["station_count"], res["connection_count"], res["line_count"]))
    for e in res["errors"]:
        print("ERROR: %s" % e)
    for w in res["warnings"]:
        print("WARNING: %s" % w)
    if res["errors"]:
        print("Data integrity check FAILED (%d errors)" % len(res["errors"]))
        return 1
    print("Data integrity check passed (%d warnings)" % len(res["warnings"]))
    return 0

    args = parser.parse_args(argv)

        return check_data(args.data_file)

    if args.max_transfers is not None and args.max_transfers < 0:
        print("--max_transfers must be non-negative")

    prefs = []
    for p in args.line_preference or []:
        prefs.extend(x.strip() for x in p.split(",") if x.strip())

    try:

        graph, s2n, n2s = build_graph(data)
    except Exception as exc:
        print("Failed to load subway data: %s" % exc)

    config = PathfindingConfig(args.mode, args.max_transfers, prefs)

    json_results = []
    if args.output_file:
        try:
            open(args.output_file, "w", encoding="utf-8").close()
        except OSError as exc:
            print("Cannot open output file: %s" % exc)

    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:

            line = sys.stdin.readline()
        except KeyboardInterrupt:
            print()
            break
        if not line:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        try:
            info = run_query(line, graph, s2n, n2s, config)
        except (ValueError, LookupError, RuntimeError) as exc:
            print(str(exc))
            continue
        except Exception as exc:
            print("Unexpected error: %s" % exc)
            continue
        text = format_json(info) if args.format == "json" else format_text(info)
        print(text)
        if args.output_file:
            try:
                if args.format == "json":
                    json_results.append(info)
                    with open(args.output_file, "w", encoding="utf-8") as fh:
                        json.dump(json_results if len(json_results) > 1 else json_results[0],
                                  fh, ensure_ascii=False, indent=2)
                else:
                    with open(args.output_file, "a", encoding="utf-8") as fh:
                        fh.write(text + "\n")
            except OSError as exc:
                print("Failed to write output file: %s" % exc)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Data loading utilities for the Beijing Subway fare calculation system.

This module locates and loads ``subway_data.json`` and builds the network
data structures:

* ``station_to_number``: a dictionary mapping station names to numbers
* ``graph``: a weighted undirected graph (see :class:`models.Graph`)
"""

import json
import os

from models import Edge, Graph

# Absolute path of the data file, valid from any working directory.
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load the subway JSON data file and return its content as a dict.

    Raises:
        FileNotFoundError: if the file does not exist.
        ValueError: if the file is not valid JSON or has the wrong structure.
    """
    if path is None:
        path = DATA_FILE_PATH
    if not os.path.isfile(path):
        raise FileNotFoundError(f"Subway data file not found: {path}")
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Subway data file is not valid JSON: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data


def build_station_mapping(data):
    """Assign a number to every station.

    Stations are numbered in the order they appear in ``data['stations']``;
    stations referenced only by connections are numbered afterwards.

    Returns:
        (station_to_number, number_to_station)
    """
    station_to_number = {}
    number_to_station = {}

    def _register(name):
        if name not in station_to_number:
            num = len(station_to_number)
            station_to_number[name] = num
            number_to_station[num] = name

    for station in data.get("stations", []):
        name = station.get("name")
        if isinstance(name, str) and name:
            _register(name)
    for conn in data.get("connections", []):
        for key in ("source", "dest"):
            name = conn.get(key)
            if isinstance(name, str) and name:
                _register(name)
    return station_to_number, number_to_station


def build_graph(data):
    """Build the subway graph from loaded data.

    Returns:
        (graph, station_to_number, number_to_station)
    """
    station_to_number, number_to_station = build_station_mapping(data)
    graph = Graph()
    for num in number_to_station:
        graph.add_vertex(num)

    for station in data.get("stations", []):
        name = station.get("name")
        if isinstance(name, str) and name:
            graph.set_station_lines(name, list(station.get("lines", []) or []))

    for conn in data.get("connections", []):
        src = conn.get("source")
        dst = conn.get("dest")
        dist = conn.get("distance")
        line = conn.get("line", "")
        if src not in station_to_number or dst not in station_to_number:
            continue
        if src == dst:
            continue
        if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist < 0:
            continue
        graph.add_edge(Edge(station_to_number[src], station_to_number[dst], dist, line))
        # Make sure line affiliation information is consistent.
        for name in (src, dst):
            lines = graph.station_lines.setdefault(name, [])
            if line and line not in lines:
                lines.append(line)

    return graph, station_to_number, number_to_station


class SubwayNetwork:
    """Convenience container bundling the graph and station mappings."""

    def __init__(self, data):
        self.data = data
        self.graph, self.station_to_number, self.number_to_station = build_graph(data)

    @classmethod
    def from_file(cls, path=DATA_FILE_PATH):
        return cls(load_subway_data(path))

    def station_names(self):
        return list(self.station_to_number.keys())

class Edge:
    """A weighted edge between two stations (by number) on a given line."""

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def other(self, v):
        """Return the endpoint different from ``v``."""
        if v == self.source:
            return self.dest
        if v == self.dest:
            return self.source
        raise ValueError(f"Vertex {v} is not an endpoint of this edge")

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}m, {self.line!r})"

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.distance, self.line) == (
            other.source, other.dest, other.distance, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.distance, self.line))


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller. ``graph`` is a dictionary
    mapping a station number to the list of outgoing edges; each undirected
    connection is stored once in each direction.
    """

    def __init__(self):
        self.graph = {}          # station number -> list[Edge]
        self.station_lines = {}  # station name -> list[str]
        self._edge_count = 0

    # ---- vertex management -------------------------------------------
    def add_vertex(self, v):
        if v not in self.graph:
            self.graph[v] = []

    def has_vertex(self, v):
        return v in self.graph

    def vertices(self):
        return list(self.graph.keys())

    def V(self):
        return len(self.graph)

    def E(self):
        return self._edge_count

    # ---- station information -----------------------------------------
    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines) if lines else []

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return len(set(self.station_lines.get(station_name, []))) > 1

    # ---- edge management ---------------------------------------------
    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        self.graph[edge.dest].append(edge.reversed())
        self._edge_count += 1

    def adj(self, v):
        return self.graph.get(v, [])

    def edges(self):
        """Return each undirected edge once."""
        result = []
        seen = set()
        for v, edges in self.graph.items():
            for e in edges:
                key = (min(e.source, e.dest), max(e.source, e.dest), e.distance, e.line)
                if key in seen:
                    continue
                seen.add(key)
                result.append(e)
        return result

    def __repr__(self):
        return f"Graph(V={self.V()}, E={self.E()})"

"""Path calculation: priority queue and Dijkstra shortest path.

The search runs on an expanded state space ``(station, current line[,
transfers])`` so that transfers at multi-line interchange stations are
counted correctly and different optimisation modes are supported.
"""

import heapq
import itertools
import math

MODES = ("shortest_distance", "least_transfer", "comprehensive")


class MinPQ:
    """Heap-based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self):
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min(self):
        if not self._heap:
            raise IndexError("min from empty priority queue")
        key, _, item = self._heap[0]
        return key, item

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)


def _norm_line(name):
    return "".join(str(name).lower().split())


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None,
                 transfer_penalty=3000, preference_factor=0.3):
        if mode not in MODES:
            raise ValueError(f"Invalid mode '{mode}', must be one of {', '.join(MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []
        self.transfer_penalty = transfer_penalty
        self.preference_factor = preference_factor

    def is_preferred(self, line):
        if not self.preferred_lines:
            return True
        n = _norm_line(line)
        return any(_norm_line(p) == n for p in self.preferred_lines)


class ShortestPath:
    """Dijkstra's shortest path from a single source station number."""

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError(f"Source station number {source} is not in the graph")
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {v: math.inf for v in graph.vertices()}
        self.dist_to[source] = 0
        # state -> [distance, transfers, non-preferred distance, parent_state, edge]
        self._info = {}
        self._best_state = {}
        self._run()

    # ------------------------------------------------------------------
    def _key(self, dist, transfers, nonpref):
        cfg = self.config
        pf = cfg.preference_factor if cfg.preferred_lines else 0.0
        if cfg.mode == "least_transfer":
            return (transfers, dist + pf * nonpref, dist)
        if cfg.mode == "comprehensive":
            return (dist + cfg.transfer_penalty * transfers + pf * nonpref, transfers, dist)
        return (dist, transfers, nonpref)

    def _run(self):
        cfg = self.config
        limited = cfg.max_transfers is not None
        start = (self.source, None, 0 if limited else None)
        self._info[start] = [0, 0, 0, None, None]
        pq = MinPQ()
        pq.insert(self._key(0, 0, 0), start)
        settled = set()

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in settled:
                continue
            settled.add(state)
            v, line, _t = state
            d, tr, npd, _, _ = self._info[state]
            for e in self.graph.adj(v):
                w = e.dest
                ntr = tr + (1 if (line is not None and e.line != line) else 0)
                if limited and ntr > cfg.max_transfers:
                    continue
                ns = (w, e.line, ntr if limited else None)
                if ns in settled:
                    continue
                nd = d + e.weight
                nnp = npd + (0 if cfg.is_preferred(e.line) else e.weight)
                nk = self._key(nd, ntr, nnp)
                old = self._info.get(ns)
                if old is None or nk < self._key(old[0], old[1], old[2]):
                    self._info[ns] = [nd, ntr, nnp, state, e]
                    pq.insert(nk, ns)

        best_keys = {}
        for state in settled:
            v = state[0]
            info = self._info[state]
            k = self._key(info[0], info[1], info[2])
            if v not in best_keys or k < best_keys[v]:
                best_keys[v] = k
                self._best_state[v] = state
        self._best_state[self.source] = start
        for v, state in self._best_state.items():
            self.dist_to[v] = self._info[state][0]

    # ------------------------------------------------------------------
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target."""
        if target not in self._best_state:
            return [], []
        edges = []
        state = self._best_state[target]
        while True:
            info = self._info[state]
            parent, edge = info[3], info[4]
            if parent is None:
                break
            edges.append(edge)
            state = parent
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the computed path to target."""
        _, edges = self.path_with_edges(target)
        return count_transfers(edges)


def count_transfers(edges):
    transfers = 0
    for prev, cur in zip(edges, edges[1:]):
        if prev.line != cur.line:
            transfers += 1
    return transfers

"""Beijing Subway fare calculation."""

import math

# (upper bound in meters (exclusive), fare in yuan)
FARE_TIERS = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_START = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP = 20000

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MIN = 5.0


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number (meters)")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters.

    0-6 km: 3 yuan; 6-12 km: 4; 12-22 km: 5; 22-32 km: 6;
    over 32 km: 7 yuan plus 1 yuan for every further 20 km.
    Boundary values enter the higher tier; fractional meters are rounded up.
    """
    _validate_distance(distance)
    d = int(math.ceil(distance))
    for upper, fare in FARE_TIERS:
        if d < upper:
            return fare
    return EXTENDED_BASE_FARE + (d - EXTENDED_START) // EXTENDED_STEP


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes from distance (meters) and transfers."""
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, int) or transfer_count < 0:
        raise ValueError("transfer_count must be a non-negative integer")
    travel = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    return round(float(travel + transfer_count * TRANSFER_TIME_MIN), 1)

"""Route summary, visualization and export formatting."""

import json

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import count_transfers


def build_segments(stations, edges, number_to_station):
    """Group consecutive edges on the same line into segments."""
    segments = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            seg = segments[-1]
            seg["stations"].append(number_to_station[e.dest])
            seg["distance"] += e.distance
        else:
            segments.append({
                "line": e.line,
                "stations": [number_to_station[e.source], number_to_station[e.dest]],
                "distance": e.distance,
            })
    return segments


def build_route_info(start, end, stations, edges, number_to_station, config=None):
    """Build a summary dict of a computed route."""
    names = [number_to_station[s] for s in stations]
    distance = sum(e.distance for e in edges)
    transfers = count_transfers(edges)
    segments = build_segments(stations, edges, number_to_station)
    transfer_stations = [seg["stations"][0] for seg in segments[1:]]
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    fare = Sum_money(distance)
    est = calculate_estimated_time(distance, transfers)
    if isinstance(distance, float) and distance.is_integer():
        distance = int(distance)
    query = {"start": start, "end": end}
    if config is not None:
        query.update({
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "line_preference": list(config.preferred_lines),
        })
    return {
        "query": query,
        "path": names,
        "station_count": len(names),
        "distance": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time": est,
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": segments,
    }


def format_distance(meters):
    if meters < 1000:
        return f"{meters} m ({meters / 1000.0:.2f} km)"
    return f"{meters / 1000.0:.2f} km ({meters} m)"


def format_text_report(info):
    """Console-formatted text report."""
    q = info["query"]
    lines = []
    sep = "=" * 50
    lines.append(sep)
    lines.append(f"Route: {q['start']} -> {q['end']}")
    if "mode" in q:
        lines.append(f"Mode: {q['mode']}")
    lines.append("-" * 50)
    lines.append(f"Stations ({info['station_count']}): " + " -> ".join(info["path"]))
    lines.append("Line segments:")
    prev = None
    for seg in info["segments"]:
        if prev is not None:
            lines.append(f"  >> Transfer at {seg['stations'][0]}: {prev['line']} -> {seg['line']}")
        stops = len(seg["stations"]) - 1
        lines.append(
            f"  [{seg['line']}] {' -> '.join(seg['stations'])} "
            f"({stops} stop{'s' if stops != 1 else ''}, {seg['distance'] / 1000.0:.2f} km)")
        prev = seg
    lines.append("Transfer stations: " + (", ".join(info["transfer_stations"]) or "None"))
    lines.append("Lines involved: " + (", ".join(info["lines"]) or "None"))
    lines.append(f"Total distance: {format_distance(info['distance'])}")
    lines.append(f"Transfers: {info['transfer_count']}")
    lines.append(f"Estimated time: {info['estimated_time']} minutes")
    lines.append(f"Fare: {info['fare']} yuan")
    lines.append(sep)
    return "\n".join(lines)


def format_json_report(info, indent=2):
    """Structured JSON report."""
    return json.dumps(info, ensure_ascii=False, indent=indent)


def format_report(info, fmt="text"):
    if fmt == "json":
        return format_json_report(info)
    return format_text_report(info)

class SubwayDataManager:
    """Manage subway data (stations, connections, lines)."""

    def __init__(self, data=None, path=DATA_FILE_PATH):
        self.path = path
        self.data = data if data is not None else load_subway_data(path)
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    # ---- queries -------------------------------------------------------
    def station_names(self):
        return [s.get("name") for s in self.data["stations"]]

    def get_station(self, name):
        for s in self.data["stations"]:
            if s.get("name") == name:
                return s
        return None

    def get_lines(self):
        lines = []
        for s in self.data["stations"]:
            for ln in s.get("lines", []) or []:
                if ln not in lines:
                    lines.append(ln)
        for c in self.data["connections"]:
            ln = c.get("line")
            if ln and ln not in lines:
                lines.append(ln)
        return lines

    def get_line_sequences(self):
        """Derive station sequences per line from connections."""
        adjacency = {}
        for c in self.data["connections"]:
            ln = c.get("line")
            a, b = c.get("source"), c.get("dest")
            adj = adjacency.setdefault(ln, {})
            adj.setdefault(a, set()).add(b)
            adj.setdefault(b, set()).add(a)
        sequences = {}
        for ln, adj in adjacency.items():
            visited = set()
            seq = []
            ends = sorted([s for s, n in adj.items() if len(n) == 1]) or sorted(adj.keys())
            for start in ends + sorted(adj.keys()):
                if start in visited:
                    continue
                cur, prev = start, None
                while cur is not None and cur not in visited:
                    visited.add(cur)
                    seq.append(cur)
                    nxt = [n for n in sorted(adj[cur]) if n != prev and n not in visited]
                    prev, cur = cur, (nxt[0] if nxt else None)
            sequences[ln] = seq
        return sequences

    def build_graph(self):
        return build_graph(self.data)

    # ---- integrity check ---------------------------------------------
    def check_integrity(self):
        errors, warnings, info = [], [], []
        stations = self.data.get("stations", [])
        connections = self.data.get("connections", [])
        names = set()
        for i, s in enumerate(stations):
            name = s.get("name") if isinstance(s, dict) else None
            if not isinstance(name, str) or not name.strip():
                errors.append(f"Station #{i} has no valid name")
                continue
            if name in names:
                warnings.append(f"Duplicate station entry: '{name}'")
            names.add(name)
            if not s.get("lines"):
                warnings.append(f"Station '{name}' has no line affiliation")
        info.append(f"Stations: {len(names)}")
        info.append(f"Connections: {len(connections)}")

        # Coverage of main lines
        all_lines = self.get_lines()
        normalized = {_norm(ln) for ln in all_lines}
        info.append(f"Lines: {len(all_lines)}")
        for main, aliases in MAIN_LINES.items():
            if _norm(main) not in normalized and not any(_norm(a) in normalized for a in aliases):
                warnings.append(f"Main line '{main}' not found in data")
        for line, groups in REPRESENTATIVE_STATIONS.items():
            for group in groups:
                if not any(n in names for n in group):
                    warnings.append(f"Representative station '{group[0]}' ({line}) not found")

        # Connections
        seen = set()
        connected = set()
        for i, c in enumerate(connections):
            src, dst = c.get("source"), c.get("dest")
            dist, line = c.get("distance"), c.get("line")
            label = f"Connection #{i} ({src} - {dst})"
            if src not in names:
                errors.append(f"{label}: unknown source station '{src}'")
            if dst not in names:
                errors.append(f"{label}: unknown destination station '{dst}'")
            if src == dst:
                errors.append(f"{label}: connects a station to itself")
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or \
                    (isinstance(dist, float) and (math.isnan(dist) or math.isinf(dist))):
                errors.append(f"{label}: invalid distance {dist!r}")
            elif dist <= 0:
                errors.append(f"{label}: non-positive distance {dist}")
            elif dist > MAX_REASONABLE_DISTANCE:
                warnings.append(f"{label}: unusually long distance {dist} m")
            elif dist < MIN_REASONABLE_DISTANCE:
                warnings.append(f"{label}: unusually short distance {dist} m")
            if not line:
                warnings.append(f"{label}: no line specified")
            key = (frozenset((src, dst)), line)
            if key in seen:
                warnings.append(f"{label}: duplicate connection on {line}")
            seen.add(key)
            connected.add(src)
            connected.add(dst)
            for n in (src, dst):
                st = self.get_station(n) if n in names else None
                if st is not None and line and line not in (st.get("lines") or []):
                    warnings.append(f"{label}: station '{n}' not listed on line '{line}'")
        for n in sorted(names - connected):
            warnings.append(f"Station '{n}' has no connections")

        if names:
            adj = {}
            for c in connections:
                a, b = c.get("source"), c.get("dest")
                adj.setdefault(a, set()).add(b)
                adj.setdefault(b, set()).add(a)
            start = next(iter(sorted(names)))
            stack, comp = [start], {start}
            while stack:
                v = stack.pop()
                for w in adj.get(v, ()):
                    if w not in comp:
                        comp.add(w)
                        stack.append(w)
            unreachable = names - comp
            if unreachable:
                warnings.append(f"Network not fully connected: {len(unreachable)} station(s) unreachable from '{start}'")
            else:
                info.append("Network is fully connected")

        return {"passed": not errors, "errors": errors, "warnings": warnings, "info": info}

Usage examples:

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
SEPARATOR_RE = re.compile(r"[;；,，]")


class QueryError(Exception):
    """User-facing query error."""


def _normalize(name):
    s = unicodedata.normalize("NFKC", name).strip().lower()
    s = " ".join(s.split())
    return s


def _strip_suffix(s):
    for suf in (" station", "站"):
        if s.endswith(suf) and len(s) > len(suf):
            return s[: -len(suf)].strip()
    return s


class StationResolver:
    """Standardize user-entered station names to official names."""

    def __init__(self, names):
        self.names = set(names)
        self.lookup = {}
        self.stripped = {}
        for n in names:
            self.lookup.setdefault(_normalize(n), n)
            self.lookup.setdefault(_normalize(n).replace(" ", ""), n)
            self.stripped.setdefault(_strip_suffix(_normalize(n)), n)

    def resolve(self, raw):
        if raw in self.names:
            return raw
        key = _normalize(raw)
        for candidate in (key, key.replace(" ", "")):
            if candidate in self.lookup:
                return self.lookup[candidate]
        stripped = _strip_suffix(key)
        if stripped in self.lookup:
            return self.lookup[stripped]
        if stripped in self.stripped:
            return self.stripped[stripped]
        return None


def parse_query(line):
    parts = SEPARATOR_RE.split(line)
    if len(parts) != 2:
        raise QueryError(FORMAT_ERROR)
    start, end = parts[0].strip(), parts[1].strip()
    if not start or not end:
        raise QueryError(FORMAT_ERROR)
    return start, end


def run_query(network, resolver, start_raw, end_raw, config):
    start = resolver.resolve(start_raw)
    if start is None:
        raise QueryError(f"Starting station '{start_raw}' does not exist, please check the station name")
    end = resolver.resolve(end_raw)
    if end is None:
        raise QueryError(f"Ending station '{end_raw}' does not exist, please check the station name")
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")
    s = network.station_to_number[start]
    t = network.station_to_number[end]
    try:
        sp = ShortestPath(network.graph, s, config)
        stations, edges = sp.path_with_edges(t)
    except Exception as exc:  # path calculation exception
        raise QueryError(f"Path calculation error: {exc}") from exc
    if not edges:
        msg = f"No path found from '{start}' to '{end}'"
        if config.max_transfers is not None:
            msg += f" within {config.max_transfers} transfer(s)"
        raise QueryError(msg)
    try:
        return build_route_info(start, end, stations, edges, network.number_to_station, config)
    except (ValueError, TypeError) as exc:
        raise QueryError(f"Fare calculation error: {exc}") from exc

def _write_output(path, fmt, results):
    try:
        with open(path, "w", encoding="utf-8") as fh:
            if fmt == "json":
                payload = results[0] if len(results) == 1 else results
                json.dump(payload, fh, ensure_ascii=False, indent=2)
                fh.write("\n")
            else:
                fh.write("\n\n".join(format_report(r, "text") for r in results))
                fh.write("\n")
        return True
    except OSError as exc:
        print(f"Failed to write output file '{path}': {exc}")
        return False

    args = parse_args(argv)
    try:
        data = load_subway_data(args.data_file)

    preferred = [s for s in preferred if s]

"""Global variable store shared across all modules of the dispatch system.

Usage::

    import globalvar
    globalvar._init()
    globalvar.set_value("riders", [])
    riders = globalvar.get_value("riders")
"""

_global_dict = None


def _init():
    """(Re)initialise the global dictionary, discarding any previous values."""
    global _global_dict
    _global_dict = {}


def is_initialized():
    """Return True when the global store has been initialised."""
    return _global_dict is not None


def set_value(key, value):
    """Store ``value`` under ``key``. Initialises the store lazily if needed."""
    global _global_dict
    if _global_dict is None:
        _init()
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` or ``default`` if missing."""
    if _global_dict is None:
        return default
    return _global_dict.get(key, default)


def keys():
    """Return a list of all keys currently stored."""
    if _global_dict is None:
        return []
    return list(_global_dict.keys())

"""Configuration handling (JSON based) for the dispatch system."""

import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 10000,
    "recruitment_cost": 1000,
    "max_riders": 20,
    "rider_speed": 50,
    "rider_spawn": [400, 400],
    "order_revenue": 0,
    "state_file": "dispatch_state.json",
    "autosave": True,
}


class Configuration:
    """Loads system parameters from a JSON file, falling back to defaults.

    Attributes:
        file_path: Path of the configuration file (may be None).
        data: Dictionary of the effective configuration.
    """

    def __init__(self, file_path=None):
        self.file_path = file_path
        self.data = dict(DEFAULT_CONFIG)
        if file_path:
            self.load(file_path)

    @property
    def initial_funds(self):
        """Initial system funds."""
        return int(self.data["initial_funds"])

    @property
    def recruitment_cost(self):
        """Cost of recruiting one rider."""
        return int(self.data["recruitment_cost"])

    def load(self, file_path):
        """Load configuration values from ``file_path`` if it exists.

        Invalid files are reported and ignored (defaults remain in effect).
        """
        if not os.path.exists(file_path):
            logger.info("Config file %s not found, using defaults", file_path)
            return
        try:
            with open(file_path, "r", encoding="utf-8") as fh:
                loaded = json.load(fh)
            if not isinstance(loaded, dict):
                raise ValueError("configuration root must be an object")
            for key, value in loaded.items():
                if key in DEFAULT_CONFIG:
                    self.data[key] = value
            self._sanitize()
        except (OSError, ValueError) as exc:
            logger.warning("Could not load config %s: %s", file_path, exc)
            self.data = dict(DEFAULT_CONFIG)

    def _sanitize(self):
        """Coerce configuration values into valid types/ranges."""
        try:
            self.data["initial_funds"] = int(self.data["initial_funds"])
            self.data["recruitment_cost"] = max(0, int(self.data["recruitment_cost"]))
            self.data["max_riders"] = max(1, int(self.data["max_riders"]))
            self.data["rider_speed"] = max(1, int(self.data["rider_speed"]))
            self.data["order_revenue"] = int(self.data["order_revenue"])
            spawn = self.data["rider_spawn"]
            sx = max(0, min(800, int(spawn[0])))
            sy = max(0, min(800, int(spawn[1])))
            # riders must start on a road
            if sx % 100 != 0 and sy % 100 != 0:
                sx = (sx // 100) * 100
            self.data["rider_spawn"] = [sx, sy]
        except (TypeError, ValueError, IndexError, KeyError) as exc:
            logger.warning("Invalid configuration values (%s); using defaults", exc)
            self.data = dict(DEFAULT_CONFIG)

    def get(self, key, default=None):
        """Return the configuration value for ``key``."""
        return self.data.get(key, default)

"""Grid map definitions: 800x800 coordinate system split into 8x8 cells.

Streets run along every multiple of 100 (both horizontally and vertically).
Restaurants and customers are located at cell centres (``x50`` coordinates).
"""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIMENSION = MAP_SIZE // CELL_SIZE


class GridMap:
    """Static helper describing the grid map."""

    map_size = MAP_SIZE
    cell_size = CELL_SIZE
    grid_dimension = GRID_DIMENSION

    @staticmethod
    def clamp(value):
        """Clamp a coordinate into the range [0, MAP_SIZE]."""
        return max(0, min(MAP_SIZE, int(value)))

    @staticmethod
    def to_center(value):
        """Convert a raw coordinate into its grid cell centre.

        ``center = (v // 100) * 100 + 50``; values out of range are clamped
        and the upper boundary (800) maps to the last cell (750).
        """
        v = GridMap.clamp(value)
        if v >= MAP_SIZE:
            v = MAP_SIZE - 1
        return (v // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2

    @staticmethod
    def to_center_point(x, y):
        """Convert a raw point to the centre point of its grid cell."""
        return GridMap.to_center(x), GridMap.to_center(y)

    @staticmethod
    def cell_of(x, y):
        """Return the (column, row) cell indices containing the point."""
        col = min(GRID_DIMENSION - 1, max(0, int(x) // CELL_SIZE))
        row = min(GRID_DIMENSION - 1, max(0, int(y) // CELL_SIZE))
        return col, row

    @staticmethod
    def in_bounds(x, y):
        """True if the point lies within the map."""
        return 0 <= x <= MAP_SIZE and 0 <= y <= MAP_SIZE

    @staticmethod
    def is_horizontal_street(y):
        """True if ``y`` lies on a horizontal street."""
        return int(y) % CELL_SIZE == 0

    @staticmethod
    def is_vertical_street(x):
        """True if ``x`` lies on a vertical street."""
        return int(x) % CELL_SIZE == 0

    @staticmethod
    def is_road(x, y):
        """True if the point lies on any street."""
        return GridMap.is_horizontal_street(y) or GridMap.is_vertical_street(x)

"""Grid-based path planning restricted to streets."""

import math

from grid_map import CELL_SIZE


class Route:
    """A planned route: ordered list of axis-aligned waypoints."""

    def __init__(self, points, speed=50):
        self.points = list(points)
        self.speed = max(1, int(speed))

    @property
    def length(self):
        """Total route length in map units."""
        return PathPlanner.route_length(self.points)

    @property
    def estimated_time(self):
        """Estimated travel time in ticks."""
        return int(math.ceil(self.length / self.speed)) if self.points else 0

    def to_dict(self):
        """Serialise the route."""
        return {
            "points": [list(p) for p in self.points],
            "speed": self.speed,
            "length": self.length,
            "estimated_time": self.estimated_time,
        }


class PathPlanner:
    """Plans street-following routes on the 8x8 grid.

    Riders travel along streets (lines at multiples of 100). Destinations
    located in the middle of a cell are entered from the closest adjacent
    horizontal street. Riders do not block each other, so no path conflicts
    or deadlocks can occur.
    """

    @staticmethod
    def plan(start, target):
        """Plan a route from ``start`` to ``target``.

        Returns a list of waypoints including both start and target. Every
        consecutive pair of waypoints is horizontally or vertically aligned.
        """
        x, y = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        cs = CELL_SIZE
        points = [(x, y)]
        if (x, y) == (tx, ty):
            return points

        # Step 1: if inside a cell (e.g. at a restaurant), go to a horizontal
        # street, preferring the direction of the target.
        if x % cs and y % cs:
            if ty > y:
                ny = (y // cs + 1) * cs
            elif ty < y:
                ny = (y // cs) * cs
            else:
                ny = (y // cs) * cs if y % cs <= cs // 2 else (y // cs + 1) * cs
            y = ny
            points.append((x, y))

        # Entry street for the target.
        if ty % cs == 0:
            ey = ty
        elif y <= ty:
            ey = (ty // cs) * cs
        else:
            ey = (ty // cs + 1) * cs

        if not (y % cs == 0 and y == ey):
            # Need vertical travel -> must be on a vertical street.
            if x % cs:
                if tx % cs == 0:
                    nx = tx
                else:
                    lo = (tx // cs) * cs
                    hi = lo + cs
                    nx = lo if abs(x - lo) <= abs(x - hi) else hi
                x = nx
                points.append((x, y))
            y = ey
            points.append((x, y))
        if x != tx:
            x = tx
            points.append((x, y))
        if y != ty:
            y = ty
            points.append((x, y))
        return PathPlanner._dedupe(points)

    @staticmethod
    def plan_multi(start, targets):
        """Plan a continuous route through several destinations in order."""
        points = [(int(start[0]), int(start[1]))]
        current = points[0]
        for target in targets:
            leg = PathPlanner.plan(current, target)
            points.extend(leg[1:])
            current = (int(target[0]), int(target[1]))
        return PathPlanner._dedupe(points)

    @staticmethod
    def build_route(start, targets, speed=50):
        """Build a :class:`Route` through ``targets`` starting at ``start``."""
        return Route(PathPlanner.plan_multi(start, targets), speed)

    @staticmethod
    def route_length(points):
        """Sum of Manhattan distances between consecutive waypoints."""
        total = 0
        for (x1, y1), (x2, y2) in zip(points, points[1:]):
            total += abs(x2 - x1) + abs(y2 - y1)
        return total

    @staticmethod
    def distance(a, b):
        """Street distance between two points."""
        return PathPlanner.route_length(PathPlanner.plan(a, b))

    @staticmethod
    def is_special_case(start, target):
        """True when the start point is inside a cell directly above/below the target."""
        x, y = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        return x == tx and x % CELL_SIZE != 0 and y != ty

    @staticmethod
    def _dedupe(points):
        result = []
        for p in points:
            p = (int(p[0]), int(p[1]))
            if not result or result[-1] != p:
                result.append(p)
        return result

"""Order data model."""

import globalvar
from grid_map import GridMap
from path_planner import PathPlanner

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def _to_int(value, label):
    if isinstance(value, bool):
        raise ValueError(f"{label} must be an integer")
    if isinstance(value, int):
        return value
    if isinstance(value, float):
        if value.is_integer():
            return int(value)
        raise ValueError(f"{label} must be an integer")
    if isinstance(value, str):
        try:
            return int(value.strip())
        except ValueError:
            raise ValueError(f"{label} must be an integer") from None
    raise ValueError(f"{label} must be an integer")


def _parse_point(pos, label):
    try:
        if len(pos) != 2:
            raise ValueError
    except (TypeError, ValueError):
        raise ValueError(f"{label} position must contain two coordinates") from None
    return _to_int(pos[0], f"{label} x"), _to_int(pos[1], f"{label} y")


class Order:
    """A delivery order from a restaurant to a customer.

    Coordinates are automatically converted to grid cell centres.

    Attributes:
        id: Unique order id.
        status: "pending", "assigned", "picking_up", "delivering", "completed".
    """

    def __init__(self, res_pos, cus_pos):
        rx, ry = _parse_point(res_pos, "restaurant")
        cx, cy = _parse_point(cus_pos, "customer")
        self.raw_input = [rx, ry, cx, cy]
        self.restaurant_x, self.restaurant_y = GridMap.to_center_point(rx, ry)
        self.customer_x, self.customer_y = GridMap.to_center_point(cx, cy)

        counter = int(globalvar.get_value("order_id_counter", 0) or 0) + 1
        globalvar.set_value("order_id_counter", counter)
        self.id = counter
        self.status = STATUS_PENDING
        self.rider_id = None
        self.start_time = int(globalvar.get_value("current_time", 0) or 0)
        self.assign_time = None
        self.accept_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.efficiency = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)

        orders = globalvar.get_value("orders")
        if not isinstance(orders, list):
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(self)

    @property
    def order_id(self):
        """Alias of :attr:`id`."""
        return self.id

    @property
    def res_pos(self):
        """Restaurant position (grid centre)."""
        return (self.restaurant_x, self.restaurant_y)

    @property
    def cus_pos(self):
        """Customer position (grid centre)."""
        return (self.customer_x, self.customer_y)

    def as_list(self):
        """Return ``[restaurant_x, restaurant_y, customer_x, customer_y]``."""
        return [self.restaurant_x, self.restaurant_y, self.customer_x, self.customer_y]

    def assign_to(self, rider_id):
        """Assign the order to the rider with ``rider_id``."""
        self.rider_id = rider_id
        if self.status == STATUS_PENDING:
            self.status = STATUS_ASSIGNED
        self.assign_time = int(globalvar.get_value("current_time", 0) or 0)

    def is_completed(self):
        """True once the order has been delivered."""
        return self.status == STATUS_COMPLETED

    def complete(self, time):
        """Mark the order as delivered at ``time``."""
        self.status = STATUS_COMPLETED
        self.completion_time = int(time)
        self.time_taken = self.completion_time - self.start_time
        if self.time_taken > 0:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = None

    def to_dict(self):
        """Serialise the order."""
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "raw_input": list(self.raw_input),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "assign_time": self.assign_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "efficiency": self.efficiency,
        }

    def __repr__(self):
        return (f"Order(id={self.id}, res={self.res_pos}, cus={self.cus_pos}, "
                f"status={self.status}, rider={self.rider_id})")

"""Rider model: movement, state machine and status logging."""

import logging
from enum import IntEnum

import globalvar
from config import DEFAULT_CONFIG
from grid_map import GridMap
from order_model import (Order, STATUS_COMPLETED, STATUS_DELIVERING,
                         STATUS_PICKING_UP)
from path_planner import PathPlanner

logger = logging.getLogger(__name__)


class RiderState(IntEnum):
    """Rider status enumeration."""

    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2


STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


def _now():
    return int(globalvar.get_value("current_time", 0) or 0)


class Rider:
    """A delivery rider.

    Attributes:
        rider_id: Unique id.
        x, y: Current coordinates.
        speed: Units moved per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: Assigned orders (processed in FIFO order).
        special_case: True while following a special-case route.
        log: List of status-change dicts (keys: time, event, order_id, x, y, state).
    """

    def __init__(self, rider_id):
        cfg = globalvar.get_value("config")
        if not isinstance(cfg, dict):
            cfg = DEFAULT_CONFIG
        spawn = cfg.get("rider_spawn", DEFAULT_CONFIG["rider_spawn"])
        self.rider_id = int(rider_id)
        self.x = int(spawn[0])
        self.y = int(spawn[1])
        self.speed = max(1, int(cfg.get("rider_speed", DEFAULT_CONFIG["rider_speed"])))
        self.state = int(RiderState.IDLE)
        self.orders = []
        self.special_case = False
        self.route = []
        self.log = []
        self.distance_travelled = 0
        self.completed_orders = 0
        self._current_order = None
        self._record("spawn", None)

    # ------------------------------------------------------------------ helpers
    @property
    def position(self):
        """Current position as a tuple."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    def active_orders(self):
        """Orders not yet completed."""
        return [o for o in self.orders if getattr(o, "status", None) != STATUS_COMPLETED]

    def is_idle(self):
        """True when the rider is idle and has no pending orders."""
        return self.state == RiderState.IDLE and not self.active_orders()

    def last_position(self):
        """Position where the rider will be after finishing all assigned orders."""
        active = self.active_orders()
        if active:
            return active[-1].cus_pos
        return self.position

    def estimate_workload(self):
        """Estimated remaining travel distance for all assigned orders."""
        active = self.active_orders()
        if not active:
            return 0
        total = PathPlanner.route_length([self.position] + list(self.route))
        pos = self.route[-1] if self.route else self.position
        for idx, order in enumerate(active):
            if idx == 0 and order is self._current_order:
                if self.state == RiderState.PICKING_UP:
                    total += order.distance
                pos = order.cus_pos
                continue
            total += PathPlanner.distance(pos, order.res_pos) + order.distance
            pos = order.cus_pos
        return total

    def _record(self, event, order_id):
        entry = {
            "time": _now(),
            "event": event,
            "order_id": order_id,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        }
        self.log.append(entry)
        logger.debug("Rider %s: %s", self.rider_id, entry)
        return entry

    def _plan_to(self, target):
        points = PathPlanner.plan(self.position, target)
        self.special_case = PathPlanner.is_special_case(self.position, target)
        self.route = points[1:]

    def _advance(self, budget):
        """Move along the route using up to ``budget`` units. Returns leftover."""
        while budget > 0 and self.route:
            tx, ty = self.route[0]
            dist = abs(tx - self.x) + abs(ty - self.y)
            if dist <= budget:
                self.x, self.y = tx, ty
                budget -= dist
                self.distance_travelled += dist
                self.route.pop(0)
            else:
                if tx != self.x:
                    self.x += budget if tx > self.x else -budget
                else:
                    self.y += budget if ty > self.y else -budget
                self.distance_travelled += budget
                budget = 0
        return budget

    def _purge_completed(self):
        self.orders[:] = [o for o in self.orders
                          if getattr(o, "status", None) != STATUS_COMPLETED]

    def _coerce_orders(self):
        for idx, order in enumerate(self.orders):
            if not isinstance(order, Order) and not hasattr(order, "res_pos"):
                try:
                    if len(order) == 4:
                        new = Order((order[0], order[1]), (order[2], order[3]))
                        new.assign_to(self.rider_id)
                        self.orders[idx] = new
                except TypeError:
                    logger.warning("Rider %s: discarding invalid order %r",
                                   self.rider_id, order)
                    self.orders[idx] = None
        self.orders[:] = [o for o in self.orders if o is not None]

    def _start_order(self, order):
        self._current_order = order
        if getattr(order, "rider_id", None) is None:
            order.rider_id = self.rider_id
        if order.pickup_time is not None:
            self.state = int(RiderState.DELIVERING)
            order.status = STATUS_DELIVERING
            self._plan_to(order.cus_pos)
            return
        self.state = int(RiderState.PICKING_UP)
        order.status = STATUS_PICKING_UP
        if order.accept_time is None:
            order.accept_time = _now()
            self._record("accepted_order", order.id)
        self._plan_to(order.res_pos)

    def _finish_order(self, order):
        now = _now()
        order.complete(now)
        self._record("delivered_to_customer", order.id)
        if order in self.orders:
            self.orders.remove(order)
        self.completed_orders += 1
        self.state = int(RiderState.IDLE)
        self._current_order = None
        self.route = []
        self.special_case = False
        cfg = globalvar.get_value("config")
        revenue = 0
        if isinstance(cfg, dict):
            revenue = int(cfg.get("order_revenue", 0) or 0)
        if revenue:
            funds = int(globalvar.get_value("funds", 0) or 0)
            globalvar.set_value("funds", funds + revenue)

    # ------------------------------------------------------------------ public
    def move(self):
        """Advance the rider by one tick.

        Handles all state transitions (idle -> picking up -> delivering ->
        idle) and writes the corresponding log entries.
        """
        self._coerce_orders()
        self._purge_completed()
        if not self.orders:
            if self.state != RiderState.IDLE:
                self.state = int(RiderState.IDLE)
                self.route = []
                self._current_order = None
                self.special_case = False
            return

        order = self.orders[0]
        if self.state == RiderState.IDLE or self._current_order is not order:
            self._start_order(order)

        budget = self.speed
        for _ in range(10):
            target = order.res_pos if self.state == RiderState.PICKING_UP else order.cus_pos
            if not self.route and self.position != target:
                self._plan_to(target)
            budget = self._advance(budget)
            if self.route:
                break
            if self.state == RiderState.PICKING_UP:
                self.state = int(RiderState.DELIVERING)
                order.pickup_time = _now()
                order.status = STATUS_DELIVERING
                self._record("arrived_at_merchant", order.id)
                self._plan_to(order.cus_pos)
                continue
            self._finish_order(order)
            break

        self.x = GridMap.clamp(self.x)
        self.y = GridMap.clamp(self.y)

    def reset_route(self):
        """Recovery helper: drop the current route so it is re-planned."""
        self.route = []
        self.special_case = False

    def to_dict(self):
        """Serialise the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [getattr(o, "id", None) for o in self.orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "completed_orders": self.completed_orders,
            "log": list(self.log),
        }

    def __repr__(self):
        return (f"Rider(id={self.rider_id}, pos=({self.x},{self.y}), "
                f"state={self.state}, orders={len(self.orders)})")

class SystemManager:
    """Manages funds, riders, simulated time and system state."""

    def __init__(self):

    # -------------------------------------------------------------- properties
    @property

    @property

        orders = globalvar.get_value("orders")

        return orders

        return int(globalvar.get_value("rider_id_counter", 0) or 0)

    @property
    def running(self):
        """Main loop running flag."""
        return bool(globalvar.get_value("running", True))

    @running.setter
    def running(self, value):
        globalvar.set_value("running", bool(value))

    @property
    def current_time(self):
        """Current simulated time (ticks)."""
        return int(globalvar.get_value("current_time", 0) or 0)

    @property
    def recruitment_cost(self):
        """Cost of recruiting a rider."""
        return int(globalvar.get_value("recruitment_cost",
                                       DEFAULT_CONFIG["recruitment_cost"]))

    @property
    def max_riders(self):
        """Maximum number of riders."""
        return int(self.config.get("max_riders", DEFAULT_CONFIG["max_riders"]))

    # ----------------------------------------------------------------- riders
    def can_recruit(self):
        """True if funds allow recruiting another rider and the cap is not reached."""
        return (self.funds >= self.recruitment_cost
                and len(self.riders) < self.max_riders)

    def recruit_rider(self, force=False):
        """Recruit a new rider, deducting the recruitment cost.

        Args:
            force: Recruit even if funds are insufficient (funds go negative).

        Returns:
            The new :class:`Rider` or None if recruitment was not possible.
        """
        if not force and not self.can_recruit():
            return None
        riders = self.riders
        max_existing = max((r.rider_id for r in riders), default=0)
        new_id = max(self.rider_id_counter, max_existing) + 1
        globalvar.set_value("rider_id_counter", new_id)
        rider = Rider(new_id)
        self.funds = self.funds - self.recruitment_cost
        riders.append(rider)
        logger.info("Recruited rider %s; funds now %s", new_id, self.funds)
        return rider

    def get_rider(self, rider_id):
        """Return the rider with ``rider_id`` or None."""
        for rider in self.riders:
            if rider.rider_id == rider_id:
                return rider
        return None

    # -------------------------------------------------------------- simulation
    def tick(self, count=1):
        """Advance the simulation by ``count`` ticks.

        Returns:
            List of ``(rider_id, log_entry)`` tuples for events that occurred.
        """
        events = []
        for _ in range(max(0, int(count))):
            globalvar.set_value("current_time", self.current_time + 1)
            for rider in list(self.riders):
                before = len(rider.log)
                try:
                    rider.move()
                except Exception as exc:  # pylint: disable=broad-except
                    logger.error("Rider %s failed to move: %s", rider.rider_id, exc)
                    rider.reset_route()
                    self.check_consistency()
                for entry in rider.log[before:]:
                    events.append((rider.rider_id, entry))
        return events

    def has_pending_work(self):
        """True if any rider still has orders to deliver."""
        return any(r.active_orders() for r in self.riders)

    def check_consistency(self):
        """Detect and repair inconsistent state. Returns list of issues fixed."""
        issues = []
        for rider in self.riders:
            if rider.state not in (0, 1, 2):
                issues.append(f"rider {rider.rider_id} invalid state {rider.state}")
                rider.state = 0
                rider.reset_route()
            if not rider.active_orders() and rider.state != 0:
                issues.append(f"rider {rider.rider_id} busy without orders")
                rider.state = 0
                rider.reset_route()
            cx = max(0, min(800, int(rider.x)))
            cy = max(0, min(800, int(rider.y)))
            if (cx, cy) != (rider.x, rider.y):
                issues.append(f"rider {rider.rider_id} out of bounds")
                rider.x, rider.y = cx, cy
                rider.reset_route()
        if issues:
            logger.warning("Consistency issues fixed: %s", issues)
        return issues

    # ------------------------------------------------------------- persistence
    def to_dict(self):
        """Serialise the full system state."""
        return {
            "config": self.config,
            "funds": self.funds,
            "current_time": self.current_time,
            "rider_id_counter": self.rider_id_counter,
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.orders],
        }

    def save_state(self, path):
        """Write the system state to ``path`` as JSON. Returns True on success."""
        try:
            with open(path, "w", encoding="utf-8") as fh:
                json.dump(self.to_dict(), fh, indent=2)
            return True
        except (OSError, TypeError, ValueError) as exc:
            logger.error("Failed to save state to %s: %s", path, exc)
            return False

"""Order management: creation, assignment and statistics."""

import logging

import globalvar
from order_model import Order
from path_planner import PathPlanner
from system import SystemManager

logger = logging.getLogger(__name__)


def find_best_rider_for_order(order):
    """Select the best rider for ``order``.

    Strategy:
      1. The nearest idle rider (street distance to the restaurant); ties are
         broken by fewer completed orders (load balancing) then rider id.
      2. If no rider is idle and funds allow, recruit a new rider.
         If there are no riders at all, a rider is always recruited.
      3. Otherwise the busy rider with the least estimated workload.

    The order is NOT appended to the rider's order list.
    """
    riders = globalvar.get_value("riders")
    if not isinstance(riders, list):
        riders = []
        globalvar.set_value("riders", riders)
    res = order.res_pos

    idle = [r for r in riders if r.is_idle()]
    if idle:
        return min(idle, key=lambda r: (PathPlanner.distance(r.position, res),
                                        r.completed_orders, r.rider_id))

    manager = SystemManager()
    if not riders or manager.can_recruit():
        rider = manager.recruit_rider(force=not riders)
        if rider is not None:
            return rider

    return min(riders, key=lambda r: (
        r.estimate_workload() + PathPlanner.distance(r.last_position(), res),
        len(r.active_orders()), r.rider_id))


class OrderManager:
    """Handles order creation, assignment and reporting."""

    def __init__(self, system=None):
        self.system = system or SystemManager()

    def create_order(self, coords):
        """Create an order from ``[restaurant_x, restaurant_y, customer_x, customer_y]``."""
        if len(coords) != 4:
            raise ValueError("an order needs exactly 4 coordinates")
        return Order((coords[0], coords[1]), (coords[2], coords[3]))

    def assign_order(self, order):
        """Assign ``order`` to the best rider.

        Returns:
            Tuple ``(rider, new_rider_added, route)`` where route is the
            planned :class:`Route` from the rider's start to the customer.
        """
        count_before = len(self.system.riders)
        rider = find_best_rider_for_order(order)
        new_rider = len(self.system.riders) > count_before
        start = rider.last_position()
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        route = PathPlanner.build_route(start, [order.res_pos, order.cus_pos], rider.speed)
        logger.info("Order %s assigned to rider %s", order.id, rider.rider_id)
        return rider, new_rider, route

    def place_order(self, coords):
        """Create and assign an order. Returns a result dict."""
        order = self.create_order(coords)
        rider, new_rider, route = self.assign_order(order)
        return {
            "order": order,
            "rider": rider,
            "new_rider": new_rider,
            "route": route,
        }

    def all_orders(self):
        """Return all orders."""
        return list(self.system.orders)

    def active_orders(self):
        """Return orders not yet completed."""
        return [o for o in self.system.orders if not o.is_completed()]

    def statistics(self):
        """Return delivery statistics."""
        done = [o for o in self.system.orders if o.is_completed()]
        avg_time = (sum(o.time_taken for o in done) / len(done)) if done else 0
        effs = [o.efficiency for o in done if o.efficiency]
        return {
            "total": len(self.system.orders),
            "completed": len(done),
            "active": len(self.system.orders) - len(done),
            "average_time": round(avg_time, 2),
            "average_efficiency": round(sum(effs) / len(effs), 2) if effs else 0,
        }

def _idx(value):
    return max(0, min(SIZE - 1, int(round(value / STEP))))


class DisplayManager:
    """Formats system state into text. All methods return strings."""

    LEGEND = ("Legend: R=restaurant C=customer @=rider start digit=rider id "
              "*=route +=intersection -,|=streets (y grows downward)")

    def __init__(self, system):
        self.system = system

    # ------------------------------------------------------------------ maps
    @staticmethod
    def _base_grid():
        grid = []
        for r in range(SIZE):
            row = []
            for c in range(SIZE):
                if r % 2 == 0 and c % 2 == 0:
                    row.append("+")
                elif r % 2 == 0:
                    row.append("-")
                elif c % 2 == 0:
                    row.append("|")
                else:
                    row.append(" ")
            grid.append(row)
        return grid

    @staticmethod
    def _draw_route(grid, points):
        for (x1, y1), (x2, y2) in zip(points, points[1:]):
            c1, r1, c2, r2 = _idx(x1), _idx(y1), _idx(x2), _idx(y2)
            if r1 == r2:
                for c in range(min(c1, c2), max(c1, c2) + 1):
                    grid[r1][c] = "*"
            else:
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    grid[r][c1] = "*"

    @staticmethod
    def _render(grid):
        header = "     " + " ".join(str(c // 2) if c % 2 == 0 else " "
                                    for c in range(SIZE))
        lines = [header + "   (x100)"]
        for r, row in enumerate(grid):
            label = f"{(r // 2) * 100:>3}" if r % 2 == 0 else "   "
            lines.append(f"{label}  " + " ".join(row))
        return "\n".join(lines)

    def render_map(self, route_points=None, highlight_order=None, rider=None):
        """Render the 8x8 grid with riders, active orders and an optional route."""
        grid = self._base_grid()
        if route_points:
            self._draw_route(grid, route_points)
        orders = [highlight_order] if highlight_order else [
            o for o in self.system.orders if o.status != STATUS_COMPLETED]
        for order in orders:
            grid[_idx(order.restaurant_y)][_idx(order.restaurant_x)] = "R"
            grid[_idx(order.customer_y)][_idx(order.customer_x)] = "C"
        if route_points:
            sx, sy = route_points[0]
            grid[_idx(sy)][_idx(sx)] = "@"
        riders = [rider] if rider is not None else self.system.riders
        for r in riders:
            if route_points and r is rider:
                continue
            grid[_idx(r.y)][_idx(r.x)] = str(r.rider_id % 10)
        return self._render(grid) + "\n" + self.LEGEND

    # --------------------------------------------------------------- reports
    def format_order_result(self, result):
        """Format the result of a ``start`` command."""
        order, rider, route = result["order"], result["rider"], result["route"]
        lines = [
            f"Order #{order.id} created at t={order.start_time}",
            f"  Input: {order.raw_input}",
            f"  Restaurant grid position: {order.res_pos}",
            f"  Customer grid position:   {order.cus_pos}",
        ]
        if result["new_rider"]:
            lines.append(f"  New rider added: Yes (Rider #{rider.rider_id}, cost "
                         f"{self.system.recruitment_cost}, funds now {self.system.funds})")
        else:
            lines.append("  New rider added: No")
        lines.append(f"  Assigned rider: #{rider.rider_id} at {rider.position} "
                     f"({rider.state_name}, {len(rider.active_orders())} order(s) queued)")
        pts = " -> ".join(f"({x},{y})" for x, y in route.points)
        lines.append(f"  Route: {pts}")
        lines.append(f"  Speed: {route.speed} units/tick, length: {route.length} units, "
                     f"estimated time: {route.estimated_time} ticks")
        lines.append(self.render_map(route.points, order, rider))
        return "\n".join(lines)

    def format_status(self):
        """Snapshot of funds, riders and their orders."""
        riders = self.system.riders
        lines = [
            f"=== Status at t={self.system.current_time} ===",
            f"Funds: {self.system.funds}",
            f"Riders: {len(riders)}",
        ]
        if not riders:

    def format_orders(self):

"""Command parsing, input validation and command history."""

import re

try:  # optional line editing / history support
    import readline  # noqa: F401  pylint: disable=unused-import
    _HAS_READLINE = True
except ImportError:  # pragma: no cover - platform dependent
    _HAS_READLINE = False


class InputManager:
    """Parses user commands and validates order input."""

    ALIASES = {
        "start": "start", "n": "start", "new": "start",
        "status": "status", "s": "status",
        "orders": "orders", "o": "orders",
        "riders": "riders", "r": "riders",
        "map": "map", "m": "map",
        "tick": "tick", "t": "tick",
        "run": "run",
        "history": "history", "h": "history",
        "save": "save",
        "help": "help", "?": "help",
        "quit": "quit", "exit": "quit", "q": "quit",
    }

    def __init__(self, max_history=500):
        self.command_history = []
        self.max_history = max_history
        self.has_readline = _HAS_READLINE

    def read_command(self, prompt="> "):
        """Read a line from the user and record it in history."""
        line = input(prompt)
        self.remember(line)
        return line

    def remember(self, line):
        """Add a non-empty line to history (bounded)."""
        line = line.strip()
        if line:
            self.command_history.append(line)
            if len(self.command_history) > self.max_history:
                del self.command_history[0]

    def parse(self, line):
        """Parse ``line`` into ``(command, args)``. Unknown commands -> ('unknown', [word])."""
        parts = line.strip().split(None, 1)
        if not parts:
            return "", []
        word = parts[0].lower()
        rest = parts[1] if len(parts) > 1 else ""
        command = self.ALIASES.get(word)
        if command is None:
            return "unknown", [parts[0]]
        args = [a for a in re.split(r"[\s,\[\]()]+", rest) if a]
        return command, args

    @staticmethod
    def parse_coordinates(value):
        """Validate order input.

        Accepts a string like ``"100 100 300 300"`` / ``"[100,100,300,300]"`` or
        a list of tokens. Returns a list of 4 ints in range 0-800.

        Raises:
            ValueError: with a user-friendly message.
        """
        if isinstance(value, str):
            tokens = [t for t in re.split(r"[\s,\[\]()]+", value) if t]
        else:
            tokens = list(value)
        if len(tokens) != 4:
            raise ValueError(f"expected 4 integers, got {len(tokens)}")
        coords = []
        for token in tokens:
            try:
                number = int(str(token).strip())
            except ValueError:
                raise ValueError(f"'{token}' is not an integer") from None
            if not 0 <= number <= 800:
                raise ValueError(f"coordinate {number} out of range 0-800")
            coords.append(number)
        return coords

    @staticmethod
    def parse_positive_int(token, default=1, maximum=100000):
        """Parse a positive integer argument."""
        if token is None:
            return default
        try:
            number = int(token)
        except ValueError:
            raise ValueError(f"'{token}' is not an integer") from None
        if number < 1 or number > maximum:
            raise ValueError(f"value must be between 1 and {maximum}")
        return number

"""Command-line entry point of the Food Delivery Order Dispatch simulator."""

import argparse
import logging
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from config import Configuration  # noqa: E402
from display import DisplayManager  # noqa: E402
from input_manager import InputManager  # noqa: E402
from order import OrderManager  # noqa: E402
from system import SystemManager, init_system  # noqa: E402

logger = logging.getLogger("dispatch")


class Application:
    """Main loop: each iteration represents one tick of simulated time."""

    def __init__(self, config_path=None, autosave=None):
        self.configuration = Configuration(config_path)
        init_system(self.configuration.data)
        self.system = SystemManager()
        self.order_manager = OrderManager(self.system)
        self.display = DisplayManager(self.system)
        self.input = InputManager()
        self.autosave = (self.configuration.get("autosave", True)
                         if autosave is None else autosave)

    def handle(self, line):
        """Handle one command line. Returns True if a tick should follow."""
        command, args = self.input.parse(line)
        if command == "":
            return True
        if command == "quit":
            self.system.running = False
            return False
        if command == "help":
            print(self.display.help_text())
        elif command == "start":
            self._start(args)
        elif command == "status":
            print(self.display.format_status())
        elif command == "orders":
            print(self.display.format_orders())
            stats = self.order_manager.statistics()
            print(f"Total {stats['total']}, completed {stats['completed']}, active "
                  f"{stats['active']}, avg time {stats['average_time']}, avg efficiency "
                  f"{stats['average_efficiency']} units/tick")
        elif command == "riders":
            print(self.display.format_rider_logs())
        elif command == "map":
            print(self.display.render_map())
        elif command == "history":
            for i, item in enumerate(self.input.command_history, 1):
                print(f"{i:>4}  {item}")
        elif command == "save":
            path = self.configuration.get("state_file", "dispatch_state.json")
            ok = self.system.save_state(path)
            print(f"State saved to {path}" if ok else "Saving failed.")
        elif command == "tick":
            try:
                n = self.input.parse_positive_int(args[0] if args else None)
            except ValueError as exc:
                print(f"Error: {exc}")
                return False
            self._advance(n)
            return False
        elif command == "run":
            ticks = 0
            while self.system.has_pending_work() and ticks < 10000:
                self._advance(1)
                ticks += 1
            print(f"Ran {ticks} tick(s); all orders delivered."
                  if not self.system.has_pending_work() else "Stopped after 10000 ticks.")
            return False
        else:
            print(f"Unknown command '{args[0] if args else ''}'. Type 'help'.")
        return True

    def _start(self, args):
        try:
            if args:
                coords = self.input.parse_coordinates(args)
            else:
                text = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
                coords = self.input.parse_coordinates(text)
        except ValueError as exc:
            print(f"Invalid input: {exc}")
            return
        try:
            result = self.order_manager.place_order(coords)
        except Exception as exc:  # pylint: disable=broad-except
            logger.exception("Order placement failed")
            print(f"Error placing order: {exc}")
            self.system.check_consistency()
            return
        print(self.display.format_order_result(result))

    def _advance(self, n):
        events = self.system.tick(n)
        if events:
            print(self.display.format_events(events))

    def run(self):
        """Run the interactive main loop."""
        print("Food Delivery Order Dispatch System")
        print(f"Funds: {self.system.funds}. Type 'help' for commands.")
        while self.system.running:
            try:
                line = self.input.read_command(f"[t={self.system.current_time}] > ")
            except (EOFError, KeyboardInterrupt):
                print()
                line = "quit"
            try:
                advance = self.handle(line)
            except Exception as exc:  # pylint: disable=broad-except
                logger.exception("Unexpected error")
                print(f"Unexpected error: {exc}; recovering.")
                self.system.check_consistency()
                advance = False
            if advance and self.system.running:
                self._advance(1)
        if self.autosave:
            path = self.configuration.get("state_file", "dispatch_state.json")
            if self.system.save_state(path):
                print(f"State saved to {path}")
        print("Goodbye.")


def main(argv=None):
    """Program entry point."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    default_cfg = os.path.join(os.path.dirname(os.path.abspath(__file__)), "config.json")

    app.run()
    return 0


if __name__ == "__main__":

import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG = {
    "initial_funds": 1000,
    "recruitment_cost": 300,
    "rider_speed": 10,
    "initial_rider_x": 435,
    "initial_rider_y": 385,
    "map_size": 800,
    "cell_size": 100,
    "state_file": "dispatch_state.json",
    "autosave": False,
}


def load_config(path=None):
    """Load configuration from a JSON file, merged over the defaults.

    Missing files or invalid content fall back to the defaults.
    """
    cfg = dict(DEFAULT_CONFIG)
    if path and os.path.isfile(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                for key, value in data.items():
                    if key in cfg:
                        cfg[key] = value
        except (OSError, ValueError) as exc:
            logger.warning("Could not load config %s: %s", path, exc)
    return cfg


def save_config(cfg, path):
    """Write the configuration dictionary to ``path`` as JSON."""
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(cfg, fh, indent=2)

"""Grid based path planning for riders."""

import math

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE


def _clamp(value, low, high):
    return max(low, min(high, value))


class PathPlanner:
    """Plans right-angle routes that prefer travelling along streets.

    Streets are the horizontal/vertical lines at multiples of 100.  A route
    from an arbitrary point first moves vertically onto the nearest
    horizontal street, travels horizontally along that street to the target
    column, then moves vertically to the target.  When the rider is directly
    above or below the target (special case) it moves straight vertically.
    """

    @staticmethod
    def to_cell_center(x, y):
        """Convert a coordinate pair to the centre of its grid cell."""
        x = _clamp(int(x), 0, MAP_SIZE - 1)
        y = _clamp(int(y), 0, MAP_SIZE - 1)
        return ((x // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2,
                (y // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2)

    @staticmethod
    def to_cell(x, y):
        """Return the (col, row) grid cell index of a coordinate."""
        col = _clamp(int(x) // CELL_SIZE, 0, GRID_DIM - 1)
        row = _clamp(int(y) // CELL_SIZE, 0, GRID_DIM - 1)
        return col, row

    @staticmethod
    def is_vertical_special(start, end):
        """True if the start is directly above or below the end point."""
        return start[0] == end[0] and start[1] != end[1]

    @staticmethod
    def nearest_street(value):
        """Return the nearest street line (multiple of 100) to ``value``."""
        return _clamp(((int(value) + CELL_SIZE // 2) // CELL_SIZE) * CELL_SIZE,
                      0, MAP_SIZE)

    @staticmethod
    def plan_route(start, end):
        """Return the list of waypoints (excluding ``start``) to ``end``.

        Every consecutive segment is axis aligned.
        """
        start = (int(start[0]), int(start[1]))
        end = (int(end[0]), int(end[1]))
        if start == end:
            return []
        sx, sy = start
        tx, ty = end
        if sx == tx or sy == ty:
            return [end]
        street_y = PathPlanner.nearest_street(sy)
        candidates = [(sx, street_y), (tx, street_y), (tx, ty)]
        route = []
        prev = start
        for pt in candidates:
            if pt != prev:
                route.append(pt)
                prev = pt
        return route

    @staticmethod
    def multi_route(start, destinations):
        """Plan a continuous route through several destinations in order."""
        route = []
        current = (int(start[0]), int(start[1]))
        for dest in destinations:
            route.extend(PathPlanner.plan_route(current, dest))
            current = (int(dest[0]), int(dest[1]))
        return route

    @staticmethod
    def route_length(start, waypoints):
        """Total Manhattan length of a route."""
        total = 0
        prev = start
        for pt in waypoints:
            total += abs(pt[0] - prev[0]) + abs(pt[1] - prev[1])
            prev = pt
        return total

    @staticmethod
    def estimate_ticks(length, speed):
        """Number of ticks needed to travel ``length`` at ``speed``."""
        if speed <= 0:
            return 0
        return int(math.ceil(length / float(speed)))

    @staticmethod
    def sample_points(start, waypoints, step=10):
        """Yield points along a route every ``step`` units (for drawing)."""
        points = [tuple(start)]
        prev = tuple(start)
        for pt in waypoints:
            dx = pt[0] - prev[0]
            dy = pt[1] - prev[1]
            dist = abs(dx) + abs(dy)
            n = max(1, dist // step)
            for i in range(1, n + 1):
                points.append((prev[0] + dx * i // n, prev[1] + dy * i // n))
            prev = pt
        return points

    @staticmethod
    def euclidean(a, b):
        """Euclidean distance between two points."""
        return math.sqrt((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2)

"""Order data model."""

import globalvar
from path_planner import PathPlanner


class Order:
    """A delivery order from a restaurant to a customer.

    Coordinates are automatically converted to grid-cell centres.

    Attributes:
        id: unique order id.
        res_pos / cus_pos: restaurant and customer positions (cell centres).
        status: "pending", "assigned", "picking_up", "delivering", "completed".
        rider_id: id of the assigned rider or None.
        start_time, pickup_time, completion_time, time_taken: tick values.
        distance: Euclidean distance restaurant -> customer.
        travel_distance: distance actually travelled by the rider.
    """

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.res_pos = PathPlanner.to_cell_center(res_pos[0], res_pos[1])
        self.cus_pos = PathPlanner.to_cell_center(cus_pos[0], cus_pos[1])
        self.raw_input = [int(res_pos[0]), int(res_pos[1]),
                          int(cus_pos[0]), int(cus_pos[1])]
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.assign_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.distance = round(PathPlanner.euclidean(self.res_pos,
                                                    self.cus_pos), 2)
        self.travel_distance = 0

    @classmethod
    def reset_counter(cls):
        """Reset the order id counter (used on system init)."""
        cls._next_id = 1

    def assign_to(self, rider_id):
        """Record the assignment of this order to rider ``rider_id``."""
        self.rider_id = rider_id
        self.assign_time = globalvar.get_value("tick", 0)
        if self.status == "pending":
            self.status = "assigned"

    def to_list(self):
        """Return the order as [restaurant_x, restaurant_y, customer_x, customer_y]."""
        return [self.res_pos[0], self.res_pos[1],
                self.cus_pos[0], self.cus_pos[1]]

    def to_dict(self):
        """Serialisable representation."""
        return {
            "id": self.id,
            "restaurant": list(self.res_pos),
            "customer": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "start_time": self.start_time,
            "assign_time": self.assign_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken,
            "distance": self.distance,
            "travel_distance": self.travel_distance,
        }

    def __repr__(self):
        return "Order(id=%d, res=%s, cus=%s, status=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status)

"""Rider model: movement, state and orders."""

import globalvar
from config import DEFAULT_CONFIG
from path_planner import PathPlanner

IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}


class RiderState:
    """Rider state enumeration."""
    IDLE = IDLE
    PICKING_UP = PICKING_UP
    DELIVERING = DELIVERING


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``; the rider
    handles them sequentially in ``move()``.
    """

    def __init__(self, rider_id):
        cfg = globalvar.get_value("config") or DEFAULT_CONFIG
        self.rider_id = rider_id
        self.x = int(cfg.get("initial_rider_x", 435))
        self.y = int(cfg.get("initial_rider_y", 385))
        self.speed = int(cfg.get("rider_speed", 10))
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.distance_travelled = 0
        self.log = []
        self._record("spawn", None)

    # ------------------------------------------------------------------
    def _record(self, event, order_id):
        self.log.append({
            "time": globalvar.get_value("tick", 0),
            "x": self.x,
            "y": self.y,
            "event": event,
            "order_id": order_id,
            "state": self.state,
        })

    @property
    def position(self):
        """Current (x, y) position."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    def is_idle(self):
        """True if the rider has no work."""
        return self.state == IDLE and not self.orders

    # ------------------------------------------------------------------
    def _advance(self, target, order):
        """Move up to ``speed`` units towards ``target`` along streets."""
        remaining = self.speed
        target = (int(target[0]), int(target[1]))
        guard = 0
        while remaining > 0 and (self.x, self.y) != target and guard < 16:
            guard += 1
            pos = (self.x, self.y)
            self.special_case = PathPlanner.is_vertical_special(pos, target)
            route = PathPlanner.plan_route(pos, target)
            if not route:
                break
            wx, wy = route[0]
            dist = abs(wx - self.x) + abs(wy - self.y)
            if dist == 0:
                break
            step = min(remaining, dist)
            if wx != self.x:
                self.x += step if wx > self.x else -step
            else:
                self.y += step if wy > self.y else -step
            remaining -= step
            self.distance_travelled += step
            if order is not None:
                order.travel_distance += step

    def move(self):
        """Advance the rider by one tick.

        Handles state transitions (idle -> picking up -> delivering -> idle)
        and records the corresponding log entries.
        """
        if not self.orders:
            if self.state != IDLE:
                self.state = IDLE
            self.special_case = False
            return
        order = self.orders[0]
        tick = globalvar.get_value("tick", 0)
        if self.state == IDLE:
            self.state = PICKING_UP
            if getattr(order, "rider_id", None) is None:
                order.rider_id = self.rider_id
            order.status = "picking_up"
            if getattr(order, "assign_time", None) is None:
                order.assign_time = tick
            self._record("accepted_order", order.id)

        target = order.res_pos if self.state == PICKING_UP else order.cus_pos
        self._advance(target, order)

        if (self.x, self.y) == (int(target[0]), int(target[1])):
            if self.state == PICKING_UP:
                self.state = DELIVERING
                order.status = "delivering"
                order.pickup_time = tick
                self._record("arrived_at_merchant", order.id)
            elif self.state == DELIVERING:
                order.status = "completed"
                order.completion_time = tick
                start = order.start_time if order.start_time is not None else 0
                order.time_taken = tick - start
                self.orders.pop(0)
                self.state = IDLE
                self.special_case = False
                self._record("delivered_to_customer", order.id)

    def to_dict(self):
        """Serialisable representation."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [o.id for o in self.orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%d,%d), state=%d, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state, len(self.orders))

"""Order assignment logic (OrderManager)."""

import math

import globalvar


def _distance(rider, order):
    rx, ry = order.res_pos
    return math.sqrt((rider.x - rx) ** 2 + (rider.y - ry) ** 2)


def find_best_rider_for_order(order):
    """Return the rider closest (Euclidean) to the order's restaurant.

    Idle riders are preferred; if none are idle, the closest rider overall is
    chosen (ties broken by queue length, then rider id).  Returns None when
    there are no riders.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    idle = [r for r in riders if r.state == 0 and not r.orders]
    pool = idle if idle else riders
    return min(pool, key=lambda r: (_distance(r, order), len(r.orders),
                                    r.rider_id))


class OrderManager:
    """Handles creation, assignment and completion queries of orders."""

    def __init__(self):
        if globalvar.get_value("orders") is None:
            globalvar.set_value("orders", [])

    @property
    def orders(self):
        """All orders."""
        return globalvar.get_value("orders", [])

    def create_order(self, rx, ry, cx, cy):
        """Create and register a new order."""
        from order_model import Order
        order = Order((rx, ry), (cx, cy))
        self.orders.append(order)
        return order

    def assign(self, order):
        """Assign ``order`` to the best rider; return the rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        if order not in rider.orders:
            rider.orders.append(order)
        order.assign_to(rider.rider_id)
        return rider

    def pending_orders(self):
        """Orders not yet assigned to a rider."""
        return [o for o in self.orders if o.status == "pending"]

    def completed_orders(self):
        """Orders that have been delivered."""
        return [o for o in self.orders if o.status == "completed"]

"""System manager: funds, riders, recruitment and simulation ticks."""

import json
import logging

import globalvar
from config import load_config
from order import OrderManager
from order_model import Order
from rider import Rider, IDLE

logger = logging.getLogger(__name__)


class SystemManager:
    """Core manager of funds, riders and the simulation clock."""

    def __init__(self, config=None):
        self.config = config or globalvar.get_value("config") or load_config()
        self.recruitment_cost = int(self.config.get("recruitment_cost", 300))
        self.order_manager = OrderManager()

    # ---------------------------------------------------------- accessors
    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", value)

    @property
    def riders(self):
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def orders(self):
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    @property
    def tick(self):
        return globalvar.get_value("tick", 0)

    @property
    def running(self):
        return bool(globalvar.get_value("running", False))

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    # ---------------------------------------------------------- riders
    def recruit_rider(self):
        """Create a new rider, paying the recruitment cost."""
        if self.funds < self.recruitment_cost:
            return None
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        self.funds = self.funds - self.recruitment_cost
        logger.info("Recruited rider %d", counter)
        return rider

    def idle_riders(self):
        """Riders without work."""
        return [r for r in self.riders if r.state == IDLE and not r.orders]

    def check_recruitment(self):
        """Recruit a rider if funds allow and pending orders exceed idle riders."""
        pending = len([o for o in self.orders if o.status == "pending"])
        if self.funds >= self.recruitment_cost and pending > len(self.idle_riders()):
            return self.recruit_rider()
        return None

    # ---------------------------------------------------------- orders
    def create_order(self, rx, ry, cx, cy):
        """Create an order, possibly recruit, and assign it.

        Returns (order, assigned_rider_or_None, new_rider_or_None).
        """
        for v in (rx, ry, cx, cy):
            if not isinstance(v, int) or v < 0 or v > 800:
                raise ValueError("Coordinates must be integers in 0-800")
        order = Order((rx, ry), (cx, cy))
        self.orders.append(order)
        new_rider = self.check_recruitment()
        rider = self.order_manager.assign(order)
        return order, rider, new_rider

    def assign_pending(self):
        """Try to assign orders still pending."""
        for order in [o for o in self.orders if o.status == "pending"]:
            self.check_recruitment()
            self.order_manager.assign(order)

    # ---------------------------------------------------------- simulation
    def advance(self, n=1):
        """Advance the simulation by ``n`` ticks."""
        for _ in range(max(0, int(n))):
            globalvar.set_value("tick", self.tick + 1)
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception as exc:  # error recovery
                    logger.error("Rider %s failed to move: %s",
                                 rider.rider_id, exc)
            self.assign_pending()
            self.check_consistency()

    def check_consistency(self):
        """Repair inconsistent state; returns list of issues fixed."""
        issues = []
        ids = {r.rider_id for r in self.riders}
        for rider in self.riders:
            if not rider.orders and rider.state != IDLE:
                rider.state = IDLE
                issues.append("rider %s reset to idle" % rider.rider_id)
            rider.orders[:] = [o for o in rider.orders
                               if o.status != "completed"]
        for order in self.orders:
            if order.status in ("assigned", "picking_up", "delivering") and \
                    order.rider_id not in ids:
                order.status = "pending"
                order.rider_id = None
                issues.append("order %d returned to pending" % order.id)
        for issue in issues:
            logger.warning(issue)
        return issues

    def find_rider(self, rider_id):
        """Return the rider with ``rider_id`` or None."""
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    # ---------------------------------------------------------- persistence
    def to_dict(self):
        """Full serialisable snapshot of the system."""
        return {
            "config": self.config,
            "funds": self.funds,
            "tick": self.tick,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.orders],
        }

    def save_state(self, path=None):
        """Save the snapshot to a JSON file; returns the path."""
        path = path or self.config.get("state_file", "dispatch_state.json")
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, indent=2)
        return path


def init_system(config_path=None):
    """Initialise global state and return a SystemManager."""
    globalvar._init()
    cfg = load_config(config_path)
    globalvar.set_value("config", cfg)
    globalvar.set_value("funds", int(cfg.get("initial_funds", 1000)))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    Order.reset_counter()
    return SystemManager(cfg)

"""Status display and ASCII map rendering."""

from path_planner import PathPlanner, GRID_DIM
from rider import STATE_NAMES


class DisplayManager:
    """Builds textual views of the system."""

    def render_map(self, system, route=None, orders=None):
        """Render the 8x8 ASCII map.

        'R' rider, 'S' restaurant, 'C' customer, '#' route/street, '.' empty.
        ``route`` is (start, waypoints); ``orders`` restricts S/C markers.
        """
        grid = [["." for _ in range(GRID_DIM)] for _ in range(GRID_DIM)]
        if route:
            start, waypoints = route
            for pt in PathPlanner.sample_points(start, waypoints):
                c, r = PathPlanner.to_cell(*pt)
                grid[r][c] = "#"
        if orders is None:
            orders = [o for o in system.orders if o.status != "completed"]
        for o in orders:
            c, r = PathPlanner.to_cell(*o.cus_pos)
            grid[r][c] = "C"
        for o in orders:
            c, r = PathPlanner.to_cell(*o.res_pos)
            grid[r][c] = "S"
        for rider in system.riders:
            c, r = PathPlanner.to_cell(rider.x, rider.y)
            grid[r][c] = "R"
        lines = ["    " + " ".join(str(i) for i in range(GRID_DIM)) + "   (x/100)"]
        for r in range(GRID_DIM):
            lines.append("%2d  " % r + " ".join(grid[r]))
        lines.append("(y/100)  R=Rider S=Restaurant C=Customer #=Route .=Empty")
        return "\n".join(lines)

    def rider_route(self, rider, order):
        """Return (start, waypoints, eta_ticks) for ``order`` on ``rider``."""
        start = (rider.x, rider.y)
        eta_before = 0
        prev = start
        for o in rider.orders:
            if o is order:
                break
            wps = []
            if o.status in ("assigned", "pending", "picking_up"):
                wps = PathPlanner.multi_route(prev, [o.res_pos, o.cus_pos])
            else:
                wps = PathPlanner.plan_route(prev, o.cus_pos)
            eta_before += PathPlanner.route_length(prev, wps)
            prev = o.cus_pos
        waypoints = PathPlanner.multi_route(prev, [order.res_pos, order.cus_pos])
        length = PathPlanner.route_length(prev, waypoints)
        eta = PathPlanner.estimate_ticks(eta_before + length, rider.speed)
        return prev, waypoints, eta, length

    def show_start_result(self, system, order, rider, new_rider):
        """Text output for the start command."""
        lines = [
            "Order #%d created at tick %d" % (order.id, order.start_time),
            "Restaurant grid coordinates: (%d, %d)" % order.res_pos,
            "Customer grid coordinates:   (%d, %d)" % order.cus_pos,
            "New rider added: %s" % (
                "yes (rider %d)" % new_rider.rider_id if new_rider else "no"),
        ]
        if rider is None:
            lines.append("No rider available; order is pending "
                         "(funds: %d)." % system.funds)
            lines.append(self.render_map(system, orders=[order]))
            return "\n".join(lines)
        start, waypoints, eta, length = self.rider_route(rider, order)

"""Command parsing and input validation."""

import logging

logger = logging.getLogger(__name__)

HELP_TEXT = """Commands:
  start [rx ry cx cy]  (s)   create an order (coordinates 0-800)
  status               (st)  system snapshot
  orders               (o)   all orders
  riders               (r)   rider status-change history
  map                  (m)   show map
  tick [n]             (t)   advance simulation n ticks (default 1)
  save [path]                save state to JSON
  history              (hi)  command history
  help                 (h)   this help
  quit / exit          (q)   exit"""

SHORTCUTS = {
    "s": "start", "st": "status", "o": "orders", "r": "riders",
    "m": "map", "t": "tick", "hi": "history", "h": "help", "?": "help",
    "q": "quit", "exit": "quit",
}


class InputManager:
    """Parses commands and dispatches them to the system."""

    def __init__(self, system, display, input_func=input):
        self.system = system
        self.display = display
        self.input_func = input_func
        self.command_history = []

    @staticmethod
    def parse_coordinates(text):
        """Parse four integer coordinates in 0-800 from text.

        Accepts formats like ``1 2 3 4``, ``1,2,3,4`` or ``[1, 2, 3, 4]``.
        Raises ValueError with a friendly message on failure.
        """
        cleaned = text.replace("[", " ").replace("]", " ").replace(",", " ")
        parts = cleaned.split()
        if len(parts) != 4:
            raise ValueError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y")
        values = []
        for p in parts:
            try:
                v = int(p)
            except ValueError:
                raise ValueError("'%s' is not an integer" % p)
            if v < 0 or v > 800:
                raise ValueError("Coordinate %d out of range 0-800" % v)
            values.append(v)
        return values

    def handle(self, line):
        """Process one command line and return the output text."""
        line = (line or "").strip()
        if not line:
            return ""
        self.command_history.append(line)
        if len(self.command_history) > 1000:
            del self.command_history[:-1000]
        tokens = line.split(None, 1)
        cmd = tokens[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        arg = tokens[1] if len(tokens) > 1 else ""
        try:
            if cmd == "start":
                if not arg:
                    arg = self.input_func(
                        "Enter restaurant_x restaurant_y customer_x customer_y: ")
                coords = self.parse_coordinates(arg)
                order, rider, new_rider = self.system.create_order(*coords)
                return self.display.show_start_result(self.system, order,
                                                      rider, new_rider)
            if cmd == "status":
                return self.display.show_status(self.system)
            if cmd == "orders":
                return self.display.show_orders(self.system)
            if cmd == "riders":
                return self.display.show_riders(self.system)
            if cmd == "map":
                return self.display.render_map(self.system)
            if cmd == "tick":
                n = int(arg) if arg else 1
                if n < 0 or n > 100000:
                    raise ValueError("tick count must be 0-100000")
                self.system.advance(n)
                return "Advanced %d tick(s). Now tick %d." % (n, self.system.tick)
            if cmd == "save":
                path = self.system.save_state(arg or None)
                return "State saved to %s" % path
            if cmd == "history":
                return "\n".join("%3d  %s" % (i + 1, c)
                                 for i, c in enumerate(self.command_history))
            if cmd == "help":
                return HELP_TEXT
            if cmd == "quit":
                self.system.stop()
                return "Goodbye."
            return "Unknown command '%s'. Type 'help'." % tokens[0]
        except ValueError as exc:
            return "Error: %s" % exc
        except (EOFError, KeyboardInterrupt):
            return "Input cancelled."
        except Exception as exc:  # graceful recovery
            logger.exception("Command failed")
            self.system.check_consistency()
            return "Unexpected error: %s (state recovered)" % exc

"""Entry point of the Food Delivery Order Dispatch System simulator."""

import logging
import sys

import globalvar
from display import DisplayManager
from input_manager import InputManager
from system import init_system

try:  # command history / line editing where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None


def main(argv=None):
    """Run the interactive main loop. Each iteration is one simulation tick."""
    argv = sys.argv[1:] if argv is None else argv
    logging.basicConfig(level=logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")
    config_path = argv[0] if argv else "config.json"
    system = init_system(config_path)
    display = DisplayManager()
    im = InputManager(system, display)
    print("Food Delivery Dispatch Simulator. Type 'help' for commands.")
    while globalvar.get_value("running", False):
        try:
            line = input("[t=%d] > " % system.tick)
        except (EOFError, KeyboardInterrupt):
            print()
            system.stop()
            break
        out = im.handle(line)
        if out:
            print(out)
        if system.running:
            system.advance(1)
    if system.config.get("autosave"):
        try:
            print("State saved to %s" % system.save_state())
        except OSError as exc:
            print("Could not save state: %s" % exc)
    return 0


if __name__ == "__main__":
    sys.exit(main())

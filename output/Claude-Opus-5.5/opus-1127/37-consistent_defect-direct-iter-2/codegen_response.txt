"""Core data models of the lottery system."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A lottery participant."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None

    def __str__(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "Not specified"
        return (f"{self.name} | Quantity: {self.quantity} | "
                f"Avoidance groups: {avoid} | Lottery person: {drawer}")

DEFAULT_LIST = [

    "Li Si,Technology Department",
    "Wang Wu,Technology Department",
    "Zhao Liu,Technology Department",
    "Sun Qi,Product Department",
    "Zhou Ba,Product Department",
    "Wu Jiu,Product Department",
    "Zheng Shi,Product Department",
    "Feng Yi,Marketing Department",
    "Chen Er,Marketing Department",
    "Chu San,Marketing Department",
    "Wei Si,Marketing Department",
    "Jiang Wu,HR Department",
    "Shen Liu,HR Department",
    "Han Qi,HR Department",
    "Yang Ba,Cheng Ru Group",
    "Zhu Jiu,Cheng Ru Group",
    "Qin Shi,Cheng Ru Group",
    "You Yi,Cheng Ru Group",
    "Xu Er,Finance Department",
    "He San,Finance Department",
]

_SEPARATOR_PATTERN = re.compile(r"[,;，；]")


class ListManager:
    """Loads and validates participant lists."""

    def __init__(self) -> None:
        self.users: List[User] = []

    # ------------------------------------------------------------------ parsing
    def _parse_lines(self, lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.replace("\ufeff", "").strip()
            if not line:
                continue
            parts = _SEPARATOR_PATTERN.split(line)
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    # ------------------------------------------------------------------ loading
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST)
        self.users = users
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        users, errors = self._load_from_path(path)
        if users:
            self.users = users
        return users, errors

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        if path is None or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        content = None
        for encoding in ("utf-8-sig", "gbk"):
            try:
                with open(path, "r", encoding=encoding) as f:
                    content = f.read()
                break
            except UnicodeDecodeError:
                continue
            except OSError as exc:
                return [], [f"File Error: cannot read file {path}: {exc}"]
        if content is None:
            return [], [f"File Error: unsupported file encoding: {path}"]
        return self._parse_lines(content.splitlines())

    # --------------------------------------------------------------- validation
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize configuration management."""

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prize library and the lottery round order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        name = str(name).strip()
        try:
            qty = int(str(quantity).strip())
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError(f"Prize quantity must be a positive integer: {quantity}")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize already exists: {name}")
        groups: List[str] = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer: Optional[str] = None
        if associated_drawer is not None and str(associated_drawer).strip():
            drawer = str(associated_drawer).strip()
        prize = Prize(name, qty, groups, drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine."""

import random
from typing import List, Optional

from lottery_system.models import Prize, User

LI_BIN = "Li Bin"
CHEN_QIXIAN = "Chen Qixian"
JIN_DUO = "Jin Duo"
MA_JIAQI = "Ma Jiaqi"
CHENG_RU_GROUP = "Cheng Ru Group"


def _norm(text) -> str:
    return "".join(str(text or "").lower().split())


def is_cheng_ru(user: User) -> bool:
    return _norm(user.department) in ("chengrugroup", "chengru")


class LotteryManager:
    """Applies lottery rules and draws winners."""

    def __init__(self, users, rng=None) -> None:
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.winners: List[tuple] = []
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.SystemRandom()

    # ----------------------------------------------------------------- helpers
    def is_winner(self, user: User) -> bool:
        return user.name in self.winner_names

    @staticmethod
    def _is_drawer(prize: Prize, name: str) -> bool:
        return _norm(prize.associated_drawer) == _norm(name)

    def is_fallback_active(self, prize: Prize) -> bool:
        return (self._is_drawer(prize, MA_JIAQI)
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    # -------------------------------------------------------------------- pool
    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = [u for u in self.users if not self.is_winner(u)]
        if self.is_fallback_active(prize):
            cheng_ru = [u for u in candidates if is_cheng_ru(u)]
            if cheng_ru:
                return cheng_ru
        if self._is_drawer(prize, LI_BIN):
            return candidates
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        if not avoid:
            return candidates
        primary = [u for u in candidates if _norm(u.department) not in avoid]
        if primary:
            return primary
        # Avoidance group members remain eligible, but are drawn after others.
        return [u for u in candidates if _norm(u.department) in avoid]

    # ------------------------------------------------------------------- draw
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winner_names.add(user.name)
        self.winners.append((user, prize))
        return user

    def record_round(self, prize: Prize, round_winners) -> None:
        if self._is_drawer(prize, CHEN_QIXIAN) or self._is_drawer(prize, JIN_DUO):
            self.last_two_draw_results.append(any(is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            self.last_two_draw_results = []

    def describe_rules(self, prize: Prize) -> str:
        lines = [f"Lottery person: {prize.associated_drawer or 'Not specified'}"]
        if self.is_fallback_active(prize):
            lines.append("Fallback (Cheng Ru group protection) ACTIVE: prize pool is "
                         "limited to Cheng Ru group members")
        elif self._is_drawer(prize, LI_BIN):
            lines.append("Special rule (Li Bin): only previous winners are excluded; "
                         "avoidance groups are not applied")
        elif prize.avoid_groups:
            lines.append("Previous winners excluded; members of avoidance groups ("
                         + ", ".join(prize.avoid_groups)
                         + ") are drawn only after all other members")
        else:
            lines.append("Previous winners excluded; no avoidance groups")
        return "\n".join(lines)

"""Result management: recording, summarising and exporting winners."""

import csv
import os
import re
from dataclasses import dataclass, field
from datetime import datetime
from typing import Dict, List, Optional

from lottery_system.models import Prize, User

_LEVEL_WORDS = [
    (0, ("grand", "special", "super")),
    (1, ("first", "1st")),
    (2, ("second", "2nd")),
    (3, ("third", "3rd")),
    (4, ("fourth", "4th")),
    (5, ("fifth", "5th")),
    (6, ("sixth", "6th")),
]


def prize_level(name: str) -> int:
    lowered = str(name).lower()
    for level, words in _LEVEL_WORDS:
        for w in words:
            if re.search(r"\b" + re.escape(w) + r"\b", lowered):
                return level
    return 100


@dataclass
class Winner:
    user: User
    prize_name: str
    win_time: datetime = field(default_factory=datetime.now)
    sequence: int = 0


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> Optional[Winner]:
        if any(w.user.name == user.name for w in self.winners):
            return None
        name = prize.name if isinstance(prize, Prize) or hasattr(prize, "name") else str(prize)
        winner = Winner(user, name, datetime.now(), len(self.winners) + 1)
        self.winners.append(winner)
        return winner

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        names: List[str] = []
        for p in prizes or []:
            n = p.name if hasattr(p, "name") else str(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        ordered = sorted(enumerate(names), key=lambda t: (prize_level(t[1]), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, n in ordered:
            result[n] = sorted((w for w in self.winners if w.prize_name == n),
                               key=lambda w: (w.win_time, w.sequence))
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winning results to export"
        filename = f"awards_result_{datetime.now():%Y%m%d}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in self.get_winners_by_prize(prizes).items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, prize_name,
                                         w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""

import re
import sys

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class _Exit(Exception):
    pass


def ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        raise _Exit()


class App:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.lottery = None
        self.results = ResultManager()
        self.drawn_prizes = []

    # ---------------------------------------------------------------- lists
    def _accept_users(self, users, errors) -> None:
        if errors:
            print("Detected invalid data:")
            for e in errors:
                print("  " + e)
            if not users:
                print("No valid data loaded.")
                return
            choice = ask("Remove invalid data and continue (r) or terminate upload (t)? [r/t]: ").lower()
            if choice != "r":
                print("Upload terminated.")
                return
        valid, dups = self.list_manager.validate_users(users)
        if dups:
            print("Detected duplicate names:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
            choice = ask("Remove duplicates (keep first occurrence) (r) or terminate upload (t)? [r/t]: ").lower()
            if choice != "r":
                print("Upload terminated.")
                return
        self.users = valid
        print(f"List loaded: {len(valid)} valid participants.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        self._accept_users(users, errors)

    def load_custom(self) -> None:
        path = ask("Enter txt list path (each line 'Name,Department Group'): ")
        users, errors = self.list_manager.load_custom_list(path)
        self._accept_users(users, errors)

    def show_groups(self) -> None:
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Valid groups:")
        for i, (dep, cnt) in enumerate(info.items(), 1):
            print(f"  {i}. {dep}: {cnt} people")

    # --------------------------------------------------------------- prizes
    def load_default_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Default prizes loaded.")

    def add_prize(self) -> None:
        name = ask("Prize name: ")
        qty = ask("Quantity: ")
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid = []
        if groups:
            print("Groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = ask("Avoidance groups (serial numbers or names, comma separated, empty for none): ")
        for token in re.split(r"[,;，；\s]+", raw) if raw else []:
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                g = groups[int(token) - 1]
            else:
                g = token
            if g not in avoid:
                avoid.append(g)
        drawer = ask("Associated lottery person (empty for none): ")
        try:
            prize = self.prize_manager.add_prize(name, qty, avoid, drawer or None)
            print(f"Prize added: {prize}")
        except ValueError as exc:
            print(f"Error: {exc}")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")

    def reorder(self) -> None:
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = ask("Enter new order as serial numbers (e.g. 3 1 2): ")
        try:
            order = [int(t) for t in re.split(r"[,;，；\s]+", raw) if t]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.show_prizes()
        else:
            print("Invalid order: must be a permutation of all serial numbers.")

    # -------------------------------------------------------------- lottery
    def run_lottery(self) -> None:
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        self.lottery = LotteryManager(self.users)
        self.results = ResultManager()
        self.drawn_prizes = prizes
        print("Commands: Enter = draw next, 'pause' = pause, 'continue' = resume, "
              "'terminate' = end current round")
        for rnd, prize in enumerate(prizes, 1):
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            print(self.lottery.describe_rules(prize))
            print("=" * 60)
            round_winners = []
            terminated = False
            for i in range(1, prize.quantity + 1):
                paused = False
                while True:
                    cmd = ask("(paused) > " if paused else f"[Draw #{i}] > ").lower()
                    if cmd == "terminate":
                        terminated = True
                        break
                    if cmd == "pause":
                        paused = True
                        print("Lottery paused. Enter 'continue' to resume.")
                        continue
                    if cmd == "continue":
                        if paused:
                            print("Lottery resumed.")
                        paused = False
                        continue
                    if paused:
                        print("Lottery is paused. Enter 'continue' or 'terminate'.")
                        continue
                    break
                if terminated:
                    print(f"Round '{prize.name}' terminated.")
                    break
                print(f"Currently drawing the {i}th winner...")
                user = self.lottery.draw_one(prize)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                self.results.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            self.lottery.record_round(prize, round_winners)
            print(f"\nRound {rnd} results ({prize.name}):")
            if not round_winners:
                print("  No winners.")
            for n, u in enumerate(round_winners, 1):
                print(f"  {n}. {u.name} ({u.department})")
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self) -> None:
        if not self.results.winners:
            print("No winners yet.")
            return
        print("\nComplete winner list:")
        for prize_name, winners in self.results.get_winners_by_prize(self.drawn_prizes).items():
            print(f"[{prize_name}]")
            if not winners:
                print("  (none)")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {prize_name} - "
                      f"{w.win_time:%Y-%m-%d %H:%M:%S}")

    def export(self) -> None:
        ok, msg = self.results.export_to_csv(self.drawn_prizes)
        print(msg)

    # ----------------------------------------------------------------- menu
    def run(self) -> None:
        actions = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.show_groups),
            "4": ("Load default prizes", self.load_default_prizes),
            "5": ("Add custom prize", self.add_prize),
            "6": ("View prizes", self.show_prizes),
            "7": ("Reorder prizes", self.reorder),
            "8": ("Start lottery", self.run_lottery),
            "9": ("Show winner summary", self.show_summary),
            "10": ("Export results", self.export),
        }
        print("=== Company Team Building Smart Lottery System ===")
        try:
            while True:
                print()
                for k, (label, _) in actions.items():
                    print(f"{k}. {label}")
                print("0. Exit")
                choice = ask("Select: ")
                if choice == "0":
                    break
                action = actions.get(choice)
                if action:
                    action[1]()
                else:
                    print("Invalid choice.")
        except (_Exit, KeyboardInterrupt):
            print()
        print("Goodbye.")


def main() -> int:
    App().run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

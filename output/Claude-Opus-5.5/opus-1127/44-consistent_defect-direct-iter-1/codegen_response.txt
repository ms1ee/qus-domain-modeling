"""Data loading utilities for the Beijing Subway fare calculation system.

The subway network is stored in ``subway_data.json`` located next to this
module.  The format is::

    {"stations": [{"name": str, "lines": [str]}],
     "connections": [{"source": str, "dest": str,
                      "distance": int (meters), "line": str}]}
"""

import json
import os

from models import Edge, Graph

# Absolute path of the bundled data file (valid from any working directory).
DATA_FILE_PATH = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "subway_data.json"
)


def load_subway_data(path=DATA_FILE_PATH) -> dict:
    """Load and return the JSON content of the subway data file."""
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    if not isinstance(data, dict):
        raise ValueError("Subway data file must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data


def build_station_mapping(data):
    """Build ``station_to_number`` and ``number_to_station`` mappings.

    Stations are numbered in the order they appear in the ``stations`` list;
    stations referenced only in connections are appended afterwards.
    """
    station_to_number = {}
    number_to_station = {}

    def _add(name):
        if name not in station_to_number:
            idx = len(station_to_number)
            station_to_number[name] = idx
            number_to_station[idx] = name

    for st in data.get("stations", []):
        name = st.get("name")
        if name:
            _add(name)
    for conn in data.get("connections", []):
        for key in ("source", "dest"):
            name = conn.get(key)
            if name:
                _add(name)
    return station_to_number, number_to_station


def build_graph(data):
    """Build a directed weighted ``Graph`` from the loaded data.

    Returns ``(graph, station_to_number, number_to_station)``.
    """
    station_to_number, number_to_station = build_station_mapping(data)
    graph = Graph()
    for idx in number_to_station:
        graph.add_vertex(idx)
    for st in data.get("stations", []):
        name = st.get("name")
        if name:
            graph.set_station_lines(name, st.get("lines", []) or [])
    for conn in data.get("connections", []):
        src = conn.get("source")
        dst = conn.get("dest")
        if src is None or dst is None:
            continue
        try:
            distance = int(conn.get("distance", 0))
        except (TypeError, ValueError):
            continue
        graph.add_edge(
            Edge(station_to_number[src], station_to_number[dst], distance,
                 conn.get("line"))
        )
    return graph, station_to_number, number_to_station


def build_adjacency_dict(data):
    """Return a ``graph`` dictionary: station name -> list of (neighbor, distance, line)."""
    graph = {}
    for st in data.get("stations", []):
        if st.get("name"):
            graph.setdefault(st["name"], [])
    for conn in data.get("connections", []):
        src, dst = conn.get("source"), conn.get("dest")
        if src is None or dst is None:
            continue
        graph.setdefault(src, []).append((dst, conn.get("distance", 0), conn.get("line")))
        graph.setdefault(dst, [])
    return graph

"""Graph data structures for the subway network."""


class Edge:
    """A directed weighted edge between two station numbers."""

    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return "Edge(%r -> %r, %r m, %r)" % (self.source, self.dest, self.weight, self.line)

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.weight, self.line) == (
            other.source, other.dest, other.weight, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.weight, self.line))


class Graph:
    """Weighted directed graph. Station numbers are assigned by the caller."""

    def __init__(self):
        self.adj = {}             # station number -> list[Edge]
        self.station_lines = {}   # station name -> list[str]
        self._edge_count = 0

    def add_vertex(self, v):
        if v not in self.adj:
            self.adj[v] = []

    def add_edge(self, edge):
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj[edge.source].append(edge)
        self._edge_count += 1

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines) if lines else []

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    def out_edges(self, v):
        return self.adj.get(v, [])

    def vertices(self):
        return list(self.adj.keys())

    def edges(self):
        result = []
        for lst in self.adj.values():
            result.extend(lst)
        return result

    def V(self):
        return len(self.adj)

    def E(self):
        return self._edge_count

    def update_edge_distance(self, source, dest, distance, line=None):
        """Update the distance of matching edges. Returns number updated."""
        count = 0
        for e in self.adj.get(source, []):
            if e.dest == dest and (line is None or e.line == line):
                e.distance = distance
                e.weight = distance
                count += 1
        return count

"""Dijkstra-based path computation over the subway graph."""

import heapq
import itertools

INF = float("inf")

MODE_SHORTEST_DISTANCE = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"
VALID_MODES = (MODE_SHORTEST_DISTANCE, MODE_LEAST_TRANSFER, MODE_COMPREHENSIVE)

# Equivalent distance penalty (meters) for one transfer in comprehensive mode.
TRANSFER_PENALTY = 3000
# Weight multiplier applied to edges on non-preferred lines when preferences exist.
NON_PREFERRED_FACTOR = 1.3


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode=MODE_SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode: %r" % (mode,))
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def del_min(self):
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


class ShortestPath:
    """Dijkstra shortest paths from a single source station number.

    The search runs over states (station, current line, [transfers]) so that
    transfers at interchange stations are tracked correctly.
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {v: INF for v in graph.vertices()}
        self.dist_to[source] = 0
        self._transfers = {source: 0}
        self._best_state = {}
        self._parent = {}
        self._run()

    def _edge_weight(self, edge):
        w = edge.weight
        prefs = self.config.preferred_lines
        if prefs and edge.line not in prefs:
            w = w * NON_PREFERRED_FACTOR
        return w

    def _cost(self, adj_dist, transfers):
        mode = self.config.mode
        if mode == MODE_LEAST_TRANSFER:
            return (transfers, adj_dist)
        if mode == MODE_COMPREHENSIVE:
            return (adj_dist + transfers * TRANSFER_PENALTY, transfers)
        return (adj_dist, transfers)

    def _run(self):
        max_tr = self.config.max_transfers
        limited = max_tr is not None
        start = (self.source, None, 0)
        info = {start: (0, 0, 0)}  # state -> (adjusted dist, actual dist, transfers)
        visited = set()
        pq = MinPQ()
        pq.insert(self._cost(0, 0), start)
        while not pq.is_empty():
            _, state = pq.del_min()
            if state in visited:
                continue
            visited.add(state)
            node, line, _ = state
            adj, actual, tr = info[state]
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = actual
                self._transfers[node] = tr
            for e in self.graph.out_edges(node):
                ntr = tr + (1 if (line is not None and e.line != line) else 0)
                if limited and ntr > max_tr:
                    continue
                nstate = (e.dest, e.line, ntr if limited else 0)
                if nstate in visited:
                    continue
                nadj = adj + self._edge_weight(e)
                nact = actual + e.weight
                ncost = self._cost(nadj, ntr)
                old = info.get(nstate)
                if old is None or ncost < self._cost(old[0], old[2]):
                    info[nstate] = (nadj, nact, ntr)
                    self._parent[nstate] = (state, e)
                    pq.insert(ncost, nstate)

    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, INF)

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target."""
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while state in self._parent:
            prev, e = self._parent[state]
            edges.append(e)
            state = prev
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._best_state:
            return 0
        return self._transfers.get(target, 0)

"""Beijing Subway fare calculation."""

import math

# (upper bound in meters, inclusive flag, fare)
FARE_TABLE = [
    (6000, True, 3),
    (12000, True, 4),
    (22000, False, 5),
    (32000, True, 6),
]
BASE_EXTENDED_FARE = 6
EXTENDED_START = 32000
EXTENDED_STEP = 20000

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5


def _validate(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be finite")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance) -> int:
    """Return the fare (yuan) for a distance given in meters."""
    _validate(distance)
    d = math.ceil(distance)
    for upper, inclusive, fare in FARE_TABLE:
        if (inclusive and d <= upper) or (not inclusive and d < upper):
            return fare
    extra = math.ceil((d - EXTENDED_START) / EXTENDED_STEP)
    return BASE_EXTENDED_FARE + extra


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated travel time in minutes (distance in meters)."""
    _validate(distance)
    if transfer_count is None or transfer_count < 0:
        transfer_count = 0
    return float(distance / 1000.0 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER)

"""Data integrity checks and update mechanisms."""

import json

from data_loader import DATA_FILE_PATH

MAIN_LINES = [
    "Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
    "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
    "Fangshan Line", "Airport Line",
]

MAX_REASONABLE_DISTANCE = 30000


class SubwayDataManager:
    def __init__(self, data):
        self.data = data
        self.data.setdefault("stations", [])
        self.data.setdefault("connections", [])

    def _station(self, name):
        for st in self.data["stations"]:
            if st.get("name") == name:
                return st
        return None

    def add_line(self, line_name, station_names, distances):
        """Add a line: station_names in order, distances between consecutive stations."""
        if len(distances) != len(station_names) - 1:
            raise ValueError("distances must have len(station_names) - 1 entries")
        for name in station_names:
            st = self._station(name)
            if st is None:
                self.data["stations"].append({"name": name, "lines": [line_name]})
            elif line_name not in st.setdefault("lines", []):
                st["lines"].append(line_name)
        for i, d in enumerate(distances):
            if d <= 0:
                raise ValueError("distance must be positive")
            a, b = station_names[i], station_names[i + 1]
            self.data["connections"].append({"source": a, "dest": b, "distance": int(d), "line": line_name})
            self.data["connections"].append({"source": b, "dest": a, "distance": int(d), "line": line_name})

    def modify_station(self, old_name, new_name=None, lines=None):
        st = self._station(old_name)
        if st is None:
            raise KeyError("Station '%s' does not exist" % old_name)
        if lines is not None:
            st["lines"] = list(lines)
        if new_name and new_name != old_name:
            st["name"] = new_name
            for c in self.data["connections"]:
                if c.get("source") == old_name:
                    c["source"] = new_name
                if c.get("dest") == old_name:
                    c["dest"] = new_name

    def update_distance(self, source, dest, distance, line=None):
        if distance <= 0:
            raise ValueError("distance must be positive")
        count = 0
        for c in self.data["connections"]:
            if c.get("source") == source and c.get("dest") == dest and (line is None or c.get("line") == line):
                c["distance"] = int(distance)
                count += 1
        return count

    def save(self, path=DATA_FILE_PATH):
        with open(path, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

    def check_integrity(self):
        """Return (ok, errors, warnings)."""
        errors, warnings = [], []
        names = set()
        all_lines = set()
        for st in self.data["stations"]:
            name = st.get("name")
            if not name:
                errors.append("Station entry without a name: %r" % (st,))
                continue
            if name in names:
                warnings.append("Duplicate station name: %s" % name)
            names.add(name)
            lines = st.get("lines") or []
            if not lines:
                warnings.append("Station '%s' has no line information" % name)
            all_lines.update(lines)
        for line in MAIN_LINES:
            if line not in all_lines:
                warnings.append("Main line '%s' not covered by station data" % line)
        connected = set()
        pairs = set()
        for c in self.data["connections"]:
            s, d = c.get("source"), c.get("dest")
            if s not in names:
                errors.append("Connection source '%s' is not a known station" % s)
            if d not in names:
                errors.append("Connection destination '%s' is not a known station" % d)
            if s == d:
                errors.append("Self-loop connection at '%s'" % s)
            dist = c.get("distance")
            if not isinstance(dist, (int, float)) or isinstance(dist, bool) or dist <= 0:
                errors.append("Invalid distance %r for %s -> %s" % (dist, s, d))
            elif dist > MAX_REASONABLE_DISTANCE:
                warnings.append("Unusually large distance %r m for %s -> %s" % (dist, s, d))
            connected.add(s)
            connected.add(d)
            pairs.add((s, d, c.get("line")))
        for s, d, line in pairs:
            if (d, s, line) not in pairs:
                warnings.append("Connection %s -> %s (%s) has no reverse direction" % (s, d, line))
        for name in names:
            if name not in connected:
                warnings.append("Station '%s' has no connections" % name)
        return (not errors), errors, warnings

"""Path visualization and export."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return "%d meters" % meters
    return "%.2f km (%d meters)" % (meters / 1000.0, meters)


def build_route_info(sp, target, number_to_station, query=None):
    stations, edges = sp.path_with_edges(target)
    if not stations:
        raise ValueError("No path found")
    names = [number_to_station.get(s, str(s)) for s in stations]
    distance = sum(e.weight for e in edges)
    transfers = sp.get_transfer_count(target)
    fare = Sum_money(distance)
    segments = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["stations"].append(number_to_station.get(e.dest, str(e.dest)))
            segments[-1]["distance"] += e.weight
        else:
            segments.append({
                "line": e.line,
                "stations": [number_to_station.get(e.source, str(e.source)),
                             number_to_station.get(e.dest, str(e.dest))],
                "distance": e.weight,
            })
    transfer_stations = [seg["stations"][0] for seg in segments[1:]]
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    return {
        "query": query or {"start": names[0], "end": names[-1]},
        "path": names,
        "distance": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time": round(calculate_estimated_time(distance, transfers), 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": segments,
    }


def summarize(info):
    return {
        "distance": info["distance"],
        "fare": info["fare"],
        "transfer_count": info["transfer_count"],
        "estimated_time": info["estimated_time"],
    }


def format_text(info):
    out = []
    q = info.get("query", {})
    out.append("=" * 50)
    out.append("Route: %s -> %s" % (q.get("start", info["path"][0]), q.get("end", info["path"][-1])))
    if q.get("mode"):
        out.append("Mode: %s" % q["mode"])
    out.append("-" * 50)
    for i, seg in enumerate(info["segments"], 1):
        out.append("Segment %d [%s]: %s (%s)" % (i, seg["line"], " -> ".join(seg["stations"]),
                                                format_distance(seg["distance"])))
        if i < len(info["segments"]):
            out.append("  Transfer at: %s" % seg["stations"][-1])
    out.append("-" * 50)
    out.append("Station sequence (from end to start): %s" % " <- ".join(reversed(info["path"])))
    out.append("Stations passed: %d" % len(info["path"]))
    out.append("Total distance: %s" % format_distance(info["distance"]))
    out.append("Transfers: %d%s" % (info["transfer_count"],
                                     (" (at " + ", ".join(info["transfer_stations"]) + ")")
                                     if info["transfer_stations"] else ""))
    out.append("Lines involved: %s" % ", ".join(str(l) for l in info["lines"]))
    out.append("Estimated time: %.1f minutes" % info["estimated_time"])
    out.append("Fare: %d yuan" % info["fare"])
    out.append("=" * 50)
    return "\n".join(out)


def format_json(info):
    data = {
        "query": info.get("query", {}),
        "path": info["path"],
        "distance": info["distance"],
        "fare": info["fare"],
        "estimated_time": info["estimated_time"],
        "transfer_count": info["transfer_count"],
        "transfer_stations": info["transfer_stations"],
        "lines": info["lines"],
    }
    return json.dumps(data, ensure_ascii=False, indent=2)

"""Command-line interface for the Beijing Subway fare calculation system."""

import argparse
import difflib
import os
import re
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data, build_graph  # noqa: E402
from data_manager import SubwayDataManager  # noqa: E402
from pathfinding import PathfindingConfig, ShortestPath, VALID_MODES  # noqa: E402
from report import build_route_info, format_text, format_json  # noqa: E402

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    p.add_argument("--data_file", default=DATA_FILE_PATH)
    return p.parse_args(argv)


def parse_query(line):
    parts = [x.strip() for x in re.split(r"[;,；，]", line)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERROR)
    return parts[0], parts[1]


def resolve_station(name, station_to_number, role):
    if name in station_to_number:
        return name
    lower = {k.lower(): k for k in station_to_number}
    if name.lower() in lower:
        return lower[name.lower()]
    matches = difflib.get_close_matches(name, list(station_to_number), n=1, cutoff=0.5)
    if matches:
        print("Note: %s station '%s' not found, using nearest match '%s'" % (role, name, matches[0]))
        return matches[0]
    raise ValueError("%s station '%s' does not exist, please check the station name" % (role, name))


def run_check(data):
    ok, errors, warnings = SubwayDataManager(data).check_integrity()
    print("Data integrity check: %d stations, %d connections" %
          (len(data["stations"]), len(data["connections"])))
    for e in errors:
        print("ERROR: " + e)
    for w in warnings:
        print("WARNING: " + w)
    print("Result: %s" % ("PASSED" if ok else "FAILED"))
    return 0 if ok else 1


def handle_query(line, graph, s2n, n2s, config, args):
    start, end = parse_query(line)
    start = resolve_station(start, s2n, "Starting")
    end = resolve_station(end, s2n, "Ending")
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    try:
        sp = ShortestPath(graph, s2n[start], config)
    except Exception as exc:
        raise RuntimeError("Path calculation error: %s" % exc)
    if not sp.has_path_to(s2n[end]):
        raise RuntimeError("No path found from '%s' to '%s' under current settings" % (start, end))
    try:
        info = build_route_info(sp, s2n[end], n2s,
                                {"start": start, "end": end, "mode": config.mode})
    except (TypeError, ValueError) as exc:
        raise RuntimeError("Fare calculation error: %s" % exc)
    text = format_json(info) if args.format == "json" else format_text(info)
    print(text)
    if args.output_file:
        try:
            with open(args.output_file, "a", encoding="utf-8") as f:
                f.write(text + "\n")
        except OSError as exc:
            print("Failed to write output file: %s" % exc)


def main(argv=None):
    args = parse_args(argv)
    try:
        data = load_subway_data(args.data_file)
    except (OSError, ValueError) as exc:
        print("Failed to load subway data: %s" % exc)
        return 1
    if args.check_data:
        return run_check(data)
    graph, s2n, n2s = build_graph(data)
    prefs = [x.strip() for x in args.line_preference.split(",")] if args.line_preference else []
    config = PathfindingConfig(args.mode, args.max_transfers, prefs)
    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (or 'quit').")
    while True:
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("quit", "exit", "q"):
            break
        try:
            handle_query(line, graph, s2n, n2s, config, args)
        except (ValueError, RuntimeError) as exc:
            print(exc)
        except Exception as exc:  # last-resort handler
            print("Unexpected error: %s" % exc)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Core graph data structures for the Beijing Subway network.

The network is modelled as a weighted undirected graph:

* vertices are station numbers (integers assigned by the caller);
* every edge stores the distance between two adjacent stations (meters)
  and the name of the subway line serving that connection.

Undirected edges are stored as two directed half-edges, one in each
adjacency list, so that edges returned by path searches are always
oriented in the direction of travel.
"""


class Edge:
    """A connection between two adjacent stations on a given line."""

    def __init__(self, source, dest, distance, line):
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise TypeError("Edge distance must be a number")
        if distance < 0:
            raise ValueError("Edge distance cannot be negative")
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        """Weight used by the shortest-path algorithm (= distance in meters)."""
        return self.distance

    @weight.setter
    def weight(self, value):
        self.distance = value

    def other(self, vertex):
        """Return the endpoint opposite to ``vertex``."""
        if vertex == self.source:
            return self.dest
        if vertex == self.dest:
            return self.source
        raise ValueError("Vertex %r is not an endpoint of this edge" % (vertex,))

    def reversed(self):
        """Return a new edge with source and destination swapped."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph stored as adjacency lists.

    Attributes:
        graph: dict mapping station number -> list of outgoing ``Edge``.
        station_lines: dict mapping station name -> list of line names.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self._edges = []

    # ----------------------------------------------------------- vertices
    def add_vertex(self, vertex):
        if vertex not in self.graph:
            self.graph[vertex] = []

    def has_vertex(self, vertex):
        return vertex in self.graph

    def vertices(self):
        return list(self.graph.keys())

    def V(self):
        return len(self.graph)

    # -------------------------------------------------------------- lines
    def set_station_lines(self, station_name, lines):
        """Record the lines serving a station (duplicates removed, order kept)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines or []))

    def get_station_lines(self, station_name):
        return list(self.station_lines.get(station_name, []))

    def is_transfer_station(self, station_name):
        """A station served by more than one line is a transfer station."""
        return len(self.station_lines.get(station_name, [])) > 1

    def all_lines(self):
        lines = set()
        for line_list in self.station_lines.values():
            lines.update(line_list)
        for edge in self._edges:
            lines.add(edge.line)
        return sorted(lines)

    # -------------------------------------------------------------- edges
    def add_edge(self, edge):
        """Add an undirected edge (stored as two oriented half-edges)."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.graph[edge.source].append(edge)
        if edge.source != edge.dest:
            self.graph[edge.dest].append(edge.reversed())
        self._edges.append(edge)

    def adj(self, vertex):
        """Edges leaving ``vertex`` (oriented away from it)."""
        return self.graph.get(vertex, [])

    def edges(self):
        return list(self._edges)

    def E(self):
        return len(self._edges)

    def degree(self, vertex):
        return len(self.graph.get(vertex, []))

    def __repr__(self):
        return "Graph(V=%d, E=%d)" % (self.V(), self.E())

"""Loading of the subway network data file."""

import json
import os

DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    Raises:
        FileNotFoundError / OSError if the file cannot be read.
        ValueError if the content is not valid JSON or has a bad structure.
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise ValueError("'stations' and 'connections' must be lists")
    return data


def build_network(data=None, path=DATA_FILE_PATH):
    """Convenience helper returning a ``SubwayNetwork`` built from data."""
    from data_manager import SubwayNetwork

    if data is None:
        data = load_subway_data(path)
    return SubwayNetwork(data)

"""Beijing Subway fare calculation.

Fare structure (distance in kilometers):
    [0, 6)   -> 3 yuan
    [6, 12)  -> 4 yuan
    [12, 22) -> 5 yuan
    [22, 32) -> 6 yuan
    >= 32    -> 7 yuan, +1 yuan for every further 20 km
Fractional distances (meters) are rounded up before the lookup.
"""

import math

# (exclusive upper bound in meters, fare in yuan)
FARE_TABLE = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_START_METERS = 32000
EXTENDED_BASE_FARE = 7
EXTENDED_STEP_METERS = 20000

AVERAGE_SPEED_KMH = 35.0
TRANSFER_TIME_MINUTES = 5.0


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise TypeError("Distance must be a number (meters)")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare (yuan, int) for a travel distance given in meters."""
    _validate_distance(distance)
    meters = math.ceil(distance)
    for upper, fare in FARE_TABLE:
        if meters < upper:
            return fare
    return EXTENDED_BASE_FARE + (meters - EXTENDED_START_METERS) // EXTENDED_STEP_METERS


def calculate_fare(distance):
    """Alias of :func:`Sum_money`."""
    return Sum_money(distance)


def calculate_estimated_time(distance, transfer_count):
    """Estimate travel time in minutes from distance (meters) and transfers."""
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, (int, float)):
        raise TypeError("Transfer count must be a number")
    if transfer_count < 0:
        raise ValueError("Transfer count cannot be negative")
    riding = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    total = riding + transfer_count * TRANSFER_TIME_MINUTES
    return round(float(total), 1)

"""Shortest path computation (Dijkstra with a binary-heap priority queue).

The search runs over states ``(station, current_line[, transfers])`` so
that transfers at multi-line interchange stations are handled exactly.
Each query mode defines a lexicographic cost key:

* shortest_distance: (distance, transfers, non-preferred distance)
* least_transfer:    (transfers, weighted distance, distance)
* comprehensive:     (distance + transfer penalty + preference penalty, ...)

All key components are non-negative and additive, so Dijkstra's algorithm
remains correct.  Complexity: O((V + E) log V) on the state graph.
"""

import heapq
import itertools
import math

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY_METERS = 4000
NON_PREFERRED_PENALTY_RATIO = 0.3


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError("Invalid mode %r, expected one of %s" % (mode, ", ".join(VALID_MODES)))
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])

    def __repr__(self):
        return "PathfindingConfig(mode=%r, max_transfers=%r, preferred_lines=%r)" % (
            self.mode, self.max_transfers, self.preferred_lines)


class MinPQ:
    """Minimum priority queue backed by a binary heap."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def del_min(self):
        if not self._heap:
            raise IndexError("del_min from empty priority queue")
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def min(self):
        if not self._heap:
            raise IndexError("min of empty priority queue")
        key, _, item = self._heap[0]
        return key, item

    def is_empty(self):
        return not self._heap

    def size(self):
        return len(self._heap)

    def __len__(self):
        return len(self._heap)


class ShortestPath:
    """Single-source shortest paths from ``source`` (a station number)."""

    def __init__(self, graph, source, config=None):
        if not graph.has_vertex(source):
            raise ValueError("Source station %r is not in the graph" % (source,))
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self._preferred = set(self.config.preferred_lines)
        self.dist_to = {v: math.inf for v in graph.vertices()}
        self._best_state = {}
        self._parent = {}
        self._costs = {}
        self._run()

    # ---------------------------------------------------------------- core
    def _key(self, dist, transfers, nonpref):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, dist + nonpref * NON_PREFERRED_PENALTY_RATIO, dist)
        if mode == "comprehensive":
            score = dist + transfers * TRANSFER_PENALTY_METERS + nonpref * NON_PREFERRED_PENALTY_RATIO
            return (score, transfers, dist)
        return (dist, transfers, nonpref)

    def _run(self):
        max_t = self.config.max_transfers
        track = max_t is not None
        start = (self.source, None, 0)
        best_key = {start: self._key(0, 0, 0)}
        self._costs[start] = (0, 0, 0)
        pq = MinPQ()
        pq.insert(best_key[start], start)
        settled = set()

        while not pq.is_empty():
            key, state = pq.del_min()
            if state in settled or key > best_key[state]:
                continue
            settled.add(state)
            vertex, line, _ = state
            dist, transfers, nonpref = self._costs[state]
            if vertex not in self._best_state:
                self._best_state[vertex] = state
                self.dist_to[vertex] = dist

            for edge in self.graph.adj(vertex):
                new_transfers = transfers + (1 if line is not None and edge.line != line else 0)
                if track and new_transfers > max_t:
                    continue
                new_state = (edge.dest, edge.line, new_transfers if track else 0)
                if new_state in settled:
                    continue
                new_dist = dist + edge.weight
                new_nonpref = nonpref
                if self._preferred and edge.line not in self._preferred:
                    new_nonpref += edge.weight
                new_key = self._key(new_dist, new_transfers, new_nonpref)
                old = best_key.get(new_state)
                if old is None or new_key < old:
                    best_key[new_state] = new_key
                    self._costs[new_state] = (new_dist, new_transfers, new_nonpref)
                    self._parent[new_state] = (state, edge)
                    pq.insert(new_key, new_state)

    # ------------------------------------------------------------- queries
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, math.inf)

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edge) from source to target.

        Returns ([], []) when the target is unreachable.
        """
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while state in self._parent:
            prev, edge = self._parent[state]
            edges.append(edge)
            state = prev
        edges.reverse()
        path = [self.source] + [e.dest for e in edges]
        return path, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the path (-1 if unreachable)."""
        if target not in self._best_state:
            return -1
        _, edges = self.path_with_edges(target)
        count = 0
        for prev, cur in zip(edges, edges[1:]):
            if prev.line != cur.line:
                count += 1
        return count

    def get_lines(self, target):
        """Ordered list of distinct lines used along the path."""
        _, edges = self.path_with_edges(target)
        lines = []
        for edge in edges:
            if not lines or lines[-1] != edge.line:
                if edge.line not in lines:
                    lines.append(edge.line)
        return lines

def normalize_name(name):
    """Normalized key used for tolerant station/line name matching."""
    if not isinstance(name, str):
        return ""
    text = unicodedata.normalize("NFKC", name).strip().casefold()
    text = text.replace("’", "'").replace("‘", "'").replace("`", "'")
    text = re.sub(r"\s+", "", text)
    for suffix in ("subwaystation", "station", "站"):
        if text.endswith(suffix) and len(text) > len(suffix):
            text = text[: -len(suffix)]
            break
    return text


class SubwayNetwork:
    """Holds the raw data plus derived graph and station mappings."""

    def __init__(self, data):
        self.data = {
            "stations": [dict(s) for s in data.get("stations", [])],
            "connections": [dict(c) for c in data.get("connections", [])],
        }
        self.rebuild()

    # ------------------------------------------------------------ building
    def rebuild(self):
        """(Re)build the graph and lookup structures from ``self.data``."""
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        declared_lines = {}

        for station in self.data["stations"]:
            name = station.get("name")
            if not isinstance(name, str) or not name:
                continue
            if name not in self.station_to_number:
                number = len(self.station_to_number)
                self.station_to_number[name] = number
                self.number_to_station[number] = name
                declared_lines[name] = []
            declared_lines[name].extend(station.get("lines") or [])

        connection_lines = {}
        for conn in self.data["connections"]:
            for key in ("source", "dest"):
                name = conn.get(key)
                if not isinstance(name, str) or not name:
                    continue
                if name not in self.station_to_number:
                    number = len(self.station_to_number)
                    self.station_to_number[name] = number
                    self.number_to_station[number] = name
                    declared_lines.setdefault(name, [])
                connection_lines.setdefault(name, []).append(conn.get("line"))

        for name, number in self.station_to_number.items():
            self.graph.add_vertex(number)
            lines = declared_lines.get(name) or [l for l in connection_lines.get(name, []) if l]
            self.graph.set_station_lines(name, lines)

        for conn in self.data["connections"]:
            src, dst = conn.get("source"), conn.get("dest")
            dist = conn.get("distance")
            if src not in self.station_to_number or dst not in self.station_to_number:
                continue
            if isinstance(dist, bool) or not isinstance(dist, (int, float)) or dist < 0:
                continue
            self.graph.add_edge(Edge(self.station_to_number[src], self.station_to_number[dst],
                                     dist, conn.get("line")))

        self._alias = {}
        for name in self.station_to_number:
            self._alias.setdefault(normalize_name(name), name)
        self._line_alias = {}
        for line in self.all_lines():
            self._line_alias.setdefault(normalize_name(line), line)

    # ------------------------------------------------------------- lookups
    def all_lines(self):
        return self.graph.all_lines()

    def resolve_station(self, name):
        """Return the official station name matching ``name`` or None."""
        if not isinstance(name, str):
            return None
        stripped = name.strip()
        if stripped in self.station_to_number:
            return stripped
        return self._alias.get(normalize_name(stripped))

    def resolve_line(self, line):
        if not isinstance(line, str):
            return None
        if line.strip() in self._line_alias.values():
            return line.strip()
        key = normalize_name(line)
        if key in self._line_alias:
            return self._line_alias[key]
        alt = normalize_name("Line " + line.strip())
        return self._line_alias.get(alt)

    def station_name(self, number):
        return self.number_to_station[number]

    def station_number(self, name):
        return self.station_to_number[name]

    def is_transfer_station(self, name):
        return self.graph.is_transfer_station(name)

    def stations_on_line(self, line):
        return [n for n, lines in self.graph.station_lines.items() if line in lines]

    # ------------------------------------------------------------- updates
    def _ensure_station(self, name, line=None):
        for station in self.data["stations"]:
            if station.get("name") == name:
                lines = station.setdefault("lines", [])
                if line and line not in lines:
                    lines.append(line)
                return
        self.data["stations"].append({"name": name, "lines": [line] if line else []})

    def add_station(self, name, lines=None):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Station name must be a non-empty string")
        name = name.strip()
        if name in self.station_to_number:
            raise ValueError("Station '%s' already exists" % name)
        self.data["stations"].append({"name": name, "lines": list(lines or [])})
        self.rebuild()

    def add_line(self, line_name, stations, distances):
        """Add a new line given its ordered stations and inter-station distances."""
        if not isinstance(line_name, str) or not line_name.strip():
            raise ValueError("Line name must be a non-empty string")
        if len(stations) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(stations) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError("Distances must be positive numbers")
        for name in stations:
            self._ensure_station(name, line_name)
        for (a, b), d in zip(zip(stations, stations[1:]), distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        official = self.resolve_station(name)
        if official is None:
            raise KeyError("Station '%s' does not exist" % name)
        if new_name is not None:
            new_name = new_name.strip()
            if not new_name:
                raise ValueError("New station name cannot be empty")
            if new_name != official and new_name in self.station_to_number:
                raise ValueError("Station '%s' already exists" % new_name)
        for station in self.data["stations"]:
            if station.get("name") == official:
                if new_name:
                    station["name"] = new_name
                if lines is not None:
                    station["lines"] = list(lines)
        if new_name:
            for conn in self.data["connections"]:
                if conn.get("source") == official:
                    conn["source"] = new_name
                if conn.get("dest") == official:
                    conn["dest"] = new_name
        self.rebuild()

    def update_distance(self, source, dest, distance, line=None):
        """Update the distance between two adjacent stations.

        If no connection exists and ``line`` is given, a new one is added.
        Returns the number of connections updated/added.
        """
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        a, b = self.resolve_station(source), self.resolve_station(dest)
        if a is None:
            raise KeyError("Station '%s' does not exist" % source)
        if b is None:
            raise KeyError("Station '%s' does not exist" % dest)
        count = 0
        for conn in self.data["connections"]:
            if {conn.get("source"), conn.get("dest")} == {a, b} and (line is None or conn.get("line") == line):
                conn["distance"] = distance
                count += 1
        if count == 0:
            if line is None:
                raise KeyError("No connection between '%s' and '%s'" % (a, b))
            self._ensure_station(a, line)
            self._ensure_station(b, line)
            self.data["connections"].append({"source": a, "dest": b, "distance": distance, "line": line})
            count = 1
        self.rebuild()
        return count

    def save(self, path):
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(self.data, handle, ensure_ascii=False, indent=2)

    # ----------------------------------------------------- integrity check
    def check_integrity(self):
        """Run data integrity checks; returns a report dict."""
        errors, warnings = [], []
        declared = {}
        for i, station in enumerate(self.data["stations"]):
            name = station.get("name")
            if not isinstance(name, str) or not name.strip():
                errors.append("Station #%d has an invalid name" % i)
                continue
            if name in declared:
                warnings.append("Duplicate station entry: '%s'" % name)
            lines = station.get("lines")
            if not isinstance(lines, list) or not lines:
                warnings.append("Station '%s' has no line information" % name)
            declared[name] = set(lines or [])

        seen_pairs = set()
        connected = set()
        for i, conn in enumerate(self.data["connections"]):
            src, dst, dist, line = conn.get("source"), conn.get("dest"), conn.get("distance"), conn.get("line")
            label = "Connection #%d (%s - %s)" % (i, src, dst)
            if src not in declared:
                errors.append("%s: unknown source station '%s'" % (label, src))
            if dst not in declared:
                errors.append("%s: unknown destination station '%s'" % (label, dst))
            if src == dst:
                errors.append("%s: connects a station to itself" % label)
            if not line:
                errors.append("%s: missing line name" % label)
            if isinstance(dist, bool) or not isinstance(dist, (int, float)):
                errors.append("%s: distance is not a number" % label)
            elif dist <= 0 or dist > MAX_VALID_DISTANCE:
                errors.append("%s: invalid distance %s m" % (label, dist))
            elif dist < MIN_REASONABLE_DISTANCE or dist > MAX_REASONABLE_DISTANCE:
                warnings.append("%s: unusual distance %s m" % (label, dist))
            if src in declared and line and line not in declared[src]:
                warnings.append("%s: station '%s' is not declared on %s" % (label, src, line))
            if dst in declared and line and line not in declared[dst]:
                warnings.append("%s: station '%s' is not declared on %s" % (label, dst, line))
            key = (frozenset((src, dst)), line)
            if key in seen_pairs:
                warnings.append("%s: duplicate connection on %s" % (label, line))
            seen_pairs.add(key)
            connected.add(src)
            connected.add(dst)

        for name in declared:
            if name not in connected:
                warnings.append("Station '%s' has no connections" % name)

        # Connectivity
        components = self._count_components()
        if components > 1:
            warnings.append("Network is not fully connected (%d components)" % components)

        # Coverage of main lines
        line_keys = {normalize_name(l) for l in self.all_lines()}
        missing_lines = []
        for main, aliases in MAIN_LINES.items():
            if not any(normalize_name(a) in line_keys for a in aliases):
                missing_lines.append(main)
        if missing_lines:
            warnings.append("Main lines not found: %s" % ", ".join(missing_lines))
        missing_stations = []
        for names in REPRESENTATIVE_STATIONS:
            if not any(self.resolve_station(n) for n in names):
                missing_stations.append(names[0])
        if missing_stations:
            warnings.append("Representative stations not found: %s" % ", ".join(missing_stations))

        stats = {
            "stations": len(self.station_to_number),
            "connections": len(self.data["connections"]),
            "lines": len(self.all_lines()),
            "transfer_stations": sum(1 for n in self.graph.station_lines if self.graph.is_transfer_station(n)),
            "components": components,
            "main_lines_covered": len(MAIN_LINES) - len(missing_lines),
            "main_lines_total": len(MAIN_LINES),
        }
        return {"valid": not errors, "errors": errors, "warnings": warnings, "stats": stats}

    def _count_components(self):
        seen = set()
        components = 0
        for v in self.graph.vertices():
            if v in seen:
                continue
            components += 1
            queue = deque([v])
            seen.add(v)
            while queue:
                u = queue.popleft()
                for e in self.graph.adj(u):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        queue.append(e.dest)
        return components


def format_integrity_report(report):
    lines = ["Data integrity check report", "=" * 30]
    stats = report["stats"]
    lines.append("Stations: %d" % stats["stations"])
    lines.append("Connections: %d" % stats["connections"])
    lines.append("Lines: %d" % stats["lines"])
    lines.append("Transfer stations: %d" % stats["transfer_stations"])
    lines.append("Connected components: %d" % stats["components"])
    lines.append("Main lines covered: %d/%d" % (stats["main_lines_covered"], stats["main_lines_total"]))
    lines.append("Errors: %d" % len(report["errors"]))
    for err in report["errors"]:
        lines.append("  [ERROR] " + err)
    lines.append("Warnings: %d" % len(report["warnings"]))
    for warn in report["warnings"]:
        lines.append("  [WARNING] " + warn)
    lines.append("Result: %s" % ("PASSED" if report["valid"] else "FAILED"))
    return "\n".join(lines)

"""Route result construction and text / JSON report formatting."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


class FareCalculationError(Exception):
    pass


def build_segments(network, path, edges):
    """Group consecutive edges on the same line into ride segments."""
    segments = []
    for edge in edges:
        src = network.station_name(edge.source)
        dst = network.station_name(edge.dest)
        if segments and segments[-1]["line"] == edge.line:
            seg = segments[-1]
            seg["stations"].append(dst)
            seg["distance"] += edge.weight
        else:
            segments.append({"line": edge.line, "stations": [src, dst], "distance": edge.weight})
    for seg in segments:
        seg["stops"] = len(seg["stations"]) - 1
    return segments


def format_distance(meters):
    if meters >= 1000:
        return "%d meters (%.2f km)" % (round(meters), meters / 1000.0)
    return "%d meters (%.3f km)" % (round(meters), meters / 1000.0)


def build_route_result(network, sp, start, end, config):
    """Build the full route summary dict for a computed ShortestPath."""
    target = network.station_number(end)
    path, edges = sp.path_with_edges(target)
    if not path:
        return None
    names = [network.station_name(v) for v in path]
    distance = sum(e.weight for e in edges)
    segments = build_segments(network, path, edges)
    transfers = []
    for prev, cur in zip(segments, segments[1:]):
        transfers.append({"station": prev["stations"][-1], "from_line": prev["line"], "to_line": cur["line"]})
    lines = []
    for seg in segments:
        if seg["line"] not in lines:
            lines.append(seg["line"])
    try:
        fare = Sum_money(distance)
        est = calculate_estimated_time(distance, len(transfers))
    except (TypeError, ValueError) as exc:
        raise FareCalculationError(str(exc))
    return {
        "query": {
            "start": start,
            "end": end,
            "mode": config.mode,
            "max_transfers": config.max_transfers,
            "line_preference": list(config.preferred_lines),
        },
        "path": names,
        "station_count": len(names),
        "segments": segments,
        "distance": distance,
        "distance_meters": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time": est,
        "estimated_time_minutes": est,
        "transfer_count": len(transfers),
        "transfers": transfers,
        "transfer_stations": [t["station"] for t in transfers],
        "lines": lines,
    }


def format_text(result):
    q = result["query"]
    out = []
    out.append("=" * 50)
    out.append("Route: %s -> %s" % (q["start"], q["end"]))
    out.append("Mode: %s" % q["mode"])
    out.append("-" * 50)
    out.append("Stations (%d): %s" % (result["station_count"], " -> ".join(result["path"])))
    out.append("Line segments:")
    for i, seg in enumerate(result["segments"]):
        out.append("  [%s] %s -> %s (%d stops, %s)" % (
            seg["line"], seg["stations"][0], seg["stations"][-1], seg["stops"], format_distance(seg["distance"])))
        out.append("      " + " -> ".join(seg["stations"]))
        if i < len(result["transfers"]):
            t = result["transfers"][i]
            out.append("  Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    out.append("-" * 50)
    out.append("Total distance: %s" % format_distance(result["distance"]))
    out.append("Transfers: %d" % result["transfer_count"])
    if result["transfer_stations"]:
        out.append("Transfer stations: %s" % ", ".join(result["transfer_stations"]))
    out.append("Lines involved: %s" % ", ".join(str(l) for l in result["lines"]))
    out.append("Estimated time: %.1f minutes" % result["estimated_time"])
    out.append("Fare: %d yuan" % result["fare"])
    out.append("=" * 50)
    return "\n".join(out)


def format_json(results):
    return json.dumps(results, ensure_ascii=False, indent=2)

"""Command-line interface for the Beijing Subway fare calculation system.

Usage:
    python main.py [--mode MODE] [--format text|json] [--output_file PATH]
                   [--max_transfers N] [--line_preference "Line 1"]
                   [--check_data] [--query "Start; End"]

Enter queries as 'Starting Station; Ending Station'. Type 'quit' to exit.
"""

import argparse
import os
import re
import sys

_HERE = os.path.dirname(os.path.abspath(__file__))
if _HERE not in sys.path:
    sys.path.insert(0, _HERE)

from data_loader import DATA_FILE_PATH, load_subway_data  # noqa: E402
from data_manager import SubwayNetwork, format_integrity_report  # noqa: E402
from pathfinding import VALID_MODES, PathfindingConfig, ShortestPath  # noqa: E402
from visualization import FareCalculationError, build_route_result, format_json, format_text  # noqa: E402

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
SEPARATOR_RE = re.compile(r"\s*(?:;|；|,|，|->|→)\s*")
QUIT_COMMANDS = {"q", "quit", "exit"}


class QueryError(Exception):
    pass


def parse_query(text):
    """Split 'Start; End' into (start, end); raises QueryError on bad format."""
    if not isinstance(text, str):
        raise QueryError(FORMAT_ERROR)
    parts = SEPARATOR_RE.split(text.strip())
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise QueryError(FORMAT_ERROR)
    return parts[0].strip(), parts[1].strip()


def process_query(network, text, config):
    """Validate input, compute route and return the result dict.

    Raises QueryError with a user-friendly message on any failure.
    """
    start_raw, end_raw = parse_query(text)
    start = network.resolve_station(start_raw)
    if start is None:
        raise QueryError("Starting station '%s' does not exist, please check the station name" % start_raw)
    end = network.resolve_station(end_raw)
    if end is None:
        raise QueryError("Ending station '%s' does not exist, please check the station name" % end_raw)
    if start == end:
        raise QueryError("Starting station cannot be the same as the ending station")
    try:
        sp = ShortestPath(network.graph, network.station_number(start), config)
    except Exception as exc:  # path calculation failure
        raise QueryError("Path calculation error: %s" % exc)
    try:
        result = build_route_result(network, sp, start, end, config)
    except FareCalculationError as exc:
        raise QueryError("Fare calculation error: %s" % exc)
    except Exception as exc:
        raise QueryError("Path calculation error: %s" % exc)
    if result is None:
        msg = "No route found from '%s' to '%s'" % (start, end)
        if config.max_transfers is not None:
            msg += " within %d transfer(s)" % config.max_transfers
        raise QueryError(msg)
    return result


def _non_negative_int(value):
    try:
        n = int(value)
    except ValueError:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    if n < 0:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    return n


def build_parser():
    parser = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    parser.add_argument("--output_file", help="Export query results to this file")
    parser.add_argument("--format", choices=["text", "json"], default="text", help="Output format")
    parser.add_argument("--mode", choices=list(VALID_MODES), default="shortest_distance",
                        help="Path query mode")
    parser.add_argument("--max_transfers", type=_non_negative_int, default=None,
                        help="Maximum number of transfers")
    parser.add_argument("--line_preference", default=None,
                        help="Preferred line(s), comma separated (e.g. 'Line 1')")
    parser.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    parser.add_argument("--data_file", default=DATA_FILE_PATH, help="Path of subway data JSON file")
    parser.add_argument("--query", default=None, help="Run a single query 'Start; End' and exit")
    return parser


def _resolve_preferences(network, raw):
    preferred = []
    if not raw:
        return preferred
    for item in re.split(r"[,，;；]", raw):
        item = item.strip()
        if not item:
            continue
        resolved = network.resolve_line(item)
        if resolved is None:
            print("Warning: line '%s' not found in network data, preference ignored" % item)
        elif resolved not in preferred:
            preferred.append(resolved)
    return preferred


class OutputWriter:
    def __init__(self, path, fmt):
        self.path = path
        self.fmt = fmt
        self.results = []

    def render(self, result):
        return format_json(result) if self.fmt == "json" else format_text(result)

    def add(self, result):
        self.results.append(result)
        if not self.path:
            return
        if self.fmt == "json":
            content = format_json(self.results[0] if len(self.results) == 1 else self.results)
        else:
            content = "\n\n".join(format_text(r) for r in self.results)
        try:
            with open(self.path, "w", encoding="utf-8") as handle:
                handle.write(content + "\n")
        except OSError as exc:
            print("Error: cannot write output file '%s': %s" % (self.path, exc))


def main(argv=None):
    parser = build_parser()
    args = parser.parse_args(argv)

    try:
        data = load_subway_data(args.data_file)
        network = SubwayNetwork(data)
    except (OSError, ValueError) as exc:
        print("Error: failed to load subway data: %s" % exc)
        return 1

    if args.check_data:
        report = network.check_integrity()
        print(format_integrity_report(report))
        return 0 if report["valid"] else 1

    try:
        config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers,
                                   preferred_lines=_resolve_preferences(network, args.line_preference))
    except ValueError as exc:
        print("Error: %s" % exc)
        return 1

    writer = OutputWriter(args.output_file, args.format)

    def handle(text):
        try:
            result = process_query(network, text, config)
        except QueryError as exc:
            print("Error: %s" % exc)
            return False
        print(writer.render(result))
        writer.add(result)
        if args.output_file:
            print("Result exported to %s" % args.output_file)
        return True

    if args.query is not None:
        return 0 if handle(args.query) else 1

    print("Beijing Subway Fare Calculator")
    print("Enter 'Starting Station; Ending Station' to query, 'quit' to exit.")
    while True:
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        text = line.strip()
        if not text:
            continue
        if text.lower() in QUIT_COMMANDS:
            break
        if text.lower() in ("help", "h", "?"):
            print("Format: Starting Station; Ending Station  (separators: ; , ->)")
            continue
        handle(text)
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

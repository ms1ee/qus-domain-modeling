import os
from typing import Dict, List, Tuple

from lottery_system.models import User

DEFAULT_LIST_TEXT = """\
Li Bin,Technology Department
Chen Qixian,Product Department
Jin Duo,Marketing Department
Ma Jiaqi,Cheng Ru group
Zhang Wei,Technology Department
Wang Fang,Technology Department
Liu Yang,Technology Department
Zhao Lei,Product Department
Sun Li,Product Department
Zhou Jie,Marketing Department
Wu Min,Marketing Department
Zheng Hao,Cheng Ru group
Xu Jing,Cheng Ru group
Huang Lei,Cheng Ru group
Lin Xue,Human Resources Department
He Tao,Human Resources Department
Gao Yan,Finance Department
Luo Bin,Finance Department
"""


class ListManager:
    """Loads and validates participant lists.

    Custom list format (one person per line): ``Name,Department``.
    Blank lines and lines starting with ``#`` are ignored.
    """

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"Unable to read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"Unable to read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            normalized = line.replace("，", ",")
            if "," in normalized:
                parts = [p.strip() for p in normalized.split(",")]
            elif "\t" in normalized:
                parts = [p.strip() for p in normalized.split("\t")]
            else:
                parts = [normalized]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {idx}: '{line}' "
                    f"(expected 'Name,Department')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

"""Prize configuration management."""

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    def __init__(self):
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self._prizes}
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize already exists: {name}")
                continue
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None,
                  associated_drawer: Optional[str] = None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else name
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid prize quantity: {quantity}")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        groups = []
        for g in (avoid_groups or []):
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name, qty, groups, associated_drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def set_avoid_groups(self, index: int, groups) -> bool:
        if not 0 <= index < len(self._prizes):
            return False
        cleaned = []
        for g in groups or []:
            g = str(g).strip()
            if g and g not in cleaned:
                cleaned.append(g)
        self._prizes[index].avoid_groups = cleaned
        return True

    def reorder_prizes(self, order) -> bool:
        n = len(self._prizes)
        try:
            positions = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(positions) != n or sorted(positions) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[p - 1] for p in positions]
        return True

"""Lottery rule engine and round control."""

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User

CHENG_RU_GROUP = "Cheng Ru group"
RELAXED_DRAWER = "Li Bin"
FALLBACK_WATCH_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"


def _norm(s) -> str:
    return " ".join(str(s or "").split()).lower()


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []
        self._rng = random.Random()

    # ---- rule helpers ----
    @staticmethod
    def _is_cheng_ru(user: User) -> bool:
        return _norm(user.department) == _norm(CHENG_RU_GROUP)

    def _fallback_active(self, prize: Prize) -> bool:
        return (
            _norm(prize.associated_drawer) == _norm(FALLBACK_TARGET_DRAWER)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        not_won = [u for u in self.users if u.name not in self.winner_names]

        if self._fallback_active(prize):
            forced = [u for u in not_won if self._is_cheng_ru(u)]
            if forced:
                return forced
            # No Cheng Ru members left; fall through to normal rules.

        if _norm(prize.associated_drawer) == _norm(RELAXED_DRAWER):
            # Relaxed rules: avoidance groups are not applied.
            return not_won

        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in not_won if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        if _norm(prize.associated_drawer) == _norm(RELAXED_DRAWER) and not self._fallback_active(prize):
            parts.append(f"special drawer {RELAXED_DRAWER}: relaxed filtering "
                         f"(avoidance groups ignored)")
        elif prize.avoid_groups:
            parts.append("avoidance groups: " + ", ".join(prize.avoid_groups))
        if self._fallback_active(prize):
            parts.append(f"fallback active: pool limited to '{CHENG_RU_GROUP}' members")
        return "; ".join(parts)

    def _record_winner(self, user: User) -> bool:
        if user.name in self.winner_names:
            return False
        self.winner_names.add(user.name)
        self.winners.append(user)
        return True

    def _update_fallback_state(self, prize: Prize, results: List[User]) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in FALLBACK_WATCH_DRAWERS}:
            self.last_two_draw_results.append(any(self._is_cheng_ru(u) for u in results))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_TARGET_DRAWER):
            self.last_two_draw_results = []

    # ---- drawing ----
    def draw(self, prize: Prize, result_manager=None) -> List[User]:
        """Non-interactive draw of a full round."""
        return self.run_round(prize, result_manager=result_manager,
                              interactive=False, output_func=lambda *a, **k: None)

    def run_round(self, prize: Prize, result_manager=None,
                  input_func: Callable[[str], str] = input,
                  output_func: Callable[..., None] = print,
                  interactive: bool = True) -> List[User]:
        output_func("")
        output_func("=" * 60)
        output_func(f"Prize: {prize.name} | Quantity: {prize.quantity}"
                    f" | Drawer: {prize.associated_drawer or 'N/A'}")
        output_func(f"Rules: {self.describe_rules(prize)}")
        output_func("=" * 60)

        results: List[User] = []
        terminated = False
        while len(results) < prize.quantity:
            pool = self._get_eligible_pool(prize)
            if not pool:
                output_func("No eligible candidates remaining for this prize.")
                break
            n = len(results) + 1
            if interactive:
                cmd = self._safe_input(
                    input_func,
                    f"[Enter] draw {_ordinal(n)} winner | 'pause' | 'terminate': ")
                if cmd == "pause":
                    output_func("Lottery paused.")
                    while True:
                        cmd2 = self._safe_input(
                            input_func, "Paused. Type 'continue' to resume or 'terminate': ")
                        if cmd2 == "continue":
                            output_func("Lottery resumed.")
                            break
                        if cmd2 == "terminate":
                            terminated = True
                            break
                        output_func("Unknown command while paused.")
                elif cmd == "terminate":
                    terminated = True
                if terminated:
                    output_func(f"Round '{prize.name}' terminated and skipped.")
                    break
                # pool may be stale only if state changed; recompute to be safe
                pool = self._get_eligible_pool(prize)
                if not pool:
                    output_func("No eligible candidates remaining for this prize.")
                    break
            output_func(f"Currently drawing the {_ordinal(n)} winner...")
            winner = self._rng.choice(pool)
            if not self._record_winner(winner):
                continue
            if result_manager is not None:
                result_manager.add_winner(winner, prize)
            results.append(winner)
            output_func(f"Congratulations {winner.name} ({winner.department}) "
                        f"for winning {prize.name}")

        self._update_fallback_state(prize, results)

        output_func("-" * 60)
        output_func(f"Round results for {prize.name}:")
        if results:
            for i, u in enumerate(results, 1):
                output_func(f"  {i}. {u.name} ({u.department})")
        else:
            output_func("  (no winners)")
        return results

    @staticmethod
    def _safe_input(input_func, prompt: str) -> str:
        try:
            return (input_func(prompt) or "").strip().lower()
        except EOFError:
            return "terminate"

"""Result recording, summary and export."""

import csv
import os
from dataclasses import dataclass, field
from datetime import datetime
from typing import Dict, List, Tuple

from lottery_system.models import Prize, User

_LEVEL_KEYWORDS = [
    ("special", 0), ("grand", 1), ("first", 2), ("second", 3),
    ("third", 4), ("fourth", 5), ("fifth", 6), ("sixth", 7),
]


def prize_level(name: str) -> int:
    lower = (name or "").lower()
    for kw, rank in _LEVEL_KEYWORDS:
        if f"{kw} prize" in lower or lower.startswith(kw):
            return rank
    return 100


@dataclass
class Winner:
    user: User
    prize: Prize
    time: datetime = field(default_factory=datetime.now)
    seq: int = 0


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self._seq = 0

    def add_winner(self, user: User, prize: Prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self._seq += 1
        self.winners.append(Winner(user, prize, datetime.now(), self._seq))
        return True

    def _sorted_prize_names(self, prizes) -> List[str]:
        names: List[str] = []
        for p in prizes or []:
            if p.name not in names:
                names.append(p.name)
        for w in self.winners:
            if w.prize.name not in names:
                names.append(w.prize.name)
        indexed = list(enumerate(names))
        indexed.sort(key=lambda t: (prize_level(t[1]), t[0]))
        return [n for _, n in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for name in self._sorted_prize_names(prizes):
            ws = [w for w in self.winners if w.prize.name == name]
            ws.sort(key=lambda w: w.seq)
            result[name] = ws
        return result

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        if not self.winners:
            return False, "No winner records to export"
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", newline="", encoding="utf-8") as f:
                writer = csv.writer(f)
                writer.writerow(["Order", "Name", "Department", "Prize Name", "Win Time"])
                order = 0
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        order += 1
                        writer.writerow([order, w.user.name, w.user.department,
                                         name, w.time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class LotteryCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.users = []
        self.lottery = None
        self.results = ResultManager()

    # ---- helpers ----
    @staticmethod
    def ask(prompt):
        try:
            return input(prompt).strip()
        except EOFError:
            return "0"

    def _set_users(self, users):
        self.users = users
        self.lottery = LotteryManager(users)
        self.results = ResultManager()
        print(f"Loaded {len(users)} valid participants.")

    def _process_loaded(self, users, errors):
        valid, dups = self.list_manager.validate_users(users)
        if not errors and not dups:
            self._set_users(valid)
            return
        if errors:
            print("Format errors detected:")
            for e in errors:
                print("  " + e)
        if dups:
            print("Duplicate names detected:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
        choice = self.ask("Type 'r' to remove invalid data and continue, "
                          "or 't' to terminate upload: ").lower()
        if choice == "r":
            self._set_users(valid)
        else:
            print("Upload terminated.")

    # ---- actions ----
    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._process_loaded(users, errors)

    def upload_custom(self):
        path = self.ask("Enter path of txt list (format per line: Name,Department): ")
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            choice = self.ask("No valid data. Press Enter to return.")
            return
        self._process_loaded(users, errors)

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Valid groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            avoid = ", ".join(p.avoid_groups) if p.avoid_groups else "None"
            print(f"  {i}. {p.name} x{p.quantity} | Drawer: {p.associated_drawer or 'N/A'}"
                  f" | Avoid: {avoid}")

    def load_presets(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        self.show_prizes()

    def _select_groups(self):
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = self.ask("Avoidance groups (numbers or names, comma separated, blank for none): ")
        selected = []
        for token in raw.replace("，", ",").split(","):
            token = token.strip()
            if not token:
                continue
            if token.isdigit() and 1 <= int(token) <= len(groups):
                g = groups[int(token) - 1]
            else:
                g = token
            if g not in selected:
                selected.append(g)
        return selected

    def add_custom_prize(self):
        name = self.ask("Prize name: ")
        qty = self.ask("Quantity: ")
        drawer = self.ask("Associated lottery person (blank for none): ") or None
        groups = self._select_groups()
        try:
            p = self.prize_manager.add_prize(name, qty, groups, drawer)
            print(f"Prize added: {p.name}")
        except ValueError as exc:
            print(f"Error: {exc}")

    def configure_avoidance(self):
        self.show_prizes()
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            return
        idx = self.ask("Prize number: ")
        if not idx.isdigit() or not 1 <= int(idx) <= len(prizes):
            print("Invalid prize number.")
            return
        groups = self._select_groups()
        self.prize_manager.set_avoid_groups(int(idx) - 1, groups)
        print("Avoidance groups updated.")

    def reorder(self):
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = self.ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.show_prizes()
        else:
            print("Invalid order; must be a permutation of current serial numbers.")

    def start_lottery(self):
        if not self.users or self.lottery is None:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        for p in prizes:
            self.lottery.run_round(p, result_manager=self.results)
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        grouped = self.results.get_winners_by_prize(self.prize_manager.get_prizes())
        print("\n===== Complete Winner List =====")
        any_w = False
        for name, ws in grouped.items():
            if not ws:
                continue
            any_w = True
            print(f"[{name}]")
            for w in ws:
                print(f"  {w.user.name} ({w.user.department}) - {name}")
        if not any_w:
            print("  (no winners yet)")

    def export(self):
        ok, msg = self.results.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        menu = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.upload_custom),
            "3": ("View groups", self.view_groups),
            "4": ("Load preset prizes", self.load_presets),
            "5": ("Add custom prize", self.add_custom_prize),
            "6": ("Configure avoidance groups", self.configure_avoidance),
            "7": ("Reorder prizes", self.reorder),
            "8": ("View prizes", self.show_prizes),
            "9": ("Start lottery", self.start_lottery),
            "10": ("Show winner summary", self.show_summary),
            "11": ("Export results", self.export),
        }
        print("=== Team Building Smart Lottery System ===")
        while True:
            print()
            for k, (label, _) in menu.items():
                print(f"{k}. {label}")
            print("0. Exit")
            choice = self.ask("Select: ")
            if choice == "0":
                print("Goodbye.")
                break
            item = menu.get(choice)
            if item:
                item[1]()
            else:
                print("Invalid option.")


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

"""
Data loading utilities for the Beijing Subway fare calculation system.

The subway network data is stored in ``subway_data.json`` located next to
this module.  ``DATA_FILE_PATH`` is an absolute path, so it is valid no
matter what the current working directory is.

Expected JSON layout::

    {
        "stations":    [{"name": str, "lines": [str, ...]}, ...],
        "connections": [{"source": str, "dest": str,
                         "distance": int (meters), "line": str}, ...]
    }
"""

import json
import os

# Absolute path of src/subway_data.json (independent of the working directory)
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON data file (defaults to ``DATA_FILE_PATH``)
    :return: the parsed JSON content (a dict)
    :raises FileNotFoundError: if the file does not exist
    :raises ValueError: if the file is not valid JSON or has a wrong structure
    """
    if path is None:
        path = DATA_FILE_PATH
    if not os.path.isfile(path):
        raise FileNotFoundError(f"Subway data file not found: {path}")
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
    except json.JSONDecodeError as exc:
        raise ValueError(f"Subway data file '{path}' is not valid JSON: {exc}") from exc

    if not isinstance(data, dict):
        raise ValueError("Subway data must be a JSON object")
    # Be tolerant: missing sections default to empty lists.
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    if not isinstance(data["stations"], list) or not isinstance(data["connections"], list):
        raise ValueError("'stations' and 'connections' must be JSON arrays")
    return data


def save_subway_data(data, path=DATA_FILE_PATH):
    """Write subway data (same layout as the input file) back to disk."""
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(data, fh, ensure_ascii=False, indent=2)

"""
Core data structures: Edge, Graph (weighted undirected graph using
adjacency lists) and MinPQ (binary-heap based priority queue).
"""


class Edge:
    """A weighted, directed edge between two station numbers on a line."""

    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        """Return the edge in the opposite direction."""
        return Edge(self.dest, self.source, self.distance, self.line)

    def other(self, v):
        """Return the endpoint different from ``v``."""
        return self.dest if v == self.source else self.source

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.weight, self.line) == (
            other.source, other.dest, other.weight, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.weight, self.line))

    def __repr__(self):
        return f"Edge({self.source!r}->{self.dest!r}, {self.weight}m, {self.line!r})"


class Graph:
    """Weighted undirected graph with adjacency lists.

    Station numbers are assigned by the caller.  Adding an edge stores it in
    both directions.  ``graph`` maps station number -> list of outgoing Edges.
    ``station_lines`` maps station name -> list of line names.
    """

    def __init__(self):
        self.graph = {}
        self.station_lines = {}
        self.edge_count = 0

    # ------------------------------------------------------------------ nodes
    @property
    def vertex_count(self):
        return len(self.graph)

    def add_vertex(self, v):
        if v not in self.graph:
            self.graph[v] = []

    def vertices(self):
        return list(self.graph.keys())

    def set_station_lines(self, station_name, lines):
        """Record which lines serve a station (duplicates removed, order kept)."""
        self.station_lines[station_name] = list(dict.fromkeys(lines or []))

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

    # ------------------------------------------------------------------ edges
    def _find(self, u, v, line):
        for e in self.graph.get(u, []):
            if e.dest == v and e.line == line:
                return e
        return None

    def add_edge(self, edge):
        """Add an undirected edge. Returns False if an identical edge
        (same endpoints and line) already exists, in which case its weight
        is updated."""
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        existing = self._find(edge.source, edge.dest, edge.line)
        if existing is not None:
            self.update_edge_weight(edge.source, edge.dest, edge.weight, edge.line)
            return False
        self.graph[edge.source].append(edge)
        if edge.source != edge.dest:
            self.graph[edge.dest].append(edge.reversed())
        self.edge_count += 1
        return True

    def update_edge_weight(self, u, v, distance, line=None):
        """Update the weight of edge(s) between u and v (optionally only on
        a given line). Returns number of undirected edges updated."""
        count = 0
        for a, b in ((u, v), (v, u)):
            for e in self.graph.get(a, []):
                if e.dest == b and (line is None or e.line == line):
                    e.weight = distance
                    e.distance = distance
                    if a == u:
                        count += 1
        return count

    def remove_edge(self, u, v, line=None):
        removed = 0
        for a, b in ((u, v), (v, u)):
            if a not in self.graph:
                continue
            before = len(self.graph[a])
            self.graph[a] = [e for e in self.graph[a]
                             if not (e.dest == b and (line is None or e.line == line))]
            if a == u:
                removed = before - len(self.graph[a])
        self.edge_count -= removed
        return removed

    def adjacent(self, v):
        return self.graph.get(v, [])

    def edges(self):
        """Return each undirected edge once."""
        seen = set()
        result = []
        for u, lst in self.graph.items():
            for e in lst:
                key = (min(repr(e.source), repr(e.dest)), max(repr(e.source), repr(e.dest)), e.line)
                if key in seen:
                    continue
                seen.add(key)
                result.append(e)
        return result


class MinPQ:
    """Minimum priority queue implemented as a binary heap.

    Items are inserted with a comparable key; ties are broken by insertion
    order so that items themselves never need to be comparable.
    """

    def __init__(self):
        self.heap = []
        self._counter = 0

    @property
    def size(self):
        return len(self.heap)

    def __len__(self):
        return len(self.heap)

    def is_empty(self):
        return not self.heap

    def insert(self, key, item):
        self.heap.append((key, self._counter, item))
        self._counter += 1
        self._swim(len(self.heap) - 1)

    push = insert

    def min(self):
        if not self.heap:
            raise IndexError("priority queue is empty")
        key, _, item = self.heap[0]
        return key, item

    def del_min(self):
        if not self.heap:
            raise IndexError("priority queue is empty")
        top = self.heap[0]
        last = self.heap.pop()
        if self.heap:
            self.heap[0] = last
            self._sink(0)
        return top[0], top[2]

    pop = del_min

    def _less(self, i, j):
        return self.heap[i][:2] < self.heap[j][:2]

    def _swim(self, k):
        while k > 0:
            parent = (k - 1) // 2
            if self._less(k, parent):
                self.heap[k], self.heap[parent] = self.heap[parent], self.heap[k]
                k = parent
            else:
                break

    def _sink(self, k):
        n = len(self.heap)
        while True:
            left = 2 * k + 1
            if left >= n:
                break
            child = left
            if left + 1 < n and self._less(left + 1, left):
                child = left + 1
            if self._less(child, k):
                self.heap[k], self.heap[child] = self.heap[child], self.heap[k]
                k = child
            else:
                break

"""
Shortest path computation (Dijkstra with a binary-heap priority queue).

The search runs over states ``(station, current line, transfers)`` so that
transfers at multi-line interchange stations are handled correctly and so
that maximum-transfer limits, line preferences and different query modes
can be supported.  Complexity is O((V+E) log V) in the state space.
"""

from models import MinPQ

VALID_MODES = ("shortest_distance", "least_transfer", "comprehensive")

# Cost in meters assigned to one transfer in "comprehensive" mode.
TRANSFER_PENALTY_METERS = 3000
# Effective weight multiplier for preferred lines in "comprehensive" mode.
PREFERRED_LINE_DISCOUNT = 0.8


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}', expected one of {', '.join(VALID_MODES)}")
        if max_transfers is not None:
            if isinstance(max_transfers, bool) or not isinstance(max_transfers, int) or max_transfers < 0:
                raise ValueError("max_transfers must be a non-negative integer")
        self.mode = mode
        self.max_transfers = max_transfers
        if isinstance(preferred_lines, str):
            preferred_lines = [preferred_lines]
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class ShortestPath:
    """Dijkstra single-source shortest paths from ``source`` (station number)."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config if config is not None else PathfindingConfig()
        self.dist_to = {}
        self.edge_to = {}
        self._labels = {}
        self._parent = {}
        self._best_state = {}
        self._run()

    # ------------------------------------------------------------------ cost
    def _cost(self, label):
        dist, transfers, nonpref, eff = label
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, dist, nonpref)
        if mode == "comprehensive":
            return (eff + transfers * TRANSFER_PENALTY_METERS, transfers, dist)
        return (dist, transfers, nonpref)

    def _run(self):
        mode = getattr(self.config, "mode", "shortest_distance")
        if mode not in VALID_MODES:
            raise ValueError(f"Invalid mode '{mode}'")
        max_t = getattr(self.config, "max_transfers", None)
        preferred = set(getattr(self.config, "preferred_lines", None) or [])
        track_t = max_t is not None

        start = (self.source, None, 0)
        self._labels[start] = (0, 0, 0, 0.0)
        pq = MinPQ()
        pq.insert(self._cost(self._labels[start]), start)
        done = set()

        while not pq.is_empty():
            _, state = pq.del_min()
            if state in done:
                continue
            done.add(state)
            station, line, _t = state
            label = self._labels[state]
            if station not in self._best_state:
                self._best_state[station] = state
                self.dist_to[station] = label[0]
                if state in self._parent:
                    self.edge_to[station] = self._parent[state][1]

            for e in self.graph.adjacent(station):
                is_transfer = line is not None and e.line != line
                nt = label[1] + (1 if is_transfer else 0)
                if track_t and nt > max_t:
                    continue
                ns = (e.dest, e.line, nt if track_t else 0)
                if ns in done:
                    continue
                w = e.weight
                pref = e.line in preferred
                new_label = (label[0] + w, nt,
                             label[2] + (0 if pref else w),
                             label[3] + (w * PREFERRED_LINE_DISCOUNT if pref else w))
                new_cost = self._cost(new_label)
                if ns not in self._labels or new_cost < self._cost(self._labels[ns]):
                    self._labels[ns] = new_label
                    self._parent[ns] = (state, e)
                    pq.insert(new_cost, ns)

    # --------------------------------------------------------------- queries
    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to.get(target, float("inf"))

    def path_with_edges(self, target):
        """Return (list of station numbers, list of Edges) from source to
        target. Both lists are empty if target is unreachable."""
        if target not in self._best_state:
            return [], []
        edges = []
        state = self._best_state[target]
        while state in self._parent:
            prev, e = self._parent[state]
            edges.append(e)
            state = prev
        edges.reverse()
        stations = [self.source] + [e.dest for e in edges]
        return stations, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        """Number of line changes along the path to target (0 if none/unreachable)."""
        _, edges = self.path_with_edges(target)
        count = 0
        for prev, cur in zip(edges, edges[1:]):
            if prev.line != cur.line:
                count += 1
        return count

    def get_lines(self, target):
        """Lines used along the path, in order of first use."""
        _, edges = self.path_with_edges(target)
        return list(dict.fromkeys(e.line for e in edges))

"""
Beijing Subway fare calculation.

Fare structure (distance rounded up to whole kilometers):
    0-6 km   : 3 yuan
    6-12 km  : 4 yuan
    12-22 km : 5 yuan
    22-32 km : 6 yuan
    > 32 km  : +1 yuan for every additional 20 km (or part thereof)
"""

import math
from dataclasses import dataclass

# (upper bound km inclusive, fare yuan)
FARE_TIERS = [
    (6, 3),
    (12, 4),
    (22, 5),
    (32, 6),
]
EXTENDED_BASE_KM = 32
EXTENDED_BASE_FARE = 6
EXTENDED_INCREMENT_KM = 20
EXTENDED_INCREMENT_YUAN = 1

AVERAGE_SPEED_KMH = 35.0        # average operating speed incl. stops
TRANSFER_TIME_MINUTES = 5.0     # average time per transfer


def _validate_distance(distance):
    if isinstance(distance, bool) or not isinstance(distance, (int, float)):
        raise ValueError(f"Distance must be a number, got {distance!r}")
    if math.isnan(distance) or math.isinf(distance):
        raise ValueError("Distance must be a finite number")
    if distance < 0:
        raise ValueError("Distance cannot be negative")


def Sum_money(distance):
    """Return the fare (yuan) for a distance given in meters."""
    _validate_distance(distance)
    km = math.ceil(round(distance, 6) / 1000.0)
    for upper, fare in FARE_TIERS:
        if km <= upper:
            return fare
    extra_km = km - EXTENDED_BASE_KM
    increments = math.ceil(extra_km / EXTENDED_INCREMENT_KM)
    return EXTENDED_BASE_FARE + increments * EXTENDED_INCREMENT_YUAN


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes for a distance (meters) and number of transfers."""
    _validate_distance(distance)
    if isinstance(transfer_count, bool) or not isinstance(transfer_count, (int, float)) or transfer_count < 0:
        raise ValueError("Transfer count must be a non-negative number")
    travel = distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0
    return round(float(travel + transfer_count * TRANSFER_TIME_MINUTES), 1)


@dataclass
class FareResult:
    fare: int
    distance_km: float


def calculate_fare(distance):
    """Return a FareResult for a distance in meters."""
    return FareResult(fare=Sum_money(distance), distance_km=round(distance / 1000.0, 2))

def _norm_line(name):
    return re.sub(r"\s+", "", str(name)).lower()


def _norm_station(name):
    s = " ".join(str(name).split()).lower()
    return s


def _norm_station_loose(name):
    s = _norm_station(name).replace("’", "'")
    if s.endswith("站") and len(s) > 1:
        s = s[:-1]
    if s.endswith(" station") and len(s) > len(" station"):
        s = s[: -len(" station")]
    return s.replace(" ", "")


@dataclass
class Station:
    name: str
    number: int
    lines: list = field(default_factory=list)

    @property
    def is_transfer_station(self):
        return len(self.lines) > 1


@dataclass
class Line:
    name: str
    station_sequence: list = field(default_factory=list)


@dataclass
class DataIntegrityCheck:
    station_coverage_valid: bool = True
    connection_integrity_valid: bool = True
    distance_accuracy_valid: bool = True
    errors: list = field(default_factory=list)
    warnings: list = field(default_factory=list)
    stats: dict = field(default_factory=dict)

    @property
    def is_valid(self):
        return self.station_coverage_valid and self.connection_integrity_valid and self.distance_accuracy_valid

    def summary(self):
        ok = lambda b: "PASS" if b else "FAIL"
        lines = ["Data Integrity Check Report", "=" * 40]
        for k, v in self.stats.items():
            lines.append(f"{k}: {v}")
        lines.append("-" * 40)
        lines.append(f"Station coverage:      {ok(self.station_coverage_valid)}")
        lines.append(f"Connection integrity:  {ok(self.connection_integrity_valid)}")
        lines.append(f"Distance accuracy:     {ok(self.distance_accuracy_valid)}")
        if self.errors:
            lines.append("Errors:")
            lines.extend(f"  - {e}" for e in self.errors)
        if self.warnings:
            lines.append("Warnings:")
            lines.extend(f"  - {w}" for w in self.warnings)
        lines.append("-" * 40)
        lines.append("Overall: " + ("PASS" if self.is_valid else "FAIL"))
        return "\n".join(lines)


class SubwayNetwork:
    """The Beijing Subway network: stations, lines, connections and graph."""

    def __init__(self, data=None, data_source=DATA_FILE_PATH):
        self.data_source = data_source
        if data is None:
            data = load_subway_data(data_source)
        self.station_to_number = {}
        self.number_to_station = {}
        self.stations = {}
        self.connections = []
        self.graph = Graph()
        self.load_warnings = []
        self._exact_index = {}
        self._loose_index = {}
        self._build(data)

    # ------------------------------------------------------------- building
    def _add_station(self, name, lines=()):
        name = str(name).strip()
        if name in self.stations:
            st = self.stations[name]
            for ln in lines:
                if ln not in st.lines:
                    st.lines.append(ln)
        else:
            number = len(self.station_to_number)
            while number in self.number_to_station:
                number += 1
            st = Station(name=name, number=number, lines=list(dict.fromkeys(lines)))
            self.stations[name] = st
            self.station_to_number[name] = number
            self.number_to_station[number] = name
            self.graph.add_vertex(number)
        self.graph.set_station_lines(name, st.lines)
        self._index_station(name)
        return st

    def _index_station(self, name):
        self._exact_index[_norm_station(name)] = name
        self._loose_index[_norm_station_loose(name)] = name

    def _rebuild_index(self):
        self._exact_index.clear()
        self._loose_index.clear()
        for name in self.stations:
            self._index_station(name)

    def _build(self, data):
        for s in data.get("stations", []):
            if not isinstance(s, dict) or not s.get("name"):
                self.load_warnings.append(f"Invalid station entry skipped: {s!r}")
                continue
            self._add_station(s["name"], s.get("lines", []) or [])
        for c in data.get("connections", []):
            try:
                self._add_connection(c["source"], c["dest"], c["distance"], c.get("line"))
            except (KeyError, TypeError, ValueError) as exc:
                self.load_warnings.append(f"Invalid connection skipped: {c!r} ({exc})")

    def _add_connection(self, source, dest, distance, line):
        source, dest = str(source).strip(), str(dest).strip()
        if isinstance(distance, bool) or not isinstance(distance, (int, float)):
            raise ValueError("distance must be numeric")
        for n in (source, dest):
            if n not in self.stations:
                self.load_warnings.append(f"Station '{n}' referenced by a connection but not defined; added")
            self._add_station(n, [line] if line else [])
        self.connections.append({"source": source, "dest": dest, "distance": distance, "line": line})
        self.graph.add_edge(Edge(self.station_to_number[source], self.station_to_number[dest], distance, line))

    # --------------------------------------------------------------- lookups
    def resolve_station(self, name):
        """Return the official station name matching ``name`` or None."""
        if name is None:
            return None
        raw = str(name).strip().strip("'\"")
        if not raw:
            return None
        if raw in self.stations:
            return raw
        found = self._exact_index.get(_norm_station(raw))
        if found:
            return found
        return self._loose_index.get(_norm_station_loose(raw))

    def station_name(self, number):
        return self.number_to_station.get(number)

    @property
    def line_names(self):
        names = []
        for st in self.stations.values():
            names.extend(st.lines)
        for c in self.connections:
            if c["line"]:
                names.append(c["line"])
        return list(dict.fromkeys(names))

    def resolve_line(self, name):
        key = _norm_line(name)
        for ln in self.line_names:
            if _norm_line(ln) == key:
                return ln
        return None

    def line_sequence(self, line):
        adj = defaultdict(list)
        for c in self.connections:
            if c["line"] == line:
                adj[c["source"]].append(c["dest"])
                adj[c["dest"]].append(c["source"])
        if not adj:
            return [n for n, st in self.stations.items() if line in st.lines]
        ends = [n for n in adj if len(set(adj[n])) == 1]
        start = ends[0] if ends else next(iter(adj))
        seq, visited, stack = [], set(), [start]
        while stack:
            n = stack.pop()
            if n in visited:
                continue
            visited.add(n)
            seq.append(n)
            for nb in reversed(adj[n]):
                if nb not in visited:
                    stack.append(nb)
        return seq

    @property
    def lines(self):
        return {ln: Line(ln, self.line_sequence(ln)) for ln in self.line_names}

    def transfer_stations(self):
        return [n for n, st in self.stations.items() if st.is_transfer_station]

    # ----------------------------------------------------------- pathfinding
    def find_route(self, start_name, end_name, config=None):
        """Return (ShortestPath, target number) for a query."""
        s = self.resolve_station(start_name)
        e = self.resolve_station(end_name)
        if s is None:
            raise KeyError(f"Starting station '{start_name}' does not exist")
        if e is None:
            raise KeyError(f"Ending station '{end_name}' does not exist")
        sp = ShortestPath(self.graph, self.station_to_number[s], config or PathfindingConfig())
        return sp, self.station_to_number[e]

    # ---------------------------------------------------------- data update
    def add_line(self, line_name, station_sequence, distances):
        """Add a new line given ordered station names and the distances
        (meters) between consecutive stations."""
        if len(station_sequence) < 2:
            raise ValueError("A line needs at least two stations")
        if len(distances) != len(station_sequence) - 1:
            raise ValueError("Number of distances must equal number of stations minus one")
        for d in distances:
            if isinstance(d, bool) or not isinstance(d, (int, float)) or d <= 0:
                raise ValueError(f"Invalid distance {d!r}")
        for name in station_sequence:
            self._add_station(name, [line_name])
        for a, b, d in zip(station_sequence, station_sequence[1:], distances):
            self._add_connection(a, b, d, line_name)

    def update_station(self, name, new_name=None, lines=None):
        """Rename a station and/or replace its line list."""
        official = self.resolve_station(name)
        if official is None:
            raise KeyError(f"Station '{name}' does not exist")
        st = self.stations[official]
        if lines is not None:
            st.lines = list(dict.fromkeys(lines))
        if new_name and new_name != official:
            if new_name in self.stations:
                raise ValueError(f"Station '{new_name}' already exists")
            number = st.number
            del self.stations[official]
            del self.station_to_number[official]
            self.graph.station_lines.pop(official, None)
            st.name = new_name
            self.stations[new_name] = st
            self.station_to_number[new_name] = number
            self.number_to_station[number] = new_name
            for c in self.connections:
                if c["source"] == official:
                    c["source"] = new_name
                if c["dest"] == official:
                    c["dest"] = new_name
            self._rebuild_index()
        self.graph.set_station_lines(st.name, st.lines)
        return st

    def update_distance(self, station_a, station_b, distance, line=None):
        """Update the distance between two adjacent stations."""
        if isinstance(distance, bool) or not isinstance(distance, (int, float)) or distance <= 0:
            raise ValueError("Distance must be a positive number")
        a, b = self.resolve_station(station_a), self.resolve_station(station_b)
        if a is None or b is None:
            raise KeyError("Station does not exist")
        updated = 0
        for c in self.connections:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                updated += 1
        if not updated:
            raise KeyError(f"No connection between '{a}' and '{b}'")
        self.graph.update_edge_weight(self.station_to_number[a], self.station_to_number[b], distance, line)
        return updated

    def to_dict(self):
        return {
            "stations": [{"name": st.name, "lines": list(st.lines)} for st in self.stations.values()],
            "connections": [dict(c) for c in self.connections],
        }

    def save(self, path=None):
        save_subway_data(self.to_dict(), path or self.data_source)

    # ------------------------------------------------------ integrity check
    def check_integrity(self):
        result = DataIntegrityCheck()
        result.stats = {
            "Stations": len(self.stations),
            "Connections": len(self.connections),
            "Lines": len(self.line_names),
            "Transfer stations": len(self.transfer_stations()),
        }
        for w in self.load_warnings:
            result.warnings.append(w)

        # --- station coverage
        line_keys = {_norm_line(ln): ln for ln in self.line_names}
        for main in MAIN_LINES:
            candidates = [_norm_line(main)] + [_norm_line(a) for a in LINE_ALIASES.get(main, [])]
            m = re.match(r"line(\d+)$", _norm_line(main))
            if m:
                candidates.append(f"{m.group(1)}号线")
            matched = None
            for key, ln in line_keys.items():
                for cand in candidates:
                    if key == cand or (key.startswith(cand) and not key[len(cand):len(cand) + 1].isdigit()):
                        matched = ln
                        break
                if matched:
                    break
            if matched is None:
                result.station_coverage_valid = False
                result.errors.append(f"Main line '{main}' is missing from the data")
            elif len(self.line_sequence(matched)) < 2:
                result.station_coverage_valid = False
                result.errors.append(f"Main line '{main}' has fewer than two stations")
        for main, names in REPRESENTATIVE_STATIONS.items():
            for n in names:
                if self.resolve_station(n) is None:
                    result.warnings.append(f"Representative station '{n}' of {main} not found")

        # --- connection integrity
        connected_stations = set()
        for c in self.connections:
            connected_stations.update((c["source"], c["dest"]))
            if c["source"] == c["dest"]:
                result.connection_integrity_valid = False
                result.errors.append(f"Self-loop connection at '{c['source']}'")
            if not c["line"]:
                result.warnings.append(f"Connection {c['source']} - {c['dest']} has no line")
            else:
                for n in (c["source"], c["dest"]):
                    if c["line"] not in self.stations[n].lines:
                        result.warnings.append(f"Station '{n}' is not declared on line '{c['line']}'")
        for n in self.stations:
            if n not in connected_stations:
                result.connection_integrity_valid = False
                result.errors.append(f"Station '{n}' has no connections")
        if self.stations:
            start = next(iter(self.station_to_number.values()))
            seen = {start}
            dq = deque([start])
            while dq:
                v = dq.popleft()
                for e in self.graph.adjacent(v):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        dq.append(e.dest)
            if len(seen) != len(self.stations):
                result.connection_integrity_valid = False
                result.errors.append(
                    f"Network is not connected: {len(self.stations) - len(seen)} station(s) unreachable")

        # --- distance accuracy
        pair_distances = {}
        for c in self.connections:
            d = c["distance"]
            label = f"{c['source']} - {c['dest']} ({c['line']})"
            if d <= 0:
                result.distance_accuracy_valid = False
                result.errors.append(f"Non-positive distance {d} for {label}")
                continue
            if d < MIN_REASONABLE_DISTANCE or d > MAX_REASONABLE_DISTANCE:
                result.warnings.append(f"Unusual distance {d} m for {label}")
            key = (frozenset((c["source"], c["dest"])), c["line"])
            if key in pair_distances and pair_distances[key] != d:
                result.distance_accuracy_valid = False
                result.errors.append(
                    f"Conflicting distances for {label}: {pair_distances[key]} vs {d}")
            pair_distances.setdefault(key, d)
        return result

"""
Route summary generation, path visualization and export (text / JSON).
"""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    """Short distances in meters, long distances in kilometers."""
    if meters < 1000:
        return f"{int(round(meters))} m"
    return f"{meters / 1000.0:.2f} km"


def build_route_result(network, sp, target, config=None):
    """Build a structured result dict for the path from sp.source to target.
    Returns None if no path exists."""
    stations, edges = sp.path_with_edges(target)
    if not stations or (not edges and sp.source != target):
        return None
    names = [network.station_name(n) for n in stations]
    total = sum(e.weight for e in edges)

    segments, transfers = [], []
    for i, e in enumerate(edges):
        from_name = network.station_name(e.source)
        to_name = network.station_name(e.dest)
        if segments and segments[-1]["line"] == e.line:
            seg = segments[-1]
            seg["to"] = to_name
            seg["stations"].append(to_name)
            seg["distance_meters"] += e.weight
        else:
            if segments:
                transfers.append({"station": from_name,
                                  "from_line": segments[-1]["line"], "to_line": e.line})
            segments.append({"line": e.line, "from": from_name, "to": to_name,
                             "stations": [from_name, to_name], "distance_meters": e.weight})
    for seg in segments:
        seg["stops"] = len(seg["stations"]) - 1

    transfer_count = len(transfers)
    fare = Sum_money(total)
    est = calculate_estimated_time(total, transfer_count)
    lines = list(dict.fromkeys(s["line"] for s in segments))

    query = {"start": names[0], "end": names[-1]}
    if config is not None:
        query.update({"mode": config.mode, "max_transfers": config.max_transfers,
                      "line_preference": list(config.preferred_lines)})
    return {
        "query": query,
        "path": names,
        "station_count": len(names),
        "segments": segments,
        "transfers": transfers,
        "distance": {"meters": total, "kilometers": round(total / 1000.0, 2)},
        "fare": fare,
        "estimated_time_minutes": est,
        "transfer_count": transfer_count,
        "lines": lines,
    }


def format_text(result):
    """Console-formatted text report."""
    sep = "=" * 60
    out = [sep, f"Route: {result['query']['start']} -> {result['query']['end']}"]
    if "mode" in result["query"]:
        out.append(f"Mode: {result['query']['mode']}")
    out.append("-" * 60)
    out.append(f"Station sequence ({result['station_count']} stations):")
    out.append("  " + " -> ".join(result["path"]))
    out.append("Line segments:")
    transfers = result["transfers"]
    for i, seg in enumerate(result["segments"]):
        out.append(f"  {i + 1}. [{seg['line']}] {seg['from']} -> {seg['to']} "
                   f"({seg['stops']} stop(s), {format_distance(seg['distance_meters'])})")
        if i < len(transfers):
            t = transfers[i]
            out.append(f"     >> Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
    if transfers:
        out.append("Transfer stations: " + ", ".join(
            f"{t['station']} ({t['from_line']} -> {t['to_line']})" for t in transfers))
    else:
        out.append("Transfer stations: none (direct route)")
    out.append("Lines involved: " + ", ".join(str(l) for l in result["lines"]))
    m = result["distance"]["meters"]
    out.append(f"Total distance: {format_distance(m)} ({m} meters / {result['distance']['kilometers']:.2f} km)")
    out.append(f"Transfers: {result['transfer_count']}")
    out.append(f"Estimated time: {result['estimated_time_minutes']} minutes")
    out.append(f"Fare: {result['fare']} yuan")
    out.append(sep)
    return "\n".join(out)


def format_json(result):
    """JSON report for programmatic integration."""
    return json.dumps(result, ensure_ascii=False, indent=2)

"""
Command-line interface for the Beijing Subway fare calculation system.

Usage:
    python main.py [--output_file FILE] [--format text|json]
                   [--mode shortest_distance|least_transfer|comprehensive]
                   [--max_transfers N] [--line_preference "Line 1"] [--check_data]

Enter queries as "Starting Station; Ending Station". Type "quit" to exit.
"""

import argparse
import json
import sys

from network import SubwayNetwork
from pathfinding import PathfindingConfig, VALID_MODES
from report import build_route_result, format_text, format_json

FORMAT_ERROR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
EXIT_WORDS = {"quit", "exit", "q"}


class InputError(Exception):
    """Raised for invalid user input."""


def _non_negative_int(value):
    try:
        v = int(value)
    except ValueError:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    if v < 0:
        raise argparse.ArgumentTypeError("must be a non-negative integer")
    return v


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculation system")
    p.add_argument("--output_file", default=None, help="Export query results to this file")
    p.add_argument("--format", default="text", choices=["text", "json"], help="Output format")
    p.add_argument("--mode", default="shortest_distance", choices=list(VALID_MODES), help="Path query mode")
    p.add_argument("--max_transfers", type=_non_negative_int, default=None, help="Maximum number of transfers")
    p.add_argument("--line_preference", default=None,
                   help="Preferred line(s), comma separated (e.g. 'Line 1')")
    p.add_argument("--check_data", action="store_true", help="Run data integrity check and exit")
    return p


def parse_query(raw):
    """Split raw input into (start, end). Raises InputError."""
    if raw is None:
        raise InputError(FORMAT_ERROR)
    text = raw.strip().replace("；", ";")
    if text.count(";") != 1:
        raise InputError(FORMAT_ERROR)
    start, end = (part.strip() for part in text.split(";"))
    if not start or not end:
        raise InputError(FORMAT_ERROR)
    return start, end


def validate_query(network, raw):
    """Return official (start, end) station names. Raises InputError."""
    start, end = parse_query(raw)
    s = network.resolve_station(start)
    if s is None:
        raise InputError(f"Starting station '{start}' does not exist, please check the station name")
    e = network.resolve_station(end)
    if e is None:
        raise InputError(f"Ending station '{end}' does not exist, please check the station name")
    if s == e:
        raise InputError("Starting station cannot be the same as the ending station")
    return s, e


def process_query(network, raw, config, fmt="text"):
    """Process one query. Returns (result dict, rendered string)."""
    start, end = validate_query(network, raw)
    try:
        sp, target = network.find_route(start, end, config)
    except Exception as exc:
        raise InputError(f"Path calculation error: {exc}")
    if not sp.has_path_to(target):
        extra = ""
        if config.max_transfers is not None:
            extra = f" with at most {config.max_transfers} transfer(s)"
        raise InputError(f"No route found from '{start}' to '{end}'{extra}")
    try:
        result = build_route_result(network, sp, target, config)
    except ValueError as exc:
        raise InputError(f"Fare calculation error: {exc}")
    if result is None:
        raise InputError(f"No route found from '{start}' to '{end}'")
    rendered = format_json(result) if fmt == "json" else format_text(result)
    return result, rendered


def _write_output(path, fmt, results, texts):
    with open(path, "w", encoding="utf-8") as fh:
        if fmt == "json":
            json.dump(results, fh, ensure_ascii=False, indent=2)
        else:
            fh.write("\n\n".join(texts) + "\n")


def main(argv=None):
    args = build_parser().parse_args(argv)
    try:
        network = SubwayNetwork()
    except (OSError, ValueError) as exc:
        print(f"Error: failed to load subway data: {exc}")
        return 1

    if args.check_data:
        check = network.check_integrity()
        print(check.summary())
        return 0 if check.is_valid else 1

    preferred = []
    if args.line_preference:
        for item in args.line_preference.replace("，", ",").split(","):
            item = item.strip()
            if not item:
                continue
            resolved = network.resolve_line(item)
            if resolved is None:
                print(f"Warning: line '{item}' not found in data, preference ignored")
            else:
                preferred.append(resolved)
    config = PathfindingConfig(mode=args.mode, max_transfers=args.max_transfers, preferred_lines=preferred)

    print("Beijing Subway Fare Calculation System")
    print("Enter 'Starting Station; Ending Station' to query, or 'quit' to exit.")
    results, texts = [], []
    while True:
        try:
            raw = input("\nQuery> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if raw.strip().lower() in EXIT_WORDS:
            break
        if not raw.strip():
            print(FORMAT_ERROR)
            continue
        try:
            result, rendered = process_query(network, raw, config, args.format)
        except InputError as exc:
            print(exc)
            continue
        except Exception as exc:  # last-resort safety net
            print(f"Unexpected error: {exc}")
            continue
        print(rendered)
        if args.output_file:
            results.append(result)
            texts.append(format_text(result) if args.format == "text" else rendered)
            try:
                _write_output(args.output_file, args.format, results, texts)
                print(f"Result saved to {args.output_file}")
            except OSError as exc:
                print(f"Error: could not write output file: {exc}")
    print("Goodbye!")
    return 0


if __name__ == "__main__":
    sys.exit(main())

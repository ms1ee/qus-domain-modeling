"""Core data models of the lottery system."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A participant of the lottery."""
    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
]


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if self.associated_drawer is not None:
            self.associated_drawer = str(self.associated_drawer).strip() or None

    @property
    def level(self) -> str:
        if " - " in self.name:
            return self.name.split(" - ", 1)[0].strip()
        return self.name

    @property
    def level_rank(self) -> int:
        return prize_level_rank(self.name)

    def rule_description(self) -> str:
        parts = []
        if self.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("Avoidance groups: none")
        if self.associated_drawer:
            parts.append(f"Lottery person: {self.associated_drawer}")
        else:
            parts.append("Lottery person: not specified")
        parts.append("Previous winners are excluded")
        return "; ".join(parts)


def prize_level_rank(name: str) -> int:
    """Lower value means higher prize level."""
    level = name.split(" - ", 1)[0].lower() if " - " in name else name.lower()
    for kw, rank in LEVEL_KEYWORDS:
        if kw in level:
            return rank
    return 99

from lottery_system.models import User

    "Qin Lei,Finance Department",
    "You Ming,Finance Department",
    "Xu Jing,Operations Department",
    "He Ping,Operations Department",

class ListManager:
    def __init__(self):
        self.source_type = None
        self.file_path = None

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff")
            if not line:
                continue
            # accept Chinese full-width comma as well
            line = line.replace("，", ",")
            parts = line.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Format Error: Line {idx} '{raw.rstrip()}' - expected 'Name,Department Group'"
                )
                continue
            name, dept = parts[0].strip(), parts[1].strip()
            if not name or not dept:
                errors.append(
                    f"Format Error: Line {idx} '{raw.rstrip()}' - name and department must not be empty"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    def load_default_list(self):
        self.source_type = "default"
        self.file_path = None
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path):
        self.source_type = "custom"
        self.file_path = path
        return self._load_from_path(path)

    def _load_from_path(self, path):
        if not path or not os.path.isfile(path):
            return [], [f"File Error: file '{path}' not found"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.readlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.readlines()
            except Exception as e:  # pragma: no cover
                return [], [f"File Error: cannot read '{path}': {e}"]
        except OSError as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        return self._parse_lines(lines)

    def validate_users(self, users):
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            key = u.name.strip()
            if key in seen:
                duplicates.append(u)
            else:
                seen.add(key)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

"""Prize configuration management."""

from typing import List

from lottery_system.models import Prize

PRESET_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], None),
    ("Second Prize - Shopping Cards", 2, [], None),
    ("First Prize - 1 Day Annual Leave", 1, [], None),
]


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors = []
        for name, qty, avoid, drawer in PRESET_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name must not be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity of prize '{name}' must be positive")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        prize = Prize(name, qty, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        n = len(self.prizes)
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

"""Winner records, summary and export."""

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import prize_level_rank


class Winner:
    def __init__(self, user, prize, time=None, order=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.order = order

    @property
    def prize_name(self) -> str:
        return self.prize.name if hasattr(self.prize, "name") else str(self.prize)

    def __repr__(self):
        return f"Winner({self.user.name}, {self.user.department}, {self.prize_name})"


def _pname(p) -> str:
    return p.name if hasattr(p, "name") else str(p)


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    @property
    def total_winners(self) -> int:
        return len(self.winners)

    def add_winner(self, user, prize):
        w = Winner(user, prize, datetime.now(), len(self.winners) + 1)
        self.winners.append(w)
        return w

    def has_won(self, user) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        names = []
        for p in prizes or []:
            n = _pname(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        indexed = list(enumerate(names))
        indexed.sort(key=lambda t: (prize_level_rank(t[1]), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, n in indexed:
            ws = [w for w in self.winners if w.prize_name == n]
            ws.sort(key=lambda w: (w.time, w.order))
            result[n] = ws
        return result

    def export_to_csv(self, prizes, directory=None):
        if not self.winners:
            return False, "No winners to export"
        directory = directory or os.getcwd()
        path = os.path.join(directory, f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for pname, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, pname,
                                         w.time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and draw execution."""

import random
from typing import List, Optional

from lottery_system.services.result_manager import ResultManager

SPECIAL_DRAWER = "Li Bin"
TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
TARGET_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(s) -> str:
    return " ".join(str(s or "").strip().lower().split())


def is_protected_group(dept) -> bool:
    d = _norm(dept)
    return d in ("cheng ru group", "cheng ru")


class LotteryManager:
    def __init__(self, users, result_manager: Optional[ResultManager] = None, rng=None):
        self.users = list(users)
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winner_names = set(w.user.name for w in self.result_manager.winners)
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.SystemRandom()

    # ---- rule helpers ----
    def is_winner(self, user) -> bool:
        return user.name in self.winner_names

    @staticmethod
    def _drawer_is(prize, name) -> bool:
        return _norm(getattr(prize, "associated_drawer", None)) == _norm(name)

    def is_fallback_active(self, prize) -> bool:
        if not self._drawer_is(prize, TARGET_DRAWER):
            return False
        if len(self.last_two_draw_results) < 2:
            return False
        return not self.last_two_draw_results[-1] and not self.last_two_draw_results[-2]

    def _get_eligible_pool(self, prize):
        not_won = [u for u in self.users if not self.is_winner(u)]
        if self.is_fallback_active(prize):
            protected = [u for u in not_won if is_protected_group(u.department)]
            if protected:
                return protected
        if self._drawer_is(prize, SPECIAL_DRAWER):
            return not_won
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in not_won if _norm(u.department) not in avoid]

    def describe_rules(self, prize) -> str:
        desc = prize.rule_description()
        if self._drawer_is(prize, SPECIAL_DRAWER):
            desc += f"; Special rule: drawn by {SPECIAL_DRAWER}, avoidance groups NOT applied"
        if self.is_fallback_active(prize):
            desc += (f"; Fallback active: pool limited to '{PROTECTED_GROUP}' members "
                     f"(previous rounds by {', '.join(TRIGGER_DRAWERS)} had none)")
        return desc

    # ---- drawing ----
    def record_winner(self, user, prize):
        if self.is_winner(user):
            raise ValueError(f"{user.name} has already won")
        self.winner_names.add(user.name)
        return self.result_manager.add_winner(user, prize)

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.record_winner(user, prize)
        return user

    def finish_round(self, prize, round_winners):
        drawer = _norm(getattr(prize, "associated_drawer", None))
        if drawer in {_norm(n) for n in TRIGGER_DRAWERS}:
            contains = any(is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            self.last_two_draw_results = []

    def remaining_quantity(self, prize) -> int:
        drawn = sum(1 for w in self.result_manager.winners if w.prize_name == prize.name)
        return max(0, prize.quantity - drawn)

"""Command-line interface for the lottery system."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryCLI:
    def __init__(self, input_func=input, output_func=print):
        self.input = input_func
        self.out = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None
        self.status = "idle"

    def ask(self, prompt):
        try:
            return self.input(prompt).strip()
        except EOFError:
            return "0"

    # ---------- list ----------
    def _set_users(self, users):
        self.users = users
        self.lottery = LotteryManager(users, self.result_manager)

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        self._set_users(valid)
        self.out(f"Default list loaded: {len(valid)} participants.")

    def load_custom(self):
        path = self.ask("Enter txt file path: ")
        users, errors = self.list_manager.load_custom_list(path)
        valid, dups = self.list_manager.validate_users(users)
        if not users and errors:
            for e in errors:
                self.out("  " + e)
            self.out("Upload failed.")
            return
        if errors or dups:
            self.out("Invalid data detected:")
            for e in errors:
                self.out("  " + e)
            for d in dups:
                self.out(f"  Duplicate Name: {d.name},{d.department}")
            choice = self.ask("Remove invalid data and continue (r) or terminate upload (t)? ").lower()
            if choice != "r":
                self.out("Upload terminated.")
                return
        self._set_users(valid)
        self.out(f"Custom list loaded: {len(valid)} valid participants.")

    def show_groups(self):
        if not self.users:
            self.out("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self.out("Current valid groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            self.out(f"  {i}. {g}: {c} people")

    # ---------- prizes ----------
    def load_preset_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            self.out("  " + e)
        self.out("Preset prize library loaded.")

    def add_custom_prize(self):
        name = self.ask("Prize name: ")
        qty = self.ask("Quantity: ")
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid = []
        if groups:
            for i, g in enumerate(groups, 1):
                self.out(f"  {i}. {g}")
            sel = self.ask("Avoidance groups (numbers or names, comma-separated, empty for none): ")
        else:
            sel = self.ask("Avoidance groups (names, comma-separated, empty for none): ")
        for item in [s.strip() for s in sel.replace("，", ",").split(",") if s.strip()]:
            if item.isdigit() and 1 <= int(item) <= len(groups):
                g = groups[int(item) - 1]
            else:
                g = item
            if g not in avoid:
                avoid.append(g)
        drawer = self.ask("Associated lottery person (empty for none): ") or None
        try:
            p = self.prize_manager.add_prize(name, qty, avoid, drawer)
            self.out(f"Prize added: {p.name} x{p.quantity}")
        except ValueError as e:
            self.out(f"Error: {e}")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            self.out(f"  {i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def reorder(self):
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        s = self.ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in s.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            self.out("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.out("Prize order updated.")
            self.show_prizes()
        else:
            self.out("Invalid order; must be a permutation of all serial numbers.")

    # ---------- lottery ----------
    def _control(self):
        """Return True to draw, False to terminate round."""
        cmd = self.ask("[Enter]=draw | pause | terminate > ").lower()
        if cmd == "terminate":
            return False
        if cmd == "pause":
            self.status = "paused"
            self.out("Lottery paused. Type 'continue' to resume or 'terminate' to end this round.")
            while True:
                c = self.ask("(paused) > ").lower()
                if c == "continue":
                    self.status = "running"
                    self.out("Lottery resumed.")
                    return True
                if c in ("terminate", "0"):
                    self.status = "running"
                    return False
        return True

    def run_lottery(self):
        if not self.users:
            self.out("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("Please configure prizes first.")
            return
        self.status = "running"
        for rnd, prize in enumerate(prizes, 1):
            remaining = self.lottery.remaining_quantity(prize)
            if remaining == 0:
                continue
            self.out(f"\n===== Round {rnd}: {prize.name} (quantity: {prize.quantity}) =====")
            self.out("Rules: " + self.lottery.describe_rules(prize))
            round_winners = []
            already = prize.quantity - remaining
            for i in range(remaining):
                if not self._control():
                    self.out("Current round terminated, skipping.")
                    break
                self.out(f"Currently drawing the {ordinal(already + i + 1)} winner...")
                user = self.lottery.draw_one(prize)
                if user is None:
                    self.out("No eligible participants left for this prize.")
                    break
                round_winners.append(user)
                self.out(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            self.lottery.finish_round(prize, round_winners)
            self.out(f"--- Round {rnd} results ---")
            if round_winners:
                for i, u in enumerate(round_winners, 1):
                    self.out(f"  {i}. {u.name} ({u.department})")
            else:
                self.out("  No winners in this round.")
        self.status = "finished"
        self.out("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        data = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        self.out("===== Complete Winner List =====")
        if not self.result_manager.winners:
            self.out("No winners yet.")
            return
        for pname, ws in data.items():
            if not ws:
                continue
            self.out(f"[{pname}]")
            for w in ws:
                self.out(f"  {w.user.name} ({w.user.department}) - {w.time:%Y-%m-%d %H:%M:%S}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self.out(msg)

    # ---------- main loop ----------
    def run(self):
        actions = {
            "1": self.load_default, "2": self.load_custom, "3": self.show_groups,
            "4": self.load_preset_prizes, "5": self.add_custom_prize, "6": self.show_prizes,
            "7": self.reorder, "8": self.run_lottery, "9": self.show_summary, "10": self.export,
        }
        while True:
            self.out("\n==== Team Building Smart Lottery System ====")
            self.out("1. Load default list      2. Upload custom list    3. View groups")
            self.out("4. Load preset prizes     5. Add custom prize      6. View prizes")
            self.out("7. Reorder prizes         8. Start lottery         9. Show results")
            self.out("10. Export results        0. Exit")
            c = self.ask("Choose: ")
            if c == "0":
                self.out("Bye.")
                return
            act = actions.get(c)
            if act:
                act()
            else:
                self.out("Invalid choice.")


def main():
    LotteryCLI().run()


if __name__ == "__main__":
    main()

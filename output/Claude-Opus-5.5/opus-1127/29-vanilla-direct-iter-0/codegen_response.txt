"""Grid based path planning for riders."""

MAP_SIZE = 800
CELL_SIZE = 100
GRID_DIM = MAP_SIZE // CELL_SIZE


def to_grid_center(value):
    """Convert a raw coordinate (0-800) into the center of its grid cell.

    ``center = (v // 100) * 100 + 50``; the boundary value 800 is clamped
    into the last cell so the result always lies inside the map.
    """
    v = int(value)
    v = max(0, min(MAP_SIZE - 1, v))
    return (v // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2


def cell_index(value):
    """Return the grid index (0-7) for a coordinate."""
    return max(0, min(GRID_DIM - 1, int(value) // CELL_SIZE))


class PathPlanner:
    """Right angle (Manhattan) path planning on the 8x8 grid.

    The rider first travels horizontally until it is aligned with the target
    column, then vertically.  When the rider is directly above or below the
    target (same x) the special case of pure vertical movement applies.
    Riders do not block each other, so paths can never deadlock.
    """

    @staticmethod
    def is_special_case(start, target):
        """True when the rider is directly above/below the target."""
        return start[0] == target[0] and start[1] != target[1]

    @staticmethod
    def plan_path(start, target):
        """Return the list of waypoints from ``start`` to ``target``."""
        start = (int(start[0]), int(start[1]))
        target = (int(target[0]), int(target[1]))
        if start == target:
            return [start]
        if start[0] == target[0] or start[1] == target[1]:
            return [start, target]
        corner = (target[0], start[1])
        return [start, corner, target]

    @staticmethod
    def plan_multi(start, destinations):
        """Plan a continuous path through several destinations."""
        path = [(int(start[0]), int(start[1]))]
        current = path[0]
        for dest in destinations:
            seg = PathPlanner.plan_path(current, dest)
            path.extend(seg[1:])
            current = (int(dest[0]), int(dest[1]))
        return path

    @staticmethod
    def path_length(path):
        """Total Manhattan length of a waypoint path."""
        total = 0
        for a, b in zip(path, path[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    @staticmethod
    def step(x, y, target, speed):
        """Advance from (x, y) toward ``target`` by at most ``speed`` units.

        Returns ``(new_x, new_y, special_case)``.
        """
        special = PathPlanner.is_special_case((x, y), target)
        remaining = max(0, int(speed))
        tx, ty = int(target[0]), int(target[1])
        x, y = int(x), int(y)
        # horizontal leg first
        if remaining and x != tx:
            d = min(remaining, abs(tx - x))
            x += d if tx > x else -d
            remaining -= d
        # vertical leg
        if remaining and x == tx and y != ty:
            d = min(remaining, abs(ty - y))
            y += d if ty > y else -d
            remaining -= d
        return x, y, special

    @staticmethod
    def sample_points(path, step=10):
        """Yield points along a waypoint path every ``step`` units."""
        if not path:
            return
        yield path[0]
        for a, b in zip(path, path[1:]):
            x, y = a
            while (x, y) != tuple(b):
                x, y, _ = PathPlanner.step(x, y, b, step)
                yield (x, y)

def _next_order_id():

    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique order id.
        res_pos: (x, y) restaurant position.
        cus_pos: (x, y) customer position.
        status: "pending", "assigned", "picking_up", "delivering", "completed".
        rider_id: id of the assigned rider or None.
    """

    def __init__(self, res_pos, cus_pos):
        self.id = _next_order_id()
        self.res_pos = (int(res_pos[0]), int(res_pos[1]))
        self.cus_pos = (int(cus_pos[0]), int(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.created_time = globalvar.get_value("time", 0) or 0
        self.assign_time = None
        self.pickup_time = None
        self.complete_time = None
        self.distance = None
        self.estimated_time = None

    def assign_to(self, rider_id):
        """Record the assignment of this order to rider ``rider_id``."""
        self.rider_id = rider_id
        self.status = "assigned"
        self.assign_time = globalvar.get_value("time", 0) or 0

    @property
    def time_taken(self):
        """Ticks from creation to completion, or None if not completed."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.created_time

    def to_dict(self):
        """Serialize the order to a dict."""
        return {
            "id": self.id,
            "res_pos": list(self.res_pos),
            "cus_pos": list(self.cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_time": self.created_time,
            "assign_time": self.assign_time,
            "pickup_time": self.pickup_time,
            "complete_time": self.complete_time,
            "distance": self.distance,
            "estimated_time": self.estimated_time,

        """Rebuild an order from a dict without consuming a new id."""

"""Rider model: movement, state machine and event log."""

import logging

import globalvar
from path_planner import PathPlanner

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

logger = logging.getLogger(__name__)


def _order_positions(order):
    """Return (restaurant, customer) positions for an order-like object."""
    if hasattr(order, "res_pos"):
        return tuple(order.res_pos), tuple(order.cus_pos)
    return (order[0], order[1]), (order[2], order[3])


def _order_id(order):
    return getattr(order, "id", None)


def _set(order, attr, value):
    try:
        setattr(order, attr, value)
    except (AttributeError, TypeError):
        pass


class Rider:
    """A delivery rider.

    Attributes:
        rider_id, x, y, speed, state (0 idle, 1 picking up, 2 delivering),
        orders (list of assigned orders), special_case (bool),
        log (list of event dicts with at least "event" and "order_id").
    """

    def __init__(self, rider_id, x=435, y=385, speed=10):
        self.rider_id = rider_id
        self.x = x
        self.y = y
        self.speed = speed
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._log_event("spawn", None)

    # ------------------------------------------------------------------
    def _log_event(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("time", 0) or 0,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })
        logger.debug("rider %s %s order %s at (%s,%s)",
                     self.rider_id, event, order_id, self.x, self.y)

    @property
    def state_name(self):
        """Readable state name."""
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def position(self):
        """Current (x, y) position."""
        return (self.x, self.y)

    def is_idle(self):
        """True when the rider has no work."""
        return self.state == IDLE and not self.orders

    def current_target(self):
        """The point the rider is currently heading to, or None."""
        if not self.orders or self.state == IDLE:
            return None
        res, cus = _order_positions(self.orders[0])
        return res if self.state == PICKING_UP else cus

    # ------------------------------------------------------------------
    def _accept(self, order):
        self.state = PICKING_UP
        _set(order, "status", "picking_up")
        if getattr(order, "rider_id", None) is None:
            _set(order, "rider_id", self.rider_id)
        self._log_event("accepted_order", _order_id(order))

    def _deliver(self, order):
        if order in self.orders:
            self.orders.remove(order)
        _set(order, "status", "completed")
        _set(order, "complete_time", globalvar.get_value("time", 0) or 0)
        self.state = IDLE
        self.special_case = False
        self._log_event("delivered_to_customer", _order_id(order))

    def _handle_arrival(self, order):
        res, cus = _order_positions(order)
        if self.state == PICKING_UP and self.position == tuple(res):
            self.state = DELIVERING
            _set(order, "status", "delivering")
            _set(order, "pickup_time", globalvar.get_value("time", 0) or 0)
            self._log_event("arrived_at_merchant", _order_id(order))
        if self.state == DELIVERING and self.position == tuple(cus):
            self._deliver(order)

    def move(self):
        """Advance the rider by one tick, performing state transitions."""
        if self.state == IDLE:
            if not self.orders:
                self.special_case = False
                return
            self._accept(self.orders[0])
        if not self.orders:
            self.state = IDLE
            return
        order = self.orders[0]
        target = self.current_target()
        if self.position != tuple(target):
            self.x, self.y, self.special_case = PathPlanner.step(
                self.x, self.y, target, self.speed)
        if self.position == tuple(target):
            self._handle_arrival(order)

    # ------------------------------------------------------------------
    def to_dict(self):
        """Serialize the rider (orders are stored by id)."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [_order_id(o) for o in self.orders],
            "special_case": self.special_case,
            "log": self.log,
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%s,%s), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state, len(self.orders))

"""System initialization and resource management."""

import json
import logging
import os

import globalvar
from rider import Rider
from order_model import Order

INITIAL_FUNDS = 1000
RIDER_COST = 300
DEFAULT_STATE_FILE = "dispatch_state.json"

logger = logging.getLogger(__name__)


def init_system(initial_funds=INITIAL_FUNDS):
    """Initialize all global state with default values."""
    globalvar._init()
    globalvar.set_value("funds", initial_funds)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("orders", [])
    globalvar.set_value("pending_orders", [])
    globalvar.set_value("time", 0)
    globalvar.set_value("order_id_counter", 0)


class SystemManager:
    """Manages funds, riders, simulated time and system state."""

    def __init__(self, initial_funds=INITIAL_FUNDS, reset=True):
        if reset or globalvar.get_value("riders") is None:
            init_system(initial_funds)

    # properties ---------------------------------------------------------
    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        return globalvar.get_value("riders", [])

    @property
    def time(self):
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        return globalvar.get_value("running", True)

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    # riders -------------------------------------------------------------
    def idle_riders(self):
        """List of riders without any work."""
        return [r for r in self.riders if r.is_idle()]

    def recruit_rider(self):
        """Create a new rider, paying the recruitment cost."""
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        self.riders.append(rider)
        globalvar.set_value("funds", self.funds - RIDER_COST)
        logger.info("Recruited rider %s", counter)
        return rider

    def auto_recruit(self):
        """Recruit riders while funds >= cost and pending > idle riders.

        Returns the list of newly created riders.
        """
        new = []
        pending = globalvar.get_value("pending_orders", [])
        while self.funds >= RIDER_COST and len(pending) > len(self.idle_riders()):
            new.append(self.recruit_rider())
        return new

    def get_rider(self, rider_id):
        """Find a rider by id."""
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    # simulation ---------------------------------------------------------
    def tick(self, order_manager=None):
        """Advance the simulation by one tick."""
        globalvar.set_value("time", self.time + 1)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception:  # pragma: no cover - recovery path
                logger.exception("Error moving rider %s; resetting", rider.rider_id)
                rider.state = 0
        if order_manager is not None:
            self.auto_recruit()
            order_manager.dispatch_pending()

    def check_consistency(self):
        """Repair inconsistent rider states; returns list of fixes."""
        fixes = []
        for r in self.riders:
            if r.state != 0 and not r.orders:
                r.state = 0
                fixes.append("rider %s reset to idle" % r.rider_id)
            if r.state not in (0, 1, 2):
                r.state = 0
                fixes.append("rider %s invalid state reset" % r.rider_id)
        return fixes

    # persistence --------------------------------------------------------
    def save(self, path=DEFAULT_STATE_FILE):
        """Save the full system state to a JSON file."""
        data = {
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "order_id_counter": globalvar.get_value("order_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in globalvar.get_value("orders", [])],
            "pending_orders": [o.id for o in globalvar.get_value("pending_orders", [])],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

    def load(self, path=DEFAULT_STATE_FILE):
        """Load system state from a JSON file."""
        if not os.path.exists(path):
            raise FileNotFoundError(path)
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        init_system(data.get("funds", INITIAL_FUNDS))
        globalvar.set_value("time", data.get("time", 0))
        globalvar.set_value("rider_id_counter", data.get("rider_id_counter", 0))
        globalvar.set_value("order_id_counter", data.get("order_id_counter", 0))
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        by_id = {o.id: o for o in orders}
        globalvar.set_value("orders", orders)
        globalvar.set_value("pending_orders",
                            [by_id[i] for i in data.get("pending_orders", []) if i in by_id])
        riders = []
        for rd in data.get("riders", []):
            r = Rider(rd["rider_id"], rd["x"], rd["y"], rd.get("speed", 10))
            r.state = rd.get("state", 0)
            r.special_case = rd.get("special_case", False)
            r.orders = [by_id[i] for i in rd.get("orders", []) if i in by_id]
            r.log = rd.get("log", r.log)
            riders.append(r)
        globalvar.set_value("riders", riders)
        self.check_consistency()
        return path

"""Order creation, assignment and completion handling."""

import math

import globalvar
from order_model import Order
from path_planner import PathPlanner, to_grid_center


def _euclid(rider, pos):
    return math.sqrt((rider.x - pos[0]) ** 2 + (rider.y - pos[1]) ** 2)


def find_best_rider_for_order(order):
    """Return the rider closest (Euclidean) to the order's restaurant.

    Idle riders are preferred; if none are idle, the least loaded rider
    (ties broken by distance) is chosen.  Returns None if no riders exist.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    res = order.res_pos
    idle = [r for r in riders if r.state == 0 and not r.orders]
    if idle:
        return min(idle, key=lambda r: (_euclid(r, res), r.rider_id))
    return min(riders, key=lambda r: (len(r.orders), _euclid(r, res), r.rider_id))


class OrderManager:
    """Handles order creation, assignment and completion records."""

    @staticmethod
    def convert(coords):
        """Convert raw [rx, ry, cx, cy] to grid-center positions."""
        rx, ry, cx, cy = (to_grid_center(v) for v in coords)
        return (rx, ry), (cx, cy)

    def create_order(self, coords):
        """Create an order from four raw coordinates and queue it."""
        res, cus = self.convert(coords)
        order = Order(res, cus)
        globalvar.get_value("orders").append(order)
        globalvar.get_value("pending_orders").append(order)
        return order

    def assign(self, order, rider):
        """Assign ``order`` to ``rider``."""
        start = rider.position
        # if rider already has orders, its route starts at the last customer
        if rider.orders:
            last = rider.orders[-1]
            start = tuple(last.cus_pos)
        path = PathPlanner.plan_multi(start, [order.res_pos, order.cus_pos])
        order.distance = PathPlanner.path_length(path)
        order.estimated_time = math.ceil(order.distance / max(1, rider.speed))
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        pending = globalvar.get_value("pending_orders", [])
        if order in pending:
            pending.remove(order)
        return path

    def dispatch_pending(self):
        """Assign all pending orders; returns list of (order, rider, path)."""
        results = []
        for order in list(globalvar.get_value("pending_orders", [])):
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            path = self.assign(order, rider)
            results.append((order, rider, path))
        return results

    @staticmethod
    def all_orders():
        return globalvar.get_value("orders", [])

    @staticmethod
    def completed_orders():
        return [o for o in globalvar.get_value("orders", []) if o.status == "completed"]

        for i, row in enumerate(grid):

        self.history.append(line)
        parts = line.split()
        cmd = SHORTCUTS.get(parts[0].lower(), parts[0].lower())
        return cmd, parts[1:]

    @staticmethod
    def parse_coordinates(text):
        """Parse four integer coordinates in range 0-800.

        Raises ValueError with a readable message on invalid input.
        """
        tokens = [t for t in re.split(r"[\s,\[\]()]+", (text or "").strip()) if t]
        if len(tokens) != 4:
            raise ValueError("Please enter exactly 4 coordinates: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for t in tokens:
            try:
                v = int(t)
            except ValueError:
                raise ValueError("'%s' is not an integer" % t)
            if not 0 <= v <= 800:
                raise ValueError("Coordinate %d out of range 0-800" % v)
            values.append(v)
        return values

"""Command-line entry point of the food delivery dispatch simulator."""

import argparse
import logging
import sys

import globalvar
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from path_planner import PathPlanner
from system import SystemManager, load_config

try:  # command history / line editing where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

HELP = """Commands:
  start (s)     input an order: rx ry cx cy (0-800)
  status (st)   system snapshot
  orders (o)    all orders
  riders (r)    rider status change log
  map (m)       ASCII map
  tick (t) [n]  advance simulation n ticks
  run           run until all orders are delivered
  save/load [f] persist state to JSON
  history       command history
  quit/exit (q) exit"""


class App:
    """Main application tying together all managers."""

    def __init__(self, funds=1000):
        self.system = SystemManager(funds)
        self.orders = OrderManager()
        self.inputs = InputManager()
        self.display = DisplayManager()

    def out(self, text):
        print(text)

    def cmd_start(self, args):
        text = " ".join(args) if args else input("Enter rx ry cx cy: ")
        coords = self.inputs.parse_coordinates(text)
        order = self.orders.create_order(coords)
        new = self.system.auto_recruit()
        results = self.orders.dispatch_pending()
        self.out("Order %d: restaurant %s -> customer %s" % (order.id, order.res_pos, order.cus_pos))
        self.out("New rider added: %s" % (", ".join(str(r.rider_id) for r in new) if new else "no"))
        for o, rider, path in results:
            if o is order:
                self.out("Assigned to rider %s (speed %d, distance %d, est. time %d ticks)" % (
                    rider.rider_id, rider.speed, o.distance, o.estimated_time))
                self.out(self.display.render_map([rider], [o], path))
                break
        else:
            self.out("Order is pending: no rider available.")

    def cmd_tick(self, args):
        n = int(args[0]) if args else 1
        for _ in range(max(0, n)):
            self.system.tick(self.orders)
        self.out("Time is now %d" % self.system.time)

    def cmd_run(self, args):
        limit = 10000
        while limit and any(o.status != "completed" for o in self.orders.all_orders()):
            self.system.tick(self.orders)
            limit -= 1
        self.out("All deliverable orders done at time %d" % self.system.time)

    def handle(self, line):
        cmd, args = self.inputs.parse_command(line)
        if cmd is None:
            return
        try:
            if cmd == "start":
                self.cmd_start(args)
            elif cmd == "status":
                self.out(self.display.status(self.system.funds, self.system.riders, self.system.time))
            elif cmd == "orders":
                self.out(self.display.orders_table(self.orders.all_orders()))
            elif cmd == "riders":
                self.out(self.display.rider_logs(self.system.riders))
            elif cmd == "map":
                self.out(self.display.render_map(self.system.riders, self.orders.all_orders()))
            elif cmd == "tick":
                self.cmd_tick(args)
                return
            elif cmd == "run":
                self.cmd_run(args)
                return
            elif cmd == "save":
                self.out("Saved to %s" % self.system.save(*args[:1]))
            elif cmd == "load":
                self.out("Loaded %s" % self.system.load(*args[:1]))
            elif cmd == "history":
                self.out("\n".join("%d %s" % (i + 1, h) for i, h in enumerate(self.inputs.history)))
            elif cmd == "help":
                self.out(HELP)
            elif cmd == "quit":
                self.system.stop()
                return
            else:
                self.out("Unknown command '%s'. Type 'help'." % cmd)
                return
        except ValueError as exc:
            self.out("Error: %s" % exc)
            return
        except Exception as exc:  # recovery
            logging.exception("Unexpected error")
            self.out("Unexpected error: %s (state recovered)" % exc)
            self.system.check_consistency()
            return
        # each processed command iteration represents one tick
        self.system.tick(self.orders)

    def loop(self):
        self.out("Food Delivery Dispatch Simulator. Type 'help' for commands.")
        while globalvar.get_value("running", True):
            try:
                line = input("> ")
            except (EOFError, KeyboardInterrupt):
                break
            self.handle(line)
        self.out("Bye.")


def main(argv=None):
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", help="JSON config file with system parameters")
    parser.add_argument("--debug", action="store_true", help="enable debug logging")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if args.debug else logging.WARNING)
    funds = 1000
    if args.config:
        try:
            funds = int(load_config(args.config).get("funds", 1000))
        except (OSError, ValueError) as exc:
            print("Could not load config: %s" % exc)
    App(funds).loop()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Global variable store shared by all modules of the dispatch simulator.

Typical keys:
    "riders"            -> list[Rider]
    "rider_id_counter"  -> int
    "order_id_counter"  -> int
    "running"           -> bool
    "funds"             -> int
    "tick"              -> int
    "orders"            -> list[Order]   (all orders ever created)
    "pending_orders"    -> list[Order]   (orders waiting for a rider)
    "config"            -> Configuration
"""

_global_dict = None


def _init():
    """(Re)initialise the global dictionary."""
    global _global_dict
    _global_dict = {}


def _ensure():
    """Make sure the global dictionary exists."""
    if _global_dict is None:
        _init()


def set_value(key, value):
    """Store ``value`` under ``key``."""
    _ensure()
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` (``default`` if missing)."""
    _ensure()
    return _global_dict.get(key, default)


def has_key(key):
    """Return True when ``key`` is present in the store."""
    _ensure()
    return key in _global_dict


def keys():
    """Return a list of all stored keys."""
    _ensure()
    return list(_global_dict.keys())

import json
import logging
import os

logger = logging.getLogger(__name__)

DEFAULT_CONFIG_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                                   "config.json")

DEFAULTS = {
    "initial_funds": 300,
    "recruit_threshold": 300,
    "recruit_cost": 300,
    "max_riders": 50,
    "rider_speed": 50,
    "rider_start_x": 400,
    "rider_start_y": 400,
    "base_fee": 20,
    "fee_per_100": 5,
    "map_size": 800,
    "cell_size": 100,
    "max_log_entries": 5000,
    "max_history": 500,
    "state_file": "dispatch_state.json",
    "autosave_on_exit": True,
}


class Configuration:
    """System parameters loaded from a JSON configuration file."""

    def __init__(self, file_path=None):
        self.file_path = file_path or DEFAULT_CONFIG_PATH
        self.values = dict(DEFAULTS)
        self.load()

    @property
    def recruit_threshold(self):
        """Funds threshold for automatic rider recruitment."""
        return int(self.values.get("recruit_threshold", 300))

    def load(self):
        """Load the configuration file, silently falling back to defaults."""
        if not self.file_path or not os.path.exists(self.file_path):
            return
        try:
            with open(self.file_path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                for key, value in data.items():
                    if key in DEFAULTS and value is not None:
                        self.values[key] = value
        except (OSError, ValueError) as exc:
            logger.warning("Could not read config %s: %s", self.file_path, exc)

    def save(self, path=None):
        """Write the configuration to disk."""
        target = path or self.file_path
        with open(target, "w", encoding="utf-8") as fh:
            json.dump(self.values, fh, indent=4)

    def get(self, key, default=None):
        """Return a configuration value."""
        return self.values.get(key, DEFAULTS.get(key, default))

    def set(self, key, value):
        """Set a configuration value."""
        self.values[key] = value

    def to_dict(self):
        """Return a copy of all configuration values."""
        return dict(self.values)

"""Grid map and coordinate conversion utilities (8x8 grid of 100x100 cells)."""

MAP_SIZE = 800
CELL_SIZE = 100


def clamp_coordinate(value, map_size=MAP_SIZE):
    """Clamp a coordinate into [0, map_size]."""
    value = int(value)
    return max(0, min(map_size, value))


def to_grid_index(value, cell_size=CELL_SIZE, map_size=MAP_SIZE):
    """Convert a raw coordinate into a grid cell index (0..7)."""
    value = clamp_coordinate(value, map_size)
    max_index = map_size // cell_size - 1
    return min(value // cell_size, max_index)


def to_cell_center(value, cell_size=CELL_SIZE, map_size=MAP_SIZE):
    """Convert a raw coordinate to the center of its grid cell.

    ``center = (v // 100) * 100 + 50`` with the boundary value 800 mapped
    into the last cell.
    """
    return to_grid_index(value, cell_size, map_size) * cell_size + cell_size // 2


class GridCell:
    """One cell of the grid."""

    def __init__(self, grid_x, grid_y, cell_size=CELL_SIZE):
        self.grid_x = grid_x
        self.grid_y = grid_y
        self.center_x = grid_x * cell_size + cell_size // 2
        self.center_y = grid_y * cell_size + cell_size // 2

    def __repr__(self):
        return "GridCell(%d, %d)" % (self.grid_x, self.grid_y)


class GridMap:
    """The whole map: 800x800 units split into 8x8 cells."""

    def __init__(self, width=MAP_SIZE, height=MAP_SIZE, cell_size=CELL_SIZE):
        self.width = width
        self.height = height
        self.cell_size = cell_size
        self.columns = width // cell_size
        self.rows = height // cell_size
        self.cells = {(gx, gy): GridCell(gx, gy, cell_size)
                      for gx in range(self.columns) for gy in range(self.rows)}

    def cell_at(self, x, y):
        """Return the GridCell containing (x, y)."""
        gx = to_grid_index(x, self.cell_size, self.width)
        gy = to_grid_index(y, self.cell_size, self.height)
        return self.cells[(gx, gy)]

    def in_bounds(self, x, y):
        """Return True if (x, y) is inside the map."""
        return 0 <= x <= self.width and 0 <= y <= self.height

    def is_street(self, x, y):
        """Streets are the grid lines (x or y multiple of the cell size)."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

"""Grid based path planning along streets."""

import math

from grid_map import CELL_SIZE, MAP_SIZE


def path_length(points):
    """Total Manhattan length of a poly-line."""
    total = 0
    for (ax, ay), (bx, by) in zip(points, points[1:]):
        total += abs(bx - ax) + abs(by - ay)
    return total


class PathPoint:
    """A waypoint of a route."""

    __slots__ = ("x", "y", "sequence")

    def __init__(self, x, y, sequence=0):
        self.x = int(x)
        self.y = int(y)
        self.sequence = sequence

    def as_tuple(self):
        """Return (x, y)."""
        return (self.x, self.y)

    def __repr__(self):
        return "PathPoint(%d, %d, #%d)" % (self.x, self.y, self.sequence)


class Route:
    """A planned route: ordered waypoints plus distance and time estimate."""

    def __init__(self, points, speed=50, stops=None):
        pts = [(int(x), int(y)) for x, y in points]
        self.points = [PathPoint(x, y, i) for i, (x, y) in enumerate(pts)]
        self.speed = max(1, int(speed))
        self.stops = list(stops or [])
        self.total_distance = path_length(pts)
        self.estimated_time = int(math.ceil(self.total_distance / self.speed))

    @property
    def coordinates(self):
        """Waypoints as a list of tuples."""
        return [p.as_tuple() for p in self.points]

    def to_dict(self):
        """Serializable representation."""
        return {"points": self.coordinates, "speed": self.speed,
                "total_distance": self.total_distance,
                "estimated_time": self.estimated_time}


class PathPlanner:
    """Plans paths that follow streets (grid lines).

    Restaurants/customers sit at cell centers and are reached by a short
    vertical spur from the horizontal street directly above or below them.
    """

    def __init__(self, cell_size=CELL_SIZE, map_size=MAP_SIZE):
        self.cell_size = cell_size
        self.map_size = map_size

    def is_on_street(self, x, y):
        """True if the point lies on a grid line."""
        return x % self.cell_size == 0 or y % self.cell_size == 0

    @staticmethod
    def is_directly_above_or_below(start, target):
        """Special case: same x, different y."""
        return start[0] == target[0] and start[1] != target[1]

    def _clamp(self, v):
        return max(0, min(self.map_size, v))

    def _line_toward(self, a, b):
        """Nearest grid line from ``a`` in the direction of ``b``."""
        cs = self.cell_size
        if a % cs == 0:
            return a
        if b > a:
            return self._clamp((a // cs + 1) * cs)
        return self._clamp((a // cs) * cs)

    def _street_route(self, p, e):
        """Route between two points that both lie on streets."""
        px, py = p
        ex, ey = e
        if p == e:
            return []
        cs = self.cell_size
        p_h, p_v = py % cs == 0, px % cs == 0
        e_h, e_v = ey % cs == 0, ex % cs == 0
        if p_h and e_h and py == ey:
            return [e]
        if p_v and e_v and px == ex:
            return [e]
        if p_v and e_h:
            return [(px, ey), e]
        if p_h and e_v:
            return [(ex, py), e]
        if p_h and e_h:
            v = self._line_toward(px, ex)
            return [(v, py), (v, ey), e]
        h = self._line_toward(py, ey)
        return [(px, h), (ex, h), e]

    @staticmethod
    def _simplify(points):
        """Remove duplicate and collinear intermediate points."""
        dedup = []
        for pt in points:
            if not dedup or dedup[-1] != pt:
                dedup.append(pt)
        if len(dedup) < 3:
            return dedup
        result = [dedup[0]]
        for i in range(1, len(dedup) - 1):
            a, b, c = result[-1], dedup[i], dedup[i + 1]
            if a[0] == b[0] == c[0] and min(a[1], c[1]) <= b[1] <= max(a[1], c[1]):
                continue
            if a[1] == b[1] == c[1] and min(a[0], c[0]) <= b[0] <= max(a[0], c[0]):
                continue
            result.append(b)
        result.append(dedup[-1])
        return result

    def find_path(self, start, target):
        """Return a list of waypoints (including start) from start to target."""
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        cs = self.cell_size
        points = [(sx, sy)]
        if (sx, sy) == (tx, ty):
            return points
        cur = (sx, sy)
        if not self.is_on_street(sx, sy):
            # leave the cell vertically towards the target
            cur = (sx, self._line_toward(sy, ty))
            points.append(cur)
            if cur == (tx, ty):
                return self._simplify(points)
        if self.is_on_street(tx, ty):
            entry, spur = (tx, ty), False
        else:
            if cur[1] < ty:
                ey = (ty // cs) * cs
            elif cur[1] > ty:
                ey = (ty // cs + 1) * cs
            else:
                ey = (ty // cs) * cs
            entry, spur = (tx, self._clamp(ey)), True
        points.extend(self._street_route(cur, entry))
        if spur:
            points.append((tx, ty))
        return self._simplify(points)

    def distance(self, start, target):
        """Street distance between two points."""
        return path_length(self.find_path(start, target))

    def plan_route(self, start, destinations, speed=50):
        """Plan a continuous route through several destinations."""
        points = [(int(start[0]), int(start[1]))]
        cur = points[0]
        for dest in destinations:
            seg = self.find_path(cur, dest)
            points.extend(seg[1:])
            cur = (int(dest[0]), int(dest[1]))
        return Route(self._simplify(points) if len(points) > 1 else points,
                     speed, stops=[tuple(d) for d in destinations])

"""Order data model."""

import globalvar
from grid_map import (CELL_SIZE, MAP_SIZE, clamp_coordinate, to_cell_center,
                      to_grid_index)
from path_planner import PathPlanner

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"


def _to_int(value):
    if isinstance(value, bool):
        raise ValueError("coordinates must be integers")
    if isinstance(value, float):
        if not value.is_integer():
            raise ValueError("coordinates must be integers")
        return int(value)
    return int(value)


def _next_order_id():
    counter = globalvar.get_value("order_id_counter", 0) or 0
    counter += 1
    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A delivery order from a restaurant to a customer.

    Input coordinates (0-800) are converted to grid cell centers.
    """

    def __init__(self, res_pos, cus_pos, start_time=None, order_id=None):
        if len(res_pos) != 2 or len(cus_pos) != 2:
            raise ValueError("positions must be (x, y) pairs")
        rx, ry = (clamp_coordinate(_to_int(v), MAP_SIZE) for v in res_pos)
        cx, cy = (clamp_coordinate(_to_int(v), MAP_SIZE) for v in cus_pos)
        self.raw_restaurant = (rx, ry)
        self.raw_customer = (cx, cy)
        self.restaurant = (to_cell_center(rx), to_cell_center(ry))
        self.customer = (to_cell_center(cx), to_cell_center(cy))
        self.restaurant_grid = (to_grid_index(rx), to_grid_index(ry))
        self.customer_grid = (to_grid_index(cx), to_grid_index(cy))
        if order_id is None:
            self.id = _next_order_id()
        else:
            self.id = int(order_id)
            if self.id > (globalvar.get_value("order_id_counter", 0) or 0):
                globalvar.set_value("order_id_counter", self.id)
        if start_time is None:
            start_time = globalvar.get_value("tick", 0) or 0
        self.start_time = int(start_time)
        self.status = STATUS_PENDING
        self.rider_id = None
        self.accept_time = None
        self.pickup_time = None
        self.completion_time = None
        self.time_taken = None
        self.active_ticks = 0
        self.fee = 0
        self.efficiency = None
        self.distance = PathPlanner(CELL_SIZE, MAP_SIZE).distance(
            self.restaurant, self.customer)

    # convenience accessors -------------------------------------------------
    @property
    def order_id(self):
        """Alias of ``id``."""
        return self.id

    @property
    def restaurant_x(self):
        return self.restaurant[0]

    @property
    def restaurant_y(self):
        return self.restaurant[1]

    @property
    def customer_x(self):
        return self.customer[0]

    @property
    def customer_y(self):
        return self.customer[1]

    def as_list(self):
        """Return [restaurant_x, restaurant_y, customer_x, customer_y]."""
        return [self.restaurant[0], self.restaurant[1],
                self.customer[0], self.customer[1]]

    # lifecycle -------------------------------------------------------------
    def assign_to(self, rider_id):
        """Assign the order to a rider (by id)."""
        self.rider_id = rider_id
        if self.status == STATUS_PENDING:
            self.status = STATUS_ASSIGNED

    def mark_accepted(self, tick, rider_id=None):
        """Rider accepted the order and heads to the restaurant."""
        if rider_id is not None and self.rider_id is None:
            self.rider_id = rider_id
        self.accept_time = tick
        self.status = STATUS_PICKING_UP

    def mark_picked_up(self, tick):
        """Rider picked up the food."""
        self.pickup_time = tick
        self.status = STATUS_DELIVERING

    def complete(self, tick):
        """Order delivered: record completion time and efficiency."""
        self.completion_time = tick
        elapsed = tick - self.start_time
        self.time_taken = elapsed if elapsed > 0 else self.active_ticks
        self.status = STATUS_COMPLETED
        if self.time_taken:
            self.efficiency = round(self.distance / self.time_taken, 2)
        else:
            self.efficiency = float(self.distance)

    def is_completed(self):
        return self.status == STATUS_COMPLETED

    # persistence -----------------------------------------------------------
    def to_dict(self):
        return {
            "id": self.id, "raw_restaurant": list(self.raw_restaurant),
            "raw_customer": list(self.raw_customer),
            "restaurant": list(self.restaurant), "customer": list(self.customer),
            "start_time": self.start_time, "status": self.status,
            "rider_id": self.rider_id, "accept_time": self.accept_time,
            "pickup_time": self.pickup_time,
            "completion_time": self.completion_time,
            "time_taken": self.time_taken, "active_ticks": self.active_ticks,
            "distance": self.distance, "fee": self.fee,
            "efficiency": self.efficiency,
        }

    @classmethod
    def from_dict(cls, data):
        order = cls(tuple(data["raw_restaurant"]), tuple(data["raw_customer"]),
                    start_time=data.get("start_time", 0), order_id=data["id"])
        for key in ("status", "rider_id", "accept_time", "pickup_time",
                    "completion_time", "time_taken", "active_ticks", "fee",
                    "efficiency"):
            if key in data:
                setattr(order, key, data[key])
        return order

    def __repr__(self):
        return "Order(#%d %s->%s %s)" % (self.id, self.restaurant,
                                         self.customer, self.status)

"""Rider model: state, movement and status log."""

import logging
from enum import IntEnum

import globalvar
from grid_map import CELL_SIZE, MAP_SIZE, clamp_coordinate
from path_planner import PathPlanner

logger = logging.getLogger(__name__)


class RiderState(IntEnum):
    """Rider status enumeration."""
    IDLE = 0
    PICKING_UP = 1
    DELIVERING = 2


IDLE = 0
PICKING_UP = 1
DELIVERING = 2

STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}


def _config_value(key, default):
    cfg = globalvar.get_value("config")
    if cfg is None:
        return default
    try:
        value = cfg.get(key, default)
    except Exception:  # pragma: no cover - defensive
        return default
    return default if value is None else value


class Rider:
    """A delivery rider moving along the streets of the grid."""

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = int(rider_id)
        self.map_size = int(_config_value("map_size", MAP_SIZE))
        self.cell_size = int(_config_value("cell_size", CELL_SIZE))
        if x is None:
            x = _config_value("rider_start_x", 400)
        if y is None:
            y = _config_value("rider_start_y", 400)
        if speed is None:
            speed = _config_value("rider_speed", 50)
        self.x = clamp_coordinate(x, self.map_size)
        self.y = clamp_coordinate(y, self.map_size)
        self.speed = max(1, int(speed))
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.path = []
        self.current_order = None
        self.target = None
        self.distance_travelled = 0
        self.delivered_count = 0
        self._planner = PathPlanner(self.cell_size, self.map_size)
        self._record("spawn", None)

    # helpers ---------------------------------------------------------------
    @property
    def state_name(self):
        return STATE_NAMES.get(self.state, "unknown")

    @property
    def position(self):
        return (self.x, self.y)

    @staticmethod
    def _now():
        return globalvar.get_value("tick", 0) or 0

    def _record(self, event, order_id):
        entry = {"time": self._now(), "event": event, "order_id": order_id,
                 "x": self.x, "y": self.y, "state": int(self.state),
                 "state_name": self.state_name}
        self.log.append(entry)
        limit = int(_config_value("max_log_entries", 5000))
        if limit > 0 and len(self.log) > limit:
            del self.log[:len(self.log) - limit]

    def active_orders(self):
        """Orders assigned to this rider that are not completed yet."""
        return [o for o in self.orders if getattr(o, "status", "") != "completed"]

    def _next_order(self):
        for order in self.orders:
            if getattr(order, "status", "") != "completed":
                return order
        return None

    def _plan_to(self, target):
        self.target = (int(target[0]), int(target[1]))
        self.special_case = self._planner.is_directly_above_or_below(
            (self.x, self.y), self.target)
        self.path = self._planner.find_path((self.x, self.y), self.target)[1:]

    def remaining_route(self):
        """Waypoints still to be travelled (current leg)."""
        return [(self.x, self.y)] + list(self.path)

    def _advance(self):
        remaining = self.speed
        while remaining > 0 and self.path:
            tx, ty = self.path[0]
            dx, dy = tx - self.x, ty - self.y
            dist = abs(dx) + abs(dy)
            if dist <= remaining:
                self.x, self.y = tx, ty
                self.path.pop(0)
                remaining -= dist
                self.distance_travelled += dist
                continue
            if dx != 0:
                step = min(abs(dx), remaining)
                self.x += step if dx > 0 else -step
            else:
                step = min(abs(dy), remaining)
                self.y += step if dy > 0 else -step
            remaining -= step
            self.distance_travelled += step

    def _reset_to_idle(self):
        self.state = IDLE
        self.current_order = None
        self.target = None
        self.path = []
        self.special_case = False

    # tick ------------------------------------------------------------------
    def move(self):
        """Advance the rider by one tick.

        Handles state transitions (idle -> picking up -> delivering -> idle)
        and records the corresponding log entries.
        """
        tick = self._now()
        if self.state == IDLE:
            order = self._next_order()
            if order is None:
                self.special_case = False
                return
            if order.rider_id is None:
                order.assign_to(self.rider_id)
            order.mark_accepted(tick, self.rider_id)
            self.current_order = order
            self.state = PICKING_UP
            self._plan_to(order.restaurant)
            self._record("accepted_order", order.id)
            return

        order = self.current_order
        if order is None or order not in self.orders or order.status == "completed":
            logger.warning("Rider %d inconsistent state; resetting", self.rider_id)
            self._reset_to_idle()
            return
        order.active_ticks += 1
        if self.target is None:
            self.target = order.restaurant if self.state == PICKING_UP else order.customer
        if not self.path and (self.x, self.y) != self.target:
            self._plan_to(self.target)
        self._advance()
        if self.path or (self.x, self.y) != self.target:
            return
        if self.state == PICKING_UP:
            order.mark_picked_up(tick)
            self.state = DELIVERING
            self._plan_to(order.customer)
            self._record("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            order.complete(tick)
            self.orders.remove(order)
            self.delivered_count += 1
            self._reset_to_idle()
            self._record("delivered_to_customer", order.id)

    # persistence -----------------------------------------------------------
    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": int(self.state),
                "orders": [o.id for o in self.orders],
                "current_order": self.current_order.id if self.current_order else None,

_PLANNER = PathPlanner(CELL_SIZE, MAP_SIZE)


def _active_orders(rider):
    return [o for o in rider.orders if getattr(o, "status", "") != "completed"]


def find_best_rider_for_order(order, riders=None):
    """Return the best rider for ``order`` (closest rider first).

    Idle riders are preferred and ranked by street distance to the restaurant.
    If all riders are busy, the rider with the smallest workload (and then the
    shortest distance from its last drop-off) wins, which balances load.
    Returns None if there are no riders.
    """
    if riders is None:
        riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    target = order.restaurant

    idle = [r for r in riders if r.state == IDLE and not _active_orders(r)]
    if idle:
        def idle_key(r):
            manhattan = abs(r.x - target[0]) + abs(r.y - target[1])
            return (_PLANNER.distance((r.x, r.y), target), manhattan, r.rider_id)
        return min(idle, key=idle_key)

    def busy_key(r):
        active = _active_orders(r)
        start = active[-1].customer if active else (r.x, r.y)
        return (len(active), _PLANNER.distance(start, target), r.rider_id)
    return min(riders, key=busy_key)


class OrderManager:
    """Handles order creation, assignment and bookkeeping."""

    def __init__(self, planner=None):
        self.planner = planner or PathPlanner(CELL_SIZE, MAP_SIZE)
        if globalvar.get_value("orders") is None:
            globalvar.set_value("orders", [])
        if globalvar.get_value("pending_orders") is None:
            globalvar.set_value("pending_orders", [])

    @staticmethod
    def all_orders():
        return globalvar.get_value("orders") or []

    @staticmethod
    def pending_orders():
        return globalvar.get_value("pending_orders") or []

    def completed_orders(self):
        return [o for o in self.all_orders() if o.status == "completed"]

    @staticmethod
    def compute_fee(order):
        cfg = globalvar.get_value("config")
        base = int(cfg.get("base_fee", 20)) if cfg else 20
        per = int(cfg.get("fee_per_100", 5)) if cfg else 5
        return base + per * (order.distance // 100)

    def create_order(self, rx, ry, cx, cy):
        """Create and register an order (coordinates already validated)."""
        order = Order((rx, ry), (cx, cy))
        order.fee = self.compute_fee(order)
        orders = self.all_orders()
        orders.append(order)
        globalvar.set_value("orders", orders)
        pending = self.pending_orders()
        pending.append(order)
        globalvar.set_value("pending_orders", pending)
        logger.debug("Created %r", order)
        return order

    def assign_order(self, order):
        """Assign an order to the best rider; returns the rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        pending = self.pending_orders()
        if order in pending:
            pending.remove(order)
        logger.debug("Assigned order %d to rider %d", order.id, rider.rider_id)
        return rider

    def assign_pending(self):
        """Try to assign all waiting orders; returns list of (order, rider)."""
        results = []
        for order in list(self.pending_orders()):
            if order.status == "completed":
                self.pending_orders().remove(order)
                continue
            rider = self.assign_order(order)
            if rider is None:
                break
            results.append((order, rider))
        return results

    def planned_route_for(self, rider, order):
        """Full planned route of ``rider`` up to delivering ``order``."""
        stops = []
        cur = rider.current_order
        if cur is not None and cur.status != "completed":
            if rider.state == PICKING_UP:
                stops.append(cur.restaurant)
            stops.append(cur.customer)
        for o in rider.orders:
            if o is cur or o.status == "completed":
                continue
            stops.append(o.restaurant)
            stops.append(o.customer)
            if o is order:
                break
        if order is cur:
            pass
        elif order not in rider.orders:
            stops.extend([order.restaurant, order.customer])
        return self.planner.plan_route((rider.x, rider.y), stops, rider.speed)

    def _idx(self, v):
        return max(0, min(self.n - 1, int(round(v / self.res))))

    def render_map(self, riders, orders=None, route=None):
        """Render the map.

        '#' streets, '.' cell interior, 'S' restaurant, 'C' customer,
        'R' rider, '*' planned route.
        """
        step = self.cell_size // self.res
        grid = [["#" if (r % step == 0 or c % step == 0) else "."
                 for c in range(self.n)] for r in range(self.n)]
        if route is not None:
            pts = route.coordinates if hasattr(route, "coordinates") else list(route)
            for (ax, ay), (bx, by) in zip(pts, pts[1:]):
                ai, aj = self._idx(ax), self._idx(ay)
                bi, bj = self._idx(bx), self._idx(by)
                if ai == bi:
                    for j in range(min(aj, bj), max(aj, bj) + 1):
                        grid[j][ai] = "*"
                elif aj == bj:
                    for i in range(min(ai, bi), max(ai, bi) + 1):
                        grid[aj][i] = "*"
                else:
                    grid[aj][ai] = "*"
                    grid[bj][bi] = "*"
        for order in orders or []:
            if order.status == "completed":
                continue
            if order.status in ("pending", "assigned", "picking_up"):
                grid[self._idx(order.restaurant[1])][self._idx(order.restaurant[0])] = "S"
            grid[self._idx(order.customer[1])][self._idx(order.customer[0])] = "C"
        for rider in riders or []:
            grid[self._idx(rider.y)][self._idx(rider.x)] = "R"

        header = "     " + " ".join(str(i // step) if i % step == 0 else " "
                                    for i in range(self.n))
        lines = [header]
        for j, row in enumerate(grid):
            label = "%3d" % (j * self.res) if j % step == 0 else "   "
            lines.append(label + "  " + " ".join(row))

class InputError(ValueError):

class InputManager:
    """Parses commands, supports shortcuts and keeps command history."""

    SHORTCUTS = {
        "n": "start", "new": "start", "s": "status", "o": "orders",
        "r": "riders", "m": "map", "t": "tick", "h": "history",
        "q": "quit", "exit": "quit", "?": "help",
    }
    COMMANDS = ("start", "status", "orders", "riders", "map", "tick",
                "history", "save", "load", "help", "quit")

    def __init__(self, max_history=500):
        self.command_history = []
        self.max_history = max_history

    def add_history(self, line):
        self.command_history.append(line)
        if len(self.command_history) > self.max_history:
            del self.command_history[:len(self.command_history) - self.max_history]

    def _resolve_history(self, line):
        if not self.command_history:
            raise InputError("command history is empty")
        if line == "!!":
            return self.command_history[-1]
        try:
            idx = int(line[1:])
        except ValueError:
            raise InputError("use !! or !<number> to repeat a command")
        if idx < 1 or idx > len(self.command_history):
            raise InputError("no history entry %d" % idx)
        return self.command_history[idx - 1]

    def parse(self, line):
        """Return (command, args)."""
        line = (line or "").strip()
        if not line:
            return "", []
        if line.startswith("!"):
            line = self._resolve_history(line)
            print("> " + line)
        self.add_history(line)
        tokens = line.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if not tokens:
            return "", []
        cmd = tokens[0].lower()
        cmd = self.SHORTCUTS.get(cmd, cmd)
        return cmd, tokens[1:]

    @staticmethod
    def validate_coordinates(tokens):
        """Validate four integer coordinates in range 0-800."""
        if isinstance(tokens, str):
            tokens = tokens.replace(",", " ").replace("[", " ").replace("]", " ").split()
        tokens = list(tokens)
        if len(tokens) != 4:
            raise InputError("expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y (got %d)" % len(tokens))
        values = []
        for tok in tokens:
            try:
                value = int(str(tok).strip())
            except ValueError:
                raise InputError("'%s' is not an integer" % tok)
            if not MIN_COORD <= value <= MAX_COORD:
                raise InputError("coordinate %d out of range %d-%d"
                                 % (value, MIN_COORD, MAX_COORD))
            values.append(value)
        return values

    def prompt_coordinates(self, reader=input):
        """Ask the user for the four coordinates."""
        line = reader("Enter restaurant_x restaurant_y customer_x customer_y "
                      "(0-800): ")
        return self.validate_coordinates(line)

    @staticmethod
    def help_text():
        return "\n".join([
            "Commands (shortcut in brackets):",
            "  start [rx ry cx cy] (n)  create an order (coordinates 0-800)",
            "  status (s)               funds and rider states",
            "  orders (o)               all order details",
            "  riders (r)               rider status change log",
            "  map (m)                  ASCII map",
            "  tick [n] (t)             advance simulated time n ticks",
            "  history (h)              command history; !! or !<n> repeats",
            "  save [file] / load [file] persist / restore state (JSON)",
            "  help (?)                 this help",
            "  quit / exit (q)          leave the program",
            "Each processed command advances the simulation by one tick.",
        ])

class SystemManager:
    """Core class managing funds, riders, time and user commands."""

    def __init__(self, config=None):
        self.config = config if config is not None else Configuration()
        init_system(self.config)
        cs = int(self.config.get("cell_size", 100))
        ms = int(self.config.get("map_size", 800))
        self.grid_map = globalvar.get_value("grid_map")
        self.order_manager = OrderManager(PathPlanner(cs, ms))
        self.display = DisplayManager(ms, cs)
        self.input_manager = InputManager(int(self.config.get("max_history", 500)))
        self.last_route = None

    # properties ------------------------------------------------------------
    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @property
    def riders(self):
        return globalvar.get_value("riders") or []

    @property
    def running(self):
        return bool(globalvar.get_value("running", False))

    @running.setter
    def running(self, value):
        globalvar.set_value("running", bool(value))

    @property
    def current_tick(self):
        return globalvar.get_value("tick", 0)

    # funds & riders --------------------------------------------------------
    def add_funds(self, amount):
        globalvar.set_value("funds", int(self.funds + amount))

    def recruit_rider(self):
        """Create a new rider with an auto-incremented id."""
        counter = (globalvar.get_value("rider_id_counter", 0) or 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        riders = self.riders
        riders.append(rider)
        globalvar.set_value("riders", riders)
        logger.info("Recruited rider %d", counter)
        return rider

    def check_recruitment(self):
        """Recruit a rider automatically when funds >= threshold."""
        threshold = int(self.config.get("recruit_threshold", 300))
        cost = int(self.config.get("recruit_cost", threshold))
        max_riders = int(self.config.get("max_riders", 50))
        if self.funds >= threshold and len(self.riders) < max_riders:
            rider = self.recruit_rider()
            self.add_funds(-cost)
            return rider
        return None

    def remove_rider(self, rider_id):
        """Remove an idle rider; its orders go back to the pending queue."""
        for rider in list(self.riders):
            if rider.rider_id == rider_id:
                for o in rider.active_orders():
                    o.rider_id = None
                    o.status = "pending"
                    self.order_manager.pending_orders().append(o)
                self.riders.remove(rider)
                return True
        return False

    # orders ----------------------------------------------------------------
    def place_order(self, rx, ry, cx, cy):
        """Create an order, collect its fee, recruit if possible, assign it."""
        order = self.order_manager.create_order(rx, ry, cx, cy)
        self.add_funds(order.fee)
        recruited = self.check_recruitment()
        rider = self.order_manager.assign_order(order)
        route = self.order_manager.planned_route_for(rider, order) if rider else None
        self.last_route = (rider, route) if rider else None
        return {"order": order, "recruited": recruited, "rider": rider,
                "route": route}

    # time ------------------------------------------------------------------
    def advance_tick(self):
        """Advance the simulation by one tick; returns new log entries."""
        tick = self.current_tick + 1
        globalvar.set_value("tick", tick)
        try:
            self.order_manager.assign_pending()
        except Exception:
            logger.exception("Assignment failed; recovering")
            self.check_consistency()
        events = []
        for rider in list(self.riders):
            before = len(rider.log)
            try:
                rider.move()
            except Exception:
                logger.exception("Rider %d failed to move; recovering",
                                 rider.rider_id)
                self.check_consistency()
            for entry in rider.log[before:] if len(rider.log) >= before else []:
                events.append((rider.rider_id, entry))
        return events

    def check_consistency(self):
        """Repair inconsistent state; returns list of fixes applied."""
        fixes = []
        if not isinstance(self.funds, int):
            globalvar.set_value("funds", int(self.funds or 0))
            fixes.append("funds type")
        ms = int(self.config.get("map_size", 800))
        pending = self.order_manager.pending_orders()
        for o in list(pending):
            if o.status == "completed" or o.rider_id is not None:
                pending.remove(o)
                fixes.append("pending order %d" % o.id)
        for r in self.riders:
            r.x = max(0, min(ms, int(r.x)))
            r.y = max(0, min(ms, int(r.y)))
            r.orders = [o for o in r.orders if o.status != "completed"]
            if r.state != IDLE and (r.current_order is None or
                                    r.current_order not in r.orders):
                r.state = IDLE
                r.current_order = None
                r.path = []
                r.target = None
                fixes.append("rider %d reset" % r.rider_id)
        for f in fixes:
            logger.warning("Consistency fix: %s", f)
        return fixes

    # persistence -----------------------------------------------------------
    def save_state(self, path=None):
        path = path or self.config.get("state_file", "dispatch_state.json")
        data = {
            "config": self.config.to_dict(),
            "funds": self.funds, "tick": self.current_tick,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "order_id_counter": globalvar.get_value("order_id_counter", 0),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in self.order_manager.all_orders()],
            "pending_orders": [o.id for o in self.order_manager.pending_orders()],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

    def load_state(self, path=None):
        path = path or self.config.get("state_file", "dispatch_state.json")
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        init_system(self.config)
        globalvar.set_value("tick", int(data.get("tick", 0)))
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        by_id = {o.id: o for o in orders}
        globalvar.set_value("orders", orders)
        globalvar.set_value("pending_orders",
                            [by_id[i] for i in data.get("pending_orders", [])
                             if i in by_id])
        riders = [Rider.from_dict(d, by_id) for d in data.get("riders", [])]
        globalvar.set_value("riders", riders)
        globalvar.set_value("funds", int(data.get("funds", 0)))
        globalvar.set_value("rider_id_counter", int(data.get("rider_id_counter",
                                                             len(riders))))
        globalvar.set_value("order_id_counter",
                            max([int(data.get("order_id_counter", 0))] +
                                [o.id for o in orders]))
        self.check_consistency()
        return path

    # commands --------------------------------------------------------------
    def handle_command(self, line):
        """Process one command; returns True if time should advance."""
        try:
            cmd, args = self.input_manager.parse(line)
        except InputError as exc:
            print("Error: %s" % exc)
            return False
        if not cmd:
            return False
        handlers = {
            "start": self.cmd_start, "status": self.cmd_status,
            "orders": self.cmd_orders, "riders": self.cmd_riders,
            "map": self.cmd_map, "tick": self.cmd_tick,
            "history": self.cmd_history, "save": self.cmd_save,
            "load": self.cmd_load, "help": self.cmd_help, "quit": self.cmd_quit,
        }
        handler = handlers.get(cmd)
        if handler is None:
            print("Unknown command '%s'. Type 'help' for the list." % cmd)
            return False
        try:
            return handler(args)
        except InputError as exc:
            print("Error: %s" % exc)
        except (OSError, ValueError, KeyError) as exc:
            print("Error: %s" % exc)
        except Exception as exc:  # pragma: no cover - recovery path
            logger.exception("Command failed")
            fixes = self.check_consistency()
            print("Unexpected error (%s); state recovered (%d fixes)."
                  % (exc, len(fixes)))
        return False

    def cmd_start(self, args):
        if args:
            coords = self.input_manager.validate_coordinates(args)
        else:
            try:
                coords = self.input_manager.prompt_coordinates()
            except EOFError:
                raise InputError("no coordinates given")
        result = self.place_order(*coords)
        order = result["order"]

"""Entry point for the food delivery dispatch simulator."""

import argparse
import logging
import sys

from config import Configuration
from system import SystemManager


def main(argv=None):
    """Parse arguments and run the interactive simulator."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch simulator")
    parser.add_argument("--config", help="path to JSON configuration file")
    parser.add_argument("--load", nargs="?", const="", default=None,
                        help="load saved state (optional file path)")
    parser.add_argument("--debug", action="store_true", help="verbose logging")
    args = parser.parse_args(argv)

    logging.basicConfig(level=logging.DEBUG if args.debug else logging.WARNING,
                        format="%(asctime)s %(levelname)s %(name)s: %(message)s")

    manager = SystemManager(Configuration(args.config))
    if args.load is not None:
        try:
            manager.load_state(args.load or None)
            print("State loaded.")
        except (OSError, ValueError, KeyError) as exc:
            print("Could not load state: %s" % exc)
    manager.run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

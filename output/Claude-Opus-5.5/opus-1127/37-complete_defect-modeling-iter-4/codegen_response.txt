from __future__ import annotations

from typing import List, Optional


def normalize(text: Optional[str]) -> str:
    """Normalize a string for case/whitespace-insensitive comparison."""
    if text is None:
        return ""
    return " ".join(str(text).split()).lower()


class User:
    """A lottery participant."""

    def __init__(self, name: str, department: str):
        self.name = name.strip() if isinstance(name, str) else name
        self.department = department.strip() if isinstance(department, str) else department

    def __eq__(self, other) -> bool:
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self) -> int:
        return hash((self.name, self.department))

    def __repr__(self) -> str:
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


class Prize:
    """A prize configured for a lottery round."""

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
    ):
        self.name = name.strip() if isinstance(name, str) else name
        self.quantity = quantity
        self.avoid_groups: List[str] = list(avoid_groups) if avoid_groups else []
        if isinstance(associated_drawer, str):
            associated_drawer = associated_drawer.strip() or None
        self.associated_drawer: Optional[str] = associated_drawer

    def rule_description(self) -> str:
        parts = ["Previous winners are excluded"]
        if normalize(self.associated_drawer) == "li bin":
            parts.append("lottery person Li Bin: avoidance groups NOT applied")
        elif self.avoid_groups:
            parts.append("avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("no avoidance groups")
        if self.associated_drawer:
            parts.append(f"lottery person: {self.associated_drawer}")
        return "; ".join(parts)

    def __repr__(self) -> str:
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity!r}, "
            f"avoid_groups={self.avoid_groups!r}, associated_drawer={self.associated_drawer!r})"
        )

class ListManager:
    """Loads, validates and analyses participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LIST_LINES)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not str(path).strip():
            return [], ["File Error: no file path given"]
        path = str(path).strip()
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        if not path.lower().endswith(".txt"):
            return [], [f"File Error: only .txt files are supported: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
            except Exception as exc:  # pragma: no cover
                return [], [f"File Error: cannot read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"File Error: cannot read file {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            line = line.replace("\uff0c", ",")
            parts = line.split(",")
            if len(parts) != 2:
                errors.append(
                    f"Format Error: Line {idx}: '{raw}' (expected 'Name,Department Group')"
                )
                continue
            name, dept = parts[0].strip(), parts[1].strip()
            if not name or not dept:
                errors.append(
                    f"Format Error: Line {idx}: '{raw}' (name and department must not be empty)"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for u in users:
            if u.name in seen:
                duplicates.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Manages the prize library and lottery sequence."""

    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Prize name must not be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be a positive integer")
        if any(p.name == name.strip() for p in self.prizes):
            raise ValueError(f"Prize '{name.strip()}' already exists")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        prize = Prize(name.strip(), qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def get_prize(self, index: int) -> Optional[Prize]:
        if 0 <= index < len(self.prizes):
            return self.prizes[index]
        return None

    def set_avoid_groups(self, index: int, groups: List[str]) -> bool:
        prize = self.get_prize(index)
        if prize is None:
            return False
        prize.avoid_groups = []
        for g in groups:
            g = str(g).strip()
            if g and g not in prize.avoid_groups:
                prize.avoid_groups.append(g)
        return True

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

from __future__ import annotations

import random
from typing import Iterable, List, Optional

from lottery_system.models import Prize, User, normalize

SPECIAL_PERSON = "Li Bin"
PROTECTED_GROUP = "Cheng Ru group"
TRIGGER_PERSONS = ["Chen Qixian", "Jin Duo"]
TARGET_PERSON = "Ma Jiaqi"


def is_protected_group(department: str) -> bool:
    return normalize(department) in ("cheng ru group", "cheng ru")


class LotteryManager:
    """Applies the lottery rules and draws winners."""

    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.special_person_name = SPECIAL_PERSON
        self.protected_group_name = PROTECTED_GROUP
        self.trigger_person_names = list(TRIGGER_PERSONS)
        self.target_person_name = TARGET_PERSON

    # ---- winner tracking -------------------------------------------------
    def mark_winner(self, user: User) -> None:
        self.winner_names.add(user.name)

    def is_winner(self, user: User) -> bool:
        return user.name in self.winner_names

    # ---- rules -----------------------------------------------------------
    def is_fallback_active(self, prize: Prize) -> bool:
        return (
            normalize(prize.associated_drawer) == normalize(self.target_person_name)
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        base = [u for u in self.users if u.name not in self.winner_names]
        if self.is_fallback_active(prize):
            protected = [u for u in base if is_protected_group(u.department)]
            if protected:
                return protected
        if normalize(prize.associated_drawer) == normalize(self.special_person_name):
            return base
        avoid = {normalize(g) for g in prize.avoid_groups}
        return [u for u in base if normalize(u.department) not in avoid]

    def draw_winner(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        winner = random.choice(pool)
        self.mark_winner(winner)
        return winner

    def record_round_result(self, prize: Prize, winners: Iterable[User]) -> None:
        drawer = normalize(prize.associated_drawer)
        if not drawer:
            return
        if drawer in {normalize(p) for p in self.trigger_person_names}:
            contains = any(is_protected_group(u.department) for u in winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        else:
            self.last_two_draw_results = []

"""Result management and export."""

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User


class Winner:
    """A winning record."""

    def __init__(self, user: User, prize: Prize, draw_order: int):
        self.user = user
        self.prize = prize
        self.draw_order = draw_order
        self.win_time = datetime.now()

    def __repr__(self) -> str:
        return f"Winner(user={self.user!r}, prize={self.prize.name!r}, order={self.draw_order})"


class ResultManager:
    """Keeps the winner list and exports it."""

    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> bool:
        if any(w.user.name == user.name for w in self.winners):
            return False
        self.winners.append(Winner(user, prize, len(self.winners) + 1))
        return True

    def count_for_prize(self, prize) -> int:
        return sum(1 for w in self.winners if w.prize.name == prize.name)

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {p.name: [] for p in prizes}
        for w in self.winners:
            if w.prize.name in result:
                result[w.prize.name].append(w)
        return result

    def export_to_csv(self, prizes):
        grouped = self.get_winners_by_prize(prizes)
        rows = [w for p in prizes for w in grouped.get(p.name, [])]
        # include winners whose prize may not be in the list anymore
        listed = set(id(w) for w in rows)
        rows += [w for w in self.winners if id(w) not in listed]
        if not rows:
            return False, "No winners to export."
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize"])
                for w in rows:
                    writer.writerow([w.user.name, w.user.department, w.prize.name])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from __future__ import annotations

import sys

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class ExitProgram(Exception):
    pass


def ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        raise ExitProgram()


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryApp:
    def __init__(self):
        self.status = "idle"
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None

    # ---------------- list --------------------
    def set_users(self, users):
        self.users = users
        self.lottery = None
        self.result_manager = ResultManager()
        print(f"{len(users)} valid participants loaded.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        valid, dups = self.list_manager.validate_users(users)
        for e in errors:
            print("  " + e)
        self.set_users(valid)

    def load_custom(self):
        path = ask("Enter path of txt list file: ")
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            print("Upload failed.")
            return
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Validation issues detected:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            while True:
                c = ask("Remove invalid data and continue (r) or terminate upload (t)? ").lower()
                if c == "r":
                    break
                if c == "t":
                    print("Upload terminated.")
                    return
        self.set_users(valid)

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for g, c in info.items():
            print(f"  {g}: {c} people")

    # ---------------- prizes --------------------
    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def load_presets(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prize library loaded.")

    def choose_groups(self):
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if not groups:
            text = ask("Enter avoidance groups (comma separated, empty for none): ")
            return [g.strip() for g in text.replace("\uff0c", ",").split(",") if g.strip()]
        for i, g in enumerate(groups, 1):
            print(f"  {i}. {g}")
        text = ask("Select avoidance groups by number (comma separated, empty for none): ")
        result = []
        for part in text.replace("\uff0c", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                result.append(groups[int(part) - 1])
            else:
                print(f"  Ignored invalid selection: {part}")
        return result

    def add_prize(self):
        name = ask("Prize name: ")
        qty = ask("Quantity: ")
        drawer = ask("Associated lottery person (empty for none): ") or None
        groups = self.choose_groups()
        try:
            self.prize_manager.add_prize(name, qty, groups, drawer)
            print("Prize added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def configure_avoidance(self):
        self.view_prizes()
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            return
        idx = ask("Select prize number: ")
        if not idx.isdigit() or not (1 <= int(idx) <= len(prizes)):
            print("Invalid selection.")
            return
        groups = self.choose_groups()
        self.prize_manager.set_avoid_groups(int(idx) - 1, groups)
        print("Avoidance groups updated.")

    def reorder(self):
        self.view_prizes()
        if not self.prize_manager.get_prizes():
            return
        text = ask("Enter new order as serial numbers (e.g. 3,1,2): ")
        parts = [x.strip() for x in text.replace("\uff0c", ",").replace(" ", ",").split(",") if x.strip()]
        if self.prize_manager.reorder_prizes(parts):
            print("Prize order updated.")
            self.view_prizes()
        else:
            print("Invalid order.")

    # ---------------- lottery --------------------
    def start_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)
        for w in self.result_manager.winners:
            self.lottery.mark_winner(w.user)
        self.status = "running"
        for rnd, prize in enumerate(prizes, 1):
            remaining = prize.quantity - self.result_manager.count_for_prize(prize)
            if remaining <= 0:
                continue
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity} (remaining {remaining})")
            print(f"Rules: {prize.rule_description()}")
            if self.lottery.is_fallback_active(prize):
                print("Fallback active: prize pool limited to Cheng Ru group members.")
            round_winners = []
            terminated = False
            for i in range(remaining):
                while True:
                    cmd = ask("[Enter]=draw | pause | terminate: ").lower()
                    if cmd == "pause":
                        print("Lottery paused. Type 'continue' to resume or 'terminate' to skip round.")
                        while True:
                            c = ask("> ").lower()
                            if c in ("continue", "terminate"):
                                break
                        cmd = c
                    if cmd == "terminate":
                        terminated = True
                        break
                    if cmd in ("", "continue", "draw"):
                        break
                    print("Unknown command.")
                if terminated:
                    print("Current round terminated.")
                    break
                print(f"Currently drawing the {ordinal(i + 1)} winner...")
                user = self.lottery.draw_winner(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                self.result_manager.add_winner(user, prize)
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            self.lottery.record_round_result(prize, round_winners)
            print(f"-- Round {rnd} results --")
            if round_winners:
                for n, u in enumerate(round_winners, 1):
                    print(f"  {n}. {u.name} ({u.department})")
            else:
                print("  No winners this round.")
        self.status = "finished"
        print("\nAll rounds completed.")
        self.show_all()

    def show_all(self):
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("Complete winner list:")
        any_w = False
        for pname, ws in grouped.items():
            for w in ws:
                any_w = True
                print(f"  {w.user.name}, {w.user.department}, {pname}")
        if not any_w:
            print("  (none)")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.view_groups),
            "4": ("Load preset prize library", self.load_presets),
            "5": ("Add custom prize", self.add_prize),
            "6": ("Configure avoidance groups", self.configure_avoidance),
            "7": ("Reorder prizes", self.reorder),
            "8": ("View prizes", self.view_prizes),
            "9": ("Start lottery", self.start_lottery),
            "10": ("View all winners", self.show_all),
            "11": ("Export results", self.export),
        }
        try:
            while True:
                print("\n===== Team Building Smart Lottery System =====")
                for k, (label, _) in actions.items():
                    print(f"{k}. {label}")
                print("0. Exit")
                c = ask("Choose: ")
                if c == "0":
                    break
                if c in actions:
                    actions[c][1]()
                else:
                    print("Invalid option.")
        except (ExitProgram, KeyboardInterrupt):
            pass
        print("Goodbye.")


def main():
    LotteryApp().run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

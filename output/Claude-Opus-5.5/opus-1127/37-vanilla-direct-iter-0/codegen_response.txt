"""Core data models for the lottery system."""

from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import List, Optional


@dataclass(frozen=True)
class User:
    """A participant of the lottery."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = [str(g).strip() for g in self.avoid_groups if str(g).strip()]
        if self.associated_drawer is not None:
            drawer = str(self.associated_drawer).strip()
            self.associated_drawer = drawer or None

    def __str__(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "Not specified"
        return (f"{self.name} | Quantity: {self.quantity} | "
                f"Lottery person: {drawer} | Avoidance groups: {avoid}")


_LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
    ("sixth", 6),
]

UNRANKED_LEVEL = 99


def prize_level(name: str) -> int:
    """Return the level of a prize: smaller numbers mean higher prizes.

    Prizes whose names do not contain a recognised level keyword are
    considered lower than all ranked prizes.
    """
    lower = (name or "").lower()
    for keyword, level in _LEVEL_KEYWORDS:
        if re.search(rf"\b{keyword}\b", lower):
            return level
    return UNRANKED_LEVEL

import os
from typing import Dict, List, Tuple

from lottery_system.models import User

DEFAULT_LIST_TEXT = """\

Li Si,Technology Department
Wang Wu,Technology Department
Zhao Liu,Technology Department
Sun Qi,Product Department
Zhou Ba,Product Department
Wu Jiu,Product Department
Zheng Shi,Product Department
Feng Yi,Marketing Department
Chen Er,Marketing Department
Chu San,Marketing Department
Wei Si,Human Resources Department
Jiang Wu,Human Resources Department
Shen Liu,Finance Department
Han Qi,Finance Department
Yang Ba,Cheng Ru Group
Zhu Jiu,Cheng Ru Group
Qin Shi,Cheng Ru Group
You Yi,Cheng Ru Group
Xu Er,Cheng Ru Group

class ListManager:
    """Loads and validates participant lists."""

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        """Load the built-in default team building list."""
        return self._parse_lines(DEFAULT_LIST_TEXT.splitlines())

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        """Load a custom txt list with lines in the form ``Name,Department``."""
        if path is None or not str(path).strip():
            return [], ["No file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as fh:
                lines = fh.read().splitlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as fh:
                    lines = fh.read().splitlines()
            except Exception as exc:  # pragma: no cover - defensive
                return [], [f"Unable to read file {path}: {exc}"]
        except OSError as exc:
            return [], [f"Unable to read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for number, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line:
                continue
            normalized = line.replace("\uff0c", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {number}: '{line}' "
                    f"(expected 'Name,Department Group')"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        """Split users into valid users and duplicates (by name).

        The first occurrence of a name is kept as valid; later occurrences
        are reported as duplicates.
        """
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        """Return a mapping department -> number of members."""
        info: Dict[str, int] = {}
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Manages the prizes and their lottery order."""

    def __init__(self) -> None:
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, quantity, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, quantity, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def _find(self, name: str) -> Optional[Prize]:
        for prize in self._prizes:
            if prize.name == name:
                return prize
        return None

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = (name or "").strip() if isinstance(name, str) else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        if isinstance(quantity, bool):
            raise ValueError("Prize quantity must be a positive integer")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Prize quantity must be a positive integer")
        if qty <= 0 or (isinstance(quantity, float) and quantity != qty):
            raise ValueError("Prize quantity must be a positive integer")
        if self._find(name) is not None:
            raise ValueError(f"Prize '{name}' already exists")
        prize = Prize(name, qty, list(avoid_groups or []), associated_drawer)
        self._prizes.append(prize)
        return prize

    def set_avoid_groups(self, index: int, groups) -> bool:
        """Set avoidance groups of the prize at 0-based ``index``."""
        if not 0 <= index < len(self._prizes):
            return False
        self._prizes[index].avoid_groups = [g.strip() for g in groups if g and g.strip()]
        return True

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self._prizes)
        try:
            positions = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(positions) != n or sorted(positions) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[p - 1] for p in positions]
        return True

"""Lottery rule engine."""

from __future__ import annotations

import random
from typing import List, Optional

from lottery_system.models import Prize, User

CHENG_RU_GROUP = "Cheng Ru Group"
SPECIAL_DRAWER = "Li Bin"
FALLBACK_WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"


def _norm(text) -> str:
    return " ".join(str(text or "").strip().lower().split())


def is_cheng_ru_member(user: User) -> bool:
    return "cheng ru" in _norm(user.department)


class LotteryManager:
    """Applies the lottery rules and performs the draws."""

    def __init__(self, users, result_manager=None, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users)
        self.result_manager = result_manager
        self.rng = rng or random.SystemRandom()
        self.winners: List[User] = []
        self.winner_names = set()
        # Whether each of the last consecutive rounds by the watched drawers
        # (Chen Qixian, Jin Duo) contained Cheng Ru group members.
        self.last_two_draw_results: List[bool] = []
        self._last_two_drawers: List[str] = []

    # ------------------------------------------------------------------ rules
    def is_fallback_triggered(self, prize: Prize) -> bool:
        if _norm(prize.associated_drawer) != _norm(FALLBACK_TARGET_DRAWER):
            return False
        recent = self.last_two_draw_results[-2:]
        return len(recent) == 2 and not any(recent)

    def _not_won(self) -> List[User]:
        return [u for u in self.users if u.name not in self.winner_names]

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = self._not_won()
        if self.is_fallback_triggered(prize):
            pool = [u for u in candidates if is_cheng_ru_member(u)]
            if pool:
                return pool
            # No Cheng Ru member left: fall back to normal rules.
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return candidates
        avoid = {_norm(g) for g in prize.avoid_groups}
        return [u for u in candidates if _norm(u.department) not in avoid]

    def describe_rules(self, prize: Prize) -> str:
        parts = ["Previous winners are excluded"]
        if self.is_fallback_triggered(prize):
            parts.append(
                f"FALLBACK ACTIVE: the previous two lottery persons "
                f"({', '.join(FALLBACK_WATCHED_DRAWERS)}) drew no {CHENG_RU_GROUP} member, "
                f"so the prize pool is limited to {CHENG_RU_GROUP} members "
                f"(avoidance groups ignored)"
            )
        elif _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            parts.append(f"Special lottery person {SPECIAL_DRAWER}: avoidance groups are not applied")
        elif prize.avoid_groups:
            parts.append("Avoidance groups excluded: " + ", ".join(prize.avoid_groups))
        else:
            parts.append("No avoidance groups")
        return "; ".join(parts)

    # ------------------------------------------------------------------ draws
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self._register_winner(user, prize)
        return user

    def _register_winner(self, user: User, prize: Prize) -> None:
        if user.name in self.winner_names:
            raise RuntimeError(f"{user.name} has already won a prize")
        self.winners.append(user)
        self.winner_names.add(user.name)
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)

    def finish_round(self, prize: Prize, results) -> None:
        """Update the fallback tracking state after a round."""
        drawer = _norm(prize.associated_drawer)
        watched = [_norm(d) for d in FALLBACK_WATCHED_DRAWERS]
        if drawer in watched:
            contains = any(is_cheng_ru_member(u) for u in results)
            self.last_two_draw_results.append(contains)
            self._last_two_drawers.append(prize.associated_drawer or "")
            self.last_two_draw_results = self.last_two_draw_results[-2:]
            self._last_two_drawers = self._last_two_drawers[-2:]
        else:
            self.last_two_draw_results = []
            self._last_two_drawers = []

    def draw_prize(self, prize: Prize) -> List[User]:
        """Draw all winners of a prize at once."""
        results: List[User] = []
        for _ in range(prize.quantity):
            user = self.draw_one(prize)
            if user is None:
                break
            results.append(user)
        self.finish_round(prize, results)
        return results

"""Result management: storage, summary and export."""

from __future__ import annotations

import csv
import os
from dataclasses import dataclass
from datetime import datetime
from typing import Dict, List

from lottery_system.models import Prize, User, prize_level

TIME_FORMAT = "%Y-%m-%d %H:%M:%S"


@dataclass
class Winner:
    user: User
    prize_name: str
    time: datetime
    sequence: int

    @property
    def time_str(self) -> str:
        return self.time.strftime(TIME_FORMAT)


def _prize_name(prize) -> str:
    return prize.name if isinstance(prize, Prize) else str(prize)


class ResultManager:
    def __init__(self) -> None:
        self.winners: List[Winner] = []
        self._names = set()
        self._seq = 0

    def add_winner(self, user: User, prize):
        """Record a winner; returns the Winner, or None if already a winner."""
        if user.name in self._names:
            return None
        self._seq += 1
        winner = Winner(user, _prize_name(prize), datetime.now(), self._seq)
        self.winners.append(winner)
        self._names.add(user.name)
        return winner

    def has_won(self, name: str) -> bool:
        return name in self._names

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        names = []
        for p in prizes or []:
            n = _prize_name(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        ordered = sorted(enumerate(names), key=lambda t: (prize_level(t[1]), t[0]))
        result: Dict[str, List[Winner]] = {}
        for _, n in ordered:
            lst = [w for w in self.winners if w.prize_name == n]
            lst.sort(key=lambda w: (w.time, w.sequence))
            result[n] = lst
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8", newline="") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in grouped.items():
                    for w in winners:
                        writer.writerow([w.user.name, w.user.department, prize_name, w.time_str])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""

from __future__ import annotations

from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryCLI:
    def __init__(self, input_func=input, output_func=print):
        self._input = input_func
        self.out = output_func
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery = None
        self.completed = set()

    def ask(self, prompt: str) -> str:
        return self._input(prompt).strip()

    # ------------------------------------------------------------- main menu
    def run(self) -> None:
        self.out("=" * 60)
        self.out(" Company Team Building Event Smart Lottery System")
        self.out("=" * 60)
        try:
            while True:
                self.out("\nMain Menu:")
                self.out(" 1. List management")
                self.out(" 2. Prize configuration")
                self.out(" 3. Start lottery")
                self.out(" 4. View results summary")
                self.out(" 5. Export results")
                self.out(" 0. Exit")
                choice = self.ask("Select: ")
                if choice == "1":
                    self.list_menu()
                elif choice == "2":
                    self.prize_menu()
                elif choice == "3":
                    self.start_lottery()
                elif choice == "4":
                    self.show_summary()
                elif choice == "5":
                    self.export_results()
                elif choice == "0":
                    self.out("Goodbye!")
                    return
                else:
                    self.out("Invalid choice.")
        except (EOFError, KeyboardInterrupt):
            self.out("\nExiting.")

    # ------------------------------------------------------------- lists
    def list_menu(self) -> None:
        while True:
            self.out("\nList Management:")
            self.out(" 1. Load built-in default list")
            self.out(" 2. Upload custom txt list (each line: Name,Department Group)")
            self.out(" 3. View group information")
            self.out(" 4. View all members")
            self.out(" 0. Back")
            choice = self.ask("Select: ")
            if choice == "1":
                users, errors = self.list_manager.load_default_list()
                self._accept_list(users, errors)
            elif choice == "2":
                path = self.ask("Enter txt file path: ")
                users, errors = self.list_manager.load_custom_list(path)
                self._accept_list(users, errors)
            elif choice == "3":
                self.show_groups()
            elif choice == "4":
                if not self.users:
                    self.out("No list loaded.")
                for i, u in enumerate(self.users, 1):
                    self.out(f" {i}. {u.name} ({u.department})")
            elif choice == "0":
                return
            else:
                self.out("Invalid choice.")

    def _accept_list(self, users, errors) -> None:
        valid, duplicates = self.list_manager.validate_users(users)
        if errors or duplicates:
            self.out("Problems detected in the list:")
            for e in errors:
                self.out("  " + e)
            for d in duplicates:
                self.out(f"  Duplicate name: {d.name} ({d.department})")
            if not users and errors:
                self.out("Upload terminated: no data loaded.")
                return
            while True:
                c = self.ask("Enter 'r' to remove invalid data and continue, 't' to terminate upload: ").lower()
                if c == "r":
                    break
                if c == "t":
                    self.out("Upload terminated.")
                    return
                self.out("Invalid input.")
        if not valid:
            self.out("No valid members in the list.")
            return
        if self.result_manager.winners:
            c = self.ask("Loading a new list clears existing results. Continue? (y/n): ").lower()
            if c != "y":
                self.out("Cancelled.")
                return
        self.users = valid
        self.lottery = None
        self.result_manager = ResultManager()
        self.completed = set()
        self.out(f"Loaded {len(valid)} valid members.")
        self.show_groups()

    def show_groups(self) -> None:
        if not self.users:
            self.out("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        self.out("Valid groups:")
        for i, (dept, cnt) in enumerate(info.items(), 1):
            self.out(f" {i}. {dept}: {cnt} people")

    # ------------------------------------------------------------- prizes
    def prize_menu(self) -> None:
        while True:
            self.out("\nPrize Configuration:")
            self.out(" 1. Load preset prize library")
            self.out(" 2. Add custom prize")
            self.out(" 3. Set avoidance groups for a prize")
            self.out(" 4. View prizes (lottery order)")
            self.out(" 5. Reorder prizes")
            self.out(" 0. Back")
            choice = self.ask("Select: ")
            if choice == "1":
                errors = self.prize_manager.load_default_prizes()
                for e in errors:
                    self.out("  " + e)
                self.out("Preset prizes loaded.")
                self.show_prizes()
            elif choice == "2":
                self.add_custom_prize()
            elif choice == "3":
                self.set_avoid_groups()
            elif choice == "4":
                self.show_prizes()
            elif choice == "5":
                self.reorder()
            elif choice == "0":
                return
            else:
                self.out("Invalid choice.")

    def _select_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if not groups:
            raw = self.ask("No list loaded. Enter avoidance group names separated by commas (blank for none): ")
            return [g.strip() for g in raw.replace("\uff0c", ",").split(",") if g.strip()]
        self.out("Available groups:")
        for i, g in enumerate(groups, 1):
            self.out(f" {i}. {g}")
        while True:
            raw = self.ask("Select avoidance groups by number, comma separated (blank for none): ")
            if not raw:
                return []
            try:
                idxs = [int(x) for x in raw.replace("\uff0c", ",").split(",") if x.strip()]
            except ValueError:
                self.out("Invalid input.")
                continue
            if all(1 <= i <= len(groups) for i in idxs):
                result = []
                for i in idxs:
                    if groups[i - 1] not in result:
                        result.append(groups[i - 1])
                return result
            self.out("Number out of range.")

    def add_custom_prize(self) -> None:
        name = self.ask("Prize name: ")
        qty = self.ask("Quantity: ")
        drawer = self.ask("Associated lottery person (blank for none): ") or None
        groups = self._select_groups()
        try:
            prize = self.prize_manager.add_prize(name, qty, groups, drawer)
        except ValueError as exc:
            self.out(f"Error: {exc}")
            return
        self.out(f"Added: {prize}")

    def set_avoid_groups(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("No prizes configured.")
            return
        self.show_prizes()
        raw = self.ask("Prize number: ")
        try:
            idx = int(raw) - 1
        except ValueError:
            self.out("Invalid input.")
            return
        if not 0 <= idx < len(prizes):
            self.out("Number out of range.")
            return
        groups = self._select_groups()
        self.prize_manager.set_avoid_groups(idx, groups)
        self.out(f"Updated: {self.prize_manager.get_prizes()[idx]}")

    def show_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("No prizes configured.")
            return
        self.out("Prizes (lottery order):")
        for i, p in enumerate(prizes, 1):
            status = " [drawn]" if p.name in self.completed else ""
            self.out(f" {i}. {p}{status}")

    def reorder(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if len(prizes) < 2:
            self.out("Need at least two prizes to reorder.")
            return
        self.show_prizes()
        raw = self.ask("Enter new order as current serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("\uff0c", ",").replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            self.out("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            self.out("Prizes reordered.")
            self.show_prizes()
        else:
            self.out("Invalid order: must contain each serial number exactly once.")

    # ------------------------------------------------------------- lottery
    def start_lottery(self) -> None:
        if not self.users:
            self.out("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            self.out("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users, self.result_manager)
        pending = [p for p in prizes if p.name not in self.completed]
        if not pending:
            self.out("All rounds have been completed.")
            return
        total = len(pending)
        for i, prize in enumerate(pending, 1):
            self._run_round(prize, i, total)
            self.completed.add(prize.name)
        self.out("\nAll rounds completed!")
        self.show_summary()

    def _run_round(self, prize, idx, total) -> None:
        self.out("\n" + "=" * 60)
        self.out(f"Round {idx}/{total}: {prize.name}")
        self.out(f"Quantity: {prize.quantity}")
        self.out(f"Lottery person: {prize.associated_drawer or 'Not specified'}")
        self.out(f"Rules: {self.lottery.describe_rules(prize)}")
        self.out("Commands: [Enter]=draw next, 'auto'=draw all remaining, 'pause', 'continue', 'terminate'")
        self.out("=" * 60)
        results: List[User] = []
        auto = False
        terminated = False
        while len(results) < prize.quantity and not terminated:
            if not auto:
                cmd = self.ask(f"[{prize.name}] command: ").lower()
                if cmd == "pause":
                    self.out("Lottery paused.")
                    while True:
                        c = self.ask("Paused. Type 'continue' to resume or 'terminate' to end this round: ").lower()
                        if c == "continue":
                            self.out("Lottery resumed.")
                            break
                        if c == "terminate":
                            terminated = True
                            break
                        self.out("Invalid command while paused.")
                    continue
                if cmd == "terminate":
                    terminated = True
                    continue
                if cmd == "continue":
                    self.out("Lottery is not paused.")
                    continue
                if cmd == "auto":
                    auto = True
                elif cmd not in ("", "draw"):
                    self.out("Invalid command.")
                    continue
            self.out(f"Currently drawing the {ordinal(len(results) + 1)} winner...")
            user = self.lottery.draw_one(prize)
            if user is None:
                self.out("No eligible participants left for this prize.")
                break
            results.append(user)
            self.out(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
        if terminated:
            self.out(f"Round '{prize.name}' terminated; skipping to next round.")
        self.lottery.finish_round(prize, results)
        self.out(f"\nResults of round '{prize.name}':")
        if not results:
            self.out("  No winners.")
        for n, u in enumerate(results, 1):
            self.out(f"  {n}. {u.name} ({u.department})")

    # ------------------------------------------------------------- results
    def show_summary(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not self.result_manager.winners:
            self.out("No winners yet.")
            return
        self.out("\n===== Complete Winner List =====")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            self.out(f"{prize_name}:")
            for n, w in enumerate(winners, 1):
                self.out(f"  {n}. {w.user.name} ({w.user.department}) - {prize_name} - {w.time_str}")

    def export_results(self) -> None:
        if not self.result_manager.winners:
            self.out("No winners to export.")
            return
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        self.out(msg)

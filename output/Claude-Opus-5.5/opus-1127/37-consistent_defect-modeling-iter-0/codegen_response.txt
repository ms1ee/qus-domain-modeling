"""Core data models for the lottery system."""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    """A lottery participant."""

    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


def infer_prize_level(name: str) -> int:
    """Infer a prize level from its name (smaller number = higher level)."""
    lowered = (name or "").lower()
    if "grand" in lowered or "special" in lowered:
        return 0
    if "first" in lowered or "1st" in lowered:
        return 1
    if "second" in lowered or "2nd" in lowered:
        return 2
    if "third" in lowered or "3rd" in lowered:
        return 3
    if "fourth" in lowered or "4th" in lowered:
        return 4
    if "fifth" in lowered or "5th" in lowered:
        return 5
    return 99


@dataclass
class Prize:
    """A prize to be drawn in one lottery round."""

    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None
    level: Optional[int] = None

    def __post_init__(self) -> None:
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if self.level is None:
            self.level = infer_prize_level(self.name)

    def describe_rules(self) -> str:
        parts = []
        drawer = self.associated_drawer or "Not specified"
        parts.append(f"Lottery person: {drawer}")
        if self.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("Avoidance groups: None")
        parts.append("Previous winners are excluded")
        if drawer.strip().lower() == "li bin":
            parts.append("Special rule: Li Bin draws ignore avoidance groups")
        if drawer.strip().lower() == "ma jiaqi":
            parts.append(
                "Fallback rule: limited to Cheng Ru group if Chen Qixian and "
                "Jin Duo both drew no Cheng Ru group member"
            )
        return "; ".join(parts)

    def __str__(self) -> str:
        return f"{self.name} x{self.quantity}"

DEFAULT_LIST_LINES = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Wang Wu,Technology Department",
    "Zhao Liu,Technology Department",
    "Sun Qi,Product Department",
    "Zhou Ba,Product Department",
    "Wu Jiu,Product Department",
    "Zheng Shi,Marketing Department",
    "Feng Yi,Marketing Department",

    "Shen Yue,Cheng Ru Group",
    "Han Xue,Cheng Ru Group",
    "Yang Fan,Cheng Ru Group",
    "Qin Feng,Finance Department",
    "You Lin,Finance Department",
    "Xu Ning,Operations Department",
    "He Tian,Operations Department",

_SEPARATORS = re.compile(r"[,;，；]")


class ListManager:
    """Loads, validates and analyses participant lists."""

    def __init__(self) -> None:
        self.users: List[User] = []

    # ------------------------------------------------------------------ loading
    def _parse_lines(self, lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff")
            if not line:
                continue
            parts = [p.strip() for p in _SEPARATORS.split(line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', "
                    f"got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not str(path).strip():
            return [], ["File Error: no file path provided"]
        path = str(path).strip().strip('"').strip("'")
        if not os.path.isfile(path):
            return [], [f"File Error: file not found '{path}'"]
        if not path.lower().endswith(".txt"):
            return [], [f"File Error: only .txt files are supported '{path}'"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            try:
                with open(path, "r", encoding="utf-8-sig") as fh:
                    lines = fh.readlines()
            except UnicodeDecodeError:
                with open(path, "r", encoding="gbk", errors="replace") as fh:
                    lines = fh.readlines()
        except OSError as exc:
            return [], [f"File Error: cannot read '{path}': {exc}"]
        return self._parse_lines(lines)

    # --------------------------------------------------------------- validation
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users or []:
            key = user.name.strip()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: "OrderedDict[str, int]" = OrderedDict()
        for user in users or []:
            info[user.department] = info.get(user.department, 0) + 1
        return dict(info)

"""Prize library management."""

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 5),
    ("Second Prize - Shopping Cards", 3),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    """Manages the prize library and lottery round order."""

    def __init__(self) -> None:
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        existing = {p.name for p in self.prizes}
        for name, qty in DEFAULT_PRIZES:
            if name in existing:
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            self.prizes.append(Prize(name, qty, [], None))
            existing.add(name)
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Prize quantity must be an integer")
        if qty <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        groups = [g.strip() for g in (avoid_groups or []) if g and str(g).strip()]
        # remove duplicates, keep order
        groups = list(dict.fromkeys(groups))
        drawer: Optional[str] = None
        if associated_drawer is not None and str(associated_drawer).strip():
            drawer = str(associated_drawer).strip()
        prize = Prize(str(name).strip(), qty, groups, drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Winner records, summaries and export."""

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List

from lottery_system.models import User, infer_prize_level


class Winner:
    """A single winning record."""

    def __init__(self, user: User, prize_name: str, order: int, time=None):
        self.user = user
        self.prize_name = prize_name
        self.order = order
        self.time = time or datetime.now()

    @property
    def time_str(self) -> str:
        return self.time.strftime("%Y-%m-%d %H:%M:%S")

    def __repr__(self) -> str:
        return f"Winner({self.user.name}, {self.user.department}, {self.prize_name})"


class ResultManager:
    """Maintains the winner list."""

    def __init__(self) -> None:
        self.winners: List[Winner] = []

    @property
    def total_winners(self) -> int:
        return len(self.winners)

    def has_won(self, user) -> bool:
        return any(w.user.name == user.name for w in self.winners)

    def add_winner(self, user, prize):
        prize_name = getattr(prize, "name", prize)
        if self.has_won(user):
            return False
        self.winners.append(Winner(user, str(prize_name), len(self.winners) + 1))
        return True

    def clear(self) -> None:
        self.winners = []

    def _sorted_prizes(self, prizes):
        indexed = list(enumerate(prizes or []))
        indexed.sort(
            key=lambda t: (
                t[1].level if getattr(t[1], "level", None) is not None
                else infer_prize_level(t[1].name),
                t[0],
            )
        )
        return [p for _, p in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for prize in self._sorted_prizes(prizes):
            result.setdefault(prize.name, [])
        for prize_name in result:
            items = [w for w in self.winners if w.prize_name == prize_name]
            items.sort(key=lambda w: (w.time, w.order))
            result[prize_name] = items
        return result

    def export_to_csv(self, prizes):
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        grouped = self.get_winners_by_prize(prizes)
        listed = {w for ws in grouped.values() for w in ws}
        extras = [w for w in self.winners if w not in listed]
        try:
            with open(path, "w", encoding="utf-8", newline="") as fh:
                writer = csv.writer(fh)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for ws in grouped.values():
                    for w in ws:
                        writer.writerow([w.user.name, w.user.department, w.prize_name, w.time_str])
                for w in extras:
                    writer.writerow([w.user.name, w.user.department, w.prize_name, w.time_str])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and draw execution."""

from __future__ import annotations

import random
from typing import List, Optional

from lottery_system.models import Prize, User

SPECIAL_PERSON = "li bin"
MONITORED_PERSONS = ("chen qixian", "jin duo")
TARGET_PERSON = "ma jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(text) -> str:
    return " ".join(str(text or "").strip().lower().split())


def is_protected_group(department) -> bool:
    key = _norm(department).replace(" ", "")
    if key.endswith("group"):
        key = key[: -len("group")]
    return key == "chengru"


class LotteryManager:
    """Applies the rule engine and draws winners."""

    def __init__(self, users, result_manager=None, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users or [])
        self.result_manager = result_manager
        self.winners: List[User] = []
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()

    # ------------------------------------------------------------ rule engine
    def _fallback_triggered(self) -> bool:
        if len(self.last_two_draw_results) < 2:
            return False
        return not any(self.last_two_draw_results[-2:])

    def _already_won(self, user) -> bool:
        if user.name in self.winner_names:
            return True
        if self.result_manager is not None and self.result_manager.has_won(user):
            return True
        return False

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        available = [u for u in self.users if not self._already_won(u)]
        drawer = _norm(prize.associated_drawer)

        if drawer == TARGET_PERSON and self._fallback_triggered():
            protected = [u for u in available if is_protected_group(u.department)]
            if protected:
                return protected

        if drawer == SPECIAL_PERSON:
            return available

        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        if not avoid:
            return available
        preferred = [u for u in available if _norm(u.department) not in avoid]
        if preferred:
            return preferred
        # Avoidance group members are only drawn after all other members.
        return available

    def is_fallback_active(self, prize: Prize) -> bool:
        return _norm(prize.associated_drawer) == TARGET_PERSON and self._fallback_triggered()

    # ---------------------------------------------------------------- drawing
    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.winners.append(user)
        self.winner_names.add(user.name)
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return user

    def finish_round(self, prize: Prize, round_winners) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in MONITORED_PERSONS:
            hit = any(is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(hit)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == TARGET_PERSON:
            self.last_two_draw_results = []

    def draw_round(self, prize: Prize) -> List[User]:
        result = []
        for _ in range(prize.quantity):
            user = self.draw_one(prize)
            if user is None:
                break
            result.append(user)
        self.finish_round(prize, result)
        return result

"""Command-line interface for the lottery system."""

from __future__ import annotations

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


def ask(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        return "exit"


class LotteryApp:
    def __init__(self) -> None:
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = LotteryManager([], self.result_manager)
        self.running = True

    # ---------------------------------------------------------------- list
    def _set_users(self, users) -> None:
        self.users = users
        self.result_manager.clear()
        self.lottery = LotteryManager(users, self.result_manager)
        print(f"{len(users)} valid participants loaded.")

    def load_default(self) -> None:
        users, errors = self.list_manager.load_default_list()
        for e in errors:
            print(e)
        valid, dups = self.list_manager.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name} ({d.department})")
        self._set_users(valid)

    def load_custom(self) -> None:
        path = ask("Enter txt file path: ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        valid, dups = self.list_manager.validate_users(users)
        if errors:
            print("Problems found:")
            for e in errors:
                print("  " + e)
        if dups:
            print("Duplicate names found:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
        if not users and errors:
            print("No valid data loaded.")
            return
        if errors or dups:
            choice = ask("Type 'remove' to remove invalid data and continue, "
                         "or 'terminate' to cancel upload: ").strip().lower()
            if choice not in ("remove", "r"):
                print("Upload terminated.")
                return
        self._set_users(valid)

    def view_groups(self) -> None:
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        print("Department groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    # ---------------------------------------------------------------- prizes
    def load_prizes(self) -> None:
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print(e)
        print("Preset prizes loaded.")

    def add_prize(self) -> None:
        name = ask("Prize name: ").strip()
        qty = ask("Quantity: ").strip()
        drawer = ask("Associated lottery person (empty for none): ").strip() or None
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid = []
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
            sel = ask("Avoidance group numbers, comma separated (empty for none): ").strip()
            for part in sel.replace("，", ",").split(","):
                part = part.strip()
                if not part:
                    continue
                if part.isdigit() and 1 <= int(part) <= len(groups):
                    avoid.append(groups[int(part) - 1])
                else:
                    print(f"Ignored invalid selection: {part}")
        else:
            sel = ask("Avoidance group names, comma separated (empty for none): ").strip()
            avoid = [p.strip() for p in sel.split(",") if p.strip()]
        try:
            prize = self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize added: {prize.name}")
        except ValueError as exc:
            print(f"Error: {exc}")

    def view_prizes(self) -> None:
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | {p.describe_rules()}")

    def reorder(self) -> None:
        self.view_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = ask("New order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.view_prizes()
        else:
            print("Invalid order, unchanged.")

    # ---------------------------------------------------------------- lottery
    def run_lottery(self) -> None:
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        print("Commands during draw: Enter=draw next, 'pause', 'continue', 'terminate'.")
        for rnd, prize in enumerate(prizes, 1):
            print(f"\n===== Round {rnd}: {prize.name} (Quantity: {prize.quantity}) =====")
            print(f"Rules: {prize.describe_rules()}")
            if self.lottery.is_fallback_active(prize):
                print("Fallback triggered: prize pool limited to Cheng Ru group members.")
            round_winners = []
            terminated = False
            for i in range(1, prize.quantity + 1):
                cmd = ask(f"[Enter] draw {ordinal(i)} winner > ").strip().lower()
                if cmd == "pause":
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to skip.")
                    while True:
                        cmd = ask("(paused) > ").strip().lower()
                        if cmd in ("continue", "terminate", "exit"):
                            break
                if cmd in ("terminate", "exit"):
                    print("Current round terminated.")
                    terminated = True
                    break
                print(f"Currently drawing the {ordinal(i)} winner...")
                user = self.lottery.draw_one(prize)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            self.lottery.finish_round(prize, round_winners)
            print(f"--- Round {rnd} results ---")
            if round_winners:
                for i, u in enumerate(round_winners, 1):
                    print(f"  {i}. {u.name} ({u.department}) - {prize.name}")
            else:
                print("  No winners.")
            if terminated and cmd == "exit":
                break
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self) -> None:
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("===== Complete Winner List =====")
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        for prize_name, winners in grouped.items():
            print(f"[{prize_name}]")
            if not winners:
                print("  (none)")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {prize_name} - {w.time_str}")

    def export(self) -> None:
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ---------------------------------------------------------------- main loop
    def run(self) -> None:
        actions = {
            "1": self.load_default,
            "2": self.load_custom,
            "3": self.view_groups,
            "4": self.load_prizes,
            "5": self.add_prize,
            "6": self.view_prizes,
            "7": self.reorder,
            "8": self.run_lottery,
            "9": self.show_summary,
            "10": self.export,
        }
        while self.running:
            print("\n==== Team Building Smart Lottery System ====")
            print("1. Load default list      2. Upload custom list")
            print("3. View groups            4. Load preset prizes")
            print("5. Add custom prize       6. View prizes")
            print("7. Reorder prizes         8. Start lottery")
            print("9. View winner summary    10. Export results")
            print("0. Exit")
            choice = ask("Select: ").strip().lower()
            if choice in ("0", "exit", "quit"):
                print("Goodbye!")
                break
            action = actions.get(choice)
            if action:
                action()
            else:
                print("Invalid option.")


def main() -> None:
    LotteryApp().run()


if __name__ == "__main__":
    main()

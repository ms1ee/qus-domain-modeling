GRID_DIM = MAP_SIZE // CELL_SIZE

    "initial_funds": 10000,

def load_config(path="config.json"):
    """Load a JSON configuration file and merge it with the defaults.

    Missing or malformed files silently fall back to the defaults.
    """
    cfg = dict(DEFAULT_CONFIG)
    if path and os.path.isfile(path):
        try:
            with open(path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
            if isinstance(data, dict):
                for key, value in data.items():
                    if key in DEFAULT_CONFIG:
                        cfg[key] = value
        except (OSError, ValueError) as exc:
            logger.warning("Could not read config file %s: %s", path, exc)
    return cfg

"""Grid based path planning.

Coordinate system: an 800x800 map split in 8x8 cells of 100x100 units.
Streets run along every line where ``x`` or ``y`` is a multiple of 100.
Restaurants and customers sit at cell centers (``x % 100 == 50`` and
``y % 100 == 50``).  Riders travel along streets and only leave them for the
final hop into a cell center (and the first hop back out of it).
"""

from config import CELL_SIZE, MAP_SIZE


def to_grid_center(value):
    """Convert a raw coordinate (0-800) to the center of its grid cell.

    ``center = (v // 100) * 100 + 50``; out of range values are clamped so
    that 800 maps to 750 and negative values map to 50.
    """
    v = int(value)
    v = max(0, min(MAP_SIZE - 1, v))
    return (v // CELL_SIZE) * CELL_SIZE + CELL_SIZE // 2


def to_grid_point(pos):
    """Convert an ``(x, y)`` pair to the corresponding cell center."""
    return (to_grid_center(pos[0]), to_grid_center(pos[1]))


def on_street(x, y):
    """Return True if the point lies on a street."""
    return x % CELL_SIZE == 0 or y % CELL_SIZE == 0


def manhattan(a, b):
    """Manhattan distance between two points."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


class PathPlanner:
    """Computes street-following, right-angled paths on the grid."""

    def __init__(self):
        self.last_special = False

    @staticmethod
    def _choose_line(a, b):
        """Choose a street line (multiple of 100) near ``a`` towards ``b``."""
        low = (int(a) // CELL_SIZE) * CELL_SIZE
        high = low + CELL_SIZE
        candidates = [c for c in (low, high) if 0 <= c <= MAP_SIZE]
        return min(candidates, key=lambda c: (abs(a - c) + abs(c - b), c))

    def street_route(self, a, b):
        """Route between two street points; returns waypoints (excl. ``a``)."""
        ax, ay = a
        bx, by = b
        if (ax, ay) == (bx, by):
            return []
        a_h = ay % CELL_SIZE == 0
        a_v = ax % CELL_SIZE == 0
        b_h = by % CELL_SIZE == 0
        b_v = bx % CELL_SIZE == 0
        if a_h and b_h and ay == by:
            return [(bx, by)]
        if a_v and b_v and ax == bx:
            return [(bx, by)]
        if a_v and b_h:
            return [(ax, by), (bx, by)]
        if a_h and b_v:
            return [(bx, ay), (bx, by)]
        if a_h and b_h:
            vx = self._choose_line(ax, bx)
            return [(vx, ay), (vx, by), (bx, by)]
        # both on vertical streets with different x
        hy = self._choose_line(ay, by)
        return [(ax, hy), (bx, hy), (bx, by)]

    def plan(self, start, target):
        """Plan a path from ``start`` to ``target``.

        Returns the list of waypoints (excluding ``start``) ending at
        ``target``.  Sets ``self.last_special`` when the special case
        "rider directly above/below the target" was handled.
        """
        x, y = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        self.last_special = False
        if (x, y) == (tx, ty):
            return []
        raw = []
        # Special case: start off-street inside the target's own cell.
        if (not on_street(x, y) and x // CELL_SIZE == tx // CELL_SIZE
                and y // CELL_SIZE == ty // CELL_SIZE):
            raw = [(tx, y), (tx, ty)]
            return self._dedupe((x, y), raw)
        # Step 1: leave an off-street position onto a horizontal street.
        if not on_street(x, y):
            if ty > y:
                ny = (y // CELL_SIZE + 1) * CELL_SIZE
            else:
                ny = (y // CELL_SIZE) * CELL_SIZE
            raw.append((x, ny))
            y = ny
        # Step 2: compute the entrance point of the target cell.
        if on_street(tx, ty):
            entrance = (tx, ty)
        else:
            below = (ty // CELL_SIZE) * CELL_SIZE
            above = below + CELL_SIZE
            ey = below if abs(y - below) <= abs(y - above) else above
            entrance = (tx, ey)
        # Special case: directly above / below the target on a non-street
        # column; the rider must side-step to a vertical street.
        if x == tx and x % CELL_SIZE != 0 and y != entrance[1]:
            self.last_special = True
        raw.extend(self.street_route((x, y), entrance))
        if entrance != (tx, ty):
            raw.append((tx, ty))
        return self._dedupe((int(start[0]), int(start[1])), raw)

    def plan_multi(self, start, destinations):
        """Plan a continuous path through several destinations in order."""
        path = []
        current = (int(start[0]), int(start[1]))
        special = False
        for dest in destinations:
            seg = self.plan(current, dest)
            special = special or self.last_special
            path.extend(seg)
            current = (int(dest[0]), int(dest[1]))
        self.last_special = special
        return path

    @staticmethod
    def _dedupe(start, points):
        """Remove consecutive duplicate points (and points equal to start)."""
        result = []
        prev = start
        for p in points:
            if p != prev:
                result.append(p)
                prev = p
        return result

    @staticmethod
    def path_length(start, path):
        """Total length of a path starting at ``start``."""
        total = 0
        prev = start
        for p in path:
            total += manhattan(prev, p)
            prev = p
        return total

def _validate_pos(pos, name):
    """Validate a coordinate pair and return it as an integer tuple."""
    try:
        if len(pos) != 2:
            raise ValueError
        return (int(pos[0]), int(pos[1]))
    except (TypeError, ValueError):
        raise ValueError("%s must be a pair of integers" % name)


def _next_order_id():
    """Return the next unique order id."""
    counter = globalvar.get_value("order_id_counter", 0) or 0
    counter += 1
    globalvar.set_value("order_id_counter", counter)
    return counter


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique order id.
        raw_res / raw_cus: coordinates as entered.
        res_pos / cus_pos: grid cell centers of restaurant / customer.
        status: "pending", "assigned", "picking_up", "delivering",
            "completed".
        rider_id: id of the assigned rider (None until assigned).
    """

    def __init__(self, res_pos, cus_pos):
        self.raw_res = _validate_pos(res_pos, "res_pos")
        self.raw_cus = _validate_pos(cus_pos, "cus_pos")
        self.res_pos = to_grid_point(self.raw_res)
        self.cus_pos = to_grid_point(self.raw_cus)
        self.id = _next_order_id()
        self.status = "pending"
        self.rider_id = None
        self.created_time = globalvar.get_value("time", 0) or 0
        self.assigned_time = None
        self.accepted_time = None
        self.picked_time = None
        self.completed_time = None
        self.time_taken = None
        self.distance = manhattan(self.res_pos, self.cus_pos)
        planner = PathPlanner()
        self.route_distance = planner.path_length(
            self.res_pos, planner.plan(self.res_pos, self.cus_pos))

    def assign_to(self, rider_id):

        self.rider_id = rider_id

            "created_time": self.created_time,
            "assigned_time": self.assigned_time,
            "picked_time": self.picked_time,
            "completed_time": self.completed_time,
            "time_taken": self.time_taken,
            "distance": self.distance,

        }

"""Rider model: movement, state machine and order handling."""

import logging

import globalvar
from config import DEFAULT_CONFIG
from path_planner import PathPlanner, to_grid_point

logger = logging.getLogger(__name__)

IDLE = 0
PICKING_UP = 1
DELIVERING = 2
STATE_NAMES = {IDLE: "idle", PICKING_UP: "picking up", DELIVERING: "delivering"}

SPAWN_POINTS = [
    (400, 400), (0, 0), (800, 800), (0, 800), (800, 0), (400, 0),
    (0, 400), (800, 400), (400, 800), (200, 200), (600, 600),
    (200, 600), (600, 200),
]


def _order_points(order):
    """Return ``(res_pos, cus_pos)`` of an Order or a 4-element sequence."""
    if hasattr(order, "res_pos") and hasattr(order, "cus_pos"):
        return tuple(order.res_pos), tuple(order.cus_pos)
    vals = list(order)
    return (to_grid_point(vals[0:2]), to_grid_point(vals[2:4]))


def _set_attr(order, name, value):
    """Set an attribute on an order object if possible."""
    try:
        setattr(order, name, value)
    except AttributeError:
        pass


class Rider:
    """A delivery rider.

    Attributes:
        rider_id: unique id.
        x, y: current position.
        speed: units moved per tick.
        state: 0 idle, 1 picking up, 2 delivering.
        orders: assigned orders (first one is the active one).
        special_case: True while handling a special path case.
        log: list of event dicts (keys include "event" and "order_id").
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = rider_id
        sx, sy = SPAWN_POINTS[int(rider_id) % len(SPAWN_POINTS)]
        self.x = sx if x is None else int(x)
        self.y = sy if y is None else int(y)
        if speed is None:
            cfg = globalvar.get_value("config") or DEFAULT_CONFIG
            speed = cfg.get("rider_speed", DEFAULT_CONFIG["rider_speed"])
        self.speed = max(1, int(speed))
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.path = []
        self.distance_travelled = 0
        self.completed_count = 0
        self._planner = PathPlanner()
        self._record("spawn", None)

    # ------------------------------------------------------------------
    def _record(self, event, order):
        """Append a log entry for a status change."""
        order_id = None
        if order is not None:
            order_id = getattr(order, "id", None)
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("time", 0) or 0,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })
        logger.debug("Rider %s: %s (order %s) at (%s, %s)",
                     self.rider_id, event, order_id, self.x, self.y)

    def _plan_to(self, target):
        """Plan a path from the current position to ``target``."""
        self.path = self._planner.plan((self.x, self.y), target)
        self.special_case = self._planner.last_special

    def _accept(self, order):
        """Idle -> picking up: accept the first order in the queue."""
        res, _ = _order_points(order)
        self.state = PICKING_UP
        now = globalvar.get_value("time", 0) or 0
        _set_attr(order, "status", "picking_up")
        _set_attr(order, "accepted_time", now)
        if getattr(order, "rider_id", "x") is None:
            _set_attr(order, "rider_id", self.rider_id)
        self._plan_to(res)
        self._record("accepted_order", order)

    def _on_arrival(self):
        """Handle reaching the end of the current path."""
        if not self.orders:
            return
        order = self.orders[0]
        res, cus = _order_points(order)
        now = globalvar.get_value("time", 0) or 0
        if self.state == PICKING_UP:
            if (self.x, self.y) != res:
                self._plan_to(res)
                return
            self.state = DELIVERING
            _set_attr(order, "status", "delivering")
            _set_attr(order, "picked_time", now)
            self._record("arrived_at_merchant", order)
            self._plan_to(cus)
        elif self.state == DELIVERING:
            if (self.x, self.y) != cus:
                self._plan_to(cus)
                return
            self._complete(order, now)

    def _complete(self, order, now):
        """Delivering -> idle: finish the active order."""
        _set_attr(order, "status", "completed")
        _set_attr(order, "completed_time", now)
        created = getattr(order, "created_time", None)
        if created is not None:
            _set_attr(order, "time_taken", now - created)
        self.orders.pop(0)
        self.state = IDLE
        self.special_case = False
        self.path = []
        self.completed_count += 1
        cfg = globalvar.get_value("config") or DEFAULT_CONFIG
        fee = cfg.get("delivery_fee", 0)
        if globalvar.get_value("funds") is not None and fee:
            globalvar.set_value("funds", globalvar.get_value("funds") + fee)
        self._record("delivered_to_customer", order)

    def _step(self):
        """Advance along the path by at most ``speed`` units."""
        budget = self.speed
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            dx = tx - self.x
            dy = ty - self.y
            dist = abs(dx) + abs(dy)
            if dist <= budget:
                self.x, self.y = tx, ty
                budget -= dist
                self.distance_travelled += dist
                self.path.pop(0)
            else:
                step = budget
                if dx != 0:
                    mv = min(step, abs(dx))
                    self.x += mv if dx > 0 else -mv
                    step -= mv
                if step > 0 and dy != 0:
                    mv = min(step, abs(dy))
                    self.y += mv if dy > 0 else -mv
                self.distance_travelled += budget
                budget = 0

    def move(self):
        """Advance the rider by one tick, performing state transitions."""
        if self.state == IDLE:
            if not self.orders:
                return
            self._accept(self.orders[0])
        if not self.orders:
            # Inconsistent state: busy without orders -> recover.
            self.state = IDLE
            self.path = []
            self.special_case = False
            return
        self._step()
        if not self.path:
            self._on_arrival()

    # ------------------------------------------------------------------
    @property
    def state_name(self):
        """Human readable state."""
        return STATE_NAMES.get(self.state, "unknown")

    def remaining_path(self):
        """Current position followed by the remaining waypoints."""
        return [(self.x, self.y)] + list(self.path)

    def to_dict(self):
        """Serializable representation of the rider."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "orders": [getattr(o, "id", None) for o in self.orders],
            "special_case": self.special_case,
            "distance_travelled": self.distance_travelled,
            "completed_count": self.completed_count,
            "log": list(self.log),
        }

    def __repr__(self):
        return "Rider(id=%s, pos=(%s, %s), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state, len(self.orders))

import json

import globalvar

def init_system(config=None):

    if config:

    globalvar.set_value("config", cfg)
    globalvar.set_value("funds", int(cfg["initial_funds"]))

def _ensure_initialized():
    """Fill in any missing global keys with default values."""
    defaults = {
        "config": dict(DEFAULT_CONFIG),
        "funds": DEFAULT_CONFIG["initial_funds"],
        "riders": [],
        "rider_id_counter": 0,
        "running": True,
        "time": 0,
        "orders": [],
        "pending_orders": [],
        "order_id_counter": 0,
    }
    for key, value in defaults.items():
        if globalvar.get_value(key) is None:
            globalvar.set_value(key, value)
    riders = globalvar.get_value("riders")
    if riders:
        counter = globalvar.get_value("rider_id_counter") or 0
        max_id = max(r.rider_id for r in riders)
        if max_id > counter:
            globalvar.set_value("rider_id_counter", max_id)


class SystemManager:
    """Manages funds, riders, simulated time and system state."""

    def __init__(self):
        _ensure_initialized()

    @property
    def config(self):
        return globalvar.get_value("config")

    @property
    def funds(self):
        return globalvar.get_value("funds")

    @property
    def riders(self):
        return globalvar.get_value("riders")

    @property
    def time(self):
        return globalvar.get_value("time")

    @property
    def running(self):
        return globalvar.get_value("running")

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    def idle_riders(self):
        """Riders that are idle and have no queued orders."""
        return [r for r in self.riders if r.state == IDLE and not r.orders]

    def can_recruit(self):
        """True when funds allow recruiting and the rider cap is not hit."""
        cfg = self.config
        return (self.funds >= cfg["rider_cost"]
                and len(self.riders) < cfg["max_riders"])

    def recruit_rider(self):
        """Recruit a new rider if possible; returns it or None."""
        if not self.can_recruit():
            return None
        counter = (globalvar.get_value("rider_id_counter") or 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        rider = Rider(counter)
        globalvar.set_value("funds", self.funds - self.config["rider_cost"])
        self.riders.append(rider)
        logger.info("Recruited rider %s", counter)
        return rider

    def check_consistency(self):
        """Repair inconsistent rider states."""
        for rider in self.riders:
            if rider.state != IDLE and not rider.orders:
                logger.warning("Rider %s busy without orders; reset",
                               rider.rider_id)
                rider.state = IDLE
                rider.path = []
                rider.special_case = False
            if rider.state not in (0, 1, 2):
                rider.state = IDLE
                rider.path = []

    def tick(self, count=1):
        """Advance the simulation by ``count`` ticks."""
        from order import OrderManager
        manager = OrderManager(self)
        for _ in range(max(0, int(count))):
            globalvar.set_value("time", self.time + 1)
            manager.process_pending()
            for rider in list(self.riders):
                try:
                    rider.move()
                except Exception as exc:  # error recovery
                    logger.error("Rider %s failed to move: %s",
                                 rider.rider_id, exc)
                    rider.path = []
            self.check_consistency()

    def snapshot(self):
        """Serializable state of the whole system."""
        orders = globalvar.get_value("orders") or []
        return {
            "config": self.config,
            "funds": self.funds,
            "time": self.time,
            "rider_id_counter": globalvar.get_value("rider_id_counter"),
            "riders": [r.to_dict() for r in self.riders],
            "orders": [o.to_dict() for o in orders],
        }

    def save_state(self, path=None):
        """Write the system state to a JSON file. Returns the path or None."""
        path = path or self.config.get("state_file", "dispatch_state.json")
        try:
            with open(path, "w", encoding="utf-8") as fh:
                json.dump(self.snapshot(), fh, indent=2)
            return path
        except OSError as exc:
            logger.error("Could not save state: %s", exc)
            return None

"""Order management: creation, assignment and completion tracking."""

import logging

import globalvar
from order_model import Order
from path_planner import PathPlanner, manhattan
from rider import IDLE

def find_best_rider_for_order(order):
    """Return the most suitable rider for ``order``.

    Idle riders are preferred; among the candidates the nearest one
    (Manhattan distance to the restaurant) wins, ties are broken by the
    number of assigned orders (load balancing) and then by rider id.
    If no riders exist, one is recruited when funds allow; returns None if
    no rider can be found.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        from system import SystemManager
        return SystemManager().recruit_rider()
    res = tuple(order.res_pos)
    candidates = [r for r in riders if r.state == IDLE and not r.orders]
    if not candidates:
        candidates = riders
    return min(candidates, key=lambda r: (manhattan((r.x, r.y), res),
                                          len(r.orders), r.rider_id))

    def create_order(self, coords):
        """Create and dispatch an order from ``[rx, ry, cx, cy]``.

        Returns a dict with keys: order, rider, new_rider (Rider or None),
        route (list of points), route_length, eta.
        """
        rx, ry, cx, cy = [int(v) for v in coords]
        order = Order((rx, ry), (cx, cy))
        self.orders.append(order)
        new_rider = None
        if not self.system.idle_riders():
            new_rider = self.system.recruit_rider()
        rider = find_best_rider_for_order(order)
        if rider is not None and new_rider is None and \
                rider in self.system.riders and len(self.system.riders) == 1 \
                and rider.log and len(rider.log) == 1 and not rider.orders \
                and rider.rider_id == globalvar.get_value("rider_id_counter") \
                and rider.distance_travelled == 0 and rider.completed_count == 0:
            pass
        result = {"order": order, "rider": rider, "new_rider": new_rider,
                  "route": [], "route_length": 0, "eta": None}
        if rider is None:
            self.pending.append(order)
            return result
        self.assign(order, rider)
        route, length = self.plan_route(rider, order)
        result["route"] = route
        result["route_length"] = length
        result["eta"] = -(-length // rider.speed) if rider.speed else None
        return result

    def plan_route(self, rider, order):
        """Plan the full route of ``rider`` for ``order``.

        Starts at the rider position (or the previous order's customer if
        the rider is busy).  Returns ``(points, length)`` where ``points``
        includes the starting point.
        """
        idx = rider.orders.index(order) if order in rider.orders else -1
        if idx > 0:
            start = tuple(rider.orders[idx - 1].cus_pos)
        else:
            start = (rider.x, rider.y)
        path = self.planner.plan_multi(start, [order.res_pos, order.cus_pos])
        length = self.planner.path_length(start, path)
        return [start] + path, length

    def process_pending(self):
        """Try to assign orders still waiting for a rider."""
        pending = self.pending
        if not pending:
            return
        for order in list(pending):
            if not self.system.idle_riders():
                self.system.recruit_rider()
            rider = find_best_rider_for_order(order)
            if rider is None:
                break
            self.assign(order, rider)
            pending.remove(order)

    def completed_orders(self):
        """All completed orders."""
        return [o for o in self.orders if o.status == "completed"]

"""Status display and ASCII map visualization."""

import globalvar
from config import CELL_SIZE, MAP_SIZE
from rider import STATE_NAMES

STEP = CELL_SIZE // 2
LEGEND = ("Legend: 0-9/A-Z rider  R restaurant  C customer  * route  "
          "+ intersection  -,| street  . empty cell")


def _snap(v):
    return int(round(v / STEP)) * STEP


def expand_route(points):
    """Expand a list of waypoints into points every 50 units."""
    result = []
    if not points:
        return result
    cx, cy = _snap(points[0][0]), _snap(points[0][1])
    result.append((cx, cy))
    for p in points[1:]:
        tx, ty = _snap(p[0]), _snap(p[1])
        while cx != tx:
            cx += STEP if tx > cx else -STEP
            result.append((cx, cy))
        while cy != ty:
            cy += STEP if ty > cy else -STEP
            result.append((cx, cy))
    return result


def rider_symbol(rider_id):
    """Single character symbol for a rider."""
    rid = int(rider_id)
    if rid < 10:
        return str(rid)
    if rid < 36:
        return chr(ord("A") + rid - 10)
    return "@"


class DisplayManager:
    """Formats status information and renders the ASCII map."""

    def render_map(self, route=None, show_orders=True, orders=None):
        """Render the 8x8 grid map (at 50-unit resolution) as a string."""
        n = MAP_SIZE // STEP + 1
        grid = {}
        for i in range(n):
            for j in range(n):
                x, y = i * STEP, j * STEP
                if x % CELL_SIZE == 0 and y % CELL_SIZE == 0:
                    ch = "+"
                elif y % CELL_SIZE == 0:
                    ch = "-"
                elif x % CELL_SIZE == 0:
                    ch = "|"
                else:
                    ch = "."
                grid[(x, y)] = ch
        if route:
            for p in expand_route(route):
                if p in grid:
                    grid[p] = "*"
        if orders is None and show_orders:
            orders = [o for o in (globalvar.get_value("orders") or [])
                      if o.status != "completed"]
        for o in orders or []:
            grid[tuple(o.res_pos)] = "R"
            grid[tuple(o.cus_pos)] = "C"
        for r in globalvar.get_value("riders") or []:
            p = (_snap(r.x), _snap(r.y))
            if p in grid:
                grid[p] = rider_symbol(r.rider_id)
        lines = []
        for j in range(n - 1, -1, -1):
            y = j * STEP
            row = " ".join(grid[(i * STEP, y)] for i in range(n))
            lines.append("%4d %s" % (y, row))
        footer = "     " + " ".join(
            str(i * STEP // 100) if (i * STEP) % CELL_SIZE == 0 else " "
            for i in range(n))
        lines.append(footer + "   (x100)")
        lines.append(LEGEND)
        return "\n".join(lines)

"""Command parsing and input validation."""

from config import MAP_SIZE

try:  # command history with arrow keys where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

SHORTCUTS = {
    "s": "start", "n": "start", "new": "start",
    "st": "status", "o": "orders", "r": "riders",
    "m": "map", "t": "tick", "h": "help", "?": "help",
    "q": "quit", "exit": "quit", "hist": "history", "sv": "save",
}

COMMANDS = ("start", "status", "orders", "riders", "map", "tick",
            "help", "history", "save", "quit")


class InputManager:
    """Reads commands, keeps history and validates user input."""

    def __init__(self):
        self.history = []

    def read(self, prompt="> "):
        """Read a line from stdin (raises EOFError on end of input)."""
        line = input(prompt)
        if line.strip():
            self.history.append(line.strip())
            if len(self.history) > 1000:
                del self.history[:-1000]
        return line

    @staticmethod
    def parse_command(line):
        """Split a line into ``(command, args)`` resolving shortcuts."""
        parts = line.replace(",", " ").split()
        if not parts:
            return "", []
        cmd = parts[0].lower()
        cmd = SHORTCUTS.get(cmd, cmd)
        return cmd, parts[1:]

    @staticmethod
    def parse_coordinates(tokens):
        """Validate four integer coordinates within 0-800.

        Accepts tokens such as ``["[100,", "200,", ...]`` or a single
        string.  Raises ValueError with a friendly message on bad input.
        """
        if isinstance(tokens, str):
            tokens = [tokens]
        text = " ".join(str(t) for t in tokens)
        for ch in "[](),":
            text = text.replace(ch, " ")
        parts = text.split()
        if len(parts) != 4:
            raise ValueError("Expected 4 integers: restaurant_x restaurant_y "
                             "customer_x customer_y (got %d values)"
                             % len(parts))
        values = []
        for p in parts:
            try:
                v = int(p)
            except ValueError:
                raise ValueError("'%s' is not an integer" % p)
            if v < 0 or v > MAP_SIZE:
                raise ValueError("Coordinate %d out of range 0-%d"
                                 % (v, MAP_SIZE))
            values.append(v)
        return values

    @staticmethod
    def parse_positive_int(tokens, default=1, maximum=100000):
        """Parse an optional positive integer argument."""
        if not tokens:
            return default
        try:
            v = int(tokens[0])
        except ValueError:
            raise ValueError("'%s' is not an integer" % tokens[0])
        if v < 1 or v > maximum:
            raise ValueError("Value must be between 1 and %d" % maximum)
        return v

"""Command line entry point of the food delivery dispatch simulator."""

import logging
import os
import sys

import globalvar
from config import load_config
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from system import SystemManager, init_system

HELP = """Commands (shortcut in brackets):
  start [rx ry cx cy] (s)  create an order: restaurant + customer coords 0-800
  status (st)              system funds and rider states
  orders (o)               all orders
  riders (r)               rider status change history
  map (m)                  ASCII map
  tick [n] (t)             advance the simulation n ticks
  history (hist)           command history
  save (sv)                save state to JSON
  help (h)                 this help
  quit / exit (q)          exit the program
Every command advances the simulation by one tick."""


class Application:
    """Main application loop."""

    def __init__(self, config=None):
        globalvar._init()
        init_system(config)
        self.system = SystemManager()
        self.orders = OrderManager(self.system)
        self.display = DisplayManager()
        self.input = InputManager()

    def handle(self, cmd, args):
        """Handle one command; returns False if the clock must not tick."""
        if cmd == "":
            return True
        if cmd == "start":
            if not args:
                args = [self.input.read(
                    "Enter restaurant_x restaurant_y customer_x customer_y: ")]
            coords = self.input.parse_coordinates(args)
            result = self.orders.create_order(coords)
            print(self.display.order_result_text(result))
        elif cmd == "status":
            print(self.display.status_text())
        elif cmd == "orders":
            print(self.display.orders_text())
        elif cmd == "riders":
            print(self.display.riders_text())
        elif cmd == "map":
            print(self.display.render_map())
        elif cmd == "tick":
            n = self.input.parse_positive_int(args)
            self.system.tick(n)
            print("Advanced %d tick(s); time is now %s."
                  % (n, self.system.time))
            return False
        elif cmd == "history":
            for i, line in enumerate(self.input.history, 1):
                print("%4d  %s" % (i, line))
        elif cmd == "save":
            path = self.system.save_state()
            print("State saved to %s" % path if path else "Save failed.")
        elif cmd == "help":
            print(HELP)
        elif cmd == "quit":
            self.system.stop()
            return False
        else:
            print("Unknown command '%s'. Type 'help' for commands." % cmd)
            return False
        return True

    def run(self):
        """Run the interactive loop."""
        print("Food Delivery Dispatch Simulator (8x8 grid, 800x800 units)")
        print("Type 'help' for commands.")
        while self.system.running:
            try:
                line = self.input.read("[t=%s] > " % self.system.time)
            except (EOFError, KeyboardInterrupt):
                print()
                self.system.stop()
                break
            cmd, args = self.input.parse_command(line)
            try:
                if self.handle(cmd, args):
                    self.system.tick(1)
            except ValueError as exc:
                print("Input error: %s" % exc)
            except (EOFError, KeyboardInterrupt):
                print()
                self.system.stop()
            except Exception as exc:  # graceful recovery
                logging.getLogger(__name__).exception("Unexpected error")
                print("Error: %s (system state recovered)" % exc)
                self.system.check_consistency()
        path = self.system.save_state()
        if path:
            print("State saved to %s" % path)
        print("Goodbye.")


def main(argv=None):
    """Program entry point."""
    argv = sys.argv[1:] if argv is None else argv
    level = logging.DEBUG if "--debug" in argv else logging.WARNING
    logging.basicConfig(level=level,
                        format="%(asctime)s %(levelname)s %(name)s: %(message)s")
    cfg_path = "config.json"
    for i, a in enumerate(argv):
        if a == "--config" and i + 1 < len(argv):
            cfg_path = argv[i + 1]
    if not os.path.isfile(cfg_path):
        alt = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                           "config.json")
        cfg_path = alt
    Application(load_config(cfg_path)).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

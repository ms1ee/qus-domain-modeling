from __future__ import annotations

from typing import List, Optional


def normalize_group(name: Optional[str]) -> str:
    """Normalize a group/department name for comparisons."""
    if name is None:
        return ""
    return " ".join(str(name).strip().split()).lower()


class User:
    """A participant of the lottery."""

    def __init__(self, name: str, department: str):
        self.name = str(name).strip()
        self.department = str(department).strip()
        self.is_valid = True
        self.has_won = False

    def __eq__(self, other):
        if not isinstance(other, User):
            return NotImplemented
        return self.name == other.name and self.department == other.department

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User(name={self.name!r}, department={self.department!r})"

    def __str__(self):
        return f"{self.name} ({self.department})"


_LEVEL_KEYWORDS = [
    ("grand", 0),
    ("special", 0),
    ("first", 1),
    ("1st", 1),
    ("second", 2),
    ("2nd", 2),
    ("third", 3),
    ("3rd", 3),
    ("fourth", 4),
    ("4th", 4),
    ("fifth", 5),
    ("5th", 5),
]


class Prize:
    """A prize to be drawn in one lottery round."""

    def __init__(
        self,
        name: str,
        quantity: int,
        avoid_groups: Optional[List[str]] = None,
        associated_drawer: Optional[str] = None,
        level: Optional[int] = None,
    ):
        self.name = str(name).strip()
        self.quantity = int(quantity)
        self.avoid_groups: List[str] = [
            str(g).strip() for g in (avoid_groups or []) if str(g).strip()
        ]
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else associated_drawer
        self.associated_drawer: Optional[str] = drawer or None
        self.level = level

    def get_level(self) -> int:
        """Return a rank (smaller is higher) derived from explicit level or name."""
        if self.level is not None:
            return int(self.level)
        lowered = self.name.lower()
        for keyword, rank in _LEVEL_KEYWORDS:
            if keyword in lowered:
                return rank
        return 100

    def rule_description(self) -> str:
        parts = []
        if self.associated_drawer:
            parts.append(f"Lottery person: {self.associated_drawer}")
        else:
            parts.append("Lottery person: (not specified)")
        if self.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("Avoidance groups: none")
        parts.append("Previous winners are excluded")
        return "; ".join(parts)

    def __repr__(self):
        return (
            f"Prize(name={self.name!r}, quantity={self.quantity}, "
            f"avoid_groups={self.avoid_groups!r}, associated_drawer={self.associated_drawer!r})"
        )

class ListManager:
    """Loads and validates participant lists."""

    def __init__(self):
        self.load_mode = None
        self.source_file_path = None

    # ------------------------------------------------------------------ load
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        self.load_mode = "default"
        self.source_file_path = None
        lines = [f"{n},{d}" for n, d in DEFAULT_LIST]
        return self._parse_lines(lines)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        self.load_mode = "custom"
        self.source_file_path = path
        if not path or not str(path).strip():
            return [], ["File path is empty."]
        path = str(path).strip()
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        if not path.lower().endswith(".txt"):
            return [], [f"Only .txt files are supported: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            try:
                with open(path, "r", encoding="utf-8-sig") as f:
                    lines = f.read().splitlines()
            except UnicodeDecodeError:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.read().splitlines()
        except OSError as exc:
            return [], [f"Failed to read file {path}: {exc}"]
        return self._parse_lines(lines)

    @staticmethod
    def _parse_lines(lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff")
            if not line:
                continue
            normalized = line.replace("，", ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # -------------------------------------------------------------- validate
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        valid: List[User] = []
        duplicates: List[User] = []
        seen = set()
        for user in users or []:
            if user.name in seen:
                user.is_valid = False
                duplicates.append(user)
            else:
                seen.add(user.name)
                valid.append(user)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for user in users or []:
            info[user.department] = info.get(user.department, 0) + 1
        return info

from __future__ import annotations

from typing import List, Optional

from lottery_system.models import Prize

PRESET_PRIZES = [
    ("Third Prize - Movie Tickets", 5, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 3, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    """Maintains the prize library and the lottery round sequence."""

    def __init__(self):
        self.prizes: List[Prize] = []
        self.is_preset = False

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in PRESET_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        self.is_preset = True
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if name is None or not str(name).strip():
            raise ValueError("Prize name cannot be empty.")
        name = str(name).strip()
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if isinstance(quantity, float) and quantity != qty:
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity of prize '{name}' must be a positive integer.")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists.")
        if isinstance(avoid_groups, str):
            avoid_groups = [g for g in avoid_groups.split(",")]
        prize = Prize(name, qty, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def remove_prize(self, index: int) -> bool:
        if 1 <= index <= len(self.prizes):
            self.prizes.pop(index - 1)
            return True
        return False

    def reorder_prizes(self, order: List[int]) -> bool:
        try:
            positions = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        n = len(self.prizes)
        if len(positions) != n or sorted(positions) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in positions]
        return True

    def find_prize(self, name: str) -> Optional[Prize]:
        for p in self.prizes:
            if p.name == name:
                return p
        return None

"""Lottery rule engine and drawing logic."""

from __future__ import annotations

import random
from typing import List, Optional

from lottery_system.models import Prize, User, normalize_group

SPECIAL_PERSON = "Li Bin"
FALLBACK_TRIGGER_PERSONS = ["Chen Qixian", "Jin Duo"]
FALLBACK_TARGET_PERSON = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru Group"


def _norm_person(name) -> str:
    return normalize_group(name)


def is_protected_group(department: str) -> bool:
    d = normalize_group(department)
    return d in ("cheng ru group", "cheng ru")


class LotteryManager:
    """Applies rules and draws winners."""

    def __init__(self, users, result_manager=None, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users or [])
        self.result_manager = result_manager
        self.winners: List[User] = []
        self._winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()
        self.special_person_name = SPECIAL_PERSON

    # ------------------------------------------------------------- helpers
    def is_winner(self, user: User) -> bool:
        return user.name in self._winner_names

    def is_fallback_active(self, prize: Prize) -> bool:
        if _norm_person(prize.associated_drawer) != _norm_person(FALLBACK_TARGET_PERSON):
            return False
        if len(self.last_two_draw_results) < 2:
            return False
        return not any(self.last_two_draw_results[-2:])

    def describe_rules(self, prize: Prize) -> str:
        if _norm_person(prize.associated_drawer) == _norm_person(SPECIAL_PERSON):
            return (f"Special lottery person {SPECIAL_PERSON}: only previous winners are "
                    f"excluded, avoidance groups are NOT applied")
        if self.is_fallback_active(prize):
            return (f"Fallback triggered: pool limited to '{PROTECTED_GROUP}' members "
                    f"(previous winners excluded, avoidance groups ignored)")
        return prize.rule_description()

    # -------------------------------------------------------------- engine
    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = [u for u in self.users if not self.is_winner(u)]
        drawer = _norm_person(prize.associated_drawer)
        if drawer == _norm_person(SPECIAL_PERSON):
            return candidates
        if self.is_fallback_active(prize):
            return [u for u in candidates if is_protected_group(u.department)]
        avoid = {normalize_group(g) for g in prize.avoid_groups}
        return [u for u in candidates if normalize_group(u.department) not in avoid]

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.record_winner(user, prize)
        return user

    def record_winner(self, user: User, prize: Prize) -> bool:
        if self.is_winner(user):
            return False
        self._winner_names.add(user.name)
        self.winners.append(user)
        user.has_won = True
        if self.result_manager is not None:
            self.result_manager.add_winner(user, prize)
        return True

    def complete_round(self, prize: Prize, round_winners: List[User]) -> None:
        """Update fallback tracking after a round finishes."""
        drawer = _norm_person(prize.associated_drawer)
        triggers = {_norm_person(p) for p in FALLBACK_TRIGGER_PERSONS}
        if drawer in triggers:
            contains = any(is_protected_group(u.department) for u in round_winners)
            self.last_two_draw_results.append(contains)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm_person(FALLBACK_TARGET_PERSON):
            self.last_two_draw_results = []

    def draw_round(self, prize: Prize) -> List[User]:
        """Non-interactive full round draw."""
        result = []
        for _ in range(prize.quantity):
            u = self.draw_one(prize)
            if u is None:
                break
            result.append(u)
        self.complete_round(prize, result)
        return result

"""Result management: winner records, summary and export."""

from __future__ import annotations

import csv
import os
from datetime import datetime
from typing import Dict, List


class Winner:
    """A winning record."""

    def __init__(self, user, prize_name: str, winning_time: datetime, order: int):
        self.user = user
        self.prize_name = prize_name
        self.winning_time = winning_time
        self.winning_order = order

    @property
    def name(self):
        return self.user.name

    @property
    def department(self):
        return self.user.department

    def __repr__(self):
        return f"Winner({self.user.name!r}, {self.user.department!r}, {self.prize_name!r})"


class ResultManager:
    """Maintains the list of winners."""

    def __init__(self):
        self.winners: List[Winner] = []
        self._names = set()
        self._counter = 0
        self.export_path = None

    def add_winner(self, user, prize) -> bool:
        if user.name in self._names:
            return False
        prize_name = prize.name if hasattr(prize, "name") else str(prize)
        self._counter += 1
        self.winners.append(Winner(user, prize_name, datetime.now(), self._counter))
        self._names.add(user.name)
        return True

    def has_won(self, user) -> bool:
        return user.name in self._names

    @staticmethod
    def _sorted_prizes(prizes):
        indexed = list(enumerate(prizes or []))
        indexed.sort(key=lambda t: (t[1].get_level() if hasattr(t[1], "get_level") else 100, t[0]))
        return [p for _, p in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in self._sorted_prizes(prizes):
            name = p.name if hasattr(p, "name") else str(p)
            result[name] = []
        for w in sorted(self.winners, key=lambda w: (w.winning_time, w.winning_order)):
            result.setdefault(w.prize_name, []).append(w)
        return result

    def export_to_csv(self, prizes, path=None):
        if not self.winners:
            return False, "No winning records to export."
        if path is None:
            path = os.path.join(os.getcwd(), f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        grouped = self.get_winners_by_prize(prizes)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for prize_name, winners in grouped.items():
                    for w in winners:
                        writer.writerow([
                            w.user.name, w.user.department, prize_name,
                            w.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                        ])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        self.export_path = path
        return True, f"Results successfully exported to {path}"

"""Command-line interface of the lottery system."""

from __future__ import annotations

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class _Exit(Exception):
    pass


def _input(prompt: str) -> str:
    try:
        return input(prompt)
    except EOFError:
        raise _Exit()


def _ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suf = "th"
    else:
        suf = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suf}"


class LotteryCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = LotteryManager([], self.result_manager)
        self.status = "idle"
        self.current_round_index = 0

    # ----------------------------------------------------------- list
    def _set_users(self, users):
        self.users = users
        self.result_manager = ResultManager()
        self.lottery = LotteryManager(users, self.result_manager)
        print(f"{len(users)} valid participants loaded.")

    def _confirm_remove(self) -> bool:
        while True:
            c = _input("Type 'r' to remove invalid data and continue, 't' to terminate upload: ").strip().lower()
            if c in ("r", "remove"):
                return True
            if c in ("t", "terminate"):
                return False

    def _process_loaded(self, users, errors):
        if errors:
            print("Problems found:")
            for e in errors:
                print("  " + e)
            if not users or not self._confirm_remove():
                print("Upload terminated.")
                return
        valid, dups = self.list_manager.validate_users(users)
        if dups:
            print("Duplicate names detected:")
            for d in dups:
                print(f"  {d.name} ({d.department})")
            if not self._confirm_remove():
                print("Upload terminated.")
                return
        if not valid:
            print("No valid participants.")
            return
        self._set_users(valid)

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._process_loaded(users, errors)

    def load_custom(self):
        path = _input("Enter txt list path (each line 'Name,Department Group'): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        self._process_loaded(users, errors)

    def show_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        print("Current valid groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    # --------------------------------------------------------- prizes
    def load_preset_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Preset prize library loaded.")

    def add_custom_prize(self):
        name = _input("Prize name: ").strip()
        qty = _input("Quantity: ").strip()
        groups = list(self.list_manager.get_group_info(self.users).keys())
        avoid = []
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = _input("Avoidance groups (numbers or names, comma separated, empty for none): ").strip()
        for part in raw.replace("，", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                avoid.append(groups[int(part) - 1])
            else:
                avoid.append(part)
        drawer = _input("Associated lottery person (empty for none): ").strip() or None
        try:
            self.prize_manager.add_prize(name, qty, avoid, drawer)
            print(f"Prize '{name}' added.")
        except ValueError as exc:
            print(f"Error: {exc}")

    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def reorder(self):
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        raw = _input("Enter new order of serial numbers (e.g. 3,1,2): ").strip()
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prize order updated.")
            self.show_prizes()
        else:
            print("Invalid order; it must be a permutation of the current serial numbers.")

    # -------------------------------------------------------- lottery
    def run_lottery(self):
        prizes = self.prize_manager.get_prizes()
        if not self.users:
            print("Please load a participant list first.")
            return
        if not prizes:
            print("Please configure prizes first.")
            return
        self.status = "running"
        for idx, prize in enumerate(prizes, 1):
            self.current_round_index = idx
            print("\n" + "=" * 60)
            print(f"Round {idx}: {prize.name} | Quantity: {prize.quantity}")
            print(f"Rules: {self.lottery.describe_rules(prize)}")
            print("Commands: [Enter]=draw next, pause, continue, terminate")
            round_winners = []
            terminated = False
            paused = False
            while len(round_winners) < prize.quantity:
                cmd = _input("> ").strip().lower()
                if cmd == "pause":
                    paused = True
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to skip this round.")
                    continue
                if cmd == "terminate":
                    terminated = True
                    print("Current round terminated.")
                    break
                if cmd == "continue":
                    if paused:
                        paused = False
                        print("Lottery resumed.")
                    continue
                if paused:
                    print("Lottery is paused. Type 'continue' or 'terminate'.")
                    continue
                print(f"Currently drawing the {_ordinal(len(round_winners) + 1)} winner...")
                user = self.lottery.draw_one(prize)
                if user is None:
                    print("No eligible participants remain for this prize.")
                    break
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            self.lottery.complete_round(prize, round_winners)
            print(f"--- Round {idx} results ({prize.name}){' [terminated]' if terminated else ''} ---")
            if not round_winners:
                print("  (no winners)")
            for i, u in enumerate(round_winners, 1):
                print(f"  {i}. {u.name} ({u.department})")
        self.status = "finished"
        print("\nAll rounds completed.")
        self.show_summary()

    def show_summary(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        if not self.result_manager.winners:
            print("No winners yet.")
            return
        print("\n===== Complete Winner List =====")
        for prize_name, winners in grouped.items():
            if not winners:
                continue
            print(f"[{prize_name}]")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {prize_name} "
                      f"@ {w.winning_time.strftime('%H:%M:%S')}")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ----------------------------------------------------------- main
    def run(self):
        menu = {
            "1": ("Load default team building list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.show_groups),
            "4": ("Load preset prize library", self.load_preset_prizes),
            "5": ("Add custom prize", self.add_custom_prize),
            "6": ("View prizes", self.show_prizes),
            "7": ("Reorder prizes", self.reorder),
            "8": ("Start lottery", self.run_lottery),
            "9": ("View winner summary", self.show_summary),
            "10": ("Export results", self.export),
        }

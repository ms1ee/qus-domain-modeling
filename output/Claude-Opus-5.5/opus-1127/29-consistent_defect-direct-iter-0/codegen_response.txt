"""Global variable store shared across the dispatch system modules.

Usage::

    import globalvar
    globalvar._init()
    globalvar.set_value("riders", [])
    riders = globalvar.get_value("riders")
"""

_global_dict = {}


def _init():
    """(Re)initialise the global variable dictionary."""
    global _global_dict
    _global_dict = {}


def set_value(key, value):
    """Store ``value`` under ``key``."""
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under ``key`` (``default`` if missing)."""
    return _global_dict.get(key, default)


def has_key(key):
    """Return True if ``key`` is present in the global store."""
    return key in _global_dict


def all_values():
    """Return a shallow copy of the whole global store."""
    return dict(_global_dict)

import json

RIDER_COST = 500
RECRUIT_THRESHOLD = 300
RIDER_SPEED = 20

INPUT_MIN = 0
INPUT_MAX = 1000

MAP_CHARS = 16  # ASCII map is 16x16

DEFAULT_CONFIG_FILE = "config.json"

DEFAULTS = {
    "initial_funds": INITIAL_FUNDS,
    "rider_cost": RIDER_COST,
    "recruit_threshold": RECRUIT_THRESHOLD,
    "rider_speed": RIDER_SPEED,
    "rider_start": list(RIDER_START),
}


def load_config(path=DEFAULT_CONFIG_FILE):
    """Load a JSON configuration file, falling back to defaults.

    Unknown keys are ignored; invalid files produce a warning and defaults.
    """
    cfg = dict(DEFAULTS)
    if not path or not os.path.exists(path):
        return cfg
    try:
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        if isinstance(data, dict):
            for key in DEFAULTS:
                if key in data:
                    cfg[key] = data[key]
    except (OSError, ValueError) as exc:
        logger.warning("Could not load config %s: %s", path, exc)
    return cfg

"""Order data model."""

import itertools

import globalvar
from config import CELL_SIZE, GRID_DIM

STATUS_PENDING = "pending"
STATUS_ASSIGNED = "assigned"
STATUS_PICKING_UP = "picking_up"
STATUS_DELIVERING = "delivering"
STATUS_COMPLETED = "completed"

_id_counter = itertools.count(1)


def reset_order_ids(start=1):
    """Reset the order ID counter (used when loading state)."""
    global _id_counter
    _id_counter = itertools.count(start)


def _grid_index(value):
    """Return the clamped grid cell index (0..GRID_DIM-1) for a coordinate."""
    idx = int(value) // CELL_SIZE
    return max(0, min(GRID_DIM - 1, idx))


def to_grid_cell(value):
    """Convert a coordinate to the top-left corner of its grid cell."""
    return _grid_index(value) * CELL_SIZE


def to_grid_center(value):
    """Convert a coordinate to its grid cell center: (v // 100) * 100 + 50."""
    return _grid_index(value) * CELL_SIZE + CELL_SIZE // 2


def _now():
    t = globalvar.get_value("time")
    return t if isinstance(t, int) else 0


def _parse_pos(pos, name):
    try:
        x, y = pos
        return int(x), int(y)
    except (TypeError, ValueError):
        raise ValueError("%s must be a pair of integers, got %r" % (name, pos))


class Order:
    """A delivery order from a restaurant to a customer.

    Attributes:
        id: unique order ID.
        status: one of pending/assigned/picking_up/delivering/completed.
        raw_res_pos / raw_cus_pos: coordinates as entered.
        res_pos / cus_pos: converted grid-cell center coordinates.
        res_cell / cus_cell: top-left corner of the grid cells.
        rider_id: ID of the assigned rider (None if unassigned).
    """

    def __init__(self, res_pos, cus_pos, order_id=None):
        rx, ry = _parse_pos(res_pos, "res_pos")
        cx, cy = _parse_pos(cus_pos, "cus_pos")
        self.id = order_id if order_id is not None else next(_id_counter)
        self.raw_res_pos = (rx, ry)
        self.raw_cus_pos = (cx, cy)
        self.res_pos = (to_grid_center(rx), to_grid_center(ry))
        self.cus_pos = (to_grid_center(cx), to_grid_center(cy))
        self.res_cell = (to_grid_cell(rx), to_grid_cell(ry))
        self.cus_cell = (to_grid_cell(cx), to_grid_cell(cy))
        self.status = STATUS_PENDING
        self.rider_id = None
        self.created_time = _now()
        self.assigned_time = None
        self.accepted_time = None
        self.pickup_time = None
        self.completed_time = None
        self.travel_distance = 0

    @property
    def distance(self):
        """Manhattan distance between restaurant and customer."""
        return (abs(self.res_pos[0] - self.cus_pos[0])
                + abs(self.res_pos[1] - self.cus_pos[1]))

    @property
    def time_taken(self):
        """Ticks from creation to completion, or None if not completed."""
        if self.completed_time is None:
            return None
        return self.completed_time - self.created_time

    def assign_to(self, rider_id):
        """Record that this order is assigned to ``rider_id``."""
        self.rider_id = rider_id
        if self.status == STATUS_PENDING:
            self.status = STATUS_ASSIGNED
        self.assigned_time = _now()

    def is_completed(self):
        """Return True if the order has been delivered."""
        return self.status == STATUS_COMPLETED

    def is_active(self):
        """Return True if the order is not yet delivered."""
        return not self.is_completed()

    def to_dict(self):
        """Serialise the order to a JSON-compatible dict."""
        return {
            "id": self.id,
            "raw_res_pos": list(self.raw_res_pos),
            "raw_cus_pos": list(self.raw_cus_pos),
            "status": self.status,
            "rider_id": self.rider_id,
            "created_time": self.created_time,
            "assigned_time": self.assigned_time,
            "accepted_time": self.accepted_time,
            "pickup_time": self.pickup_time,
            "completed_time": self.completed_time,
            "travel_distance": self.travel_distance,
        }

    @classmethod
    def from_dict(cls, data):
        """Rebuild an order from :meth:`to_dict` output."""
        order = cls(data["raw_res_pos"], data["raw_cus_pos"],
                    order_id=data["id"])
        for key in ("status", "rider_id", "created_time", "assigned_time",
                    "accepted_time", "pickup_time", "completed_time",
                    "travel_distance"):
            if key in data:
                setattr(order, key, data[key])
        return order

    def __repr__(self):
        return "Order(id=%s, res=%s, cus=%s, status=%s, rider=%s)" % (
            self.id, self.res_pos, self.cus_pos, self.status, self.rider_id)

"""Grid-based path planning."""

from config import CELL_SIZE


class PathPlanner:
    """Plans Manhattan (right-angle) paths on the grid map.

    Riders move only horizontally or vertically. Paths prefer streets: if a
    rider stands on a vertical street (x is a multiple of 100) it moves
    vertically first, otherwise it moves horizontally first. When the rider
    is directly above/below (or level with) the target a straight path is
    used (the special case). Riders never block each other, so there are no
    path conflicts or deadlocks.
    """

    @staticmethod
    def is_street(x, y):
        """Return True if (x, y) lies on a street grid line."""
        return x % CELL_SIZE == 0 or y % CELL_SIZE == 0

    @staticmethod
    def is_special_case(start, target):
        """Rider directly above or below the target (same x, different y)."""
        return start[0] == target[0] and start[1] != target[1]

    @staticmethod
    def plan(start, target):
        """Return a list of waypoints from ``start`` to ``target``.

        The start point is not included; the last waypoint is the target.
        """
        sx, sy = int(start[0]), int(start[1])
        tx, ty = int(target[0]), int(target[1])
        if (sx, sy) == (tx, ty):
            return []
        if sx == tx or sy == ty:
            return [(tx, ty)]
        if sx % CELL_SIZE == 0 and sy % CELL_SIZE != 0:
            corner = (sx, ty)
        else:
            corner = (tx, sy)
        return [corner, (tx, ty)]

    @staticmethod
    def route(start, stops):
        """Plan a continuous route through several stops.

        Returns the list of points including ``start``.
        """
        points = [(int(start[0]), int(start[1]))]
        cur = points[0]
        for stop in stops:
            for wp in PathPlanner.plan(cur, stop):
                points.append(wp)
            cur = (int(stop[0]), int(stop[1]))
        return points

    @staticmethod
    def path_length(points):
        """Total Manhattan length of a polyline."""
        total = 0
        for a, b in zip(points, points[1:]):
            total += abs(a[0] - b[0]) + abs(a[1] - b[1])
        return total

    @staticmethod
    def step(pos, waypoints, budget):
        """Advance along ``waypoints`` using at most ``budget`` units.

        Returns ``(new_pos, remaining_waypoints, used)``.
        """
        x, y = int(pos[0]), int(pos[1])
        wps = list(waypoints)
        used = 0
        budget = int(budget)
        while wps and used < budget:
            tx, ty = wps[0]
            if (x, y) == (tx, ty):
                wps.pop(0)
                continue
            rem = budget - used
            if x != tx:
                d = min(rem, abs(tx - x))
                x += d if tx > x else -d
                used += d
            else:
                d = min(rem, abs(ty - y))
                y += d if ty > y else -d
                used += d
        while wps and (x, y) == tuple(wps[0]):
            wps.pop(0)
        return (x, y), wps, used

"""Rider model: movement, state machine and event log."""

import logging

import globalvar
from config import RIDER_SPEED, RIDER_START
from order_model import (Order, STATUS_COMPLETED, STATUS_DELIVERING,
                         STATUS_PICKING_UP)
from path_planner import PathPlanner

logger = logging.getLogger(__name__)

STATE_IDLE = 0
STATE_PICKING_UP = 1
STATE_DELIVERING = 2

STATE_NAMES = {
    STATE_IDLE: "idle",
    STATE_PICKING_UP: "picking up",
    STATE_DELIVERING: "delivering",
}


def _status_of(order):
    return getattr(order, "status", None)


class Rider:
    """A delivery rider.

    Orders are assigned by appending them to ``rider.orders``. Each call to
    :meth:`move` advances the rider by one tick; state transitions and log
    entries happen there.
    """

    def __init__(self, rider_id, x=None, y=None, speed=None):
        self.rider_id = rider_id
        self.x = int(RIDER_START[0] if x is None else x)
        self.y = int(RIDER_START[1] if y is None else y)
        self.speed = int(RIDER_SPEED if speed is None else speed)
        if self.speed < 1:
            self.speed = 1
        self.state = STATE_IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.current_order = None
        self.path = []
        self.ticks = 0
        self.total_distance = 0
        self._record("spawn", None)

    # ------------------------------------------------------------------ utils
    def _now(self):
        t = globalvar.get_value("time")
        return t if isinstance(t, int) else self.ticks

    def _record(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": self._now(),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    @property
    def position(self):
        """Current (x, y) position."""
        return (self.x, self.y)

    @property
    def state_name(self):
        """Human-readable state."""
        return STATE_NAMES.get(self.state, str(self.state))

    def active_orders(self):
        """Orders not yet completed."""
        return [o for o in self.orders if _status_of(o) != STATUS_COMPLETED]

    def active_order_count(self):
        """Number of orders not yet completed."""
        return len(self.active_orders())

    def is_idle(self):
        """True if the rider is idle with no outstanding orders."""
        return self.state == STATE_IDLE and self.active_order_count() == 0

    def _normalize_orders(self):
        """Convert raw ``[rx, ry, cx, cy]`` entries into Order objects."""
        for i, o in enumerate(self.orders):
            if isinstance(o, (list, tuple)) and len(o) == 4:
                order = Order((o[0], o[1]), (o[2], o[3]))
                order.assign_to(self.rider_id)
                self.orders[i] = order

    def _current_target(self):
        order = self.current_order
        if self.state == STATE_PICKING_UP:
            return tuple(order.res_pos)
        return tuple(order.cus_pos)

    def _replan(self):
        target = self._current_target()
        self.special_case = PathPlanner.is_special_case(self.position, target)
        self.path = PathPlanner.plan(self.position, target)

    def _accept_next(self):
        """Take the next outstanding order. Returns True if one was taken."""
        for order in self.orders:
            if _status_of(order) == STATUS_COMPLETED:
                continue
            self.current_order = order
            if getattr(order, "rider_id", None) is None and \
                    hasattr(order, "assign_to"):
                order.assign_to(self.rider_id)
            if _status_of(order) == STATUS_DELIVERING:
                self.state = STATE_DELIVERING
            else:
                self.state = STATE_PICKING_UP
                order.status = STATUS_PICKING_UP
                order.accepted_time = self._now()
                self._record("accepted_order", order.id)
            self._replan()
            return True
        self.current_order = None
        return False

    def _arrive(self):
        order = self.current_order
        if self.state == STATE_PICKING_UP:
            self.state = STATE_DELIVERING
            order.status = STATUS_DELIVERING
            order.pickup_time = self._now()
            self._record("arrived_at_merchant", order.id)
            self._replan()
        else:
            order.status = STATUS_COMPLETED
            order.completed_time = self._now()
            self._record("delivered_to_customer", order.id)
            self.current_order = None
            self.path = []
            self.special_case = False
            if not self._accept_next():
                self.state = STATE_IDLE

    # ---------------------------------------------------------------- moving
    def move(self):
        """Advance the rider by one tick. Returns the resulting state."""
        self.ticks += 1
        self._normalize_orders()
        if self.current_order is None or \
                _status_of(self.current_order) == STATUS_COMPLETED or \
                self.current_order not in self.orders:
            self.current_order = None
            if not self._accept_next():
                self.state = STATE_IDLE
                return self.state
        budget = self.speed
        guard = 0
        while self.current_order is not None and guard < 1000:
            guard += 1
            target = self._current_target()
            if self.position == target:
                self._arrive()
                continue
            if budget <= 0:
                break
            if not self.path or tuple(self.path[-1]) != target:
                self._replan()
            new_pos, self.path, used = PathPlanner.step(
                self.position, self.path, budget)
            if used == 0:
                self._replan()
                new_pos, self.path, used = PathPlanner.step(
                    self.position, self.path, budget)
                if used == 0:
                    logger.error("Rider %s stuck at %s", self.rider_id,
                                 self.position)
                    break
            self.x, self.y = new_pos
            budget -= used
            self.total_distance += used
            if self.current_order is not None:
                self.current_order.travel_distance = \
                    getattr(self.current_order, "travel_distance", 0) + used
        return self.state

    def remaining_route(self):
        """Points of the rider's remaining route (including current pos)."""
        stops = []
        if self.current_order is not None:
            if self.state == STATE_PICKING_UP:
                stops.append(self.current_order.res_pos)
            stops.append(self.current_order.cus_pos)
        for o in self.orders:
            if o is self.current_order or _status_of(o) == STATUS_COMPLETED:
                continue
            stops.append(o.res_pos)
            stops.append(o.cus_pos)
        return PathPlanner.route(self.position, stops)

    # ----------------------------------------------------------- persistence
    def to_dict(self):
        """Serialise the rider (orders stored as IDs)."""
        return {
            "rider_id": self.rider_id,
            "x": self.x,
            "y": self.y,
            "speed": self.speed,
            "state": self.state,
            "special_case": self.special_case,
            "orders": [getattr(o, "id", None) for o in self.orders],
            "current_order": getattr(self.current_order, "id", None),
            "log": list(self.log),
            "ticks": self.ticks,
            "total_distance": self.total_distance,
        }

    @classmethod
    def from_dict(cls, data, orders_by_id):
        """Rebuild a rider from :meth:`to_dict` output."""
        rider = cls(data["rider_id"], data["x"], data["y"], data["speed"])
        rider.state = data.get("state", STATE_IDLE)
        rider.special_case = data.get("special_case", False)
        rider.orders = [orders_by_id[i] for i in data.get("orders", [])
                        if i in orders_by_id]
        cur = data.get("current_order")
        rider.current_order = orders_by_id.get(cur) if cur is not None \
            else None
        rider.log = list(data.get("log", []))
        rider.ticks = data.get("ticks", 0)
        rider.total_distance = data.get("total_distance", 0)
        if rider.current_order is not None:
            rider._replan()
        return rider

    def __repr__(self):
        return "Rider(id=%s, pos=(%s, %s), state=%s, orders=%d)" % (
            self.rider_id, self.x, self.y, self.state_name, len(self.orders))

"""Order management: creation, assignment and completion queries."""

import logging
import math

import globalvar
from order_model import Order, STATUS_COMPLETED, STATUS_PENDING

logger = logging.getLogger(__name__)


def _restaurant_pos(order):
    if hasattr(order, "res_pos"):
        return order.res_pos
    probe = Order((order[0], order[1]), (order[2], order[3]), order_id=-1)
    return probe.res_pos


def rider_distance(rider, pos):
    """Euclidean distance √[(rx - x)² + (ry - y)²]."""
    return math.sqrt((rider.x - pos[0]) ** 2 + (rider.y - pos[1]) ** 2)


def find_best_rider_for_order(order):
    """Return the best rider for ``order`` (None if there are no riders).

    Riders with the fewest outstanding orders are preferred; ties are broken
    by the distance to the restaurant (closest first), then by rider ID.
    """
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    res = _restaurant_pos(order)

    def key(r):
        if hasattr(r, "active_order_count"):
            load = r.active_order_count()
        else:
            load = len(getattr(r, "orders", []))
        return (load, rider_distance(r, res), r.rider_id)

    return min(riders, key=key)


class OrderManager:
    """Handles order creation, assignment and completion bookkeeping."""

    def __init__(self):
        if globalvar.get_value("orders") is None:
            globalvar.set_value("orders", [])

    @property
    def orders(self):
        """All orders known to the system."""
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        return orders

    def create_order(self, coords):
        """Create an order from ``[rx, ry, cx, cy]`` and register it."""
        if len(coords) != 4:
            raise ValueError("Exactly four coordinates are required.")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        self.orders.append(order)
        logger.info("Created %r", order)
        return order

    def pending_orders(self):
        """Orders not yet assigned to a rider."""
        return [o for o in self.orders if o.status == STATUS_PENDING]

    def completed_orders(self):
        """Delivered orders."""
        return [o for o in self.orders if o.status == STATUS_COMPLETED]

    def get_order(self, order_id):
        """Return order by ID or None."""
        for o in self.orders:
            if o.id == order_id:
                return o
        return None

    def assign(self, order):
        """Assign ``order`` to the best rider. Returns the rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        logger.info("Assigned order %s to rider %s", order.id, rider.rider_id)
        return rider

    def assign_pending(self):
        """Assign all pending orders. Returns list of (order, rider)."""
        result = []
        for order in self.pending_orders():
            rider = self.assign(order)
            if rider is None:
                break
            result.append((order, rider))
        return result

"""System initialisation and the SystemManager core class."""

import json
import logging

import globalvar
import order_model
from config import (DEFAULTS, DEFAULT_SAVE_FILE)
from order_model import Order, STATUS_PENDING
from rider import Rider

logger = logging.getLogger(__name__)


def init_system(config=None):
    """Initialise the global system state.

    Sets funds (1000), an empty rider list, rider ID counter (0), running
    state (True), simulated time (0) and an empty order list.
    """
    cfg = dict(DEFAULTS)
    if config:
        cfg.update(config)
    globalvar._init()
    globalvar.set_value("funds", int(cfg["initial_funds"]))
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("config", cfg)
    globalvar.set_value("map_size", 800)
    globalvar.set_value("cell_size", 100)


class SystemManager:
    """Manages funds, riders, simulated time and system state."""

    def __init__(self, config=None, initialise=True):
        if initialise or globalvar.get_value("riders") is None:
            init_system(config)

    # ------------------------------------------------------------ properties
    @property
    def config(self):
        return globalvar.get_value("config") or dict(DEFAULTS)

    @property
    def funds(self):
        return globalvar.get_value("funds", 0)

    @funds.setter
    def funds(self, value):
        globalvar.set_value("funds", int(value))

    @property
    def riders(self):
        riders = globalvar.get_value("riders")
        if riders is None:
            riders = []
            globalvar.set_value("riders", riders)
        return riders

    @property
    def time(self):
        return globalvar.get_value("time", 0)

    @property
    def running(self):
        return bool(globalvar.get_value("running", False))

    def stop(self):
        """Stop the main loop."""
        globalvar.set_value("running", False)

    # --------------------------------------------------------------- riders
    def recruit_rider(self):
        """Create a new rider, deducting the recruitment cost."""
        counter = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", counter)
        start = self.config.get("rider_start", [400, 400])
        rider = Rider(counter, start[0], start[1],
                      self.config.get("rider_speed"))
        self.riders.append(rider)
        self.funds = self.funds - int(self.config["rider_cost"])
        logger.info("Recruited rider %s; funds now %s", counter, self.funds)
        return rider

    def get_rider(self, rider_id):
        """Return rider by ID or None."""
        for r in self.riders:
            if r.rider_id == rider_id:
                return r
        return None

    def idle_rider_count(self):
        """Number of riders with no outstanding orders."""
        return sum(1 for r in self.riders if r.is_idle())

    def pending_order_count(self):
        """Number of orders not yet assigned to a rider."""
        orders = globalvar.get_value("orders") or []
        return sum(1 for o in orders if o.status == STATUS_PENDING)

    def auto_recruit(self):
        """Recruit riders while funds ≥ threshold and pending > idle.

        Returns the list of newly recruited riders.
        """
        new = []
        threshold = int(self.config["recruit_threshold"])
        while (self.funds >= threshold
               and self.pending_order_count() > self.idle_rider_count()):
            new.append(self.recruit_rider())
        return new

    # ----------------------------------------------------------- simulation
    def tick(self):
        """Advance simulated time by one tick and move all riders."""
        globalvar.set_value("time", self.time + 1)
        for rider in list(self.riders):
            try:
                rider.move()
            except Exception as exc:  # recover from a faulty rider
                logger.exception("Rider %s failed to move: %s",
                                 rider.rider_id, exc)
                rider.current_order = None
                rider.path = []
                rider.state = 0

    def all_delivered(self):
        """True if every order has been completed."""
        orders = globalvar.get_value("orders") or []
        return all(o.is_completed() for o in orders)

    def check_consistency(self):
        """Fix and report inconsistencies. Returns list of messages."""
        issues = []
        ids = set()
        for r in self.riders:
            if r.rider_id in ids:
                issues.append("Duplicate rider id %s" % r.rider_id)
            ids.add(r.rider_id)
            if r.state not in (0, 1, 2):
                issues.append("Rider %s invalid state %s; reset to idle"
                              % (r.rider_id, r.state))
                r.state = 0
                r.current_order = None
            if r.state == 0 and r.current_order is not None:
                r.current_order = None
        return issues

    # ---------------------------------------------------------- persistence
    def save(self, path=DEFAULT_SAVE_FILE):
        """Save system state to a JSON file."""
        orders = globalvar.get_value("orders") or []
        data = {
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter", 0),
            "time": self.time,
            "config": self.config,
            "orders": [o.to_dict() for o in orders],
            "riders": [r.to_dict() for r in self.riders],
        }
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        return path

    def load(self, path=DEFAULT_SAVE_FILE):
        """Load system state from a JSON file."""
        with open(path, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        init_system(data.get("config"))
        globalvar.set_value("funds", int(data.get("funds", 0)))
        globalvar.set_value("rider_id_counter",
                            int(data.get("rider_id_counter", 0)))
        globalvar.set_value("time", int(data.get("time", 0)))
        orders = [Order.from_dict(d) for d in data.get("orders", [])]
        globalvar.set_value("orders", orders)
        by_id = {o.id: o for o in orders}
        max_id = max(by_id) if by_id else 0
        order_model.reset_order_ids(max_id + 1)
        riders = [Rider.from_dict(d, by_id) for d in data.get("riders", [])]
        globalvar.set_value("riders", riders)
        return path

"""Status display and ASCII map visualisation."""

from config import MAP_CHARS, MAP_SIZE
from order_model import STATUS_COMPLETED

SCALE = MAP_SIZE // MAP_CHARS  # 50 units per character


def _cell(v):
    return max(0, min(MAP_CHARS - 1, int(v) // SCALE))


class DisplayManager:
    """Builds textual views of the system state."""

    @staticmethod
    def render_map(riders, orders, route=None):
        """Return a 16x16 ASCII map.

        Symbols: 'R' rider, 'S' restaurant, 'C' customer, '#' street,
        '.' empty, '*' route.
        """
        grid = []
        for row in range(MAP_CHARS):
            line = []
            for col in range(MAP_CHARS):
                line.append("#" if row % 2 == 0 or col % 2 == 0 else ".")
            grid.append(line)
        if route:
            for a, b in zip(route, route[1:]):
                dist = abs(a[0] - b[0]) + abs(a[1] - b[1])
                steps = max(1, dist // 10)
                for i in range(steps + 1):
                    x = a[0] + (b[0] - a[0]) * i / steps
                    y = a[1] + (b[1] - a[1]) * i / steps
                    grid[_cell(y)][_cell(x)] = "*"
        for o in orders:
            if o.status == STATUS_COMPLETED:
                continue
            grid[_cell(o.res_pos[1])][_cell(o.res_pos[0])] = "S"
            grid[_cell(o.cus_pos[1])][_cell(o.cus_pos[0])] = "C"
        for r in riders:
            grid[_cell(r.y)][_cell(r.x)] = "R"
        header = "    " + "".join("%-2d" % (c * 2 % 100 // 2 % 10)
                                  for c in range(0, MAP_CHARS, 2))
        lines = ["    " + "".join(str(c % 10) for c in range(MAP_CHARS))]
        del header
        for i, line in enumerate(grid):
            lines.append("%3d " % (i * SCALE) + "".join(line))
        lines.append("Legend: R=rider S=restaurant C=customer #=street "
                     ".=empty *=route (1 char = %d units)" % SCALE)
        return "\n".join(lines)

    @staticmethod
    def render_status(funds, riders, time):
        """System snapshot: funds and each rider's state/orders."""
        lines = ["Time: %d" % time, "Funds: %d" % funds,
                 "Riders: %d" % len(riders)]
        for r in riders:
            ids = [str(getattr(o, "id", "?")) for o in r.orders]
            active = [str(o.id) for o in r.active_orders()]
            lines.append(
                "  Rider %d @ (%d, %d) speed=%d state=%s orders=%d "
                "[all: %s] [active: %s]" % (
                    r.rider_id, r.x, r.y, r.speed, r.state_name,
                    len(r.orders), ", ".join(ids) or "-",
                    ", ".join(active) or "-"))
        return "\n".join(lines)

    @staticmethod
    def render_orders(orders):
        """Order detail table."""
        if not orders:
            return "No orders."
        lines = ["%-4s %-6s %-12s %-12s %-6s %-6s %-10s %-6s" % (
            "ID", "Start", "Restaurant", "Customer", "Dist", "Taken",
            "Status", "Rider")]
        for o in orders:
            lines.append("%-4s %-6s %-12s %-12s %-6s %-6s %-10s %-6s" % (
                o.id, o.created_time,
                "(%d,%d)" % o.res_pos, "(%d,%d)" % o.cus_pos,
                o.distance,
                "-" if o.time_taken is None else o.time_taken,
                o.status, "-" if o.rider_id is None else o.rider_id))

    @staticmethod
    def render_rider_logs(riders):

            return "No riders."
        lines = []
        for r in riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:

                    "-" if e.get("order_id") is None else e.get("order_id"),

                rider.rider_id,
                " -> ".join("(%d,%d)" % (p[0], p[1]) for p in route)),
            "Speed: %d units/tick, route length: %d units, "
            "estimated time: %d ticks" % (rider.speed, length, eta),
            DisplayManager.render_map(riders, orders, route),
        ]
        return "\n".join(lines)

"""Command parsing and input validation."""

from config import INPUT_MAX, INPUT_MIN

ALIASES = {
    "start": "start", "s": "start", "order": "start", "new": "start",
    "status": "status", "st": "status",
    "orders": "orders", "o": "orders",
    "riders": "riders", "r": "riders",
    "map": "map", "m": "map",
    "tick": "tick", "t": "tick",
    "run": "run",
    "save": "save", "load": "load",
    "history": "history", "hist": "history",
    "help": "help", "h": "help", "?": "help",
    "quit": "quit", "q": "quit", "exit": "quit",
}


class InputManager:
    """Parses commands, validates input and keeps command history."""

    def __init__(self, max_history=500):
        self.history = []
        self.max_history = max_history

    def record(self, line):
        """Add a line to the command history (bounded)."""
        if line.strip():
            self.history.append(line.strip())
            if len(self.history) > self.max_history:
                del self.history[0]

    def parse_command(self, line):
        """Return ``(command, args)``; command is None for empty input.

        ``!N`` repeats history entry N; ``!!`` repeats the last command.
        Unknown commands are returned as ``("unknown", [word])``.
        """
        line = (line or "").strip()
        if not line:
            return None, []
        if line == "!!" and self.history:
            line = self.history[-1]
        elif line.startswith("!") and line[1:].isdigit():
            idx = int(line[1:]) - 1
            if 0 <= idx < len(self.history):
                line = self.history[idx]
        parts = line.replace(",", " ").split()
        cmd = ALIASES.get(parts[0].lower())
        self.record(line)
        if cmd is None:
            return "unknown", [parts[0]]
        return cmd, parts[1:]

    @staticmethod
    def validate_coordinates(args):
        """Validate four integer coordinates in range 0-1000.

        Raises ValueError with a friendly message on failure.
        """
        if isinstance(args, str):
            args = args.replace(",", " ").replace("[", " ") \
                .replace("]", " ").split()
        args = [a for a in args if str(a).strip() not in ("[", "]")]
        cleaned = []
        for a in args:
            s = str(a).strip().strip("[]")
            if s:
                cleaned.append(s)
        if len(cleaned) != 4:
            raise ValueError("Please enter exactly 4 integers: "
                             "restaurant_x restaurant_y customer_x customer_y")
        values = []
        for s in cleaned:
            try:
                v = int(s)
            except ValueError:
                raise ValueError("'%s' is not an integer." % s)
            if not INPUT_MIN <= v <= INPUT_MAX:
                raise ValueError("Coordinate %d out of range (%d-%d)."
                                 % (v, INPUT_MIN, INPUT_MAX))
            values.append(v)
        return values

    @staticmethod
    def parse_positive_int(args, default=1):
        """Parse an optional positive integer argument."""
        if not args:
            return default
        try:
            v = int(args[0])
        except ValueError:
            raise ValueError("'%s' is not an integer." % args[0])
        if v < 1:
            raise ValueError("Value must be positive.")
        return v

"""Command-line entry point for the food delivery dispatch simulator."""

import argparse
import logging
import sys

import globalvar
from config import DEFAULT_CONFIG_FILE, DEFAULT_SAVE_FILE, load_config
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from system import SystemManager

try:  # command history / line editing where available
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    readline = None

HELP = """Commands (shortcut in brackets):
  start [s]   rx ry cx cy   create an order (prompts if no coordinates)
  status [st]               system funds and rider states/orders
  orders [o]                details of all orders
  riders [r]                status-change log of all riders
  map [m]                   ASCII map of riders and active orders
  tick [t] [n]              advance simulation n ticks (default 1)
  run                       run until all orders are delivered
  save / load [file]        persist / restore state as JSON
  history                   show command history (!N or !! to repeat)
  help [h]                  this help
  quit / exit [q]           leave the program
Each command entered also advances the simulation by one tick."""


class Application:
    """Main loop tying together the managers."""

    def __init__(self, config=None, out=None):
        self.out = out or sys.stdout
        self.system = SystemManager(config)
        self.orders = OrderManager()
        self.inputs = InputManager()
        self.display = DisplayManager()

    def echo(self, text=""):
        print(text, file=self.out)

    def all_orders(self):
        return globalvar.get_value("orders") or []

    # -------------------------------------------------------------- commands
    def cmd_start(self, args):
        if not args:
            raw = input("Enter restaurant_x restaurant_y customer_x "
                        "customer_y: ")
            args = raw
        coords = self.inputs.validate_coordinates(args)
        order = self.orders.create_order(coords)
        new_riders = self.system.auto_recruit()
        rider = self.orders.assign(order)
        self.echo("Order %d created." % order.id)
        self.echo("  Restaurant grid cell: (%d, %d), center: (%d, %d)"
                  % (order.res_cell + order.res_pos))
        self.echo("  Customer   grid cell: (%d, %d), center: (%d, %d)"
                  % (order.cus_cell + order.cus_pos))
        if new_riders:
            self.echo("  New rider(s) added: %s (funds now %d)" % (
                ", ".join(str(r.rider_id) for r in new_riders),
                self.system.funds))
        else:
            self.echo("  No new rider added (funds %d)." % self.system.funds)
        if rider is None:
            self.echo("  No rider available; order is pending.")
            return
        self.echo("  Assigned to rider %d." % rider.rider_id)
        route = rider.remaining_route()
        if rider.current_order is None:
            from path_planner import PathPlanner
            stops = []
            for o in rider.active_orders():
                stops.extend([o.res_pos, o.cus_pos])
            route = PathPlanner.route(rider.position, stops)
        self.echo(self.display.render_route(
            rider, route, self.system.riders, self.all_orders()))

    def cmd_tick(self, n):
        for _ in range(n):
            self.step()
        self.echo("Advanced %d tick(s). Time is now %d."
                  % (n, self.system.time))

    def cmd_run(self):
        limit = 100000
        count = 0
        while not self.system.all_delivered() and count < limit:
            self.step()
            count += 1
        self.echo("Ran %d tick(s). Time is now %d." % (count,
                                                      self.system.time))

    def step(self):
        """One simulation tick, assigning leftover pending orders."""
        if self.orders.pending_orders():
            self.system.auto_recruit()
            self.orders.assign_pending()
        self.system.tick()

    def handle(self, line):
        """Process one command line. Returns False to quit."""
        cmd, args = self.inputs.parse_command(line)
        if cmd is None:
            return True
        if cmd == "quit":
            self.system.stop()
            return False
        try:
            if cmd == "start":
                self.cmd_start(args)
            elif cmd == "status":
                self.echo(self.display.render_status(
                    self.system.funds, self.system.riders, self.system.time))
            elif cmd == "orders":
                self.echo(self.display.render_orders(self.all_orders()))
            elif cmd == "riders":
                self.echo(self.display.render_rider_logs(self.system.riders))
            elif cmd == "map":
                self.echo(self.display.render_map(
                    self.system.riders, self.all_orders()))
            elif cmd == "tick":
                self.cmd_tick(self.inputs.parse_positive_int(args))
                return True
            elif cmd == "run":
                self.cmd_run()
                return True
            elif cmd == "save":
                path = self.system.save(args[0] if args
                                        else DEFAULT_SAVE_FILE)
                self.echo("State saved to %s." % path)
            elif cmd == "load":
                path = self.system.load(args[0] if args
                                        else DEFAULT_SAVE_FILE)
                self.echo("State loaded from %s." % path)
            elif cmd == "history":
                for i, h in enumerate(self.inputs.history, 1):
                    self.echo("%4d  %s" % (i, h))
            elif cmd == "help":
                self.echo(HELP)
            else:
                self.echo("Unknown command '%s'. Type 'help'." % args[0])
                return True
        except ValueError as exc:
            self.echo("Input error: %s" % exc)
            return True
        except (OSError, KeyError) as exc:
            self.echo("Error: %s" % exc)
            return True
        except Exception as exc:  # graceful recovery
            logging.exception("Unexpected error")
            self.echo("Unexpected error: %s (state recovered)" % exc)
            for msg in self.system.check_consistency():
                self.echo("  fixed: %s" % msg)
            return True
        self.step()
        return True

    def run(self):
        """Main interactive loop: each iteration is one tick."""
        self.echo("Food Delivery Dispatch Simulator. Type 'help'.")
        while self.system.running:
            try:
                line = input("dispatch> ")
            except (EOFError, KeyboardInterrupt):
                self.echo()
                break
            if not self.handle(line):
                break
        self.echo("Goodbye.")


def main(argv=None):
    """Program entry point."""
    parser = argparse.ArgumentParser(description="Food delivery dispatch "
                                                 "simulator")
    parser.add_argument("--config", default=DEFAULT_CONFIG_FILE)
    parser.add_argument("--debug", action="store_true")
    ns = parser.parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if ns.debug else logging.WARNING,
                        format="%(levelname)s %(name)s: %(message)s")
    Application(load_config(ns.config)).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Data loading utilities for the Beijing Subway fare calculation system.

The network data lives in ``subway_data.json`` next to this module.  The
path is computed from this file's location, so it is valid no matter what
the current working directory is.
"""

import json
import os

#: Absolute path of the bundled subway data file (src/subway_data.json).
DATA_FILE_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "subway_data.json")


def load_subway_data(path=DATA_FILE_PATH):
    """Load and return the JSON content of the subway data file.

    :param path: path of the JSON file (defaults to :data:`DATA_FILE_PATH`).
    :return: dict with keys ``stations`` and ``connections``.
    :raises FileNotFoundError: if the file does not exist.
    :raises ValueError: if the file is not valid JSON.
    """
    if path is None:
        path = DATA_FILE_PATH
    with open(path, "r", encoding="utf-8") as fh:
        try:
            data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid JSON in data file '{path}': {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError(f"Data file '{path}' must contain a JSON object")
    data.setdefault("stations", [])
    data.setdefault("connections", [])
    return data

from dataclasses import dataclass, field
from typing import List


class Edge:
    """A directed, weighted edge between two stations (by station number)."""

    __slots__ = ("source", "dest", "distance", "weight", "line")

    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}m, {self.line})"

    def __eq__(self, other):
        if not isinstance(other, Edge):
            return NotImplemented
        return (self.source, self.dest, self.distance, self.line) == (
            other.source, other.dest, other.distance, other.line)

    def __hash__(self):
        return hash((self.source, self.dest, self.distance, self.line))


class Graph:
    """Weighted directed graph with adjacency lists.

    Station numbers are assigned by the caller; the graph simply stores the
    edges it is given.  Each edge is usable only in its listed direction.
    """

    def __init__(self):
        self.adj_list = {}          # station number -> list[Edge]
        self.station_lines = {}     # station name -> list[str]
        self.edge_count = 0

    # ---- vertices -------------------------------------------------------
    def add_vertex(self, v):
        if v not in self.adj_list:
            self.adj_list[v] = []

    @property
    def vertex_count(self):
        return len(self.adj_list)

    def V(self):
        return self.vertex_count

    def E(self):
        return self.edge_count

    def vertices(self):
        return list(self.adj_list.keys())

    def has_vertex(self, v):
        return v in self.adj_list

    # ---- edges ----------------------------------------------------------
    def add_edge(self, edge):
        self.add_vertex(edge.source)
        self.add_vertex(edge.dest)
        self.adj_list[edge.source].append(edge)
        self.edge_count += 1

    def adj(self, v):
        return self.adj_list.get(v, [])

    def edges(self):
        for lst in self.adj_list.values():
            for e in lst:
                yield e

    def remove_edges(self, source, dest, line=None):
        lst = self.adj_list.get(source, [])
        keep = [e for e in lst if not (e.dest == dest and (line is None or e.line == line))]
        self.edge_count -= len(lst) - len(keep)
        if source in self.adj_list:
            self.adj_list[source] = keep

    # ---- station metadata ----------------------------------------------
    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines) if lines else []

    def get_station_lines(self, station_name):
        return self.station_lines.get(station_name, [])

    def is_transfer_station(self, station_name):
        return len(self.station_lines.get(station_name, [])) > 1

"""Beijing Subway fare calculation.

Fare structure (distance in kilometres, right-closed intervals):
    0  - 6   : 3 yuan
    6  - 12  : 4 yuan
    12 - 22  : 5 yuan
    22 - 32  : 6 yuan
    > 32     : +1 yuan for every additional 20 km (rounded up)
"""

import math

# (upper bound in meters, fare)
FARE_TIERS = [
    (6000, 3),
    (12000, 4),
    (22000, 5),
    (32000, 6),
]
EXTENDED_BASE_DISTANCE = 32000
EXTENDED_BASE_FARE = 6
EXTENDED_INCREMENT_DISTANCE = 20000
EXTENDED_INCREMENT_FARE = 1

MINUTES_PER_KM = 1.5
MINUTES_PER_TRANSFER = 5.0


def Sum_money(distance):
    """Return the fare (yuan, int) for a travel distance given in meters."""
    if distance is None:
        raise ValueError("Distance must not be None")
    try:
        distance = float(distance)
    except (TypeError, ValueError):
        raise ValueError(f"Invalid distance: {distance!r}")
    if math.isnan(distance) or math.isinf(distance) or distance < 0:
        raise ValueError(f"Invalid distance: {distance!r}")
    for upper, fare in FARE_TIERS:
        if distance <= upper:
            return fare
    extra = math.ceil((distance - EXTENDED_BASE_DISTANCE) / EXTENDED_INCREMENT_DISTANCE)
    return EXTENDED_BASE_FARE + extra * EXTENDED_INCREMENT_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimated travel time in minutes: 1.5 min/km plus 5 min per transfer."""
    if distance is None or distance < 0:
        raise ValueError(f"Invalid distance: {distance!r}")
    transfer_count = transfer_count or 0
    if transfer_count < 0:
        raise ValueError(f"Invalid transfer count: {transfer_count!r}")
    return distance / 1000 * MINUTES_PER_KM + transfer_count * MINUTES_PER_TRANSFER

"""Dijkstra-based path finding with transfer awareness."""

import heapq
import itertools

TRANSFER_PENALTY_METERS = 3000      # composite score penalty per transfer
PREFERRED_LINE_FACTOR = 0.7         # weight multiplier for preferred lines

MODES = ("shortest_distance", "least_transfer", "comprehensive")


class PathfindingConfig:
    """Configuration for path queries."""

    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode or "shortest_distance"
        self.max_transfers = max_transfers
        if isinstance(preferred_lines, str):
            preferred_lines = [p.strip() for p in preferred_lines.split(",") if p.strip()]
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Binary-heap minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, key, item):
        heapq.heappush(self._heap, (key, next(self._counter), item))

    def pop(self):
        key, _, item = heapq.heappop(self._heap)
        return key, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)

    @property
    def size(self):
        return len(self._heap)


class _DistDict(dict):
    """dict returning infinity for unreachable stations."""

    def __missing__(self, key):
        return float("inf")


class ShortestPath:
    """Single-source path computation over a :class:`models.Graph`.

    The search runs over states (station, line of arrival[, transfers]) so
    transfers can be counted and limited correctly.
    """

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        if self.config.mode not in MODES:
            raise ValueError(f"Unknown mode: {self.config.mode}")
        self.dist_to = _DistDict()
        self.transfers_to = {}
        self._best_state = {}
        self._parent = {}
        self._run()

    def _key(self, weighted, transfers, actual):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, weighted, actual)
        if mode == "comprehensive":
            return (weighted + transfers * TRANSFER_PENALTY_METERS, transfers, actual)
        return (weighted, transfers, actual)

    def _run(self):
        cfg = self.config
        track = cfg.max_transfers is not None
        preferred = set(cfg.preferred_lines)
        start = (self.source, None, 0 if track else None)
        info = {start: (0, 0.0, 0)}  # state -> (actual, weighted, transfers)
        best_key = {start: self._key(0.0, 0, 0)}
        self._parent[start] = (None, None)
        pq = MinPQ()
        pq.push(best_key[start], start)
        settled = set()
        while not pq.is_empty():
            key, state = pq.pop()
            if state in settled or key != best_key.get(state):
                continue
            settled.add(state)
            node, line, _ = state
            actual, weighted, transfers = info[state]
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = actual
                self.transfers_to[node] = transfers
            for e in self.graph.adj(node):
                nt = transfers + (1 if line is not None and e.line != line else 0)
                if track and nt > cfg.max_transfers:
                    continue
                factor = PREFERRED_LINE_FACTOR if (preferred and e.line in preferred) else 1.0
                nw = weighted + e.weight * factor
                na = actual + e.weight
                nstate = (e.dest, e.line, nt if track else None)
                if nstate in settled:
                    continue
                nkey = self._key(nw, nt, na)
                if nstate not in best_key or nkey < best_key[nstate]:
                    best_key[nstate] = nkey
                    info[nstate] = (na, nw, nt)
                    self._parent[nstate] = (state, e)
                    pq.push(nkey, nstate)

    def has_path_to(self, target):
        return target in self._best_state

    def distance_to(self, target):
        return self.dist_to[target]

    def path_with_edges(self, target):
        """Return (station numbers from source to target, edges used)."""
        if target not in self._best_state:
            return [], []
        nodes, edges = [], []
        state = self._best_state[target]
        while state is not None:
            nodes.append(state[0])
            prev, edge = self._parent[state]
            if edge is not None:
                edges.append(edge)
            state = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                count += 1
        return count

"""Subway network data management: building the graph, updates, integrity checks."""

import difflib
import json
from dataclasses import dataclass, field
from typing import List

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

MAIN_LINES = {
    "Line 1": ["Line 1", "1号线"],
    "Line 2": ["Line 2", "2号线"],
    "Line 4": ["Line 4", "4号线"],
    "Line 5": ["Line 5", "5号线"],
    "Line 6": ["Line 6", "6号线"],
    "Line 8": ["Line 8", "8号线"],
    "Line 10": ["Line 10", "10号线"],
    "Line 13": ["Line 13", "13号线"],
    "Line 14": ["Line 14", "14号线"],
    "Line 15": ["Line 15", "15号线"],
    "Changping Line": ["Changping", "昌平线"],
    "Yizhuang Line": ["Yizhuang", "亦庄线"],
    "Fangshan Line": ["Fangshan", "房山线"],
    "Airport Line": ["Airport", "机场线"],
}

MAX_REASONABLE_DISTANCE = 50000


@dataclass
class DataIntegrityCheck:
    station_coverage_valid: bool = True
    connection_integrity_valid: bool = True
    distance_accuracy_valid: bool = True
    messages: List[str] = field(default_factory=list)

    @property
    def valid(self):
        return (self.station_coverage_valid and self.connection_integrity_valid
                and self.distance_accuracy_valid)


class SubwayNetwork:
    def __init__(self, data_source_file=DATA_FILE_PATH, data=None):
        self.data_source_file = data_source_file
        if data is None:
            data = load_subway_data(data_source_file)
        self.stations = []          # list of {"name", "lines"}
        self.connections = []       # list of {"source","dest","distance","line"}
        self.station_to_number = {}
        self.number_to_station = {}
        self._load(data)
        self.build_graph()

    # ---- loading ---------------------------------------------------------
    def _register_station(self, name, lines=None):
        if name not in self.station_to_number:
            num = len(self.station_to_number)
            self.station_to_number[name] = num
            self.number_to_station[num] = name
            self.stations.append({"name": name, "lines": list(lines or [])})
        elif lines:
            for st in self.stations:
                if st["name"] == name:
                    for ln in lines:
                        if ln not in st["lines"]:
                            st["lines"].append(ln)
                    break

    def _load(self, data):
        for st in data.get("stations", []):
            name = st.get("name")
            if name:
                self._register_station(name, st.get("lines", []))
        for c in data.get("connections", []):
            self.connections.append(dict(c))
            for key in ("source", "dest"):
                if c.get(key) is not None and c.get(key) not in self.station_to_number:
                    self._register_station(c[key], [c["line"]] if c.get("line") else [])

    def build_graph(self):
        g = Graph()
        for st in self.stations:
            g.add_vertex(self.station_to_number[st["name"]])
            g.set_station_lines(st["name"], st["lines"])
        for c in self.connections:
            try:
                s = self.station_to_number[c["source"]]
                d = self.station_to_number[c["dest"]]
                dist = c["distance"]
            except (KeyError, TypeError):
                continue
            g.add_edge(Edge(s, d, dist, c.get("line")))
        self.graph = g
        # adjacency dictionary representation (station number -> [(dest, distance, line)])
        self.graph_dict = {v: [(e.dest, e.distance, e.line) for e in g.adj(v)] for v in g.vertices()}
        return g

    # ---- lookup ----------------------------------------------------------
    def station_names(self):
        return list(self.station_to_number.keys())

    def resolve_station(self, name):
        """Return (resolved name or None, exact match flag)."""
        name = (name or "").strip()
        if not name:
            return None, False
        if name in self.station_to_number:
            return name, True
        lowered = {n.lower(): n for n in self.station_to_number}
        if name.lower() in lowered:
            return lowered[name.lower()], True
        for suffix in ("站", " Station", " station"):
            if name.endswith(suffix) and name[: -len(suffix)] in self.station_to_number:
                return name[: -len(suffix)], True
        matches = difflib.get_close_matches(name, self.station_names(), n=1, cutoff=0.6)
        if matches:
            return matches[0], False
        return None, False

    def all_lines(self):
        lines = []
        for st in self.stations:
            for ln in st["lines"]:
                if ln not in lines:
                    lines.append(ln)
        for c in self.connections:
            ln = c.get("line")
            if ln and ln not in lines:
                lines.append(ln)
        return lines

    # ---- updates ---------------------------------------------------------
    def add_line(self, line_name, station_sequence, distances, bidirectional=True):
        if len(distances) != len(station_sequence) - 1:
            raise ValueError("distances must have exactly len(stations) - 1 entries")
        for name in station_sequence:
            self._register_station(name, [line_name])
        for a, b, d in zip(station_sequence, station_sequence[1:], distances):
            if d is None or d <= 0:
                raise ValueError(f"Invalid distance between {a} and {b}: {d}")
            self.connections.append({"source": a, "dest": b, "distance": d, "line": line_name})
            if bidirectional:
                self.connections.append({"source": b, "dest": a, "distance": d, "line": line_name})
        self.build_graph()

    def update_station(self, name, new_name=None, lines=None):
        if name not in self.station_to_number:
            raise KeyError(f"Station '{name}' does not exist")
        for st in self.stations:
            if st["name"] == name:
                if lines is not None:
                    st["lines"] = list(lines)
                if new_name and new_name != name:
                    if new_name in self.station_to_number:
                        raise ValueError(f"Station '{new_name}' already exists")
                    st["name"] = new_name
                    num = self.station_to_number.pop(name)
                    self.station_to_number[new_name] = num
                    self.number_to_station[num] = new_name
                    for c in self.connections:
                        if c.get("source") == name:
                            c["source"] = new_name
                        if c.get("dest") == name:
                            c["dest"] = new_name
                break
        self.build_graph()

    def update_distance(self, source, dest, distance, line=None, bidirectional=True):
        if distance is None or distance <= 0:
            raise ValueError(f"Invalid distance: {distance}")
        updated = 0
        for c in self.connections:
            match = (c.get("source") == source and c.get("dest") == dest) or \
                    (bidirectional and c.get("source") == dest and c.get("dest") == source)
            if match and (line is None or c.get("line") == line):
                c["distance"] = distance
                updated += 1
        if not updated:
            raise KeyError(f"No connection between '{source}' and '{dest}'")
        self.build_graph()
        return updated

    def to_dict(self):
        return {"stations": [dict(s) for s in self.stations],
                "connections": [dict(c) for c in self.connections]}

    def save(self, path=None):
        path = path or self.data_source_file
        with open(path, "w", encoding="utf-8") as fh:
            json.dump(self.to_dict(), fh, ensure_ascii=False, indent=2)

    # ---- integrity -------------------------------------------------------
    def check_integrity(self):
        result = DataIntegrityCheck()
        lines = self.all_lines()
        for main, aliases in MAIN_LINES.items():
            found = any(any(alias.lower() == ln.lower() or
                            (len(alias) > 6 and alias.lower() in ln.lower()) or
                            (alias.endswith("线") and alias in ln)
                            for alias in aliases) for ln in lines)
            if not found:
                result.station_coverage_valid = False
                result.messages.append(f"Main line missing: {main}")
        if not self.stations:
            result.station_coverage_valid = False
            result.messages.append("No station data")

        seen = set()
        for i, c in enumerate(self.connections):
            s, d = c.get("source"), c.get("dest")
            if s not in self.station_to_number or d not in self.station_to_number:
                result.connection_integrity_valid = False
                result.messages.append(f"Connection #{i} references unknown station")
            if s == d:
                result.connection_integrity_valid = False
                result.messages.append(f"Connection #{i} is a self-loop at '{s}'")
            key = (s, d, c.get("line"))
            if key in seen:
                result.messages.append(f"Duplicate connection {s} -> {d} ({c.get('line')})")
            seen.add(key)
            dist = c.get("distance")
            if not isinstance(dist, (int, float)) or isinstance(dist, bool) or dist <= 0 \
                    or dist > MAX_REASONABLE_DISTANCE:
                result.distance_accuracy_valid = False
                result.messages.append(f"Connection #{i} {s} -> {d} has invalid distance {dist!r}")
        connected = set()
        for c in self.connections:
            connected.add(c.get("source"))
            connected.add(c.get("dest"))
        for st in self.stations:
            if st["name"] not in connected:
                result.connection_integrity_valid = False
                result.messages.append(f"Station '{st['name']}' has no connections")
        return result

"""Route summary, text and JSON report generation."""

import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return f"{int(meters)} m"
    return f"{meters / 1000:.2f} km ({int(meters)} m)"


def build_route(sp, target, number_to_station):
    nodes, edges = sp.path_with_edges(target)
    if not nodes:
        return None
    names = [number_to_station[n] for n in nodes]
    distance = sum(e.distance for e in edges)
    transfers = []
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    for a, b in zip(edges, edges[1:]):
        if a.line != b.line:
            transfers.append({"station": number_to_station[a.dest],
                              "from_line": a.line, "to_line": b.line})
    segments = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["stations"].append(number_to_station[e.dest])
        else:
            segments.append({"line": e.line,
                             "stations": [number_to_station[e.source], number_to_station[e.dest]]})
    fare = Sum_money(distance)
    return {
        "path": names,
        "distance": distance,
        "fare": fare,
        "transfer_count": len(transfers),
        "transfers": transfers,
        "lines": lines,
        "segments": segments,
        "estimated_time": round(calculate_estimated_time(distance, len(transfers)), 1),
    }


def format_text(start, end, route, mode=None):
    out = [f"Route: {start} -> {end}" + (f" (mode: {mode})" if mode else "")]
    out.append("Stations: " + " -> ".join(route["path"]))
    out.append(f"Number of stations: {len(route['path'])}")
    out.append("Segments:")
    for seg in route["segments"]:
        out.append(f"  [{seg['line']}] " + " -> ".join(seg["stations"]))
    if route["transfers"]:
        out.append("Transfers:")
        for t in route["transfers"]:
            out.append(f"  Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
    else:
        out.append("Transfers: none")
    out.append(f"Lines involved: {', '.join(str(l) for l in route['lines'])}")
    out.append(f"Total distance: {format_distance(route['distance'])}")
    out.append(f"Transfer count: {route['transfer_count']}")
    out.append(f"Estimated time: {route['estimated_time']} minutes")
    out.append(f"Fare: {route['fare']} yuan")
    return "\n".join(out)


def build_json_dict(start, end, route, mode=None):
    return {
        "query": {"start": start, "end": end, "mode": mode},
        "path": route["path"],
        "distance": route["distance"],
        "distance_km": round(route["distance"] / 1000, 3),
        "fare": route["fare"],
        "estimated_time": route["estimated_time"],
        "transfer_count": route["transfer_count"],
        "transfers": route["transfers"],
        "lines": route["lines"],
    }


def format_json(start, end, route, mode=None):
    return json.dumps(build_json_dict(start, end, route, mode), ensure_ascii=False, indent=2)

def parse_query(line):
    parts = [x.strip() for x in re.split(r"[;,；，]", line)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None
    return parts[0], parts[1]


def handle_query(network, line, config, fmt):
    """Process one query line. Returns (output string, result object or None)."""
    q = parse_query(line)
    if q is None:
        return FORMAT_ERROR, None
    notes = []
    resolved = []
    for label, name in (("Starting", q[0]), ("Ending", q[1])):
        r, exact = network.resolve_station(name)
        if r is None:
            return f"{label} station '{name}' does not exist, please check the station name", None
        if not exact:
            notes.append(f"Note: {label.lower()} station '{name}' matched to '{r}'")
        resolved.append(r)
    start, end = resolved
    if start == end:
        return "Starting station cannot be the same as the ending station", None
    try:
        sp = ShortestPath(network.graph, network.station_to_number[start], config)
        route = build_route(sp, network.station_to_number[end], network.number_to_station)
    except ValueError as exc:
        return f"Fare calculation error: {exc}", None
    except Exception as exc:  # noqa: BLE001
        return f"Path calculation error: {exc}", None
    if route is None:
        return f"No route found from '{start}' to '{end}' under the current settings", None
    if fmt == "json":
        obj = build_json_dict(start, end, route, config.mode)
        body = json.dumps(obj, ensure_ascii=False, indent=2)
    else:
        obj = format_text(start, end, route, config.mode)
        body = obj
    return "\n".join(notes + [body]), obj


def write_output(path, results, fmt):
    try:
        with open(path, "w", encoding="utf-8") as fh:
            if fmt == "json":
                data = results[0] if len(results) == 1 else results
                json.dump(data, fh, ensure_ascii=False, indent=2)
            else:
                fh.write(("\n\n" + "-" * 40 + "\n\n").join(results) + "\n")
    except OSError as exc:
        print(f"Failed to write output file '{path}': {exc}")


def main(argv=None):
    args = parse_args(argv)
    try:
        network = SubwayNetwork(args.data_file)
    except (OSError, ValueError) as exc:
        print(f"Failed to load subway data: {exc}")
        return 1

    if args.check_data:
        res = network.check_integrity()
        print("Data integrity check")
        print(f"  Stations: {len(network.stations)}, connections: {len(network.connections)}")
        print(f"  Station coverage: {'OK' if res.station_coverage_valid else 'FAILED'}")
        print(f"  Connection integrity: {'OK' if res.connection_integrity_valid else 'FAILED'}")
        print(f"  Distance accuracy: {'OK' if res.distance_accuracy_valid else 'FAILED'}")
        for m in res.messages:
            print("  - " + m)
        return 0 if res.valid else 1

    if args.max_transfers is not None and args.max_transfers < 0:
        print("--max_transfers must be non-negative")
        return 1
    config = PathfindingConfig(args.mode, args.max_transfers, args.line_preference)
    results = []
    print("Beijing Subway Fare Calculator (type 'quit' to exit)")
    while True:
        try:
            line = input("Please enter 'Starting Station; Ending Station': ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in EXIT_WORDS:
            break
        output, obj = handle_query(network, line, config, args.format)
        print(output)
        if obj is not None and args.output_file:
            results.append(obj)
            write_output(args.output_file, results, args.format)
    return 0


if __name__ == "__main__":
    sys.exit(main())

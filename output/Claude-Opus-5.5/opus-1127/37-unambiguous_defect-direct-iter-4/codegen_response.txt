"""Core data models."""

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)

    def __str__(self) -> str:
        avoid = ", ".join(self.avoid_groups) if self.avoid_groups else "None"
        drawer = self.associated_drawer or "None"
        return (f"{self.name} x{self.quantity} | Avoid groups: {avoid} | "
                f"Drawer: {drawer}")

from typing import Dict, Iterable, List, Tuple

DEFAULT_LIST_PATH = os.path.join(
    os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
    "data", "default_list.txt")


class ListManager:
    """Handles loading and validation of participant lists."""

    SEPARATORS = (",", "，", "\t")

    def __init__(self):
        self.users: List[User] = []

    # ------------------------------------------------------------------ load
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        path = DEFAULT_LIST_PATH
        if not os.path.isfile(path):
            try:
                os.makedirs(os.path.dirname(path), exist_ok=True)
                with open(path, "w", encoding="utf-8") as f:
                    f.write(DEFAULT_LIST_CONTENT)
            except OSError:
                return self._parse_lines(DEFAULT_LIST_CONTENT.splitlines())
        return self._load_from_path(path)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File not found: {path}"]
        if not str(path).lower().endswith(".txt"):
            return [], [f"Unsupported file type (only .txt allowed): {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.read().splitlines()
        except FileNotFoundError:
            return [], [f"File not found: {path}"]
        except (OSError, UnicodeDecodeError) as exc:
            return [], [f"Unable to read file {path}: {exc}"]
        return self._parse_lines(lines)

    def _parse_lines(self, lines: Iterable[str]) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = None
            for sep in self.SEPARATORS:
                if sep in line:
                    parts = [p.strip() for p in line.split(sep)]
                    break
            if parts is None or len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department', "
                    f"got '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    # -------------------------------------------------------------- validate
    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        duplicates: List[User] = []
        for user in users:
            key = user.name.strip().lower()
            if key in seen:
                duplicates.append(user)
            else:
                seen.add(key)
                valid.append(user)
        return valid, duplicates

    # ---------------------------------------------------------------- groups
    def get_group_info(self, users) -> Dict[str, int]:
        info: "OrderedDict[str, int]" = OrderedDict()
        for user in users:
            info[user.department] = info.get(user.department, 0) + 1
        return dict(info)

"""Prize configuration management."""

from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
    ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
    ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
]


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty, avoid, drawer in DEFAULT_PRIZES:
            try:
                self.add_prize(name, qty, list(avoid), drawer)
            except ValueError as exc:
                errors.append(str(exc))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None,
                  associated_drawer=None) -> Prize:
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError(f"Invalid quantity for prize '{name}': {quantity}")
        if qty <= 0:
            raise ValueError(f"Quantity for prize '{name}' must be positive")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = []
        for g in avoid_groups or []:
            g = str(g).strip()
            if g and g not in groups:
                groups.append(g)
        drawer = associated_drawer.strip() if isinstance(associated_drawer, str) else None
        prize = Prize(name, qty, groups, drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def get_prize(self, name) -> Optional[Prize]:
        for p in self.prizes:
            if p.name == name:
                return p
        return None

    def set_avoid_groups(self, prize_name, groups) -> bool:
        prize = self.get_prize(prize_name)
        if prize is None:
            return False
        prize.avoid_groups = []
        for g in groups or []:
            g = str(g).strip()
            if g and g not in prize.avoid_groups:
                prize.avoid_groups.append(g)
        return True

    def reorder_prizes(self, order) -> bool:
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        n = len(self.prizes)
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

"""Lottery rule engine and round execution."""

import random
from typing import Callable, List, Optional

from lottery_system.models import Prize, User

SPECIAL_DRAWER = "Li Bin"
FALLBACK_WATCH_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "Cheng Ru group"


def _norm(text) -> str:
    return " ".join(str(text or "").strip().lower().split())


def is_protected_group(department) -> bool:
    d = _norm(department)
    if d.endswith(" group"):
        d = d[:-len(" group")].strip()
    return d == "cheng ru"


def ordinal(n: int) -> str:
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"


class LotteryManager:
    def __init__(self, users, rng: Optional[random.Random] = None):
        self.users: List[User] = list(users)
        self.winner_names = set()
        self.last_two_draw_results: List[bool] = []
        self.rng = rng or random.Random()

    # ------------------------------------------------------------ rules
    def _is_winner(self, user) -> bool:
        return _norm(user.name) in self.winner_names

    def fallback_active(self, prize) -> bool:
        return (_norm(prize.associated_drawer) == _norm(FALLBACK_TARGET_DRAWER)
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize) -> List[User]:
        non_winners = [u for u in self.users if not self._is_winner(u)]
        if self.fallback_active(prize):
            protected = [u for u in non_winners if is_protected_group(u.department)]
            if protected:
                return protected
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            return non_winners
        avoid = {_norm(g) for g in (prize.avoid_groups or [])}
        return [u for u in non_winners if _norm(u.department) not in avoid]

    def describe_rules(self, prize) -> str:
        lines = ["Previous winners are excluded (no repeated winning)."]
        if _norm(prize.associated_drawer) == _norm(SPECIAL_DRAWER):
            lines.append(f"Special drawer {SPECIAL_DRAWER}: relaxed filtering, "
                         "avoidance groups are ignored.")
        elif prize.avoid_groups:
            lines.append("Avoidance groups excluded: " + ", ".join(prize.avoid_groups))
        else:
            lines.append("No avoidance groups.")
        if self.fallback_active(prize):
            lines.append(f"Fallback active: pool limited to '{PROTECTED_GROUP}' "
                         "members (overrides avoidance).")
        return " ".join(lines)

    def mark_winner(self, user) -> None:
        self.winner_names.add(_norm(user.name))

    def draw_one(self, prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = self.rng.choice(pool)
        self.mark_winner(user)
        return user

    def update_fallback_state(self, prize, winners) -> None:
        drawer = _norm(prize.associated_drawer)
        if drawer in {_norm(d) for d in FALLBACK_WATCH_DRAWERS}:
            self.last_two_draw_results.append(
                any(is_protected_group(u.department) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == _norm(FALLBACK_TARGET_DRAWER):
            self.last_two_draw_results = []

    # ------------------------------------------------------------ rounds
    def run_round(self, prize: Prize,
                  input_func: Optional[Callable[[str], str]] = None,
                  output: Callable[[str], None] = print,
                  on_winner: Optional[Callable[[User, Prize], None]] = None):
        """Run one prize round. Returns (winners, terminated)."""
        output("=" * 60)
        output(f"Prize: {prize.name} | Quantity: {prize.quantity} | "
               f"Drawer: {prize.associated_drawer or 'None'}")
        output(f"Rules: {self.describe_rules(prize)}")
        output("=" * 60)
        winners: List[User] = []
        terminated = False
        paused = False
        for i in range(1, prize.quantity + 1):
            if input_func is not None:
                while True:
                    try:
                        prompt = ("[PAUSED] Type 'continue' or 'terminate': " if paused
                                  else f"[{i}/{prize.quantity}] Enter to draw, "
                                       "or 'pause'/'terminate': ")
                        cmd = input_func(prompt).strip().lower()
                    except EOFError:
                        cmd = "terminate"
                    if cmd == "terminate":
                        terminated = True
                        break
                    if paused:
                        if cmd == "continue":
                            paused = False
                            output("Lottery resumed.")
                            break
                        output("Lottery is paused.")
                    else:
                        if cmd in ("", "draw", "continue"):
                            break
                        if cmd == "pause":
                            paused = True
                            output("Lottery paused.")
                        else:
                            output(f"Unknown command: {cmd}")
            if terminated:
                output(f"Round '{prize.name}' terminated and skipped.")
                break
            output(f"Currently drawing the {ordinal(i)} winner...")
            user = self.draw_one(prize)
            if user is None:
                output("No eligible candidates remain for this prize.")
                break
            winners.append(user)
            if on_winner:
                on_winner(user, prize)
            output(f"Congratulations {user.name} ({user.department}) "
                   f"for winning {prize.name}")
        self.update_fallback_state(prize, winners)
        output(f"--- Results of round '{prize.name}' ---")
        if winners:
            for n, u in enumerate(winners, 1):
                output(f"{n}. {u.name} ({u.department})")
        else:
            output("No winners in this round.")
        return winners, terminated

"""Result management and export."""

import csv
import os
import re
from datetime import datetime
from typing import Dict, List

LEVEL_KEYWORDS = [
    (r"\b(grand|special)\b", 0),
    (r"\bfirst\b", 1),
    (r"\bsecond\b", 2),
    (r"\bthird\b", 3),
    (r"\bfourth\b", 4),
    (r"\bfifth\b", 5),
    (r"\bsixth\b", 6),
]


def _prize_name(prize) -> str:
    return prize.name if hasattr(prize, "name") else str(prize)


def prize_level(name: str) -> int:
    low = name.lower()
    for pattern, level in LEVEL_KEYWORDS:
        if re.search(pattern, low):
            return level
    return 100


class Winner:
    def __init__(self, user, prize, sequence: int):
        self.user = user
        self.prize = prize
        self.prize_name = _prize_name(prize)
        self.sequence = sequence
        self.win_time = datetime.now()

    def __repr__(self):
        return f"Winner({self.user!r}, {self.prize_name!r})"


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize) -> Winner:
        w = Winner(user, prize, len(self.winners) + 1)
        self.winners.append(w)
        return w

    def _ordered_prize_names(self, prizes) -> List[str]:
        names = []
        for p in prizes or []:
            n = _prize_name(p)
            if n not in names:
                names.append(n)
        for w in self.winners:
            if w.prize_name not in names:
                names.append(w.prize_name)
        total = len(names)
        indexed = list(enumerate(names))
        # higher level first; unknown-level ties: later-drawn considered higher
        indexed.sort(key=lambda t: (prize_level(t[1]), total - t[0]))
        return [n for _, n in indexed]

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for name in self._ordered_prize_names(prizes):
            result[name] = sorted(
                [w for w in self.winners if w.prize_name == name],
                key=lambda w: w.sequence)
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export."
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["No.", "Name", "Department", "Prize", "Win Time"])
                n = 0
                for prize_name, winners in self.get_winners_by_prize(prizes).items():
                    for w in winners:
                        n += 1
                        writer.writerow([n, w.user.name, w.user.department,
                                         prize_name,
                                         w.win_time.strftime("%Y-%m-%d %H:%M:%S")])
        except OSError as exc:
            return False, f"Export failed: {exc}"
        return True, f"Results successfully exported to {path}"

"""Command-line interface for the lottery system."""

from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


class LotteryApp:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users = []
        self.lottery = None

    def ask(self, prompt):
        try:
            return input(prompt)
        except EOFError:
            raise SystemExit(0)

    # ---------------------------------------------------------------- lists
    def _accept_list(self, users, errors):
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Problems detected in the list:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate name: {d.name} ({d.department})")
            choice = self.ask("Remove invalid data and continue (r) or terminate upload (t)? ").strip().lower()
            if choice != "r":
                print("Upload terminated.")
                return
        if not valid:
            print("No valid users loaded.")
            return
        self.users = valid
        self.lottery = None
        self.result_manager = ResultManager()
        print(f"Loaded {len(valid)} valid users.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        self._accept_list(users, errors)

    def load_custom(self):
        path = self.ask("Enter txt file path (each line: Name,Department): ").strip()
        users, errors = self.list_manager.load_custom_list(path)
        if not users and errors:
            for e in errors:
                print("  " + e)
            print("Upload failed.")
            return
        self._accept_list(users, errors)

    def view_groups(self):
        info = self.list_manager.get_group_info(self.users)
        if not info:
            print("No list loaded.")
            return
        print("Current groups:")
        for i, (g, c) in enumerate(info.items(), 1):
            print(f"  {i}. {g}: {c} people")

    # --------------------------------------------------------------- prizes
    def _select_groups(self):
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        raw = self.ask("Avoidance groups (numbers or names, comma-separated, blank for none): ").strip()
        result = []
        for part in raw.replace("，", ",").split(","):
            part = part.strip()
            if not part:
                continue
            if part.isdigit() and 1 <= int(part) <= len(groups):
                result.append(groups[int(part) - 1])
            else:
                result.append(part)
        return result

    def load_default_prizes(self):
        errors = self.prize_manager.load_default_prizes()
        for e in errors:
            print("  " + e)
        print("Default prizes loaded.")

    def add_prize(self):
        name = self.ask("Prize name: ")
        qty = self.ask("Quantity: ")
        groups = self._select_groups()
        drawer = self.ask("Associated drawer (blank for none): ").strip() or None
        try:
            p = self.prize_manager.add_prize(name, qty, groups, drawer)
            print(f"Added: {p}")
        except ValueError as exc:
            print(f"Error: {exc}")

    def configure_avoid(self):
        prizes = self.view_prizes()
        if not prizes:
            return
        raw = self.ask("Select prize number: ").strip()
        if not raw.isdigit() or not 1 <= int(raw) <= len(prizes):
            print("Invalid selection.")
            return
        prize = prizes[int(raw) - 1]
        self.prize_manager.set_avoid_groups(prize.name, self._select_groups())
        print(f"Updated: {prize}")

    def reorder(self):
        prizes = self.view_prizes()
        if not prizes:
            return
        raw = self.ask("Enter new order as serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in raw.replace("，", ",").replace(" ", ",").split(",") if x.strip()]
        except ValueError:
            order = []
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.view_prizes()
        else:
            print("Invalid order.")

    def view_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p}")
        return prizes

    # -------------------------------------------------------------- lottery
    def start_lottery(self):
        if not self.users:
            print("Please load a list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        if self.lottery is None:
            self.lottery = LotteryManager(self.users)
        for prize in prizes:
            self.lottery.run_round(prize, input_func=self.ask, output=print,
                                   on_winner=self.result_manager.add_winner)
        print("All rounds completed.")
        self.show_results()

    def show_results(self):
        grouped = self.result_manager.get_winners_by_prize(self.prize_manager.get_prizes())
        print("===== Complete Winner List =====")
        any_w = False
        for name, winners in grouped.items():
            if not winners:
                continue
            any_w = True
            print(f"[{name}]")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {name}")
        if not any_w:
            print("No winners yet.")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    def run(self):
        actions = {
            "1": ("Load default list", self.load_default),
            "2": ("Upload custom txt list", self.load_custom),
            "3": ("View groups", self.view_groups),
            "4": ("Load preset prizes", self.load_default_prizes),
            "5": ("Add custom prize", self.add_prize),
            "6": ("Configure prize avoidance groups", self.configure_avoid),
            "7": ("Reorder prizes", self.reorder),
            "8": ("View prizes", self.view_prizes),
            "9": ("Start lottery", self.start_lottery),
            "10": ("View results", self.show_results),
            "11": ("Export results", self.export),
            "0": ("Exit", None),
        }
        while True:
            print("\n===== Team Building Smart Lottery System =====")
            for k, (label, _) in actions.items():
                print(f"  {k}. {label}")
            choice = self.ask("Select: ").strip()
            if choice == "0":
                print("Bye!")
                return
            action = actions.get(choice)
            if action is None:
                print("Invalid choice.")
                continue
            action[1]()


def main():
    try:
        LotteryApp().run()
    except (KeyboardInterrupt, SystemExit):
        print("\nExiting.")


if __name__ == "__main__":
    main()

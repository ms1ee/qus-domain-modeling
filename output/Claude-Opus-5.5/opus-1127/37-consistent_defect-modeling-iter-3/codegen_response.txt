"""Core data models."""
from dataclasses import dataclass, field
from typing import List, Optional


def normalize_group(name: str) -> str:
    """Normalize a group name for comparison (case/space insensitive)."""
    return "".join(str(name).lower().split())


_LEVEL_KEYWORDS = [
    ("special", 0),
    ("grand", 0),
    ("first", 1),
    ("second", 2),
    ("third", 3),
    ("fourth", 4),
    ("fifth", 5),
]

UNKNOWN_LEVEL = 99


@dataclass
class User:
    name: str
    department: str

    def __str__(self) -> str:
        return f"{self.name} ({self.department})"


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: Optional[List[str]] = field(default=None)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []
        else:
            self.avoid_groups = list(self.avoid_groups)
        if self.associated_drawer is not None:
            self.associated_drawer = str(self.associated_drawer).strip() or None

    @property
    def level(self) -> int:
        """Smaller number = higher prize level."""
        lower = self.name.lower()
        for kw, lvl in _LEVEL_KEYWORDS:
            if kw in lower:
                return lvl
        return UNKNOWN_LEVEL

    def rule_description(self) -> str:
        parts = []
        if self.associated_drawer:
            parts.append(f"Drawer: {self.associated_drawer}")
        else:
            parts.append("Drawer: (not specified)")
        if self.avoid_groups:
            parts.append("Avoidance groups: " + ", ".join(self.avoid_groups))
        else:
            parts.append("Avoidance groups: none")
        parts.append("Previous winners are excluded")
        return "; ".join(parts)

DEFAULT_LIST_LINES = [
    "Zhang San,Technology Department",
    "Li Si,Technology Department",
    "Wang Wu,Technology Department",
    "Zhao Liu,Technology Department",
    "Sun Qi,Product Department",
    "Zhou Ba,Product Department",

    "Zheng Shi,Marketing Department",

    "Chen Er,Marketing Department",
    "Chu Wei,HR Department",

    "Jiang Hao,Finance Department",
    "Shen Yue,Finance Department",
    "Han Mei,Cheng Ru Group",
    "Yang Fan,Cheng Ru Group",
    "Zhu Lin,Cheng Ru Group",
    "Qin Lan,Cheng Ru Group",
    "You Tian,Operations Department",
    "Xu Ming,Operations Department",
]

_SEPARATORS = [";", "，", "；", "\t"]


class ListManager:
    def __init__(self):
        self.users: List[User] = []

    @staticmethod
    def _parse_lines(lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for idx, raw in enumerate(lines, start=1):
            line = raw.strip().lstrip("\ufeff")
            if not line:
                continue
            normalized = line
            for sep in _SEPARATORS:
                normalized = normalized.replace(sep, ",")
            parts = [p.strip() for p in normalized.split(",")]
            if len(parts) != 2:
                errors.append(
                    f"Line {idx}: Format Error - expected 'Name,Department Group', got '{line}'"
                )
                continue
            name, dept = parts
            if not name or not dept:
                errors.append(
                    f"Line {idx}: Format Error - name or department is empty in '{line}'"
                )
                continue
            users.append(User(name, dept))
        return users, errors

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        users, errors = self._parse_lines(DEFAULT_LIST_LINES)
        return users, errors

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        if not path or not os.path.isfile(str(path)):
            return [], [f"File Error - file not found: {path}"]
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8-sig") as f:
                lines = f.readlines()
        except UnicodeDecodeError:
            try:
                with open(path, "r", encoding="gbk") as f:
                    lines = f.readlines()
            except Exception as e:  # pragma: no cover
                return [], [f"File Error - cannot read file: {e}"]
        except OSError as e:
            return [], [f"File Error - cannot read file: {e}"]
        return self._parse_lines(lines)

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        duplicates: List[User] = []
        for u in users:
            if u.name in seen:
                duplicates.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, duplicates

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

"""Prize configuration management."""
from typing import List, Optional

from lottery_system.models import Prize

DEFAULT_PRIZES = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        for name, qty in DEFAULT_PRIZES:
            if any(p.name == name for p in self.prizes):
                errors.append(f"Prize '{name}' already exists, skipped")
                continue
            self.prizes.append(Prize(name, qty))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None) -> Prize:
        name = str(name).strip() if name is not None else ""
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            qty = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Prize quantity must be an integer")
        if qty <= 0:
            raise ValueError("Prize quantity must be positive")
        if any(p.name == name for p in self.prizes):
            raise ValueError(f"Prize '{name}' already exists")
        groups = [str(g).strip() for g in (avoid_groups or []) if str(g).strip()]
        prize = Prize(name, qty, groups, associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            idx = [int(i) for i in order]
        except (TypeError, ValueError):
            return False
        if len(idx) != n or sorted(idx) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in idx]
        return True

    def set_avoid_groups(self, index: int, groups) -> bool:
        if not 0 <= index < len(self.prizes):
            return False
        self.prizes[index].avoid_groups = [str(g).strip() for g in groups if str(g).strip()]
        return True

    def set_drawer(self, index: int, drawer: Optional[str]) -> bool:
        if not 0 <= index < len(self.prizes):
            return False
        self.prizes[index].associated_drawer = (drawer or "").strip() or None
        return True

"""Result management and export."""
import csv
import os
from dataclasses import dataclass
from datetime import datetime
from typing import Dict, List, Tuple

from lottery_system.models import UNKNOWN_LEVEL, Prize, User


@dataclass
class Winner:
    user: User
    prize_name: str
    winning_time: datetime
    order: int

    @property
    def name(self) -> str:
        return self.user.name

    @property
    def department(self) -> str:
        return self.user.department


def _prize_name(prize) -> str:
    return prize.name if hasattr(prize, "name") else str(prize)


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []
        self._counter = 0

    def has_won(self, name: str) -> bool:
        return any(w.user.name == name for w in self.winners)

    def add_winner(self, user, prize) -> bool:
        if self.has_won(user.name):
            return False
        self._counter += 1
        self.winners.append(Winner(user, _prize_name(prize), datetime.now(), self._counter))
        return True

    @staticmethod
    def _sorted_prize_names(prizes) -> List[str]:
        indexed = []
        for i, p in enumerate(prizes or []):
            if isinstance(p, Prize):
                lvl = p.level
            else:
                lvl = Prize(_prize_name(p), 1).level
            indexed.append((lvl, i, _prize_name(p)))
        indexed.sort(key=lambda t: (t[0], t[1]))
        names = []
        for _, _, n in indexed:
            if n not in names:
                names.append(n)
        return names

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for n in self._sorted_prize_names(prizes):
            result[n] = []
        extra = []
        for w in sorted(self.winners, key=lambda w: w.order):
            if w.prize_name not in result:
                result[w.prize_name] = []
                extra.append(w.prize_name)
            result[w.prize_name].append(w)
        if extra:
            ordered = {k: v for k, v in result.items() if k not in extra}
            extra.sort(key=lambda n: (Prize(n, 1).level, extra.index(n)))
            for n in extra:
                ordered[n] = result[n]
            result = ordered
        return result

    def get_sorted_winners(self, prizes) -> List[Winner]:
        out: List[Winner] = []
        for lst in self.get_winners_by_prize(prizes).values():
            out.extend(lst)
        return out

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        if not self.winners:
            return False, "No winners to export"
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8", newline="") as f:
                writer = csv.writer(f)
                writer.writerow(["Name", "Department", "Prize Name", "Winning Time"])
                for w in self.get_sorted_winners(prizes):
                    writer.writerow([
                        w.user.name,
                        w.user.department,
                        w.prize_name,
                        w.winning_time.strftime("%Y-%m-%d %H:%M:%S"),
                    ])
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

"""Lottery rule engine and draw execution."""
import random
from typing import Dict, List, Optional

from lottery_system.models import Prize, User, normalize_group
from lottery_system.services.result_manager import ResultManager

SPECIAL_DRAWER = "Li Bin"
PROTECTED_GROUP = "Cheng Ru Group"
MONITORED_DRAWERS = ("Chen Qixian", "Jin Duo")
TRIGGERED_DRAWER = "Ma Jiaqi"

_PROTECTED_KEYS = {normalize_group("Cheng Ru Group"), normalize_group("Cheng Ru")}


def is_protected_group(department: str) -> bool:
    return normalize_group(department) in _PROTECTED_KEYS


def _same_person(a: Optional[str], b: str) -> bool:
    return a is not None and normalize_group(a) == normalize_group(b)


class LotteryManager:
    def __init__(self, users, result_manager: Optional[ResultManager] = None):
        self.users: List[User] = list(users)
        self.result_manager = result_manager if result_manager is not None else ResultManager()
        self.winner_names = set(w.user.name for w in self.result_manager.winners)
        self.last_two_draw_results: List[bool] = []
        self.drawn_counts: Dict[str, int] = {}
        self.completed_prizes = set()

    # ---------- rule engine ----------
    def is_fallback_active(self, prize: Prize) -> bool:
        if not _same_person(prize.associated_drawer, TRIGGERED_DRAWER):
            return False
        r = self.last_two_draw_results
        return len(r) >= 2 and not r[-1] and not r[-2]

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        candidates = [u for u in self.users if u.name not in self.winner_names]
        if self.is_fallback_active(prize):
            protected = [u for u in candidates if is_protected_group(u.department)]
            if protected:
                return protected
        if _same_person(prize.associated_drawer, SPECIAL_DRAWER):
            return candidates
        avoid = {normalize_group(g) for g in (prize.avoid_groups or [])}
        if not avoid:
            return candidates
        preferred = [u for u in candidates if normalize_group(u.department) not in avoid]
        if preferred:
            return preferred
        # Avoidance group members remain eligible but are drawn after others.
        return candidates

    # ---------- draw execution ----------
    def remaining(self, prize: Prize) -> int:
        return max(0, prize.quantity - self.drawn_counts.get(prize.name, 0))

    def draw_winner(self, prize: Prize) -> Optional[User]:
        if self.remaining(prize) <= 0:
            return None
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winner_names.add(user.name)
        self.result_manager.add_winner(user, prize)
        self.drawn_counts[prize.name] = self.drawn_counts.get(prize.name, 0) + 1
        return user

    def finish_round(self, prize: Prize, round_winners: List[User]) -> None:
        self.completed_prizes.add(prize.name)
        drawer = prize.associated_drawer
        if any(_same_person(drawer, m) for m in MONITORED_DRAWERS):
            self.last_two_draw_results.append(
                any(is_protected_group(u.department) for u in round_winners)
            )
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif _same_person(drawer, TRIGGERED_DRAWER):
            self.last_two_draw_results = []

    def is_completed(self, prize: Prize) -> bool:
        return prize.name in self.completed_prizes

"""Command-line interface."""
from typing import List

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.result_manager import ResultManager


def _ask(prompt: str) -> str:
    try:
        return input(prompt).strip()
    except EOFError:
        return "0"


class LotteryCLI:
    def __init__(self):
        self.list_manager = ListManager()
        self.prize_manager = PrizeManager()
        self.result_manager = ResultManager()
        self.users: List[User] = []
        self.lottery_manager: LotteryManager = None

    # ---------- list ----------
    def _set_users(self, users):
        self.users = users
        self.result_manager = ResultManager()
        self.lottery_manager = LotteryManager(users, self.result_manager)
        print(f"{len(users)} valid participants loaded.")

    def load_default(self):
        users, errors = self.list_manager.load_default_list()
        for e in errors:
            print("  " + e)
        valid, _ = self.list_manager.validate_users(users)
        self._set_users(valid)

    def load_custom(self):
        path = _ask("Enter txt file path: ")
        users, errors = self.list_manager.load_custom_list(path)
        if errors and not users:
            for e in errors:
                print("  " + e)
            print("Upload failed.")
            return
        valid, dups = self.list_manager.validate_users(users)
        if errors or dups:
            print("Validation issues detected:")
            for e in errors:
                print("  " + e)
            for d in dups:
                print(f"  Duplicate Name: {d.name} ({d.department})")
            while True:
                choice = _ask("[r] remove all invalid data and continue, "
                              "[m] manually review entries, [t] terminate upload: ").lower()
                if choice == "t":
                    print("Upload terminated.")
                    return
                if choice == "r":
                    break
                if choice == "m":
                    valid = self._manual_review(users)
                    break
        self._set_users(valid)

    def _manual_review(self, users):
        current = list(users)
        while True:
            print("Current entries:")
            names = {}
            for i, u in enumerate(current, 1):
                names[u.name] = names.get(u.name, 0) + 1
            for i, u in enumerate(current, 1):
                mark = " [DUPLICATE]" if names[u.name] > 1 else ""
                print(f"  {i}. {u.name} ({u.department}){mark}")
            s = _ask("Enter serial numbers to remove (comma separated), or press Enter to finish: ")
            if not s:
                break
            try:
                idxs = sorted({int(x) for x in s.replace("，", ",").split(",") if x.strip()},
                              reverse=True)
            except ValueError:
                print("Invalid input.")
                continue
            for i in idxs:
                if 1 <= i <= len(current):
                    current.pop(i - 1)
        valid, dups = self.list_manager.validate_users(current)
        if dups:
            print(f"{len(dups)} remaining duplicates removed automatically.")
        return valid

    def view_groups(self):
        if not self.users:
            print("No list loaded.")
            return
        info = self.list_manager.get_group_info(self.users)
        print("Department groups:")
        for g, c in info.items():
            print(f"  {g}: {c} people")

    # ---------- prizes ----------
    def show_prizes(self):
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("No prizes configured.")
            return
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} | {p.rule_description()}")

    def _choose_groups(self) -> List[str]:
        groups = list(self.list_manager.get_group_info(self.users).keys())
        if groups:
            print("Available groups:")
            for i, g in enumerate(groups, 1):
                print(f"  {i}. {g}")
        s = _ask("Avoidance groups (serial numbers or names, comma separated, Enter for none): ")
        result = []
        for tok in s.replace("，", ",").split(","):
            tok = tok.strip()
            if not tok:
                continue
            if tok.isdigit() and 1 <= int(tok) <= len(groups):
                g = groups[int(tok) - 1]
            else:
                g = tok
            if g not in result:
                result.append(g)
        return result

    def add_prize(self):
        name = _ask("Prize name: ")
        qty = _ask("Quantity: ")
        groups = self._choose_groups()
        drawer = _ask("Associated lottery person (Enter for none): ") or None
        try:
            p = self.prize_manager.add_prize(name, qty, groups, drawer)
            print(f"Prize '{p.name}' added.")
        except ValueError as e:
            print(f"Error: {e}")

    def configure_prize(self):
        self.show_prizes()
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            return
        s = _ask("Select prize serial number to configure: ")
        if not s.isdigit() or not 1 <= int(s) <= len(prizes):
            print("Invalid selection.")
            return
        idx = int(s) - 1
        self.prize_manager.set_avoid_groups(idx, self._choose_groups())
        drawer = _ask("Associated lottery person (Enter to keep current, '-' to clear): ")
        if drawer == "-":
            self.prize_manager.set_drawer(idx, None)
        elif drawer:
            self.prize_manager.set_drawer(idx, drawer)
        print("Prize updated.")

    def reorder(self):
        self.show_prizes()
        if not self.prize_manager.get_prizes():
            return
        s = _ask("Enter new order of serial numbers (e.g. 3,1,2): ")
        try:
            order = [int(x) for x in s.replace("，", ",").split(",") if x.strip()]
        except ValueError:
            print("Invalid input.")
            return
        if self.prize_manager.reorder_prizes(order):
            print("Prizes reordered.")
            self.show_prizes()
        else:
            print("Invalid order.")

    # ---------- lottery ----------
    def run_lottery(self):
        if not self.users:
            print("Please load a participant list first.")
            return
        prizes = self.prize_manager.get_prizes()
        if not prizes:
            print("Please configure prizes first.")
            return
        lm = self.lottery_manager
        for rnd, prize in enumerate(prizes, 1):
            if lm.is_completed(prize):
                continue
            print("\n" + "=" * 60)
            print(f"Round {rnd}: {prize.name} | Quantity: {prize.quantity}")
            print(f"Rules: {prize.rule_description()}")
            if lm.is_fallback_active(prize):
                print("Fallback triggered: prize pool limited to Cheng Ru Group members!")
            print("Commands: Enter=draw next, 'pause', 'continue', 'terminate'")
            round_winners: List[User] = []
            terminated = False
            while lm.remaining(prize) > 0:
                cmd = _ask(f"[{prize.name}] > ").lower()
                if cmd == "pause":
                    print("Lottery paused. Type 'continue' to resume or 'terminate' to skip round.")
                    while True:
                        c = _ask("(paused) > ").lower()
                        if c == "continue":
                            print("Lottery resumed.")
                            break
                        if c == "terminate":
                            terminated = True
                            break
                    if terminated:
                        break
                    continue
                if cmd == "terminate":
                    terminated = True
                    break
                if cmd not in ("", "continue", "draw"):
                    print("Unknown command.")
                    continue
                n = len(round_winners) + 1
                print(f"Currently drawing the {n}th winner...")
                user = lm.draw_winner(prize)
                if user is None:
                    print("No eligible participants left for this prize.")
                    break
                round_winners.append(user)
                print(f"Congratulations {user.name} ({user.department}) for winning {prize.name}")
            if terminated:
                print("Current round terminated and skipped.")
            lm.finish_round(prize, round_winners)
            print(f"--- Results of round {rnd}: {prize.name} ---")
            if round_winners:
                for i, u in enumerate(round_winners, 1):
                    print(f"  {i}. {u.name} ({u.department})")
            else:
                print("  No winners.")
        print("\nAll rounds completed.")
        self.show_results()

    def show_results(self):
        prizes = self.prize_manager.get_prizes()
        grouped = self.result_manager.get_winners_by_prize(prizes)
        print("=" * 60)
        print("Complete winner list (prize level high to low):")
        any_w = False
        for pname, winners in grouped.items():
            if not winners:
                continue
            any_w = True
            print(f"[{pname}]")
            for w in winners:
                print(f"  {w.user.name} ({w.user.department}) - {pname} - "
                      f"{w.winning_time.strftime('%Y-%m-%d %H:%M:%S')}")
        if not any_w:
            print("  No winners yet.")

    def export(self):
        ok, msg = self.result_manager.export_to_csv(self.prize_manager.get_prizes())
        print(msg)

    # ---------- main loop ----------
    def run(self):
        actions = {
            "1": self.load_default,
            "2": self.load_custom,
            "3": self.view_groups,
            "4": self._load_default_prizes,
            "5": self.add_prize,
            "6": self.configure_prize,
            "7": self.show_prizes,
            "8": self.reorder,
            "9": self.run_lottery,
            "10": self.show_results,
            "11": self.export,

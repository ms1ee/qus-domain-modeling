- **Globals:** `globalvar` holds a dict. `_init()` resets it, and `system.init_system()` calls `_init()` and then sets funds, riders, the rider ID counter, running state, the tick, and the orders list.
- **Orders:** `Order(res_pos, cus_pos)` takes two `(x, y)` positions and snaps them to grid-cell centers. The global order ID counter is stored in `globalvar` so it resets with `_init()`.
- **Assignment:** An order is assigned by appending it to `rider.orders`. `find_best_rider_for_order` returns the nearest rider by Euclidean distance and prefers idle riders. If there are no riders, it returns `None`.
- **Spawn log:** `Rider.__init__` logs "spawn" with `order_id` of `None`.
- **Movement:** Riders move along streets, which are multiples of 100 on either axis.
  - A rider at (435, 385) is on no street, so it first moves to the nearest street line, then travels along it.
  - Cell centers like (50, 50) are not on a street, so the last leg leaves the street to reach the center.
  - Movement is simple: the rider moves in x first, then y, with at most `speed` units per tick.
  - To follow streets, the target is approached via the nearest street intersection.
- **Tick behavior:** In one `move()` call, a rider spends up to `speed` units of travel and can change state mid-tick. Pickup and delivery are handled at the start of each `move()` if the rider is already at the target. This keeps the logic simple.
- **Order status:** An order's status becomes "assigned" when assigned, "picked" at pickup, and "completed" on delivery.
- **Rider states:** A rider with orders but state 0 becomes state 1 and logs "accepted_order" on its next `move()`. On arrival at the restaurant it logs "arrived_at_merchant" and moves to state 2. On arrival at the customer it logs "delivered_to_customer", removes the order, and returns to idle.
- **Extras:** I added a minimal CLI in `main.py` with `start`, `status`, `orders`, `riders`, and `quit`. It handles recruitment and ASCII display briefly.

"""Order data model."""
import globalvar


def to_center(v):
    """Convert a coordinate to its grid cell center."""
    v = max(0, min(799, int(v)))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        cnt = globalvar.get_value("order_id_counter", 0) + 1
        globalvar.set_value("order_id_counter", cnt)
        self.id = cnt
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "created"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.completion_time = None
        d = ((self.res_pos[0] - self.cus_pos[0]) ** 2 +
             (self.res_pos[1] - self.cus_pos[1]) ** 2) ** 0.5
        self.distance = d
        orders = globalvar.get_value("orders")
        if orders is not None:
            orders.append(self)

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    @property
    def time_taken(self):
        if self.completion_time is None:
            return None
        return self.completion_time - self.start_time

"""Rider model."""
import globalvar


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0),
                         "x": self.x, "y": self.y})

    @staticmethod
    def _nearest_street(v):
        return int(round(v / 100.0)) * 100

    def _step_toward(self, tx, ty):
        """Move up to speed units toward target, preferring streets."""
        budget = self.speed
        on_h = self.y % 100 == 0
        on_v = self.x % 100 == 0
        while budget > 0 and (self.x, self.y) != (tx, ty):
            if not on_h and not on_v:
                # get onto the nearest street (vertical line x multiple of 100)
                sx = self._nearest_street(self.x)
                if self.x == tx or abs(self.x - sx) == 0:
                    on_v = True
                    continue
                d = sx - self.x
                m = min(budget, abs(d))
                self.x += m if d > 0 else -m
                budget -= m
                on_v = self.x % 100 == 0
                continue
            if self.y != ty and (on_v and (self.x % 100 == 0)) and \
                    (self.x != tx or True):
                if self.x == tx or self.y % 100 != 0 or True:
                    # travel vertically along the street
                    if self.y != ty and not (self.y % 100 == 0 and
                                             self.x != tx and
                                             self.x % 100 != 0):
                        d = ty - self.y
                        m = min(budget, abs(d))
                        self.y += m if d > 0 else -m
                        budget -= m
                        on_h = self.y % 100 == 0
                        if budget == 0:
                            break
                        continue
            if self.x != tx:
                d = tx - self.x
                m = min(budget, abs(d))
                self.x += m if d > 0 else -m
                budget -= m
                on_v = self.x % 100 == 0
                on_h = self.y % 100 == 0
                continue
            break

    def move(self):
        """Advance one tick."""
        if self.state == 0 and self.orders:
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        if self.state == 0:
            return
        order = self.orders[0]
        if self.state == 1:
            tx, ty = order.res_pos
            if (self.x, self.y) != (tx, ty):
                self._step_toward(tx, ty)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                order.status = "picked"
                self._log("arrived_at_merchant", order.id)
            return
        tx, ty = order.cus_pos
        if (self.x, self.y) != (tx, ty):
            self._step_toward(tx, ty)
        if (self.x, self.y) == (tx, ty):
            order.status = "completed"
            order.completion_time = globalvar.get_value("tick", 0)
            self.orders.pop(0)
            self.state = 0
            self._log("delivered_to_customer", order.id)

"""Order assignment."""
import globalvar


def find_best_rider_for_order(order):
    """Return the nearest rider (idle preferred), or None."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    rx, ry = order.res_pos

    def key(r):
        d = ((r.x - rx) ** 2 + (r.y - ry) ** 2) ** 0.5
        return (0 if r.state == 0 and not r.orders else 1,
                len(r.orders), d)
    return min(riders, key=key)

"""Command-line entry point."""
import globalvar
import system
from rider import Rider
from order_model import Order
from order import find_best_rider_for_order

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


def tick():
    globalvar.set_value("tick", globalvar.get_value("tick") + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def start(args):
    if len(args) != 4:
        print("Usage: start rx ry cx cy")
        return
    try:
        v = [int(a) for a in args]
    except ValueError:
        print("Coordinates must be integers.")
        return
    if any(a < 0 or a > 800 for a in v):
        print("Coordinates must be within 0-800.")
        return
    order = Order((v[0], v[1]), (v[2], v[3]))
    riders = globalvar.get_value("riders")
    added = False
    funds = globalvar.get_value("funds")
    pending = sum(len(r.orders) for r in riders) + 1
    idle = sum(1 for r in riders if r.state == 0 and not r.orders)
    if funds >= 300 and pending > idle:
        cnt = globalvar.get_value("rider_id_counter") + 1
        globalvar.set_value("rider_id_counter", cnt)
        riders.append(Rider(cnt))
        globalvar.set_value("funds", funds - 300)
        added = True
    best = find_best_rider_for_order(order)
    if best is None:
        print("No rider available.")
        return
    best.orders.append(order)
    order.assign_to(best.rider_id)
    print("Restaurant cell:", order.res_pos, "Customer cell:", order.cus_pos)
    print("New rider added:", added, "| rider", best.rider_id)


def status():
    print("Funds:", globalvar.get_value("funds"))
    for r in globalvar.get_value("riders"):
        print("Rider %d at (%d,%d) %s orders=%s" % (
            r.rider_id, r.x, r.y, STATES[r.state],
            [o.id for o in r.orders]))


def orders():
    for o in globalvar.get_value("orders"):
        print("Order %d start=%s res=%s cus=%s dist=%.1f taken=%s rider=%s" % (
            o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
            o.time_taken, o.rider_id))


def riders_log():
    for r in globalvar.get_value("riders"):
        for e in r.log:
            print("Rider %d t=%s (%s,%s) %s order=%s" % (
                r.rider_id, e["time"], e["x"], e["y"], e["event"],
                e["order_id"]))


def main():
    system.init_system()
    history = []
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if not line:
            continue
        history.append(line)
        parts = line.split()
        cmd, args = parts[0].lower(), parts[1:]
        if cmd in ("quit", "exit"):
            globalvar.set_value("running", False)
        elif cmd == "start":
            start(args)
        elif cmd == "status":
            status()
        elif cmd == "orders":
            orders()
        elif cmd == "riders":
            riders_log()
        else:
            print("Unknown command.")
            continue
        if cmd in ("start", "status", "orders", "riders"):
            tick()


if __name__ == "__main__":
    main()

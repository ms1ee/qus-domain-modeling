I had to reconcile several contradictions in the PRD, so these are the choices I made. I have not run any of this code.

- **Rider states:** I used 0 = idle, 1 = picking up, 2 = delivering. The PRD also lists 1/2/3 in one place, but the data model says 0-2.
- **Rider speed and start position:** I used speed 10 and start position (400, 400), per the recruitment section. The PRD also says 20 in one place.
- **Coordinate conversion:** The PRD gives two conversions, top-left corner and `(v // 100) * 100 + 50`. I used the top-left corner, `(v // 100) * 100`, clamped to 0-700, because it is the one the order-creation section describes. Locations are therefore multiples of 100.
- **Riders on roads:** A rider at (400, 400) sits on a grid point rather than strictly on a road. That is fine for this simulator.
- **Movement:** Riders move along axes, horizontal first, then vertical, with at most `speed` units per tick. A rider arrives exactly on the target.
- **Order flow:**
  - Assigning an order appends it to `rider.orders`.
  - On the next `move()` the rider logs `accepted_order` and moves to state 1.
  - On arrival at the restaurant it logs `arrived_at_merchant` and moves to state 2.
  - On arrival at the customer it logs `delivered_to_customer`, marks the order "completed" and goes idle or picks up the next pending order.
- **Assignment:** `find_best_rider_for_order` picks the rider with the fewest active orders, then the smallest distance to the restaurant.
- **Display, history and persistence:** `start` prints the grid coordinates, whether a rider was recruited, and an ASCII route map. Command history uses `readline` if available. State is saved to `state.json` on exit.
- **Recruitment:** A rider is hired when funds are at least 300 and pending orders exceed idle riders. Each hire costs 500, so funds can go negative.

"""Order data model."""
import itertools

import globalvar

_counter = itertools.count(1)


def to_grid(v):
    """Convert coordinate to top-left corner of its grid cell (0-700)."""
    v = int(v)
    return max(0, min(700, (v // 100) * 100))


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_grid(res_pos[0]), to_grid(res_pos[1]))
        self.cus_pos = (to_grid(cus_pos[0]), to_grid(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0)
        self.end_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Mark order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res": list(self.res_pos),
                "cus": list(self.cus_pos), "status": self.status,
                "rider": self.rider_id, "start": self.start_time,
                "end": self.end_time, "distance": self.distance}

"""Path planning."""


class PathPlanner:
    """Grid-based planner: horizontal first, then vertical."""

    @staticmethod
    def next_step(pos, target, speed):
        """Return new position moving up to speed units toward target."""
        x, y = pos
        tx, ty = target
        budget = speed
        if x != tx:
            d = min(budget, abs(tx - x))
            x += d if tx > x else -d
            budget -= d
        if budget > 0 and y != ty:
            d = min(budget, abs(ty - y))
            y += d if ty > y else -d
        return x, y

    @staticmethod
    def route(start, *points):
        """Return list of corner points through destinations."""
        path = [tuple(start)]
        cur = tuple(start)
        for p in points:
            if cur[0] != p[0] and cur[1] != p[1]:
                path.append((p[0], cur[1]))
            if cur != tuple(p):
                path.append(tuple(p))
            cur = tuple(p)
        return path

"""Rider model."""
import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self.current = None
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0),
                         "pos": (self.x, self.y)})

    def pending_orders(self):
        return [o for o in self.orders if o.status != "completed"]

    def _take_next(self):
        for o in self.orders:
            if o.status in ("pending", "assigned"):
                self.current = o
                o.status = "picking"
                self.state = 1
                self._record("accepted_order", o.id)
                return True
        return False

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self._take_next():
                return
        o = self.current
        if o is None:
            self.state = 0
            return
        target = o.res_pos if self.state == 1 else o.cus_pos
        self.x, self.y = PathPlanner.next_step(
            (self.x, self.y), target, self.speed)
        if (self.x, self.y) != tuple(target):
            return
        if self.state == 1:
            self._record("arrived_at_merchant", o.id)
            o.status = "delivering"
            self.state = 2
        else:
            o.status = "completed"
            o.end_time = globalvar.get_value("time", 0)
            self._record("delivered_to_customer", o.id)
            self.current = None
            self.state = 0
            self._take_next()

"""Order management."""
import globalvar
from order_model import Order


def find_best_rider_for_order(order):
    """Pick rider with fewest active orders, then nearest to restaurant."""
    riders = globalvar.get_value("riders", [])
    if not riders:
        return None

    def key(r):
        d = ((r.x - order.res_pos[0]) ** 2
             + (r.y - order.res_pos[1]) ** 2) ** 0.5
        return (len(r.pending_orders()), d, r.rider_id)

    return min(riders, key=key)


class OrderManager:
    """Create and assign orders."""

    @staticmethod
    def create(coords):
        o = Order(coords[0:2], coords[2:4])
        orders = globalvar.get_value("orders")
        orders.append(o)
        return o

    @staticmethod
    def assign(order):
        r = find_best_rider_for_order(order)
        if r is None:
            return None
        order.assign_to(r.rider_id)
        r.orders.append(order)
        return r

def init_system():

    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])

    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Funds, riders and recruitment."""

    @staticmethod
    def pending_count():
        return sum(1 for o in globalvar.get_value("orders", [])
                   if o.status == "pending")

    @staticmethod
    def maybe_recruit(extra_pending=0):
        """Hire a rider if conditions are met. Return True if hired."""
        riders = globalvar.get_value("riders")
        idle = sum(1 for r in riders if r.state == 0 and not r.pending_orders())
        pending = SystemManager.pending_count() + extra_pending
        if globalvar.get_value("funds") >= 300 and pending > idle:
            n = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", n)
            riders.append(Rider(n))
            globalvar.set_value("funds", globalvar.get_value("funds") - 500)
            return True
        return False

    @staticmethod
    def tick():
        globalvar.set_value("time", globalvar.get_value("time") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

"""Display."""
import globalvar
from path_planner import PathPlanner

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and map rendering."""

    @staticmethod
    def map_text(riders=(), orders=(), route=None):
        n = 16
        g = [["." for _ in range(n)] for _ in range(n)]
        for i in range(n):
            for j in range(n):
                if i % 2 == 0 or j % 2 == 0:
                    g[i][j] = "#"

        def cell(p):
            return min(n - 1, p[1] // 50), min(n - 1, p[0] // 50)

        if route:
            for a, b in zip(route, route[1:]):
                steps = max(abs(b[0] - a[0]), abs(b[1] - a[1])) // 50
                for s in range(steps + 1):
                    x = a[0] + (b[0] - a[0]) * s // max(steps, 1)
                    y = a[1] + (b[1] - a[1]) * s // max(steps, 1)
                    i, j = cell((x, y))
                    g[i][j] = "*"
        for o in orders:
            i, j = cell(o.res_pos)
            g[i][j] = "S"
            i, j = cell(o.cus_pos)
            g[i][j] = "C"
        for r in riders:
            i, j = cell((r.x, r.y))
            g[i][j] = "R"
        return "\n".join("".join(row) for row in g)

    @staticmethod
    def status_text():
        out = ["Funds: %d" % globalvar.get_value("funds"),
               "Time: %d" % globalvar.get_value("time"),
               "Riders: %d" % len(globalvar.get_value("riders"))]
        for r in globalvar.get_value("riders"):
            out.append("  Rider %d at (%d,%d) %s, orders: %s" % (
                r.rider_id, r.x, r.y, STATE_NAMES[r.state],
                [(o.id, o.status) for o in r.orders]))
        return "\n".join(out)

    @staticmethod
    def orders_text():
        out = []
        for o in globalvar.get_value("orders"):
            t = (o.end_time - o.start_time) if o.end_time is not None else "-"
            out.append("Order %d start=%s from=%s to=%s dist=%d time=%s rider=%s [%s]"
                       % (o.id, o.start_time, o.res_pos, o.cus_pos,
                          o.distance, t, o.rider_id, o.status))
        return "\n".join(out) or "No orders."

    @staticmethod
    def riders_text():
        out = []
        for r in globalvar.get_value("riders"):
            out.append("Rider %d:" % r.rider_id)
            for e in r.log:
                out.append("  t=%s %s order=%s pos=%s" % (
                    e["time"], e["event"], e["order_id"], e["pos"]))
        return "\n".join(out) or "No riders."

"""Input parsing and validation."""


class InputManager:
    """Parse commands."""

    @staticmethod
    def parse_coords(text):
        """Parse four integers in 0-1000; raise ValueError otherwise."""
        parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(parts) != 4:
            raise ValueError("Enter exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        for v in vals:
            if not 0 <= v <= 1000:
                raise ValueError("Coordinates must be within 0-1000.")
        return vals

import json

def save_state(path="state.json"):
    data = {"funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state,
                        "orders": [o.id for o in r.orders]}
                       for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
    try:
        with open(path, "w") as f:
            json.dump(data, f, indent=2)
    except OSError as e:
        print("Could not save state:", e)


def do_start():
    try:
        coords = InputManager.parse_coords(
            input("Enter restaurant_x restaurant_y customer_x customer_y: "))
    except ValueError as e:
        print("Error:", e)
        return
    order = OrderManager.create(coords)
    hired = SystemManager.maybe_recruit()
    rider = OrderManager.assign(order)
    if rider is None:
        print("No rider available; order pending.")
    print("Restaurant grid:", order.res_pos, "Customer grid:", order.cus_pos)
    print("New rider added:", hired)
    if rider:
        route = PathPlanner.route((rider.x, rider.y), order.res_pos,
                                  order.cus_pos)
        dist = sum(abs(b[0] - a[0]) + abs(b[1] - a[1])
                   for a, b in zip(route, route[1:]))
        print("Rider %d speed %d, est. time %d ticks" % (
            rider.rider_id, rider.speed, -(-dist // rider.speed)))
        print(DisplayManager.map_text([rider], [order], route))


def main():
    init_system()
    print("Commands: start, status, orders, riders, quit/exit, (enter = tick)")
    while globalvar.get_value("running"):
        try:
            cmd = input("> ").strip().lower()
        except (EOFError, KeyboardInterrupt):
            break
        try:
            if cmd in ("quit", "exit", "q"):
                break
            elif cmd in ("start", "s"):
                do_start()
            elif cmd == "status":
                print(DisplayManager.status_text())
                print(DisplayManager.map_text(
                    globalvar.get_value("riders"),
                    [o for o in globalvar.get_value("orders")
                     if o.status != "completed"]))
            elif cmd == "orders":
                print(DisplayManager.orders_text())
            elif cmd == "riders":
                print(DisplayManager.riders_text())
            elif cmd and cmd != "tick":
                print("Unknown command.")
                continue
        except Exception as e:  # recovery
            print("Error:", e)
            continue
        SystemManager.tick()
    globalvar.set_value("running", False)
    save_state()


if __name__ == "__main__":
    main()

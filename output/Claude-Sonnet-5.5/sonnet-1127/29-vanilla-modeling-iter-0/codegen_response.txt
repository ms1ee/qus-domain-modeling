"""Global state storage shared across modules."""

_global_dict = {}


def _init():
    """Reset the global dictionary."""
    global _global_dict
    _global_dict = {}


def set_value(key, value):
    """Store a value under key."""
    _global_dict[key] = value


def get_value(key, default=None):
    """Return the value stored under key, or default."""
    return _global_dict.get(key, default)

"""Order data model."""
import itertools

_counter = itertools.count(1)


def to_center(v):
    """Convert a coordinate to its grid cell center."""
    v = max(0, min(800, int(v)))
    c = (v // 100) * 100 + 50
    return min(c, 750)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos, start_time=0):
        self.id = next(_counter)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = start_time
        self.completion_time = None
        dx = self.res_pos[0] - self.cus_pos[0]
        dy = self.res_pos[1] - self.cus_pos[1]
        self.distance = (dx * dx + dy * dy) ** 0.5

    def assign_to(self, rider_id):
        """Assign this order to a rider id."""
        self.rider_id = rider_id
        self.status = "assigned"

"""Rider model and movement."""
import globalvar


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0),
                         "x": self.x, "y": self.y})

    def _step_to(self, tx, ty):
        """Move up to speed units toward target; horizontal first."""
        budget = self.speed
        dx = tx - self.x
        step = min(abs(dx), budget)
        self.x += step if dx > 0 else -step
        budget -= step
        dy = ty - self.y
        step = min(abs(dy), budget)
        self.y += step if dy > 0 else -step
        return self.x == tx and self.y == ty

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        order = self.orders[0]
        if self.state == 1:
            if self._step_to(*order.res_pos):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            if self._step_to(*order.cus_pos):
                order.status = "completed"
                order.completion_time = globalvar.get_value("tick", 0)
                self.orders.pop(0)
                self.state = 0
                self._log("delivered_to_customer", order.id)

"""Order assignment."""
import globalvar
from rider import Rider


def recruit_rider():
    """Create a new rider if funds allow."""
    funds = globalvar.get_value("funds", 0)
    if funds < 300:
        return None
    cid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", cid)
    globalvar.set_value("funds", funds - 300)
    r = Rider(cid)
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    riders.append(r)
    return r


def find_best_rider_for_order(order):
    """Return the rider closest to the order's restaurant."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return recruit_rider()
    rx, ry = order.res_pos
    return min(riders, key=lambda r: (r.x - rx) ** 2 + (r.y - ry) ** 2)


def assign_order(order):
    """Find the best rider and give the order to them."""
    rider = find_best_rider_for_order(order)
    if rider is not None:
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
    return rider

"""Command-line entry point."""
import globalvar
import system
from order_model import Order
from order import assign_order


def main():
    system.init_system()
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip().split()
        except EOFError:
            break
        if not line:
            continue
        cmd = line[0].lower()
        if cmd in ("quit", "exit"):
            globalvar.set_value("running", False)
        elif cmd == "start":
            try:
                v = [int(a) for a in line[1:]]
                if len(v) != 4 or any(not 0 <= a <= 800 for a in v):
                    raise ValueError
            except ValueError:
                print("Enter four integers in range 0-800.")
                continue
            o = Order(v[:2], v[2:], globalvar.get_value("tick", 0))
            r = assign_order(o)
            print("Order", o.id, o.res_pos, o.cus_pos,
                  "rider", r.rider_id if r else None)
        elif cmd == "status":
            print("Funds:", globalvar.get_value("funds"))
            for r in globalvar.get_value("riders"):
                print(r.rider_id, (r.x, r.y), r.state, len(r.orders))
        else:
            print("Unknown command")
            continue
        globalvar.set_value("tick", globalvar.get_value("tick", 0) + 1)
        for r in globalvar.get_value("riders"):
            r.move()


if __name__ == "__main__":
    main()

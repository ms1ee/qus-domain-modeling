"""Graph data structures."""


class Edge:
    """Undirected edge segment between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists keyed by station number."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an edge in both directions."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(Edge(edge.dest, edge.source, edge.distance, edge.line))

    def neighbors(self, n):
        return self.adj.get(n, [])

    def remove_edges_between(self, a, b, line=None):
        for x, y in ((a, b), (b, a)):
            self.adj[x] = [e for e in self.adj.get(x, [])
                           if not (e.dest == y and (line is None or e.line == line))]

    def update_distance(self, a, b, distance, line=None):
        """Update distance of existing edges; return number of directed edges changed."""
        n = 0
        for x, y in ((a, b), (b, a)):
            for e in self.adj.get(x, []):
                if e.dest == y and (line is None or e.line == line):
                    e.distance = distance
                    n += 1
        return n

    def nodes(self):
        return list(self.adj.keys())

    def V(self):
        return len(self.adj)

    def E(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""Dijkstra shortest path with priority queue."""
import heapq
import itertools

from models import Edge, Graph  # noqa: F401

MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 100000, "comprehensive": 1500}


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Heap based minimum priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}    # node -> state key
        self._info = {}    # state key -> (real dist, transfers, parent key, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        pref = set(cfg.preferred_lines or [])
        track_t = cfg.max_transfers is not None
        start = (self.source, None, 0)
        cost = {start: 0.0}
        self._info[start] = (0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, _ = st
            d, t, _, _ = self._info[st]
            if node not in self._best:
                self._best[node] = st
                self.dist_to[node] = d
            for e in self.graph.neighbors(node):
                nt = t + (1 if line is not None and e.line != line else 0)
                if track_t and nt > cfg.max_transfers:
                    continue
                key = (e.dest, e.line, nt if track_t else 0)
                if key in done:
                    continue
                w = e.distance * (0.95 if e.line in pref else 1.0)
                nc = c + w + (penalty if nt > t else 0)
                if nc < cost.get(key, float("inf")):
                    cost[key] = nc
                    self._info[key] = (d + e.distance, nt, st, e)
                    pq.push(nc, key)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        nodes, edges = [], []
        st = self._best[target]
        while st is not None:
            _, _, parent, e = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = parent
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._best:
            return 0
        return self._info[self._best[target]][1]

"""Fare and time calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance - 1e-9)
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 35 km/h average speed plus 5 min per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("negative input")
    return distance / 1000.0 / 35.0 * 60.0 + 5.0 * transfer_count + 2.0

"""Network building, integrity check and data updates."""
from models import Edge, Graph

REPRESENTATIVE = {
    "1号线": [], "2号线": [], "4号线": [], "5号线": [],
}
MAIN_LINES = ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
              "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
              "Fangshan Line", "Airport Line"]


class SubwayNetwork:
    def __init__(self, data):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        self.connections = []
        for s in data.get("stations", []):
            self.add_station(s["name"], s.get("lines", []))
        for c in data.get("connections", []):
            self.add_connection(c["source"], c["dest"], c["distance"], c.get("line", ""))

    def add_station(self, name, lines):
        if name not in self.station_to_number:
            n = len(self.station_to_number)
            self.station_to_number[name] = n
            self.number_to_station[n] = name
            self.graph.add_node(n)
        self.graph.set_station_lines(name, lines)

    def add_connection(self, a, b, dist, line):
        for nm in (a, b):
            if nm not in self.station_to_number:
                self.add_station(nm, [line] if line else [])
        for nm in (a, b):
            ls = self.graph.station_lines.setdefault(nm, [])
            if line and line not in ls:
                ls.append(line)
        self.graph.add_edge(Edge(self.station_to_number[a], self.station_to_number[b], dist, line))
        self.connections.append((a, b, dist, line))

    # ---- updates ----
    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        for nm in stations:
            self.add_station(nm, list(self.graph.station_lines.get(nm, [])))
        for i, dist in enumerate(distances):
            self.add_connection(stations[i], stations[i + 1], dist, line)

    def modify_station(self, name, lines):
        if name not in self.station_to_number:
            raise KeyError(name)
        self.graph.set_station_lines(name, lines)

    def update_distance(self, a, b, distance, line=None):
        if a not in self.station_to_number or b not in self.station_to_number:
            raise KeyError("unknown station")
        return self.graph.update_distance(self.station_to_number[a],
                                          self.station_to_number[b], distance, line)

    # ---- check ----
    def check_data(self):
        """Return list of problem strings (empty if OK)."""
        problems = []
        lines_present = set()
        for ls in self.graph.station_lines.values():
            lines_present.update(ls)
        for l in MAIN_LINES:
            if l not in lines_present:
                problems.append("Line missing: %s" % l)
        for nm, n in self.station_to_number.items():
            if not self.graph.neighbors(n):
                problems.append("Station without connections: %s" % nm)
        for a, b, d, l in self.connections:
            if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
                problems.append("Suspicious distance %s-%s (%s): %r" % (a, b, l, d))
            if a == b:
                problems.append("Self loop at %s" % a)
        # connectivity
        if self.graph.nodes():
            seen = {self.graph.nodes()[0]}
            stack = list(seen)
            while stack:
                x = stack.pop()
                for e in self.graph.neighbors(x):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        stack.append(e.dest)
            if len(seen) != self.graph.V():
                problems.append("Network not fully connected: %d of %d stations reachable"
                                % (len(seen), self.graph.V()))
        return problems

"""CLI for Beijing subway fare calculation."""
import argparse
import json
import os
import re
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data  # noqa: E402
from data_manager import SubwayNetwork  # noqa: E402
from fare_calculator import Sum_money, calculate_estimated_time  # noqa: E402
from pathfinding import PathfindingConfig, ShortestPath, MODES  # noqa: E402


def parse_input(text):
    parts = [p.strip() for p in re.split(r"[;；,，]", text)]
    parts = [p for p in parts if p != ""]
    if len(parts) != 2:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    return parts[0], parts[1]


def query(net, start, end, config):
    if start not in net.station_to_number:
        raise KeyError("Starting station '%s' does not exist, please check the station name" % start)
    if end not in net.station_to_number:
        raise KeyError("Ending station '%s' does not exist, please check the station name" % end)
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    try:
        s, t = net.station_to_number[start], net.station_to_number[end]
        sp = ShortestPath(net.graph, s, config)
        nodes, edges = sp.path_with_edges(t)
        if not nodes:
            raise RuntimeError("No route found between '%s' and '%s'" % (start, end))
        dist = sum(e.distance for e in edges)
        transfers = sp.get_transfer_count(t)
    except RuntimeError:
        raise
    except Exception as ex:
        raise RuntimeError("Path calculation failed: %s" % ex)
    try:
        fare = Sum_money(dist)
        minutes = calculate_estimated_time(dist, transfers)
    except Exception as ex:
        raise RuntimeError("Fare calculation failed: %s" % ex)
    names = [net.number_to_station[n] for n in nodes]
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = names[i + 1]
            segments[-1]["distance"] += e.distance
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(names[i])
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1],
                             "distance": e.distance, "stations": 1})
    lines = []
    for sg in segments:
        if sg["line"] not in lines:
            lines.append(sg["line"])
    return {"query": {"start": start, "end": end, "mode": config.mode},
            "path": names, "distance_m": dist, "distance_km": round(dist / 1000.0, 2),
            "fare": fare, "estimated_time_min": round(minutes, 1),
            "transfer_count": transfers, "transfer_stations": transfer_stations,
            "lines": lines, "segments": segments}


def format_text(r):
    out = ["Route: %s -> %s (mode: %s)" % (r["query"]["start"], r["query"]["end"], r["query"]["mode"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])), ""]
    for sg in r["segments"]:
        out.append("  [%s] %s -> %s (%d stops, %d m)" % (
            sg["line"], sg["from"], sg["to"], sg["stations"], sg["distance"]))
    out.append("")
    d = r["distance_m"]
    out.append("Total distance: %d m (%.2f km)" % (d, d / 1000.0))
    out.append("Transfers: %d" % r["transfer_count"])
    if r["transfer_stations"]:
        out.append("Transfer stations: %s" % ", ".join(r["transfer_stations"]))
    out.append("Lines: %s" % ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    if fmt == "json":
        return json.dumps(r, ensure_ascii=False, indent=2)
    return format_text(r)


def write_output(text, path):
    if path:
        with open(path, "a", encoding="utf-8") as f:
            f.write(text + "\n\n")


def main(argv=None):
    ap = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    ap.add_argument("--output_file", default=None)
    ap.add_argument("--format", choices=["text", "json"], default="text")
    ap.add_argument("--mode", choices=list(MODES), default="shortest_distance")
    ap.add_argument("--max_transfers", type=int, default=None)
    ap.add_argument("--line_preference", default=None)
    ap.add_argument("--check_data", action="store_true")
    args = ap.parse_args(argv)

    try:
        net = SubwayNetwork(load_subway_data(DATA_FILE_PATH))
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1

    if args.check_data:
        problems = net.check_data()
        print("Stations: %d, edges: %d" % (net.graph.V(), net.graph.E()))
        if problems:
            print("Data check found %d problem(s):" % len(problems))
            for p in problems:
                print(" - " + p)
            return 1
        print("Data check passed.")
        return 0

    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    cfg.preferred_lines = [args.line_preference] if args.line_preference else []

    print("Enter 'Starting Station; Ending Station' (or 'q' to quit):")
    while True:
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        try:
            a, b = parse_input(line)
            res = query(net, a, b, cfg)
        except (ValueError, KeyError, RuntimeError) as ex:
            msg = ex.args[0] if ex.args else str(ex)
            print("Error: %s" % msg)
            continue
        text = render(res, args.format)
        print(text)
        try:
            write_output(text, args.output_file)
        except OSError as ex:
            print("Error writing output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

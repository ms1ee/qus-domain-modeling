"""Grid based path planning."""

GRID = 100
SIZE = 800


def center(v):
    """Convert coordinate to grid cell centre."""
    v = max(0, min(SIZE, int(v)))
    cell = min(v // GRID, SIZE // GRID - 1)
    return cell * GRID + GRID // 2


class PathPlanner:
    """Right-angle path planner."""

    @staticmethod
    def plan(start, end):
        """Waypoints from start to end: horizontal first, then vertical."""
        pts = [tuple(start)]
        corner = (end[0], start[1])
        if corner != pts[-1]:
            pts.append(corner)
        if tuple(end) != pts[-1]:
            pts.append(tuple(end))
        return pts

    @staticmethod
    def step(pos, target, speed):
        """Advance pos towards target by up to speed units."""
        x, y = pos
        tx, ty = target
        left = speed
        if x != tx:
            d = min(left, abs(tx - x))
            x += d if tx > x else -d
            left -= d
        if left > 0 and y != ty:
            d = min(left, abs(ty - y))
            y += d if ty > y else -d
        return x, y

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Order data model."""
import itertools

from . import globalvar
from .pathplanner import center, PathPlanner

_ids = itertools.count(1)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_ids)
        self.raw = [int(res_pos[0]), int(res_pos[1]),
                    int(cus_pos[0]), int(cus_pos[1])]
        self.res_pos = (center(res_pos[0]), center(res_pos[1]))
        self.cus_pos = (center(cus_pos[0]), center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.end_time = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)

    def assign_to(self, rider_id):
        """Assign to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self):
        """Mark as completed."""
        self.status = "completed"
        self.end_time = globalvar.get_value("time", 0) or 0

    @property
    def duration(self):
        """Ticks taken, or None if unfinished."""
        if self.end_time is None:
            return None
        return self.end_time - self.start_time

    def to_dict(self):
        """Serialise."""
        return {"id": self.id, "res": list(self.res_pos),
                "cus": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "end_time": self.end_time, "distance": self.distance}

"""Rider model."""
from . import globalvar
from .pathplanner import PathPlanner


class Rider:
    """A delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    @property
    def pos(self):
        return (self.x, self.y)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0) or 0,
                         "x": self.x, "y": self.y})

    def _go(self, target):
        self.x, self.y = PathPlanner.step(self.pos, target, self.speed)

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
            order = self.orders[0]
            order.status = "picking_up"
            return
        order = self.orders[0]
        if self.state == 1:
            self._go(order.res_pos)
            if self.pos == tuple(order.res_pos):
                self.state = 2
                order.status = "delivering"
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            self._go(order.cus_pos)
            if self.pos == tuple(order.cus_pos):
                self.orders.pop(0)
                order.complete()
                self.state = 0
                self._log("delivered_to_customer", order.id)

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""System manager."""
import json

from . import globalvar
from .rider import Rider

RIDER_COST = 300


class SystemManager:
    """Manages funds, riders and state."""

    def __init__(self):
        init_system()

    @staticmethod
    def recruit():
        """Create riders while funds suffice. Returns list of new riders."""
        new = []
        while globalvar.get_value("funds") >= RIDER_COST:
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            r = Rider(rid)
            globalvar.get_value("riders").append(r)
            globalvar.set_value("funds",
                                globalvar.get_value("funds") - RIDER_COST)
            new.append(r)
            break
        return new

    @staticmethod
    def tick():
        """Advance simulation one tick."""
        globalvar.set_value("time", globalvar.get_value("time") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

    @staticmethod
    def save(path="state.json"):
        """Persist state to JSON."""
        data = {"funds": globalvar.get_value("funds"),
                "rider_id_counter": globalvar.get_value("rider_id_counter"),
                "time": globalvar.get_value("time"),
                "riders": [r.to_dict() for r in globalvar.get_value("riders")],
                "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", 300)
    globalvar.set_value("time", 0)

from . import globalvar

def _cost(rider, order):
    load = len(rider.orders)
    dist = PathPlanner.distance(rider.pos, order.res_pos)
    return (0 if rider.state == 0 and load == 0 else 1, dist, load,
            rider.rider_id)


def find_best_rider_for_order(order):
    """Return the closest (preferably idle) rider, or None."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    return min(riders, key=lambda r: _cost(r, order))


class OrderManager:
    """Creates, assigns and records orders."""

    @staticmethod
    def create(res_pos, cus_pos):
        """Create an order; returns (order, rider, new_rider_added)."""
        from .system import SystemManager
        order = Order(res_pos, cus_pos)
        globalvar.get_value("orders").append(order)
        added = False
        riders = globalvar.get_value("riders")
        if not riders or all(r.state != 0 or r.orders for r in riders):
            added = bool(SystemManager.recruit())
        rider = find_best_rider_for_order(order)
        if rider is not None:
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
        return order, rider, added

"""Display manager."""
from . import globalvar
from .pathplanner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text rendering."""

    @staticmethod
    def render_map(route=None, orders=None, riders=None):
        """ASCII map; '#' streets between cells."""
        grid = [["." for _ in range(8)] for _ in range(8)]
        for p in route or []:
            grid[min(7, p[1] // 100)][min(7, p[0] // 100)] = "*"
        for o in orders or []:
            if o.status != "completed":
                grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
                grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        for r in riders or []:
            grid[min(7, r.y // 100)][min(7, r.x // 100)] = "R"
        sep = "#" * 17
        lines = [sep]
        for row in grid:
            lines.append("#" + "#".join(row) + "#")
            lines.append(sep)
        return "\n".join(lines)

    @staticmethod
    def route_cells(start, end):
        """Cells traversed from start to end."""
        cells = []
        pos = tuple(start)
        cells.append(pos)
        while pos != tuple(end):
            pos = PathPlanner.step(pos, end, 50)
            cells.append(pos)
        return cells

    def status(self):
        riders = globalvar.get_value("riders")
        out = ["Time: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            out.append("  Rider %d at (%d,%d) speed %d %s orders=%s" % (
                r.rider_id, r.x, r.y, r.speed, STATES[r.state],
                [o.id for o in r.orders]))
        out.append(self.render_map(orders=globalvar.get_value("orders"),
                                   riders=riders))
        return "\n".join(out)

    def orders(self):
        orders = globalvar.get_value("orders")
        if not orders:
            return "No orders."
        out = []
        for o in orders:
            out.append("Order %d: start=%d restaurant=%s customer=%s "
                       "distance=%d time=%s rider=%s status=%s" % (
                           o.id, o.start_time, o.res_pos, o.cus_pos,
                           o.distance, o.duration, o.rider_id, o.status))
        return "\n".join(out)

    def riders(self):
        out = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                out.append("Rider %d t=%d %s order=%s at (%d,%d)" % (
                    r.rider_id, e["time"], e["event"], e["order_id"],
                    e["x"], e["y"]))
        return "\n".join(out) or "No riders."

"""Input parsing and validation."""


class InputManager:
    """Parses commands."""

    ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
               "q": "quit", "exit": "quit", "t": "tick"}

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        parts = line.strip().split()
        if not parts:
            return None, []
        self.history.append(line.strip())
        cmd = parts[0].lower()
        return self.ALIASES.get(cmd, cmd), parts[1:]

    @staticmethod
    def parse_coords(tokens):
        """Validate four integers in 0-800."""
        if len(tokens) != 4:
            raise ValueError("Need exactly 4 integers: rx ry cx cy")
        try:
            vals = [int(t) for t in tokens]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        for v in vals:
            if not 0 <= v <= 800:
                raise ValueError("Coordinates must be within 0-800")
        return vals

"""Command-line entry point."""
from . import globalvar
from .system import SystemManager, init_system
from .order import OrderManager
from .display import DisplayManager
from .input_manager import InputManager


def main():
    init_system()
    disp = DisplayManager()
    inp = InputManager()
    print("Commands: start rx ry cx cy | status | orders | riders | "
          "tick [n] | quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            cmd, args = inp.parse_command(line)
            if cmd is None:
                continue
            if cmd == "quit":
                break
            elif cmd == "start":
                if not args:
                    args = input("rx ry cx cy: ").split()
                v = inp.parse_coords(args)
                order, rider, added = OrderManager.create(v[:2], v[2:])
                print("Restaurant cell: (%d,%d) Customer cell: (%d,%d)" % (
                    order.res_pos[0] // 100, order.res_pos[1] // 100,
                    order.cus_pos[0] // 100, order.cus_pos[1] // 100))
                print("New rider added: %s" % added)
                if rider is None:
                    print("No rider available (insufficient funds).")
                else:
                    cells = (disp.route_cells(rider.pos, order.res_pos) +
                             disp.route_cells(order.res_pos, order.cus_pos))
                    total = 0
                    prev = rider.pos
                    for p in (order.res_pos, order.cus_pos):
                        total += abs(p[0] - prev[0]) + abs(p[1] - prev[1])
                        prev = p
                    print("Rider %d speed %d, estimated time %d ticks" % (
                        rider.rider_id, rider.speed,
                        -(-total // rider.speed)))
                    print(disp.render_map(route=cells,
                                          orders=[order], riders=[rider]))
                SystemManager.tick()
            elif cmd == "status":
                print(disp.status())
            elif cmd == "orders":
                print(disp.orders())
            elif cmd == "riders":
                print(disp.riders())
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                for _ in range(n):
                    SystemManager.tick()
                print("Time: %d" % globalvar.get_value("time"))
            else:
                print("Unknown command: %s" % cmd)
        except ValueError as e:
            print("Error: %s" % e)
        except Exception as e:  # recover gracefully
            print("Unexpected error: %s" % e)
    globalvar.set_value("running", False)
    try:
        SystemManager.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

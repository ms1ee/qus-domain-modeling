"""Core data models: Edge, Graph, Station, Line."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Station:
    def __init__(self, name, station_id, is_transfer=False):
        self.name = name
        self.station_id = station_id
        self.is_transfer = is_transfer


class Line:
    def __init__(self, name, station_sequence=None):
        self.name = name
        self.station_sequence = list(station_sequence or [])


class Graph:
    """Weighted directed graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, node):
        self.adjacency.setdefault(node, [])

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, [])

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def nodes(self):
        return list(self.adjacency.keys())

"""Dijkstra based path finding."""
import heapq
import itertools

MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 10 ** 7, "comprehensive": 1000}


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class MinPQ:
    """Heap based min priority queue."""

    def __init__(self):
        self.heap = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self.edge_to = {}
        self._node_state = {}
        self._prev = {}
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        track = cfg.max_transfers is not None
        start = (self.source, None, 0)
        best = {start: 0.0}
        dist = {start: 0}
        self._prev[start] = (None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in self._node_state:
                self._node_state[node] = st
                self.dist_to[node] = dist[st]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0
                if line is not None and e.line != line:
                    nt = t + 1
                    extra = penalty
                if track and nt > cfg.max_transfers:
                    continue
                w = e.weight
                if cfg.preferred_lines and e.line in cfg.preferred_lines:
                    w = w * 0.8
                ns = (e.dest, e.line, nt if track else 0)
                nc = cost + w + extra
                if ns not in done and nc < best.get(ns, float("inf")):
                    best[ns] = nc
                    dist[ns] = dist[st] + e.weight
                    self._prev[ns] = (st, e)
                    pq.push(nc, ns)
        for n, st in self._node_state.items():
            if self._prev[st][1] is not None:
                self.edge_to[n] = self._prev[st][1]

    def has_path_to(self, target):
        return target in self._node_state

    def path_with_edges(self, target):
        if target not in self._node_state:
            return [], []
        st = self._node_state[target]
        edges = []
        while True:
            prev, e = self._prev[st]
            if prev is None:
                break
            edges.append(e)
            st = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math


class FareTier:
    def __init__(self, min_distance, max_distance, price):
        self.min_distance = min_distance
        self.max_distance = max_distance
        self.price = price


FARE_TIERS = [FareTier(0, 6, 3), FareTier(6, 12, 4), FareTier(12, 22, 5), FareTier(22, 32, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters (fractions of km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    for t in FARE_TIERS:
        if km <= t.max_distance:
            return t.price
    return 6 + math.ceil((km - 32) / 20.0)


class FareCalculator:
    def calculate(self, distance):
        return Sum_money(distance)


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + 5.0 * transfer_count

"""SubwayNetwork: builds graph from data."""
import difflib
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Graph, Edge


class SubwayNetwork:
    def __init__(self, data_file=DATA_FILE_PATH, data=None):
        self.data_file = data_file
        self.data = data if data is not None else load_subway_data(data_file)
        self.build()

    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.lines = {}
        for s in self.data.get("stations", []):
            self._add_station(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in self.station_to_number:
                    self._add_station(n, [c.get("line")])
            a, b = self.station_to_number[c["source"]], self.station_to_number[c["dest"]]
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))
            seq = self.lines.setdefault(c["line"], [])
            for n in (c["source"], c["dest"]):
                if n not in seq:
                    seq.append(n)

    def _add_station(self, name, lines):
        if name in self.station_to_number:
            return
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name
        self.graph.set_station_lines(name, [l for l in lines if l])
        self.graph.add_node(num)

    def has_station(self, name):
        return name in self.station_to_number

    def is_transfer(self, name):
        return len(self.graph.station_lines.get(name, [])) > 1

    def match_station(self, name):
        if name in self.station_to_number:
            return name
        m = difflib.get_close_matches(name, list(self.station_to_number), n=1, cutoff=0.5)
        return m[0] if m else None

"""Route computation, summary, and reports."""
import json
from pathfinding import ShortestPath, PathfindingConfig
from fare_calculator import Sum_money, calculate_estimated_time


class Route:
    def __init__(self, stations, distance, transfer_count, lines, segments, transfer_stations):
        self.station_sequence = stations
        self.total_distance = distance
        self.transfer_count = transfer_count
        self.lines_involved = lines
        self.segments = segments
        self.transfer_stations = transfer_stations


class RouteQuery:
    def __init__(self, start, end, mode="shortest_distance", max_transfers=None, line_preference=None):
        self.start_station = start
        self.end_station = end
        self.mode = mode
        self.max_transfers = max_transfers
        self.line_preference = line_preference


def find_route(network, query):
    pref = [query.line_preference] if query.line_preference else []
    cfg = PathfindingConfig(query.mode, query.max_transfers, pref)
    src = network.station_to_number[query.start_station]
    dst = network.station_to_number[query.end_station]
    sp = ShortestPath(network.graph, src, cfg)
    nodes, edges = sp.path_with_edges(dst)
    if not nodes:
        return None
    names = [network.number_to_station[n] for n in nodes]
    segments = []
    transfers = []
    lines = []
    for i, e in enumerate(edges):
        if e.line not in lines:
            lines.append(e.line)
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["end"] = names[i + 1]
            segments[-1]["distance"] += e.distance
        else:
            if segments:
                transfers.append(names[i])
            segments.append({"line": e.line, "start": names[i], "end": names[i + 1],
                             "distance": e.distance})
    total = sum(e.distance for e in edges)
    return Route(names, total, len(transfers), lines, segments, transfers)


def format_distance(d):
    return "%d m" % d if d < 1000 else "%.2f km" % (d / 1000.0)


def summarize(route):
    return {"total_distance": route.total_distance,
            "fare": Sum_money(route.total_distance),
            "transfer_count": route.transfer_count,
            "estimated_time": calculate_estimated_time(route.total_distance, route.transfer_count)}


def report_text(query, route):
    s = summarize(route)
    out = ["Route: %s -> %s (mode: %s)" % (query.start_station, query.end_station, query.mode),
           "Stations (%d): %s" % (len(route.station_sequence), " -> ".join(route.station_sequence)),
           "Distance: %s" % format_distance(route.total_distance)]
    for seg in route.segments:
        out.append("  [%s] %s -> %s (%s)" % (seg["line"], seg["start"], seg["end"],
                                             format_distance(seg["distance"])))
    out.append("Transfers: %d%s" % (route.transfer_count,
               " (at " + ", ".join(route.transfer_stations) + ")" if route.transfer_stations else ""))
    out.append("Lines: " + ", ".join(route.lines_involved))
    out.append("Estimated time: %.1f minutes" % s["estimated_time"])
    out.append("Fare: %d yuan" % s["fare"])
    return "\n".join(out)


def report_json(query, route):
    s = summarize(route)
    return json.dumps({
        "query": {"start": query.start_station, "end": query.end_station, "mode": query.mode,
                  "max_transfers": query.max_transfers, "line_preference": query.line_preference},
        "path": route.station_sequence,
        "distance": s["total_distance"],
        "fare": s["fare"],
        "estimated_time": s["estimated_time"],
        "transfer_count": s["transfer_count"],
        "transfer_stations": route.transfer_stations,
        "lines": route.lines_involved,
    }, ensure_ascii=False, indent=2)


class ReportExporter:
    def __init__(self, fmt="text", output_file=None):
        self.format = fmt
        self.output_file = output_file

    def render(self, query, route):
        return report_json(query, route) if self.format == "json" else report_text(query, route)

    def export(self, query, route):
        text = self.render(query, route)
        if self.output_file:
            with open(self.output_file, "a", encoding="utf-8") as f:
                f.write(text + "\n")
        return text

"""Data management and integrity checks."""
import json
from data_loader import DATA_FILE_PATH, load_subway_data
from network import SubwayNetwork

REPRESENTATIVE = {"1号线": ["西单", "国贸"], "2号线": ["西直门", "雍和宫"], "Line 1": ["Xidan"]}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)

    def network(self):
        return SubwayNetwork(self.data_file, data=self.data)

    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: meters between consecutive ones (both directions)."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line not in known[n]["lines"]:
                    known[n]["lines"].append(line)
            else:
                self.data["stations"].append({"name": n, "lines": [line]})
        for a, b, d in zip(stations, stations[1:], distances):
            for s, t in ((a, b), (b, a)):
                self.data["connections"].append({"source": s, "dest": t, "distance": d, "line": line})

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                return True
        return False

    def update_distance(self, a, b, distance, bidirectional=True):
        n = 0
        for c in self.data["connections"]:
            if (c["source"], c["dest"]) == (a, b) or (bidirectional and (c["source"], c["dest"]) == (b, a)):
                c["distance"] = distance
                n += 1
        return n

    def save(self, path=None):
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)


class DataIntegrityChecker:
    def __init__(self, manager):
        self.manager = manager

    def check(self):
        """Return list of problem strings (empty if OK)."""
        d = self.manager.data
        problems = []
        names = set()
        for s in d.get("stations", []):
            if s["name"] in names:
                problems.append("Duplicate station: %s" % s["name"])
            names.add(s["name"])
            if not s.get("lines"):
                problems.append("Station without line: %s" % s["name"])
        if len(names) < 50:
            problems.append("Station coverage too small: %d" % len(names))
        pairs = set()
        for c in d.get("connections", []):
            for k in ("source", "dest"):
                if c[k] not in names:
                    problems.append("Connection references unknown station: %s" % c[k])
            if not isinstance(c.get("distance"), (int, float)) or c["distance"] <= 0:
                problems.append("Bad distance: %s -> %s" % (c["source"], c["dest"]))
            elif c["distance"] > 30000:
                problems.append("Suspicious distance: %s -> %s" % (c["source"], c["dest"]))
            pairs.add((c["source"], c["dest"]))
        connected = {p[0] for p in pairs} | {p[1] for p in pairs}
        for n in sorted(names - connected):
            problems.append("Isolated station: %s" % n)
        return problems

"""Input validation."""
import re


class InputValidator:
    def __init__(self, network):
        self.network = network

    def parse(self, text):
        """Return (start, end, messages). Raises ValueError with a user message on failure."""
        parts = [p.strip() for p in re.split(r"[;；,，]", text or "") if p.strip()]
        if len(parts) != 2:
            raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
        start, end = parts
        notes = []
        ms = self.network.match_station(start)
        if ms is None:
            raise ValueError("Starting station '%s' does not exist, please check the station name" % start)
        me = self.network.match_station(end)
        if me is None:
            raise ValueError("Ending station '%s' does not exist, please check the station name" % end)
        if ms != start:
            notes.append("Starting station '%s' matched to '%s'" % (start, ms))
        if me != end:
            notes.append("Ending station '%s' matched to '%s'" % (end, me))
        if ms == me:
            raise ValueError("Starting station cannot be the same as the ending station")
        return ms, me, notes

"""Command line interface."""
import argparse
import sys
from data_loader import DATA_FILE_PATH
from data_manager import DataManager, DataIntegrityChecker
from validator import InputValidator
from route_service import RouteQuery, find_route, ReportExporter


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p


class CLI:
    def __init__(self, args=None):
        self.args = build_parser().parse_args(args)
        self.manager = DataManager(DATA_FILE_PATH)
        self.network = self.manager.network()
        self.validator = InputValidator(self.network)
        self.exporter = ReportExporter(self.args.format, self.args.output_file)

    def check_data(self):
        problems = DataIntegrityChecker(self.manager).check()
        if problems:
            print("Data check found %d problem(s):" % len(problems))
            for p in problems[:50]:
                print(" -", p)
            return 1
        print("Data check passed.")
        return 0

    def query(self, line):
        try:
            start, end, notes = self.validator.parse(line)
        except ValueError as e:
            print(e)
            return
        for n in notes:
            print(n)
        q = RouteQuery(start, end, self.args.mode, self.args.max_transfers, self.args.line_preference)
        try:
            route = find_route(self.network, q)
        except Exception as e:
            print("Path calculation failed: %s" % e)
            return
        if route is None:
            print("No route found between '%s' and '%s' (check transfer limit)" % (start, end))
            return
        try:
            print(self.exporter.export(q, route))
        except Exception as e:
            print("Fare calculation or export failed: %s" % e)

    def run(self):
        if self.args.check_data:
            return self.check_data()
        print("Enter 'Starting Station; Ending Station' (empty line or 'quit' to exit)")
        while True:
            try:
                line = input("> ")
            except EOFError:
                break
            if line.strip().lower() in ("", "q", "quit", "exit"):
                break
            self.query(line)
        return 0


def main(args=None):
    return CLI(args).run()


if __name__ == "__main__":
    sys.exit(main())

import re
from collections import OrderedDict
from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Cheng Ru Group"),
    ("Zheng Shi", "Cheng Ru Group"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Operations Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Li Bin", "Marketing Department"),
    ("Liu Yang", "Finance Department"),
    ("Huang Lei", "Finance Department"),
    ("Xu Fang", "HR Department"),
    ("Gao Min", "HR Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error: cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = re.split(r"[,;，；]", line.strip())
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Format Error: Line {i}: '{line.strip()}'")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in [("Third Prize - Movie Tickets", 3),
                          ("Second Prize - Shopping Cards", 2),
                          ("First Prize - 1 Day Annual Leave", 1)]:
            try:
                self.add_prize(name, qty)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer or None)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

PROTECTED = "cheng ru"
FALLBACK_BEFORE = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET = "Ma Jiaqi"
SPECIAL = "Li Bin"


def _is_cr(user):
    return PROTECTED in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _fallback_active(self, prize):
        return (prize.associated_drawer == FALLBACK_TARGET
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize):
        won = {id(u) for u in self.winners} | {u.name for u in self.winners}
        pool = [u for u in self.users if u not in self.winners and u.name not in won]
        if self._fallback_active(prize):
            return [u for u in pool if _is_cr(u)]
        if prize.associated_drawer == SPECIAL:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def _secondary_pool(self, prize):
        """Avoided-group members: still eligible but drawn last."""
        if prize.associated_drawer == SPECIAL or self._fallback_active(prize):
            return []
        avoid = set(prize.avoid_groups or [])
        return [u for u in self.users if u.department in avoid
                and u not in self.winners]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = self._secondary_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_round(self, prize, round_winners):
        d = prize.associated_drawer
        if d in FALLBACK_BEFORE:
            self.last_two_draw_results.append(any(_is_cr(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif d == FALLBACK_TARGET:
            self.last_two_draw_results = []

    def rule_description(self, prize):
        parts = []
        if prize.associated_drawer == SPECIAL:
            parts.append("Special: avoidance groups not applied")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        if self._fallback_active(prize):
            parts.append("Fallback: pool limited to Cheng Ru group")
        parts.append("Previous winners excluded")
        return "; ".join(parts)

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name="", time=None, order=0):
        self.user = user
        self.prize_name = prize_name
        self.time = time or datetime.now()
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name, datetime.now(), len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            name = getattr(p, "name", p)
            result[name] = sorted((w for w in self.winners if w.prize_name == name),
                                  key=lambda w: w.order)
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    c = input("Load list: 1) default 2) custom txt [1]: ").strip()
    if c == "2":
        users, errors = lm._load_from_path(input("Path: ").strip())
        for e in errors:
            print(e)
        if errors and input("Errors found. Remove invalid lines and continue? (y/n): ").lower() != "y":
            return None
    else:
        users, _ = lm.load_default_list()
    users, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate removed: {d.name}")
    print("Groups:", lm.get_group_info(users))
    return users


def configure_prizes(pm, groups):
    if input("Load preset prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        try:
            name = input("Name: ").strip()
            qty = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma sep): ").split(",") if g.strip()]
            drawer = input("Lottery person: ").strip() or None
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    s = input("New order (space separated numbers, blank to keep): ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, kept.")
        except ValueError:
            print("Invalid order, kept.")


def run_round(lot, res, prize, idx):
    print(f"\n=== Round {idx}: {prize.name} x{prize.quantity} ===")
    print("Rules:", lot.rule_description(prize))
    got = []
    paused = False
    n = 0
    while n < prize.quantity:
        cmd = input("[Enter]=draw, pause/continue/terminate: ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            paused = True
            print("Paused.")
            continue
        if cmd == "continue":
            paused = False
            continue
        if paused:
            print("Paused. Type 'continue'.")
            continue
        print(f"Currently drawing the {n + 1}th winner")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible participants left.")
            break
        res.add_winner(u, prize)
        got.append(u)
        n += 1
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.finish_round(prize, got)
    print("Round results:", [u.name for u in got])


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        return
    pm = PrizeManager()
    configure_prizes(pm, lm.get_group_info(users))
    prizes = pm.get_prizes()
    lot = LotteryManager(users)
    res = ResultManager()
    for i, p in enumerate(prizes, 1):
        run_round(lot, res, p, i)
    print("\n=== Final Winners ===")
    for name, ws in res.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    if input("Export results? (y/n): ").lower() == "y":
        print(res.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()

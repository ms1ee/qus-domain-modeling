"""Graph data structures."""
import heapq


class Edge:
    """Directed weighted edge between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted directed graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, [])

    def edges_from(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self._h = []
        self._c = 0

    def push(self, priority, item):
        self._c += 1
        heapq.heappush(self._h, (priority, self._c, item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def __len__(self):
        return len(self._h)

    def is_empty(self):
        return not self._h

"""Dijkstra shortest path with transfer awareness."""
from models import MinPQ

TRANSFER_PENALTY = {"shortest_distance": 0, "comprehensive": 3000, "least_transfer": 1000000}


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}   # node -> state key
        self._info = {}   # key -> (cost, dist, transfers, prev_key, edge)
        self._run()

    def _run(self):
        cfg = self.config
        pen = TRANSFER_PENALTY.get(cfg.mode, 0)
        limit = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        best_cost = {start: 0.0}
        self._info[start] = (0.0, 0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, key = pq.pop()
            if key in done:
                continue
            done.add(key)
            node, line, _ = key
            _, d, t, _, _ = self._info[key]
            if node not in self._best:
                self._best[node] = key
                self.dist_to[node] = d
            for e in self.graph.edges_from(node):
                nt = t + (1 if line is not None and e.line != line else 0)
                if limit is not None and nt > limit:
                    continue
                w = e.weight
                if pref and e.line in pref:
                    w = w * 0.9
                ncost = cost + w + (pen if nt > t else 0)
                nkey = (e.dest, e.line, nt if limit is not None else 0)
                if ncost < best_cost.get(nkey, float("inf")):
                    best_cost[nkey] = ncost
                    self._info[nkey] = (ncost, d + e.distance, nt, key, e)
                    pq.push(ncost, nkey)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        key = self._best.get(target)
        if key is None:
            return [], []
        nodes, edges = [], []
        while key is not None:
            _, _, _, prev, e = self._info[key]
            nodes.append(key[0])
            if e is not None:
                edges.append(e)
            key = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        key = self._best.get(target)
        if key is None:
            return 0
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (fractional km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km < 22:
        return 5
    if km <= 32:
        return 6
    return 7 + (km - 32 - 1) // 20


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 1.5 min/km plus 5 min per transfer."""
    return distance / 1000.0 * 1.5 + 5 * transfer_count

"""Network building, data checks and data updates."""
import difflib

from models import Edge, Graph


class SubwayNetwork:
    def __init__(self, data):
        self.data = data
        self.build()

    def build(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        for s in self.data.get("stations", []):
            self._ensure(s["name"])
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a = self._ensure(c["source"])
            b = self._ensure(c["dest"])
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))

    def _ensure(self, name):
        if name not in self.station_to_number:
            n = len(self.station_to_number) + 1
            self.station_to_number[name] = n
            self.number_to_station[n] = name
            self.graph.add_node(n)
        return self.station_to_number[name]

    def match_station(self, name):
        """Return (matched_name, exact)."""
        if name in self.station_to_number:
            return name, True
        m = difflib.get_close_matches(name, list(self.station_to_number), n=1, cutoff=0.5)
        if not m:
            m = [k for k in self.station_to_number if name and (name in k or k in name)][:1]
        return (m[0] if m else None), False

    def lines(self):
        return sorted({c["line"] for c in self.data.get("connections", [])})

    # ---- data updates ----
    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: meters between consecutive stations (bidirectional)."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances length must be len(stations)-1")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line not in known[n]["lines"]:
                    known[n]["lines"].append(line)
            else:
                self.data["stations"].append({"name": n, "lines": [line]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line})
            self.data["connections"].append({"source": b, "dest": a, "distance": d, "line": line})
        self.build()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        ok = False
        for c in self.data["connections"]:
            if c["source"] == a and c["dest"] == b and (line is None or c["line"] == line):
                c["distance"] = distance
                ok = True
        if ok:
            self.build()
        return ok

    # ---- integrity ----
    def check_data(self):
        problems = []
        names = {s["name"] for s in self.data.get("stations", [])}
        if not names:
            problems.append("No station data")
        conns = self.data.get("connections", [])
        if not conns:
            problems.append("No connection data")
        for s in self.data.get("stations", []):
            if not s.get("lines"):
                problems.append("Station without lines: %s" % s["name"])
        for c in conns:
            for k in ("source", "dest"):
                if c[k] not in names:
                    problems.append("Connection references unknown station: %s" % c[k])
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                problems.append("Invalid distance %s-%s: %r" % (c["source"], c["dest"], d))
            elif d > 20000:
                problems.append("Suspicious distance %s-%s: %s" % (c["source"], c["dest"], d))
        connected = {c["source"] for c in conns} | {c["dest"] for c in conns}
        for n in sorted(names - connected):
            problems.append("Station without connections: %s" % n)
        return problems

"""CLI for Beijing Subway fare calculation."""
import argparse
import json
import re
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath
from subway_system import SubwayNetwork


def parse_input(text):
    parts = [p.strip() for p in re.split(r"[;；,，]", text) if p.strip()]
    if len(parts) != 2:
        return None
    return parts


def build_result(net, src, dst, config):
    s, t = net.station_to_number[src], net.station_to_number[dst]
    sp = ShortestPath(net.graph, s, config)
    nodes, edges = sp.path_with_edges(t)
    if not nodes:
        raise RuntimeError("No route found from '%s' to '%s'" % (src, dst))
    dist = sp.dist_to[t]
    transfers = sp.get_transfer_count(t)
    fare = Sum_money(dist)
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = net.number_to_station[e.dest]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(net.number_to_station[e.source])
            segments.append({"line": e.line, "from": net.number_to_station[e.source],
                             "to": net.number_to_station[e.dest], "stations": 1})
    return {
        "query": {"start": src, "end": dst, "mode": config.mode},
        "path": [net.number_to_station[n] for n in nodes],
        "distance": dist,
        "fare": fare,
        "estimated_time": round(calculate_estimated_time(dist, transfers), 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": list(dict.fromkeys(e.line for e in edges)),
        "segments": segments,
    }


def format_distance(m):
    return "%d m" % m if m < 1000 else "%.2f km" % (m / 1000.0)


def format_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Distance: %s (%d m, %.2f km)" % (format_distance(r["distance"]), r["distance"], r["distance"] / 1000.0)]
    for seg in r["segments"]:
        out.append("  [%s] %s -> %s (%d stations)" % (seg["line"], seg["from"], seg["to"], seg["stations"]))
    if r["transfer_stations"]:
        out.append("Transfers (%d): %s" % (r["transfer_count"], ", ".join(r["transfer_stations"])))
    else:
        out.append("Transfers: 0")
    out.append("Lines: %s" % ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def process(net, text, config, fmt):
    parts = parse_input(text)
    if parts is None:
        return "Input format incorrect, please use 'Starting Station; Ending Station' format"
    names = []
    for label, raw in zip(("Starting", "Ending"), parts):
        m, exact = net.match_station(raw)
        if m is None:
            return "%s station '%s' does not exist, please check the station name" % (label, raw)
        if not exact:
            print("Note: station '%s' not found, using nearest match '%s'" % (raw, m))
        names.append(m)
    if names[0] == names[1]:
        return "Starting station cannot be the same as the ending station"
    try:
        r = build_result(net, names[0], names[1], config)
    except ValueError as e:
        return "Fare calculation error: %s" % e
    except Exception as e:
        return "Path calculation error: %s" % e
    return json.dumps(r, ensure_ascii=False, indent=2) if fmt == "json" else format_text(r)


def main(argv=None):
    ap = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    ap.add_argument("--output_file")
    ap.add_argument("--format", choices=["text", "json"], default="text")
    ap.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                    default="shortest_distance")
    ap.add_argument("--max_transfers", type=int)
    ap.add_argument("--line_preference")
    ap.add_argument("--check_data", action="store_true")
    a = ap.parse_args(argv)

    try:
        net = SubwayNetwork(load_subway_data(DATA_FILE_PATH))
    except Exception as e:
        print("Failed to load data: %s" % e)
        return 1

    if a.check_data:
        probs = net.check_data()
        print("Stations: %d, lines: %s" % (len(net.station_to_number), ", ".join(net.lines())))
        if probs:
            print("Found %d problem(s):" % len(probs))
            for p in probs[:50]:
                print(" -", p)
            return 1
        print("Data check passed")
        return 0

    cfg = PathfindingConfig()
    cfg.mode = a.mode
    cfg.max_transfers = a.max_transfers
    cfg.preferred_lines = [a.line_preference] if a.line_preference else []

    outputs = []
    print("Enter 'Starting Station; Ending Station' (or 'exit' to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("exit", "quit", "q"):
            break
        res = process(net, line, cfg, a.format)
        print(res)
        outputs.append(res)
        if a.output_file:
            try:
                with open(a.output_file, "w", encoding="utf-8") as f:
                    f.write("\n\n".join(outputs) + "\n")
            except OSError as e:
                print("Cannot write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

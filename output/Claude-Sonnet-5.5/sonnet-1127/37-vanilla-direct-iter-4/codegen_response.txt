DEFAULT_LINES = [

class ListManager:
    def _parse_lines(self, lines):
        users, errors = [], []
        for i, raw in enumerate(lines, 1):
            line = raw.strip()
            if not line:
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Line {i}: Format Error: '{line}' (expected 'Name,Department')")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def load_default_list(self):
        return self._parse_lines(DEFAULT_LINES)

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except FileNotFoundError:
            return [], [f"File not found: {path}"]
        except Exception as e:
            return [], [f"Failed to read file {path}: {e}"]
        return self._parse_lines(lines)

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, qty in defaults:
            try:
                self.add_prize(name, qty)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        prize = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "cheng ru group"
SPECIAL_DRAWER = "Li Bin"
PROTECTED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_TARGET = "Ma Jiaqi"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _is_cheng_ru(self, user):
        return user.department.strip().lower() == CHENG_RU

    def _get_eligible_pool(self, prize):
        won = {id(u) for u in self.winners}
        won_names = {u.name for u in self.winners}
        pool = [u for u in self.users if id(u) not in won and u.name not in won_names]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        avoid = {g.strip().lower() for g in (prize.avoid_groups or [])}
        pool = [u for u in pool if u.department.strip().lower() not in avoid]
        if (drawer == PROTECTED_TARGET and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            won_pool = [u for u in self.users
                        if u.name not in won_names and self._is_cheng_ru(u)]
            return won_pool
        return pool

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def record_round(self, prize, round_winners):
        if prize.associated_drawer in PROTECTED_DRAWERS:
            self.last_two_draw_results.append(
                any(self._is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, win_time=None):
        self.user = user
        self.prize_name = prize_name
        self.win_time = win_time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize.name))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in reversed(list(prizes)):
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for ws in self.get_winners_by_prize(prizes).values():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{w.prize_name},"
                                f"{w.win_time:%Y-%m-%d %H:%M:%S}\n")
        except Exception as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt file > ").strip()
        if c == "1":
            users, errors = lm.load_default_list()
        elif c == "2":
            users, errors = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errors:
            print("  !", e)
        if errors:
            a = input("Errors found. (r)emove invalid and continue / (t)erminate: ").strip().lower()
            if a != "r":
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"  ! Duplicate name removed: {d.name} ({d.department})")
        if valid:
            return valid
        print("No valid users.")


def setup_prizes(pm, groups):
    pm.load_default_prizes()
    while input("Add custom prize? (y/n) > ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            qty = int(input("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        drawer = input("Associated drawer (blank for none): ").strip() or None
        try:
            pm.add_prize(name, qty, drawer and None or None, drawer)
        except ValueError as e:
            print(e)
    print("Groups:", ", ".join(groups))
    for p in pm.get_prizes():
        a = input(f"Avoid groups for '{p.name}' (comma separated, blank none): ").strip()
        if a:
            p.avoid_groups = [g.strip() for g in a.split(",") if g.strip()]
    show_prizes(pm)
    o = input("New order (e.g. 3 1 2), blank keeps: ").strip()
    if o:
        try:
            ok = pm.reorder_prizes([int(x) for x in o.replace(",", " ").split()])
        except ValueError:
            ok = False
        print("Reordered." if ok else "Invalid order, kept original.")


def show_prizes(pm):
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")


def run_round(lot, prize, rm):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | avoid: "
          f"{', '.join(prize.avoid_groups) or 'none'} | drawer: {prize.associated_drawer or 'none'}")
    got = []
    i = 1
    while i <= prize.quantity:
        cmd = input("Enter to draw (pause/continue/terminate) > ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue' > ").strip().lower() != "continue":
                pass
            continue
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        rm.add_winner(u, prize)
        got.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.record_round(prize, got)
    print("Round results:", "; ".join(f"{u.name} ({u.department})" for u in got) or "none")


def main():
    lm = ListManager()
    users = load_users(lm)
    info = lm.get_group_info(users)
    for g, c in info.items():
        print(f"  {g}: {c}")
    pm = PrizeManager()
    setup_prizes(pm, list(info))
    lot = LotteryManager(users)
    rm = ResultManager()
    for p in pm.get_prizes():
        run_round(lot, p, rm)
    print("\n=== Final winner list ===")
    for name, ws in rm.get_winners_by_prize(pm.get_prizes()).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {name}")
    ok, msg = rm.export_to_csv(pm.get_prizes())
    print(msg)


if __name__ == "__main__":
    main()

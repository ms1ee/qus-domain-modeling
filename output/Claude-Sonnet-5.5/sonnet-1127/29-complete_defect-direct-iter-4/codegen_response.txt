"""Grid-based path planning."""

CELL = 100


def to_center(v):
    """Convert a coordinate (0-800) to its grid cell center."""
    g = min(max(int(v) // CELL, 0), 7)
    return g * CELL + CELL // 2


class PathPlanner:
    """Plans right-angle paths on the grid."""

    @staticmethod
    def plan(start, target):
        """Return waypoints from start to target (horizontal then vertical).

        If the rider is directly above/below the target only a vertical
        leg is produced.
        """
        sx, sy = start
        tx, ty = target
        pts = [(sx, sy)]
        if sx != tx:
            pts.append((tx, sy))
        if sy != ty:
            pts.append((tx, ty))
        return pts

    @staticmethod
    def plan_multi(start, targets):
        """Continuous path through several destinations."""
        path = [tuple(start)]
        cur = tuple(start)
        for t in targets:
            seg = PathPlanner.plan(cur, tuple(t))
            path.extend(seg[1:])
            cur = tuple(t)
        return path

    @staticmethod
    def step(pos, target, speed):
        """Move from pos toward target by at most speed units."""
        x, y = pos
        tx, ty = target
        budget = speed
        if x != tx:
            d = min(budget, abs(tx - x))
            x += d if tx > x else -d
            budget -= d
        if budget > 0 and y != ty:
            d = min(budget, abs(ty - y))
            y += d if ty > y else -d
        return x, y

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Order data model."""
import globalvar
from path_planner import to_center, PathPlanner


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        n = (globalvar.get_value("order_id_counter", 0) or 0) + 1
        globalvar.set_value("order_id_counter", n)
        self.id = n
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.complete_time = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self):
        """Mark as completed."""
        self.status = "completed"
        self.complete_time = globalvar.get_value("time", 0) or 0

    @property
    def time_taken(self):
        if self.complete_time is None:
            return None
        return self.complete_time - self.start_time

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "complete_time": self.complete_time,
                "distance": self.distance}

"""Rider model."""
import globalvar
from path_planner import PathPlanner

IDLE, PICKING, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 50
        self.y = 50
        self.speed = 50
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0) or 0,
                         "x": self.x, "y": self.y})

    def move(self):
        """Advance one tick."""
        if self.state == IDLE and self.orders:
            self.state = PICKING
            self._log("accepted_order", self.orders[0].id)
        if self.state == IDLE or not self.orders:
            return
        order = self.orders[0]
        if self.state == PICKING:
            self.x, self.y = PathPlanner.step(
                (self.x, self.y), order.res_pos, self.speed)
            if (self.x, self.y) == tuple(order.res_pos):
                self.state = DELIVERING
                self._log("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            self.x, self.y = PathPlanner.step(
                (self.x, self.y), order.cus_pos, self.speed)
            if (self.x, self.y) == tuple(order.cus_pos):
                self._log("delivered_to_customer", order.id)
                order.complete()
                self.orders.pop(0)
                self.state = IDLE

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order management: assignment and creation."""
import globalvar
from order_model import Order
from path_planner import PathPlanner


def _effective_pos(rider):
    if rider.orders:
        return rider.orders[-1].cus_pos
    return (rider.x, rider.y)


def find_best_rider_for_order(order):
    """Return the closest rider (ties broken by lighter load), or None."""
    riders = globalvar.get_value("riders", []) or []
    best, best_key = None, None
    for r in riders:
        key = (PathPlanner.distance(_effective_pos(r), order.res_pos),
               len(r.orders), r.rider_id)
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


class OrderManager:
    """Creates and assigns orders."""

    @staticmethod
    def create_order(coords):
        """Create an order; returns (order, new_rider_added, rider)."""
        import system
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        added = system.SystemManager.auto_recruit() > 0
        rider = find_best_rider_for_order(order)
        orders = globalvar.get_value("orders", [])
        orders.append(order)
        globalvar.set_value("orders", orders)
        if rider is not None:
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
        globalvar.set_value("funds", globalvar.get_value("funds", 0) + 100)
        return order, added, rider

"""System manager."""
import json
import globalvar
from rider import Rider

RECRUIT_COST = 300


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", 300)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_id_counter", 0)


class SystemManager:
    """Funds, riders and ticking."""

    @staticmethod
    def add_rider():
        n = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", n)
        r = Rider(n)
        globalvar.get_value("riders").append(r)
        return r

    @staticmethod
    def auto_recruit():
        """Recruit riders while funds >= 300. Returns number added."""
        count = 0
        while globalvar.get_value("funds", 0) >= RECRUIT_COST:
            globalvar.set_value(
                "funds", globalvar.get_value("funds") - RECRUIT_COST)
            SystemManager.add_rider()
            count += 1
        return count

    @staticmethod
    def tick():
        """Advance simulated time one unit."""
        globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
        for r in globalvar.get_value("riders", []):
            r.move()

    @staticmethod
    def save(path="state.json"):
        data = {"funds": globalvar.get_value("funds"),
                "time": globalvar.get_value("time"),
                "riders": [r.to_dict() for r in globalvar.get_value("riders")],
                "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Display manager."""
import globalvar
from path_planner import PathPlanner

NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and ASCII map rendering."""

    @staticmethod
    def render_map(riders=(), orders=(), route=None):
        grid = [["." for _ in range(8)] for _ in range(8)]
        if route:
            for (a, b) in zip(route, route[1:]):
                n = max(abs(a[0] - b[0]), abs(a[1] - b[1])) // 50 + 1
                for i in range(n + 1):
                    x = a[0] + (b[0] - a[0]) * i // max(n, 1)
                    y = a[1] + (b[1] - a[1]) * i // max(n, 1)
                    grid[min(y // 100, 7)][min(x // 100, 7)] = "#"
        for o in orders:
            if o.status != "completed":
                gx, gy = o.res_pos[0] // 100, o.res_pos[1] // 100
                grid[gy][gx] = "S"
                gx, gy = o.cus_pos[0] // 100, o.cus_pos[1] // 100
                grid[gy][gx] = "C"
        for r in riders:
            grid[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
        lines = []
        for row in range(7, -1, -1):
            lines.append(" ".join(grid[row]))
        return "\n".join(lines)

    @staticmethod
    def status():
        riders = globalvar.get_value("riders", [])
        out = ["Time: %s  Funds: %s  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            out.append("Rider %d at (%d,%d) speed %d %s orders=%s" % (
                r.rider_id, r.x, r.y, r.speed, NAMES[r.state],
                [o.id for o in r.orders]))
        out.append(DisplayManager.render_map(
            riders, globalvar.get_value("orders", [])))
        return "\n".join(out)

    @staticmethod
    def orders():
        orders = globalvar.get_value("orders", [])
        if not orders:
            return "No orders."
        out = []
        for o in orders:
            out.append("Order %d start=%s rest=%s cust=%s dist=%d time=%s "
                       "rider=%s status=%s" % (
                           o.id, o.start_time, o.res_pos, o.cus_pos,
                           o.distance, o.time_taken, o.rider_id, o.status))
        return "\n".join(out)

    @staticmethod
    def riders_log():
        out = []
        for r in globalvar.get_value("riders", []):
            for e in r.log:
                out.append("t=%s rider %d %s order=%s at (%s,%s)" % (
                    e["time"], r.rider_id, e["event"], e["order_id"],
                    e["x"], e["y"]))
        return "\n".join(out) or "No rider records."

"""Command parsing and validation."""


class InputError(ValueError):
    """Invalid user input."""


class InputManager:
    """Parses commands."""

    ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
               "q": "quit", "exit": "quit", "t": "tick"}

    def __init__(self):
        self.history = []

    def parse(self, line):
        """Return (command, args)."""
        parts = line.strip().split()
        if not parts:
            raise InputError("empty command")
        self.history.append(line.strip())
        cmd = parts[0].lower()
        cmd = self.ALIASES.get(cmd, cmd)
        if cmd not in ("start", "status", "orders", "riders", "quit",
                       "tick", "history"):
            raise InputError("unknown command: %s" % parts[0])
        return cmd, parts[1:]

    @staticmethod
    def parse_coords(args):
        """Validate four integers in 0-800."""
        if len(args) != 4:
            raise InputError("need 4 integers: rx ry cx cy")
        vals = []
        for a in args:
            try:
                vals.append(int(a.strip("[],")))
            except ValueError:
                raise InputError("not an integer: %s" % a)
        for v in vals:
            if not 0 <= v <= 800:
                raise InputError("coordinate out of range 0-800: %d" % v)
        return vals

"""Command-line entry point."""
import globalvar
from system import init_system, SystemManager
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager, InputError
from path_planner import PathPlanner


def handle_start(args):
    coords = InputManager.parse_coords(args)
    order, added, rider = OrderManager.create_order(coords)
    out = ["Restaurant grid: (%d,%d)  Customer grid: (%d,%d)" % (
        order.res_pos[0] // 100, order.res_pos[1] // 100,
        order.cus_pos[0] // 100, order.cus_pos[1] // 100),
        "New rider added: %s" % ("yes" if added else "no")]
    if rider is None:
        out.append("No rider available.")
    else:
        route = PathPlanner.plan_multi((rider.x, rider.y),
                                       [order.res_pos, order.cus_pos])
        dist = sum(PathPlanner.distance(a, b)
                   for a, b in zip(route, route[1:]))
        out.append("Rider %d speed %d, estimated time %d ticks" % (
            rider.rider_id, rider.speed, -(-dist // rider.speed)))
        out.append(DisplayManager.render_map([rider], [order], route))
    return "\n".join(out)


def main():
    init_system()
    im = InputManager()
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except EOFError:
            break
        try:
            cmd, args = im.parse(line)
            if cmd == "quit":
                globalvar.set_value("running", False)
                continue
            if cmd == "start":
                print(handle_start(args))
            elif cmd == "status":
                print(DisplayManager.status())
            elif cmd == "orders":
                print(DisplayManager.orders())
            elif cmd == "riders":
                print(DisplayManager.riders_log())
            elif cmd == "history":
                print("\n".join(im.history))
            elif cmd == "tick":
                n = int(args[0]) if args and args[0].isdigit() else 1
                for _ in range(n):
                    SystemManager.tick()
                print("Time: %s" % globalvar.get_value("time"))
                continue
        except InputError as e:
            print("Error: %s" % e)
        except Exception as e:  # recovery
            print("Unexpected error: %s" % e)
        SystemManager.tick()
    try:
        SystemManager.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

import os
from lottery_system.models import User


class ListManager:
    DEFAULT_DATA = [
        ("Zhang San", "Technology Department"),
        ("Li Si", "Technology Department"),
        ("Wang Wu", "Product Department"),
        ("Zhao Liu", "Product Department"),
        ("Chen Qixian", "Market Department"),
        ("Jin Duo", "Market Department"),
        ("Ma Jiaqi", "Operations Department"),
        ("Li Bin", "Operations Department"),
        ("Sun Qi", "Cheng Ru group"),
        ("Zhou Ba", "Cheng Ru group"),
        ("Wu Jiu", "Cheng Ru group"),
        ("Zheng Shi", "Technology Department"),
        ("Qian Yi", "Product Department"),
        ("He Er", "Market Department"),
    ]

    def load_default_list(self):
        return [User(n, d) for n, d in self.DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        if not path or not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except Exception as e:
            return [], [f"Cannot read file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        presets = [
            ("Third Prize - Movie Tickets", 3, "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, "Ma Jiaqi"),
        ]
        for name, q, d in presets:
            try:
                self.add_prize(name, q, [], d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        self.prizes.append(Prize(name, quantity, list(avoid_groups or []), associated_drawer))

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHEN = "Chen Qixian"
JIN = "Jin Duo"
MA = "Ma Jiaqi"
LI_BIN = "Li Bin"
CHENG_RU = "cheng ru"


def _is_cheng_ru(user):
    return CHENG_RU in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _unwon(self):
        won = {w.name for w in self.winners}
        return [u for u in self.users if u.name not in won]

    def _get_eligible_pool(self, prize):
        pool = self._unwon()
        drawer = prize.associated_drawer
        if drawer == LI_BIN:
            return pool
        if (drawer == MA and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_cheng_ru(u)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def describe_rules(self, prize):
        d = prize.associated_drawer
        parts = ["Exclude previous winners"]
        if d == LI_BIN:
            parts = ["Exclude previous winners only (avoidance groups ignored)"]
        elif prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if d == MA:
            parts.append("fallback: if the last two draws (Chen Qixian, Jin Duo) had no "
                         "Cheng Ru group winner, pool limited to Cheng Ru group")
        return "; ".join(parts)

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def finish_round(self, prize, round_winners):
        if prize.associated_drawer in (CHEN, JIN):
            self.last_two_draw_results.append(any(_is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name="", time=None):
        self.user = user
        self.prize_name = prize_name
        self.time = time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for pname, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{pname},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
            return True, f"Results successfully exported to {path}"
        except Exception as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_list(lm):
    c = input("1) default list  2) custom txt file: ").strip()
    if c == "1":
        users, errs = lm.load_default_list()
    else:
        users, errs = lm.load_custom_list(input("Path: ").strip())
    for e in errs:
        print("  ", e)
    valid, dups = lm.validate_users(users)
    for d in dups:
        print(f"   Duplicate name: {d.name} ({d.department})")
    if errs or dups:
        if input("Remove invalid data and continue? (y = remove, n = terminate): ").strip().lower() != "y":
            print("Upload terminated.")
            return None
    print(f"Loaded {len(valid)} participants.")
    return valid


def show_groups(lm, users):
    if not users:
        print("No list loaded.")
        return
    for g, n in lm.get_group_info(users).items():
        print(f"  {g}: {n}")


def prize_menu(pm):
    while True:
        prizes = pm.get_prizes()
        for i, p in enumerate(prizes, 1):
            print(f"  {i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
        c = input("a) add  d) load presets  r) reorder  b) back: ").strip().lower()
        if c == "a":
            name = input("Name: ").strip()
            try:
                q = int(input("Quantity: "))
                avoid = [x.strip() for x in input("Avoid groups (comma separated): ").split(",") if x.strip()]
                drawer = input("Lottery person (optional): ").strip() or None
                pm.add_prize(name, q, avoid, drawer)
            except ValueError as e:
                print("Invalid:", e)
        elif c == "d":
            pm.load_default_prizes()
        elif c == "r":
            try:
                order = [int(x) for x in input("New order (e.g. 2,1,3): ").replace(" ", "").split(",")]
            except ValueError:
                order = []
            print("Reordered." if pm.reorder_prizes(order) else "Invalid order.")
        elif c == "b":
            return


def run_lottery(users, pm, rm):
    lot = LotteryManager(users)
    for idx, prize in enumerate(pm.get_prizes(), 1):
        print(f"\n=== Round {idx}: {prize.name} | quantity {prize.quantity} ===")
        print("Rules:", lot.describe_rules(prize))
        won = []
        n = 0
        while n < prize.quantity:
            cmd = input(f"Enter to draw winner {n + 1} (pause/terminate): ").strip().lower()
            if cmd == "pause":
                print("Paused.")
                while True:
                    c2 = input("Type 'continue' or 'terminate': ").strip().lower()
                    if c2 in ("continue", "terminate"):
                        cmd = c2
                        break
                if cmd == "terminate":
                    print("Round terminated.")
                    break
                continue
            if cmd == "terminate":
                print("Round terminated.")
                break
            print(f"Currently drawing the {n + 1}th winner...")
            u = lot.draw_one(prize)
            if u is None:
                print("No eligible participants left.")
                break
            rm.add_winner(u, prize)
            won.append(u)
            n += 1
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        lot.finish_round(prize, won)
        print("Round results:")
        for i, u in enumerate(won, 1):
            print(f"  {i}. {u.name} ({u.department})")


def summary(pm, rm):
    ordered = list(reversed(pm.get_prizes()))
    for pname, ws in rm.get_winners_by_prize(ordered).items():
        print(pname)
        for w in ws:
            print(f"  {w.user.name} ({w.user.department})  {w.time:%H:%M:%S}")
    return ordered


def main():
    lm = ListManager()
    pm = PrizeManager()
    rm = ResultManager()
    users = []
    while True:
        print("\n1 Load list  2 View groups  3 Prizes  4 Start lottery  5 Export  0 Exit")
        c = input("> ").strip()
        if c == "1":
            r = load_list(lm)
            if r is not None:
                users = r
        elif c == "2":
            show_groups(lm, users)
        elif c == "3":
            prize_menu(pm)
        elif c == "4":
            if not users or not pm.get_prizes():
                print("Load a list and configure prizes first.")
                continue
            rm = ResultManager()
            run_lottery(users, pm, rm)
            print("\n=== Final winners ===")
            summary(pm, rm)
        elif c == "5":
            print(rm.export_to_csv(list(reversed(pm.get_prizes())))[1])
        elif c == "0":
            break


if __name__ == "__main__":
    main()

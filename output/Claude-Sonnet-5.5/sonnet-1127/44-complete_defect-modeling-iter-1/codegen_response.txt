"""Core data models: Edge and Graph."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}, {self.line})"


class Graph:
    """Undirected weighted graph; add_edge stores both directions."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

    def edges(self):
        return [e for lst in self.adj.values() for e in lst]

"""Dijkstra shortest path with a heap based priority queue."""
import heapq
import itertools


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    def __init__(self):
        self.heap = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap


TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 100000, "comprehensive": 1000}


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}
        self._info = {}  # state -> (cost, dist, prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        pen = TRANSFER_PENALTY.get(cfg.mode, 0)
        maxt = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        self._info[start] = (0.0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in self._best:
                self._best[node] = st
                self.dist_to[node] = self._info[st][1]
            dist = self._info[st][1]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0
                if line is not None and e.line != line:
                    nt = t + 1
                    extra = pen
                if maxt is not None and nt > maxt:
                    continue
                w = e.weight * (0.9 if e.line in pref else 1.0)
                ns = (e.dest, e.line, nt if maxt is not None else 0)
                nc = cost + w + extra
                if ns in done:
                    continue
                if ns not in self._info or nc < self._info[ns][0]:
                    self._info[ns] = (nc, dist + e.weight, st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        st = self._best[target]
        nodes, edges = [], []
        while st is not None:
            _, _, prev, e = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation for the Beijing Subway."""
import math

FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]
AVERAGE_SPEED_KMH = 35.0
TRANSFER_MINUTES = 5.0


def Sum_money(distance):
    """Fare in yuan for a distance in meters (km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0 - 1e-9)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes for a distance in meters."""
    return distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0 + transfer_count * TRANSFER_MINUTES

"""Network construction, integrity checks and data updates."""
from data_loader import load_subway_data, DATA_FILE_PATH
from models import Graph, Edge

REQUIRED_LINES = ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8",
                  "Line 10", "Line 13", "Line 14", "Line 15"]


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.build()

    def build(self):
        names = {s["name"] for s in self.data.get("stations", [])}
        for c in self.data.get("connections", []):
            names.add(c["source"])
            names.add(c["dest"])
        self.station_to_number = {n: i for i, n in enumerate(sorted(names))}
        self.number_to_station = {i: n for n, i in self.station_to_number.items()}
        self.graph = Graph()
        for n in self.station_to_number.values():
            self.graph.add_node(n)
        for s in self.data.get("stations", []):
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            self.graph.add_edge(Edge(self.station_to_number[c["source"]],
                                     self.station_to_number[c["dest"]],
                                     c["distance"], c["line"]))

    def lines_of(self, name):
        return self.graph.station_lines.get(name, [])

    def is_transfer(self, name):
        return len(self.lines_of(name)) > 1

    def check_data(self):
        """Return list of problem strings (empty if ok)."""
        problems = []
        known = {s["name"] for s in self.data.get("stations", [])}
        lines = {l for s in self.data.get("stations", []) for l in s.get("lines", [])}
        for l in REQUIRED_LINES:
            if l not in lines:
                problems.append(f"Missing line: {l}")
        connected = set()
        for c in self.data.get("connections", []):
            connected.add(c["source"])
            connected.add(c["dest"])
            for k in ("source", "dest"):
                if c[k] not in known:
                    problems.append(f"Connection references unknown station: {c[k]}")
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
                problems.append(f"Invalid distance {d} between {c['source']} and {c['dest']}")
        for n in known - connected:
            problems.append(f"Station without connections: {n}")
        return problems

    # ---- update mechanisms ----
    def add_station_line(self, name, line):
        for s in self.data["stations"]:
            if s["name"] == name:
                if line not in s["lines"]:
                    s["lines"].append(line)
                break
        else:
            self.data["stations"].append({"name": name, "lines": [line]})
        self.build()

    def add_line(self, line, stations, distances):
        """Add a line from a station sequence and len(stations)-1 distances."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        for s in stations:
            self.add_station_line(s, line)
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line})
        self.build()

    def rename_station(self, old, new):
        for s in self.data["stations"]:
            if s["name"] == old:
                s["name"] = new
        for c in self.data["connections"]:
            if c["source"] == old:
                c["source"] = new
            if c["dest"] == old:
                c["dest"] = new
        self.build()

    def update_distance(self, a, b, distance):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                found = True
        if not found:
            raise KeyError(f"No connection between {a} and {b}")
        self.build()

"""Route reports (text / JSON)."""
import json


def format_distance(m):
    if m < 1000:
        return f"{m} m"
    return f"{m / 1000.0:.2f} km"


def build_summary(query, route):
    return {
        "query": query,
        "path": route["stations"],
        "distance_m": route["distance"],
        "distance_km": round(route["distance"] / 1000.0, 2),
        "fare": route["fare"],
        "estimated_time_min": round(route["time"], 1),
        "transfer_count": route["transfers"],
        "transfer_stations": [t[0] for t in route["transfer_details"]],
        "lines": route["lines"],
    }


def to_json(query, route):
    return json.dumps(build_summary(query, route), ensure_ascii=False, indent=2)


def to_text(query, route):
    out = [f"Route: {query['start']} -> {query['end']}",
           "Stations: " + " -> ".join(route["stations"])]
    out.append("Segments:")
    for line, a, b in route["segments"]:
        out.append(f"  {line}: {a} -> {b}")
    out.append(f"Total distance: {format_distance(route['distance'])}")
    out.append(f"Transfers: {route['transfers']}")
    for st, fl, tl in route["transfer_details"]:
        out.append(f"  Transfer at {st}: {fl} -> {tl}")
    out.append("Lines: " + ", ".join(route["lines"]))
    out.append(f"Estimated time: {route['time']:.0f} min")
    out.append(f"Fare: {route['fare']} yuan")
    return "\n".join(out)

"""CLI entry point for the Beijing Subway fare system."""
import argparse
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_manager import DataManager
from pathfinding import ShortestPath, PathfindingConfig
from fare_calculator import Sum_money, calculate_estimated_time
import report

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def parse_input(text):
    text = text.replace("；", ";")
    parts = [x.strip() for x in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERR)
    return parts[0], parts[1]


def compute_route(dm, start, end, config):
    s, e = dm.station_to_number[start], dm.station_to_number[end]
    sp = ShortestPath(dm.graph, s, config)
    nodes, edges = sp.path_with_edges(e)
    if not nodes:
        raise RuntimeError("No route found between the stations")
    dist = sum(x.distance for x in edges)
    transfers = sp.get_transfer_count(e)
    segments, lines, details = [], [], []
    for i, ed in enumerate(edges):
        a, b = dm.number_to_station[ed.source], dm.number_to_station[ed.dest]
        if segments and segments[-1][0] == ed.line:
            segments[-1][2] = b
        else:
            if segments:
                details.append((a, segments[-1][0], ed.line))
            segments.append([ed.line, a, b])
        if ed.line not in lines:
            lines.append(ed.line)
    try:
        fare = Sum_money(dist)
    except Exception as ex:
        raise ArithmeticError(f"Fare calculation failed: {ex}")
    return {"stations": [dm.number_to_station[n] for n in nodes], "distance": dist,
            "fare": fare, "time": calculate_estimated_time(dist, transfers),
            "transfers": transfers, "lines": lines, "segments": segments,
            "transfer_details": details}


def handle(dm, line, args, config):
    start, end = parse_input(line)
    if start not in dm.station_to_number:
        raise ValueError(f"Starting station '{start}' does not exist, please check the station name")
    if end not in dm.station_to_number:
        raise ValueError(f"Ending station '{end}' does not exist, please check the station name")
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    query = {"start": start, "end": end, "mode": args.mode,
             "max_transfers": args.max_transfers, "line_preference": args.line_preference}
    route = compute_route(dm, start, end, config)
    return report.to_json(query, route) if args.format == "json" else report.to_text(query, route)


def main(argv=None):
    args = parse_args(argv)
    dm = DataManager()
    if args.check_data:
        problems = dm.check_data()
        if problems:
            print("Data check found problems:")
            for p in problems:
                print(" -", p)
            return 1
        print("Data check passed.")
        return 0
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []
    print("Enter 'Starting Station; Ending Station' (empty line or 'quit' to exit)")
    while True:
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if not line or line.lower() in ("quit", "exit", "q"):
            break
        try:
            result = handle(dm, line, args, config)
        except ValueError as ex:
            print(ex)
            continue
        except Exception as ex:
            print(f"Error: {ex}")
            continue
        print(result)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(result + "\n\n")
            except OSError as ex:
                print(f"Could not write output file: {ex}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

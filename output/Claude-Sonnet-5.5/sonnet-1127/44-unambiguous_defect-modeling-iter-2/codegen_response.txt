"""Core data structures: Edge, Graph, MinPQ."""
import heapq
import itertools


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}, {self.line})"


class Graph:
    """Weighted undirected graph with adjacency lists (station numbers)."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adjacency.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adjacency[edge.source].append(edge)
        self.adjacency[edge.dest].append(edge.reversed())

    def edges_from(self, n):
        return self.adjacency.get(n, [])

    def nodes(self):
        return list(self.adjacency.keys())

    def node_count(self):
        return len(self.adjacency)

    def edge_count(self):
        return sum(len(v) for v in self.adjacency.values()) // 2


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)

"""Dijkstra shortest path with transfer-aware states."""
from models import MinPQ

SHORTEST_DISTANCE = "shortest_distance"
LEAST_TRANSFER = "least_transfer"
COMPREHENSIVE = "comprehensive"
MODES = (SHORTEST_DISTANCE, LEAST_TRANSFER, COMPREHENSIVE)

TRANSFER_PENALTY = {SHORTEST_DISTANCE: 0, LEAST_TRANSFER: 100000, COMPREHENSIVE: 1500}


class PathfindingConfig:
    def __init__(self, mode=SHORTEST_DISTANCE, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}   # node -> state
        self._info = {}   # state -> (real_dist, transfers, parent_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        start = (self.source, None)
        cost = {start: 0.0}
        self._info[start] = (0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, last = st
            if node not in self._best:
                self._best[node] = st
            real, tr, _, _ = self._info[st]
            for e in self.graph.edges_from(node):
                t = tr
                extra = 0
                if last is not None and e.line != last:
                    t += 1
                    extra = penalty
                if cfg.max_transfers is not None and t > cfg.max_transfers:
                    continue
                w = e.weight
                if e.line in cfg.preferred_lines:
                    w *= 0.9
                nc = c + w + extra
                ns = (e.dest, e.line)
                if ns in done:
                    continue
                if nc < cost.get(ns, float("inf")):
                    cost[ns] = nc
                    self._info[ns] = (real + e.weight, t, st, e)
                    pq.push(nc, ns)
        for n, st in self._best.items():
            self.dist_to[n] = self._info[st][0]

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            raise ValueError("No path to target")
        nodes, edges = [], []
        st = self._best[target]
        while st is not None:
            _, _, parent, e = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = parent
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""

FARE_TIERS = [(0, 6000, 3), (6000, 12000, 4), (12000, 22000, 5), (22000, 32000, 6)]


def Sum_money(distance) -> int:
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("Distance must be a non-negative number")
    d = int(round(distance))
    for lo, hi, fare in FARE_TIERS:
        if lo <= d < hi:
            return fare
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated minutes: ~35 km/h average, +2 min boarding, +5 min per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("Invalid distance or transfer count")
    return round(distance / 1000.0 / 35.0 * 60.0 + 2 + 5 * transfer_count, 1)


class FareCalculator:
    def calculate(self, distance):
        return Sum_money(distance)

"""Data management: loading, graph building, checks and updates."""
from data_loader import load_subway_data, DATA_FILE_PATH
from models import Graph, Edge

REPRESENTATIVE = {
    "1号线": ["苹果园", "国贸"], "2号线": ["西直门", "积水潭"],
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.build()

    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.line_stations = {}
        for s in self.data.get("stations", []):
            n = self.station_to_number.setdefault(s["name"], len(self.station_to_number))
            self.number_to_station[n] = s["name"]
            self.graph.add_node(n)
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for nm in (c["source"], c["dest"]):
                if nm not in self.station_to_number:
                    n = len(self.station_to_number)
                    self.station_to_number[nm] = n
                    self.number_to_station[n] = nm
                    self.graph.add_node(n)
                    self.graph.set_station_lines(nm, [c["line"]])
            a, b = self.station_to_number[c["source"]], self.station_to_number[c["dest"]]
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))
            seq = self.line_stations.setdefault(c["line"], [])
            for nm in (c["source"], c["dest"]):
                if nm not in seq:
                    seq.append(nm)

    def has_station(self, name):
        return name in self.station_to_number

    def station_names(self):
        return list(self.station_to_number.keys())

    def is_transfer(self, name):
        return len(self.graph.station_lines.get(name, [])) > 1

    def lines(self):
        return sorted(self.line_stations)

    def check_data(self):
        """Return (ok, list of messages)."""
        msgs, ok = [], True
        stations = self.data.get("stations", [])
        conns = self.data.get("connections", [])
        msgs.append(f"Stations: {len(stations)}, connections: {len(conns)}, lines: {len(self.line_stations)}")
        if not stations or not conns:
            return False, msgs + ["Data is empty"]
        names = set(self.station_to_number)
        listed = {s["name"] for s in stations}
        if len(listed) != len(stations):
            ok = False
            msgs.append("Duplicate station names found")
        for c in conns:
            for k in ("source", "dest"):
                if c[k] not in listed:
                    ok = False
                    msgs.append(f"Connection references unknown station: {c[k]}")
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
                ok = False
                msgs.append(f"Suspicious distance {d} between {c['source']} and {c['dest']}")
            if c["source"] == c["dest"]:
                ok = False
                msgs.append(f"Self-loop at {c['source']}")
        connected = set()
        for c in conns:
            connected.update((c["source"], c["dest"]))
        for nm in names - connected:
            ok = False
            msgs.append(f"Station without connections: {nm}")
        for line in REPRESENTATIVE and ["1号线", "2号线"]:
            pass
        msgs.append("Data check passed" if ok else "Data check found problems")
        return ok, msgs

    # ---- updates ----
    def add_line(self, line, station_names, distances):
        """Add a line; distances has len(station_names)-1 entries (meters)."""
        if len(distances) != len(station_names) - 1:
            raise ValueError("distances length must be stations-1")
        known = {s["name"]: s for s in self.data["stations"]}
        for nm in station_names:
            if nm in known:
                if line not in known[nm]["lines"]:
                    known[nm]["lines"].append(line)
            else:
                self.data["stations"].append({"name": nm, "lines": [line]})
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line})
        self.build()

    def update_station(self, name, lines):
        for s in self.data["stations"]:
            if s["name"] == name:
                s["lines"] = list(lines)
                self.build()
                return
        raise KeyError(name)

    def update_distance(self, a, b, distance):
        if distance <= 0:
            raise ValueError("distance must be positive")
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                found = True
        if not found:
            raise KeyError((a, b))
        self.build()

    def save(self, path=None):
        import json
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

"""Input validation."""
import re

SEPARATORS = r"[;；,，]"


class ValidationError(Exception):
    pass


class InputValidator:
    def __init__(self, data_manager):
        self.dm = data_manager

    def parse(self, text):
        """Return (start, end) or raise ValidationError."""
        parts = [p.strip() for p in re.split(SEPARATORS, text or "")]
        if len(parts) != 2 or not parts[0] or not parts[1]:
            raise ValidationError("Input format incorrect, please use 'Starting Station; Ending Station' format")
        start, end = parts
        if not self.dm.has_station(start):
            raise ValidationError(f"Starting station '{start}' does not exist, please check the station name")
        if not self.dm.has_station(end):
            raise ValidationError(f"Ending station '{end}' does not exist, please check the station name")
        if start == end:
            raise ValidationError("Starting station cannot be the same as the ending station")
        return start, end

"""Report generation (text and JSON)."""
import json


def segments(edges, names):
    """Group edges by line: list of dicts(line, from, to, stops)."""
    segs = []
    for e in edges:
        if segs and segs[-1]["line"] == e.line:
            segs[-1]["to"] = names[e.dest]
            segs[-1]["stops"] += 1
        else:
            segs.append({"line": e.line, "from": names[e.source], "to": names[e.dest], "stops": 1})
    return segs


def format_distance(d):
    return f"{d} m ({d / 1000:.2f} km)" if d < 1000 else f"{d / 1000:.2f} km ({d} m)"


def build_summary(start, end, mode, station_names, edges, names, distance, fare, minutes):
    segs = segments(edges, names)
    transfers = [{"station": s["to"], "from_line": s["line"], "to_line": n["line"]}
                 for s, n in zip(segs, segs[1:])]
    lines = []
    for s in segs:
        if s["line"] not in lines:
            lines.append(s["line"])
    return {
        "query": {"start": start, "end": end, "mode": mode},
        "path": station_names,
        "segments": segs,
        "distance_m": distance,
        "distance_km": round(distance / 1000, 2),
        "fare": fare,
        "estimated_time_min": minutes,
        "transfer_count": len(transfers),
        "transfers": transfers,
        "lines": lines,
    }


def render(summary, fmt="text"):
    if fmt == "json":
        return json.dumps(summary, ensure_ascii=False, indent=2)
    q = summary["query"]
    out = [f"Route: {q['start']} -> {q['end']} (mode: {q['mode']})",
           f"Stations ({len(summary['path'])}): " + " -> ".join(summary["path"]),
           f"Total distance: {format_distance(summary['distance_m'])}",
           f"Transfers: {summary['transfer_count']}",
           f"Estimated time: {summary['estimated_time_min']} min",
           "Lines: " + ", ".join(summary["lines"]),
           "Segments:"]
    for s in summary["segments"]:
        out.append(f"  [{s['line']}] {s['from']} -> {s['to']} ({s['stops']} stops)")
    for t in summary["transfers"]:
        out.append(f"  Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
    out.append(f"Fare: {summary['fare']} yuan")
    return "\n".join(out)

"""Command-line interface."""
import argparse
import sys

from data_manager import DataManager
from validator import InputValidator, ValidationError
from pathfinding import ShortestPath, PathfindingConfig, MODES, SHORTEST_DISTANCE
from fare_calculator import Sum_money, calculate_estimated_time
from report import build_summary, render


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(MODES), default=SHORTEST_DISTANCE)
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    p.add_argument("--query", default=None, help="single query 'Start; End' (non-interactive)")
    return p.parse_args(argv)


def run_query(dm, start, end, args):
    names = dm.number_to_station
    pref = [args.line_preference] if args.line_preference else []
    cfg = PathfindingConfig(args.mode, args.max_transfers, pref)
    s, t = dm.station_to_number[start], dm.station_to_number[end]
    try:
        sp = ShortestPath(dm.graph, s, cfg)
        note = ""
        if not sp.has_path_to(t) and args.max_transfers is not None:
            sp = ShortestPath(dm.graph, s, PathfindingConfig(args.mode, None, pref))
            note = "Note: no route within the transfer limit; showing best route without limit.\n"
        if not sp.has_path_to(t):
            raise ValueError("no route found")
        nodes, edges = sp.path_with_edges(t)
        dist = sp.dist_to[t]
        transfers = sp.get_transfer_count(t)
    except Exception as e:
        return f"Path calculation failed: {e}"
    try:
        fare = Sum_money(dist)
        minutes = calculate_estimated_time(dist, transfers)
    except Exception as e:
        return f"Fare calculation failed: {e}"
    summary = build_summary(start, end, args.mode, [names[n] for n in nodes], edges,
                            names, dist, fare, minutes)
    text = render(summary, args.format)
    return note + text if args.format == "text" else text


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager()
    except Exception as e:
        print(f"Failed to load data: {e}")
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1
    validator = InputValidator(dm)
    first = [True]

    def emit(text):
        print(text)
        if args.output_file:
            try:
                with open(args.output_file, "w" if first[0] else "a", encoding="utf-8") as f:
                    f.write(text + "\n")
                first[0] = False
            except OSError as e:
                print(f"Cannot write output file: {e}")

    if args.query is not None:
        try:
            st, en = validator.parse(args.query)
        except ValidationError as e:
            print(e)
            return 1
        emit(run_query(dm, st, en, args))
        return 0

    print("Beijing Subway fare query. Enter 'Start; End', or 'q' to quit.")
    while True:
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        try:
            st, en = validator.parse(line)
        except ValidationError as e:
            print(e)
            continue
        emit(run_query(dm, st, en, args))
    return 0


if __name__ == "__main__":
    sys.exit(main())

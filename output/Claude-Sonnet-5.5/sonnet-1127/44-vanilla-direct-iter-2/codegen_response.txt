"""Graph data structures."""


class Edge:
    """Undirected weighted edge between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""Fare and time calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (boundaries go to higher tier)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5

"""Dijkstra-based path finding with priority queue."""
import heapq
import itertools

TRANSFER_PENALTY = 1500  # meters-equivalent per transfer (comprehensive)


class MinPQ:
    """Heap-based minimum priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    """Single-source best path on state space (station, line, transfers)."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}   # node -> state
        self._info = {}   # state -> (cost, dist, transfers, prev_state, edge)
        self._run()

    def _edge_cost(self, e, transfer):
        c = float(e.weight)
        if e.line in self.config.preferred_lines:
            c *= 0.9
        mode = self.config.mode
        if transfer:
            if mode == "least_transfer":
                c += 1e7
            elif mode == "comprehensive":
                c += TRANSFER_PENALTY
            else:
                c += 0.001
        return c

    def _run(self):
        mt = self.config.max_transfers
        start = (self.source, None, 0)
        self._info[start] = (0.0, 0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, _ = st
            _, d, t, _, _ = self._info[st]
            if node not in self._best:
                self._best[node] = st
                self.dist_to[node] = d
            for e in self.graph.neighbors(node):
                tr = line is not None and e.line != line
                nt = t + (1 if tr else 0)
                if mt is not None and nt > mt:
                    continue
                ns = (e.dest, e.line, nt if mt is not None else 0)
                if ns in done:
                    continue
                nc = cost + self._edge_cost(e, tr)
                old = self._info.get(ns)
                if old is None or nc < old[0]:
                    self._info[ns] = (nc, d + e.weight, nt, st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        st = self._best.get(target)
        if st is None:
            return [], []
        nodes, edges = [], []
        while st is not None:
            _, _, _, prev, e = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Network management, integrity checks and data updates."""
from data_loader import load_subway_data, DATA_FILE_PATH
from models import Graph, Edge

REPRESENTATIVE_STATIONS = ["西单", "国贸", "西直门", "东直门", "公主坟", "北京南站", "朝阳门", "北京西站"]


class SubwayNetwork:
    def __init__(self, data=None):
        self.data = data if data is not None else load_subway_data(DATA_FILE_PATH)
        self.rebuild()

    def rebuild(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in self.data.get("stations", []):
            if s["name"] not in self.station_to_number:
                n = len(self.station_to_number)
                self.station_to_number[s["name"]] = n
                self.number_to_station[n] = s["name"]
                self.graph.add_node(n)
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a = self.station_to_number.get(c["source"])
            b = self.station_to_number.get(c["dest"])
            if a is None or b is None:
                continue
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        if len(distances) != len(stations) - 1:
            raise ValueError("need len(stations)-1 distances")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for i, d in enumerate(distances):
            self.data["connections"].append(
                {"source": stations[i], "dest": stations[i + 1], "distance": d, "line": line_name})
        self.rebuild()

    def update_station(self, name, lines=None, new_name=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.rebuild()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        if distance <= 0:
            raise ValueError("distance must be positive")
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self.rebuild()
        return found

    # ---- checks ----
    def check_integrity(self):
        issues = []
        names = {s["name"] for s in self.data["stations"]}
        lines = set()
        for s in self.data["stations"]:
            lines.update(s.get("lines", []))
        for m in REPRESENTATIVE_STATIONS:
            if m not in names:
                issues.append("Representative station missing: %s" % m)
        connected = set()
        for c in self.data["connections"]:
            for k in ("source", "dest"):
                if c[k] not in names:
                    issues.append("Connection references unknown station: %s" % c[k])
                connected.add(c[k])
            if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0:
                issues.append("Invalid distance %s-%s" % (c["source"], c["dest"]))
            elif c["distance"] > 30000:
                issues.append("Suspicious distance %s-%s: %s" % (c["source"], c["dest"], c["distance"]))
            if c["source"] == c["dest"]:
                issues.append("Self loop at %s" % c["source"])
        for n in names - connected:
            issues.append("Station without connections: %s" % n)
        return {"ok": not issues, "stations": len(names), "lines": len(lines),
                "connections": len(self.data["connections"]), "issues": issues}

"""Route query and report generation."""
import json
from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import ShortestPath


def format_distance(m):
    if m < 1000:
        return "%d meters" % m
    return "%.2f km (%d meters)" % (m / 1000.0, m)


def query_route(network, start, end, config):
    s = network.station_to_number[start]
    t = network.station_to_number[end]
    sp = ShortestPath(network.graph, s, config)
    nodes, edges = sp.path_with_edges(t)
    if not nodes:
        raise RuntimeError("No route found from '%s' to '%s' (check max transfers)" % (start, end))
    total = sp.dist_to[t]
    transfers = sp.get_transfer_count(t)
    try:
        fare = Sum_money(total)
    except Exception as e:
        raise ArithmeticError("Fare calculation failed: %s" % e)
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = network.number_to_station[e.dest]
            segments[-1]["stops"] += 1
        else:
            if segments:
                transfer_stations.append(network.number_to_station[e.source])
            segments.append({"line": e.line, "from": network.number_to_station[e.source],
                             "to": network.number_to_station[e.dest], "stops": 1})
    lines = []
    for sg in segments:
        if sg["line"] not in lines:
            lines.append(sg["line"])
    return {
        "query": {"start": start, "end": end, "mode": config.mode},
        "path": [network.number_to_station[n] for n in nodes],
        "distance_m": total,
        "distance_km": round(total / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(calculate_estimated_time(total, transfers), 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": segments,
    }


def format_text(r):
    out = ["Route: %s -> %s (mode: %s)" % (r["query"]["start"], r["query"]["end"], r["query"]["mode"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: %s" % format_distance(r["distance_m"]),
           "Segments:"]
    for sg in r["segments"]:
        out.append("  %s: %s -> %s (%d stops)" % (sg["line"], sg["from"], sg["to"], sg["stops"]))
    if r["transfer_stations"]:
        out.append("Transfers at: %s" % ", ".join(r["transfer_stations"]))
    out.append("Transfer count: %d" % r["transfer_count"])
    out.append("Lines involved: %s" % ", ".join(r["lines"]))
    out.append("Estimated time: %.0f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def format_json(r):
    return json.dumps(r, ensure_ascii=False, indent=2)

"""CLI entry point for the Beijing Subway fare calculator."""
import argparse
import sys

from data_manager import SubwayNetwork
from pathfinding import PathfindingConfig
from report import query_route, format_text, format_json

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_input(line, network):
    """Return (start, end) or raise ValueError with a message."""
    norm = line.replace("；", ";")
    parts = [p.strip() for p in norm.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERR)
    a, b = parts
    if a not in network.station_to_number:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % a)
    if b not in network.station_to_number:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % b)
    if a == b:
        raise ValueError("Starting station cannot be the same as the ending station")
    return a, b


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p


def main(argv=None):
    args = build_parser().parse_args(argv)
    try:
        network = SubwayNetwork()
    except Exception as e:
        print("Failed to load subway data: %s" % e)
        return 1
    if args.check_data:
        res = network.check_integrity()
        print("Stations: %d, Lines: %d, Connections: %d" % (res["stations"], res["lines"], res["connections"]))
        for i in res["issues"]:
            print("  - " + i)
        print("Data check %s" % ("passed" if res["ok"] else "found issues"))
        return 0 if res["ok"] else 1

    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    if args.line_preference:
        config.preferred_lines = [x.strip() for x in args.line_preference.split(",") if x.strip()]

    print("Enter 'Starting Station; Ending Station' (type 'exit' to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("exit", "quit", "q"):
            break
        try:
            a, b = parse_input(line, network)
        except ValueError as e:
            print(e)
            continue
        try:
            r = query_route(network, a, b, config)
        except ArithmeticError as e:
            print(e)
            continue
        except Exception as e:
            print("Path calculation error: %s" % e)
            continue
        text = format_json(r) if args.format == "json" else format_text(r)
        print(text)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(text + "\n")
            except OSError as e:
                print("Failed to write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Core data models: Edge, Graph, Station, Line, PathQuery, Route."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r,%r,%r,%r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, [])
        self.adjacency.setdefault(edge.dest, [])
        self.adjacency[edge.source].append(edge)
        self.adjacency[edge.dest].append(edge.reversed())

    def neighbors(self, number):
        return self.adjacency.get(number, [])

    def remove_edges_between(self, a, b, line=None):
        for x, y in ((a, b), (b, a)):
            self.adjacency[x] = [e for e in self.adjacency.get(x, [])
                                 if not (e.dest == y and (line is None or e.line == line))]

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1


class Station:
    def __init__(self, id, name, lines=None):
        self.id = id
        self.name = name
        self.lines = list(lines or [])

    @property
    def is_transfer(self):
        return len(self.lines) > 1


class Line:
    def __init__(self, name, station_sequence=None):
        self.name = name
        self.station_sequence = list(station_sequence or [])


class PathQuery:
    def __init__(self, start, end, mode="shortest_distance", max_transfers=None, line_preference=None):
        self.start = start
        self.end = end
        self.mode = mode
        self.max_transfers = max_transfers
        self.line_preference = line_preference


class Route:

        self.stations = stations
        self.edges = edges
        self.total_distance = total_distance
        self.transfer_count = transfer_count
        self.lines = lines
        self.transfer_stations = transfer_stations or []

"""Dijkstra shortest path with a heap-based priority queue."""
import heapq
import itertools


class MinPQ:
    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 100000, "comprehensive": 1500}


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._parent = {}  # state -> (prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        limit = cfg.max_transfers
        pref = set(cfg.preferred_lines)
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        self._parent[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in self._best_state:
                self._best_state[node] = st
                self.dist_to[node] = real[st]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0
                if line is not None and e.line != line:
                    if limit is not None and t + 1 > limit:
                        continue
                    nt = t + 1 if limit is not None else 0
                    extra = penalty
                w = e.weight * (0.9 if e.line in pref else 1.0)
                ns = (e.dest, e.line, nt)
                nc = c + w + extra
                if ns not in cost or nc < cost[ns] - 1e-12:
                    cost[ns] = nc
                    real[ns] = real[st] + e.weight
                    self._parent[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        edges = []
        while self._parent[st] is not None:
            prev, e = self._parent[st]
            edges.append(e)
            st = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Beijing subway fare calculation."""

FARE_TIERS = [(0, 6000, 3), (6000, 12000, 4), (12000, 22000, 5), (22000, 32000, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters."""
    if distance is None or distance != distance:
        raise ValueError("invalid distance")
    if distance < 0:
        raise ValueError("distance must be non-negative")
    d = int(round(distance))
    for lo, hi, fare in FARE_TIERS:
        if d < hi:
            return fare
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: ~35 km/h average plus 5 min per transfer and 2 min boarding."""
    km = max(distance, 0) / 1000.0
    return km / 35.0 * 60.0 + 5.0 * transfer_count + 2.0

"""Data management: load, graph building, integrity checks and updates."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Graph, Edge

MAIN_LINES = ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
              "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
              "Fangshan Line", "Airport Line"]


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = {"stations": [], "connections": []}
        self.graph = Graph()
        self.load()

    def load(self):
        self.data = load_subway_data(self.data_file)
        self.rebuild()

    def rebuild(self):
        g = Graph()
        n = 0
        for s in self.data.get("stations", []):
            if s["name"] not in g.station_to_number:
                g.add_station(s["name"], n)
                n += 1
            g.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for nm in (c["source"], c["dest"]):
                if nm not in g.station_to_number:
                    g.add_station(nm, n)
                    n += 1
                    g.set_station_lines(nm, [c["line"]])
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
        self.graph = g

    @property
    def station_to_number(self):
        return self.graph.station_to_number

    def has_station(self, name):
        return name in self.graph.station_to_number

    def lines(self):
        s = set()
        for ls in self.graph.station_lines.values():
            s.update(ls)
        return sorted(s)

    def suggest(self, name, limit=5):
        return [n for n in self.graph.station_to_number if name and (name in n or n in name)][:limit]

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: meters between consecutive stations."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 entries")
        known = {s["name"]: s for s in self.data["stations"]}
        for nm in stations:
            if nm in known:
                if line_name not in known[nm]["lines"]:
                    known[nm]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": nm, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.rebuild()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        if distance <= 0:
            raise ValueError("distance must be positive")
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self.rebuild()
        return found

    # ---- integrity ----
    def check_data(self):
        """Return (ok, list of messages)."""
        issues = []
        names = {s["name"] for s in self.data["stations"]}
        lines = set(self.lines())
        for l in MAIN_LINES:
            if l not in lines:
                issues.append("Missing main line: %s" % l)
        connected = set()
        for c in self.data["connections"]:
            for k in ("source", "dest"):
                if c[k] not in names:
                    issues.append("Connection references unknown station: %s" % c[k])
                connected.add(c[k])
            d = c["distance"]
            if not isinstance(d, (int, float)) or d <= 0:
                issues.append("Invalid distance %s-%s: %s" % (c["source"], c["dest"], d))
            elif d > 20000:
                issues.append("Suspicious distance %s-%s: %s" % (c["source"], c["dest"], d))
            if c["source"] == c["dest"]:
                issues.append("Self loop at %s" % c["source"])
        for n in sorted(names - connected):
            issues.append("Station without connections: %s" % n)
        return (not issues), issues

"""Route report generation (text / json) and summaries."""
import json
from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(m):
    return "%d meters (%.2f km)" % (round(m), m / 1000.0)


def summarize(route):
    return {
        "distance_m": route.total_distance,
        "distance_km": round(route.total_distance / 1000.0, 2),
        "fare": Sum_money(route.total_distance),
        "transfer_count": route.transfer_count,
        "estimated_time": round(calculate_estimated_time(route.total_distance, route.transfer_count), 1),
    }


def segments(route):
    segs = []
    for e, name in zip(route.edges, route.stations[1:]):
        if segs and segs[-1]["line"] == e.line:
            segs[-1]["to"] = name
            segs[-1]["stations"] += 1
        else:
            segs.append({"line": e.line, "from": route.stations[len(route.stations) - 1 - 0] if False else None,
                         "to": name, "stations": 1})
    # fill 'from'
    idx = 0
    for s in segs:
        s["from"] = route.stations[idx]
        idx += s["stations"]
    return segs


class RouteReport:
    def __init__(self, query, route, fmt="text", output_file=None):
        self.query = query
        self.route = route
        self.format = fmt
        self.output_file = output_file
        self.summary = summarize(route)

    def to_text(self):
        r, s = self.route, self.summary
        out = ["Route: %s -> %s" % (self.query.start, self.query.end),
               "Stations (%d): %s" % (len(r.stations), " -> ".join(r.stations)),
               "Distance: " + format_distance(r.total_distance)]
        for seg in segments(r):
            out.append("  [%s] %s -> %s (%d stops)" % (seg["line"], seg["from"], seg["to"], seg["stations"]))
        if r.transfer_stations:
            out.append("Transfers (%d): %s" % (r.transfer_count, "; ".join(
                "%s (%s -> %s)" % t for t in r.transfer_stations)))
        else:
            out.append("Transfers: 0")
        out.append("Lines: " + ", ".join(r.lines))
        out.append("Estimated time: %.0f minutes" % s["estimated_time"])
        out.append("Fare: %d yuan" % s["fare"])
        return "\n".join(out)

    def to_json(self):
        r, s = self.route, self.summary
        return json.dumps({
            "query": {"start": self.query.start, "end": self.query.end, "mode": self.query.mode,
                      "max_transfers": self.query.max_transfers,
                      "line_preference": self.query.line_preference},
            "path": r.stations,
            "distance_m": s["distance_m"],
            "distance_km": s["distance_km"],
            "fare": s["fare"],
            "estimated_time": s["estimated_time"],
            "transfer_count": s["transfer_count"],
            "transfer_stations": [t[0] for t in r.transfer_stations],
            "lines": r.lines,
            "segments": segments(r),
        }, ensure_ascii=False, indent=2)

    def render(self):
        return self.to_json() if self.format == "json" else self.to_text()

"""Command-line interface."""
import argparse
import re
import sys

from data_manager import DataManager
from models import PathQuery, Route
from pathfinding import ShortestPath, PathfindingConfig
from report import RouteReport

MODES = ("shortest_distance", "least_transfer", "comprehensive")


def find_route(dm, query):
    g = dm.graph
    s = g.station_to_number[query.start]
    t = g.station_to_number[query.end]
    pref = [query.line_preference] if query.line_preference else []
    cfg = PathfindingConfig(query.mode, query.max_transfers, pref)
    sp = ShortestPath(g, s, cfg)
    nodes, edges = sp.path_with_edges(t)
    if not nodes:
        return None
    names = [g.number_to_station[n] for n in nodes]
    lines, tstations = [], []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    for a, b in zip(edges, edges[1:]):
        if a.line != b.line:
            tstations.append((g.number_to_station[a.dest], a.line, b.line))
    return Route(names, edges, sp.dist_to[t], len(tstations), lines, tstations)


def parse_input(text):
    parts = [p.strip() for p in re.split(r"[;；,，]", text)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    return parts


def validate(dm, a, b):
    for label, n in (("Starting", a), ("Ending", b)):
        if not dm.has_station(n):
            msg = "%s station '%s' does not exist, please check the station name" % (label, n)
            sug = dm.suggest(n)
            if sug:
                msg += " (did you mean: %s?)" % ", ".join(sug)
            raise ValueError(msg)
    if a == b:
        raise ValueError("Starting station cannot be the same as the ending station")


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=MODES, default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p


def run_query(dm, text, args):
    a, b = parse_input(text)
    validate(dm, a, b)
    q = PathQuery(a, b, args.mode, args.max_transfers, args.line_preference)
    try:
        route = find_route(dm, q)
    except Exception as ex:
        raise ValueError("Path calculation failed: %s" % ex)
    if route is None:
        raise ValueError("No route found between '%s' and '%s'%s" % (
            a, b, " within the transfer limit" if args.max_transfers is not None else ""))
    try:
        rep = RouteReport(q, route, args.format, args.output_file)
    except Exception as ex:
        raise ValueError("Fare calculation failed: %s" % ex)
    return rep


def main(argv=None):
    args = build_parser().parse_args(argv)
    if args.max_transfers is not None and args.max_transfers < 0:
        print("--max_transfers must be non-negative")
        return 2
    try:
        dm = DataManager()
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1
    if args.check_data:
        ok, issues = dm.check_data()
        print("Data check passed." if ok else "Data check found %d issue(s):" % len(issues))
        for i in issues:
            print(" - " + i)
        return 0 if ok else 1
    first = True
    print("Enter 'Starting Station; Ending Station' (q to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        try:
            rep = run_query(dm, line, args)
        except ValueError as ex:
            print(ex)
            continue
        out = rep.render()
        print(out)
        if args.output_file:
            try:
                with open(args.output_file, "w" if first else "a", encoding="utf-8") as f:
                    f.write(out + "\n")
                first = False
            except OSError as ex:
                print("Could not write output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Grid based path planning."""

CELL = 100


class PathPlanner:
    """Moves along axes: horizontal first, then vertical."""

    @staticmethod
    def next_position(x, y, tx, ty, speed):
        """Advance up to `speed` units toward the target."""
        budget = speed
        if x != tx:
            step = min(budget, abs(tx - x))
            x += step if tx > x else -step
            budget -= step
        if budget > 0 and y != ty:
            step = min(budget, abs(ty - y))
            y += step if ty > y else -step
        return x, y

    @staticmethod
    def corner_points(points):
        """Return the polyline (with right-angle corners) through points."""
        if not points:
            return []
        result = [points[0]]
        for px, py in points[1:]:
            cx, cy = result[-1]
            if cx != px and cy != py:
                result.append((px, cy))
            if result[-1] != (px, py):
                result.append((px, py))
        return result

    @staticmethod
    def cells_on_path(points):
        """Return the set of grid cells (col,row) touched by the path."""
        cells = []
        poly = PathPlanner.corner_points(points)
        for (x1, y1), (x2, y2) in zip(poly, poly[1:]):
            n = max(abs(x2 - x1), abs(y2 - y1))
            for i in range(0, n + 1, 10 if n >= 10 else 1):
                px = x1 + (x2 - x1) * i // n if n else x1
                py = y1 + (y2 - y1) * i // n if n else y1
                c = (min(px // CELL, 7), min(py // CELL, 7))
                if c not in cells:
                    cells.append(c)
        return cells

    @staticmethod
    def path_length(points):
        """Manhattan length of the polyline."""
        poly = PathPlanner.corner_points(points)
        return sum(abs(b[0] - a[0]) + abs(b[1] - a[1])
                   for a, b in zip(poly, poly[1:]))

import globalvar

_counter = [0]


def to_center(v):
    """Convert a coordinate to its grid cell center."""
    v = max(0, min(int(v), 799))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        _counter[0] += 1
        self.id = _counter[0]
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.end_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self):
        """Mark delivered."""
        self.status = "completed"
        self.end_time = globalvar.get_value("time", 0) or 0

    @property
    def duration(self):
        """Ticks taken, or None."""
        if self.end_time is None:
            return None
        return self.end_time - self.start_time

    def to_dict(self):
        """Serialise."""
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "end_time": self.end_time, "distance": self.distance}

"""Rider model."""

import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0) or 0,
                         "x": self.x, "y": self.y})

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        order = self.orders[0] if self.orders else None
        if order is None:
            self.state = 0
            return
        if self.state == 1:
            tx, ty = order.res_pos
            self.x, self.y = PathPlanner.next_position(
                self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            tx, ty = order.cus_pos
            self.x, self.y = PathPlanner.next_position(
                self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                order.complete()
                self.orders.pop(0)
                self.state = 0
                self._log("delivered_to_customer", order.id)

    def to_dict(self):
        """Serialise."""
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order management."""

import math

import globalvar
from order_model import Order


def find_best_rider_for_order(order):
    """Return the closest rider (idle riders preferred), or None."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    idle = [r for r in riders if r.state == 0 and not r.orders]
    pool = idle or riders

    def key(r):
        d = math.hypot(r.x - order.res_pos[0], r.y - order.res_pos[1])
        return (d, len(r.orders), r.rider_id)

    return min(pool, key=key)


def create_order(coords):
    """Create an order from [rx, ry, cx, cy]."""
    return Order((coords[0], coords[1]), (coords[2], coords[3]))


def assign_order(order):
    """Assign order to the best rider; return the rider or None."""
    rider = find_best_rider_for_order(order)
    if rider is None:
        return None
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    orders = globalvar.get_value("orders")
    if orders is not None and order not in orders:
        orders.append(order)
    return rider

"""System manager."""

import json

import globalvar
from rider import Rider

RIDER_COST = 300


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])


def recruit_rider():
    """Create a rider, paying the cost."""
    n = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", n)
    globalvar.set_value("funds", globalvar.get_value("funds") - RIDER_COST)
    rider = Rider(n)
    globalvar.get_value("riders").append(rider)
    return rider


def maybe_recruit(pending):
    """Recruit when funds >= 300 and pending > idle riders."""
    riders = globalvar.get_value("riders")
    idle = sum(1 for r in riders if r.state == 0 and not r.orders)
    if globalvar.get_value("funds") >= RIDER_COST and pending > idle:
        return recruit_rider()
    return None


def tick():
    """Advance simulated time one unit."""
    globalvar.set_value("time", globalvar.get_value("time") + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def save(path="state.json"):
    """Persist state to JSON."""
    data = {"funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "rider_id_counter": globalvar.get_value("rider_id_counter"),
            "riders": [r.to_dict() for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
    with open(path, "w") as f:
        json.dump(data, f, indent=2)

"""Display manager."""

import globalvar
from path_planner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text rendering."""

    @staticmethod
    def status():
        """System status text."""
        riders = globalvar.get_value("riders")
        lines = ["Time: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    @staticmethod
    def orders():
        """Order list text."""
        lines = []
        for o in globalvar.get_value("orders"):
            lines.append("Order %d start=%s res=%s cus=%s dist=%d took=%s "
                         "rider=%s status=%s" % (
                             o.id, o.start_time, o.res_pos, o.cus_pos,
                             o.distance, o.duration, o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    @staticmethod
    def riders_log():
        """Rider history text."""
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("Rider %d t=%d %s at (%d,%d) order=%s" % (
                    r.rider_id, e["time"], e["event"], e["x"], e["y"],
                    e["order_id"]))
        return "\n".join(lines) or "No rider events."

    @staticmethod
    def map(route=None, order=None):
        """ASCII 8x8 map; '#' street, '.' empty, '*' route."""
        grid = [["." for _ in range(8)] for _ in range(8)]
        for c, r in PathPlanner.cells_on_path(route or []):
            grid[r][c] = "#"
        for o in globalvar.get_value("orders"):
            if o.status != "completed":
                grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
                grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        for r in globalvar.get_value("riders"):
            grid[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
        return "\n".join(" ".join(row) for row in grid)

"""Input parsing and validation."""


class InputError(ValueError):
    """Invalid user input."""


def parse_coords(text):
    """Parse four integers in 0-800."""
    text = text.replace(",", " ").replace("[", " ").replace("]", " ")
    parts = text.split()
    if len(parts) != 4:
        raise InputError("Expected 4 integers: rx ry cx cy")
    try:
        vals = [int(p) for p in parts]
    except ValueError:
        raise InputError("Coordinates must be integers")
    for v in vals:
        if not 0 <= v <= 800:
            raise InputError("Coordinates must be within 0-800")
    return vals


class InputManager:
    """Command parsing with history and shortcuts."""

    SHORTCUTS = {"s": "start", "st": "status", "o": "orders",
                 "r": "riders", "q": "quit", "t": "tick"}

    def __init__(self):
        self.history = []

    def parse(self, line):
        """Return (command, argument string)."""
        line = line.strip()
        if not line:
            return "", ""
        self.history.append(line)
        parts = line.split(None, 1)
        cmd = parts[0].lower()
        cmd = self.SHORTCUTS.get(cmd, cmd)
        return cmd, parts[1] if len(parts) > 1 else ""

"""Command-line entry point."""

import system
import globalvar
import order as order_mod
from display import DisplayManager
from input_manager import InputManager, InputError, parse_coords
from path_planner import PathPlanner


def handle_start(arg):
    """Create and dispatch an order."""
    coords = parse_coords(arg)
    o = order_mod.create_order(coords)
    new_rider = system.maybe_recruit(1)
    rider = order_mod.assign_order(o)
    if rider is None:
        return "No riders available."
    route = [(rider.x, rider.y), o.res_pos, o.cus_pos]
    length = PathPlanner.path_length(route)
    eta = -(-length // rider.speed)
    return "\n".join([
        "Order %d: restaurant grid %s, customer grid %s" % (
            o.id, (o.res_pos[0] // 100, o.res_pos[1] // 100),
            (o.cus_pos[0] // 100, o.cus_pos[1] // 100)),
        "New rider added: %s" % ("yes" if new_rider else "no"),
        "Assigned rider %d, speed %d, est. time %d ticks" % (
            rider.rider_id, rider.speed, eta),
        DisplayManager.map(route)])


def main():
    """Run the simulator."""
    system.init_system()
    im = InputManager()
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except EOFError:
            break
        try:
            cmd, arg = im.parse(line)
            ticks = 1
            if cmd == "start":
                print(handle_start(arg))
            elif cmd == "status":
                print(DisplayManager.status())
                print(DisplayManager.map())
            elif cmd == "orders":
                print(DisplayManager.orders())
            elif cmd == "riders":
                print(DisplayManager.riders_log())
            elif cmd == "tick":
                ticks = int(arg) if arg.strip() else 1
            elif cmd in ("quit", "exit"):
                globalvar.set_value("running", False)
                ticks = 0
            elif cmd == "":
                ticks = 0
            else:
                print("Unknown command. Use start/status/orders/riders/quit")
                ticks = 0
            for _ in range(ticks):
                system.tick()
        except (InputError, ValueError) as e:
            print("Error: %s" % e)
        except Exception as e:  # recover gracefully
            print("Unexpected error: %s" % e)


if __name__ == "__main__":
    main()

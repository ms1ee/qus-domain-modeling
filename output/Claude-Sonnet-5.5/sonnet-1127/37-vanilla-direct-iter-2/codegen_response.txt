DEFAULT_USERS = [

    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Li Bin", "Marketing Department"),
    ("Chen Qixian", "Operations Department"),

    ("Cheng Yi", CHENG_RU),
    ("Cheng Er", CHENG_RU),
    ("Cheng San", CHENG_RU),
    ("Qian Yi", "HR Department"),
    ("Qian Er", "HR Department"),
    ("Feng Yi", "Finance Department"),

class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_USERS], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        except UnicodeDecodeError:
            return [], [f"File Error: cannot decode '{path}'"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error - '{line}' (expected 'Name,Department')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, ["Cheng Ru Group"], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, a, d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        prize = Prize(name, quantity, list(avoid_groups or []), associated_drawer or None)
        self._prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random

from lottery_system.services.list_manager import CHENG_RU

LI_BIN = "Li Bin"
GUARD_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"


def _is_cr(dept):
    return str(dept).strip().casefold() == CHENG_RU.casefold()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []  # list of User
        self.last_two_draw_results = []

    def _winner_names(self):
        return {w.name for w in self.winners}

    def _get_eligible_pool(self, prize):
        won = self._winner_names()
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == LI_BIN:
            return pool
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_cr(u.department)]
        avoid = {g.casefold() for g in prize.avoid_groups}
        return [u for u in pool if u.department.casefold() not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_prize(self, prize, drawn):
        if prize.associated_drawer in GUARD_DRAWERS:
            self.last_two_draw_results.append(any(_is_cr(u.department) for u in drawn))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, win_time=None):
        self.user = user
        self.prize_name = prize_name
        self.win_time = win_time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        grouped = self.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            return False, "No winners to export"
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                for ws in grouped.values():
                    for w in sorted(ws, key=lambda x: x.win_time):
                        f.write(f"{w.user.name},{w.user.department},{w.prize_name},"
                                f"{w.win_time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import queue
import threading
import time

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print("  !", e)
        if errs and c == "2":
            a = input("Errors found. (r)emove invalid lines and continue / (t)erminate: ").strip().lower()
            if a.startswith("t"):
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"  ! Duplicate name removed: {d.name}")
        if valid:
            return valid
        print("No valid users.")


def config_prizes(pm):
    if input("Load default prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        try:
            name = input("Name: ").strip()
            qty = int(input("Quantity: "))
            drawer = input("Associated drawer (blank for none): ").strip() or None
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    ps = pm.get_prizes()
    for i, p in enumerate(ps, 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("New order (space-separated numbers, blank to keep): ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order; keeping default.")
        except ValueError:
            print("Invalid order; keeping default.")


def start_reader(q):
    def run():
        try:
            for line in iter(input, None):
                q.put(line.strip().lower())
        except EOFError:
            pass
    threading.Thread(target=run, daemon=True).start()


def run_lottery(users, prizes):
    lot = LotteryManager(users)
    res = ResultManager()
    q = queue.Queue()
    start_reader(q)
    print("Commands during draw: pause / continue / terminate")
    for p in prizes:
        print(f"\n=== {p.name} | quantity {p.quantity} | avoid {p.avoid_groups or 'none'} | drawer {p.associated_drawer or 'none'}")
        drawn = []
        for i in range(1, p.quantity + 1):
            stop = False
            while True:
                try:
                    cmd = q.get_nowait()
                except queue.Empty:
                    break
                if cmd == "terminate":
                    stop = True
                    break
                if cmd == "pause":
                    print("Paused. Type 'continue' to resume.")
                    while True:
                        c2 = q.get()
                        if c2 == "continue":
                            break
                        if c2 == "terminate":
                            stop = True
                            break
                    break
            if stop:
                print("Round terminated.")
                break
            print(f"Currently drawing the {i}th winner...")
            time.sleep(0.5)
            u = lot.draw_one(p)
            if u is None:
                print("No eligible participants remain.")
                break
            drawn.append(u)
            res.add_winner(u, p)
            print(f"Congratulations {u.name} ({u.department}) for winning {p.name}")
        lot.finish_prize(p, drawn)
    return res


def main():
    lm = ListManager()
    users = load_users(lm)
    print("Groups:")
    for g, c in lm.get_group_info(users).items():
        print(f"  {g}: {c}")
    pm = PrizeManager()
    config_prizes(pm)
    prizes = pm.get_prizes()
    if not prizes:
        print("No prizes configured.")
        return
    res = run_lottery(users, prizes)
    print("\n=== Final winners (high to low) ===")
    ordered = list(reversed(prizes))
    for name, ws in res.get_winners_by_prize(ordered).items():
        for w in ws:
            print(f"{w.user.name} ({w.user.department}) - {name}")
    ok, msg = res.export_to_csv(ordered)
    print(msg)


if __name__ == "__main__":
    main()

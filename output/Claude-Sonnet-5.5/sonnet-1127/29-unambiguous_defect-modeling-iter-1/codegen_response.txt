"""Grid map and coordinate helpers."""

MAP_SIZE = 800
CELL = 100


def to_center(v):
    """Convert coordinate to grid cell center, clamped to the map."""
    v = max(0, min(MAP_SIZE - 1, int(v)))
    return (v // CELL) * CELL + CELL // 2


def convert_point(x, y):
    """Convert a point to its cell center."""
    return (to_center(x), to_center(y))


def cell_of(v):
    """Grid index of a coordinate."""
    return max(0, min(MAP_SIZE // CELL - 1, int(v) // CELL))


class GridMap:
    """8x8 grid map description."""

    def __init__(self, width=MAP_SIZE, height=MAP_SIZE, cell_size=CELL):
        self.width = width
        self.height = height
        self.cell_size = cell_size

"""Grid based path planning."""
from grid_map import GridMap


class Path:
    """A path of waypoints."""

    def __init__(self, waypoints, speed=1):
        self.waypoints = waypoints
        self.total_distance = sum(
            abs(a[0] - b[0]) + abs(a[1] - b[1])
            for a, b in zip(waypoints, waypoints[1:]))
        self.estimated_time = -(-self.total_distance // max(1, speed))


class PathPlanner:
    """Plans right-angle paths (horizontal first, then vertical)."""

    def __init__(self, grid=None):
        self.grid = grid or GridMap()

    def plan(self, start, targets, speed=1):
        """Plan a path from start through each target in order."""
        pts = [tuple(start)]
        for t in targets:
            t = tuple(t)
            cur = pts[-1]
            if cur[0] != t[0] and cur[1] != t[1]:
                pts.append((t[0], cur[1]))
            if pts[-1] != t:
                pts.append(t)
        return Path(pts, speed)

"""Rider model."""
import random

import globalvar

IDLE, PICKING, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        # start on a road: one coordinate on a grid line, the other not
        self.x = random.randint(0, 8) * 100
        self.y = random.randint(0, 7) * 100 + random.randint(1, 99)
        self.speed = 50
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0) or 0,
                         "x": self.x, "y": self.y})

    def _step_to(self, tx, ty):
        budget = self.speed
        dx = max(-budget, min(budget, tx - self.x))
        self.x += dx
        budget -= abs(dx)
        dy = max(-budget, min(budget, ty - self.y))
        self.y += dy

    def move(self):
        """Advance one tick."""
        if self.state == IDLE:
            if not self.orders:
                return
            self.state = PICKING
            self._record("accepted_order", self.orders[0].id)
        order = self.orders[0] if self.orders else None
        if order is None:
            self.state = IDLE
            return
        if self.state == PICKING:
            tx, ty = order.res_pos
            self._step_to(tx, ty)
            if (self.x, self.y) == (tx, ty):
                self.state = DELIVERING
                self._record("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            tx, ty = order.cus_pos
            self._step_to(tx, ty)
            if (self.x, self.y) == (tx, ty):
                now = globalvar.get_value("tick", 0) or 0
                order.status = "completed"
                order.completion_time = now
                order.time_taken = now - order.start_time
                self._record("delivered_to_customer", order.id)
                self.orders.pop(0)
                self.state = IDLE

"""Order assignment."""
import globalvar
from rider import Rider

RECRUIT_COST = 100
MAX_RIDERS = 10


def _dist(r, order):
    return abs(r.x - order.res_pos[0]) + abs(r.y - order.res_pos[1])


def recruit_rider():
    """Create a new rider if funds allow; return it or None."""
    funds = globalvar.get_value("funds", 0)
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    if funds < RECRUIT_COST or len(riders) >= MAX_RIDERS:
        return None
    rid = (globalvar.get_value("rider_id_counter", 0) or 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    globalvar.set_value("funds", funds - RECRUIT_COST)
    r = Rider(rid)
    riders.append(r)
    return r


def find_best_rider_for_order(order):
    """Pick the nearest, least loaded rider; recruit if none is idle."""
    riders = globalvar.get_value("riders") or []
    idle = [r for r in riders if not r.orders]
    if not idle:
        new = recruit_rider()
        if new is not None:
            return new
    pool = idle or riders
    if not pool:
        return None
    return min(pool, key=lambda r: (_dist(r, order), len(r.orders),
                                    r.rider_id))

"""Display manager."""
import globalvar
from grid_map import cell_of

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Textual output."""

    def status(self):
        riders = globalvar.get_value("riders") or []
        lines = ["Tick: %s  Funds: %s  Riders: %d" % (
            globalvar.get_value("tick"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) %s speed=%d orders=%s" % (
                r.rider_id, r.x, r.y, STATE_NAMES[r.state], r.speed,
                [o.id for o in r.orders]))
        lines.append(self.map())
        return "\n".join(lines)

    def orders(self):
        orders = globalvar.get_value("orders") or []
        if not orders:
            return "No orders."
        return "\n".join(
            "Order %d start=%s rest=%s cust=%s dist=%d time=%s rider=%s %s"
            % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
               o.time_taken, o.rider_id, o.status) for o in orders)

    def riders(self):
        out = []
        for r in globalvar.get_value("riders") or []:
            for e in r.log:
                out.append("t=%d rider %d %s order=%s at (%d,%d)" % (
                    e["time"], r.rider_id, e["event"], e["order_id"],
                    e["x"], e["y"]))
        return "\n".join(out) or "No records."

    def map(self, path=None):
        """ASCII 8x8 map: R rider, S restaurant, C customer, * route."""
        g = [["." for _ in range(8)] for _ in range(8)]
        if path:
            pts = path.waypoints
            for a, b in zip(pts, pts[1:]):
                n = max(abs(a[0] - b[0]), abs(a[1] - b[1])) // 10 + 1
                for i in range(n + 1):
                    x = a[0] + (b[0] - a[0]) * i // n
                    y = a[1] + (b[1] - a[1]) * i // n
                    g[cell_of(y)][cell_of(x)] = "*"
        for o in globalvar.get_value("orders") or []:
            if o.status != "completed":
                g[cell_of(o.res_pos[1])][cell_of(o.res_pos[0])] = "S"
                g[cell_of(o.cus_pos[1])][cell_of(o.cus_pos[0])] = "C"
        for r in globalvar.get_value("riders") or []:
            g[cell_of(r.y)][cell_of(r.x)] = "R"
        return "\n".join(" ".join(row) for row in reversed(g))

"""Input parsing and validation."""

ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
           "exit": "quit", "h": "help", "st": "start"}


class InputManager:
    """Parses commands and keeps history."""

    def __init__(self):
        self.command_history = []

    def parse(self, line):
        """Return (command, args)."""
        parts = line.strip().split()
        if not parts:
            return None, []
        self.command_history.append(line.strip())
        cmd = parts[0].lower()
        return ALIASES.get(cmd, cmd), parts[1:]

    @staticmethod
    def parse_coords(args):
        """Validate four integers in 0-800."""
        if len(args) != 4:
            raise ValueError("Need 4 integers: rx ry cx cy")
        try:
            vals = [int(a) for a in args]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be within 0-800")
        return vals

"""System manager."""
import json

import globalvar
from display import DisplayManager
from grid_map import GridMap, convert_point
from input_manager import InputManager
from order import find_best_rider_for_order, recruit_rider
from order_model import Order
from path_planner import PathPlanner


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", 10000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Coordinates the simulation."""

    def __init__(self):
        init_system()
        self.grid = GridMap()
        self.planner = PathPlanner(self.grid)
        self.display = DisplayManager()
        self.input = InputManager()

    def tick(self):
        """Advance simulated time."""
        globalvar.set_value("tick", globalvar.get_value("tick") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

    def submit_order(self, vals):
        """Create and assign an order; return a report string."""
        res = convert_point(vals[0], vals[1])
        cus = convert_point(vals[2], vals[3])
        order = Order(res, cus)
        before = len(globalvar.get_value("riders"))
        rider = find_best_rider_for_order(order)
        if rider is None:
            return "No rider available (insufficient funds)."
        added = len(globalvar.get_value("riders")) > before
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        globalvar.get_value("orders").append(order)
        path = self.planner.plan((rider.x, rider.y), [res, cus], rider.speed)
        return ("Order %d: restaurant=%s customer=%s\nNew rider added: %s\n"
                "Rider %d speed=%d distance=%d est. time=%d ticks\n%s" % (
                    order.id, res, cus, added, rider.rider_id, rider.speed,
                    path.total_distance, path.estimated_time,
                    self.display.map(path)))

    def save(self, filename="state.json"):
        """Persist state as JSON."""
        data = {
            "funds": globalvar.get_value("funds"),
            "tick": globalvar.get_value("tick"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state}
                       for r in globalvar.get_value("riders")],
            "orders": [{"id": o.id, "restaurant": o.res_pos,
                        "customer": o.cus_pos, "status": o.status,
                        "rider": o.rider_id}
                       for o in globalvar.get_value("orders")],
        }
        with open(filename, "w") as f:
            json.dump(data, f)

    def handle(self, line):
        """Process one command line; return output."""
        cmd, args = self.input.parse(line)
        if cmd is None:
            return ""
        if cmd == "quit":
            globalvar.set_value("running", False)
            return "Bye."
        if cmd == "start":
            if not args:
                args = input("rx ry cx cy: ").split()
            try:
                return self.submit_order(self.input.parse_coords(args))
            except ValueError as e:
                return "Error: %s" % e
        if cmd == "status":
            return self.display.status()
        if cmd == "orders":
            return self.display.orders()
        if cmd == "riders":
            return self.display.riders()
        if cmd == "help":
            return "Commands: start, status, orders, riders, quit/exit"
        return "Unknown command: %s (try help)" % cmd

"""Entry point."""
import globalvar
from system import SystemManager


def main():
    """Run the command loop; each iteration is one tick."""
    sm = SystemManager()
    print("Food Delivery Dispatch. Type help for commands.")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            out = sm.handle(line)
        except Exception as e:  # recover gracefully
            out = "Error: %s" % e
        if out:
            print(out)
        sm.tick()
    try:
        sm.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

"""Core data models: Edge and Graph."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}, {self.line})"


class Graph:
    """Weighted undirected graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = {}  # number -> list[Edge]
        self.station_lines = {}  # name -> list[str]

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.graph.setdefault(number, [])
        self.station_lines.setdefault(name, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def has_station(self, name):
        return name in self.station_to_number

    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        self.graph.setdefault(edge.source, [])
        self.graph.setdefault(edge.dest, [])
        self.graph[edge.source] = [e for e in self.graph[edge.source]
                                   if not (e.dest == edge.dest and e.line == edge.line)]
        self.graph[edge.dest] = [e for e in self.graph[edge.dest]
                                 if not (e.dest == edge.source and e.line == edge.line)]
        self.graph[edge.source].append(edge)
        self.graph[edge.dest].append(edge.reversed())

    def neighbors(self, number):
        return self.graph.get(number, [])

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

    def station_count(self):
        return len(self.station_to_number)

"""MinPQ and Dijkstra-based ShortestPath."""
import heapq
import itertools

MODE_SHORTEST = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"

TRANSFER_PENALTY = {MODE_SHORTEST: 0.0, MODE_LEAST_TRANSFER: 1e7, MODE_COMPREHENSIVE: 2000.0}


class PathfindingConfig:
    def __init__(self, mode=MODE_SHORTEST, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    """Binary-heap priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._info = {}  # state -> (cost, dist, transfers)
        self._prev = {}  # state -> (prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0.0)
        limit = cfg.max_transfers
        start = (self.source, None, 0)
        self._info[start] = (0.0, 0, 0)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        best_cost = {}
        while not pq.is_empty():
            cost, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line, _ = state
            _, dist, tr = self._info[state]
            if node not in best_cost or cost < best_cost[node]:
                best_cost[node] = cost
                self._best_state[node] = state
                self.dist_to[node] = dist
            for e in self.graph.neighbors(node):
                ntr = tr + (1 if line is not None and e.line != line else 0)
                if limit is not None and ntr > limit:
                    continue
                w = e.distance * (0.8 if e.line in cfg.preferred_lines else 1.0)
                ncost = cost + w + (penalty if ntr > tr else 0.0)
                nstate = (e.dest, e.line, ntr if limit is not None else 0)
                if nstate in done:
                    continue
                old = self._info.get(nstate)
                if old is None or ncost < old[0]:
                    self._info[nstate] = (ncost, dist + e.distance, ntr)
                    self._prev[nstate] = (state, e)
                    pq.push(ncost, nstate)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while state in self._prev:
            state, e = self._prev[state]
            edges.append(e)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math

FARE_TIERS = [(0, 6000, 3), (6000, 12000, 4), (12000, 22000, 5), (22000, 32000, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters (left-closed, right-open tiers)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    for lo, hi, price in FARE_TIERS:
        if lo <= d < hi:
            return price
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer; distance in meters."""
    return distance / 1000.0 * 1.5 + transfer_count * 5

"""Data management and integrity checks."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE = {
    "Line 1": [], "Line 2": [], "Line 4": [], "Line 5": [], "Line 6": [], "Line 8": [],
    "Line 10": [], "Line 13": [], "Line 14": [], "Line 15": [], "Changping Line": [],
    "Yizhuang Line": [], "Fangshan Line": [], "Airport Line": [],
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = Graph()
        self.lines = {}
        self._build()

    def _build(self):
        g = Graph()
        for i, st in enumerate(sorted(self.data.get("stations", []), key=lambda s: s["name"])):
            g.add_station(st["name"], i)
            g.set_station_lines(st["name"], st.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in g.station_to_number:
                    g.add_station(n, len(g.station_to_number))
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
        self.graph = g
        self.lines = {}
        for c in self.data.get("connections", []):
            seq = self.lines.setdefault(c["line"], [])
            for n in (c["source"], c["dest"]):
                if n not in seq:
                    seq.append(n)

    def get_graph(self):
        return self.graph

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """Add a line from an ordered station list and len-1 distances."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        for s in stations:
            if not any(x["name"] == s for x in self.data["stations"]):
                self.data["stations"].append({"name": s, "lines": [line_name]})
            else:
                self.update_station(s, add_line=line_name)
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self._build()

    def update_station(self, name, new_name=None, add_line=None):
        for st in self.data["stations"]:
            if st["name"] == name:
                if add_line and add_line not in st["lines"]:
                    st["lines"].append(add_line)
                if new_name:
                    st["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self._build()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self._build()
        return found


class DataIntegrityChecker:
    def __init__(self, manager):
        self.manager = manager
        self.check_result = False

    def run(self):
        data = self.manager.data
        problems = []
        names = {s["name"] for s in data.get("stations", [])}
        lines = {l for s in data.get("stations", []) for l in s.get("lines", [])}
        for ln in REPRESENTATIVE:
            if ln not in lines:
                problems.append(f"Line missing from station data: {ln}")
        conn_nodes = set()
        for c in data.get("connections", []):
            conn_nodes.update((c["source"], c["dest"]))
            for n in (c["source"], c["dest"]):
                if n not in names:
                    problems.append(f"Connection refers to unknown station: {n}")
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                problems.append(f"Invalid distance {d} between {c['source']} and {c['dest']}")
            if c["source"] == c["dest"]:
                problems.append(f"Self loop at {c['source']}")
        for n in names - conn_nodes:
            problems.append(f"Station without connections: {n}")
        self.check_result = not problems
        return self.check_result, problems

"""Route building, summary and report generation."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


class Route:
    def __init__(self, stations, edges, graph):
        self.stations = stations
        self.edges = edges
        self.total_distance = sum(e.distance for e in edges)
        self.segments = []  # (line, [stations])
        self.transfer_stations = []
        for i, e in enumerate(edges):
            if self.segments and self.segments[-1][0] == e.line:
                self.segments[-1][1].append(stations[i + 1])
            else:
                if self.segments:
                    self.transfer_stations.append(stations[i])
                self.segments.append((e.line, [stations[i], stations[i + 1]]))
        self.transfer_count = len(self.transfer_stations)
        self.lines_involved = []
        for line, _ in self.segments:
            if line not in self.lines_involved:
                self.lines_involved.append(line)


def format_distance(meters):
    if meters < 1000:
        return f"{meters} m"
    return f"{meters / 1000:.2f} km"


def summarize(route):
    return {
        "total_distance_km": round(route.total_distance / 1000.0, 2),
        "fare": Sum_money(route.total_distance),
        "transfer_count": route.transfer_count,
        "estimated_time": round(calculate_estimated_time(route.total_distance, route.transfer_count), 1),
    }


def text_report(start, end, route, summary):
    out = [f"Route: {start} -> {end}",
           f"Stations ({len(route.stations)}): " + " -> ".join(route.stations),
           f"Total distance: {format_distance(route.total_distance)} ({route.total_distance} m / "
           f"{summary['total_distance_km']} km)"]
    for line, sts in route.segments:
        out.append(f"  [{line}] {sts[0]} -> {sts[-1]} ({len(sts) - 1} stops)")
    if route.transfer_stations:
        out.append("Transfers at: " + ", ".join(route.transfer_stations))
    out.append(f"Transfer count: {summary['transfer_count']}")
    out.append("Lines: " + ", ".join(route.lines_involved))
    out.append(f"Estimated time: {summary['estimated_time']} minutes")
    out.append(f"Fare: {summary['fare']} yuan")
    return "\n".join(out)


def json_report(start, end, mode, route, summary):
    return json.dumps({
        "query": {"start": start, "end": end, "mode": mode},
        "path": route.stations,
        "distance_m": route.total_distance,
        "distance_km": summary["total_distance_km"],
        "fare": summary["fare"],
        "estimated_time": summary["estimated_time"],
        "transfer_count": summary["transfer_count"],
        "transfer_stations": route.transfer_stations,
        "lines": route.lines_involved,
    }, ensure_ascii=False, indent=2)

"""CLI entry point for the Beijing Subway fare system."""
import argparse
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_manager import DataIntegrityChecker, DataManager  # noqa: E402
from pathfinding import PathfindingConfig, ShortestPath  # noqa: E402
from report import Route, json_report, summarize, text_report  # noqa: E402

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def validate_input(text, graph):
    """Return (start, end, error)."""
    text = text.replace("；", ";")
    parts = [x.strip() for x in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None, None, FORMAT_ERR
    s, e = parts
    if s not in graph.station_to_number:
        return None, None, f"Starting station '{s}' does not exist, please check the station name"
    if e not in graph.station_to_number:
        return None, None, f"Ending station '{e}' does not exist, please check the station name"
    if s == e:
        return None, None, "Starting station cannot be the same as the ending station"
    return s, e, None


def run_query(graph, start, end, args):
    cfg = PathfindingConfig(args.mode, args.max_transfers,
                            [args.line_preference] if args.line_preference else [])
    try:
        sp = ShortestPath(graph, graph.station_to_number[start], cfg)
        nodes, edges = sp.path_with_edges(graph.station_to_number[end])
    except Exception as ex:  # path calculation failure
        return None, f"Path calculation failed: {ex}"
    if not nodes:
        return None, "No route found between the stations (check --max_transfers)"
    names = [graph.number_to_station[n] for n in nodes]
    route = Route(names, edges, graph)
    try:
        summary = summarize(route)
    except Exception as ex:  # fare failure
        return None, f"Fare calculation failed: {ex}"
    if args.format == "json":
        return json_report(start, end, args.mode, route, summary), None
    return text_report(start, end, route, summary), None


def emit(content, args):
    print(content)
    if args.output_file:
        try:
            with open(args.output_file, "a", encoding="utf-8") as f:
                f.write(content + "\n")
        except OSError as ex:
            print(f"Cannot write output file: {ex}")


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager()
    except Exception as ex:
        print(f"Failed to load subway data: {ex}")
        return 1
    if args.check_data:
        ok, problems = DataIntegrityChecker(dm).run()
        print("Data check passed" if ok else "Data check found problems:")
        for p in problems:
            print(" - " + p)
        return 0 if ok else 1
    graph = dm.get_graph()
    print("Beijing Subway fare calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        s, e, err = validate_input(line, graph)
        if err:
            print(err)
            continue
        content, err = run_query(graph, s, e, args)
        if err:
            print(err)
            continue
        emit(content, args)
    return 0


if __name__ == "__main__":
    sys.exit(main())

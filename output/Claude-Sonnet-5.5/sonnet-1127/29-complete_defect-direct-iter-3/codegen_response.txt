"""Grid based path planning."""


class PathPlanner:
    """Plans street-preferring right-angle paths."""

    @staticmethod
    def nearest_street(v):
        """Nearest street line coordinate (multiple of 100)."""
        return min(800, max(0, int(round(v / 100.0)) * 100))

    @classmethod
    def plan(cls, start, end):
        """Return waypoint list from start to end (start excluded)."""
        sx, sy = start
        ex, ey = end
        pts = []
        cur = (sx, sy)
        on_street = sx % 100 == 0 or sy % 100 == 0
        if not on_street and (sx, sy) != (ex, ey):
            # rider is inside a cell: step to nearest horizontal street
            # (special case: directly above/below target handled the same way)
            sy2 = cls.nearest_street(sy)
            pts.append((sx, sy2))
            cur = (sx, sy2)
        if cur[0] != ex and cur[1] % 100 == 0:
            pts.append((ex, cur[1]))
            cur = (ex, cur[1])
        elif cur[0] != ex:
            pts.append((ex, cur[1]))
            cur = (ex, cur[1])
        if cur[1] != ey:
            pts.append((ex, ey))
        return pts

    @classmethod
    def full_route(cls, points):
        """Route through several consecutive destinations."""
        route = [points[0]]
        for a, b in zip(points, points[1:]):
            route.extend(cls.plan(a, b))
        return route

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

import itertools

from src import globalvar

_counter = itertools.count(1)


def to_center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(800, int(v)))
    g = min(v // 100, 7)
    return g * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.completed_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Assign order to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self):
        """Mark as completed."""
        self.status = "completed"
        self.completed_time = globalvar.get_value("time", 0) or 0

    @property
    def duration(self):
        """Time taken, or None if unfinished."""
        if self.completed_time is None:
            return None
        return self.completed_time - self.start_time

    def to_dict(self):
        """Serialise."""
        return {"id": self.id, "res": list(self.res_pos),
                "cus": list(self.cus_pos), "status": self.status,
                "rider": self.rider_id, "start": self.start_time,
                "duration": self.duration, "distance": self.distance}

"""Rider model."""
from src import globalvar
from src.path_planner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.path = []
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0),
                         "x": self.x, "y": self.y})

    def _set_path(self, target):
        self.path = PathPlanner.plan((self.x, self.y), target)

    def _advance(self):
        budget = self.speed
        while budget > 0 and self.path:
            tx, ty = self.path[0]
            dx, dy = tx - self.x, ty - self.y
            dist = abs(dx) + abs(dy)
            if dist == 0:
                self.path.pop(0)
                continue
            step = min(budget, dist)
            if dx:
                self.x += step if dx > 0 else -step
            else:
                self.y += step if dy > 0 else -step
            budget -= step
            if step == dist:
                self.path.pop(0)

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            order = self.orders[0]
            self.state = 1
            order.status = "picking_up"
            self._log("accepted_order", order.id)
            self._set_path(order.res_pos)
        if not self.orders:
            self.state = 0
            return
        order = self.orders[0]
        self._advance()
        if self.state == 1 and not self.path and \
                (self.x, self.y) == tuple(order.res_pos):
            self.state = 2
            order.status = "delivering"
            self._log("arrived_at_merchant", order.id)
            self._set_path(order.cus_pos)
        if self.state == 2 and not self.path and \
                (self.x, self.y) == tuple(order.cus_pos):
            order.complete()
            self._log("delivered_to_customer", order.id)
            self.orders.pop(0)
            self.state = 0

"""Order management."""
from src import globalvar
from src.order_model import Order
from src.path_planner import PathPlanner


def find_best_rider_for_order(order):
    """Return best rider: idle first, nearest to restaurant, least loaded."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None

    def key(r):
        d = PathPlanner.distance((r.x, r.y), order.res_pos)
        return (0 if r.state == 0 and not r.orders else 1, d,
                len(r.orders), r.rider_id)
    return min(riders, key=key)


class OrderManager:
    """Creates, assigns and completes orders."""

    def create_order(self, coords):
        """Create an order from four coordinates."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        globalvar.get_value("orders").append(order)
        return order

    def assign(self, order):
        """Assign order to the best rider; returns rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider

"""System manager."""
import json

from src import globalvar
from src.rider import Rider

RIDER_COST = 300
INITIAL_FUNDS = 300
ORDER_REVENUE = 100


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("funds", INITIAL_FUNDS)


class SystemManager:
    """Manages funds, riders and the simulation clock."""

    def __init__(self):
        init_system()

    def recruit_if_possible(self):
        """Create a rider while funds >= 300. Returns number added."""
        added = 0
        while globalvar.get_value("funds") >= RIDER_COST:
            globalvar.set_value("funds", globalvar.get_value("funds")
                                - RIDER_COST)
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            globalvar.get_value("riders").append(Rider(rid))
            added += 1
        return added

    def add_funds(self, amount):
        """Add funds (may be negative)."""
        globalvar.set_value("funds", globalvar.get_value("funds") + amount)

    def tick(self):
        """Advance one time unit."""
        globalvar.set_value("time", globalvar.get_value("time") + 1)
        for r in globalvar.get_value("riders"):
            before = len(r.orders)
            try:
                r.move()
            except Exception:  # recover to consistent state
                r.state = 0
                r.path = []
            if len(r.orders) < before:
                self.add_funds(ORDER_REVENUE)
        self.recruit_if_possible()

    def save(self, path="state.json"):
        """Persist state as JSON."""
        data = {
            "funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "rider_id_counter": globalvar.get_value("rider_id_counter"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state,
                        "log": r.log}
                       for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

from src import globalvar

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Renders status and ASCII map."""

    @staticmethod
    def render_map(route=None, riders=None, orders=None):
        """17x17 canvas at 50-unit resolution; streets are '#'."""
        n = 17
        c = [["#" if (i % 2 == 0 or j % 2 == 0) else "." for j in range(n)]
             for i in range(n)]

        def idx(p):
            return (min(16, max(0, p[0] // 50)), min(16, max(0, p[1] // 50)))
        if route:
            for a, b in zip(route, route[1:]):
                ax, ay = idx(a)
                bx, by = idx(b)
                for x in range(min(ax, bx), max(ax, bx) + 1):
                    for y in range(min(ay, by), max(ay, by) + 1):
                        c[y][x] = "*"
        for o in orders or []:
            if o.status != "completed":
                x, y = idx(o.res_pos)
                c[y][x] = "S"
                x, y = idx(o.cus_pos)
                c[y][x] = "C"
        for r in riders or []:
            x, y = idx((r.x, r.y))
            c[y][x] = "R"
        return "\n".join("".join(row) for row in c)

    def status(self):
        """System snapshot text."""
        riders = globalvar.get_value("riders")
        lines = ["Time: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("Rider %d at (%d,%d) speed %d state=%s orders=%s"
                         % (r.rider_id, r.x, r.y, r.speed, STATES[r.state],
                            [o.id for o in r.orders]))
        lines.append(self.render_map(riders=riders,
                                     orders=globalvar.get_value("orders")))
        return "\n".join(lines)

    def orders(self):
        """Order details text."""
        out = []
        for o in globalvar.get_value("orders"):
            d = o.duration
            out.append("Order %d start=%d restaurant=%s customer=%s "
                       "distance=%d time=%s rider=%s status=%s" % (
                           o.id, o.start_time, o.res_pos, o.cus_pos,
                           o.distance, d if d is not None else "-",
                           o.rider_id, o.status))
        return "\n".join(out) or "No orders."

    def riders_log(self):
        """Rider event history text."""
        out = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                out.append("Rider %d t=%s %s order=%s at (%d,%d)" % (
                    r.rider_id, e["time"], e["event"], e["order_id"],
                    e["x"], e["y"]))
        return "\n".join(out) or "No rider events."

"""Input parsing and validation."""

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "exit": "quit", "t": "tick", "h": "history"}


class InputError(ValueError):
    """Invalid user input."""


class InputManager:
    """Parses commands and keeps history."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        """Return (command, args)."""
        parts = line.strip().split()
        if not parts:
            return None, []
        self.history.append(line.strip())
        cmd = parts[0].lower()
        return ALIASES.get(cmd, cmd), parts[1:]

    @staticmethod
    def parse_coords(tokens):
        """Validate four integers in 0-800."""
        if len(tokens) == 1:
            tokens = tokens[0].strip("[]").replace(",", " ").split()
        if len(tokens) != 4:
            raise InputError("Need four integers: rx ry cx cy")
        vals = []
        for t in tokens:
            t = t.strip("[],")
            try:
                v = int(t)
            except ValueError:
                raise InputError("Not an integer: %r" % t)
            if not 0 <= v <= 800:
                raise InputError("Coordinate out of range 0-800: %d" % v)
            vals.append(v)
        return vals

"""Command-line entry point."""
from src import globalvar
from src.display import DisplayManager
from src.input_manager import InputManager, InputError
from src.order import OrderManager
from src.path_planner import PathPlanner
from src.system import SystemManager


def handle_start(args, im, om, sm, dm):
    """Process a start command."""
    coords = im.parse_coords(args if args else input(
        "Enter rx ry cx cy: ").split())
    order = om.create_order(coords)
    added = sm.recruit_if_possible()
    rider = om.assign(order)
    if rider is None:
        return "No rider available (funds %d)." % globalvar.get_value("funds")
    route = PathPlanner.full_route(
        [(rider.x, rider.y), order.res_pos, order.cus_pos])
    total = sum(PathPlanner.distance(a, b) for a, b in zip(route, route[1:]))
    ticks = -(-total // rider.speed)
    return "\n".join([
        "Restaurant grid: (%d,%d)  Customer grid: (%d,%d)" % (
            order.res_pos[0] // 100, order.res_pos[1] // 100,
            order.cus_pos[0] // 100, order.cus_pos[1] // 100),
        "New rider added: %s" % ("yes" if added else "no"),
        "Assigned rider %d, speed %d, estimated time %d ticks" % (
            rider.rider_id, rider.speed, ticks),
        dm.render_map(route=route, orders=[order])])


def main():
    """Run the interactive loop."""
    sm = SystemManager()
    sm.recruit_if_possible()
    im, om, dm = InputManager(), OrderManager(), DisplayManager()
    print("Commands: start, status, orders, riders, tick [n], history, quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            cmd, args = im.parse_command(line)
            if cmd is None:
                continue
            if cmd == "start":
                print(handle_start(args, im, om, sm, dm))
            elif cmd == "status":
                print(dm.status())
            elif cmd == "orders":
                print(dm.orders())
            elif cmd == "riders":
                print(dm.riders_log())
            elif cmd == "history":
                print("\n".join(im.history))
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                for _ in range(max(0, n)):
                    sm.tick()
                continue
            elif cmd == "quit":
                globalvar.set_value("running", False)
                continue
            else:
                print("Unknown command: %s" % cmd)
            sm.tick()
        except (InputError, ValueError) as e:
            print("Error: %s" % e)
    try:
        sm.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

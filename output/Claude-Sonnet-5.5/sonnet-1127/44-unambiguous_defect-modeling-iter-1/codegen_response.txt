"""Core data models: Edge and Graph."""


class Edge:
    """Undirected connection between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists keyed by station number."""

    def __init__(self):
        self.adjacency = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, []).append(edge.reversed())

    def neighbors(self, number):
        return self.adjacency.get(number, [])

    def remove_edges_between(self, a, b, line=None):
        for x, y in ((a, b), (b, a)):
            self.adjacency[x] = [e for e in self.adjacency.get(x, [])
                                 if not (e.dest == y and (line is None or e.line == line))]

    def has_station(self, name):
        return name in self.station_to_number

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

    @property
    def num_stations(self):
        return len(self.station_to_number)

"""Dijkstra shortest path with a heap-based priority queue."""
import heapq
import itertools

MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = {"shortest_distance": 0.001, "least_transfer": 100000.0, "comprehensive": 1500.0}


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class MinPQ:
    """Min priority queue based on heapq."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


class ShortestPath:
    """Dijkstra over states (station, line, transfers) from a source station number."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_cost = {}
        self._best_state = {}
        self._parent = {}
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0.001)
        limit = cfg.max_transfers
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        self._parent[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in self._best_cost or c < self._best_cost[node]:
                self._best_cost[node] = c
                self._best_state[node] = st
                self.dist_to[node] = real[st]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0.0
                if line is not None and e.line != line:
                    nt = t + 1
                    extra = penalty
                if limit is not None and nt > limit:
                    continue
                w = e.distance
                if e.line in cfg.preferred_lines:
                    w = w * 0.9
                key_t = nt if limit is not None else 0
                ns = (e.dest, e.line, key_t)
                nc = c + w + extra
                if ns not in cost or nc < cost[ns] - 1e-12:
                    cost[ns] = nc
                    real[ns] = real[st] + e.distance
                    self._parent[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        edges = []
        st = self._best_state[target]
        while self._parent[st] is not None:
            st, e = self._parent[st]
            edges.append(e)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""
import math

FARE_TIERS = [(0, 6000, 3), (6000, 12000, 4), (12000, 22000, 5), (22000, 32000, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0 or distance != distance:
        raise ValueError("Invalid distance: %r" % (distance,))
    d = int(math.ceil(distance))
    for lo, hi, fare in FARE_TIERS:
        if d < hi:
            return fare
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: ~35 km/h average, 5 min per transfer, 2 min boarding."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("Invalid arguments")
    return round(distance / 1000.0 / 35.0 * 60.0 + transfer_count * 5 + 2, 1)

"""Data management and integrity checking."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE_STATIONS = {
    "1号线": "苹果园", "2号线": "西直门", "4号线": "西单", "5号线": "宋家庄",
    "10号线": "知春路", "13号线": "西二旗", "八通线": "土桥",
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = None
        self.rebuild_graph()

    def rebuild_graph(self):
        g = Graph()
        names = []
        seen = set()
        for s in self.data.get("stations", []):
            if s["name"] not in seen:
                seen.add(s["name"])
                names.append(s["name"])
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in seen:
                    seen.add(n)
                    names.append(n)
        for i, n in enumerate(names):
            g.add_station(n, i)
        lines = {}
        for s in self.data.get("stations", []):
            lines.setdefault(s["name"], [])
            for l in s.get("lines", []):
                if l not in lines[s["name"]]:
                    lines[s["name"]].append(l)
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                lines.setdefault(n, [])
                if c["line"] not in lines[n]:
                    lines[n].append(c["line"])
        for n in names:
            g.set_station_lines(n, lines.get(n, []))
        for c in self.data.get("connections", []):
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
        self.graph = g

    def station_names(self):
        return set(self.graph.station_to_number)

    def line_names(self):
        out = []
        for ls in self.graph.station_lines.values():
            for l in ls:
                if l not in out:
                    out.append(l)
        return out

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """Add a line: stations is an ordered list, distances has len(stations)-1 entries."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 entries")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.rebuild_graph()

    def update_station(self, name, new_name=None, lines=None):
        found = False
        for s in self.data["stations"]:
            if s["name"] == name:
                found = True
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
        if not found:
            raise KeyError(name)
        if new_name:
            for c in self.data["connections"]:
                if c["source"] == name:
                    c["source"] = new_name
                if c["dest"] == name:
                    c["dest"] = new_name
        self.rebuild_graph()

    def update_distance(self, a, b, distance, line=None):
        if distance <= 0:
            raise ValueError("distance must be positive")
        hit = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                hit = True
        if not hit:
            raise KeyError((a, b))
        self.rebuild_graph()

    def save(self, path=None):
        import json
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)


class DataIntegrityChecker:
    def __init__(self, manager):
        self.manager = manager
        self.stationCoverageValid = False
        self.connectionValid = False
        self.distanceValid = False
        self.messages = []

    def check(self):
        d = self.manager.data
        names = {s["name"] for s in d.get("stations", [])}
        lines = set(self.manager.line_names())
        self.messages = []
        missing = [(l, s) for l, s in REPRESENTATIVE_STATIONS.items()
                   if l in lines and s not in self.manager.station_names()]
        self.stationCoverageValid = bool(names) and not missing
        for l, s in missing:
            self.messages.append("Missing representative station %s on %s" % (s, l))
        bad = [c for c in d.get("connections", [])
               if c["source"] not in names or c["dest"] not in names or c["source"] == c["dest"]]
        g = self.manager.graph
        isolated = [n for n, num in g.station_to_number.items() if not g.neighbors(num)]
        self.connectionValid = not bad and not isolated and bool(d.get("connections"))
        for c in bad[:10]:
            self.messages.append("Bad connection: %s-%s" % (c["source"], c["dest"]))
        for n in isolated[:10]:
            self.messages.append("Isolated station: %s" % n)
        badd = [c for c in d.get("connections", [])
                if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0 or c["distance"] > 30000]
        self.distanceValid = not badd
        for c in badd[:10]:
            self.messages.append("Suspicious distance: %s-%s %s" % (c["source"], c["dest"], c["distance"]))
        return self.stationCoverageValid and self.connectionValid and self.distanceValid

    def report(self):
        ok = self.check()
        out = ["Station coverage: %s" % ("OK" if self.stationCoverageValid else "FAILED"),
               "Connections: %s" % ("OK" if self.connectionValid else "FAILED"),
               "Distances: %s" % ("OK" if self.distanceValid else "FAILED")]
        out += self.messages
        out.append("Data check %s" % ("passed" if ok else "failed"))
        return "\n".join(out)

"""Route summary and report generation."""
import json
from fare_calculator import Sum_money, calculate_estimated_time


def build_summary(graph, source_name, dest_name, nodes, edges, mode="shortest_distance"):
    stations = [graph.number_to_station[n] for n in nodes]
    total = sum(e.distance for e in edges)
    transfer_stations = []
    for i in range(1, len(edges)):
        if edges[i - 1].line != edges[i].line:
            transfer_stations.append(graph.number_to_station[edges[i].source])
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    try:
        fare = Sum_money(total)
    except ValueError as ex:
        raise RuntimeError("Fare calculation failed: %s" % ex)
    tc = len(transfer_stations)
    return {
        "query": {"start": source_name, "end": dest_name, "mode": mode},
        "path": stations,
        "distance_m": total,
        "distance_km": round(total / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": calculate_estimated_time(total, tc),
        "transfer_count": tc,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": _segments(graph, nodes, edges),
    }


def _segments(graph, nodes, edges):
    segs = []
    for i, e in enumerate(edges):
        if segs and segs[-1]["line"] == e.line:
            segs[-1]["to"] = graph.number_to_station[e.dest]
            segs[-1]["stops"] += 1
            segs[-1]["distance_m"] += e.distance
        else:
            segs.append({"line": e.line, "from": graph.number_to_station[e.source],
                         "to": graph.number_to_station[e.dest], "stops": 1, "distance_m": e.distance})
    return segs


def format_distance(m):
    if m < 1000:
        return "%d m" % m
    return "%d m (%.2f km)" % (m, m / 1000.0)


def to_text(s):
    q = s["query"]
    out = ["Route: %s -> %s (mode: %s)" % (q["start"], q["end"], q["mode"]),
           "Stations (%d): %s" % (len(s["path"]), " -> ".join(s["path"])),
           "Total distance: %s" % format_distance(s["distance_m"]),
           "Estimated time: %.1f minutes" % s["estimated_time_min"],
           "Transfers: %d" % s["transfer_count"]]
    if s["transfer_stations"]:
        out.append("Transfer stations: " + ", ".join(s["transfer_stations"]))
    out.append("Lines: " + ", ".join(s["lines"]))
    out.append("Segments:")
    for g in s["segments"]:
        out.append("  [%s] %s -> %s (%d stops, %d m)" % (g["line"], g["from"], g["to"], g["stops"], g["distance_m"]))
    out.append("Fare: %d yuan" % s["fare"])
    return "\n".join(out)


def to_json(s):
    return json.dumps(s, ensure_ascii=False, indent=2)


def render(s, fmt="text"):
    return to_json(s) if fmt == "json" else to_text(s)

"""Command line interface."""
import argparse
import re
import sys

from data_manager import DataManager, DataIntegrityChecker
from pathfinding import PathfindingConfig, ShortestPath
from report import build_summary, render

SEP = re.compile(r"[;；,，]")


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def validate_input(text, graph):
    """Return (start, end, error)."""
    parts = SEP.split(text.strip())
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        return None, None, "Input format incorrect, please use 'Starting Station; Ending Station' format"
    a, b = parts[0].strip(), parts[1].strip()
    if not graph.has_station(a):
        return None, None, "Starting station '%s' does not exist, please check the station name" % a
    if not graph.has_station(b):
        return None, None, "Ending station '%s' does not exist, please check the station name" % b
    if a == b:
        return None, None, "Starting station cannot be the same as the ending station"
    return a, b, None


def query(manager, a, b, args):
    g = manager.graph
    cfg = PathfindingConfig(args.mode, args.max_transfers,
                            [args.line_preference] if args.line_preference else [])
    try:
        sp = ShortestPath(g, g.station_to_number[a], cfg)
        t = g.station_to_number[b]
        nodes, edges = sp.path_with_edges(t)
    except Exception as ex:
        raise RuntimeError("Path calculation failed: %s" % ex)
    if not nodes:
        raise RuntimeError("No route found between '%s' and '%s' (check --max_transfers)" % (a, b))
    return build_summary(g, a, b, nodes, edges, args.mode)


def main(argv=None, stdin=None, stdout=None):
    args = parse_args(argv)
    stdin = stdin or sys.stdin
    stdout = stdout or sys.stdout
    try:
        manager = DataManager()
    except Exception as ex:
        print("Failed to load subway data: %s" % ex, file=stdout)
        return 1
    if args.check_data:
        checker = DataIntegrityChecker(manager)
        print(checker.report(), file=stdout)
        return 0 if checker.check() else 1
    if args.output_file:
        try:
            open(args.output_file, "w", encoding="utf-8").close()
        except OSError as ex:
            print("Cannot write output file: %s" % ex, file=stdout)
            args.output_file = None
    print("Enter 'Starting Station; Ending Station' (or 'q' to quit):", file=stdout)
    while True:
        line = stdin.readline()
        if not line:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        a, b, err = validate_input(line, manager.graph)
        if err:
            print(err, file=stdout)
            continue
        try:
            content = render(query(manager, a, b, args), args.format)
        except Exception as ex:
            print(str(ex), file=stdout)
            continue
        print(content, file=stdout)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(content + "\n")
            except OSError as ex:
                print("Cannot write output file: %s" % ex, file=stdout)
    return 0

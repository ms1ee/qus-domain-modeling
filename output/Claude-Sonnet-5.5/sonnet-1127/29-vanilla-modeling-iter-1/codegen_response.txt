"""Grid map and coordinate helpers."""


class GridMap:
    """8x8 grid, 100x100 units per cell."""

    def __init__(self, width=800, height=800, cell_size=100):
        self.width = width
        self.height = height
        self.cell_size = cell_size

    @property
    def cols(self):
        return self.width // self.cell_size

    @property
    def rows(self):
        return self.height // self.cell_size

    def cell_of(self, x, y):
        """Return (col, row) of the cell containing the point (clamped)."""
        c = min(max(int(x) // self.cell_size, 0), self.cols - 1)
        r = min(max(int(y) // self.cell_size, 0), self.rows - 1)
        return c, r

    def to_center(self, x, y):
        """Convert a coordinate to the centre of its grid cell."""
        c, r = self.cell_of(x, y)
        half = self.cell_size // 2
        return c * self.cell_size + half, r * self.cell_size + half


GRID = GridMap()


def to_center(x, y):
    """Module level shortcut for GridMap.to_center."""
    return GRID.to_center(x, y)

"""Grid based path planning (prefers moving along streets)."""

from gridmap import GRID


class PathPlanner:
    """Plans axis-aligned waypoint paths on the grid."""

    def __init__(self, grid=GRID):
        self.grid = grid

    def plan(self, start, end):
        """Return list of axis-aligned waypoints from start to end."""
        x, y = start
        ex, ey = end
        cs = self.grid.cell_size
        pts = []
        if x % cs != 0 and x != ex:
            # Off-street: first reach the nearest vertical street.
            nx = int(round(x / float(cs))) * cs
            nx = min(max(nx, 0), self.grid.width)
            pts.append((nx, y))
            x = nx
        pts.append((x, ey))
        pts.append((ex, ey))
        out = []
        cur = start
        for p in pts:
            if p != cur:
                out.append(p)
                cur = p
        return out

    def plan_multi(self, start, stops):
        """Plan a continuous path through several destinations."""
        path = []
        cur = start
        for s in stops:
            path.extend(self.plan(cur, s))
            cur = s
        return path

"""Rider model."""

import globalvar
from pathplanner import PathPlanner

_planner = PathPlanner()


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.path = []
        self.log = []
        self._log("spawn", None)

    @property
    def id(self):
        return self.rider_id

    def _log(self, event, order_id):
        self.log.append({
            "time": globalvar.get_value("tick", 0) or 0,
            "x": self.x,
            "y": self.y,
            "event": event,
            "order_id": order_id,
        })

    def _plan_to(self, target):
        self.path = _planner.plan((self.x, self.y), target)
        # Special case: already aligned with target on one axis.
        self.special_case = len(self.path) == 1

    def _follow(self):
        budget = self.speed
        while budget > 0 and self.path:
            wx, wy = self.path[0]
            dist = abs(wx - self.x) + abs(wy - self.y)
            if dist <= budget:
                self.x, self.y = wx, wy
                budget -= dist
                self.path.pop(0)
            else:
                if wx != self.x:
                    self.x += budget if wx > self.x else -budget
                else:
                    self.y += budget if wy > self.y else -budget
                budget = 0

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            order = self.orders[0]
            self.state = 1
            self._log("accepted_order", order.id)
            self._plan_to(tuple(order.res_pos))
        if self.state == 1:
            order = self.orders[0]
            self._follow()
            if (self.x, self.y) == tuple(order.res_pos):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
                self._plan_to(tuple(order.cus_pos))
            return
        if self.state == 2:
            order = self.orders[0]
            self._follow()
            if (self.x, self.y) == tuple(order.cus_pos):
                now = globalvar.get_value("tick", 0) or 0
                order.status = "completed"
                order.completion_time = now
                order.time_taken = now - order.start_time
                self.orders.pop(0)
                self._log("delivered_to_customer", order.id)
                self.path = []
                self.special_case = False
                self.state = 0

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order data model."""

import math
import itertools

import globalvar
from gridmap import to_center

_counter = itertools.count(1)


class Order:
    """A delivery order; positions are converted to grid cell centres."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = to_center(res_pos[0], res_pos[1])
        self.cus_pos = to_center(cus_pos[0], cus_pos[1])
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.time_taken = None
        self.distance = math.hypot(self.res_pos[0] - self.cus_pos[0],
                                   self.res_pos[1] - self.cus_pos[1])

    def assign_to(self, rider_id):
        """Mark order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""Order assignment logic."""

import math

import globalvar
from order_model import Order


def find_best_rider_for_order(order):
    """Return the rider closest (Euclidean) to the restaurant, or None."""
    riders = globalvar.get_value("riders", []) or []
    best = None
    best_key = None
    for r in riders:
        d = math.hypot(r.x - order.res_pos[0], r.y - order.res_pos[1])
        key = (d, len(r.orders))
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


class OrderManager:
    """Creates and assigns orders."""

    def create_order(self, rx, ry, cx, cy):
        order = Order((rx, ry), (cx, cy))
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        return order

    def assign(self, order):
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        return rider

    def pending_count(self):
        return sum(1 for o in globalvar.get_value("orders", [])
                   if o.status == "pending")

"""System manager and initialisation."""

import json

import globalvar
from rider import Rider
from order import OrderManager

RECRUIT_COST = 300


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Manages funds, riders and the simulation clock."""

    def __init__(self):
        init_system()
        self.order_manager = OrderManager()

    @property
    def funds(self):
        return globalvar.get_value("funds")

    @property
    def riders(self):
        return globalvar.get_value("riders")

    @property
    def tick_count(self):
        return globalvar.get_value("tick")

    def idle_riders(self):
        return sum(1 for r in self.riders if r.state == 0 and not r.orders)

    def maybe_recruit(self):
        """Recruit a rider if funds allow and pending > idle."""
        if (self.funds >= RECRUIT_COST and
                self.order_manager.pending_count() > self.idle_riders()):
            cnt = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", cnt)
            globalvar.set_value("funds", self.funds - RECRUIT_COST)
            rider = Rider(cnt)
            self.riders.append(rider)
            return rider
        return None

    def submit_order(self, rx, ry, cx, cy):
        """Create order, recruit if needed, assign. Returns (order, new_rider, rider)."""
        order = self.order_manager.create_order(rx, ry, cx, cy)
        new_rider = self.maybe_recruit()
        rider = self.order_manager.assign(order)
        return order, new_rider, rider

    def tick(self, n=1):
        for _ in range(n):
            globalvar.set_value("tick", self.tick_count + 1)
            for r in self.riders:
                r.move()

    def save(self, path="state.json"):
        data = {"funds": self.funds, "tick": self.tick_count,
                "rider_id_counter": globalvar.get_value("rider_id_counter"),
                "riders": [r.to_dict() for r in self.riders],
                "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Display manager: status and ASCII map."""

import globalvar
from gridmap import GRID
from pathplanner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Renders status, orders, logs, and map."""

    def __init__(self, system):
        self.system = system
        self.planner = PathPlanner()

    def render_map(self, route_cells=(), riders=True, orders=True):
        grid = [["." for _ in range(GRID.cols)] for _ in range(GRID.rows)]
        for c, r in route_cells:
            grid[r][c] = "#"
        if orders:
            for o in globalvar.get_value("orders", []):
                if o.status == "completed":
                    continue
                c, r = GRID.cell_of(*o.res_pos)
                grid[r][c] = "S"
                if o.status != "pending" or True:
                    if o.rider_id is not None and \
                            self._picked(o):
                        pass
                c, r = GRID.cell_of(*o.cus_pos)
                grid[r][c] = "C"
        if riders:
            for rd in self.system.riders:
                c, r = GRID.cell_of(rd.x, rd.y)
                grid[r][c] = "R"
        return "\n".join(" ".join(row) for row in grid)

    def _picked(self, o):
        return False

    def route_cells(self, start, stops):
        """Cells covered by the path start -> stops."""
        cells = []
        cur = start
        pts = [start] + self.planner.plan_multi(start, stops)
        for a, b in zip(pts, pts[1:]):
            steps = max(abs(b[0] - a[0]), abs(b[1] - a[1]))
            for i in range(0, steps + 1, 10):
                x = a[0] + (0 if b[0] == a[0] else (i if b[0] > a[0] else -i))
                y = a[1] + (0 if b[1] == a[1] else (i if b[1] > a[1] else -i))
                cell = GRID.cell_of(x, y)
                if cell not in cells:
                    cells.append(cell)
        return cells

    def route_report(self, rider, order):
        start = (rider.x, rider.y)
        stops = [order.res_pos, order.cus_pos]
        pts = self.planner.plan_multi(start, stops)
        total = 0
        cur = start
        for p in pts:
            total += abs(p[0] - cur[0]) + abs(p[1] - cur[1])
            cur = p
        ticks = -(-total // rider.speed)
        cells = self.route_cells(start, stops)
        return (self.render_map(cells) +
                "\nRoute length: %d, speed: %d/tick, estimated time: %d ticks"
                % (total, rider.speed, ticks))

    def status(self):
        s = self.system
        lines = ["Tick: %d  Funds: %d  Riders: %d"
                 % (s.tick_count, s.funds, len(s.riders))]
        for r in s.riders:
            lines.append("Rider %d at (%d, %d) state=%s orders=%s (%d)"
                         % (r.rider_id, r.x, r.y, STATES[r.state],
                            [o.id for o in r.orders], len(r.orders)))
        lines.append(self.render_map())
        return "\n".join(lines)

    def orders(self):
        orders = globalvar.get_value("orders", [])
        if not orders:
            return "No orders."
        lines = []
        for o in orders:
            tt = o.time_taken if o.time_taken is not None else "-"
            lines.append("Order %d start=%d restaurant=%s customer=%s "
                         "distance=%.1f time=%s rider=%s status=%s"
                         % (o.id, o.start_time, o.res_pos, o.cus_pos,
                            o.distance, tt, o.rider_id, o.status))
        return "\n".join(lines)

    def rider_logs(self):
        lines = []
        for r in self.system.riders:
            for e in r.log:
                lines.append("t=%d rider=%d pos=(%d, %d) event=%s order=%s"
                             % (e["time"], r.rider_id, e["x"], e["y"],
                                e["event"], e["order_id"]))
        return "\n".join(lines) if lines else "No rider events."

"""Command parsing and validation."""

try:
    import readline  # noqa: F401  (command history / line editing)
except ImportError:  # pragma: no cover
    pass

ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
           "exit": "quit", "t": "tick", "st": "start"}


class InputManager:
    """Parses and executes user commands."""

    def __init__(self, system, display):
        self.system = system
        self.display = display
        self.command_history = []

    def parse_coords(self, parts):
        if len(parts) != 4:
            raise ValueError("Enter exactly four integers: rx ry cx cy")
        vals = []
        for p in parts:
            p = p.strip("[],")
            try:
                v = int(p)
            except ValueError:
                raise ValueError("Coordinates must be integers: %r" % p)
            if not 0 <= v <= 800:
                raise ValueError("Coordinate out of range 0-800: %d" % v)
            vals.append(v)
        return vals

    def handle(self, line):
        """Execute a command line; return output text."""
        line = line.strip()
        if not line:
            return ""
        self.command_history.append(line)
        parts = line.replace(",", " ").split()
        cmd = ALIASES.get(parts[0].lower(), parts[0].lower())
        args = parts[1:]
        try:
            if cmd == "start":
                if not args:
                    args = input("Enter rx ry cx cy: ").replace(
                        ",", " ").split()
                rx, ry, cx, cy = self.parse_coords(args)
                order, new_rider, rider = self.system.submit_order(
                    rx, ry, cx, cy)
                out = ["Order %d: restaurant grid %s, customer grid %s"
                       % (order.id, order.res_pos, order.cus_pos),
                       "New rider added: %s" % (
                           "yes (id %d)" % new_rider.rider_id
                           if new_rider else "no")]
                if rider is None:
                    out.append("No rider available; order pending.")
                else:
                    out.append("Assigned to rider %d" % rider.rider_id)
                    out.append(self.display.route_report(rider, order))
                return "\n".join(out)
            if cmd == "status":
                return self.display.status()
            if cmd == "orders":
                return self.display.orders()
            if cmd == "riders":
                return self.display.rider_logs()
            if cmd == "tick":
                n = int(args[0]) if args else 1
                if n < 0 or n > 100000:
                    raise ValueError("tick count out of range")
                self.system.tick(n)
                return "Advanced to tick %d" % self.system.tick_count
            if cmd == "save":
                self.system.save(args[0] if args else "state.json")
                return "Saved."
            if cmd == "history":
                return "\n".join(self.command_history)
            if cmd == "quit":
                import globalvar
                globalvar.set_value("running", False)
                return "Bye."
            if cmd == "help":
                return ("Commands: start rx ry cx cy | status | orders | "
                        "riders | tick [n] | save [file] | history | quit")
            return "Unknown command. Type 'help'."
        except ValueError as e:
            return "Error: %s" % e
        except Exception as e:  # recovery
            return "Unexpected error: %s" % e

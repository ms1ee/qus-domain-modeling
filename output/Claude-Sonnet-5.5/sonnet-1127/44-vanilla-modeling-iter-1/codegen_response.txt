"""Core data models: Edge and Graph."""


class Edge:
    """Undirected connection between two station numbers on one line."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph keyed by station numbers."""

    def __init__(self):
        self.adjacency = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, []).append(edge.reversed())

    def neighbors(self, number):
        return self.adjacency.get(number, [])

    def has_station(self, name):
        return name in self.station_to_number

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

    def update_distance(self, a, b, distance, line=None):
        """Update distance between station numbers a and b; return count of edges changed."""
        n = 0
        for u, v in ((a, b), (b, a)):
            for i, e in enumerate(self.adjacency.get(u, [])):
                if e.dest == v and (line is None or e.line == line):
                    self.adjacency[u][i] = Edge(u, v, distance, e.line)
                    n += 1
        return n

    def remove_edges_between(self, a, b):
        self.adjacency[a] = [e for e in self.adjacency.get(a, []) if e.dest != b]
        self.adjacency[b] = [e for e in self.adjacency.get(b, []) if e.dest != a]

"""Dijkstra-based path finding with transfer awareness."""
import heapq
import itertools

SHORTEST = "shortest_distance"
LEAST_TRANSFER = "least_transfer"
COMPREHENSIVE = "comprehensive"

TRANSFER_PENALTY = 1500   # meters-equivalent per transfer (comprehensive mode)
PREFERENCE_FACTOR = 0.9   # cost factor on preferred lines
DEFAULT_TRANSFER_CAP = 15


class PathfindingConfig:
    def __init__(self, mode=SHORTEST, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class MinPQ:
    """Binary-heap based min priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class ShortestPath:
    """Dijkstra over states (station, last line, transfers)."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._prev = {}
        self._run()

    def _edge_cost(self, edge, transferred):
        mode = self.config.mode
        d = float(edge.distance)
        if mode == COMPREHENSIVE and edge.line in self.config.preferred_lines:
            d *= PREFERENCE_FACTOR
        if mode == LEAST_TRANSFER:
            return d + (1e7 if transferred else 0)
        if mode == COMPREHENSIVE:
            return d + (TRANSFER_PENALTY if transferred else 0)
        return d

    def _run(self):
        cap = self.config.max_transfers
        if cap is None:
            cap = DEFAULT_TRANSFER_CAP
        start = (self.source, None, 0)
        costs = {start: 0.0}
        dists = {start: 0}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        self._best_state[self.source] = start
        self.dist_to[self.source] = 0
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, tr = st
            if node not in self._best_state:
                self._best_state[node] = st
                self.dist_to[node] = dists[st]
            for e in self.graph.neighbors(node):
                transferred = line is not None and e.line != line
                ntr = tr + (1 if transferred else 0)
                if ntr > cap:
                    continue
                ns = (e.dest, e.line, ntr)
                if ns in done:
                    continue
                if self.config.mode == SHORTEST:
                    nc = c + e.distance + (0.001 if transferred else 0)
                else:
                    nc = c + self._edge_cost(e, transferred)
                if nc < costs.get(ns, float("inf")):
                    costs[ns] = nc
                    dists[ns] = dists[st] + e.distance
                    self._prev[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        st = self._best_state.get(target)
        if st is None:
            return [], []
        edges = []
        while st in self._prev:
            st, e = self._prev[st]
            edges.append(e)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        st = self._best_state.get(target)
        return st[2] if st else 0

"""Fare and time estimation."""
import math

FARE_TIERS = [(6000, 3), (12000, 4), (22000, 5), (32000, 6)]


def Sum_money(distance):
    """Fare in yuan for distance in meters (left-closed, right-open tiers)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    for limit, fare in FARE_TIERS:
        if d < limit:
            return fare
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5


class FareCalculator:
    def calculate(self, distance):
        return Sum_money(distance)

"""Data management: build graph, integrity check, updates."""
from data_loader import load_subway_data, DATA_FILE_PATH
from models import Graph, Edge


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = {"stations": [], "connections": []}
        self.graph = Graph()

    def load(self):
        self.data = load_subway_data(self.data_file)
        self.build_graph()
        return self.graph

    def build_graph(self):
        g = Graph()
        for i, s in enumerate(self.data.get("stations", [])):
            g.add_station(s["name"], i)
            g.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a, b = c["source"], c["dest"]
            for n in (a, b):
                if n not in g.station_to_number:
                    g.add_station(n, len(g.station_to_number))
                    g.set_station_lines(n, [c.get("line")] if c.get("line") else [])
            g.add_edge(Edge(g.station_to_number[a], g.station_to_number[b],
                            c["distance"], c.get("line")))
        self.graph = g
        return g

    def lines(self):
        s = set()
        for st in self.data.get("stations", []):
            s.update(st.get("lines", []))
        for c in self.data.get("connections", []):
            if c.get("line"):
                s.add(c["line"])
        return sorted(s)

    def check_data(self):
        """Return (ok, list of messages)."""
        msgs = []
        names = [s["name"] for s in self.data.get("stations", [])]
        nameset = set(names)
        if not names:
            msgs.append("No station data")
        if len(names) != len(nameset):
            msgs.append("Duplicate station names found")
        for s in self.data.get("stations", []):
            if not s.get("lines"):
                msgs.append("Station without line: %s" % s["name"])
        conns = self.data.get("connections", [])
        if not conns:
            msgs.append("No connection data")
        for c in conns:
            for k in ("source", "dest"):
                if c[k] not in nameset:
                    msgs.append("Connection references unknown station: %s" % c[k])
            if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0:
                msgs.append("Invalid distance %s-%s" % (c["source"], c["dest"]))
            elif c["distance"] > 20000:
                msgs.append("Suspicious distance %s-%s: %s" % (c["source"], c["dest"], c["distance"]))
        connected = {c["source"] for c in conns} | {c["dest"] for c in conns}
        for n in names:
            if n not in connected:
                msgs.append("Isolated station: %s" % n)
        # connectivity
        g = self.graph
        if g.adjacency:
            start = next(iter(g.adjacency))
            seen = {start}
            stack = [start]
            while stack:
                u = stack.pop()
                for e in g.neighbors(u):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        stack.append(e.dest)
            if len(seen) != len(g.adjacency):
                msgs.append("Network not fully connected (%d of %d reachable)"
                            % (len(seen), len(g.adjacency)))
        return (not msgs), msgs

    # ---- updates ----
    def add_line(self, line_name, station_names, distances):
        """Add a new line: station_names in order, distances between consecutive stations."""
        if len(distances) != len(station_names) - 1:
            raise ValueError("distances length must be stations-1")
        existing = {s["name"]: s for s in self.data["stations"]}
        for n in station_names:
            if n in existing:
                if line_name not in existing[n]["lines"]:
                    existing[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": int(d), "line": line_name})
        self.build_graph()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build_graph()
                return True
        return False

    def update_distance(self, a, b, distance):
        n = 0
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = int(distance)
                n += 1
        if n:
            self.build_graph()
        return n > 0

    def save(self, path=None):
        import json
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

"""Route construction, summaries and report formatting."""
import json
from fare_calculator import Sum_money, calculate_estimated_time


class Route:
    def __init__(self, stations, edges, total_distance, transfer_count):
        self.stations = stations
        self.edges = edges
        self.total_distance = total_distance
        self.transfer_count = transfer_count
        self.involved_lines = []
        for e in edges:
            if e.line not in self.involved_lines:
                self.involved_lines.append(e.line)

    def segments(self):
        """List of (line, [station names]) grouped by consecutive line."""
        segs = []
        for i, e in enumerate(self.edges):
            if not segs or segs[-1][0] != e.line:
                segs.append([e.line, [self.stations[i]]])
            segs[-1][1].append(self.stations[i + 1])
        return segs

    def transfer_stations(self):
        segs = self.segments()
        return [(segs[i][1][-1], segs[i][0], segs[i + 1][0]) for i in range(len(segs) - 1)]


def format_distance(meters):
    if meters < 1000:
        return "%d m" % meters
    return "%.2f km" % (meters / 1000.0)


def summary(route):
    return {
        "total_distance_km": round(route.total_distance / 1000.0, 2),
        "total_fare": Sum_money(route.total_distance),
        "transfer_count": route.transfer_count,
        "estimated_time": round(calculate_estimated_time(route.total_distance, route.transfer_count), 1),
    }


def to_text(route, start, end):
    s = summary(route)
    out = ["Route: %s -> %s" % (start, end),
           "Stations: " + " -> ".join(route.stations)]
    for line, sts in route.segments():
        out.append("  [%s] %s -> %s (%d stops)" % (line, sts[0], sts[-1], len(sts) - 1))
    out.append("Total distance: %s (%d m / %.2f km)" % (
        format_distance(route.total_distance), route.total_distance, s["total_distance_km"]))
    ts = route.transfer_stations()
    if ts:
        out.append("Transfers (%d):" % len(ts))
        for st, a, b in ts:
            out.append("  %s: %s -> %s" % (st, a, b))
    else:
        out.append("Transfers: none")
    out.append("Lines: " + ", ".join(str(l) for l in route.involved_lines))
    out.append("Estimated time: %.1f minutes" % s["estimated_time"])
    out.append("Fare: %d yuan" % s["total_fare"])
    return "\n".join(out)


def to_json(route, start, end, mode=None, max_transfers=None, line_preference=None):
    s = summary(route)
    d = {
        "query": {"start": start, "end": end, "mode": mode,
                  "max_transfers": max_transfers, "line_preference": line_preference},
        "path": route.stations,
        "distance": route.total_distance,
        "distance_km": s["total_distance_km"],
        "fare": s["total_fare"],
        "estimated_time": s["estimated_time"],
        "transfer_count": route.transfer_count,
        "transfer_stations": [{"station": a, "from_line": b, "to_line": c}
                              for a, b, c in route.transfer_stations()],
        "lines": route.involved_lines,
    }
    return json.dumps(d, ensure_ascii=False, indent=2)

"""CLI for the Beijing Subway fare calculation system."""
import argparse
import sys

from data_manager import DataManager
from data_loader import DATA_FILE_PATH
from pathfinding import PathfindingConfig, ShortestPath
from report import Route, to_text, to_json

PROMPT = "Enter 'Starting Station; Ending Station' (or 'q' to quit): "


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def parse_input(text):
    """Return (start, end, error)."""
    text = text.replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        return None, None, "Input format incorrect, please use 'Starting Station; Ending Station' format"
    return parts[0].strip(), parts[1].strip(), None


def query(graph, start, end, args):
    """Run a query; return output string or raise ValueError with message."""
    if start not in graph.station_to_number:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % start)
    if end not in graph.station_to_number:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % end)
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    prefs = [args.line_preference] if args.line_preference else []
    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    cfg.preferred_lines = prefs
    try:
        s = graph.station_to_number[start]
        t = graph.station_to_number[end]
        sp = ShortestPath(graph, s, cfg)
        nodes, edges = sp.path_with_edges(t)
    except Exception as ex:
        raise ValueError("Path calculation failed: %s" % ex)
    if not nodes:
        raise ValueError("No route found from '%s' to '%s' under the given constraints" % (start, end))
    names = [graph.number_to_station[n] for n in nodes]
    route = Route(names, edges, sp.dist_to[t], sp.get_transfer_count(t))
    try:
        if args.format == "json":
            return to_json(route, start, end, args.mode, args.max_transfers, args.line_preference)
        return to_text(route, start, end)
    except Exception as ex:
        raise ValueError("Fare calculation failed: %s" % ex)


def emit(text, args):
    print(text)
    if args.output_file:
        try:
            with open(args.output_file, "a", encoding="utf-8") as f:
                f.write(text + "\n\n")
        except OSError as ex:
            print("Cannot write output file: %s" % ex)


def main(argv=None):
    args = parse_args(argv)
    dm = DataManager(DATA_FILE_PATH)
    try:
        graph = dm.load()
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("Lines: %s" % ", ".join(str(l) for l in dm.lines()))
        print("Stations: %d" % len(graph.station_to_number))
        if ok:
            print("Data check passed.")
            return 0
        print("Data check found %d issue(s):" % len(msgs))
        for m in msgs[:50]:
            print("  - " + m)
        return 1
    while True:
        try:
            line = input(PROMPT)
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        start, end, err = parse_input(line)
        if err:
            print(err)
            continue
        try:
            emit(query(graph, start, end, args), args)
        except ValueError as ex:
            print(ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

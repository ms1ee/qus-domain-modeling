"""Graph data structures."""


class Edge:
    """Undirected weighted edge between two station numbers."""

    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def other(self, v):
        return self.dest if v == self.source else self.source

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists keyed by station number."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}
        self.edge_count = 0

    def add_node(self, v):
        self.adj.setdefault(v, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        if edge.dest != edge.source:
            self.adj[edge.dest].append(edge)
        self.edge_count += 1

    def neighbors(self, v):
        return self.adj.get(v, [])

    def nodes(self):
        return list(self.adj.keys())

    def edges(self):
        seen = set()
        for lst in self.adj.values():
            for e in lst:
                if id(e) not in seen:
                    seen.add(id(e))
                    yield e

    def V(self):
        return len(self.adj)

    def E(self):
        return self.edge_count

"""Dijkstra-based path finding with several query modes."""
import heapq
import itertools

INF = float("inf")
TRANSFER_PENALTY = 1000  # metres-equivalent per transfer in comprehensive mode
PREFERENCE_FACTOR = 0.9


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, key, item):
        heapq.heappush(self._h, (key, next(self._c), item))

    def pop(self):
        key, _, item = heapq.heappop(self._h)
        return key, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {v: INF for v in graph.nodes()}
        self.dist_to[source] = 0
        self._best = {}   # node -> state
        self._info = {}   # state -> (dist, transfers, prev_state, edge)
        self._run()

    def _key(self, dist, transfers, score):
        mode = self.config.mode
        if mode == "least_transfer":
            return (transfers, dist)
        if mode == "comprehensive":
            return (score, dist)
        return (dist,)

    def _run(self):
        cfg = self.config
        maxt = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None)
        self._info[start] = (0, 0, None, None)
        pq = MinPQ()
        pq.push(self._key(0, 0, 0), (start, 0, 0, 0))
        done = set()
        while not pq.is_empty():
            _, (state, dist, tr, score) = pq.pop()
            if state in done:
                continue
            done.add(state)
            v, line = state
            if v not in self._best:
                self._best[v] = state
                self.dist_to[v] = dist
            for e in self.graph.neighbors(v):
                w = e.other(v)
                ntr = tr
                if line is not None and e.line != line:
                    ntr += 1
                if maxt is not None and ntr > maxt:
                    continue
                nstate = (w, e.line)
                if nstate in done:
                    continue
                nd = dist + e.weight
                factor = PREFERENCE_FACTOR if e.line in pref else 1.0
                nscore = score + e.weight * factor + (ntr - tr) * TRANSFER_PENALTY
                key = self._key(nd, ntr, nscore)
                old = self._info.get(nstate)
                if old is not None and old[4] <= key:
                    continue
                self._info[nstate] = (nd, ntr, state, e, key)
                pq.push(key, (nstate, nd, ntr, nscore))

    def has_path_to(self, target):
        return target in self._best

    def _states(self, target):
        st = self._best.get(target)
        out = []
        while st is not None:
            out.append(st)
            st = self._info[st][2]
        out.reverse()
        return out

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        states = self._states(target)
        nodes = [s[0] for s in states]
        edges = [self._info[s][3] for s in states[1:]]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                count += 1
        return count

"""Beijing subway fare rules."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in metres (left-closed, right-open tiers)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5.0

"""Network building, integrity checks and data updates."""
from data_loader import load_subway_data, DATA_FILE_PATH
from models import Graph, Edge

MAIN_LINES = ["1号线", "2号线", "4号线", "5号线", "6号线", "8号线", "10号线",
              "13号线", "14号线", "15号线", "昌平线", "亦庄线", "房山线", "机场线"]


class SubwayNetwork:
    def __init__(self, data=None):
        self.data = data if data is not None else load_subway_data(DATA_FILE_PATH)
        self.rebuild()

    def rebuild(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in self.data.get("stations", []):
            self._add_station(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in self.station_to_number:
                    self._add_station(n, [c.get("line")])
            self.graph.add_edge(Edge(self.station_to_number[c["source"]],
                                     self.station_to_number[c["dest"]],
                                     c["distance"], c.get("line")))

    def _add_station(self, name, lines):
        if name in self.station_to_number:
            return
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name
        self.graph.add_node(num)
        self.graph.set_station_lines(name, lines)

    # ---- updates ----
    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: len(stations)-1 metres."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances length mismatch")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line not in known[n]["lines"]:
                    known[n]["lines"].append(line)
            else:
                self.data["stations"].append({"name": n, "lines": [line]})
        for i, d in enumerate(distances):
            self.data["connections"].append(
                {"source": stations[i], "dest": stations[i + 1], "distance": d, "line": line})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.rebuild()
                return True
        return False

    def update_distance(self, a, b, distance):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                found = True
        if found:
            self.rebuild()
        return found

    # ---- check ----
    def check_data(self):
        """Return (ok, list of messages)."""
        msgs, ok = [], True
        names = {s["name"] for s in self.data.get("stations", [])}
        if not names:
            return False, ["No station data"]
        msgs.append("Stations: %d" % len(names))
        present = set()
        for s in self.data["stations"]:
            present.update(s.get("lines", []))
        missing = [l for l in MAIN_LINES if l not in present]
        if missing:
            msgs.append("Warning: lines not found: " + ", ".join(missing))
        bad = 0
        connected = set()
        for c in self.data.get("connections", []):
            connected.update((c["source"], c["dest"]))
            if c["source"] not in names or c["dest"] not in names:
                bad += 1
                msgs.append("Connection refers to unknown station: %s-%s" % (c["source"], c["dest"]))
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
                bad += 1
                msgs.append("Suspicious distance: %s-%s %r" % (c["source"], c["dest"], d))
        isolated = names - connected
        if isolated:
            msgs.append("Warning: stations without connections: %d" % len(isolated))
        if bad:
            ok = False
        msgs.append("Connections: %d" % len(self.data.get("connections", [])))
        msgs.append("Data check " + ("passed" if ok else "FAILED"))
        return ok, msgs

"""Route reports (text and JSON)."""
import json
from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(m):
    return "%d meters" % m if m < 1000 else "%.2f kilometers" % (m / 1000.0)


def build_summary(net, start, end, nodes, edges, dist, transfers):
    names = [net.number_to_station[n] for n in nodes]
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = names[i + 1]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(names[i])
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1], "stations": 1})
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    return {
        "query": {"start": start, "end": end},
        "path": names,
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": Sum_money(dist),
        "estimated_time_min": round(calculate_estimated_time(dist, transfers), 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": segments,
    }


def to_json(summary):
    return json.dumps(summary, ensure_ascii=False, indent=2)


def to_text(s):
    out = ["Route: %s -> %s" % (s["query"]["start"], s["query"]["end"]),
           "Stations: " + " -> ".join(s["path"]),
           "Total distance: " + format_distance(s["distance_m"]) +
           " (%.2f km)" % s["distance_km"]]
    for seg in s["segments"]:
        out.append("  [%s] %s -> %s (%d segments)" % (seg["line"], seg["from"], seg["to"], seg["stations"]))
    out.append("Transfers: %d" % s["transfer_count"])
    if s["transfer_stations"]:
        out.append("Transfer stations: " + ", ".join(s["transfer_stations"]))
    out.append("Lines: " + ", ".join(str(l) for l in s["lines"]))
    out.append("Estimated time: %.1f minutes" % s["estimated_time_min"])
    out.append("Fare: %d yuan" % s["fare"])
    return "\n".join(out)

"""CLI entry point for the Beijing subway fare calculator."""
import argparse
import sys

from data_manager import SubwayNetwork
from pathfinding import ShortestPath, PathfindingConfig, INF
import report

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def validate(net, text):
    """Return (start, end, error)."""
    parts = text.replace("；", ";").split(";")
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        return None, None, FORMAT_ERR
    a, b = parts[0].strip(), parts[1].strip()
    if a not in net.station_to_number:
        return None, None, "Starting station '%s' does not exist, please check the station name" % a
    if b not in net.station_to_number:
        return None, None, "Ending station '%s' does not exist, please check the station name" % b
    if a == b:
        return None, None, "Starting station cannot be the same as the ending station"
    return a, b, None


def query(net, a, b, args):
    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    cfg.preferred_lines = [args.line_preference] if args.line_preference else []
    try:
        sp = ShortestPath(net.graph, net.station_to_number[a], cfg)
        t = net.station_to_number[b]
        if sp.dist_to.get(t, INF) == INF:
            return None, "No route found between '%s' and '%s' under the given constraints" % (a, b)
        nodes, edges = sp.path_with_edges(t)
        transfers = sp.get_transfer_count(t)
        dist = sum(e.weight for e in edges)
    except Exception as ex:
        return None, "Path calculation error: %s" % ex
    try:
        summary = report.build_summary(net, a, b, nodes, edges, dist, transfers)
    except Exception as ex:
        return None, "Fare calculation error: %s" % ex
    return summary, None


def main(argv=None):
    args = parse_args(argv)
    try:
        net = SubwayNetwork()
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1
    if args.check_data:
        ok, msgs = net.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1
    first = True
    print("Enter 'Starting Station; Ending Station' (or 'exit' to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("exit", "quit", "q"):
            break
        if not line:
            continue
        a, b, err = validate(net, line)
        if err:
            print(err)
            continue
        summary, err = query(net, a, b, args)
        if err:
            print(err)
            continue
        out = report.to_json(summary) if args.format == "json" else report.to_text(summary)
        print(out)
        if args.output_file:
            try:
                with open(args.output_file, "w" if first else "a", encoding="utf-8") as f:
                    f.write(out + "\n")
                first = False
            except OSError as ex:
                print("Cannot write output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Graph data structures."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.graph = {}          # number -> list[Edge]
        self.station_lines = {}  # name -> list[str]

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, n):
        self.graph.setdefault(n, [])

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.graph[edge.source].append(edge)
        self.graph[edge.dest].append(Edge(edge.dest, edge.source, edge.distance, edge.line))

    def adj(self, n):
        return self.graph.get(n, [])

    def nodes(self):
        return list(self.graph.keys())

"""Dijkstra shortest path with a heap-based priority queue."""
import heapq

TRANSFER_PENALTY = 1000000   # least_transfer mode
COMPREHENSIVE_PENALTY = 1500  # meters-equivalent per transfer


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class MinPQ:
    def __init__(self):
        self._h = []
        self._c = 0

    def push(self, key, item):
        self._c += 1
        heapq.heappush(self._h, (key, self._c, item))

    def pop(self):
        k, _, item = heapq.heappop(self._h)
        return k, item

    def __len__(self):
        return len(self._h)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._parent = {}   # state -> (prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        limit = cfg.max_transfers
        track = limit is not None
        if cfg.mode == "least_transfer":
            pen = TRANSFER_PENALTY
        elif cfg.mode == "comprehensive":
            pen = COMPREHENSIVE_PENALTY
        else:
            pen = 0
        pref = set(cfg.preferred_lines) if cfg.mode == "comprehensive" else set()
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        self._parent[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        best_cost = {}
        while len(pq):
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, tr = st
            if node not in best_cost:
                best_cost[node] = c
                self._best_state[node] = st
                self.dist_to[node] = real[st]
            for e in self.graph.adj(node):
                ntr = tr
                add = e.weight
                if line is not None and e.line != line:
                    ntr += 1
                    add += pen
                    if track and ntr > limit:
                        continue
                if e.line in pref:
                    add = add * 0.9
                key_tr = ntr if (track or pen) else 0
                ns = (e.dest, e.line, key_tr if track else (ntr if pen else 0))
                nc = c + add
                if ns not in cost or nc < cost[ns] - 1e-9:
                    cost[ns] = nc
                    real[ns] = real[st] + e.weight
                    self._parent[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self.dist_to

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        edges = []
        while self._parent[st] is not None:
            prev, e = self._parent[st]
            edges.append(e)
            st = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        n = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                n += 1
        return n

"""Fare and time calculation."""
import math

FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]


def Sum_money(distance):
    """Fare in yuan for distance in meters (distance rounded up to whole km)."""
    if distance is None or distance < 0:
        raise ValueError("invalid distance")
    km = math.ceil(distance / 1000.0)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 35 km/h average speed plus 5 min per transfer."""
    return distance / 1000.0 / 35.0 * 60.0 + 5.0 * transfer_count

def check_data(data):
    """Return list of problem strings (empty if OK)."""
    problems = []
    stations = {s["name"]: s for s in data.get("stations", [])}
    if not stations:
        problems.append("No stations")
    lines = set()
    for s in stations.values():
        lines.update(s.get("lines", []))
    if not lines:
        problems.append("No lines found")
    deg = {n: 0 for n in stations}
    for c in data.get("connections", []):
        for k in ("source", "dest"):
            if c[k] not in stations:
                problems.append("Connection references unknown station: %s" % c[k])
            else:
                deg[c[k]] += 1
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0:
            problems.append("Bad distance %s-%s: %r" % (c["source"], c["dest"], d))
        elif d > 30000:
            problems.append("Suspicious distance %s-%s: %r" % (c["source"], c["dest"], d))
    for n, k in deg.items():
        if k == 0:
            problems.append("Isolated station: %s" % n)
    return problems


def add_line(data, line_name, station_names, distances):
    """Add a new line; distances has len(station_names)-1 entries."""
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances length mismatch")
    index = {s["name"]: s for s in data["stations"]}
    for n in station_names:
        if n in index:
            if line_name not in index[n]["lines"]:
                index[n]["lines"].append(line_name)
        else:
            s = {"name": n, "lines": [line_name]}
            data["stations"].append(s)
            index[n] = s
    for a, b, d in zip(station_names, station_names[1:], distances):
        data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})


def modify_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    ok = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
            c["distance"] = distance
            ok = True
    return ok

"""CLI for the Beijing Subway fare calculation system."""
import argparse
import json
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph
from pathfinding import PathfindingConfig, ShortestPath
from fare_calculator import Sum_money, calculate_estimated_time
import data_manager


def build_graph(data):
    station_to_number = {}
    number_to_station = {}
    g = Graph()
    for s in data["stations"]:
        if s["name"] not in station_to_number:
            n = len(station_to_number)
            station_to_number[s["name"]] = n
            number_to_station[n] = s["name"]
            g.add_node(n)
        g.set_station_lines(s["name"], s.get("lines", []))
    for c in data["connections"]:
        if c["source"] in station_to_number and c["dest"] in station_to_number:
            g.add_edge(Edge(station_to_number[c["source"]], station_to_number[c["dest"]],
                            c["distance"], c["line"]))
    return g, station_to_number, number_to_station


def format_distance(m):
    return "%d m" % m if m < 1000 else "%.2f km (%d m)" % (m / 1000.0, m)


def query(g, s2n, n2s, start, end, config):
    src, dst = s2n[start], s2n[end]
    try:
        sp = ShortestPath(g, src, config)
        nodes, edges = sp.path_with_edges(dst)
    except Exception as e:
        raise RuntimeError("Path calculation failed: %s" % e)
    if not nodes:
        raise RuntimeError("No route found between '%s' and '%s'" % (start, end))
    dist = sp.dist_to[dst]
    transfers = sp.get_transfer_count(dst)
    try:
        fare = Sum_money(dist)
        minutes = calculate_estimated_time(dist, transfers)
    except Exception as e:
        raise RuntimeError("Fare calculation failed: %s" % e)
    segments = []
    transfer_stations = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["stations"].append(n2s[e.dest])
        else:
            if segments:
                transfer_stations.append(n2s[e.source])
            segments.append({"line": e.line, "stations": [n2s[e.source], n2s[e.dest]]})
    lines = []
    for sg in segments:
        if sg["line"] not in lines:
            lines.append(sg["line"])
    return {
        "query": {"start": start, "end": end, "mode": config.mode},
        "path": [n2s[n] for n in nodes],
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(minutes, 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": segments,
    }


def format_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"]))]
    for sg in r["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (sg["line"], sg["stations"][0],
                                                   sg["stations"][-1], len(sg["stations"]) - 1))
    out.append("Total distance: %s" % format_distance(r["distance_m"]))
    out.append("Transfers: %d" % r["transfer_count"])
    if r["transfer_stations"]:
        out.append("Transfer stations: %s" % ", ".join(r["transfer_stations"]))
    out.append("Lines: %s" % ", ".join(r["lines"]))
    out.append("Estimated time: %.0f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    if fmt == "json":
        return json.dumps(r, ensure_ascii=False, indent=2)
    return format_text(r)


def parse_input(line):
    line = line.replace("；", ";")
    parts = line.split(";")
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        return None
    return parts[0].strip(), parts[1].strip()


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except Exception as e:
        print("Failed to load data: %s" % e)
        return 1
    if args.check_data:
        problems = data_manager.check_data(data)
        if problems:
            print("Data check found %d problem(s):" % len(problems))
            for x in problems:
                print(" -", x)
            return 1
        print("Data check passed: %d stations, %d connections" %
              (len(data["stations"]), len(data["connections"])))
        return 0

    g, s2n, n2s = build_graph(data)
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []

    print("Beijing Subway Fare Calculator. Input 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        parsed = parse_input(line)
        if not parsed:
            print("Input format incorrect, please use 'Starting Station; Ending Station' format")
            continue
        start, end = parsed
        if start not in s2n:
            print("Starting station '%s' does not exist, please check the station name" % start)
            continue
        if end not in s2n:
            print("Ending station '%s' does not exist, please check the station name" % end)
            continue
        if start == end:
            print("Starting station cannot be the same as the ending station")
            continue
        try:
            r = query(g, s2n, n2s, start, end, config)
        except RuntimeError as e:
            print(e)
            continue
        text = render(r, args.format)
        print(text)
        if args.output_file:
            try:
                with open(args.output_file, "w", encoding="utf-8") as f:
                    f.write(text + "\n")
                print("Result saved to %s" % args.output_file)
            except OSError as e:
                print("Failed to write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

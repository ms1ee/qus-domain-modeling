import re
from lottery_system.models import User


class ListManager:
    DEFAULT_DATA = [
        ("Zhang San", "Technology Department"),
        ("Li Si", "Technology Department"),
        ("Wang Wu", "Technology Department"),
        ("Zhao Liu", "Product Department"),
        ("Sun Qi", "Product Department"),
        ("Zhou Ba", "Product Department"),
        ("Wu Jiu", "Marketing Department"),
        ("Zheng Shi", "Marketing Department"),
        ("Chen Yi", "Cheng Ru Group"),
        ("Liu Er", "Cheng Ru Group"),
        ("Huang San", "Cheng Ru Group"),
        ("Lin Si", "Operations Department"),
        ("He Wu", "Operations Department"),
        ("Gao Liu", "Finance Department"),
        ("Luo Qi", "Finance Department"),
    ]

    def load_default_list(self):
        return [User(n, d) for n, d in self.DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except FileNotFoundError:
            return [], [f"File not found: {path}"]
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in re.split(r"[,;，；]", line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, a, d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        self.prizes.append(Prize(name, quantity, list(avoid_groups or []),
                                 associated_drawer))

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        if (not isinstance(order, list) or len(order) != n
                or sorted(order) != list(range(1, n + 1))):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random


def _is_cheng_ru(dept):
    return "cheng ru" in dept.lower()


class LotteryManager:
    SPECIAL_DRAWER = "Li Bin"
    MONITORED = ("Chen Qixian", "Jin Duo")
    PROTECTED_DRAWER = "Ma Jiaqi"

    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {u.name for u in self.winners}
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == self.SPECIAL_DRAWER:
            return pool
        if (drawer == self.PROTECTED_DRAWER
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_cheng_ru(u.department)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_round(self, prize, winners):
        if prize.associated_drawer in self.MONITORED:
            self.last_two_draw_results.append(
                any(_is_cheng_ru(u.department) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def rule_description(self, prize):
        if prize.associated_drawer == self.SPECIAL_DRAWER:
            return "Only previous winners excluded; avoidance groups ignored"
        d = "Previous winners excluded"
        if prize.avoid_groups:
            d += "; avoiding groups: " + ", ".join(prize.avoid_groups)
        if prize.associated_drawer == self.PROTECTED_DRAWER:
            d += "; Cheng Ru group protection may apply"
        return d

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, time=None):
        self.user = user
        self.prize_name = prize_name
        self.time = time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize.name))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def rank(prize):
    n = prize.name.lower()
    for i, k in enumerate(("first", "second", "third")):
        if k in n:
            return i
    return 3


def load_users(lm):
    users, errors = [], []
    if input("Load (1) default list or (2) custom txt? [1]: ").strip() == "2":
        users, errors = lm.load_custom_list(input("Path: ").strip())
    else:
        users, errors = lm.load_default_list()
    for e in errors:
        print("  ", e)
    if errors and input("Errors found. Remove invalid lines and continue? (y/n): ").lower() != "y":
        return []
    users, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate removed: {d.name} ({d.department})")
    print("Groups:")
    for g, c in lm.get_group_info(users).items():
        print(f"  {g}: {c}")
    return users


def configure_prizes(pm):
    if input("Load preset prizes? (y/n) [y]: ").strip().lower() != "n":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            q = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            drawer = input("Lottery person: ").strip() or None
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    ps = pm.get_prizes()
    for i, p in enumerate(ps, 1):
        print(f"{i}. {p.name} x{p.quantity} drawer={p.associated_drawer} avoid={p.avoid_groups}")
    o = input("New order (space separated numbers, blank to keep): ").split()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o]):
                print("Invalid order; kept.")
        except ValueError:
            print("Invalid order; kept.")


def run_round(lot, res, prize, idx):
    print(f"\n=== Round {idx}: {prize.name} x{prize.quantity} (drawer: {prize.associated_drawer}) ===")
    print("Rules:", lot.rule_description(prize))
    winners = []
    for k in range(1, prize.quantity + 1):
        while True:
            cmd = input("Enter to draw (pause/continue/terminate): ").strip().lower()
            if cmd == "pause":
                while input("Paused. Type 'continue': ").strip().lower() != "continue":
                    pass
                continue
            break
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing winner #{k}")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible participants left.")
            break
        winners.append(u)
        res.add_winner(u, prize)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.finish_round(prize, winners)
    print("Round results:")
    for i, u in enumerate(winners, 1):
        print(f"  {i}. {u.name} ({u.department})")


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No participants. Exiting.")
        return
    pm = PrizeManager()
    configure_prizes(pm)
    prizes = pm.get_prizes()
    lot, res = LotteryManager(users), ResultManager()
    for i, p in enumerate(prizes, 1):
        run_round(lot, res, p, i)
    print("\n=== Final winners ===")
    ordered = sorted(prizes, key=rank)
    for name, ws in res.get_winners_by_prize(ordered).items():
        for w in ws:
            print(f"{name}: {w.user.name} ({w.user.department}) {w.time:%H:%M:%S}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(res.export_to_csv(ordered)[1])


if __name__ == "__main__":
    main()

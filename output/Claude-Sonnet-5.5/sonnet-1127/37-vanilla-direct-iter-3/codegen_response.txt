from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Cheng Ru group"),
    ("Zheng Shi", "Cheng Ru group"),
    ("Feng Yi", "Cheng Ru group"),
    ("Chen Er", "Marketing Department"),
    ("Chu San", "Marketing Department"),
    ("Wei Si", "Operations Department"),
    ("Jiang Wu", "Operations Department"),
    ("Shen Liu", "Operations Department"),
    ("Han Qi", "Human Resources"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error - '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in (("Third Prize - Movie Tickets", 3),
                          ("Second Prize - Shopping Cards", 2),
                          ("First Prize - 1 Day Annual Leave", 1)):
            try:
                self.add_prize(name, qty)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            ok = sorted(order) == list(range(1, n + 1))
        except TypeError:
            return False
        if not ok:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

PROTECTED_GROUP = "Cheng Ru group"
SPECIAL_DRAWER = "Li Bin"
TRACKED_DRAWERS = ("Chen Qixian", "Jin Duo")
GUARDED_DRAWER = "Ma Jiaqi"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _winner_names(self):
        return {u.name for u in self.winners}

    def _get_eligible_pool(self, prize):
        won = self._winner_names()
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == GUARDED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if u.department == PROTECTED_GROUP]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_round(self, prize, round_winners):
        if prize.associated_drawer in TRACKED_DRAWERS:
            self.last_two_draw_results.append(
                any(u.department == PROTECTED_GROUP for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, win_time=None):
        self.user = user
        self.prize_name = prize_name
        self.win_time = win_time or datetime.now()


def _level(name):
    n = name.lower()
    for i, k in enumerate(("first", "second", "third"), 1):
        if k in n:
            return i
    return 99


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        names = []
        for p in prizes:
            if p.name not in names:
                names.append(p.name)
        names.sort(key=_level)
        return {n: [w for w in self.winners if w.prize_name == n] for n in names}

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for n, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{n},"
                                f"{w.win_time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    c = input("Load list: 1) default 2) custom txt [1]: ").strip()
    if c == "2":
        users, errors = lm.load_custom_list(input("Path: ").strip())
        if errors:
            for e in errors:
                print(e)
            if input("Errors found. Remove invalid data and continue? (y/n): ").lower() != "y":
                return None
    else:
        users, _ = lm.load_default_list()
    valid, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate name removed: {d.name} ({d.department})")
    return valid


def show_groups(lm, users):
    for g, c in lm.get_group_info(users).items():
        print(f"  {g}: {c}")


def configure_prizes(pm, groups):
    if input("Load default prizes? (y/n) [y]: ").strip().lower() != "n":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            qty = int(input("Quantity: "))
            avoid = [g.strip() for g in input(
                f"Avoid groups (comma separated, from {list(groups)}): ").split(",") if g.strip()]
            drawer = input("Associated drawer (optional): ").strip() or None
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    o = input("New order (space separated numbers, Enter to keep): ").split()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o]):
                print("Invalid order, keeping original.")
        except ValueError:
            print("Invalid order, keeping original.")


def run_round(prize, lot, rm):
    avoid = ", ".join(prize.avoid_groups) or "none"
    print(f"\n=== {prize.name} | quantity {prize.quantity} | avoid: {avoid} | "
          f"drawer: {prize.associated_drawer or 'n/a'} ===")
    got = []
    i = 1
    while i <= prize.quantity:
        cmd = input("Enter to draw (pause/continue/terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        rm.add_winner(u, prize)
        got.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, got)
    print("Round results:")
    for u in got:
        print(f"  {u.name} ({u.department})")


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No users loaded.")
        return
    show_groups(lm, users)
    pm = PrizeManager()
    configure_prizes(pm, lm.get_group_info(users).keys())
    lot = LotteryManager(users)
    rm = ResultManager()
    for p in pm.get_prizes():
        run_round(p, lot, rm)
    print("\n=== Final winners ===")
    for n, ws in rm.get_winners_by_prize(pm.get_prizes()).items():
        for w in ws:
            print(f"{n}: {w.user.name} ({w.user.department})")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(rm.export_to_csv(pm.get_prizes())[1])


if __name__ == "__main__":
    main()

I had to guess at several details the PRD and Interface leave open. Nothing has been run or tested.

- **Custom list format:** `load_custom_list` accepts both a comma and a semicolon as the separator.
- **Fallback rule:** `last_two_draw_results` holds, for each of the last two draws by Chen Qixian and Jin Duo, whether the winner was a Cheng Ru group member. When Ma Jiaqi draws and the list has two entries that are both `False`, the pool is limited to unwon Cheng Ru members. If that pool is empty, it falls back to the normal pool.
- **Result recording:** `LotteryManager.draw` records the result only for Chen Qixian and Jin Duo.
- **Li Bin rule:** For Li Bin, only previous winners are excluded and avoidance groups are ignored.
- **Avoidance ordering:** `_get_eligible_pool` returns non-avoided users first and avoided users after them. The PRD says avoided members remain eligible but are drawn after others, while section 2.4 says they are excluded. Because `_get_eligible_pool` is a public interface, I kept the ordering behaviour, with the exclusion applied when drawing: `draw_one` picks from the non-avoided users and only uses the avoided ones if no one else is left.
- **Cheng Ru restriction:** Under the fallback, Cheng Ru members are returned even if they are an avoidance group.
- **Group matching:** `_get_eligible_pool` returns only the non-avoided users, and avoided users are appended after them. For an avoidance group that is also Cheng Ru, the fallback overrides it.
- **Reorder:** `reorder_prizes` returns `False` unless the order is a valid permutation of the 1-based positions.
- **Default data:** The built-in list and prize library are small samples I made up. The default prizes load with no errors.

class ListManager:
    DEFAULT = [
        ("Zhang San", "Technology Department"),
        ("Li Si", "Technology Department"),
        ("Wang Wu", "Product Department"),
        ("Zhao Liu", "Product Department"),
        ("Chen Qixian", "Operations Department"),
        ("Jin Duo", "Operations Department"),
        ("Ma Jiaqi", "Marketing Department"),
        ("Li Bin", "Marketing Department"),
        ("Sun Qi", "Cheng Ru group"),
        ("Zhou Ba", "Cheng Ru group"),
        ("Wu Jiu", "Cheng Ru group"),
        ("Zheng Shi", "Finance Department"),
    ]

    def load_default_list(self):
        return [User(n, d) for n, d in self.DEFAULT], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        if not os.path.exists(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except Exception as e:
            return [], [f"Read error: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            sep = "," if "," in line else (";" if ";" in line else None)
            if sep is None:
                errors.append(f"Format Error: Line {i}: {line}")
                continue
            parts = line.split(sep)
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Format Error: Line {i}: {line}")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for n, q in [("Third Prize - Movie Tickets", 3),
                     ("Second Prize - Shopping Cards", 2),
                     ("First Prize - 1 Day Annual Leave", 1)]:
            try:
                self.add_prize(n, q)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or int(quantity) <= 0:
            raise ValueError("Invalid prize")
        p = Prize(name, int(quantity), list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "Cheng Ru group"
TRACKED = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
SPECIAL = "Li Bin"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _base(self, prize):
        won = {w.name for w in self.winners}
        return [u for u in self.users if u.name not in won]

    def _get_eligible_pool(self, prize):
        pool = self._base(prize)
        drawer = prize.associated_drawer
        if (drawer == FALLBACK_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            cr = [u for u in pool if u.department == CHENG_RU]
            if cr:
                return cr
        if drawer == SPECIAL:
            return pool
        return [u for u in pool if u.department not in prize.avoid_groups]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            pool = [u for u in self._base(prize)]
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        if prize.associated_drawer in TRACKED:
            self.last_two_draw_results.append(user.department == CHENG_RU)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        return user

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize, time=None):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize))

    def get_winners_by_prize(self, prizes):
        res = {}
        for p in prizes:
            res[p.name] = [w for w in self.winners
                           if getattr(w.prize, "name", w.prize) == p.name]
        return res

    def export_to_csv(self, prizes):
        try:
            path = os.path.join(os.getcwd(),
                                f"awards_result_{datetime.now():%Y%m%d}.txt")
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for p in prizes:
                    for w in grouped[p.name]:
                        f.write(f"{w.user.name},{w.user.department},{p.name},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
            return True, f"Results successfully exported to {path}"
        except Exception as e:
            return False, f"Export failed: {e}"

from lottery_system.services.prize_manager import PrizeManager

def main():

    users, _ = lm.load_default_list()
    users, _ = lm.validate_users(users)
    pm = PrizeManager()
    pm.load_default_prizes()
    lot = LotteryManager(users)
    rm = ResultManager()
    for p in pm.get_prizes():

            if u is None:
                break
            rm.add_winner(u, p)
            print(f"Congratulations {u.name} ({u.department}) for winning {p.name}")
    print(rm.export_to_csv(pm.get_prizes())[1])


if __name__ == "__main__":
    main()

The PRD is internally inconsistent, so I made these choices:

- **Rider state:** 0 = idle, 1 = picking up, 2 = delivering.
- **Rider speed:** 20 units/tick, per the Rider module. The recruitment text says 10, and the Rider spec says 20, so I used 20.
- **Recruitment:** a new rider spawns at (400, 400) and costs 500.
- **Coordinate conversion:** I used the top-left corner of the grid cell, i.e. `(v // 100) * 100`, clamped to 0–700. This follows the order-creation text.
- **Rider spawn position:** (400, 400) sits on a grid point, which the PRD calls a restaurant/customer location. I kept it as specified.

Order handling:
- Assigning an order appends it to `rider.orders`.
- `move()` picks the first non-completed order and sets the state to 1 if the rider is idle.
- `move()` logs "accepted_order" when the rider first takes an order.
- Each tick moves the rider toward the restaurant, then the customer.
- Movement is axis-aligned: horizontal first, then vertical.
- On arrival at the restaurant the state becomes 2 and "arrived_at_merchant" is logged.
- On arrival at the customer the order is marked completed and "delivered_to_customer" is logged.
- After a delivery the rider moves on to the next pending order, or goes idle.

The interface exposes `Order.assign_to(rider_id)` and `find_best_rider_for_order(order)`. The best rider is the one with the fewest pending orders, with ties broken by distance.

"""Order data model."""
import itertools

from src import globalvar

_counter = itertools.count(1)


def to_grid(v):
    """Convert coordinate to top-left corner of its grid cell."""
    v = int(v)
    return max(0, min(700, (v // 100) * 100))


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_grid(res_pos[0]), to_grid(res_pos[1]))
        self.cus_pos = (to_grid(cus_pos[0]), to_grid(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.end_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Record the rider this order is assigned to."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def to_list(self):
        """Return [rx, ry, cx, cy]."""
        return [self.res_pos[0], self.res_pos[1],
                self.cus_pos[0], self.cus_pos[1]]

"""Rider model."""
from src import globalvar


class Rider:
    """A delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._current = None
        self._record("spawn", None)

    def _record(self, event, order):
        self.log.append({
            "event": event,
            "order_id": getattr(order, "id", None),
            "time": globalvar.get_value("time", 0) or 0,
            "pos": (self.x, self.y),
        })

    def _pending(self):
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def _step(self, target):
        """Move up to speed units toward target, horizontal first."""
        budget = self.speed
        tx, ty = target
        dx = max(-budget, min(budget, tx - self.x))
        self.x += dx
        budget -= abs(dx)
        if budget > 0:
            dy = max(-budget, min(budget, ty - self.y))
            self.y += dy
        return (self.x, self.y) == (tx, ty)

    def move(self):
        """Advance one tick."""
        if self._current is None or self._current.status == "completed":
            self._current = self._pending()
            if self._current is None:
                self.state = 0
                return
            self.state = 1
            self._current.status = "picking_up"
            self._record("accepted_order", self._current)
        order = self._current
        if self.state == 1:
            if self._step(order.res_pos):
                self.state = 2
                self._record("arrived_at_merchant", order)
        elif self.state == 2:
            if self._step(order.cus_pos):
                order.status = "completed"
                order.end_time = globalvar.get_value("time", 0) or 0
                self._record("delivered_to_customer", order)
                self._current = None
                self.state = 0

"""Order assignment."""
from src import globalvar


def _pending_count(rider):
    return sum(1 for o in rider.orders if o.status != "completed")


def find_best_rider_for_order(order):
    """Return rider with fewest pending orders, ties by distance."""
    riders = globalvar.get_value("riders", [])
    if not riders:
        return None
    rx, ry = order.res_pos
    return min(riders, key=lambda r: (
        _pending_count(r),
        ((r.x - rx) ** 2 + (r.y - ry) ** 2) ** 0.5))

"""Command-line entry point."""
from src import globalvar, system
from src.order_model import Order
from src.order import find_best_rider_for_order
from src.rider import Rider


def recruit_if_needed():
    """Hire a rider when funds and demand allow. Returns True if hired."""
    riders = globalvar.get_value("riders")
    pending = sum(1 for o in globalvar.get_value("orders")
                  if o.status == "pending")
    idle = sum(1 for r in riders if r.state == 0 and not r.orders)
    funds = globalvar.get_value("funds")
    if funds >= 300 and pending > idle:
        rid = globalvar.get_value("rider_id_counter") + 1
        globalvar.set_value("rider_id_counter", rid)
        riders.append(Rider(rid))
        globalvar.set_value("funds", funds - 500)
        return True
    return False


def tick():
    """Advance simulation by one tick."""
    globalvar.set_value("time", globalvar.get_value("time") + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def cmd_start(args):
    """Create an order from four integers."""
    try:
        v = [int(a) for a in args]
    except ValueError:
        print("Coordinates must be integers.")
        return
    if len(v) != 4 or any(not 0 <= n <= 1000 for n in v):
        print("Usage: start rx ry cx cy (0-1000)")
        return
    order = Order((v[0], v[1]), (v[2], v[3]))
    globalvar.get_value("orders").append(order)
    hired = recruit_if_needed()
    rider = find_best_rider_for_order(order)
    if rider is None:
        print("No rider available.")
        return
    rider.orders.append(order)
    order.assign_to(rider.rider_id)
    print("Order %d: restaurant %s customer %s" % (
        order.id, order.res_pos, order.cus_pos))
    print("New rider added:", hired)
    print("Assigned rider %d speed %d" % (rider.rider_id, rider.speed))
    print("Estimated ticks:", -(-(abs(rider.x - order.res_pos[0])
          + abs(rider.y - order.res_pos[1]) + order.distance) // rider.speed))


def cmd_status():
    """Print system status."""
    print("Funds:", globalvar.get_value("funds"),
          "Time:", globalvar.get_value("time"))
    names = {0: "idle", 1: "picking up", 2: "delivering"}
    for r in globalvar.get_value("riders"):
        print("Rider %d at (%d,%d) %s orders=%s" % (
            r.rider_id, r.x, r.y, names[r.state], [o.id for o in r.orders]))
    grid = [["." for _ in range(16)] for _ in range(16)]
    for r in globalvar.get_value("riders"):
        grid[min(15, r.y // 50)][min(15, r.x // 50)] = "R"
    for row in grid:
        print("".join(row))


def cmd_orders():
    """Print orders."""
    for o in globalvar.get_value("orders"):
        taken = (o.end_time - o.start_time) if o.end_time is not None else "-"
        print("Order %d start=%d from=%s to=%s dist=%d time=%s rider=%s %s" % (
            o.id, o.start_time, o.res_pos, o.cus_pos, o.distance, taken,
            o.rider_id, o.status))


def cmd_riders():
    """Print rider logs."""
    for r in globalvar.get_value("riders"):
        for e in r.log:
            print("Rider %d t=%d %s pos=%s order=%s" % (
                r.rider_id, e["time"], e["event"], e["pos"], e["order_id"]))


def main():
    """Run the command loop; each command is one tick."""
    system.init_system()
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        parts = line.split()
        if not parts:
            tick()
            continue
        c = parts[0].lower()
        if c == "start":
            cmd_start(parts[1:])
        elif c == "status":
            cmd_status()
        elif c == "orders":
            cmd_orders()
        elif c == "riders":
            cmd_riders()
        elif c in ("quit", "exit"):
            globalvar.set_value("running", False)
            continue
        else:
            print("Unknown command.")
        tick()


if __name__ == "__main__":
    main()

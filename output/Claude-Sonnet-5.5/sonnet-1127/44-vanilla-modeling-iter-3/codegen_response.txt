"""Core data models: Edge and Graph."""


class Edge:
    """Undirected-edge half: source/dest are station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph keyed by station numbers."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, node):
        self.adjacency.setdefault(node, [])

    def add_edge(self, edge):
        """Add the edge in both directions."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adjacency[edge.source].append(edge)
        self.adjacency[edge.dest].append(Edge(edge.dest, edge.source, edge.distance, edge.line))

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def nodes(self):
        return list(self.adjacency.keys())

"""Priority queue and Dijkstra-based shortest path with transfer awareness."""
import heapq
import itertools

MODE_SHORTEST = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"

TRANSFER_PENALTY_COMPREHENSIVE = 1500  # meters-equivalent per transfer
TRANSFER_PENALTY_LEAST = 10_000_000


class MinPQ:
    """Heap-based minimum priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def insert(self, key, item):
        heapq.heappush(self.heap, (key, next(self._counter), item))

    def del_min(self):
        key, _, item = heapq.heappop(self.heap)
        return key, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class PathfindingConfig:
    def __init__(self, mode=MODE_SHORTEST, max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines) if preferred_lines else []


class ShortestPath:
    """Dijkstra over (station, current line) states."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}
        self._parent = {}
        self._trans = {}
        self._run()

    def _penalty(self):
        if self.config.mode == MODE_LEAST_TRANSFER:
            return TRANSFER_PENALTY_LEAST
        if self.config.mode == MODE_COMPREHENSIVE:
            return TRANSFER_PENALTY_COMPREHENSIVE
        return 0

    def _run(self):
        cfg = self.config
        penalty = self._penalty()
        start = (self.source, None)
        cost = {start: 0.0}
        dist = {start: 0}
        trans = {start: 0}
        self._parent[start] = None
        done = set()
        pq = MinPQ()
        pq.insert(0.0, start)
        best_cost = {}
        while not pq.is_empty():
            c, state = pq.del_min()
            if state in done:
                continue
            done.add(state)
            node, line = state
            if node not in best_cost or c < best_cost[node]:
                best_cost[node] = c
                self._best[node] = state
                self.dist_to[node] = dist[state]
            for e in self.graph.neighbors(node):
                nt = trans[state] + (1 if line is not None and e.line != line else 0)
                if cfg.max_transfers is not None and nt > cfg.max_transfers:
                    continue
                step = float(e.distance)
                if e.line in cfg.preferred_lines:
                    step *= 0.95
                nc = c + step + (penalty if nt > trans[state] else 0)
                ns = (e.dest, e.line)
                if ns in done:
                    continue
                if ns not in cost or nc < cost[ns]:
                    cost[ns] = nc
                    dist[ns] = dist[state] + e.distance
                    trans[ns] = nt
                    self._parent[ns] = (state, e)
                    pq.insert(nc, ns)
        self._trans = trans

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        state = self._best[target]
        nodes = [state[0]]
        edges = []
        while self._parent[state] is not None:
            prev, e = self._parent[state]
            edges.append(e)
            nodes.append(prev[0])
            state = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._best:
            return 0
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""
import math

FARE_TIERS = [(0, 6000, 3), (6000, 12000, 4), (12000, 22000, 5), (22000, 32000, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters (left-closed, right-open tiers)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    for lo, hi, price in FARE_TIERS:
        if lo <= d < hi:
            return price
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer. distance in meters."""
    return distance / 1000.0 * 1.5 + transfer_count * 5

"""Data management: build graph, integrity check, updates."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE = {
    "Line 1": [], "Line 2": [], "Line 4": [], "Line 5": [], "Line 6": [],
    "Line 8": [], "Line 10": [], "Line 13": [], "Line 14": [], "Line 15": [],
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = None
        self.station_to_number = {}
        self.number_to_station = {}
        self.build()

    def build(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        for s in self.data.get("stations", []):
            self._add_station(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in self.station_to_number:
                    self._add_station(n, [c.get("line")])
            self.graph.add_edge(Edge(self.station_to_number[c["source"]],
                                     self.station_to_number[c["dest"]],
                                     c["distance"], c.get("line")))

    def _add_station(self, name, lines):
        if name in self.station_to_number:
            return
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name
        self.graph.set_station_lines(name, lines)
        self.graph.add_node(num)

    def has_station(self, name):
        return name in self.station_to_number

    def is_transfer(self, name):
        return len(self.graph.station_lines.get(name, [])) > 1

    def check_data(self):
        """Return (ok, list of messages)."""
        msgs = []
        stations = self.data.get("stations", [])
        conns = self.data.get("connections", [])
        names = {s["name"] for s in stations}
        lines = {l for s in stations for l in s.get("lines", [])}
        for ln in REPRESENTATIVE:
            if ln not in lines:
                msgs.append("Line missing from station data: %s" % ln)
        connected = set()
        for c in conns:
            for n in (c["source"], c["dest"]):
                connected.add(n)
                if n not in names:
                    msgs.append("Connection references unknown station: %s" % n)
            if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0:
                msgs.append("Invalid distance: %s - %s" % (c["source"], c["dest"]))
            elif c["distance"] > 20000:
                msgs.append("Suspicious distance: %s - %s (%s m)" % (c["source"], c["dest"], c["distance"]))
        for n in sorted(names - connected):
            msgs.append("Station without connections: %s" % n)
        return (not msgs), msgs

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 entries")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.build()

    def update_station(self, name, lines):
        for s in self.data["stations"]:
            if s["name"] == name:
                s["lines"] = list(lines)
                self.build()
                return
        raise KeyError(name)

    def update_distance(self, a, b, distance):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                found = True
        if not found:
            raise KeyError("%s-%s" % (a, b))
        self.build()

"""Route analysis, report generation and export."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return "%d m" % meters
    return "%.2f km" % (meters / 1000.0)


def build_summary(dm, start, end, nodes, edges):
    names = [dm.number_to_station[n] for n in nodes]
    total = sum(e.distance for e in edges)
    segments = []
    transfers = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = dm.number_to_station[e.dest]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfers.append(dm.number_to_station[e.source])
            segments.append({"line": e.line, "from": dm.number_to_station[e.source],
                             "to": dm.number_to_station[e.dest], "stations": 1})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    tc = len(transfers)
    return {
        "query": {"start": start, "end": end},
        "path": names,
        "distance_m": total,
        "distance_km": round(total / 1000.0, 2),
        "fare": Sum_money(total),
        "estimated_time_min": round(calculate_estimated_time(total, tc), 1),
        "transfer_count": tc,
        "transfer_stations": transfers,
        "lines": lines,
        "segments": segments,
    }


def render(summary, fmt="text"):
    if fmt == "json":
        return json.dumps(summary, ensure_ascii=False, indent=2)
    q = summary["query"]
    out = ["Route: %s -> %s" % (q["start"], q["end"]),
           "Stations (%d): %s" % (len(summary["path"]), " -> ".join(summary["path"])),
           "Total distance: %s" % format_distance(summary["distance_m"]),
           "Lines: %s" % ", ".join(summary["lines"]),
           "Transfers: %d" % summary["transfer_count"]]
    for s in summary["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (s["line"], s["from"], s["to"], s["stations"]))
    for t in summary["transfer_stations"]:
        out.append("  Transfer at: %s" % t)
    out.append("Estimated time: %.1f minutes" % summary["estimated_time_min"])
    out.append("Fare: %d yuan" % summary["fare"])
    return "\n".join(out)


class ReportExporter:
    def __init__(self, fmt="text", output_file=None):
        self.format = fmt
        self.output_file = output_file

    def export(self, summary):
        text = render(summary, self.format)
        if self.output_file:
            with open(self.output_file, "a", encoding="utf-8") as f:
                f.write(text + "\n\n")
        return text

"""Command-line interface for the Beijing Subway fare system."""
import argparse
import sys

from data_manager import DataManager
from pathfinding import PathfindingConfig, ShortestPath
from report import ReportExporter, build_summary

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def validate(dm, text):
    """Return (start, end, error)."""
    text = text.replace("；", ";")
    parts = [x.strip() for x in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None, None, FORMAT_ERR
    s, e = parts
    if not dm.has_station(s):
        return None, None, "Starting station '%s' does not exist, please check the station name" % s
    if not dm.has_station(e):
        return None, None, "Ending station '%s' does not exist, please check the station name" % e
    if s == e:
        return None, None, "Starting station cannot be the same as the ending station"
    return s, e, None


def run_query(dm, start, end, config):
    try:
        sp = ShortestPath(dm.graph, dm.station_to_number[start], config)
        target = dm.station_to_number[end]
        nodes, edges = sp.path_with_edges(target)
    except Exception as ex:  # path calculation failure
        return None, "Path calculation failed: %s" % ex
    if not nodes:
        return None, "No route found between '%s' and '%s' under the given constraints" % (start, end)
    try:
        return build_summary(dm, start, end, nodes, edges), None
    except Exception as ex:  # fare failure
        return None, "Fare calculation failed: %s" % ex


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager()
    except Exception as ex:
        print("Failed to load data: %s" % ex)
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("Data check passed." if ok else "Data check found issues:")
        for m in msgs:
            print(" - " + m)
        return 0 if ok else 1
    config = PathfindingConfig(args.mode, args.max_transfers,
                               [args.line_preference] if args.line_preference else [])
    exporter = ReportExporter(args.format, args.output_file)
    print("Enter 'Starting Station; Ending Station' (or 'exit' to quit):")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("exit", "quit", "q"):
            break
        if not line:
            continue
        s, e, err = validate(dm, line)
        if err:
            print(err)
            continue
        summary, err = run_query(dm, s, e, config)
        if err:
            print(err)
            continue
        print(exporter.export(summary))
    return 0


if __name__ == "__main__":
    sys.exit(main())

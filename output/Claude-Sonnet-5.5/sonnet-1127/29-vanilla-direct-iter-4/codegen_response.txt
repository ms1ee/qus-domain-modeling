I had to guess at several details the Interface leaves open, so these choices are mine rather than the PRD's. Nothing has been run.

- **Modules:** every module sits at the top level of `src/`, as the Interface lists them (`globalvar`, `system`, `rider`, `order_model`, `order`). I added `pathplanner`, `display`, `inputmanager` and `main` for the remaining PRD modules.
- **Globals:** `globalvar` is a key/value store. `system.init_system()` sets funds to 1000, the rider list to empty, the rider ID counter to 0, running to True, time to 0 and the order list to empty.
- **Rider start:** a new rider spawns at (435, 385), which is on a road, and logs a "spawn" event.
- **Movement per tick:**
  - A rider with orders and state 0 moves to state 1 and logs "accepted_order".
  - In state 1 it walks toward the restaurant. On arrival it logs "arrived_at_merchant" and switches to state 2.
  - In state 2 it walks toward the customer. On arrival it logs "delivered_to_customer", sets the order to "completed", removes it from the rider's list and returns to idle.
  - If the rider is already at its target when it accepts an order, the arrival is handled in the same tick.
- **Pathfinding:** a rider off the street grid first moves to the nearest grid line, then travels along the streets, then enters the target cell centre.
  - Each tick moves along one axis at a time and never overshoots the target.
  - The pathfinder takes the rider's coordinates and a target and returns the next point.
  - Streets are the lines where x or y is a multiple of 100.
  - Restaurants and customers sit at cell centres like (50, 50), so they are not on the grid lines. The rider approaches along a street, then steps in perpendicular.
- **Order assignment:** `find_best_rider_for_order` picks the nearest rider by Euclidean distance, preferring idle riders. It returns None if there are no riders.
- **Starting an order:** the `start` command converts the four input coordinates to cell centres and creates an order. It recruits a rider if funds are at least 300 and pending orders exceed idle riders, assigns the order, and prints the route on the ASCII map. Each command advances time by one tick.
- **Other commands:** `status`, `orders`, `riders` (the event log), `tick [n]`, `help`, and `quit`/`exit`. Shortcuts: s=status, o=orders, r=riders, q=quit. Command history comes from `readline` when it is available.
- **Persistence:** a `save` command writes state to JSON.

import globalvar


def init_system():
    """Initialize core parameters."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Manage funds, riders and time."""

    @staticmethod
    def recruit_if_needed():
        """Recruit a rider if conditions hold. Returns True if added."""
        from rider import Rider
        orders = globalvar.get_value("orders", [])
        riders = globalvar.get_value("riders", [])
        pending = sum(1 for o in orders if o.status == "pending")
        idle = sum(1 for r in riders if r.state == 0 and not r.orders)
        if globalvar.get_value("funds", 0) >= 300 and pending > idle:
            rid = globalvar.get_value("rider_id_counter", 0) + 1
            globalvar.set_value("rider_id_counter", rid)
            globalvar.set_value("funds", globalvar.get_value("funds") - 300)
            riders.append(Rider(rid))
            return True
        return False

    @staticmethod
    def tick():
        """Advance one time unit."""
        globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
        for r in globalvar.get_value("riders", []):
            r.move()

"""Grid path planning."""


class PathPlanner:
    """Street-based pathfinding."""

    @staticmethod
    def _step(a, b, d):
        if abs(b - a) <= d:
            return b
        return a + d if b > a else a - d

    @staticmethod
    def next_point(x, y, tx, ty, speed):
        """Return the next position after moving up to speed units."""
        left = speed
        while left > 0 and (x, y) != (tx, ty):
            on_v = x % 100 == 0
            on_h = y % 100 == 0
            if on_v and on_h:
                pass
            if not on_v and not on_h:
                # go to nearest vertical street horizontally
                gx = round(x / 100) * 100
                if abs(gx - x) == 50:
                    gx = x - 50
                nx = PathPlanner._step(x, gx, left)
                left -= abs(nx - x)
                x = nx
                continue
            if y == ty or x == tx:
                if y == ty and x != tx:
                    nx = PathPlanner._step(x, tx, left)
                    left -= abs(nx - x)
                    x = nx
                else:
                    ny = PathPlanner._step(y, ty, left)
                    left -= abs(ny - y)
                    y = ny
                continue
            if on_v:
                # travel vertically along street toward target row street
                gy = ty if ty % 100 == 0 else (ty // 100) * 100
                if y == gy:
                    nx = PathPlanner._step(x, tx, left)
                    left -= abs(nx - x)
                    x = nx
                else:
                    ny = PathPlanner._step(y, gy, left)
                    left -= abs(ny - y)
                    y = ny
            else:
                gx = tx if tx % 100 == 0 else (tx // 100) * 100
                if x == gx:
                    ny = PathPlanner._step(y, ty, left)
                    left -= abs(ny - y)
                    y = ny
                else:
                    nx = PathPlanner._step(x, gx, left)
                    left -= abs(nx - x)
                    x = nx
        return x, y

    @staticmethod
    def route(x, y, tx, ty, speed=10):
        """Return list of points visited, tick by tick."""
        pts = [(x, y)]
        for _ in range(10000):
            if (x, y) == (tx, ty):
                break
            x, y = PathPlanner.next_point(x, y, tx, ty, speed)
            pts.append((x, y))
        return pts

"""Rider class."""
import globalvar
from pathplanner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("time", 0),
            "x": self.x,
            "y": self.y,
            "state": self.state,
            "rider_id": self.rider_id,
        })

    def _target(self):
        o = self.orders[0]
        return o.res_pos if self.state == 1 else o.cus_pos

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._record("accepted_order", self.orders[0].id)
        if not self.orders:
            self.state = 0
            return
        tx, ty = self._target()
        self.x, self.y = PathPlanner.next_point(
            self.x, self.y, tx, ty, self.speed)
        if (self.x, self.y) == (tx, ty):
            order = self.orders[0]
            if self.state == 1:
                self._record("arrived_at_merchant", order.id)
                self.state = 2
                order.status = "delivering"
            else:
                self.orders.pop(0)
                order.complete(globalvar.get_value("time", 0))
                self.state = 0
                self._record("delivered_to_customer", order.id)
                if self.orders:
                    self.state = 1
                    self._record("accepted_order", self.orders[0].id)

"""Order data model."""
import globalvar

_counter = [0]


def to_center(v):
    """Convert coordinate to grid cell centre."""
    v = max(0, min(800, int(v)))
    if v == 800:
        v = 799
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        _counter[0] += 1
        self.id = _counter[0]
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0)
        self.end_time = None
        dx = self.res_pos[0] - self.cus_pos[0]
        dy = self.res_pos[1] - self.cus_pos[1]
        self.distance = (dx * dx + dy * dy) ** 0.5

    def assign_to(self, rider_id):
        """Assign to a rider."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def complete(self, t):
        """Mark completed."""
        self.status = "completed"
        self.end_time = t

    def to_dict(self):
        return {"id": self.id, "res": list(self.res_pos),
                "cus": list(self.cus_pos), "status": self.status,
                "rider": self.rider_id, "start": self.start_time,
                "end": self.end_time, "distance": self.distance}

"""Order management."""
import math
import globalvar


def find_best_rider_for_order(order):
    """Return nearest rider (idle preferred), or None."""
    riders = globalvar.get_value("riders", [])
    if not riders:
        return None
    rx, ry = order.res_pos

    def key(r):
        d = math.hypot(r.x - rx, r.y - ry)
        return (0 if (r.state == 0 and not r.orders) else 1,
                len(r.orders), d, r.rider_id)
    # distance first among idle riders; load balance otherwise
    idle = [r for r in riders if r.state == 0 and not r.orders]
    if idle:
        return min(idle, key=lambda r: (math.hypot(r.x - rx, r.y - ry),
                                        r.rider_id))
    return min(riders, key=key)


class OrderManager:
    """Create and dispatch orders."""

    @staticmethod
    def create(coords):
        from order_model import Order
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        globalvar.get_value("orders").append(order)
        return order

    @staticmethod
    def dispatch(order):
        """Assign order to best rider. Returns (rider, recruited)."""
        from system import SystemManager
        recruited = SystemManager.recruit_if_needed()
        rider = find_best_rider_for_order(order)
        if rider is not None:
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
        return rider, recruited

"""Display manager."""
import globalvar
from pathplanner import PathPlanner


class DisplayManager:
    """Status and map rendering."""

    @staticmethod
    def status():
        lines = ["Time: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(globalvar.get_value("riders")))]
        names = {0: "idle", 1: "picking up", 2: "delivering"}
        for r in globalvar.get_value("riders"):
            lines.append("Rider %d at (%d,%d) %s, orders: %s" % (
                r.rider_id, r.x, r.y, names[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    @staticmethod
    def orders():
        os_ = globalvar.get_value("orders")
        if not os_:
            return "No orders."
        out = []
        for o in os_:
            took = (o.end_time - o.start_time) if o.end_time is not None \
                else "-"
            out.append("Order %d start=%d res=%s cus=%s dist=%.1f "
                       "time=%s rider=%s status=%s" % (
                           o.id, o.start_time, o.res_pos, o.cus_pos,
                           o.distance, took, o.rider_id, o.status))
        return "\n".join(out)

    @staticmethod
    def riders_log():
        out = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                out.append("t=%d rider=%d %s order=%s at (%d,%d)" % (
                    e["time"], r.rider_id, e["event"], e["order_id"],
                    e["x"], e["y"]))
        return "\n".join(out) if out else "No records."

    @staticmethod
    def map(route=None):
        """Render 8x8 map; each cell 1 char, streets between."""
        size = 17
        g = [["#" if (i % 2 == 0 or j % 2 == 0) else "." for j in range(size)]
             for i in range(size)]

        def cell(p):
            return (min(p[1] // 100, 7) * 2 + 1, min(p[0] // 100, 7) * 2 + 1)

        def street(p):
            return (min(round(p[1] / 50), 16), min(round(p[0] / 50), 16))
        if route:
            for p in route:
                i, j = street(p)
                g[i][j] = "*"
        for o in globalvar.get_value("orders"):
            if o.status != "completed":
                i, j = cell(o.res_pos)
                g[i][j] = "S"
                i, j = cell(o.cus_pos)
                g[i][j] = "C"
        for r in globalvar.get_value("riders"):
            i, j = street((r.x, r.y))
            g[i][j] = "R"
        return "\n".join("".join(row) for row in g)

    @staticmethod
    def route_for(rider, order):
        pts = PathPlanner.route(rider.x, rider.y, *order.res_pos, rider.speed)
        pts += PathPlanner.route(*order.res_pos, *order.cus_pos,
                                 rider.speed)[1:]
        return pts

"""Command parsing."""
import json
import globalvar
from order import OrderManager
from system import SystemManager
from display import DisplayManager

try:
    import readline  # noqa: F401
except ImportError:
    pass

ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
           "exit": "quit", "t": "tick"}


def parse_coords(text):
    """Parse four ints in 0-800. Raises ValueError."""
    cleaned = text.replace("[", " ").replace("]", " ").replace(",", " ")
    parts = cleaned.split()
    if len(parts) != 4:
        raise ValueError("Need exactly four integers.")
    try:
        vals = [int(p) for p in parts]
    except ValueError:
        raise ValueError("Coordinates must be integers.")
    for v in vals:
        if not 0 <= v <= 800:
            raise ValueError("Coordinates must be within 0-800.")
    return vals


class InputManager:
    """Handle commands."""

    def handle(self, line):
        """Process a line; return output string."""
        parts = line.strip().split(None, 1)
        if not parts:
            return ""
        cmd = ALIASES.get(parts[0].lower(), parts[0].lower())
        arg = parts[1] if len(parts) > 1 else ""
        try:
            if cmd == "quit":
                globalvar.set_value("running", False)
                return "Bye."
            if cmd == "status":
                return DisplayManager.status() + "\n" + DisplayManager.map()
            if cmd == "orders":
                return DisplayManager.orders()
            if cmd == "riders":
                return DisplayManager.riders_log()
            if cmd == "help":
                return ("Commands: start x1 y1 x2 y2, status, orders, "
                        "riders, tick [n], save [file], quit")
            if cmd == "tick":
                n = int(arg) if arg.strip() else 1
                for _ in range(max(0, n)):
                    SystemManager.tick()
                return DisplayManager.status()
            if cmd == "save":
                return self.save(arg.strip() or "state.json")
            if cmd == "start":
                if not arg:
                    arg = input("Enter res_x res_y cus_x cus_y: ")
                return self.start(parse_coords(arg))
            return "Unknown command. Type help."
        except ValueError as e:
            return "Error: %s" % e
        except Exception as e:  # recovery
            return "Unexpected error: %s" % e

    def start(self, coords):
        order = OrderManager.create(coords)
        rider, recruited = OrderManager.dispatch(order)
        out = ["Restaurant grid: %s, Customer grid: %s" % (
            (order.res_pos[0] // 100, order.res_pos[1] // 100),
            (order.cus_pos[0] // 100, order.cus_pos[1] // 100)),
            "New rider added: %s" % recruited]
        if rider is None:
            out.append("No rider available; order pending.")
        else:
            pts = DisplayManager.route_for(rider, order)
            out.append("Order %d assigned to rider %d, speed %d, "
                       "estimated time %d ticks" % (
                           order.id, rider.rider_id, rider.speed,
                           len(pts) - 1))
            out.append(DisplayManager.map(pts))
        SystemManager.tick()
        return "\n".join(out)

    def save(self, path):
        data = {
            "funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state,
                        "orders": [o.id for o in r.orders],
                        "special_case": r.special_case}
                       for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        with open(path, "w") as f:
            json.dump(data, f, indent=2)
        return "Saved to %s" % path

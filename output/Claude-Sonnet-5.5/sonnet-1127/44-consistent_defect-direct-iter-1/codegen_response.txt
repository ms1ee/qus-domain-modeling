"""Graph data structures."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted directed graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}
        self.edges = []

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, [])
        self.edges.append(edge)

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

"""Heap based priority queue."""
import heapq
import itertools


class MinPQ:
    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, key, item):
        heapq.heappush(self._h, (key, next(self._c), item))

    def pop(self):
        key, _, item = heapq.heappop(self._h)
        return key, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)

"""Dijkstra shortest path with transfer awareness."""
from min_pq import MinPQ

MODES = ("shortest_distance", "least_transfer", "comprehensive")


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._node_state = {}
        self._parent = {}
        self._run()

    def _penalty(self):
        m = self.config.mode
        if m == "least_transfer":
            return 1e7
        if m == "comprehensive":
            return 1500.0
        return 0.0

    def _run(self):
        cfg = self.config
        limit = cfg.max_transfers
        pen = self._penalty()
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        best = {start: 0.0}
        real = {start: 0}
        self._parent[start] = (None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done or cost > best.get(st, float("inf")):
                continue
            done.add(st)
            node, line, tr = st
            if node not in self._node_state:
                self._node_state[node] = st
                self.dist_to[node] = real[st]
            for e in self.graph.neighbors(node):
                is_tr = line is not None and e.line != line
                ntr = tr + (1 if is_tr else 0)
                if limit is not None and ntr > limit:
                    continue
                c = cost + e.weight * (0.9 if e.line in pref else 1.0) + (pen if is_tr else 0.0)
                ns = (e.dest, e.line, ntr if limit is not None else 0)
                if c < best.get(ns, float("inf")):
                    best[ns] = c
                    real[ns] = real[st] + e.distance
                    self._parent[ns] = (st, e)
                    pq.push(c, ns)

    def has_path_to(self, target):
        return target in self._node_state

    def path_with_edges(self, target):
        st = self._node_state.get(target)
        if st is None:
            return [], []
        nodes, edges = [], []
        while st is not None:
            nodes.append(st[0])
            p, e = self._parent[st]
            if e is not None:
                edges.append(e)
            st = p
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("invalid distance")
    d = math.ceil(distance)
    if d <= 6000:
        return 3
    if d <= 12000:
        return 4
    if d < 22000:
        return 5
    if d <= 32000:
        return 6
    return 7 + (d - 32000 - 1) // 20000


def calculate_estimated_time(distance, transfer_count):
    return distance / 1000.0 * 1.5 + transfer_count * 5.0

"""Network building, integrity checks and data updates."""
from models import Edge, Graph

REPRESENTATIVE = {
    "1号线": "天安门东", "2号线": "西直门", "4号线": "西单", "5号线": "东单",
}


class SubwayNetwork:
    def __init__(self, data):
        self.data = data
        self.build()

    def build(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        names = [s["name"] for s in self.data.get("stations", [])]
        for c in self.data.get("connections", []):
            names.append(c["source"])
            names.append(c["dest"])
        for n in names:
            if n not in self.station_to_number:
                i = len(self.station_to_number)
                self.station_to_number[n] = i
                self.number_to_station[i] = n
                self.graph.add_node(i)
        for s in self.data.get("stations", []):
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            self.graph.add_edge(Edge(self.station_to_number[c["source"]],
                                     self.station_to_number[c["dest"]],
                                     c["distance"], c["line"]))

    # ---- updates ----
    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: len(stations)-1 values (bidirectional)."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distance count mismatch")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line not in known[n]["lines"]:
                    known[n]["lines"].append(line)
            else:
                self.data["stations"].append({"name": n, "lines": [line]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line})
            self.data["connections"].append({"source": b, "dest": a, "distance": d, "line": line})
        self.build()

    def modify_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build()
                return True
        return False

    def update_distance(self, source, dest, distance, line=None):
        ok = False
        for c in self.data["connections"]:
            if c["source"] == source and c["dest"] == dest and (line is None or c["line"] == line):
                c["distance"] = distance
                ok = True
        if ok:
            self.build()
        return ok

    # ---- checks ----
    def check_data(self):
        problems = []
        names = {s["name"] for s in self.data.get("stations", [])}
        lines = set()
        for s in self.data.get("stations", []):
            lines.update(s.get("lines", []))
        report = ["Stations: %d, Connections: %d, Lines: %d" % (
            len(names), len(self.data.get("connections", [])), len(lines))]
        for ln, st in REPRESENTATIVE.items():
            if ln in lines and st not in names:
                problems.append("Representative station %s missing for %s" % (st, ln))
        connected = set()
        for c in self.data.get("connections", []):
            for k in ("source", "dest"):
                if c[k] not in names:
                    problems.append("Connection station %s not in station list" % c[k])
                connected.add(c[k])
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                problems.append("Bad distance %s->%s: %r" % (c["source"], c["dest"], d))
            if c["source"] == c["dest"]:
                problems.append("Self loop at %s" % c["source"])
        for n in sorted(names - connected):
            problems.append("Station %s has no connections" % n)
        if problems:
            report.append("Problems found (%d):" % len(problems))
            report.extend("  - " + p for p in problems[:50])
        else:
            report.append("Data check passed.")
        return not problems, "\n".join(report)

"""Beijing Subway fare calculation CLI."""
import argparse
import difflib
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import SubwayNetwork
from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_input(text):
    for sep in (";", "；", ",", "，"):
        if sep in text:
            parts = [p.strip() for p in text.split(sep)]
            if len(parts) == 2 and all(parts):
                return parts
            return None
    return None


def match_station(net, name):
    if name in net.station_to_number:
        return name
    m = difflib.get_close_matches(name, list(net.station_to_number), n=1, cutoff=0.4)
    if m:
        return m[0]
    for n in net.station_to_number:
        if name in n or n in name:
            return n
    return None


def format_distance(d):
    return "%d m" % d if d < 1000 else "%.2f km (%d m)" % (d / 1000.0, d)


def build_result(net, start, end, config):
    s, t = net.station_to_number[start], net.station_to_number[end]
    sp = ShortestPath(net.graph, s, config)
    if not sp.has_path_to(t):
        return None
    nodes, edges = sp.path_with_edges(t)
    dist = sp.dist_to[t]
    transfers = sp.get_transfer_count(t)
    try:
        fare = Sum_money(dist)
    except Exception as e:
        raise RuntimeError("Fare calculation failed: %s" % e)
    names = [net.number_to_station[n] for n in nodes]
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = names[i + 1]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(names[i])
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1], "stations": 1})
    return {
        "query": {"start": start, "end": end, "mode": config.mode},
        "path": names,
        "distance": dist,
        "fare": fare,
        "estimated_time": calculate_estimated_time(dist, transfers),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": sorted({e.line for e in edges}),
        "segments": segments,
    }


def render_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: " + format_distance(r["distance"])]
    for sg in r["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (sg["line"], sg["from"], sg["to"], sg["stations"]))
    if r["transfer_stations"]:
        out.append("Transfers (%d): %s" % (r["transfer_count"], ", ".join(r["transfer_stations"])))
    else:
        out.append("Transfers: 0")
    out.append("Lines: " + ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    if fmt == "json":
        return json.dumps(r, ensure_ascii=False, indent=2)
    return render_text(r)


def run_query(net, text, config, fmt):
    parts = parse_input(text)
    if parts is None:
        return None, FORMAT_ERR
    a, b = parts
    sa, sb = match_station(net, a), match_station(net, b)
    if sa is None:
        return None, "Starting station '%s' does not exist, please check the station name" % a
    if sb is None:
        return None, "Ending station '%s' does not exist, please check the station name" % b
    if sa == sb:
        return None, "Starting station cannot be the same as the ending station"
    try:
        r = build_result(net, sa, sb, config)
    except RuntimeError as e:
        return None, str(e)
    except Exception as e:
        return None, "Path calculation error: %s" % e
    if r is None:
        return None, "No route found between '%s' and '%s' under the current settings" % (sa, sb)
    return render(r, fmt), None


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        net = SubwayNetwork(load_subway_data(DATA_FILE_PATH))
    except Exception as e:
        print("Failed to load data: %s" % e)
        return 1
    if args.check_data:
        ok, rep = net.check_data()
        print(rep)
        return 0 if ok else 1

    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []

    print("Beijing Subway fare calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            text = input("> ").strip()
        except EOFError:
            break
        if text.lower() in ("q", "quit", "exit"):
            break
        if not text:
            continue
        res, err = run_query(net, text, config, args.format)
        if err:
            print(err)
            continue
        print(res)
        if args.output_file:
            try:
                with open(args.output_file, "w", encoding="utf-8") as f:
                    f.write(res + "\n")
                print("Saved to %s" % args.output_file)
            except OSError as e:
                print("Could not write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    name: str
    department: str
    has_won: bool = field(default=False, compare=False)


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None


def prize_level(name: str) -> int:
    """Lower number = higher prize level."""
    n = name.lower()
    for i, key in enumerate(["special", "first", "second", "third"]):
        if key in n:
            return i
    return 99

import os
from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Product Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Cheng Ru group"),
    ("Zhou Ba", "Cheng Ru group"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Li Bin", "Operations Department"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Finance Department"),
    ("Ma Jiaqi", "Finance Department"),
    ("Qian Yi", "Technology Department"),
    ("Feng Er", "Product Department"),
    ("Han Mei", "Cheng Ru group"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Cannot read file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in [("Third Prize - Movie Tickets", 3),
                          ("Second Prize - Shopping Cards", 2),
                          ("First Prize - 1 Day Annual Leave", 1)]:
            try:
                self.add_prize(name, qty)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "cheng ru group"
SPECIAL_DRAWER = "Li Bin"
CHECKED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _has_won(self, u):
        return u.has_won or any(w is u or w.name == u.name for w in self.winners)

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if not self._has_won(u)]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if u.department.lower() == CHENG_RU]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        u.has_won = True
        self.winners.append(u)
        return u

    def finish_round(self, prize, round_winners):
        if prize.associated_drawer in CHECKED_DRAWERS:
            has = any(w.department.lower() == CHENG_RU for w in round_winners)
            self.last_two_draw_results.append(has)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime
from lottery_system.models import prize_level


class Winner:
    def __init__(self, user, prize, time=None, order=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize, datetime.now(), len(self.winners) + 1)
        self.winners.append(w)
        return w

    def _sorted_prizes(self, prizes):
        return sorted(prizes, key=lambda p: prize_level(p.name))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in self._sorted_prizes(prizes):
            result[p.name] = [w for w in self.winners if w.prize.name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    c = input("Load list: 1) default 2) custom txt [1]: ").strip()
    if c == "2":
        users, errors = lm.load_custom_list(input("Path: ").strip())
        for e in errors:
            print(e)
        if errors and input("Continue with valid data? (y/n): ").lower() != "y":
            return []
    else:
        users, _ = lm.load_default_list()
    valid, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate removed: {d.name}")
    for g, n in lm.get_group_info(valid).items():
        print(f"  {g}: {n}")
    return valid


def setup_prizes(pm):
    pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        try:
            name = input("Name: ").strip()
            qty = int(input("Quantity: "))
            av = [g.strip() for g in input("Avoid groups (comma): ").split(",") if g.strip()]
            dr = input("Drawer (blank none): ").strip() or None
            pm.add_prize(name, qty, av, dr)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(i, p.name, p.quantity, p.avoid_groups, p.associated_drawer)
    o = input("New order (e.g. 3 1 2, blank keep): ").split()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o]):
                print("Invalid order, kept.")
        except ValueError:
            print("Invalid order, kept.")


def main():
    users = load_users(ListManager())
    if not users:
        print("No participants.")
        return
    pm = PrizeManager()
    setup_prizes(pm)
    lot = LotteryManager(users)
    rm = ResultManager()
    prizes = pm.get_prizes()
    for r, p in enumerate(prizes, 1):
        print(f"\n=== Round {r}: {p.name} x{p.quantity} | avoid: {p.avoid_groups} | drawer: {p.associated_drawer}")
        won = []
        i = 0
        while i < p.quantity:
            cmd = input("Enter to draw (pause/terminate): ").strip().lower()
            if cmd == "pause":
                while input("Paused. Type 'continue': ").strip().lower() != "continue":
                    pass
                continue
            if cmd == "terminate":
                print("Round terminated.")
                break
            print(f"Currently drawing the {i + 1}th winner")
            u = lot.draw_one(p)
            if u is None:
                print("No eligible participants left.")
                break
            rm.add_winner(u, p)
            won.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {p.name}")
            i += 1
        lot.finish_round(p, won)
    print("\n=== Final Results ===")
    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    ok, msg = rm.export_to_csv(prizes)
    print(msg)


if __name__ == "__main__":
    main()

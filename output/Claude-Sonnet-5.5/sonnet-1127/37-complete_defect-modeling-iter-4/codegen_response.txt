from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    name: str
    department: str
    has_won: bool = field(default=False, compare=False)

    def __hash__(self):
        return hash((self.name, self.department))


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

from collections import OrderedDict
from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Operations Department"),
    ("Ma Jiaqi", "Technology Department"),
    ("Li Bin", "Administration Department"),
    ("Liu Yi", "Cheng Ru group"),
    ("Qian Er", "Cheng Ru group"),
    ("Feng San", "Cheng Ru group"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error - '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for n, q in [("Third Prize - Movie Tickets", 3),
                     ("Second Prize - Shopping Cards", 2),
                     ("First Prize - 1 Day Annual Leave", 1)]:
            try:
                self.add_prize(n, q)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or int(quantity) <= 0:
            raise ValueError("Invalid prize name or quantity")
        self.prizes.append(Prize(name, int(quantity), list(avoid_groups or []),
                                 associated_drawer or None))

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

SPECIAL_DRAWER = "Li Bin"
MONITORED = ("Chen Qixian", "Jin Duo")
TARGET = "Ma Jiaqi"
PROTECTED = "cheng ru"


def _is_protected(user):
    return PROTECTED in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _available(self):
        return [u for u in self.users if not u.has_won and u not in self.winners]

    def _get_eligible_pool(self, prize):
        pool = self._available()
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == TARGET and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_protected(u)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        u.has_won = True
        self.winners.append(u)
        return u

    def finish_round(self, prize, round_winners):
        drawer = prize.associated_drawer
        if drawer in MONITORED:
            self.last_two_draw_results.append(
                any(_is_protected(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == TARGET:
            self.last_two_draw_results = []

    def rule_description(self, prize):
        parts = []
        if prize.associated_drawer:
            parts.append(f"Drawer: {prize.associated_drawer}")
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("Avoidance rules ignored")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        parts.append("Previous winners excluded")
        return "; ".join(parts)

import datetime


class Winner:
    def __init__(self, user, prize_name, order=0):
        self.user = user
        self.prize_name = prize_name
        self.order = order
        self.timestamp = datetime.datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name, len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        res = {}
        for p in prizes:
            res[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return res

    def export_to_csv(self, prizes):
        path = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in grouped[p.name]:
                        f.write(f"{w.user.name},{w.user.department},{w.prize_name}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("1) Default list  2) Custom txt list: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print("  !", e)
        if errs:
            a = input("Errors found. (r)emove invalid data and continue / (t)erminate: ").strip().lower()
            if a == "t":
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"  ! Duplicate name removed: {d.name}")
        if not valid:
            print("No valid users.")
            continue
        print("Groups:")
        for g, n in lm.get_group_info(valid).items():
            print(f"  {g}: {n}")
        return valid


def config_prizes(pm, groups):
    if input("Load preset prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            q = int(input("Quantity: "))
            avoid = [g.strip() for g in input(f"Avoid groups (comma sep, from {list(groups)}): ").split(",") if g.strip()]
            drawer = input("Drawer (blank for none): ").strip() or None
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    for p in pm.get_prizes():
        pass
    show_prizes(pm)
    o = input("Enter new order as serial numbers (blank keeps): ").replace(",", " ").split()
    if o and not pm.reorder_prizes(o):
        print("Invalid order, keeping default.")


def show_prizes(pm):
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} drawer={p.associated_drawer} avoid={p.avoid_groups}")


def run(lot, pm, rm):
    for r, prize in enumerate(pm.get_prizes(), 1):
        print(f"\n=== Round {r}: {prize.name} | Quantity {prize.quantity} ===")
        print("Rules:", lot.rule_description(prize))
        won = []
        i = 1
        while i <= prize.quantity:
            cmd = input("Enter to draw (pause/continue/terminate): ").strip().lower()
            if cmd == "pause":
                while input("Paused. Type 'continue': ").strip().lower() != "continue":
                    pass
                continue
            if cmd == "terminate":
                print("Round terminated.")
                break
            print(f"Currently drawing winner #{i}...")
            u = lot.draw_one(prize)
            if u is None:
                print("No eligible participants left.")
                break
            rm.add_winner(u, prize)
            won.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
            i += 1
        lot.finish_round(prize, won)
        print(f"Round {r} results:")
        for k, u in enumerate(won, 1):
            print(f"  {k}. {u.name} ({u.department})")
    print("\n=== All winners ===")
    grouped = rm.get_winners_by_prize(pm.get_prizes())
    for p, ws in grouped.items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {p}")


def main():
    lm = ListManager()
    users = load_users(lm)
    pm = PrizeManager()
    config_prizes(pm, lm.get_group_info(users).keys())
    if not pm.get_prizes():
        print("No prizes configured.")
        return
    lot = LotteryManager(users)
    rm = ResultManager()
    run(lot, pm, rm)
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(rm.export_to_csv(pm.get_prizes())[1])


if __name__ == "__main__":
    main()

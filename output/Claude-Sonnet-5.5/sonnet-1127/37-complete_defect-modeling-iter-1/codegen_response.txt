from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Li Bin", "Administration Department"),
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Liu Yi", "Cheng Ru group"),
    ("Xu Er", "Cheng Ru group"),
    ("Yang San", "Cheng Ru group"),
    ("Huang Si", "Administration Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error - '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        presets = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, q in presets:
            try:
                self.add_prize(name, q)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        quantity = int(quantity)
        if quantity <= 0:
            raise ValueError("Quantity must be positive")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

PROTECTED_GROUP = "Cheng Ru group"
CHECK_PERSONS = ("Chen Qixian", "Jin Duo")
FALLBACK_PERSON = "Ma Jiaqi"
SPECIAL_PERSON = "Li Bin"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winner_names = set()
        self.last_two_draw_results = []

    def _fallback_active(self, prize):
        return (
            prize.associated_drawer == FALLBACK_PERSON
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        )

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if u.name not in self.winner_names]
        if self._fallback_active(prize):
            return [u for u in pool if u.department == PROTECTED_GROUP]
        if prize.associated_drawer == SPECIAL_PERSON:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def describe_rules(self, prize):
        parts = ["Exclude previous winners"]
        if prize.associated_drawer == SPECIAL_PERSON:
            parts.append("avoidance groups ignored (Li Bin)")
        elif prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if self._fallback_active(prize):
            parts.append(f"restricted to {PROTECTED_GROUP} (fallback)")
        return "; ".join(parts)

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winner_names.add(u.name)
        return u

    def finish_round(self, prize, winners):
        """Record whether a checked drawer's round included protected group members."""
        if prize.associated_drawer in CHECK_PERSONS:
            has = any(w.department == PROTECTED_GROUP for w in winners)
            self.last_two_draw_results.append(has)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, order=0):
        self.user = user
        self.prize_name = prize_name
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name, len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        res = {}
        for p in prizes:
            res[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return res

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            by = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in by.get(p.name, []):
                        f.write(f"{w.user.name},{w.user.department},{w.prize_name}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    print("1. Default list  2. Custom txt list")
    if input("Choose: ").strip() == "2":
        users, errors = lm.load_custom_list(input("Path: ").strip())
        for e in errors:
            print("  !", e)
        if errors and input("Errors found. Remove invalid data and continue? (y/n): ").lower() != "y":
            return []
    else:
        users, _ = lm.load_default_list()
    users, dups = lm.validate_users(users)
    for d in dups:
        print(f"  ! Duplicate removed: {d.name}")
    print("Groups:")
    for g, c in lm.get_group_info(users).items():
        print(f"  {g}: {c}")
    return users


def config_prizes(pm, groups):
    if input("Load preset prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        try:
            name = input("Name: ").strip()
            q = int(input("Quantity: "))
            drawer = input("Lottery person (blank for none): ").strip() or None
            av = [a.strip() for a in input(f"Avoid groups, comma-separated {groups}: ").split(",") if a.strip()]
            pm.add_prize(name, q, av, drawer)
        except ValueError as e:
            print("Invalid:", e)
    ps = pm.get_prizes()
    for i, p in enumerate(ps, 1):
        print(f"{i}. {p.name} x{p.quantity} drawer={p.associated_drawer} avoid={p.avoid_groups}")
    o = input("New order (e.g. 2,1,3) or Enter to keep: ").strip()
    if o:
        ok = pm.reorder_prizes([x for x in o.replace(" ", "").split(",") if x])
        print("Reordered." if ok else "Invalid order, kept original.")


def run_round(lot, rm, prize):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | drawer {prize.associated_drawer} ===")
    print("Rules:", lot.describe_rules(prize))
    won = []
    i = 1
    while i <= prize.quantity:
        cmd = input("[Enter]=draw, pause, terminate: ").strip().lower()
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible participants remain.")
            break
        rm.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, won)
    print("Round results:", [w.name for w in won])


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No participants.")
        return
    pm = PrizeManager()
    config_prizes(pm, sorted(lm.get_group_info(users)))
    prizes = pm.get_prizes()
    lot, rm = LotteryManager(users), ResultManager()
    for p in prizes:
        run_round(lot, rm, p)
    print("\n=== All Winners ===")
    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    if input("Export results? (y/n): ").lower() == "y":
        print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()

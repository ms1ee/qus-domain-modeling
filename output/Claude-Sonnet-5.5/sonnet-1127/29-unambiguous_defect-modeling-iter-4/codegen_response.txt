"""System management: initialization, recruitment and ticking."""
import json

import globalvar

INITIAL_FUNDS = 10000
RIDER_COST = 100
MAX_RIDERS = 20


def init_system():
    """Initialise all global parameters."""
    globalvar._init()
    globalvar.set_value("funds", INITIAL_FUNDS)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("history", [])


def recruit_rider():
    """Recruit a new rider if funds allow; return it or None."""
    from rider import Rider
    funds = globalvar.get_value("funds", 0)
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    if funds < RIDER_COST or len(riders) >= MAX_RIDERS:
        return None
    rid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    globalvar.set_value("funds", funds - RIDER_COST)
    r = Rider(rid)
    riders.append(r)
    return r


def tick(n=1):
    """Advance simulation by n ticks."""
    for _ in range(n):
        globalvar.set_value("tick", globalvar.get_value("tick", 0) + 1)
        for r in list(globalvar.get_value("riders", [])):
            r.move()


def save_state(path="state.json"):
    """Persist system state as JSON."""
    data = {
        "funds": globalvar.get_value("funds"),
        "tick": globalvar.get_value("tick"),
        "riders": [{"id": r.rider_id, "x": r.x, "y": r.y, "speed": r.speed,
                    "state": r.state, "log": r.log}
                   for r in globalvar.get_value("riders", [])],
        "orders": [o.to_dict() for o in globalvar.get_value("orders", [])],
    }
    with open(path, "w") as f:
        json.dump(data, f, indent=2)

"""Rider model."""
import globalvar


class Rider:
    """A delivery rider moving on the grid."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"time": globalvar.get_value("tick", 0),
                         "x": self.x, "y": self.y, "state": self.state,
                         "event": event, "order_id": order_id})

    def _step_towards(self, tx, ty):
        """Move up to speed units along streets (x first, then y)."""
        budget = self.speed
        dx = tx - self.x
        if dx and budget:
            s = min(abs(dx), budget)
            self.x += s if dx > 0 else -s
            budget -= s
        dy = ty - self.y
        if dy and budget:
            s = min(abs(dy), budget)
            self.y += s if dy > 0 else -s

    def move(self):
        """Advance one tick, handling state transitions."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        order = self.orders[0] if self.orders else None
        if order is None:
            self.state = 0
            return
        if self.state == 1:
            tx, ty = order.res_pos
            self._step_towards(tx, ty)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
                order.status = "picked_up"
        elif self.state == 2:
            tx, ty = order.cus_pos
            self._step_towards(tx, ty)
            if (self.x, self.y) == (tx, ty):
                now = globalvar.get_value("tick", 0)
                order.status = "completed"
                order.completion_time = now
                order.time_taken = now - order.start_time
                self.orders.pop(0)
                self.state = 0
                self._log("delivered_to_customer", order.id)

"""Order data model."""
import itertools

import globalvar

_counter = itertools.count(1)


def to_center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(799, int(v)))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.completion_time = None
        self.time_taken = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""Order management: assignment logic."""
import globalvar
import system
from order_model import Order


def _dist(rider, order):
    return abs(rider.x - order.res_pos[0]) + abs(rider.y - order.res_pos[1])


def find_best_rider_for_order(order):
    """Pick the best rider: idle & nearest; recruit if none is idle."""
    riders = globalvar.get_value("riders") or []
    idle = [r for r in riders if r.state == 0 and not r.orders]
    if idle:
        return min(idle, key=lambda r: (_dist(r, order), r.rider_id))
    new = system.recruit_rider()
    if new is not None:
        return new
    if not riders:
        return None
    return min(riders, key=lambda r: (len(r.orders), _dist(r, order),
                                      r.rider_id))


def create_order(rx, ry, cx, cy):
    """Create an order, assign it and return (order, rider, recruited)."""
    before = len(globalvar.get_value("riders") or [])
    order = Order((rx, ry), (cx, cy))
    rider = find_best_rider_for_order(order)
    if rider is None:
        return order, None, False
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    globalvar.get_value("orders").append(order)
    recruited = len(globalvar.get_value("riders")) > before
    return order, rider, recruited

"""Path planning on the grid."""


class PathPlanner:
    """Manhattan (street-following) path planner."""

    @staticmethod
    def plan(start, targets):
        """Return waypoint list from start through targets (x first, then y)."""
        pts = [tuple(start)]
        cur = tuple(start)
        for t in targets:
            t = tuple(t)
            if cur[0] != t[0]:
                pts.append((t[0], cur[1]))
            if cur[1] != t[1]:
                pts.append(t)
            cur = t
        return pts

"""ASCII display helpers."""
import globalvar
from path_planner import PathPlanner

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


def _cell(v):
    return max(0, min(7, v // 100))


def render_map(rider=None, order=None):
    """Render 8x8 map. R rider, S restaurant, C customer, . empty, * route."""
    grid = [["." for _ in range(8)] for _ in range(8)]
    if rider is not None and order is not None:
        pts = PathPlanner.plan((rider.x, rider.y),
                               [order.res_pos, order.cus_pos])
        for a, b in zip(pts, pts[1:]):
            steps = max(abs(a[0] - b[0]), abs(a[1] - b[1])) // 50
            for i in range(steps + 1):
                x = a[0] + (b[0] - a[0]) * i // max(steps, 1)
                y = a[1] + (b[1] - a[1]) * i // max(steps, 1)
                grid[_cell(y)][_cell(x)] = "*"
    for o in globalvar.get_value("orders", []):
        if o.status != "completed":
            if o.status in ("pending", "assigned"):
                grid[_cell(o.res_pos[1])][_cell(o.res_pos[0])] = "S"
            grid[_cell(o.cus_pos[1])][_cell(o.cus_pos[0])] = "C"
    for r in globalvar.get_value("riders", []):
        grid[_cell(r.y)][_cell(r.x)] = "R"
    return "\n".join(" ".join(row) for row in reversed(grid))


def render_status():
    lines = ["Tick: %d  Funds: %d  Riders: %d" % (
        globalvar.get_value("tick", 0), globalvar.get_value("funds", 0),
        len(globalvar.get_value("riders", [])))]
    for r in globalvar.get_value("riders", []):
        lines.append("Rider %d at (%d,%d) %s speed=%d orders=%s" % (
            r.rider_id, r.x, r.y, STATE_NAMES[r.state], r.speed,
            [o.id for o in r.orders]))
    return "\n".join(lines)


def render_orders():
    orders = globalvar.get_value("orders", [])
    if not orders:
        return "No orders."
    return "\n".join(
        "Order %d start=%d rest=%s cust=%s dist=%d time=%s rider=%s [%s]" % (
            o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
            o.time_taken, o.rider_id, o.status) for o in orders)


def render_riders():
    lines = []
    for r in globalvar.get_value("riders", []):
        for e in r.log:
            lines.append("t=%d rider %d (%d,%d) %s order=%s" % (
                e["time"], r.rider_id, e["x"], e["y"], e["event"],
                e["order_id"]))
    return "\n".join(lines) or "No records."

"""Command line entry point."""
import system
import display
import order as order_mod


def parse_start(text):
    """Parse four ints in 0-800; raise ValueError on bad input."""
    parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
    if len(parts) != 4:
        raise ValueError("need exactly four integers")
    vals = [int(p) for p in parts]
    if any(v < 0 or v > 800 for v in vals):
        raise ValueError("coordinates must be within 0-800")
    return vals


def main():
    system.init_system()
    import globalvar
    history = []
    aliases = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
               "t": "tick"}
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if not line:
            continue
        history.append(line)
        cmd, _, rest = line.partition(" ")
        cmd = aliases.get(cmd.lower(), cmd.lower())
        try:
            if cmd == "start":
                if not rest:
                    rest = input("restaurant_x restaurant_y customer_x "
                                 "customer_y: ")
                v = parse_start(rest)
                o, r, new = order_mod.create_order(*v)
                if r is None:
                    print("No rider available.")
                    continue
                print("Restaurant %s Customer %s new rider: %s" % (
                    o.res_pos, o.cus_pos, new))
                print(display.render_map(r, o))
                total = (abs(r.x - o.res_pos[0]) + abs(r.y - o.res_pos[1])
                         + o.distance)
                print("Rider %d speed %d est. time %d ticks" % (
                    r.rider_id, r.speed, -(-total // r.speed)))
            elif cmd == "tick":
                system.tick(int(rest) if rest else 1)
                print(display.render_map())
            elif cmd == "status":
                print(display.render_status())
                print(display.render_map())
            elif cmd == "orders":
                print(display.render_orders())
            elif cmd == "riders":
                print(display.render_riders())
            elif cmd == "history":
                print("\n".join(history))
            elif cmd in ("quit", "exit"):
                globalvar.set_value("running", False)
            else:
                print("Unknown command. Use start/tick/status/orders/"
                      "riders/quit.")
        except ValueError as e:
            print("Error: %s" % e)


if __name__ == "__main__":
    main()

class User:
    def __init__(self, name, department):
        self.name = name
        self.department = department
        self.has_won = False

    def __eq__(self, other):
        return (isinstance(other, User) and self.name == other.name
                and self.department == other.department)

    def __hash__(self):
        return hash((self.name, self.department))

    def __repr__(self):
        return f"User({self.name!r}, {self.department!r})"


class Prize:
    def __init__(self, name, quantity, avoid_groups=None, associated_drawer=None):
        self.name = name
        self.quantity = quantity
        self.avoid_groups = list(avoid_groups) if avoid_groups else []
        self.associated_drawer = associated_drawer

    def __repr__(self):
        return f"Prize({self.name!r}, {self.quantity})"

import os
from collections import OrderedDict
from lottery_system.models import User

DEFAULT_DATA = [
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Operations Department"),
    ("Li Bin", "Administration Department"),
    ("Zhang Wei", "Technology Department"),
    ("Wang Fang", "Technology Department"),
    ("Liu Yang", "Technology Department"),
    ("Zhao Lei", "Product Department"),
    ("Sun Li", "Product Department"),
    ("Zhou Min", "Product Department"),
    ("Wu Jing", "Operations Department"),
    ("Zheng Hao", "Operations Department"),
    ("Cheng Ru", "Cheng Ru Group"),
    ("Cheng Yu", "Cheng Ru Group"),
    ("Cheng Xin", "Cheng Ru Group"),
    ("Xu Ting", "Administration Department"),
    ("Hu Jun", "Administration Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        if not os.path.isfile(path):
            return [], [f"File Error: cannot read file '{path}'"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: {e}"]
        for i, line in enumerate(lines, 1):
            s = line.strip()
            if not s or s.startswith("#"):
                continue
            s = s.replace("，", ",")
            parts = [p.strip() for p in s.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        presets = [
            ("Third Prize - Movie Tickets", 3, [], None),
            ("Second Prize - Shopping Cards", 2, [], None),
            ("First Prize - 1 Day Annual Leave", 1, [], None),
        ]
        for name, q, av, dr in presets:
            try:
                self.add_prize(name, q, av, dr)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name.strip(), quantity, avoid_groups or [], associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            ok = len(order) == n and sorted(order) == list(range(1, n + 1))
        except TypeError:
            return False
        if not ok:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "Cheng Ru Group"
LI_BIN = "Li Bin"
MA = "Ma Jiaqi"
TRACKED = ("Chen Qixian", "Jin Duo")


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = set()
        self.last_two_draw_results = []

    def _is_cheng_ru(self, user):
        return user.department.strip().lower() == CHENG_RU.lower()

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users
                if u not in self.winners and u.name not in {w.name for w in self.winners}]
        drawer = prize.associated_drawer
        if drawer == MA and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if self._is_cheng_ru(u)]
        if drawer == LI_BIN:
            return pool  # relaxed: avoidance groups ignored
        avoid = {g.strip().lower() for g in prize.avoid_groups}
        return [u for u in pool if u.department.strip().lower() not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.add(u)
        u.has_won = True
        return u

    def record_round(self, prize, round_winners):
        """Track whether tracked drawers' results contained Cheng Ru members."""
        if prize.associated_drawer in TRACKED:
            self.last_two_draw_results.append(
                any(self._is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime


class Winner:
    def __init__(self, user, prize_name, order=0):
        self.user = user
        self.prize_name = prize_name
        self.order = order
        self.time = datetime.datetime.now()


def _level(name):
    low = name.lower()
    for i, k in enumerate(("special", "first", "second", "third")):
        if low.startswith(k):
            return i
    return 99


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize.name, len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        ordered = sorted(prizes, key=lambda p: _level(p.name))
        result = {}
        for p in ordered:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for pname, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{pname}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt file: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        users, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name: {d.name} ({d.department})")
        if errs or dups:
            a = input("Remove invalid data and continue? (y = remove, n = terminate upload): ").strip().lower()
            if a != "y":
                continue
        if users:
            return users
        print("No valid users.")


def show_groups(lm, users):
    for g, n in lm.get_group_info(users).items():
        print(f"  {g}: {n}")


def configure_prizes(pm, users, lm):
    if input("Load preset prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Prize name: ")
        try:
            q = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma separated, blank none): ").split(",") if g.strip()]
            dr = input("Associated drawer (blank none): ").strip() or None
            pm.add_prize(name, q, avoid, dr)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    s = input("New order as numbers (blank keep): ").split()
    if s:
        try:
            ok = pm.reorder_prizes([int(x) for x in s])
        except ValueError:
            ok = False
        print("Reordered." if ok else "Invalid order, keeping original.")


def describe(p):
    d = f"Avoid groups: {', '.join(p.avoid_groups) or 'none'}"
    if p.associated_drawer:
        d += f"; Drawer: {p.associated_drawer}"
    return d


def run():
    lm_list = ListManager()
    users = load_users(lm_list)
    print("Groups:")
    show_groups(lm_list, users)
    pm = PrizeManager()
    configure_prizes(pm, users, lm_list)
    prizes = pm.get_prizes()
    lot = LotteryManager(users)
    rm = ResultManager()
    for idx, p in enumerate(prizes, 1):
        print(f"\n=== Round {idx}: {p.name} (quantity {p.quantity}) | {describe(p)} ===")
        round_winners = []
        n = 0
        while n < p.quantity:
            cmd = input("Enter to draw / pause / terminate: ").strip().lower()
            if cmd == "pause":
                print("Paused.")
                while input("Type 'continue' to resume: ").strip().lower() != "continue":
                    pass
                continue
            if cmd == "terminate":
                print("Round terminated.")
                break
            print(f"Currently drawing the {n + 1}th winner...")
            u = lot.draw_one(p)
            if u is None:
                print("No eligible participants left.")
                break
            n += 1
            rm.add_winner(u, p)
            round_winners.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {p.name}")
        lot.record_round(p, round_winners)
        print(f"Round {idx} results:")
        for i, u in enumerate(round_winners, 1):
            print(f"  {i}. {u.name} ({u.department})")
    print("\n=== Final Winner List ===")
    for pname, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {pname}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    run()

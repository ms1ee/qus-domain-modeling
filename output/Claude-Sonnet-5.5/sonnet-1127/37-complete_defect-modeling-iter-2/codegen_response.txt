class User:
    def __init__(self, name, department):
        self.name = name
        self.department = department
        self.has_won = False

    def __repr__(self):
        return f"User({self.name!r}, {self.department!r})"

    def __eq__(self, other):
        return (isinstance(other, User) and self.name == other.name
                and self.department == other.department)

    def __hash__(self):
        return hash((self.name, self.department))


class Prize:
    def __init__(self, name, quantity, avoid_groups=None, associated_drawer=None):
        self.name = name
        self.quantity = quantity
        self.avoid_groups = list(avoid_groups) if avoid_groups else []
        self.associated_drawer = associated_drawer

    def __repr__(self):
        return f"Prize({self.name!r}, {self.quantity})"

    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Operations Department"),

    ("Cheng Ru B", "Cheng Ru Group"),
    ("Cheng Ru C", "Cheng Ru Group"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error reading file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, q in defaults:
            try:
                self.add_prize(name, q)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, avoid_groups or [], associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

PROTECTED_GROUP = "Cheng Ru Group"
MONITORED_DRAWERS = ("Chen Qixian", "Jin Duo")
TARGET_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _is_protected(self, dept):
        return dept.strip().lower() == PROTECTED_GROUP.lower()

    def _fallback_active(self, prize):
        return (prize.associated_drawer == TARGET_DRAWER
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize):
        won = {w.name for w in self.winners}
        pool = [u for u in self.users if u.name not in won]
        if self._fallback_active(prize):
            return [u for u in pool if self._is_protected(u.department)]
        if prize.associated_drawer == SPECIAL_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def describe_rules(self, prize):
        parts = ["Exclude previous winners"]
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("Special drawer: avoidance groups ignored")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        if self._fallback_active(prize):
            parts.append(f"Fallback: pool limited to {PROTECTED_GROUP}")
        return "; ".join(parts)

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        u.has_won = True
        self.winners.append(u)
        return u

    def finish_round(self, prize, round_winners):
        if prize.associated_drawer in MONITORED_DRAWERS:
            self.last_two_draw_results.append(
                any(self._is_protected(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime


class Winner:
    def __init__(self, user, prize):
        self.user = user
        self.prize = prize


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize is p
                              or getattr(w.prize, "name", w.prize) == p.name]
        return result

    def export_to_csv(self, prizes):
        fname = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(fname, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in grouped[p.name]:
                        f.write(f"{w.user.name},{w.user.department},{p.name}\n")
            return True, f"Results successfully exported to {fname}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    c = input("Load list: 1) default 2) custom txt [1]: ").strip()
    if c == "2":
        users, errors = lm.load_custom_list(input("Path: ").strip())
        for e in errors:
            print(e)
        if errors and input("Invalid data found. Remove invalid lines and continue? (y/n): ").lower() != "y":
            return []
    else:
        users, _ = lm.load_default_list()
    valid, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate name removed: {d.name}")
    return valid


def configure_prizes(pm, groups):
    if input("Load preset prizes? (y/n) [y]: ").strip().lower() != "n":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n) [n]: ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            q = int(input("Quantity: "))
            av = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            dr = input("Drawer (optional): ").strip() or None
            pm.add_prize(name, q, av, dr)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    o = input("New order (space separated numbers, blank to keep): ").split()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o]):
                print("Invalid order, keeping default.")
        except ValueError:
            print("Invalid order, keeping default.")


def run_round(lot, res, prize, num):
    print(f"\n=== Round {num}: {prize.name} (x{prize.quantity}) ===")
    print("Rules:", lot.describe_rules(prize))
    won = []
    i = 0
    paused = False
    while i < prize.quantity:
        cmd = input("Enter to draw (pause/continue/terminate): ").strip().lower()
        if cmd == "pause":
            paused = True
            print("Paused.")
            continue
        if cmd == "continue":
            paused = False
            continue
        if cmd == "terminate":
            print("Round terminated.")
            break
        if paused:
            print("Paused. Type 'continue' to resume.")
            continue
        print(f"Currently drawing the {i + 1}th winner")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible participants left.")
            break
        res.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, won)


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No participants.")
        return
    for g, c in lm.get_group_info(users).items():
        print(f"{g}: {c}")
    pm = PrizeManager()
    configure_prizes(pm, lm.get_group_info(users))
    lot = LotteryManager(users)
    res = ResultManager()
    for n, p in enumerate(pm.get_prizes(), 1):
        run_round(lot, res, p, n)
    print("\n=== Final Winners ===")
    for p in pm.get_prizes():
        for w in res.get_winners_by_prize(pm.get_prizes())[p.name]:
            print(f"{w.user.name}, {w.user.department}, {p.name}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(res.export_to_csv(pm.get_prizes())[1])


if __name__ == "__main__":
    main()

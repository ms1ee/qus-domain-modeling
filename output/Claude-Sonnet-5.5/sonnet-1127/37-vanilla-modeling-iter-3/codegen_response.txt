from typing import List, Tuple, Dict

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Product Department"),
    ("Zhao Liu", "Product Department"),
    ("Chen Qixian", "Market Department"),
    ("Jin Duo", "Market Department"),
    ("Ma Jiaqi", "Operations Department"),
    ("Li Bin", "Operations Department"),
    ("Sun Qi", "Cheng Ru Group"),
    ("Zhou Ba", "Cheng Ru Group"),
    ("Wu Jiu", "Cheng Ru Group"),
    ("Zheng Shi", "Technology Department"),
    ("Qian Yi", "Product Department"),
    ("Feng Er", "Market Department"),
]


class ListManager:
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error: cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Format Error: Line {i}: '{line}' should be 'Name,Department Group'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        dups: List[User] = []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from typing import List, Optional

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        presets = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, qty in presets:
            try:
                self.add_prize(name, qty)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or quantity is None or int(quantity) <= 0:
            raise ValueError("Invalid prize name or quantity")
        p = Prize(name, int(quantity), list(avoid_groups or []), associated_drawer or None)
        self.prizes.append(p)
        return p

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random
from typing import List

from lottery_system.models import User, Prize

PROTECTED_GROUP = "cheng ru"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


def _is_protected(dept: str) -> bool:
    return PROTECTED_GROUP in dept.lower()


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        won = {w.name for w in self.winners}
        pool = [u for u in self.users if not u.has_won and u.name not in won]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if drawer == FALLBACK_DRAWER and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if _is_protected(u.department)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def rule_description(self, prize: Prize) -> str:
        parts = ["Exclude previous winners"]
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts = ["Exclude previous winners only (avoidance groups ignored for Li Bin)"]
        elif prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer == FALLBACK_DRAWER:
            parts.append("fallback: Cheng Ru group protection may apply")
        return "; ".join(parts)

    def draw_one(self, prize: Prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        u.has_won = True
        self.winners.append(u)
        return u

    def finish_round(self, prize: Prize, winners: List[User]) -> None:
        if prize.associated_drawer in WATCHED_DRAWERS:
            self.last_two_draw_results.append(
                any(_is_protected(w.department) for w in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime
from typing import Dict, List


class Winner:
    def __init__(self, user, prize, time=None, order=0):
        self.user = user
        self.prize = prize
        self.time = time or datetime.now()
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize):
        w = Winner(user, prize, datetime.now(), len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize.name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def setup_list(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt file > ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        if errs:
            a = input("Errors found. (r)emove invalid and continue / (t)erminate: ").strip().lower()
            if a == "t":
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name}")
        if valid:
            return valid
        print("No valid users.")


def setup_prizes(pm, groups):
    if input("Load preset prizes? (y/n) ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n) ").strip().lower() == "y":
        name = input("Prize name: ").strip()
        try:
            qty = int(input("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        drawer = input("Lottery person (blank for none): ").strip() or None
        print("Groups:", ", ".join(groups))
        av = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
        try:
            pm.add_prize(name, qty, av, drawer)
        except ValueError as e:
            print(e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    o = input("New order (e.g. 3 1 2), blank to keep: ").split()
    if o and not pm.reorder_prizes(o):
        print("Invalid order, keeping original.")


def run_round(idx, prize, lot, res):
    print(f"\n=== Round {idx}: {prize.name} | Quantity: {prize.quantity} | "
          f"Drawer: {prize.associated_drawer or '-'} ===")
    print("Rules:", lot.rule_description(prize))
    won = []
    for n in range(1, prize.quantity + 1):
        while True:
            cmd = input("Enter to draw (pause / terminate): ").strip().lower()
            if cmd == "pause":
                while input("Paused. Type 'continue': ").strip().lower() != "continue":
                    pass
                continue
            break
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing the {n}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible participants left.")
            break
        res.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.finish_round(prize, won)


def main():
    lm = ListManager()
    users = setup_list(lm)
    for g, c in lm.get_group_info(users).items():
        print(f"{g}: {c}")
    pm = PrizeManager()
    setup_prizes(pm, list(lm.get_group_info(users)))
    lot = LotteryManager(users)
    res = ResultManager()
    for i, p in enumerate(pm.get_prizes(), 1):
        run_round(i, p, lot, res)
    print("\n=== Final Results ===")
    def level(p):
        n = p.name.lower()
        for k, name in enumerate(("first", "second", "third")):
            if n.startswith(name):
                return k
        return 3
    ordered = sorted(pm.get_prizes(), key=level)
    for name, ws in res.get_winners_by_prize(ordered).items():
        for w in ws:
            print(f"{name}: {w.user.name} ({w.user.department})")
    if input("Export results? (y/n) ").strip().lower() == "y":
        print(res.export_to_csv(ordered)[1])


if __name__ == "__main__":
    main()

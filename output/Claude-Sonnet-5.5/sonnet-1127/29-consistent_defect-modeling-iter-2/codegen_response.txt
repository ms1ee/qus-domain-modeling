MAX_COORD = 800


class PathPlanner:
    """Plans right-angle paths along streets."""

    @staticmethod
    def next_position(x, y, tx, ty, speed):
        """Move up to `speed` units from (x, y) toward (tx, ty).

        Moves horizontally first, then vertically (right-angle turns).
        """
        budget = speed
        if x != tx:
            step = min(budget, abs(tx - x))
            x += step if tx > x else -step
            budget -= step
        if budget > 0 and y != ty:
            step = min(budget, abs(ty - y))
            y += step if ty > y else -step
        return x, y

    @staticmethod
    def plan_route(points):
        """Return waypoints connecting a list of points with right angles."""
        route = []
        for i, p in enumerate(points):
            if i == 0:
                route.append(p)
                continue
            prev = route[-1]
            if prev[0] != p[0] and prev[1] != p[1]:
                route.append((p[0], prev[1]))
            route.append(p)
        return route

    @staticmethod
    def route_length(points):
        """Manhattan length of a route through points."""
        return sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                   for a, b in zip(points, points[1:]))

"""Rider model."""

import globalvar
from pathplanner import PathPlanner

IDLE, PICKING_UP, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._accepted = set()
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("tick", 0),
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def current_order(self):
        """First order not yet completed, or None."""
        for o in self.orders:
            if getattr(o, "status", None) != "completed":
                return o
        return None

    def pending_count(self):
        """Number of uncompleted orders."""
        return sum(1 for o in self.orders
                   if getattr(o, "status", None) != "completed")

    def move(self):
        """Advance one tick."""
        order = self.current_order()
        if order is None:
            self.state = IDLE
            return
        if self.state == IDLE:
            self.state = PICKING_UP
        if order.id not in self._accepted:
            self._accepted.add(order.id)
            self._record("accepted_order", order.id)
        if self.state == PICKING_UP:
            tx, ty = order.res_pos
            self.x, self.y = PathPlanner.next_position(
                self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                self.state = DELIVERING
                self._record("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            tx, ty = order.cus_pos
            self.x, self.y = PathPlanner.next_position(
                self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                order.status = "completed"
                now = globalvar.get_value("tick", 0)
                order.completion_time = now
                order.time_taken = now - order.start_time
                self.state = PICKING_UP if self.current_order() else IDLE
                self._record("delivered_to_customer", order.id)

"""Order model."""

import math
import globalvar


def to_cell(v):
    """Convert coordinate to top-left corner of its grid cell."""
    v = max(0, min(int(v), 800))
    return min(v // 100 * 100, 700)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        n = globalvar.get_value("order_id_counter", 0) + 1
        globalvar.set_value("order_id_counter", n)
        self.id = n
        self.res_pos = (to_cell(res_pos[0]), to_cell(res_pos[1]))
        self.cus_pos = (to_cell(cus_pos[0]), to_cell(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0)
        self.completion_time = None
        self.time_taken = None
        self.distance = math.hypot(self.res_pos[0] - self.cus_pos[0],
                                   self.res_pos[1] - self.cus_pos[1])

    def assign_to(self, rider_id):
        """Mark as assigned to a rider."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""Order management and assignment."""

import math
import globalvar
from order_model import Order


def find_best_rider_for_order(order):
    """Rider with fewest pending orders, ties broken by distance."""
    riders = globalvar.get_value("riders", [])
    if not riders:
        return None
    rx, ry = order.res_pos
    return min(riders, key=lambda r: (
        r.pending_count(), math.hypot(r.x - rx, r.y - ry), r.rider_id))


class OrderManager:
    """Creates, assigns, and tracks orders."""

    def create_order(self, rx, ry, cx, cy):
        order = Order((rx, ry), (cx, cy))
        globalvar.get_value("orders").append(order)
        return order

    def assign(self, order):
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider

    def pending_orders(self):
        return [o for o in globalvar.get_value("orders", [])
                if o.status == "pending"]

"""System management."""

import json
import globalvar
from rider import Rider
from order import OrderManager

RECRUIT_COST = 500
RECRUIT_MIN_FUNDS = 300


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)


class SystemManager:
    """Core manager for funds, riders and ticks."""

    def __init__(self):
        init_system()
        self.order_manager = OrderManager()

    def recruit_if_needed(self):
        """Hire a rider when conditions are met. Returns the Rider or None."""
        funds = globalvar.get_value("funds")
        riders = globalvar.get_value("riders")
        pending = len(self.order_manager.pending_orders())
        idle = sum(1 for r in riders if r.state == 0 and not r.current_order())
        if funds >= RECRUIT_MIN_FUNDS and pending > idle:
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            globalvar.set_value("funds", funds - RECRUIT_COST)
            rider = Rider(rid)
            riders.append(rider)
            return rider
        return None

    def submit_order(self, rx, ry, cx, cy):
        """Create order, maybe recruit, assign. Returns (order, rider, new)."""
        order = self.order_manager.create_order(rx, ry, cx, cy)
        new = self.recruit_if_needed()
        rider = self.order_manager.assign(order)
        if rider is not None:
            rider.log.append({
                "event": "order_received", "order_id": order.id,
                "time": globalvar.get_value("tick"),
                "x": rider.x, "y": rider.y, "state": rider.state})
        return order, rider, new

    def tick(self):
        globalvar.set_value("tick", globalvar.get_value("tick") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

    def save(self, path="state.json"):
        data = {
            "funds": globalvar.get_value("funds"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "state": r.state, "log": r.log}
                       for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Display manager."""

import globalvar
from pathplanner import PathPlanner

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text rendering."""

    @staticmethod
    def render_map(riders=(), orders=(), route=None):
        g = [["." for _ in range(16)] for _ in range(16)]
        for i in range(16):
            for j in range(16):
                if i % 2 == 1 or j % 2 == 1:
                    g[i][j] = "#"

        def cell(p):
            return min(p[1] // 50, 15), min(p[0] // 50, 15)

        if route:
            for a, b in zip(route, route[1:]):
                (r1, c1), (r2, c2) = cell(a), cell(b)
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        g[r][c] = "*"
        for o in orders:
            if o.status != "completed":
                r, c = cell(o.res_pos)
                g[r][c] = "S"
                r, c = cell(o.cus_pos)
                g[r][c] = "C"
        for rd in riders:
            r, c = cell((rd.x, rd.y))
            g[r][c] = "R"
        return "\n".join("".join(row) for row in g)

    def show_status(self):
        riders = globalvar.get_value("riders")
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) %s, orders: %s" % (
                r.rider_id, r.x, r.y, STATE_NAMES[r.state],
                [(o.id, o.status) for o in r.orders]))
        lines.append(self.render_map(riders, globalvar.get_value("orders")))
        return "\n".join(lines)

    @staticmethod
    def show_orders():
        orders = globalvar.get_value("orders")
        if not orders:
            return "No orders."
        return "\n".join(
            "Order %d: start=%d restaurant=%s customer=%s dist=%.1f "
            "time=%s rider=%s status=%s" % (
                o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                o.time_taken, o.rider_id, o.status) for o in orders)

    @staticmethod
    def show_riders():
        lines = []
        for r in globalvar.get_value("riders"):
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%s (%s,%s) %s order=%s" % (
                    e.get("time"), e.get("x"), e.get("y"),
                    e["event"], e["order_id"]))
        return "\n".join(lines) or "No riders."

    def show_route(self, rider, order):
        pts = [(rider.x, rider.y), order.res_pos, order.cus_pos]
        route = PathPlanner.plan_route(pts)
        length = PathPlanner.route_length(pts)
        t = -(-length // rider.speed)
        return "%s\nSpeed: %d units/tick, length %d, estimated time %d ticks" % (
            self.render_map([rider], [order], route), rider.speed, length, t)

class InputManager:
    """Parses commands."""

    COMMANDS = {"start": "start", "s": "start", "status": "status",
                "orders": "orders", "o": "orders", "riders": "riders",
                "r": "riders", "quit": "quit", "exit": "quit", "q": "quit"}

    def __init__(self):
        self.history = []

    def parse_command(self, text):
        text = text.strip().lower()
        if text:
            self.history.append(text)
        return self.COMMANDS.get(text)

    @staticmethod
    def parse_coordinates(text):
        """Parse four integers in 0..1000; raises ValueError."""
        parts = text.replace(",", " ").replace("[", " ").replace(
            "]", " ").split()
        if len(parts) != 4:
            raise ValueError("Expected 4 integers")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        if any(v < 0 or v > 1000 for v in vals):
            raise ValueError("Coordinates must be within 0-1000")
        return vals

"""Command line entry point."""

import globalvar
from system import SystemManager
from display import DisplayManager
from inputmanager import InputManager


def main():
    sysm = SystemManager()
    disp = DisplayManager()
    inp = InputManager()
    print("Commands: start, status, orders, riders, quit")
    while globalvar.get_value("running"):
        try:
            cmd = inp.parse_command(input("> "))
        except (EOFError, KeyboardInterrupt):
            break
        try:
            if cmd == "quit":
                break
            elif cmd == "start":
                vals = inp.parse_coordinates(
                    input("rx ry cx cy: "))
                order, rider, new = sysm.submit_order(*vals)
                print("Restaurant:", order.res_pos, "Customer:", order.cus_pos)
                print("New rider added:", bool(new))
                if rider:
                    print(disp.show_route(rider, order))
            elif cmd == "status":
                print(disp.show_status())
            elif cmd == "orders":
                print(disp.show_orders())
            elif cmd == "riders":
                print(disp.show_riders())
            else:
                print("Unknown command")
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
        sysm.tick()
    try:
        sysm.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

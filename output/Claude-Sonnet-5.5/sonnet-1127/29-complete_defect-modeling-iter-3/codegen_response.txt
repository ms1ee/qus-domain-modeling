"""System initialization and management."""
import globalvar

RECRUIT_COST = 300
DELIVERY_FEE = 100


def init_system():
    """Initialize all global state."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", 0)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("grid_size", 8)
    globalvar.set_value("cell_size", 100)


def add_rider():
    """Create a new rider with an auto-incremented id."""
    from rider import Rider
    rid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    r = Rider(rid)
    globalvar.get_value("riders").append(r)
    return r


def check_recruit():
    """Recruit a rider if funds >= 300. Returns the new rider or None."""
    funds = globalvar.get_value("funds", 0)
    if funds >= RECRUIT_COST:
        globalvar.set_value("funds", funds - RECRUIT_COST)
        return add_rider()
    return None


def tick():
    """Advance the simulation one tick."""
    globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
    for r in list(globalvar.get_value("riders")):
        r.move()

"""Grid-based path planning."""


class PathPlanner:
    """Plans right-angle paths on the grid."""

    def __init__(self, grid_size=8, cell_size=100):
        self.grid_size = grid_size
        self.cell_size = cell_size

    def next_step(self, x, y, tx, ty, speed):
        """Return the next position moving horizontally first, then vertically."""
        if x != tx:
            d = min(speed, abs(tx - x))
            return x + (d if tx > x else -d), y
        if y != ty:
            d = min(speed, abs(ty - y))
            return x, y + (d if ty > y else -d)
        return x, y

    def path(self, points, speed=100):
        """Return list of positions through successive points."""
        out = [points[0]]
        x, y = points[0]
        for tx, ty in points[1:]:
            while (x, y) != (tx, ty):
                x, y = self.next_step(x, y, tx, ty, speed)
                out.append((x, y))
        return out

"""Order data model."""
import globalvar

_counter = [0]


def to_center(v):
    """Convert a coordinate to its grid cell center."""
    v = max(0, min(int(v), 799))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        _counter[0] += 1
        self.id = _counter[0]
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.start_time = globalvar.get_value("time", 0) or 0
        self.completion_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))
        self.time_taken = None
        self.rider_id = None
        self.status = "pending"

    def assign_to(self, rider_id):
        """Assign to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

"""Rider model."""
import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 100
        self.state = 0
        self.orders = []
        self.special_case = False
        self.planner = PathPlanner()
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0) or 0,
                         "x": self.x, "y": self.y})

    def _step(self, target):
        self.x, self.y = self.planner.next_step(
            self.x, self.y, target[0], target[1], self.speed)
        return (self.x, self.y) == tuple(target)

    def move(self):
        """Advance one tick."""
        if not self.orders:
            return
        order = self.orders[0]
        if self.state == 0:
            self.state = 1
            self._log("accepted_order", order.id)
            if order.status != "completed":
                order.status = "picking_up"
        if self.state == 1:
            if self._step(order.res_pos):
                self.state = 2
                order.status = "delivering"
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            if self._step(order.cus_pos):
                now = globalvar.get_value("time", 0) or 0
                order.status = "completed"
                order.completion_time = now
                order.time_taken = now - order.start_time
                self._log("delivered_to_customer", order.id)
                self.orders.pop(0)
                self.state = 0
                funds = globalvar.get_value("funds", 0) or 0
                globalvar.set_value("funds", funds + 100)

import globalvar
from order_model import Order


def find_best_rider_for_order(order):
    """Return the closest rider (idle preferred, then least loaded)."""
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    if not riders:
        import system
        system.add_rider()
        riders = globalvar.get_value("riders")

    def key(r):
        d = abs(r.x - order.res_pos[0]) + abs(r.y - order.res_pos[1])
        return (len(r.orders), d, r.rider_id)

    return min(riders, key=key)


def create_and_assign(res, cus):
    """Create an order and assign it. Returns (order, rider)."""
    order = Order(res, cus)
    globalvar.get_value("orders").append(order)
    rider = find_best_rider_for_order(order)
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    return order, rider

"""Display helpers."""
import globalvar

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


def render_map(riders=(), orders=(), route=()):
    """Render an 8x8 ASCII map."""
    g = [["." for _ in range(8)] for _ in range(8)]
    for p in route:
        cx, cy = min(p[0] // 100, 7), min(p[1] // 100, 7)
        g[cy][cx] = "#"
    for o in orders:
        g[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
        g[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
    for r in riders:
        g[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
    return "\n".join(" ".join(row) for row in reversed(g))


def status_text():
    """Return system status text."""
    riders = globalvar.get_value("riders")
    lines = ["Time: %s  Funds: %s  Riders: %d" % (
        globalvar.get_value("time"), globalvar.get_value("funds"),
        len(riders))]
    for r in riders:
        lines.append("Rider %d at (%d,%d) %s orders=%s" % (
            r.rider_id, r.x, r.y, STATES[r.state],
            [o.id for o in r.orders]))
    return "\n".join(lines)


def orders_text():
    """Return order details."""
    lines = []
    for o in globalvar.get_value("orders"):
        lines.append("Order %d start=%s res=%s cus=%s dist=%s time=%s rider=%s %s"
                     % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                        o.time_taken, o.rider_id, o.status))
    return "\n".join(lines) or "No orders."


def riders_text():
    """Return rider event logs."""
    lines = []
    for r in globalvar.get_value("riders"):
        for e in r.log:
            lines.append("Rider %d t=%s (%s,%s) %s order=%s" % (
                r.rider_id, e["time"], e["x"], e["y"], e["event"],
                e["order_id"]))
    return "\n".join(lines) or "No riders."

"""Command-line entry point."""
import json
import system
import globalvar
import order as order_mod
import display


def parse_coords(text):
    """Parse four integers in 0-800."""
    parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
    if len(parts) != 4:
        raise ValueError("Enter exactly four integers.")
    try:
        vals = [int(p) for p in parts]
    except ValueError:
        raise ValueError("Coordinates must be integers.")
    if any(v < 0 or v > 800 for v in vals):
        raise ValueError("Coordinates must be within 0-800.")
    return vals


def save(path="state.json"):
    """Persist state to JSON."""
    data = {"funds": globalvar.get_value("funds"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "state": r.state} for r in globalvar.get_value("riders")],
            "orders": [{"id": o.id, "res": o.res_pos, "cus": o.cus_pos,
                        "status": o.status, "rider": o.rider_id}
                       for o in globalvar.get_value("orders")]}
    try:
        with open(path, "w") as f:
            json.dump(data, f)
    except OSError:
        pass


def do_start():
    """Handle order input."""
    vals = parse_coords(input("Enter rx ry cx cy: "))
    before = len(globalvar.get_value("riders"))
    order, rider = order_mod.create_and_assign(vals[:2], vals[2:])
    added = len(globalvar.get_value("riders")) > before
    print("Restaurant grid:", order.res_pos[0] // 100, order.res_pos[1] // 100,
          "Customer grid:", order.cus_pos[0] // 100, order.cus_pos[1] // 100)
    print("New rider added:", added)
    route = rider.planner.path([(rider.x, rider.y), order.res_pos,
                                order.cus_pos], rider.speed)
    for _ in range(1000):
        if order.status == "completed":
            break
        system.tick()
    system.check_recruit()
    print(display.render_map([rider], [order], route))
    print("Rider %d speed=%d time=%s" % (rider.rider_id, rider.speed,
                                         order.time_taken))


def main():
    """Run the main loop."""
    system.init_system()
    history = []
    while globalvar.get_value("running"):
        try:
            cmd = input("> ").strip().lower()
        except EOFError:
            break
        if not cmd:
            continue
        history.append(cmd)
        try:
            if cmd in ("quit", "exit", "q"):
                globalvar.set_value("running", False)
            elif cmd in ("start", "s"):
                do_start()
            elif cmd == "status":
                print(display.status_text())
            elif cmd in ("orders", "o"):
                print(display.orders_text())
            elif cmd in ("riders", "r"):
                print(display.riders_text())
            elif cmd == "history":
                print("\n".join(history))
            else:
                print("Unknown command.")
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
    save()


if __name__ == "__main__":
    main()

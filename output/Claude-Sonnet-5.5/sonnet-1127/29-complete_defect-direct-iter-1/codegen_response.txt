"""Grid based path planning."""

GRID = 100
SIZE = 800


def to_center(v):
    """Convert a coordinate to its grid cell centre."""
    v = max(0, min(int(v), SIZE))
    return min(v // GRID, 7) * GRID + 50


class PathPlanner:
    """Simple right-angle path planner."""

    @staticmethod
    def route(start, end):
        """Waypoints from start to end (horizontal then vertical)."""
        pts = [tuple(start)]
        corner = (end[0], start[1])
        if corner != pts[-1] and corner != tuple(end):
            pts.append(corner)
        if tuple(end) != pts[-1]:
            pts.append(tuple(end))
        return pts

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def step(pos, target, speed):
        """Advance up to speed units toward target; returns new (x, y)."""
        x, y = pos
        left = speed
        dx = target[0] - x
        if dx and left > 0:
            m = min(abs(dx), left)
            x += m if dx > 0 else -m
            left -= m
        dy = target[1] - y
        if dy and left > 0:
            m = min(abs(dy), left)
            y += m if dy > 0 else -m
        return x, y

"""Order data model."""

import globalvar
from path_planner import to_center, PathPlanner


class Order:
    """A delivery order."""

    _next_id = 1

    def __init__(self, res_pos, cus_pos):
        self.id = Order._next_id
        Order._next_id += 1
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.complete_time = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)

    def assign_to(self, rider_id):
        """Mark order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, now):
        """Mark as completed."""
        self.status = "completed"
        self.complete_time = now

    @property
    def duration(self):
        """Ticks taken, or None."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.start_time

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "complete_time": self.complete_time,
                "distance": self.distance}

"""Rider model."""

import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id, x=50, y=50, speed=50):
        self.rider_id = rider_id
        self.x = x
        self.y = y
        self.speed = speed
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0) or 0,
                         "x": self.x, "y": self.y})

    @property
    def pos(self):
        return (self.x, self.y)

    def move(self):
        """Advance one tick, handling state transitions."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._record("accepted_order", self.orders[0].id)
        order = self.orders[0]
        target = order.res_pos if self.state == 1 else order.cus_pos
        # special case: aligned on a vertical/horizontal line with target
        self.special_case = (self.x == target[0] or self.y == target[1])
        self.x, self.y = PathPlanner.step(self.pos, target, self.speed)
        if self.pos != tuple(target):
            return
        if self.state == 1:
            self._record("arrived_at_merchant", order.id)
            self.state = 2
        else:
            now = globalvar.get_value("time", 0) or 0
            order.complete(now)
            self._record("delivered_to_customer", order.id)
            self.orders.pop(0)
            self.state = 0
            self.special_case = False

"""Order management."""

import globalvar
from order_model import Order
from path_planner import PathPlanner


def find_best_rider_for_order(order):
    """Return the closest rider (ties: lighter load, lower id) or None."""
    riders = globalvar.get_value("riders", []) or []
    best = None
    best_key = None
    for r in riders:
        key = (PathPlanner.distance(r.pos, order.res_pos),
               len(r.orders), r.rider_id)
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


class OrderManager:
    """Creates and assigns orders."""

    def create_order(self, coords):
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        return order

    def assign(self, order):
        """Assign order to best rider; returns rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider

    def assign_pending(self):
        for o in globalvar.get_value("orders", []) or []:
            if o.status == "pending":
                self.assign(o)

"""System management."""

import json
import globalvar
from rider import Rider

RIDER_COST = 300
ORDER_FEE = 100
INITIAL_FUNDS = 300


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", INITIAL_FUNDS)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Manages funds, riders and time."""

    def check_recruit(self):
        """Recruit riders while funds >= 300. Returns number added."""
        added = 0
        while globalvar.get_value("funds", 0) >= RIDER_COST:
            globalvar.set_value("funds",
                                globalvar.get_value("funds") - RIDER_COST)
            cid = globalvar.get_value("rider_id_counter", 0) + 1
            globalvar.set_value("rider_id_counter", cid)
            globalvar.get_value("riders").append(Rider(cid))
            added += 1
        return added

    def add_funds(self, amount):
        globalvar.set_value("funds", globalvar.get_value("funds", 0) + amount)

    def tick(self):
        """Advance one time unit."""
        globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
        for r in globalvar.get_value("riders", []):
            r.move()

    def busy(self):
        return any(r.orders for r in globalvar.get_value("riders", []))

    def save(self, path):
        data = {
            "funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "riders": [{"rider_id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state,
                        "special_case": r.special_case,
                        "orders": [o.id for o in r.orders]}
                       for r in globalvar.get_value("riders", [])],
            "orders": [o.to_dict()
                       for o in globalvar.get_value("orders", [])],
        }
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

"""Display manager."""

import globalvar
from path_planner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text output helpers."""

    def _canvas(self):
        n = 17
        c = [["#"] * n for _ in range(n)]
        for i in range(1, n, 2):
            for j in range(1, n, 2):
                c[i][j] = "."
        return c

    @staticmethod
    def _idx(v):
        return max(0, min(16, int(round(v / 50.0))))

    def render_map(self, route=None):
        c = self._canvas()
        if route:
            for a, b in zip(route, route[1:]):
                x0, y0, x1, y1 = (self._idx(a[0]), self._idx(a[1]),
                                  self._idx(b[0]), self._idx(b[1]))
                for x in range(min(x0, x1), max(x0, x1) + 1):
                    for y in range(min(y0, y1), max(y0, y1) + 1):
                        c[y][x] = "-" if y0 == y1 else "|"
        for o in globalvar.get_value("orders", []) or []:
            if o.status == "completed":
                continue
            if o.status in ("pending", "assigned"):
                c[self._idx(o.res_pos[1])][self._idx(o.res_pos[0])] = "S"
            c[self._idx(o.cus_pos[1])][self._idx(o.cus_pos[0])] = "C"
        for r in globalvar.get_value("riders", []) or []:
            c[self._idx(r.y)][self._idx(r.x)] = "R"
        return "\n".join("".join(row) for row in c)

    def route_info(self, rider, order):
        pts = [rider.pos] + PathPlanner.route(rider.pos, order.res_pos)[1:]
        pts += PathPlanner.route(order.res_pos, order.cus_pos)[1:]
        dist = sum(PathPlanner.distance(a, b) for a, b in zip(pts, pts[1:]))
        t = -(-dist // rider.speed) if rider.speed else 0
        return pts, dist, t

    def status(self):
        lines = ["Time: %s  Funds: %s  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(globalvar.get_value("riders", [])))]
        for r in globalvar.get_value("riders", []):
            lines.append("Rider %d at (%d,%d) speed=%d state=%s orders=%s" % (
                r.rider_id, r.x, r.y, r.speed, STATES[r.state],
                [o.id for o in r.orders]))
        lines.append(self.render_map())
        return "\n".join(lines)

    def orders(self):
        os_ = globalvar.get_value("orders", [])
        if not os_:
            return "No orders."
        lines = []
        for o in os_:
            d = o.duration
            lines.append("Order %d start=%s restaurant=%s customer=%s "
                         "distance=%d time=%s rider=%s status=%s" % (
                             o.id, o.start_time, o.res_pos, o.cus_pos,
                             o.distance, d if d is not None else "-",
                             o.rider_id if o.rider_id else "-", o.status))
        return "\n".join(lines)

    def riders(self):
        lines = []
        for r in globalvar.get_value("riders", []):
            for e in r.log:
                lines.append("Rider %d t=%s %s order=%s at (%d,%d)" % (
                    r.rider_id, e["time"], e["event"], e["order_id"],
                    e["x"], e["y"]))
        return "\n".join(lines) or "No riders."

"""Input parsing and validation."""

import re

ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
           "exit": "quit", "h": "help", "?": "help", "t": "tick",
           "st": "start"}


class InputManager:
    """Parses commands and coordinates."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        parts = line.strip().split(None, 1)
        if not parts:
            return None, ""
        self.history.append(line.strip())
        cmd = parts[0].lower()
        return ALIASES.get(cmd, cmd), parts[1] if len(parts) > 1 else ""

    @staticmethod
    def parse_coords(text):
        """Return four ints in 0..800 or raise ValueError."""
        tokens = [t for t in re.split(r"[\s,\[\]]+", text) if t]
        if len(tokens) != 4:
            raise ValueError("Expected 4 integers: rx ry cx cy")
        vals = []
        for t in tokens:
            try:
                vals.append(int(t))
            except ValueError:
                raise ValueError("'%s' is not an integer" % t)
        for v in vals:
            if not 0 <= v <= 800:
                raise ValueError("Coordinate %d out of range 0-800" % v)
        return vals

"""Command line entry point."""

import logging
import globalvar
from system import init_system, SystemManager, ORDER_FEE
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager

try:
    import readline  # noqa: F401
except ImportError:
    pass

HELP = ("Commands: start [rx ry cx cy] (st), status (s), orders (o), "
        "riders (r), tick [n] (t), run, save [file], help (h), quit (q)")


def handle_start(args, sm, om, dm, im):
    if not args.strip():
        args = input("Enter restaurant_x restaurant_y customer_x "
                     "customer_y: ")
    coords = im.parse_coords(args)
    sm.add_funds(ORDER_FEE)
    added = sm.check_recruit()
    order = om.create_order(coords)
    rider = om.assign(order)
    print("Restaurant grid: (%d,%d)  Customer grid: (%d,%d)" % (
        order.res_pos[0] // 100, order.res_pos[1] // 100,
        order.cus_pos[0] // 100, order.cus_pos[1] // 100))
    print("New rider added: %s" % ("yes" if added else "no"))
    if rider is None:
        print("No rider available; order %d is pending." % order.id)
        return
    pts, dist, t = dm.route_info(rider, order)
    print("Order %d assigned to rider %d (speed %d, distance %d, "
          "estimated time %d ticks)" % (order.id, rider.rider_id,
                                        rider.speed, dist, t))
    print(dm.render_map(pts))


def main():
    logging.basicConfig(level=logging.WARNING)
    init_system()
    sm, om, dm, im = SystemManager(), OrderManager(), DisplayManager(), \
        InputManager()
    print(HELP)
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            cmd, args = im.parse_command(line)
            if cmd is None:
                continue
            if cmd == "quit":
                globalvar.set_value("running", False)
            elif cmd == "help":
                print(HELP)
            elif cmd == "start":
                handle_start(args, sm, om, dm, im)
            elif cmd == "status":
                print(dm.status())
            elif cmd == "orders":
                print(dm.orders())
            elif cmd == "riders":
                print(dm.riders())
            elif cmd == "tick":
                n = int(args) if args.strip() else 1
                for _ in range(max(0, n)):
                    om.assign_pending()
                    sm.tick()
                print(dm.status())
            elif cmd == "run":
                guard = 0
                while guard < 10000:
                    om.assign_pending()
                    if not sm.busy():
                        break
                    sm.tick()
                    guard += 1
                print(dm.status())
            elif cmd == "save":
                sm.save(args.strip() or "state.json")
                print("Saved.")
            else:
                print("Unknown command '%s'. Type help." % cmd)
        except ValueError as e:
            print("Error: %s" % e)
        except Exception as e:  # recover gracefully
            logging.exception("unexpected error")
            print("Unexpected error: %s" % e)
    print("Bye.")


if __name__ == "__main__":
    main()

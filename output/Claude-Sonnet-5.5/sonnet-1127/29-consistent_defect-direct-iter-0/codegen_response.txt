I had to resolve several contradictions in the PRD, and I did not run any of this code. The choices below are my own reading of it.

- **Rider states:** I used 0 = idle, 1 = picking up, 2 = delivering, as in the data model. Rider speed is fixed at 20 units/tick, and a new rider starts at (400, 400).
- **Order coordinates:** An input coordinate becomes `(v // 100) * 100`, which is the top-left corner of its grid cell, and is clamped to 0–700. The PRD also mentions cell centres (+50), which contradicts this. Corners are on the 100-unit grid lines and so are reachable by riders moving along streets.
- **Rider movement:** A rider starts on the road at (400, 400) and moves along x first, then y. Since the rider is on a street line when x is a multiple of 100, this is simple Manhattan movement. Each tick it advances by at most `speed`.
- **Pickup then delivery:** After a rider reaches the restaurant it logs `arrived_at_merchant` and switches to delivering. Reaching the customer logs `delivered_to_customer`, marks the order "completed" and keeps it in `rider.orders`.
- **Queued orders:** If the rider has more orders waiting, it goes back to picking up. Otherwise it goes idle.
- **Order assignment:** `find_best_rider_for_order` picks the rider with the fewest unfinished orders and breaks ties by distance.
- **Initial state:** `move()` on an idle rider picks up the first non-completed order. It logs `accepted_order` and sets the state to picking up.
- **Recruitment:** `start` creates a rider when funds are at least 300 and pending orders exceed idle riders. If there are no riders at all, the order cannot be assigned without one, so recruitment is forced in that case. It costs 500, so funds can go negative.
- **Persistence:** State is saved to `state.json`.

    globalvar._init()
    globalvar.set_value("funds", 1000)

    globalvar.set_value("rider_id_counter", 0)

class SystemManager:
    """Manage funds, riders and system state."""

    def recruit_if_needed(self, force=False):
        """Recruit a rider when conditions are met. Returns new rider or None."""
        from rider import Rider
        funds = globalvar.get_value("funds")
        riders = globalvar.get_value("riders")
        pending = sum(1 for o in globalvar.get_value("orders")
                      if o.status != "completed")
        idle = sum(1 for r in riders if r.state == 0)
        if force or (funds >= 300 and pending > idle):
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            r = Rider(rid)
            riders.append(r)
            globalvar.set_value("funds", funds - 500)
            return r
        return None

"""Order data model."""
import globalvar


def to_cell(v):
    """Convert a coordinate to its grid cell top-left corner (0-700)."""
    return max(0, min(700, (int(v) // 100) * 100))


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        n = (globalvar.get_value("order_counter") or 0) + 1
        globalvar.set_value("order_counter", n)
        self.id = n
        self.res_pos = [to_cell(res_pos[0]), to_cell(res_pos[1])]
        self.cus_pos = [to_cell(cus_pos[0]), to_cell(cus_pos[1])]
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time") or 0
        self.end_time = None
        self.picked = False
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))
        orders = globalvar.get_value("orders")
        if orders is not None:
            orders.append(self)

    def assign_to(self, rider_id):
        """Assign order to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res": self.res_pos, "cus": self.cus_pos,
                "status": self.status, "rider": self.rider_id,
                "start": self.start_time, "end": self.end_time,
                "distance": self.distance}

"""Rider class."""
import globalvar


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time") or 0,
                         "pos": (self.x, self.y)})

    def current_order(self):
        """First unfinished order."""
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def _step_toward(self, tx, ty):
        budget = self.speed
        dx = tx - self.x
        step = min(abs(dx), budget)
        self.x += step if dx > 0 else -step
        budget -= step
        dy = ty - self.y
        step = min(abs(dy), budget)
        self.y += step if dy > 0 else -step

    def move(self):
        """Advance one tick."""
        order = self.current_order()
        if self.state == 0:
            if order is None:
                return
            self.state = 1
            order.status = "picking_up"
            self._record("accepted_order", order.id)
        if order is None:
            self.state = 0
            return
        if self.state == 1:
            tx, ty = order.res_pos
            self._step_toward(tx, ty)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                order.status = "delivering"
                order.picked = True
                self._record("arrived_at_merchant", order.id)
        elif self.state == 2:
            tx, ty = order.cus_pos
            self._step_toward(tx, ty)
            if (self.x, self.y) == (tx, ty):
                order.status = "completed"
                order.end_time = globalvar.get_value("time") or 0
                self._record("delivered_to_customer", order.id)
                nxt = self.current_order()
                if nxt is not None:
                    self.state = 1
                    nxt.status = "picking_up"
                    self._record("accepted_order", nxt.id)
                else:
                    self.state = 0

"""Order management and assignment."""
import globalvar
from order_model import Order


def _pending(r):
    return sum(1 for o in r.orders if o.status != "completed")


def find_best_rider_for_order(order):
    """Pick rider with fewest pending orders, ties by distance."""
    riders = globalvar.get_value("riders")
    if not riders:
        return None
    rx, ry = order.res_pos
    return min(riders, key=lambda r: (
        _pending(r), ((r.x - rx) ** 2 + (r.y - ry) ** 2) ** 0.5))


class OrderManager:
    """Create and assign orders."""

    def create_order(self, res, cus):
        """Create an order, recruit if needed, and assign it."""
        from system import SystemManager
        order = Order(res, cus)
        new = SystemManager().recruit_if_needed(
            force=not globalvar.get_value("riders"))
        rider = find_best_rider_for_order(order)
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        return order, rider, new is not None

"""Path planning."""


class PathPlanner:
    """Grid path planner: horizontal first, then vertical."""

    def plan(self, start, points):
        """Return corner waypoints from start through points."""
        path = [tuple(start)]
        cur = tuple(start)
        for p in points:
            p = tuple(p)
            if cur[0] != p[0] and cur[1] != p[1]:
                path.append((p[0], cur[1]))
            if path[-1] != p:
                path.append(p)
            cur = p
        return path

"""Display manager."""
import globalvar

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status display and ASCII map."""

    def status(self):
        riders = globalvar.get_value("riders")
        lines = ["Funds: %d" % globalvar.get_value("funds"),
                 "Time: %d" % globalvar.get_value("time"),
                 "Riders: %d" % len(riders)]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) %s, orders: %s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [(o.id, o.status) for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        lines = []
        for o in globalvar.get_value("orders"):
            t = (o.end_time - o.start_time) if o.end_time is not None else "-"
            lines.append("Order %d start=%d res=%s cus=%s dist=%d time=%s "
                         "rider=%s status=%s" % (
                             o.id, o.start_time, o.res_pos, o.cus_pos,
                             o.distance, t, o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    def riders_log(self):
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("Rider %d t=%d pos=%s %s order=%s" % (
                    r.rider_id, e["time"], e["pos"], e["event"],
                    e["order_id"]))
        return "\n".join(lines) or "No riders."

    def map(self, path=None, riders=None, orders=None):
        """16x16 ASCII map; each cell is 50 units."""
        n = 16
        g = [["." for _ in range(n)] for _ in range(n)]
        for i in range(n):
            for j in range(n):
                if i % 2 == 0 or j % 2 == 0:
                    g[i][j] = "#"
        def cell(x, y):
            return min(n - 1, y // 50), min(n - 1, x // 50)
        if path:
            for a, b in zip(path, path[1:]):
                for t in range(0, 101, 10):
                    x = a[0] + (b[0] - a[0]) * t // 100
                    y = a[1] + (b[1] - a[1]) * t // 100
                    i, j = cell(x, y)
                    g[i][j] = "*"
        for o in orders or globalvar.get_value("orders"):
            if o.status == "completed":
                continue
            i, j = cell(*o.res_pos)
            g[i][j] = "S"
            i, j = cell(*o.cus_pos)
            g[i][j] = "C"
        for r in riders if riders is not None else globalvar.get_value("riders"):
            i, j = cell(r.x, r.y)
            g[i][j] = "R"
        return "\n".join("".join(row) for row in g)

"""Input parsing and validation."""


class InputManager:
    """Parse commands."""

    def parse_coords(self, text):
        parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(parts) != 4:
            raise ValueError("Enter four integers: rx ry cx cy")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        for v in vals:
            if not 0 <= v <= 1000:
                raise ValueError("Coordinates must be within 0-1000")
        return vals

"""Command-line entry point."""
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import globalvar
from system import init_system
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager
from pathplanner import PathPlanner


def save_state(path="state.json"):
    data = {"funds": globalvar.get_value("funds"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "state": r.state} for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
    try:
        with open(path, "w") as f:
            json.dump(data, f, indent=2)
    except OSError:
        pass


def tick():
    globalvar.set_value("time", globalvar.get_value("time") + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def main():
    init_system()
    disp, inp, om = DisplayManager(), InputManager(), OrderManager()
    history = []
    print("Commands: start, status, orders, riders, tick [n], history, quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        if not line:
            continue
        history.append(line)
        cmd, _, rest = line.partition(" ")
        cmd = cmd.lower()
        try:
            if cmd in ("quit", "exit", "q"):
                globalvar.set_value("running", False)
            elif cmd == "start":
                if not rest:
                    rest = input("Enter rx ry cx cy: ")
                vals = inp.parse_coords(rest)
                order, rider, new = om.create_order(vals[:2], vals[2:])
                print("Restaurant cell:", order.res_pos,
                      "Customer cell:", order.cus_pos)
                print("New rider added:", new)
                path = PathPlanner().plan(
                    (rider.x, rider.y), [order.res_pos, order.cus_pos])
                total = sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                            for a, b in zip(path, path[1:]))
                print("Speed %d, est. time %d ticks" % (
                    rider.speed, -(-total // rider.speed)))
                print(disp.map(path=path))
                tick()
            elif cmd == "tick":
                for _ in range(int(rest) if rest else 1):
                    tick()
                print("Time:", globalvar.get_value("time"))
            elif cmd == "status":
                print(disp.status())
                print(disp.map())
            elif cmd == "orders":
                print(disp.orders())
            elif cmd == "riders":
                print(disp.riders_log())
            elif cmd == "history":
                print("\n".join(history))
            else:
                print("Unknown command")
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
        save_state()


if __name__ == "__main__":
    main()

"""Core data models: Edge and Graph."""


class Edge:
    """Undirected-edge record; source/dest are station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists."""

    def __init__(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = {}
        self.station_lines = {}

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.graph.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.graph.setdefault(edge.source, []).append(edge)
        self.graph.setdefault(edge.dest, []).append(edge.reversed())

    def neighbors(self, number):
        return self.graph.get(number, [])

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

    def node_count(self):
        return len(self.graph)

"""Dijkstra shortest path with a heap-based priority queue."""
import heapq
import itertools

TRANSFER_PENALTY_LEAST = 1000000
TRANSFER_PENALTY_COMPREHENSIVE = 1000


class MinPQ:
    """Min priority queue on a binary heap."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    @property
    def size(self):
        return len(self._heap)

    def is_empty(self):
        return not self._heap


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    """Single-source search; states are (station, line) to track transfers."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._labels = {}  # state -> (cost, dist, transfers, prev_state, edge)
        self._run()

    def _penalty(self):
        m = self.config.mode
        if m == "least_transfer":
            return TRANSFER_PENALTY_LEAST
        if m == "comprehensive":
            return TRANSFER_PENALTY_COMPREHENSIVE
        return 0

    def _preferred(self, line):
        for p in self.config.preferred_lines:
            if p and (p == line or p.casefold() in str(line).casefold()):
                return True
        return False

    def _edge_cost(self, edge):
        c = edge.weight
        if self.config.preferred_lines and self._preferred(edge.line):
            c *= 0.9
        return c

    def _run(self):
        start = (self.source, None)
        self._labels[start] = (0, 0, 0, None, None)
        pq = MinPQ()
        pq.push(0, start)
        done = set()
        pen = self._penalty()
        maxt = self.config.max_transfers
        while not pq.is_empty():
            cost, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line = state
            _, dist, trans, _, _ = self._labels[state]
            for e in self.graph.neighbors(node):
                nt = trans
                add = self._edge_cost(e)
                if line is not None and e.line != line:
                    nt += 1
                    add += pen
                    if maxt is not None and nt > maxt:
                        continue
                ns = (e.dest, e.line)
                if ns in done:
                    continue
                nc = cost + add
                old = self._labels.get(ns)
                if old is None or nc < old[0]:
                    self._labels[ns] = (nc, dist + e.weight, nt, state, e)
                    pq.push(nc, ns)
        for state, lab in self._labels.items():
            n = state[0]
            b = self._best_state.get(n)
            if b is None or lab[0] < self._labels[b][0]:
                self._best_state[n] = state
        for n, st in self._best_state.items():
            self.dist_to[n] = self._labels[st][1]

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        st = self._best_state.get(target)
        if st is None:
            return [], []
        edges = []
        while st is not None:
            lab = self._labels[st]
            if lab[4] is not None:
                edges.append(lab[4])
            st = lab[3]
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                count += 1
        return count

"""Fare and time estimation."""
import math

FARE_TIERS = [(0, 6, 3), (6, 12, 4), (12, 22, 5), (22, 32, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters (km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0 - 1e-9)
    for lo, hi, fare in FARE_TIERS:
        if km <= hi:
            return fare
    return 6 + int(math.ceil((km - 32) / 20.0))


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: ~35 km/h average speed plus 5 min per transfer."""
    return round(distance / 1000.0 / 35.0 * 60.0 + 5.0 * transfer_count, 1)

"""Data management: building the graph, integrity check, updates."""
import json

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REQUIRED_LINES = ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
                  "Line 13", "Line 14", "Line 15", "Changping", "Yizhuang", "Fangshan", "Airport"]
ALT_NAMES = {"Line 1": ["1号线"], "Line 2": ["2号线"], "Line 4": ["4号线"], "Line 5": ["5号线"],
             "Line 6": ["6号线"], "Line 8": ["8号线"], "Line 10": ["10号线"], "Line 13": ["13号线"],
             "Line 14": ["14号线"], "Line 15": ["15号线"], "Changping": ["昌平"],
             "Yizhuang": ["亦庄"], "Fangshan": ["房山"], "Airport": ["机场"]}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = None
        self.build_graph()

    def build_graph(self):
        g = Graph()
        for i, st in enumerate(self.data.get("stations", [])):
            g.add_station(st["name"], i)
            g.set_station_lines(st["name"], st.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in g.station_to_number:
                    g.add_station(n, len(g.station_to_number))
                    g.set_station_lines(n, [c["line"]])
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
        self.graph = g
        return g

    def line_names(self):
        names = set()
        for st in self.data.get("stations", []):
            names.update(st.get("lines", []))
        for c in self.data.get("connections", []):
            names.add(c["line"])
        return names

    def check_data(self):
        """Return list of problem strings (empty if data is fine)."""
        problems = []
        names = self.line_names()
        for req in REQUIRED_LINES:
            keys = [req.casefold()] + [a.casefold() for a in ALT_NAMES.get(req, [])]
            if not any(any(k in n.casefold() for k in keys) for n in names):
                problems.append("Missing line: %s" % req)
        known = {s["name"] for s in self.data.get("stations", [])}
        connected = set()
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                connected.add(n)
                if n not in known:
                    problems.append("Connection references unknown station: %s" % n)
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
                problems.append("Bad distance %r between %s and %s" % (d, c["source"], c["dest"]))
        for n in sorted(known - connected):
            problems.append("Station without connection: %s" % n)
        return problems

    def save(self):
        with open(self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)

    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        existing = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in existing:
                if line_name not in existing[n]["lines"]:
                    existing[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.build_graph()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build_graph()
                return True
        return False

    def update_distance(self, a, b, distance):
        ok = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                ok = True
        if ok:
            self.build_graph()
        return ok

"""CLI for the Beijing Subway fare calculation system."""
import argparse
import json
import sys

from data_manager import DataManager
from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_input(text):
    text = text.replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        raise ValueError(FORMAT_ERR)
    return parts[0].strip(), parts[1].strip()


def fmt_distance(m):
    return "%d m" % m if m < 1000 else "%.2f km" % (m / 1000.0)


def build_result(dm, start, end, config):
    g = dm.graph
    s, t = g.station_to_number[start], g.station_to_number[end]
    sp = ShortestPath(g, s, config)
    if t not in sp.dist_to:
        raise RuntimeError("No route found (check --max_transfers setting)")
    nodes, edges = sp.path_with_edges(t)
    stations = [g.number_to_station[n] for n in nodes]
    dist = sp.dist_to[t]
    transfers = []
    segments = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = stations[i + 1]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfers.append({"station": stations[i], "from_line": segments[-1]["line"],
                                  "to_line": e.line})
            segments.append({"line": e.line, "from": stations[i], "to": stations[i + 1], "stations": 1})
    tc = sp.get_transfer_count(t)
    try:
        fare = Sum_money(dist)
    except Exception as ex:
        raise RuntimeError("Fare calculation failed: %s" % ex)
    return {
        "query": {"start": start, "end": end, "mode": config.mode,
                  "max_transfers": config.max_transfers,
                  "line_preference": config.preferred_lines},
        "path": stations,
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": calculate_estimated_time(dist, tc),
        "transfer_count": tc,
        "transfers": transfers,
        "segments": segments,
        "lines": list(dict.fromkeys(e.line for e in edges)),
    }


def format_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: %s" % fmt_distance(r["distance_m"]),
           "Lines: %s" % ", ".join(r["lines"])]
    for sg in r["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (sg["line"], sg["from"], sg["to"], sg["stations"]))
    out.append("Transfers: %d" % r["transfer_count"])
    for t in r["transfers"]:
        out.append("  Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    out.append("Estimated time: %s minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    return json.dumps(r, ensure_ascii=False, indent=2) if fmt == "json" else format_text(r)


def run_query(dm, start, end, config, fmt, output_file):
    g = dm.graph
    if start not in g.station_to_number:
        print("Starting station '%s' does not exist, please check the station name" % start)
        return
    if end not in g.station_to_number:
        print("Ending station '%s' does not exist, please check the station name" % end)
        return
    if start == end:
        print("Starting station cannot be the same as the ending station")
        return
    try:
        r = build_result(dm, start, end, config)
    except Exception as ex:
        print("Path calculation error: %s" % ex)
        return
    text = render(r, fmt)
    print(text)
    if output_file:
        try:
            with open(output_file, "a", encoding="utf-8") as f:
                f.write(text + "\n")
            print("Result saved to %s" % output_file)
        except OSError as ex:
            print("Could not write output file: %s" % ex)


def main(argv=None):
    ap = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    ap.add_argument("--output_file")
    ap.add_argument("--format", choices=["text", "json"], default="text")
    ap.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                    default="shortest_distance")
    ap.add_argument("--max_transfers", type=int)
    ap.add_argument("--line_preference")
    ap.add_argument("--check_data", action="store_true")
    args = ap.parse_args(argv)
    try:
        dm = DataManager()
    except Exception as ex:
        print("Failed to load data: %s" % ex)
        return 1
    if args.check_data:
        probs = dm.check_data()
        if probs:
            print("Data check found %d problem(s):" % len(probs))
            for p in probs:
                print(" -", p)
            return 1
        print("Data check passed: %d stations." % len(dm.graph.station_to_number))
        return 0
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []
    print("Enter 'Starting Station; Ending Station' (q to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        try:
            a, b = parse_input(line)
        except ValueError as ex:
            print(ex)
            continue
        run_query(dm, a, b, config, args.format, args.output_file)
    return 0


if __name__ == "__main__":
    sys.exit(main())

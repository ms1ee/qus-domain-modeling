import os
import re
from collections import OrderedDict
from lottery_system.models import User

DEFAULT_USERS = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Product Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Marketing Department"),
    ("Zhou Ba", "Marketing Department"),
    ("Wu Jiu", "HR Department"),
    ("Zheng Shi", "HR Department"),
    ("Li Bin", "Technology Department"),
    ("Chen Qixian", "Product Department"),
    ("Jin Duo", "Marketing Department"),
    ("Ma Jiaqi", "Technology Department"),
    ("Cheng Ru", "Cheng Ru group"),
    ("Liu Yang", "Cheng Ru group"),
    ("Huang Lei", "Cheng Ru group"),
    ("Xu Fang", "Finance Department"),
    ("Gao Min", "Finance Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_USERS], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = re.split(r"[,;，；]", line.strip())
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Format Error: Line {i}: '{line.strip()}' "
                              "(expected 'Name,Department')")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, a, d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        if (not isinstance(order, (list, tuple)) or len(order) != n
                or sorted(order) != list(range(1, n + 1))):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

FALLBACK_GROUP = "Cheng Ru group"
SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winner_names = set()
        self.last_two_draw_results = []

    def _fallback_active(self, prize):
        return (prize.associated_drawer == FALLBACK_DRAWER
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if u.name not in self.winner_names]
        if self._fallback_active(prize):
            return [u for u in pool if u.department == FALLBACK_GROUP]
        if prize.associated_drawer == SPECIAL_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def rule_description(self, prize):
        parts = []
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("Special drawer: only previous winners excluded")
        elif prize.avoid_groups:
            parts.append("Avoid groups (drawn last): " + ", ".join(prize.avoid_groups))
        else:
            parts.append("Previous winners excluded")
        if self._fallback_active(prize):
            parts.append(f"Fallback: pool limited to {FALLBACK_GROUP}")
        return "; ".join(parts)

    def draw_next(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool and not self._fallback_active(prize):
            # avoidance group members are drawn after other members
            pool = [u for u in self.users if u.name not in self.winner_names]
        if not pool:
            return None
        user = random.choice(pool)
        self.winner_names.add(user.name)
        return user

    def finish_round(self, prize, winners):
        drawer = prize.associated_drawer
        if drawer in WATCHED_DRAWERS:
            has = any(u.department == FALLBACK_GROUP for u in winners)
            self.last_two_draw_results.append(has)
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == FALLBACK_DRAWER:
            self.last_two_draw_results = []

import os
from collections import OrderedDict
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, winning_time=None, order=0):
        self.user = user
        self.prize_name = prize_name
        self.winning_time = winning_time or datetime.now()
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name, datetime.now(), len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        result = OrderedDict()
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return dict(result)

    def export_to_csv(self, prizes):
        grouped = self.get_winners_by_prize(prizes)
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.winning_time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt file > ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        valid, dups = lm.validate_users(users)
        for e in errs:
            print(e)
        for d in dups:
            print(f"Duplicate name: {d.name} ({d.department})")
        if errs or dups:
            a = input("Remove invalid data and continue? (y = continue, n = terminate/reload) ").strip().lower()
            if a != "y":
                if c == "1":
                    continue
                continue
        if not valid:
            print("No valid participants.")
            continue
        return valid


def show_groups(lm, users):
    for g, n in lm.get_group_info(users).items():
        print(f"  {g}: {n}")


def configure_prizes(pm, groups):
    pm.load_default_prizes()
    while True:
        print("Prizes:")
        for i, p in enumerate(pm.get_prizes(), 1):
            print(f" {i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
        c = input("a) add prize  v) view groups  o) reorder  s) set avoid groups  d) done > ").strip().lower()
        if c == "a":
            try:
                name = input("Name: ").strip()
                q = int(input("Quantity: "))
                drawer = input("Drawer (blank none): ").strip() or None
                av = [x.strip() for x in input("Avoid groups (comma separated): ").split(",") if x.strip()]
                pm.add_prize(name, q, av, drawer)
            except ValueError as e:
                print("Invalid:", e)
        elif c == "v":
            for g, n in groups.items():
                print(f"  {g}: {n}")
        elif c == "s":
            try:
                i = int(input("Prize number: "))
                av = [x.strip() for x in input("Avoid groups (comma separated): ").split(",") if x.strip()]
                pm.get_prizes()[i - 1].avoid_groups = av
            except (ValueError, IndexError):
                print("Invalid input")
        elif c == "o":
            try:
                order = [int(x) for x in input("New order (e.g. 2 1 3): ").split()]
            except ValueError:
                order = []
            print("Reordered" if pm.reorder_prizes(order) else "Invalid order")
        elif c == "d":
            if pm.get_prizes():
                return


def run_round(lot, rm, prize, idx):
    print(f"\n=== Round {idx}: {prize.name} | quantity {prize.quantity} ===")
    print("Rules:", lot.rule_description(prize))
    winners = []
    paused = False
    while len(winners) < prize.quantity:
        cmd = input("Enter to draw (pause / continue / terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            paused = True
            print("Paused. Type 'continue' to resume.")
            continue
        if cmd == "continue":
            paused = False
            continue
        if paused:
            print("Paused. Type 'continue' to resume.")
            continue
        print(f"Currently drawing winner #{len(winners) + 1}...")
        u = lot.draw_next(prize)
        if u is None:
            print("No eligible participants left.")
            break
        rm.add_winner(u, prize)
        winners.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.finish_round(prize, winners)
    print("Round results:")
    for i, u in enumerate(winners, 1):
        print(f" {i}. {u.name} ({u.department})")


def main():
    lm = ListManager()
    users = load_users(lm)
    groups = lm.get_group_info(users)
    print("Groups:")
    show_groups(lm, users)
    pm = PrizeManager()
    configure_prizes(pm, groups)
    lot = LotteryManager(users)
    rm = ResultManager()
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        run_round(lot, rm, p, i)
    print("\n=== Final winners (high to low prize level) ===")
    ordered = list(reversed(prizes))
    for name, ws in rm.get_winners_by_prize(ordered).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {name}")
    ok, msg = rm.export_to_csv(ordered)
    print(msg)


if __name__ == "__main__":
    main()

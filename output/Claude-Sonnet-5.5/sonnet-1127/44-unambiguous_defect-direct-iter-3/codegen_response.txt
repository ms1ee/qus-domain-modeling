"""Graph data structures."""


class Edge:
    """Undirected edge between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph; node numbers are assigned by the caller."""

    def __init__(self):
        self.graph = {}          # number -> list[Edge]
        self.station_lines = {}  # name -> list[str]

    def add_node(self, number):
        self.graph.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.graph[edge.source].append(edge)
        self.graph[edge.dest].append(edge.reversed())

    def adj(self, node):
        return self.graph.get(node, [])

    def nodes(self):
        return list(self.graph.keys())

    def edge_count(self):
        return sum(len(v) for v in self.graph.values()) // 2

"""Dijkstra shortest path with priority queue."""
import heapq
import itertools

from models import Graph  # noqa: F401

MODES = ("shortest_distance", "least_transfer", "comprehensive")


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap minimum priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}   # node -> state
        self._info = {}   # state -> (cost, dist, prev_state, edge, transfers)
        self._run()

    def _penalty(self):
        mode = self.config.mode
        if mode == "least_transfer":
            return 100000.0
        if mode == "comprehensive":
            return 1000.0
        return 0.0

    def _run(self):
        pen = self._penalty()
        limit = self.config.max_transfers
        prefs = set(self.config.preferred_lines or [])

        def key(node, line, t):
            return (node, line, t if limit is not None else 0)

        start = key(self.source, None, 0)
        self._info[start] = (0.0, 0, None, None, 0)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, _ = st
            _, d, _, _, t = self._info[st]
            cur = self._best.get(node)
            if cur is None or (cost, d) < (self._info[cur][0], self._info[cur][1]):
                self._best[node] = st
                self.dist_to[node] = d
            for e in self.graph.adj(node):
                nt = t + (1 if line is not None and e.line != line else 0)
                if limit is not None and nt > limit:
                    continue
                w = e.weight * (0.9 if e.line in prefs else 1.0)
                ncost = cost + w + (pen if nt > t else 0)
                ns = key(e.dest, e.line, nt)
                if ns in done:
                    continue
                old = self._info.get(ns)
                if old is None or ncost < old[0]:
                    self._info[ns] = (ncost, d + e.distance, st, e, nt)
                    pq.push(ncost, ns)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        st = self._best.get(target)
        if st is None:
            return [], []
        edges = []
        while st is not None:
            _, _, prev, e, _ = self._info[st]
            if e is not None:
                edges.append(e)
            st = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""
import math

FARE_TIERS = [(6000, 3), (12000, 4), (22000, 5), (32000, 6)]


def Sum_money(distance):
    """Fare in yuan for distance in meters."""
    if distance is None or distance != distance or distance < 0:
        raise ValueError("invalid distance: %r" % (distance,))
    d = math.ceil(distance)
    for limit, fare in FARE_TIERS:
        if d < limit:
            return fare
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 35 km/h average plus 5 min per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("negative input")
    return distance / 1000.0 / 35.0 * 60.0 + 5.0 * transfer_count

"""Network building, integrity check, and data updates."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE = {
    "1号线": [], "2号线": [], "4号线": [], "5号线": [], "6号线": [], "8号线": [],
    "10号线": [], "13号线": [], "14号线": [], "15号线": [],
}


class DataManager:
    def __init__(self, path=DATA_FILE_PATH):
        self.data = load_subway_data(path)
        self.rebuild()

    def rebuild(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in self.data.get("stations", []):
            self._num(s["name"])
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a, b = self._num(c["source"]), self._num(c["dest"])
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))
            for n in (c["source"], c["dest"]):
                ls = self.graph.station_lines.setdefault(n, [])
                if c["line"] not in ls:
                    ls.append(c["line"])
        for n in self.station_to_number.values():
            self.graph.add_node(n)

    def _num(self, name):
        if name not in self.station_to_number:
            n = len(self.station_to_number)
            self.station_to_number[name] = n
            self.number_to_station[n] = name
        return self.station_to_number[name]

    def is_transfer(self, name):
        return len(self.graph.station_lines.get(name, [])) > 1

    def check_data(self):
        """Return (ok, list of messages)."""
        msgs = []
        ok = True
        stations = self.data.get("stations", [])
        conns = self.data.get("connections", [])
        names = {s["name"] for s in stations}
        if not stations or not conns:
            return False, ["Data is empty"]
        msgs.append("Stations: %d, connections: %d" % (len(names), len(conns)))
        lines = set()
        for s in stations:
            lines.update(s.get("lines", []))
        for c in conns:
            lines.add(c["line"])
        missing = [l for l in REPRESENTATIVE if l not in lines
                   and l.replace("号线", "") not in {x.replace("号线", "") for x in lines}]
        if missing:
            msgs.append("Warning: lines not found: " + ", ".join(missing))
        bad = [c for c in conns if c["source"] not in names or c["dest"] not in names]
        if bad:
            ok = False
            msgs.append("%d connections reference unknown stations" % len(bad))
        badd = [c for c in conns if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0]
        if badd:
            ok = False
            msgs.append("%d connections have invalid distance" % len(badd))
        loops = [c for c in conns if c["source"] == c["dest"]]
        if loops:
            ok = False
            msgs.append("%d self-loop connections" % len(loops))
        connected = {n for n, e in self.graph.graph.items() if e}
        iso = [self.number_to_station[n] for n in self.graph.graph if n not in connected]
        if iso:
            ok = False
            msgs.append("Isolated stations: " + ", ".join(iso[:10]))
        msgs.append("Data check " + ("passed" if ok else "FAILED"))
        return ok, msgs

    # updates
    def add_line(self, line_name, station_names, distances):
        if len(distances) != len(station_names) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in station_names:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                s = {"name": n, "lines": [line_name]}
                self.data["stations"].append(s)
                known[n] = s
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": int(d), "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.rebuild()
                return True
        return False

    def update_distance(self, a, b, distance):
        if distance <= 0:
            raise ValueError("distance must be positive")
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = int(distance)
                found = True
        if found:
            self.rebuild()
        return found

"""Route report generation."""
import json

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


def compute_route(dm, start, end, config=None):
    """Compute route summary dict."""
    config = config or PathfindingConfig()
    s, e = dm.station_to_number[start], dm.station_to_number[end]
    sp = ShortestPath(dm.graph, s, config)
    nodes, edges = sp.path_with_edges(e)
    if not nodes:
        raise RuntimeError("No route found between '%s' and '%s'" % (start, end))
    dist = sp.dist_to[e]
    transfers = []
    segments = []
    for ed in edges:
        if segments and segments[-1]["line"] == ed.line:
            segments[-1]["to"] = dm.number_to_station[ed.dest]
            segments[-1]["stops"] += 1
        else:
            if segments:
                transfers.append({"station": dm.number_to_station[ed.source],
                                  "from_line": segments[-1]["line"], "to_line": ed.line})
            segments.append({"line": ed.line, "from": dm.number_to_station[ed.source],
                             "to": dm.number_to_station[ed.dest], "stops": 1})
    try:
        fare = Sum_money(dist)
    except Exception as ex:
        raise RuntimeError("Fare calculation failed: %s" % ex)
    return {
        "query": {"start": start, "end": end, "mode": config.mode,
                  "max_transfers": config.max_transfers,
                  "line_preference": list(config.preferred_lines)},
        "path": [dm.number_to_station[n] for n in nodes],
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(calculate_estimated_time(dist, len(transfers)), 1),
        "transfer_count": len(transfers),
        "transfers": transfers,
        "segments": segments,
        "lines": list(dict.fromkeys(sg["line"] for sg in segments)),
    }


def format_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: %d m (%.2f km)" % (r["distance_m"], r["distance_km"]),
           "Segments:"]
    for sg in r["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (sg["line"], sg["from"], sg["to"], sg["stops"]))
    out.append("Transfers: %d" % r["transfer_count"])
    for t in r["transfers"]:
        out.append("  Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    out.append("Lines: " + ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def format_json(r):
    return json.dumps(r, ensure_ascii=False, indent=2)

"""CLI entry point for the Beijing Subway fare system."""
import argparse
import re
import sys

from data_manager import DataManager
from pathfinding import PathfindingConfig, MODES
from report import compute_route, format_json, format_text

SEP = re.compile(r"[;；,，]")


def parse_query(text, dm):
    """Return (start, end) or raise ValueError with a message."""
    parts = [p.strip() for p in SEP.split(text)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    a, b = parts
    if a not in dm.station_to_number:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % a)
    if b not in dm.station_to_number:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % b)
    if a == b:
        raise ValueError("Starting station cannot be the same as the ending station")
    return a, b


def run_query(dm, a, b, config, fmt):
    try:
        r = compute_route(dm, a, b, config)
    except RuntimeError as ex:
        return None, "Path calculation error: %s" % ex
    except Exception as ex:
        return None, "Path calculation error: %s" % ex
    return r, (format_json(r) if fmt == "json" else format_text(r))


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        dm = DataManager()
    except Exception as ex:
        print("Failed to load data: %s" % ex)
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1

    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    if args.line_preference:
        config.preferred_lines = [x.strip() for x in SEP.split(args.line_preference) if x.strip()]

    if args.output_file:
        open(args.output_file, "w", encoding="utf-8").close()
    print("Enter 'Starting Station; Ending Station' (or 'q' to quit):")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        try:
            a, b = parse_query(line, dm)
        except ValueError as ex:
            print(ex)
            continue
        r, text = run_query(dm, a, b, config, args.format)
        print(text)
        if r is not None and args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(text + "\n\n")
            except OSError as ex:
                print("Cannot write output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

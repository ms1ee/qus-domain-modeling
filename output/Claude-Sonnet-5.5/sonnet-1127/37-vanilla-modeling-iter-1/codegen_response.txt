from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class User:
    name: str
    department: str
    has_won: bool = field(default=False, compare=False)

    def __hash__(self):
        return hash((self.name, self.department))


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Product Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Marketing Department"),
    ("Zhou Ba", "Marketing Department"),
    ("Wu Jiu", "Cheng Ru Group"),
    ("Zheng Shi", "Cheng Ru Group"),
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Li Bin", "Operations Department"),
    ("Qian Yi", "Operations Department"),
    ("Feng Er", "Cheng Ru Group"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]:
            try:
                self.add_prize(name, qty)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Invalid prize name or quantity")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            ok = len(order) == n and sorted(order) == list(range(1, n + 1))
        except TypeError:
            return False
        if not ok:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

LI_BIN = "Li Bin"
MONITORED = ("Chen Qixian", "Jin Duo")
TARGET = "Ma Jiaqi"
PROTECTED_GROUP = "cheng ru group"


class LotteryManager:
    def __init__(self, users):
        self.users = users
        self.winners = []
        self.last_two_draw_results = []

    def _is_won(self, u):
        return u.has_won or u in self.winners

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if not self._is_won(u)]
        drawer = prize.associated_drawer
        if drawer == LI_BIN:
            return pool
        if (drawer == TARGET and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if u.department.lower() == PROTECTED_GROUP]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        u.has_won = True
        self.winners.append(u)
        return u

    def draw_round(self, prize, control=None, on_win=None):
        """control(i) may return 'terminate' to stop the round."""
        results = []
        for i in range(1, prize.quantity + 1):
            if control and control(i) == "terminate":
                break
            u = self.draw_one(prize)
            if u is None:
                break
            results.append(u)
            if on_win:
                on_win(u)
        self.finish_round(prize, results)
        return results

    def finish_round(self, prize, results):
        if prize.associated_drawer in MONITORED:
            has = any(u.department.lower() == PROTECTED_GROUP for u in results)
            self.last_two_draw_results.append(has)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, time=None, order=0):
        self.user = user
        self.prize_name = prize_name
        self.time = time or datetime.now()
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name, datetime.now(), len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        res = {}
        for p in prizes:
            res[p.name] = sorted(
                [w for w in self.winners if w.prize_name == p.name],
                key=lambda w: w.order)
        return res

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import sys
import os

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def level_of(name):
    n = name.lower()
    for i, k in enumerate(("first", "second", "third")):
        if k in n:
            return 3 - i
    return 0


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt file: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        if errs and c == "2":
            a = input("Errors found. (r)emove invalid lines and continue / (t)erminate: ").lower()
            if a.startswith("t"):
                continue
        users, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name}")
        if users:
            return users
        print("No valid users.")


def setup_prizes(pm, users, lm):
    if input("Load default prizes? (y/n): ").lower().startswith("y"):
        for e in pm.load_default_prizes():
            print(e)
    while input("Add custom prize? (y/n): ").lower().startswith("y"):
        name = input("Name: ").strip()
        try:
            qty = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            drawer = input("Drawer (optional): ").strip() or None
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Invalid input:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("New order (space-separated numbers, Enter to keep): ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, keeping original.")
        except ValueError:
            print("Invalid order, keeping original.")


def main():
    lm = ListManager()
    users = load_users(lm)
    for g, c in lm.get_group_info(users).items():
        print(f"{g}: {c}")
    pm = PrizeManager()
    setup_prizes(pm, users, lm)
    prizes = pm.get_prizes()
    engine = LotteryManager(users)
    rm = ResultManager()

    for r, prize in enumerate(prizes, 1):
        print(f"\n=== Round {r}: {prize.name}, quantity {prize.quantity} ===")
        rule = "Excludes previous winners"
        if prize.associated_drawer == "Li Bin":
            rule += "; avoidance groups not applied (Li Bin)"
        elif prize.avoid_groups:
            rule += "; avoids " + ", ".join(prize.avoid_groups)
        print("Rules:", rule)

        def control(i):
            while True:
                cmd = input(f"Currently drawing winner #{i} (Enter=draw, pause, terminate): ").strip().lower()
                if cmd == "terminate":
                    return "terminate"
                if cmd == "pause":
                    while input("Paused. Type 'continue': ").strip().lower() != "continue":
                        pass
                    continue
                return None

        def on_win(u):
            rm.add_winner(u, prize)
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")

        res = engine.draw_round(prize, control, on_win)
        if len(res) < prize.quantity:
            print("Round ended early or pool exhausted.")

    print("\n=== Final Results ===")
    ordered = sorted(prizes, key=lambda p: -level_of(p.name))
    for name, ws in rm.get_winners_by_prize(ordered).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    ok, msg = rm.export_to_csv(ordered)
    print(msg)


if __name__ == "__main__":
    main()

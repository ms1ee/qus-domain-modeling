import os
from collections import OrderedDict
from lottery_system.models import User


class ListManager:
    DEFAULT_DATA = [
        ("Zhang Wei", "Technology Department"),
        ("Wang Fang", "Technology Department"),
        ("Li Na", "Technology Department"),
        ("Liu Yang", "Product Department"),
        ("Chen Qixian", "Product Department"),
        ("Zhao Min", "Product Department"),
        ("Jin Duo", "Operations Department"),
        ("Ma Jiaqi", "Operations Department"),
        ("Li Bin", "Operations Department"),
        ("Sun Li", "Cheng Ru Group"),
        ("Zhou Jie", "Cheng Ru Group"),
        ("Wu Tao", "Cheng Ru Group"),
    ]

    def load_default_list(self):
        return [User(n, d) for n, d in self.DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, raw in enumerate(lines, 1):
            line = raw.strip()
            if not line:
                continue
            parts = None
            for sep in (",", "，", "\t"):
                if sep in line:
                    parts = [p.strip() for p in line.split(sep)]
                    break
            if parts is None or len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        presets = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
        ]
        for name, q, avoid, drawer in presets:
            try:
                self.add_prize(name, q, avoid, drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        prize = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        if not isinstance(order, (list, tuple)) or len(order) != n:
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "cheng ru"


def is_cheng_ru(user):
    return CHENG_RU in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {u.name for u in self.winners}
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if (drawer == "Ma Jiaqi" and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if is_cheng_ru(u)]
        if drawer == "Li Bin":
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def record_round_result(self, round_winners):
        self.last_two_draw_results.append(any(is_cheng_ru(u) for u in round_winners))
        self.last_two_draw_results = self.last_two_draw_results[-2:]

    @staticmethod
    def describe_rules(prize):
        parts = []
        if prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer == "Li Bin":
            parts.append("relaxed filtering (Li Bin)")
        if prize.associated_drawer == "Ma Jiaqi":
            parts.append("Cheng Ru group protection may apply")
        parts.append("previous winners excluded")
        return "; ".join(parts)

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize=None, order=0):
        self.user = user
        self.prize = prize
        self.order = order
        self.time = datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize, len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners
                              if w.prize is not None and w.prize.name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for pname, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{pname}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    choice = input("Load list: 1) default  2) custom txt file: ").strip()
    if choice == "2":
        users, errors = lm._load_from_path(input("Path: ").strip())
    else:
        users, errors = lm.load_default_list()
    for e in errors:
        print(e)
    if errors:
        a = input("Errors found. (r)emove invalid and continue / (t)erminate: ").strip().lower()
        if a != "r":
            return []
    valid, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate name removed: {d.name} ({d.department})")
    return valid


def configure_prizes(pm):
    if input("Load preset prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            q = int(input("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
        drawer = input("Associated drawer: ").strip() or None
        try:
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print(e)
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("New order (space separated numbers, blank to keep): ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, keeping default")
        except ValueError:
            print("Invalid order, keeping default")


def run():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No valid participants.")
        return
    for g, c in lm.get_group_info(users).items():
        print(f"Group {g}: {c} people")
    pm = PrizeManager()
    configure_prizes(pm)
    prizes = pm.get_prizes()
    engine = LotteryManager(users)
    rm = ResultManager()
    for idx, prize in enumerate(prizes, 1):
        print(f"\n=== Round {idx}: {prize.name}, quantity {prize.quantity} ===")
        print("Rules:", engine.describe_rules(prize))
        round_winners = []
        n = 0
        while n < prize.quantity:
            cmd = input("Enter to draw (pause / continue / terminate): ").strip().lower()
            if cmd == "terminate":
                print("Round terminated.")
                break
            if cmd == "pause":
                while input("Paused. Type 'continue': ").strip().lower() != "continue":
                    pass
                continue
            print(f"Currently drawing the {n + 1}th winner")
            u = engine.draw_one(prize)
            if u is None:
                print("No eligible participants left.")
                break
            n += 1
            rm.add_winner(u, prize)
            round_winners.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        engine.record_round_result(round_winners)
        print("Round results:", ", ".join(u.name for u in round_winners) or "none")
    print("\n=== Final winners ===")
    for pname, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {pname}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    run()

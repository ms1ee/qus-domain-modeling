class GridMap:
    """8x8 grid, 100-unit cells, 800x800 map."""

    def __init__(self):
        self.width = 800
        self.height = 800
        self.cell_size = 100
        self.grid_dimension = 8

    def clamp(self, v):
        return max(0, min(self.width, int(v)))

    def to_cell_corner(self, x, y):
        """Convert coordinates to the top-left corner of their grid cell."""
        cx = (self.clamp(x) // self.cell_size) * self.cell_size
        cy = (self.clamp(y) // self.cell_size) * self.cell_size
        return cx, cy

    def to_cell_center(self, x, y):
        cx, cy = self.to_cell_corner(x, y)
        return cx + 50, cy + 50

    def is_road(self, x, y):
        """Points at integer hundreds are locations; others are roads."""
        return not (x % 100 == 0 and y % 100 == 0)

class PathPlanner:
    """Right-angle, street-preferring path planner."""

    def plan(self, start, end):
        """Return waypoint list (excluding start) from start to end."""
        sx, sy = start
        ex, ey = end
        if (sx, sy) == (ex, ey):
            return []
        if sx == ex or sy == ey:
            return [(ex, ey)]
        # Prefer to stay on a street: if on a horizontal street move
        # horizontally first; otherwise vertical first (vertical street
        # or directly above/below special case).
        if sy % 100 == 0:
            return [(ex, sy), (ex, ey)]
        return [(sx, ey), (ex, ey)]

    def plan_multi(self, start, points):
        """Continuous path through several destinations."""
        path = []
        cur = start
        for p in points:
            path.extend(self.plan(cur, p))
            cur = p
        return path

    @staticmethod
    def length(start, path):
        total = 0
        cur = start
        for p in path:
            total += abs(p[0] - cur[0]) + abs(p[1] - cur[1])
            cur = p
        return total

    def step(self, pos, target, dist):
        """Move up to dist units from pos toward target along the path."""
        path = self.plan(pos, target)
        x, y = pos
        remaining = dist
        for wx, wy in path:
            d = abs(wx - x) + abs(wy - y)
            if d <= remaining:
                remaining -= d
                x, y = wx, wy
            else:
                if wx != x:
                    x += remaining if wx > x else -remaining
                else:
                    y += remaining if wy > y else -remaining
                remaining = 0
            if remaining == 0:
                break
        return x, y

"""Order data model."""
import itertools

import globalvar
from gridmap import GridMap

_counter = itertools.count(1)
_grid = GridMap()


def reset_counter():
    global _counter
    _counter = itertools.count(1)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = _grid.to_cell_corner(*res_pos)
        self.cus_pos = _grid.to_cell_corner(*cus_pos)
        self.start_time = globalvar.get_value("tick", 0)
        self.completion_time = None
        self.rider_id = None
        self.status = "pending"
        self.distance = float(abs(self.res_pos[0] - self.cus_pos[0])
                              + abs(self.res_pos[1] - self.cus_pos[1]))
        self.time_taken = None

    def assign_to(self, rider_id):
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick):
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "start_time": self.start_time,
                "completion_time": self.completion_time,
                "rider_id": self.rider_id, "status": self.status,
                "distance": self.distance, "time_taken": self.time_taken}

"""Rider model."""
import globalvar
from pathplanner import PathPlanner

_planner = PathPlanner()


class Rider:
    """Delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"time": globalvar.get_value("tick", 0),
                         "event": event, "order_id": order_id,
                         "x": self.x, "y": self.y, "state": self.state})

    def active_order(self):
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def pending_count(self):
        return sum(1 for o in self.orders if o.status != "completed")

    def move(self):
        """Advance one tick."""
        order = self.active_order()
        if order is None:
            self.state = 0
            return
        if self.state == 0:
            self.state = 1
            self._log("accepted_order", order.id)
        target = order.res_pos if self.state == 1 else order.cus_pos
        self.special_case = self.x % 100 != 0 and self.y % 100 != 0
        self.x, self.y = _planner.step((self.x, self.y), target, self.speed)
        if (self.x, self.y) != tuple(target):
            return
        if self.state == 1:
            self.state = 2
            self._log("arrived_at_merchant", order.id)
        else:
            order.complete(globalvar.get_value("tick", 0))
            self.state = 0
            self._log("delivered_to_customer", order.id)
            nxt = self.active_order()
            if nxt is not None:
                self.state = 1
                self._log("accepted_order", nxt.id)

"""Order management."""
import math

import globalvar
from order_model import Order


def find_best_rider_for_order(order):
    """Rider with fewest current orders, ties by distance to restaurant."""
    riders = globalvar.get_value("riders", [])
    best = None
    best_key = None
    for r in riders:
        d = math.hypot(r.x - order.res_pos[0], r.y - order.res_pos[1])
        key = (r.pending_count(), d)
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


class OrderManager:
    """Creates, assigns and tracks orders."""

    def create_order(self, rx, ry, cx, cy):
        order = Order((rx, ry), (cx, cy))
        globalvar.get_value("orders").append(order)
        return order

    def assign(self, order):
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider

    def unassigned(self):
        return [o for o in globalvar.get_value("orders")
                if o.status == "pending"]

"""System manager."""
import json

import globalvar
from gridmap import GridMap
from order import OrderManager
from order_model import reset_counter
from pathplanner import PathPlanner
from rider import Rider

SAVE_FILE = "state.json"


def init_system():
    """Initialise globals and return a SystemManager."""
    globalvar._init()
    reset_counter()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    return SystemManager()


class SystemManager:
    """Manages funds, riders and time."""

    def __init__(self):
        self.grid = GridMap()
        self.order_manager = OrderManager()
        self.planner = PathPlanner()

    @property
    def funds(self):
        return globalvar.get_value("funds")

    @property
    def riders(self):
        return globalvar.get_value("riders")

    def maybe_recruit(self):
        """Recruit a rider when funds>=300 and pending > idle riders."""
        pending = len(self.order_manager.unassigned())
        idle = sum(1 for r in self.riders if r.pending_count() == 0)
        if self.funds >= 300 and pending > idle:
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            globalvar.set_value("funds", self.funds - 500)
            self.riders.append(Rider(rid))
            return True
        return False

    def submit_order(self, rx, ry, cx, cy):
        """Create an order; returns (order, rider, recruited)."""
        order = self.order_manager.create_order(rx, ry, cx, cy)
        recruited = self.maybe_recruit()
        rider = self.order_manager.assign(order)
        return order, rider, recruited

    def dispatch_pending(self):
        for o in self.order_manager.unassigned():
            if not self.riders:
                self.maybe_recruit()
            self.order_manager.assign(o)

    def tick(self, n=1):
        for _ in range(n):
            globalvar.set_value("tick", globalvar.get_value("tick") + 1)
            self.dispatch_pending()
            for r in self.riders:
                r.move()

    def save(self, path=SAVE_FILE):
        data = {
            "funds": self.funds,
            "rider_id_counter": globalvar.get_value("rider_id_counter"),
            "tick": globalvar.get_value("tick"),
            "riders": [{"rider_id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state,
                        "special_case": r.special_case,
                        "orders": [o.id for o in r.orders],
                        "log": r.log} for r in self.riders],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Display manager."""
import globalvar

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text rendering of status and map."""

    SIZE = 16

    def _cell(self, x, y):
        return (min(self.SIZE - 1, max(0, y // 50)),
                min(self.SIZE - 1, max(0, x // 50)))

    def render_map(self, route=None):
        grid = [["#" if (i % 2 == 0 or j % 2 == 0) and False else "."
                 for j in range(self.SIZE)] for i in range(self.SIZE)]
        for i in range(self.SIZE):
            for j in range(self.SIZE):
                if i % 2 == 0 and j % 2 == 0:
                    grid[i][j] = "#"
        if route:
            for (x1, y1), (x2, y2) in zip(route, route[1:]):
                steps = max(abs(x2 - x1), abs(y2 - y1)) // 25 + 1
                for s in range(steps + 1):
                    x = x1 + (x2 - x1) * s // steps
                    y = y1 + (y2 - y1) * s // steps
                    i, j = self._cell(x, y)
                    grid[i][j] = "*"
        for o in globalvar.get_value("orders", []):
            if o.status != "completed":
                i, j = self._cell(*o.res_pos)
                grid[i][j] = "S"
                i, j = self._cell(*o.cus_pos)
                grid[i][j] = "C"
        for r in globalvar.get_value("riders", []):
            i, j = self._cell(r.x, r.y)
            grid[i][j] = "R"
        return "\n".join("".join(row) for row in grid)

    def status(self):
        riders = globalvar.get_value("riders", [])
        lines = ["Time: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) %s, %d order(s): %s" % (
                r.rider_id, r.x, r.y, STATE_NAMES[r.state], len(r.orders),
                [(o.id, o.status) for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        orders = globalvar.get_value("orders", [])
        if not orders:
            return "No orders."
        lines = []
        for o in orders:
            lines.append(
                "Order %d: start=%d restaurant=%s customer=%s dist=%.1f "
                "time_taken=%s rider=%s status=%s" % (
                    o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                    o.time_taken if o.time_taken is not None else "-",
                    o.rider_id if o.rider_id is not None else "-", o.status))
        return "\n".join(lines)

    def rider_logs(self):
        lines = []
        for r in globalvar.get_value("riders", []):
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%d %s at (%d,%d) order=%s" % (
                    e["time"], e["event"], e["x"], e["y"],
                    e["order_id"] if e["order_id"] is not None else "-"))
        return "\n".join(lines) if lines else "No riders."

"""Input parsing and validation."""

try:
    import readline  # noqa: F401  (command history)
except ImportError:  # pragma: no cover
    readline = None

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "n": "tick"}


class InputManager:
    """Parses commands and validates coordinates."""

    def parse_command(self, line):
        parts = line.strip().split()
        if not parts:
            return None, []
        cmd = parts[0].lower()
        return ALIASES.get(cmd, cmd), parts[1:]

    def parse_coords(self, text):
        """Parse four integers in 0-1000; raises ValueError."""
        text = text.replace("[", " ").replace("]", " ").replace(",", " ")
        parts = text.split()
        if len(parts) != 4:
            raise ValueError("Need exactly four integers: rx ry cx cy")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        for v in vals:
            if not 0 <= v <= 1000:
                raise ValueError("Coordinates must be within 0-1000")
        return vals

"""Command-line entry point."""
import math

import globalvar
from display import DisplayManager
from inputmanager import InputManager
from system import init_system

HELP = ("Commands: start (s), status (st), orders (o), riders (r), "
        "tick [n] (n), help, quit/exit (q)")


def do_start(sm, disp, im):
    try:
        vals = im.parse_coords(input("Enter rx ry cx cy: "))
    except ValueError as e:
        print("Error:", e)
        return
    order, rider, recruited = sm.submit_order(*vals)
    print("Restaurant grid: %s  Customer grid: %s" % (order.res_pos,
                                                     order.cus_pos))
    print("New rider added: %s" % ("yes" if recruited else "no"))
    if rider is None:
        print("No rider available yet; order %d queued." % order.id)
        return
    start = (rider.x, rider.y)
    route = [start] + sm.planner.plan_multi(start,
                                            [order.res_pos, order.cus_pos])
    total = sm.planner.length(start, route[1:])
    print("Order %d assigned to rider %d, speed %d, distance %d, "
          "est. time %d ticks" % (order.id, rider.rider_id, rider.speed,
                                  total, math.ceil(total / rider.speed)))
    print(disp.render_map(route))


def main():
    sm = init_system()
    disp = DisplayManager()
    im = InputManager()
    print("Food Delivery Dispatch System")
    print(HELP)
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            cmd, args = im.parse_command(line)
            if cmd is None:
                continue
            if cmd in ("quit", "exit"):
                break
            elif cmd == "start":
                do_start(sm, disp, im)
            elif cmd == "status":
                print(disp.status())
                print(disp.render_map())
            elif cmd == "orders":
                print(disp.orders())
            elif cmd == "riders":
                print(disp.rider_logs())
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                sm.tick(max(1, n))
                print("Time:", globalvar.get_value("tick"))
                continue
            elif cmd == "help":
                print(HELP)
            else:
                print("Unknown command. " + HELP)
                continue
        except Exception as e:  # graceful recovery
            print("Error:", e)
            continue
        sm.tick()
    globalvar.set_value("running", False)
    try:
        sm.save()
    except OSError as e:
        print("Could not save state:", e)


if __name__ == "__main__":
    main()

"""Grid based path planning."""

CELL = 100
MAP_SIZE = 800


def to_center(v):
    """Convert a raw coordinate (0-800) to its grid cell center."""
    cell = int(v) // CELL
    cell = max(0, min(MAP_SIZE // CELL - 1, cell))
    return cell * CELL + CELL // 2


class PathPlanner:
    """Street-preferring right-angle path planner."""

    @staticmethod
    def horizontal_first(pos):
        """Decide the first axis: follow a street if on one."""
        x, y = pos
        if y % CELL == 0:
            return True
        if x % CELL == 0:
            return False
        return True

    def plan(self, start, end):
        """Return list of waypoints from start to end (inclusive)."""
        sx, sy = start
        ex, ey = end
        pts = [(sx, sy)]
        if self.horizontal_first(start):
            corner = (ex, sy)
        else:
            corner = (sx, ey)
        if corner != pts[-1] and corner != (ex, ey):
            pts.append(corner)
        if (ex, ey) != pts[-1]:
            pts.append((ex, ey))
        return pts

    def distance(self, a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    def step(self, pos, target, speed):
        """Move up to speed units from pos toward target."""
        x, y = pos
        tx, ty = target
        left = speed
        axes = ["x", "y"] if self.horizontal_first(pos) else ["y", "x"]
        for axis in axes:
            if left <= 0:
                break
            if axis == "x":
                d = min(left, abs(tx - x))
                x += d if tx > x else -d
            else:
                d = min(left, abs(ty - y))
                y += d if ty > y else -d
            left -= d
        return x, y

import itertools

try:
    from src import globalvar
    from src.path_planner import to_center
except ImportError:  # pragma: no cover
    import globalvar
    from path_planner import to_center

_counter = itertools.count(1)


class Order:
    """A delivery order; positions are converted to grid cell centers."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0]) +
                         abs(self.res_pos[1] - self.cus_pos[1]))
        self.time_taken = None
        self.efficiency = None

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        """Serialize to dict."""
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken,
                "efficiency": self.efficiency}

"""Rider model."""

try:
    from src import globalvar
    from src.path_planner import PathPlanner
except ImportError:  # pragma: no cover
    import globalvar
    from path_planner import PathPlanner

IDLE, PICKING, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider moving on the grid."""

    def __init__(self, rider_id, speed=50):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = speed
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.planner = PathPlanner()
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0),
                         "x": self.x, "y": self.y, "state": self.state})

    def move(self):
        """Advance one tick; handle state transitions and logging."""
        if self.state == IDLE:
            if not self.orders:
                return
            self.state = PICKING
            self._log("accepted_order", self.orders[0].id)
            if self.orders[0].status == "assigned":
                self.orders[0].status = "picking_up"
        order = self.orders[0]
        target = order.res_pos if self.state == PICKING else order.cus_pos
        self.x, self.y = self.planner.step((self.x, self.y), target,
                                           self.speed)
        if (self.x, self.y) != tuple(target):
            return
        if self.state == PICKING:
            self.state = DELIVERING
            order.status = "delivering"
            self._log("arrived_at_merchant", order.id)
        else:
            now = globalvar.get_value("tick", 0)
            order.status = "completed"
            order.completion_time = now
            order.time_taken = max(0, now - order.start_time)
            order.efficiency = (order.distance / order.time_taken
                                if order.time_taken else float(order.distance))
            self.orders.pop(0)
            self.state = IDLE
            self._log("delivered_to_customer", order.id)

def find_best_rider_for_order(order):
    """Return the closest rider (ties: fewer orders, lower id) or None."""
    riders = globalvar.get_value("riders", []) or []
    best, best_key = None, None
    for r in riders:
        d = abs(r.x - order.res_pos[0]) + abs(r.y - order.res_pos[1])
        key = (d, len(r.orders), r.rider_id)
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


class OrderManager:
    """Creates, assigns and records orders."""

    def create_order(self, coords):
        """Create an order from [rx, ry, cx, cy] and register it."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        return order

    def assign(self, order):
        """Assign to best rider; returns rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is not None:
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
        return rider

"""System manager and initialization."""

import json

try:
    from src import globalvar
    from src.rider import Rider
    from src.order import OrderManager
    from src.path_planner import PathPlanner
except ImportError:  # pragma: no cover
    import globalvar
    from rider import Rider
    from order import OrderManager
    from path_planner import PathPlanner

RIDER_COST = 300


class SystemManager:
    """Manages funds, riders, ticks and orders."""

    def __init__(self):
        self.order_manager = OrderManager()
        self.planner = PathPlanner()

    @property
    def funds(self):
        return globalvar.get_value("funds")

    def recruit_if_needed(self):
        """Create a new rider if funds >= 300. Returns rider or None."""
        if globalvar.get_value("funds") >= RIDER_COST:
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            globalvar.set_value("funds", globalvar.get_value("funds")
                                - RIDER_COST)
            rider = Rider(rid)
            globalvar.get_value("riders").append(rider)
            return rider
        return None

    def submit_order(self, coords):
        """Create an order; returns (order, rider, new_rider)."""
        order = self.order_manager.create_order(coords)
        new_rider = self.recruit_if_needed()
        rider = self.order_manager.assign(order)
        return order, rider, new_rider

    def tick(self):
        """Advance simulation one tick."""
        globalvar.set_value("tick", globalvar.get_value("tick") + 1)
        for r in globalvar.get_value("riders"):
            before = len(r.orders)
            r.move()
            if len(r.orders) < before:
                globalvar.set_value("funds",
                                    globalvar.get_value("funds") + 100)

    def save(self, path="state.json"):
        """Persist state as JSON."""
        data = {"funds": globalvar.get_value("funds"),
                "tick": globalvar.get_value("tick"),
                "rider_id_counter": globalvar.get_value("rider_id_counter"),
                "riders": [{"rider_id": r.rider_id, "x": r.x, "y": r.y,
                            "speed": r.speed, "state": r.state,
                            "orders": [o.id for o in r.orders],
                            "special_case": r.special_case, "log": r.log}
                           for r in globalvar.get_value("riders")],
                "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)


def init_system():
    """Initialize global state and return a SystemManager."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    return SystemManager()

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


def _ch(v):
    return max(0, min(16, int(v) // 50))


class DisplayManager:
    """Renders status and maps."""

    def render_map(self, riders=(), orders=(), route=None):
        grid = [["#" if (i % 2 == 0 or j % 2 == 0) else "."
                 for j in range(17)] for i in range(17)]
        if route:
            for a, b in zip(route, route[1:]):
                ax, ay, bx, by = _ch(a[0]), _ch(a[1]), _ch(b[0]), _ch(b[1])
                for cx in range(min(ax, bx), max(ax, bx) + 1):
                    for cy in range(min(ay, by), max(ay, by) + 1):
                        grid[cy][cx] = "*"
        for o in orders:
            if o.status != "completed":
                grid[_ch(o.res_pos[1])][_ch(o.res_pos[0])] = "S"
                grid[_ch(o.cus_pos[1])][_ch(o.cus_pos[0])] = "C"
        for r in riders:
            grid[_ch(r.y)][_ch(r.x)] = "R"
        return "\n".join("".join(row) for row in grid)

    def status(self, riders):
        lines = ["Funds: %s" % globalvar.get_value("funds"),
                 "Tick: %s" % globalvar.get_value("tick"),
                 "Riders: %d" % len(riders)]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) speed %d state %s orders %s"
                         % (r.rider_id, r.x, r.y, r.speed, STATES[r.state],
                            [o.id for o in r.orders]))
        return "\n".join(lines)

    def orders(self, orders):
        if not orders:
            return "No orders."
        lines = []
        for o in orders:
            lines.append("Order %d: start %s restaurant %s customer %s "
                         "distance %s time %s rider %s status %s"
                         % (o.id, o.start_time, o.res_pos, o.cus_pos,
                            o.distance, o.time_taken, o.rider_id, o.status))
        return "\n".join(lines)

    def riders_log(self, riders):
        lines = []
        for r in riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%s %s order=%s pos=(%s,%s) state=%s"
                             % (e["time"], e["event"], e["order_id"],
                                e["x"], e["y"], STATES[e["state"]]))
        return "\n".join(lines) if lines else "No riders."

"""Command parsing and validation."""

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "exit": "quit", "quit": "quit", "start": "start",
           "status": "status", "orders": "orders", "riders": "riders",
           "h": "history", "history": "history"}


class InputManager:
    """Parses commands and validates coordinates."""

    def __init__(self):
        self.command_history = []

    def parse_command(self, line):
        """Return (command, args) or (None, []) if blank/unknown."""
        parts = line.strip().split()
        if not parts:
            return None, []
        self.command_history.append(line.strip())
        return ALIASES.get(parts[0].lower(), "unknown"), parts[1:]

    @staticmethod
    def parse_coords(tokens):
        """Validate four integers in 0-800; raises ValueError."""
        if len(tokens) != 4:
            raise ValueError("Expected 4 integers: rx ry cx cy")
        try:
            vals = [int(t) for t in tokens]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        for v in vals:
            if v < 0 or v > 800:
                raise ValueError("Coordinates must be within 0-800")
        return vals

"""Command-line entry point."""

import math

try:
    from src import globalvar
    from src.system import init_system
    from src.display import DisplayManager
    from src.input_manager import InputManager
except ImportError:  # pragma: no cover
    import globalvar
    from system import init_system
    from display import DisplayManager
    from input_manager import InputManager


def handle_start(system, display, inp, args):
    """Process the start command."""
    if not args:
        args = input("Enter rx ry cx cy: ").split()
    coords = inp.parse_coords(args)
    pre = (coords[0], coords[1])
    order, rider, new_rider = system.submit_order(coords)
    out = ["Restaurant grid: (%d,%d)  Customer grid: (%d,%d)" % (
        pre[0] // 100 if pre[0] < 800 else 7, pre[1] // 100 if pre[1] < 800 else 7,
        order.cus_pos[0] // 100, order.cus_pos[1] // 100),
        "New rider added: %s" % ("yes (#%d)" % new_rider.rider_id
                                 if new_rider else "no")]
    if rider is None:
        out.append("No rider available; order %d pending." % order.id)
        return "\n".join(out)
    route = system.planner.plan((rider.x, rider.y), order.res_pos)
    route += system.planner.plan(order.res_pos, order.cus_pos)[1:]
    dist = sum(system.planner.distance(a, b) for a, b in zip(route, route[1:]))
    out.append("Order %d assigned to rider %d, speed %d, est. time %d ticks"
               % (order.id, rider.rider_id, rider.speed,
                  math.ceil(dist / rider.speed)))
    out.append(display.render_map(globalvar.get_value("riders"), [order],
                                  route))
    return "\n".join(out)


def main():
    """Run the interactive loop."""
    system = init_system()
    display = DisplayManager()
    inp = InputManager()
    print("Food Delivery Dispatch. Commands: start, status, orders, riders, quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        cmd, args = inp.parse_command(line)
        if cmd is None:
            continue
        try:
            if cmd == "start":
                print(handle_start(system, display, inp, args))
            elif cmd == "status":
                print(display.status(globalvar.get_value("riders")))
                print(display.render_map(globalvar.get_value("riders"),
                                         globalvar.get_value("orders")))
            elif cmd == "orders":
                print(display.orders(globalvar.get_value("orders")))
            elif cmd == "riders":
                print(display.riders_log(globalvar.get_value("riders")))
            elif cmd == "history":
                print("\n".join(inp.command_history))
            elif cmd == "quit":
                globalvar.set_value("running", False)
                break
            else:
                print("Unknown command.")
        except ValueError as e:
            print("Error: %s" % e)
        except Exception as e:  # recover gracefully
            print("Unexpected error: %s" % e)
        system.tick()
    try:
        system.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

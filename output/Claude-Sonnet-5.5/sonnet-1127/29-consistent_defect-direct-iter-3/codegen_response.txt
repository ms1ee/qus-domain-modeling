I had to resolve several contradictions in the PRD, so here are the choices I made, then the files.

- **Rider states:** 0 = idle, 1 = picking up, 2 = delivering. Speed is 20 and the spawn point is (400, 400).
- **Coordinates:** Positions are snapped to the top-left corner of the grid cell (`//100*100`). Restaurant and customer points sit on multiples of 100, and riders move freely along the grid lines.
- **Rider movement:** Each tick, a rider moves toward its current target with Manhattan movement of up to `speed` units. It goes horizontally first, then vertically.
- **State transitions in `move()`:**
  - Idle with an unfinished order: log "accepted_order" and set state to 1.
  - Arrival at the restaurant: log "arrived_at_merchant" and set state to 2.
  - Arrival at the customer: mark the order completed, log "delivered_to_customer", and set state to idle, or to 1 if more orders are pending.
- **Order assignment:** `find_best_rider_for_order` picks the rider with the fewest unfinished orders, breaking ties by distance. It returns `None` if there are no riders.
- **Appending to `rider.orders`:** The order appears in the list when assigned. The rider's `move()` detects the new unfinished order and logs "accepted_order".
- **Spawn:** A rider logs "spawn" in its constructor.

"""Order data model."""
import itertools

import globalvar

_counter = itertools.count(1)


def to_grid(v):
    """Convert coordinate to its grid cell top-left corner (0-800)."""
    v = int(v)
    return max(0, min(800, (v // 100) * 100))


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_grid(res_pos[0]), to_grid(res_pos[1]))
        self.cus_pos = (to_grid(cus_pos[0]), to_grid(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.end_time = None
        self.picked_up = False

    @property
    def distance(self):
        """Manhattan distance restaurant -> customer."""
        return (abs(self.res_pos[0] - self.cus_pos[0])
                + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Record assignment to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def as_list(self):
        """Order as [rx, ry, cx, cy]."""
        return [*self.res_pos, *self.cus_pos]

"""Rider class."""
import globalvar


def _step(x, y, tx, ty, speed):
    """Move up to speed units toward target, horizontal then vertical."""
    left = speed
    dx = tx - x
    s = min(abs(dx), left)
    x += s if dx > 0 else -s
    left -= s
    dy = ty - y
    s = min(abs(dy), left)
    y += s if dy > 0 else -s
    return x, y


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("time", 0) or 0,
            "x": self.x,
            "y": self.y,
        })

    def current_order(self):
        """First unfinished order or None."""
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def pending_count(self):
        """Number of unfinished orders."""
        return sum(1 for o in self.orders if o.status != "completed")

    def move(self):
        """Advance one tick."""
        order = self.current_order()
        if order is None:
            self.state = 0
            return
        if self.state == 0:
            self.state = 1
            order.picked_up = False
            self._record("accepted_order", order.id)
        if self.state == 1:
            tx, ty = order.res_pos
            self.x, self.y = _step(self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                order.picked_up = True
                self._record("arrived_at_merchant", order.id)
        elif self.state == 2:
            tx, ty = order.cus_pos
            self.x, self.y = _step(self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                order.status = "completed"
                order.end_time = globalvar.get_value("time", 0) or 0
                self._record("delivered_to_customer", order.id)
                nxt = self.current_order()
                if nxt is None:
                    self.state = 0
                else:
                    self.state = 1
                    self._record("accepted_order", nxt.id)

"""Order assignment."""
import globalvar


def find_best_rider_for_order(order):
    """Pick rider with fewest pending orders, ties by distance."""
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None

    def key(r):
        d = ((r.x - order.res_pos[0]) ** 2
             + (r.y - order.res_pos[1]) ** 2) ** 0.5
        return (r.pending_count(), d, r.rider_id)

    return min(riders, key=key)


def assign_order(order):
    """Assign order to best rider."""
    rider = find_best_rider_for_order(order)
    if rider is None:
        return None
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    return rider

"""System initialization and main loop."""
import globalvar
from rider import Rider
from order import assign_order
from order_model import Order


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])


def maybe_recruit():
    """Recruit a rider if funds >= 300 and pending > idle riders."""
    riders = globalvar.get_value("riders")
    orders = globalvar.get_value("orders")
    pending = sum(1 for o in orders if o.rider_id is None)
    idle = sum(1 for r in riders if r.pending_count() == 0)
    if globalvar.get_value("funds") >= 300 and pending > idle:
        rid = globalvar.get_value("rider_id_counter") + 1
        globalvar.set_value("rider_id_counter", rid)
        globalvar.set_value("funds", globalvar.get_value("funds") - 500)
        riders.append(Rider(rid))
        return True
    return False


def submit_order(coords):
    """Create an order, recruit if needed, assign. Returns (order, rider, recruited)."""
    order = Order(coords[:2], coords[2:])
    globalvar.get_value("orders").append(order)
    recruited = maybe_recruit()
    rider = assign_order(order)
    return order, rider, recruited


def tick():
    """Advance simulation one tick."""
    globalvar.set_value("time", globalvar.get_value("time") + 1)
    for r in globalvar.get_value("riders"):
        r.move()

"""Command-line entry point."""
import globalvar
import system

HELP = "Commands: start, status, orders, riders, tick [n], help, quit/exit"


def parse_coords(text):
    """Parse four integers in 0-1000."""
    parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
    if len(parts) != 4:
        raise ValueError("need exactly four integers")
    vals = [int(p) for p in parts]
    for v in vals:
        if not 0 <= v <= 1000:
            raise ValueError("coordinates must be within 0-1000")
    return vals


def render_map(points_route=None):
    """ASCII map of 16x16."""
    g = [["#" if (i % 2 == 0 or j % 2 == 0) else "." for j in range(16)]
         for i in range(16)]
    for o in globalvar.get_value("orders"):
        if o.status != "completed":
            rx, ry = o.res_pos
            cx, cy = o.cus_pos
            g[min(15, ry // 50)][min(15, rx // 50)] = "S"
            g[min(15, cy // 50)][min(15, cx // 50)] = "C"
    for r in globalvar.get_value("riders"):
        g[min(15, r.y // 50)][min(15, r.x // 50)] = "R"
    return "\n".join("".join(row) for row in g)


def show_status():
    print("Time:", globalvar.get_value("time"),
          "Funds:", globalvar.get_value("funds"))
    riders = globalvar.get_value("riders")
    print("Riders:", len(riders))
    names = {0: "idle", 1: "picking up", 2: "delivering"}
    for r in riders:
        print("  #%d at (%d,%d) %s orders=%s" % (
            r.rider_id, r.x, r.y, names[r.state], [o.id for o in r.orders]))
    print(render_map())


def show_orders():
    for o in globalvar.get_value("orders"):
        t = (o.end_time - o.start_time) if o.end_time is not None else "-"
        print("Order %d start=%s from=%s to=%s dist=%d time=%s rider=%s %s" % (
            o.id, o.start_time, o.res_pos, o.cus_pos, o.distance, t,
            o.rider_id, o.status))


def show_riders():
    for r in globalvar.get_value("riders"):
        print("Rider", r.rider_id)
        for e in r.log:
            print("  t=%s (%s,%s) %s order=%s" % (
                e["time"], e["x"], e["y"], e["event"], e["order_id"]))


def main():
    system.init_system()
    print(HELP)
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        parts = line.split(None, 1)
        if not parts:
            system.tick()
            continue
        cmd = parts[0].lower()
        try:
            if cmd in ("quit", "exit", "q"):
                break
            elif cmd in ("start", "s"):
                text = parts[1] if len(parts) > 1 else input(
                    "Enter restaurant x y customer x y: ")
                order, rider, rec = system.submit_order(parse_coords(text))
                print("Restaurant:", order.res_pos, "Customer:", order.cus_pos)
                print("New rider added:", rec)
                print("Assigned rider:", rider.rider_id if rider else None,
                      "speed:", rider.speed if rider else "-")
                if rider:
                    print("Estimated time:", (abs(rider.x - order.res_pos[0])
                          + abs(rider.y - order.res_pos[1]) + order.distance)
                          // rider.speed + 1, "ticks")
                print(render_map())
            elif cmd == "status":
                show_status()
            elif cmd == "orders":
                show_orders()
            elif cmd == "riders":
                show_riders()
            elif cmd == "tick":
                n = int(parts[1]) if len(parts) > 1 else 1
                for _ in range(n):
                    system.tick()
                show_status()
            elif cmd == "help":
                print(HELP)
            else:
                print("Unknown command. " + HELP)
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
    print("Bye.")


if __name__ == "__main__":
    main()

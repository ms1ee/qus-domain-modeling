"""Global variable storage shared across the whole program."""
import sys

_STORE_NAME = "_food_dispatch_globals"


def _store():
    """Return the single shared dict (survives duplicate module imports)."""
    return sys.__dict__.setdefault(_STORE_NAME, {})


def _init():
    """Reset all global variables to their initial values."""
    s = _store()
    s.clear()
    s.update({
        "funds": 1000,
        "riders": [],
        "rider_id_counter": 0,
        "running": True,
        "tick": 0,
        "orders": [],
        "order_id_counter": 0,
    })


def set_value(key, value):
    """Set a global value."""
    _store()[key] = value


def get_value(key, default=None):
    """Get a global value."""
    return _store().get(key, default)

"""Order data model."""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import globalvar  # noqa: E402

CELL = 100


def to_grid(v):
    """Convert a coordinate to the top-left corner of its grid cell."""
    return (int(v) // CELL) * CELL


class Order:
    """A delivery order: restaurant position -> customer position."""

    def __init__(self, res_pos, cus_pos):
        n = (globalvar.get_value("order_id_counter") or 0) + 1
        globalvar.set_value("order_id_counter", n)
        self.id = n
        self.res_pos = (to_grid(res_pos[0]), to_grid(res_pos[1]))
        self.cus_pos = (to_grid(cus_pos[0]), to_grid(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick") or 0
        self.completion_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    @property
    def time_taken(self):
        """Ticks from creation to completion (None if not done)."""
        if self.completion_time is None:
            return None
        return self.completion_time - self.start_time

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        if self.status == "pending":
            self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "restaurant": list(self.res_pos),
                "customer": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""Rider model and movement."""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import globalvar  # noqa: E402

IDLE, PICKING_UP, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick") or 0,
                         "x": self.x, "y": self.y, "state": self.state})

    def current_order(self):
        """First order not yet completed, or None."""
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def pending_count(self):
        return sum(1 for o in self.orders if o.status != "completed")

    def _step_toward(self, tx, ty):
        """Move along streets (horizontal first, then vertical)."""
        budget = self.speed
        dx = tx - self.x
        if dx and budget:
            m = min(abs(dx), budget)
            self.x += m if dx > 0 else -m
            budget -= m
        dy = ty - self.y
        if dy and budget:
            m = min(abs(dy), budget)
            self.y += m if dy > 0 else -m
            budget -= m

    def move(self):
        """Advance one tick."""
        order = self.current_order()
        if self.state == IDLE:
            if order is None:
                return
            self.state = PICKING_UP
            order.status = "picking_up"
            self._record("accepted_order", order.id)
        if order is None:
            self.state = IDLE
            return
        if self.state == PICKING_UP:
            tx, ty = order.res_pos
            self._step_toward(tx, ty)
            if (self.x, self.y) == (tx, ty):
                self.state = DELIVERING
                order.status = "delivering"
                self._record("arrived_at_merchant", order.id)
            return
        if self.state == DELIVERING:
            tx, ty = order.cus_pos
            self._step_toward(tx, ty)
            if (self.x, self.y) == (tx, ty):
                order.status = "completed"
                order.completion_time = globalvar.get_value("tick") or 0
                self.state = IDLE
                self._record("delivered_to_customer", order.id)

"""Order assignment."""
import math
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import globalvar  # noqa: E402


def find_best_rider_for_order(order):
    """Rider with fewest current orders; ties by distance to restaurant."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None

    def key(r):
        d = math.hypot(r.x - order.res_pos[0], r.y - order.res_pos[1])
        return (r.pending_count(), d, r.rider_id)

    return min(riders, key=key)


def assign_order(order):
    """Assign order to best rider; returns rider or None."""
    rider = find_best_rider_for_order(order)
    if rider is not None:
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
    return rider

"""System manager and initialization."""
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import globalvar  # noqa: E402
from rider import Rider  # noqa: E402
import order as order_mod  # noqa: E402
from order_model import Order  # noqa: E402


def init_system():
    """Initialize global state."""
    globalvar._init()


class SystemManager:
    """Manages funds, riders, orders and the clock."""

    def __init__(self):
        init_system()

    @property
    def funds(self):
        return globalvar.get_value("funds")

    @property
    def riders(self):
        return globalvar.get_value("riders")

    @property
    def orders(self):
        return globalvar.get_value("orders")

    @property
    def tick(self):
        return globalvar.get_value("tick")

    def pending_orders(self):
        return [o for o in self.orders if o.status == "pending"]

    def idle_riders(self):
        return [r for r in self.riders if r.state == 0 and not r.pending_count()]

    def maybe_recruit(self, extra_pending=0):
        """Recruit a rider if funds>=300 and pending > idle."""
        if (self.funds >= 300 and
                len(self.pending_orders()) + extra_pending > len(self.idle_riders())):
            n = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", n)
            self.riders.append(Rider(n))
            globalvar.set_value("funds", self.funds - 500)
            return True
        return False

    def create_order(self, res_pos, cus_pos):
        """Create order, recruit if needed, assign. Returns (order, rider, hired)."""
        o = Order(res_pos, cus_pos)
        self.orders.append(o)
        hired = self.maybe_recruit()
        rider = order_mod.assign_order(o)
        return o, rider, hired

    def step(self):
        """Advance one tick."""
        globalvar.set_value("tick", self.tick + 1)
        for r in self.riders:
            r.move()
        for o in self.orders:
            if o.status == "pending":
                order_mod.assign_order(o)

    def save(self, path="state.json"):
        data = {"funds": self.funds, "tick": self.tick,
                "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                            "speed": r.speed, "state": r.state,
                            "orders": [o.id for o in r.orders]}
                           for r in self.riders],
                "orders": [o.to_dict() for o in self.orders]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

class PathPlanner:
    """Street-following planner with right-angle turns."""

    def plan(self, start, end):
        """Waypoints from start to end: horizontal then vertical."""
        pts = [tuple(start)]
        corner = (end[0], start[1])
        if corner != pts[-1] and corner != tuple(end):
            pts.append(corner)
        if tuple(end) != pts[-1]:
            pts.append(tuple(end))
        return pts

    def plan_multi(self, start, destinations):
        """Continuous path through several destinations."""
        path = [tuple(start)]
        cur = tuple(start)
        for d in destinations:
            path.extend(self.plan(cur, d)[1:])
            cur = tuple(d)
        return path

"""Display manager."""
from path_planner import PathPlanner

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}
SIZE = 16
UNIT = 50


def _cell(v):
    return max(0, min(SIZE - 1, int(v) // UNIT))


class DisplayManager:
    """Textual status and ASCII map rendering."""

    def render_map(self, riders=(), orders=(), route=None):
        grid = [["." for _ in range(SIZE)] for _ in range(SIZE)]
        if route:
            for a, b in zip(route, route[1:]):
                x0, y0 = _cell(a[0]), _cell(a[1])
                x1, y1 = _cell(b[0]), _cell(b[1])
                for x in range(min(x0, x1), max(x0, x1) + 1):
                    for y in range(min(y0, y1), max(y0, y1) + 1):
                        grid[y][x] = "#"
        for o in orders:
            if o.status != "completed":
                grid[_cell(o.res_pos[1])][_cell(o.res_pos[0])] = "S"
                grid[_cell(o.cus_pos[1])][_cell(o.cus_pos[0])] = "C"
        for r in riders:
            grid[_cell(r.y)][_cell(r.x)] = "R"
        return "\n".join(" ".join(row) for row in grid)

    def route_text(self, rider, order):
        pl = PathPlanner()
        route = pl.plan_multi((rider.x, rider.y), [order.res_pos, order.cus_pos])
        length = sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                     for a, b in zip(route, route[1:]))
        t = -(-length // rider.speed)
        return (self.render_map([rider], [order], route)
                + "\nRoute: %s\nSpeed: %d units/tick, estimated time: %d ticks"
                % (" -> ".join(map(str, route)), rider.speed, t))

    def status_text(self, system):
        lines = ["Tick: %d  Funds: %d  Riders: %d"
                 % (system.tick, system.funds, len(system.riders))]
        for r in system.riders:
            lines.append("Rider %d at (%d,%d) state=%s orders=%s"
                         % (r.rider_id, r.x, r.y, STATE_NAMES[r.state],
                            [(o.id, o.status) for o in r.orders]))
        lines.append(self.render_map(system.riders, system.orders))
        return "\n".join(lines)

    def orders_text(self, system):
        if not system.orders:
            return "No orders."
        return "\n".join(
            "Order %d start=%d restaurant=%s customer=%s distance=%s time=%s rider=%s status=%s"
            % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
               o.time_taken, o.rider_id, o.status) for o in system.orders)

    def riders_text(self, system):
        lines = []
        for r in system.riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%d (%d,%d) state=%d event=%s order=%s"
                             % (e["time"], e["x"], e["y"], e["state"],
                                e["event"], e["order_id"]))
        return "\n".join(lines) or "No riders."

class InputManager:
    """Parses commands and validates coordinates."""

    ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
               "q": "quit", "exit": "quit", "quit": "quit", "start": "start",
               "status": "status", "orders": "orders", "riders": "riders"}

    def __init__(self):
        self.history = []

    def parse_command(self, text):
        text = text.strip().lower()
        if text:
            self.history.append(text)
        return self.ALIASES.get(text)

    def parse_coordinates(self, text):
        """Return 4 ints or raise ValueError with clear message."""
        parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(parts) != 4:
            raise ValueError("Please enter exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        for v in vals:
            if not 0 <= v <= 1000:
                raise ValueError("Coordinates must be within 0-1000.")
        return vals

"""Command-line entry point."""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import globalvar  # noqa: E402
from system import SystemManager  # noqa: E402
from display import DisplayManager  # noqa: E402
from input_manager import InputManager  # noqa: E402


def main():
    system = SystemManager()
    display = DisplayManager()
    inp = InputManager()
    print("Food Delivery Dispatch. Commands: start, status, orders, riders, quit")
    while globalvar.get_value("running"):
        try:
            cmd = inp.parse_command(input("> "))
        except (EOFError, KeyboardInterrupt):
            break
        try:
            if cmd == "quit":
                globalvar.set_value("running", False)
                break
            elif cmd == "start":
                c = inp.parse_coordinates(input("restaurant_x restaurant_y customer_x customer_y: "))
                o, rider, hired = system.create_order((c[0], c[1]), (c[2], c[3]))
                print("Restaurant grid: %s, Customer grid: %s" % (o.res_pos, o.cus_pos))
                print("New rider added: %s" % hired)
                if rider:
                    print(display.route_text(rider, o))
                else:
                    print("No rider available; order pending.")
            elif cmd == "status":
                print(display.status_text(system))
            elif cmd == "orders":
                print(display.orders_text(system))
            elif cmd == "riders":
                print(display.riders_text(system))
            else:
                print("Unknown command.")
        except ValueError as e:
            print("Error: %s" % e)
        except Exception as e:  # recover gracefully
            print("Unexpected error: %s" % e)
        system.step()
    try:
        system.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

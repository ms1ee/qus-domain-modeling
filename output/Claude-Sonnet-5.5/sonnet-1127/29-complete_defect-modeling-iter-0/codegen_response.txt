"""Grid map and path planning."""

CELL = 100
DIM = 8


class GridMap:
    """8x8 grid of 100-unit cells."""

    def __init__(self):
        self.width = 800
        self.height = 800
        self.cellSize = CELL
        self.gridDimension = DIM

    @staticmethod
    def center(v):
        """Convert a coordinate to its grid cell center."""
        v = max(0, min(799, int(v)))
        return (v // CELL) * CELL + 50


class PathPlanner:
    """Right-angle path planning."""

    def __init__(self, grid=None):
        self.grid = grid or GridMap()

    def plan(self, start, end):
        """Return corner points from start to end (horizontal first)."""
        pts = [tuple(start)]
        if start[0] != end[0] and start[1] != end[1]:
            pts.append((end[0], start[1]))
        if tuple(end) != pts[-1]:
            pts.append(tuple(end))
        return pts

    def next_position(self, pos, target, speed):
        """Advance from pos toward target by at most speed units."""
        x, y = pos
        tx, ty = target
        if x != tx:
            step = min(speed, abs(tx - x))
            x += step if tx > x else -step
        elif y != ty:
            step = min(speed, abs(ty - y))
            y += step if ty > y else -step
        return x, y

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Order data model."""
import itertools

import globalvar
from pathplanner import GridMap, PathPlanner

_ids = itertools.count(1)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_ids)
        self.res_pos = (GridMap.center(res_pos[0]), GridMap.center(res_pos[1]))
        self.cus_pos = (GridMap.center(cus_pos[0]), GridMap.center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)
        self.time_taken = None

    def assign_to(self, rider_id):
        """Record the rider this order is assigned to."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick):
        """Mark as completed."""
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time

"""Rider model."""
import globalvar
from pathplanner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self.planner = PathPlanner()
        self._record("spawn", None)

    @property
    def pos(self):
        return (self.x, self.y)

    def _record(self, event, order_id):
        self.log.append({
            "event": event, "order_id": order_id,
            "time": globalvar.get_value("tick", 0) or 0,
            "x": self.x, "y": self.y, "state": self.state,
        })

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._record("accepted_order", self.orders[0].id)
            return
        order = self.orders[0]
        target = order.res_pos if self.state == 1 else order.cus_pos
        if self.pos != target:
            self.x, self.y = self.planner.next_position(
                self.pos, target, self.speed)
        if self.pos == target:
            if self.state == 1:
                self.state = 2
                self._record("arrived_at_merchant", order.id)
            else:
                order.complete(globalvar.get_value("tick", 0) or 0)
                self.state = 0
                self.orders.pop(0)
                self._record("delivered_to_customer", order.id)

"""Order assignment."""
import globalvar
from pathplanner import PathPlanner


def find_best_rider_for_order(order):
    """Return the closest rider (ties: fewer orders), or None."""
    riders = globalvar.get_value("riders", []) or []
    best = None
    best_key = None
    for r in riders:
        key = (PathPlanner.distance(r.pos, order.res_pos), len(r.orders))
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


def assign_order(order):
    """Assign order to best rider; returns the rider or None."""
    rider = find_best_rider_for_order(order)
    if rider is not None:
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
    return rider

"""System initialization and management."""
import globalvar
from rider import Rider

RIDER_COST = 300


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", 0)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])


def recruit_rider():
    """Create a new rider if funds allow; return it or None."""
    funds = globalvar.get_value("funds", 0)
    if funds < RIDER_COST:
        return None
    globalvar.set_value("funds", funds - RIDER_COST)
    rid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    rider = Rider(rid)
    globalvar.get_value("riders").append(rider)
    return rider


def tick():
    """Advance simulation one tick."""
    globalvar.set_value("tick", globalvar.get_value("tick", 0) + 1)
    for r in globalvar.get_value("riders"):
        r.move()

"""Display helpers."""
import globalvar
from pathplanner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and map rendering."""

    @staticmethod
    def map_text(riders=(), orders=(), route=None):
        grid = [["." for _ in range(8)] for _ in range(8)]
        if route:
            for a, b in zip(route, route[1:]):
                for i in range(0, 101, 10):
                    px = a[0] + (b[0] - a[0]) * i // 100
                    py = a[1] + (b[1] - a[1]) * i // 100
                    grid[min(7, py // 100)][min(7, px // 100)] = "#"
        for o in orders:
            grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
            grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        for r in riders:
            grid[min(7, r.y // 100)][min(7, r.x // 100)] = "R"
        return "\n".join(" ".join(row) for row in grid)

    def status(self):
        riders = globalvar.get_value("riders")
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        lines.append(self.map_text(
            riders, [o for r in riders for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        orders = globalvar.get_value("orders")
        if not orders:
            return "No orders."
        lines = []
        for o in orders:
            lines.append(
                "Order %d start=%d rest=%s cust=%s dist=%d time=%s rider=%s [%s]"
                % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                   o.time_taken, o.rider_id, o.status))
        return "\n".join(lines)

    def riders(self):
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("Rider %d t=%d (%d,%d) %s order=%s" % (
                    r.rider_id, e["time"], e["x"], e["y"], e["event"],
                    e["order_id"]))
        return "\n".join(lines) or "No riders."

"""Command parsing and input validation."""
import globalvar
import system
from display import DisplayManager
from order import assign_order
from order_model import Order
from pathplanner import PathPlanner

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "t": "tick"}


class InputManager:
    """Handles commands."""

    def __init__(self):
        self.commandHistory = []
        self.display = DisplayManager()
        self.planner = PathPlanner()

    @staticmethod
    def parse_coords(text):
        """Parse four ints in 0-800; raises ValueError."""
        parts = text.replace(",", " ").replace("[", " ").replace(
            "]", " ").split()
        if len(parts) != 4:
            raise ValueError("Need exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be in 0-800.")
        return vals

    def start_order(self, vals):
        order = Order((vals[0], vals[1]), (vals[2], vals[3]))
        globalvar.set_value("funds", globalvar.get_value("funds") + 300)
        globalvar.get_value("orders").append(order)
        added = False
        riders = globalvar.get_value("riders")
        if not any(r.state == 0 and not r.orders for r in riders):
            added = system.recruit_rider() is not None
        rider = assign_order(order)
        if rider is None:
            return "No rider available."
        route = self.planner.plan(rider.pos, order.res_pos)
        route += self.planner.plan(order.res_pos, order.cus_pos)[1:]
        total = sum(PathPlanner.distance(a, b)
                    for a, b in zip(route, route[1:]))
        eta = -(-total // rider.speed)
        return "\n".join([
            "Restaurant grid: (%d,%d)  Customer grid: (%d,%d)" % (
                order.res_pos[0] // 100, order.res_pos[1] // 100,
                order.cus_pos[0] // 100, order.cus_pos[1] // 100),
            "New rider added: %s" % added,
            "Order %d -> rider %d, speed %d, distance %d, est. time %d ticks"
            % (order.id, rider.rider_id, rider.speed, total, eta),
            DisplayManager.map_text([rider], [order], route)])

    def handle(self, line):
        """Process a command line; return output text."""
        line = line.strip()
        if not line:
            return ""
        self.commandHistory.append(line)
        parts = line.split(None, 1)
        cmd = ALIASES.get(parts[0].lower(), parts[0].lower())
        arg = parts[1] if len(parts) > 1 else ""
        try:
            if cmd == "start":
                return self.start_order(self.parse_coords(arg))
            if cmd == "status":
                return self.display.status()
            if cmd == "orders":
                return self.display.orders()
            if cmd == "riders":
                return self.display.riders()
            if cmd == "tick":
                n = int(arg) if arg else 1
                for _ in range(max(0, n)):
                    system.tick()
                return "Advanced to tick %d" % globalvar.get_value("tick")
            if cmd == "history":
                return "\n".join(self.commandHistory)
            if cmd in ("quit", "exit"):
                globalvar.set_value("running", False)
                return "Bye."
            return "Unknown command: %s" % cmd
        except ValueError as e:
            return "Error: %s" % e

"""Entry point: python src/main.py"""
import globalvar
import system
from input_manager import InputManager


def main():
    system.init_system()
    im = InputManager()
    print("Commands: start x1 y1 x2 y2 | status | orders | riders | "
          "tick [n] | history | quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        out = im.handle(line)
        if out:
            print(out)
        if globalvar.get_value("running"):
            system.tick()


if __name__ == "__main__":
    main()

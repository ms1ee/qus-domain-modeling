from lottery_system.models import User

DEFAULT_DATA = [
    ("Li Bin", "Technology Department"),
    ("Zhang San", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Chen Qixian", "Product Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Zhao Liu", "Marketing Department"),
    ("Sun Qi", "Product Department"),
    ("Cheng Ru", "Cheng Ru group"),
    ("Cheng Yi", "Cheng Ru group"),
    ("Cheng Er", "Cheng Ru group"),
    ("Zhou Ba", "Finance Department"),
    ("Wu Jiu", "Finance Department"),
    ("Zheng Shi", "HR Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error: cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, qty in defaults:
            try:
                self.add_prize(name, qty)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        self.prizes.append(Prize(name, quantity, list(avoid_groups or []), associated_drawer))

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHEN = "Chen Qixian"
JIN = "Jin Duo"
MA = "Ma Jiaqi"
SPECIAL = "Li Bin"


def is_cheng_ru(dept):
    return "cheng ru" in dept.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {u.name for u in self.winners}
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == MA and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if is_cheng_ru(u.department)]
        if drawer == SPECIAL:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        w = random.choice(pool)
        self.winners.append(w)
        return w

    def finish_round(self, prize, round_winners):
        if prize.associated_drawer in (CHEN, JIN):
            self.last_two_draw_results.append(
                any(is_cheng_ru(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize):
        self.user = user
        self.prize = prize


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize is p or
                              getattr(w.prize, "name", w.prize) == p.name]
        return result

    def export_to_csv(self, prizes):
        grouped = self.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            return False, "No winners to export"
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for pname, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{pname}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import sys
import time

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def poll_command():
    try:
        import select
        r, _, _ = select.select([sys.stdin], [], [], 0)
        if r:
            return sys.stdin.readline().strip().lower()
    except Exception:
        pass
    return None


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        if errs and c == "2":
            a = input("Errors found. (r)emove invalid and continue / (t)erminate: ").strip().lower()
            if a == "t":
                continue
        users, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name} ({d.department})")
        return users


def run_round(idx, prize, lot, results):
    print(f"\n=== Round {idx}: {prize.name} x{prize.quantity} ===")
    rule = f"Avoid groups: {', '.join(prize.avoid_groups) or 'none'}; drawer: {prize.associated_drawer or 'n/a'}"
    print("Rules:", rule)
    print("(type pause / continue / terminate during drawing)")
    won = []
    for i in range(1, prize.quantity + 1):
        cmd = poll_command()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            print("Paused. Type 'continue' to resume or 'terminate'.")
            while True:
                c = input("> ").strip().lower()
                if c in ("continue", "terminate"):
                    cmd = c
                    break
            if cmd == "terminate":
                print("Round terminated.")
                break
        print(f"Currently drawing the {i}th winner...")
        time.sleep(0.3)
        w = lot.draw_one(prize)
        if w is None:
            print("No eligible participants left.")
            break
        won.append(w)
        results.add_winner(w, prize)
        print(f"Congratulations {w.name} ({w.department}) for winning {prize.name}")
    lot.finish_round(prize, won)


def main():
    lmgr = ListManager()
    users = load_users(lmgr)
    print("Groups:")
    for g, c in lmgr.get_group_info(users).items():
        print(f"  {g}: {c}")
    pm = PrizeManager()
    if input("Load preset prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            q = int(input("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        av = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
        dr = input("Drawer (optional): ").strip() or None
        pm.add_prize(name, q, av, dr)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    o = input("New order (e.g. 3 1 2) or Enter to keep: ").split()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o]):
                print("Invalid order, keeping default.")
        except ValueError:
            print("Invalid order, keeping default.")
    lot = LotteryManager(users)
    results = ResultManager()
    for i, p in enumerate(pm.get_prizes(), 1):
        run_round(i, p, lot, results)
    print("\n=== Final winners ===")
    for pname, ws in results.get_winners_by_prize(pm.get_prizes()).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {pname}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(results.export_to_csv(pm.get_prizes())[1])


if __name__ == "__main__":
    main()

"""Graph data structures."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph; node ids are station numbers."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""Dijkstra shortest path with a heap based priority queue."""
import heapq
import itertools

TRANSFER_PENALTY = 1000.0
PREFERENCE_FACTOR = 0.8


class MinPQ:
    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._transfers = {}
        self._final = {}
        self._parent = {}
        self._run()

    def _cost(self, eff, dist, tr):
        mode = self.config.mode
        if mode == "least_transfer":
            return (tr, dist)
        if mode == "comprehensive":
            return (eff + tr * TRANSFER_PENALTY,)
        return (dist,)

    def _run(self):
        cfg = self.config
        limit = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        best = {start: self._cost(0, 0, 0)}
        self._parent[start] = (None, None)
        pq = MinPQ()
        pq.push(best[start], (start, 0.0, 0))
        done = set()
        while not pq.is_empty():
            cost, (state, eff, dist) = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line, tr = state
            if node not in self._final:
                self._final[node] = state
                self.dist_to[node] = dist
                self._transfers[node] = tr
            for e in self.graph.neighbors(node):
                ntr = tr + (1 if line is not None and e.line != line else 0)
                if limit is not None and ntr > limit:
                    continue
                w = e.weight * (PREFERENCE_FACTOR if e.line in pref else 1.0)
                neff = eff + w
                ndist = dist + e.weight
                if limit is not None:
                    ns = (e.dest, e.line, ntr)
                else:
                    ns = (e.dest, e.line, 0)
                    ntr_key = ntr
                c = self._cost(neff, ndist, ntr)
                if ns in done:
                    continue
                if ns not in best or c < best[ns]:
                    best[ns] = c
                    self._parent[ns] = (state, e)
                    pq.push(c, (ns, neff, ndist))
        # with no transfer limit, state transfers are not tracked in key;
        # recompute transfers along reconstructed paths
        self._recompute = limit is None

    def has_path_to(self, target):
        return target in self._final

    def path_with_edges(self, target):
        if target not in self._final:
            return [], []
        state = self._final[target]
        edges = []
        while state is not None:
            parent, e = self._parent[state]
            if e is not None:
                edges.append(e)
            state = parent
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                count += 1
        return count

"""Fare calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (left-closed, right-open tiers)."""
    if distance is None or isinstance(distance, bool):
        raise ValueError("invalid distance")
    d = float(distance)
    if d != d or d < 0:
        raise ValueError("invalid distance")
    d = math.ceil(d)
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5.0

"""Network construction, integrity checks and data updates."""
from models import Edge, Graph

REPRESENTATIVE = {
    "Line 1": ["Tiananmen East"],
}


def build_graph(data):
    """Return (graph, station_to_number, number_to_station)."""
    graph = Graph()
    s2n = {}
    n2s = {}
    for st in data.get("stations", []):
        if st["name"] not in s2n:
            n = len(s2n)
            s2n[st["name"]] = n
            n2s[n] = st["name"]
            graph.add_node(n)
        graph.set_station_lines(st["name"], st.get("lines", []))
    for c in data.get("connections", []):
        for nm in (c["source"], c["dest"]):
            if nm not in s2n:
                n = len(s2n)
                s2n[nm] = n
                n2s[n] = nm
                graph.add_node(n)
                graph.set_station_lines(nm, [c["line"]])
        graph.add_edge(Edge(s2n[c["source"]], s2n[c["dest"]], c["distance"], c["line"]))
    return graph, s2n, n2s


def check_data(data):
    """Return list of problem strings (empty if OK)."""
    problems = []
    names = set()
    for st in data.get("stations", []):
        if not st.get("name"):
            problems.append("Station without name")
        if not st.get("lines"):
            problems.append("Station %s has no lines" % st.get("name"))
        names.add(st.get("name"))
    if not names:
        problems.append("No stations")
    linked = set()
    for c in data.get("connections", []):
        for k in ("source", "dest"):
            if c[k] not in names:
                problems.append("Connection references unknown station %s" % c[k])
            linked.add(c[k])
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0:
            problems.append("Bad distance %s-%s: %r" % (c["source"], c["dest"], d))
        elif d > 20000:
            problems.append("Suspicious distance %s-%s: %r" % (c["source"], c["dest"], d))
        if c["source"] == c["dest"]:
            problems.append("Self loop at %s" % c["source"])
    for n in names - linked:
        problems.append("Station %s has no connections" % n)
    lines = set()
    for st in data.get("stations", []):
        lines.update(st.get("lines", []))
    for c in data.get("connections", []):
        lines.add(c["line"])
    for ln, sts in REPRESENTATIVE.items():
        if ln in lines:
            for s in sts:
                if s not in names:
                    problems.append("Representative station %s missing on %s" % (s, ln))
    return problems


def add_line(data, line_name, station_names, distances):
    """Add a new line; distances has len(station_names)-1 entries."""
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances length mismatch")
    existing = {s["name"]: s for s in data["stations"]}
    for nm in station_names:
        if nm in existing:
            if line_name not in existing[nm]["lines"]:
                existing[nm]["lines"].append(line_name)
        else:
            s = {"name": nm, "lines": [line_name]}
            data["stations"].append(s)
            existing[nm] = s
    for a, b, d in zip(station_names, station_names[1:], distances):
        data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})


def update_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    ok = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
            c["distance"] = distance
            ok = True
    return ok

"""Route report generation (text and JSON)."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


def _fmt_distance(m):
    if m < 1000:
        return "%d m" % m
    return "%.2f km" % (m / 1000.0)


def build_summary(start, end, nodes, edges, n2s):
    total = sum(e.distance for e in edges)
    stations = [n2s[n] for n in nodes]
    segments = []
    transfers = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = n2s[e.dest]
            segments[-1]["distance"] += e.distance
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfers.append({"station": n2s[e.source],
                                  "from_line": segments[-1]["line"], "to_line": e.line})
            segments.append({"line": e.line, "from": n2s[e.source], "to": n2s[e.dest],
                             "distance": e.distance, "stations": 1})
    tc = len(transfers)
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    return {
        "query": {"start": start, "end": end},
        "path": stations,
        "distance_meters": total,
        "distance_km": round(total / 1000.0, 2),
        "distance_text": _fmt_distance(total),
        "fare": Sum_money(total),
        "estimated_time_minutes": round(calculate_estimated_time(total, tc), 1),
        "transfer_count": tc,
        "transfers": transfers,
        "lines": lines,
        "segments": segments,
    }


def format_text(s):
    out = ["Route: %s -> %s" % (s["query"]["start"], s["query"]["end"]),
           "Stations (%d): %s" % (len(s["path"]), " -> ".join(s["path"])),
           "Total distance: %s" % s["distance_text"]]
    for seg in s["segments"]:
        out.append("  [%s] %s -> %s (%d stops, %s)" % (
            seg["line"], seg["from"], seg["to"], seg["stations"], _fmt_distance(seg["distance"])))
    out.append("Transfers: %d" % s["transfer_count"])
    for t in s["transfers"]:
        out.append("  Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    out.append("Lines: %s" % ", ".join(s["lines"]))
    out.append("Estimated time: %.1f minutes" % s["estimated_time_minutes"])
    out.append("Fare: %d yuan" % s["fare"])
    return "\n".join(out)


def format_json(s):
    return json.dumps(s, ensure_ascii=False, indent=2)

"""CLI for Beijing Subway fare calculation."""
import argparse
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import build_graph, check_data
from pathfinding import PathfindingConfig, ShortestPath
from report import build_summary, format_json, format_text

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_query(text):
    """Return (start, end) or raise ValueError with a message."""
    text = text.replace("；", ";")
    parts = [p.strip() for p in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERR)
    return parts[0], parts[1]


def run_query(text, graph, s2n, n2s, config, fmt):
    start, end = parse_query(text)
    if start not in s2n:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % start)
    if end not in s2n:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % end)
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    try:
        sp = ShortestPath(graph, s2n[start], config)
        if s2n[end] not in sp.dist_to:
            raise ValueError("No route found between '%s' and '%s' under current constraints" % (start, end))
        nodes, edges = sp.path_with_edges(s2n[end])
    except ValueError:
        raise
    except Exception as ex:
        raise ValueError("Path calculation failed: %s" % ex)
    try:
        summary = build_summary(start, end, nodes, edges, n2s)
    except Exception as ex:
        raise ValueError("Fare calculation failed: %s" % ex)
    return format_json(summary) if fmt == "json" else format_text(summary)


def main(argv=None):
    ap = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    ap.add_argument("--output_file", default=None)
    ap.add_argument("--format", choices=["text", "json"], default="text")
    ap.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                    default="shortest_distance")
    ap.add_argument("--max_transfers", type=int, default=None)
    ap.add_argument("--line_preference", default=None)
    ap.add_argument("--check_data", action="store_true")
    args = ap.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except Exception as ex:
        print("Failed to load data: %s" % ex)
        return 1

    if args.check_data:
        problems = check_data(data)
        if problems:
            print("Data check found %d problem(s):" % len(problems))
            for p in problems:
                print(" - " + p)
            return 1
        print("Data check passed: %d stations, %d connections" % (
            len(data["stations"]), len(data["connections"])))
        return 0

    graph, s2n, n2s = build_graph(data)
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    if args.line_preference:
        config.preferred_lines = [args.line_preference]

    print("Enter 'Starting Station; Ending Station' (empty line or 'q' to quit)")
    while True:
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if not line or line.lower() in ("q", "quit", "exit"):
            break
        try:
            result = run_query(line, graph, s2n, n2s, config, args.format)
        except ValueError as ex:
            print(ex)
            continue
        print(result)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(result + "\n\n")
            except OSError as ex:
                print("Cannot write output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

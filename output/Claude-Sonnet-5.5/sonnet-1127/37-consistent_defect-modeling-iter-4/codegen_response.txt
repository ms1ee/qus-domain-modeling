from lottery_system.models import User

DEFAULT_PEOPLE = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Product Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Cheng Ru group"),
    ("Zhou Ba", "Cheng Ru group"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Li Bin", "Operations Department"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Finance Department"),
    ("Ma Jiaqi", "Finance Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_PEOPLE], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error: cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            norm = line.replace(";", ",").replace("；", ",").replace("，", ",")
            parts = [p.strip() for p in norm.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in [("Third Prize - Movie Tickets", 3),
                          ("Second Prize - Shopping Cards", 2),
                          ("First Prize - 1 Day Annual Leave", 1)]:
            self.prizes.append(Prize(name, qty, [], None))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        self.prizes.append(Prize(name, quantity, list(avoid_groups or []), associated_drawer))

    def get_prizes(self):
        return self.prizes

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU_GROUPS = {"Cheng Ru group", "Cheng Ru", "Cheng Ru Group"}
TRIGGER_DRAWERS = ("Chen Qixian", "Jin Duo")
TARGET_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _winner_names(self):
        return {u.name for u in self.winners}

    def _get_eligible_pool(self, prize):
        won = self._winner_names()
        base = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return base
        if (drawer == TARGET_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            pool = [u for u in base if u.department in CHENG_RU_GROUPS]
            if pool:
                return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in base if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def record_round(self, prize, round_winners):
        if prize.associated_drawer in TRIGGER_DRAWERS:
            has = any(u.department in CHENG_RU_GROUPS for u in round_winners)
            self.last_two_draw_results.append(has)
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def describe_rules(self, prize):
        parts = []
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("Only previous winners excluded; avoidance groups not applied")
        else:
            parts.append("Previous winners excluded")
            if prize.avoid_groups:
                parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer == TARGET_DRAWER:
            parts.append("fallback: Cheng Ru group protection may apply")
        return "; ".join(parts)

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize, order=0, time=None):
        self.user = user
        self.prize = prize
        self.order = order
        self.time = time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize, len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners
                              if w.prize is p or getattr(w.prize, "name", w.prize) == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users():
    lm = ListManager()
    while True:
        c = input("Load list: 1) default 2) custom txt file > ").strip()
        if c == "2":
            users, errors = lm.load_custom_list(input("Path: ").strip())
        else:
            users, errors = lm.load_default_list()
        for e in errors:
            print(e)
        if errors and input("Errors found. Remove invalid data and continue? (y/n) ").strip().lower() != "y":
            if c == "2":
                print("Upload terminated.")
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name}")
        if not valid:
            print("No valid users.")
            continue
        print("Groups:")
        for g, n in lm.get_group_info(valid).items():
            print(f"  {g}: {n}")
        return valid


def setup_prizes(groups):
    pm = PrizeManager()
    if input("Load preset prizes? (y/n) ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n) ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            qty = int(input("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        if qty <= 0 or not name:
            print("Invalid prize")
            continue
        drawer = input("Associated lottery person (blank for none): ").strip() or None
        avoid = [a.strip() for a in input("Avoid groups (comma separated): ").split(",") if a.strip()]
        pm.add_prize(name, qty, avoid, drawer)
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        print(f"{i}. {p.name} x{p.quantity}")
    order = input("New order (e.g. 3 1 2), blank to keep: ").split()
    if order and not pm.reorder_prizes(order):
        print("Invalid order, keeping original.")
    return pm


def run_round(lot, rm, prize):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | rules: {lot.describe_rules(prize)}")
    won = []
    paused = False
    i = 1
    while i <= prize.quantity:
        cmd = input("Enter to draw (pause/continue/terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            paused = True
            print("Paused. Type 'continue' to resume.")
            continue
        if paused:
            if cmd == "continue":
                paused = False
                print("Resumed.")
            continue
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible participants remain.")
            break
        rm.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.record_round(prize, won)
    print("Round results:")
    for n, u in enumerate(won, 1):
        print(f"  {n}. {u.name} ({u.department})")


def main():
    users = load_users()
    pm = setup_prizes(set(u.department for u in users))
    prizes = pm.get_prizes()
    lot = LotteryManager(users)
    rm = ResultManager()
    for p in prizes:
        run_round(lot, rm, p)
    print("\n=== Final winners (high to low prize) ===")
    for name, ws in rm.get_winners_by_prize(list(reversed(prizes))).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    ok, msg = rm.export_to_csv(list(reversed(prizes)))
    print(msg)


if __name__ == "__main__":
    main()

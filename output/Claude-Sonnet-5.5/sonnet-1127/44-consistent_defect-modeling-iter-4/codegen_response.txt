"""Core data models: Edge and Graph (weighted directed graph)."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted directed graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, number):
        self.adjacency.setdefault(number, [])

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, [])

    def neighbors(self, number):
        return self.adjacency.get(number, [])

    def nodes(self):
        return list(self.adjacency.keys())

"""MinPQ and ShortestPath (Dijkstra over (station, line) states)."""
import heapq
import itertools

MODE_SHORTEST = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"

TRANSFER_PENALTY = {
    MODE_SHORTEST: 0,
    MODE_LEAST_TRANSFER: 1000000,
    MODE_COMPREHENSIVE: 1500,
}


class PathfindingConfig:
    def __init__(self):
        self.mode = MODE_SHORTEST
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Heap based min priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    def is_empty(self):
        return not self._heap


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._prev = {}   # state -> (prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        track = cfg.max_transfers is not None
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line, t = state
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = real[state]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0
                if line is not None and e.line != line:
                    extra = penalty
                    if track:
                        nt = t + 1
                        if nt > cfg.max_transfers:
                            continue
                w = e.weight
                if e.line in cfg.preferred_lines:
                    w *= 0.9
                ns = (e.dest, e.line, nt if track else 0)
                nc = c + w + extra
                if ns not in done and nc < cost.get(ns, float("inf")):
                    cost[ns] = nc
                    real[ns] = real[state] + e.distance
                    self._prev[ns] = (state, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while state in self._prev:
            state, e = self._prev[state]
            edges.append(e)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (fractions rounded up)."""
    if distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    if d <= 6000:
        return 3
    if d <= 12000:
        return 4
    if d < 22000:
        return 5
    if d <= 32000:
        return 6
    return 7 + (d - 32001) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + 5.0 * transfer_count

"""Data management: loading, graph building, integrity check, updates."""
import json

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE = {
    "Line 1": "苹果园", "Line 2": "西直门", "Line 4": "安河桥北", "Line 5": "宋家庄",
    "Line 10": "知春路", "Line 13": "西二旗",
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.build()

    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        names = [s["name"] for s in self.data.get("stations", [])]
        for c in self.data.get("connections", []):
            names.extend([c["source"], c["dest"]])
        for n in names:
            if n not in self.station_to_number:
                num = len(self.station_to_number)
                self.station_to_number[n] = num
                self.number_to_station[num] = n
                self.graph.add_node(num)
        for s in self.data.get("stations", []):
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            self.graph.add_edge(Edge(self.station_to_number[c["source"]],
                                     self.station_to_number[c["dest"]],
                                     c["distance"], c["line"]))

    def station_names(self):
        return list(self.station_to_number.keys())

    def lines_of(self, name):
        return self.graph.station_lines.get(name, [])

    def line_names(self):
        res = set()
        for s in self.data.get("stations", []):
            res.update(s.get("lines", []))
        return sorted(res)

    def is_transfer(self, name):
        return len(self.lines_of(name)) > 1

    def check_data(self):
        """Return (ok, list of messages)."""
        msgs = []
        names = {s["name"] for s in self.data.get("stations", [])}
        lines = set(self.line_names())
        for line, st in REPRESENTATIVE.items():
            if line in lines and st not in names:
                msgs.append("Representative station '%s' of %s missing" % (st, line))
        if not names:
            msgs.append("No stations defined")
        for c in self.data.get("connections", []):
            for k in ("source", "dest"):
                if c[k] not in names:
                    msgs.append("Connection refers to unknown station '%s'" % c[k])
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                msgs.append("Invalid distance %r between %s and %s" % (d, c["source"], c["dest"]))
            elif d > 30000:
                msgs.append("Suspicious distance %r between %s and %s" % (d, c["source"], c["dest"]))
        connected = set()
        for c in self.data.get("connections", []):
            connected.add(c["source"])
            connected.add(c["dest"])
        for n in names - connected:
            msgs.append("Station '%s' has no connections" % n)
        return (not msgs), msgs

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: len(stations)-1 values (bidirectional)."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 entries")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
            self.data["connections"].append({"source": b, "dest": a, "distance": d, "line": line_name})
        self.build()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build()
                return True
        return False

    def update_distance(self, source, dest, distance, line=None):
        found = False
        for c in self.data["connections"]:
            if c["source"] == source and c["dest"] == dest and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self.build()
        return found

    def save(self, path=None):
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)

"""Input validation."""
import difflib
import re


class InputValidator:
    def __init__(self, station_names):
        self.station_names = list(station_names)
        self._set = set(self.station_names)

    def parse(self, text):
        """Return (start, end) or raise ValueError."""
        parts = [p.strip() for p in re.split(r"[;,；，]", text or "")]
        if len(parts) != 2 or not parts[0] or not parts[1]:
            raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
        return parts[0], parts[1]

    def match(self, name):
        """Return exact or nearest station name, or None."""
        name = name.strip()
        if name in self._set:
            return name
        m = difflib.get_close_matches(name, self.station_names, n=1, cutoff=0.5)
        return m[0] if m else None

    def validate(self, text):
        """Return (start, end) matched station names, or raise ValueError."""
        s, e = self.parse(text)
        ms = self.match(s)
        if ms is None:
            raise ValueError("Starting station '%s' does not exist, please check the station name" % s)
        me = self.match(e)
        if me is None:
            raise ValueError("Ending station '%s' does not exist, please check the station name" % e)
        if ms == me:
            raise ValueError("Starting station cannot be the same as the ending station")
        return ms, me

"""Route, RouteSummary and ReportExporter."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


class Route:
    def __init__(self, stations, edges, transfer_stations=None):
        self.stations = stations
        self.edges = edges
        self.totalDistance = sum(e.distance for e in edges)
        self.transfer_stations = transfer_stations or []
        self.transferCount = len(self.transfer_stations)
        lines = []
        for e in edges:
            if e.line not in lines:
                lines.append(e.line)
        self.involvedLines = lines

    def segments(self):
        """List of (line, first station, last station)."""
        segs = []
        for i, e in enumerate(self.edges):
            if segs and segs[-1][0] == e.line:
                segs[-1][2] = self.stations[i + 1]
            else:
                segs.append([e.line, self.stations[i], self.stations[i + 1]])
        return segs


class RouteSummary:
    def __init__(self, route):
        self.route = route
        self.totalDistance = route.totalDistance
        self.transferCount = route.transferCount
        self.fare = Sum_money(route.totalDistance)
        self.estimatedTime = calculate_estimated_time(route.totalDistance, route.transferCount)


def format_distance(d):
    return "%d m" % d if d < 1000 else "%.2f km" % (d / 1000.0)


class ReportExporter:
    def __init__(self, fmt="text", output_file=None):
        self.format = fmt
        self.outputFile = output_file

    def render(self, start, end, mode, summary):
        return self.render_json(start, end, mode, summary) if self.format == "json" \
            else self.render_text(start, end, mode, summary)

    def render_text(self, start, end, mode, summary):
        r = summary.route
        out = ["Route: %s -> %s (mode: %s)" % (start, end, mode),
               "Stations (%d): %s" % (len(r.stations), " -> ".join(r.stations)),
               "Segments:"]
        for line, a, b in r.segments():
            out.append("  [%s] %s -> %s" % (line, a, b))
        out.append("Distance: %d m (%s)" % (r.totalDistance, format_distance(r.totalDistance)))
        if r.transfer_stations:
            out.append("Transfers (%d): %s" % (r.transferCount, ", ".join(r.transfer_stations)))
        else:
            out.append("Transfers: 0")
        out.append("Lines: %s" % ", ".join(r.involvedLines))
        out.append("Estimated time: %.1f minutes" % summary.estimatedTime)
        out.append("Fare: %d yuan" % summary.fare)
        return "\n".join(out)

    def render_json(self, start, end, mode, summary):
        r = summary.route
        return json.dumps({
            "query": {"start": start, "end": end, "mode": mode},
            "path": r.stations,
            "distance": r.totalDistance,
            "fare": summary.fare,
            "estimated_time": round(summary.estimatedTime, 1),
            "transfer_count": r.transferCount,
            "transfer_stations": r.transfer_stations,
            "lines": r.involvedLines,
        }, ensure_ascii=False, indent=2)

    def export(self, content):
        if self.outputFile:
            with open(self.outputFile, "a", encoding="utf-8") as f:
                f.write(content + "\n")

"""Command line interface."""
import argparse
import sys

from data_manager import DataManager
from pathfinding import PathfindingConfig, ShortestPath
from report import Route, RouteSummary, ReportExporter
from validator import InputValidator


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p


class CLI:
    def __init__(self, args, data_manager=None):
        self.args = args
        self.dm = data_manager or DataManager()
        self.validator = InputValidator(self.dm.station_names())
        self.exporter = ReportExporter(args.format, args.output_file)

    def compute(self, start, end):
        cfg = PathfindingConfig()
        cfg.mode = self.args.mode
        cfg.max_transfers = self.args.max_transfers
        cfg.preferred_lines = [self.args.line_preference] if self.args.line_preference else []
        s, t = self.dm.station_to_number[start], self.dm.station_to_number[end]
        sp = ShortestPath(self.dm.graph, s, cfg)
        nodes, edges = sp.path_with_edges(t)
        if not nodes:
            raise RuntimeError("No route found between '%s' and '%s'" % (start, end))
        names = [self.dm.number_to_station[n] for n in nodes]
        transfers = [names[i + 1] for i in range(len(edges) - 1) if edges[i].line != edges[i + 1].line]
        return RouteSummary(Route(names, edges, transfers))

    def handle(self, text):
        try:
            start, end = self.validator.validate(text)
        except ValueError as e:
            print(e)
            return
        try:
            summary = self.compute(start, end)
        except Exception as e:
            print("Path calculation failed: %s" % e)
            return
        try:
            content = self.exporter.render(start, end, self.args.mode, summary)
        except Exception as e:
            print("Fare calculation failed: %s" % e)
            return
        print(content)
        try:
            self.exporter.export(content)
        except OSError as e:
            print("Could not write output file: %s" % e)

    def run(self):
        if self.args.check_data:
            ok, msgs = self.dm.check_data()
            print("Data check passed" if ok else "Data check found issues:")
            for m in msgs:
                print(" - " + m)
            return 0 if ok else 1
        print("Enter 'Starting Station; Ending Station' (q to quit)")
        while True:
            try:
                text = input("> ")
            except EOFError:
                break
            if text.strip().lower() in ("q", "quit", "exit"):
                break
            if not text.strip():
                continue
            self.handle(text)
        return 0


def main(argv=None):
    args = build_parser().parse_args(argv)
    return CLI(args).run()

import os
from lottery_system.models import User

DEFAULT_MEMBERS = [
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Operations Department"),
    ("Li Bin", "Technology Department"),
    ("Zhang Wei", "Technology Department"),
    ("Wang Fang", "Product Department"),
    ("Liu Yang", "Product Department"),
    ("Zhao Min", "Cheng Ru Group"),
    ("Sun Li", "Cheng Ru Group"),
    ("Zhou Jie", "Cheng Ru Group"),
    ("Wu Gang", "Operations Department"),
    ("Zheng Hua", "Operations Department"),
    ("Xu Lei", "Marketing Department"),
    ("He Ping", "Marketing Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_MEMBERS], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read {path}: {e}"]
        for i, raw in enumerate(lines, 1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in line.replace("，", ",").split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}' (expected 'name,department')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        presets = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, q in presets:
            try:
                self.add_prize(name, q)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer or None)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "cheng ru"
FIRST_DRAWERS = ("Chen Qixian", "Jin Duo")
GUARDED_DRAWER = "Ma Jiaqi"
RELAXED_DRAWER = "Li Bin"


def _is_cheng_ru(user):
    return CHENG_RU in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winner_names = set()
        self.last_two_draw_results = []

    def _fallback_active(self, prize):
        return (prize.associated_drawer == GUARDED_DRAWER
                and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:]))

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if u.name not in self.winner_names]
        if self._fallback_active(prize):
            return [u for u in pool if _is_cheng_ru(u)]
        if prize.associated_drawer == RELAXED_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winner_names.add(u.name)
        return u

    def finish_round(self, prize, winners):
        d = prize.associated_drawer
        if d in FIRST_DRAWERS:
            self.last_two_draw_results.append(any(_is_cheng_ru(w) for w in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif d == GUARDED_DRAWER:
            self.last_two_draw_results = []

    def rule_description(self, prize):
        parts = []
        if prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer:
            parts.append("drawer: " + prize.associated_drawer)
        if prize.associated_drawer == RELAXED_DRAWER:
            parts.append("relaxed filtering (avoidance groups ignored)")
        if self._fallback_active(prize):
            parts.append("fallback: pool limited to Cheng Ru group")
        parts.append("previous winners excluded")
        return "; ".join(parts)

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, order=0, time=None):
        self.user = user
        self.prize_name = prize_name
        self.order = order
        self.time = time or datetime.now()


def _level(name):
    n = name.lower()
    for i, k in enumerate(("special", "first", "second", "third")):
        if k in n:
            return i
    return 99


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name, len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        idx = list(enumerate(prizes))
        idx.sort(key=lambda t: (_level(t[1].name), t[0]))
        result = {}
        for _, p in idx:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize,WinningOrder,WinningTime\n")
                for pname, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{pname},{w.order},"
                                f"{w.time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import queue
import sys
import threading
import time

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


class Input:
    def __init__(self):
        self.q = queue.Queue()
        threading.Thread(target=self._read, daemon=True).start()

    def _read(self):
        while True:
            line = sys.stdin.readline()
            if not line:
                self.q.put(None)
                return
            self.q.put(line.strip())

    def ask(self, prompt=""):
        print(prompt, end="", flush=True)
        v = self.q.get()
        if v is None:
            raise EOFError
        return v

    def poll(self):
        try:
            v = self.q.get_nowait()
        except queue.Empty:
            return ""
        if v is None:
            raise EOFError
        return v.lower()


def load_list(inp, lm):
    c = inp.ask("1) default list  2) custom txt file: ")
    if c == "1":
        users, errs = lm.load_default_list()
    elif c == "2":
        users, errs = lm.load_custom_list(inp.ask("Path: "))
    else:
        print("Invalid choice.")
        return None
    if errs:
        print("Problems found:")
        for e in errs:
            print("  " + e)
        if inp.ask("Remove invalid data and continue? (y/n): ").lower() != "y":
            print("Upload terminated.")
            return None
    valid, dups = lm.validate_users(users)
    if dups:
        print("Duplicate names:")
        for d in dups:
            print(f"  {d.name} ({d.department})")
        if inp.ask("Remove duplicates and continue? (y/n): ").lower() != "y":
            print("Upload terminated.")
            return None
    print(f"Loaded {len(valid)} participants.")
    return valid


def show_groups(lm, users):
    for g, n in lm.get_group_info(users).items():
        print(f"  {g}: {n}")


def show_prizes(pm):
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"  {i}. {p.name} x{p.quantity} avoid={p.avoid_groups or '-'} drawer={p.associated_drawer or '-'}")


def prize_menu(inp, pm, users, lm):
    while True:
        print("\nPrizes: 1 load presets, 2 add custom, 3 reorder, 4 list, 0 back")
        c = inp.ask("> ")
        if c == "1":
            for e in pm.load_default_prizes():
                print(e)
        elif c == "2":
            name = inp.ask("Prize name: ")
            try:
                q = int(inp.ask("Quantity: "))
            except ValueError:
                print("Invalid quantity.")
                continue
            drawer = inp.ask("Lottery person (blank for none): ") or None
            if users:
                print("Groups:")
                show_groups(lm, users)
            av = inp.ask("Avoid groups (comma separated, blank none): ")
            avoid = [a.strip() for a in av.split(",") if a.strip()]
            try:
                pm.add_prize(name, q, avoid, drawer)
            except ValueError as e:
                print(e)
        elif c == "3":
            show_prizes(pm)
            raw = inp.ask("New order (e.g. 3 1 2): ")
            if not pm.reorder_prizes(raw.replace(",", " ").split()):
                print("Invalid order.")
        elif c == "4":
            show_prizes(pm)
        elif c == "0":
            return


def run_lottery(inp, users, pm, rm):
    lot = LotteryManager(users)
    prizes = pm.get_prizes()
    print("Commands during lottery: pause / continue / terminate")
    for rno, prize in enumerate(prizes, 1):
        print(f"\n=== Round {rno}: {prize.name} x{prize.quantity} ===")
        print("Rules: " + lot.rule_description(prize))
        winners = []
        for i in range(1, prize.quantity + 1):
            cmd = inp.poll()
            if cmd == "pause":
                print("Paused. Type 'continue' to resume or 'terminate' to skip round.")
                while cmd not in ("continue", "terminate"):
                    cmd = inp.ask("") .lower()
            if cmd == "terminate":
                print("Round terminated.")
                break
            print(f"Currently drawing the {i}th winner...")
            time.sleep(0.5)
            u = lot.draw_one(prize)
            if u is None:
                print("No eligible participants remain.")
                break
            winners.append(u)
            rm.add_winner(u, prize)
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        lot.finish_round(prize, winners)
        print(f"Round {rno} results:")
        for k, u in enumerate(winners, 1):
            print(f"  {k}. {u.name} ({u.department})")
    print("\n=== Final winners ===")
    show_results(rm, prizes)


def show_results(rm, prizes):
    for pname, ws in rm.get_winners_by_prize(prizes).items():
        print(pname)
        for w in ws:
            print(f"  {w.user.name} ({w.user.department})")


def main():
    inp = Input()
    lm = ListManager()
    pm = PrizeManager()
    rm = ResultManager()
    users = []
    try:
        while True:
            print("\n1 load list, 2 view groups, 3 prizes, 4 start lottery, 5 results/export, 0 exit")
            c = inp.ask("> ")
            if c == "1":
                r = load_list(inp, lm)
                if r is not None:
                    users = r
            elif c == "2":
                show_groups(lm, users) if users else print("No list loaded.")
            elif c == "3":
                prize_menu(inp, pm, users, lm)
            elif c == "4":
                if not users or not pm.get_prizes():
                    print("Load a list and configure prizes first.")
                else:
                    rm = ResultManager()
                    run_lottery(inp, users, pm, rm)
                    if inp.ask("Export results? (y/n): ").lower() == "y":
                        print(rm.export_to_csv(pm.get_prizes())[1])
            elif c == "5":
                show_results(rm, pm.get_prizes())
                if rm.winners and inp.ask("Export? (y/n): ").lower() == "y":
                    print(rm.export_to_csv(pm.get_prizes())[1])
            elif c == "0":
                break
    except EOFError:
        pass


if __name__ == "__main__":
    main()

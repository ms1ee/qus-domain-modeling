"""System manager and initialization."""
import json

import globalvar

RIDER_COST = 300
MAX_RIDERS = 20


def init_system():
    """Initialise global state."""
    globalvar._init()


def recruit_rider():
    """Create a new rider if funds allow. Returns the Rider or None."""
    from rider import Rider
    funds = globalvar.get_value("funds", 0)
    riders = globalvar.get_value("riders")
    if funds < RIDER_COST or len(riders) >= MAX_RIDERS:
        return None
    rid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    globalvar.set_value("funds", funds - RIDER_COST)
    r = Rider(rid)
    riders.append(r)
    return r


def tick():
    """Advance simulated time by one tick."""
    globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
    for r in globalvar.get_value("riders"):
        r.move()


class SystemManager:
    """Facade over system-level operations."""

    def __init__(self):
        init_system()

    recruit = staticmethod(recruit_rider)
    tick = staticmethod(tick)

    @staticmethod
    def save(path="config.json"):
        """Persist state as JSON."""
        data = {
            "funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "riders": [r.to_dict() for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Order data model."""
import itertools

import globalvar

_ids = itertools.count(1)


def to_center(v):
    """Convert coordinate to grid cell centre."""
    v = max(0, min(799, int(v)))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_ids)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("time", 0)
        self.complete_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Mark order assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self):
        """Mark order complete."""
        self.status = "completed"
        self.complete_time = globalvar.get_value("time", 0)

    @property
    def duration(self):
        """Ticks taken, or None."""
        if self.complete_time is None:
            return None
        return self.complete_time - self.create_time

    def to_dict(self):
        """Serialise."""
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "create_time": self.create_time,
                "complete_time": self.complete_time,
                "distance": self.distance}

"""Rider model."""
import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 50 + 100 * (rider_id % 8) if rider_id % 8 != 7 else 50
        self.x = 0
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0),
                         "pos": (self.x, self.y)})

    def _current(self):
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def _step_to(self, target):
        nxt = PathPlanner.step((self.x, self.y), target, self.speed)
        self.x, self.y = nxt
        return nxt == tuple(target)

    def move(self):
        """Advance one tick."""
        order = self._current()
        if order is None:
            self.state = 0
            return
        if self.state == 0:
            self.state = 1
            self._log("accepted_order", order.id)
        if self.state == 1:
            if self._step_to(order.res_pos):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            if self._step_to(order.cus_pos):
                order.complete()
                self._log("delivered_to_customer", order.id)
                self.orders.remove(order)
                self.state = 0

    def to_dict(self):
        """Serialise."""
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Grid path planning."""


class PathPlanner:
    """Manhattan pathfinding on the grid."""

    @staticmethod
    def step(pos, target, speed):
        """Move up to `speed` units: horizontal first, then vertical."""
        x, y = pos
        tx, ty = target
        budget = speed
        dx = max(-budget, min(budget, tx - x))
        x += dx
        budget -= abs(dx)
        dy = max(-budget, min(budget, ty - y))
        y += dy
        return (x, y)

    @staticmethod
    def route(start, via, end):
        """Corner-point route start -> via -> end."""
        pts = [tuple(start)]
        for p in (via, end):
            c = pts[-1]
            if c[0] != p[0] and c[1] != p[1]:
                pts.append((p[0], c[1]))
            if pts[-1] != tuple(p):
                pts.append(tuple(p))
        return pts

"""Order assignment."""
import globalvar
import system
from order_model import Order


def _cost(rider, order):
    dist = abs(rider.x - order.res_pos[0]) + abs(rider.y - order.res_pos[1])
    return dist + 400 * len(rider.orders), rider.rider_id


def find_best_rider_for_order(order):
    """Pick nearest, least loaded rider; recruit one if none exists."""
    riders = globalvar.get_value("riders")
    idle = [r for r in riders if not r.orders]
    if not idle:
        new = system.recruit_rider()
        if new is not None:
            return new
    if not riders:
        return None
    return min(riders, key=lambda r: _cost(r, order))


class OrderManager:
    """Order creation and dispatch."""

    @staticmethod
    def create(coords):
        """Create and assign order. Returns (order, rider, new_rider_added)."""
        before = len(globalvar.get_value("riders"))
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        globalvar.get_value("orders").append(order)
        rider = find_best_rider_for_order(order)
        added = len(globalvar.get_value("riders")) > before
        if rider is not None:
            rider.orders.append(order)
            order.assign_to(rider.rider_id)
        return order, rider, added

"""Display manager."""
import globalvar
from path_planner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text output."""

    @staticmethod
    def map_text(route=None, riders=True):
        """ASCII 8x8 map. R rider, S restaurant, C customer, * route."""
        grid = [["." for _ in range(8)] for _ in range(8)]
        if route:
            for a, b in zip(route, route[1:]):
                for i in range(101):
                    px = a[0] + (b[0] - a[0]) * i // 100
                    py = a[1] + (b[1] - a[1]) * i // 100
                    grid[min(7, py // 100)][min(7, px // 100)] = "*"
        for o in globalvar.get_value("orders"):
            if o.status != "completed":
                grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
                grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        if riders:
            for r in globalvar.get_value("riders"):
                grid[min(7, r.y // 100)][min(7, r.x // 100)] = "R"
        return "\n".join(" ".join(row) for row in reversed(grid))

    @staticmethod
    def status_text():
        """System snapshot."""
        rs = globalvar.get_value("riders")
        lines = ["Time: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(rs))]
        for r in rs:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    @staticmethod
    def orders_text():
        """All orders."""
        lines = []
        for o in globalvar.get_value("orders"):
            lines.append("Order %d start=%s res=%s cus=%s dist=%d time=%s "
                         "rider=%s status=%s" % (
                             o.id, o.create_time, o.res_pos, o.cus_pos,
                             o.distance, o.duration, o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    @staticmethod
    def riders_text():
        """Rider event logs."""
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("t=%s rider %d %s order=%s pos=%s" % (
                    e["time"], r.rider_id, e["event"], e["order_id"],
                    e["pos"]))
        return "\n".join(lines) or "No riders."

    @staticmethod
    def route_text(order, rider):
        """Route description for a rider."""
        route = PathPlanner.route((rider.x, rider.y), order.res_pos,
                                  order.cus_pos)
        total = sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                    for a, b in zip(route, route[1:]))
        t = -(-total // rider.speed)
        return ("Route: %s\nSpeed: %d  Estimated time: %d ticks\n%s" % (
            " -> ".join(map(str, route)), rider.speed, t,
            DisplayManager.map_text(route)))

"""Command parsing."""


class InputManager:
    """Parse and validate user input."""

    @staticmethod
    def parse_coords(text):
        """Parse four integers in 0-800. Raises ValueError."""
        cleaned = text.replace("[", " ").replace("]", " ").replace(",", " ")
        parts = cleaned.split()
        if len(parts) != 4:
            raise ValueError("Need exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be within 0-800.")
        return vals

    ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
               "q": "quit", "exit": "quit", "t": "tick"}

    @classmethod
    def command(cls, line):
        """Normalise command word."""
        w = line.strip().lower()
        return cls.ALIASES.get(w, w)

"""Entry point: python src/main.py"""
import globalvar
import system
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager


def main():
    """Run the command loop."""
    system.init_system()
    history = []
    print("Commands: start(s) status(st) orders(o) riders(r) tick(t) "
          "history(h) quit(q)")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        cmd = InputManager.command(line)
        if cmd:
            history.append(cmd)
        try:
            if cmd == "start":
                coords = InputManager.parse_coords(
                    input("Enter res_x res_y cus_x cus_y: "))
                order, rider, added = OrderManager.create(coords)
                if rider is None:
                    print("No rider available (insufficient funds).")
                else:
                    print("Restaurant:", order.res_pos, "Customer:",
                          order.cus_pos, "New rider added:", added)
                    print(DisplayManager.route_text(order, rider))
            elif cmd == "status":
                print(DisplayManager.status_text())
                print(DisplayManager.map_text())
            elif cmd == "orders":
                print(DisplayManager.orders_text())
            elif cmd == "riders":
                print(DisplayManager.riders_text())
            elif cmd == "tick":
                system.tick()
                continue
            elif cmd == "history":
                print("\n".join(history))
            elif cmd == "quit":
                globalvar.set_value("running", False)
            elif cmd:
                print("Unknown command.")
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
        if globalvar.get_value("running"):
            system.tick()
    try:
        system.SystemManager.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

"""Core data models: Station, Line, Edge, Graph, MinPQ."""
import heapq


class Station:
    def __init__(self, name, station_id, is_transfer=False):
        self.name = name
        self.station_id = station_id
        self.is_transfer = is_transfer


class Line:
    def __init__(self, name, station_sequence=None):
        self.name = name
        self.station_sequence = list(station_sequence or [])


class Edge:
    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r,%r,%r,%r)" % (self.source, self.dest, self.distance, self.line)


Connection = Edge


class MinPQ:
    """Heap based min priority queue of (priority, item)."""

    def __init__(self):
        self.heap = []
        self._n = 0

    def push(self, priority, item):
        self._n += 1
        heapq.heappush(self.heap, (priority, self._n, item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class Graph:
    """Weighted directed graph; station numbers are given by the caller."""

    def __init__(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.adjacency = {}
        self.station_lines = {}

    @property
    def graph(self):
        return self.adjacency

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, [])

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def nodes(self):
        return list(self.adjacency.keys())

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

"""Dijkstra shortest path over (station, line) states."""
from models import MinPQ

TRANSFER_PENALTY = 1000
LEAST_TRANSFER_WEIGHT = 1000000
PREFERRED_FACTOR = 0.9


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._cost = {}      # state -> cost
        self._real = {}      # state -> (distance, transfers)
        self._prev = {}      # state -> (prev_state, edge)
        self._best = {}      # node -> state
        self._run()

    def _edge_cost(self, edge, transfer):
        c = self.config
        d = edge.distance
        if edge.line in c.preferred_lines:
            d = d * PREFERRED_FACTOR
        if c.mode == "least_transfer":
            return d + (LEAST_TRANSFER_WEIGHT if transfer else 0)
        if c.mode == "comprehensive":
            return d + (TRANSFER_PENALTY if transfer else 0)
        return d

    def _run(self):
        start = (self.source, None)
        self._cost[start] = 0
        self._real[start] = (0, 0)
        pq = MinPQ()
        pq.push(0, start)
        done = set()
        mt = self.config.max_transfers
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line = st
            if node not in self._best:
                self._best[node] = st
                self.dist_to[node] = self._real[st][0]
            dist, tr = self._real[st]
            for e in self.graph.neighbors(node):
                transfer = line is not None and e.line != line
                ntr = tr + (1 if transfer else 0)
                if mt is not None and ntr > mt:
                    continue
                ns = (e.dest, e.line)
                nc = cost + self._edge_cost(e, transfer)
                if ns not in done and nc < self._cost.get(ns, float("inf")):
                    self._cost[ns] = nc
                    self._real[ns] = (dist + e.distance, ntr)
                    self._prev[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best

    def distance_to(self, target):
        return self.dist_to.get(target, float("inf"))

    def path_with_edges(self, target):
        st = self._best.get(target)
        if st is None:
            return [], []
        edges = []
        while st in self._prev:
            st, e = self._prev[st]
            edges.append(e)
        edges.reverse()
        return [self.source] + [e.dest for e in edges], edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        st = self._best.get(target)
        return self._real[st][1] if st else 0

"""Fare calculation."""
import math


class FareTier:
    def __init__(self, min_distance, max_distance, price):
        self.min_distance = min_distance
        self.max_distance = max_distance
        self.price = price


FARE_TIERS = [FareTier(0, 6, 3), FareTier(6, 12, 4), FareTier(12, 22, 5), FareTier(22, 32, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km < 22:
        return 5
    if km <= 32:
        return 6
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km, 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5


class FareCalculator:
    def calculate(self, distance):
        return Sum_money(distance)

"""Data management and integrity checks."""
import difflib

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Graph, Edge


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = Graph()
        self.build()

    def build(self):
        g = Graph()
        for s in self.data.get("stations", []):
            if s["name"] not in g.station_to_number:
                g.add_station(s["name"], len(g.station_to_number))
            g.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in g.station_to_number:
                    g.add_station(n, len(g.station_to_number))
                    g.station_lines.setdefault(n, [c.get("line")])
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c.get("line")))
        self.graph = g

    def lines(self):
        result = {}
        for s, ls in self.graph.station_lines.items():
            for l in ls:
                result.setdefault(l, []).append(s)
        return result

    def match_station(self, name):
        """Return (station, exact) or (None, False)."""
        if name in self.graph.station_to_number:
            return name, True
        m = difflib.get_close_matches(name, list(self.graph.station_to_number), n=1, cutoff=0.5)
        return (m[0], False) if m else (None, False)

    # updates
    def add_line(self, line_name, stations, distances):
        if len(distances) != len(stations) - 1:
            raise ValueError("need len(stations)-1 distances")
        for s in stations:
            lines = self.data["stations"]
            ex = next((x for x in lines if x["name"] == s), None)
            if ex is None:
                self.data["stations"].append({"name": s, "lines": [line_name]})
            elif line_name not in ex["lines"]:
                ex["lines"].append(line_name)
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
            self.data["connections"].append({"source": b, "dest": a, "distance": d, "line": line_name})
        self.build()

    def update_station(self, old_name, new_name=None, lines=None):
        found = False
        for s in self.data["stations"]:
            if s["name"] == old_name:
                found = True
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
        if not found:
            raise KeyError(old_name)
        if new_name:
            for c in self.data["connections"]:
                if c["source"] == old_name:
                    c["source"] = new_name
                if c["dest"] == old_name:
                    c["dest"] = new_name
        self.build()

    def update_distance(self, a, b, distance):
        if distance <= 0:
            raise ValueError("distance must be positive")
        n = 0
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                n += 1
        if not n:
            raise KeyError((a, b))
        self.build()


REPRESENTATIVE = ["西直门", "国贸", "天安门东", "北京站", "西单"]


class DataChecker:
    def __init__(self, manager):
        self.manager = manager

    def check(self):
        g = self.manager.graph
        issues = []
        lines = self.manager.lines()
        if len(g.station_to_number) == 0:
            issues.append("No stations loaded")
        for name in REPRESENTATIVE:
            if name not in g.station_to_number:
                issues.append("Representative station missing: %s" % name)
        names = set(g.station_to_number)
        for c in self.manager.data.get("connections", []):
            if c["source"] not in names or c["dest"] not in names:
                issues.append("Connection references unknown station: %s-%s" % (c["source"], c["dest"]))
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
                issues.append("Bad distance %s-%s: %s" % (c["source"], c["dest"], d))
        for s in names:
            if not g.adjacency.get(g.station_to_number[s]):
                issues.append("Station without outgoing connection: %s" % s)
        return {"ok": not issues, "issues": issues, "stations": len(names),
                "lines": len(lines)}

"""Route, RouteSummary, Query and route computation."""
from pathfinding import ShortestPath, PathfindingConfig
from fare_calculator import Sum_money, calculate_estimated_time

MODES = ("shortest_distance", "least_transfer", "comprehensive")


class Query:
    def __init__(self, start_station, end_station, mode="shortest_distance",
                 max_transfers=None, line_preference=None):
        self.start_station = start_station
        self.end_station = end_station
        self.mode = mode
        self.max_transfers = max_transfers
        self.line_preference = line_preference


class Route:
    def __init__(self, stations, total_distance, transfer_count, lines_involved,
                 segments=None, transfer_stations=None):
        self.stations = stations
        self.total_distance = total_distance
        self.transfer_count = transfer_count
        self.lines_involved = lines_involved
        self.segments = segments or []  # (line, [stations])
        self.transfer_stations = transfer_stations or []


class RouteSummary:
    def __init__(self, route):
        self.total_distance = route.total_distance
        self.fare = Sum_money(route.total_distance)
        self.transfer_count = route.transfer_count
        self.estimated_time = calculate_estimated_time(route.total_distance, route.transfer_count)


def find_route(graph, query):
    s = graph.station_to_number[query.start_station]
    t = graph.station_to_number[query.end_station]
    cfg = PathfindingConfig(query.mode, query.max_transfers,
                            [query.line_preference] if query.line_preference else [])
    sp = ShortestPath(graph, s, cfg)
    nodes, edges = sp.path_with_edges(t)
    if not nodes:
        return None
    names = [graph.number_to_station[n] for n in nodes]
    segments, transfers, lines = [], [], []
    for i, e in enumerate(edges):
        if not segments or segments[-1][0] != e.line:
            if segments:
                transfers.append(names[i])
            segments.append((e.line, [names[i]]))
        segments[-1][1].append(names[i + 1])
        if e.line not in lines:
            lines.append(e.line)
    return Route(names, sum(e.distance for e in edges), sp.get_transfer_count(t),
                 lines, segments, transfers)

"""Report generation (text and JSON)."""
import json


def format_distance(d):
    return "%d m" % d if d < 1000 else "%.2f km" % (d / 1000.0)


class Report:
    def __init__(self, query, route, summary, fmt="text", output_file=None):
        self.query = query
        self.route = route
        self.summary = summary
        self.format = fmt
        self.output_file = output_file
        self.content = self.render()

    def render(self):
        return self.to_json() if self.format == "json" else self.to_text()

    def to_text(self):
        r, s = self.route, self.summary
        out = ["Route: %s -> %s" % (self.query.start_station, self.query.end_station),
               "Stations (%d): %s" % (len(r.stations), " -> ".join(r.stations))]
        for line, sts in r.segments:
            out.append("  [%s] %s -> %s (%d stops)" % (line, sts[0], sts[-1], len(sts) - 1))
        out.append("Distance: %d m (%.2f km) [%s]" % (r.total_distance, r.total_distance / 1000.0,
                                                    format_distance(r.total_distance)))
        out.append("Transfers: %d%s" % (r.transfer_count,
                   " (at " + ", ".join(r.transfer_stations) + ")" if r.transfer_stations else ""))
        out.append("Lines: %s" % ", ".join(r.lines_involved))
        out.append("Estimated time: %.1f minutes" % s.estimated_time)
        out.append("Fare: %d yuan" % s.fare)
        return "\n".join(out)

    def to_json(self):
        r, s = self.route, self.summary
        return json.dumps({
            "query": {"start": self.query.start_station, "end": self.query.end_station,
                      "mode": self.query.mode, "max_transfers": self.query.max_transfers,
                      "line_preference": self.query.line_preference},
            "path": r.stations,
            "distance": r.total_distance,
            "fare": s.fare,
            "estimated_time": s.estimated_time,
            "transfer_count": r.transfer_count,
            "transfer_stations": r.transfer_stations,
            "lines": r.lines_involved,
        }, ensure_ascii=False, indent=2)

    def save(self):
        if self.output_file:
            with open(self.output_file, "a", encoding="utf-8") as f:
                f.write(self.content + "\n")

"""Command line interface."""
import argparse
import re
import sys

from data_manager import DataManager, DataChecker
from route import Query, RouteSummary, find_route, MODES
from report import Report


class CLI:
    def __init__(self, args, manager=None):
        self.args = args
        self.output_file = args.output_file
        self.format = args.format
        self.check_data = args.check_data
        self.manager = manager or DataManager()

    def parse_input(self, line):
        parts = [p.strip() for p in re.split(r"[;；,，]", line)]
        if len(parts) != 2 or not parts[0] or not parts[1]:
            raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
        return parts

    def process(self, line):
        """Returns output text or raises ValueError with message."""
        a, b = self.parse_input(line)
        start, ex1 = self.manager.match_station(a)
        if start is None:
            raise ValueError("Starting station '%s' does not exist, please check the station name" % a)
        end, ex2 = self.manager.match_station(b)
        if end is None:
            raise ValueError("Ending station '%s' does not exist, please check the station name" % b)
        if start == end:
            raise ValueError("Starting station cannot be the same as the ending station")
        notes = []
        if not ex1:
            notes.append("Starting station '%s' matched to '%s'" % (a, start))
        if not ex2:
            notes.append("Ending station '%s' matched to '%s'" % (b, end))
        q = Query(start, end, self.args.mode, self.args.max_transfers, self.args.line_preference)
        try:
            route = find_route(self.manager.graph, q)
        except Exception as e:
            raise ValueError("Path calculation failed: %s" % e)
        if route is None:
            raise ValueError("No route found between '%s' and '%s'" % (start, end))
        try:
            summary = RouteSummary(route)
        except Exception as e:
            raise ValueError("Fare calculation failed: %s" % e)
        rep = Report(q, route, summary, self.format, self.output_file)
        rep.save()
        text = rep.content
        if notes and self.format == "text":
            text = "\n".join(notes) + "\n" + text
        return text

    def run_check(self):
        res = DataChecker(self.manager).check()
        print("Stations: %d, Lines: %d" % (res["stations"], res["lines"]))
        if res["ok"]:
            print("Data check passed")
        else:
            print("Data check found %d issue(s):" % len(res["issues"]))
            for i in res["issues"][:50]:
                print(" - " + i)
        return 0 if res["ok"] else 1

    def run(self):
        if self.check_data:
            return self.run_check()
        print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (q to quit)")
        while True:
            try:
                line = input("> ").strip()
            except EOFError:
                break
            if line.lower() in ("q", "quit", "exit"):
                break
            if not line:
                continue
            try:
                print(self.process(line))
            except ValueError as e:
                print(e)
            except Exception as e:
                print("Unexpected error: %s" % e)
        return 0


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p


def main(argv=None):
    args = build_parser().parse_args(argv)
    return CLI(args).run()


if __name__ == "__main__":
    sys.exit(main())

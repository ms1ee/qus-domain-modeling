I made several design choices where the PRD and Interface leave details open. I have not run any of this code.

- **Order assignment:** the interface gives only `rider.orders.append(order)`, so `find_best_rider_for_order` just picks a rider. `Rider.move` handles the rest. If there are no riders it spawns one itself, so tests can call it without setup.
- **Order state:** `Order.status` starts as "pending", becomes "assigned" and finally "completed".
- **Rider logging:** a rider's log gets "spawn" when it is created. On its first `move`, a rider with a pending order logs "accepted_order" and enters state 1. The log also records "arrived_at_merchant" (state 2) and "delivered_to_customer" (state 0, order completed and removed from `orders`).
- **Movement:** riders move along streets, using Manhattan-style stepping. A rider moves up to `speed` units per tick and always stays on a road or at a destination. Restaurants and customers sit at the grid-cell centers (`(v//100)*100+50`), as the PRD's formula specifies.
- **Coordinates:** `clamp` limits input to 0–799 so the centre stays within the 8x8 map.
- **Files:** `main.py` provides the command loop, and the other modules are small and follow the interface.

"""System manager and initialization."""
import globalvar

RIDER_COST = 100


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", 10000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("order_counter", 0)


class SystemManager:
    """Manages funds and riders."""

    @staticmethod
    def recruit():
        """Create a new rider if funds allow; return it or None."""
        from rider import Rider
        funds = globalvar.get_value("funds", 0)
        if funds < RIDER_COST:
            return None
        rid = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", rid)
        globalvar.set_value("funds", funds - RIDER_COST)
        r = Rider(rid)
        globalvar.get_value("riders").append(r)
        return r

"""Grid path planning."""


def center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(799, int(v)))
    return (v // 100) * 100 + 50


class PathPlanner:
    """Simple L-shaped path planner."""

    @staticmethod
    def step(x, y, tx, ty, speed):
        """Move up to speed units toward target, horizontal first."""
        left = speed
        dx = tx - x
        if dx and left:
            m = min(abs(dx), left)
            x += m if dx > 0 else -m
            left -= m
        dy = ty - y
        if dy and left:
            m = min(abs(dy), left)
            y += m if dy > 0 else -m
        return x, y

"""Order data model."""
import globalvar
from pathplanner import center


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        cnt = globalvar.get_value("order_counter", 0) + 1
        globalvar.set_value("order_counter", cnt)
        self.id = cnt
        self.res_pos = (center(res_pos[0]), center(res_pos[1]))
        self.cus_pos = (center(cus_pos[0]), center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0)
        self.end_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))
        orders = globalvar.get_value("orders")
        if orders is not None:
            orders.append(self)

    def assign_to(self, rider_id):
        """Mark assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

"""Rider model."""
import globalvar
from pathplanner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = (rider_id * 100) % 800
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0),
                         "pos": (self.x, self.y)})

    def move(self):
        """Advance one tick."""
        if not self.orders:
            self.state = 0
            return
        order = self.orders[0]
        if self.state == 0:
            self.state = 1
            order.assign_to(self.rider_id)
            self._record("accepted_order", order.id)
        if self.state == 1:
            tx, ty = order.res_pos
            self.x, self.y = PathPlanner.step(self.x, self.y, tx, ty,
                                              self.speed)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                self._record("arrived_at_merchant", order.id)
        elif self.state == 2:
            tx, ty = order.cus_pos
            self.x, self.y = PathPlanner.step(self.x, self.y, tx, ty,
                                              self.speed)
            if (self.x, self.y) == (tx, ty):
                order.status = "completed"
                order.end_time = globalvar.get_value("time", 0)
                self.orders.pop(0)
                self.state = 0
                self._record("delivered_to_customer", order.id)

"""Order assignment."""
import globalvar
from system import SystemManager


def find_best_rider_for_order(order):
    """Pick the rider with the lowest load, then nearest to restaurant."""
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    idle = [r for r in riders if not r.orders]
    if not idle:
        new = SystemManager.recruit()
        if new is not None:
            return new
    if not riders:
        return None

    def key(r):
        d = abs(r.x - order.res_pos[0]) + abs(r.y - order.res_pos[1])
        return (len(r.orders), d, r.rider_id)

    return min(riders, key=key)

"""Command-line interface."""
import json
import globalvar
from system import init_system
from order import find_best_rider_for_order
from order_model import Order

NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


def tick():
    """Advance simulated time one tick."""
    globalvar.set_value("time", globalvar.get_value("time") + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def render(route=None):
    """ASCII map."""
    grid = [["." for _ in range(8)] for _ in range(8)]
    for o in globalvar.get_value("orders"):
        if o.status != "completed":
            grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "R"
            grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
    for r in globalvar.get_value("riders"):
        gx, gy = min(7, r.x // 100), min(7, r.y // 100)
        grid[gy][gx] = "B"
    return "\n".join(" ".join(row) for row in reversed(grid))


def handle(cmd):
    """Process a command; return False on quit."""
    cmd = cmd.strip().lower()
    if cmd in ("quit", "exit", "q"):
        globalvar.set_value("running", False)
        return False
    if cmd in ("status", "s"):
        print("Funds:", globalvar.get_value("funds"))
        for r in globalvar.get_value("riders"):
            print(r.rider_id, (r.x, r.y), NAMES[r.state],
                  [o.id for o in r.orders])
        print(render())
    elif cmd == "orders":
        for o in globalvar.get_value("orders"):
            t = (o.end_time - o.start_time) if o.end_time is not None else "-"
            print(o.id, o.start_time, o.res_pos, o.cus_pos, o.distance, t,
                  o.rider_id)
    elif cmd == "riders":
        for r in globalvar.get_value("riders"):
            for e in r.log:
                print(r.rider_id, e)
    elif cmd == "save":
        data = {"funds": globalvar.get_value("funds"),
                "riders": [{"id": r.rider_id, "x": r.x, "y": r.y}
                           for r in globalvar.get_value("riders")],
                "orders": [{"id": o.id, "res": o.res_pos, "cus": o.cus_pos,
                            "status": o.status}
                           for o in globalvar.get_value("orders")]}
        with open("state.json", "w") as f:
            json.dump(data, f)
        print("saved")
    elif cmd in ("start", "st"):
        raw = input("Enter rx ry cx cy: ").replace(",", " ").split()
        try:
            v = [int(a) for a in raw]
        except ValueError:
            print("Error: integers required")
            return True
        if len(v) != 4 or any(a < 0 or a > 800 for a in v):
            print("Error: need 4 integers in 0-800")
            return True
        before = len(globalvar.get_value("riders"))
        o = Order((v[0], v[1]), (v[2], v[3]))
        r = find_best_rider_for_order(o)
        if r is None:
            print("No rider available")
            return True
        r.orders.append(o)
        o.assign_to(r.rider_id)
        print("Restaurant:", o.res_pos, "Customer:", o.cus_pos)
        print("New rider added:", len(globalvar.get_value("riders")) > before)
        print("Rider", r.rider_id, "speed", r.speed, "ETA ticks",
              (abs(r.x - o.res_pos[0]) + abs(r.y - o.res_pos[1])
               + o.distance) // r.speed + 1)
        print(render())
    elif cmd:
        print("Unknown command. Use start/status/orders/riders/quit")
    return True


def main():
    """Run the simulator."""
    init_system()
    while globalvar.get_value("running"):
        try:
            cmd = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            if not handle(cmd):
                break
        except Exception as exc:  # recover gracefully
            print("Error:", exc)
        tick()


if __name__ == "__main__":
    main()

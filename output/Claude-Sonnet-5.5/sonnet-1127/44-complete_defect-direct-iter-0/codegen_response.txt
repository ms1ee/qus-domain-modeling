"""Graph data structures."""


class Edge:
    """An undirected track segment between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists.

    Station numbers are assigned by the caller.
    """

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, node):
        self.adj.setdefault(node, [])

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, node):
        return self.adj.get(node, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""Dijkstra shortest path with a heap based priority queue."""
import heapq
import itertools


class MinPQ:
    """Minimum priority queue backed by a binary heap."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


class PathfindingConfig:
    MODES = ("shortest_distance", "least_transfer", "comprehensive")

    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    """Single-source shortest paths; state is (station, line, transfers)."""

    TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 1000000, "comprehensive": 1500}

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._prev = {}      # state -> (prev_state, edge)
        self._info = {}      # state -> (actual distance, transfers)
        self._run()

    def _run(self):
        cfg = self.config
        limited = cfg.max_transfers is not None
        penalty = self.TRANSFER_PENALTY.get(cfg.mode, 0)
        preferred = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        cost = {start: 0.0}
        self._info[start] = (0, 0)
        self._prev[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line, _ = state
            actual, trans = self._info[state]
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = actual
            for e in self.graph.neighbors(node):
                nt = trans
                if line is not None and e.line != line:
                    nt += 1
                if limited and nt > cfg.max_transfers:
                    continue
                w = float(e.distance)
                if e.line in preferred:
                    w *= 0.9
                if nt > trans:
                    w += penalty
                key = (e.dest, e.line, nt if limited else 0)
                nc = c + w
                if nc < cost.get(key, float("inf")):
                    cost[key] = nc
                    self._info[key] = (actual + e.distance, nt)
                    self._prev[key] = (state, e)
                    pq.push(nc, key)

    def has_path_to(self, target):
        return target in self.dist_to

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        nodes, edges = [], []
        while state is not None:
            nodes.append(state[0])
            p = self._prev[state]
            if p is None:
                break
            edges.append(p[1])
            state = p[0]
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        if target not in self._best_state:
            return 0
        return self._info[self._best_state[target]][1]

"""Fare and time estimation."""
import math

AVERAGE_SPEED_KMH = 35.0
TRANSFER_MINUTES = 5.0


def Sum_money(distance):
    """Fare in yuan for a distance in meters (distance rounded up to km)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km <= 22:
        return 5
    if km <= 32:
        return 6
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes for a distance in meters."""
    return distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0 + transfer_count * TRANSFER_MINUTES

"""Data integrity checks and update helpers."""


def check_data(data):
    """Return (ok, list of messages)."""
    problems = []
    stations = data.get("stations", [])
    conns = data.get("connections", [])
    names = [s["name"] for s in stations]
    if len(names) != len(set(names)):
        problems.append("Duplicate station names found")
    nameset = set(names)
    lines = set()
    linked = set()
    for s in stations:
        lines.update(s.get("lines", []))
    for c in conns:
        for k in ("source", "dest"):
            if c[k] not in nameset:
                problems.append("Connection references unknown station '%s'" % c[k])
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0:
            problems.append("Invalid distance %r between %s and %s" % (d, c["source"], c["dest"]))
        elif d > 30000:
            problems.append("Suspicious distance %r between %s and %s" % (d, c["source"], c["dest"]))
        linked.add(c["source"])
        linked.add(c["dest"])
    for n in names:
        if n not in linked:
            problems.append("Station '%s' has no connections" % n)
    return (not problems), problems + [
        "Stations: %d, connections: %d, lines: %d" % (len(names), len(conns), len(lines))]


def add_line(data, line, station_names, distances):
    """Add a line; distances has len(station_names)-1 entries."""
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances must have one fewer entry than stations")
    index = {s["name"]: s for s in data["stations"]}
    for n in station_names:
        if n in index:
            if line not in index[n]["lines"]:
                index[n]["lines"].append(line)
        else:
            st = {"name": n, "lines": [line]}
            data["stations"].append(st)
            index[n] = st
    for i, d in enumerate(distances):
        data["connections"].append(
            {"source": station_names[i], "dest": station_names[i + 1], "distance": int(d), "line": line})


def update_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    found = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
            c["distance"] = int(distance)
            found = True
    return found

"""CLI for the Beijing Subway fare calculation system."""
import argparse
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data  # noqa: E402
from models import Edge, Graph  # noqa: E402
from pathfinding import PathfindingConfig, ShortestPath  # noqa: E402
from fare_calculator import Sum_money, calculate_estimated_time  # noqa: E402
from data_manager import check_data  # noqa: E402

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def build_graph(data):
    graph = Graph()
    station_to_number = {}
    for s in data["stations"]:
        if s["name"] not in station_to_number:
            station_to_number[s["name"]] = len(station_to_number)
        graph.set_station_lines(s["name"], s.get("lines", []))
        graph.add_node(station_to_number[s["name"]])
    for c in data["connections"]:
        if c["source"] in station_to_number and c["dest"] in station_to_number:
            graph.add_edge(Edge(station_to_number[c["source"]], station_to_number[c["dest"]],
                                c["distance"], c["line"]))
    return graph, station_to_number


def format_distance(m):
    return "%d m" % m if m < 1000 else "%.2f km" % (m / 1000.0)


def query(graph, station_to_number, start, end, config):
    """Return result dict; raises ValueError on problems."""
    number_to_station = {v: k for k, v in station_to_number.items()}
    sp = ShortestPath(graph, station_to_number[start], config)
    target = station_to_number[end]
    nodes, edges = sp.path_with_edges(target)
    if not nodes:
        raise ValueError("No route found between '%s' and '%s'" % (start, end))
    distance = sp.dist_to[target]
    transfers = sp.get_transfer_count(target)
    try:
        fare = Sum_money(distance)
    except Exception as e:
        raise ValueError("Fare calculation failed: %s" % e)
    segments = []
    transfer_stations = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = number_to_station[e.dest]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(number_to_station[e.source])
            segments.append({"line": e.line, "from": number_to_station[e.source],
                             "to": number_to_station[e.dest], "stations": 1})
    return {
        "query": {"start": start, "end": end, "mode": config.mode},
        "path": [number_to_station[n] for n in nodes],
        "distance_m": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(calculate_estimated_time(distance, transfers), 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": list(dict.fromkeys(e.line for e in edges)),
        "segments": segments,
    }


def render_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: %s" % format_distance(r["distance_m"]),
           "Lines: %s" % ", ".join(r["lines"])]
    for seg in r["segments"]:
        out.append("  [%s] %s -> %s (%d segments)" % (seg["line"], seg["from"], seg["to"], seg["stations"]))
    if r["transfer_stations"]:
        out.append("Transfer at: %s" % ", ".join(r["transfer_stations"]))
    out.append("Transfers: %d" % r["transfer_count"])
    out.append("Estimated time: %.1f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    if fmt == "json":
        return json.dumps(r, ensure_ascii=False, indent=2)
    return render_text(r)


def process_line(line, graph, s2n, config, fmt):
    """Return (ok, message)."""
    line = line.replace("；", ";")
    parts = [p.strip() for p in line.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return False, FORMAT_ERR
    a, b = parts
    if a not in s2n:
        return False, "Starting station '%s' does not exist, please check the station name" % a
    if b not in s2n:
        return False, "Ending station '%s' does not exist, please check the station name" % b
    if a == b:
        return False, "Starting station cannot be the same as the ending station"
    try:
        return True, render(query(graph, s2n, a, b, config), fmt)
    except ValueError as e:
        return False, str(e)
    except Exception as e:
        return False, "Path calculation error: %s" % e


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(PathfindingConfig.MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except Exception as e:
        print("Failed to load data: %s" % e)
        return 1
    if args.check_data:
        ok, msgs = check_data(data)
        print("\n".join(msgs))
        print("Data check passed" if ok else "Data check found problems")
        return 0 if ok else 1

    graph, s2n = build_graph(data)
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    if args.line_preference:
        config.preferred_lines = [args.line_preference]

    while True:
        try:
            line = input("Enter 'Starting Station; Ending Station' (q to quit): ")
        except EOFError:
            break
        if line.strip().lower() in ("q", "quit", "exit"):
            break
        if not line.strip():
            continue
        ok, msg = process_line(line, graph, s2n, config, args.format)
        print(msg)
        if ok and args.output_file:
            try:
                with open(args.output_file, "w", encoding="utf-8") as f:
                    f.write(msg + "\n")
            except OSError as e:
                print("Could not write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

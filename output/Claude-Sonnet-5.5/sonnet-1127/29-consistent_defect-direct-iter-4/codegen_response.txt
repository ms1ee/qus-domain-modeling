"""System initialization and management."""
import json

import globalvar


def init_system():
    """Initialize global parameters."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Manages funds, riders and running state."""

    def __init__(self):
        init_system()

    def tick(self):
        """Advance simulated time one unit and move all riders."""
        globalvar.set_value("time", globalvar.get_value("time") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

    def save(self, path="config.json"):
        """Persist system state to JSON."""
        data = {
            "funds": globalvar.get_value("funds"),
            "rider_id_counter": globalvar.get_value("rider_id_counter"),
            "riders": [r.to_dict() for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

"""Order data model."""
import itertools

import globalvar

_ids = itertools.count(1)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_ids)
        self.res_pos = tuple(res_pos)
        self.cus_pos = tuple(cus_pos)
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.end_time = None

    def assign_to(self, rider_id):
        """Mark order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    @property
    def distance(self):
        """Manhattan distance restaurant -> customer."""
        return (abs(self.res_pos[0] - self.cus_pos[0])
                + abs(self.res_pos[1] - self.cus_pos[1]))

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "end_time": self.end_time}

"""Rider model."""
import globalvar

PENDING = ("pending", "assigned")


class Rider:
    """A delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = 0
        self.orders = []
        self.special_case = False
        self.current = None
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0),
                         "x": self.x, "y": self.y})

    def active_orders(self):
        """Orders not yet completed."""
        return [o for o in self.orders if o.status != "completed"]

    def _accept_next(self):
        for o in self.orders:
            if o.status in PENDING:
                self.current = o
                o.status = "picking_up"
                self.state = 1
                self._log("accepted_order", o.id)
                return True
        return False

    def _step(self, tx, ty):
        budget = self.speed
        dx = tx - self.x
        m = min(abs(dx), budget)
        self.x += m if dx > 0 else -m
        budget -= m
        dy = ty - self.y
        m = min(abs(dy), budget)
        self.y += m if dy > 0 else -m

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            self._accept_next()
            return
        o = self.current
        if o is None:
            self.state = 0
            return
        if self.state == 1:
            self._step(*o.res_pos)
            if (self.x, self.y) == tuple(o.res_pos):
                self.state = 2
                o.status = "delivering"
                self._log("arrived_at_merchant", o.id)
        else:
            self._step(*o.cus_pos)
            if (self.x, self.y) == tuple(o.cus_pos):
                o.status = "completed"
                o.end_time = globalvar.get_value("time", 0)
                self._log("delivered_to_customer", o.id)
                self.current = None
                self.state = 0
                self._accept_next()

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order management and assignment."""
import globalvar
from order_model import Order
from rider import Rider


def find_best_rider_for_order(order):
    """Rider with fewest active orders, ties broken by distance."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None
    rx, ry = order.res_pos

    def key(r):
        d = ((r.x - rx) ** 2 + (r.y - ry) ** 2) ** 0.5
        return (len(r.active_orders()), d, r.rider_id)

    return min(riders, key=key)


def to_grid(v):
    """Convert coordinate to top-left corner of its grid cell."""
    v = max(0, min(800, v))
    return (v // 100) * 100


class OrderManager:
    """Creates, recruits for and assigns orders."""

    def create_order(self, rx, ry, cx, cy):
        """Create an order; returns (order, rider, recruited)."""
        order = Order((to_grid(rx), to_grid(ry)), (to_grid(cx), to_grid(cy)))
        order.start_time = globalvar.get_value("time", 0)
        globalvar.get_value("orders").append(order)
        recruited = self.maybe_recruit()
        rider = find_best_rider_for_order(order)
        if rider is None:
            return order, None, recruited
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return order, rider, recruited

    def maybe_recruit(self):
        """Hire a rider if funds and demand allow."""
        riders = globalvar.get_value("riders")
        pending = len([o for o in globalvar.get_value("orders")
                       if o.status in ("pending", "assigned")])
        idle = len([r for r in riders if r.state == 0
                    and not r.active_orders()])
        funds = globalvar.get_value("funds")
        if funds >= 300 and pending > idle:
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            riders.append(Rider(rid))
            globalvar.set_value("funds", funds - 500)
            return True
        return False

class PathPlanner:
    """Right-angle path planner (horizontal first, then vertical)."""

    @staticmethod
    def plan(start, end):
        """Return corner points from start to end."""
        pts = [tuple(start)]
        corner = (end[0], start[1])
        if corner != pts[-1] and corner != tuple(end):
            pts.append(corner)
        if tuple(end) != pts[-1]:
            pts.append(tuple(end))
        return pts

    @classmethod
    def plan_multi(cls, points):
        """Continuous path through several destinations."""
        path = [tuple(points[0])]
        for a, b in zip(points, points[1:]):
            path.extend(cls.plan(a, b)[1:])
        return path

    @staticmethod
    def length(path):
        return sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                   for a, b in zip(path, path[1:]))

"""Display helpers."""
import globalvar

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and map rendering."""

    def status(self):
        riders = globalvar.get_value("riders")
        lines = ["Time: %s  Funds: %s  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        lines = []
        for o in globalvar.get_value("orders"):
            took = (o.end_time - o.start_time) if o.end_time is not None \
                else "-"
            lines.append("Order %d start=%s res=%s cus=%s dist=%d time=%s "
                         "rider=%s status=%s" % (
                             o.id, o.start_time, o.res_pos, o.cus_pos,
                             o.distance, took, o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    def riders_log(self):
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("Rider %d t=%s %s at (%d,%d) order=%s" % (
                    r.rider_id, e["time"], e["event"], e["x"], e["y"],
                    e["order_id"]))
        return "\n".join(lines) or "No riders."

    def map(self, path=None):
        """16x16 ASCII map; each cell is 50 units."""
        g = [["#" if (i % 2 == 0 or j % 2 == 0) else "." for j in range(16)]
             for i in range(16)]

        def cell(x, y):
            return min(15, y // 50), min(15, x // 50)

        if path:
            for a, b in zip(path, path[1:]):
                (r1, c1), (r2, c2) = cell(*a), cell(*b)
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        g[r][c] = "*"
        for o in globalvar.get_value("orders"):
            if o.status != "completed":
                r, c = cell(*o.res_pos)
                g[r][c] = "S"
                r, c = cell(*o.cus_pos)
                g[r][c] = "C"
        for rd in globalvar.get_value("riders"):
            r, c = cell(rd.x, rd.y)
            g[r][c] = "R"
        return "\n".join("".join(row) for row in g)

"""Command parsing and validation."""


class InputManager:
    """Parses user input."""

    def __init__(self):
        self.history = []

    def parse(self, line):
        """Return (command, args) or raise ValueError."""
        line = line.strip()
        if not line:
            raise ValueError("empty command")
        self.history.append(line)
        parts = line.split()
        cmd = parts[0].lower()
        aliases = {"s": "status", "o": "orders", "r": "riders",
                   "q": "quit", "exit": "quit"}
        return aliases.get(cmd, cmd), parts[1:]

    @staticmethod
    def parse_coords(text):
        """Parse four integers within 0-1000."""
        parts = text.replace(",", " ").replace("[", " ").replace(
            "]", " ").split()
        if len(parts) != 4:
            raise ValueError("need exactly 4 integers")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("coordinates must be integers")
        if any(v < 0 or v > 1000 for v in vals):
            raise ValueError("coordinates must be within 0-1000")
        return vals

"""Command-line entry point."""
import globalvar
from system import SystemManager
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager
from path_planner import PathPlanner


def main():
    sysm = SystemManager()
    om, dm, im = OrderManager(), DisplayManager(), InputManager()
    print("Commands: start, status, orders, riders, tick, quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            cmd, args = im.parse(line)
            if cmd == "start":
                text = " ".join(args) or input("res_x res_y cus_x cus_y: ")
                o, rider, rec = om.create_order(*im.parse_coords(text))
                print("Restaurant:", o.res_pos, "Customer:", o.cus_pos)
                print("New rider added:", rec)
                if rider:
                    path = PathPlanner.plan_multi(
                        [(rider.x, rider.y), o.res_pos, o.cus_pos])
                    dist = PathPlanner.length(path)
                    print("Rider %d speed %d, est. time %d ticks" % (
                        rider.rider_id, rider.speed,
                        -(-dist // rider.speed)))
                    print(dm.map(path))
                else:
                    print("No rider available.")
            elif cmd == "status":
                print(dm.status())
                print(dm.map())
            elif cmd == "orders":
                print(dm.orders())
            elif cmd == "riders":
                print(dm.riders_log())
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                for _ in range(n):
                    sysm.tick()
                continue
            elif cmd == "quit":
                globalvar.set_value("running", False)
                sysm.save()
                break
            else:
                print("Unknown command.")
                continue
            sysm.tick()
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)


if __name__ == "__main__":
    main()

"""System initialization and core operations."""
import globalvar

RIDER_COST = 300
ORDER_INCOME = 100


def init_system():
    """Initialize all global parameters."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", 0)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("grid_size", 8)
    globalvar.set_value("cell_size", 100)


def recruit_rider(force=False):
    """Create a rider if funds >= cost (or forced). Returns the rider or None."""
    from rider import Rider
    funds = globalvar.get_value("funds", 0)
    if funds < RIDER_COST and not force:
        return None
    globalvar.set_value("funds", funds - RIDER_COST)
    rid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    rider = Rider(rid)
    globalvar.get_value("riders").append(rider)
    return rider


def tick():
    """Advance simulation by one tick."""
    globalvar.set_value("tick", globalvar.get_value("tick", 0) + 1)
    for r in globalvar.get_value("riders", []):
        r.move()

"""Order data model."""
import itertools
import globalvar

_counter = itertools.count(1)


def to_center(v):
    """Convert a coordinate to its grid cell center."""
    return min(max(int(v) // 100, 0), 7) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.start_time = globalvar.get_value("tick", 0)
        self.completion_time = None
        self.rider_id = None
        self.status = "pending"
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0]) +
                         abs(self.res_pos[1] - self.cus_pos[1]))

    @property
    def time_taken(self):
        if self.completion_time is None:
            return None
        return self.completion_time - self.start_time

    def assign_to(self, rider_id):
        """Mark order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

"""Rider model and movement."""
import globalvar


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0),
                         "x": self.x, "y": self.y})

    def _step_toward(self, tx, ty):
        """Move up to speed units: horizontal first, then vertical."""
        budget = self.speed
        dx = tx - self.x
        if dx and budget:
            s = min(abs(dx), budget)
            self.x += s if dx > 0 else -s
            budget -= s
        dy = ty - self.y
        if dy and budget:
            s = min(abs(dy), budget)
            self.y += s if dy > 0 else -s

    def move(self):
        """Advance one tick."""
        if self.state == 0 and self.orders:
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        if self.state == 0 or not self.orders:
            return
        order = self.orders[0]
        target = order.res_pos if self.state == 1 else order.cus_pos
        self._step_toward(*target)
        if (self.x, self.y) == tuple(target):
            if self.state == 1:
                self._log("arrived_at_merchant", order.id)
                self.state = 2
            else:
                order.status = "completed"
                order.completion_time = globalvar.get_value("tick", 0)
                self._log("delivered_to_customer", order.id)
                self.orders.pop(0)
                self.state = 0

"""Order assignment."""
import globalvar


def find_best_rider_for_order(order):
    """Return the closest rider (idle riders and lighter loads preferred on ties)."""
    riders = globalvar.get_value("riders", [])
    if not riders:
        return None
    rx, ry = order.res_pos[0], order.res_pos[1]
    return min(riders, key=lambda r: (abs(r.x - rx) + abs(r.y - ry),
                                      len(r.orders), r.rider_id))


def create_and_assign(res_xy, cus_xy):
    """Create an order, recruit if needed, assign it. Returns (order, rider, recruited)."""
    import system
    from order_model import Order
    order = Order(res_xy, cus_xy)
    globalvar.get_value("orders").append(order)
    globalvar.set_value("funds", globalvar.get_value("funds", 0) + system.ORDER_INCOME)
    recruited = system.recruit_rider() is not None
    if not globalvar.get_value("riders"):
        system.recruit_rider(force=True)
        recruited = True
    rider = find_best_rider_for_order(order)
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    return order, rider, recruited

"""Command-line interface for the dispatch simulator."""
import globalvar
import system
import order as order_mod

NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


def parse_coords(parts):
    """Validate four integers in 0-800."""
    if len(parts) != 4:
        raise ValueError("Expected 4 integers: rx ry cx cy")
    try:
        vals = [int(p) for p in parts]
    except ValueError:
        raise ValueError("Coordinates must be integers")
    if any(v < 0 or v > 800 for v in vals):
        raise ValueError("Coordinates must be within 0-800")
    return vals


def draw_map(rider=None, order=None):
    grid = [["." for _ in range(8)] for _ in range(8)]
    for r in globalvar.get_value("riders", []):
        grid[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
    if order:
        grid[order.res_pos[1] // 100][order.res_pos[0] // 100] = "S"
        grid[order.cus_pos[1] // 100][order.cus_pos[0] // 100] = "C"
    return "\n".join(" ".join(row) for row in grid)


def cmd_start(args):
    vals = parse_coords(args)
    o, r, recruited = order_mod.create_and_assign(vals[:2], vals[2:])
    dist = abs(r.x - o.res_pos[0]) + abs(r.y - o.res_pos[1]) + o.distance
    print("Restaurant grid: (%d,%d) Customer grid: (%d,%d)" % (
        o.res_pos[0] // 100, o.res_pos[1] // 100,
        o.cus_pos[0] // 100, o.cus_pos[1] // 100))
    print("New rider added:", recruited)
    print("Rider %d speed %d, est. time %d ticks" % (
        r.rider_id, r.speed, -(-dist // r.speed)))
    print(draw_map(order=o))


def cmd_status():
    print("Funds:", globalvar.get_value("funds"), "Tick:", globalvar.get_value("tick"))
    riders = globalvar.get_value("riders")
    print("Riders:", len(riders))
    for r in riders:
        print("  #%d (%d,%d) %s orders=%s" % (
            r.rider_id, r.x, r.y, NAMES[r.state], [o.id for o in r.orders]))
    print(draw_map())


def cmd_orders():
    for o in globalvar.get_value("orders"):
        print("Order %d start=%d rest=%s cust=%s dist=%d time=%s rider=%s %s" % (
            o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
            o.time_taken, o.rider_id, o.status))


def cmd_riders():
    for r in globalvar.get_value("riders"):
        for e in r.log:
            print("Rider %d t=%d (%d,%d) %s order=%s" % (
                r.rider_id, e["time"], e["x"], e["y"], e["event"], e["order_id"]))


def main():
    system.init_system()
    history = []
    aliases = {"s": "start", "st": "status", "o": "orders", "r": "riders", "q": "quit"}
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if not line:
            continue
        history.append(line)
        parts = line.split()
        cmd = aliases.get(parts[0].lower(), parts[0].lower())
        try:
            if cmd == "start":
                if len(parts) == 1:
                    parts += input("rx ry cx cy: ").split()
                cmd_start(parts[1:])
            elif cmd == "status":
                cmd_status()
            elif cmd == "orders":
                cmd_orders()
            elif cmd == "riders":
                cmd_riders()
            elif cmd == "history":
                print("\n".join(history))
            elif cmd in ("quit", "exit"):
                globalvar.set_value("running", False)
                continue
            else:
                print("Unknown command")
                continue
        except ValueError as e:
            print("Error:", e)
            continue
        system.tick()


if __name__ == "__main__":
    main()

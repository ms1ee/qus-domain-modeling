"""Graph data structures."""


class Edge:
    """Undirected edge endpoint description (station numbers)."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""Dijkstra-based path finding with transfer awareness."""
import heapq
import itertools


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class ShortestPath:
    TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 100000, "comprehensive": 1500}

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}      # station -> state key
        self._label = {}     # state key -> (dist, transfers, prev key, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = self.TRANSFER_PENALTY.get(cfg.mode, 0)
        limit = cfg.max_transfers
        prefs = set(cfg.preferred_lines or [])
        use_pref = cfg.mode != "shortest_distance" or False
        start = (self.source, None, 0)
        self._label[start] = (0, 0, None, None)
        cost_of = {start: 0}
        pq = MinPQ()
        pq.push(0, start)
        done = set()
        while not pq.is_empty():
            cost, key = pq.pop()
            if key in done:
                continue
            done.add(key)
            st, line, _t = key
            dist, trans, _, _ = self._label[key]
            if st not in self._best:
                self._best[st] = key
                self.dist_to[st] = dist
            for e in self.graph.neighbors(st):
                nt = trans + (1 if line is not None and e.line != line else 0)
                if limit is not None and nt > limit:
                    continue
                nkey = (e.dest, e.line, nt if limit is not None else 0)
                if nkey in done:
                    continue
                w = e.weight
                if use_pref and prefs and e.line in prefs:
                    w = w * 0.9
                ncost = cost + w + (penalty if nt > trans else 0)
                if ncost < cost_of.get(nkey, float("inf")):
                    cost_of[nkey] = ncost
                    self._label[nkey] = (dist + e.weight, nt, key, e)
                    pq.push(ncost, nkey)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        key = self._best[target]
        nodes, edges = [], []
        while key is not None:
            _, _, prev, e = self._label[key]
            nodes.append(key[0])
            if e is not None:
                edges.append(e)
            key = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for i in range(1, len(edges)) if edges[i].line != edges[i - 1].line)

"""Fare and time estimation."""


def Sum_money(distance) -> int:
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = int(round(distance))
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated minutes: ~35 km/h average speed plus 5 min per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("negative input")
    return round(distance / 1000.0 / 35.0 * 60.0 + 5.0 * transfer_count, 1)

"""Network construction, integrity checks and data updates."""
from models import Graph, Edge
from data_loader import load_subway_data, DATA_FILE_PATH

MAIN_LINE_KEYS = ["1", "2", "4", "5", "6", "8", "10", "13", "14", "15"]


class SubwayNetwork:
    def __init__(self, data=None):
        self.data = data if data is not None else load_subway_data(DATA_FILE_PATH)
        self.rebuild()

    def rebuild(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        for s in self.data.get("stations", []):
            self._number(s["name"])
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a, b = self._number(c["source"]), self._number(c["dest"])
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))
            for n in (c["source"], c["dest"]):
                ls = self.graph.station_lines.setdefault(n, [])
                if c["line"] not in ls:
                    ls.append(c["line"])
        for n in self.station_to_number.values():
            self.graph.add_node(n)

    def _number(self, name):
        if name not in self.station_to_number:
            n = len(self.station_to_number)
            self.station_to_number[name] = n
            self.number_to_station[n] = name
        return self.station_to_number[name]

    # ---- integrity check ----
    def check_data(self):
        """Return (ok, list of messages)."""
        msgs = []
        ok = True
        names = {s["name"] for s in self.data.get("stations", [])}
        lines = set()
        for s in self.data.get("stations", []):
            lines.update(s.get("lines", []))
        msgs.append("Stations: %d, connections: %d, lines: %d" % (
            len(names), len(self.data.get("connections", [])), len(lines)))
        for k in MAIN_LINE_KEYS:
            if not any(k in "".join(ch if ch.isdigit() else " " for ch in l).split() for l in lines):
                msgs.append("Warning: main line %s not found" % k)
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in names:
                    ok = False
                    msgs.append("Connection references unknown station: %s" % n)
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                ok = False
                msgs.append("Invalid distance %s between %s and %s" % (d, c["source"], c["dest"]))
            elif d > 30000:
                msgs.append("Warning: suspicious long distance %s-%s: %s" % (c["source"], c["dest"], d))
        connected = {n for n in self.graph.nodes() if self.graph.neighbors(n)}
        for name, n in self.station_to_number.items():
            if n not in connected:
                msgs.append("Warning: station without connections: %s" % name)
        msgs.append("Data check " + ("passed" if ok else "FAILED"))
        return ok, msgs

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        if len(distances) != len(stations) - 1:
            raise ValueError("need len(stations)-1 distances")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for i, d in enumerate(distances):
            self.data["connections"].append(
                {"source": stations[i], "dest": stations[i + 1], "distance": d, "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.rebuild()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self.rebuild()
        return found

"""CLI entry point for the Beijing Subway fare calculator."""
import argparse
import json
import re
import sys

from data_manager import SubwayNetwork
from pathfinding import PathfindingConfig, ShortestPath
from fare_calculator import Sum_money, calculate_estimated_time

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_input(text):
    parts = [p.strip() for p in re.split(r"[;；,，]", text)]
    parts = [p for p in parts if p != ""]
    if len(parts) != 2:
        raise ValueError(FORMAT_ERR)
    return parts[0], parts[1]


def validate(net, start, end):
    if start not in net.station_to_number:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % start)
    if end not in net.station_to_number:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % end)
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")


def query(net, start, end, config):
    sp = ShortestPath(net.graph, net.station_to_number[start], config)
    t = net.station_to_number[end]
    try:
        nodes, edges = sp.path_with_edges(t)
    except Exception as e:
        raise RuntimeError("Path calculation failed: %s" % e)
    if not nodes:
        raise RuntimeError("No route found between '%s' and '%s' under the given constraints" % (start, end))
    distance = sum(e.distance for e in edges)
    transfers = sp.get_transfer_count(t)
    try:
        fare = Sum_money(distance)
        minutes = calculate_estimated_time(distance, transfers)
    except Exception as e:
        raise RuntimeError("Fare calculation failed: %s" % e)
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        name_to = net.number_to_station[e.dest]
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = name_to
            segments[-1]["stations"] += 1
            segments[-1]["distance"] += e.distance
        else:
            if segments:
                transfer_stations.append({"station": net.number_to_station[e.source],
                                          "from_line": segments[-1]["line"], "to_line": e.line})
            segments.append({"line": e.line, "from": net.number_to_station[e.source],
                             "to": name_to, "stations": 1, "distance": e.distance})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    return {
        "query": {"start": start, "end": end, "mode": config.mode,
                  "max_transfers": config.max_transfers,
                  "line_preference": config.preferred_lines},
        "path": [net.number_to_station[n] for n in nodes],
        "distance_m": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": minutes,
        "transfer_count": transfers,
        "lines": lines,
        "segments": segments,
        "transfers": transfer_stations,
    }


def format_text(r):
    d = r["distance_m"]
    dist = "%d m (%.2f km)" % (d, d / 1000.0)
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: " + dist]
    for s in r["segments"]:
        out.append("  [%s] %s -> %s (%d stops, %.2f km)" % (
            s["line"], s["from"], s["to"], s["stations"], s["distance"] / 1000.0))
    out.append("Transfers: %d" % r["transfer_count"])
    for t in r["transfers"]:
        out.append("  Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    out.append("Lines: " + ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    if fmt == "json":
        return json.dumps(r, ensure_ascii=False, indent=2)
    return format_text(r)


def main(argv=None):
    ap = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    ap.add_argument("--output_file")
    ap.add_argument("--format", choices=["text", "json"], default="text")
    ap.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                    default="shortest_distance")
    ap.add_argument("--max_transfers", type=int)
    ap.add_argument("--line_preference")
    ap.add_argument("--check_data", action="store_true")
    args = ap.parse_args(argv)

    try:
        net = SubwayNetwork()
    except Exception as e:
        print("Failed to load subway data: %s" % e)
        return 1
    if args.check_data:
        ok, msgs = net.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1

    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    if args.line_preference:
        cfg.preferred_lines = [p.strip() for p in args.line_preference.split(",") if p.strip()]

    outputs = []
    while True:
        try:
            text = input("Enter 'Starting Station; Ending Station' (q to quit): ")
        except EOFError:
            break
        text = text.strip()
        if text.lower() in ("q", "quit", "exit"):
            break
        if not text:
            continue
        try:
            start, end = parse_input(text)
            validate(net, start, end)
            res = query(net, start, end, cfg)
        except (ValueError, RuntimeError) as e:
            print("Error: %s" % e)
            continue
        except Exception as e:
            print("Unexpected error: %s" % e)
            continue
        s = render(res, args.format)
        print(s)
        outputs.append(s)
        if args.output_file:
            try:
                with open(args.output_file, "w", encoding="utf-8") as f:
                    f.write("\n\n".join(outputs) + "\n")
            except OSError as e:
                print("Cannot write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

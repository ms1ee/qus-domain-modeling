"""Grid map and path planning."""


class GridMap:
    """8x8 grid, 100-unit cells."""

    def __init__(self, width=800, height=800, cell_size=100):
        self.width = width
        self.height = height
        self.cell_size = cell_size

    def center(self, v):
        """Convert a coordinate to its cell center, clamped to the map."""
        v = max(0, min(int(v), self.width - 1))
        return (v // self.cell_size) * self.cell_size + self.cell_size // 2


class PathPlanner:
    """Right-angle path planning."""

    def __init__(self, grid=None):
        self.grid = grid or GridMap()

    def plan(self, start, *targets):
        """Return a list of waypoints through targets using L-shaped legs."""
        pts = [tuple(start)]
        cur = tuple(start)
        for t in targets:
            t = tuple(t)
            if cur[0] != t[0] and cur[1] != t[1]:
                pts.append((t[0], cur[1]))
            if pts[-1] != t:
                pts.append(t)
            cur = t
        return pts

    @staticmethod
    def length(points):
        """Manhattan length of a polyline."""
        return sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                   for a, b in zip(points, points[1:]))

"""Rider model."""
import globalvar


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order):
        self.log.append({
            "event": event,
            "order_id": getattr(order, "id", None),
            "time": globalvar.get_value("tick", 0) or 0,
            "x": self.x, "y": self.y, "state": self.state,
        })

    def _step(self, target):
        """Move toward target along x then y; return True if reached."""
        budget = self.speed
        dx = target[0] - self.x
        if dx:
            m = min(budget, abs(dx))
            self.x += m if dx > 0 else -m
            budget -= m
        dy = target[1] - self.y
        if dy and budget:
            m = min(budget, abs(dy))
            self.y += m if dy > 0 else -m
        return (self.x, self.y) == tuple(target)

    def move(self):
        """Advance one tick."""
        if not self.orders:
            self.state = 0
            return
        order = self.orders[0]
        if self.state == 0:
            self.state = 1
            self._log("accepted_order", order)
            order.status = "picking_up"
            return
        if self.state == 1:
            if self._step(order.res_pos):
                self.state = 2
                order.status = "delivering"
                self._log("arrived_at_merchant", order)
        elif self.state == 2:
            if self._step(order.cus_pos):
                self._log("delivered_to_customer", order)
                order.complete(globalvar.get_value("tick", 0) or 0)
                self.orders.pop(0)
                self.state = 0

"""Order model."""
import itertools
import globalvar
from pathplanner import GridMap

_ids = itertools.count(1)
_grid = GridMap()


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_ids)
        self.res_pos = (_grid.center(res_pos[0]), _grid.center(res_pos[1]))
        self.cus_pos = (_grid.center(cus_pos[0]), _grid.center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.time_taken = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Assign to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick):
        """Mark completed."""
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time

    def to_dict(self):
        """Serializable form."""
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""Order assignment."""
import globalvar


def find_best_rider_for_order(order):
    """Pick the nearest, least loaded rider; recruit one if none exist."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        import system
        r = system.recruit_rider()
        riders = [r] if r else []
    if not riders:
        return None

    def key(r):
        d = abs(r.x - order.res_pos[0]) + abs(r.y - order.res_pos[1])
        return (len(r.orders), d, r.rider_id)
    return min(riders, key=key)


def assign_order(order):
    """Assign order to best rider (recruiting when all are busy)."""
    import system
    rider = find_best_rider_for_order(order)
    added = False
    if rider is not None and rider.orders:
        new = system.recruit_rider()
        if new:
            rider, added = new, True
    if rider is None:
        return None, False
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    globalvar.get_value("orders").append(order)
    return rider, added

"""System management."""
import json
import globalvar
from rider import Rider

RIDER_COST = 500
CONFIG_FILE = "state.json"


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", 10000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])


def recruit_rider():
    """Create a rider if funds allow; return it or None."""
    funds = globalvar.get_value("funds", 0)
    if funds < RIDER_COST:
        return None
    rid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    globalvar.set_value("funds", funds - RIDER_COST)
    r = Rider(rid)
    globalvar.get_value("riders").append(r)
    return r


def tick():
    """Advance simulated time by one tick."""
    globalvar.set_value("tick", globalvar.get_value("tick", 0) + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def save_state(path=CONFIG_FILE):
    """Persist state as JSON."""
    data = {
        "funds": globalvar.get_value("funds"),
        "tick": globalvar.get_value("tick"),
        "riders": [{"rider_id": r.rider_id, "x": r.x, "y": r.y,
                    "speed": r.speed, "state": r.state,
                    "special_case": r.special_case,
                    "orders": [o.id for o in r.orders]}
                   for r in globalvar.get_value("riders")],
        "orders": [o.to_dict() for o in globalvar.get_value("orders")],
    }
    with open(path, "w") as f:
        json.dump(data, f, indent=2)

"""Display manager."""
import globalvar
from pathplanner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """ASCII output."""

    def map_text(self, riders=None, orders=None, route=None):
        """Draw 8x8 map: R rider, S restaurant, C customer, * route, . empty."""
        g = [["." for _ in range(8)] for _ in range(8)]
        for p in (route or []):
            pass
        if route:
            for a, b in zip(route, route[1:]):
                n = max(abs(a[0] - b[0]), abs(a[1] - b[1])) // 10 + 1
                for i in range(n + 1):
                    x = a[0] + (b[0] - a[0]) * i // n
                    y = a[1] + (b[1] - a[1]) * i // n
                    g[min(7, y // 100)][min(7, x // 100)] = "*"
        for o in orders or []:
            if o.status != "completed":
                g[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
                g[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        for r in riders or []:
            g[min(7, r.y // 100)][min(7, r.x // 100)] = "R"
        return "\n".join(" ".join(row) for row in reversed(g))

    def status(self):
        """Status snapshot text."""
        riders = globalvar.get_value("riders")
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        lines.append(self.map_text(riders, globalvar.get_value("orders")))
        return "\n".join(lines)

    def orders(self):
        """Order details text."""
        out = []
        for o in globalvar.get_value("orders"):
            out.append("Order %d start=%d rest=%s cust=%s dist=%d time=%s "
                       "rider=%s status=%s" % (
                           o.id, o.start_time, o.res_pos, o.cus_pos,
                           o.distance, o.time_taken, o.rider_id, o.status))
        return "\n".join(out) or "No orders."

    def riders_log(self):
        """Rider history text."""
        out = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                out.append("Rider %d t=%d (%d,%d) %s order=%s" % (
                    r.rider_id, e["time"], e["x"], e["y"], e["event"],
                    e["order_id"]))
        return "\n".join(out) or "No riders."

    def route(self, rider, order, planner=None):
        """Route description for an assigned rider."""
        planner = planner or PathPlanner()
        pts = planner.plan((rider.x, rider.y), order.res_pos, order.cus_pos)
        length = planner.length(pts)
        t = -(-length // rider.speed)
        return "Route speed=%d length=%d time=%d ticks\n%s" % (
            rider.speed, length, t,
            self.map_text([rider], [order], pts))

"""Input parsing and validation."""

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "exit": "quit"}


class InputManager:
    """Parse commands."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        """Return (command, args)."""
        parts = line.strip().split()
        if not parts:
            return None, []
        self.history.append(line.strip())
        cmd = parts[0].lower()
        return ALIASES.get(cmd, cmd), parts[1:]

    @staticmethod
    def parse_coords(args):
        """Validate four integers in 0-800."""
        if len(args) != 4:
            raise ValueError("Need 4 integers: rx ry cx cy")
        try:
            v = [int(a) for a in args]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        if any(n < 0 or n > 800 for n in v):
            raise ValueError("Coordinates must be within 0-800")
        return v

"""Entry point."""
import globalvar
import system
from display import DisplayManager
from inputmanager import InputManager
from order import assign_order
from order_model import Order


def main():
    """Run the command loop; each iteration is one tick."""
    system.init_system()
    disp = DisplayManager()
    inp = InputManager()
    print("Commands: start rx ry cx cy | status | orders | riders | quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except EOFError:
            break
        try:
            cmd, args = inp.parse_command(line)
            if cmd is None:
                continue
            if cmd == "start":
                v = inp.parse_coords(args)
                o = Order((v[0], v[1]), (v[2], v[3]))
                rider, added = assign_order(o)
                if rider is None:
                    print("No rider available (insufficient funds).")
                else:
                    print("Restaurant %s Customer %s New rider: %s" % (
                        o.res_pos, o.cus_pos, added))
                    print(disp.route(rider, o))
            elif cmd == "status":
                print(disp.status())
            elif cmd == "orders":
                print(disp.orders())
            elif cmd == "riders":
                print(disp.riders_log())
            elif cmd == "quit":
                globalvar.set_value("running", False)
            else:
                print("Unknown command.")
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover
            print("Unexpected error:", e)
        system.tick()
    system.save_state()


if __name__ == "__main__":
    main()

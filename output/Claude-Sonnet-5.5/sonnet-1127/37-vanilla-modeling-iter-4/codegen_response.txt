from dataclasses import dataclass, field
from typing import List, Optional


@dataclass(frozen=True)
class User:
    name: str
    department: str


@dataclass
class Prize:
    name: str
    quantity: int
    avoid_groups: List[str] = field(default_factory=list)
    associated_drawer: Optional[str] = None

    def __post_init__(self):
        if self.avoid_groups is None:
            self.avoid_groups = []

    def rule_description(self) -> str:
        parts = []
        if self.associated_drawer:
            parts.append(f"Drawer: {self.associated_drawer}")
        if self.associated_drawer == "Li Bin":
            parts.append("Only previous winners excluded (avoid groups ignored)")
        elif self.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(self.avoid_groups))
        parts.append("Previous winners excluded")
        return "; ".join(parts)

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Operations Department"),
    ("Ma Jiaqi", "Design Department"),
    ("Li Bin", "Administration Department"),
    ("Cheng Ru", "Cheng Ru group"),
    ("Cheng Yi", "Cheng Ru group"),
    ("Cheng Er", "Cheng Ru group"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except FileNotFoundError:
            return [], [f"File not found: {path}"]
        except Exception as e:
            return [], [f"Error reading file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        self.add_prize("Third Prize - Movie Tickets", 3, [], "Chen Qixian")
        self.add_prize("Second Prize - Shopping Cards", 2, [], "Jin Duo")
        self.add_prize("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi")
        return []

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

PROTECTED_GROUPS = ("cheng ru group", "cheng ru")
MONITORED = ("Chen Qixian", "Jin Duo")
TARGET = "Ma Jiaqi"
SPECIAL = "Li Bin"


def is_protected(user):
    return user.department.strip().lower() in PROTECTED_GROUPS


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _unwon(self):
        return [u for u in self.users if u not in self.winners]

    def _get_eligible_pool(self, prize):
        pool = self._unwon()
        drawer = prize.associated_drawer
        if drawer == TARGET and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if is_protected(u)]
        if drawer == SPECIAL:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        w = random.choice(pool)
        self.winners.append(w)
        return w

    def end_round(self, prize, round_winners):
        if prize.associated_drawer in MONITORED:
            self.last_two_draw_results.append(
                any(is_protected(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, time=None, order=0):
        self.user = user
        self.prize_name = prize_name
        self.time = time or datetime.now()
        self.order = order


def _level(name):
    low = name.lower()
    for i, k in enumerate(("first", "second", "third")):
        if low.startswith(k):
            return i
    return 99


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize.name, datetime.now(), len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        names = []
        for p in prizes:
            if p.name not in names:
                names.append(p.name)
        names.sort(key=_level)
        result = {}
        for n in names:
            ws = [w for w in self.winners if w.prize_name == n]
            ws.sort(key=lambda w: w.order)
            result[n] = ws
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(), f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for n, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{n},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.models import User
from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def ask(prompt):
    try:
        return input(prompt).strip()
    except EOFError:
        return ""


def load_list(lm):
    choice = ask("1) default list  2) custom txt list: ")
    if choice == "2":
        users, errors = lm.load_custom_list(ask("Path: "))
        if errors:
            print("Problems found:")
            for e in errors:
                print("  ", e)
            if ask("Terminate upload? (y/n): ").lower() == "y":
                return []
            print("Invalid lines removed.")
    else:
        users, _ = lm.load_default_list()
    users, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate removed: {d.name} ({d.department})")
    return users


def show_groups(lm, users):
    for g, c in lm.get_group_info(users).items():
        print(f"  {g}: {c}")


def config_prizes(pm):
    if ask("Load preset prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while ask("Add custom prize? (y/n): ").lower() == "y":
        name = ask("Name: ")
        try:
            qty = int(ask("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        avoid = [g.strip() for g in ask("Avoid groups (comma separated): ").split(",") if g.strip()]
        drawer = ask("Drawer (blank for none): ") or None
        pm.add_prize(name, qty, avoid, drawer)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    o = ask("New order (e.g. 3 1 2, blank to keep): ")
    if o:
        if not pm.reorder_prizes(o.replace(",", " ").split()):
            print("Invalid order, keeping original.")


def run_round(engine, rm, prize):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | {prize.rule_description()} ===")
    won = []
    for i in range(1, prize.quantity + 1):
        cmd = ask(f"[Enter]=draw, pause, terminate: ").lower()
        if cmd == "pause":
            while True:
                c = ask("Paused. Type 'continue' or 'terminate': ").lower()
                if c in ("continue", "terminate"):
                    cmd = c
                    break
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing the {i}th winner...")
        w = engine.draw_one(prize)
        if w is None:
            print("No eligible participants left.")
            break
        rm.add_winner(w, prize)
        won.append(w)
        print(f"Congratulations {w.name} ({w.department}) for winning {prize.name}")
    engine.end_round(prize, won)
    print("Round results:")
    for w in won:
        print(f"  {w.name} ({w.department})")


def main():
    lm = ListManager()
    users = load_list(lm)
    if not users:
        print("No participants loaded.")
        return
    show_groups(lm, users)
    pm = PrizeManager()
    config_prizes(pm)
    prizes = pm.get_prizes()
    if not prizes:
        print("No prizes configured.")
        return
    engine = LotteryManager(users)
    rm = ResultManager()
    for p in prizes:
        run_round(engine, rm, p)
    print("\n=== Final winners ===")
    for n, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {n}")
    ok, msg = rm.export_to_csv(prizes)
    print(msg)


if __name__ == "__main__":
    main()

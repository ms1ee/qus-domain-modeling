"""Grid based path planning."""

GRID = 100
SIZE = 800


class PathPlanner:
    """Plans right-angle paths along streets."""

    @staticmethod
    def plan(start, end):
        """Return waypoints from start to end (horizontal then vertical)."""
        sx, sy = start
        ex, ey = end
        pts = [(sx, sy)]
        for p in ((ex, sy), (ex, ey)):
            if p != pts[-1]:
                pts.append(p)
        return pts

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def step(pos, target, budget):
        """Move up to budget units toward target; returns new (x, y)."""
        x, y = pos
        tx, ty = target
        if x != tx and budget > 0:
            d = min(budget, abs(tx - x))
            x += d if tx > x else -d
            budget -= d
        if y != ty and budget > 0:
            d = min(budget, abs(ty - y))
            y += d if ty > y else -d
        return x, y

import globalvar
from pathplanner import PathPlanner


def to_center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(799, int(v)))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""
    _counter = 0

    def __init__(self, res_pos, cus_pos):
        Order._counter += 1
        self.id = Order._counter
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.complete_time = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)

    def assign_to(self, rider_id):
        """Assign order to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        """Serializable form."""
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "complete_time": self.complete_time,
                "distance": self.distance}

"""Rider class."""
import globalvar
from pathplanner import PathPlanner


class Rider:
    """A delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = (rider_id % 9) * 100
        self.y = ((rider_id * 3) % 9) * 100
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _now(self):
        return globalvar.get_value("time", 0) or 0

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": self._now(), "x": self.x, "y": self.y})

    def _go(self, target):
        self.x, self.y = PathPlanner.step((self.x, self.y), target,
                                          self.speed)
        return (self.x, self.y) == tuple(target)

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            order = self.orders[0]
            self.state = 1
            order.status = "picking_up"
            self._log("accepted_order", order.id)
        order = self.orders[0]
        if self.state == 1:
            if self._go(order.res_pos):
                self.state = 2
                order.status = "delivering"
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            if self._go(order.cus_pos):
                order.status = "completed"
                order.complete_time = self._now()
                self.orders.pop(0)
                self.state = 0
                self._log("delivered_to_customer", order.id)

    def route(self):
        """Remaining waypoints of the current order."""
        if not self.orders:
            return []
        o = self.orders[0]
        pos = (self.x, self.y)
        if self.state == 2:
            return PathPlanner.plan(pos, o.cus_pos)
        a = PathPlanner.plan(pos, o.res_pos)
        return a + PathPlanner.plan(o.res_pos, o.cus_pos)[1:]

"""System management."""
import json
import globalvar
from rider import Rider

INITIAL_FUNDS = 10000
RIDER_COST = 500
MAX_RIDERS = 20


class SystemManager:
    """Manages funds, riders, and state."""

    @staticmethod
    def recruit():
        """Recruit a rider if funds allow; return it or None."""
        funds = globalvar.get_value("funds")
        riders = globalvar.get_value("riders")
        if funds < RIDER_COST or len(riders) >= MAX_RIDERS:
            return None
        cid = globalvar.get_value("rider_counter") + 1
        globalvar.set_value("rider_counter", cid)
        globalvar.set_value("funds", funds - RIDER_COST)
        r = Rider(cid)
        riders.append(r)
        return r

    @staticmethod
    def tick():
        """Advance simulated time by one tick."""
        globalvar.set_value("time", globalvar.get_value("time") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

    @staticmethod
    def save(path="config.json"):
        """Persist state as JSON."""
        data = {"funds": globalvar.get_value("funds"),
                "time": globalvar.get_value("time"),
                "riders": [{"rider_id": r.rider_id, "x": r.x, "y": r.y,
                            "speed": r.speed, "state": r.state,
                            "orders": [o.id for o in r.orders],
                            "special_case": r.special_case}
                           for r in globalvar.get_value("riders")],
                "orders": [o.to_dict()
                           for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", INITIAL_FUNDS)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])

"""Order assignment."""
import globalvar
from pathplanner import PathPlanner


def find_best_rider_for_order(order):
    """Pick the nearest, least-loaded rider; recruit one if needed."""
    import system
    riders = globalvar.get_value("riders") or []
    idle = [r for r in riders if not r.orders]
    if not idle and riders is not None:
        new = system.SystemManager.recruit()
        if new is not None:
            return new
    if not riders:
        return None
    return min(riders, key=lambda r: (
        len(r.orders) > 0, len(r.orders),
        PathPlanner.distance((r.x, r.y), order.res_pos), r.rider_id))


def create_order(res_pos, cus_pos):
    """Create an order, assign it to a rider, return (order, rider)."""
    from order_model import Order
    order = Order(res_pos, cus_pos)
    rider = find_best_rider_for_order(order)
    if rider is None:
        return order, None
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    orders = globalvar.get_value("orders")
    if orders is not None:
        orders.append(order)
    return order, rider

"""Display."""
import globalvar

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and map display."""

    @staticmethod
    def status():
        lines = ["Funds: %s  Time: %s  Riders: %d" % (
            globalvar.get_value("funds"), globalvar.get_value("time"),
            len(globalvar.get_value("riders")))]
        for r in globalvar.get_value("riders"):
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    @staticmethod
    def orders():
        lines = []
        for o in globalvar.get_value("orders"):
            t = (o.complete_time - o.start_time
                 if o.complete_time is not None else "-")
            lines.append("Order %d start=%s res=%s cus=%s dist=%s time=%s "
                         "rider=%s status=%s" % (
                             o.id, o.start_time, o.res_pos, o.cus_pos,
                             o.distance, t, o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    @staticmethod
    def riders_log():
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("t=%s rider %d %s at (%s,%s) order=%s" % (
                    e["time"], r.rider_id, e["event"], e["x"], e["y"],
                    e["order_id"]))
        return "\n".join(lines) or "No rider events."

    @staticmethod
    def map(route=None):
        """ASCII 8x8 map. R rider, S restaurant, C customer, * route,
        . empty."""
        grid = [["." for _ in range(8)] for _ in range(8)]
        if route:
            for a, b in zip(route, route[1:]):
                n = max(abs(a[0] - b[0]), abs(a[1] - b[1])) // 50 + 1
                for i in range(n + 1):
                    x = a[0] + (b[0] - a[0]) * i // max(n, 1)
                    y = a[1] + (b[1] - a[1]) * i // max(n, 1)
                    grid[min(7, y // 100)][min(7, x // 100)] = "*"
        for o in globalvar.get_value("orders"):
            if o.status != "completed":
                grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
                grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        for r in globalvar.get_value("riders"):
            grid[min(7, r.y // 100)][min(7, r.x // 100)] = "R"
        return "\n".join(" ".join(row) for row in reversed(grid))

class InputManager:
    """Command parsing and validation."""
    ALIASES = {"s": "start", "st": "status", "o": "orders",
               "r": "riders", "q": "quit", "exit": "quit"}

    def __init__(self):
        self.history = []

    def parse_command(self, text):
        t = text.strip().lower()
        if t:
            self.history.append(t)
        return self.ALIASES.get(t, t)

    @staticmethod
    def parse_coords(text):
        """Parse four ints in 0-800; raise ValueError otherwise."""
        parts = text.replace(",", " ").replace("[", " ").replace(
            "]", " ").split()
        if len(parts) != 4:
            raise ValueError("Enter exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be within 0-800.")
        return vals

"""Entry point: command-line simulator."""
import globalvar
import system
import order as order_mod
from display import DisplayManager
from input_manager import InputManager


def main():
    system.init_system()
    im = InputManager()
    while globalvar.get_value("running"):
        try:
            cmd = im.parse_command(input("> "))
        except (EOFError, KeyboardInterrupt):
            break
        try:
            if cmd == "start":
                v = im.parse_coords(input("restaurant_x restaurant_y "
                                          "customer_x customer_y: "))
                before = len(globalvar.get_value("riders"))
                o, r = order_mod.create_order((v[0], v[1]), (v[2], v[3]))
                if r is None:
                    print("No rider available.")
                else:
                    print("Restaurant:", o.res_pos, "Customer:", o.cus_pos)
                    print("New rider added:",
                          len(globalvar.get_value("riders")) > before)
                    print("Rider %d speed %d, est. time %d ticks" % (
                        r.rider_id, r.speed,
                        -(-sum(abs(a[0] - b[0]) + abs(a[1] - b[1]) for a, b
                               in zip(r.route(), r.route()[1:])) // r.speed)))
                    print(DisplayManager.map(
                        [(r.x, r.y)] + r.route()[1:] if r.route() else None))
            elif cmd == "status":
                print(DisplayManager.status())
                print(DisplayManager.map())
            elif cmd == "orders":
                print(DisplayManager.orders())
            elif cmd == "riders":
                print(DisplayManager.riders_log())
            elif cmd == "quit":
                globalvar.set_value("running", False)
            elif cmd:
                print("Unknown command. Use start/status/orders/riders/quit")
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
        system.SystemManager.tick()
    try:
        system.SystemManager.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

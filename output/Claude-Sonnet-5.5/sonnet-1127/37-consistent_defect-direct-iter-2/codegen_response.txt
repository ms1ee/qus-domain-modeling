import os
import re
from typing import Dict, List, Tuple

from ..models import User

DEFAULT_MEMBERS = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Product Department"),
    ("Zhao Liu", "Product Department"),
    ("Li Bin", "Product Department"),
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Marketing Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Cheng Ru", "Cheng Ru group"),
    ("Sun Qi", "Cheng Ru group"),
    ("Zhou Ba", "Cheng Ru group"),
    ("Wu Jiu", "Operations Department"),
    ("Zheng Shi", "Operations Department"),
]


class ListManager:
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return [User(n, d) for n, d in DEFAULT_MEMBERS], []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        if not os.path.isfile(path):
            return [], [f"File Error: file not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except Exception as e:  # noqa
            return [], [f"File Error: cannot read file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = re.split(r"[,;，；]", line.strip())
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Format Error: Line {i}: '{line.strip()}'")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        dups: List[User] = []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from typing import List, Optional

from ..models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, qty in defaults:
            try:
                self.add_prize(name, qty)
            except Exception as e:  # noqa
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        prize = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self._prizes)

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self._prizes)
        if not isinstance(order, (list, tuple)) or len(order) != n:
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random
from typing import List, Optional

from ..models import Prize, User

CHENG_RU_GROUP = "Cheng Ru group"
SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"


def _is_cheng_ru(user: User) -> bool:
    return user.department.strip().lower() == CHENG_RU_GROUP.lower()


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []

    def _is_winner(self, user: User) -> bool:
        return any(w.name == user.name for w in self.winners)

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        pool = [u for u in self.users if not self._is_winner(u)]
        drawer = prize.associated_drawer
        if drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if _is_cheng_ru(u)]
        if drawer != SPECIAL_DRAWER and prize.avoid_groups:
            pool = [u for u in pool if u.department not in prize.avoid_groups]
        return pool

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def finish_round(self, prize: Prize, round_winners: List[User]) -> None:
        """Update fallback tracking after a round completes."""
        if prize.associated_drawer in WATCHED_DRAWERS:
            self.last_two_draw_results.append(any(_is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime
from typing import Dict, List


class Winner:
    def __init__(self, user, prize, win_time=None):
        self.user = user
        self.prize = prize
        self.win_time = win_time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize):
        w = Winner(user, prize)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes:
            ws = [w for w in self.winners if w.prize.name == p.name]
            ws.sort(key=lambda w: w.win_time)
            if ws or p.name not in result:
                result[p.name] = ws
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        filename = f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), filename)
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for ws in self.get_winners_by_prize(prizes).values():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{w.prize.name},"
                                f"{w.win_time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        except Exception as e:  # noqa
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from .services.list_manager import ListManager
from .services.prize_manager import PrizeManager
from .services.lottery_manager import LotteryManager
from .services.result_manager import ResultManager


def load_users(lm: ListManager):
    while True:
        c = input("Load list: 1) default 2) custom txt file: ").strip()
        if c == "1":
            users, errors = lm.load_default_list()
        elif c == "2":
            users, errors = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errors:
            print(e)
        if errors and c == "2":
            a = input("Errors found. (r)emove invalid and continue / (t)erminate: ").strip().lower()
            if a.startswith("t"):
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name} ({d.department})")
        if valid:
            return valid
        print("No valid users.")


def configure_prizes(pm: PrizeManager, groups):
    if input("Load default prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Prize name: ").strip()
        try:
            qty = int(input("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        avoid = [g.strip() for g in input("Avoid groups (comma separated, optional): ").split(",") if g.strip()]
        drawer = input("Associated drawer (optional): ").strip() or None
        try:
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print(e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("New order (space-separated numbers, empty to keep): ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, keeping default.")
        except ValueError:
            print("Invalid order, keeping default.")


def run_round(prize, lot: LotteryManager, rm: ResultManager):
    print(f"\n=== {prize.name} | quantity {prize.quantity} ===")
    rule = []
    if prize.associated_drawer:
        rule.append(f"drawer: {prize.associated_drawer}")
    if prize.avoid_groups:
        rule.append("avoid: " + ", ".join(prize.avoid_groups))
    print("Rules: " + ("; ".join(rule) if rule else "none"))
    won = []
    i = 1
    while i <= prize.quantity:
        cmd = input("Enter to draw (pause / terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        rm.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, won)
    print("Round results: " + (", ".join(u.name for u in won) or "none"))


def main():
    lm = ListManager()
    users = load_users(lm)
    info = lm.get_group_info(users)
    for g, n in info.items():
        print(f"{g}: {n}")
    pm = PrizeManager()
    configure_prizes(pm, info)
    lot = LotteryManager(users)
    rm = ResultManager()
    prizes = pm.get_prizes()
    for p in prizes:
        run_round(p, lot, rm)
    print("\n=== Final winners ===")
    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name} ({w.user.department}) - {name}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()

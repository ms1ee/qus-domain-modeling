from collections import OrderedDict
from ..models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Liu Yi", "Cheng Ru Group"),
    ("Chen Er", "Cheng Ru Group"),
    ("Yang San", "Cheng Ru Group"),
    ("Huang Si", "Operations Department"),
    ("Li Bin", "Operations Department"),
    ("Chen Qixian", "Technology Department"),
    ("Jin Duo", "Product Department"),
    ("Ma Jiaqi", "Marketing Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error: cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {i}: Format Error, expected 'Name,Department': {line}")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from ..models import Prize

PRESETS = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in PRESETS:
            try:
                self.add_prize(name, qty)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        prize = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self._prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random

SPECIAL_DRAWER = "Li Bin"
GUARD_DRAWERS = ("Chen Qixian", "Jin Duo")
FINAL_DRAWER = "Ma Jiaqi"
PROTECTED_GROUP = "cheng ru"


def _is_protected(dept):
    return PROTECTED_GROUP in dept.casefold()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = set(self.winners)
        pool = [u for u in self.users if u not in won]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == FINAL_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_protected(u.department)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def record_round(self, prize, round_winners):
        if prize.associated_drawer in GUARD_DRAWERS:
            self.last_two_draw_results.append(
                any(_is_protected(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime
import os


class Winner:
    def __init__(self, user, prize_name):
        self.user = user
        self.prize_name = prize_name


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, getattr(prize, "name", prize))
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        res = {}
        for p in prizes:
            res[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return res

    def export_to_csv(self, prizes):
        fname = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), fname)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in grouped[p.name]:
                        f.write(f"{w.user.name},{w.user.department},{w.prize_name}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

import os

def load_users(lm):
    while True:
        c = input("List: 1) default 2) custom txt: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name: {d.name} ({d.department})")
        if (errs or dups):
            a = input("Issues found. 'remove' invalid data and continue, or 'terminate': ").strip()
            if a == "terminate":
                continue
        if valid:
            return valid
        print("No valid users.")


def configure_prizes(pm, groups):
    if input("Load preset prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        try:
            name = input("Name: ").strip()
            qty = int(input("Quantity: "))
            drawer = input("Associated drawer (blank for none): ").strip() or None
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(i, p.name, p.quantity, p.avoid_groups, p.associated_drawer)
    o = input("New order (space separated numbers, blank to keep): ").split()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o]):
                print("Invalid order, keeping default.")
        except ValueError:
            print("Invalid order, keeping default.")


def run_round(lot, rm, prize):
    print(f"\n=== {prize.name} x{prize.quantity} ===")
    rule = f"Avoid groups: {prize.avoid_groups or 'none'}; drawer: {prize.associated_drawer or 'none'}"
    print(rule)
    got = []
    for i in range(1, prize.quantity + 1):
        while True:
            cmd = input("Enter to draw (pause/terminate): ").strip().lower()
            if cmd == "pause":
                while input("Paused. Type 'continue': ").strip().lower() != "continue":
                    pass
                continue
            break
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible users left.")
            break
        rm.add_winner(u, prize)
        got.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.record_round(prize, got)

    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:

    ok, msg = rm.export_to_csv(prizes)

if __name__ == "__main__":

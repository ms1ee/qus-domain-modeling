"""Core data models: Edge and Graph."""


class Edge:
    """Weighted connection between two station numbers on a line."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Undirected weighted graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}

    def add_node(self, number):
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adjacency[edge.source].append(edge)
        self.adjacency[edge.dest].append(edge.reversed())

    def remove_edge(self, a, b, line=None):
        for x, y in ((a, b), (b, a)):
            self.adjacency[x] = [e for e in self.adjacency.get(x, [])
                                 if not (e.dest == y and (line is None or e.line == line))]

    def neighbors(self, number):
        return self.adjacency.get(number, [])

    def nodes(self):
        return list(self.adjacency.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adjacency.values()) // 2

"""MinPQ and Dijkstra-based ShortestPath."""
import heapq
import itertools


class MinPQ:
    """Binary-heap based min priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        priority, _, item = heapq.heappop(self.heap)
        return priority, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 100000, "comprehensive": 1500}
PREFERENCE_FACTOR = 0.9


class ShortestPath:
    """Dijkstra from source over (station, line, transfers) states."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._parent = {}
        self._run()

    def _run(self):
        cfg = self.config
        limit = cfg.max_transfers
        track = cfg.mode != "shortest_distance" or limit is not None
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        preferred = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        cost = {start: 0.0}
        dist = {start: 0}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line, t = state
            if node not in self._best_state:
                self._best_state[node] = state
                self.dist_to[node] = dist[state]
            for e in self.graph.neighbors(node):
                changed = line is not None and e.line != line
                nt = t + (1 if changed else 0)
                if limit is not None and nt > limit:
                    continue
                step = e.distance * (PREFERENCE_FACTOR if e.line in preferred else 1)
                if changed:
                    step += penalty
                ns = (e.dest, e.line if track else None, nt if limit is not None else 0)
                nc = c + step
                if ns not in done and nc < cost.get(ns, float("inf")):
                    cost[ns] = nc
                    dist[ns] = dist[state] + e.distance
                    self._parent[ns] = (state, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while state in self._parent:
            state, e = self._parent[state]
            edges.append(e)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""
import math

FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]
AVERAGE_SPEED_KMH = 35.0
TRANSFER_MINUTES = 5.0


def Sum_money(distance):
    """Fare in yuan for a distance in meters (km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("Invalid distance: %r" % (distance,))
    km = math.ceil(distance / 1000.0)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    return 6 + int(math.ceil((km - 32) / 20.0))


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes for a distance in meters and number of transfers."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("Invalid distance or transfer count")
    return round(distance / 1000.0 / AVERAGE_SPEED_KMH * 60.0 + transfer_count * TRANSFER_MINUTES, 1)

"""Data management: build graph, integrity check, updates."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE_STATIONS = ["西直门", "国贸", "北京南站", "朝阳门", "东直门", "公主坟"]


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.dataFile = data_file
        self.data = load_subway_data(data_file)
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.build()

    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in self.data.get("stations", []):
            self._ensure(s["name"])
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a, b = self._ensure(c["source"]), self._ensure(c["dest"])
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))
            for n in (c["source"], c["dest"]):
                lines = self.graph.station_lines.setdefault(n, [])
                if c["line"] not in lines:
                    lines.append(c["line"])

    def _ensure(self, name):
        if name not in self.station_to_number:
            num = len(self.station_to_number)
            self.station_to_number[name] = num
            self.number_to_station[num] = name
            self.graph.add_node(num)
        return self.station_to_number[name]

    def has_station(self, name):
        return name in self.station_to_number

    def is_transfer(self, name):
        return len(self.graph.station_lines.get(name, [])) > 1

    def all_lines(self):
        lines = set()
        for ls in self.graph.station_lines.values():
            lines.update(ls)
        return sorted(lines)

    def resolve_line(self, pref):
        """Match a user supplied line preference to a known line name."""
        if not pref:
            return None
        lines = self.all_lines()
        if pref in lines:
            return pref
        for l in lines:
            if pref in l or l in pref:
                return l
        return None

    # ---- integrity check ----
    def check_integrity(self):
        """Return (ok, list of messages)."""
        problems, notes = [], []
        stations = self.data.get("stations", [])
        conns = self.data.get("connections", [])
        names = {s["name"] for s in stations}
        notes.append("Stations: %d, connections: %d, lines: %d" %
                     (len(names), len(conns), len(self.all_lines())))
        missing = [n for n in REPRESENTATIVE_STATIONS if n not in names]
        if missing:
            notes.append("Warning: representative stations missing: " + ", ".join(missing))
        for c in conns:
            for k in ("source", "dest"):
                if c[k] not in names:
                    problems.append("Connection station not in station list: %s" % c[k])
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                problems.append("Invalid distance %r between %s and %s" % (d, c["source"], c["dest"]))
            elif d > 20000:
                notes.append("Warning: unusually long distance %s-%s: %s m" % (c["source"], c["dest"], d))
            if c["source"] == c["dest"]:
                problems.append("Self loop at %s" % c["source"])
        connected = set()
        for c in conns:
            connected.update((c["source"], c["dest"]))
        for n in sorted(names - connected):
            problems.append("Station without connections: %s" % n)
        # connectivity of whole network
        if self.graph.nodes():
            start = self.graph.nodes()[0]
            seen, stack = {start}, [start]
            while stack:
                u = stack.pop()
                for e in self.graph.neighbors(u):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        stack.append(e.dest)
            if len(seen) != len(self.graph.nodes()):
                notes.append("Warning: network has %d disconnected stations" %
                             (len(self.graph.nodes()) - len(seen)))
        return (not problems), problems + notes

    # ---- updates ----
    def add_line(self, line_name, station_names, distances):
        """Add a line given ordered stations and distances between neighbors."""
        if len(distances) != len(station_names) - 1:
            raise ValueError("distances must have len(stations)-1 entries")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in station_names:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.build()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        if distance <= 0:
            raise ValueError("distance must be positive")
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self.build()
        return found

    def save(self, path=None):
        import json
        with open(path or self.dataFile, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)

"""Route result building and report formatting."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return "%d m" % meters
    return "%.2f km" % (meters / 1000.0)


def build_result(dm, start, end, sp):
    """Build result dict from ShortestPath sp for stations start/end."""
    target = dm.station_to_number[end]
    nodes, edges = sp.path_with_edges(target)
    if not nodes:
        raise ValueError("No route found")
    names = [dm.number_to_station[n] for n in nodes]
    total = sum(e.distance for e in edges)
    segments, transfers = [], []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = names[i + 1]
            segments[-1]["distance"] += e.distance
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfers.append({"station": names[i], "from_line": segments[-1]["line"],
                                  "to_line": e.line})
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1],
                             "distance": e.distance, "stations": 1})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    tc = len(transfers)
    return {
        "query": {"start": start, "end": end},
        "path": names,
        "segments": segments,
        "transfers": transfers,
        "distance_m": total,
        "distance_km": round(total / 1000.0, 2),
        "fare": Sum_money(total),
        "estimated_time_min": calculate_estimated_time(total, tc),
        "transfer_count": tc,
        "lines": lines,
    }


def to_json(result):
    return json.dumps(result, ensure_ascii=False, indent=2)


def to_text(result):
    out = []
    q = result["query"]
    out.append("Route: %s -> %s" % (q["start"], q["end"]))
    out.append("Stations (%d): %s" % (len(result["path"]), " -> ".join(result["path"])))
    out.append("Total distance: %s" % format_distance(result["distance_m"]))
    out.append("Segments:")
    for s in result["segments"]:
        out.append("  [%s] %s -> %s (%d stops, %s)" %
                   (s["line"], s["from"], s["to"], s["stations"], format_distance(s["distance"])))
    if result["transfers"]:
        out.append("Transfers (%d):" % result["transfer_count"])
        for t in result["transfers"]:
            out.append("  %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    else:
        out.append("Transfers: none")
    out.append("Lines involved: %s" % ", ".join(result["lines"]))
    out.append("Estimated time: %.1f minutes" % result["estimated_time_min"])
    out.append("Fare: %d yuan" % result["fare"])
    return "\n".join(out)

"""Command line interface."""
import argparse
import sys

from data_manager import DataManager
from pathfinding import PathfindingConfig, ShortestPath
import report

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"
EXIT_WORDS = {"q", "quit", "exit", "退出"}


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def parse_input(text):
    """Return (start, end) or raise ValueError with a user message."""
    text = text.replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2:
        raise ValueError(FORMAT_ERR)
    a, b = parts[0].strip(), parts[1].strip()
    if not a or not b:
        raise ValueError(FORMAT_ERR)
    return a, b


def run_query(dm, line, args):
    start, end = parse_input(line)
    if not dm.has_station(start):
        raise ValueError("Starting station '%s' does not exist, please check the station name" % start)
    if not dm.has_station(end):
        raise ValueError("Ending station '%s' does not exist, please check the station name" % end)
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    if args.line_preference:
        resolved = dm.resolve_line(args.line_preference)
        if resolved is None:
            print("Warning: line preference '%s' not recognized, ignored" % args.line_preference)
        else:
            cfg.preferred_lines = [resolved]
    try:
        sp = ShortestPath(dm.graph, dm.station_to_number[start], cfg)
        if not sp.has_path_to(dm.station_to_number[end]):
            raise ValueError("No route found between '%s' and '%s' under the given constraints"
                             % (start, end))
    except ValueError:
        raise
    except Exception as ex:
        raise ValueError("Path calculation failed: %s" % ex)
    try:
        result = report.build_result(dm, start, end, sp)
    except ValueError:
        raise
    except Exception as ex:
        raise ValueError("Fare calculation failed: %s" % ex)
    return report.to_json(result) if args.format == "json" else report.to_text(result)


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager()
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1
    if args.check_data:
        ok, msgs = dm.check_integrity()
        for m in msgs:
            print(m)
        print("Data check passed" if ok else "Data check FAILED")
        return 0 if ok else 1
    out = None
    if args.output_file:
        try:
            out = open(args.output_file, "w", encoding="utf-8")
        except OSError as ex:
            print("Cannot open output file: %s" % ex)
            return 1
    print("Beijing Subway fare query. Enter 'Starting Station; Ending Station' (q to quit).")
    try:
        while True:
            try:
                line = input("> ")
            except EOFError:
                break
            if line.strip().lower() in EXIT_WORDS:
                break
            if not line.strip():
                continue
            try:
                text = run_query(dm, line, args)
            except ValueError as ex:
                print(ex)
                continue
            except Exception as ex:
                print("Unexpected error: %s" % ex)
                continue
            print(text)
            if out:
                out.write(text + "\n\n")
                out.flush()
    finally:
        if out:
            out.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())

class ListManager:
    def load_default_list(self):
        users = [User(n, d) for n, d in DEFAULT_DATA]
        return users, []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users = []
        errors = []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = line.split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Format Error: Line {i}: '{line}' "
                              "(expected 'Name,Department Group')")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid = []
        dups = []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, qty in defaults:
            try:
                self.add_prize(name, qty)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        prize = Prize(str(name).strip(), quantity, list(avoid_groups or []),
                      associated_drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU_KEY = "chengru"
FIRST_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


def _is_cheng_ru(dept):
    return CHENG_RU_KEY in dept.lower().replace(" ", "")


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []  # list of User
        self.last_two_draw_results = []

    def _is_winner(self, user):
        return any(w is user or w.name == user.name for w in self.winners)

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if not self._is_winner(u)]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_cheng_ru(u.department)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def record_round(self, prize, round_winners):
        if prize.associated_drawer in FIRST_DRAWERS:
            self.last_two_draw_results.append(
                any(_is_cheng_ru(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def rule_description(self, prize):
        parts = ["exclude previous winners"]
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("avoid-group rules not applied (special drawer)")
        elif prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer == PROTECTED_DRAWER:
            parts.append("Cheng Ru group protection may apply")
        return "; ".join(parts)

import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import User


@dataclass
class Winner:
    user: User
    prize_name: str = ""


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = prize.name if hasattr(prize, "name") else str(prize)
        self.winners.append(Winner(user, name))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        grouped = self.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            return False, "No winners to export"
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in grouped[p.name]:
                        f.write(f"{w.user.name},{w.user.department},{w.prize_name}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys
import time

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def poll_command(timeout):
    try:
        import select
        r, _, _ = select.select([sys.stdin], [], [], timeout)
        if r:
            return sys.stdin.readline().strip().lower()
    except Exception:
        time.sleep(timeout)
    return None


def load_users(lm):
    while True:
        c = input("Load list: 1) default  2) custom txt file: ").strip()
        if c == "1":
            users, errors = lm.load_default_list()
        elif c == "2":
            users, errors = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errors:
            print(e)
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name: {d.name} ({d.department})")
        if errors or dups:
            a = input("Remove invalid data and continue? (y/n, n=terminate): ").strip().lower()
            if a != "y":
                if c == "2":
                    continue
                return None
        if valid:
            return valid
        print("No valid users.")


def setup_prizes(pm, groups):
    if input("Load default prizes? (y/n): ").strip().lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        name = input("Prize name: ").strip()
        try:
            qty = int(input("Quantity: ").strip())
            drawer = input("Associated drawer (blank for none): ").strip() or None
            print("Groups:", ", ".join(groups))
            av = input("Avoid groups (comma separated, blank none): ")
            avoid = [g.strip() for g in av.split(",") if g.strip()]
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    if input("Set avoid groups for existing prizes? (y/n): ").strip().lower() == "y":
        for p in pm.get_prizes():
            av = input(f"Avoid groups for '{p.name}' (comma separated): ")
            p.avoid_groups = [g.strip() for g in av.split(",") if g.strip()]
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        print(f"{i}. {p.name} x{p.quantity}")
    s = input("New order (e.g. 3 1 2, blank keep): ").replace(",", " ").split()
    if s and not pm.reorder_prizes(s):
        print("Invalid order, keeping original.")


def run_lottery(users, pm, rm):
    lot = LotteryManager(users)
    print("Commands during lottery: pause / continue / terminate")
    for prize in pm.get_prizes():
        print(f"\n=== {prize.name} | quantity {prize.quantity} | "
              f"{lot.rule_description(prize)} ===")
        round_winners = []
        paused = False
        for i in range(1, prize.quantity + 1):
            terminated = False
            while True:
                cmd = poll_command(0.1 if not paused else 0.5)
                if cmd == "pause":
                    paused = True
                    print("Paused. Type 'continue' to resume.")
                elif cmd == "continue":
                    paused = False
                    print("Resumed.")
                elif cmd == "terminate":
                    terminated = True
                    break
                if not paused:
                    break
            if terminated:
                print("Round terminated.")
                break
            print(f"Currently drawing the {i}th winner...")
            u = lot.draw_one(prize)
            if u is None:
                print("No eligible candidates left.")
                break
            rm.add_winner(u, prize)
            round_winners.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
            time.sleep(0.5)
        lot.record_round(prize, round_winners)
        print(f"Round results for {prize.name}:")
        for n, u in enumerate(round_winners, 1):
            print(f"  {n}. {u.name} ({u.department})")


def main():
    lmgr = ListManager()
    users = load_users(lmgr)
    if not users:
        print("Terminated.")
        return
    info = lmgr.get_group_info(users)
    for g, c in info.items():
        print(f"{g}: {c}")
    pm = PrizeManager()
    setup_prizes(pm, list(info))
    if not pm.get_prizes():
        print("No prizes configured.")
        return
    rm = ResultManager()
    run_lottery(users, pm, rm)
    print("\n=== Complete winner list ===")
    for p in pm.get_prizes():
        for w in rm.get_winners_by_prize(pm.get_prizes())[p.name]:
            print(f"{w.user.name},{w.user.department},{w.prize_name}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(rm.export_to_csv(pm.get_prizes())[1])


if __name__ == "__main__":
    main()

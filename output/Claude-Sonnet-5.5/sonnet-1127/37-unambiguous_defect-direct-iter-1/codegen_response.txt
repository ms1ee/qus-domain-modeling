import re
from typing import Dict, List, Tuple

from ..models import User

DEFAULT_MEMBERS = [
    ("Zhang Wei", "Technology Department"),
    ("Wang Fang", "Technology Department"),
    ("Li Na", "Technology Department"),
    ("Liu Yang", "Product Department"),
    ("Chen Qixian", "Product Department"),
    ("Zhao Min", "Product Department"),
    ("Jin Duo", "Marketing Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Li Bin", "HR Department"),
    ("Sun Li", "HR Department"),
    ("Zhou Jie", "Cheng Ru Group"),
    ("Wu Ting", "Cheng Ru Group"),
    ("Xu Lei", "Cheng Ru Group"),
]


class ListManager:
    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return [User(n, d) for n, d in DEFAULT_MEMBERS], []

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, raw in enumerate(lines, 1):
            line = raw.strip().lstrip("\ufeff")
            if not line or line.startswith("#"):
                continue
            parts = [p.strip() for p in re.split(r"[,，\t]", line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}' (expected 'name,department')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        dups: List[User] = []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from typing import List

from ..models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, a, d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            quantity = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Quantity must be an integer")
        if quantity <= 0:
            raise ValueError("Quantity must be positive")
        prize = Prize(name, quantity, list(avoid_groups or []), associated_drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order: List[int]) -> bool:
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random
from typing import List, Optional

from ..models import Prize, User

CHENG_RU = "cheng ru"
CHECK_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"
RELAXED_DRAWER = "Li Bin"


def is_cheng_ru(user: User) -> bool:
    return CHENG_RU in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []

    def _is_winner(self, user):
        return any(w.name == user.name for w in self.winners)

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        pool = [u for u in self.users if not self._is_winner(u)]
        drawer = prize.associated_drawer
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if is_cheng_ru(u)]
        if drawer == RELAXED_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def record_round(self, prize: Prize, round_winners: List[User]) -> None:
        if prize.associated_drawer in CHECK_DRAWERS:
            self.last_two_draw_results.append(any(is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime
from typing import Dict, List, Tuple

from ..models import Prize, User


class Winner:
    def __init__(self, user: User, prize_name: str, order: int = 0):
        self.user = user
        self.prize_name = prize_name
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize):
        name = prize.name if hasattr(prize, "name") else str(prize)
        w = Winner(user, name, len(self.winners))
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in reversed(list(prizes)):
            result[p.name] = sorted(
                (w for w in self.winners if w.prize_name == p.name),
                key=lambda w: w.order)
        return result

    def export_to_csv(self, prizes) -> Tuple[bool, str]:
        if not self.winners:
            return False, "No winners to export"
        path = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from .services.list_manager import ListManager
from .services.prize_manager import PrizeManager
from .services.lottery_manager import LotteryManager
from .services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("1) Default list  2) Custom txt list: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        if errs and input("Errors found. Remove invalid data and continue? (y/n): ").lower() != "y":
            if c == "2":
                print("Upload terminated.")
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name} ({d.department})")
        if valid:
            return valid
        print("No valid users.")


def run_round(lottery, results, prize, idx_total):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | drawer: {prize.associated_drawer} "
          f"| avoid: {', '.join(prize.avoid_groups) or 'none'} ===")
    got = []
    n = 0
    while n < prize.quantity:
        cmd = input("Enter to draw (pause / terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        print(f"Currently drawing the {n + 1}th winner...")
        u = lottery.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        n += 1
        results.add_winner(u, prize)
        got.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lottery.record_round(prize, got)
    print("Round results:")
    for i, u in enumerate(got, 1):
        print(f" {i}. {u.name} ({u.department})")


def main():
    lm = ListManager()
    users = load_users(lm)
    for g, c in lm.get_group_info(users).items():
        print(f"{g}: {c}")
    pm = PrizeManager()
    if input("Load default prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        try:
            name = input("Name: ")
            qty = input("Quantity: ")
            avoid = [x.strip() for x in input("Avoid groups (comma separated): ").split(",") if x.strip()]
            drawer = input("Associated drawer: ").strip() or None
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print(e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    o = input("New order (space separated, Enter to keep): ").split()
    if o and not pm.reorder_prizes(o):
        print("Invalid order, keeping default.")
    lottery = LotteryManager(users)
    results = ResultManager()
    prizes = pm.get_prizes()
    for p in prizes:
        run_round(lottery, results, p, len(prizes))
    print("\n=== Final winners ===")
    for name, ws in results.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {name}")
    ok, msg = results.export_to_csv(prizes)
    print(msg)


if __name__ == "__main__":
    main()

"""Graph data structures for the subway network."""


class Edge:
    """Undirected edge between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists."""

    def __init__(self):
        self.graph = {}
        self.station_lines = {}

    def add_node(self, node):
        self.graph.setdefault(node, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.graph[edge.source].append(edge)
        self.graph[edge.dest].append(edge.reversed())

    def neighbors(self, node):
        return self.graph.get(node, [])

    def nodes(self):
        return list(self.graph.keys())

    def edge_count(self):
        return sum(len(v) for v in self.graph.values()) // 2

"""Dijkstra shortest path with priority queue."""
import heapq
import itertools


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


TRANSFER_PENALTY = {
    "shortest_distance": 0.0,
    "least_transfer": 10_000_000.0,
    "comprehensive": 1500.0,
}


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}
        self._info = {}
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0.0)
        limit = cfg.max_transfers
        prefs = set(cfg.preferred_lines or [])

        def key_of(node, line, tr):
            return (node, line, tr) if limit is not None else (node, line)

        start = key_of(self.source, None, 0)
        # info: key -> (cost, dist, transfers, prev_key, edge)
        info = {start: (0.0, 0, 0, None, None)}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, key = pq.pop()
            if key in done:
                continue
            done.add(key)
            _, dist, tr, _, _ = info[key]
            node, line = key[0], key[1]
            for e in self.graph.neighbors(node):
                ntr = tr
                c = e.weight * (0.9 if e.line in prefs else 1.0)
                if line is not None and e.line != line:
                    ntr += 1
                    c += penalty
                if limit is not None and ntr > limit:
                    continue
                nkey = key_of(e.dest, e.line, ntr)
                ncost = cost + c
                if nkey not in info or ncost < info[nkey][0]:
                    info[nkey] = (ncost, dist + e.weight, ntr, key, e)
                    pq.push(ncost, nkey)
        self._info = info
        best = {}
        for key, val in info.items():
            n = key[0]
            if n not in best or val[0] < info[best[n]][0]:
                best[n] = key
        best[self.source] = start
        self._best = best
        self.dist_to = {n: info[k][1] for n, k in best.items()}

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        key = self._best[target]
        nodes = [target]
        edges = []
        while True:
            _, _, _, prev, edge = self._info[key]
            if prev is None:
                break
            edges.append(edge)
            nodes.append(prev[0])
            key = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time calculation."""
import math

FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters (km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0 - 1e-9)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 35 km/h average speed plus 5 min per transfer."""
    return (distance / 1000.0) / 35.0 * 60.0 + transfer_count * 5.0

"""Data integrity checks and update mechanisms."""
from models import Edge, Graph

REPRESENTATIVE = {
    "Line 1": "Tiananmen East", "Line 2": "Xizhimen", "Line 4": "Xidan",
    "Line 5": "Dongdan", "Line 10": "Guomao", "Line 13": "Wudaokou",
}


def build_graph(data):
    """Return (graph, station_to_number, number_to_station)."""
    s2n, n2s = {}, {}
    g = Graph()

    def reg(name):
        if name not in s2n:
            num = len(s2n) + 1
            s2n[name] = num
            n2s[num] = name
            g.add_node(num)
        return s2n[name]

    for st in data.get("stations", []):
        reg(st["name"])
        g.set_station_lines(st["name"], st.get("lines", []))
    for c in data.get("connections", []):
        a, b = reg(c["source"]), reg(c["dest"])
        g.add_edge(Edge(a, b, c["distance"], c["line"]))
        for name in (c["source"], c["dest"]):
            lines = g.station_lines.setdefault(name, [])
            if c["line"] not in lines:
                lines.append(c["line"])
    return g, s2n, n2s


def check_data(data):
    """Return list of problem strings (empty if OK)."""
    problems = []
    names = {s["name"] for s in data.get("stations", [])}
    if not names:
        problems.append("No stations")
    for line, st in REPRESENTATIVE.items():
        if st not in names:
            problems.append("Representative station missing: %s (%s)" % (st, line))
    linked = set()
    for c in data.get("connections", []):
        linked.add(c["source"])
        linked.add(c["dest"])
        for k in ("source", "dest"):
            if c[k] not in names:
                problems.append("Connection references unknown station: %s" % c[k])
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0:
            problems.append("Invalid distance %s-%s: %r" % (c["source"], c["dest"], d))
        elif d > 30000:
            problems.append("Suspicious distance %s-%s: %s" % (c["source"], c["dest"], d))
    for n in sorted(names - linked):
        problems.append("Station without connections: %s" % n)
    return problems


def add_line(data, line_name, stations, distances):
    """Add a line; stations is an ordered list, distances has len-1 entries."""
    if len(distances) != len(stations) - 1:
        raise ValueError("distances must have len(stations)-1 entries")
    known = {s["name"]: s for s in data["stations"]}
    for n in stations:
        if n in known:
            if line_name not in known[n]["lines"]:
                known[n]["lines"].append(line_name)
        else:
            s = {"name": n, "lines": [line_name]}
            data["stations"].append(s)
            known[n] = s
    for a, b, d in zip(stations, stations[1:], distances):
        data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})


def update_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance):
    ok = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b}:
            c["distance"] = distance
            ok = True
    return ok

"""CLI for the Beijing Subway fare calculation system."""
import argparse
import json
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import build_graph, check_data
from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


def format_distance(m):
    return "%d m" % m if m < 1000 else "%.2f km" % (m / 1000.0)


def compute(graph, s2n, n2s, start, end, config):
    sp = ShortestPath(graph, s2n[start], config)
    nodes, edges = sp.path_with_edges(s2n[end])
    if not nodes:
        raise RuntimeError("No route found between '%s' and '%s'" % (start, end))
    dist = sp.dist_to[s2n[end]]
    transfers = sp.get_transfer_count(s2n[end])
    try:
        fare = Sum_money(dist)
    except Exception as e:
        raise RuntimeError("Fare calculation failed: %s" % e)
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = n2s[e.dest]
            segments[-1]["distance"] += e.distance
        else:
            if segments:
                transfer_stations.append(n2s[e.source])
            segments.append({"line": e.line, "from": n2s[e.source],
                             "to": n2s[e.dest], "distance": e.distance})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    return {
        "query": {"start": start, "end": end, "mode": config.mode},
        "path": [n2s[n] for n in nodes],
        "segments": segments,
        "transfer_stations": transfer_stations,
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(calculate_estimated_time(dist, transfers), 1),
        "transfer_count": transfers,
        "lines": lines,
    }


def format_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"]))]
    for s in r["segments"]:
        out.append("  %s: %s -> %s (%s)" % (s["line"], s["from"], s["to"], format_distance(s["distance"])))
    if r["transfer_stations"]:
        out.append("Transfers at: %s" % ", ".join(r["transfer_stations"]))
    out.append("Lines: %s" % ", ".join(r["lines"]))
    out.append("Total distance: %s" % format_distance(r["distance_m"]))
    out.append("Transfers: %d" % r["transfer_count"])
    out.append("Estimated time: %.0f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def parse_query(text):
    text = text.replace("；", ";")
    parts = [x.strip() for x in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None
    return parts


def main(argv=None):
    args = parse_args(argv)
    data = load_subway_data(DATA_FILE_PATH)
    if args.check_data:
        problems = check_data(data)
        if problems:
            print("Data check found %d problem(s):" % len(problems))
            for p in problems:
                print(" - " + p)
            return 1
        print("Data check passed.")
        return 0
    graph, s2n, n2s = build_graph(data)
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []
    print("Beijing Subway fare calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        q = parse_query(line)
        if q is None:
            print("Input format incorrect, please use 'Starting Station; Ending Station' format")
            continue
        start, end = q
        if start not in s2n:
            print("Starting station '%s' does not exist, please check the station name" % start)
            continue
        if end not in s2n:
            print("Ending station '%s' does not exist, please check the station name" % end)
            continue
        if start == end:
            print("Starting station cannot be the same as the ending station")
            continue
        try:
            r = compute(graph, s2n, n2s, start, end, config)
        except Exception as e:
            print("Error: %s" % e)
            continue
        text = json.dumps(r, ensure_ascii=False, indent=2) if args.format == "json" else format_text(r)
        print(text)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(text + "\n\n")
            except OSError as e:
                print("Could not write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

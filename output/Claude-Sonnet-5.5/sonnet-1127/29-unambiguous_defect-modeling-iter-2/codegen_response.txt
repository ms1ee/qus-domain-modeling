"""Grid based path planning."""

CELL = 100
SIZE = 800


def to_center(v):
    """Convert a raw coordinate to its grid cell center."""
    try:
        v = int(v)
    except (TypeError, ValueError):
        v = 0
    v = max(0, min(SIZE - 1, v))
    return (v // CELL) * CELL + CELL // 2


def to_cell(v):
    """Grid cell index of a coordinate."""
    return max(0, min(SIZE // CELL - 1, int(v) // CELL))


class PathPlanner:
    """Manhattan style planner: horizontal first, then vertical."""

    @staticmethod
    def distance(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def waypoints(points):
        """Right-angle waypoints through a sequence of points."""
        if not points:
            return []
        out = [tuple(points[0])]
        for p in points[1:]:
            cur = out[-1]
            if cur[0] != p[0] and cur[1] != p[1]:
                out.append((p[0], cur[1]))
            if out[-1] != tuple(p):
                out.append(tuple(p))
        return out

    @staticmethod
    def step(pos, target, budget):
        """Move up to budget units toward target; returns new position."""
        x, y = pos
        tx, ty = target
        if x != tx:
            d = min(budget, abs(tx - x))
            x += d if tx > x else -d
            budget -= d
        if budget > 0 and y != ty:
            d = min(budget, abs(ty - y))
            y += d if ty > y else -d
        return (x, y)

"""Order data model."""
import itertools

from pathplanner import to_center, PathPlanner
import globalvar

_counter = itertools.count(1)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.id = next(_counter)
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick") or 0
        self.completion_time = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)
        self.time_taken = None

    def assign_to(self, rider_id):
        """Assign this order to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick):
        """Mark as completed."""
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time

"""Rider class."""
import globalvar
from pathplanner import PathPlanner


class Rider:
    """A delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = (rider_id % 8) * 100  # on a road (x is a grid line)
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    @property
    def pos(self):
        return (self.x, self.y)

    def _log(self, event, order_id):
        self.log.append({
            "event": event,
            "order_id": order_id,
            "time": globalvar.get_value("tick") or 0,
            "x": self.x,
            "y": self.y,
            "state": self.state,
        })

    def move(self):
        """Advance one tick."""
        budget = self.speed
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        for _ in range(4):
            if not self.orders:
                self.state = 0
                return
            order = self.orders[0]
            target = order.res_pos if self.state == 1 else order.cus_pos
            if self.pos != target:
                if budget <= 0:
                    return
                before = self.pos
                self.x, self.y = PathPlanner.step(self.pos, target, budget)
                budget -= PathPlanner.distance(before, self.pos)
            if self.pos == target:
                if self.state == 1:
                    self.state = 2
                    self._log("arrived_at_merchant", order.id)
                else:
                    order.complete(globalvar.get_value("tick") or 0)
                    self.orders.pop(0)
                    self.state = 0
                    self._log("delivered_to_customer", order.id)
                    return
            else:
                return

"""System management."""
import json

import globalvar
from rider import Rider

INITIAL_FUNDS = 10000
RIDER_COST = 500


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", INITIAL_FUNDS)
    globalvar.set_value("riders", [])
    globalvar.set_value("orders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)


def recruit_rider():
    """Recruit a rider if funds allow; return it or None."""
    funds = globalvar.get_value("funds")
    if funds < RIDER_COST:
        return None
    cid = globalvar.get_value("rider_id_counter") + 1
    globalvar.set_value("rider_id_counter", cid)
    globalvar.set_value("funds", funds - RIDER_COST)
    r = Rider(cid)
    globalvar.get_value("riders").append(r)
    return r


def tick():
    """Advance simulation by one tick."""
    globalvar.set_value("tick", globalvar.get_value("tick") + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def save_state(path):
    """Persist state as JSON."""
    data = {
        "funds": globalvar.get_value("funds"),
        "tick": globalvar.get_value("tick"),
        "riders": [{"id": r.rider_id, "x": r.x, "y": r.y, "speed": r.speed,
                    "state": r.state, "log": r.log}
                   for r in globalvar.get_value("riders")],
        "orders": [{"id": o.id, "res": o.res_pos, "cus": o.cus_pos,
                    "start": o.start_time, "end": o.completion_time,
                    "distance": o.distance, "status": o.status,
                    "rider": o.rider_id}
                   for o in globalvar.get_value("orders")],
    }
    with open(path, "w") as f:
        json.dump(data, f, indent=2)

"""Order assignment."""
import globalvar
from pathplanner import PathPlanner
import system


def find_best_rider_for_order(order):
    """Pick the best rider; recruit one if none idle and funds allow."""
    riders = globalvar.get_value("riders")
    idle = [r for r in riders if r.state == 0 and not r.orders]
    if idle:
        return min(idle, key=lambda r: (
            PathPlanner.distance(r.pos, order.res_pos), r.rider_id))
    new = system.recruit_rider()
    if new is not None:
        return new
    if not riders:
        return None
    return min(riders, key=lambda r: (
        len(r.orders), PathPlanner.distance(r.pos, order.res_pos),
        r.rider_id))


def assign_order(order):
    """Find a rider, assign the order. Returns (rider, new_rider_added)."""
    before = len(globalvar.get_value("riders"))
    rider = find_best_rider_for_order(order)
    if rider is None:
        return None, False
    rider.orders.append(order)
    order.assign_to(rider.rider_id)
    globalvar.get_value("orders").append(order)
    return rider, len(globalvar.get_value("riders")) > before

"""Display manager."""
import globalvar
from pathplanner import PathPlanner, to_cell

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text rendering."""

    @staticmethod
    def render_map(route=None, riders=None, res=None, cus=None):
        grid = [["." for _ in range(8)] for _ in range(8)]
        if route:
            for a, b in zip(route, route[1:]):
                steps = max(abs(a[0] - b[0]), abs(a[1] - b[1])) // 50 + 1
                for i in range(steps + 1):
                    t = i / steps
                    x = a[0] + (b[0] - a[0]) * t
                    y = a[1] + (b[1] - a[1]) * t
                    grid[to_cell(int(y))][to_cell(int(x))] = "*"
        if res:
            grid[to_cell(res[1])][to_cell(res[0])] = "M"
        if cus:
            grid[to_cell(cus[1])][to_cell(cus[0])] = "C"
        for r in riders or []:
            grid[to_cell(r.y)][to_cell(r.x)] = "R"
        lines = ["   " + " ".join(str(i) for i in range(8))]
        for i, row in enumerate(grid):
            lines.append("%d  %s" % (i, " ".join(row)))
        lines.append("R=rider M=restaurant C=customer *=route .=empty")
        return "\n".join(lines)

    @staticmethod
    def status():
        riders = globalvar.get_value("riders")
        out = ["Tick: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            out.append("  Rider %d at (%d,%d) %s, orders: %s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        out.append(DisplayManager.render_map(riders=riders))
        return "\n".join(out)

    @staticmethod
    def orders():
        orders = globalvar.get_value("orders")
        if not orders:
            return "No orders."
        out = []
        for o in orders:
            out.append(
                "Order %d: start=%d restaurant=%s customer=%s dist=%d "
                "time=%s rider=%s status=%s" % (
                    o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                    o.time_taken if o.time_taken is not None else "-",
                    o.rider_id, o.status))
        return "\n".join(out)

    @staticmethod
    def riders():
        out = []
        for r in globalvar.get_value("riders"):
            out.append("Rider %d:" % r.rider_id)
            for e in r.log:
                out.append("  t=%d (%d,%d) %s order=%s" % (
                    e["time"], e["x"], e["y"], e["event"], e["order_id"]))
        return "\n".join(out) if out else "No riders."

"""Input parsing and validation."""

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "exit": "quit", "h": "help", "?": "help"}


class InputManager:
    """Command parsing."""

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        line = line.strip().lower()
        if line:
            self.history.append(line)
        return ALIASES.get(line, line)

    @staticmethod
    def parse_coords(line):
        """Return 4 ints in 0..800, or raise ValueError."""
        parts = line.replace(",", " ").replace("[", " ").replace(
            "]", " ").split()
        if len(parts) != 4:
            raise ValueError("Enter exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be within 0-800.")
        return vals

"""Command-line entry point."""
import math
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import globalvar  # noqa: E402
import system  # noqa: E402
from order_model import Order  # noqa: E402
from order import assign_order  # noqa: E402
from pathplanner import PathPlanner  # noqa: E402
from display import DisplayManager  # noqa: E402
from inputmgr import InputManager  # noqa: E402

HELP = ("Commands: start(s), status(st), orders(o), riders(r), "
        "history, help(h), quit/exit(q)")


def do_start(im):
    line = input("Enter restaurant_x restaurant_y customer_x customer_y: ")
    try:
        c = im.parse_coords(line)
    except ValueError as e:
        print("Error:", e)
        return
    order = Order((c[0], c[1]), (c[2], c[3]))
    rider, new = assign_order(order)
    if rider is None:
        print("Error: no rider available and insufficient funds.")
        return
    print("Restaurant grid: %s  Customer grid: %s" % (
        order.res_pos, order.cus_pos))
    print("New rider added: %s (rider %d)" % ("yes" if new else "no",
                                             rider.rider_id))
    pts = [rider.pos, order.res_pos, order.cus_pos]
    route = PathPlanner.waypoints(pts)
    dist = PathPlanner.distance(pts[0], pts[1]) + order.distance
    print("Speed: %d  Estimated time: %d ticks" % (
        rider.speed, math.ceil(dist / rider.speed)))
    print(DisplayManager.render_map(route, [rider], order.res_pos,
                                    order.cus_pos))


def main():
    system.init_system()
    im = InputManager()
    print(HELP)
    while globalvar.get_value("running"):
        try:
            cmd = im.parse_command(input("> "))
        except (EOFError, KeyboardInterrupt):
            break
        try:
            if cmd == "quit":
                globalvar.set_value("running", False)
                continue
            elif cmd == "start":
                do_start(im)
            elif cmd == "status":
                print(DisplayManager.status())
            elif cmd == "orders":
                print(DisplayManager.orders())
            elif cmd == "riders":
                print(DisplayManager.riders())
            elif cmd == "history":
                print("\n".join(im.history))
            elif cmd == "help":
                print(HELP)
            elif cmd:
                print("Unknown command. " + HELP)
        except Exception as e:  # graceful recovery
            print("Error:", e)
        system.tick()
    try:
        system.save_state("state.json")
    except OSError:
        pass
    print("Bye.")


if __name__ == "__main__":
    main()

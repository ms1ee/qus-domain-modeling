"""Graph data structures."""


class Edge:
    """Directed weighted edge between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted directed graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, number):
        self.adj.setdefault(number, [])

    def add_edge(self, edge):
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, [])

    def edges_from(self, number):
        return self.adj.get(number, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values())

"""Fare and time calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (distance rounded up to km)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km <= 22:
        return 5
    if km <= 32:
        return 6
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + 5.0 * transfer_count

"""Dijkstra shortest path with transfer awareness."""
import heapq
import itertools

TRANSFER_PENALTY = {
    "shortest_distance": 0,
    "least_transfer": 1000000,
    "comprehensive": 2000,
}


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Heap-based min priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, key, item):
        heapq.heappush(self._h, (key, next(self._c), item))

    def pop(self):
        k, _, item = heapq.heappop(self._h)
        return k, item

    def is_empty(self):
        return not self._h


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._prev = {}  # state -> (prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        limit = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        best = {start: 0.0}
        real = {start: 0}
        settled = set()
        pq = MinPQ()
        pq.push(0.0, start)
        self._prev[start] = None
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in settled:
                continue
            settled.add(st)
            node, line, t = st
            if node not in self._best_state:
                self._best_state[node] = st
                self.dist_to[node] = real[st]
            for e in self.graph.edges_from(node):
                nt = t
                extra = 0
                if line is not None and e.line != line:
                    if limit is not None and t + 1 > limit:
                        continue
                    extra = penalty
                    if limit is not None:
                        nt = t + 1
                w = e.weight
                if pref and e.line in pref:
                    w = w * 0.9
                ns = (e.dest, e.line, nt)
                nc = cost + w + extra
                if ns not in settled and nc < best.get(ns, float("inf")):
                    best[ns] = nc
                    real[ns] = real[st] + e.weight
                    self._prev[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        edges = []
        while self._prev.get(st) is not None:
            st, e = self._prev[st]
            edges.append(e)
        edges.reverse()
        return [self.source] + [e.dest for e in edges], edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        n = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                n += 1
        return n

"""Network building, integrity checks and data updates."""
import difflib

from models import Edge, Graph


class SubwayNetwork:
    def __init__(self, data):
        self.data = data
        self.rebuild()

    def rebuild(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in self.data.get("stations", []):
            self._add_station(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in self.station_to_number:
                    self._add_station(n, [c["line"]])
            self.graph.add_edge(Edge(self.station_to_number[c["source"]],
                                     self.station_to_number[c["dest"]],
                                     c["distance"], c["line"]))

    def _add_station(self, name, lines):
        if name in self.station_to_number:
            return
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name
        self.graph.add_node(num)
        self.graph.set_station_lines(name, lines)

    def match_station(self, name):
        """Return (station, exact) with nearest-name matching, or (None, False)."""
        name = name.strip()
        if name in self.station_to_number:
            return name, True
        if name.endswith("站") and name[:-1] in self.station_to_number:
            return name[:-1], True
        m = difflib.get_close_matches(name, list(self.station_to_number), n=1, cutoff=0.5)
        return (m[0], False) if m else (None, False)

    # ---- updates ----
    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters; bidirectional."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distance count mismatch")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line not in known[n]["lines"]:
                    known[n]["lines"].append(line)
            else:
                self.data["stations"].append({"name": n, "lines": [line]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line})
            self.data["connections"].append({"source": b, "dest": a, "distance": d, "line": line})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.rebuild()
                return True
        return False

    def update_distance(self, source, dest, distance, line=None):
        found = False
        for c in self.data["connections"]:
            if c["source"] == source and c["dest"] == dest and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self.rebuild()
        return found

    # ---- check ----
    def check_data(self):
        problems = []
        names = set(s["name"] for s in self.data.get("stations", []))
        if not names:
            problems.append("No stations loaded")
        for c in self.data.get("connections", []):
            if c["source"] not in names or c["dest"] not in names:
                problems.append("Connection refers to unknown station: %s -> %s" % (c["source"], c["dest"]))
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                problems.append("Invalid distance: %s -> %s (%r)" % (c["source"], c["dest"], d))
            elif d > 30000:
                problems.append("Suspicious distance: %s -> %s (%r)" % (c["source"], c["dest"], d))
        connected = set()
        for c in self.data.get("connections", []):
            connected.add(c["source"])
            connected.add(c["dest"])
        for n in sorted(names - connected):
            problems.append("Station has no connections: %s" % n)
        lines = set()
        for s in self.data.get("stations", []):
            lines.update(s.get("lines", []))
        return problems, len(names), len(self.data.get("connections", [])), sorted(lines)

"""Route reports and summaries."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(d):
    if d < 1000:
        return "%d m" % d
    return "%.2f km (%d m)" % (d / 1000.0, d)


def build_result(network, src, dst, path, edges, transfers):
    names = [network.number_to_station[n] for n in path]
    dist = sum(e.distance for e in edges)
    segments = []
    transfer_stations = []
    for e in edges:
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = network.number_to_station[e.dest]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(network.number_to_station[e.source])
            segments.append({"line": e.line, "from": network.number_to_station[e.source],
                             "to": network.number_to_station[e.dest], "stations": 1})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    return {
        "query": {"start": src, "end": dst},
        "path": names,
        "segments": segments,
        "transfer_stations": transfer_stations,
        "distance": dist,
        "fare": Sum_money(dist),
        "estimated_time": round(calculate_estimated_time(dist, transfers), 1),
        "transfer_count": transfers,
        "lines": lines,
    }


def to_json(result):
    return json.dumps(result, ensure_ascii=False, indent=2)


def to_text(result):
    out = []
    q = result["query"]
    out.append("Route: %s -> %s" % (q["start"], q["end"]))
    out.append("Stations (%d): %s" % (len(result["path"]), " -> ".join(result["path"])))
    for s in result["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (s["line"], s["from"], s["to"], s["stations"]))
    if result["transfer_stations"]:
        out.append("Transfers at: %s" % ", ".join(result["transfer_stations"]))
    out.append("Lines: %s" % ", ".join(result["lines"]))
    out.append("Total distance: %s" % format_distance(result["distance"]))
    out.append("Transfer count: %d" % result["transfer_count"])
    out.append("Estimated time: %.1f minutes" % result["estimated_time"])
    out.append("Fare: %d yuan" % result["fare"])
    return "\n".join(out)

"""CLI for the Beijing Subway fare calculator."""
import argparse
import re
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import SubwayNetwork
from pathfinding import PathfindingConfig, ShortestPath
from report import build_result, to_json, to_text

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_input(line):
    parts = [p.strip() for p in re.split(r"[;；,，]", line) if p.strip()]
    if len(parts) != 2:
        return None
    return parts


def run_query(network, line, args, config):
    parts = parse_input(line)
    if parts is None:
        return FORMAT_ERR
    a, b = parts
    s, exact_s = network.match_station(a)
    if s is None:
        return "Starting station '%s' does not exist, please check the station name" % a
    t, exact_t = network.match_station(b)
    if t is None:
        return "Ending station '%s' does not exist, please check the station name" % b
    notes = []
    if not exact_s:
        notes.append("Starting station '%s' matched to '%s'" % (a, s))
    if not exact_t:
        notes.append("Ending station '%s' matched to '%s'" % (b, t))
    if s == t:
        return "Starting station cannot be the same as the ending station"
    try:
        sn, tn = network.station_to_number[s], network.station_to_number[t]
        sp = ShortestPath(network.graph, sn, config)
        path, edges = sp.path_with_edges(tn)
        if not path:
            return "No route found between '%s' and '%s' under current settings" % (s, t)
        transfers = sp.get_transfer_count(tn)
    except Exception as ex:
        return "Path calculation error: %s" % ex
    try:
        result = build_result(network, s, t, path, edges, transfers)
    except Exception as ex:
        return "Fare calculation error: %s" % ex
    text = to_json(result) if args.format == "json" else to_text(result)
    if args.format != "json" and notes:
        text = "\n".join(notes) + "\n" + text
    return text


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        network = SubwayNetwork(load_subway_data(DATA_FILE_PATH))
    except Exception as ex:
        print("Failed to load data: %s" % ex)
        return 1

    if args.check_data:
        problems, ns, nc, lines = network.check_data()
        print("Stations: %d, connections: %d, lines: %d" % (ns, nc, len(lines)))
        for pr in problems:
            print("Problem: " + pr)
        print("Data check passed" if not problems else "Data check found %d problem(s)" % len(problems))
        return 0 if not problems else 1

    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []

    print("Enter 'Starting Station; Ending Station' (empty line or 'q' to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        if not line.strip() or line.strip().lower() in ("q", "quit", "exit"):
            break
        out = run_query(network, line, args, config)
        print(out)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(out + "\n\n")
            except OSError as ex:
                print("Cannot write output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

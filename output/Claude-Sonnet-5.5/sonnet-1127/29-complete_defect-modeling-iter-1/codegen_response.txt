class GridMap:
    """8x8 grid over an 800x800 coordinate system."""

    def __init__(self, width=800, height=800, cell_size=100):
        self.width = width
        self.height = height
        self.cell_size = cell_size
        self.grid_dimension = width // cell_size

    def cell_index(self, v):
        """Return the grid index (0..7) of a coordinate."""
        return max(0, min(self.grid_dimension - 1, int(v) // self.cell_size))

    def center(self, v):
        """Return the cell-centre coordinate for a coordinate."""
        return self.cell_index(v) * self.cell_size + self.cell_size // 2

    def valid(self, v):
        """Check coordinate range 0-800."""
        return isinstance(v, int) and 0 <= v <= self.width

"""Grid-based path planning."""
from gridmap import GRID


class PathPlanner:
    """Right-angle path planner along streets."""

    def __init__(self, grid=GRID):
        self.grid = grid

    def step(self, x, y, tx, ty, speed):
        """Move up to speed units toward target: horizontal first, then
        vertical. If aligned in x (directly above/below) go vertical."""
        remaining = speed
        if x != tx and remaining > 0:
            d = min(remaining, abs(tx - x))
            x += d if tx > x else -d
            remaining -= d
        if y != ty and remaining > 0:
            d = min(remaining, abs(ty - y))
            y += d if ty > y else -d
        return x, y

    def waypoints(self, points):
        """Expand a list of points into an L-shaped polyline."""
        out = [points[0]]
        for (px, py) in points[1:]:
            lx, ly = out[-1]
            if lx != px and ly != py:
                out.append((px, ly))
            if out[-1] != (px, py):
                out.append((px, py))
        return out

    def route_cells(self, points):
        """Return ordered grid cells (cx, cy) crossed by the route."""
        wp = self.waypoints(points)
        cells = []

        def add(x, y):
            c = (self.grid.cell_index(x), self.grid.cell_index(y))
            if not cells or cells[-1] != c:
                cells.append(c)

        add(*wp[0])
        for (ax, ay), (bx, by) in zip(wp, wp[1:]):
            n = max(abs(bx - ax), abs(by - ay))
            for i in range(10, n + 10, 10):
                f = min(i, n)
                add(ax + (bx - ax) * f // n, ay + (by - ay) * f // n)
        return cells

    @staticmethod
    def length(points):
        """Manhattan length of a route."""
        return sum(abs(b[0] - a[0]) + abs(b[1] - a[1])
                   for a, b in zip(points, points[1:]))

"""Order data model."""
import itertools

import globalvar
from gridmap import GRID

_ids = itertools.count(1)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_ids)
        self.res_pos = (GRID.center(res_pos[0]), GRID.center(res_pos[1]))
        self.cus_pos = (GRID.center(cus_pos[0]), GRID.center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))
        self.time_taken = None

    def assign_to(self, rider_id):
        """Assign this order to a rider id."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""Rider class."""
import globalvar
from path_planner import PathPlanner

_planner = PathPlanner()


class Rider:
    """A delivery rider. state: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 0
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0) or 0,
                         "x": self.x, "y": self.y})

    def move(self):
        """Advance one tick."""
        if self.state == 0 and self.orders:
            self.state = 1
            self.orders[0].status = "picking_up"
            self._record("accepted_order", self.orders[0].id)
        if self.state == 0:
            return
        order = self.orders[0]
        budget = self.speed
        while budget > 0 and self.state != 0:
            target = order.res_pos if self.state == 1 else order.cus_pos
            self.special_case = self.x == target[0] or self.y == target[1]
            nx, ny = _planner.step(self.x, self.y, target[0], target[1],
                                   budget)
            budget -= abs(nx - self.x) + abs(ny - self.y)
            self.x, self.y = nx, ny
            if (self.x, self.y) != tuple(target):
                break
            if self.state == 1:
                self.state = 2
                order.status = "delivering"
                self._record("arrived_at_merchant", order.id)
            else:
                now = globalvar.get_value("tick", 0) or 0
                order.status = "completed"
                order.completion_time = now
                order.time_taken = now - order.start_time
                self._record("delivered_to_customer", order.id)
                self.orders.pop(0)
                self.state = 0
                self.special_case = False
        if self.state == 0 and (self.x, self.y) == tuple(order.cus_pos):
            return

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case, "log": self.log}

"""Order management and assignment."""
import globalvar
from order_model import Order
from gridmap import GRID


def _dist(rider, order):
    return abs(rider.x - order.res_pos[0]) + abs(rider.y - order.res_pos[1])


def find_best_rider_for_order(order):
    """Closest rider first; idle/less-loaded riders win ties for balance."""
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None
    return min(riders, key=lambda r: (_dist(r, order) + 200 * len(r.orders),
                                      len(r.orders), r.rider_id))


class OrderManager:
    """Creates, assigns and completes orders."""

    def create_order(self, coords):
        """Validate coords and create an order."""
        if len(coords) != 4:
            raise ValueError("Need 4 integers: rx ry cx cy")
        for c in coords:
            if not GRID.valid(c):
                raise ValueError("Coordinates must be integers in 0-800")
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        orders = globalvar.get_value("orders")
        if orders is None:
            orders = []
            globalvar.set_value("orders", orders)
        orders.append(order)
        return order

    def assign(self, order):
        """Assign order to best rider; returns rider or None."""
        rider = find_best_rider_for_order(order)
        if rider is not None:
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
        return rider

    def assign_pending(self):
        """Try assigning all unassigned orders."""
        for o in globalvar.get_value("orders", []) or []:
            if o.status == "pending":
                self.assign(o)

"""System manager and initialisation."""
import json

import globalvar
from rider import Rider
from order import OrderManager
from gridmap import GRID

RIDER_COST = 300
ORDER_FEE = 100
INITIAL_FUNDS = 300


class SystemManager:
    """Manages funds, riders and the simulation clock."""

    def __init__(self):
        self.grid = GRID
        self.order_manager = OrderManager()

    @property
    def funds(self):
        return globalvar.get_value("funds")

    @funds.setter
    def funds(self, v):
        globalvar.set_value("funds", v)

    @property
    def running(self):
        return globalvar.get_value("running")

    @running.setter
    def running(self, v):
        globalvar.set_value("running", v)

    @property
    def tick_count(self):
        return globalvar.get_value("tick")

    def recruit(self):
        """Create riders while funds >= 300. Returns number added."""
        added = 0
        while self.funds >= RIDER_COST:
            self.funds -= RIDER_COST
            rid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", rid)
            globalvar.get_value("riders").append(Rider(rid))
            added += 1
        return added

    def submit_order(self, coords):
        """Create an order; returns (order, rider, riders_added)."""
        order = self.order_manager.create_order(coords)
        self.funds += ORDER_FEE
        added = self.recruit()
        rider = self.order_manager.assign(order)
        return order, rider, added

    def tick(self):
        """Advance simulation by one tick."""
        globalvar.set_value("tick", self.tick_count + 1)
        self.order_manager.assign_pending()
        for r in globalvar.get_value("riders"):
            r.move()

    def save(self, path="state.json"):
        data = {"funds": self.funds, "tick": self.tick_count,
                "rider_id_counter": globalvar.get_value("rider_id_counter"),
                "riders": [r.to_dict() for r in globalvar.get_value("riders")],
                "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)


def init_system():
    """Initialise global state and return a SystemManager."""
    globalvar._init()
    globalvar.set_value("riders", [])
    globalvar.set_value("orders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("funds", INITIAL_FUNDS)
    globalvar.set_value("tick", 0)
    return SystemManager()

"""Display manager."""
import globalvar
from gridmap import GRID
from path_planner import PathPlanner

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and ASCII map rendering."""

    def __init__(self, system):
        self.system = system
        self.planner = PathPlanner()

    def render_map(self, route=None):
        n = GRID.grid_dimension
        size = 2 * n + 1
        g = [["#"] * size for _ in range(size)]
        for i in range(n):
            for j in range(n):
                g[2 * j + 1][2 * i + 1] = "."
        if route:
            for cx, cy in route:
                g[2 * cy + 1][2 * cx + 1] = "*"
        for o in globalvar.get_value("orders", []):
            if o.status != "completed":
                if o.status in ("pending", "assigned", "picking_up"):
                    g[2 * GRID.cell_index(o.res_pos[1]) + 1][
                        2 * GRID.cell_index(o.res_pos[0]) + 1] = "S"
                g[2 * GRID.cell_index(o.cus_pos[1]) + 1][
                    2 * GRID.cell_index(o.cus_pos[0]) + 1] = "C"
        for r in globalvar.get_value("riders"):
            g[2 * GRID.cell_index(r.y) + 1][2 * GRID.cell_index(r.x) + 1] = "R"
        return "\n".join("".join(row) for row in g)

    def route_info(self, rider, order):
        pts = [(rider.x, rider.y), order.res_pos, order.cus_pos]
        dist = self.planner.length(pts)
        t = -(-dist // rider.speed)
        cells = self.planner.route_cells(pts)
        return (self.render_map(cells) + "\nRider %d speed %d, distance %d, "
                "estimated time %d ticks" % (rider.rider_id, rider.speed,
                                             dist, t))

    def status(self):
        riders = globalvar.get_value("riders")
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (
            self.system.tick_count, self.system.funds, len(riders))]
        for r in riders:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATE_NAMES[r.state],
                [o.id for o in r.orders]))
        lines.append(self.render_map())
        return "\n".join(lines)

    def orders(self):
        os_ = globalvar.get_value("orders")
        if not os_:
            return "No orders."
        lines = []
        for o in os_:
            lines.append("Order %d start=%s restaurant=%s customer=%s "
                         "distance=%d time_taken=%s rider=%s status=%s" % (
                             o.id, o.start_time, o.res_pos, o.cus_pos,
                             o.distance, o.time_taken, o.rider_id, o.status))
        return "\n".join(lines)

    def riders_log(self):
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("t=%d rider %d (%d,%d) %s order=%s" % (
                    e["time"], r.rider_id, e["x"], e["y"], e["event"],
                    e["order_id"]))
        return "\n".join(lines) or "No rider records."

"""Command parsing."""
from display import DisplayManager

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "exit": "quit", "h": "help", "t": "tick"}


class InputManager:
    """Parses commands and dispatches them."""

    def __init__(self, system, input_fn=input):
        self.system = system
        self.display = DisplayManager(system)
        self.command_history = []
        self.input_fn = input_fn

    def handle(self, line):
        """Process a command line; returns output text."""
        line = line.strip()
        if not line:
            return ""
        self.command_history.append(line)
        parts = line.split()
        cmd = ALIASES.get(parts[0].lower(), parts[0].lower())
        args = parts[1:]
        try:
            if cmd == "start":
                return self._start(args)
            if cmd == "status":
                return self.display.status()
            if cmd == "orders":
                return self.display.orders()
            if cmd == "riders":
                return self.display.riders_log()
            if cmd == "tick":
                n = int(args[0]) if args else 1
                for _ in range(max(0, n)):
                    self.system.tick()
                return "Advanced %d tick(s)." % n
            if cmd == "history":
                return "\n".join(self.command_history)
            if cmd == "save":
                self.system.save()
                return "Saved."
            if cmd == "quit":
                self.system.running = False
                return "Bye."
            if cmd == "help":
                return ("Commands: start [rx ry cx cy], status, orders, "
                        "riders, tick [n], history, save, quit/exit")
            return "Unknown command '%s'. Type help." % parts[0]
        except ValueError as e:
            return "Error: %s" % e
        except Exception as e:  # recovery
            return "Unexpected error: %s" % e

    def _start(self, args):
        if not args:
            args = self.input_fn("Enter rx ry cx cy: ").split()
        try:
            coords = [int(a) for a in args]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        order, rider, added = self.system.submit_order(coords)
        from gridmap import GRID
        out = ["Order %d: restaurant cell (%d,%d), customer cell (%d,%d)" % (
            order.id, GRID.cell_index(order.res_pos[0]),
            GRID.cell_index(order.res_pos[1]),
            GRID.cell_index(order.cus_pos[0]),
            GRID.cell_index(order.cus_pos[1])),
            "New rider added: %s" % ("yes" if added else "no")]
        if rider:
            out.append(self.display.route_info(rider, order))
        else:
            out.append("No rider available; order pending.")
        return "\n".join(out)

"""Entry point: python src/main.py"""
import globalvar
from system import init_system
from input_manager import InputManager


def main():
    system = init_system()
    im = InputManager(system)
    print("Food Delivery Dispatch Simulator. Type 'help'.")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        out = im.handle(line)
        if out:
            print(out)
        if globalvar.get_value("running"):
            system.tick()
    try:
        system.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

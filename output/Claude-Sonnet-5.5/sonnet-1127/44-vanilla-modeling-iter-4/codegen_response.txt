"""Core data models: Edge and Graph."""
from collections import defaultdict


class Edge:
    """Undirected weighted edge between station numbers."""

    def __init__(self, source, dest, distance, line=None):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph keyed by station numbers."""

    def __init__(self):
        self.adjacency = defaultdict(list)
        self.station_lines = {}
        self.station_to_number = {}
        self.number_to_station = {}

    def register_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an edge in both directions (duplicates are ignored)."""
        self._add_directed(edge)
        self._add_directed(Edge(edge.dest, edge.source, edge.distance, edge.line))

    def _add_directed(self, e):
        for x in self.adjacency[e.source]:
            if x.dest == e.dest and x.line == e.line and x.distance == e.distance:
                return
        self.adjacency[e.source].append(e)

    def remove_edges_between(self, a, b, line=None):
        for u, v in ((a, b), (b, a)):
            self.adjacency[u] = [e for e in self.adjacency[u]
                                 if not (e.dest == v and (line is None or e.line == line))]

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def nodes(self):
        s = set(self.adjacency.keys())
        s.update(self.number_to_station.keys())
        return s

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

"""MinPQ and Dijkstra-based ShortestPath."""
import heapq
import itertools


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


TRANSFER_PENALTY = {
    "shortest_distance": 0.001,   # only tie-breaker
    "least_transfer": 10_000_000.0,
    "comprehensive": 1500.0,      # meters-equivalent per transfer
}


class ShortestPath:
    """Dijkstra from a source station number; state = (station, line, transfers)."""

    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._parent = {}
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0.001)
        use_pref = cfg.mode != "shortest_distance" and cfg.preferred_lines
        limit = cfg.max_transfers
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        self._parent[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in self.dist_to:
                self.dist_to[node] = real[st]
                self._best_state[node] = st
            for e in self.graph.neighbors(node):
                change = 1 if (line is not None and e.line != line) else 0
                nt_real = t + change
                if limit is not None and nt_real > limit:
                    continue
                nt = nt_real if limit is not None else 0
                w = e.weight
                if use_pref and e.line in cfg.preferred_lines:
                    w *= 0.9
                nst = (e.dest, e.line, nt)
                nc = c + w + change * penalty
                if nst not in cost or nc < cost[nst]:
                    cost[nst] = nc
                    real[nst] = real[st] + e.weight
                    self._parent[nst] = (st, e)
                    pq.push(nc, nst)

    def has_path_to(self, target):
        return target in self.dist_to

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        nodes, edges = [], []
        while st is not None:
            nodes.append(st[0])
            p = self._parent[st]
            if p is None:
                break
            edges.append(p[1])
            st = p[0]
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        n = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                n += 1
        return n

"""Fare and time estimation."""
import math

FARE_TIERS = [(0, 6000, 3), (6000, 12000, 4), (12000, 22000, 5), (22000, 32000, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters (left-closed, right-open tiers)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    for lo, hi, price in FARE_TIERS:
        if lo <= d < hi:
            return price
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5

"""Data management: building the graph, integrity checks and updates."""
import json

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

MAIN_LINES = ["1", "2", "4", "5", "6", "8", "10", "13", "14", "15",
              "昌平", "亦庄", "房山", "机场", "Changping", "Yizhuang", "Fangshan", "Airport"]


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = Graph()
        self.lines = {}
        self.build()

    def build(self):
        g = Graph()
        self.graph = g
        data = self.data
        num = 0
        for s in data.get("stations", []):
            if s["name"] not in g.station_to_number:
                num += 1
                g.register_station(s["name"], num)
            g.set_station_lines(s["name"], s.get("lines", []))
        lines = {}
        for c in data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in g.station_to_number:
                    num += 1
                    g.register_station(n, num)
                    g.set_station_lines(n, [c["line"]])
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
            seq = lines.setdefault(c["line"], [])
            for n in (c["source"], c["dest"]):
                if n not in seq:
                    seq.append(n)
        for s in data.get("stations", []):
            for l in s.get("lines", []):
                seq = lines.setdefault(l, [])
                if s["name"] not in seq:
                    seq.append(s["name"])
        self.lines = lines

    @property
    def station_to_number(self):
        return self.graph.station_to_number

    def has_station(self, name):
        return name in self.graph.station_to_number

    def normalize_station(self, name):
        name = (name or "").strip()
        if name in self.station_to_number:
            return name
        if name.endswith("站") and name[:-1] in self.station_to_number:
            return name[:-1]
        if name + "站" in self.station_to_number:
            return name + "站"
        return None

    def check_data(self):
        """Return (ok, messages)."""
        msgs, ok = [], True
        names = set(s["name"] for s in self.data.get("stations", []))
        if not names:
            return False, ["No station data"]
        msgs.append("Stations: %d, lines: %d" % (len(names), len(self.lines)))
        for l in MAIN_LINES:
            pass
        found = [m for m in MAIN_LINES if any(m in ln for ln in self.lines)]
        if len(found) < 5:
            msgs.append("WARNING: few main lines recognized")
        conns = self.data.get("connections", [])
        for c in conns:
            if c["source"] not in names or c["dest"] not in names:
                ok = False
                msgs.append("ERROR: connection references unknown station: %s - %s"
                            % (c["source"], c["dest"]))
            if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0:
                ok = False
                msgs.append("ERROR: invalid distance %s - %s: %r"
                            % (c["source"], c["dest"], c["distance"]))
            elif c["distance"] > 30000:
                msgs.append("WARNING: suspicious distance %s - %s: %s"
                            % (c["source"], c["dest"], c["distance"]))
        connected = set()
        for c in conns:
            connected.add(c["source"])
            connected.add(c["dest"])
        for n in sorted(names - connected):
            msgs.append("WARNING: station without connections: %s" % n)
        # connectivity
        g = self.graph
        if g.station_to_number:
            start = next(iter(g.station_to_number.values()))
            seen, stack = {start}, [start]
            while stack:
                u = stack.pop()
                for e in g.neighbors(u):
                    if e.dest not in seen:
                        seen.add(e.dest)
                        stack.append(e.dest)
            if len(seen) < len(connected):
                msgs.append("WARNING: network has %d disconnected stations"
                            % (len(connected) - len(seen)))
        msgs.append("Data check " + ("passed" if ok else "FAILED"))
        return ok, msgs

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters values."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": int(d), "line": line_name})
        self.build()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        if distance <= 0:
            raise ValueError("distance must be positive")
        hit = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = int(distance)
                hit = True
        if hit:
            self.build()
        return hit

    def save(self, path=None):
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

"""Route building and report generation."""
import json

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath


def compute_route(dm, start, end, mode="shortest_distance", max_transfers=None, line_preference=None):
    g = dm.graph
    cfg = PathfindingConfig(mode, max_transfers, [line_preference] if line_preference else [])
    sp = ShortestPath(g, g.station_to_number[start], cfg)
    target = g.station_to_number[end]
    nodes, edges = sp.path_with_edges(target)
    if not nodes:
        raise LookupError("No route found from '%s' to '%s'" % (start, end))
    names = [g.number_to_station[n] for n in nodes]
    dist = sum(e.weight for e in edges)
    transfers = []
    for i, (a, b) in enumerate(zip(edges, edges[1:])):
        if a.line != b.line:
            transfers.append({"station": names[i + 1], "from": a.line, "to": b.line})
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    segments = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["end"] = names[i + 1]
            segments[-1]["stations"] += 1
        else:
            segments.append({"line": e.line, "start": names[i], "end": names[i + 1], "stations": 1})
    tc = len(transfers)
    try:
        fare = Sum_money(dist)
    except Exception as ex:
        raise ArithmeticError("Fare calculation failed: %s" % ex)
    return {
        "query": {"start": start, "end": end, "mode": mode,
                  "max_transfers": max_transfers, "line_preference": line_preference},
        "path": names,
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(calculate_estimated_time(dist, tc), 1),
        "transfer_count": tc,
        "transfers": transfers,
        "lines": lines,
        "segments": segments,
    }


def format_distance(m):
    return "%d m" % m if m < 1000 else "%.2f km" % (m / 1000.0)


def format_text(r):
    q = r["query"]
    out = ["Route: %s -> %s (mode: %s)" % (q["start"], q["end"], q["mode"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: %s (%d m / %.2f km)" % (format_distance(r["distance_m"]),
                                                    r["distance_m"], r["distance_km"])]
    out.append("Line segments:")
    for s in r["segments"]:
        out.append("  [%s] %s -> %s (%d segments)" % (s["line"], s["start"], s["end"], s["stations"]))
    if r["transfers"]:
        out.append("Transfers (%d):" % r["transfer_count"])
        for t in r["transfers"]:
            out.append("  %s: %s -> %s" % (t["station"], t["from"], t["to"]))
    else:
        out.append("Transfers: 0")
    out.append("Lines: " + ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def format_json(r):
    return json.dumps(r, ensure_ascii=False, indent=2)

"""Beijing Subway fare calculation CLI."""
import argparse
import sys

from data_loader import DATA_FILE_PATH
from data_manager import DataManager
from report import compute_route, format_json, format_text

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def validate_input(dm, text):
    """Return (start, end, error)."""
    text = text.replace("；", ";")
    parts = [x.strip() for x in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None, None, FORMAT_ERR
    s = dm.normalize_station(parts[0])
    if s is None:
        return None, None, "Starting station '%s' does not exist, please check the station name" % parts[0]
    e = dm.normalize_station(parts[1])
    if e is None:
        return None, None, "Ending station '%s' does not exist, please check the station name" % parts[1]
    if s == e:
        return None, None, "Starting station cannot be the same as the ending station"
    return s, e, None


def run_query(dm, args, text):
    s, e, err = validate_input(dm, text)
    if err:
        return None, err
    try:
        r = compute_route(dm, s, e, args.mode, args.max_transfers, args.line_preference)
    except LookupError as ex:
        return None, "Path calculation error: %s" % ex
    except ArithmeticError as ex:
        return None, str(ex)
    except Exception as ex:
        return None, "Path calculation error: %s" % ex
    return (format_json(r) if args.format == "json" else format_text(r)), None


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager(DATA_FILE_PATH)
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1
    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        result, err = run_query(dm, args, line)
        if err:
            print(err)
            continue
        print(result)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(result + "\n\n")
            except OSError as ex:
                print("Cannot write output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

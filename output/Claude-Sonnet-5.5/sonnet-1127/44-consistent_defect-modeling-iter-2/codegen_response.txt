"""Core data models: Station, Line, Edge, Graph."""


class Station:
    def __init__(self, id, name, is_transfer=False):
        self.id = id
        self.name = name
        self.is_transfer = is_transfer


class Line:
    def __init__(self, name, station_sequence=None):
        self.name = name
        self.station_sequence = list(station_sequence or [])


class Edge:
    """Directed weighted connection between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


Connection = Edge


class Graph:
    """Weighted directed graph. Station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}

    def add_station(self, number, name):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, [])

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def nodes(self):
        return list(self.adjacency.keys())

    def edges(self):
        for lst in self.adjacency.values():
            for e in lst:
                yield e

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

"""MinPQ, PathfindingConfig and ShortestPath (Dijkstra)."""
import heapq
import itertools

MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = 3000


class MinPQ:
    """Heap based min priority queue."""

    def __init__(self):
        self.heap = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}
        self._rec = {}
        self._run()

    def _edge_cost(self, edge, transfer):
        mode = self.config.mode
        d = edge.distance
        pref = edge.line in self.config.preferred_lines
        if mode == "least_transfer":
            return (1000000 if transfer else 0) + d * (0.95 if pref else 1)
        if mode == "comprehensive":
            return (TRANSFER_PENALTY if transfer else 0) + d * (0.9 if pref else 1)
        return d

    def _run(self):
        limit = self.config.max_transfers
        start = (self.source, None, 0)
        self._rec[start] = (0, 0, 0, None, None)  # cost, dist, transfers, prev, edge
        pq = MinPQ()
        pq.push(0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, _ = st
            _, dist, tr, _, _ = self._rec[st]
            for e in self.graph.neighbors(node):
                transfer = line is not None and e.line != line
                ntr = tr + (1 if transfer else 0)
                if limit is not None and ntr > limit:
                    continue
                nst = (e.dest, e.line, ntr if limit is not None else 0)
                ncost = cost + self._edge_cost(e, transfer)
                old = self._rec.get(nst)
                if old is None or ncost < old[0]:
                    self._rec[nst] = (ncost, dist + e.distance, ntr, st, e)
                    pq.push(ncost, nst)
        for st, rec in self._rec.items():
            n = st[0]
            if n not in self._best or rec[0] < self._rec[self._best[n]][0]:
                self._best[n] = st
        self.dist_to = {n: self._rec[s][1] for n, s in self._best.items()}

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        st = self._best[target]
        nodes, edges = [], []
        while st is not None:
            rec = self._rec[st]
            nodes.append(st[0])
            if rec[4] is not None:
                edges.append(rec[4])
            st = rec[3]
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math


class FareTier:
    def __init__(self, min_distance, max_distance, price):
        self.min_distance = min_distance
        self.max_distance = max_distance
        self.price = price


def Sum_money(distance):
    """Fare in yuan for a distance in meters (fractions round up to km)."""
    if distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km < 22:
        return 5
    if km <= 32:
        return 6
    return 7 + (km - 33) // 20


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000 * 1.5 + transfer_count * 5


class FareCalculator:
    def calculate(self, distance):
        return Sum_money(distance)

"""Network data management and integrity checks."""
import difflib

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE = {
    "Line 1": "Tiananmen East",
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = Graph()
        self.lines = {}
        self._build()

    def _build(self):
        g = Graph()
        for i, s in enumerate(self.data.get("stations", [])):
            g.add_station(i, s["name"])
            g.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in g.station_to_number:
                    g.add_station(len(g.station_to_number), n)
                    g.set_station_lines(n, [c["line"]])
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
        self.graph = g
        lines = {}
        for c in self.data.get("connections", []):
            seq = lines.setdefault(c["line"], [])
            for n in (c["source"], c["dest"]):
                if n not in seq:
                    seq.append(n)
        self.lines = lines

    @property
    def station_names(self):
        return list(self.graph.station_to_number)

    def find_station(self, name):
        """Return (matched_name or None, exact)."""
        name = name.strip()
        if name in self.graph.station_to_number:
            return name, True
        m = difflib.get_close_matches(name, self.station_names, n=1, cutoff=0.5)
        return (m[0] if m else None), False

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """Add a bidirectional line; distances has len(stations)-1 items."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
            self.data["connections"].append({"source": b, "dest": a, "distance": d, "line": line_name})
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        self._build()

    def update_station(self, old_name, new_name):
        for s in self.data["stations"]:
            if s["name"] == old_name:
                s["name"] = new_name
        for c in self.data["connections"]:
            if c["source"] == old_name:
                c["source"] = new_name
            if c["dest"] == old_name:
                c["dest"] = new_name
        self._build()

    def update_distance(self, a, b, distance, line=None):
        hit = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                hit = True
        if hit:
            self._build()
        return hit

    # ---- checks ----
    def check_data(self):
        """Return (ok, list of messages)."""
        issues = []
        names = set(self.graph.station_to_number)
        for line in ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
                     "Line 13", "Line 14", "Line 15"]:
            if line not in self.lines:
                issues.append("Missing line: %s" % line)
        for s in self.data["stations"]:
            if not s.get("lines"):
                issues.append("Station without line: %s" % s["name"])
        for c in self.data["connections"]:
            if c["source"] not in names or c["dest"] not in names:
                issues.append("Connection with unknown station: %s-%s" % (c["source"], c["dest"]))
            if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0:
                issues.append("Bad distance: %s-%s" % (c["source"], c["dest"]))
            if c["source"] == c["dest"]:
                issues.append("Self loop at %s" % c["source"])
        for n, num in self.graph.station_to_number.items():
            has_in = any(e.dest == num for e in self.graph.edges())
            if not self.graph.neighbors(num) and not has_in:
                issues.append("Isolated station: %s" % n)
        return not issues, issues

"""Route, RouteReport, Query."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


class Query:
    def __init__(self, start_station, end_station, mode="shortest_distance",
                 max_transfers=None, line_preference=None):
        self.start_station = start_station
        self.end_station = end_station
        self.mode = mode
        self.max_transfers = max_transfers
        self.line_preference = line_preference


class Route:
    def __init__(self, stations, total_distance, transfer_count, involved_lines,
                 transfer_stations=None, segments=None):
        self.stations = stations
        self.total_distance = total_distance
        self.transfer_count = transfer_count
        self.involved_lines = involved_lines
        self.transfer_stations = transfer_stations or []
        self.segments = segments or []  # (line, first, last)


def build_route(graph, nodes, edges):
    names = [graph.number_to_station[n] for n in nodes]
    lines, tstations, segs = [], [], []
    for i, e in enumerate(edges):
        if e.line not in lines:
            lines.append(e.line)
        if i > 0 and edges[i - 1].line != e.line:
            tstations.append(names[i])
        if not segs or segs[-1][0] != e.line or (i > 0 and edges[i - 1].line != e.line):
            segs.append([e.line, names[i], names[i + 1]])
        else:
            segs[-1][2] = names[i + 1]
    return Route(names, sum(e.distance for e in edges), len(tstations), lines, tstations,
                 [tuple(s) for s in segs])


def format_distance(d):
    if d < 1000:
        return "%d m" % d
    return "%.2f km (%d m)" % (d / 1000, d)


class RouteReport:
    def __init__(self, route, query=None, fmt="text", output_file=None):
        self.route = route
        self.query = query
        self.format = fmt
        self.output_file = output_file
        self.fare = Sum_money(route.total_distance)
        self.estimated_time = calculate_estimated_time(route.total_distance, route.transfer_count)

    def summary(self):
        return {"distance": self.route.total_distance, "fare": self.fare,
                "transfer_count": self.route.transfer_count,
                "estimated_time": self.estimated_time}

    def to_text(self):
        r = self.route
        out = []
        if self.query:
            out.append("Route: %s -> %s (mode: %s)" % (self.query.start_station,
                                                       self.query.end_station, self.query.mode))
        out.append("Stations (%d): %s" % (len(r.stations), " -> ".join(r.stations)))
        for line, a, b in r.segments:
            out.append("  [%s] %s -> %s" % (line, a, b))
        out.append("Total distance: %s" % format_distance(r.total_distance))
        if r.transfer_stations:
            out.append("Transfers (%d): %s" % (r.transfer_count, ", ".join(r.transfer_stations)))
        else:
            out.append("Transfers: 0")
        out.append("Lines: %s" % ", ".join(r.involved_lines))
        out.append("Estimated time: %.1f minutes" % self.estimated_time)
        out.append("Fare: %d yuan" % self.fare)
        return "\n".join(out)

    def to_json(self):
        r = self.route
        q = self.query
        return json.dumps({
            "query": {"start": q.start_station, "end": q.end_station, "mode": q.mode,
                      "max_transfers": q.max_transfers,
                      "line_preference": q.line_preference} if q else None,
            "path": r.stations,
            "distance": r.total_distance,
            "fare": self.fare,
            "estimated_time": self.estimated_time,
            "transfer_count": r.transfer_count,
            "transfer_stations": r.transfer_stations,
            "lines": r.involved_lines,
        }, ensure_ascii=False, indent=2)

    def render(self):
        return self.to_json() if self.format == "json" else self.to_text()

    def save(self):
        if self.output_file:
            with open(self.output_file, "a", encoding="utf-8") as f:
                f.write(self.render() + "\n")

"""Command line interface."""
import argparse
import re
import sys

from data_manager import DataManager
from pathfinding import MODES, PathfindingConfig, ShortestPath
from report import Query, RouteReport, build_route

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


class CLI:
    def __init__(self, data_manager=None, output_file=None, fmt="text", mode="shortest_distance",
                 max_transfers=None, line_preference=None):
        self.dm = data_manager or DataManager()
        self.output_file = output_file
        self.format = fmt
        self.mode = mode
        self.max_transfers = max_transfers
        self.line_preference = line_preference

    def parse_input(self, text):
        parts = [p.strip() for p in re.split(r"[;；,，]", text)]
        if len(parts) != 2 or not all(parts):
            raise ValueError(FORMAT_ERR)
        return parts

    def process(self, text):
        """Return the output string for one query line."""
        try:
            a, b = self.parse_input(text)
        except ValueError as e:
            return str(e)
        s, exact_s = self.dm.find_station(a)
        if s is None:
            return "Starting station '%s' does not exist, please check the station name" % a
        t, exact_t = self.dm.find_station(b)
        if t is None:
            return "Ending station '%s' does not exist, please check the station name" % b
        notes = []
        if not exact_s:
            notes.append("Starting station '%s' matched to '%s'" % (a, s))
        if not exact_t:
            notes.append("Ending station '%s' matched to '%s'" % (b, t))
        if s == t:
            return "Starting station cannot be the same as the ending station"
        g = self.dm.graph
        cfg = PathfindingConfig(self.mode, self.max_transfers,
                                [self.line_preference] if self.line_preference else [])
        try:
            sp = ShortestPath(g, g.station_to_number[s], cfg)
            tn = g.station_to_number[t]
            nodes, edges = sp.path_with_edges(tn)
            if not nodes:
                return "No route found from '%s' to '%s' under the given constraints" % (s, t)
            route = build_route(g, nodes, edges)
        except Exception as e:
            return "Path calculation failed: %s" % e
        try:
            rep = RouteReport(route, Query(s, t, self.mode, self.max_transfers,
                                           self.line_preference), self.format, self.output_file)
        except Exception as e:
            return "Fare calculation failed: %s" % e
        try:
            rep.save()
        except OSError as e:
            notes.append("Could not write output file: %s" % e)
        text_out = rep.render()
        if notes and self.format != "json":
            text_out = "\n".join(notes) + "\n" + text_out
        return text_out

    def run(self):
        print("Beijing Subway fare calculator. Enter 'Start; End' (empty line or 'quit' to exit).")
        while True:
            try:
                line = input("> ")
            except EOFError:
                break
            if line.strip().lower() in ("", "quit", "exit", "q"):
                break
            print(self.process(line))


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p


def main(argv=None):
    args = build_parser().parse_args(argv)
    try:
        dm = DataManager()
    except Exception as e:
        print("Failed to load data: %s" % e)
        return 1
    if args.check_data:
        ok, issues = dm.check_data()
        print("Data check passed" if ok else "Data check found issues:")
        for i in issues:
            print(" - " + i)
        return 0 if ok else 1
    CLI(dm, args.output_file, args.format, args.mode, args.max_transfers,
        args.line_preference).run()
    return 0


if __name__ == "__main__":
    sys.exit(main())

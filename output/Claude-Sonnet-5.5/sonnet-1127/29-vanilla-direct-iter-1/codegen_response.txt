_state = {}


def _init():
    """Reset all global values to their defaults."""
    _state.clear()
    _state["funds"] = 1000
    _state["riders"] = []
    _state["rider_id_counter"] = 0
    _state["running"] = True
    _state["time"] = 0
    _state["orders"] = []
    _state["history"] = []


def set_value(key, value):
    """Set a global value."""
    if not _state:
        _init()
    _state[key] = value


def get_value(key, default=None):
    """Get a global value (None/default if missing)."""
    if not _state:
        _init()
    return _state.get(key, default)

"""System initialisation and management."""
import json

import globalvar

RECRUIT_COST = 300


def init_system():
    """Initialise all global parameters."""
    globalvar._init()
    return True


def recruit_rider():
    """Create a new rider, deducting cost. Returns Rider or None."""
    from rider import Rider
    funds = globalvar.get_value("funds")
    if funds < RECRUIT_COST:
        return None
    rid = globalvar.get_value("rider_id_counter")
    globalvar.set_value("rider_id_counter", rid + 1)
    globalvar.set_value("funds", funds - RECRUIT_COST)
    r = Rider(rid)
    globalvar.get_value("riders").append(r)
    return r


def tick():
    """Advance the simulation by one tick."""
    globalvar.set_value("time", globalvar.get_value("time") + 1)
    for r in list(globalvar.get_value("riders")):
        try:
            r.move()
        except Exception as exc:  # error recovery
            r.state = 0
            r.orders = []
            print("Recovered rider %s: %s" % (r.rider_id, exc))


def save_state(path="state.json"):
    """Persist state to a JSON file."""
    data = {
        "funds": globalvar.get_value("funds"),
        "time": globalvar.get_value("time"),
        "riders": [r.to_dict() for r in globalvar.get_value("riders")],
        "orders": [o.to_dict() for o in globalvar.get_value("orders")],
    }
    with open(path, "w") as f:
        json.dump(data, f, indent=2)

"""Grid path planning."""


class PathPlanner:
    """Right-angle grid pathfinding (horizontal first, then vertical)."""

    @staticmethod
    def next_position(x, y, tx, ty, speed):
        """Move up to `speed` units from (x,y) toward (tx,ty)."""
        budget = speed
        dx = tx - x
        step = min(abs(dx), budget)
        x += step if dx > 0 else -step
        budget -= step
        dy = ty - y
        step = min(abs(dy), budget)
        y += step if dy > 0 else -step
        return x, y

    @staticmethod
    def waypoints(points):
        """Return corner list for a multi-destination path."""
        pts = [points[0]]
        for (x, y) in points[1:]:
            px, py = pts[-1]
            if px != x and py != y:
                pts.append((x, py))
            pts.append((x, y))
        return pts

    @staticmethod
    def cells(points):
        """Grid cells (col,row) traversed by the path."""
        wps = PathPlanner.waypoints(points)
        out = []
        for (ax, ay), (bx, by) in zip(wps, wps[1:]):
            n = max(abs(bx - ax), abs(by - ay)) // 10 + 1
            for i in range(n + 1):
                px = ax + (bx - ax) * i // n
                py = ay + (by - ay) * i // n
                c = (min(px // 100, 7), min(py // 100, 7))
                if c not in out:
                    out.append(c)
        return out

"""Order data model."""
import globalvar

_counter = [0]


def to_center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(int(v), 799))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        _counter[0] += 1
        self.id = _counter[0]
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("time") or 0
        self.complete_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Assign order to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self):
        """Mark completed."""
        self.status = "completed"
        self.complete_time = globalvar.get_value("time") or 0

    @property
    def duration(self):
        if self.complete_time is None:
            return None
        return self.complete_time - self.create_time

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "create_time": self.create_time,
                "complete_time": self.complete_time,
                "distance": self.distance}

"""Rider model."""
import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider. states: 0 idle, 1 picking up, 2 delivering."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time") or 0,
                         "x": self.x, "y": self.y})

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        order = self.orders[0]
        tx, ty = order.res_pos if self.state == 1 else order.cus_pos
        self.special_case = (self.x == tx and self.y != ty)
        self.x, self.y = PathPlanner.next_position(
            self.x, self.y, tx, ty, self.speed)
        if (self.x, self.y) != (tx, ty):
            return
        if self.state == 1:
            self.state = 2
            self._log("arrived_at_merchant", order.id)
        else:
            self._log("delivered_to_customer", order.id)
            order.complete()
            self.orders.pop(0)
            self.state = 0
            self.special_case = False

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order management: creation and assignment."""
import math

import globalvar
import system
from order_model import Order


def find_best_rider_for_order(order):
    """Return the closest rider to the restaurant (None if no riders)."""
    riders = globalvar.get_value("riders")
    if not riders:
        return None
    rx, ry = order.res_pos
    return min(riders, key=lambda r: (math.hypot(r.x - rx, r.y - ry),
                                      len(r.orders), r.rider_id))


class OrderManager:
    """Handles order creation, assignment, completion."""

    @staticmethod
    def create_order(coords):
        """Create and assign an order. Returns (order, rider, recruited)."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        riders = globalvar.get_value("riders")
        idle = sum(1 for r in riders if r.state == 0 and not r.orders)
        pending = 1 + sum(1 for o in globalvar.get_value("orders")
                          if o.status == "pending")
        recruited = False
        if globalvar.get_value("funds") >= 300 and pending > idle:
            recruited = system.recruit_rider() is not None
        rider = find_best_rider_for_order(order)
        globalvar.get_value("orders").append(order)
        if rider is None:
            return order, None, recruited
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return order, rider, recruited

"""Display helpers."""
import globalvar
from path_planner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and map rendering."""

    @staticmethod
    def map_text(path_pts=None, extra=None):
        grid = [["#" if (x + y) % 2 == 0 else "." for x in range(8)]
                for y in range(8)]
        if path_pts:
            for cx, cy in PathPlanner.cells(path_pts):
                grid[cy][cx] = "*"
        for o in globalvar.get_value("orders"):
            if o.status != "completed":
                rx, ry = o.res_pos
                cx, cy = o.cus_pos
                grid[ry // 100][rx // 100] = "S"
                grid[cy // 100][cx // 100] = "C"
        for r in globalvar.get_value("riders"):
            grid[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
        return "\n".join(" ".join(row) for row in grid)

    @staticmethod
    def status_text():
        lines = ["Time: %s  Funds: %s  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(globalvar.get_value("riders")))]
        for r in globalvar.get_value("riders"):
            lines.append("  Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    @staticmethod
    def orders_text():
        os_ = globalvar.get_value("orders")
        if not os_:
            return "No orders."
        return "\n".join(
            "Order %d: start=%s res=%s cus=%s dist=%d time=%s rider=%s [%s]" % (
                o.id, o.create_time, o.res_pos, o.cus_pos, o.distance,
                o.duration, o.rider_id, o.status) for o in os_)

    @staticmethod
    def riders_text():
        lines = []
        for r in globalvar.get_value("riders"):
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%s (%s,%s) %s order=%s" % (
                    e["time"], e["x"], e["y"], e["event"], e["order_id"]))
        return "\n".join(lines) or "No riders."

class InputManager:
    """Parses user input."""

    ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
               "exit": "quit", "t": "tick", "h": "help"}

    def __init__(self):
        self.history = []

    def parse_command(self, line):
        cmd = line.strip().lower()
        if cmd:
            self.history.append(cmd)
        parts = cmd.split()
        if not parts:
            return "", []
        return self.ALIASES.get(parts[0], parts[0]), parts[1:]

    @staticmethod
    def parse_coords(line):
        """Parse four ints in 0-800. Raises ValueError with message."""
        text = line.replace(",", " ").replace("[", " ").replace("]", " ")
        parts = text.split()
        if len(parts) != 4:
            raise ValueError("Enter exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be within 0-800.")
        return vals

"""Command-line entry point."""
import globalvar
import system
from display import DisplayManager
from input_manager import InputManager
from order import OrderManager
from path_planner import PathPlanner

HELP = ("Commands: start, status(s), orders(o), riders(r), tick [n](t), "
        "help(h), quit/exit(q)")


def handle_start(im):
    try:
        coords = im.parse_coords(input("Enter rx ry cx cy: "))
    except ValueError as e:
        print("Error:", e)
        return
    order, rider, recruited = OrderManager.create_order(coords)
    print("Restaurant grid:", (order.res_pos[0] // 100, order.res_pos[1] // 100))
    print("Customer grid:", (order.cus_pos[0] // 100, order.cus_pos[1] // 100))
    print("New rider added:", recruited)
    if rider is None:
        print("No rider available; order pending.")
        return
    pts = [(rider.x, rider.y), order.res_pos, order.cus_pos]
    total = (abs(rider.x - pts[1][0]) + abs(rider.y - pts[1][1])
             + order.distance)
    print("Rider %d speed %d, estimated time %d ticks" % (
        rider.rider_id, rider.speed, -(-total // rider.speed)))
    print(DisplayManager.map_text(pts))


def main():
    system.init_system()
    im = InputManager()
    print(HELP)
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        cmd, args = im.parse_command(line)
        try:
            if cmd == "start":
                handle_start(im)
            elif cmd == "status":
                print(DisplayManager.status_text())
                print(DisplayManager.map_text())
            elif cmd == "orders":
                print(DisplayManager.orders_text())
            elif cmd == "riders":
                print(DisplayManager.riders_text())
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                for _ in range(max(n, 0)):
                    system.tick()
                print("Time:", globalvar.get_value("time"))
            elif cmd == "help":
                print(HELP)
            elif cmd == "quit":
                globalvar.set_value("running", False)
                continue
            elif cmd:
                print("Unknown command. " + HELP)
                continue
            else:
                continue
        except ValueError:
            print("Error: invalid argument.")
            continue
        if cmd in ("start",):
            system.tick()
    try:
        system.save_state()
    except OSError:
        pass


if __name__ == "__main__":
    main()

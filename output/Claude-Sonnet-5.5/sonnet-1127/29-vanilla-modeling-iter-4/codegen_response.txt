class GridMap:
    """8x8 grid, 100 unit cells, 800x800 coordinate system."""

    def __init__(self, width=800, height=800, cell_size=100):
        self.width = width
        self.height = height
        self.cell_size = cell_size
        self.grid_dimension = width // cell_size

    def cell_of(self, x, y):
        """Return (col,row) cell index of a coordinate, clamped."""
        d = self.grid_dimension - 1
        cx = min(max(int(x) // self.cell_size, 0), d)
        cy = min(max(int(y) // self.cell_size, 0), d)
        return cx, cy

    def center_of(self, x, y):
        """Return the center point of the cell containing (x, y)."""
        cx, cy = self.cell_of(x, y)
        return cx * self.cell_size + self.cell_size // 2, \
            cy * self.cell_size + self.cell_size // 2

    def in_range(self, v):
        return 0 <= v <= self.width

"""Grid based path planning."""


class PathPlanner:
    """Right-angle movement planner: horizontal first, then vertical."""

    def next_position(self, x, y, tx, ty, speed):
        """Move up to `speed` units toward the target; return (x, y)."""
        remaining = speed
        dx = tx - x
        step = min(abs(dx), remaining)
        x += step if dx > 0 else -step
        remaining -= step
        dy = ty - y
        step = min(abs(dy), remaining)
        y += step if dy > 0 else -step
        return x, y

    def plan(self, start, end):
        """Return waypoints [start, corner, end] of the L-shaped route."""
        pts = [tuple(start)]
        if start[0] != end[0] and start[1] != end[1]:
            pts.append((end[0], start[1]))
        if tuple(end) != pts[-1]:
            pts.append(tuple(end))
        return pts

    def plan_multi(self, points):
        """Continuous route through several destinations."""
        route = []
        for a, b in zip(points, points[1:]):
            seg = self.plan(a, b)
            route.extend(seg if not route else seg[1:])
        return route or [tuple(points[0])] if points else []

    def path_length(self, points):
        return sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                   for a, b in zip(points, points[1:]))

"""Order data model."""
import itertools

import globalvar
from grid_map import GridMap

_counter = itertools.count(1)


class Order:
    """A delivery order. Positions are converted to grid cell centers."""

    def __init__(self, res_pos, cus_pos):
        g = GridMap()
        self.id = next(_counter)
        self.res_pos = g.center_of(res_pos[0], res_pos[1])
        self.cus_pos = g.center_of(cus_pos[0], cus_pos[1])
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))
        self.time_taken = None

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, tick):
        self.completion_time = tick
        self.time_taken = tick - self.start_time
        self.status = "completed"

    def to_dict(self):
        return {"id": self.id, "restaurant": list(self.res_pos),
                "customer": list(self.cus_pos), "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken,
                "rider_id": self.rider_id, "status": self.status}

"""Rider model."""
import globalvar
from path_planner import PathPlanner

IDLE, PICKING, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.planner = PathPlanner()
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0) or 0,
                         "x": self.x, "y": self.y})

    def _target(self):
        order = self.orders[0]
        return order.res_pos if self.state == PICKING else order.cus_pos

    def move(self):
        """Advance one tick."""
        if self.state == IDLE:
            if not self.orders:
                return
            self.state = PICKING
            self.orders[0].status = "picking"
            self._record("accepted_order", self.orders[0].id)
        if not self.orders:
            self.state = IDLE
            return
        tx, ty = self._target()
        self.x, self.y = self.planner.next_position(
            self.x, self.y, tx, ty, self.speed)
        if (self.x, self.y) != (tx, ty):
            return
        order = self.orders[0]
        if self.state == PICKING:
            self.state = DELIVERING
            order.status = "delivering"
            self._record("arrived_at_merchant", order.id)
        else:
            tick = globalvar.get_value("tick", 0) or 0
            order.complete(tick)
            self._record("delivered_to_customer", order.id)
            self.orders.pop(0)
            self.state = IDLE

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order management and assignment."""
import math

import globalvar
from order_model import Order


def find_best_rider_for_order(order):
    """Return the rider closest (Euclidean) to the restaurant, or None."""
    riders = globalvar.get_value("riders", []) or []
    best, best_key = None, None
    rx, ry = order.res_pos
    for r in riders:
        d = math.hypot(r.x - rx, r.y - ry)
        key = (d, len(r.orders), r.rider_id)
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


class OrderManager:
    """Creates, assigns and stores orders."""

    def __init__(self):
        self.orders = []

    def create(self, coords):
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        self.orders.append(order)
        globalvar.set_value("orders", self.orders)
        return order

    def assign(self, order):
        rider = find_best_rider_for_order(order)
        if rider is None:
            return None
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        return rider

    def pending_count(self):
        return sum(1 for o in self.orders if o.status == "pending")

    def unstarted_count(self):
        return sum(1 for o in self.orders if o.status in ("pending", "assigned"))

"""System manager."""
import json

import globalvar
from grid_map import GridMap
from order import OrderManager
from rider import Rider

RIDER_COST = 300


def init_system():
    """Initialise global state."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Manages funds, riders, orders and time."""

    def __init__(self):
        init_system()
        self.grid = GridMap()
        self.order_manager = OrderManager()

    @property
    def funds(self):
        return globalvar.get_value("funds")

    @property
    def riders(self):
        return globalvar.get_value("riders")

    @property
    def tick(self):
        return globalvar.get_value("tick")

    @property
    def running(self):
        return globalvar.get_value("running")

    def idle_riders(self):
        return [r for r in self.riders if r.state == 0 and not r.orders]

    def recruit_if_needed(self):
        """Hire a rider if funds and demand allow. Returns the new rider."""
        if (self.funds >= RIDER_COST and
                self.order_manager.unstarted_count() > len(self.idle_riders())):
            cnt = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", cnt)
            globalvar.set_value("funds", self.funds - RIDER_COST)
            rider = Rider(cnt)
            self.riders.append(rider)
            return rider
        return None

    def submit_order(self, coords):
        """Create order, recruit if needed, assign. Returns (order, rider, new)."""
        order = self.order_manager.create(coords)
        new_rider = self.recruit_if_needed()
        rider = self.order_manager.assign(order)
        return order, rider, new_rider

    def step(self, n=1):
        for _ in range(n):
            globalvar.set_value("tick", self.tick + 1)
            for r in self.riders:
                r.move()

    def stop(self):
        globalvar.set_value("running", False)

    def save(self, path="config.json"):
        data = {"funds": self.funds, "tick": self.tick,
                "rider_id_counter": globalvar.get_value("rider_id_counter"),
                "riders": [dict(r.to_dict(), log=r.log) for r in self.riders],
                "orders": [o.to_dict() for o in self.order_manager.orders]}
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)

"""Status and map display."""

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Formats system information."""

    def __init__(self, system):
        self.system = system

    def map_text(self, route=None, order=None):
        n = self.system.grid.grid_dimension
        g = self.system.grid
        cells = [["." for _ in range(n)] for _ in range(n)]
        for p in route or []:
            cx, cy = g.cell_of(*p)
            cells[cy][cx] = "#"
        for r in self.system.riders:
            cx, cy = g.cell_of(r.x, r.y)
            cells[cy][cx] = "R"
        for o in self.system.order_manager.orders:
            if o.status == "completed":
                continue
            cx, cy = g.cell_of(*o.res_pos)
            cells[cy][cx] = "S"
            cx, cy = g.cell_of(*o.cus_pos)
            cells[cy][cx] = "C"
        return "\n".join(" ".join(row) for row in cells)

    def route_text(self, rider, order):
        from path_planner import PathPlanner
        pp = PathPlanner()
        pts = pp.plan_multi([(rider.x, rider.y), order.res_pos, order.cus_pos])
        length = pp.path_length(pts)
        time = -(-length // rider.speed)
        path = " -> ".join("(%d,%d)" % p for p in pts)
        return "%s\nSpeed: %d, estimated time: %d ticks\n%s" % (
            path, rider.speed, time, self.map_text(route=pts))

    def status_text(self):
        s = self.system
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (s.tick, s.funds, len(s.riders))]
        for r in s.riders:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATE_NAMES[r.state],
                [o.id for o in r.orders]))
        lines.append(self.map_text())
        return "\n".join(lines)

    def orders_text(self):
        orders = self.system.order_manager.orders
        if not orders:
            return "No orders."
        lines = []
        for o in orders:
            lines.append("Order %d start=%d restaurant=%s customer=%s dist=%d "
                         "time=%s rider=%s status=%s" % (
                             o.id, o.start_time, o.res_pos, o.cus_pos,
                             o.distance, o.time_taken, o.rider_id, o.status))
        return "\n".join(lines)

    def riders_text(self):
        if not self.system.riders:
            return "No riders."
        lines = []
        for r in self.system.riders:
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%d (%d,%d) %s order=%s" % (
                    e["time"], e["x"], e["y"], e["event"], e["order_id"]))
        return "\n".join(lines)

from display import DisplayManager


class InputManager:
    """Parses commands and runs them against the system."""

    def __init__(self, system):
        self.system = system
        self.display = DisplayManager(system)
        self.history = []
        self.aliases = {"s": "status", "o": "orders", "r": "riders",
                        "q": "quit", "exit": "quit"}

    def parse_coords(self, text):
        parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(parts) != 4:
            raise ValueError("Enter exactly four integers: rx ry cx cy")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        for v in vals:
            if not 0 <= v <= 800:
                raise ValueError("Coordinates must be within 0-800")
        return vals

    def handle(self, line, prompt=input):
        """Process one command; returns output text."""
        cmd = line.strip().lower()
        if not cmd:
            return ""
        self.history.append(cmd)
        cmd = self.aliases.get(cmd, cmd)
        if cmd == "quit":
            self.system.stop()
            return "Bye."
        if cmd == "status":
            return self.display.status_text()
        if cmd == "orders":
            return self.display.orders_text()
        if cmd == "riders":
            return self.display.riders_text()
        if cmd == "history":
            return "\n".join(self.history)
        if cmd == "start":
            try:
                coords = self.parse_coords(prompt("Enter rx ry cx cy: "))
            except ValueError as e:
                return "Error: %s" % e
            order, rider, new = self.system.submit_order(coords)
            out = ["Restaurant cell: %s  Customer cell: %s" % (
                order.res_pos, order.cus_pos)]
            out.append("New rider added: %s" % ("yes (#%d)" % new.rider_id
                                                 if new else "no"))
            if rider is None:
                out.append("No rider available; order pending.")
            else:
                out.append("Assigned to rider %d" % rider.rider_id)
                out.append(self.display.route_text(rider, order))
            return "\n".join(out)
        return "Unknown command. Use start/status/orders/riders/quit."

"""Entry point: each loop iteration is one tick."""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from input_manager import InputManager  # noqa: E402
from system import SystemManager  # noqa: E402


def main():
    system = SystemManager()
    im = InputManager(system)
    while system.running:
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            out = im.handle(line)
        except Exception as e:  # graceful recovery
            out = "Error: %s" % e
        if out:
            print(out)
        system.step()
    try:
        system.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

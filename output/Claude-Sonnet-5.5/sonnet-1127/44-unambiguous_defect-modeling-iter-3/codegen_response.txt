"""Core data models: Station, Line, Edge, Graph."""


class Station:
    def __init__(self, id, name, lines=None):
        self.id = id
        self.name = name
        self.lines = list(lines or [])

    @property
    def is_transfer(self):
        return len(self.lines) > 1


class Line:
    def __init__(self, name, station_sequence=None):
        self.name = name
        self.station_sequence = list(station_sequence or [])


class Edge:
    """Undirected connection between two station numbers (stored per direction)."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists keyed by station number."""

    def __init__(self):
        self.adjacency = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, []).append(
            Edge(edge.dest, edge.source, edge.distance, edge.line))

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def edges_between(self, a, b):
        return [e for e in self.adjacency.get(a, []) if e.dest == b]

    def nodes(self):
        return list(self.adjacency.keys())

"""MinPQ and Dijkstra-based ShortestPath with transfer-aware modes."""
import heapq
import itertools

MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 100000, "comprehensive": 1500}


class MinPQ:
    """Binary-heap priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}   # node -> state
        self._prev = {}   # state -> (prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        limit = cfg.max_transfers
        track = penalty > 0 or limit is not None
        pref = set(cfg.preferred_lines)
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line, t = state
            if node not in self._best:
                self._best[node] = state
                self.dist_to[node] = real[state]
            for e in self.graph.neighbors(node):
                nt = t
                if track and line is not None and e.line != line:
                    nt = t + 1
                    if limit is not None and nt > limit:
                        continue
                w = e.distance * (0.9 if e.line in pref else 1.0)
                if track and line is not None and e.line != line:
                    w += penalty
                nline = e.line if track else None
                ns = (e.dest, nline, nt)
                nc = c + w
                if ns not in done and nc < cost.get(ns, float("inf")):
                    cost[ns] = nc
                    real[ns] = real[state] + e.distance
                    self._prev[ns] = (state, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        state = self._best[target]
        edges = []
        nodes = [state[0]]
        while state in self._prev:
            state, e = self._prev[state]
            edges.append(e)
            nodes.append(state[0])
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""

FARE_TIERS = [(0, 6000, 3), (6000, 12000, 4), (12000, 22000, 5), (22000, 32000, 6)]


def Sum_money(distance) -> int:
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("Distance must be non-negative")
    d = int(round(distance))
    for lo, hi, fare in FARE_TIERS:
        if d < hi:
            return fare
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated minutes: ~35 km/h average, plus 5 min per transfer, 2 min boarding."""
    return round(distance / 1000.0 / 35.0 * 60.0 + 5.0 * transfer_count + 2.0, 1)


class FareCalculator:
    def calculate(self, distance):
        return Sum_money(distance)

"""Data management: building the graph, integrity checks, updates."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Graph, Edge

REQUIRED_LINES = ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
                  "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
                  "Fangshan Line", "Airport Line"]


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = None
        self.build_graph()

    def build_graph(self):
        g = Graph()
        for i, s in enumerate(self.data.get("stations", [])):
            g.add_station(s["name"], i)
            g.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a = g.station_to_number.get(c["source"])
            b = g.station_to_number.get(c["dest"])
            if a is None or b is None:
                continue
            g.add_edge(Edge(a, b, c["distance"], c["line"]))
        self.graph = g

    @property
    def station_to_number(self):
        return self.graph.station_to_number

    def has_station(self, name):
        return name in self.graph.station_to_number

    def line_sequences(self):
        seqs = {}
        for c in self.data.get("connections", []):
            seq = seqs.setdefault(c["line"], [])
            for n in (c["source"], c["dest"]):
                if n not in seq:
                    seq.append(n)
        return seqs

    def check_data(self):
        """Return (ok, messages)."""
        msgs = []
        ok = True
        stations = self.data.get("stations", [])
        names = {s["name"] for s in stations}
        all_lines = set()
        for s in stations:
            all_lines.update(s.get("lines", []))
        for c in self.data.get("connections", []):
            all_lines.add(c["line"])
        for ln in REQUIRED_LINES:
            if ln not in all_lines:
                ok = False
                msgs.append("Missing line: %s" % ln)
        for c in self.data.get("connections", []):
            if c["source"] not in names or c["dest"] not in names:
                ok = False
                msgs.append("Connection with unknown station: %s - %s" % (c["source"], c["dest"]))
            d = c.get("distance")
            if not isinstance(d, int) or d <= 0 or d > 30000:
                ok = False
                msgs.append("Suspicious distance %r: %s - %s" % (d, c["source"], c["dest"]))
        for name, num in self.graph.station_to_number.items():
            if not self.graph.neighbors(num):
                ok = False
                msgs.append("Isolated station: %s" % name)
        msgs.append("Stations: %d, connections: %d, lines: %d" % (
            len(stations), len(self.data.get("connections", [])), len(all_lines)))
        msgs.append("Data check passed" if ok else "Data check found problems")
        return ok, msgs

    # ----- update mechanisms -----
    def add_line(self, line_name, station_names, distances):
        """Add a line: consecutive stations with the given distances (len = n-1)."""
        if len(distances) != len(station_names) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        existing = {s["name"]: s for s in self.data["stations"]}
        for n in station_names:
            if n in existing:
                if line_name not in existing[n]["lines"]:
                    existing[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(station_names, station_names[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": int(d), "line": line_name})
        self.build_graph()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build_graph()
                return True
        return False

    def update_distance(self, a, b, distance):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = int(distance)
                found = True
        if found:
            self.build_graph()
        return found

"""Route, Query and PathReport."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


class Query:
    def __init__(self, start, end, mode="shortest_distance", max_transfers=None, line_preference=None):
        self.start = start
        self.end = end
        self.mode = mode
        self.max_transfers = max_transfers
        self.line_preference = line_preference


class Route:
    def __init__(self, stations, total_distance, edges):
        self.stations = stations
        self.total_distance = total_distance
        self.edges = edges
        self.transfers = []  # (station, from_line, to_line)
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                self.transfers.append((stations[i], edges[i - 1].line, edges[i].line))
        self.transfer_count = len(self.transfers)
        self.involved_lines = []
        for e in edges:
            if e.line not in self.involved_lines:
                self.involved_lines.append(e.line)

    def segments(self):
        segs = []
        for i, e in enumerate(self.edges):
            if segs and segs[-1]["line"] == e.line:
                segs[-1]["to"] = self.stations[i + 1]
                segs[-1]["count"] += 1
            else:
                segs.append({"line": e.line, "from": self.stations[i],
                             "to": self.stations[i + 1], "count": 1})
        return segs


class PathReport:
    def __init__(self, query, route, fmt="text"):
        self.query = query
        self.route = route
        self.format = fmt
        d = route.total_distance
        self.distance_km = round(d / 1000.0, 2)
        self.fare = Sum_money(d)
        self.estimated_time = calculate_estimated_time(d, route.transfer_count)

    def distance_text(self):
        d = self.route.total_distance
        return "%d m (%.2f km)" % (d, d / 1000.0)

    def to_dict(self):
        r = self.route
        return {
            "query": {"start": self.query.start, "end": self.query.end,
                      "mode": self.query.mode, "max_transfers": self.query.max_transfers,
                      "line_preference": self.query.line_preference},
            "path": r.stations,
            "distance_meters": r.total_distance,
            "distance_km": self.distance_km,
            "fare": self.fare,
            "estimated_time_minutes": self.estimated_time,
            "transfer_count": r.transfer_count,
            "transfers": [{"station": s, "from_line": a, "to_line": b} for s, a, b in r.transfers],
            "lines": r.involved_lines,
        }

    def to_json(self):
        return json.dumps(self.to_dict(), ensure_ascii=False, indent=2)

    def to_text(self):
        r = self.route
        out = ["Route: %s -> %s (mode: %s)" % (self.query.start, self.query.end, self.query.mode),
               "Stations (%d): %s" % (len(r.stations), " -> ".join(r.stations)),
               "Segments:"]
        for s in r.segments():
            out.append("  %s: %s -> %s (%d stops)" % (s["line"], s["from"], s["to"], s["count"]))
        out.append("Total distance: %s" % self.distance_text())
        out.append("Transfers: %d" % r.transfer_count)
        for s, a, b in r.transfers:
            out.append("  Transfer at %s: %s -> %s" % (s, a, b))
        out.append("Lines: %s" % ", ".join(r.involved_lines))
        out.append("Estimated time: %.0f minutes" % self.estimated_time)
        out.append("Fare: %d yuan" % self.fare)
        return "\n".join(out)

    def render(self):
        return self.to_json() if self.format == "json" else self.to_text()

"""Command-line interface for the Beijing Subway fare calculator."""
import argparse
import re
import sys

from data_manager import DataManager
from pathfinding import PathfindingConfig, ShortestPath
from report import Query, Route, PathReport

SEPARATORS = re.compile(r"[;；,，]")
FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_input(text):
    """Return (start, end) or raise ValueError with a user message."""
    parts = [p.strip() for p in SEPARATORS.split(text)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERR)
    return parts[0], parts[1]


def validate(dm, start, end):
    if not dm.has_station(start):
        raise ValueError("Starting station '%s' does not exist, please check the station name" % start)
    if not dm.has_station(end):
        raise ValueError("Ending station '%s' does not exist, please check the station name" % end)
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")


def run_query(dm, query, fmt="text"):
    """Compute a report for the query; raises ValueError on problems."""
    validate(dm, query.start, query.end)
    g = dm.graph
    cfg = PathfindingConfig(query.mode, query.max_transfers,
                            [query.line_preference] if query.line_preference else [])
    try:
        sp = ShortestPath(g, g.station_to_number[query.start], cfg)
        tgt = g.station_to_number[query.end]
        nodes, edges = sp.path_with_edges(tgt)
    except Exception as e:  # path calculation failure
        raise ValueError("Path calculation failed: %s" % e)
    if not nodes:
        raise ValueError("No route found between '%s' and '%s' (check --max_transfers)"
                         % (query.start, query.end))
    route = Route([g.number_to_station[n] for n in nodes], sp.dist_to[tgt], edges)
    try:
        return PathReport(query, route, fmt)
    except Exception as e:  # fare calculation failure
        raise ValueError("Fare calculation failed: %s" % e)


def build_parser():
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None, help="write results to this file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p


def main(argv=None):
    args = build_parser().parse_args(argv)
    try:
        dm = DataManager()
    except Exception as e:
        print("Failed to load subway data: %s" % e)
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1
    if args.max_transfers is not None and args.max_transfers < 0:
        print("--max_transfers must be non-negative")
        return 2
    if args.output_file:
        try:
            open(args.output_file, "w", encoding="utf-8").close()
        except OSError as e:
            print("Cannot write output file: %s" % e)
            return 2
    print("Beijing Subway fare calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        try:
            start, end = parse_input(line)
            q = Query(start, end, args.mode, args.max_transfers, args.line_preference)
            rep = run_query(dm, q, args.format)
            text = rep.render()
        except ValueError as e:
            print(e)
            continue
        print(text)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(text + "\n\n")
            except OSError as e:
                print("Cannot write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Global variable storage."""
_state = {}


def _init():
    """Reset all global variables to their initial values."""
    global _state
    _state = {
        "riders": [],
        "rider_id_counter": 0,
        "running": True,
        "funds": 0,
        "tick": 0,
        "orders": [],
        "history": [],
    }


def set_value(key, value):
    """Set a global value."""
    if not _state:
        _init()
    _state[key] = value


def get_value(key, default=None):
    """Get a global value."""
    if not _state:
        _init()
    return _state.get(key, default)

"""Grid path planning."""

CELL = 100
SIZE = 800


def to_center(v):
    """Convert a coordinate to its grid cell center."""
    v = max(0, min(SIZE, int(v)))
    return min(v // CELL, SIZE // CELL - 1) * CELL + CELL // 2


class PathPlanner:
    """Right-angle path planner (horizontal first, then vertical)."""

    @staticmethod
    def waypoints(start, end):
        """Return corner points from start to end."""
        (sx, sy), (ex, ey) = start, end
        pts = [(sx, sy)]
        if sx != ex and sy != ey:
            pts.append((ex, sy))
        if (ex, ey) != pts[-1]:
            pts.append((ex, ey))
        return pts

    @staticmethod
    def route(points):
        """Multi-destination route through successive points."""
        full = [points[0]]
        for a, b in zip(points, points[1:]):
            full.extend(PathPlanner.waypoints(a, b)[1:])
        return full

    @staticmethod
    def step(x, y, tx, ty, speed):
        """Move up to `speed` units toward target along axes."""
        budget = speed
        if x != tx:
            d = min(budget, abs(tx - x))
            x += d if tx > x else -d
            budget -= d
        if budget > 0 and y != ty:
            d = min(budget, abs(ty - y))
            y += d if ty > y else -d
        return x, y

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Rider model."""
import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 50
        self.y = 50
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0),
                         "x": self.x, "y": self.y})

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._log("accepted_order", self.orders[0].id)
        order = self.orders[0]
        if self.state == 1:
            tx, ty = order.res_pos
            self.x, self.y = PathPlanner.step(self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            tx, ty = order.cus_pos
            self.x, self.y = PathPlanner.step(self.x, self.y, tx, ty, self.speed)
            if (self.x, self.y) == (tx, ty):
                order.status = "completed"
                order.complete_time = globalvar.get_value("tick", 0)
                self.orders.pop(0)
                self.state = 0
                self._log("delivered_to_customer", order.id)

"""Order model."""
import globalvar
from path_planner import to_center


class Order:
    """A delivery order."""
    _counter = 0

    def __init__(self, res_pos, cus_pos):
        Order._counter += 1
        self.id = Order._counter
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.create_time = globalvar.get_value("tick", 0)
        self.complete_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Mark order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        """Serializable form."""
        return {"id": self.id, "res": list(self.res_pos),
                "cus": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start": self.create_time,
                "end": self.complete_time, "distance": self.distance}

"""Order management."""
import globalvar
from order_model import Order
from path_planner import PathPlanner


def find_best_rider_for_order(order):
    """Return the closest rider (idle riders preferred), or None."""
    riders = globalvar.get_value("riders", [])
    if not riders:
        return None

    def key(r):
        d = PathPlanner.distance((r.x, r.y), order.res_pos)
        return (len(r.orders), d, r.rider_id)
    return min(riders, key=key)


class OrderManager:
    """Create and assign orders."""

    def create_order(self, coords):
        """Create order from [rx, ry, cx, cy], assign it; return (order, rider)."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        globalvar.get_value("orders").append(order)
        rider = find_best_rider_for_order(order)
        if rider is not None:
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
        return order, rider

"""System management."""
import globalvar
from rider import Rider

RIDER_COST = 300


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", 300)


class SystemManager:
    """Manage funds, riders and ticks."""

    def recruit(self):
        """Recruit a rider if funds suffice; return the rider or None."""
        funds = globalvar.get_value("funds", 0)
        if funds < RIDER_COST:
            return None
        globalvar.set_value("funds", funds - RIDER_COST)
        rid = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", rid)
        r = Rider(rid)
        globalvar.get_value("riders").append(r)
        return r

    def tick(self):
        """Advance simulation one tick."""
        globalvar.set_value("tick", globalvar.get_value("tick", 0) + 1)
        for r in globalvar.get_value("riders"):
            before = len(r.orders)
            r.move()
            if len(r.orders) < before:
                globalvar.set_value("funds", globalvar.get_value("funds") + 100)

"""Display helpers."""
import globalvar
from path_planner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Render status and maps."""

    def render_map(self, route=None, orders=None):
        grid = [["." for _ in range(8)] for _ in range(8)]
        for px, py in self._cells(route or []):
            grid[py][px] = "#"
        for o in orders or []:
            grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
            grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        for r in globalvar.get_value("riders", []):
            grid[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
        return "\n".join(" ".join(row) for row in grid)

    @staticmethod
    def _cells(points):
        cells = set()
        for (ax, ay), (bx, by) in zip(points, points[1:]):
            ax, ay, bx, by = ax // 100, ay // 100, bx // 100, by // 100
            for x in range(min(ax, bx), max(ax, bx) + 1):
                for y in range(min(ay, by), max(ay, by) + 1):
                    cells.add((x, y))
        if len(points) == 1:
            cells.add((points[0][0] // 100, points[0][1] // 100))
        return cells

    def status(self):
        riders = globalvar.get_value("riders", [])
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick", 0), globalvar.get_value("funds", 0), len(riders))]
        for r in riders:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state], [o.id for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        lines = []
        for o in globalvar.get_value("orders", []):
            took = (o.complete_time - o.create_time) if o.complete_time is not None else "-"
            lines.append("Order %d start=%d res=%s cus=%s dist=%d time=%s rider=%s status=%s" % (
                o.id, o.create_time, o.res_pos, o.cus_pos, o.distance, took,
                o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    def riders(self):
        lines = []
        for r in globalvar.get_value("riders", []):
            for e in r.log:
                lines.append("t=%s rider=%d %s order=%s at (%s,%s)" % (
                    e["time"], r.rider_id, e["event"], e["order_id"], e["x"], e["y"]))
        return "\n".join(lines) or "No riders."

"""Input parsing and validation."""

ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
           "q": "quit", "exit": "quit", "t": "tick"}


class InputManager:
    """Parse commands and coordinates."""

    def __init__(self):
        self.history = []

    def parse_command(self, text):
        text = text.strip().lower()
        if text:
            self.history.append(text)
        return ALIASES.get(text, text)

    @staticmethod
    def parse_coords(text):
        """Parse four integers in 0-800; raise ValueError otherwise."""
        parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(parts) != 4:
            raise ValueError("Expected four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        for v in vals:
            if not 0 <= v <= 800:
                raise ValueError("Coordinates must be within 0-800.")
        return vals

"""Command-line entry point."""
import json
import globalvar
from system import init_system, SystemManager
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager
from path_planner import PathPlanner


def save(path="state.json"):
    data = {"funds": globalvar.get_value("funds"),
            "tick": globalvar.get_value("tick"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y, "state": r.state,
                        "log": r.log} for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
    try:
        with open(path, "w") as f:
            json.dump(data, f, indent=2)
    except OSError as e:
        print("Could not save state:", e)


def main():
    init_system()
    sysm, om, disp, im = SystemManager(), OrderManager(), DisplayManager(), InputManager()
    while globalvar.get_value("running"):
        try:
            cmd = im.parse_command(input("> "))
        except EOFError:
            break
        try:
            if cmd == "quit":
                globalvar.set_value("running", False)
            elif cmd == "start":
                coords = im.parse_coords(input("restaurant_x restaurant_y customer_x customer_y: "))
                added = None
                if not globalvar.get_value("riders") or all(
                        r.state != 0 for r in globalvar.get_value("riders")):
                    added = sysm.recruit()
                order, rider = om.create_order(coords)
                if rider is None:
                    print("No rider available (insufficient funds).")
                    continue
                print("Restaurant grid:", (order.res_pos[0] // 100, order.res_pos[1] // 100),
                      "Customer grid:", (order.cus_pos[0] // 100, order.cus_pos[1] // 100))
                print("New rider added:", bool(added))
                route = PathPlanner.route([(rider.x, rider.y), order.res_pos, order.cus_pos])
                n = 0
                while order.status != "completed" and n < 2000:
                    sysm.tick()
                    n += 1
                print(disp.render_map(route, [order]))
                print("Rider %d speed=%d time=%d ticks" % (rider.rider_id, rider.speed, n))
            elif cmd == "tick":
                sysm.tick()
                print(disp.status())
            elif cmd == "status":
                print(disp.status())
                print(disp.render_map(None, globalvar.get_value("orders")))
            elif cmd == "orders":
                print(disp.orders())
            elif cmd == "riders":
                print(disp.riders())
            elif cmd:
                print("Unknown command. Use start/status/orders/riders/tick/quit.")
        except ValueError as e:
            print("Error:", e)
        except EOFError:
            break
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
    save()


if __name__ == "__main__":
    main()

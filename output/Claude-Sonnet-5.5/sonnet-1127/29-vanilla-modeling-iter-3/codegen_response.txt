"""Order data model."""
import itertools

_counter = itertools.count(1)


class Order:
    """A delivery order. res_pos/cus_pos are (x, y) tuples."""

    def __init__(self, res_pos, cus_pos, start_time=None):
        import globalvar
        self.id = next(_counter)
        self.res_pos = tuple(res_pos)
        self.cus_pos = tuple(cus_pos)
        self.status = "pending"
        self.rider_id = None
        self.start_time = (globalvar.get_value("tick", 0)
                           if start_time is None else start_time)
        self.completion_time = None
        self.time_taken = None
        dx = self.res_pos[0] - self.cus_pos[0]
        dy = self.res_pos[1] - self.cus_pos[1]
        self.distance = (dx * dx + dy * dy) ** 0.5

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, time):
        """Mark as completed and record efficiency data."""
        self.status = "completed"
        self.completion_time = time
        self.time_taken = time - self.start_time

    def to_dict(self):
        return {"id": self.id, "restaurant": list(self.res_pos),
                "customer": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""Grid based path planning."""
import math

CELL = 100
SIZE = 800


class GridMap:
    """8x8 grid map of 100-unit cells."""
    width = SIZE
    height = SIZE
    cell_size = CELL


class Route:
    """A list of waypoints and estimated time."""

    def __init__(self, waypoints, estimated_time):
        self.waypoints = waypoints
        self.estimated_time = estimated_time


def to_center(v):
    """Convert coordinate to its grid cell center."""
    v = max(0, min(SIZE - 1, int(v)))
    return (v // CELL) * CELL + 50


class PathPlanner:
    """Plans right-angle routes."""

    def plan(self, start, targets, speed=10):
        """Route from start through targets (horizontal first, then vertical)."""
        pts = [tuple(start)]
        cur = tuple(start)
        for t in targets:
            t = tuple(t)
            if cur[0] != t[0] and cur[1] != t[1]:
                pts.append((t[0], cur[1]))
            if cur != t:
                pts.append(t)
            cur = t
        length = sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                     for a, b in zip(pts, pts[1:]))
        return Route(pts, int(math.ceil(length / float(speed))) if speed else 0)

    def step(self, x, y, tx, ty, budget):
        """Advance up to budget units toward target. Returns (x, y)."""
        if x != tx:
            d = min(budget, abs(tx - x))
            x += d if tx > x else -d
            budget -= d
        if budget > 0 and y != ty:
            d = min(budget, abs(ty - y))
            y += d if ty > y else -d
        return x, y

"""Rider model."""
import globalvar
from pathplanner import PathPlanner

_planner = PathPlanner()


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("tick", 0),
                         "x": self.x, "y": self.y, "state": self.state})

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.state = 1
            self._record("accepted_order", self.orders[0].id)
        order = self.orders[0] if self.orders else None
        if order is None:
            self.state = 0
            return
        if self.state == 1:
            tx, ty = order.res_pos
            self._advance(tx, ty)
            if (self.x, self.y) == (tx, ty):
                self.state = 2
                self._record("arrived_at_merchant", order.id)
        elif self.state == 2:
            tx, ty = order.cus_pos
            self._advance(tx, ty)
            if (self.x, self.y) == (tx, ty):
                order.complete(globalvar.get_value("tick", 0))
                self.orders.pop(0)
                self.state = 0
                self.special_case = False
                self._record("delivered_to_customer", order.id)

    def _advance(self, tx, ty):
        # directly above/below the target: only vertical movement
        self.special_case = (self.x == tx and self.y != ty)
        self.x, self.y = _planner.step(self.x, self.y, tx, ty, self.speed)

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order management: creation and assignment."""
import globalvar
from order_model import Order
from pathplanner import to_center


def create_order(rx, ry, cx, cy):
    """Create an order from raw coordinates (converted to cell centers)."""
    order = Order((to_center(rx), to_center(ry)),
                  (to_center(cx), to_center(cy)))
    orders = globalvar.get_value("orders")
    if orders is None:
        orders = []
        globalvar.set_value("orders", orders)
    orders.append(order)
    return order


def find_best_rider_for_order(order):
    """Nearest idle rider (Euclidean to restaurant); fallback least loaded."""
    riders = globalvar.get_value("riders") or []
    if not riders:
        return None

    def dist(r):
        return ((r.x - order.res_pos[0]) ** 2 +
                (r.y - order.res_pos[1]) ** 2) ** 0.5

    idle = [r for r in riders if r.state == 0 and not r.orders]
    if idle:
        return min(idle, key=lambda r: (dist(r), r.rider_id))
    return min(riders, key=lambda r: (len(r.orders), dist(r), r.rider_id))


def assign_order(order):
    """Assign order to best rider; returns rider or None."""
    rider = find_best_rider_for_order(order)
    if rider is not None:
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
    return rider

"""System manager."""
import json
import globalvar
from rider import Rider
import order as order_mod
from pathplanner import PathPlanner

RIDER_COST = 300


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])


def recruit_rider():
    """Create a new rider, deducting cost."""
    cnt = globalvar.get_value("rider_id_counter", 0)
    rider = Rider(cnt)
    globalvar.set_value("rider_id_counter", cnt + 1)
    globalvar.set_value("funds", globalvar.get_value("funds") - RIDER_COST)
    globalvar.get_value("riders").append(rider)
    return rider


class SystemManager:
    """Facade over the global state."""

    def __init__(self):
        init_system()
        self.planner = PathPlanner()

    def tick(self, n=1):
        for _ in range(n):
            globalvar.set_value("tick", globalvar.get_value("tick") + 1)
            for r in globalvar.get_value("riders"):
                r.move()

    def submit_order(self, rx, ry, cx, cy):
        """Returns (order, rider, recruited, route)."""
        o = order_mod.create_order(rx, ry, cx, cy)
        riders = globalvar.get_value("riders")
        pending = sum(1 for x in globalvar.get_value("orders")
                      if x.status == "pending")
        idle = sum(1 for r in riders if r.state == 0 and not r.orders)
        recruited = False
        if globalvar.get_value("funds") >= RIDER_COST and pending > idle:
            recruit_rider()
            recruited = True
        rider = order_mod.assign_order(o)
        route = None
        if rider is not None:
            route = self.planner.plan((rider.x, rider.y),
                                      [o.res_pos, o.cus_pos], rider.speed)
        return o, rider, recruited, route

    def save(self, path="config.json"):
        data = {"funds": globalvar.get_value("funds"),
                "tick": globalvar.get_value("tick"),
                "rider_id_counter": globalvar.get_value("rider_id_counter"),
                "riders": [r.to_dict() for r in globalvar.get_value("riders")],
                "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Display manager."""
import globalvar

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text rendering."""

    def render_map(self, route=None):
        grid = [["." for _ in range(8)] for _ in range(8)]

        def cell(p):
            return (min(7, max(0, p[1] // 100)), min(7, max(0, p[0] // 100)))

        if route:
            for a, b in zip(route.waypoints, route.waypoints[1:]):
                (r1, c1), (r2, c2) = cell(a), cell(b)
                for r in range(min(r1, r2), max(r1, r2) + 1):
                    for c in range(min(c1, c2), max(c1, c2) + 1):
                        grid[r][c] = "#"
        for o in globalvar.get_value("orders") or []:
            if o.status != "completed":
                r, c = cell(o.res_pos)
                grid[r][c] = "S"
                r, c = cell(o.cus_pos)
                grid[r][c] = "C"
        for rd in globalvar.get_value("riders") or []:
            r, c = cell((rd.x, rd.y))
            grid[r][c] = "R"
        return "\n".join(" ".join(row) for row in grid)

    def status(self):
        riders = globalvar.get_value("riders")
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("  Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        orders = globalvar.get_value("orders")
        if not orders:
            return "No orders."
        lines = []
        for o in orders:
            lines.append(
                "Order %d start=%s from=%s to=%s dist=%.1f time=%s rider=%s [%s]"
                % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                   o.time_taken if o.time_taken is not None else "-",
                   o.rider_id if o.rider_id is not None else "-", o.status))
        return "\n".join(lines)

    def riders_history(self):
        lines = []
        for r in globalvar.get_value("riders"):
            lines.append("Rider %d:" % r.rider_id)
            for e in r.log:
                lines.append("  t=%d (%d,%d) state=%s event=%s order=%s" % (
                    e["time"], e["x"], e["y"], STATES[e["state"]],
                    e["event"], e["order_id"]))
        return "\n".join(lines) if lines else "No riders."

"""Input manager: command parsing and validation."""
import globalvar
from display import DisplayManager

try:
    import readline  # noqa: F401
except ImportError:  # pragma: no cover
    pass

ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
           "t": "tick", "h": "history"}


class InputManager:
    """Reads and executes commands."""

    def __init__(self, system):
        self.system = system
        self.display = DisplayManager()
        self.command_history = []

    def parse_coords(self, text):
        parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
        if len(parts) != 4:
            raise ValueError("Need exactly 4 integers: rx ry cx cy")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        for v in vals:
            if not 0 <= v <= 800:
                raise ValueError("Coordinates must be within 0-800")
        return vals

    def execute(self, line):
        """Execute one command line; returns output text."""
        line = line.strip()
        if not line:
            return ""
        self.command_history.append(line)
        cmd, _, rest = line.partition(" ")
        cmd = ALIASES.get(cmd.lower(), cmd.lower())
        if cmd in ("quit", "exit"):
            globalvar.set_value("running", False)
            return "Bye."
        if cmd == "start":
            if not rest.strip():
                rest = input("Enter rx ry cx cy: ")
            vals = self.parse_coords(rest)
            o, rider, recruited, route = self.system.submit_order(*vals)
            out = ["Order %d: restaurant cell center %s, customer cell center %s"
                   % (o.id, o.res_pos, o.cus_pos),
                   "New rider added: %s" % ("yes" if recruited else "no")]
            if rider is None:
                out.append("No rider available; order pending.")
            else:
                out.append("Assigned rider %d, speed %d, est. time %d ticks"
                           % (rider.rider_id, rider.speed, route.estimated_time))
                out.append(self.display.render_map(route))
            return "\n".join(out)
        if cmd == "status":
            return self.display.status() + "\n" + self.display.render_map()
        if cmd == "orders":
            return self.display.orders()
        if cmd == "riders":
            return self.display.riders_history()
        if cmd == "history":
            return "\n".join(self.command_history)
        if cmd == "tick":
            n = int(rest) if rest.strip().isdigit() else 1
            self.system.tick(n)
            return "Advanced %d tick(s)." % n
        if cmd == "save":
            self.system.save(rest.strip() or "config.json")
            return "Saved."
        return "Unknown command. Use start, status, orders, riders, tick, save, quit."

"""Entry point."""
import globalvar
from system import SystemManager
from inputmgr import InputManager


def main():
    system = SystemManager()
    im = InputManager(system)
    print("Food Delivery Dispatch. Commands: start, status, orders, "
          "riders, tick [n], save, quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            out = im.execute(line)
            if out:
                print(out)
        except ValueError as e:
            print("Error: %s" % e)
        except Exception as e:  # recover gracefully
            print("Unexpected error: %s" % e)
        if globalvar.get_value("running") and line.strip() and \
                not line.strip().lower().startswith(("tick", "t ", "t")):
            system.tick()


if __name__ == "__main__":
    main()

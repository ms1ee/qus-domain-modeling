"""Core data structures: Edge and Graph (weighted directed graph)."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Directed weighted graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, number):
        self.adjacency.setdefault(number, [])

    def add_edge(self, edge):
        self.adjacency.setdefault(edge.source, []).append(edge)
        self.adjacency.setdefault(edge.dest, [])

    def neighbors(self, number):
        return self.adjacency.get(number, [])

    def nodes(self):
        return list(self.adjacency.keys())

"""Dijkstra shortest path with MinPQ, transfer-aware."""
import heapq
import itertools

TRANSFER_PENALTY = 1000.0      # meters-equivalent per transfer (comprehensive)
LEAST_TRANSFER_WEIGHT = 1e7
PREFERRED_FACTOR = 0.9


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self.edge_to = {}
        self._best_state = {}
        self._run()

    def _step_cost(self, edge, transferred):
        cfg = self.config
        d = float(edge.distance)
        if edge.line in cfg.preferred_lines:
            d *= PREFERRED_FACTOR
        if cfg.mode == "least_transfer":
            return d + (LEAST_TRANSFER_WEIGHT if transferred else 0.0)
        if cfg.mode == "comprehensive":
            return d + (TRANSFER_PENALTY if transferred else 0.0)
        return d

    def _run(self):
        limit = self.config.max_transfers
        start = (self.source, None, 0)
        cost = {start: 0.0}
        dist = {start: 0}
        prev = {start: (None, None)}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            for e in self.graph.neighbors(node):
                transferred = line is not None and e.line != line
                nt = t + (1 if transferred else 0)
                if limit is not None and nt > limit:
                    continue
                ns = (e.dest, e.line, nt if limit is not None else 0)
                if ns in done:
                    continue
                nc = c + self._step_cost(e, transferred)
                if nc < cost.get(ns, float("inf")):
                    cost[ns] = nc
                    dist[ns] = dist[st] + e.distance
                    prev[ns] = (st, e)
                    pq.push(nc, ns)
        best = {}
        for st, c in cost.items():
            n = st[0]
            if n not in best or c < cost[best[n]]:
                best[n] = st
        self._prev = prev
        self._best_state = best
        for n, st in best.items():
            self.dist_to[n] = dist[st]

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        nodes, edges = [], []
        while st is not None:
            nodes.append(st[0])
            p, e = self._prev[st]
            if e is not None:
                edges.append(e)
            st = p
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (fractional km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0 - 1e-12)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km < 22:
        return 5
    if km <= 32:
        return 6
    return 7 + (km - 33) // 20


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5.0

"""Data management: building the graph, checks and updates."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE = {
    "1号线": [], "2号线": [], "4号线": [], "5号线": [], "10号线": [],
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.build()

    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in self.data.get("stations", []):
            self._add_station(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in self.station_to_number:
                    self._add_station(n, [c.get("line")])
            self.graph.add_edge(Edge(self.station_to_number[c["source"]],
                                     self.station_to_number[c["dest"]],
                                     c["distance"], c.get("line")))

    def _add_station(self, name, lines):
        if name in self.station_to_number:
            return
        num = len(self.station_to_number)
        self.station_to_number[name] = num
        self.number_to_station[num] = name
        self.graph.add_node(num)
        self.graph.set_station_lines(name, lines)

    def is_transfer(self, name):
        return len(self.graph.station_lines.get(name, [])) > 1

    def lines(self):
        result = {}
        for s in self.data.get("stations", []):
            for l in s.get("lines", []):
                result.setdefault(l, []).append(s["name"])
        return result

    # ---- integrity check ----
    def check_data(self):
        """Return (ok, list of messages)."""
        msgs = []
        ok = True
        names = set(self.station_to_number)
        lines = self.lines()
        wanted = ["1号线", "2号线", "4号线", "5号线", "6号线", "8号线", "10号线",
                  "13号线", "14号线", "15号线", "昌平线", "亦庄线", "房山线", "首都机场线"]
        missing = [w for w in wanted if not any(w in l or l in w for l in lines)]
        msgs.append("Stations: %d, lines: %d" % (len(names), len(lines)))
        if missing:
            msgs.append("Lines possibly missing: " + ", ".join(missing))
        conns = self.data.get("connections", [])
        stset = {s["name"] for s in self.data.get("stations", [])}
        for c in conns:
            if c["source"] not in stset or c["dest"] not in stset:
                ok = False
                msgs.append("Connection references unknown station: %s -> %s" % (c["source"], c["dest"]))
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                ok = False
                msgs.append("Invalid distance: %s -> %s (%r)" % (c["source"], c["dest"], d))
        connected = set()
        for c in conns:
            connected.add(c["source"])
            connected.add(c["dest"])
        isolated = stset - connected
        if isolated:
            ok = False
            msgs.append("Stations without connections: %d" % len(isolated))
        msgs.append("Data check " + ("passed" if ok else "FAILED"))
        return ok, msgs

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: meters between consecutive stations."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
            self.data["connections"].append({"source": b, "dest": a, "distance": d, "line": line_name})
        self.build()

    def modify_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                found = True
        if found:
            self.build()
        return found

"""Route reports: text and JSON."""
import json
from fare_calculator import Sum_money, calculate_estimated_time


class Route:
    def __init__(self, station_sequence, total_distance, transfer_count, involved_lines, edges):
        self.station_sequence = station_sequence
        self.total_distance = total_distance
        self.transfer_count = transfer_count
        self.involved_lines = involved_lines
        self.edges = edges


class PathSummary:
    def __init__(self, route):
        self.route = route
        self.total_distance = route.total_distance
        self.fare = Sum_money(route.total_distance)
        self.transfer_count = route.transfer_count
        self.estimated_time = calculate_estimated_time(route.total_distance, route.transfer_count)


def make_route(names, edges, total):
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    tc = sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)
    return Route(names, total, tc, lines, edges)


def _dist_text(m):
    return "%d m" % m if m < 1000 else "%.2f km (%d m)" % (m / 1000.0, m)


class ReportExporter:
    def __init__(self, fmt="text", output_file=None):
        self.format = fmt
        self.output_file = output_file
        self._started = False

    def render(self, start, end, summary, names=None):
        if self.format == "json":
            return self.to_json(start, end, summary)
        return self.to_text(start, end, summary)

    def to_text(self, start, end, summary):
        r = summary.route
        out = ["Route: %s -> %s" % (start, end),
               "Stations (%d): %s" % (len(r.station_sequence), " -> ".join(r.station_sequence))]
        # line segments and transfers
        segs = []
        for i, e in enumerate(r.edges):
            if segs and segs[-1][0] == e.line:
                segs[-1][2] = r.station_sequence[i + 1]
            else:
                segs.append([e.line, r.station_sequence[i], r.station_sequence[i + 1]])
        for line, a, b in segs:
            out.append("  [%s] %s -> %s" % (line, a, b))
        if len(segs) > 1:
            out.append("Transfer stations: " + ", ".join(s[1] for s in segs[1:]))
        out.append("Lines: " + ", ".join(r.involved_lines))
        out.append("Total distance: " + _dist_text(r.total_distance))
        out.append("Transfers: %d" % summary.transfer_count)
        out.append("Estimated time: %.1f minutes" % summary.estimated_time)
        out.append("Fare: %d yuan" % summary.fare)
        return "\n".join(out)

    def to_json(self, start, end, summary):
        r = summary.route
        return json.dumps({
            "query": {"start": start, "end": end},
            "path": r.station_sequence,
            "distance": summary.total_distance,
            "fare": summary.fare,
            "estimated_time": round(summary.estimated_time, 2),
            "transfer_count": summary.transfer_count,
            "lines": r.involved_lines,
        }, ensure_ascii=False, indent=2)

    def export(self, text):
        if not self.output_file:
            return
        mode = "a" if self._started else "w"
        with open(self.output_file, mode, encoding="utf-8") as f:
            f.write(text + "\n")
        self._started = True

"""CLI entry point for the Beijing Subway fare calculator."""
import argparse
import difflib
import re
import sys

from data_manager import DataManager
from pathfinding import PathfindingConfig, ShortestPath
from report import PathSummary, ReportExporter, make_route

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def resolve_station(name, dm):
    if name in dm.station_to_number:
        return name
    m = difflib.get_close_matches(name, list(dm.station_to_number), n=1, cutoff=0.5)
    return m[0] if m else None


def process_query(text, dm, args, exporter):
    parts = [p.strip() for p in re.split(r"[;；,，]", text) if p.strip()]
    if len(parts) != 2:
        return FORMAT_ERR
    s_raw, e_raw = parts
    s = resolve_station(s_raw, dm)
    if s is None:
        return "Starting station '%s' does not exist, please check the station name" % s_raw
    e = resolve_station(e_raw, dm)
    if e is None:
        return "Ending station '%s' does not exist, please check the station name" % e_raw
    if s == e:
        return "Starting station cannot be the same as the ending station"
    cfg = PathfindingConfig(args.mode, args.max_transfers,
                            [args.line_preference] if args.line_preference else [])
    try:
        sp = ShortestPath(dm.graph, dm.station_to_number[s], cfg)
        t = dm.station_to_number[e]
        nodes, edges = sp.path_with_edges(t)
    except Exception as ex:
        return "Path calculation failed: %s" % ex
    if not nodes:
        return "No route found from '%s' to '%s' under the current settings" % (s, e)
    try:
        route = make_route([dm.number_to_station[n] for n in nodes], edges, sp.dist_to[t])
        summary = PathSummary(route)
    except Exception as ex:
        return "Fare calculation failed: %s" % ex
    out = exporter.render(s, e, summary)
    try:
        exporter.export(out)
    except OSError as ex:
        out += "\n(Could not write output file: %s)" % ex
    return out


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager()
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1
    exporter = ReportExporter(args.format, args.output_file)
    print("Enter 'Starting Station; Ending Station' (or 'q' to quit):")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        print(process_query(line, dm, args, exporter))
    return 0


if __name__ == "__main__":
    sys.exit(main())

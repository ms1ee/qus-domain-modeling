"""Grid path planning."""


class PathPlanner:
    """Axis-aligned (right-angle) path planner."""

    @staticmethod
    def plan(start, end, horizontal_first=True):
        """Return list of waypoints from start to end (inclusive)."""
        sx, sy = start
        ex, ey = end
        if sx == ex or sy == ey:
            return [(sx, sy), (ex, ey)]
        corner = (ex, sy) if horizontal_first else (sx, ey)
        return [(sx, sy), corner, (ex, ey)]

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    @staticmethod
    def step(pos, target, speed):
        """Move up to speed units toward target: horizontal first, then vertical."""
        x, y = pos
        tx, ty = target
        left = speed
        if x != tx:
            d = min(left, abs(tx - x))
            x += d if tx > x else -d
            left -= d
        if left > 0 and y != ty:
            d = min(left, abs(ty - y))
            y += d if ty > y else -d
        return x, y

"""Rider model."""
import globalvar
from path_planner import PathPlanner


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = 50
        self.speed = 50
        self.state = 0
        self.orders = []
        self.special_case = False
        self.current = None
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({
            "event": event, "order_id": order_id,
            "time": globalvar.get_value("time", 0) or 0,
            "pos": (self.x, self.y)})

    @property
    def pos(self):
        return (self.x, self.y)

    def _go(self, target):
        self.x, self.y = PathPlanner.step(self.pos, target, self.speed)
        return self.pos == tuple(target)

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            pending = [o for o in self.orders if o.status != "completed"]
            if not pending:
                return
            self.current = pending[0]
            self.state = 1
            self._log("accepted_order", self.current.id)
            if hasattr(self.current, "status"):
                self.current.status = "picking_up"
            return
        order = self.current
        if order is None:
            self.state = 0
            return
        if self.state == 1:
            if self._go(order.res_pos):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
                order.status = "delivering"
        elif self.state == 2:
            if self._go(order.cus_pos):
                self._log("delivered_to_customer", order.id)
                order.complete(globalvar.get_value("time", 0) or 0)
                if order in self.orders:
                    self.orders.remove(order)
                self.current = None
                self.state = 0

"""Order model."""
import itertools

_counter = itertools.count(1)


def to_center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(799, int(v)))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        import globalvar
        self.id = next(_counter)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0) or 0
        self.end_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Assign to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def complete(self, now):
        """Mark completed."""
        self.status = "completed"
        self.end_time = now

    def to_dict(self):
        return {"id": self.id, "res": list(self.res_pos),
                "cus": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "end_time": self.end_time, "distance": self.distance}

"""System management."""
import json
import globalvar
from rider import Rider

RIDER_COST = 500
INITIAL_FUNDS = 10000
MAX_RIDERS = 12


def init_system():
    """Initialize global state."""
    globalvar.set_value("funds", INITIAL_FUNDS)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("time", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Manages funds, riders and persistence."""

    @staticmethod
    def recruit_rider():
        """Recruit a rider if funds allow; return it or None."""
        riders = globalvar.get_value("riders")
        funds = globalvar.get_value("funds")
        if riders is None or funds is None:
            init_system()
            riders = globalvar.get_value("riders")
            funds = globalvar.get_value("funds")
        if funds < RIDER_COST or len(riders) >= MAX_RIDERS:
            return None
        rid = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", rid)
        globalvar.set_value("funds", funds - RIDER_COST)
        r = Rider(rid)
        riders.append(r)
        return r

    @staticmethod
    def tick():
        """Advance simulated time and move riders."""
        globalvar.set_value("time", (globalvar.get_value("time", 0) or 0) + 1)
        for r in globalvar.get_value("riders", []):
            r.move()

    @staticmethod
    def save(path="state.json"):
        """Persist state to JSON."""
        data = {
            "funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state,
                        "log": r.log} for r in globalvar.get_value("riders", [])],
            "orders": [o.to_dict() for o in globalvar.get_value("orders", [])],
        }
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Order management and assignment."""
import globalvar
from order_model import Order
from path_planner import PathPlanner


def _cost(rider, order):
    base = PathPlanner.distance(rider.pos, order.res_pos)
    load = len([o for o in rider.orders if o.status != "completed"])
    return base + load * 400 + (0 if rider.state == 0 else 200)


def find_best_rider_for_order(order):
    """Pick best rider (distance + load); recruit if none or all busy."""
    from system import SystemManager
    riders = globalvar.get_value("riders")
    if riders is None:
        riders = []
        globalvar.set_value("riders", riders)
    if not any(r.state == 0 and not r.orders for r in riders):
        SystemManager.recruit_rider()
    if not riders:
        return None
    return min(riders, key=lambda r: (_cost(r, order), r.rider_id))


class OrderManager:
    """Creates and dispatches orders."""

    @staticmethod
    def create_order(coords):
        """Create order from [rx, ry, cx, cy] and dispatch it."""
        before = len(globalvar.get_value("riders", []))
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        rider = find_best_rider_for_order(order)
        if rider is None:
            return order, None, False
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        globalvar.get_value("orders").append(order)
        added = len(globalvar.get_value("riders")) > before
        return order, rider, added

"""Display manager."""
import globalvar
from path_planner import PathPlanner

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and map rendering."""

    @staticmethod
    def render_map(riders=(), orders=(), path=None):
        grid = [["." for _ in range(8)] for _ in range(8)]
        if path:
            for a, b in zip(path, path[1:]):
                n = max(abs(b[0] - a[0]), abs(b[1] - a[1]))
                for i in range(n + 1):
                    x = a[0] + (b[0] - a[0]) * i // max(n, 1)
                    y = a[1] + (b[1] - a[1]) * i // max(n, 1)
                    grid[min(7, y // 100)][min(7, x // 100)] = "*"
        for o in orders:
            if o.status != "completed":
                grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
                grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
        for r in riders:
            grid[min(7, r.y // 100)][min(7, r.x // 100)] = "R"
        return "\n".join(" ".join(row) for row in grid)

    @staticmethod
    def status():
        riders = globalvar.get_value("riders", [])
        lines = ["Time: %s  Funds: %s  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        lines.append(DisplayManager.render_map(
            riders, globalvar.get_value("orders", [])))
        return "\n".join(lines)

    @staticmethod
    def orders():
        out = []
        for o in globalvar.get_value("orders", []):
            took = (o.end_time - o.start_time) if o.end_time is not None else "-"
            out.append("Order %d start=%s res=%s cus=%s dist=%d time=%s rider=%s [%s]"
                       % (o.id, o.start_time, o.res_pos, o.cus_pos,
                          o.distance, took, o.rider_id, o.status))
        return "\n".join(out) or "No orders."

    @staticmethod
    def riders():
        out = []
        for r in globalvar.get_value("riders", []):
            for e in r.log:
                out.append("t=%s rider %d %s order=%s at %s" % (
                    e["time"], r.rider_id, e["event"], e["order_id"], e["pos"]))
        return "\n".join(out) or "No riders."

    @staticmethod
    def route(order, rider):
        pts = [rider.pos, order.res_pos, order.cus_pos]
        path = PathPlanner.plan(pts[0], pts[1])[:-1] + PathPlanner.plan(pts[1], pts[2])
        total = PathPlanner.distance(pts[0], pts[1]) + order.distance
        eta = -(-total // rider.speed)
        return "%s\nSpeed %d, estimated time %d ticks" % (
            DisplayManager.render_map([rider], [order], path), rider.speed, eta)

"""Input parsing and validation."""


class InputManager:
    """Parses commands and validates coordinates."""

    ALIASES = {"s": "start", "st": "status", "o": "orders", "r": "riders",
               "q": "quit", "exit": "quit"}

    @staticmethod
    def parse_command(text):
        t = text.strip().lower()
        return InputManager.ALIASES.get(t, t)

    @staticmethod
    def parse_coords(text):
        """Parse four integers in 0-800; raise ValueError otherwise."""
        parts = text.replace("[", " ").replace("]", " ").replace(",", " ").split()
        if len(parts) != 4:
            raise ValueError("Please enter exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be within 0-800.")
        return vals

"""Command-line entry point."""
import globalvar
import system
from system import SystemManager
from order import OrderManager
from display import DisplayManager
from input_manager import InputManager


def main():
    globalvar._init()
    system.init_system()
    history = []
    print("Commands: start(s), status(st), orders(o), riders(r), quit(q)")
    while globalvar.get_value("running"):
        try:
            cmd = InputManager.parse_command(input("> "))
        except (EOFError, KeyboardInterrupt):
            break
        history.append(cmd)
        try:
            if cmd == "start":
                coords = InputManager.parse_coords(
                    input("Enter rx ry cx cy: "))
                order, rider, added = OrderManager.create_order(coords)
                if rider is None:
                    print("No rider available.")
                else:
                    print("Restaurant %s, customer %s, new rider added: %s"
                          % (order.res_pos, order.cus_pos, added))
                    print(DisplayManager.route(order, rider))
            elif cmd == "status":
                print(DisplayManager.status())
            elif cmd == "orders":
                print(DisplayManager.orders())
            elif cmd == "riders":
                print(DisplayManager.riders())
            elif cmd == "quit":
                globalvar.set_value("running", False)
            elif cmd:
                print("Unknown command.")
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
        SystemManager.tick()
    try:
        SystemManager.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang Wei", "Technology Department"),
    ("Wang Fang", "Technology Department"),
    ("Li Na", "Technology Department"),
    ("Liu Yang", "Technology Department"),
    ("Chen Jing", "Product Department"),
    ("Zhao Lei", "Product Department"),
    ("Sun Li", "Product Department"),
    ("Zhou Tao", "Market Department"),
    ("Wu Qiang", "Market Department"),
    ("Zheng Xin", "Market Department"),
    ("Feng Jun", "Cheng Ru group"),
    ("Cai Ling", "Cheng Ru group"),
    ("Xu Hao", "Cheng Ru group"),
    ("Li Bin", "Operations Department"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Finance Department"),
    ("Ma Jiaqi", "Finance Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users = []
        errors = []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.replace("，", ",").split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: expected 'name,department' but got '{line.strip()}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid = []
        dups = []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize

PRESETS = [
    ("Third Prize - Movie Tickets", 3),
    ("Second Prize - Shopping Cards", 2),
    ("First Prize - 1 Day Annual Leave", 1),
]


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in PRESETS:
            try:
                self.add_prize(name, qty)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if len(order) != n or sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "cheng ru"
PRE_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


def _is_cheng_ru(dept):
    return CHENG_RU in dept.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = set()
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if u.name not in self.winners]
        drawer = getattr(prize, "associated_drawer", None)
        if drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if _is_cheng_ru(u.department)]
        if drawer == SPECIAL_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def rule_description(self, prize):
        parts = ["Exclude previous winners"]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            parts.append("Li Bin: relaxed filtering (avoidance groups ignored)")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        if drawer == PROTECTED_DRAWER:
            parts.append("Cheng Ru protection may apply")
        return "; ".join(parts)

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.add(u.name)
        return u

    def finish_round(self, prize, winners):
        """Record fallback state after a round."""
        if prize.associated_drawer in PRE_DRAWERS:
            self.last_two_draw_results.append(any(_is_cheng_ru(u.department) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime


class Winner:
    def __init__(self, user, prize, order=0):
        self.user = user
        self.prize = prize
        self.order = order
        self.time = datetime.datetime.now()


def _level(name):
    n = name.lower()
    for i, k in enumerate(("special", "first", "second", "third")):
        if k in n:
            return i
    return 99


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize, len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        ordered = sorted(prizes, key=lambda p: _level(p.name))
        result = {}
        for p in ordered:
            ws = [w for w in self.winners if w.prize is p or w.prize.name == p.name]
            ws.sort(key=lambda w: w.order)
            result[p.name] = ws
        return result

    def export_to_csv(self, prizes):
        fname = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(fname, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for pname, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{pname}\n")
            return True, f"Results successfully exported to {fname}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def show_groups(lm, users):
    for g, c in lm.get_group_info(users).items():
        print(f"  {g}: {c}")


def load_list(lm):
    c = input("1) default list  2) custom txt file: ").strip()
    if c == "2":
        path = input("Path: ").strip()
        users, errors = lm.load_custom_list(path)
    else:
        users, errors = lm.load_default_list()
    for e in errors:
        print(e)
    if errors and c == "2":
        a = input("Errors found. (r)emove invalid lines and continue / (t)erminate upload: ").strip().lower()
        if a != "r":
            print("Upload terminated.")
            return None
    valid, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate name: {d.name} ({d.department}) removed")
    print(f"Loaded {len(valid)} valid participants.")
    show_groups(lm, valid)
    return valid


def run_lottery(users, prizes, results):
    engine = LotteryManager(users)
    for idx, prize in enumerate(prizes, 1):
        print(f"\n=== Round {idx}: {prize.name} x{prize.quantity} ===")
        print("Rules:", engine.rule_description(prize))
        won = []
        terminated = False
        for n in range(1, prize.quantity + 1):
            while True:
                cmd = input("Enter to draw (pause / continue / terminate): ").strip().lower()
                if cmd == "pause":
                    while input("Paused. Type 'continue' to resume: ").strip().lower() != "continue":
                        pass
                    continue
                if cmd == "terminate":
                    terminated = True
                break
            if terminated:
                print("Round terminated, skipping.")
                break
            print(f"Currently drawing the {n}th winner...")
            u = engine.draw_one(prize)
            if u is None:
                print("No eligible participants left.")
                break
            results.add_winner(u, prize)
            won.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        engine.finish_round(prize, won)
        print(f"Round {idx} results:")
        for i, u in enumerate(won, 1):
            print(f"  {i}. {u.name} ({u.department})")
    print("\n=== Final winner list ===")
    for pname, ws in results.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {pname}")


def main():
    lm = ListManager()
    pm = PrizeManager()
    pm.load_default_prizes()
    users = lm.load_default_list()[0]
    results = ResultManager()
    while True:
        print("\n1 Load list  2 View groups  3 List prizes  4 Add prize  5 Set avoid groups")
        print("6 Reorder prizes  7 Start lottery  8 Export  0 Exit")
        c = input("> ").strip()
        if c == "1":
            r = load_list(lm)
            if r is not None:
                users = r
        elif c == "2":
            show_groups(lm, users)
        elif c == "3":
            for i, p in enumerate(pm.get_prizes(), 1):
                print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
        elif c == "4":
            try:
                name = input("Name: ").strip()
                q = int(input("Quantity: "))
                d = input("Drawer (optional): ").strip() or None
                av = [x.strip() for x in input("Avoid groups (comma separated): ").split(",") if x.strip()]
                pm.add_prize(name, q, av, d)
            except ValueError as e:
                print("Invalid input:", e)
        elif c == "5":
            try:
                i = int(input("Prize number: "))
                av = [x.strip() for x in input("Avoid groups (comma separated): ").split(",") if x.strip()]
                pm.get_prizes()[i - 1].avoid_groups = av
            except (ValueError, IndexError):
                print("Invalid input.")
        elif c == "6":
            order = input("New order (e.g. 3 1 2): ").replace(",", " ").split()
            print("Reordered." if pm.reorder_prizes(order) else "Invalid order.")
        elif c == "7":
            results = ResultManager()
            run_lottery(users, pm.get_prizes(), results)
        elif c == "8":
            print(results.export_to_csv(pm.get_prizes())[1])
        elif c == "0":
            break


if __name__ == "__main__":
    main()

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Chen Qixian", "Marketing Department"),
    ("Jin Duo", "Technology Department"),
    ("Ma Jiaqi", "Product Department"),
    ("Li Bin", "Operations Department"),
    ("Liu Yi", "Cheng Ru Group"),
    ("Qian Er", "Cheng Ru Group"),
    ("Feng San", "Cheng Ru Group"),
    ("He Si", "Operations Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error at Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, a, d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        self._prizes.append(Prize(name, quantity, list(avoid_groups or []), associated_drawer))

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "cheng ru group"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


def _is_cheng_ru(user):
    return user.department.strip().lower() == CHENG_RU


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = set()
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if u.name not in self.winners]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_cheng_ru(u)]
        avoid = {g.strip().lower() for g in (prize.avoid_groups or [])}
        return [u for u in pool if u.department.strip().lower() not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.add(u.name)
        return u

    def finish_round(self, prize, round_winners):
        if prize.associated_drawer in WATCHED_DRAWERS:
            self.last_two_draw_results.append(any(_is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    @staticmethod
    def describe_rules(prize):
        parts = []
        if prize.associated_drawer:
            parts.append(f"Drawer: {prize.associated_drawer}")
        if prize.associated_drawer == SPECIAL_DRAWER:
            parts.append("only previous winners excluded")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        parts.append("previous winners excluded")
        return "; ".join(parts)

import datetime
import os


class Winner:
    def __init__(self, user, prize_name):
        self.user = user
        self.prize_name = prize_name


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        grouped = self.get_winners_by_prize(prizes)
        if not any(grouped.values()):
            return False, "No winners to export"
        fname = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), fname)
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for name, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def ask(msg):
    try:
        return input(msg).strip()
    except EOFError:
        return ""


def load_users(lm):
    while True:
        c = ask("Load list: 1) default 2) custom txt > ")
        if c == "2":
            users, errors = lm.load_custom_list(ask("Path: "))
        else:
            users, errors = lm.load_default_list()
        for e in errors:
            print(e)
        if errors and ask("Errors found. Remove invalid data and continue? (y/n) ").lower() != "y":
            print("Upload terminated.")
            if c == "2":
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name} ({d.department})")
        if valid:
            return valid
        print("No valid users.")


def show_groups(lm, users):
    for g, n in lm.get_group_info(users).items():
        print(f"  {g}: {n}")


def setup_prizes(pm):
    if ask("Load default prizes? (y/n) ").lower() == "y":
        pm.load_default_prizes()
    while ask("Add custom prize? (y/n) ").lower() == "y":
        name = ask("Prize name: ")
        try:
            q = int(ask("Quantity: "))
            avoid = [g.strip() for g in ask("Avoid groups (comma separated, optional): ").split(",") if g.strip()]
            drawer = ask("Associated drawer (optional): ") or None
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    o = ask("Enter new order as numbers (e.g. 3 1 2), blank to keep: ")
    if o:
        try:
            ok = pm.reorder_prizes([int(x) for x in o.replace(",", " ").split()])
        except ValueError:
            ok = False
        print("Reordered." if ok else "Invalid order, keeping default.")


def run_round(lot, prize, rm):
    print(f"\n=== {prize.name} | quantity {prize.quantity} ===")
    print("Rules:", lot.describe_rules(prize))
    won = []
    i = 1
    while i <= prize.quantity:
        cmd = ask("Enter to draw, 'pause', or 'terminate': ").lower()
        if cmd == "pause":
            while ask("Paused. Type 'continue': ").lower() not in ("continue",):
                pass
            continue
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing winner #{i}")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        rm.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, won)


def main():
    lm = ListManager()
    users = load_users(lm)
    show_groups(lm, users)
    pm = PrizeManager()
    setup_prizes(pm)
    prizes = pm.get_prizes()
    lot = LotteryManager(users)
    rm = ResultManager()
    for p in prizes:
        run_round(lot, p, rm)
    print("\n=== All Winners ===")
    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    if ask("Export results? (y/n) ").lower() == "y":
        print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()

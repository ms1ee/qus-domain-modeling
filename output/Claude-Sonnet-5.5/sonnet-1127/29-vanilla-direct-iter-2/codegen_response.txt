I had to guess several details the Interface leaves open, and I haven't run any of this code.

- **Coordinates.** `Order` takes `res_pos` and `cus_pos` as raw (x, y) tuples and converts them to grid-cell centres, so `(120, 340)` becomes `(150, 350)`. Rider positions stay raw integers.
- **Movement.** Riders move along the grid streets, which sit at multiples of 100. The spawn point (435, 385) is on a road: 435 and 385 are not multiples of 100, but a restaurant or customer point at a cell centre like (150, 350) is not on a street either. To reach a target, a rider first moves to the nearest street line and then follows it. Each rider builds a waypoint list (street-aligned vertical and horizontal segments ending at the target) and follows it at 10 units per tick. This is a simplification of the PRD's pathfinding.
- **Rider logging.** `Rider.log` entries are dicts with `event`, `order_id`, `time`, `x` and `y`. `Rider(rider_id)` appends a `"spawn"` entry with `order_id` set to `None`.
- **Accepting orders.** Orders are assigned by appending to `rider.orders`. In `move()`, an idle rider with orders accepts the first one, logs `"accepted_order"`, and enters state 1.
- **Pickup and delivery.** On reaching the restaurant the rider logs `"arrived_at_merchant"` and enters state 2. On reaching the customer it logs `"delivered_to_customer"`, marks the order `"completed"`, removes it from `orders`, and returns to idle.
- **Globals.** `globalvar` holds a dict. `system.init_system()` sets funds to 1000, an empty rider list, the ID counter to 0, the running flag, time 0 and an empty order list.
- **Assignment.** `find_best_rider_for_order` takes the nearest rider by Euclidean distance to the restaurant. If there are no riders and funds are at least 300, it recruits one. It returns `None` if no rider can be found. Ties are broken by fewer orders, which gives some load balancing.
- **Interface.** I added a minimal CLI in `main.py` with `start`, `status`, `orders`, `riders` and `quit`, plus a JSON save and a simple map display. The `start` command does not render the route map the PRD asks for.

import globalvar

    from rider import Rider
    cid = globalvar.get_value("rider_id_counter", 0) + 1

    globalvar.set_value("funds", globalvar.get_value("funds", 0) - 300)

"""Rider model."""
import math
import globalvar


def _route(x, y, tx, ty):
    """Waypoints along streets from (x, y) to (tx, ty)."""
    pts = []
    cx, cy = x, y
    on_v = cx % 100 == 0
    on_h = cy % 100 == 0
    if not on_v and not on_h:
        # go to nearest horizontal street
        ny = round(cy / 100) * 100
        ny = min(max(ny, 0), 800)
        pts.append((cx, ny))
        cy = ny
    if cy % 100 == 0 and cx % 100 != 0:
        # on horizontal street: move horizontally to target column street
        nx = (tx // 100) * 100 if tx % 100 else tx
        # choose the street next to target
        nx = round(tx / 100) * 100
        pts.append((nx, cy))
        cx = nx
    # now on vertical street: move vertically to target row
    pts.append((cx, ty))
    cy = ty
    pts.append((tx, ty))
    return pts


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.current = None
        self.path = []
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0),
                         "x": self.x, "y": self.y})

    def _set_target(self, tx, ty):
        self.path = _route(self.x, self.y, tx, ty)

    def _advance(self):
        """Move up to speed units along path. Return True if path done."""
        budget = self.speed
        while self.path and budget > 0:
            px, py = self.path[0]
            d = math.hypot(px - self.x, py - self.y)
            if d <= budget:
                self.x, self.y = px, py
                budget -= d
                self.path.pop(0)
            else:
                self.x += round((px - self.x) / d * budget)
                self.y += round((py - self.y) / d * budget)
                budget = 0
        return not self.path

    def move(self):
        """Advance one tick."""
        if self.state == 0:
            if not self.orders:
                return
            self.current = self.orders[0]
            self.state = 1
            self._log("accepted_order", self.current.id)
            self._set_target(*self.current.res_pos)
        if self.state == 1:
            if self._advance():
                self.state = 2
                self._log("arrived_at_merchant", self.current.id)
                self._set_target(*self.current.cus_pos)
            return
        if self.state == 2:
            if self._advance():
                o = self.current
                self._log("delivered_to_customer", o.id)
                o.status = "completed"
                o.finish_time = globalvar.get_value("time", 0)
                if o in self.orders:
                    self.orders.remove(o)
                self.current = None
                self.state = 0

"""Order model."""
import globalvar

_counter = [0]


def to_center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(int(v), 799))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        _counter[0] += 1
        self.id = _counter[0]
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0)
        self.finish_time = None

    def assign_to(self, rider_id):
        """Record assignment."""
        self.rider_id = rider_id
        self.status = "assigned"

    @property
    def distance(self):
        return abs(self.res_pos[0] - self.cus_pos[0]) + \
            abs(self.res_pos[1] - self.cus_pos[1])

"""Order assignment."""
import math
import globalvar


def find_best_rider_for_order(order):
    """Return the nearest rider (recruiting if none exists), or None."""
    riders = globalvar.get_value("riders", [])
    if not riders and globalvar.get_value("funds", 0) >= 300:
        import system
        system.create_rider()
        riders = globalvar.get_value("riders", [])
    if not riders:
        return None
    rx, ry = order.res_pos
    return min(riders, key=lambda r: (math.hypot(r.x - rx, r.y - ry),
                                      len(r.orders)))

"""Command-line interface."""
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import globalvar
import system
from order_model import Order
from order import find_best_rider_for_order


def show_map():
    grid = [["." for _ in range(8)] for _ in range(8)]
    for o in globalvar.get_value("orders"):
        if o.status != "completed":
            grid[o.res_pos[1] // 100][o.res_pos[0] // 100] = "S"
            grid[o.cus_pos[1] // 100][o.cus_pos[0] // 100] = "C"
    for r in globalvar.get_value("riders"):
        grid[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
    for row in reversed(grid):
        print(" ".join(row))


def cmd_start(args):
    if len(args) != 4:
        print("Usage: start x1 y1 x2 y2")
        return
    try:
        v = [int(a) for a in args]
    except ValueError:
        print("Coordinates must be integers.")
        return
    if any(n < 0 or n > 800 for n in v):
        print("Coordinates must be within 0-800.")
        return
    o = Order(v[:2], v[2:])
    pending = [x for x in globalvar.get_value("orders") if x.status == "pending"]
    idle = [r for r in globalvar.get_value("riders") if r.state == 0 and not r.orders]
    added = False
    if globalvar.get_value("funds") >= 300 and len(pending) + 1 > len(idle):
        system.create_rider()
        added = True
    rider = find_best_rider_for_order(o)
    globalvar.get_value("orders").append(o)
    if rider is None:
        print("No rider available.")
        return
    o.assign_to(rider.rider_id)
    rider.orders.append(o)
    print("Restaurant grid:", o.res_pos, "Customer grid:", o.cus_pos)
    print("New rider added:", added, "Assigned rider:", rider.rider_id)
    print("Speed:", rider.speed, "Est. ticks:",
          (abs(rider.x - o.res_pos[0]) + abs(rider.y - o.res_pos[1]) + o.distance)
          // rider.speed)


def cmd_status():
    print("Funds:", globalvar.get_value("funds"))
    names = ["idle", "picking up", "delivering"]
    for r in globalvar.get_value("riders"):
        print("Rider %d (%d,%d) %s orders=%s" % (
            r.rider_id, r.x, r.y, names[r.state], [o.id for o in r.orders]))
    show_map()


def cmd_orders():
    for o in globalvar.get_value("orders"):
        t = None if o.finish_time is None else o.finish_time - o.start_time
        print("Order %d start=%d res=%s cus=%s dist=%d time=%s rider=%s %s" % (
            o.id, o.start_time, o.res_pos, o.cus_pos, o.distance, t,
            o.rider_id, o.status))


def cmd_riders():
    for r in globalvar.get_value("riders"):
        for e in r.log:
            print("Rider %d t=%d %s order=%s at (%d,%d)" % (
                r.rider_id, e["time"], e["event"], e["order_id"], e["x"], e["y"]))


def save():
    data = {"funds": globalvar.get_value("funds"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y, "state": r.state}
                       for r in globalvar.get_value("riders")],
            "orders": [{"id": o.id, "res": o.res_pos, "cus": o.cus_pos,
                        "status": o.status}
                       for o in globalvar.get_value("orders")]}
    try:
        with open("state.json", "w") as f:
            json.dump(data, f)
    except OSError:
        pass


def main():
    system.init_system()
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        parts = line.split()
        if not parts:
            system.tick()
            continue
        c = parts[0].lower()
        try:
            if c == "start":
                cmd_start(parts[1:])
            elif c == "status":
                cmd_status()
            elif c == "orders":
                cmd_orders()
            elif c == "riders":
                cmd_riders()
            elif c in ("quit", "exit"):
                globalvar.set_value("running", False)
            else:
                print("Unknown command.")
        except Exception as e:
            print("Error:", e)
        system.tick()
    save()


if __name__ == "__main__":
    main()

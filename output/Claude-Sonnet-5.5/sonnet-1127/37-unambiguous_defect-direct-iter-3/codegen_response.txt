import re
from typing import Dict, List, Tuple

from lottery_system.models import User

DEFAULT_LINES = [
    "Zhang Wei,Technology Department",
    "Li Na,Technology Department",
    "Wang Fang,Technology Department",
    "Liu Yang,Product Department",
    "Chen Jing,Product Department",
    "Zhao Lei,Product Department",
    "Sun Li,Cheng Ru Group",
    "Zhou Min,Cheng Ru Group",
    "Wu Gang,Cheng Ru Group",
    "Zheng Hao,Marketing Department",
    "Huang Ting,Marketing Department",
    "Xu Qiang,Operations Department",
    "Gao Yan,Operations Department",
    "Li Bin,Operations Department",
    "Chen Qixian,Technology Department",
    "Jin Duo,Product Department",
    "Ma Jiaqi,Marketing Department",
]


class ListManager:
    def _parse_lines(self, lines) -> Tuple[List[User], List[str]]:
        users: List[User] = []
        errors: List[str] = []
        for i, raw in enumerate(lines, 1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            parts = [p.strip() for p in re.split(r"[,\t，]", line)]
            parts = [p for p in parts if p != ""] if len(parts) > 2 else parts
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error - '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def load_default_list(self) -> Tuple[List[User], List[str]]:
        return self._parse_lines(DEFAULT_LINES)

    def _load_from_path(self, path) -> Tuple[List[User], List[str]]:
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except FileNotFoundError:
            return [], [f"File not found: {path}"]
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Cannot read file {path}: {e}"]
        return self._parse_lines(lines)

    def load_custom_list(self, path) -> Tuple[List[User], List[str]]:
        return self._load_from_path(path)

    def validate_users(self, users) -> Tuple[List[User], List[User]]:
        seen = set()
        valid: List[User] = []
        dups: List[User] = []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users) -> Dict[str, int]:
        info: Dict[str, int] = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from typing import List

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes: List[Prize] = []

    def load_default_prizes(self) -> List[str]:
        errors: List[str] = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, q in defaults:
            try:
                self.add_prize(name, q)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        try:
            q = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Quantity must be an integer")
        if q <= 0:
            raise ValueError("Quantity must be positive")
        prize = Prize(str(name).strip(), q, list(avoid_groups or []),
                      associated_drawer or None)
        self.prizes.append(prize)
        return prize

    def get_prizes(self) -> List[Prize]:
        return list(self.prizes)

    def reorder_prizes(self, order) -> bool:
        n = len(self.prizes)
        try:
            order = [int(x) for x in order]
        except (TypeError, ValueError):
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random
from typing import List, Optional

from lottery_system.models import Prize, User

CHENG_RU = "cheng ru"
TRACKED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"
RELAXED_DRAWER = "Li Bin"


def _is_cheng_ru(user: User) -> bool:
    return CHENG_RU in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users: List[User] = list(users)
        self.winners: List[User] = []
        self.last_two_draw_results: List[bool] = []

    def _get_eligible_pool(self, prize: Prize) -> List[User]:
        won = set(self.winners)
        pool = [u for u in self.users if u not in won]
        drawer = prize.associated_drawer
        if drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if _is_cheng_ru(u)]
        if drawer == RELAXED_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize: Prize) -> Optional[User]:
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def finish_prize(self, prize: Prize, round_winners) -> None:
        if prize.associated_drawer in TRACKED_DRAWERS:
            self.last_two_draw_results.append(
                any(_is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime
import os
from typing import Dict, List


class Winner:
    def __init__(self, user, prize_name, order=0):
        self.user = user
        self.prize_name = prize_name
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners: List[Winner] = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        w = Winner(user, name, len(self.winners))
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes) -> Dict[str, List[Winner]]:
        result: Dict[str, List[Winner]] = {}
        for p in prizes:
            name = getattr(p, "name", p)
            result[name] = [w for w in self.winners if w.prize_name == name]
        return result

    def export_to_csv(self, prizes):
        date = datetime.date.today().strftime("%Y%m%d")
        path = os.path.join(os.getcwd(), f"awards_result_{date}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for name, ws in grouped.items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("1) default list  2) custom txt list: ").strip()
        if c == "1":
            users, errors = lm.load_default_list()
        elif c == "2":
            users, errors = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errors:
            print("Warning:", e)
        if errors and input("Errors found. Remove invalid data and continue? (y/n): ").lower() != "y":
            if c == "2":
                print("Upload terminated.")
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate removed: {d.name} ({d.department})")
        if valid:
            return valid
        print("No valid users.")


def configure_prizes(pm, groups):
    if input("Load preset prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        name = input("Name: ").strip()
        qty = input("Quantity: ").strip()
        avoid = [g.strip() for g in input(
            f"Avoid groups (comma separated, from {list(groups)}): ").split(",") if g.strip()]
        drawer = input("Associated drawer (blank for none): ").strip() or None
        try:
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Error:", e)
    for p in pm.get_prizes():
        if p.avoid_groups == [] and p.name.startswith("Third") and not p.associated_drawer:
            pass
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    o = input("Enter new order as numbers (blank to keep): ").split()
    if o and not pm.reorder_prizes(o):
        print("Invalid order, keeping original.")


def run_round(lot, res, prize):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | avoid {prize.avoid_groups or 'none'}"
          f" | drawer {prize.associated_drawer or 'none'} ===")
    got = []
    n = 0
    while n < prize.quantity:
        cmd = input("Enter to draw (pause / terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
        print(f"Currently drawing the {n + 1}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible participants left.")
            break
        res.add_winner(u, prize)
        got.append(u)
        n += 1
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.finish_prize(prize, got)
    print("Round results:")
    for i, u in enumerate(got, 1):
        print(f"  {i}. {u.name} ({u.department})")


def main():
    lm = ListManager()
    users = load_users(lm)
    for d, c in lm.get_group_info(users).items():
        print(f"  {d}: {c}")
    pm = PrizeManager()
    configure_prizes(pm, lm.get_group_info(users))
    lot = LotteryManager(users)
    res = ResultManager()
    prizes = pm.get_prizes()
    for p in prizes:
        run_round(lot, res, p)
    ordered = list(reversed(prizes))
    print("\n=== Final winners ===")
    for name, ws in res.get_winners_by_prize(ordered).items():
        for w in ws:
            print(f"{name}: {w.user.name} ({w.user.department})")
    ok, msg = res.export_to_csv(ordered)
    print(msg)


if __name__ == "__main__":
    main()

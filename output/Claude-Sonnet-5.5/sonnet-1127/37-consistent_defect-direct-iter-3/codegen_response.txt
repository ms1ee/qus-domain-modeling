import os
from collections import OrderedDict

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Cheng Ru Group"),
    ("Zheng Shi", "Cheng Ru Group"),
    ("Chen Yi", "Cheng Ru Group"),
    ("Li Bin", "Operations Department"),
    ("Qian Er", "Operations Department"),
    ("Feng San", "Marketing Department"),
    ("Wei Si", "Marketing Department"),
    ("Jiang Wu", "Marketing Department"),
    ("Han Liu", "Human Resources"),
    ("Yang Qi", "Human Resources"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        if not os.path.isfile(path):
            return [], [f"File not found: {path}"]
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Cannot read file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = line.replace(";", ",").replace("；", ",").replace("，", ",").split(",")
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Line {i}: Format Error - '{line.strip()}'")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, ["Product Department"], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, ["Technology Department"], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, list(a), d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        name = (name or "").strip()
        if not name:
            raise ValueError("Prize name cannot be empty")
        try:
            quantity = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Quantity must be an integer")
        if quantity <= 0:
            raise ValueError("Quantity must be positive")
        drawer = associated_drawer.strip() if associated_drawer else None
        prize = Prize(name, quantity, list(avoid_groups or []), drawer or None)
        self._prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random

CHENG_RU_GROUP = "Cheng Ru Group"
SPECIAL_DRAWER = "Li Bin"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"


def _is_cheng_ru(dept):
    return dept.strip().lower() in ("cheng ru group", "cheng ru")


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _won_names(self):
        return {u.name for u in self.winners}

    def _fallback_active(self, prize):
        r = self.last_two_draw_results
        return (prize.associated_drawer == PROTECTED_DRAWER
                and len(r) >= 2 and not r[-1] and not r[-2])

    def _get_eligible_pool(self, prize):
        won = self._won_names()
        pool = [u for u in self.users if u.name not in won]
        if self._fallback_active(prize):
            return [u for u in pool if _is_cheng_ru(u.department)]
        if prize.associated_drawer == SPECIAL_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool and not self._fallback_active(prize) \
                and prize.associated_drawer != SPECIAL_DRAWER:
            # avoidance-group members are drawn after the others
            won = self._won_names()
            pool = [u for u in self.users if u.name not in won]
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def finish_round(self, prize, winners):
        """Record whether a watched drawer's round contained Cheng Ru members."""
        if prize.associated_drawer in WATCHED_DRAWERS:
            self.last_two_draw_results.append(
                any(_is_cheng_ru(u.department) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, win_time=None):
        self.user = user
        self.prize_name = prize_name
        self.win_time = win_time or datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for name, ws in grouped.items():
                    for w in sorted(ws, key=lambda x: x.win_time):
                        f.write(f"{w.user.name},{w.user.department},{name},"
                                f"{w.win_time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def _rank(p):
    n = p.name.lower()
    for i, k in enumerate(("first", "second", "third")):
        if k in n:
            return i
    return 99


def load_users(lm):
    choice = input("Load list: 1) default  2) custom txt [1]: ").strip()
    if choice == "2":
        path = input("Path: ").strip()
        users, errors = lm.load_custom_list(path)
    else:
        users, errors = lm.load_default_list()
    for e in errors:
        print("Warning:", e)
    if errors and choice == "2":
        if input("Errors found. Continue with valid data? (y/n): ").strip().lower() != "y":
            return None
    users, dups = lm.validate_users(users)
    for d in dups:
        print(f"Duplicate removed: {d.name} ({d.department})")
    print("Groups:")
    for g, c in lm.get_group_info(users).items():
        print(f"  {g}: {c}")
    return users


def configure_prizes(pm, groups):
    if input("Load default prizes? (y/n) [y]: ").strip().lower() != "n":
        for e in pm.load_default_prizes():
            print("Warning:", e)
    while input("Add custom prize? (y/n) [n]: ").strip().lower() == "y":
        name = input("Name: ")
        qty = input("Quantity: ")
        avoid = [a.strip() for a in input("Avoid groups (comma separated): ").split(",") if a.strip()]
        drawer = input("Associated drawer: ").strip() or None
        try:
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Error:", e)
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("Enter new order as numbers (blank to keep): ").strip()
    if s:
        try:
            ok = pm.reorder_prizes([int(x) for x in s.replace(",", " ").split()])
        except ValueError:
            ok = False
        print("Reordered." if ok else "Invalid order, keeping original.")


def run_round(prize, lot, res):
    print(f"\n=== {prize.name} | quantity {prize.quantity} ===")
    print(f"Rules: avoid groups={prize.avoid_groups or 'none'}, drawer={prize.associated_drawer or 'n/a'}")
    won = []
    while len(won) < prize.quantity:
        cmd = input("Enter to draw (pause / terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        print(f"Currently drawing the {len(won) + 1}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        res.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.finish_round(prize, won)
    print("Round results:")
    for i, u in enumerate(won, 1):
        print(f"  {i}. {u.name} ({u.department})")


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No users loaded.")
        return
    pm = PrizeManager()
    configure_prizes(pm, lm.get_group_info(users))
    prizes = pm.get_prizes()
    lot, res = LotteryManager(users), ResultManager()
    for p in prizes:
        run_round(p, lot, res)
    print("\n=== Final winners ===")
    ordered = sorted(prizes, key=_rank)
    for name, ws in res.get_winners_by_prize(ordered).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {name}")
    ok, msg = res.export_to_csv(ordered)
    print(msg)


if __name__ == "__main__":
    main()

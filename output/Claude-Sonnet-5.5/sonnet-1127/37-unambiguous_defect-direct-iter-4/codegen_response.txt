from lottery_system.models import User

DEFAULT_USERS = [
    ("Zhang Wei", "Technology Department"),
    ("Wang Fang", "Technology Department"),

    ("Chen Qixian", "Product Department"),
    ("Zhao Min", "Product Department"),
    ("Sun Li", "Product Department"),
    ("Li Bin", "Marketing Department"),
    ("Zhou Jie", "Marketing Department"),
    ("Wu Qiang", "Marketing Department"),
    ("Jin Duo", "Operations Department"),
    ("Ma Jiaqi", "Operations Department"),
    ("Xu Lei", "Operations Department"),
    ("Cheng Ru", "Cheng Ru Group"),
    ("Gao Peng", "Cheng Ru Group"),
    ("Lin Na", "Cheng Ru Group"),
    ("He Ping", "Finance Department"),
    ("Luo Yi", "Finance Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_USERS], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, raw in enumerate(lines, 1):
            line = raw.strip().lstrip("\ufeff").strip()
            if not line or line.startswith("#"):
                continue
            line = line.replace("，", ",")
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error: '{line}' (expected 'name,department')")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, q in [("Third Prize - Movie Tickets", 3),
                        ("Second Prize - Shopping Cards", 2),
                        ("First Prize - 1 Day Annual Leave", 1)]:
            try:
                self.add_prize(name, q)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        prize = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(prize)
        return prize

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"
RELAXED_DRAWER = "Li Bin"


def is_cheng_ru(department):
    d = str(department).lower().replace(" ", "")
    return d.startswith("chengru") or "程如" in d


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {u.name for u in self.winners}
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == RELAXED_DRAWER:
            return pool
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if is_cheng_ru(u.department)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def record_round_result(self, prize, winners):
        if prize.associated_drawer in CHENG_RU_DRAWERS:
            self.last_two_draw_results.append(
                any(is_cheng_ru(w.department) for w in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize, order=0):
        self.user = user
        self.prize = prize
        self.prize_name = prize.name if hasattr(prize, "name") else str(prize)
        self.order = order


def _rank(name):
    n = name.lower()
    for i, k in enumerate(("first", "second", "third")):
        if k in n:
            return i
    return 3


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize, len(self.winners))
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        ordered = sorted(prizes, key=lambda p: _rank(p.name))
        result = {}
        for p in ordered:
            result[p.name] = sorted(
                [w for w in self.winners if w.prize_name == p.name],
                key=lambda w: w.order)
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    c = input("1) default list  2) custom txt file [1]: ").strip()
    if c == "2":
        users, errors = lm.load_custom_list(input("Path: ").strip())
    else:
        users, errors = lm.load_default_list()
    for e in errors:
        print("  !", e)
    if errors and input("Terminate upload? (y/N): ").strip().lower() == "y":
        return []
    valid, dups = lm.validate_users(users)
    for d in dups:
        print(f"  ! Duplicate removed: {d.name}")
    print("Groups:")
    for g, n in lm.get_group_info(valid).items():
        print(f"  {g}: {n}")
    return valid


def config_prizes(pm, users):
    if input("Load default prizes? (Y/n): ").strip().lower() != "n":
        pm.load_default_prizes()
    while input("Add custom prize? (y/N): ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            q = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma-separated): ").split(",") if g.strip()]
            drawer = input("Associated drawer: ").strip() or None
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print("  !", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    s = input("New order (e.g. 3 1 2), blank to keep: ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("  ! Invalid order, kept original")
        except ValueError:
            print("  ! Invalid order, kept original")


def run_round(lot, prize, res):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | avoid: "
          f"{', '.join(prize.avoid_groups) or 'none'} | drawer: {prize.associated_drawer or '-'}")
    got = []
    while len(got) < prize.quantity:
        cmd = input("[Enter]=draw, pause, continue, terminate: ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        print(f"Currently drawing the {len(got) + 1}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        res.add_winner(u, prize)
        got.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.record_round_result(prize, got)


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No users loaded.")
        return
    pm = PrizeManager()
    config_prizes(pm, users)
    lot = LotteryManager(users)
    res = ResultManager()
    for p in pm.get_prizes():
        run_round(lot, p, res)
    print("\n=== Final Results ===")
    for name, ws in res.get_winners_by_prize(pm.get_prizes()).items():
        for w in ws:
            print(f"{name}: {w.user.name} ({w.user.department})")
    print(res.export_to_csv(pm.get_prizes())[1])


if __name__ == "__main__":
    main()

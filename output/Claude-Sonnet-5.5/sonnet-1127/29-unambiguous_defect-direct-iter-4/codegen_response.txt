"""Global state storage."""

_state = {}


def _init():
    """Reset all global variables to initial values."""
    _state.clear()
    _state.update({
        "funds": 10000,
        "riders": [],
        "rider_id_counter": 0,
        "running": True,
        "time": 0,
        "orders": [],
    })


def set_value(key, value):
    """Set a global value."""
    if not _state:
        _init()
    _state[key] = value


def get_value(key, default=None):
    """Get a global value."""
    if not _state:
        _init()
    return _state.get(key, default)

"""Grid-based path planning."""

CELL = 100
SIZE = 800


def to_center(v):
    """Convert coordinate to grid cell center."""
    v = max(0, min(SIZE - 1, int(v)))
    return (v // CELL) * CELL + CELL // 2


def center_pos(x, y):
    """Convert a point to its cell center."""
    return (to_center(x), to_center(y))


class PathPlanner:
    """Right-angle path planner."""

    @staticmethod
    def route(start, *targets):
        """Waypoints through targets; moves vertically first then horizontally
        alternately chosen so that the path follows streets where possible."""
        pts = [tuple(start)]
        cur = tuple(start)
        for t in targets:
            t = tuple(t)
            if cur[0] != t[0] and cur[1] != t[1]:
                pts.append((cur[0], t[1]) if cur[0] % CELL == 0 else (t[0], cur[1]))
            if pts[-1] != t:
                pts.append(t)
            cur = t
        return pts

    @staticmethod
    def step(pos, target, speed):
        """Advance pos toward target by speed units (x axis first, or
        y first when already aligned on a vertical street)."""
        x, y = pos
        tx, ty = target
        left = speed
        order = ("x", "y") if x % CELL != 0 or x == tx else ("y", "x")
        if x % CELL == 0 and x != tx:
            order = ("y", "x")
        for axis in order:
            if left <= 0:
                break
            if axis == "x":
                d = min(left, abs(tx - x))
                x += d if tx > x else -d
            else:
                d = min(left, abs(ty - y))
                y += d if ty > y else -d
            left -= d
        return (x, y)

    @staticmethod
    def distance(a, b):
        """Manhattan distance."""
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

"""Rider model."""
import globalvar
from path_planner import PathPlanner

IDLE, PICKING, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 100
        self.y = 50
        self.speed = 50
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    @property
    def pos(self):
        return (self.x, self.y)

    def _log(self, event, order_id):
        self.log.append({
            "time": globalvar.get_value("time", 0),
            "event": event,
            "order_id": order_id,
            "x": self.x,
            "y": self.y,
        })

    def _go(self, target):
        self.x, self.y = PathPlanner.step(self.pos, target, self.speed)
        return self.pos == tuple(target)

    def move(self):
        """Advance one tick."""
        if self.state == IDLE:
            if not self.orders:
                return
            order = self.orders[0]
            self.state = PICKING
            if hasattr(order, "status"):
                order.status = "picking_up"
            self._log("accepted_order", order.id)
        if not self.orders:
            self.state = IDLE
            return
        order = self.orders[0]
        if self.state == PICKING:
            if self._go(order.res_pos):
                self.state = DELIVERING
                order.status = "delivering"
                self._log("arrived_at_merchant", order.id)
        elif self.state == DELIVERING:
            if self._go(order.cus_pos):
                now = globalvar.get_value("time", 0)
                order.status = "completed"
                order.complete_time = now
                order.duration = now - order.start_time
                self._log("delivered_to_customer", order.id)
                self.orders.pop(0)
                self.state = IDLE

"""Order model."""
import itertools
import globalvar
from path_planner import center_pos, PathPlanner

_counter = itertools.count(1)


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = center_pos(*res_pos)
        self.cus_pos = center_pos(*cus_pos)
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("time", 0)
        self.complete_time = None
        self.duration = None
        self.distance = PathPlanner.distance(self.res_pos, self.cus_pos)

    def assign_to(self, rider_id):
        """Mark order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "complete_time": self.complete_time,
                "distance": self.distance}

"""System management."""
import json
import globalvar
from rider import Rider

RIDER_COST = 100


def init_system():
    """Initialize global state."""
    globalvar._init()


def recruit_rider():
    """Create a rider if funds allow; return it or None."""
    funds = globalvar.get_value("funds")
    if funds < RIDER_COST:
        return None
    cid = globalvar.get_value("rider_id_counter") + 1
    globalvar.set_value("rider_id_counter", cid)
    globalvar.set_value("funds", funds - RIDER_COST)
    r = Rider(cid)
    globalvar.get_value("riders").append(r)
    return r


class SystemManager:
    """Manage funds, riders and ticks."""

    def __init__(self):
        init_system()

    def tick(self):
        globalvar.set_value("time", globalvar.get_value("time") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

    def save(self, path="config.json"):
        data = {
            "funds": globalvar.get_value("funds"),
            "time": globalvar.get_value("time"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "speed": r.speed, "state": r.state,
                        "log": r.log} for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
        }
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Order management and assignment."""
import globalvar
import system
from path_planner import PathPlanner
from order_model import Order


def find_best_rider_for_order(order):
    """Pick the best rider: idle & nearest first, load balanced."""
    riders = globalvar.get_value("riders")
    if not riders:
        system.recruit_rider()
        riders = globalvar.get_value("riders")
        if not riders:
            return None

    def key(r):
        return (len(r.orders), PathPlanner.distance(r.pos, order.res_pos),
                r.rider_id)
    return min(riders, key=key)


class OrderManager:
    """Create and assign orders."""

    def create_order(self, coords):
        """Returns (order, rider, new_rider_added)."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        riders = globalvar.get_value("riders")
        added = False
        if not any(r.state == 0 and not r.orders for r in riders):
            if system.recruit_rider():
                added = True
        rider = find_best_rider_for_order(order)
        if rider is None:
            return order, None, added
        rider.orders.append(order)
        order.assign_to(rider.rider_id)
        globalvar.get_value("orders").append(order)
        return order, rider, added

"""Display."""
import globalvar
from path_planner import PathPlanner

STATE = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Status and map display."""

    def map(self, route=None, riders=True):
        n = 17
        g = [["+" if (i % 2 == 0 and j % 2 == 0) else
              ("." if (i % 2 == 0 or j % 2 == 0) else " ")
              for j in range(n)] for i in range(n)]
        if route:
            for a, b in zip(route, route[1:]):
                for t in range(0, 101):
                    x = a[0] + (b[0] - a[0]) * t // 100
                    y = a[1] + (b[1] - a[1]) * t // 100
                    g[min(y // 50, 16)][min(x // 50, 16)] = "*"
            g[route[-1][1] // 50][route[-1][0] // 50] = "C"
            if len(route) > 2:
                pass
        for o in globalvar.get_value("orders"):
            if o.status != "completed":
                g[o.res_pos[1] // 50][o.res_pos[0] // 50] = "R"
                g[o.cus_pos[1] // 50][o.cus_pos[0] // 50] = "C"
        if riders:
            for r in globalvar.get_value("riders"):
                g[min(r.y // 50, 16)][min(r.x // 50, 16)] = "B"
        return "\n".join(" ".join(row) for row in g)

    def status(self):
        rs = globalvar.get_value("riders")
        lines = ["Time: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("time"), globalvar.get_value("funds"),
            len(rs))]
        for r in rs:
            lines.append("Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATE[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        os_ = globalvar.get_value("orders")
        if not os_:
            return "No orders."
        lines = []
        for o in os_:
            lines.append(
                "Order %d start=%d res=%s cus=%s dist=%d time=%s rider=%s [%s]"
                % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                   o.duration if o.duration is not None else "-",
                   o.rider_id, o.status))
        return "\n".join(lines)

    def riders_log(self):
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("t=%s rider=%d %s order=%s at (%d,%d)" % (
                    e["time"], r.rider_id, e["event"], e["order_id"],
                    e["x"], e["y"]))
        return "\n".join(lines) or "No records."

"""Command parsing and loop."""
import globalvar
from display import DisplayManager
from order import OrderManager
from path_planner import PathPlanner
from system import SystemManager

try:
    import readline  # noqa: F401  (command history)
except ImportError:
    pass

ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
           "n": "start", "h": "help"}


def parse_coords(text):
    """Parse four integers in 0-800; raises ValueError."""
    parts = text.replace(",", " ").replace("[", " ").replace("]", " ").split()
    if len(parts) != 4:
        raise ValueError("Enter exactly four integers.")
    vals = [int(p) for p in parts]
    if any(v < 0 or v > 800 for v in vals):
        raise ValueError("Coordinates must be within 0-800.")
    return vals


class InputManager:
    """Interactive command loop."""

    def __init__(self):
        self.sm = SystemManager()
        self.disp = DisplayManager()
        self.om = OrderManager()

    def handle(self, cmd):
        cmd = ALIASES.get(cmd, cmd)
        if cmd in ("quit", "exit"):
            globalvar.set_value("running", False)
        elif cmd == "status":
            print(self.disp.status())
            print(self.disp.map())
        elif cmd == "orders":
            print(self.disp.orders())
        elif cmd == "riders":
            print(self.disp.riders_log())
        elif cmd == "start":
            try:
                coords = parse_coords(input("Enter rx ry cx cy: "))
            except ValueError as e:
                print("Invalid input: %s" % e)
                return
            order, rider, added = self.om.create_order(coords)
            if rider is None:
                print("No rider available.")
                return
            print("Restaurant grid pos: %s, customer grid pos: %s" % (
                order.res_pos, order.cus_pos))
            print("New rider added: %s" % added)
            route = PathPlanner.route(rider.pos, order.res_pos, order.cus_pos)
            dist = sum(PathPlanner.distance(a, b)
                       for a, b in zip(route, route[1:]))
            print("Rider %d speed %d, estimated time %d ticks" % (
                rider.rider_id, rider.speed, -(-dist // rider.speed)))
            print(self.disp.map(route))
        elif cmd == "help":
            print("start(n) status(s) orders(o) riders(r) quit(q)")
        elif cmd:
            print("Unknown command '%s'. Type help." % cmd)

    def run(self):
        print("Food Delivery Dispatch. Type help.")
        while globalvar.get_value("running"):
            try:
                cmd = input("> ").strip().lower()
            except (EOFError, KeyboardInterrupt):
                break
            try:
                self.handle(cmd)
                self.sm.tick()
            except Exception as e:  # recover gracefully
                print("Error: %s" % e)
        self.sm.save()

"""Graph data structures."""
import heapq
import itertools


class Edge:
    """Undirected edge between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return f"Edge({self.source}, {self.dest}, {self.distance}, {self.line!r})"


class MinPQ:
    """Heap based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        priority, _, item = heapq.heappop(self._heap)
        return priority, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


class Graph:
    """Weighted undirected graph (adjacency lists keyed by station number)."""

    def __init__(self):
        self.graph = {}
        self.station_to_number = {}
        self.number_to_station = {}
        self.station_lines = {}

    def add_station(self, name, number=None):
        if name in self.station_to_number:
            return self.station_to_number[name]
        if number is None:
            number = len(self.station_to_number)
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.graph.setdefault(number, [])
        return number

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.graph.setdefault(edge.source, []).append(edge)
        self.graph.setdefault(edge.dest, []).append(edge.reversed())

    def adj(self, node):
        return self.graph.get(node, [])

    def set_distance(self, a, b, distance, line=None):
        """Update distance of edges between a and b; returns True if found."""
        found = False
        for u, v in ((a, b), (b, a)):
            for e in self.graph.get(u, []):
                if e.dest == v and (line is None or e.line == line):
                    e.distance = distance
                    e.weight = distance
                    found = True
        return found

    def is_transfer_station(self, name):
        return len(self.station_lines.get(name, [])) > 1

    def V(self):
        return len(self.graph)

    def E(self):
        return sum(len(v) for v in self.graph.values()) // 2

"""Dijkstra shortest path with transfer awareness."""
from models import MinPQ

MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = 3000  # meters-equivalent per transfer (comprehensive)
PREFERRED_FACTOR = 0.9


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._transfers = {}
        self._best_state = {}
        self._info = {}  # state -> (cost, dist, transfers, parent, edge)
        self._run()

    def _step_cost(self, edge, transferred):
        mode = self.config.mode
        d = edge.distance
        if edge.line in self.config.preferred_lines:
            d_eff = d * PREFERRED_FACTOR
        else:
            d_eff = d
        t = 1 if transferred else 0
        if mode == "least_transfer":
            return t * 1e9 + d_eff
        if mode == "comprehensive":
            return d_eff + t * TRANSFER_PENALTY
        return d_eff

    def _run(self):
        maxt = self.config.max_transfers
        bounded = maxt is not None

        def key(node, line, t):
            return (node, line, t) if bounded else (node, line)

        start = (self.source, None, 0) if bounded else (self.source, None)
        self._info[start] = (0.0, 0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            _, dist, t, _, _ = self._info[st]
            node, line = st[0], st[1]
            for e in self.graph.adj(node):
                transferred = line is not None and e.line != line
                nt = t + (1 if transferred else 0)
                if bounded and nt > maxt:
                    continue
                nst = key(e.dest, e.line, nt)
                if nst in done:
                    continue
                ncost = cost + self._step_cost(e, transferred)
                old = self._info.get(nst)
                if old is None or ncost < old[0]:
                    self._info[nst] = (ncost, dist + e.distance, nt, st, e)
                    pq.push(ncost, nst)
        for st, (cost, dist, t, _, _) in self._info.items():
            node = st[0]
            cur = self._best_state.get(node)
            if cur is None or cost < self._info[cur][0]:
                self._best_state[node] = st
        for node, st in self._best_state.items():
            self.dist_to[node] = self._info[st][1]
            self._transfers[node] = self._info[st][2]

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        st = self._best_state.get(target)
        if st is None:
            return [], []
        nodes, edges = [], []
        while st is not None:
            _, _, _, parent, e = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = parent
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        return self._transfers.get(target, 0)

"""Fare and time calculation."""
import math


def Sum_money(distance) -> int:
    """Fare in yuan for a distance in meters (left-closed, right-open tiers)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count) -> float:
    """Minutes: 1.5 per km plus 5 per transfer."""
    return distance / 1000.0 * 1.5 + 5.0 * transfer_count

"""Graph building, integrity check and data update helpers."""
from models import Graph, Edge


def build_graph(data):
    g = Graph()
    for s in data.get("stations", []):
        g.add_station(s["name"])
        g.set_station_lines(s["name"], s.get("lines", []))
    for c in data.get("connections", []):
        a = g.add_station(c["source"])
        b = g.add_station(c["dest"])
        for n in (c["source"], c["dest"]):
            lines = g.station_lines.setdefault(n, [])
            if c["line"] not in lines:
                lines.append(c["line"])
        g.add_edge(Edge(a, b, c["distance"], c["line"]))
    return g


def check_data(data):
    """Return (ok, list of messages)."""
    msgs = []
    stations = data.get("stations", [])
    conns = data.get("connections", [])
    names = {s["name"] for s in stations}
    if not stations:
        msgs.append("No stations found")
    if len(names) != len(stations):
        msgs.append("Duplicate station names found")
    lines = {c["line"] for c in conns}
    if not lines:
        msgs.append("No lines found")
    connected = set()
    for c in conns:
        if c["source"] not in names or c["dest"] not in names:
            msgs.append(f"Connection references unknown station: {c['source']} - {c['dest']}")
        if c["source"] == c["dest"]:
            msgs.append(f"Self connection: {c['source']}")
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
            msgs.append(f"Suspicious distance {d}: {c['source']} - {c['dest']} ({c['line']})")
        connected.add(c["source"])
        connected.add(c["dest"])
    for n in names - connected:
        msgs.append(f"Station without connections: {n}")
    return (not msgs), msgs


def add_line(data, line_name, station_names, distances):
    """Add a new line: consecutive stations with given distances."""
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances must have len(stations)-1 entries")
    existing = {s["name"]: s for s in data["stations"]}
    for n in station_names:
        if n in existing:
            if line_name not in existing[n]["lines"]:
                existing[n]["lines"].append(line_name)
        else:
            data["stations"].append({"name": n, "lines": [line_name]})
    for i, d in enumerate(distances):
        data["connections"].append({"source": station_names[i], "dest": station_names[i + 1],
                                    "distance": int(d), "line": line_name})


def update_station(data, old_name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == old_name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == old_name:
                        c["source"] = new_name
                    if c["dest"] == old_name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    found = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
            c["distance"] = int(distance)
            found = True
    return found

"""CLI for the Beijing Subway fare calculator."""
import argparse
import json
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import build_graph, check_data
from pathfinding import PathfindingConfig, ShortestPath
from fare_calculator import Sum_money, calculate_estimated_time

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def format_distance(d):
    if d < 1000:
        return f"{d} m"
    return f"{d / 1000:.2f} km"


def parse_query(text):
    text = text.replace("；", ";")
    parts = [p.strip() for p in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERR)
    return parts[0], parts[1]


def analyze(graph, start, end, config):
    s, t = graph.station_to_number[start], graph.station_to_number[end]
    sp = ShortestPath(graph, s, config)
    if t not in sp.dist_to:
        raise RuntimeError(f"No route found from '{start}' to '{end}'")
    nodes, edges = sp.path_with_edges(t)
    names = [graph.number_to_station[n] for n in nodes]
    dist = sp.dist_to[t]
    transfers_at = []
    for i in range(len(edges) - 1):
        if edges[i].line != edges[i + 1].line:
            transfers_at.append({"station": names[i + 1], "from_line": edges[i].line,
                                 "to_line": edges[i + 1].line})
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    segments = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = names[i + 1]
            segments[-1]["stops"] += 1
        else:
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1], "stops": 1})
    tc = len(transfers_at)
    try:
        fare = Sum_money(dist)
    except Exception as ex:
        raise RuntimeError(f"Fare calculation failed: {ex}")
    return {"start": start, "end": end, "path": names, "distance": dist, "fare": fare,
            "time": round(calculate_estimated_time(dist, tc), 1), "transfer_count": tc,
            "transfers": transfers_at, "lines": lines, "segments": segments}


def render_text(r, mode):
    out = [f"Route: {r['start']} -> {r['end']} (mode: {mode})",
           "Stations: " + " -> ".join(r["path"]),
           f"Total stations: {len(r['path'])}",
           f"Distance: {format_distance(r['distance'])} ({r['distance'] / 1000:.2f} km)"]
    out.append("Segments:")
    for sg in r["segments"]:
        out.append(f"  [{sg['line']}] {sg['from']} -> {sg['to']} ({sg['stops']} stops)")
    out.append(f"Transfers: {r['transfer_count']}")
    for t in r["transfers"]:
        out.append(f"  Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
    out.append("Lines: " + ", ".join(r["lines"]))
    out.append(f"Estimated time: {r['time']} minutes")
    out.append(f"Fare: {r['fare']} yuan")
    return "\n".join(out)


def render_json(r, mode):
    obj = {"query": {"start": r["start"], "end": r["end"], "mode": mode},
           "path": r["path"], "distance_meters": r["distance"],
           "distance_km": round(r["distance"] / 1000, 2), "fare": r["fare"],
           "estimated_time_minutes": r["time"], "transfer_count": r["transfer_count"],
           "transfers": r["transfers"], "lines": r["lines"], "segments": r["segments"]}
    return json.dumps(obj, ensure_ascii=False, indent=2)


def run_query(graph, text, args, config):
    try:
        start, end = parse_query(text)
    except ValueError as e:
        return str(e), False
    if start not in graph.station_to_number:
        return f"Starting station '{start}' does not exist, please check the station name", False
    if end not in graph.station_to_number:
        return f"Ending station '{end}' does not exist, please check the station name", False
    if start == end:
        return "Starting station cannot be the same as the ending station", False
    try:
        r = analyze(graph, start, end, config)
    except Exception as e:
        return f"Path calculation error: {e}", False
    return (render_json(r, args.mode) if args.format == "json" else render_text(r, args.mode)), True


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except Exception as e:
        print(f"Failed to load subway data: {e}")
        return 1
    if args.check_data:
        ok, msgs = check_data(data)
        print(f"Stations: {len(data.get('stations', []))}, "
              f"connections: {len(data.get('connections', []))}")
        for m in msgs[:50]:
            print("  - " + m)
        print("Data check passed" if ok else f"Data check found {len(msgs)} issue(s)")
        return 0 if ok else 1

    graph = build_graph(data)
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []

    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            text = input("Query> ").strip()
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if not text:
            continue
        if text.lower() in ("q", "quit", "exit"):
            break
        result, ok = run_query(graph, text, args, config)
        print(result)
        if ok and args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(result + "\n\n")
            except OSError as e:
                print(f"Failed to write output file: {e}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

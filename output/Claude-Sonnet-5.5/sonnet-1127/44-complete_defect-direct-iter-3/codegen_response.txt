"""Graph data structures."""


class Edge:
    """Undirected-edge record between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists keyed by station number."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, node):
        self.adj.setdefault(node, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, node):
        return self.adj.get(node, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

    def update_distance(self, a, b, distance, line=None):
        """Update distance of edges between a and b (optionally for one line)."""
        found = False
        for u, v in ((a, b), (b, a)):
            for e in self.adj.get(u, []):
                if e.dest == v and (line is None or e.line == line):
                    e.distance = distance
                    found = True
        return found

"""Dijkstra shortest path with a heap-based priority queue."""
import heapq
import itertools

TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 1000000, "comprehensive": 1500}
PREFERRED_FACTOR = 0.9


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap min priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._paths = {}
        self._transfers = {}
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        maxt = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        # state: (node, line, transfers); transfers only distinguishes when limited
        start = (self.source, None, 0)
        parent = {start: (None, None)}
        real = {start: 0}
        best = {start[:2] if maxt is None else start: 0.0}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            key = st[:2] if maxt is None else st
            if key in done:
                continue
            done.add(key)
            node = st[0]
            if node not in self.dist_to:
                edges = []
                s = st
                while parent[s][0] is not None:
                    ps, e = parent[s]
                    edges.append(e)
                    s = ps
                edges.reverse()
                self.dist_to[node] = real[st]
                self._paths[node] = edges
                self._transfers[node] = st[2]
            for e in self.graph.neighbors(node):
                t = st[2]
                if st[1] is not None and e.line != st[1]:
                    t += 1
                if maxt is not None and t > maxt:
                    continue
                w = e.weight * (PREFERRED_FACTOR if e.line in pref else 1.0)
                if t > st[2]:
                    w += penalty
                ns = (e.dest, e.line, t)
                nkey = ns[:2] if maxt is None else ns
                if nkey in done:
                    continue
                nc = cost + w
                if nc < best.get(nkey, float("inf")):
                    best[nkey] = nc
                    parent[ns] = (st, e)
                    real[ns] = real[st] + e.weight
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self.dist_to

    def path_with_edges(self, target):
        if target not in self.dist_to:
            return [], []
        edges = list(self._paths[target])
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._paths:
            return 0
        edges = self._paths[target]
        return sum(1 for i in range(1, len(edges)) if edges[i].line != edges[i - 1].line)

"""Fare and time calculation."""
import math

FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]


def Sum_money(distance):
    """Fare in yuan for distance in meters (distance rounded up to whole km)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    return 6 + int(math.ceil((km - 32) / 20.0))


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 35 km/h average speed plus 5 minutes per transfer."""
    return round(distance / 1000.0 / 35.0 * 60.0 + transfer_count * 5.0, 1)

"""Data integrity checks and update helpers."""

REQUIRED_LINES = ["Line 1", "Line 2", "Line 4", "Line 5", "Line 6", "Line 8", "Line 10",
                  "Line 13", "Line 14", "Line 15", "Changping Line", "Yizhuang Line",
                  "Fangshan Line", "Airport Line"]


def check_data(data):
    """Return list of problem strings (empty if fine) and info lines."""
    problems = []
    names = {s["name"] for s in data.get("stations", [])}
    lines_present = set()
    for s in data.get("stations", []):
        lines_present.update(s.get("lines", []))
    for c in data.get("connections", []):
        lines_present.add(c.get("line"))
    for ln in REQUIRED_LINES:
        if not any(ln.lower() == p.lower() or ln.lower() in str(p).lower() for p in lines_present):
            problems.append("Missing line coverage: %s" % ln)
    linked = set()
    for c in data.get("connections", []):
        for k in ("source", "dest"):
            if c[k] not in names:
                problems.append("Connection references unknown station: %s" % c[k])
            linked.add(c[k])
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0:
            problems.append("Invalid distance %s between %s and %s" % (d, c["source"], c["dest"]))
        elif d > 20000:
            problems.append("Suspicious distance %s between %s and %s" % (d, c["source"], c["dest"]))
    for n in names - linked:
        problems.append("Station without connections: %s" % n)
    return problems


def add_line(data, line_name, station_names, distances):
    """Add a new line; distances has len(station_names)-1 entries (meters)."""
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances length mismatch")
    existing = {s["name"]: s for s in data["stations"]}
    for n in station_names:
        if n in existing:
            if line_name not in existing[n]["lines"]:
                existing[n]["lines"].append(line_name)
        else:
            data["stations"].append({"name": n, "lines": [line_name]})
    for i, d in enumerate(distances):
        data["connections"].append({"source": station_names[i], "dest": station_names[i + 1],
                                    "distance": d, "line": line_name})


def modify_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    ok = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
            c["distance"] = distance
            ok = True
    return ok

"""CLI for Beijing Subway fare calculation."""
import argparse
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import data_loader
import data_manager
from models import Edge, Graph
from pathfinding import PathfindingConfig, ShortestPath
from fare_calculator import Sum_money, calculate_estimated_time


class SubwaySystem:
    def __init__(self, data):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in data["stations"]:
            self._num(s["name"])
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in data["connections"]:
            a, b = self._num(c["source"]), self._num(c["dest"])
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))
            for n in (c["source"], c["dest"]):
                self.graph.station_lines.setdefault(n, [])
                if c["line"] not in self.graph.station_lines[n]:
                    self.graph.station_lines[n].append(c["line"])

    def _num(self, name):
        if name not in self.station_to_number:
            n = len(self.station_to_number)
            self.station_to_number[name] = n
            self.number_to_station[n] = name
            self.graph.add_node(n)
        return self.station_to_number[name]

    def query(self, start, end, config):
        sp = ShortestPath(self.graph, self.station_to_number[start], config)
        t = self.station_to_number[end]
        nodes, edges = sp.path_with_edges(t)
        if not nodes:
            raise RuntimeError("No route found between '%s' and '%s'" % (start, end))
        dist = sp.dist_to[t]
        transfers = sp.get_transfer_count(t)
        try:
            fare = Sum_money(dist)
        except Exception as e:
            raise ArithmeticError("Fare calculation failed: %s" % e)
        names = [self.number_to_station[n] for n in nodes]
        tstations = []
        for i in range(1, len(edges)):
            if edges[i].line != edges[i - 1].line:
                tstations.append({"station": names[i], "from_line": edges[i - 1].line,
                                  "to_line": edges[i].line})
        lines = []
        segments = []
        for i, e in enumerate(edges):
            if e.line not in lines:
                lines.append(e.line)
            if segments and segments[-1]["line"] == e.line:
                segments[-1]["to"] = names[i + 1]
                segments[-1]["stations"] += 1
            else:
                segments.append({"line": e.line, "from": names[i], "to": names[i + 1], "stations": 1})
        return {"query": {"start": start, "end": end, "mode": config.mode},
                "path": names, "distance_m": dist, "distance_km": round(dist / 1000.0, 2),
                "fare": fare, "estimated_time": calculate_estimated_time(dist, transfers),
                "transfer_count": transfers, "lines": lines,
                "transfer_stations": tstations, "segments": segments}


def fmt_distance(m):
    if m < 1000:
        return "%d meters (%.2f km)" % (m, m / 1000.0)
    return "%.2f km (%d meters)" % (m / 1000.0, m)


def format_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"]))]
    for s in r["segments"]:
        out.append("  Take %s from %s to %s (%d stops)" % (s["line"], s["from"], s["to"], s["stations"]))
    out.append("Total distance: " + fmt_distance(r["distance_m"]))
    out.append("Transfers: %d" % r["transfer_count"])
    for t in r["transfer_stations"]:
        out.append("  Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    out.append("Lines: " + ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    if fmt == "json":
        return json.dumps(r, ensure_ascii=False, indent=2)
    return format_text(r)


def parse_input(line, system):
    """Return (start, end) or raise ValueError with a user message."""
    line = line.replace("；", ";")
    parts = [p.strip() for p in line.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    a, b = parts
    if a not in system.station_to_number:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % a)
    if b not in system.station_to_number:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % b)
    if a == b:
        raise ValueError("Starting station cannot be the same as the ending station")
    return a, b


def write_output(path, text):
    try:
        with open(path, "a", encoding="utf-8") as f:
            f.write(text + "\n")
    except OSError as e:
        print("Failed to write output file: %s" % e)


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        data = data_loader.load_subway_data(data_loader.DATA_FILE_PATH)
    except (OSError, ValueError) as e:
        print("Failed to load data: %s" % e)
        return 1

    if args.check_data:
        problems = data_manager.check_data(data)
        print("Stations: %d, connections: %d" % (len(data["stations"]), len(data["connections"])))
        if problems:
            print("Data check found %d problem(s):" % len(problems))
            for x in problems:
                print(" - " + x)
        else:
            print("Data check passed.")
        return 0

    system = SubwaySystem(data)
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    if args.line_preference:
        config.preferred_lines = [args.line_preference]

    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (or 'quit').")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("quit", "exit", "q"):
            break
        if not line:
            continue
        try:
            a, b = parse_input(line, system)
        except ValueError as e:
            print(e)
            continue
        try:
            result = system.query(a, b, config)
        except ArithmeticError as e:
            print(e)
            continue
        except Exception as e:
            print("Path calculation error: %s" % e)
            continue
        text = render(result, args.format)
        print(text)
        if args.output_file:
            write_output(args.output_file, text)
    return 0


if __name__ == "__main__":
    sys.exit(main())

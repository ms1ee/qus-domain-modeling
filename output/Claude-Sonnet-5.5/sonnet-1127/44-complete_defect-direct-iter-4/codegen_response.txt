"""Graph data structures."""


class Edge:
    """Undirected weighted edge between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def other(self, node):
        return self.dest if node == self.source else self.source

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists keyed by station number."""

    def __init__(self):
        self.graph = {}          # number -> list[Edge]
        self.adj = self.graph
        self.station_lines = {}  # station name -> list of lines
        self.edges = []

    def add_node(self, node):
        self.graph.setdefault(node, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.graph[edge.source].append(edge)
        self.graph[edge.dest].append(edge)
        self.edges.append(edge)

    def neighbors(self, node):
        return self.graph.get(node, [])

    def nodes(self):
        return list(self.graph.keys())

"""Dijkstra shortest path with priority queue, transfer limits and line preference."""
import heapq
import itertools


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}   # node -> state key
        self._info = {}   # state -> (cost, dist, prev_state, edge)
        self._run()

    def _transfer_penalty(self):
        mode = self.config.mode
        if mode == "least_transfer":
            return 10_000_000.0
        if mode == "comprehensive":
            return 1000.0
        return 0.001

    def _run(self):
        limit = self.config.max_transfers
        pref = set(self.config.preferred_lines or [])
        pen = self._transfer_penalty()
        start = (self.source, None, 0)
        self._info[start] = (0.0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, last, t = st
            if node not in self._best:
                self._best[node] = st
                self.dist_to[node] = self._info[st][1]
            dist = self._info[st][1]
            for e in self.graph.neighbors(node):
                nt = t
                c = e.weight * (0.9 if e.line in pref else 1.0)
                if last is not None and e.line != last:
                    nt += 1
                    c += pen
                if limit is not None and nt > limit:
                    continue
                ns = (e.other(node), e.line, nt if limit is not None else 0)
                if ns in done:
                    continue
                ncost = cost + c
                old = self._info.get(ns)
                if old is None or ncost < old[0]:
                    self._info[ns] = (ncost, dist + e.weight, st, e)
                    pq.push(ncost, ns)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        nodes, edges = [], []
        st = self._best[target]
        while st is not None:
            _, _, prev, e = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                count += 1
        return count

"""Fare and time estimation."""
import math

FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]


def Sum_money(distance):
    """Fare in yuan for a distance in meters (distance rounded up to km)."""
    if distance is None or distance < 0:
        raise ValueError("invalid distance")
    km = math.ceil(distance / 1000.0)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: 35 km/h average speed plus 5 minutes per transfer."""
    return distance / 1000.0 / 35.0 * 60.0 + transfer_count * 5.0

"""Network construction, integrity checks and data updates."""
from models import Edge, Graph

REPRESENTATIVE = {
    "1号线": [], "2号线": [], "4号线": [], "5号线": [], "6号线": [], "8号线": [],
    "10号线": [], "13号线": [], "14号线": [], "15号线": [],
}


def build_network(data):
    """Return (graph, station_to_number, number_to_station)."""
    graph = Graph()
    s2n, n2s = {}, {}
    for st in data.get("stations", []):
        name = st["name"]
        if name not in s2n:
            s2n[name] = len(s2n)
            n2s[s2n[name]] = name
            graph.add_node(s2n[name])
        graph.set_station_lines(name, st.get("lines", []))
    for c in data.get("connections", []):
        for nm in (c["source"], c["dest"]):
            if nm not in s2n:
                s2n[nm] = len(s2n)
                n2s[s2n[nm]] = nm
                graph.add_node(s2n[nm])
                graph.station_lines.setdefault(nm, [c["line"]])
        graph.add_edge(Edge(s2n[c["source"]], s2n[c["dest"]], c["distance"], c["line"]))
    return graph, s2n, n2s


def check_data(data):
    """Return list of problem strings (empty if data is fine) and a summary list."""
    problems = []
    names = {s["name"] for s in data.get("stations", [])}
    if not names:
        problems.append("No station data")
    lines = set()
    for s in data.get("stations", []):
        lines.update(s.get("lines", []))
        if not s.get("lines"):
            problems.append("Station without line: %s" % s["name"])
    for c in data.get("connections", []):
        if c["source"] not in names or c["dest"] not in names:
            problems.append("Connection references unknown station: %s-%s" % (c["source"], c["dest"]))
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
            problems.append("Suspicious distance %s: %s-%s" % (d, c["source"], c["dest"]))
        if c["source"] == c["dest"]:
            problems.append("Self loop at %s" % c["source"])
    connected = set()
    for c in data.get("connections", []):
        connected.add(c["source"])
        connected.add(c["dest"])
    for n in sorted(names - connected):
        problems.append("Station has no connections: %s" % n)
    for key in REPRESENTATIVE:
        if not any(key in l or l in key for l in lines):
            problems.append("Main line missing: %s" % key)
    summary = ["Stations: %d" % len(names), "Connections: %d" % len(data.get("connections", [])),
               "Lines: %d" % len(lines)]
    return problems, summary


def add_line(data, line_name, stations, distances):
    """Add a new line: stations (ordered names) and distances between neighbours."""
    if len(distances) != len(stations) - 1:
        raise ValueError("distances must have len(stations)-1 items")
    existing = {s["name"]: s for s in data["stations"]}
    for n in stations:
        if n in existing:
            if line_name not in existing[n]["lines"]:
                existing[n]["lines"].append(line_name)
        else:
            data["stations"].append({"name": n, "lines": [line_name]})
    for a, b, d in zip(stations, stations[1:], distances):
        data["connections"].append({"source": a, "dest": b, "distance": int(d), "line": line_name})


def update_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    found = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
            c["distance"] = int(distance)
            found = True
    return found

"""Route summary, text and JSON reports."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


def format_distance(meters):
    if meters < 1000:
        return "%d meters" % meters
    return "%.2f km (%d meters)" % (meters / 1000.0, meters)


def build_summary(start, end, names, edges, distance, transfers):
    fare = Sum_money(distance)
    time_min = calculate_estimated_time(distance, transfers)
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = names[i + 1]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(names[i])
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1], "stations": 1})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    return {
        "query": {"start": start, "end": end},
        "path": names,
        "distance_m": distance,
        "distance_km": round(distance / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(time_min, 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": segments,
    }


def to_text(s):
    out = ["Route: %s -> %s" % (s["query"]["start"], s["query"]["end"]),
           "Stations (%d): %s" % (len(s["path"]), " -> ".join(s["path"])),
           "Total distance: %s" % format_distance(s["distance_m"])]
    for seg in s["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (seg["line"], seg["from"], seg["to"], seg["stations"]))
    if s["transfer_stations"]:
        out.append("Transfers (%d): %s" % (s["transfer_count"], ", ".join(s["transfer_stations"])))
    else:
        out.append("Transfers: 0")
    out.append("Lines: %s" % ", ".join(s["lines"]))
    out.append("Estimated time: %.0f minutes" % s["estimated_time_min"])
    out.append("Fare: %d yuan" % s["fare"])
    return "\n".join(out)


def to_json(s):
    return json.dumps(s, ensure_ascii=False, indent=2)

"""CLI entry point for the Beijing subway fare system."""
import argparse
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data  # noqa: E402
from data_manager import build_network, check_data  # noqa: E402
from pathfinding import PathfindingConfig, ShortestPath  # noqa: E402
import report  # noqa: E402

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def parse_query(text):
    """Return (start, end, error)."""
    text = text.replace("；", ";")
    parts = [x.strip() for x in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None, None, FORMAT_ERR
    return parts[0], parts[1], None


def query(start, end, graph, s2n, n2s, config, fmt):
    if start not in s2n:
        return None, "Starting station '%s' does not exist, please check the station name" % start
    if end not in s2n:
        return None, "Ending station '%s' does not exist, please check the station name" % end
    if start == end:
        return None, "Starting station cannot be the same as the ending station"
    try:
        sp = ShortestPath(graph, s2n[start], config)
        nodes, edges = sp.path_with_edges(s2n[end])
        if not nodes:
            return None, "No route found between '%s' and '%s' (check transfer limit)" % (start, end)
        dist = sp.dist_to[s2n[end]]
        transfers = sp.get_transfer_count(s2n[end])
    except Exception as ex:
        return None, "Path calculation error: %s" % ex
    try:
        summary = report.build_summary(start, end, [n2s[n] for n in nodes], edges, dist, transfers)
    except Exception as ex:
        return None, "Fare calculation error: %s" % ex
    return (report.to_json(summary) if fmt == "json" else report.to_text(summary)), None


def main(argv=None):
    args = parse_args(argv)
    data = load_subway_data(DATA_FILE_PATH)
    if args.check_data:
        problems, summary = check_data(data)
        print("\n".join(summary))
        if problems:
            print("Problems found:")
            for pr in problems:
                print("  - " + pr)
        else:
            print("Data check passed.")
        return 0
    graph, s2n, n2s = build_network(data)
    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    if args.line_preference:
        config.preferred_lines = [args.line_preference]
    results = []
    print("Beijing Subway Fare Calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        start, end, err = parse_query(line)
        if err:
            print(err)
            continue
        out, err = query(start, end, graph, s2n, n2s, config, args.format)
        if err:
            print(err)
            continue
        print(out)
        results.append(out)
        if args.output_file:
            try:
                with open(args.output_file, "w", encoding="utf-8") as f:
                    f.write("\n\n".join(results) + "\n")
            except OSError as ex:
                print("Cannot write output file: %s" % ex)
    return 0


if __name__ == "__main__":
    sys.exit(main())

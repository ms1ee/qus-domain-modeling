class ListManager:
    def _parse(self, text):
        users, errors = [], []
        for n, raw in enumerate(text.splitlines(), 1):
            line = raw.strip()
            if not line:
                continue
            parts = [p.strip() for p in re.split(r"[,\uff0c\t]", line)]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {n}: Format Error: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def load_default_list(self):
        return self._parse(DEFAULT_LIST)

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        try:
            with open(path, "r", encoding="utf-8") as f:
                text = f.read()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        return self._parse(text)

    def validate_users(self, users):
        seen, valid, dups = set(), [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, q in defaults:
            try:
                self.add_prize(name, q)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Prize quantity must be a positive integer")
        self._prizes.append(Prize(name, quantity, avoid_groups or [], associated_drawer))

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        if not isinstance(order, (list, tuple)) or sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, order=0):
        self.user = user
        self.prize_name = prize_name
        self.order = order
        self.time = datetime.now()


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        w = Winner(user, prize.name, len(self.winners) + 1)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = sorted(
                (w for w in self.winners if w.prize_name == p.name),
                key=lambda w: w.order)
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import random

CHENG_RU = "cheng ru"


def _is_cheng_ru(user):
    return CHENG_RU in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winner_names = set()
        self.last_two_draw_results = []

    def _available(self):
        return [u for u in self.users
                if not getattr(u, "has_won", False) and u.name not in self.winner_names]

    def _get_eligible_pool(self, prize):
        avail = self._available()
        drawer = prize.associated_drawer
        if drawer == "Ma Jiaqi" and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in avail if _is_cheng_ru(u)]
        if drawer == "Li Bin":
            return avail
        avoid = {g.lower() for g in prize.avoid_groups}
        return [u for u in avail if u.department.lower() not in avoid]

    def describe_rules(self, prize):
        parts = ["exclude previous winners"]
        if prize.associated_drawer == "Li Bin":
            parts = ["relaxed rules (avoidance groups ignored)"]
        elif prize.avoid_groups:
            parts.append("avoid groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer:
            parts.append(f"drawer: {prize.associated_drawer}")
        return "; ".join(parts)

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        u.has_won = True
        self.winner_names.add(u.name)
        return u

    def finish_round(self, prize, winners):
        if prize.associated_drawer in ("Chen Qixian", "Jin Duo"):
            self.last_two_draw_results.append(any(_is_cheng_ru(u) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    def draw_round(self, prize, result_manager, control=None, on_win=None):
        """control(i) may return 'terminate' to end the round."""
        won = []
        for i in range(1, prize.quantity + 1):
            if control and control(i) == "terminate":
                break
            if on_win:
                on_win(i, None)
            u = self.draw_one(prize)
            if u is None:
                break
            result_manager.add_winner(u, prize)
            won.append(u)
            if on_win:
                on_win(i, u)
        self.finish_round(prize, won)
        return won

def load_users(lm):
    while True:
        c = input("1) Default list  2) Custom txt list: ").strip()
        if c == "1":
            users, errors = lm.load_default_list()
        elif c == "2":
            users, errors = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errors:
            print("  !", e)
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"  ! Duplicate name: {d.name}")
        if errors or dups:
            a = input("Remove invalid data and continue? (y=continue / n=terminate): ").strip().lower()
            if a != "y":
                if c == "2":
                    continue
                return []
        return valid


def control(i):
    cmd = input(f"[Drawing winner #{i}] Enter to draw (pause/terminate): ").strip().lower()
    if cmd == "terminate":
        return "terminate"
    if cmd == "pause":
        while input("Paused. Type 'continue': ").strip().lower() != "continue":
            pass
    return None

from lottery_system.models import User

DEFAULT_DATA = [
    ("Li Bin", "Technology Department"),
    ("Chen Qixian", "Product Department"),
    ("Jin Duo", "Technology Department"),
    ("Ma Jiaqi", "Operations Department"),
    ("Wang Fang", "Cheng Ru"),
    ("Zhao Lei", "Cheng Ru"),
    ("Sun Li", "Product Department"),
    ("Zhou Ming", "Technology Department"),
    ("Wu Hao", "Operations Department"),
    ("Zheng Yu", "Product Department"),
    ("Qian Xin", "Cheng Ru"),
    ("Liu Yang", "Technology Department"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"Cannot read file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            sep = "," if "," in line else ("，" if "，" in line else None)
            parts = [p.strip() for p in line.split(sep)] if sep else []
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {i}: Format Error: expected 'name,department' but got '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        for name, qty in [("Third Prize - Movie Tickets", 3),
                          ("Second Prize - Shopping Cards", 2),
                          ("First Prize - 1 Day Annual Leave", 1)]:
            try:
                self.add_prize(name, qty)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name.strip(), quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "Cheng Ru"
SPECIAL_DRAWER = "Li Bin"
TRACKED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {u.name for u in self.winners}
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if u.department == CHENG_RU]
        if drawer == SPECIAL_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_round(self, prize, winners):
        if prize.associated_drawer in TRACKED_DRAWERS:
            self.last_two_draw_results.append(
                any(u.department == CHENG_RU for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime
from dataclasses import dataclass


@dataclass
class Winner:
    user: object
    prize_name: str
    order: int = 0


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        self.winners.append(Winner(user, prize.name, len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        res = {}
        for p in prizes:
            res[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return res

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("1) default list  2) custom txt list: ").strip()
        if c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            users, errs = lm.load_default_list()
        for e in errs:
            print(e)
        if errs:
            a = input("Errors found. (r)emove invalid and continue / (t)erminate: ").strip().lower()
            if a != "r":
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate removed: {d.name}")
        if valid:
            return valid
        print("No valid users.")


def setup_prizes(pm, groups):
    pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        try:
            name = input("Name: ")
            qty = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            drawer = input("Associated drawer (blank for none): ").strip() or None
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print(f"Invalid: {e}")
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("New order (e.g. 3 1 2) or blank: ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, keeping default.")
        except ValueError:
            print("Invalid order, keeping default.")


def run():
    lm = ListManager()
    users = load_users(lm)
    info = lm.get_group_info(users)
    for g, n in info.items():
        print(f"{g}: {n}")
    pm = PrizeManager()
    setup_prizes(pm, info)
    lot = LotteryManager(users)
    rm = ResultManager()
    prizes = pm.get_prizes()
    for p in prizes:
        print(f"\n== {p.name} | quantity {p.quantity} | avoid {p.avoid_groups} | drawer {p.associated_drawer}")
        round_winners = []
        i = 0
        while i < p.quantity:
            cmd = input("Enter to draw (pause/continue/terminate): ").strip().lower()
            if cmd == "terminate":
                print("Round terminated.")
                break
            if cmd == "pause":
                while input("Paused. Type 'continue': ").strip().lower() != "continue":
                    pass
                continue
            print(f"Currently drawing the {i + 1}th winner")
            u = lot.draw_one(p)
            if u is None:
                print("No eligible candidates left.")
                break
            rm.add_winner(u, p)
            round_winners.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {p.name}")
            i += 1
        lot.finish_round(p, round_winners)
    print("\n=== Final Results ===")
    for name, ws in rm.get_winners_by_prize(list(reversed(prizes))).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {name}")
    print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    run()

"""Grid based path planning."""

CELL = 100


class Path:
    """A polyline path."""

    def __init__(self, waypoints, speed=1):
        self.waypoints = list(waypoints)
        self.total_distance = sum(
            abs(a[0] - b[0]) + abs(a[1] - b[1])
            for a, b in zip(self.waypoints, self.waypoints[1:]))
        self.estimated_time = -(-self.total_distance // max(1, speed))


class PathPlanner:
    """Plans right-angle paths, preferring streets (grid lines)."""

    @staticmethod
    def _nearest_line(v):
        return min(max(0, round(v / CELL) * CELL), 800)

    def plan(self, start, end, speed=1):
        """Plan a path start -> end with at most right-angle turns.

        If the start is off a street, first snap to the nearest street
        line, travel along streets, then leave the street to the target.
        """
        sx, sy = start
        ex, ey = end
        pts = [(sx, sy)]
        if sx == ex or sy == ey:  # straight line (directly above/below)
            pts.append((ex, ey))
        else:
            pts.append((ex, sy))  # horizontal first, then vertical
            pts.append((ex, ey))
        clean = [pts[0]]
        for p in pts[1:]:
            if p != clean[-1]:
                clean.append(p)
        return Path(clean, speed)

    def plan_multi(self, points, speed=1):
        """Plan a continuous path through several destinations."""
        wps = [tuple(points[0])]
        for p in points[1:]:
            seg = self.plan(wps[-1], tuple(p), speed)
            wps.extend(seg.waypoints[1:])
        return Path(wps, speed)

"""Rider model."""
import globalvar
from path_planner import PathPlanner

IDLE, PICKING, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 0
        self.y = (rider_id % 8) * 100 + 0  # on a street line
        self.speed = 50
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.path = None
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({
            "event": event, "order_id": order_id,
            "time": globalvar.get_value("tick", 0) or 0,
            "x": self.x, "y": self.y})

    def _step_towards(self, target):
        """Move up to speed units (x first, then y)."""
        budget = self.speed
        tx, ty = target
        dx = max(-budget, min(budget, tx - self.x))
        self.x += dx
        budget -= abs(dx)
        dy = max(-budget, min(budget, ty - self.y))
        self.y += dy

    def move(self):
        """Advance one tick."""
        if not self.orders:
            self.state = IDLE
            return
        order = self.orders[0]
        if self.state == IDLE:
            self.state = PICKING
            self.path = PathPlanner().plan(
                (self.x, self.y), tuple(order.res_pos), self.speed)
            self._log("accepted_order", order.id)
        if self.state == PICKING:
            self._step_towards(order.res_pos)
            if (self.x, self.y) == tuple(order.res_pos):
                self.state = DELIVERING
                self.path = PathPlanner().plan(
                    (self.x, self.y), tuple(order.cus_pos), self.speed)
                self._log("arrived_at_merchant", order.id)
            return
        if self.state == DELIVERING:
            self._step_towards(order.cus_pos)
            if (self.x, self.y) == tuple(order.cus_pos):
                self._log("delivered_to_customer", order.id)
                order.complete(globalvar.get_value("tick", 0) or 0)
                self.orders.pop(0)
                self.state = IDLE
                self.path = None

    def to_dict(self):
        return {"rider_id": self.rider_id, "x": self.x, "y": self.y,
                "speed": self.speed, "state": self.state,
                "orders": [o.id for o in self.orders],
                "special_case": self.special_case}

"""Order model."""
import globalvar


def to_center(v):
    """Convert coordinate to grid cell center (clamped to the map)."""
    v = max(0, min(int(v), 799))
    return (v // 100) * 100 + 50


class Order:
    """A delivery order."""

    _counter = 0

    def __init__(self, res_pos, cus_pos):
        Order._counter += 1
        self.id = Order._counter
        self.res_pos = [to_center(res_pos[0]), to_center(res_pos[1])]
        self.cus_pos = [to_center(cus_pos[0]), to_center(cus_pos[1])]
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.time_taken = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0]) +
                         abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Mark the order as assigned to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"
        self.start_time = globalvar.get_value("tick", 0) or 0

    def complete(self, tick):
        """Mark delivered and record efficiency data."""
        self.status = "completed"
        self.completion_time = tick
        self.time_taken = tick - self.start_time

    def to_dict(self):
        return {"id": self.id, "res_pos": self.res_pos,
                "cus_pos": self.cus_pos, "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "completion_time": self.completion_time,
                "distance": self.distance, "time_taken": self.time_taken}

"""System management: funds, riders, persistence."""
import json
import globalvar
from rider import Rider

INITIAL_FUNDS = 10000
RECRUIT_COST = 500
MAX_RIDERS = 20


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", INITIAL_FUNDS)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])
    globalvar.set_value("recruit_cost", RECRUIT_COST)


def recruit_rider():
    """Recruit a rider if funds allow; return Rider or None."""
    funds = globalvar.get_value("funds", 0)
    riders = globalvar.get_value("riders")
    if funds < RECRUIT_COST or len(riders) >= MAX_RIDERS:
        return None
    rid = globalvar.get_value("rider_id_counter", 0) + 1
    globalvar.set_value("rider_id_counter", rid)
    globalvar.set_value("funds", funds - RECRUIT_COST)
    r = Rider(rid)
    riders.append(r)
    return r


def tick():
    """Advance simulated time by one tick."""
    globalvar.set_value("tick", globalvar.get_value("tick", 0) + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def save(path="config.json"):
    """Persist system data as JSON."""
    data = {"funds": globalvar.get_value("funds"),
            "tick": globalvar.get_value("tick"),
            "riders": [r.to_dict() for r in globalvar.get_value("riders")],
            "orders": [o.to_dict() for o in globalvar.get_value("orders")]}
    with open(path, "w") as f:
        json.dump(data, f, indent=2)

"""Order management and assignment."""
import globalvar
import system
from order_model import Order


def _dist(r, order):
    return abs(r.x - order.res_pos[0]) + abs(r.y - order.res_pos[1])


def find_best_rider_for_order(order):
    """Pick the nearest rider (Manhattan), preferring idle, balancing load.

    Recruits a new rider if none is idle and funds allow.
    """
    riders = globalvar.get_value("riders") or []
    if not any(r.state == 0 and not r.orders for r in riders):
        system.recruit_rider()
        riders = globalvar.get_value("riders")
    if not riders:
        return None
    return min(riders, key=lambda r: (len(r.orders), _dist(r, order),
                                      r.rider_id))


class OrderManager:
    """Creates and dispatches orders."""

    def create_order(self, coords):
        """Create order from [rx, ry, cx, cy], assign it; return
        (order, rider, new_rider_added)."""
        before = len(globalvar.get_value("riders"))
        order = Order(coords[0:2], coords[2:4])
        rider = find_best_rider_for_order(order)
        if rider is None:
            return order, None, False
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
        globalvar.get_value("orders").append(order)
        return order, rider, len(globalvar.get_value("riders")) > before

"""Display helpers."""
import globalvar

STATES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Text rendering."""

    def status(self):
        lines = ["Funds: %s" % globalvar.get_value("funds"),
                 "Tick: %s" % globalvar.get_value("tick"),
                 "Riders: %d" % len(globalvar.get_value("riders"))]
        for r in globalvar.get_value("riders"):
            lines.append("  Rider %d at (%d,%d) %s orders=%s" % (
                r.rider_id, r.x, r.y, STATES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        orders = globalvar.get_value("orders")
        if not orders:
            return "No orders."
        return "\n".join(
            "Order %d start=%s rest=%s cust=%s dist=%d time=%s rider=%s %s" % (
                o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                o.time_taken, o.rider_id, o.status) for o in orders)

    def riders_log(self):
        lines = []
        for r in globalvar.get_value("riders"):
            for e in r.log:
                lines.append("Rider %d t=%s (%s,%s) %s order=%s" % (
                    r.rider_id, e["time"], e["x"], e["y"], e["event"],
                    e["order_id"]))
        return "\n".join(lines) or "No records."

    def map(self, path=None, order=None):
        """ASCII 8x8 map; R rider, S restaurant, C customer, * route."""
        g = [["." for _ in range(8)] for _ in range(8)]
        if path:
            pts = path.waypoints
            for a, b in zip(pts, pts[1:]):
                n = max(abs(a[0] - b[0]), abs(a[1] - b[1])) // 50
                for i in range(n + 1):
                    x = a[0] + (b[0] - a[0]) * i // max(n, 1)
                    y = a[1] + (b[1] - a[1]) * i // max(n, 1)
                    g[min(7, y // 100)][min(7, x // 100)] = "*"
        if order:
            g[order.res_pos[1] // 100][order.res_pos[0] // 100] = "S"
            g[order.cus_pos[1] // 100][order.cus_pos[0] // 100] = "C"
        for r in globalvar.get_value("riders"):
            g[min(7, r.y // 100)][min(7, r.x // 100)] = "R"
        return "\n".join(" ".join(row) for row in reversed(g))

"""Command parsing and main loop."""
import globalvar
import system
from display import DisplayManager
from order import OrderManager
from path_planner import PathPlanner

ALIASES = {"s": "status", "o": "orders", "r": "riders", "q": "quit",
           "st": "start", "h": "history"}


class InputManager:
    """Parses and executes commands."""

    def __init__(self):
        self.command_history = []
        self.display = DisplayManager()
        self.orders = OrderManager()

    def parse_coords(self, parts):
        if len(parts) != 4:
            raise ValueError("Need 4 integers: rx ry cx cy")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers")
        if any(v < 0 or v > 800 for v in vals):
            raise ValueError("Coordinates must be within 0-800")
        return vals

    def start(self, parts):
        vals = self.parse_coords(parts)
        order, rider, added = self.orders.create_order(vals)
        if rider is None:
            return "No rider available (insufficient funds)."
        planner = PathPlanner()
        path = planner.plan_multi(
            [(rider.x, rider.y), tuple(order.res_pos), tuple(order.cus_pos)],
            rider.speed)
        return ("Restaurant: %s Customer: %s\nNew rider added: %s\n"
                "Rider %d speed=%d distance=%d time=%d ticks\n%s" % (
                    order.res_pos, order.cus_pos, added, rider.rider_id,
                    rider.speed, path.total_distance, path.estimated_time,
                    self.display.map(path, order)))

    def execute(self, line):
        parts = line.split()
        if not parts:
            return ""
        self.command_history.append(line)
        cmd = ALIASES.get(parts[0].lower(), parts[0].lower())
        if cmd in ("quit", "exit"):
            globalvar.set_value("running", False)
            return "Bye."
        if cmd == "start":
            out = self.start(parts[1:])
        elif cmd == "status":
            out = self.display.status() + "\n" + self.display.map()
        elif cmd == "orders":
            out = self.display.orders()
        elif cmd == "riders":
            out = self.display.riders_log()
        elif cmd == "history":
            out = "\n".join(self.command_history)
        else:
            raise ValueError("Unknown command: %s" % cmd)
        system.tick()
        return out

    def loop(self):
        while globalvar.get_value("running"):
            try:
                line = input("> ")
            except EOFError:
                break
            try:
                print(self.execute(line))
            except ValueError as e:
                print("Error:", e)
            except Exception as e:  # recover gracefully
                print("Unexpected error:", e)

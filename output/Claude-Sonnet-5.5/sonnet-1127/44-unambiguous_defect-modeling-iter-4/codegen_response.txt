"""Core data models: Station, Line, Edge, Graph."""


class Station:
    def __init__(self, id, name, is_transfer=False):
        self.id = id
        self.name = name
        self.is_transfer = is_transfer


class Line:
    def __init__(self, name, station_sequence=None):
        self.name = name
        self.station_sequence = list(station_sequence or [])


class Edge:
    """Undirected-edge half; source/dest are station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph. Station numbers are assigned by the caller."""

    def __init__(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = {}  # number -> list[Edge]
        self.station_lines = {}  # name -> list[str]

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.graph.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an edge in both directions."""
        self.graph.setdefault(edge.source, []).append(edge)
        self.graph.setdefault(edge.dest, []).append(edge.reversed())

    def neighbors(self, number):
        return self.graph.get(number, [])

    def remove_edges_between(self, a, b, line=None):
        for x, y in ((a, b), (b, a)):
            self.graph[x] = [e for e in self.graph.get(x, [])
                             if not (e.dest == y and (line is None or e.line == line))]

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1

"""Dijkstra shortest path with priority queue, transfer-aware."""
import heapq
import itertools

MODES = ("shortest_distance", "least_transfer", "comprehensive")
TRANSFER_PENALTY = {"shortest_distance": 0, "least_transfer": 5000, "comprehensive": 1500}


class PathfindingConfig:
    def __init__(self, mode="shortest_distance", max_transfers=None, preferred_lines=None):
        self.mode = mode
        self.max_transfers = max_transfers
        self.preferred_lines = list(preferred_lines or [])


class MinPQ:
    """Heap-based minimum priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self.edge_to = {}
        self._prev = {}
        self._best_state = {}
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        limit = cfg.max_transfers
        track = limit is not None
        start = (self.source, None, 0)
        cost = {start: 0.0}
        dist = {start: 0}
        best = {self.source: (0.0, start)}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, tr = st
            for e in self.graph.neighbors(node):
                ntr = tr
                extra = 0
                if line is not None and e.line != line:
                    ntr += 1
                    extra = penalty
                    if track and ntr > limit:
                        continue
                w = e.weight
                if e.line in self.config.preferred_lines:
                    w *= 0.9
                nst = (e.dest, e.line, ntr if track else 0)
                nc = c + w + extra
                if nc < cost.get(nst, float("inf")):
                    cost[nst] = nc
                    dist[nst] = dist[st] + e.weight
                    self._prev[nst] = (st, e)
                    pq.push(nc, nst)
                    if nc < best.get(e.dest, (float("inf"), None))[0]:
                        best[e.dest] = (nc, nst)
        for node, (c, st) in best.items():
            self._best_state[node] = st
            self.dist_to[node] = dist[st]

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        edges = []
        while st in self._prev:
            st, e = self._prev[st]
            edges.append(e)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        count = 0
        for a, b in zip(edges, edges[1:]):
            if a.line != b.line:
                count += 1
        return count

"""Fare calculation for Beijing Subway."""

FARE_TIERS = {"0-6km": 3, "6-12km": 4, "12-22km": 5, "22-32km": 6, "32km+": 7}


def Sum_money(distance):
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = int(round(distance))
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: ~35 km/h running speed plus 5 min per transfer."""
    return distance / 1000.0 / 35.0 * 60.0 + 5.0 * transfer_count


class FareCalculator:
    def __init__(self):
        self.fare_tiers = dict(FARE_TIERS)

    def calculate(self, distance):
        return Sum_money(distance)

"""Data management and integrity checking."""
import json

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph

REPRESENTATIVE = {
    "Line 1": "Tiananmen East", "Line 2": "Xizhimen", "Line 4": "Xidan",
    "Line 5": "Dongdan", "Line 10": "Guomao", "Line 13": "Wudaokou",
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = Graph()
        self.build()

    def build(self):
        g = Graph()
        names = []
        seen = set()
        for s in self.data.get("stations", []):
            if s["name"] not in seen:
                seen.add(s["name"])
                names.append(s["name"])
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in seen:
                    seen.add(n)
                    names.append(n)
        for i, n in enumerate(names):
            g.add_station(n, i)
        for s in self.data.get("stations", []):
            g.set_station_lines(s["name"], s.get("lines", []))
        derived = {}
        for c in self.data.get("connections", []):
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
            for n in (c["source"], c["dest"]):
                derived.setdefault(n, [])
                if c["line"] not in derived[n]:
                    derived[n].append(c["line"])
        for n in names:
            if n not in g.station_lines:
                g.set_station_lines(n, derived.get(n, []))
        self.graph = g

    def station_names(self):
        return list(self.graph.station_to_number)

    def lines(self):
        result = {}
        for n, ls in self.graph.station_lines.items():
            for l in ls:
                result.setdefault(l, []).append(n)
        return result

    def save(self):
        with open(self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)

    # ---- updates ----
    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters."""
        if len(distances) != len(stations) - 1:
            raise ValueError("need len(stations)-1 distances")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": int(d), "line": line_name})
        self.build()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = int(distance)
                found = True
        if found:
            self.build()
        return found


class DataIntegrityChecker:
    def __init__(self, manager):
        self.manager = manager
        self.station_coverage_valid = False
        self.connection_valid = False
        self.distance_valid = False

    def run(self):
        msgs = []
        g = self.manager.graph
        lines = self.manager.lines()
        missing = [f"{l}:{s}" for l, s in REPRESENTATIVE.items() if s not in g.station_to_number]
        self.station_coverage_valid = not missing and len(g.station_to_number) > 0
        if missing:
            msgs.append("Missing representative stations: " + ", ".join(missing))
        msgs.append("Stations: %d, lines: %d" % (len(g.station_to_number), len(lines)))
        bad = []
        for c in self.manager.data.get("connections", []):
            if c["source"] not in g.station_to_number or c["dest"] not in g.station_to_number:
                bad.append(c)
        isolated = [n for n, i in g.station_to_number.items() if not g.neighbors(i)]
        self.connection_valid = not bad and not isolated
        if isolated:
            msgs.append("Isolated stations: " + ", ".join(isolated[:10]))
        baddist = [c for c in self.manager.data.get("connections", [])
                   if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0
                   or c["distance"] > 20000]
        self.distance_valid = not baddist
        if baddist:
            msgs.append("%d connections have suspicious distances" % len(baddist))
        ok = self.station_coverage_valid and self.connection_valid and self.distance_valid
        msgs.append("Data check " + ("PASSED" if ok else "FAILED"))
        return ok, msgs

"""Route reports: text and JSON."""
import json

from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import ShortestPath, PathfindingConfig


class Route:
    def __init__(self, stations, edges, total_distance, transfer_count, involved_lines):
        self.station_sequence = stations
        self.edges = edges
        self.total_distance = total_distance
        self.transfer_count = transfer_count
        self.involved_lines = involved_lines


def build_route(graph, start, end, config):
    s, e = graph.station_to_number[start], graph.station_to_number[end]
    sp = ShortestPath(graph, s, config)
    if e not in sp.dist_to:
        raise ValueError("No route found between %s and %s" % (start, end))
    nodes, edges = sp.path_with_edges(e)
    lines = []
    for ed in edges:
        if ed.line not in lines:
            lines.append(ed.line)
    names = [graph.number_to_station[n] for n in nodes]
    return Route(names, edges, sp.dist_to[e], sp.get_transfer_count(e), lines)


def segments(route):
    """Group consecutive edges by line -> list of (line, from, to, stops)."""
    segs = []
    names = route.station_sequence
    for i, ed in enumerate(route.edges):
        if segs and segs[-1][0] == ed.line:
            segs[-1][2] = names[i + 1]
            segs[-1][3] += 1
        else:
            segs.append([ed.line, names[i], names[i + 1], 1])
    return segs


def summary(route, query):
    fare = Sum_money(route.total_distance)
    t = calculate_estimated_time(route.total_distance, route.transfer_count)
    segs = segments(route)
    transfers = [segs[i][2] for i in range(len(segs) - 1)]
    return {
        "query": query,
        "path": route.station_sequence,
        "distance_meters": route.total_distance,
        "distance_km": round(route.total_distance / 1000.0, 2),
        "fare": fare,
        "estimated_time_minutes": round(t),
        "transfer_count": route.transfer_count,
        "transfer_stations": transfers,
        "lines": route.involved_lines,
        "segments": [{"line": s[0], "from": s[1], "to": s[2], "stops": s[3]} for s in segs],
    }


def format_text(info):
    q = info["query"]
    out = ["Route: %s -> %s (mode: %s)" % (q["start"], q["end"], q["mode"])]
    out.append("Stations (%d): %s" % (len(info["path"]), " -> ".join(info["path"])))
    m = info["distance_meters"]
    out.append("Distance: %d m (%.2f km)" % (m, m / 1000.0))
    for s in info["segments"]:
        out.append("  [%s] %s -> %s (%d stops)" % (s["line"], s["from"], s["to"], s["stops"]))
    if info["transfer_stations"]:
        out.append("Transfers: %d at %s" % (info["transfer_count"],
                                            ", ".join(info["transfer_stations"])))
    else:
        out.append("Transfers: 0")
    out.append("Lines: " + ", ".join(info["lines"]))
    out.append("Estimated time: %d minutes" % info["estimated_time_minutes"])
    out.append("Fare: %d yuan" % info["fare"])
    return "\n".join(out)


def format_json(info):
    return json.dumps(info, ensure_ascii=False, indent=2)


def render(info, fmt):
    return format_json(info) if fmt == "json" else format_text(info)

"""Command-line interface."""
import argparse
import re
import sys

from data_manager import DataManager, DataIntegrityChecker
from pathfinding import PathfindingConfig, MODES
from report import build_route, summary, render

SEP = re.compile(r"[;；,，]")


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=list(MODES), default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def parse_query(text, names):
    """Return (start, end) or raise ValueError with a user-facing message."""
    parts = [x.strip() for x in SEP.split(text)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    a, b = parts
    if a not in names:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % a)
    if b not in names:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % b)
    if a == b:
        raise ValueError("Starting station cannot be the same as the ending station")
    return a, b


def run_query(dm, text, args, config):
    a, b = parse_query(text, dm.graph.station_to_number)
    try:
        route = build_route(dm.graph, a, b, config)
    except ValueError as e:
        raise ValueError("Path calculation failed: %s" % e)
    try:
        info = summary(route, {"start": a, "end": b, "mode": args.mode,
                               "max_transfers": args.max_transfers,
                               "line_preference": args.line_preference})
    except Exception as e:
        raise ValueError("Fare calculation failed: %s" % e)
    return render(info, args.format)


def write_output(path, content):
    try:
        with open(path, "a", encoding="utf-8") as f:
            f.write(content + "\n")
    except OSError as e:
        print("Cannot write output file: %s" % e)


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager()
    except Exception as e:
        print("Failed to load subway data: %s" % e)
        return 1
    if args.check_data:
        ok, msgs = DataIntegrityChecker(dm).run()
        print("\n".join(msgs))
        return 0 if ok else 1
    config = PathfindingConfig(args.mode, args.max_transfers,
                               [args.line_preference] if args.line_preference else [])
    print("Enter 'Starting Station; Ending Station' (empty line or 'q' to quit):")
    while True:
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if not line or line.lower() in ("q", "quit", "exit"):
            break
        try:
            out = run_query(dm, line, args, config)
        except ValueError as e:
            print(e)
            continue
        except Exception as e:
            print("Unexpected error: %s" % e)
            continue
        print(out)
        if args.output_file:
            write_output(args.output_file, out)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Graph data structures."""


class Edge:
    """Edge between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return f"Edge({self.source}->{self.dest}, {self.distance}, {self.line})"


class Graph:
    """Weighted undirected graph; station numbers are chosen by the caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(Edge(edge.dest, edge.source, edge.distance, edge.line))

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""Dijkstra shortest path with priority queue."""
import heapq
import itertools

MODE_SHORTEST = "shortest_distance"
MODE_LEAST_TRANSFER = "least_transfer"
MODE_COMPREHENSIVE = "comprehensive"

TRANSFER_PENALTY = {MODE_SHORTEST: 0, MODE_LEAST_TRANSFER: 10000, MODE_COMPREHENSIVE: 1500}


class PathfindingConfig:
    def __init__(self):
        self.mode = MODE_SHORTEST
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap based min priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._parent = {}   # state -> (prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0)
        limit = cfg.max_transfers
        prefs = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        cost = {start: 0.0}
        dist = {start: 0}
        settled = set()
        pq = MinPQ()
        pq.push(0.0, start)
        while not pq.is_empty():
            c, st = pq.pop()
            if st in settled:
                continue
            settled.add(st)
            node, line, t = st
            if node not in self._best_state:
                self._best_state[node] = st
                self.dist_to[node] = dist[st]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0
                if line is not None and e.line != line:
                    extra = penalty
                    if limit is not None:
                        nt = t + 1
                        if nt > limit:
                            continue
                w = e.weight * (0.9 if e.line in prefs else 1.0)
                ns = (e.dest, e.line, nt)
                if ns in settled:
                    continue
                nc = c + w + extra
                if nc < cost.get(ns, float("inf")):
                    cost[ns] = nc
                    dist[ns] = dist[st] + e.weight
                    self._parent[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        nodes = [st[0]]
        edges = []
        while st in self._parent:
            st, e = self._parent[st]
            nodes.append(st[0])
            edges.append(e)
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time calculation."""
import math


def Sum_money(distance) -> int:
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated minutes: ~2 min per km plus 5 min per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("negative input")
    return round(distance / 1000.0 * 2.0 + transfer_count * 5.0, 1)

"""Network construction, integrity checks and updates."""
from models import Edge, Graph

REPRESENTATIVE = {
    "1号线": [], "2号线": [], "4号线": [], "5号线": [], "6号线": [], "8号线": [],
    "10号线": [], "13号线": [], "14号线": [], "15号线": [],
}


class SubwayNetwork:
    def __init__(self, data):
        self.data = data
        self.build()

    def build(self):
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        for s in self.data.get("stations", []):
            self.add_station(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            self._connect(c["source"], c["dest"], c["distance"], c["line"])

    def add_station(self, name, lines):
        if name not in self.station_to_number:
            n = len(self.station_to_number)
            self.station_to_number[name] = n
            self.number_to_station[n] = name
            self.graph.add_node(n)
        self.graph.set_station_lines(name, lines)

    def _connect(self, a, b, d, line):
        if a not in self.station_to_number or b not in self.station_to_number:
            return False
        self.graph.add_edge(Edge(self.station_to_number[a], self.station_to_number[b], d, line))
        return True

    # --- update mechanisms ---
    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distance count mismatch")
        for s in stations:
            lines = list(self.graph.station_lines.get(s, []))
            if line not in lines:
                lines.append(line)
            self.add_station(s, lines)
        for a, b, d in zip(stations, stations[1:], distances):
            self._connect(a, b, d, line)

    def modify_station(self, name, lines):
        if name not in self.station_to_number:
            raise KeyError(name)
        self.graph.set_station_lines(name, lines)

    def update_distance(self, a, b, distance, line=None):
        if distance <= 0:
            raise ValueError("distance must be positive")
        na, nb = self.station_to_number[a], self.station_to_number[b]
        found = False
        for u, v in ((na, nb), (nb, na)):
            for e in self.graph.adj[u]:
                if e.dest == v and (line is None or e.line == line):
                    e.distance = e.weight = distance
                    found = True
        if not found:
            raise KeyError("no such connection")

    # --- integrity ---
    def check_integrity(self):
        problems = []
        lines = set()
        for ls in self.graph.station_lines.values():
            lines.update(ls)
        for ln in REPRESENTATIVE:
            if ln not in lines:
                problems.append(f"Line missing: {ln}")
        names = set(self.station_to_number)
        for c in self.data.get("connections", []):
            if c["source"] not in names or c["dest"] not in names:
                problems.append(f"Connection with unknown station: {c['source']} - {c['dest']}")
            if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0:
                problems.append(f"Bad distance: {c['source']} - {c['dest']}")
        for name, n in self.station_to_number.items():
            if not self.graph.adj.get(n):
                problems.append(f"Isolated station: {name}")
        return problems

"""Route report generation."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


def build_summary(network, start, end, nodes, edges):
    dist = sum(e.weight for e in edges)
    transfers = []
    for a, b in zip(edges, edges[1:]):
        if a.line != b.line:
            transfers.append({"station": network.number_to_station[a.dest],
                              "from_line": a.line, "to_line": b.line})
    lines = []
    for e in edges:
        if e.line not in lines:
            lines.append(e.line)
    fare = Sum_money(dist)
    return {
        "query": {"start": start, "end": end},
        "path": [network.number_to_station[n] for n in nodes],
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": calculate_estimated_time(dist, len(transfers)),
        "transfer_count": len(transfers),
        "transfers": transfers,
        "lines": lines,
        "segments": _segments(network, edges),
    }


def _segments(network, edges):
    segs = []
    for e in edges:
        if segs and segs[-1]["line"] == e.line:
            segs[-1]["to"] = network.number_to_station[e.dest]
            segs[-1]["stations"] += 1
        else:
            segs.append({"line": e.line, "from": network.number_to_station[e.source],
                         "to": network.number_to_station[e.dest], "stations": 1})
    return segs


def format_text(s):
    out = [f"Route: {s['query']['start']} -> {s['query']['end']}",
           "Stations: " + " -> ".join(s["path"])]
    d = s["distance_m"]
    out.append(f"Total distance: {d} m ({s['distance_km']} km)")
    for seg in s["segments"]:
        out.append(f"  [{seg['line']}] {seg['from']} -> {seg['to']} ({seg['stations']} stops)")
    if s["transfers"]:
        for t in s["transfers"]:
            out.append(f"Transfer at {t['station']}: {t['from_line']} -> {t['to_line']}")
    else:
        out.append("No transfers")
    out.append("Lines: " + ", ".join(s["lines"]))
    out.append(f"Transfers: {s['transfer_count']}")
    out.append(f"Estimated time: {s['estimated_time_min']} minutes")
    out.append(f"Fare: {s['fare']} yuan")
    return "\n".join(out)


def format_json(s):
    return json.dumps(s, ensure_ascii=False, indent=2)

"""CLI entry point for the Beijing Subway fare calculator."""
import argparse
import os
import re
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data  # noqa: E402
from data_manager import SubwayNetwork  # noqa: E402
from pathfinding import PathfindingConfig, ShortestPath  # noqa: E402
from reporter import build_summary, format_json, format_text  # noqa: E402


def parse_input(line):
    parts = [p.strip() for p in re.split(r"[;；,，]", line)]
    parts = [p for p in parts if p != ""]
    if len(parts) != 2 or len(re.split(r"[;；,，]", line)) != 2:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    return parts[0], parts[1]


def query(network, start, end, config):
    if start not in network.station_to_number:
        raise ValueError(f"Starting station '{start}' does not exist, please check the station name")
    if end not in network.station_to_number:
        raise ValueError(f"Ending station '{end}' does not exist, please check the station name")
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    s, t = network.station_to_number[start], network.station_to_number[end]
    try:
        sp = ShortestPath(network.graph, s, config)
        nodes, edges = sp.path_with_edges(t)
    except Exception as ex:
        raise RuntimeError(f"Path calculation failed: {ex}")
    if not nodes:
        raise ValueError("No route found (check max transfer limit)")
    try:
        return build_summary(network, start, end, nodes, edges)
    except Exception as ex:
        raise RuntimeError(f"Fare calculation failed: {ex}")


def main(argv=None):
    ap = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    ap.add_argument("--output_file")
    ap.add_argument("--format", choices=["text", "json"], default="text")
    ap.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                    default="shortest_distance")
    ap.add_argument("--max_transfers", type=int)
    ap.add_argument("--line_preference")
    ap.add_argument("--check_data", action="store_true")
    args = ap.parse_args(argv)

    try:
        network = SubwayNetwork(load_subway_data(DATA_FILE_PATH))
    except Exception as ex:
        print(f"Failed to load data: {ex}")
        return 1

    if args.check_data:
        problems = network.check_integrity()
        print(f"Stations: {len(network.station_to_number)}, edges: {network.graph.edge_count()}")
        if problems:
            for p in problems:
                print("Problem:", p)
            return 1
        print("Data check passed.")
        return 0

    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    config.preferred_lines = [args.line_preference] if args.line_preference else []

    results = []
    while True:
        try:
            line = input("Enter 'Starting Station; Ending Station' (q to quit): ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        try:
            start, end = parse_input(line)
            summary = query(network, start, end, config)
        except (ValueError, RuntimeError) as ex:
            print(ex)
            continue
        text = format_json(summary) if args.format == "json" else format_text(summary)
        print(text)
        results.append(text)
    if args.output_file and results:
        try:
            with open(args.output_file, "w", encoding="utf-8") as f:
                f.write("\n\n".join(results) + "\n")
        except OSError as ex:
            print(f"Cannot write output file: {ex}")
            return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

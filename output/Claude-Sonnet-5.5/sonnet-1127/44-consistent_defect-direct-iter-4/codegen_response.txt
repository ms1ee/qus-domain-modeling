"""Graph data structures: Edge, Graph, MinPQ."""
import heapq


class Edge:
    """Directed weighted edge between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted directed graph with adjacency lists. Numbers assigned by caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, number):
        self.adj.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, [])

    def neighbors(self, number):
        return self.adj.get(number, [])

    def nodes(self):
        return list(self.adj.keys())

    def edges(self):
        for lst in self.adj.values():
            for e in lst:
                yield e

    def edge_count(self):
        return sum(len(v) for v in self.adj.values())


class MinPQ:
    """Heap based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._n = 0

    def push(self, priority, item):
        self._n += 1
        heapq.heappush(self._heap, (priority, self._n, item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)

"""Dijkstra based path finding with transfer awareness."""
from models import MinPQ

MODES = ("shortest_distance", "least_transfer", "comprehensive")


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._parent = {}   # state -> (prev_state, edge)
        self._run()

    def _penalty(self):
        mode = self.config.mode
        if mode == "least_transfer":
            return 10_000_000.0
        if mode == "comprehensive":
            return 1500.0
        return 0.0

    def _run(self):
        cfg = self.config
        penalty = self._penalty()
        limit = cfg.max_transfers
        prefs = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        best = {start: 0.0}
        real = {start: 0}
        self._parent[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        node_cost = {}
        while not pq.is_empty():
            cost, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line, tr = state
            if node not in node_cost or cost < node_cost[node]:
                node_cost[node] = cost
                self._best_state[node] = state
                self.dist_to[node] = real[state]
            for e in self.graph.neighbors(node):
                ntr = tr
                step = float(e.weight)
                if line is not None and e.line != line:
                    ntr = tr + 1
                    if limit is not None and ntr > limit:
                        continue
                    step += penalty
                if e.line in prefs:
                    step *= 0.9
                key_tr = ntr if limit is not None else 0
                ns = (e.dest, e.line, key_tr)
                nc = cost + step
                if nc < best.get(ns, float("inf")):
                    best[ns] = nc
                    real[ns] = real[state] + e.distance
                    self._parent[ns] = (state, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        state = self._best_state[target]
        edges = []
        while self._parent[state] is not None:
            prev, e = self._parent[state]
            edges.append(e)
            state = prev
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (fractional km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0 - 1e-9)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km <= 22:
        return 5
    if km <= 32:
        return 6
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km, 5 per transfer."""
    return distance / 1000.0 * 1.5 + transfer_count * 5

"""Graph construction, integrity checks and data updates."""
from models import Edge, Graph


def build_graph(data):
    """Return (graph, station_to_number, number_to_station)."""
    graph = Graph()
    s2n = {}
    n2s = {}

    def reg(name):
        if name not in s2n:
            s2n[name] = len(s2n)
            n2s[s2n[name]] = name
            graph.add_node(s2n[name])

    for st in data.get("stations", []):
        reg(st["name"])
        graph.set_station_lines(st["name"], st.get("lines", []))
    for c in data.get("connections", []):
        reg(c["source"])
        reg(c["dest"])
        graph.add_edge(Edge(s2n[c["source"]], s2n[c["dest"]], c["distance"], c["line"]))
    for name in s2n:
        if name not in graph.station_lines:
            graph.set_station_lines(name, [])
    return graph, s2n, n2s


def check_data(data):
    """Return list of report lines and an ok flag."""
    report = []
    ok = True
    stations = {s["name"] for s in data.get("stations", [])}
    conns = data.get("connections", [])
    report.append("Stations: %d, connections: %d" % (len(stations), len(conns)))
    if not stations:
        ok = False
        report.append("ERROR: no stations")
    lines = set()
    for s in data.get("stations", []):
        lines.update(s.get("lines", []))
    report.append("Lines: %d" % len(lines))
    bad = [c for c in conns if c["source"] not in stations or c["dest"] not in stations]
    if bad:
        ok = False
        report.append("ERROR: %d connections reference unknown stations" % len(bad))
    badd = [c for c in conns if not isinstance(c["distance"], (int, float)) or c["distance"] <= 0]
    if badd:
        ok = False
        report.append("ERROR: %d connections with invalid distance" % len(badd))
    connected = set()
    for c in conns:
        connected.add(c["source"])
        connected.add(c["dest"])
    iso = stations - connected
    if iso:
        report.append("WARNING: %d stations without connections" % len(iso))
    cline = {c["line"] for c in conns}
    missing = lines - cline
    if missing:
        report.append("WARNING: lines without connections: %s" % ", ".join(sorted(missing)))
    report.append("Data check " + ("passed" if ok else "FAILED"))
    return report, ok


def add_line(data, line_name, station_names, distances):
    """Add a new line: consecutive stations with bidirectional connections."""
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances must have len(stations)-1 items")
    existing = {s["name"]: s for s in data["stations"]}
    for n in station_names:
        if n in existing:
            if line_name not in existing[n]["lines"]:
                existing[n]["lines"].append(line_name)
        else:
            s = {"name": n, "lines": [line_name]}
            data["stations"].append(s)
            existing[n] = s
    for a, b, d in zip(station_names, station_names[1:], distances):
        data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        data["connections"].append({"source": b, "dest": a, "distance": d, "line": line_name})


def update_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None, both=True):
    n = 0
    for c in data["connections"]:
        if line is not None and c["line"] != line:
            continue
        if (c["source"] == a and c["dest"] == b) or (both and c["source"] == b and c["dest"] == a):
            c["distance"] = distance
            n += 1
    return n

"""CLI for the Beijing Subway fare calculator."""
import argparse
import difflib
import json
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from data_manager import build_graph, check_data
from fare_calculator import Sum_money, calculate_estimated_time
from pathfinding import PathfindingConfig, ShortestPath

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_query(text):
    for sep in (";", "；", ",", "，"):
        if sep in text:
            parts = [p.strip() for p in text.split(sep)]
            if len(parts) == 2 and parts[0] and parts[1]:
                return parts
            return None
    return None


def match_station(name, names):
    if name in names:
        return name, False
    subs = [n for n in names if name in n or n in name]
    if subs:
        return min(subs, key=lambda n: abs(len(n) - len(name))), True
    close = difflib.get_close_matches(name, names, n=1, cutoff=0.0)
    return (close[0], True) if close else (None, True)


def fmt_distance(d):
    return "%d m" % d if d < 1000 else "%.2f km" % (d / 1000.0)


def build_result(start, end, nodes, edges, n2s, sp, target):
    dist = sp.dist_to[target]
    transfers = sp.get_transfer_count(target)
    segments = []
    transfer_stations = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = n2s[e.dest]
            segments[-1]["stations"] += 1
        else:
            if segments:
                transfer_stations.append(n2s[e.source])
            segments.append({"line": e.line, "from": n2s[e.source], "to": n2s[e.dest], "stations": 1})
    lines = []
    for s in segments:
        if s["line"] not in lines:
            lines.append(s["line"])
    return {
        "query": {"start": start, "end": end},
        "path": [n2s[n] for n in nodes],
        "distance": dist,
        "fare": Sum_money(dist),
        "estimated_time": round(calculate_estimated_time(dist, transfers), 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": lines,
        "segments": segments,
    }


def format_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"])),
           "Total distance: %d m (%.2f km) [%s]" % (r["distance"], r["distance"] / 1000.0, fmt_distance(r["distance"]))]
    for s in r["segments"]:
        out.append("  %s: %s -> %s (%d stops)" % (s["line"], s["from"], s["to"], s["stations"]))
    if r["transfer_stations"]:
        out.append("Transfer stations: " + ", ".join(r["transfer_stations"]))
    out.append("Transfers: %d" % r["transfer_count"])
    out.append("Lines: " + ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def run_query(text, ctx):
    q = parse_query(text)
    if q is None:
        return FORMAT_ERR, False
    a, b = q
    names = ctx["names"]
    s_name, s_fuzzy = match_station(a, names)
    e_name, e_fuzzy = match_station(b, names)
    if s_name is None:
        return "Starting station '%s' does not exist, please check the station name" % a, False
    if e_name is None:
        return "Ending station '%s' does not exist, please check the station name" % b, False
    if s_name == e_name:
        return "Starting station cannot be the same as the ending station", False
    notes = []
    if s_fuzzy:
        notes.append("Starting station '%s' matched to '%s'" % (a, s_name))
    if e_fuzzy:
        notes.append("Ending station '%s' matched to '%s'" % (b, e_name))
    try:
        s2n = ctx["s2n"]
        cfg = ctx["config"]
        sp = ShortestPath(ctx["graph"], s2n[s_name], cfg)
        target = s2n[e_name]
        nodes, edges = sp.path_with_edges(target)
        if not nodes or target not in sp.dist_to:
            return "No route found from '%s' to '%s' (check transfer limit)" % (s_name, e_name), False
    except Exception as ex:  # path calculation errors
        return "Path calculation failed: %s" % ex, False
    try:
        r = build_result(s_name, e_name, nodes, edges, ctx["n2s"], sp, target)
    except Exception as ex:
        return "Fare calculation failed: %s" % ex, False
    if ctx["format"] == "json":
        body = json.dumps(r, ensure_ascii=False, indent=2)
    else:
        body = format_text(r)
    if notes:
        if ctx["format"] == "json":
            r["notes"] = notes
            body = json.dumps(r, ensure_ascii=False, indent=2)
        else:
            body = "\n".join(notes) + "\n" + body
    return body, True


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1
    if args.check_data:
        report, ok = check_data(data)
        print("\n".join(report))
        return 0 if ok else 1

    graph, s2n, n2s = build_graph(data)
    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    cfg.preferred_lines = [args.line_preference] if args.line_preference else []
    ctx = {"graph": graph, "s2n": s2n, "n2s": n2s, "names": list(s2n.keys()),
           "config": cfg, "format": args.format}

    out_f = open(args.output_file, "w", encoding="utf-8") if args.output_file else None
    try:
        print("Enter 'Starting Station; Ending Station' (or 'q' to quit):")
        while True:
            try:
                line = input("> ")
            except EOFError:
                break
            line = line.strip()
            if not line:
                continue
            if line.lower() in ("q", "quit", "exit"):
                break
            msg, ok = run_query(line, ctx)
            print(msg)
            if out_f and ok:
                out_f.write(msg + "\n\n")
                out_f.flush()
    finally:
        if out_f:
            out_f.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())

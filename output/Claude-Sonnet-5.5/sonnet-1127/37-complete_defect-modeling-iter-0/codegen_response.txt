from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Marketing Department"),
    ("Zheng Shi", "Marketing Department"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Operations Department"),
    ("Ma Jiaqi", "Design Department"),
    ("Li Bin", "Design Department"),
    ("Cheng Ru A", "Cheng Ru Group"),
    ("Cheng Ru B", "Cheng Ru Group"),
    ("Cheng Ru C", "Cheng Ru Group"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"Error: cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        presets = [
            ("Third Prize - Movie Tickets", 3, [], None),
            ("Second Prize - Shopping Cards", 2, [], None),
            ("First Prize - 1 Day Annual Leave", 1, [], None),
        ]
        for name, q, avoid, drawer in presets:
            try:
                self.add_prize(name, q, avoid, drawer)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or int(quantity) <= 0:
            raise ValueError("Invalid prize name or quantity")
        p = Prize(name, int(quantity), list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

SPECIAL_DRAWER = "Li Bin"
FALLBACK_GROUP = "cheng ru"
MONITORED = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"


def _is_cheng_ru(dept):
    return FALLBACK_GROUP in dept.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = set()
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if u.name not in self.winners and not u.has_won]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        avoid = set(prize.avoid_groups or [])
        normal = [u for u in pool if u.department not in avoid]
        if (drawer == FALLBACK_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            forced = [u for u in pool if _is_cheng_ru(u.department)]
            if forced:
                return forced
        return normal

    def draw(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        u.has_won = True
        self.winners.add(u.name)
        return u

    def finish_round(self, prize, winners):
        if prize.associated_drawer in MONITORED:
            self.last_two_draw_results.append(
                any(_is_cheng_ru(w.department) for w in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import datetime


class Winner:
    def __init__(self, user, prize_name, order=0):
        self.user = user
        self.prize_name = prize_name
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name, len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        res = {}
        for p in prizes:
            res[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return res

    def export_to_csv(self, prizes):
        path = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        try:
            by = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in by[p.name]:
                        f.write(f"{w.user.name},{w.user.department},{p.name}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def setup_list(lm):
    while True:
        c = input("List: 1) default 2) custom txt: ").strip()
        if c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
            for e in errs:
                print(e)
            if errs and input("Terminate upload? (y/n): ").strip().lower() == "y":
                continue
        else:
            users, _ = lm.load_default_list()
        users, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate removed: {d.name}")
        if users:
            return users
        print("No valid users.")


def setup_prizes(pm):
    if input("Load default prizes? (y/n): ").strip().lower() != "n":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").strip().lower() == "y":
        try:
            name = input("Name: ").strip()
            q = int(input("Quantity: "))
            drawer = input("Drawer (blank for none): ").strip() or None
            avoid = [g.strip() for g in input("Avoid groups (comma sep): ").split(",") if g.strip()]
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print(f"Invalid input: {e}")
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity}")
    o = input("New order (e.g. 2,1,3) or blank: ").strip()
    if o:
        try:
            if not pm.reorder_prizes([int(x) for x in o.split(",")]):
                print("Invalid order; keeping original.")
        except ValueError:
            print("Invalid order; keeping original.")


def run_round(num, prize, mgr, rm):
    avoid = ", ".join(prize.avoid_groups) or "none"
    print(f"\n=== Round {num}: {prize.name} x{prize.quantity} | drawer: "
          f"{prize.associated_drawer or 'n/a'} | avoid: {avoid} ===")
    won = []
    i = 1
    while i <= prize.quantity:
        cmd = input("Enter to draw (pause/terminate): ").strip().lower()
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing the {i}th winner...")
        u = mgr.draw(prize)
        if u is None:
            print("No eligible participants.")
            break
        rm.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    mgr.finish_round(prize, won)


def main():
    lm = ListManager()
    users = setup_list(lm)
    for d, c in lm.get_group_info(users).items():
        print(f"{d}: {c}")
    pm = PrizeManager()
    setup_prizes(pm)
    mgr = LotteryManager(users)
    rm = ResultManager()
    prizes = pm.get_prizes()
    for n, p in enumerate(prizes, 1):
        run_round(n, p, mgr, rm)
    print("\n=== All Winners ===")
    for p, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {p}")
    if input("Export results? (y/n): ").strip().lower() == "y":
        print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()

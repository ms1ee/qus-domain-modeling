class GridMap:
    """8x8 grid, 800x800 units, 100-unit cells."""

    def __init__(self):
        self.width = 800
        self.height = 800
        self.cell_size = 100
        self.grid_dimension = 8

    def to_grid(self, v):
        """Convert a coordinate to the top-left corner of its cell."""
        v = max(0, min(int(v), self.width))
        return min((v // self.cell_size) * self.cell_size, self.width)

    def is_poi(self, x, y):
        """True for restaurant/customer points (integer hundreds)."""
        return x % 100 == 0 and y % 100 == 0

"""Grid-based path planning."""


class PathPlanner:
    """Plans movement along streets (axis aligned)."""

    def next_position(self, x, y, tx, ty, speed):
        """Return position after moving up to `speed` units toward target.

        Moves horizontally first, then vertically (right-angle turn).
        If already aligned on x, moves vertically directly.
        """
        budget = speed
        if x != tx:
            step = min(budget, abs(tx - x))
            x += step if tx > x else -step
            budget -= step
        if budget > 0 and y != ty:
            step = min(budget, abs(ty - y))
            y += step if ty > y else -step
        return x, y

    def plan(self, start, goals, speed=20):
        """Return list of waypoints (per tick) through the goals in order."""
        x, y = start
        pts = [(x, y)]
        for gx, gy in goals:
            guard = 0
            while (x, y) != (gx, gy) and guard < 10000:
                x, y = self.next_position(x, y, gx, gy, speed)
                pts.append((x, y))
                guard += 1
        return pts

"""Rider model."""
import globalvar
from path_planner import PathPlanner

IDLE, PICKING_UP, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self.history = []
        self.planner = PathPlanner()
        self._record("spawn", None)

    def _record(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id})
        self.history.append({
            "time": globalvar.get_value("tick", 0), "x": self.x,
            "y": self.y, "state": self.state, "order_id": order_id,
            "event": event})

    def active_order(self):
        """First order not yet completed."""
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def active_count(self):
        return sum(1 for o in self.orders if o.status != "completed")

    def move(self):
        """Advance one tick."""
        order = self.active_order()
        if order is None:
            self.state = IDLE
            return
        if self.state == IDLE:
            self.state = PICKING_UP
            order.status = "picking_up"
            self._record("accepted_order", order.id)
        target = order.res_pos if self.state == PICKING_UP else order.cus_pos
        self.special_case = (self.x == target[0] or self.y == target[1])
        self.x, self.y = self.planner.next_position(
            self.x, self.y, target[0], target[1], self.speed)
        if (self.x, self.y) != tuple(target):
            return
        if self.state == PICKING_UP:
            self.state = DELIVERING
            order.status = "delivering"
            self._record("arrived_at_merchant", order.id)
        else:
            order.status = "completed"
            now = globalvar.get_value("tick", 0)
            order.completion_time = now
            order.time_taken = now - order.start_time
            self.state = IDLE
            self._record("delivered_to_customer", order.id)
            nxt = self.active_order()
            if nxt is not None:
                self.state = PICKING_UP
                nxt.status = "picking_up"
                self._record("accepted_order", nxt.id)

"""Order management."""
import globalvar
from order_model import Order
from rider import Rider


def find_best_rider_for_order(order):
    """Rider with fewest active orders, ties broken by distance."""
    riders = globalvar.get_value("riders", [])
    if not riders:
        return None
    rx, ry = order.res_pos
    return min(riders, key=lambda r: (
        r.active_count(),
        ((r.x - rx) ** 2 + (r.y - ry) ** 2) ** 0.5))


class OrderManager:
    """Creates, assigns, and tracks orders."""

    def pending_count(self):
        return sum(1 for o in globalvar.get_value("orders", [])
                   if o.status == "pending")

    def idle_riders(self):
        return [r for r in globalvar.get_value("riders", [])
                if r.active_count() == 0]

    def maybe_recruit(self, pending):
        """Recruit a rider if conditions hold. Returns True if added."""
        funds = globalvar.get_value("funds")
        if funds >= 300 and pending > len(self.idle_riders()):
            cid = globalvar.get_value("rider_id_counter") + 1
            globalvar.set_value("rider_id_counter", cid)
            globalvar.set_value("funds", funds - 500)
            globalvar.get_value("riders").append(Rider(cid))
            return True
        return False

    def create_order(self, coords):
        """Create and assign an order. Returns (order, new_rider_added)."""
        order = Order((coords[0], coords[1]), (coords[2], coords[3]))
        globalvar.get_value("orders").append(order)
        added = self.maybe_recruit(self.pending_count())
        rider = find_best_rider_for_order(order)
        if rider is not None:
            order.assign_to(rider.rider_id)
            rider.orders.append(order)
        return order, added

"""Display and map visualization."""
import globalvar

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


class DisplayManager:
    """Renders status and maps."""

    def render_map(self, route=None, order=None):
        grid = [["." for _ in range(16)] for _ in range(16)]
        for i in range(16):
            for j in range(16):
                if i % 2 == 0 or j % 2 == 0:
                    grid[j][i] = "#"

        def cell(p):
            return min(p[0] // 50, 15), min(p[1] // 50, 15)

        for p in route or []:
            cx, cy = cell(p)
            grid[cy][cx] = "*"
        orders = [order] if order else [
            o for o in globalvar.get_value("orders", [])
            if o.status != "completed"]
        for o in orders:
            sx, sy = cell(o.res_pos)
            grid[sy][sx] = "S"
            cx, cy = cell(o.cus_pos)
            grid[cy][cx] = "C"
        for r in globalvar.get_value("riders", []):
            cx, cy = cell((r.x, r.y))
            grid[cy][cx] = "R"
        return "\n".join("".join(row) for row in grid)

    def status(self):
        riders = globalvar.get_value("riders", [])
        lines = ["Tick: %d  Funds: %d  Riders: %d" % (
            globalvar.get_value("tick", 0), globalvar.get_value("funds"),
            len(riders))]
        for r in riders:
            lines.append("Rider %d at (%d,%d) %s, orders: %s" % (
                r.rider_id, r.x, r.y, STATE_NAMES[r.state],
                [o.id for o in r.orders]))
        return "\n".join(lines)

    def orders(self):
        lines = []
        for o in globalvar.get_value("orders", []):
            lines.append(
                "Order %d start=%d res=%s cus=%s dist=%d time=%s rider=%s [%s]"
                % (o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                   o.time_taken, o.rider_id, o.status))
        return "\n".join(lines) or "No orders."

    def riders_log(self):
        lines = []
        for r in globalvar.get_value("riders", []):
            for h in r.history:
                lines.append("Rider %d t=%d (%d,%d) state=%d order=%s %s" % (
                    r.rider_id, h["time"], h["x"], h["y"], h["state"],
                    h["order_id"], h["event"]))
        return "\n".join(lines) or "No riders."

class InputManager:
    """Parses user input."""

    COMMANDS = {"start": "start", "s": "start", "status": "status",
                "orders": "orders", "o": "orders", "riders": "riders",
                "r": "riders", "map": "map", "m": "map", "tick": "tick",
                "t": "tick", "run": "run", "quit": "quit", "exit": "quit",
                "q": "quit", "help": "help", "h": "help"}

    def parse_command(self, line):
        parts = line.strip().split()
        if not parts:
            return None, []
        cmd = self.COMMANDS.get(parts[0].lower())
        return cmd, parts[1:]

    def parse_coords(self, text):
        """Parse four integers in 0-1000; raises ValueError."""
        text = text.replace("[", " ").replace("]", " ").replace(",", " ")
        parts = text.split()
        if len(parts) != 4:
            raise ValueError("Enter exactly four integers.")
        try:
            vals = [int(p) for p in parts]
        except ValueError:
            raise ValueError("Coordinates must be integers.")
        for v in vals:
            if not 0 <= v <= 1000:
                raise ValueError("Coordinates must be within 0-1000.")
        return vals

"""System manager."""
import json

import globalvar
from grid_map import GridMap
from order import OrderManager
from path_planner import PathPlanner
from display import DisplayManager
from input_manager import InputManager


def init_system():
    """Initialize global state."""
    globalvar._init()
    globalvar.set_value("funds", 1000)
    globalvar.set_value("riders", [])
    globalvar.set_value("rider_id_counter", 0)
    globalvar.set_value("running", True)
    globalvar.set_value("tick", 0)
    globalvar.set_value("orders", [])


class SystemManager:
    """Coordinates all modules."""

    def __init__(self):
        init_system()
        self.grid = GridMap()
        self.orders = OrderManager()
        self.planner = PathPlanner()
        self.display = DisplayManager()
        self.input = InputManager()

    def tick(self):
        globalvar.set_value("tick", globalvar.get_value("tick") + 1)
        for r in globalvar.get_value("riders"):
            r.move()

    def busy(self):
        return any(o.status != "completed"
                   for o in globalvar.get_value("orders"))

    def start_order(self, coords):
        order, added = self.orders.create_order(coords)
        rider = next((r for r in globalvar.get_value("riders")
                      if r.rider_id == order.rider_id), None)
        out = ["Restaurant: %s  Customer: %s" % (order.res_pos, order.cus_pos),
               "New rider added: %s" % ("yes" if added else "no")]
        if rider is None:
            out.append("No rider available; order pending.")
            return "\n".join(out)
        route = self.planner.plan((rider.x, rider.y),
                                  [order.res_pos, order.cus_pos], rider.speed)
        out.append("Rider %d speed %d units/tick, estimated time %d ticks"
                   % (rider.rider_id, rider.speed, len(route) - 1))
        out.append(self.display.render_map(route, order))
        return "\n".join(out)

    def save(self, path="state.json"):
        data = {
            "funds": globalvar.get_value("funds"),
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "state": r.state, "history": r.history}
                       for r in globalvar.get_value("riders")],
            "orders": [{"id": o.id, "res": o.res_pos, "cus": o.cus_pos,
                        "status": o.status, "rider": o.rider_id,
                        "start": o.start_time, "time": o.time_taken}
                       for o in globalvar.get_value("orders")]}
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

"""Command-line entry point."""
import globalvar
from system import SystemManager

try:
    import readline  # noqa: F401  (command history)
except ImportError:
    pass


def main():
    sm = SystemManager()
    print("Food Delivery Dispatch. Commands: start, status, orders, riders,"
          " map, tick [n], run, quit")
    while globalvar.get_value("running"):
        try:
            line = input("> ")
        except (EOFError, KeyboardInterrupt):
            break
        try:
            cmd, args = sm.input.parse_command(line)
            if cmd is None:
                if line.strip():
                    print("Unknown command. Type 'help'.")
            elif cmd == "start":
                text = " ".join(args) or input(
                    "Enter rx ry cx cy: ")
                print(sm.start_order(sm.input.parse_coords(text)))
            elif cmd == "status":
                print(sm.display.status())
            elif cmd == "orders":
                print(sm.display.orders())
            elif cmd == "riders":
                print(sm.display.riders_log())
            elif cmd == "map":
                print(sm.display.render_map())
            elif cmd == "tick":
                n = int(args[0]) if args else 1
                for _ in range(max(n, 0)):
                    sm.tick()
                print(sm.display.status())
            elif cmd == "run":
                guard = 0
                while sm.busy() and guard < 10000:
                    sm.tick()
                    guard += 1
                print(sm.display.status())
            elif cmd == "help":
                print("start [rx ry cx cy], status, orders, riders, map,"
                      " tick [n], run, quit")
            elif cmd == "quit":
                globalvar.set_value("running", False)
        except ValueError as e:
            print("Error:", e)
        except Exception as e:  # recover gracefully
            print("Unexpected error:", e)
    try:
        sm.save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

"""Graph data structures."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""MinPQ and Dijkstra-based ShortestPath."""
import heapq
import itertools

MODES = ("shortest_distance", "least_transfer", "comprehensive")


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def is_empty(self):
        return not self._h

    def __len__(self):
        return len(self._h)


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._parent = {}   # state -> (prev_state, edge)
        self._run()

    def _penalty(self):
        m = self.config.mode
        if m == "least_transfer":
            return 100000
        if m == "comprehensive":
            return 1500
        return 0

    def _run(self):
        cfg = self.config
        pen = self._penalty()
        limit = cfg.max_transfers
        prefs = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        self._parent[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        node_cost = {}
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in node_cost or c < node_cost[node]:
                node_cost[node] = c
                self._best_state[node] = st
                self.dist_to[node] = real[st]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0
                if line is not None and e.line != line:
                    extra = pen
                    if limit is not None:
                        nt = t + 1
                        if nt > limit:
                            continue
                w = e.distance * (0.9 if e.line in prefs else 1.0)
                ns = (e.dest, e.line, nt)
                nc = c + w + extra
                if ns not in done and nc < cost.get(ns, float("inf")):
                    cost[ns] = nc
                    real[ns] = real[st] + e.distance
                    self._parent[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self.dist_to

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        edges = []
        while self._parent[st] is not None:
            st, e = self._parent[st]
            edges.append(e)
        edges.reverse()
        nodes = [self.source] + [e.dest for e in edges]
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare and time estimation."""


def Sum_money(distance):
    """Fare in yuan for a distance in meters."""
    d = int(round(float(distance)))
    if d < 0:
        raise ValueError("distance cannot be negative")
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes: ~2.5 min/km plus 5 min per transfer."""
    return round(distance / 1000.0 * 2.5 + transfer_count * 5.0, 1)

"""Data management: build graph, validate, update."""
from data_loader import load_subway_data, DATA_FILE_PATH
from models import Graph, Edge
from pathfinding import ShortestPath

REPRESENTATIVE = {
    "Line 1": "Tian'anmen East", "Line 2": "Xizhimen", "Line 4": "Xidan",
    "Line 10": "Guomao", "Line 13": "Wudaokou",
}


class SubwaySystem:
    def __init__(self, path=DATA_FILE_PATH):
        self.data = load_subway_data(path)
        self.build()

    def build(self):
        self.graph = Graph()
        self.station_to_number = {}
        self.number_to_station = {}
        for s in self.data.get("stations", []):
            self._ensure(s["name"])
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        for c in self.data.get("connections", []):
            a, b = self._ensure(c["source"]), self._ensure(c["dest"])
            self.graph.add_edge(Edge(a, b, c["distance"], c["line"]))

    def _ensure(self, name):
        if name not in self.station_to_number:
            n = len(self.station_to_number)
            self.station_to_number[name] = n
            self.number_to_station[n] = name
            self.graph.add_node(n)
            self.graph.station_lines.setdefault(name, [])
        return self.station_to_number[name]

    def lines(self):
        return sorted({c["line"] for c in self.data.get("connections", [])})

    def find_route(self, start, end, config=None):
        sp = ShortestPath(self.graph, self.station_to_number[start], config)
        t = self.station_to_number[end]
        if t not in sp.dist_to:
            return None
        nodes, edges = sp.path_with_edges(t)
        return {"nodes": nodes, "edges": edges, "distance": sp.dist_to[t],
                "transfers": sp.get_transfer_count(t)}

    # ---- updates ----
    def add_line(self, line, stations, distances):
        """stations: ordered names; distances: list of len(stations)-1."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances length mismatch")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line not in known[n]["lines"]:
                    known[n]["lines"].append(line)
            else:
                self.data["stations"].append({"name": n, "lines": [line]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append(
                {"source": a, "dest": b, "distance": d, "line": line})
        self.build()

    def update_station(self, name, lines):
        for s in self.data["stations"]:
            if s["name"] == name:
                s["lines"] = list(lines)
                self.build()
                return True
        return False

    def update_distance(self, a, b, distance, line=None):
        ok = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
                c["distance"] = distance
                ok = True
        if ok:
            self.build()
        return ok

    # ---- checks ----
    def check_data(self):
        problems = []
        names = {s["name"] for s in self.data["stations"]}
        lines = set(self.lines())
        for ln, st in REPRESENTATIVE.items():
            if ln in lines and st not in names:
                problems.append("Representative station '%s' of %s missing" % (st, ln))
        for c in self.data["connections"]:
            for k in ("source", "dest"):
                if c[k] not in names:
                    problems.append("Connection references unknown station '%s'" % c[k])
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0 or d > 20000:
                problems.append("Suspicious distance %s between %s and %s" % (d, c["source"], c["dest"]))
        connected = set()
        for c in self.data["connections"]:
            connected.add(c["source"]); connected.add(c["dest"])
        for n in names - connected:
            problems.append("Station '%s' has no connections" % n)
        return problems

"""Route reports (text / JSON)."""
import json
from fare_calculator import Sum_money, calculate_estimated_time


def build_report(system, start, end, route):
    nodes, edges = route["nodes"], route["edges"]
    names = [system.number_to_station[n] for n in nodes]
    segments = []
    transfers = []
    for i, e in enumerate(edges):
        if segments and segments[-1]["line"] == e.line:
            segments[-1]["to"] = names[i + 1]
            segments[-1]["distance"] += e.distance
            segments[-1]["stops"] += 1
        else:
            if segments:
                transfers.append({"station": names[i], "from_line": segments[-1]["line"],
                                  "to_line": e.line})
            segments.append({"line": e.line, "from": names[i], "to": names[i + 1],
                             "distance": e.distance, "stops": 1})
    dist = route["distance"]
    return {
        "query": {"start": start, "end": end},
        "path": names,
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": Sum_money(dist),
        "estimated_time_min": calculate_estimated_time(dist, route["transfers"]),
        "transfer_count": route["transfers"],
        "lines": sorted({s["line"] for s in segments}),
        "segments": segments,
        "transfers": transfers,
    }


def format_json(rep):
    return json.dumps(rep, ensure_ascii=False, indent=2)


def format_text(rep):
    L = ["Route: %s -> %s" % (rep["query"]["start"], rep["query"]["end"]),
         "Stations (%d): %s" % (len(rep["path"]), " -> ".join(rep["path"])),
         "Total distance: %d m (%.2f km)" % (rep["distance_m"], rep["distance_km"])]
    for s in rep["segments"]:
        L.append("  [%s] %s -> %s (%d stops, %d m)" % (s["line"], s["from"], s["to"], s["stops"], s["distance"]))
    if rep["transfers"]:
        for t in rep["transfers"]:
            L.append("Transfer at %s: %s -> %s" % (t["station"], t["from_line"], t["to_line"]))
    else:
        L.append("No transfers")
    L.append("Transfers: %d" % rep["transfer_count"])
    L.append("Lines: %s" % ", ".join(rep["lines"]))
    L.append("Estimated time: %.1f minutes" % rep["estimated_time_min"])
    L.append("Fare: %d yuan" % rep["fare"])
    return "\n".join(L)

"""CLI entry point."""
import argparse
import re
import sys

from subway_system import SubwaySystem
from pathfinding import PathfindingConfig
from report import build_report, format_json, format_text

SEP = re.compile(r"[;；,，、]")


def parse_query(line, system):
    """Return (start, end) or raise ValueError with message."""
    parts = [p.strip() for p in SEP.split(line)]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError("Input format incorrect, please use 'Starting Station; Ending Station' format")
    a, b = parts
    if a not in system.station_to_number:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % a)
    if b not in system.station_to_number:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % b)
    if a == b:
        raise ValueError("Starting station cannot be the same as the ending station")
    return a, b


def run_query(system, a, b, config, fmt):
    try:
        route = system.find_route(a, b, config)
    except Exception as e:
        raise RuntimeError("Path calculation failed: %s" % e)
    if route is None:
        raise RuntimeError("No route found between '%s' and '%s' (check transfer limit)" % (a, b))
    try:
        rep = build_report(system, a, b, route)
    except Exception as e:
        raise RuntimeError("Fare calculation failed: %s" % e)
    return format_json(rep) if fmt == "json" else format_text(rep)


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        system = SubwaySystem()
    except Exception as e:
        print("Failed to load data: %s" % e)
        return 1

    if args.check_data:
        problems = system.check_data()
        if problems:
            print("Data check found %d problem(s):" % len(problems))
            for x in problems:
                print(" -", x)
            return 1
        print("Data check passed: %d stations, %d lines." % (
            len(system.station_to_number), len(system.lines())))
        return 0

    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    if args.line_preference:
        cfg.preferred_lines = [x.strip() for x in args.line_preference.split(",") if x.strip()]

    outputs = []
    print("Enter 'Starting Station; Ending Station' (or 'q' to quit).")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if not line:
            continue
        if line.lower() in ("q", "quit", "exit"):
            break
        try:
            a, b = parse_query(line, system)
            out = run_query(system, a, b, cfg, args.format)
        except (ValueError, RuntimeError) as e:
            print(e)
            continue
        print(out)
        outputs.append(out)

    if args.output_file and outputs:
        try:
            with open(args.output_file, "w", encoding="utf-8") as f:
                f.write("\n\n".join(outputs) + "\n")
            print("Results written to %s" % args.output_file)
        except OSError as e:
            print("Cannot write output file: %s" % e)
            return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Core data models: Edge and Graph."""


class Edge:
    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph with adjacency lists.
    Station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_node(self, node):
        self.adjacency.setdefault(node, [])

    def add_edge(self, edge):
        """Add an undirected edge (stored in both directions)."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adjacency[edge.source].append(edge)
        self.adjacency[edge.dest].append(Edge(edge.dest, edge.source, edge.distance, edge.line))

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def nodes(self):
        return list(self.adjacency.keys())

    def has_node(self, node):
        return node in self.adjacency

"""MinPQ and Dijkstra-based ShortestPath."""
import heapq
import itertools


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class MinPQ:
    """Binary-heap based min priority queue."""

    def __init__(self):
        self.heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self.heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self.heap)
        return p, item

    def is_empty(self):
        return not self.heap

    def __len__(self):
        return len(self.heap)


TRANSFER_PENALTY = {"shortest_distance": 0.0, "least_transfer": 1e7, "comprehensive": 1500.0}
PREFERRED_FACTOR = 0.9


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}      # node -> state
        self._info = {}      # state -> (cost, dist, prev_state, edge)
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0.0)
        limit = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        self._info[start] = (0.0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in self._best:
                self._best[node] = st
                self.dist_to[node] = self._info[st][1]
            dist = self._info[st][1]
            for e in self.graph.neighbors(node):
                nt = t
                if line is not None and e.line != line:
                    nt += 1
                if limit is not None and nt > limit:
                    continue
                if limit is None:
                    nt_key = 0
                    if penalty == 0.0:
                        nt_key = 0
                else:
                    nt_key = nt
                w = e.weight * (PREFERRED_FACTOR if e.line in pref else 1.0)
                c = cost + w + (penalty if nt != t else 0.0)
                nst = (e.dest, e.line, nt_key)
                if nst in done:
                    continue
                old = self._info.get(nst)
                if old is None or c < old[0]:
                    self._info[nst] = (c, dist + e.distance, st, e)
                    pq.push(c, nst)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        nodes, edges = [], []
        st = self._best[target]
        while st is not None:
            _, _, prev, e = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math

FARE_RULES = [(6, 3), (12, 4), (22, 5), (32, 6)]


def Sum_money(distance) -> int:
    """Fare in yuan for a distance in meters (km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    for limit, fare in FARE_RULES:
        if km <= limit:
            return fare
    return 6 + int(math.ceil((km - 32) / 20.0))


def calculate_estimated_time(distance, transfer_count) -> float:
    """Estimated minutes: 35 km/h average plus 5 minutes per transfer."""
    if distance < 0 or transfer_count < 0:
        raise ValueError("invalid arguments")
    return round(distance / 1000.0 / 35.0 * 60.0 + transfer_count * 5.0, 1)

"""SubwayNetwork and DataManager."""
import json

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph


class SubwayNetwork:
    def __init__(self, data):
        self.data = data
        self.station_to_number = {}
        self.number_to_station = {}
        self.graph = Graph()
        self.lines = {}
        self._build()

    def _number(self, name):
        if name not in self.station_to_number:
            n = len(self.station_to_number)
            self.station_to_number[name] = n
            self.number_to_station[n] = name
            self.graph.add_node(n)
        return self.station_to_number[name]

    def _build(self):
        for s in self.data.get("stations", []):
            self._number(s["name"])
            self.graph.set_station_lines(s["name"], s.get("lines", []))
        seen = set()
        for c in self.data.get("connections", []):
            a, b = self._number(c["source"]), self._number(c["dest"])
            key = (min(a, b), max(a, b), c.get("line"))
            if key in seen or a == b:
                continue
            seen.add(key)
            self.graph.add_edge(Edge(a, b, c["distance"], c.get("line")))
            for nm in (c["source"], c["dest"]):
                ls = self.graph.station_lines.setdefault(nm, [])
                if c.get("line") not in ls:
                    ls.append(c.get("line"))
            seq = self.lines.setdefault(c.get("line"), [])
            for nm in (c["source"], c["dest"]):
                if nm not in seq:
                    seq.append(nm)

    def has_station(self, name):
        return name in self.station_to_number

    def is_transfer(self, name):
        return len(self.graph.station_lines.get(name, [])) > 1


REPRESENTATIVE = {
    "1号线": "天安门东", "2号线": "雍和宫", "4号线": "西单", "5号线": "东单",
}


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.network = SubwayNetwork(self.data)

    def rebuild(self):
        self.network = SubwayNetwork(self.data)

    def save(self, path=None):
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)

    def check_data(self):
        """Return list of issue strings (empty if data is fine)."""
        issues = []
        names = set()
        for s in self.data.get("stations", []):
            if not s.get("name"):
                issues.append("Station without name")
            if s["name"] in names:
                issues.append("Duplicate station: %s" % s["name"])
            names.add(s["name"])
            if not s.get("lines"):
                issues.append("Station without lines: %s" % s["name"])
        if not names:
            issues.append("No station data")
        linked = set()
        for c in self.data.get("connections", []):
            for k in ("source", "dest"):
                if c[k] not in names:
                    issues.append("Connection references unknown station: %s" % c[k])
                linked.add(c[k])
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0:
                issues.append("Invalid distance %s-%s: %r" % (c["source"], c["dest"], d))
            elif d > 20000:
                issues.append("Suspicious distance %s-%s: %s" % (c["source"], c["dest"], d))
        for n in sorted(names - linked):
            issues.append("Station without connections: %s" % n)
        return issues

    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: len(stations)-1 meters."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                st = {"name": n, "lines": [line_name]}
                self.data["stations"].append(st)
                known[n] = st
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.rebuild()

    def update_station(self, name, new_name=None, lines=None):
        for s in self.data["stations"]:
            if s["name"] == name:
                if lines is not None:
                    s["lines"] = list(lines)
                if new_name:
                    s["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.rebuild()
                return True
        return False

    def update_distance(self, a, b, distance):
        if distance <= 0:
            raise ValueError("distance must be positive")
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                found = True
        if found:
            self.rebuild()
        return found

"""Route building and report generation."""
import json

from fare_calculator import Sum_money, calculate_estimated_time


class Route:
    def __init__(self, stations, edges, total_distance, transfer_count,
                 involved_lines, estimated_time, transfer_stations, segments):
        self.stationSequence = stations
        self.edges = edges
        self.totalDistance = total_distance
        self.transferCount = transfer_count
        self.involvedLines = involved_lines
        self.estimatedTime = estimated_time
        self.transferStations = transfer_stations
        self.segments = segments  # list of (line, [stations])


def build_route(network, sp, target):
    nodes, edges = sp.path_with_edges(target)
    if not nodes:
        raise ValueError("no path")
    names = [network.number_to_station[n] for n in nodes]
    total = sum(e.distance for e in edges)
    lines, segments, transfers = [], [], []
    for i, e in enumerate(edges):
        if e.line not in lines:
            lines.append(e.line)
        if segments and segments[-1][0] == e.line:
            segments[-1][1].append(names[i + 1])
        else:
            if segments:
                transfers.append(names[i])
            segments.append((e.line, [names[i], names[i + 1]]))
    t = len(transfers)
    return Route(names, edges, total, t, lines,
                 calculate_estimated_time(total, t), transfers, segments)


def format_distance(meters):
    if meters < 1000:
        return "%d meters" % meters
    return "%.2f km" % (meters / 1000.0)


def text_report(route, fare, start, end):
    out = ["Route: %s -> %s" % (start, end),
           "Stations (%d): %s" % (len(route.stationSequence), " -> ".join(route.stationSequence)),
           "Total distance: %s (%.2f km)" % (format_distance(route.totalDistance), route.totalDistance / 1000.0)]
    for line, sts in route.segments:
        out.append("  [%s] %s -> %s (%d stops)" % (line, sts[0], sts[-1], len(sts) - 1))
    if route.transferStations:
        out.append("Transfers: %d at %s" % (route.transferCount, ", ".join(route.transferStations)))
    else:
        out.append("Transfers: 0")
    out.append("Lines: %s" % ", ".join(route.involvedLines))
    out.append("Estimated time: %.1f minutes" % route.estimatedTime)
    out.append("Fare: %d yuan" % fare)
    return "\n".join(out)


def json_report(route, fare, start, end, mode):
    return json.dumps({
        "query": {"start": start, "end": end, "mode": mode},
        "path": route.stationSequence,
        "distance_meters": route.totalDistance,
        "distance_km": round(route.totalDistance / 1000.0, 2),
        "fare": fare,
        "estimated_time": route.estimatedTime,
        "transfer_count": route.transferCount,
        "transfer_stations": route.transferStations,
        "lines": route.involvedLines,
    }, ensure_ascii=False, indent=2)


def make_report(route, fare, start, end, mode, fmt="text"):
    if fmt == "json":
        return json_report(route, fare, start, end, mode)
    return text_report(route, fare, start, end)

"""CLI entry point for the Beijing Subway fare system."""
import argparse
import sys

from fare_calculator import Sum_money
from network import DataManager
from pathfinding import PathfindingConfig, ShortestPath
from report import build_route, make_report


def parse_input(text):
    text = text.replace("；", ";")
    parts = [p.strip() for p in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        return None
    return parts


def run_query(dm, start, end, args):
    net = dm.network
    if not net.has_station(start):
        return "Starting station '%s' does not exist, please check the station name" % start
    if not net.has_station(end):
        return "Ending station '%s' does not exist, please check the station name" % end
    if start == end:
        return "Starting station cannot be the same as the ending station"
    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    cfg.preferred_lines = [args.line_preference] if args.line_preference else []
    try:
        s, t = net.station_to_number[start], net.station_to_number[end]
        sp = ShortestPath(net.graph, s, cfg)
        route = build_route(net, sp, t)
    except Exception as e:
        return "Path calculation failed: %s" % (
            "no route found (check the transfer limit)" if str(e) == "no path" else e)
    try:
        fare = Sum_money(route.totalDistance)
    except Exception as e:
        return "Fare calculation failed: %s" % e
    return make_report(route, fare, start, end, args.mode, args.format)


def emit(msg, args):
    print(msg)
    if args.output_file:
        try:
            with open(args.output_file, "a", encoding="utf-8") as f:
                f.write(msg + "\n")
        except OSError as e:
            print("Cannot write output file: %s" % e)


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)
    try:
        dm = DataManager()
    except Exception as e:
        print("Failed to load subway data: %s" % e)
        return 1
    if args.check_data:
        issues = dm.check_data()
        if issues:
            print("Data check found %d issue(s):" % len(issues))
            for i in issues:
                print(" - " + i)
            return 1
        print("Data check passed: %d stations, %d connections" % (
            len(dm.data["stations"]), len(dm.data["connections"])))
        return 0
    print("Beijing Subway fare query. Input 'Starting Station; Ending Station' (q to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        line = line.strip()
        if line.lower() in ("q", "quit", "exit"):
            break
        if not line:
            continue
        parts = parse_input(line)
        if parts is None:
            print("Input format incorrect, please use 'Starting Station; Ending Station' format")
            continue
        emit(run_query(dm, parts[0], parts[1], args), args)
    return 0


if __name__ == "__main__":
    sys.exit(main())

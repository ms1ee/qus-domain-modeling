"""Order data model."""
import itertools

_counter = itertools.count(1)


def to_center(v):
    """Convert a coordinate to its grid cell center."""
    v = max(0, min(800, int(v)))
    cell = min(v // 100, 7)
    return cell * 100 + 50


class Order:
    """A delivery order; positions are converted to grid cell centers."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_counter)
        self.res_pos = (to_center(res_pos[0]), to_center(res_pos[1]))
        self.cus_pos = (to_center(cus_pos[0]), to_center(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = 0
        self.end_time = None
        self.distance = (abs(self.res_pos[0] - self.cus_pos[0])
                         + abs(self.res_pos[1] - self.cus_pos[1]))

    def assign_to(self, rider_id):
        """Assign this order to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

    def to_dict(self):
        return {"id": self.id, "res_pos": list(self.res_pos),
                "cus_pos": list(self.cus_pos), "status": self.status,
                "rider_id": self.rider_id, "start_time": self.start_time,
                "end_time": self.end_time, "distance": self.distance}

"""Rider model and movement."""
import globalvar


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 435
        self.y = 385
        self.speed = 10
        self.state = 0
        self.orders = []
        self.special_case = False
        self.log = []
        self._log("spawn", None)

    def _log(self, event, order_id):
        self.log.append({"event": event, "order_id": order_id,
                         "time": globalvar.get_value("time", 0),
                         "x": self.x, "y": self.y})

    def _step_to(self, tx, ty):
        """Move up to speed units: horizontal first, then vertical."""
        budget = self.speed
        dx = tx - self.x
        if dx and budget > 0:
            s = min(abs(dx), budget)
            self.x += s if dx > 0 else -s
            budget -= s
        dy = ty - self.y
        if dy and budget > 0:
            s = min(abs(dy), budget)
            self.y += s if dy > 0 else -s
        return self.x == tx and self.y == ty

    def move(self):
        """Advance one tick."""
        if not self.orders:
            self.state = 0
            return
        order = self.orders[0]
        if self.state == 0:
            self.state = 1
            self._log("accepted_order", order.id)
            return
        if self.state == 1:
            if self._step_to(*order.res_pos):
                self.state = 2
                self._log("arrived_at_merchant", order.id)
        elif self.state == 2:
            if self._step_to(*order.cus_pos):
                order.status = "completed"
                order.end_time = globalvar.get_value("time", 0)
                self.orders.pop(0)
                self.state = 0
                self._log("delivered_to_customer", order.id)

"""Order assignment."""
import math
import globalvar
from rider import Rider


def find_best_rider_for_order(order):
    """Return the closest rider (Euclidean) to the restaurant, or None."""
    riders = globalvar.get_value("riders", []) or []
    best = None
    best_key = None
    for r in riders:
        d = math.hypot(r.x - order.res_pos[0], r.y - order.res_pos[1])
        key = (d, len(r.orders))
        if best_key is None or key < best_key:
            best, best_key = r, key
    return best


def recruit_if_needed(pending):
    """Recruit a rider if funds allow and pending orders exceed idle riders."""
    riders = globalvar.get_value("riders")
    funds = globalvar.get_value("funds", 0)
    idle = sum(1 for r in riders if r.state == 0 and not r.orders)
    if funds >= 300 and pending > idle:
        rid = globalvar.get_value("rider_id_counter", 0) + 1
        globalvar.set_value("rider_id_counter", rid)
        globalvar.set_value("funds", funds - 300)
        riders.append(Rider(rid))
        return True
    return False


def dispatch(order):
    """Register an order, recruit if needed, assign. Returns (rider, hired)."""
    orders = globalvar.get_value("orders")
    order.start_time = globalvar.get_value("time", 0)
    orders.append(order)
    hired = recruit_if_needed(1)
    rider = find_best_rider_for_order(order)
    if rider is not None:
        order.assign_to(rider.rider_id)
        rider.orders.append(order)
    return rider, hired

"""Display helpers."""
import globalvar

STATE_NAMES = {0: "idle", 1: "picking up", 2: "delivering"}


def render_map(rider=None, order=None):
    """ASCII 8x8 map; route drawn for the rider's first order."""
    grid = [["." for _ in range(8)] for _ in range(8)]
    for r in globalvar.get_value("riders", []):
        grid[min(r.y // 100, 7)][min(r.x // 100, 7)] = "R"
    if rider is not None and rider.orders:
        o = rider.orders[0]
        pts = [(rider.x, rider.y), o.res_pos, o.cus_pos]
        for (x1, y1), (x2, y2) in zip(pts, pts[1:]):
            for x in range(min(x1, x2) // 100, max(x1, x2) // 100 + 1):
                c = grid[min(y1 // 100, 7)][min(x, 7)]
                if c == ".":
                    grid[min(y1 // 100, 7)][min(x, 7)] = "#"
            for y in range(min(y1, y2) // 100, max(y1, y2) // 100 + 1):
                c = grid[min(y, 7)][min(x2 // 100, 7)]
                if c == ".":
                    grid[min(y, 7)][min(x2 // 100, 7)] = "#"
    for o in globalvar.get_value("orders", []):
        if o.status != "completed":
            rx, ry = o.res_pos[0] // 100, o.res_pos[1] // 100
            cx, cy = o.cus_pos[0] // 100, o.cus_pos[1] // 100
            grid[ry][rx] = "S"
            grid[cy][cx] = "C"
    return "\n".join(" ".join(row) for row in grid)


def status_text():
    lines = ["Time: %d  Funds: %d  Riders: %d" % (
        globalvar.get_value("time", 0), globalvar.get_value("funds", 0),
        len(globalvar.get_value("riders", [])))]
    for r in globalvar.get_value("riders", []):
        lines.append("Rider %d at (%d,%d) %s, orders: %s" % (
            r.rider_id, r.x, r.y, STATE_NAMES[r.state],
            [o.id for o in r.orders]))
    return "\n".join(lines)


def orders_text():
    lines = []
    for o in globalvar.get_value("orders", []):
        taken = (o.end_time - o.start_time) if o.end_time is not None else "-"
        lines.append("Order %d start=%d res=%s cus=%s dist=%d time=%s "
                     "rider=%s status=%s" % (
                         o.id, o.start_time, o.res_pos, o.cus_pos,
                         o.distance, taken, o.rider_id, o.status))
    return "\n".join(lines) or "No orders."


def riders_text():
    lines = []
    for r in globalvar.get_value("riders", []):
        for e in r.log:
            lines.append("Rider %d t=%d %s order=%s at (%d,%d)" % (
                r.rider_id, e["time"], e["event"], e["order_id"],
                e["x"], e["y"]))
    return "\n".join(lines) or "No riders."

"""Command-line entry point."""
import json
import globalvar
import system
import display
from order_model import Order
import order as order_mod


def tick():
    globalvar.set_value("time", globalvar.get_value("time", 0) + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def parse_coords(parts):
    if len(parts) != 4:
        raise ValueError("Need exactly four integers.")
    vals = [int(p) for p in parts]
    if any(v < 0 or v > 800 for v in vals):
        raise ValueError("Coordinates must be within 0-800.")
    return vals


def save(path="state.json"):
    data = {"funds": globalvar.get_value("funds"),
            "orders": [o.to_dict() for o in globalvar.get_value("orders")],
            "riders": [{"id": r.rider_id, "x": r.x, "y": r.y,
                        "state": r.state, "log": r.log}
                       for r in globalvar.get_value("riders")]}
    with open(path, "w") as f:
        json.dump(data, f, indent=2)


def main():
    system.init_system()
    while globalvar.get_value("running"):
        try:
            line = input("> ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        parts = line.split()
        if not parts:
            tick()
            continue
        cmd = parts[0].lower()
        try:
            if cmd in ("quit", "exit", "q"):
                globalvar.set_value("running", False)
            elif cmd in ("start", "s"):
                v = parse_coords(parts[1:])
                o = Order(v[:2], v[2:])
                rider, hired = order_mod.dispatch(o)
                print("Restaurant:", o.res_pos, "Customer:", o.cus_pos)
                print("New rider added:", hired)
                if rider:
                    eta = (abs(rider.x - o.res_pos[0])
                           + abs(rider.y - o.res_pos[1]) + o.distance)
                    print("Rider %d speed %d, est. time %d ticks" % (
                        rider.rider_id, rider.speed,
                        -(-eta // rider.speed)))
                    print(display.render_map(rider))
                else:
                    print("No rider available (insufficient funds).")
                tick()
            elif cmd == "status":
                print(display.status_text())
                print(display.render_map())
            elif cmd == "orders":
                print(display.orders_text())
            elif cmd == "riders":
                print(display.riders_text())
            elif cmd == "tick":
                n = int(parts[1]) if len(parts) > 1 else 1
                for _ in range(n):
                    tick()
            else:
                print("Unknown command. Use start/status/orders/riders/"
                      "tick/quit.")
        except ValueError as e:
            print("Error:", e)
    try:
        save()
    except OSError:
        pass


if __name__ == "__main__":
    main()

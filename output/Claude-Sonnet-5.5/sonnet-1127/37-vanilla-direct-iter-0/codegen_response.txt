from collections import OrderedDict
from lottery_system.models import User


class ListManager:
    DEFAULT = [
        ("Zhang San", "Technology Department"), ("Li Si", "Technology Department"),
        ("Wang Wu", "Technology Department"), ("Zhao Liu", "Product Department"),
        ("Sun Qi", "Product Department"), ("Zhou Ba", "Product Department"),
        ("Wu Jiu", "Marketing Department"), ("Zheng Shi", "Marketing Department"),
        ("Chen Qixian", "Cheng Ru group"), ("Jin Duo", "Cheng Ru group"),
        ("Ma Jiaqi", "Operations Department"), ("Li Bin", "Operations Department"),
        ("Liu Yi", "Cheng Ru group"), ("Huang Er", "Cheng Ru group"),
        ("Xu San", "HR Department"), ("Gao Si", "HR Department"),
    ]

    def load_default_list(self):
        return [User(n, d) for n, d in self.DEFAULT], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"File Error: cannot read '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        for name, q in [("Third Prize - Movie Tickets", 3),
                        ("Second Prize - Shopping Cards", 2),
                        ("First Prize - 1 Day Annual Leave", 1)]:
            try:
                self.add_prize(name, q)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer or None)
        self._prizes.append(p)
        return p

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, win_time=None):
        self.user = user
        self.prize_name = prize_name
        self.win_time = win_time or datetime.now()


def _level(name):
    n = name.lower()
    for i, k in enumerate(("special", "first", "second", "third")):
        if k in n:
            return i
    return 99


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        w = Winner(user, name)
        self.winners.append(w)
        return w

    def get_winners_by_prize(self, prizes):
        ordered = sorted(enumerate(prizes), key=lambda t: (_level(t[1].name), t[0]))
        result = {}
        for _, p in ordered:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for pname, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{pname},"
                                f"{w.win_time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import random

CHECK_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


def _is_cr(dept):
    return "cheng ru" in dept.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {u.name for u in self.winners}
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if _is_cr(u.department)]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_round(self, prize, round_winners):
        if prize.associated_drawer in CHECK_DRAWERS:
            self.last_two_draw_results.append(
                any(_is_cr(u.department) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

    @staticmethod
    def describe(prize):
        d = f"Quantity: {prize.quantity}"
        d += f", Avoid groups: {', '.join(prize.avoid_groups) or 'None'}"
        if prize.associated_drawer:
            d += f", Drawer: {prize.associated_drawer}"
            if prize.associated_drawer == SPECIAL_DRAWER:
                d += " (only previous winners excluded)"
        return d

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt file: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        if errs and input("Errors found. Remove invalid lines and continue? (y/n): ").lower() != "y":
            if input("Terminate upload and retry? (y/n): ").lower() == "y":
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name} ({d.department})")
        if valid:
            return valid
        print("No valid users.")


def setup_prizes(pm):
    if input("Load default prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        try:
            name = input("Name: ").strip()
            q = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            drawer = input("Associated drawer (optional): ").strip() or None
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        print(f"{i}. {p.name} x{p.quantity}")
    s = input("New order (space separated numbers, Enter to keep): ").split()
    if s:
        try:
            ok = pm.reorder_prizes([int(x) for x in s])
        except ValueError:
            ok = False
        print("Reordered." if ok else "Invalid order, keeping default.")


def run_round(lot, rm, prize):
    print(f"\n=== {prize.name} === {lot.describe(prize)}")
    got = []
    i = 1
    while i <= prize.quantity:
        cmd = input(f"Press Enter to draw winner {i} (pause/terminate): ").strip().lower()
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        if cmd == "terminate":
            print("Round terminated.")
            break
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        rm.add_winner(u, prize)
        got.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, got)
    print("Round results:", ", ".join(u.name for u in got) or "none")


def main():
    lm = ListManager()
    users = load_users(lm)
    for g, n in lm.get_group_info(users).items():
        print(f"{g}: {n}")
    pm = PrizeManager()
    setup_prizes(pm)
    lot, rm = LotteryManager(users), ResultManager()
    prizes = pm.get_prizes()
    for p in prizes:
        run_round(lot, rm, p)
    print("\n=== Final Winners ===")
    for pname, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{pname}")
    print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()

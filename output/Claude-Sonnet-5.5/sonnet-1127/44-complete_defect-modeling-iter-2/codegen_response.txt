"""Core data structures: Edge, Graph, MinPQ."""
import heapq
import itertools


class Edge:
    """Directed view of a connection between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph; station numbers are assigned by the caller."""

    def __init__(self):
        self.adjacency = {}
        self.station_lines = {}
        self.station_to_number = {}
        self.number_to_station = {}

    def add_station(self, name, number):
        self.station_to_number[name] = number
        self.number_to_station[number] = name
        self.adjacency.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add an undirected edge (both directions)."""
        for e in (edge, edge.reversed()):
            lst = self.adjacency.setdefault(e.source, [])
            if not any(x.dest == e.dest and x.line == e.line for x in lst):
                lst.append(e)
            self.adjacency.setdefault(e.dest, [])

    def remove_edges(self, a, b, line=None):
        for x, y in ((a, b), (b, a)):
            self.adjacency[x] = [e for e in self.adjacency.get(x, [])
                                 if not (e.dest == y and (line is None or e.line == line))]

    def neighbors(self, node):
        return self.adjacency.get(node, [])

    def nodes(self):
        return list(self.adjacency.keys())

    def is_transfer(self, name):
        return len(self.station_lines.get(name, [])) > 1


class MinPQ:
    """Heap based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)

"""Dijkstra shortest path with transfer limits and line preference."""
from models import MinPQ

TRANSFER_PENALTY = {
    "shortest_distance": 0.0,
    "least_transfer": 1000000.0,
    "comprehensive": 1000.0,
}


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best_state = {}
        self._pred = {}
        self._run()

    def _run(self):
        cfg = self.config
        penalty = TRANSFER_PENALTY.get(cfg.mode, 0.0)
        limit = cfg.max_transfers
        pref = set(cfg.preferred_lines or [])
        start = (self.source, None, 0)
        cost = {start: 0.0}
        real = {start: 0}
        self._pred[start] = None
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            c, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, t = st
            if node not in self._best_state:
                self._best_state[node] = st
                self.dist_to[node] = real[st]
            for e in self.graph.neighbors(node):
                nt = t
                extra = 0.0
                if line is not None and e.line != line:
                    if limit is not None:
                        nt = t + 1
                        if nt > limit:
                            continue
                    extra += penalty
                    if limit is None and penalty == 0.0:
                        pass
                w = e.weight * (0.9 if e.line in pref else 1.0)
                # tiny tie-break so transfers are never free when counted
                nc = c + w + extra
                ns = (e.dest, e.line, nt)
                if ns not in cost or nc < cost[ns]:
                    cost[ns] = nc
                    real[ns] = real[st] + e.weight
                    self._pred[ns] = (st, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best_state

    def path_with_edges(self, target):
        if target not in self._best_state:
            return [], []
        st = self._best_state[target]
        edges = []
        while self._pred[st] is not None:
            prev, e = self._pred[st]
            edges.append(e)
            st = prev
        edges.reverse()
        return [self.source] + [e.dest for e in edges], edges

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for a, b in zip(edges, edges[1:]) if a.line != b.line)

"""Fare calculation."""
import math

FARE_TIERS = [(6, 3), (12, 4), (22, 5), (32, 6)]
EXTENDED_STEP_KM = 20
EXTENDED_STEP_FARE = 1
AVG_SPEED_KMH = 35.0
TRANSFER_MINUTES = 5.0


def Sum_money(distance):
    """Fare in yuan for a distance in meters (km rounded up)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    for limit, fare in FARE_TIERS:
        if km <= limit:
            return fare
    return FARE_TIERS[-1][1] + math.ceil((km - 32) / EXTENDED_STEP_KM) * EXTENDED_STEP_FARE


def calculate_estimated_time(distance, transfer_count):
    """Estimated minutes for distance in meters."""
    return round(distance / 1000.0 / AVG_SPEED_KMH * 60.0 + transfer_count * TRANSFER_MINUTES, 1)

"""Data management: build graph, integrity check, updates."""
from data_loader import DATA_FILE_PATH, load_subway_data
from models import Graph, Edge

REPRESENTATIVE_STATIONS = ["西直门", "国贸", "北京站", "天安门东", "东直门", "公主坟"]


class DataManager:
    def __init__(self, data_file=DATA_FILE_PATH):
        self.data_file = data_file
        self.data = load_subway_data(data_file)
        self.graph = None
        self.build_graph()

    def build_graph(self):
        g = Graph()
        num = 0
        for st in self.data.get("stations", []):
            if st["name"] not in g.station_to_number:
                g.add_station(st["name"], num)
                num += 1
            g.set_station_lines(st["name"], st.get("lines", []))
        for c in self.data.get("connections", []):
            for n in (c["source"], c["dest"]):
                if n not in g.station_to_number:
                    g.add_station(n, num)
                    num += 1
                    g.set_station_lines(n, [c["line"]])
            g.add_edge(Edge(g.station_to_number[c["source"]], g.station_to_number[c["dest"]],
                            c["distance"], c["line"]))
        self.graph = g
        return g

    def line_names(self):
        names = []
        for st in self.data.get("stations", []):
            for l in st.get("lines", []):
                if l not in names:
                    names.append(l)
        for c in self.data.get("connections", []):
            if c["line"] not in names:
                names.append(c["line"])
        return names

    def resolve_line(self, text):
        if not text:
            return None
        names = self.line_names()
        if text in names:
            return text
        low = text.lower().replace(" ", "")
        for n in names:
            if n.lower().replace(" ", "") == low:
                return n
        return text

    def check_data(self):
        """Return (ok, messages)."""
        msgs = []
        ok = True
        names = {s["name"] for s in self.data.get("stations", [])}
        if not names:
            return False, ["No station data"]
        msgs.append("Stations: %d, lines: %d" % (len(names), len(self.line_names())))
        missing = [s for s in REPRESENTATIVE_STATIONS if s not in names]
        if missing:
            msgs.append("Warning: representative stations missing: " + ", ".join(missing))
        conns = self.data.get("connections", [])
        msgs.append("Connections: %d" % len(conns))
        for c in conns:
            for k in ("source", "dest"):
                if c[k] not in names:
                    ok = False
                    msgs.append("Connection refers to unknown station: %s" % c[k])
            d = c.get("distance")
            if not isinstance(d, (int, float)) or d <= 0 or d > 30000:
                ok = False
                msgs.append("Suspicious distance %r between %s and %s" % (d, c["source"], c["dest"]))
        connected = set()
        for c in conns:
            connected.add(c["source"])
            connected.add(c["dest"])
        isolated = names - connected
        if isolated:
            ok = False
            msgs.append("Stations without connections: " + ", ".join(sorted(isolated)))
        msgs.append("Data check " + ("passed" if ok else "FAILED"))
        return ok, msgs

    # ---- update mechanisms (in-memory; optionally saved) ----
    def add_line(self, line_name, stations, distances):
        """stations: ordered names; distances: meters between consecutive stations."""
        if len(distances) != len(stations) - 1:
            raise ValueError("distances must have len(stations)-1 items")
        known = {s["name"]: s for s in self.data["stations"]}
        for n in stations:
            if n in known:
                if line_name not in known[n]["lines"]:
                    known[n]["lines"].append(line_name)
            else:
                self.data["stations"].append({"name": n, "lines": [line_name]})
        for a, b, d in zip(stations, stations[1:], distances):
            self.data["connections"].append({"source": a, "dest": b, "distance": d, "line": line_name})
        self.build_graph()

    def update_station(self, name, new_name=None, lines=None):
        for st in self.data["stations"]:
            if st["name"] == name:
                if lines is not None:
                    st["lines"] = list(lines)
                if new_name:
                    st["name"] = new_name
                    for c in self.data["connections"]:
                        if c["source"] == name:
                            c["source"] = new_name
                        if c["dest"] == name:
                            c["dest"] = new_name
                self.build_graph()
                return True
        return False

    def update_distance(self, a, b, distance):
        found = False
        for c in self.data["connections"]:
            if {c["source"], c["dest"]} == {a, b}:
                c["distance"] = distance
                found = True
        if found:
            self.build_graph()
        return found

    def save(self, path=None):
        import json
        with open(path or self.data_file, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=1)

"""Route construction and report generation."""
import json
from fare_calculator import Sum_money, calculate_estimated_time


class TransferInfo:
    def __init__(self, station_name, line_names):
        self.station_name = station_name
        self.line_names = line_names


class Route:
    def __init__(self, stations, total_distance, transfers, lines, segments, estimated_time):
        self.station_sequence = stations
        self.total_distance = total_distance
        self.transfers = transfers
        self.transfer_count = len(transfers)
        self.involved_lines = lines
        self.segments = segments  # list of (line, [stations])
        self.estimated_time = estimated_time


def build_route(graph, nodes, edges):
    names = [graph.number_to_station[n] for n in nodes]
    total = sum(e.distance for e in edges)
    transfers, lines, segments = [], [], []
    for i, e in enumerate(edges):
        if e.line not in lines:
            lines.append(e.line)
        if segments and segments[-1][0] == e.line:
            segments[-1][1].append(names[i + 1])
        else:
            segments.append((e.line, [names[i], names[i + 1]]))
        if i + 1 < len(edges) and edges[i + 1].line != e.line:
            transfers.append(TransferInfo(names[i + 1], [e.line, edges[i + 1].line]))
    return Route(names, total, transfers, lines, segments,
                 calculate_estimated_time(total, len(transfers)))


def format_distance(meters):
    if meters < 1000:
        return "%d m" % meters
    return "%.2f km (%d m)" % (meters / 1000.0, meters)


def text_report(start, end, route, fare):
    out = ["Route: %s -> %s" % (start, end),
           "Stations (%d): %s" % (len(route.station_sequence), " -> ".join(route.station_sequence)),
           "Total distance: %s" % format_distance(route.total_distance)]
    for line, sts in route.segments:
        out.append("  [%s] %s -> %s (%d stops)" % (line, sts[0], sts[-1], len(sts) - 1))
    if route.transfers:
        for t in route.transfers:
            out.append("Transfer at %s: %s" % (t.station_name, " -> ".join(t.line_names)))
    else:
        out.append("No transfers")
    out.append("Lines: %s" % ", ".join(route.involved_lines))
    out.append("Transfers: %d" % route.transfer_count)
    out.append("Estimated time: %.1f minutes" % route.estimated_time)
    out.append("Fare: %d yuan" % fare)
    return "\n".join(out)


def json_report(start, end, mode, route, fare):
    d = {
        "query": {"start": start, "end": end, "mode": mode},
        "path": route.station_sequence,
        "distance_m": route.total_distance,
        "distance_km": round(route.total_distance / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": route.estimated_time,
        "transfer_count": route.transfer_count,
        "transfers": [{"station": t.station_name, "lines": t.line_names} for t in route.transfers],
        "lines": route.involved_lines,
    }
    return json.dumps(d, ensure_ascii=False, indent=2)


def generate(start, end, mode, route, fmt):
    fare = Sum_money(route.total_distance)
    if fmt == "json":
        return json_report(start, end, mode, route, fare)
    return text_report(start, end, route, fare)

"""Command-line interface."""
import argparse
import sys

from data_manager import DataManager
from pathfinding import ShortestPath, PathfindingConfig
import report

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def parse_args(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file", default=None)
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int, default=None)
    p.add_argument("--line_preference", default=None)
    p.add_argument("--check_data", action="store_true")
    return p.parse_args(argv)


def parse_query(text):
    """Return (start, end) or raise ValueError with a message."""
    text = text.replace("；", ";")
    parts = [x.strip() for x in text.split(";")]
    if len(parts) != 2 or not parts[0] or not parts[1]:
        raise ValueError(FORMAT_ERR)
    return parts[0], parts[1]


def run_query(dm, args, text):
    start, end = parse_query(text)
    g = dm.graph
    if start not in g.station_to_number:
        raise ValueError("Starting station '%s' does not exist, please check the station name" % start)
    if end not in g.station_to_number:
        raise ValueError("Ending station '%s' does not exist, please check the station name" % end)
    if start == end:
        raise ValueError("Starting station cannot be the same as the ending station")
    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    if args.line_preference:
        cfg.preferred_lines = [dm.resolve_line(args.line_preference)]
    try:
        sp = ShortestPath(g, g.station_to_number[start], cfg)
        nodes, edges = sp.path_with_edges(g.station_to_number[end])
    except Exception as e:
        raise ValueError("Path calculation failed: %s" % e)
    if not edges:
        raise ValueError("No route found between '%s' and '%s' under the given constraints" % (start, end))
    try:
        route = report.build_route(g, nodes, edges)
        return report.generate(start, end, args.mode, route, args.format)
    except Exception as e:
        raise ValueError("Fare calculation failed: %s" % e)


def main(argv=None):
    args = parse_args(argv)
    try:
        dm = DataManager()
    except Exception as e:
        print("Failed to load subway data: %s" % e)
        return 1
    if args.check_data:
        ok, msgs = dm.check_data()
        print("\n".join(msgs))
        return 0 if ok else 1
    print("Beijing Subway fare calculator. Enter 'Starting Station; Ending Station' (q to quit).")
    while True:
        try:
            text = input("> ").strip()
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if not text:
            continue
        if text.lower() in ("q", "quit", "exit"):
            break
        try:
            result = run_query(dm, args, text)
        except ValueError as e:
            print(e)
            continue
        except Exception as e:
            print("Unexpected error: %s" % e)
            continue
        print(result)
        if args.output_file:
            try:
                with open(args.output_file, "a", encoding="utf-8") as f:
                    f.write(result + "\n\n")
            except OSError as e:
                print("Cannot write output file: %s" % e)
    return 0


if __name__ == "__main__":
    sys.exit(main())

from lottery_system.models import User


class ListManager:
    DEFAULT_DATA = [
        ("Zhang San", "Technology Department"),
        ("Li Si", "Technology Department"),
        ("Wang Wu", "Technology Department"),
        ("Zhao Liu", "Product Department"),
        ("Sun Qi", "Product Department"),
        ("Zhou Ba", "Product Department"),
        ("Wu Jiu", "Cheng Ru group"),
        ("Zheng Shi", "Cheng Ru group"),
        ("Feng Yi", "Cheng Ru group"),
        ("Chen Qixian", "Operations Department"),
        ("Jin Duo", "Operations Department"),
        ("Ma Jiaqi", "Marketing Department"),
        ("Li Bin", "Marketing Department"),
        ("Han Mei", "Human Resources"),
        ("Lin Tao", "Human Resources"),
    ]

    def load_default_list(self):
        return [User(n, d) for n, d in self.DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except OSError as e:
            return [], [f"Error: cannot read file '{path}': {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Format Error: Line {i}: '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, q in defaults:
            try:
                self.add_prize(name, q)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        try:
            quantity = int(quantity)
        except (TypeError, ValueError):
            raise ValueError("Quantity must be an integer")
        if not name or quantity <= 0:
            raise ValueError("Invalid prize name or quantity")
        self.prizes.append(Prize(name, quantity, list(avoid_groups or []),
                                 associated_drawer or None))

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        if not isinstance(order, (list, tuple)) or len(order) != n:
            return False
        if sorted(order) != list(range(1, n + 1)):
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "Cheng Ru group"
SPECIAL_DRAWER = "Li Bin"
TRACKED_DRAWERS = ("Chen Qixian", "Jin Duo")
PROTECTED_DRAWER = "Ma Jiaqi"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _available(self):
        won = {w.name for w in self.winners}
        return [u for u in self.users if u.name not in won]

    def _get_eligible_pool(self, prize):
        pool = self._available()
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (drawer == PROTECTED_DRAWER and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if u.department == CHENG_RU]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_round(self, prize, round_winners):
        drawer = prize.associated_drawer
        if drawer in TRACKED_DRAWERS:
            self.last_two_draw_results.append(
                any(u.department == CHENG_RU for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]
        elif drawer == PROTECTED_DRAWER:
            self.last_two_draw_results = []

import datetime
import os


class Winner:
    def __init__(self, user, prize_name=""):
        self.user = user
        self.prize_name = prize_name


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        fname = f"awards_result_{datetime.date.today().strftime('%Y%m%d')}.txt"
        path = os.path.join(os.getcwd(), fname)
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in grouped.get(p.name, []):
                        f.write(f"{w.user.name},{w.user.department},{p.name}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("1) Default list  2) Custom txt list: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        if errs and input("Errors found. Remove invalid data and continue? (y/n): ").lower() != "y":
            if c == "2":
                print("Upload terminated.")
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate name removed: {d.name}")
        if valid:
            return valid
        print("No valid users.")


def setup_prizes(pm):
    if input("Load default prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        name = input("Prize name: ").strip()
        q = input("Quantity: ").strip()
        avoid = [g.strip() for g in input("Avoid groups (comma separated, optional): ").split(",") if g.strip()]
        drawer = input("Associated drawer (optional): ").strip() or None
        try:
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print(e)
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        print(f"{i}. {p.name} x{p.quantity}")
    s = input("New order (e.g. 2 1 3), blank to keep: ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, keeping default.")
        except ValueError:
            print("Invalid order, keeping default.")


def run_round(lot, res, prize):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | drawer {prize.associated_drawer or '-'} "
          f"| avoid {', '.join(prize.avoid_groups) or 'none'} ===")
    won = []
    i = 1
    while i <= prize.quantity:
        cmd = input("Enter to draw (pause/terminate): ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while input("Paused. Type 'continue': ").strip().lower() != "continue":
                pass
            continue
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        res.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, won)


def main():
    lm = ListManager()
    users = load_users(lm)
    for g, n in lm.get_group_info(users).items():
        print(f"{g}: {n}")
    pm = PrizeManager()
    setup_prizes(pm)
    prizes = pm.get_prizes()
    lot = LotteryManager(users)
    res = ResultManager()
    for p in prizes:
        run_round(lot, res, p)
    print("\n=== Complete winner list ===")
    for name, ws in res.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    print(res.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()

"""Graph data structures."""


class Edge:
    """Undirected edge between two station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.line = line

    @property
    def weight(self):
        return self.distance

    def reversed(self):
        return Edge(self.dest, self.source, self.distance, self.line)

    def __repr__(self):
        return "Edge(%r, %r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted undirected graph using adjacency lists keyed by station number."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, n):
        self.adj.setdefault(n, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        """Add edge in both directions."""
        self.add_node(edge.source)
        self.add_node(edge.dest)
        self.adj[edge.source].append(edge)
        self.adj[edge.dest].append(edge.reversed())

    def neighbors(self, n):
        return self.adj.get(n, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values()) // 2

"""Fare and time calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters (left-closed, right-open tiers)."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    d = math.ceil(distance)
    if d < 6000:
        return 3
    if d < 12000:
        return 4
    if d < 22000:
        return 5
    if d < 32000:
        return 6
    return 7 + (d - 32000) // 20000


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km plus 5 per transfer. distance in meters."""
    return distance / 1000.0 * 1.5 + transfer_count * 5

"""Dijkstra shortest path with priority queue."""
import heapq
import itertools

TRANSFER_PENALTY = 1000  # meters-equivalent used in comprehensive mode
PREFERENCE_FACTOR = 0.9


class MinPQ:
    """Heap based minimum priority queue."""

    def __init__(self):
        self._h = []
        self._c = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._h, (priority, next(self._c), item))

    def pop(self):
        p, _, item = heapq.heappop(self._h)
        return p, item

    def __len__(self):
        return len(self._h)

    def is_empty(self):
        return not self._h


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}   # node -> state
        self._info = {}   # state -> (prev_state, edge, dist, transfers)
        self._run()

    def _edge_cost(self, edge, transferred):
        c = float(edge.distance)
        if edge.line in self.config.preferred_lines:
            c *= PREFERENCE_FACTOR
        if transferred:
            mode = self.config.mode
            if mode == "least_transfer":
                c += 1e7
            elif mode == "comprehensive":
                c += TRANSFER_PENALTY
        return c

    def _run(self):
        limit = self.config.max_transfers
        start = (self.source, None, 0)
        self._info[start] = (None, None, 0, 0)
        costs = {start: 0.0}
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, st = pq.pop()
            if st in done:
                continue
            done.add(st)
            node, line, _ = st
            _, _, dist, tr = self._info[st]
            if node not in self._best:
                self._best[node] = st
                self.dist_to[node] = dist
            for e in self.graph.neighbors(node):
                transferred = line is not None and e.line != line
                ntr = tr + (1 if transferred else 0)
                if limit is not None and ntr > limit:
                    continue
                ns = (e.dest, e.line, ntr if limit is not None else 0)
                if ns in done:
                    continue
                nc = cost + self._edge_cost(e, transferred)
                if nc < costs.get(ns, float("inf")):
                    costs[ns] = nc
                    self._info[ns] = (st, e, dist + e.distance, ntr)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        nodes, edges = [], []
        st = self._best[target]
        while st is not None:
            prev, e, _, _ = self._info[st]
            nodes.append(st[0])
            if e is not None:
                edges.append(e)
            st = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        _, edges = self.path_with_edges(target)
        return sum(1 for i in range(1, len(edges)) if edges[i].line != edges[i - 1].line)

"""Data integrity checks and update mechanisms."""

REPRESENTATIVE = {
    "Line 1": ["苹果园", "国贸", "四惠"],
    "Line 2": ["西直门", "建国门"],
    "Line 4": ["安河桥北", "宣武门"],
    "Line 5": ["天通苑北", "宋家庄"],
    "Line 10": ["知春路", "劲松"],
    "Line 13": ["西直门", "东直门"],
}


def check_data(data):
    """Return (ok, list of messages)."""
    msgs = []
    names = {s["name"] for s in data.get("stations", [])}
    if not names:
        msgs.append("No stations found")
    for line, sts in REPRESENTATIVE.items():
        for s in sts:
            if s not in names:
                msgs.append("Warning: representative station '%s' (%s) missing" % (s, line))
    adj = {}
    lines_of = {s["name"]: set(s.get("lines", [])) for s in data.get("stations", [])}
    for c in data.get("connections", []):
        a, b, d = c.get("source"), c.get("dest"), c.get("distance")
        if a not in names or b not in names:
            msgs.append("Connection endpoint missing: %s - %s" % (a, b))
            continue
        if not isinstance(d, (int, float)) or d <= 0:
            msgs.append("Invalid distance %s between %s and %s" % (d, a, b))
        if c.get("line") not in lines_of.get(a, set()) or c.get("line") not in lines_of.get(b, set()):
            msgs.append("Line %s not listed for %s or %s" % (c.get("line"), a, b))
        adj.setdefault(a, set()).add(b)
        adj.setdefault(b, set()).add(a)
    for n in names:
        if n not in adj:
            msgs.append("Station '%s' has no connections" % n)
    if names and adj:
        start = next(iter(adj))
        seen, stack = {start}, [start]
        while stack:
            for m in adj[stack.pop()]:
                if m not in seen:
                    seen.add(m)
                    stack.append(m)
        if len(seen) != len(adj):
            msgs.append("Network not fully connected: %d of %d stations reachable" % (len(seen), len(adj)))
    hard = [m for m in msgs if not m.startswith("Warning")]
    return (not hard), msgs


def add_line(data, line_name, station_names, distances):
    """Add a new line; distances has len(station_names)-1 entries (meters)."""
    if len(distances) != len(station_names) - 1:
        raise ValueError("distances length mismatch")
    existing = {s["name"]: s for s in data["stations"]}
    for n in station_names:
        if n in existing:
            if line_name not in existing[n]["lines"]:
                existing[n]["lines"].append(line_name)
        else:
            data["stations"].append({"name": n, "lines": [line_name]})
    for i, d in enumerate(distances):
        data["connections"].append({"source": station_names[i], "dest": station_names[i + 1],
                                    "distance": int(d), "line": line_name})


def modify_station(data, name, new_name=None, lines=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    ok = False
    for c in data["connections"]:
        if {c["source"], c["dest"]} == {a, b} and (line is None or c["line"] == line):
            c["distance"] = int(distance)
            ok = True
    return ok

"""CLI for Beijing Subway fare calculation."""
import argparse
import json
import sys

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph
from pathfinding import PathfindingConfig, ShortestPath
from fare_calculator import Sum_money, calculate_estimated_time
import data_manager

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def build_graph(data):
    station_to_number = {}
    number_to_station = {}
    graph = Graph()
    for s in data["stations"]:
        if s["name"] not in station_to_number:
            n = len(station_to_number)
            station_to_number[s["name"]] = n
            number_to_station[n] = s["name"]
            graph.add_node(n)
        graph.set_station_lines(s["name"], s.get("lines", []))
    for c in data["connections"]:
        a, b = station_to_number.get(c["source"]), station_to_number.get(c["dest"])
        if a is None or b is None:
            continue
        graph.add_edge(Edge(a, b, c["distance"], c["line"]))
    return graph, station_to_number, number_to_station


def format_distance(d):
    return "%d m" % d if d < 1000 else "%.2f km" % (d / 1000.0)


def segments(names, edges):
    segs = []
    for i, e in enumerate(edges):
        if segs and segs[-1]["line"] == e.line:
            segs[-1]["to"] = names[i + 1]
            segs[-1]["distance"] += e.distance
        else:
            segs.append({"line": e.line, "from": names[i], "to": names[i + 1], "distance": e.distance})
    return segs


def query(graph, s2n, n2s, start, end, config):
    sp = ShortestPath(graph, s2n[start], config)
    target = s2n[end]
    nodes, edges = sp.path_with_edges(target)
    if not nodes:
        raise RuntimeError("No route found between '%s' and '%s'" % (start, end))
    names = [n2s[n] for n in nodes]
    dist = sp.dist_to[target]
    tc = sp.get_transfer_count(target)
    try:
        fare = Sum_money(dist)
    except Exception as ex:
        raise RuntimeError("Fare calculation failed: %s" % ex)
    segs = segments(names, edges)
    transfers = [segs[i]["from"] for i in range(1, len(segs))]
    return {
        "query": {"start": start, "end": end},
        "path": names,
        "distance_m": dist,
        "distance_km": round(dist / 1000.0, 2),
        "fare": fare,
        "estimated_time_min": round(calculate_estimated_time(dist, tc), 1),
        "transfer_count": tc,
        "transfer_stations": transfers,
        "lines": list(dict.fromkeys(s["line"] for s in segs)),
        "segments": segs,
    }


def render_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"]))]
    for s in r["segments"]:
        out.append("  [%s] %s -> %s (%s)" % (s["line"], s["from"], s["to"], format_distance(s["distance"])))
    out.append("Total distance: %s (%d m / %.2f km)" % (format_distance(r["distance_m"]), r["distance_m"], r["distance_km"]))
    if r["transfer_stations"]:
        out.append("Transfers: %d at %s" % (r["transfer_count"], ", ".join(r["transfer_stations"])))
    else:
        out.append("Transfers: 0")
    out.append("Lines: %s" % ", ".join(r["lines"]))
    out.append("Estimated time: %.1f minutes" % r["estimated_time_min"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def parse_input(line, s2n):
    """Return (start, end, error)."""
    text = line.strip().replace("；", ";")
    parts = text.split(";")
    if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
        return None, None, FORMAT_ERR
    a, b = parts[0].strip(), parts[1].strip()
    if a not in s2n:
        return None, None, "Starting station '%s' does not exist, please check the station name" % a
    if b not in s2n:
        return None, None, "Ending station '%s' does not exist, please check the station name" % b
    if a == b:
        return None, None, "Starting station cannot be the same as the ending station"
    return a, b, None


def emit(text, out_file):
    print(text)
    if out_file:
        try:
            with open(out_file, "a", encoding="utf-8") as f:
                f.write(text + "\n")
        except OSError as ex:
            print("Cannot write output file: %s" % ex)


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except Exception as ex:
        print("Failed to load subway data: %s" % ex)
        return 1

    if args.check_data:
        ok, msgs = data_manager.check_data(data)
        for m in msgs:
            print(m)
        print("Data check %s" % ("passed" if ok else "failed"))
        return 0 if ok else 1

    config = PathfindingConfig()
    config.mode = args.mode
    config.max_transfers = args.max_transfers
    if args.line_preference:
        config.preferred_lines = [x.strip() for x in args.line_preference.split(",") if x.strip()]

    graph, s2n, n2s = build_graph(data)
    print("Enter 'Starting Station; Ending Station' (q to quit)")
    while True:
        try:
            line = input("> ")
        except EOFError:
            break
        if line.strip().lower() in ("q", "quit", "exit"):
            break
        if not line.strip():
            continue
        a, b, err = parse_input(line, s2n)
        if err:
            print(err)
            continue
        try:
            r = query(graph, s2n, n2s, a, b, config)
        except Exception as ex:
            print("Path calculation error: %s" % ex)
            continue
        if args.format == "json":
            emit(json.dumps(r, ensure_ascii=False, indent=2), args.output_file)
        else:
            emit(render_text(r), args.output_file)
    return 0


if __name__ == "__main__":
    sys.exit(main())

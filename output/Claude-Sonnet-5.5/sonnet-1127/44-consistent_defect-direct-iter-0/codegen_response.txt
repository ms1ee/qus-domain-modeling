"""Graph data structures: Edge, Graph, MinPQ."""
import heapq
import itertools


class Edge:
    """Directed weighted edge between station numbers."""

    def __init__(self, source, dest, distance, line):
        self.source = source
        self.dest = dest
        self.distance = distance
        self.weight = distance
        self.line = line

    def __repr__(self):
        return "Edge(%r->%r, %r, %r)" % (self.source, self.dest, self.distance, self.line)


class Graph:
    """Weighted directed graph; station numbers assigned by the caller."""

    def __init__(self):
        self.adj = {}
        self.station_lines = {}

    def add_node(self, number):
        self.adj.setdefault(number, [])

    def set_station_lines(self, station_name, lines):
        self.station_lines[station_name] = list(lines)

    def add_edge(self, edge):
        self.adj.setdefault(edge.source, []).append(edge)
        self.adj.setdefault(edge.dest, [])

    def edges_from(self, number):
        return self.adj.get(number, [])

    def nodes(self):
        return list(self.adj.keys())

    def edge_count(self):
        return sum(len(v) for v in self.adj.values())


class MinPQ:
    """Heap based minimum priority queue."""

    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def push(self, priority, item):
        heapq.heappush(self._heap, (priority, next(self._counter), item))

    def pop(self):
        p, _, item = heapq.heappop(self._heap)
        return p, item

    def is_empty(self):
        return not self._heap

    def __len__(self):
        return len(self._heap)

"""Dijkstra shortest path with transfer awareness."""
from models import MinPQ

TRANSFER_PENALTY = 1500  # meters-equivalent penalty per transfer (comprehensive)
BIG = 10 ** 7


class PathfindingConfig:
    def __init__(self):
        self.mode = "shortest_distance"
        self.max_transfers = None
        self.preferred_lines = []


class ShortestPath:
    def __init__(self, graph, source, config=None):
        self.graph = graph
        self.source = source
        self.config = config or PathfindingConfig()
        self.dist_to = {}
        self._best = {}    # node -> state
        self._info = {}    # state -> (cost, dist, transfers, prev_state, edge)
        self._run()

    def _step_cost(self, edge, transfer):
        mode = self.config.mode
        w = float(edge.weight)
        if edge.line in self.config.preferred_lines:
            w *= 0.9
        if mode == "least_transfer":
            return w / 1000.0 + (BIG if transfer else 0)
        if mode == "comprehensive":
            return w + (TRANSFER_PENALTY if transfer else 0)
        return w

    def _run(self):
        maxt = self.config.max_transfers
        start = (self.source, None)
        self._info[start] = (0.0, 0, 0, None, None)
        pq = MinPQ()
        pq.push(0.0, start)
        done = set()
        while not pq.is_empty():
            cost, state = pq.pop()
            if state in done:
                continue
            done.add(state)
            node, line = state
            _, dist, tr, _, _ = self._info[state]
            if node not in self._best:
                self._best[node] = state
                self.dist_to[node] = dist
            for e in self.graph.edges_from(node):
                transfer = line is not None and e.line != line
                nt = tr + (1 if transfer else 0)
                if maxt is not None and nt > maxt:
                    continue
                ns = (e.dest, e.line)
                if ns in done:
                    continue
                nc = cost + self._step_cost(e, transfer)
                old = self._info.get(ns)
                if old is None or nc < old[0]:
                    self._info[ns] = (nc, dist + e.weight, nt, state, e)
                    pq.push(nc, ns)

    def has_path_to(self, target):
        return target in self._best

    def path_with_edges(self, target):
        if target not in self._best:
            return [], []
        state = self._best[target]
        nodes, edges = [], []
        while state is not None:
            nodes.append(state[0])
            _, _, _, prev, e = self._info[state]
            if e is not None:
                edges.append(e)
            state = prev
        nodes.reverse()
        edges.reverse()
        return nodes, edges

    def path_to(self, target):
        return self.path_with_edges(target)[0]

    def get_transfer_count(self, target):
        if target not in self._best:
            return 0
        return self._info[self._best[target]][2]

"""Fare and time calculation."""
import math


def Sum_money(distance):
    """Fare in yuan for a distance in meters."""
    if distance is None or distance < 0:
        raise ValueError("distance must be non-negative")
    km = math.ceil(distance / 1000.0)
    if km <= 6:
        return 3
    if km <= 12:
        return 4
    if km < 22:
        return 5
    if km <= 32:
        return 6
    return 6 + math.ceil((km - 32) / 20.0)


def calculate_estimated_time(distance, transfer_count):
    """Minutes: 1.5 per km, 5 per transfer."""
    return distance / 1000.0 * 1.5 + 5 * transfer_count

"""Data integrity checks and update helpers."""


def check_data(data):
    """Return list of problem strings (empty if OK)."""
    problems = []
    names = {s["name"] for s in data.get("stations", [])}
    if not names:
        problems.append("No stations found")
    lines = set()
    for s in data.get("stations", []):
        lines.update(s.get("lines", []))
    connected = set()
    for c in data.get("connections", []):
        for k in ("source", "dest"):
            if c[k] not in names:
                problems.append("Connection references unknown station: %s" % c[k])
        d = c.get("distance")
        if not isinstance(d, (int, float)) or d <= 0:
            problems.append("Invalid distance %s->%s: %r" % (c["source"], c["dest"], d))
        connected.add(c["source"])
        connected.add(c["dest"])
    for n in sorted(names - connected):
        problems.append("Station without connections: %s" % n)
    return problems


def add_line(data, line, station_names, distances):
    """Add a line with consecutive stations (both directions)."""
    known = {s["name"]: s for s in data["stations"]}
    for n in station_names:
        if n in known:
            if line not in known[n]["lines"]:
                known[n]["lines"].append(line)
        else:
            s = {"name": n, "lines": [line]}
            data["stations"].append(s)
            known[n] = s
    for i in range(len(station_names) - 1):
        a, b, d = station_names[i], station_names[i + 1], distances[i]
        data["connections"].append({"source": a, "dest": b, "distance": d, "line": line})
        data["connections"].append({"source": b, "dest": a, "distance": d, "line": line})


def update_station(data, name, lines=None, new_name=None):
    for s in data["stations"]:
        if s["name"] == name:
            if lines is not None:
                s["lines"] = list(lines)
            if new_name:
                s["name"] = new_name
                for c in data["connections"]:
                    if c["source"] == name:
                        c["source"] = new_name
                    if c["dest"] == name:
                        c["dest"] = new_name
            return True
    return False


def update_distance(data, a, b, distance, line=None):
    n = 0
    for c in data["connections"]:
        if c["source"] == a and c["dest"] == b and (line is None or c["line"] == line):
            c["distance"] = distance
            n += 1
    return n

"""CLI for the Beijing Subway fare calculator."""
import argparse
import difflib
import json
import os
import re
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from data_loader import DATA_FILE_PATH, load_subway_data
from models import Edge, Graph
from pathfinding import PathfindingConfig, ShortestPath
from fare_calculator import Sum_money, calculate_estimated_time
import data_manager

FORMAT_ERR = "Input format incorrect, please use 'Starting Station; Ending Station' format"


def build_graph(data):
    station_to_number = {}
    number_to_station = {}
    g = Graph()
    for s in data["stations"]:
        if s["name"] not in station_to_number:
            n = len(station_to_number)
            station_to_number[s["name"]] = n
            number_to_station[n] = s["name"]
            g.add_node(n)
        g.set_station_lines(s["name"], s.get("lines", []))
    for c in data["connections"]:
        if c["source"] in station_to_number and c["dest"] in station_to_number:
            g.add_edge(Edge(station_to_number[c["source"]], station_to_number[c["dest"]],
                            c["distance"], c["line"]))
    return g, station_to_number, number_to_station


def parse_input(text):
    parts = [p.strip() for p in re.split(r"[;；,，]", text) if p.strip()]
    if len(parts) != 2 or re.search(r"[;；,，].*[;；,，].*[;；,，]", text):
        return None
    return parts


def resolve(name, names):
    if name in names:
        return name, False
    m = difflib.get_close_matches(name, names, n=1, cutoff=0.5)
    if m:
        return m[0], True
    return None, False


def segments(edges, number_to_station):
    segs = []
    for e in edges:
        if segs and segs[-1]["line"] == e.line:
            segs[-1]["to"] = number_to_station[e.dest]
            segs[-1]["distance"] += e.distance
            segs[-1]["stations"] += 1
        else:
            segs.append({"line": e.line, "from": number_to_station[e.source],
                         "to": number_to_station[e.dest], "distance": e.distance, "stations": 1})
    return segs


def fmt_distance(d):
    return "%d m" % d if d < 1000 else "%d m (%.2f km)" % (d, d / 1000.0)


def query(ctx, start, end, args):
    g, s2n, n2s = ctx
    cfg = PathfindingConfig()
    cfg.mode = args.mode
    cfg.max_transfers = args.max_transfers
    cfg.preferred_lines = [args.line_preference] if args.line_preference else []
    try:
        sp = ShortestPath(g, s2n[start], cfg)
        nodes, edges = sp.path_with_edges(s2n[end])
    except Exception as ex:
        raise RuntimeError("Path calculation failed: %s" % ex)
    if not nodes:
        raise RuntimeError("No route found from '%s' to '%s'" % (start, end))
    dist = sp.dist_to[s2n[end]]
    transfers = sp.get_transfer_count(s2n[end])
    try:
        fare = Sum_money(dist)
        minutes = calculate_estimated_time(dist, transfers)
    except Exception as ex:
        raise RuntimeError("Fare calculation failed: %s" % ex)
    segs = segments(edges, n2s)
    transfer_stations = [segs[i]["to"] for i in range(len(segs) - 1)]
    return {
        "query": {"start": start, "end": end, "mode": args.mode},
        "path": [n2s[n] for n in nodes],
        "distance": dist,
        "fare": fare,
        "estimated_time": round(minutes, 1),
        "transfer_count": transfers,
        "transfer_stations": transfer_stations,
        "lines": [s["line"] for s in segs],
        "segments": segs,
    }


def render_text(r):
    out = ["Route: %s -> %s" % (r["query"]["start"], r["query"]["end"]),
           "Stations (%d): %s" % (len(r["path"]), " -> ".join(r["path"]))]
    for s in r["segments"]:
        out.append("  [%s] %s -> %s (%d stations, %s)" % (
            s["line"], s["from"], s["to"], s["stations"], fmt_distance(s["distance"])))
    out.append("Total distance: %s" % fmt_distance(r["distance"]))
    if r["transfer_stations"]:
        out.append("Transfers: %d at %s" % (r["transfer_count"], ", ".join(r["transfer_stations"])))
    else:
        out.append("Transfers: 0")
    out.append("Lines: %s" % ", ".join(dict.fromkeys(r["lines"])))
    out.append("Estimated time: %.1f minutes" % r["estimated_time"])
    out.append("Fare: %d yuan" % r["fare"])
    return "\n".join(out)


def render(r, fmt):
    if fmt == "json":
        return json.dumps(r, ensure_ascii=False, indent=2)
    return render_text(r)


def handle(line, ctx, args):
    parsed = parse_input(line)
    if not parsed:
        return FORMAT_ERR
    names = list(ctx[1].keys())
    resolved = []
    notes = []
    for label, raw in zip(("Starting", "Ending"), parsed):
        n, fuzzy = resolve(raw, names)
        if n is None:
            return "%s station '%s' does not exist, please check the station name" % (label, raw)
        if fuzzy:
            notes.append("%s station '%s' matched to '%s'" % (label, raw, n))
        resolved.append(n)
    if resolved[0] == resolved[1]:
        return "Starting station cannot be the same as the ending station"
    try:
        r = query(ctx, resolved[0], resolved[1], args)
    except RuntimeError as ex:
        return str(ex)
    text = render(r, args.format)
    if args.format == "text" and notes:
        text = "\n".join(notes) + "\n" + text
    if args.output_file:
        try:
            with open(args.output_file, "a", encoding="utf-8") as f:
                f.write(text + "\n")
        except OSError as ex:
            text += "\nFailed to write output file: %s" % ex
    return text


def main(argv=None):
    p = argparse.ArgumentParser(description="Beijing Subway fare calculator")
    p.add_argument("--output_file")
    p.add_argument("--format", choices=["text", "json"], default="text")
    p.add_argument("--mode", choices=["shortest_distance", "least_transfer", "comprehensive"],
                   default="shortest_distance")
    p.add_argument("--max_transfers", type=int)
    p.add_argument("--line_preference")
    p.add_argument("--check_data", action="store_true")
    args = p.parse_args(argv)

    try:
        data = load_subway_data(DATA_FILE_PATH)
    except (OSError, ValueError) as ex:
        print("Failed to load data: %s" % ex)
        return 1
    if args.check_data:
        problems = data_manager.check_data(data)
        print("Stations: %d, connections: %d" % (len(data["stations"]), len(data["connections"])))
        if problems:
            for x in problems[:100]:
                print("PROBLEM:", x)
            return 1
        print("Data check passed")
        return 0

    ctx = build_graph(data)
    print("Beijing Subway fare calculator. Enter 'Start; End' (empty line or 'q' to quit).")
    while True:
        try:
            line = input("> ").strip()
        except EOFError:
            break
        if not line or line.lower() in ("q", "quit", "exit"):
            break
        print(handle(line, ctx, args))
    return 0


if __name__ == "__main__":
    sys.exit(main())

from collections import OrderedDict
from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang Wei", "Technology Department"),
    ("Wang Fang", "Technology Department"),
    ("Li Na", "Technology Department"),
    ("Liu Yang", "Technology Department"),
    ("Chen Qixian", "Product Department"),
    ("Zhao Min", "Product Department"),
    ("Sun Li", "Product Department"),
    ("Jin Duo", "Marketing Department"),
    ("Zhou Jie", "Marketing Department"),
    ("Wu Qiang", "Marketing Department"),
    ("Ma Jiaqi", "Operations Department"),
    ("Li Bin", "Operations Department"),
    ("Xu Lei", "Operations Department"),
    ("Cheng Ru", "Cheng Ru group"),
    ("Hu Jun", "Cheng Ru group"),
    ("Guo Mei", "Cheng Ru group"),
    ("Lin Tao", "Cheng Ru group"),
]


class ListManager:
    def load_default_list(self):
        return [User(n, d) for n, d in DEFAULT_DATA], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Cannot read file {path}: {e}"]
        for i, raw in enumerate(lines, 1):
            line = raw.strip()
            if not line or line.startswith("#"):
                continue
            sep = "," if "," in line else ("，" if "，" in line else None)
            parts = line.split(sep) if sep else line.split()
            parts = [p.strip() for p in parts]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(
                    f"Line {i}: Format Error: expected 'name,department', got '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = OrderedDict()
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return dict(info)

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, [], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, a, d)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer or None)
        self._prizes.append(p)
        return p

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random

CHENG_RU_KEY = "cheng ru"
FALLBACK_SOURCES = ("Chen Qixian", "Jin Duo")
FALLBACK_TARGET = "Ma Jiaqi"
RELAXED_DRAWER = "Li Bin"


def is_cheng_ru(user):
    return CHENG_RU_KEY in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {id(u) for u in self.winners}
        won_names = {u.name for u in self.winners}
        pool = [u for u in self.users if id(u) not in won and u.name not in won_names]
        drawer = prize.associated_drawer
        if (drawer == FALLBACK_TARGET and len(self.last_two_draw_results) >= 2
                and not any(self.last_two_draw_results[-2:])):
            return [u for u in pool if is_cheng_ru(u)]
        if drawer == RELAXED_DRAWER:
            return pool
        avoid = {g.strip().lower() for g in prize.avoid_groups}
        return [u for u in pool if u.department.strip().lower() not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        return u

    def finish_round(self, prize, winners):
        if prize.associated_drawer in FALLBACK_SOURCES:
            self.last_two_draw_results.append(any(is_cheng_ru(u) for u in winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, order=0):
        self.user = user
        self.prize_name = prize_name
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = prize if isinstance(prize, str) else prize.name
        self.winners.append(Winner(user, name, len(self.winners)))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in reversed(list(prizes)):
            ws = [w for w in self.winners if w.prize_name == p.name]
            ws.sort(key=lambda w: w.order)
            result[p.name] = ws
        return result

    def export_to_csv(self, prizes):
        if not self.winners:
            return False, "No winners to export"
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for name, ws in self.get_winners_by_prize(prizes).items():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{name}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("1) default list  2) custom txt list: ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print("  !", e)
        if errs and input("Errors found. Remove invalid data and continue? (y/n): ").lower() != "y":
            if c == "2":
                print("Upload terminated.")
                continue
        valid, dups = lm.validate_users(users)
        for d in dups:
            print(f"  ! Duplicate removed: {d.name}")
        if valid:
            return valid
        print("No valid users.")


def configure_prizes(pm, groups):
    if input("Load default prizes? (y/n): ").lower() == "y":
        pm.load_default_prizes()
    while input("Add custom prize? (y/n): ").lower() == "y":
        try:
            name = input("Name: ").strip()
            qty = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            drawer = input("Associated drawer: ").strip() or None
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    prizes = pm.get_prizes()
    for i, p in enumerate(prizes, 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("New order (e.g. 2 1 3) or Enter to keep: ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, kept original.")
        except ValueError:
            print("Invalid order, kept original.")


def run_round(lot, prize, rm):
    print(f"\n=== {prize.name} | quantity {prize.quantity} | avoid {prize.avoid_groups} "
          f"| drawer {prize.associated_drawer} ===")
    won = []
    paused = False
    while len(won) < prize.quantity:
        cmd = input(f"Enter to draw #{len(won) + 1} (pause/continue/terminate): ").strip().lower()
        if cmd == "pause":
            paused = True
            print("Paused. Type 'continue' to resume.")
            continue
        if cmd == "continue":
            paused = False
            continue
        if cmd == "terminate":
            print("Round terminated.")
            break
        if paused:
            print("Paused. Type 'continue' to resume.")
            continue
        print(f"Currently drawing winner #{len(won) + 1}...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        won.append(u)
        rm.add_winner(u, prize)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
    lot.finish_round(prize, won)


def main():
    lm = ListManager()
    users = load_users(lm)
    for g, n in lm.get_group_info(users).items():
        print(f"  {g}: {n}")
    pm = PrizeManager()
    configure_prizes(pm, None)
    lot = LotteryManager(users)
    rm = ResultManager()
    prizes = pm.get_prizes()
    for p in prizes:
        run_round(lot, p, rm)
    print("\n=== Final Results ===")
    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{name}: {w.user.name} ({w.user.department})")
    ok, msg = rm.export_to_csv(prizes)
    print(msg)


if __name__ == "__main__":
    main()

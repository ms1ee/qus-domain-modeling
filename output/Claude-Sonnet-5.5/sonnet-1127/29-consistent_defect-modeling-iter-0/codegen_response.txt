Everything lives under `src/`, but I have not run any of it. Where the PRD contradicts itself, I picked one value and noted it below.

- **Rider speed and state:** The PRD gives speed as both 10 and 20, and states as both 0–2 and 1–3. I used speed 20 and states 0 = idle, 1 = picking up, 2 = delivering.
- **Order coordinates:** Order coordinates snap to the top-left corner of their grid cell and are clamped to 0–700, so an input of 800 lands in the last cell.
- **Rider movement:** Riders move horizontally first, then vertically, up to their speed each tick.
- **Assignment:** `find_best_rider_for_order` picks the rider with the fewest unfinished orders, then the nearest one by straight-line distance. It returns `None` if there are no riders.
- **Recruitment:** `assign_order` recruits a rider (cost 500) when funds are at least 300 and pending orders exceed idle riders. It also always recruits when no rider exists yet, so the first order can be assigned.
- **Main loop:** Each command you enter advances the simulation by one tick.
- **Coordinate validation:** Input coordinates are accepted from 0 to 1000, as the PRD's validation rule says.
- **Not implemented:** JSON persistence, command history, shortcut keys, the 16x16 route map, and the installation script and `requirements.txt` are not included. `status`, `orders` and `riders` print plain text, and `start` prints an ASCII map of the assigned rider's route.

"""Order data model."""
import itertools

import globalvar

_ids = itertools.count(1)


def to_grid(v):
    """Convert a coordinate to the top-left corner of its grid cell."""
    v = int(v)
    return max(0, min(700, (v // 100) * 100))


class Order:
    """A delivery order."""

    def __init__(self, res_pos, cus_pos):
        self.id = next(_ids)
        self.res_pos = (to_grid(res_pos[0]), to_grid(res_pos[1]))
        self.cus_pos = (to_grid(cus_pos[0]), to_grid(cus_pos[1]))
        self.status = "pending"
        self.rider_id = None
        self.start_time = globalvar.get_value("tick", 0) or 0
        self.completion_time = None
        self.distance = float(
            abs(self.res_pos[0] - self.cus_pos[0])
            + abs(self.res_pos[1] - self.cus_pos[1]))

    @property
    def time_taken(self):
        if self.completion_time is None:
            return None
        return self.completion_time - self.start_time

    def assign_to(self, rider_id):
        """Assign this order to a rider."""
        self.rider_id = rider_id
        self.status = "assigned"

"""Rider model."""
import globalvar

IDLE, PICKING_UP, DELIVERING = 0, 1, 2


class Rider:
    """A delivery rider."""

    def __init__(self, rider_id):
        self.rider_id = rider_id
        self.x = 400
        self.y = 400
        self.speed = 20
        self.state = IDLE
        self.orders = []
        self.special_case = False
        self.log = []
        self._record("spawn", None)

    def _record(self, event, order):
        self.log.append({
            "event": event,
            "order_id": order.id if order is not None else None,
            "time": globalvar.get_value("tick", 0) or 0,
            "x": self.x,
            "y": self.y,
        })

    def current_order(self):
        """First order not yet completed."""
        for o in self.orders:
            if o.status != "completed":
                return o
        return None

    def pending_count(self):
        return sum(1 for o in self.orders if o.status != "completed")

    def move(self):
        """Advance one tick."""
        budget = self.speed
        for _ in range(10):
            order = self.current_order()
            if order is None:
                self.state = IDLE
                return
            if self.state == IDLE:
                self.state = PICKING_UP
                self._record("accepted_order", order)
            target = order.res_pos if self.state == PICKING_UP else order.cus_pos
            if (self.x, self.y) == tuple(target):
                if self.state == PICKING_UP:
                    self.state = DELIVERING
                    self._record("arrived_at_merchant", order)
                    continue
                order.status = "completed"
                order.completion_time = globalvar.get_value("tick", 0) or 0
                self.state = IDLE
                self._record("delivered_to_customer", order)
                return
            if budget <= 0:
                return
            budget -= self._step(target, budget)

    def _step(self, target, budget):
        """Move along streets: horizontal first then vertical."""
        used = 0
        dx = target[0] - self.x
        if dx:
            s = min(abs(dx), budget)
            self.x += s if dx > 0 else -s
            used += s
            budget -= s
        dy = target[1] - self.y
        if dy and budget > 0:
            s = min(abs(dy), budget)
            self.y += s if dy > 0 else -s
            used += s
        return used

"""Order assignment."""
import math

import globalvar
from rider import Rider
from order_model import Order

RECRUIT_COST = 500
RECRUIT_MIN_FUNDS = 300


def find_best_rider_for_order(order):
    """Rider with fewest orders, ties broken by distance to restaurant."""
    riders = globalvar.get_value("riders", []) or []
    if not riders:
        return None

    def key(r):
        d = math.hypot(r.x - order.res_pos[0], r.y - order.res_pos[1])
        return (r.pending_count(), d, r.rider_id)

    return min(riders, key=key)


def maybe_recruit():
    """Recruit a rider if conditions hold. Returns the new rider or None."""
    riders = globalvar.get_value("riders")
    funds = globalvar.get_value("funds")
    orders = globalvar.get_value("orders", [])
    pending = sum(1 for o in orders if o.status != "completed")
    idle = sum(1 for r in riders if r.pending_count() == 0)
    if not riders or (funds >= RECRUIT_MIN_FUNDS and pending > idle):
        if funds < RECRUIT_MIN_FUNDS and riders:
            return None
        rid = globalvar.get_value("rider_id_counter") + 1
        globalvar.set_value("rider_id_counter", rid)
        globalvar.set_value("funds", funds - RECRUIT_COST)
        r = Rider(rid)
        riders.append(r)
        return r
    return None


def assign_order(order):
    """Register, possibly recruit, and assign the order.

    Returns (rider, new_rider_or_None).
    """
    globalvar.get_value("orders").append(order)
    new = maybe_recruit()
    rider = find_best_rider_for_order(order)
    if rider is None:
        return None, new
    order.assign_to(rider.rider_id)
    rider.orders.append(order)
    return rider, new


def create_order(rx, ry, cx, cy):
    return Order((rx, ry), (cx, cy))

"""Command-line entry point."""
import globalvar
from system import init_system
from order import create_order, assign_order


def render_map(rider, order):
    """ASCII 16x16 map with the route."""
    g = [['#' if (i % 2 == 0 or j % 2 == 0) else '.' for j in range(16)]
         for i in range(16)]

    def cell(p):
        return min(15, p[1] // 50), min(15, p[0] // 50)
    r, c = cell(order.res_pos)
    g[r][c] = 'S'
    r, c = cell(order.cus_pos)
    g[r][c] = 'C'
    r, c = cell((rider.x, rider.y))
    g[r][c] = 'R'
    return "\n".join("".join(row) for row in g)


def tick():
    globalvar.set_value("tick", globalvar.get_value("tick") + 1)
    for r in globalvar.get_value("riders"):
        r.move()


def cmd_start():
    try:
        vals = [int(v) for v in input("rx ry cx cy: ").split()]
    except ValueError:
        print("Error: integers required")
        return
    if len(vals) != 4 or any(v < 0 or v > 1000 for v in vals):
        print("Error: need 4 integers in 0-1000")
        return
    order = create_order(*vals)
    rider, new = assign_order(order)
    print("Restaurant:", order.res_pos, "Customer:", order.cus_pos)
    print("New rider added:", new is not None)
    if rider:
        print("Rider", rider.rider_id, "speed", rider.speed,
              "est. time", (abs(rider.x - order.res_pos[0])
                            + abs(rider.y - order.res_pos[1])
                            + order.distance) / rider.speed)
        print(render_map(rider, order))


def main():
    init_system()
    names = {0: "idle", 1: "picking up", 2: "delivering"}
    while globalvar.get_value("running"):
        try:
            cmd = input("> ").strip().lower()
        except EOFError:
            break
        if cmd in ("quit", "exit", "q"):
            globalvar.set_value("running", False)
            continue
        if cmd in ("start", "s"):
            cmd_start()
        elif cmd == "status":
            print("Funds:", globalvar.get_value("funds"))
            for r in globalvar.get_value("riders"):
                print(r.rider_id, (r.x, r.y), names[r.state],
                      [o.id for o in r.orders])
        elif cmd == "orders":
            for o in globalvar.get_value("orders"):
                print(o.id, o.start_time, o.res_pos, o.cus_pos, o.distance,
                      o.time_taken, o.rider_id, o.status)
        elif cmd == "riders":
            for r in globalvar.get_value("riders"):
                for e in r.log:
                    print(r.rider_id, e)
        elif cmd:
            print("Unknown command")
        tick()


if __name__ == "__main__":
    main()

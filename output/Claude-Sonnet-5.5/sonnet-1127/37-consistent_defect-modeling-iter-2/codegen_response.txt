import re
from lottery_system.models import User


class ListManager:
    def load_default_list(self):
        data = [
            ("Zhang San", "Technology Department"),
            ("Li Si", "Technology Department"),
            ("Wang Wu", "Technology Department"),
            ("Zhao Liu", "Product Department"),
            ("Sun Qi", "Product Department"),
            ("Zhou Ba", "Product Department"),
            ("Wu Jiu", "Marketing Department"),
            ("Zheng Shi", "Marketing Department"),
            ("Li Bin", "Operations Department"),
            ("Chen Qixian", "Operations Department"),
            ("Jin Duo", "Finance Department"),
            ("Ma Jiaqi", "Finance Department"),
            ("Cheng Ru", "Cheng Ru group"),
            ("Liu Yang", "Cheng Ru group"),
            ("Huang Mei", "Cheng Ru group"),
        ]
        return [User(n, d) for n, d in data], []

    def load_custom_list(self, path):
        return self._load_from_path(path)

    def _load_from_path(self, path):
        users = []
        errors = []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error: cannot read file {path}: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = re.split(r"[,;，；]", line.strip())
            if len(parts) != 2 or not parts[0].strip() or not parts[1].strip():
                errors.append(f"Format Error: Line {i}: '{line.strip()}'")
                continue
            users.append(User(parts[0].strip(), parts[1].strip()))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid = []
        dups = []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self.prizes = []

    def load_default_prizes(self):
        errors = []
        presets = [
            ("Third Prize - Movie Tickets", 3),
            ("Second Prize - Shopping Cards", 2),
            ("First Prize - 1 Day Annual Leave", 1),
        ]
        for name, q in presets:
            try:
                self.add_prize(name, q)
            except ValueError as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or not str(name).strip():
            raise ValueError("Prize name cannot be empty")
        if not isinstance(quantity, int) or quantity <= 0:
            raise ValueError("Quantity must be a positive integer")
        p = Prize(name, quantity, list(avoid_groups or []), associated_drawer)
        self.prizes.append(p)
        return p

    def get_prizes(self):
        return list(self.prizes)

    def reorder_prizes(self, order):
        n = len(self.prizes)
        try:
            if sorted(order) != list(range(1, n + 1)):
                return False
        except TypeError:
            return False
        self.prizes = [self.prizes[i - 1] for i in order]
        return True

import random

CHENG_RU = "cheng ru"


def _is_cheng_ru(user):
    return CHENG_RU in user.department.lower()


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        won = {w.name for w in self.winners}
        pool = [u for u in self.users if u.name not in won]
        drawer = prize.associated_drawer
        if drawer == "Ma Jiaqi" and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            return [u for u in pool if _is_cheng_ru(u)]
        if drawer == "Li Bin":
            return pool
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def _avoided_members(self, prize):
        won = {w.name for w in self.winners}
        avoid = set(prize.avoid_groups or [])
        return [u for u in self.users
                if u.name not in won and u.department in avoid]

    def rule_description(self, prize):
        parts = []
        if prize.associated_drawer:
            parts.append(f"Drawer: {prize.associated_drawer}")
        if prize.associated_drawer == "Li Bin":
            parts.append("Avoidance rules not applied")
        elif prize.avoid_groups:
            parts.append("Avoid groups: " + ", ".join(prize.avoid_groups))
        if prize.associated_drawer == "Ma Jiaqi" and len(self.last_two_draw_results) >= 2 \
                and not any(self.last_two_draw_results[-2:]):
            parts.append("Cheng Ru group protection active")
        parts.append("Previous winners excluded")
        return "; ".join(parts)

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool and prize.associated_drawer != "Li Bin":
            fallback_active = (prize.associated_drawer == "Ma Jiaqi"
                               and len(self.last_two_draw_results) >= 2
                               and not any(self.last_two_draw_results[-2:]))
            if not fallback_active:
                pool = self._avoided_members(prize)
        if not pool:
            return None
        user = random.choice(pool)
        self.winners.append(user)
        return user

    def record_round(self, prize, round_winners):
        if prize.associated_drawer in ("Chen Qixian", "Jin Duo"):
            self.last_two_draw_results.append(
                any(_is_cheng_ru(u) for u in round_winners))
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from datetime import datetime


class Winner:
    def __init__(self, user, prize_name, time=None, order=0):
        self.user = user
        self.prize_name = prize_name
        self.time = time or datetime.now()
        self.order = order


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name, datetime.now(), len(self.winners) + 1))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(os.getcwd(),
                            f"awards_result_{datetime.now():%Y%m%d}.txt")
        try:
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize Name,Winning Time\n")
                for ws in self.get_winners_by_prize(prizes).values():
                    for w in ws:
                        f.write(f"{w.user.name},{w.user.department},{w.prize_name},"
                                f"{w.time:%Y-%m-%d %H:%M:%S}\n")
        except OSError as e:
            return False, f"Export failed: {e}"
        return True, f"Results successfully exported to {path}"

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    while True:
        c = input("Load list: 1) default 2) custom txt file > ").strip()
        if c == "1":
            users, errs = lm.load_default_list()
        elif c == "2":
            users, errs = lm.load_custom_list(input("Path: ").strip())
        else:
            continue
        for e in errs:
            print(e)
        if errs and c == "2":
            a = input("Errors found. 'remove' invalid lines and continue, or 'terminate': ").strip()
            if a == "terminate":
                continue
        users, dups = lm.validate_users(users)
        for d in dups:
            print(f"Duplicate removed: {d.name} ({d.department})")
        if users:
            return users
        print("No valid users.")


def configure_prizes(pm, groups):
    if input("Load preset prizes? (y/n) > ").lower().startswith("y"):
        pm.load_default_prizes()
    while input("Add custom prize? (y/n) > ").lower().startswith("y"):
        name = input("Name: ").strip()
        try:
            q = int(input("Quantity: "))
            avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
            drawer = input("Lottery person (blank for none): ").strip() or None
            pm.add_prize(name, q, avoid, drawer)
        except ValueError as e:
            print("Invalid:", e)
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")
    s = input("New order (space separated numbers, blank to keep) > ").split()
    if s:
        try:
            if not pm.reorder_prizes([int(x) for x in s]):
                print("Invalid order, keeping default.")
        except ValueError:
            print("Invalid order, keeping default.")


def main():
    lm = ListManager()
    users = load_users(lm)
    groups = lm.get_group_info(users)
    for g, n in groups.items():
        print(f"Group {g}: {n}")
    pm = PrizeManager()
    configure_prizes(pm, groups)
    prizes = pm.get_prizes()
    engine = LotteryManager(users)
    rm = ResultManager()
    for r, prize in enumerate(prizes, 1):
        print(f"\n=== Round {r}: {prize.name}, quantity {prize.quantity} ===")
        print("Rules:", engine.rule_description(prize))
        round_winners = []
        i = 1
        paused = False
        while i <= prize.quantity:
            cmd = input("Enter to draw (pause/continue/terminate) > ").strip().lower()
            if cmd == "pause":
                paused = True
                print("Paused.")
                continue
            if cmd == "continue":
                paused = False
                print("Resumed.")
                continue
            if paused:
                print("Paused; type 'continue'.")
                continue
            if cmd == "terminate":
                print("Round terminated.")
                break
            print(f"Currently drawing the {i}th winner")
            u = engine.draw_one(prize)
            if u is None:
                print("No eligible participants left.")
                break
            rm.add_winner(u, prize)
            round_winners.append(u)
            print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
            i += 1
        engine.record_round(prize, round_winners)
        print("Round results:")
        for k, u in enumerate(round_winners, 1):
            print(f" {k}. {u.name} ({u.department})")
    print("\n=== Final winners ===")
    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name}, {w.user.department}, {name}")
    ok, msg = rm.export_to_csv(prizes)
    print(msg)


if __name__ == "__main__":
    main()

from lottery_system.models import User

DEFAULT_DATA = [
    ("Zhang San", "Technology Department"),
    ("Li Si", "Technology Department"),
    ("Wang Wu", "Technology Department"),
    ("Zhao Liu", "Product Department"),
    ("Sun Qi", "Product Department"),
    ("Zhou Ba", "Product Department"),
    ("Wu Jiu", "Cheng Ru group"),
    ("Zheng Shi", "Cheng Ru group"),
    ("Chen Qixian", "Operations Department"),
    ("Jin Duo", "Operations Department"),
    ("Ma Jiaqi", "Marketing Department"),
    ("Li Bin", "Marketing Department"),
    ("Qian Yi", "Cheng Ru group"),
    ("Feng Er", "Product Department"),
]


class ListManager:
    def load_default_list(self):
        users = [User(n, d) for n, d in DEFAULT_DATA]
        return users, []

    def load_custom_list(self, path):
        users, errors = self._load_from_path(path)
        valid, dups = self.validate_users(users)
        errors = list(errors)
        for d in dups:
            errors.append(f"Duplicate name: {d.name} ({d.department})")
        return valid, errors

    def _load_from_path(self, path):
        users, errors = [], []
        try:
            with open(path, "r", encoding="utf-8") as f:
                lines = f.read().splitlines()
        except (OSError, UnicodeDecodeError) as e:
            return [], [f"Error reading file: {e}"]
        for i, line in enumerate(lines, 1):
            if not line.strip():
                continue
            parts = [p.strip() for p in line.split(",")]
            if len(parts) != 2 or not parts[0] or not parts[1]:
                errors.append(f"Line {i}: Format Error - '{line}'")
                continue
            users.append(User(parts[0], parts[1]))
        return users, errors

    def validate_users(self, users):
        seen = set()
        valid, dups = [], []
        for u in users:
            if u.name in seen:
                dups.append(u)
            else:
                seen.add(u.name)
                valid.append(u)
        return valid, dups

    def get_group_info(self, users):
        info = {}
        for u in users:
            info[u.department] = info.get(u.department, 0) + 1
        return info

from lottery_system.models import Prize


class PrizeManager:
    def __init__(self):
        self._prizes = []

    def load_default_prizes(self):
        errors = []
        defaults = [
            ("Third Prize - Movie Tickets", 3, [], "Chen Qixian"),
            ("Second Prize - Shopping Cards", 2, [], "Jin Duo"),
            ("First Prize - 1 Day Annual Leave", 1, ["Cheng Ru group"], "Ma Jiaqi"),
        ]
        for n, q, a, d in defaults:
            try:
                self.add_prize(n, q, a, d)
            except Exception as e:
                errors.append(str(e))
        return errors

    def add_prize(self, name, quantity, avoid_groups=None, associated_drawer=None):
        if not name or int(quantity) <= 0:
            raise ValueError("Invalid prize name or quantity")
        self._prizes.append(
            Prize(name, int(quantity), list(avoid_groups or []), associated_drawer)
        )

    def get_prizes(self):
        return list(self._prizes)

    def reorder_prizes(self, order):
        n = len(self._prizes)
        if sorted(order) != list(range(1, n + 1)):
            return False
        self._prizes = [self._prizes[i - 1] for i in order]
        return True

import random

PROTECTED_GROUP = "Cheng Ru group"
WATCHED_DRAWERS = ("Chen Qixian", "Jin Duo")
FALLBACK_DRAWER = "Ma Jiaqi"
SPECIAL_DRAWER = "Li Bin"


class LotteryManager:
    def __init__(self, users):
        self.users = list(users)
        self.winners = []
        self._winner_names = set()
        self.last_two_draw_results = []

    def _get_eligible_pool(self, prize):
        pool = [u for u in self.users if u.name not in self._winner_names]
        drawer = prize.associated_drawer
        if drawer == SPECIAL_DRAWER:
            return pool
        if (
            drawer == FALLBACK_DRAWER
            and len(self.last_two_draw_results) >= 2
            and not any(self.last_two_draw_results[-2:])
        ):
            return [u for u in pool if u.department == PROTECTED_GROUP]
        avoid = set(prize.avoid_groups or [])
        return [u for u in pool if u.department not in avoid]

    def draw_one(self, prize):
        pool = self._get_eligible_pool(prize)
        if not pool:
            return None
        u = random.choice(pool)
        self.winners.append(u)
        self._winner_names.add(u.name)
        return u

    def finish_round(self, prize, winners):
        if prize.associated_drawer in WATCHED_DRAWERS:
            self.last_two_draw_results.append(
                any(w.department == PROTECTED_GROUP for w in winners)
            )
            self.last_two_draw_results = self.last_two_draw_results[-2:]

import os
from dataclasses import dataclass
from datetime import datetime

from lottery_system.models import User


@dataclass
class Winner:
    user: User
    prize_name: str = ""


class ResultManager:
    def __init__(self):
        self.winners = []

    def add_winner(self, user, prize):
        name = getattr(prize, "name", prize)
        self.winners.append(Winner(user, name))

    def get_winners_by_prize(self, prizes):
        result = {}
        for p in prizes:
            result[p.name] = [w for w in self.winners if w.prize_name == p.name]
        return result

    def export_to_csv(self, prizes):
        path = os.path.join(
            os.getcwd(), f"awards_result_{datetime.now().strftime('%Y%m%d')}.txt"
        )
        try:
            grouped = self.get_winners_by_prize(prizes)
            with open(path, "w", encoding="utf-8") as f:
                f.write("Name,Department,Prize\n")
                for p in prizes:
                    for w in grouped[p.name]:
                        f.write(f"{w.user.name},{w.user.department},{p.name}\n")
            return True, f"Results successfully exported to {path}"
        except OSError as e:
            return False, f"Export failed: {e}"

from lottery_system.services.list_manager import ListManager
from lottery_system.services.prize_manager import PrizeManager
from lottery_system.services.lottery_manager import LotteryManager
from lottery_system.services.result_manager import ResultManager


def load_users(lm):
    c = input("1) default list  2) custom txt file: ").strip()
    if c == "2":
        path = input("Path: ").strip()
        users, errors = lm.load_custom_list(path)
        if errors:
            print("Problems found:")
            for e in errors:
                print("  ", e)
            if input("Terminate upload? (y/N): ").strip().lower() == "y":
                return []
            print("Invalid data removed, continuing.")
        return users
    users, _ = lm.load_default_list()
    return users


def show_groups(lm, users):
    for g, n in lm.get_group_info(users).items():
        print(f"  {g}: {n}")


def configure_prizes(pm):
    if input("Load preset prizes? (Y/n): ").strip().lower() != "n":
        pm.load_default_prizes()
    while input("Add custom prize? (y/N): ").strip().lower() == "y":
        name = input("Name: ").strip()
        try:
            qty = int(input("Quantity: "))
        except ValueError:
            print("Invalid quantity")
            continue
        avoid = [g.strip() for g in input("Avoid groups (comma separated): ").split(",") if g.strip()]
        drawer = input("Associated drawer (optional): ").strip() or None
        try:
            pm.add_prize(name, qty, avoid, drawer)
        except ValueError as e:
            print(e)
    show_prizes(pm)
    s = input("New order by serial numbers (space separated, Enter to keep): ").split()
    if s:
        try:
            ok = pm.reorder_prizes([int(x) for x in s])
        except ValueError:
            ok = False
        print("Reordered." if ok else "Invalid order, keeping original.")
        show_prizes(pm)


def show_prizes(pm):
    for i, p in enumerate(pm.get_prizes(), 1):
        print(f"{i}. {p.name} x{p.quantity} avoid={p.avoid_groups} drawer={p.associated_drawer}")


def run_round(prize, lot, rm):
    print(f"\n=== {prize.name} | quantity {prize.quantity} ===")
    print(f"Rules: avoid groups {prize.avoid_groups or 'none'}; drawer {prize.associated_drawer or 'n/a'}")
    won = []
    i = 1
    while i <= prize.quantity:
        cmd = input("[Enter]=draw, pause, terminate: ").strip().lower()
        if cmd == "terminate":
            print("Round terminated.")
            break
        if cmd == "pause":
            while True:
                c = input("Paused. Type 'continue' or 'terminate': ").strip().lower()
                if c == "continue":
                    break
                if c == "terminate":
                    cmd = "terminate"
                    break
            if cmd == "terminate":
                print("Round terminated.")
                break
        print(f"Currently drawing the {i}th winner...")
        u = lot.draw_one(prize)
        if u is None:
            print("No eligible candidates left.")
            break
        rm.add_winner(u, prize)
        won.append(u)
        print(f"Congratulations {u.name} ({u.department}) for winning {prize.name}")
        i += 1
    lot.finish_round(prize, won)


def main():
    lm = ListManager()
    users = load_users(lm)
    if not users:
        print("No users loaded.")
        return
    show_groups(lm, users)
    pm = PrizeManager()
    configure_prizes(pm)
    prizes = pm.get_prizes()
    if not prizes:
        print("No prizes configured.")
        return
    lot = LotteryManager(users)
    rm = ResultManager()
    for p in prizes:
        run_round(p, lot, rm)
    print("\n=== Complete winner list ===")
    for name, ws in rm.get_winners_by_prize(prizes).items():
        for w in ws:
            print(f"{w.user.name},{w.user.department},{name}")
    if input("Export results? (Y/n): ").strip().lower() != "n":
        print(rm.export_to_csv(prizes)[1])


if __name__ == "__main__":
    main()
